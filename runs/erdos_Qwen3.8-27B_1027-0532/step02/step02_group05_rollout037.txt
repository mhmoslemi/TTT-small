    N = len(h)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def grad_corr(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)

        k = j - (N - 1)

        else:

            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_corr(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def ap_obj(g, alpha):
    h = antipal_h(g)
    return smooth_max_val(compute_correlations(h), alpha)


def ap_grad(g, alpha):
    h = antipal_h(g)
    grad_h = grad_corr(h, alpha)
    half = len(g)
    grad_g = grad_h[:half].copy()
    grad_g -= grad_h[half + 1:][::-1]
    return grad_g


def to_antipal(h):
    n = len(h)
    if n % 2 != 1:
        n = n - 1
        h = h[:n]
    m = n // 2
    g = (h[:m] + (1.0 - h[m + 1:][::-1])) / 2.0
    g = np.clip(g, 0.0, 1.0)
    return antipal_h(g), g


def lbfbsb_opt(g0, alpha, maxiter, maxls=50):
    n = len(g0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        ap_obj, g0, method='L-BFGS-B', bounds=bounds,
        jac=ap_grad, args=(alpha,),
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0,
                 'maxls': maxls, 'maxfun': max(100, maxiter * 5)}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        return np.clip(res.x, 0.0, 1.0)
    return None


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def grad_full(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def full_obj(h, alpha):
    return smooth_max_val(compute_correlations(h), alpha)


def full_lbfbsb(h0, alpha, target_sum, maxiter, maxls=40):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N

    def fun(x):
        return smooth_max_val(compute_correlations(x), alpha) + 500.0 * (np.sum(x) - target_sum) ** 2

    def jac(x):
        g = grad_full(x, alpha)
        g += 1000.0 * (np.sum(x) - target_sum)
        return g

    res = minimize(
        fun, h0, method='L-BFGS-B', bounds=bounds,
        jac=jac,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0,
                 'maxls': maxls, 'maxfun': max(100, maxiter * 5)}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    init = np.array(initial_h_values, dtype=float)
    best_h, best_g = to_antipal(init)
    best_score = evaluate_score(best_h)
    n = len(best_h)
    m = len(best_g)

    # Phase 1: L-BFGS-B graduated alpha on anti-palindromic
    alphas_phase1 = [10, 20, 50, 100, 200, 400, 800, 1500, 3000, 6000, 12000, 25000, 50000]
    for alpha in alphas_phase1:
        if elapsed() > budget_s - 400:
            break
        maxiter = max(50, min(600, int(30000 // m)))
        g_new = lbfbsb_opt(best_g, alpha, maxiter)
        if g_new is not None:
            h_new = antipal_h(g_new)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_g = g_new
                best_h = h_new

    # Phase 2: Diverse random restarts
    n_restarts = 120
    for r in range(n_restarts):
        if elapsed() > budget_s - 350:
            break
        restart_type = r % 4
        if restart_type == 0:
            scale = 0.03 + 0.12 * rng.rand()
            g_pert = best_g + rng.randn(m) * scale
        elif restart_type == 1:
            n_blocks = rng.randint(2, 8)
            g_pert = np.zeros(m)
            for b in range(n_blocks):
                s_idx = b * m // n_blocks
                e_idx = (b + 1) * m // n_blocks
                g_pert[s_idx:e_idx] = 0.3 + 0.4 * rng.rand()
            g_pert = g_pert + rng.randn(m) * 0.05
        elif restart_type == 2:
            k_mode = rng.randint(1, 6)
            phase = rng.rand() * 2.0 * np.pi
            amp = 0.3 + 0.2 * rng.rand()
            g_pert = 0.5 + amp * np.sin(2.0 * np.pi * k_mode * np.arange(m) / m + phase)
        else:
            g_pert = (rng.rand(m) < 0.5).astype(float)
            g_pert = 0.1 * g_pert + 0.9 * best_g

        g_pert = np.clip(g_pert, 0.0, 1.0)
        alpha = int(50 + 5000 * rng.rand() ** 2)
        maxiter = max(30, min(150, int(8000 // m)))
        g_new = lbfbsb_opt(g_pert, alpha, maxiter, maxls=30)
        if g_new is not None:
            h_new = antipal_h(g_new)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_g = g_new
                best_h = h_new

    # Phase 3: Multi-resolution (downscale then upscale)
    for n_low in [51, 101, 201]:
        if n_low >= n or elapsed() > budget_s - 300:
            continue
        m_low = n_low // 2
        x_full = np.linspace(0, 2, n, endpoint=False)
        x_low = np.linspace(0, 2, n_low, endpoint=False)
        h_interp = np.interp(x_low, x_full, best_h)
        _, g_low = to_antipal(h_interp)
        for alpha in [50, 500, 3000]:
            if elapsed() > budget_s - 250:
                break
            maxiter = max(30, min(200, int(12000 // m_low)))
            g_new = lbfbsb_opt(g_low, alpha, maxiter)
            if g_new is not None:
                h_low_new = antipal_h(g_new)
                x_up = np.linspace(0, 2, n, endpoint=False)
                h_up = np.interp(x_up, x_low, h_low_new)
                h_ap, g_up = to_antipal(h_up)
                s_up = evaluate_score(h_ap)
                if s_up < best_score:
                    best_score = s_up
                    best_g = g_up
                    best_h = h_ap

    # Phase 4: Higher resolution attempt
    n_high = 2 * n + 1
    if n_high <= 2500 and elapsed() < budget_s - 250:
        m_high = n_high // 2
        x_full = np.linspace(0, 2, n, endpoint=False)
        x_high = np.linspace(0, 2, n_high, endpoint=False)
        h_high = np.interp(x_high, x_full, best_h)
        _, g_high = to_antipal(h_high)
        for alpha in [20, 100, 500, 2000]:
            if elapsed() > budget_s - 200:
                break
            maxiter = max(20, min(120, int(8000 // m_high)))
            g_new = lbfbsb_opt(g_high, alpha, maxiter)
            if g_new is not None:
                h_new = antipal_h(g_new)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_g = g_new
                    best_h = h_new
                    n = n_high
                    m = m_high

    # Phase 5: Try without anti-palindromic constraint (full optimization)
    if elapsed() < budget_s - 200:
        n_full = len(best_h)
        target_full = n_full / 2.0
        h_start = best_h.copy()
        for alpha in [100, 500, 2000]:
            if elapsed() > budget_s - 150:
                break
            maxiter = max(30, min(150, int(5000 // n_full)))
            h_new = full_lbfbsb(h_start, alpha, target_full, maxiter, maxls=30)
            if h_new is not None:
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new
                    h_start = h_new.copy()
                    # Convert back to anti-palindromic for subsequent phases
                    _, best_g = to_antipal(best_h)
                    m = len(best_g)
                    n = len(best_h)

    # Phase 6: Final high-alpha polish
    for alpha in [10000, 30000, 80000, 200000]:
        if elapsed() > budget_s - 20:
            break
        maxiter = max(100, int(60000 // len(best_g)))
        g_new = lbfbsb_opt(best_g, alpha, maxiter, maxls=100)
        if g_new is not None:
            h_new = antipal_h(g_new)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_g = g_new
                best_h = h_new

    # Phase 7: Very fine local search with many small restarts
    n_fine = 60
    for r in range(n_fine):
        if elapsed() > budget_s - 10:
            break
        scale = 0.01 + 0.04 * rng.rand()
        g_pert = best_g + rng.randn(len(best_g)) * scale
        g_pert = np.clip(g_pert, 0.0, 1.0)
        alpha = 5000 + int(50000 * rng.rand())
        maxiter = max(50, min(200, int(10000 // len(best_g))))
        g_new = lbfbsb_opt(g_pert, alpha, maxiter, maxls=50)
        if g_new is not None:
            h_new = antipal_h(g_new)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_g = g_new
                best_h = h_new

    best_h = antipal_h(best_g)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
