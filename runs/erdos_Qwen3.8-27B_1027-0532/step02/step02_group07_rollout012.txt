    """Project h onto {x : 0 <= x <= 1, sum(x) = target} using water-filling."""
    n = len(h)
    h = h.copy()
    # Water-filling: find threshold tau such that sum(clip(h - tau, 0, 1)) = target
    # clip(h - tau, 0, 1) = max(0, min(1, h - tau))
    # = (h - tau) for tau < h < tau+1
    # = 1 for h > tau + 1
    # = 0 for h < tau
    
    lo, hi = -1.0, 1.0
    for _ in range(100):

        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2

def fourier_move(h, rng, n_harmonics=5):
    """Add a sinusoidal perturbation and optimize the amplitude."""
    n = len(h)
    x = np.linspace(0, 2, n, endpoint=False)
    best_h_local = h.copy()
    best_score_local = evaluate_score(h)
    
    for k in range(1, n_harmonics + 1):
        for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
            direction = np.sin(2*np.pi*k*x + phase)
            # Optimize amplitude a in [-0.5, 0.5]
            # h_new = clip(h + a * direction, 0, 1), then project sum
            best_a = 0.0
            best_s = best_score_local
            for a in np.linspace(-0.3, 0.3, 30):
                h_try = np.clip(h + a * direction, 0, 1)
                h_try = project_sum(h_try, n/2.0)
                s = evaluate_score(h_try)
                if s < best_s:
                    best_s = s
                    best_a = a
            if best_a != 0:
                h_new = np.clip(h + best_a * direction, 0, 1)
                h_new = project_sum(h_new, n/2.0)
                if best_s < best_score_local:
                    best_score_local = best_s
                    best_h_local = h_new
    
    return best_h_local, best_score_local

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15 * n:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_obj_fn(h, alpha):
    return smooth_max(compute_correlations(h), alpha)

def constraint_eq_fn(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac_fn(h):
    return np.ones(len(h))

def slsqp_opt(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            lambda x: smooth_max(compute_correlations(x), alpha),
            h0, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq_fn, 'jac': constraint_eq_jac_fn}],
            options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None
