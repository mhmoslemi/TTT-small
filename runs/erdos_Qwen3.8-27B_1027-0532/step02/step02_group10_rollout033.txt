import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = min(0.5, 1.0 / (alpha * dx * 3.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-13:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 150:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        improved = False
        s = step
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-14:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = s
            break
        step = min(s * 1.3, 1.0)
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.3, 1.0 / (alpha * dx * 5.0 + 1.0))
    prev_sm = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        if prev_sm is not None:
            ov_x = compute_overlaps(x_new)
            sm_x, _ = smooth_max_and_weights(ov_x, alpha)
            if sm_x > prev_sm + 1e-9:
                t_k = 1.0
            prev_sm = sm_x
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x


def iterative_lp_optimize(h, target, max_iters, deadline, top_k=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    no_improve_count = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        overlaps = compute_overlaps(h)
        K = min(top_k, len(overlaps))
        top_k_idx = np.argsort(-overlaps)[:K]

        A_ub_list = []
        b_ub_list = []

        for k_idx in top_k_idx:
            m = k_idx - (n - 1)
            grad = np.zeros(n)
            if m >= 0:
                if n - m > 0:
                    grad[:n - m] += (1.0 - h[m:])
                    grad[m:] -= h[:n - m]
            else:
                p = -m
                if n - p > 0:
                    grad[p:] += (1.0 - h[:n - p])
                    grad[:n - p] -= h[p:]

            const = overlaps[k_idx] - np.dot(grad, h)
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-const)

        for i in range(n):
            row = np.zeros(n + 1)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(1.0)

        for i in range(n):
            row = np.zeros(n + 1)
            row[i] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(0.0)

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        c = np.zeros(n + 1)
        c[n] = 1.0

        bounds = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break

        if not res.success or res.x is None:
            break

        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        h_lp = project_onto_constraint(h_lp, target)

        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_trial = (1.0 - alpha) * h + alpha * h_lp
            h_trial = project_onto_constraint(h_trial, target)
            score = evaluate_score(h_trial)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                no_improve_count = 0
                break

        if not improved:
            no_improve_count += 1
            h = (1.0 - 0.05) * h + 0.05 * h_lp
            h = project_onto_constraint(h, target)
            if no_improve_count > 50:
                break

    return best_h, best_score


def structured_perturbation(h, rng, n):
    mode = rng.randint(5)
    pert = h.copy()
    if mode == 0:
        L = rng.randint(n // 10, n // 3)
        start = rng.randint(0, n - L)
        pert[start:start + L] = 1.0 - pert[start:start + L]
    elif mode == 1:
        freq = rng.randint(2, n // 4)
        amp = rng.uniform(0.05, 0.2)
        pert += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
    elif mode == 2:
        shift = rng.randint(1, n // 5)
        pert = np.roll(h, shift)
    elif mode == 3:
        n_blocks = rng.randint(3, 8)
        block_size = n // n_blocks
        for b in range(n_blocks):
            if rng.rand() < 0.3:
                s = b * block_size
                e = min(s + block_size, n)
                pert[s:e] = 1.0 - pert[s:e]
    else:
        scale = rng.uniform(0.02, 0.1)
        pert += rng.randn(n) * scale

    return np.clip(pert, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    best_h = project_onto_constraint(init.copy(), target)
    best_score = evaluate_score(best_h)

    remaining = budget_s - (time.time() - t0)

    # Phase 1: Iterative LP (45% of budget)
    lp_budget = 0.45 * remaining
    dl = time.time() + lp_budget
    best_h, best_score = iterative_lp_optimize(best_h, target, 300, dl, top_k=100)
    best_h = project_onto_constraint(best_h, target)
    best_score = evaluate_score(best_h)

    # Phase 1b: Second round of LP from slightly perturbed points
    if time.time() - t0 < budget_s * 0.55:
        for trial in range(3):
            if time.time() - t0 > budget_s * 0.55:
                break
            pert_h = structured_perturbation(best_h, rng, n)
            pert_h = project_onto_constraint(pert_h, target)
            dl = time.time() + min(40, (budget_s * 0.55 - (time.time() - t0)) / max(1, 3 - trial))
            h_trial, s_trial = iterative_lp_optimize(pert_h, target, 100, dl, top_k=80)
            h_trial = project_onto_constraint(h_trial, target)
            s_trial = evaluate_score(h_trial)
            if s_trial < best_score:
                best_score = s_trial
                best_h = h_trial.copy()

    elapsed = time.time() - t0
    print("After LP phase: score={:.6f}, elapsed={:.1f}s".format(best_score, elapsed))

    # Phase 2: PGD/FISTA refinement (25% of budget)
    remaining2 = budget_s * 0.75 - (time.time() - t0)
    if remaining2 > 30:
        alphas_pgd = [100, 300, 800, 2000, 5000]
        per_alpha = max(5, remaining2 / len(alphas_pgd))
        for alpha in alphas_pgd:
            if time.time() - t0 > budget_s * 0.75:
                break
            dl = time.time() + min(per_alpha, 25)
            h = pgd_optimize(best_h, alpha, target, 6000, dl)
            h = project_onto_constraint(h, target)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

        if time.time() - t0 < budget_s * 0.75:
            for alpha in [200, 1000, 3000]:
                if time.time() - t0 > budget_s * 0.73:
                    break
                dl = time.time() + 15
                h = fista_optimize(best_h, alpha, target, 4000, dl)
                h = project_onto_constraint(h, target)
                s = evaluate_score(h)
                if s < best_score:
                    best_score = s
                    best_h = h.copy()

    elapsed = time.time() - t0
    print("After PGD phase: score={:.6f}, elapsed={:.1f}s".format(best_score, elapsed))

    # Phase 3: Random restarts with structured perturbations (15% of budget)
    if time.time() - t0 < budget_s * 0.90:
        n_restarts = 50
        per_restart = max(3.0, (budget_s * 0.90 - (time.time() - t0)) / n_restarts)
        for i in range(n_restarts):
            if time.time() - t0 > budget_s * 0.90:
                break
            pert = structured_perturbation(best_h, rng, n)
            pert = project_onto_constraint(pert, target)
            alpha = 300 + 700 * rng.randint(5)
            dl = time.time() + min(per_restart, 12)
            h = pgd_optimize(pert, alpha, target, 2000, dl)
            h = project_onto_constraint(h, target)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

            if i % 10 == 5:
                pert = structured_perturbation(best_h, rng, n)
                pert = project_onto_constraint(pert, target)
                dl2 = time.time() + 8
                h2, s2 = iterative_lp_optimize(pert, target, 30, dl2, top_k=60)
                h2 = project_onto_constraint(h2, target)
                s2 = evaluate_score(h2)
                if s2 < best_score:
                    best_score = s2
                    best_h = h2.copy()

    elapsed = time.time() - t0
    print("After restarts: score={:.6f}, elapsed={:.1f}s".format(best_score, elapsed))

    # Phase 4: Resolution increase (10% of budget)
    if time.time() - t0 < budget_s * 0.97:
        for n_try in [n + 100, 2 * n]:
            if n_try > 2000 or time.time() - t0 > budget_s * 0.96:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_onto_constraint(h_init, n_try / 2.0)
            for alpha in [200, 1000, 3000]:
                if time.time() - t0 > budget_s * 0.96:
                    break
                dl = time.time() + 15
                h = pgd_optimize(h_init, alpha, n_try / 2.0, 4000, dl)
                h = project_onto_constraint(h, n_try / 2.0)
                s = evaluate_score(h)
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
                    h_init = h.copy()

    # Phase 5: Final polish with high-alpha PGD
    if time.time() - t0 < budget_s - 10:
        for alpha in [8000, 20000]:
            if time.time() - t0 > budget_s - 8:
                break
            nf = len(best_h)
            dl = time.time() + 5
            h = pgd_optimize(best_h, alpha, nf / 2.0, 3000, dl)
            h = project_onto_constraint(h, nf / 2.0)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    elapsed = time.time() - t0
    print("Final: score={:.6f}, n={}, elapsed={:.1f}s".format(best_score, len(best_h), elapsed))
    return (best_h, best_score, len(best_h))
