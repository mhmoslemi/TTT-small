```python
def compute_corr_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-15:
            continue
        k = m - (n - 1)  # actual shift
        
        if k >= 0:
            # Part 1: (1-h[j-k]) for j in [k, n-1]
            if k < n:
                j_lo = k
                j_hi = n - 1
                idx = np.arange(j_lo, j_hi + 1) - k
                grad[j_lo:j_hi+1] += dx * wm * (1.0 - h[idx])
            # Part 2: -h[j+k] for j in [0, n-1-k]
            if k < n:
                j_lo2 = 0
                j_hi2 = n - 1 - k
                idx2 = np.arange(j_lo2, j_hi2 + 1) + k
                grad[j_lo2:j_hi2+1] -= dx * wm * h[idx2]
        else:
            kp = -k  # positive
            # Part 1: (1-h[j+kp]) for j in [0, n-1-kp]
            if kp < n:
                j_lo = 0
                j_hi = n - 1 - kp
                idx = np.arange(j_lo, j_hi + 1) + kp
                grad[j_lo:j_hi+1] += dx * wm * (1.0 - h[idx])
            # Part 2: -h[j-kp] for j in [kp, n-1]
            if kp < n:
                j_lo2 = kp
                j_hi2 = n - 1
                idx2 = np.arange(j_lo2, j_hi2 + 1) - kp
                grad[j_lo2:j_hi2+1] -= dx * wm * h[idx2]
    
    return grad
```
