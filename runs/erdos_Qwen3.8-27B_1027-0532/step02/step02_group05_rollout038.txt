import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def obj_grad_anti(g, alpha):
    h = antipal_h(g)
    N = len(h)
    dx = 2.0 / N
    corr = compute_correlations(h)
    corr_dx = corr * dx
    val = smooth_max_val(corr_dx, alpha)
    m = np.max(corr_dx)
    ex = np.exp(alpha * (corr_dx - m))
    p = ex / np.sum(ex)
    grad_h = np.zeros(N)
    for k in range(2 * N - 1):
        w = p[k]
        if w < 1e-14:
            continue
        kk = k - (N - 1)
        if kk >= 0:
            grad_h[kk:] += w * (1.0 - h[:N - kk])
            grad_h[:N - kk] -= w * h[kk:]
        else:
            kp = -kk
            grad_h[:N - kp] += w * (1.0 - h[kp:])
            grad_h[kp:] -= w * h[:N - kp]
    half = len(g)
    grad_g = grad_h[:half] - grad_h[half + 1:][::-1]
    return val, grad_g * dx

def lbfgsb_anti(g0, alpha, maxiter=1000):
    half = len(g0)

        return v

        _, g = obj_grad_anti(x, alpha)
        return g

                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0})
