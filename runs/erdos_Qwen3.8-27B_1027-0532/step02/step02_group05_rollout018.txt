    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def solve_1d_min_max(base, slopes):
    ts = np.linspace(0.0, 1.0, 49)
    vals = np.max(base[None, :] + slopes[None, :] * ts[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(len(ts) - 1, best_idx + 1)]
    for _ in range(60):
        mid = (lo + hi) / 2.0
        f_mid = np.max(base + slopes * mid)
        active = (base + slopes * mid) >= (f_mid - 1e-12)
        if not np.any(active):
            active = (base + slopes * mid) >= (f_mid - 1e-8)
        if not np.any(active):
            break
        min_s = float(np.min(slopes[active]))
        max_s = float(np.max(slopes[active]))
        if min_s >= -1e-12:
            hi = mid
        elif max_s <= 1e-12:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0


def cd_antipal(g, max_sweeps, t_start, time_limit, rng):
    m = len(g)
    N = 2 * m + 1
    best_g = g.copy()
    best_score = evaluate_score(antipal_h(g))
    ks = np.arange(-(N - 1), N)
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        h = antipal_h(best_g)
        corr = compute_correlations(h)
        max_change = 0.0
        order = rng.permutation(m)
        for i in order:
            N1i = N - 1 - i
            s = np.zeros(2 * N - 1)
            idx1 = i + ks
            mask1 = (idx1 >= 0) & (idx1 < N)
            s[mask1] += (1.0 - h[idx1[mask1]])
            idx2 = i - ks
            mask2 = (idx2 >= 0) & (idx2 < N)
            s[mask2] -= h[idx2[mask2]]
            idx3 = N1i + ks
            mask3 = (idx3 >= 0) & (idx3 < N)
            s[mask3] -= (1.0 - h[idx3[mask3]])
            idx4 = N1i - ks
            mask4 = (idx4 >= 0) & (idx4 < N)
            s[mask4] += h[idx4[mask4]]
            base = corr - s * best_g[i]
            t_new = solve_1d_min_max(base, s)
            t_new = float(np.clip(t_new, 0.0, 1.0))
            change = abs(t_new - best_g[i])
            max_change = max(max_change, change)
            best_g[i] = t_new
            h = antipal_h(best_g)
            corr = compute_correlations(h)
        if max_change < 1e-13:
            break
        if sweep % 50 == 0:
            sc = evaluate_score(antipal_h(best_g))
            if sc < best_score:
                best_score = sc
    h_final = antipal_h(best_g)
    score = evaluate_score(h_final)
    if score < best_score:
        best_score = score
    return best_g, best_score


def cd_full(h, max_sweeps, t_start, time_limit, rng):
    N = len(h)
    target = N / 2.0
    best_h = h.copy()

    ks = np.arange(-(N - 1), N)

            break
        corr = compute_correlations(best_h)
        max_change = 0.0

        for i in order:
            s = np.zeros(2 * N - 1)
            idx1 = i + ks
            mask1 = (idx1 >= 0) & (idx1 < N)
            s[mask1] += (1.0 - best_h[idx1[mask1]])
            idx2 = i - ks
            mask2 = (idx2 >= 0) & (idx2 < N)
            s[mask2] -= best_h[idx2[mask2]]
            base = corr - s * best_h[i]
            t_new = solve_1d_min_max(base, s)
            t_new = float(np.clip(t_new, 0.0, 1.0))

        best_score = score
    return best_h, best_score


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def slsqp_fun(x, alpha):
    return smooth_max_val(compute_correlations(x), alpha)


def slsqp_jac(x, alpha):
    return grad_obj(x, alpha)


def slsqp_eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def slsqp_eq_jac(x):
    return np.ones(len(x))


def slsqp_polish(h, alpha, maxiter):
    N = len(h)
    bounds = [(0.0, 1.0)] * N
    try:
        res = minimize(
            slsqp_fun, h, method='SLSQP', bounds=bounds,
            jac=slsqp_jac, args=(alpha,),
            constraints=[{'type': 'eq', 'fun': slsqp_eq_fun,
                          'jac': slsqp_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(np.clip(res.x, 0.0, 1.0), N / 2.0)
    except Exception:
        pass
    return h
