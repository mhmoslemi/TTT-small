        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

def active_set_gradient(h, overlaps, K, dx):
    """Compute gradient using only the top K active shifts"""
    n = len(h)
    m = np.max(overlaps)
    threshold = m - 0.01 / dx  # within 0.01 of max (in original units)
    # Actually let's use: top K by value
    idx = np.argsort(overlaps)[-K:]
    weights = np.zeros(len(overlaps))
    max_val = overlaps[idx[-1]]
    # Weight by how close to max
    for i in idx:
        weights[i] = (overlaps[i] / max_val) if max_val > 0 else 0
    weights /= weights.sum()
    return compute_gradient(h, weights, dx)

def pgd_with_active_set(h, target, max_iters, deadline, K=10, alpha=5000):
    """PGD with adaptive method: smooth max for most, active set for refinement"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = 0.3
    prev_score = None
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        score = np.max(overlaps)
        
        if prev_score is not None:
            if score < prev_score - 1e-12:
                stall = 0
            else:
                stall += 1
                if stall > 100:
                    break
        prev_score = score
        
        # Use smooth max for gradient
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        grad = compute_gradient(h, weights, dx)
        
        # Backtracking line search
        improved = False
        s = step
        for bt in range(20):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            if np.max(ov_t) < score - 1e-14:
                h = h_trial
                step = min(s * 1.5, 1.0)
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = max(s, 1e-10)
        # Occasionally reset alpha
        if it % 200 == 199:
            alpha *= 1.5
    return h

def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    L = alpha * dx * 3.0 + 1.0
    step = 1.0 / L
    prev_sm = None
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        
        # Check for increase, reset momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
        # Adaptive: increase alpha over time
        alpha *= 1.01
        step = 1.0 / (alpha * dx * 3.0 + 1.0)
    
    return x

        delta = new_score - current_score

            h = h_trial

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

            w_full /= w_full.sum()
            weights = w_full

        grad = compute_gradient(h, weights, dx)

        score_new = evaluate_score(x_new)
        if prev_score is not None:
            if score_new < prev_score - 1e-12:
                stalls = 0
            else:
                stalls += 1
                if stalls > 200:
                    break
        prev_score = score_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
        alpha *= 1.02
        step = 1.0 / (alpha * dx * 3.0 + 1.0)
    return x


def block_sim_anneal(h, target, deadline, rng, temp_start=0.004, temp_end=1e-7, max_block_frac=0.15):
    n = len(h)
    h = h.copy()
    current_score = evaluate_score(h)
    best_h = h.copy()
    best_score = current_score
    it = 0
    min_block = 3
    while time.time() < deadline:
        it += 1
        frac = min(it / 8000.0, 1.0)
        T = temp_start * (temp_end / temp_start) ** frac
        max_block = max(min_block, int(n * max_block_frac * (1.0 - frac) + min_block))
        bsz = rng.randint(min_block, max_block + 1)
        if bsz >= n:
            bsz = n // 4
        start = rng.randint(0, n - bsz + 1)
        noise_scale = 0.03 + 0.08 * (1.0 - frac)
        h_trial = h.copy()
        h_trial[start:start + bsz] += rng.randn(bsz) * noise_scale
        h_trial = project_onto_constraint(h_trial, target)
        new_score = evaluate_score(h_trial)
        delta = new_score - current_score
        if delta < -1e-15 or (T > temp_end * 1.5 and rng.random() < np.exp(-delta / T)):
            h = h_trial
            current_score = new_score
            if current_score < best_score - 1e-15:
                best_score = current_score
                best_h = h.copy()
    return best_h, best_score


def sga_optimize(h, target, deadline, rng, alpha, temp_start=0.003, noise_scale_base=2.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    current_score = evaluate_score(h)
    best_h = h.copy()
    best_score = current_score
    step = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 5000.0, 1.0)
        T = temp_start * (1e-5 / temp_start) ** frac
        overlaps = compute_overlaps(h)
        _, weights = smooth_max_and_weights(overlaps, alpha)
        grad = compute_gradient(h, weights, dx)
        noise = rng.randn(n) * T * noise_scale_base
        h_trial = project_onto_constraint(h - step * grad + noise, target)
        new_score = evaluate_score(h_trial)
        delta = new_score - current_score
        if delta < -1e-15 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_trial
            current_score = new_score
            step = min(step * 1.03, 0.5)
            if current_score < best_score - 1e-15:
                best_score = current_score
                best_h = h.copy()
        else:
            step = max(step * 0.95, 0.02)
    return best_h, best_score


def multistart_optimize(h, target, deadline, rng, n_starts=5, pgd_time_each=15):
    n = len(h)
    best_h = h.copy()
    best_score = evaluate_score(h)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        pert_scale = 0.02 + 0.04 * s
        pert = h + rng.randn(n) * pert_scale
        pert = project_onto_constraint(pert, target)
        alpha = 200 * (s + 1)
        dl = min(deadline, time.time() + pgd_time_each)
        h_trial = pgd_optimize(pert, alpha, target, 3000, dl, K=5)
        h_trial = project_onto_constraint(h_trial, target)
        sc = evaluate_score(h_trial)
        if sc < best_score:
            best_score = sc
            best_h = h_trial.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0

    best_h = project_onto_constraint(init.copy(), target)
    best_score = evaluate_score(best_h)

    # Phase 1: PGD sweep at original resolution (12% budget)
    deadline1 = t0 + budget_s * 0.12
    for alpha in [100, 500, 2000, 8000, 30000]:
        if time.time() > deadline1:
            break
        dl = min(deadline1, time.time() + 20)
        h = pgd_optimize(best_h, alpha, target, 5000, dl, K=5)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 2: FISTA at original resolution (8% budget)
    deadline2 = t0 + budget_s * 0.20
    if time.time() < deadline2:
        for alpha in [200, 1000, 5000]:
            if time.time() > deadline2:
                break
            h = fista_optimize(best_h, alpha, target, 4000, time.time() + min(15, deadline2 - time.time()))
            h = project_onto_constraint(h, target)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

    # Phase 3: Block SA at original resolution (15% budget)
    deadline3 = t0 + budget_s * 0.35
    if time.time() < deadline3:
        h, s = block_sim_anneal(best_h, target, deadline3, rng, temp_start=0.005)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 4: SGA at original resolution (10% budget)
    deadline4 = t0 + budget_s * 0.45
    if time.time() < deadline4:
        h, s = sga_optimize(best_h, target, deadline4, rng, alpha=2000, temp_start=0.003)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 5: Upsample to 2n and optimize (25% budget)
    deadline5 = t0 + budget_s * 0.70
    n2 = 2 * n
    if time.time() < deadline5:
        target2 = n2 / 2.0
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_onto_constraint(h2, target2)
        s2 = evaluate_score(h2)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()
        # Optimize at 2n
        for alpha in [200, 1000, 5000, 20000]:
            if time.time() > deadline5:
                break
            dl = min(deadline5, time.time() + 20)
            h2 = pgd_optimize(best_h, alpha, target2, 4000, dl, K=5)
            h2 = project_onto_constraint(h2, target2)
            s2 = evaluate_score(h2)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()
        # FISTA at 2n
        if time.time() < deadline5:
            h2 = fista_optimize(best_h, 3000, target2, 3000, deadline5)
            h2 = project_onto_constraint(h2, target2)
            s2 = evaluate_score(h2)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()

    # Phase 6: SA at best resolution (15% budget)
    deadline6 = t0 + budget_s * 0.85
    if time.time() < deadline6:
        nf = len(best_h)
        tf = nf / 2.0
        h, s = block_sim_anneal(best_h, tf, deadline6, rng, temp_start=0.002, max_block_frac=0.1)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 7: Try 3n upsample if time permits (8% budget)
    deadline7 = t0 + budget_s * 0.93
    n3 = 3 * n
    if time.time() < deadline7 and n3 <= 2000:
        target3 = n3 / 2.0
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n3, endpoint=False)
        h3 = np.interp(x_new, x_old, best_h)
        h3 = project_onto_constraint(h3, target3)
        for alpha in [500, 3000, 10000]:
            if time.time() > deadline7:
                break
            dl = min(deadline7, time.time() + 15)
            h3 = pgd_optimize(h3, alpha, target3, 3000, dl, K=3)
            h3 = project_onto_constraint(h3, target3)
            s3 = evaluate_score(h3)
            if s3 < best_score:
                best_score = s3
                best_h = h3.copy()

    # Phase 8: Final high-alpha polish (5% budget)
    deadline8 = t0 + budget_s * 0.97
    if time.time() < deadline8:
        nf = len(best_h)
        tf = nf / 2.0
        for alpha in [50000, 100000, 200000]:
            if time.time() > deadline8:
                break
            h = pgd_optimize(best_h, alpha, tf, 3000, time.time() + 8, K=3)
            h = project_onto_constraint(h, tf)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

    # Final multistart if time remains
    if time.time() < budget_s * 0.98:
        nf = len(best_h)
        tf = nf / 2.0
        h, s = multistart_optimize(best_h, tf, t0 + budget_s * 0.98, rng, n_starts=3, pgd_time_each=8)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
