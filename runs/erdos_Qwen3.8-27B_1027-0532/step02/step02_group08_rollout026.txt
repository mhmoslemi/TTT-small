def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n

def project_sum_box(h, target):

    h = np.asarray(h, dtype=float)
    if abs(np.sum(h) - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def exact_max_grad(h):
    corr = compute_correlations(h)
    k_star = np.argmax(corr)
    r = k_star - (len(h) - 1)
    n = len(h)
    grad = np.zeros(n)
    for m in range(n):
        val = 0.0
        if 0 <= m + r < n:
            val += (1.0 - h[m + r])
        if 0 <= m - r < n:
            val -= h[m - r]
        grad[m] = val
    return corr[k_star], grad

        _, grad = exact_max_grad(h_nest)
        v = momentum * v - lr * grad
        h_curr = h_curr + v
        h_curr = project_sum_box(h_curr, n / 2.0)
    return h_curr


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = np.clip(alpha * (corrs - m), -600, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def lbfgsb_step(h, alpha, maxiter, bounds):
    res = minimize(

        bounds=bounds, args=(alpha,), jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0,
                 'maxls': 50, 'maxfun': maxiter * 30}
    )
    return res.x


def topk_gradient(h, k_count):
    corr = compute_correlations(h)
    top_k = np.argsort(corr)[-k_count:]
    n = len(h)
    grad = np.zeros(n)
    for kk in top_k:
        r = kk - (n - 1)
        for m in range(n):
            val = 0.0
            if 0 <= m + r < n:
                val += (1.0 - h[m + r])
            if 0 <= m - r < n:
                val -= h[m - r]
            grad[m] += val
    return grad / k_count


def topk_nesterov(h0, n_steps, lr0, momentum, k_count=5):

        v = momentum * v - lr0 * grad
        h_curr = h_curr + v
        h_curr = project_sum_box(h_curr, n / 2.0)
    return h_curr
