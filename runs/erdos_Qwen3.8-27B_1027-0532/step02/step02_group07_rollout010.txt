def subgradient_optimize(h, alpha, n_iters=100, momentum=0.9, time_budget=30.0):
    """Projected subgradient with Nesterov momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    y = h.copy()  # Nesterov lookahead
    
    t0 = time.time()
    step = 1.0 / np.sqrt(n)
    
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        
        # Compute gradient at y
        grad, corr, max_corr = get_gradient(y, alpha)
        
        # Nesterov update
        v = momentum * v - step * grad
        y_new = y + v
        
        # Project to feasible set
        y_new = np.clip(y_new, 0.0, 1.0)
        y_new = project_sum(y_new, target)
        
        # Backtracking line search
        f_old = np.max(corr)
        f_new = np.max(compute_correlations(y_new))
        
        if f_new < f_old:
            h = y_new
            y = h + momentum * (y_new - h + step * grad)  # Hmm, this isn't quite right
            # Let me use a simpler scheme
            y = y_new
        else:
            step *= 0.5
            y = h + momentum * (y - h)  # revert
    
    return h

def sharpen(h, power):
    """Push values toward 0 or 1 using a power transform."""
    # h_new = h^p / (h^p + (1-h)^p) for p > 1
    # This maps [0,1] to [0,1] and pushes toward extremes
    h_safe = np.clip(h, 1e-10, 1 - 1e-10)
    hp = h_safe ** power
    (1-hp) = (1 - h_safe) ** power
    h_new = hp / (hp + (1 - h_safe) ** power)
    return h_new

def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            # Exact 1D minimization
            h_j = h[j]
            m = 2 * n - 1
            p = np.arange(m)
            
            idx1 = j + p - (n - 1)
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = j - p + (n - 1)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            slopes = np.zeros(m)
            if np.any(valid1):
                slopes[valid1] += (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                slopes[valid2] -= h[idx2[valid2]]
            
            a = corr - h_j * slopes
            
            # Binary search for optimal V
            zero_mask = (np.abs(slopes) < 1e-14)
            pos_mask = slopes > 1e-14
            neg_mask = slopes < -1e-14
            
            lo_V = float(np.max(np.minimum(a, a + slopes)))
            hi_V = float(np.max(corr))
            
            if lo_V >= hi_V - 1e-14:
                t_opt = h_j
            else:
                for _ in range(40):
                    mid_V = (lo_V + hi_V) * 0.5
                    feasible = True
                    if np.any(zero_mask):
                        if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                            feasible = False
                    if feasible:
                        lo_t = 0.0
                        if np.any(neg_mask):
                            lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
                        hi_t = 1.0
                        if np.any(pos_mask):
                            hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
                        if lo_t > hi_t + 1e-14:
                            feasible = False
                    
                    if feasible:
                        hi_V = mid_V
                    else:
                        lo_V = mid_V
                
                V_star = (lo_V + hi_V) * 0.5
                lo_t = 0.0
                hi_t = 1.0
                if np.any(neg_mask):
                    lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
                if np.any(pos_mask):
                    hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
                lo_t = max(lo_t, 0.0)
                hi_t = min(hi_t, 1.0)
                if lo_t > hi_t:
                    t_opt = h_j
                else:
                    t_opt = (lo_t + hi_t) * 0.5
            
            t_opt = float(np.clip(t_opt, 0.0, 1.0))
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    h = project_sum(h, n / 2.0)
    return h

**Alternative: Use the dual problem**

The primal is: min_h max_k f_k(h) s.t. sum(h) = n/2, 0≤h≤1

The Lagrangian dual (informal, since it's non-convex) would involve dual variables for each constraint f_k(h) ≤ t.

A "dual ascent" approach:
1. Maintain dual variables λ_k ≥ 0, sum λ_k = 1
2. At each iteration, solve: min_h sum_k λ_k * f_k(h) s.t. sum(h) = n/2, 0≤h≤1
3. Update λ: λ_k ← λ_k * exp(eta * f_k(h)) / Z (exponential weights)

Step 2 is minimizing a weighted sum of quadratic functions, which is a QP! This could be solved efficiently.

Actually, f_k(h) = sum_i h[i](1-h[i+k]) is a quadratic in h (specifically, it's linear in h minus a quadratic term). So sum_k λ_k f_k(h) is also quadratic in h. Minimizing it subject to linear constraints (sum = n/2, 0≤h≤1) is a convex QP if the quadratic form is positive semi-definite.

Let's check: f_k(h) = sum_i h[i] - sum_i h[i]h[i+k]
The quadratic part is -sum_i h[i]h[i+k], which is -h^T A_k h / 2 where A_k is a symmetric matrix.
The matrix A_k has A_k[i,j] = 1/2 if j-i = k or i-j = k (for the appropriate k).

The Hessian of sum_k λ_k f_k(h) is -sum_k λ_k A_k (plus the linear part doesn't contribute to the Hessian).

For this to be a minimization (convex), we need -sum_k λ_k A_k to be PSD, i.e., sum_k λ_k A_k to be NSD.

A_k is related to the autocorrelation matrix at lag k. For k=0, A_0 = I (identity). For k≠0, A_k is a shifted identity.

sum_k λ_k A_k = λ_0 I + sum_{k≠0} λ_k A_k

This is a circulant-like matrix. Its eigenvalues are related to the Fourier transform of the λ_k sequence. Specifically, if we define Λ(ω) = sum_k λ_k e^{ikω}, then the eigenvalues of sum_k λ_k A_k are Λ(2πm/n) for m=0,...,n-1 (approximately, for a circulant matrix).

For the QP to be convex, we need all eigenvalues of sum_k λ_k A_k to be ≤ 0, i.e., Λ(ω) ≤ 0 for all ω.

Since λ_k ≥ 0 and sum λ_k = 1, we have Λ(0) = sum λ_k = 1 > 0. So the matrix is NOT NSD in general, and the QP is NOT convex!

So this approach doesn't directly work. The problem is fundamentally non-convex.

However, we could still use this as a heuristic: at each iteration, solve the (potentially non-convex) QP using an interior point method or QP solver, and use the exponential weight update for the dual variables.

Actually, wait. The slope depends on the current values of h (specifically h[i+k] and h[i-k]). So when we update the block, the slopes for the next block change. This is fine for coordinate descent - we just recompute.

But there's a subtlety: the slope for h[i] at lag k is (1-h[i+k]) - h[i-k], which depends on h[i+k] and h[i-k]. If i+k or i-k is in the same block, then the "slope" actually depends on the variables we're optimizing. This breaks the linearity!

Let me check: if i and i+k are both in the block {start, ..., end-1}, then h[i+k] is a variable, and the term h[i]*(1-h[i+k]) is bilinear in the block variables. So the LP approach only works if the block elements don't interact with each other through the correlation.

For a block of size b, the correlation c[k] involves h[i] and h[i+k]. If both i and i+k are in the block, we get a quadratic term h[i]*h[i+k].

So the LP approach only works for "independent" blocks where no two elements in the block differ by a lag k that contributes to the correlation. But every pair of elements (i, j) with |i-j| < n contributes to some c[k].

So the LP approach doesn't work directly for blocks of size > 1 (unless the block is very small and we ignore the intra-block interactions, which would be an approximation).

For block size 1, it's exact (the 1D case we already have).

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
            constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
        )
        return res.x if res.x is not None else None
    except Exception:
        return None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(44):
        mid_V = (lo_V + hi_V) * 0.5

        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible and np.any(neg_mask):
            lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        elif feasible and np.any(pos_mask):
            hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=300, tol=1e-13, time_budget=60.0, order='forward'):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break

        if order == 'forward' or (order == 'zigzag' and sweep % 2 == 0):
            indices = range(n)
        elif order == 'reverse' or (order == 'zigzag' and sweep % 2 == 1):
            indices = range(n - 1, -1, -1)
        else:
            indices = list(range(n))
            np.random.shuffle(indices)

        max_change = 0.0
        for j in indices:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))

        if sweep % 25 == 24:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def projected_gradient_step(h, n_steps=200, step_scale=1.0, time_budget=30.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    t0 = time.time()
    step = step_scale / np.sqrt(n)

    for it in range(n_steps):
        if time.time() - t0 > time_budget:
            break

        corr = compute_correlations(h)
        max_c = np.max(corr)

        weights = np.exp(50.0 * (corr - max_c))
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        grad = np.zeros(n)
        for p in range(2 * n - 1):
            if weights[p] < 1e-14:
                continue
            k = p - (n - 1)
            wv = weights[p]
            if k >= 0:
                if n - 1 - k >= 0:
                    grad[:n - k] += wv * (1.0 - h[k:])
                if k < n:
                    grad[k:] -= wv * h[:n - k]
            else:
                ka = -k
                grad[ka:] += wv * (1.0 - h[:n - ka])
                grad[:n - ka] -= wv * h[ka:]

        grad *= dx

        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target)

        corr_new = compute_correlations(h_new)
        if np.max(corr_new) < max_c:
            h = h_new
            step = min(step * 1.05, step_scale)
        else:
            step *= 0.7

    return h

        h_pg = projected_gradient_step(best_h, n_steps=300, step_scale=0.3,
                                        time_budget=min(20, time_left() - 15))
        h_cd = coordinate_descent(h_pg, n_sweeps=300, tol=1e-14,
                                   time_budget=min(15, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 30:
        h_cd = coordinate_descent(best_h, n_sweeps=600, tol=1e-15,
                                   time_budget=min(25, time_left() - 5), order='zigzag')
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 20:
        n_final = len(best_h)
        if n_final <= 1500:
            bounds = [(0.0, 1.0)] * n_final
            for alpha in [20000, 100000, 500000]:
                if time_left() < 10:
                    break
                res_x = slsqp_optimize(best_h, alpha, 800, bounds)
                if res_x is not None:
                    h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s = evaluate_score(h_new)
                        if s < best_score:
                            best_score = s
                            best_h = h_new

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
