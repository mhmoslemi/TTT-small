    N = len(h)

    max_val = np.max(corr)
    # Find active lags

    grad = np.zeros(N)

        g = corr_grad_for_lag(h, k)

    return np.correlate(h, 1.0 - h, mode='full')

    return np.max(corr) * dx

def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_obj(h, alpha):

    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)

        if w < 1e-14:

    """Compute subgradient of exact max correlation and the max value."""
    N = len(h)
    corr = compute_correlations(h)
    max_val = float(np.max(corr))
    active = np.where(corr >= max_val - epsilon)[0]
    grad = np.zeros(N)
    for idx in active:
        k = int(idx) - (N - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:N - k])
            grad[:N - k] -= h[k:]
        else:
            kp = -k
            grad[:N - kp] += (1.0 - h[kp:])
            grad[kp:] -= h[:N - kp]
    grad /= float(len(active))
    return grad, max_val

        if res_x is not None and np.all(np.isfinite(res_x)):
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), len(best_h) / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 7: Final subgradient polish
    if elapsed() < budget_s - 20:
        h_new, s = subgrad_descent(best_h, len(best_h) / 2.0, 2000, 1e-6, 0.2 / len(best_h), True)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
