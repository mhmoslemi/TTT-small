def coordinate_descent(h, n_sweeps=100, tol=1e-10, time_budget=60.0, use_pairs=False):

    h = h.copy()
    corr = compute_correlations(h)

        if time.time() - t0 > time_budget:
            break

        if use_pairs and n > 2:

                change_j = t_opt_j - h[j]

                    corr = corr + change_j * slopes_j
                    max_change = max(max_change, abs(change_j))
                t_opt_k, slopes_k = optimize_coordinate(h, j + 1, corr)
                change_k = t_opt_k - h[j + 1]
                if abs(change_k) > 1e-13:
                    h[j + 1] = t_opt_k

                    max_change = max(max_change, abs(change_k))

                change = t_opt - h[n - 1]
                if abs(change) > 1e-13:
                    h[n - 1] = t_opt

                change = t_opt - h[j]

            corr = compute_correlations(h)
        if max_change < tol:

def optimize_pair_2d(h, j, k, corr):
    """Optimize h[j] and h[k] simultaneously using a 2D search."""
    n = len(h)
    m = 2 * n - 1
    p = np.arange(m)
    
    # Slopes for j
    idx1j = j + p - (n - 1)
    v1j = (idx1j >= 0) & (idx1j < n)
    idx2j = j - p + (n - 1)
    v2j = (idx2j >= 0) & (idx2j < n)
    slopes_j = np.zeros(m)
    if np.any(v1j):
        slopes_j[v1j] += (1.0 - h[idx1j[v1j]])
    if np.any(v2j):
        slopes_j[v2j] -= h[idx2j[v2j]]
    
    # Slopes for k
    idx1k = k + p - (n - 1)
    v1k = (idx1k >= 0) & (idx1k < n)
    idx2k = k - p + (n - 1)
    v2k = (idx2k >= 0) & (idx2k < n)
    slopes_k = np.zeros(m)
    if np.any(v1k):
        slopes_k[v1k] += (1.0 - h[idx1k[v1k]])
    if np.any(v2k):
        slopes_k[v2k] -= h[idx2k[v2k]]
    
    # corr_new[t] = corr[t] + dx * slopes_j[t] + dy * slopes_k[t]
    # where dx = h[j]_new - h[j], dy = h[k]_new - h[k]
    
    a = corr - h[j] * slopes_j - h[k] * slopes_k
    # corr_new = a + x * slopes_j + y * slopes_k where x=h[j], y=h[k]
    
    # We want to minimize max_t (a[t] + x*slopes_j[t] + y*slopes_k[t])
    # subject to 0<=x<=1, 0<=y<=1
    
    # Grid search: 20x20 grid
    best_V = np.inf
    best_x, best_y = h[j], h[k]
    
    x_vals = np.linspace(0, 1, 20)
    y_vals = np.linspace(0, 1, 20)
    
    for x in x_vals:
        for y in y_vals:
            V = np.max(a + x * slopes_j + y * slopes_k)
            if V < best_V:
                best_V = V
                best_x, best_y = x, y
    
    # Refine around best with finer grid
    x_lo = max(0, best_x - 1/20)
    x_hi = min(1, best_x + 1/20)
    y_lo = max(0, best_y - 1/20)
    y_hi = min(1, best_y + 1/20)

    y_vals2 = np.linspace(y_lo, y_hi, 10)

        for y in y_vals2:
            V = np.max(a + x * slopes_j + y * slopes_k)
            if V < best_V:
                best_V = V
                best_x, best_y = x, y
    
    return best_x, best_y


def slsqp_optimize(h0, alpha, maxiter, bounds):
    def obj(h):
        return smooth_max(compute_correlations(h), alpha)
    
    def obj_jac(h):
        corr = compute_correlations(h)
        m = 2 * len(h) - 1
        mvals = np.argmax(corr)
        # Subgradient of smooth max
        weights = np.exp(alpha * (corr - corr[mvals]))
        weights /= np.sum(weights)
        # d(corr[k])/d(h[j]) for each k
        # This is expensive, so we use numerical gradient
        return None  # Let SLSQP use numerical

        obj, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - len(h)/2.0}],
        options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
    )
    return res.x if res.x is not None else None

    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=100, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 25 == 24:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def slsqp_objective(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_objective, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def subgradient_step(h, eta):
    corr = compute_correlations(h)
    k_star = int(np.argmax(corr))
    shift = k_star - (len(h) - 1)
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j + shift < n:
            val += (1.0 - h[j + shift])
        if 0 <= j - shift < n:
            val -= h[j - shift]
        grad[j] = val
    h_new = h - eta * grad
    h_new = project_sum(h_new, n / 2.0)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_sum(init, n / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: Quick CD from initial
    if time_left() > 40:
        h_cd = coordinate_descent(best_h, n_sweeps=150, time_budget=min(35, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: Subgradient + CD hybrid
    if time_left() > 60:
        h_sg = best_h.copy()
        t_start = time.time()
        eta = 0.01
        for it in range(200):
            if time.time() - t_start > min(40, time_left() - 20):
                break
            h_sg = subgradient_step(h_sg, eta)
            if it % 10 == 9:
                h_sg = coordinate_descent(h_sg, n_sweeps=10, time_budget=2.0)
                eta *= 0.9
        s = evaluate_score(h_sg)
        if s < best_score:
            best_score = s
            best_h = h_sg

    # Phase 3: Simulated annealing with diverse perturbations
    n_cur = len(best_h)
    n_anneal = 150
    T_start = 0.003
    T_end = 0.00005
    for i in range(n_anneal):
        if time_left() < 25:
            break
        frac = i / max(n_anneal - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        h_pert = best_h.copy()
        pert_type = rng.randint(0, 6)

        if pert_type == 0:
            scale = 0.015 + 0.06 * (1.0 - frac)
            h_pert += rng.randn(n_cur) * scale
        elif pert_type == 1:
            n_swap = rng.randint(1, 4)
            for _ in range(n_swap):
                i1, i2 = rng.choice(n_cur, 2, replace=False)
                h_pert[i1], h_pert[i2] = h_pert[i2], h_pert[i1]
        elif pert_type == 2:
            b = rng.randint(2, 10)
            start = rng.randint(0, n_cur - b)
            h_pert[start:start + b] = 1.0 - h_pert[start:start + b]
        elif pert_type == 3:
            for _ in range(rng.randint(1, 5)):
                j = rng.randint(n_cur)
                h_pert[j] = rng.uniform(0.0, 1.0)
        elif pert_type == 4:
            corr = compute_correlations(best_h)
            k_worst = int(np.argmax(corr))
            shift = k_worst - (n_cur - 1)
            n_pert = rng.randint(2, 8)
            for _ in range(n_pert):
                j = rng.randint(n_cur)
                if 0 <= j + shift < n_cur:
                    h_pert[j] = max(0.0, h_pert[j] - rng.uniform(0.02, 0.1))
                elif 0 <= j - shift < n_cur:
                    h_pert[j] = min(1.0, h_pert[j] + rng.uniform(0.02, 0.1))
        else:
            b = rng.randint(3, 20)
            i1 = rng.randint(0, n_cur - b)
            i2 = rng.randint(0, n_cur - b)
            if abs(i1 - i2) > b:
                tmp = h_pert[i1:i1 + b].copy()
                h_pert[i1:i1 + b] = h_pert[i2:i2 + b]
                h_pert[i2:i2 + b] = tmp

        h_pert = project_sum(h_pert, n_cur / 2.0)
        tb = min(2.5, max(0.5, (time_left() - 20) / max(n_anneal - i, 1)))
        h_cd = coordinate_descent(h_pert, n_sweeps=40, time_budget=tb)
        s = evaluate_score(h_cd)

        if s < best_score:
            best_score = s
            best_h = h_cd
        elif rng.random() < np.exp(-(s - best_score) / T):
            best_h = h_cd

    # Phase 4: Coarse-to-fine multi-resolution
    n_cur = len(best_h)
    half_n = max(50, n_cur // 2)
    if time_left() > 80 and half_n >= 50:
        x_full = np.linspace(0.0, 2.0, n_cur, endpoint=False)
        x_half = np.linspace(0.0, 2.0, half_n, endpoint=False)
        h_half = np.interp(x_half, x_full, best_h)
        h_half = project_sum(h_half, half_n / 2.0)
        h_half = coordinate_descent(h_half, n_sweeps=100, time_budget=min(20, time_left() - 30))

        x_full2 = np.linspace(0.0, 2.0, n_cur, endpoint=False)
        h_interp = np.interp(x_full2, x_half, h_half)
        h_interp = project_sum(h_interp, n_cur / 2.0)
        h_interp = coordinate_descent(h_interp, n_sweeps=80, time_budget=min(25, time_left() - 20))
        s = evaluate_score(h_interp)
        if s < best_score:
            best_score = s
            best_h = h_interp

    # Phase 5: Higher resolution
    for mult in [1.5, 2.0]:
        n_try = int(n_cur * mult)
        if n_try > 2000 or n_try < 100:
            continue
        if time_left() < 60:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)
        tb_cd = min(35, time_left() - 30)
        h_cd = coordinate_descent(h_interp, n_sweeps=150, time_budget=tb_cd)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 6: More perturbations from current best
    n_cur = len(best_h)
    n_perturb2 = 80
    for i in range(n_perturb2):
        if time_left() < 15:
            break
        frac = i / max(n_perturb2 - 1, 1)
        h_pert = best_h.copy()
        scale = 0.005 + 0.04 * (1.0 - frac)
        strategy = i % 4

        if strategy == 0:
            h_pert += rng.randn(n_cur) * scale
        elif strategy == 1:
            n_mod = rng.randint(2, 6)
            for _ in range(n_mod):
                j = rng.randint(n_cur)
                h_pert[j] = 1.0 - h_pert[j]
        elif strategy == 2:
            b = rng.randint(2, 8)
            start = rng.randint(0, n_cur - b)
            shift_amt = rng.uniform(-0.1, 0.1)
            h_pert[start:start + b] += shift_amt
        else:
            corr = compute_correlations(best_h)
            active = np.where(corr > 0.95 * np.max(corr))[0]
            for ak in active[:3]:
                shift = ak - (n_cur - 1)
                for j in range(0, n_cur, rng.randint(5, 20)):
                    if 0 <= j + shift < n_cur:
                        h_pert[j] = max(0.0, min(1.0, h_pert[j] - 0.03))

        h_pert = project_sum(h_pert, n_cur / 2.0)
        tb = min(2.0, max(0.3, (time_left() - 10) / max(n_perturb2 - i, 1)))
        h_cd = coordinate_descent(h_pert, n_sweeps=30, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 7: SLSQP polish with multiple alphas
    if time_left() > 25:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [300, 1000, 5000, 20000]:
            if time_left() < 8:
                break
            maxiter = max(80, int(300000.0 / n_final))
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 8: Final intensive CD
    if time_left() > 15:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(50, time_left() - 5))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 9: One more SLSQP + CD cycle if time allows
    if time_left() > 20:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        res_x = slsqp_optimize(best_h, 50000, max(100, int(200000.0 / n_final)), bounds)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
            if np.all(np.isfinite(h_new)):
                h_new = coordinate_descent(h_new, n_sweeps=100, time_budget=min(15, time_left() - 5))
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
