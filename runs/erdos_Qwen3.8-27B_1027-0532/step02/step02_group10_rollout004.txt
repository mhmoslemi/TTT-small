            continue
        offset = m - (n - 1)

        i_end = min(n, n - offset)
        if i_start < i_end:

Let me try yet another angle: **Use the fact that the problem is a game.**

We can view this as a zero-sum game:
- Player 1 chooses a shift k
- Player 2 chooses h
- Payoff is f(k, h) = ∫ h(x)(1-h(x+k)) dx

Player 1 wants to maximize, Player 2 wants to minimize.

The value of the game is C₅. By von Neumann's minimax theorem (if we relax to mixed strategies), the value is:
min_{h} max_k f(k, h) = max_{p} min_h E_p[f(k, h)]

where p is a distribution over shifts. The second form is:
max_p min_h ∫ p(k) f(k, h) dk

This is convex in p (for fixed h) and... well, non-convex in h (bilinear).

But we can use **mirror descent** or **multiplicative weights** on the p side:
1. Initialize p uniformly
2. Given p, find h* = argmin_h ∫ p(k) f(k, h) dk (this is still non-convex in h, but the objective is a weighted sum of bilinear forms, which is a quadratic form)
3. Update p: p(k) ∝ exp(η * f(k, h*))
4. Repeat

Step 2 is the hard part. ∫ p(k) f(k, h) dk = ∫ h(x) dx - ∫ h(x) [∫ p(k) h(x+k) dk] dx = 1 - ∫ h(x) g(x) dx

where g(x) = ∫ p(k) h(x+k) dk is the "blurred" version of h.

So we're minimizing 1 - ∫ h(x) g(x) dx, which is equivalent to maximizing ∫ h(x) g(x) dx.

But g depends on h! So it's still non-convex.

However, if we **freeze g** (use the current h to compute g, then optimize h with g fixed), we get a linear problem:
maximize ∫ h(x) g(x) dx subject to 0 ≤ h ≤ 1, ∫h = 1

The solution is: h(x) = 1 where g(x) is largest (top n/2 values), 0 elsewhere.

   b) Optimize h to minimize F_p(h) using Nesterov (this is a smooth, non-convex problem)
   c) Update p(k) ∝ p(k) * exp(η * f(k, h))
3. At the end, the max overlap is what we report

The advantage of this approach is that the inner optimization (step b) is a smooth problem (no max to approximate), which might be easier to optimize.

def multiplicative_weights_optimize(h, n_iter_p, n_inner, eta, target, deadline):

    K = 2 * n - 1  # number of shifts
    p = np.ones(K) / K
    h = h.copy()
    
    for it in range(n_iter_p):
        if time.time() > deadline:
            break

        ov = compute_overlaps(h)
        F = np.dot(p, ov)

        # This is equivalent to smooth max with uniform "focus" on all shifts weighted by p
        # We can use Nesterov with a custom objective
        
        # The gradient of F_p(h) is: sum_k p(k) * grad f(k, h)
        # We can compute this directly
        dx = 2.0 / n
        step = 1.0 / (4.0 * dx)
        x = h.copy()
        
        for inner in range(n_inner):
            if time.time() > deadline:
                break
            # Compute gradient of F_p
            grad = np.zeros(n)
            for m in range(K):
                pm = p[m]
                if pm < 1e-15:
                    continue
                offset = m - (n - 1)
                i_start = max(0, -offset)
                i_end = min(n, n - offset)
                if i_start < i_end:
                    grad[i_start:i_end] += pm * dx * (1.0 - x[i_start + offset:i_end + offset])
                i_start2 = max(0, offset)
                i_end2 = min(n, n + offset)
                if i_start2 < i_end2:
                    grad[i_start2:i_end2] -= pm * dx * x[i_start2 - offset:i_end2 - offset]
            
            # Backtracking
            s = step
            improved = False
            for bt in range(30):

                ov_new = compute_overlaps(x_new)
                F_new = np.dot(p, ov_new)
                if F_new < F - 1e-16:
                    h = x_new
                    F = F_new
                    improved = True
                    break
                s *= 0.5
                if s < 1e-14:
                    break
            if not improved:
                break
            step = min(s * 1.5, 1.0 / (2.0 * dx))
        
        # Step c: Update p
        ov = compute_overlaps(h)
        p = p * np.exp(eta * (ov - np.max(ov)))
        p = np.maximum(p, 1e-20)
        p /= p.sum()
    
    return h, p

Specifically:
F_p(h) = ∑_k p(k) [∫ h(x) dx - ∫ h(x)h(x+k) dx]
= 1 - ∑_k p(k) ∫ h(x)h(x+k) dx
= 1 - h^T M h (in discrete form)

where M is a matrix that encodes the weighted autocorrelation.

So we want to maximize h^T M h subject to 0 ≤ h ≤ 1 and 1^T h = n/2.

This is a **quadratic programming problem** (maximize a quadratic subject to linear constraints). If M is positive semidefinite, this is convex (equivalently, minimize -h^T M h is convex). But M might not be PSD.

However, we can still use projected gradient or other methods to solve this QP efficiently.

This is h^T A h where A is a banded matrix (bandwidth depends on the shifts with non-zero p).

If p is concentrated on a few shifts, A is very banded, and the QP can be solved efficiently.

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_start = max(0, -offset)
        i_end = min(n, n - offset)
        if i_start < i_end:
            grad[i_start:i_end] += w * dx * (1.0 - h[i_start + offset:i_end + offset])
        i_start2 = max(0, offset)
        i_end2 = min(n, n + offset)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= w * dx * h[i_start2 - offset:i_end2 - offset]
    return grad
