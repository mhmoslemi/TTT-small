2. If f(0) ≤ f(1) and the right derivative at 0 ≤ 0, minimum is at 0
3. If f(1) ≤ f(0) and the left derivative at 1 ≥ 0, minimum is at 1
4. Otherwise, binary search for the point where the subgradient crosses 0

The right derivative at 0 is: max{a_L : b_L = f(0)} (the maximum slope among active constraints)
The left derivative at 1 is: min{a_L : a_L + b_L = f(1)} (the minimum slope among active constraints)

For the binary search: at point t, the subgradient is [min{a_L : active}, max{a_L : active}].
If max(active a_L) < 0, the function is decreasing, so increase t.
If min(active a_L) > 0, the function is increasing, so decrease t.
Otherwise, 0 is in the subgradient and t is optimal.

2. i = m. Then the term is h[m + j - (n-1)] * (1-h[m]).
   Contribution: h[m+j-(n-1)] * (1-h[m]) = h[m+j-(n-1)] - h[m+j-(n-1)]*h[m]

So the total contribution of h[m] to corr[j] is:
[1-h[m-j+(n-1)]] * h[m] * I(0≤m-j+(n-1)<n) - h[m+j-(n-1)] * h[m] * I(0≤m+j-(n-1)<n)

= h[m] * { (1-h[m-j+(n-1)]) * I(0≤m-j+(n-1)<n) - h[m+j-(n-1)] * I(0≤m+j-(n-1)<n) }

Let L = j - (n-1) (the lag). Then:
m - j + (n-1) = m - L
m + j - (n-1) = m + L

So a_L = (1-h[m-L]) * I(0≤m-L<n) - h[m+L] * I(0≤m+L<n)

And corr[j] = a_L * h[m] + b_L where b_L = corr[j] - a_L * h[m]

This matches what I had before. Good.

Now, one issue: when I compute `corr = compute_correlations(h)` at the start of each sweep, and then modify h[m] during the sweep, the `corr` values become stale for subsequent coordinates. For coordinate descent, the standard approach is to recompute the objective/gradient after each coordinate update. But that would be O(n^2) per coordinate (to recompute all correlations), making the total O(n^3) per sweep.

The alternative is to update the correlations incrementally. When h[m] changes by δ, all correlations change by:
Δcorr[j] = δ * a_L(j) (where a_L(j) is the coefficient we computed)

def coordinate_descent(h, n_sweeps):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)  # current correlations
    
    for sweep in range(n_sweeps):
        for m in range(n):
            hm = h[m]
            L = np.arange(-(n-1), n)
            m_plus_L = m + L
            m_minus_L = m - L
            valid_plus = (m_plus_L >= 0) & (m_plus_L < n)
            valid_minus = (m_minus_L >= 0) & (m_minus_L < n)
            
            a = np.zeros(2*n - 1)
            a[valid_plus] = 1.0 - h[m_plus_L[valid_plus]]
            a[valid_minus] -= h[m_minus_L[valid_minus]]
            
            b = corr - a * hm
            t_new = minimize_1d_convex(a, b)
            
            # Update correlations
            delta = t_new - hm
            if abs(delta) > 1e-15:
                corr += delta * a
                h[m] = t_new
        
        h = project_sum(h, n / 2.0)
        corr = compute_correlations(h)  # recompute after projection
    
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def penalty_obj_grad(h, alpha, mu):
    obj, grad = smooth_max_obj_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad


def minimize_1d_convex(a, b):
    f0 = np.max(b)
    active0 = (b >= f0 - 1e-12)
    if np.max(a[active0]) >= 0:
        return 0.0
    f1 = np.max(a + b)
    active1 = (a + b >= f1 - 1e-12)
    if np.min(a[active1]) <= 0:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        val = a * mid + b
        f_mid = np.max(val)
        active = (val >= f_mid - 1e-12)
        min_d = np.min(a[active])
        max_d = np.max(a[active])
        if min_d > 0:
            hi = mid
        elif max_d < 0:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0


def coordinate_descent(h, n_sweeps):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    L_range = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        for m in range(n):
            hm = h[m]
            m_plus_L = m + L_range
            m_minus_L = m - L_range
            valid_plus = (m_plus_L >= 0) & (m_plus_L < n)
            valid_minus = (m_minus_L >= 0) & (m_minus_L < n)
            a = np.zeros(2 * n - 1)
            a[valid_plus] = 1.0 - h[m_plus_L[valid_plus]]
            a[valid_minus] -= h[m_minus_L[valid_minus]]
            b = corr - a * hm
            t_new = minimize_1d_convex(a, b)
            delta = t_new - hm
            if abs(delta) > 1e-15:
                corr += delta * a
                h[m] = t_new
        h = project_sum(h, n / 2.0)
        corr = compute_correlations(h)
    return h


def lbfgsb_optimize(h0, alpha, mu, maxiter, bounds):
    res = minimize(
        penalty_obj_grad, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu), jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0,
                 'maxls': 50, 'maxfun': maxiter * 20}
    )
    return res.x
