import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(300):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad_full(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad

def penalty_obj_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad_full(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

def slsqp_obj_fun(h, alpha):
    return lse_val(compute_overlaps(h), alpha)

def slsqp_grad_fun(h, alpha):
    return lse_grad_full(h, alpha)

def sum_con_fun(h):
    return np.sum(h) - len(h) / 2.0

def sum_con_jac(h):
    return np.ones(len(h))

def lbfgsb_run(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(penalty_obj_grad, h0, args=(alpha, penalty),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16})
        return res.x
    except Exception:
        return None

def slsqp_run(h0, alpha, maxiter, bounds):
    try:
        res = minimize(slsqp_obj_fun, h0, args=(alpha,),
                       method='SLSQP', jac=slsqp_grad_fun, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return res.x
    except Exception:
        return None

def binary_swap_search(h0, t0, time_limit, rng, n_cand=80, tabu_len=200):

    h = (h0 >= 0.5).astype(float)
    n_target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > n_target:
        ones_idx = np.where(h == 1)[0]
        rem = ones_idx[rng.choice(len(ones_idx), size=n_ones - n_target, replace=False)]
        h[rem] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h == 0)[0]
        add = zeros_idx[rng.choice(len(zeros_idx), size=n_target - n_ones, replace=False)]
        h[add] = 1

    c = compute_overlaps(h).astype(float)
    best_max = float(np.max(c))
    best_h = h.copy()

    h_ext = np.zeros(2 * n, dtype=np.float64)
    h_ext[n - 1:n - 1 + n] = h

    temp = 0.5
    tabu_list = []
    tabu_set = set()
    it = 0
    stagnation = 0
    eval_count = 0

    while time.time() - t0 < time_limit:
        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break

        K = min(n_cand, len(ones_arr) * len(zeros_arr))
        if K < 2:
            break

        a_arr = ones_arr[rng.randint(len(ones_arr), size=K)]
        b_arr = zeros_arr[rng.randint(len(zeros_arr), size=K)]

        best_new_max = best_max + 1.0
        best_ab = (-1, -1)

            b = int(b_arr[ki])
            key = (min(a, b), max(a, b))
            if key in tabu_set:

            delta = (h_ext[n + a + 1:n + a + n] + h_ext[a + 1:n + a + 1][::-1]
                     - h_ext[n + b + 1:n + b + n] - h_ext[b + 1:n + b + 1][::-1])

            new_max_pos = float(np.max(c[n:] + delta))
            new_max_neg = float(np.max(c[:n - 1] + delta[::-1]))

                best_delta = delta
                best_ab = (a, b)

            tabu_list = []

            h[b] = 1
            h_ext[n - 1 + a] = 0
            h_ext[n - 1 + b] = 1
            c[n:] += delta
            c[:n - 1] += delta[::-1]

            if best_new_max < best_max:
                best_max = best_new_max
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1

            key = (min(a, b), max(a, b))

            tabu_list.append(key)
            if len(tabu_list) > tabu_len:

            c[:] = compute_overlaps(h).astype(float)
            h_ext[:] = 0
            h_ext[n - 1:n - 1 + n] = h
            best_max = float(np.max(c))
            stagnation = 0

    bounds = [(0.0, 1.0)] * n

    alpha_list = [300, 1000, 3000, 8000, 20000]
    for alpha in alpha_list:
        if time.time() - t0 > time_limit:
            break
        penalty = alpha * 20.0
        maxiter = max(80, 120000 // n)
        res_x = lbfgsb_run(h, alpha, penalty, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    if time.time() - t0 < time_limit - 10:
        res_x = slsqp_run(best_h, 5000, 300, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_score = evaluate_score(h_parent)

    def time_left():
        return budget_s - 15.0 - (time.time() - t0)

    def update_best(h_new, score_new):
        nonlocal_best = [best_h, best_score]
        return h_new, score_new

    # Phase 1: Binary 2-swap search at n=424
    n_bin = 424
    if time_left() > 120:
        h_bin_init = np.interp(
            np.linspace(0, 2, n_bin, endpoint=False),
            np.linspace(0, 2, n_parent, endpoint=False),
            h_parent
        )

        phases_binary = [
            (h_bin_init, 0.25, 42),
            (make_random_binary(n_bin, rng), 0.20, 123),
            (make_structured_binary(n_bin, rng, 777), 0.15, 999),
            (1.0 - h_bin_init, 0.20, 55),
            (make_random_binary(n_bin, np.random.RandomState(seed + 1)), 0.20, 2024),
        ]

        for init_h, frac, s_seed in phases_binary:
            if time_left() < 30:
                break
            t_limit = time_left() * frac * 0.9
            r_local = np.random.RandomState(s_seed)
            h_candidate, s_candidate = binary_swap_search(
                init_h, t0, t0 + t_limit, r_local, n_cand=100, tabu_len=300
            )
            if s_candidate < best_score:
                best_score = s_candidate
                best_h = h_candidate.copy()

    # Phase 2: Binary search at n=parent (425) with continuous values
    if time_left() > 60:
        t_limit = time_left() * 0.25
        h_cand, s_cand = binary_swap_search(
            best_h if len(best_h) != n_parent else h_parent,
            t0, t0 + t_limit,
            np.random.RandomState(seed + 42),
            n_cand=80, tabu_len=200
        )
        if s_cand < best_score:
            best_score = s_cand
            best_h = h_cand.copy()

    # Phase 3: Continuous polishing
    if time_left() > 60:
        t_limit = time_left() * 0.35
        h_polished, s_polished = continuous_polish(best_h, t0, t0 + t_limit)
        if s_polished < best_score:
            best_score = s_polished
            best_h = h_polished.copy()

    # Phase 4: Try the parent directly with continuous optimization
    if time_left() > 60:
        t_limit = time_left() * 0.3
        h_pol2, s_pol2 = continuous_polish(h_parent, t0, t0 + t_limit)
        if s_pol2 < best_score:
            best_score = s_pol2
            best_h = h_pol2.copy()

    # Phase 5: Multi-resolution upsampling
    if time_left() > 90:
        n_up = len(best_h) * 2
        if n_up <= 2000:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_box_sum(h_up, n_up / 2.0)
            t_limit = time_left() * 0.4
            h_up_pol, s_up = continuous_polish(h_up, t0, t0 + t_limit)
            if s_up < best_score:
                best_score = s_up
                best_h = h_up_pol.copy()

    # Phase 6: Final high-precision refinement
    if time_left() > 20:
        n_f = len(best_h)
        h_f = project_box_sum(best_h.copy(), n_f / 2.0)
        bounds_f = [(0.0, 1.0)] * n_f
        for alpha_f in [10000, 30000]:
            if time_left() < 10:
                break
            pen_f = alpha_f * 50.0
            res_x = lbfgsb_run(h_f, alpha_f, pen_f, 500, bounds_f)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_f / 2.0)
                s_new = evaluate_score(h_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_h = h_new.copy()
                    h_f = h_new.copy()

    # Final projection and validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
