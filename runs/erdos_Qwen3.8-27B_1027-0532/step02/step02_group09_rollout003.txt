import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    lo = -2.0
    hi = 3.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val_and_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = float(np.max(overlaps))
    shifted = overlaps - m
    exp_vals = np.exp(alpha * shifted)
    wsum = float(np.sum(exp_vals))
    weights = exp_vals / wsum
    val = m + np.log(wsum) / alpha
    grad = np.zeros(n)
    threshold = 1e-7
    sig_idx = np.where(weights > threshold)[0]
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return val, grad


def penalty_obj_func(h, alpha, penalty):
    val, grad = lse_val_and_grad(h, alpha)
    s = float(np.sum(h)) - len(h) / 2.0
    return val + penalty * s * s

    val, grad = lse_val_and_grad(h, alpha)
    s = float(np.sum(h)) - len(h) / 2.0
    return val, grad + 2.0 * penalty * s


def slsqp_obj_func(h, alpha):
    val, _ = lse_val_and_grad(h, alpha)
    return val


def slsqp_grad_func(h, alpha):
    _, grad = lse_val_and_grad(h, alpha)
    return grad


def sum_eq_fun(h):
    return np.sum(h) - len(h) / 2.0


def sum_eq_jac(h):
    return np.ones(len(h))


def lbfgsb_run(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_obj_func, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=penalty_obj_grad, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14, 'maxls': 50}
        )
        return res.x
    except Exception:
        return None


def slsqp_run(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj_func, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad_func, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def polyak_subgrad_step(h, alpha, target, lr_scale):
    n = len(h)
    val, grad = lse_val_and_grad(h, alpha)
    gnorm2 = float(np.dot(grad, grad))
    if gnorm2 < 1e-20:
        return h, val
    lr = lr_scale * val / (gnorm2 + 1e-10)
    lr = min(lr, 0.5)
    h_new = project_box_sum(h - lr * grad, target)
    return h_new, val


def subgrad_polyak(h0, alpha, target, n_iters, lr_scale):

    return best_h, best_score


def coordinate_descent(h0, n_passes, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()

    return best_h, best_score
