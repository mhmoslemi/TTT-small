import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0 - 1e-8
        else:
            mask = x > 1e-8
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_max_corr(h, top_k=1):
    corr = compute_corr(h)
    n = len(h)
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        k = k_idx - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            l = -k
            grad[:n-l] += (1.0 - h[l:])
            grad[l:] -= h[:n-l]
    return grad / top_k

def subgradient_phase(h, n_iter, target_sum, focus_top_k=1, 
                      init_step=0.01, step_decay=0.999, min_step=1e-5):
    h = h.copy()
    step = init_step
    best_local = h.copy()
    best_score = compute_c5(h)
    stall = 0
    
    for it in range(n_iter):
        g = gradient_max_corr(h, top_k=focus_top_k)
        h_new = h - step * g
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_local = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                step *= step_decay
                if step < min_step:
                    break
            h = h_new  # still update to move
    
    return best_local, best_score

def slsqp_refine(h, target_sum, maxiter=500):
    n = len(h)
    
    def objective(x):
        corr = compute_corr(x)
        c_max = corr.max()
        # Log-sum-exp smoothing
        alpha = 50.0
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new
    except Exception:
        return h

def slsqp_refine_multi_alpha(h, target_sum, alphas=(10, 50, 200, 500)):
    current_best = h.copy()
    current_score = compute_c5(h)
    n = len(h)

            sc = compute_c5(h_new)

                current_best = h_new
        except Exception:
            pass
    
    return current_best

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_old = np.linspace(0.0, 1.0, n_src)

    h_out = np.interp(x_new, x_old, h_src)

def smooth_grad(h, alpha):
    corr = compute_corr(h)
    c_max = corr.max()
    log_weights = alpha * (corr - c_max) - np.log(len(corr))  # log softmax
    # For numerical stability
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    weights /= weights.sum()
    
    n = len(h)
    grad = np.zeros(n)
    
    # Only compute for correlations with significant weight
    # (to save time, only top M)
    threshold = log_weights.max() - 10  # weight > e^(-10) relative to max
    active = np.where(log_weights > threshold)[0]
    
    for k_idx in active:
        k = k_idx - (n - 1)
        w = weights[k_idx]
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            l = -k
            grad[:n-l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n-l]
    
    return grad

If m+k = j (m = j-k): term is h[j-k](1-h[j])

So the dependence on h[i] and h[j] is:
corr_k = B_k + h[i]*(1-h[i+k]) + h[j]*(1-h[j+k]) + h[i-k]*(1-h[i]) + h[j-k]*(1-h[j]) - (corrections for double counting)

Where B_k is the sum over m ∉ {i,j,i-k,j-k} of h[m](1-h[m+k]).

The corrections are needed when {i, j, i-k, j-k} has overlaps. This is getting complex.

Let me simplify: for the purpose of coordinate descent, I'll just compute the coefficients numerically.

For a given k, corr_k as a function of h[i] = x, h[j] = s-x (where s = h[i]+h[j]):
corr_k = A_k * x + C_k * (s-x) + D_k = (A_k - C_k) * x + C_k * s + D_k

Where:
- A_k = coefficient of h[i] in corr_k (with h[j] = 0)
- C_k = coefficient of h[j] in corr_k (with h[i] = 0)
- D_k = corr_k with h[i] = 0, h[j] = 0

We can compute these by:
- A_k = corr_k(h with h[i]=1, h[j]=0) - corr_k(h with h[i]=0, h[j]=0)
- C_k = corr_k(h with h[i]=0, h[j]=1) - corr_k(h with h[i]=0, h[j]=0)
- D_k = corr_k(h with h[i]=0, h[j]=0)

This requires 3 correlation computations per k, which is O(n) each, so O(n²) per step. For n=400, that's 160,000 operations per step. With, say, 10,000 steps, that's 1.6 billion operations. This might be too slow.

Let me optimize: instead of computing full correlations, just compute the change in corr_k when h[i] or h[j] changes.

When h[i] changes from 0 to x:
Δcorr_k = x * (1 - h[i+k]) [if i+k valid and i+k ≠ j]
        + (h[i-k] - h[i-k]) * ... 

Hmm, this is still complex. Let me just compute the gradients directly.

For coordinate descent, I'll use a simpler approach:
1. Compute the full correlation vector
2. For coordinate i, compute ∂corr_k/∂h[i] for all k
3. The objective max_k corr_k has subgradient = average of ∇corr_k over active k
4. Do a 1D minimization of the max function in the direction of the projected subgradient

    lo = h.min() - 2.0

        else:
            mask = x > 1e-7
        if mask.any():

            x = np.clip(x, 0.0, 1.0)
    return x

def grad_at_lag(h, k, n):
    """Gradient of corr_k w.r.t. h"""
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        l = -k
        g[:n-l] += (1.0 - h[l:])
        g[l:] -= h[:n-l]
    return g

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_gradient(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    c_max = corr.max()
    log_w = alpha * (corr - c_max)
    log_w -= log_w.max()

        wk = w[k_idx]
        g += wk * grad_at_lag(h, k, n)
    return g

def projected_gradient_descent(h, target_sum, n_iter, alpha_init=50.0, 
                                 step_init=0.01, alpha_growth=1.1):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_init
    step = step_init
    stall = 0
    
    for it in range(n_iter):
        g = smooth_gradient(h, alpha)
        # Project gradient to maintain sum constraint
        g_proj = g - np.mean(g)  # This makes sum of steps = 0 approximately
        
        h_new = h - step * g_proj
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
        
        if stall % 100 == 0:
            step *= 0.95
            alpha *= alpha_growth
        
        if step < 1e-8:
            break
    
    return best_h, best_score

def slsqp_round(h, target_sum, alpha, maxiter=500):
    n = len(h)
    
    def obj(x):
        return smooth_objective(x, alpha)
    
    def obj_grad(x):
        return smooth_gradient(x, alpha)
    
    def cons_fun(x):
        return np.sum(x) - target_sum
    
    def cons_jac(x):
        return np.ones(n)

    try:

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h

The proper approach for the projected gradient method with the constraint sum(h) = target:
1. Compute gradient g
2. The projected direction d should satisfy: d is in the tangent space, i.e., sum(d) = 0, and h + t*d stays in [0,1]^n for small t > 0.
3. Simplest: d = g - mean(g), but then we need to check box constraints.

Actually, the simplest correct approach is:
1. h_new = h - step * g (unconstrained step)
2. h_proj = project_feasible(h_new, target) (proper projection)
3. If f(h_proj) < f(h), accept; otherwise reduce step.

This is what I had before. Let me keep it simple.

One more idea: **use the structure of the correlation matrix**. The correlation vector corr has 2n-1 entries. The gradient of corr[k] with respect to h is:
∂corr[k]/∂h[j] = (1-h[j+k]) - h[j-k] (with boundary conditions)

This has a very specific structure: it's a "difference of shifted h". This means the gradient is smooth and has bounded norm, which is good for convergence.
