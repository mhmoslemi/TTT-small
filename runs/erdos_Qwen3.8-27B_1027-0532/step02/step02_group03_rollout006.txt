    """Subgradient method with exact max subgradient."""
    dx = 2.0 / n
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        corr = _corr_of(x)
        k_star = int(np.argmax(corr))
        g = _grad_max_corr_vectorized(x, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_sum_box(x - alpha * g, target_sum)
        cv = _c5_of(x_new)
        if cv < best_val:
            best_val = cv
            best = x_new.copy()
        x = x_new
    
    return best, best_val

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha0=0.3, lse_alpha=100.0):
    """Nesterov accelerated gradient with LSE-smoothed objective."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        corr = _corr_of(y)
        max_c = np.max(corr)
        # LSE-smoothed max
        shifted = lse_alpha * (corr - max_c)
        exp_vals = np.exp(np.clip(shifted, -50, 50))
        sum_exp = exp_vals.sum()
        if sum_exp < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = exp_vals / sum_exp
        
        # Gradient of LSE-smoothed max
        g = np.zeros(n)
        # Only use top few for efficiency
        top_k = np.argsort(weights)[::-1][:10]
        for idx in top_k:
            g += weights[idx] * _grad_max_corr_vectorized(y, idx, dx)
        
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_sum_box(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_sum_box(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _pair_coord_descent(h, target_sum, max_rounds=50, tol=1e-10):
    """Optimize pairs of adjacent coordinates."""
    n = len(h)
    h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        for j in range(0, n - 1, 2):
            # Fix s = h[j] + h[j+1]
            s = h[j] + h[j + 1]
            if s < 0 or s > 2:
                continue
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_lo >= t_hi:
                continue
            
            # Current value of C5
            old_c5 = _c5_of(h)
            
            # Try a grid of t values
            best_t = h[j]
            best_c = old_c5
            n_trials = 20
            t_vals = np.linspace(t_lo, t_hi, n_trials)
            for t in t_vals:
                h[j] = t
                h[j + 1] = s - t
                c = _c5_of(h)
                if c < best_c:
                    best_c = c
                    best_t = t
            
            h[j] = best_t
            h[j + 1] = s - best_t
            
            if best_c < old_c5 - tol:
                improved = True
        
        if not improved:
            break
    
    return h

def _slsqp_smooth(h0, n, target_sum, lse_alpha, maxiter=300):
    """SLSQP with LSE-smoothed objective."""
    global _g_target_sum, _g_n
    _g_target_sum = target_sum
    _g_n = n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = lse_alpha * (corr - m)
        return (m + np.log(np.sum(np.exp(np.clip(shifted, -50, 50)))) / lse_alpha) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

_g_n = 0

def _slsqp_smooth2(h0, n, target_sum, lse_alpha, maxiter=300):

    def obj(x):

        shifted = lse_alpha * (corr - m)

def _triple_coord_descent(h, target_sum, max_rounds=20):
    """Optimize triples of adjacent coordinates."""
    n = len(h)
    h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        for j in range(0, n - 2, 3):
            s = h[j] + h[j+1] + h[j+2]
            if s < 0 or s > 3:
                continue
            
            old_c5 = _c5_of(h)
            best_c = old_c5
            best_triplet = (h[j], h[j+1], h[j+2])
            
            # Grid search over the simplex
            n1 = 5
            for a in np.linspace(0, min(1, s), n1):
                for b in np.linspace(0, min(1, s-a), n1):
                    c_val = s - a - b
                    if c_val < 0 or c_val > 1:
                        continue
                    h[j] = a
                    h[j+1] = b
                    h[j+2] = c_val
                    c = _c5_of(h)
                    if c < best_c:
                        best_c = c
                        best_triplet = (a, b, c_val)
            
            h[j], h[j+1], h[j+2] = best_triplet
            if best_c < old_c5 - 1e-10:
                improved = True
        
        if not improved:
            break
    
    return h

def _fourier_h(params, n, M):
    """Evaluate Fourier series h(x) at n points on [0,2]."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, M+1):
        h += params[2*(k-1)] * np.cos(k * np.pi * x)
        h += params[2*(k-1)+1] * np.sin(k * np.pi * x)
    return h

def _fourier_optimize(n, M, maxiter=500):

    def obj(params):
        h = _fourier_h(params, n, M)
        if np.any(h < -0.01) or np.any(h > 1.01):
            return 10.0  # penalty for violation
        h = np.clip(h, 0, 1)
        # Fix sum
        h = _fix_sum(h, n/2.0)
        return _c5_of(h)
    
    # Initial: all zeros
    p0 = np.zeros(2*M)
    res = minimize(obj, p0, method='Nelder-Mead', options={'maxiter': maxiter})
    h = _fourier_h(res.x, n, M)
    h = _fix_sum(np.clip(h, 0, 1), n/2.0)
    return h, _c5_of(h)

from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_sum_box(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_at_lag(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _nesterov(h0, n, target_sum, n_iters, alpha0=0.3, lse_alpha=50.0, top_m=8):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = _corr_of(y)
        max_c = np.max(corr)
        shifted = lse_alpha * (corr - max_c)
        exp_vals = np.exp(np.clip(shifted, -50, 50))
        total = exp_vals.sum()
        if total < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = exp_vals / total
        top_idx = np.argsort(weights)[::-1][:top_m]
        g = np.zeros(n)
        for idx in top_idx:
            g += weights[idx] * _grad_at_lag(y, idx, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_sum_box(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_sum_box(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        return (m + np.log(np.sum(np.exp(np.clip(shifted, -50, 50)))) / alpha) * dx
    
    def con_fun(x):
        return x.sum() - target_sum
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _pair_cd(h, target_sum, max_rounds=30):
    n = len(h)
    h = h.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(0, n - 1, 2):
            s = h[j] + h[j + 1]
            if s < 0 or s > 2:
                continue
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_lo >= t_hi:
                continue
            old_c5 = _c5_of(h)
            best_t = h[j]
            best_c = old_c5
            for t in np.linspace(t_lo, t_hi, 15):

            h[j + 1] = s - best_t

                improved = True
        if not improved:
            break
    return h

        h += params[2*(k-1)+1] * np.sin(k * np.pi * x)
    return h
