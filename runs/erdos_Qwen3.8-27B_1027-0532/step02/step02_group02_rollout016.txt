    """Compute a gradient direction for the max of the correlation.
    Returns gradient of the max (at argmax) or weighted average of top-k."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k lags
    top_idx = np.argsort(corr)[-n_top:]
    if n_top == 1:
        lag = top_idx[0] - (n - 1)
        return _grad_at_lag(h, lag), corr
    else:
        # Weighted average based on how close to max
        max_val = corr[top_idx[0]]
        weights = np.array([0.5 ** ((max_val - corr[i]) * 10) for i in top_idx])
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = idx - (n - 1)
            grad += w * _grad_at_lag(h, lag)
        return grad, corr

    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad
