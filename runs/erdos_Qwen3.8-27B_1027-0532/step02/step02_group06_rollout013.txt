**Another key insight**: The projection to the constraint set {0≤h≤1, sum(h)=n/2} via bisection gives a unique solution, but the projection operator is not smooth (it's the projection onto a polytope intersected with a hyperplane). This can cause issues with gradient-based methods.

Alternative: Use a **soft constraint** approach:
- Minimize f_β(h) + (μ/2)(sum(h) - n/2)² + (ν/2)∑max(0, -h[i])² + (ν/2)∑max(0, h[i]-1)²
- Then do gradient descent without projection
- The penalty terms keep h approximately feasible
- At the end, project to get the exact feasible solution

This avoids the non-smoothness of the projection and might converge faster.

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, corr, beta):
    n = len(h)
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx_max = np.argmax(corr)
        w = np.zeros(len(corr))
        w[idx_max] = 1.0
    else:
        w /= w_sum
    active = np.where(w > 1e-12)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k in active:
        w_k = w[k]
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w_k * (1.0 - 2.0 * h)
        elif lag > 0:
            na = n - lag
            grad[lag:] += w_k * (1.0 - h[:na])
            grad[:na] -= w_k * h[lag:]
        else:
            la = -lag
            na = n - la
            grad[:na] += w_k * (1.0 - h[la:])
            grad[la:] -= w_k * h[:na]
    return grad

def inner_loop(h, beta_lo, beta_hi, max_iters, deadline, seed):
    """Anneal from beta_lo to beta_hi, update h in place, return best found."""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    h_prev = h.copy()
    mu = 0.85
    best_h = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.RandomState(seed)
    step = 0.3
    stall = 0
    total_corr = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(max_iters, 1)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        
        corr = compute_correlations(h)
        y = h + mu * (h - h_prev)
        y = project_to_feasible(y, target_sum)
        corr_y = compute_correlations(y)
        g = grad_smooth_max(y, corr_y, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        
        improved = False
        for t in range(30):
            cand = h - step * g
            cand = project_to_feasible(cand, target_sum)
            s = evaluate_score(cand)
            if s < best_score:
                best_score = s
                best_h = cand.copy()
                h_prev, h = h, cand
                stall = 0
                improved = True
                break
            curr_s = evaluate_score(h)
            if s < curr_s:
                h_prev, h = h, cand
                stall = 0
                improved = True
                break
            step *= 0.6
        
        if not improved:
            stall += 1
            if stall > 200:
                eps = 0.03 + 0.05 * min(stall / 500.0, 1.0)
                h = h + rng.randn(n) * eps / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                h_prev = h.copy()
                step = 0.3
                stall = 0
    
    return best_h, best_score

def slsqp_refine(h, beta, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def objective(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    
    def obj_grad(hv):
        corr = compute_correlations(hv)
        return grad_smooth_max(hv, corr, beta)
    
    res = minimize(objective, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                  'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def grad_smooth_max(h, corr, beta):
    n = len(h)
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx_max = np.argmax(corr)
        w = np.zeros(len(corr))
        w[idx_max] = 1.0
    else:
        w /= w_sum
    active = np.where(w > 1e-12)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k in active:
        w_k = w[k]
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w_k * (1.0 - 2.0 * h)
        elif lag > 0:
            na = n - lag
            grad[lag:] += w_k * (1.0 - h[:na])
            grad[:na] -= w_k * h[lag:]
        else:
            la = -lag
            na = n - la
            grad[:na] += w_k * (1.0 - h[la:])
            grad[la:] -= w_k * h[:na]
    return grad


def inner_loop(h, beta_lo, beta_hi, max_iters, deadline, seed):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    h_prev = h.copy()
    mu = 0.88
    best_h = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.RandomState(seed)
    step = 0.4
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / float(max(max_iters, 1))
        beta = beta_lo + (beta_hi - beta_lo) * frac

        corr = compute_correlations(h)
        y = h + mu * (h - h_prev)
        y = project_to_feasible(y, target_sum)
        corr_y = compute_correlations(y)
        g = grad_smooth_max(y, corr_y, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm

        improved = False
        curr_score = evaluate_score(h)
        for trial in range(35):
            cand = h - step * g
            cand = project_to_feasible(cand, target_sum)
            s = evaluate_score(cand)
            if s < best_score:
                best_score = s
                best_h = cand.copy()
                h_prev = h.copy()
                h = cand
                stall = 0
                improved = True
                break
            if s < curr_score:
                h_prev = h.copy()
                h = cand
                curr_score = s
                stall = 0
                improved = True
                break
            step *= 0.55

        if not improved:
            stall += 1
            step = max(step, 0.01)
            if stall > 300:
                eps = 0.03 + 0.06 * min(float(stall) / 800.0, 1.0)
                h = h + rng.randn(n) * eps / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                h_prev = h.copy()
                step = 0.3
                mu = 0.88
                stall = 0

        if it % 2000 == 1999:
            mu = max(0.82, mu - 0.02)

    return best_h, best_score


def _slsqp_obj(v, hv, beta):
    corr = compute_correlations(hv)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def _slsqp_grad(v, hv, beta):
    corr = compute_correlations(hv)
    return grad_smooth_max(hv, corr, beta)


def _slsqp_con_fun(v, hv, target_sum):
    return np.sum(hv) - target_sum


def _slsqp_con_jac(v, hv, target_sum):
    return np.ones(len(hv))


def slsqp_refine(h, beta, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def objective(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

    def obj_grad(hv):
        corr = compute_correlations(hv)
        return grad_smooth_max(hv, corr, beta)

    def cfun(hv):
        return np.sum(hv) - target_sum

    def cjac(hv):
        return np.ones(n)

    res = minimize(objective, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def make_fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False)
    f = np.zeros(n)
    for k in range(1, 8):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        f += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    f += 0.3 * np.cos(2.0 * np.pi * x)
    h = 1.0 / (1.0 + np.exp(-2.0 * f))
    h = project_to_feasible(h, n / 2.0)
    return h


def make_block_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_blocks = rng.randint(4, 12)
    edges = np.sort(rng.choice(np.arange(1, n), n_blocks - 1, replace=False))
    boundaries = np.concatenate([[0], edges, [n]])
    for j in range(n_blocks):
        s = boundaries[j]
        e = boundaries[j + 1]
        h[s:e] = rng.choice([0.0, 0.25, 0.5, 0.75, 1.0])
    h = project_to_feasible(h, n / 2.0)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    else:
        init = make_fourier_init(425, seed)
    n = len(init)
    init = project_to_feasible(init, n / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)
    rng = np.random.RandomState(seed)

    # Phase 1: Annealed gradient (low to medium beta)
    t1 = t0 + budget_s * 0.18
    h1, s1 = inner_loop(best_h, 3.0, 30.0, 8000, t1, seed)
    if s1 < best_score:
        best_score, best_h = s1, h1

    # Phase 2: Medium to high beta
    t2 = t0 + budget_s * 0.32
    h2, s2 = inner_loop(best_h, 15.0, 80.0, 6000, t2, seed + 100)
    if s2 < best_score:
        best_score, best_h = s2, h2

    # Phase 3: SLSQP at several betas
    if time.time() < deadline - 40:
        for b in [20, 50, 100, 200]:
            if time.time() > deadline - 20:
                break
            h_s = slsqp_refine(best_h, b, maxiter=300)
            if h_s is not None:
                s_s = evaluate_score(h_s)
                if s_s < best_score:
                    best_score, best_h = s_s, h_s

    # Phase 4: Basin hopping with diverse perturbations
    if time.time() < deadline - 60:
        t_bp_end = min(deadline - 40, time.time() + budget_s * 0.25)
        scales = [0.01, 0.015, 0.02, 0.03, 0.04, 0.06, 0.08, 0.10, 0.13, 0.16]
        for trial in range(50):
            if time.time() > t_bp_end:
                break
            scale = scales[trial % len(scales)]
            h_pert = best_h + rng.randn(len(best_h)) * scale / np.sqrt(len(best_h))
            h_pert = project_to_feasible(h_pert, len(best_h) / 2.0)
            t_inner = min(t_bp_end - time.time() - 3.0, 20.0)
            if t_inner < 4:
                break
            h_b, s_b = inner_loop(h_pert, 5.0, 60.0, 2500, time.time() + t_inner, seed + 2000 + trial)
            if s_b < best_score:
                best_score, best_h = s_b, h_b

    # Phase 5: Try alternative initializations
    if time.time() < deadline - 80:
        alt_starts = []
        ns_try = n
        h_alt1 = make_fourier_init(ns_try, seed + 7777)
        alt_starts.append(h_alt1)
        h_alt2 = make_block_init(ns_try, seed + 8888)
        alt_starts.append(h_alt2)

        for ai, h_alt in enumerate(alt_starts):
            if time.time() > deadline - 50:
                break
            t_inner = min(deadline - time.time() - 40, 60.0)
            if t_inner < 15:
                break
            h_a, s_a = inner_loop(h_alt, 3.0, 50.0, 3000, time.time() + t_inner, seed + 3000 + ai)
            if s_a < best_score:
                best_score, best_h = s_a, h_a

    # Phase 6: Multi-resolution (upsample and refine)
    if time.time() < deadline - 80:
        for mult in [2, 3]:
            n_new = len(best_h) * mult
            if n_new > 2500 or time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2.0, n_new, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_to_feasible(h_up, n_new / 2.0)
            t_inner = min(deadline - time.time() - 40, 90.0)
            if t_inner < 25:
                break
            h_up, s_up = inner_loop(h_up, 3.0, 40.0, 4000, time.time() + t_inner, seed + 6000 + n_new)
            if s_up < best_score:
                best_score, best_h = s_up, h_up
                if time.time() < deadline - 25:
                    h_sf = slsqp_refine(best_h, 50, maxiter=150)
                    if h_sf is not None:
                        s_sf = evaluate_score(h_sf)
                        if s_sf < best_score:
                            best_score, best_h = s_sf, h_sf

    # Phase 7: Final high-beta annealing polish
    if time.time() < deadline - 20:
        t_polish = deadline - 10
        h_p, s_p = inner_loop(best_h, 30.0, 200.0, 3000, t_polish, seed + 9000)
        if s_p < best_score:
            best_score, best_h = s_p, h_p

    # Final SLSQP
    if time.time() < deadline - 8:
        for b in [50, 150, 400]:
            if time.time() > deadline - 3:
                break
            h_f = slsqp_refine(best_h, b, maxiter=100)
            if h_f is not None:
                s_f = evaluate_score(h_f)
                if s_f < best_score:
                    best_score, best_h = s_f, h_f

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
