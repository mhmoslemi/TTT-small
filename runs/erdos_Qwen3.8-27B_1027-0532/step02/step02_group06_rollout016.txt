import cvxpy as cp

    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    """Get lags that are within threshold_frac of the max"""
    max_val = np.max(corr)
    active = np.where(corr >= max_val - threshold_frac * max_val)[0]
    return active

def lp_optimize(h, max_iters=50, t0=None, time_limit=100, thresh_frac=0.003):
    """Iteratively solve LP subproblems"""
    n = len(h)
    h = project_to_feasible(h.copy(), n / 2.0)
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for it in range(max_iters):
        if t0 is not None and time.time() - t0 > t0 + time_limit:
            break
        
        corr = compute_correlations(h)
        active = get_active_lags(corr, thresh_frac)
        
        h_new = solve_lp_subproblem(h, active, n)
        if h_new is None:
            break
        
        h_new = project_to_feasible(h_new, n / 2.0)
        score = evaluate_score(h_new)
        
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        # Damping: blend old and new
        h = 0.3 * h + 0.7 * h_new
        h = project_to_feasible(h, n / 2.0)
    
    return best_h, best_score

def subgradient_optimize(h, n_top=10, alpha=20, max_iters=10000, 
                         step_init=0.5, t0=None, time_limit=100):
    """Subgradient method with soft-max"""
    n = len(h)
    target_sum = n / 2.0
    h = project_to_feasible(h.copy(), target_sum)
    rng = np.random.RandomState(42)
    
    best_h = h.copy()
    best_score = evaluate_score(h)
    step = step_init
    stall = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iters):
        if t0 is not None and time.time() - t0 > t0 + time_limit:
            break
        
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        
        m = top_vals[0]
        shifted = top_vals - m
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            if lag == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:] += w * (1.0 - h[:-lag])
                grad[:n-lag] -= w * h[lag:]
            else:
                la = -lag
                grad[:n-la] += w * (1.0 - h[la:])
                grad[la:] -= w * h[:n-la]
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        grad_norm = np.sqrt(np.sum(momentum * momentum))
        if grad_norm < 1e-12:
            break
        
        momentum_normalized = momentum / grad_norm
        
        improved = False
        for trial in range(15):
            h_new = h - step * momentum_normalized
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.5
        
        if not improved:
            stall += 1
            if stall > 100:
                h = best_h + rng.randn(n) * 0.05 / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init
                stall = 0
                momentum = np.zeros(n)
    
    return best_h, best_score

def upsample(h, n_new):
    """Upsample h to n_new points"""
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def solve_lp_subproblem(h_current, active_lags, n, eps=0.15):
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h_var >= np.maximum(h_current - eps, 0.0),
        h_var <= np.minimum(h_current + eps, 1.0),
        cp.sum(h_var) == n / 2.0
    ]
    
    for idx in active_lags:
        lag = int(idx - (n - 1))
        lin_coeff = np.zeros(n)
        const = 0.0
        
        if lag == 0:
            lin_coeff[:] = 1.0 - 2.0 * h_current
            const = np.sum(h_current ** 2)
        elif lag > 0:
            # Vectorized: for i in range(n-lag)
            # lin_coeff[i] += 1 - h_current[i+lag]
            # lin_coeff[i+lag] += -h_current[i]
            # const += h_current[i] * h_current[i+lag]
            idx1 = np.arange(n - lag)
            idx2 = idx1 + lag
            np.add.at(lin_coeff, idx1, 1.0 - h_current[idx2])
            np.add.at(lin_coeff, idx2, -h_current[idx1])
            const = np.sum(h_current[idx1] * h_current[idx2])
        else:
            la = -lag
            idx1 = np.arange(n - la)
            idx2 = idx1 + la
            # f = h[i+la]*(1-h[i])
            # coeff of h[i]: -h[i+la]
            # coeff of h[i+la]: 1-h[i]
            np.add.at(lin_coeff, idx1, -h_current[idx2])
            np.add.at(lin_coeff, idx2, 1.0 - h_current[idx1])
            const = np.sum(h_current[idx1] * h_current[idx2])
        
        constraints.append(lin_coeff @ h_var + const <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if h_var.value is not None and np.all(np.isfinite(h_var.value)):
            return np.array(h_var.value).flatten()
    except:
        pass
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if h_var.value is not None and np.all(np.isfinite(h_var.value)):
            return np.array(h_var.value).flatten()
    except:
        pass
    
    return None

import numpy as np
import time
import cvxpy as cp


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_linearized_constraint(h_current, lag, n):
    lin_coeff = np.zeros(n)
    if lag == 0:
        lin_coeff[:] = 1.0 - 2.0 * h_current
        const = np.sum(h_current * h_current)
    elif lag > 0:
        lin_coeff[:n - lag] += (1.0 - h_current[lag:])
        lin_coeff[lag:] += (-h_current[:n - lag])
        const = np.sum(h_current[:n - lag] * h_current[lag:])
    else:
        la = -lag
        lin_coeff[la:] += (1.0 - h_current[:n - la])
        lin_coeff[:n - la] += (-h_current[la:])
        const = np.sum(h_current[:n - la] * h_current[la:])
    return lin_coeff, const


def solve_lp(h_current, active_indices, n, eps=None, solver=None):
    h_var = cp.Variable(n)
    t = cp.Variable()
    if eps is not None and eps > 0:
        lb = np.maximum(h_current - eps, 0.0)
        ub = np.minimum(h_current + eps, 1.0)
    else:
        lb = np.zeros(n)
        ub = np.ones(n)
    constraints = [h_var >= lb, h_var <= ub, cp.sum(h_var) == n / 2.0]
    for idx in active_indices:
        lag = int(idx - (n - 1))
        lc, ct = build_linearized_constraint(h_current, lag, n)
        constraints.append(lc @ h_var + ct <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    solvers = [solver] if solver is not None else [cp.ECOS, cp.SCS, cp.CVXOPT]
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if h_var.value is not None and np.all(np.isfinite(h_var.value)):
                return np.array(h_var.value).flatten()
        except Exception:
            continue
        prob = cp.Problem(cp.Minimize(t), constraints)
    return None


def get_active_indices(corr, thresh_frac):
    max_val = np.max(corr)
    threshold = max_val - thresh_frac * max_val
    return np.where(corr >= threshold)[0]


def lp_phase(h, t0, time_limit, thresh_start=0.05, thresh_end=0.002, max_iters=40):
    n = len(h)
    h = project_to_feasible(h.copy(), n / 2.0)
    best_h = h.copy()
    best_score = evaluate_score(h)
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        frac = thresh_start + (thresh_end - thresh_start) * (it / max(max_iters - 1, 1))
        corr = compute_correlations(h)
        active = get_active_indices(corr, frac)
        if len(active) > 300:
            top = np.argsort(-corr)[:300]
            active = top
        eps = 0.15 * (1.0 - 0.5 * it / max(max_iters - 1, 1))
        h_lp = solve_lp(h, active, n, eps=eps)
        if h_lp is None:
            eps2 = eps * 0.5
            h_lp = solve_lp(h, active, n, eps=eps2)
        if h_lp is None:
            break
        h_lp = project_to_feasible(h_lp, n / 2.0)
        score_lp = evaluate_score(h_lp)
        if score_lp < best_score:
            best_score = score_lp
            best_h = h_lp.copy()
            h = h_lp
        else:
            h = 0.5 * h + 0.5 * h_lp
            h = project_to_feasible(h, n / 2.0)
            score_mid = evaluate_score(h)
            if score_mid < best_score:
                best_score = score_mid
                best_h = h.copy()
    return best_h, best_score


def subgrad_phase(h, t0, time_limit, n_top=15, alpha=30, step_init=0.3):
    n = len(h)
    target_sum = n / 2.0
    h = project_to_feasible(h.copy(), target_sum)
    rng = np.random.RandomState(12345)
    best_h = h.copy()
    best_score = evaluate_score(h)
    step = step_init
    stall = 0
    momentum = np.zeros(n)
    beta = 0.85
    for it in range(100000):
        if time.time() - t0 > t0 + time_limit:
            break
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        m = top_vals[0]
        shifted = top_vals - m
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            if lag == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] += w * (-h[:n - lag])
            else:
                la = -lag
                grad[la:] += w * (1.0 - h[:n - la])
                grad[:n - la] += w * (-h[la:])
        momentum = beta * momentum + (1.0 - beta) * grad
        gnorm = np.sqrt(np.sum(momentum * momentum))
        if gnorm < 1e-14:
            break
        gnormed = momentum / gnorm
        improved = False
        for trial in range(20):
            h_new = h - step * gnormed
            h_new = project_to_feasible(h_new, target_sum)
            sc = evaluate_score(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.45
        if not improved:
            stall += 1
            if stall > 150:
                h = best_h + rng.randn(n) * (0.04 + 0.06 * min(stall / 500.0, 1.0)) / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init
                stall = 0
                momentum = np.zeros(n)
        if it > 0 and it % 5000 == 0:
            step = max(step_init * 0.97 ** (it // 5000), step_init * 0.05)
    return best_h, best_score


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
