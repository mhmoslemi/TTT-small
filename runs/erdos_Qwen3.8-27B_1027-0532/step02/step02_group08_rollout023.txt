import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_h_grad(h, alpha):
    """Smooth max of correlations with gradient."""
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    # Gradient computation
    n_corr = len(corrs)
    w_rev = weights[::-1]
    # conv1[j] = sum_m g[m] * w_rev[j-m] = sum_m (1-h[m]) * weights[n_corr-1-j+m]
    # We want the gradient at the center positions
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad

    m_top = np.max(top_corrs)
    shifted = alpha * (top_corrs - m_top)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights_top = exp_shifted / sum_exp
    obj = m_top + np.log(sum_exp) / alpha
    # Build full weights vector (zeros except at top-m positions)
    weights = np.zeros(len(corrs))

def project_sum(h, target):
    """Project h to satisfy sum(h) = target while keeping 0<=h<=1."""
    n = len(h)
    h = h.copy()

        diff = target - s
        if abs(diff) < 1e-14:

        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)

    )

def momentum_descent(h0, alpha, m, n_iter, lr, momentum, bounds, target_sum, m_top_param=None):
    """Momentum-based gradient descent with projection."""
    h = h0.copy()
    v = np.zeros_like(h)
    for i in range(n_iter):

            obj, grad = top_m_smooth_max_h_grad(h, alpha, m_top_param)
        else:
            obj, grad = smooth_max_h_grad(h, alpha)
        v = momentum * v - lr * grad

        h = project_sum(h, target_sum)
    return h

def local_search(h, n_perturbations, rng, n, bounds):
    """Try small local modifications."""
    best_h = h.copy()
    best_score = evaluate_score(h)
    for _ in range(n_perturbations):
        h_try = h.copy()
        # Choose a random contiguous block to modify
        block_len = rng.randint(3, max(4, n // 50))
        start = rng.randint(0, n - block_len)
        # Add small perturbation to the block
        pert = rng.randn(block_len) * 0.05
        h_try[start:start+block_len] += pert
        h_try = np.clip(h_try, 0, 1)

        s = evaluate_score(h_try)

def fft_convolve(a, b):
    """Compute np.convolve(a, b, mode='full') using FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]


def compute_correlations(h):
    g = 1.0 - h
    return fft_convolve(h, g[::-1])


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_h_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = fft_convolve(h, g[::-1])
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    shifted = np.clip(shifted, -60, 60)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    conv1 = fft_convolve(g, weights)
    conv2 = fft_convolve(h, weights[::-1])
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def top_m_smooth_max_h_grad(h, alpha, m):
    n = len(h)
    g = 1.0 - h

                        best_score = s
                        best_h = h_new.copy()
        except Exception:
            pass

    # Phase 6: Very high alpha polish
    for alpha in [50000, 100000, 200000]:
        if time_left() < 15:
            break
        res = minimize(
            smooth_max_combined, best_h, method='L-BFGS-B',
            bounds=bounds_best, args=(alpha,), jac=True,
            options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0}
        )
        if res.x is not None:
            h_new = project_sum(np.clip(res.x, 0, 1), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 7: Try nearby resolutions
    for n_adj in [n_best + 10, n_best - 10, n_best + 30, n_best - 30, n_best + 50]:
        if time_left() < 20:
            break
        if n_adj < 200 or n_adj > 3000:
            continue
        h_init = upscale(best_h, n_adj)
        h_init = project_sum(h_init, n_adj / 2.0)
        bounds_adj = [(0.0, 1.0)] * n_adj
        for alpha in [2000, 5000, 15000]:
            if time_left() < 10:
                break
            res = minimize(
                smooth_max_combined, h_init, method='L-BFGS-B',
                bounds=bounds_adj, args=(alpha,), jac=True,
                options={'maxiter': max(30, min(200, 80000 // n_adj)),
                         'ftol': 1e-16, 'gtol': 1e-12, 'iprint': 0}
            )
            if res.x is not None:
                h_new = project_sum(np.clip(res.x, 0, 1), n_adj / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_init = h_new.copy()

    # Final: one more polish at best resolution
    n_final = len(best_h)
    best_h = project_sum(best_h, n_final / 2.0)
    if time_left() > 10:
        bounds_final = [(0.0, 1.0)] * n_final
        res = minimize(
            smooth_max_combined, best_h, method='L-BFGS-B',
            bounds=bounds_final, args=(50000,), jac=True,
            options={'maxiter': 100, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0}
        )
        if res.x is not None:
            h_new = project_sum(np.clip(res.x, 0, 1), n_final / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
