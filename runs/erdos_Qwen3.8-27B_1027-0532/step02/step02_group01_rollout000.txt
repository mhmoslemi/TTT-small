    remaining = budget_s - (time.time() - start_time)
    n_sa = 50000
    t_sa_est = n_sa * 0.00005
    if remaining > t_sa_est + 60:
        tl = min(remaining - 60, 120)
        h_sa, c5_sa = binary_sa(to_binary(best_h), n_sa, 0.5, 0.01,
                                rng, tl, start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_soft = h_sa + 0.03 * rng.standard_normal(n0)
        h_soft = project_onto_constraint(h_soft, target_sum)
        if time.time() - start_time < time_limit - 50:
            tl2 = min(time_limit - (time.time() - start_time) - 30, 80)
            h_opt, c5_opt = slsqp_step(h_soft, 0.001, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_bin2 = to_binary(h_opt)
            tl3 = min(time_limit - (time.time() - start_time) - 20, 100)
            h_bi3, c5_bi3 = binary_local_search_first_improvement(
                h_bin2, 30, tl3, start_time, rng)
            if c5_bi3 < best_c5:
                best_c5 = c5_bi3
                best_h = h_bi3.copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        for restart in range(8):
            if time.time() - start_time > time_limit - 30:
                break
            strategy = restart % 4
            if strategy == 0:
                h_pert = best_h + 0.04 * rng.standard_normal(n0)
            elif strategy == 1:
                h_pert = perturb_binary_swap(to_binary(best_h), max(3, n0 // 10), rng)
                h_pert = h_pert + 0.02 * rng.standard_normal(n0)
            elif strategy == 2:
                h_rand = project_onto_constraint(
                    np.clip(0.5 + 0.4 * rng.standard_normal(n0), 0, 1), target_sum)
                alpha = rng.uniform(0.4, 0.7)
                h_pert = alpha * best_h + (1.0 - alpha) * h_rand
            else:
                h_pert = best_h.copy()
                nflip = max(2, n0 // 12)
                idx = rng.choice(n0, size=nflip, replace=False)
                h_pert[idx] = 1.0 - h_pert[idx]

            h_pert = project_onto_constraint(h_pert, target_sum)
            h_opt, c5_opt = slsqp_step(h_pert, 0.005, 100)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

            h_bin_r = to_binary(best_h)
            tl_r = min(remaining - 60, 80)
            h_bi_r, c5_bi_r = binary_local_search_first_improvement(
                h_bin_r, 20, tl_r, start_time, rng)
            if c5_bi_r < best_c5:
                best_c5 = c5_bi_r
                best_h = h_bi_r.copy()

    remaining = budget_s - (time.time() - start_time) - 30
    if remaining > 200:
        n1 = 2 * n0
        h_up = upsample(best_h, 2)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.015 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        tl_up = min(remaining - 30, 400)
        h_u, c5_u = slsqp_step(h_up, 0.01, 150)
        if c5_u < best_c5:
            best_c5 = c5_u
            best_h = h_u.copy()
        h_u2, c5_u2 = slsqp_step(h_u, 0.002, 200)
        if c5_u2 < best_c5:
            best_c5 = c5_u2
            best_h = h_u2.copy()
        h_bin_up = to_binary(best_h)
        tl_ub = min(budget_s - (time.time() - start_time) - 20, 150)
        h_bi_up, c5_bi_up = binary_local_search_first_improvement(
            h_bin_up, 30, tl_ub, start_time, rng)
        if c5_bi_up < best_c5:
            best_c5 = c5_bi_up
            best_h = h_bi_up.copy()

    remaining = budget_s - (time.time() - start_time) - 20
    if remaining > 80:
        n_cur = len(best_h)
        h_polish = best_h + 0.003 * rng.standard_normal(n_cur)
        h_polish = project_onto_constraint(h_polish, n_cur / 2.0)
        tl_p = min(remaining - 10, 60)
        h_opt, c5_opt = slsqp_step(h_polish, 0.0001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    remaining = budget_s - (time.time() - start_time) - 10
    if remaining > 50:
        n_cur = len(best_h)
        h_bin_final = to_binary(best_h)
        tl_f = min(remaining - 5, 40)
        h_bi_f, c5_bi_f = binary_local_search_first_improvement(
            h_bin_final, 20, tl_f, start_time, rng)
        if c5_bi_f < best_c5:
            best_c5 = c5_bi_f
            best_h = h_bi_f.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
