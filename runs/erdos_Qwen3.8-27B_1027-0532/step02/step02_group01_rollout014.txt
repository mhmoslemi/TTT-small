        if s > target_sum:

                best_score = scores[j_min]
                best_i = ii
                best_j = j_min
                best_delta = D + B[j_min]
        
        if best_score < current_c5 - 1e-12:
            # Apply the swap
            i_pos = ones[best_i]
            j_pos = zeros[best_j]
            h[i_pos] = 0.0
            h[j_pos] = 1.0
            c += best_delta
            current_c5 = best_score
            # Update ones and zeros
            ones = ones[np.ones(len(ones), dtype=bool)]
            ones[best_i] = -1  # mark for removal
            ones = ones[ones >= 0]
            zeros = zeros[np.ones(len(zeros), dtype=bool)]
            zeros[best_j] = -1
            zeros = zeros[zeros >= 0]
        else:
            break
    
    return h

def sa_with_delta(h_start, n_iter=50000, t_start=1.0, t_end=0.001, rng=None):
    """Simulated annealing with fast delta evaluation."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h_start)
    h = np.round(h_start).astype(float)
    # Ensure exactly n/2 ones
    n_ones_target = n // 2
    n_ones = int(h.sum())
    if n_ones > n_ones_target:
        ones = np.where(h == 1.0)[0]
        to_flip = rng.choice(ones, size=n_ones - n_ones_target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < n_ones_target:
        zeros = np.where(h == 0.0)[0]
        to_flip = rng.choice(zeros, size=n_ones_target - n_ones, replace=False)
        h[to_flip] = 1.0
    
    c = compute_corr_vector(h).astype(float)
    current_c5 = c.max()
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    k = np.arange(-(n-1), n)
    
    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))
        
        # Random swap
        ii = rng.integers(len(ones))
        jj = rng.integers(len(zeros))
        i_pos = ones[ii]
        j_pos = zeros[jj]
        
        # Compute delta
        idx_plus_i = i_pos + k
        idx_minus_i = i_pos - k
        idx_plus_j = j_pos + k
        idx_minus_j = j_pos - k
        
        delta = np.zeros(2*n-1)
        vp = (idx_plus_i >= 0) & (idx_plus_i < n)
        vm = (idx_minus_i >= 0) & (idx_minus_i < n)
        delta[vp] += h[idx_plus_i[vp]]
        delta[vm] += h[idx_minus_i[vm]]
        vp = (idx_plus_j >= 0) & (idx_plus_j < n)
        vm = (idx_minus_j >= 0) & (idx_minus_j < n)
        delta[vp] -= h[idx_plus_j[vp]]
        delta[vm] -= h[idx_minus_j[vm]]
        
        new_c5 = (c + delta).max()
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(t, 1e-10)):
            # Accept
            h[i_pos] = 0.0
            h[j_pos] = 1.0
            c += delta
            current_c5 = new_c5
            # Update ones/zeros
            ones[ii] = j_pos
            zeros[jj] = i_pos
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        # else: reject, h unchanged
    
    return best_h, best_c5

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def make_binary(h, n_target, rng):
    """Convert h to a binary vector with exactly n_target/2 ones."""
    n = len(h)
    h_bin = np.round(h).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1.0)[0]
        to_flip = rng.choice(ones, size=n_ones - target, replace=False)
        h_bin[to_flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h_bin == 0.0)[0]
        to_flip = rng.choice(zeros, size=target - n_ones, replace=False)
        h_bin[to_flip] = 1.0
    return h_bin

def fast_local_search_binary(h, max_iters=20000, time_limit=60, verbose=False):
    """Fast local search using O(n^2 * n) swap evaluation per step."""
    n = len(h)
    h = h.copy().astype(float)
    start = time.time()
    
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = c.max()
    
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    k_vec = np.arange(-(n-1), n)
    two_n_minus_1 = 2*n - 1
    
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        ones_arr = ones
        zeros_arr = zeros
        n_ones = len(ones_arr)
        n_zeros = len(zeros_arr)
        
        # Compute A[i] = h[ones[i] + k] + h[ones[i] - k] for all ones
        # Shape: (n_ones, 2n-1)
        idx_p = ones_arr[:, None] + k_vec[None, :]
        idx_m = ones_arr[:, None] - k_vec[None, :]
        A = np.zeros((n_ones, two_n_minus_1))
        vp = (idx_p >= 0) & (idx_p < n)
        vm = (idx_m >= 0) & (idx_m < n)
        A[vp] += h[idx_p[vp]]
        A[vm] += h[idx_m[vm]]
        
        # Compute B[j] = -(h[zeros[j] + k] + h[zeros[j] - k]) for all zeros
        idx_pz = zeros_arr[:, None] + k_vec[None, :]
        idx_mz = zeros_arr[:, None] - k_vec[None, :]
        B = np.zeros((n_zeros, two_n_minus_1))
        vpz = (idx_pz >= 0) & (idx_pz < n)
        vmz = (idx_mz >= 0) & (idx_mz < n)
        B[vpz] -= h[idx_pz[vpz]]
        B[vmz] -= h[idx_mz[vmz]]
        
        # Find best swap: min over (i,j) of max(c + A[i] + B[j])
        best_score = current_max  # only accept if strictly better
        best_i, best_j = -1, -1
        best_delta = None
        
        for ii in range(n_ones):
            D = c + A[ii]  # (2n-1,)
            # scores[jj] = max(D + B[jj])
            scores = np.max(D[None, :] + B, axis=1)
            jm = np.argmin(scores)
            if scores[jm] < best_score - 1e-12:
                best_score = scores[jm]
                best_i = ii
                best_j = jm
                best_delta = D + B[jm]
        
        if best_i >= 0:
            i_pos = int(ones[best_i])
            j_pos = int(zeros[best_j])
            h[i_pos] = 0.0
            h[j_pos] = 1.0
            c += best_delta
            current_max = best_score
            # Update ones and zeros arrays
            ones = np.delete(ones, best_i)
            zeros = np.delete(zeros, best_j)
            zeros = np.append(zeros, i_pos)
            ones = np.append(ones, j_pos)
            stall = 0
        else:
            stall += 1
            break
    
    return h, compute_c5(h)

def sa_binary(h_start, n_iter=100000, t_start=1.0, t_end=0.001, rng=None):
    """Simulated annealing on binary solutions with fast delta."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h_start)
    h = make_binary(h_start, n, rng)
    
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = c.max()
    best_max = current_max
    best_h = h.copy()

def smooth_max_grad(h, eps):

    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)

            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

            f0 = rng.choice(zeros_idx, n_flip_0, replace=False)

        h_kick = make_binary(h_kick, rng)
        tl = min(remaining - 10, 15)
        h_ls, c5_ls = fast_local_search_binary(h_kick, max_iters=10000, time_limit=tl)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)

    n0 = len(h0)
    h_bin = make_binary(h0, rng)
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        h_lf, c5_lf = fast_local_search_binary(h_bin, max_iters=50000, time_limit=min(remaining - 30, 55))
        if c5_lf < best_c5:
            best_c5 = c5_lf
            best_h = h_lf

    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        h_sa, _ = sa_binary(best_h, n_iter=200000, t_start=3.0, t_end=0.001, rng=rng)
        h_lf2, c5_lf2 = fast_local_search_binary(h_sa, max_iters=30000, time_limit=min(remaining - 20, 20))
        if c5_lf2 < best_c5:
            best_c5 = c5_lf2
            best_h = h_lf2

    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        for t_s in [1.0, 2.0, 5.0, 8.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 40:
                break
            h_sa2, _ = sa_binary(best_h, n_iter=120000, t_start=t_s, t_end=0.005, rng=rng)
            h_lf3, c5_lf3 = fast_local_search_binary(h_sa2, max_iters=20000, time_limit=min(remaining - 15, 12))
            if c5_lf3 < best_c5:
                best_c5 = c5_lf3
                best_h = h_lf3

    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        best_h, best_c5 = multi_kick_local_search(best_h, n_kicks=15, kick_size=6,
                                                   time_limit=budget_s - 30,
                                                   start_time=start_time, rng=rng)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        for eps in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 25:
                break
            h_soft = best_h + 0.03 * rng.standard_normal(n0)
            h_soft = project_onto_constraint(h_soft, n0 / 2.0)
            h_opt, c5_opt = slsqp_refine(h_soft, eps, 150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        h_rand = (rng.random(n0) < 0.5).astype(float)
        h_rand = make_binary(h_rand, rng)
        h_sa_r, _ = sa_binary(h_rand, n_iter=150000, t_start=8.0, t_end=0.01, rng=rng)
        h_lf_r, c5_r = fast_local_search_binary(h_sa_r, max_iters=25000, time_limit=min(remaining - 20, 25))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_lf_r

    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        n1 = 2 * n0
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[i // 2]
        h_up_bin = make_binary(h_up, rng)
        h_lf_up, c5_up = fast_local_search_binary(h_up_bin, max_iters=25000, time_limit=min(remaining - 50, 50))
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_lf_up
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            h_sa_up, _ = sa_binary(best_h, n_iter=80000, t_start=2.0, t_end=0.01, rng=rng)
            h_lf_up2, c5_up2 = fast_local_search_binary(h_sa_up, max_iters=15000, time_limit=min(remaining - 20, 20))
            if c5_up2 < best_c5:
                best_c5 = c5_up2
                best_h = h_lf_up2
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            n_cur = len(best_h)
            h_soft_up = best_h + 0.02 * rng.standard_normal(n_cur)
            h_soft_up = project_onto_constraint(h_soft_up, n_cur / 2.0)
            h_opt_up, c5_opt_up = slsqp_refine(h_soft_up, 0.0005, 150)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up

    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_cur = len(best_h)
        h_polish = best_h + 0.005 * rng.standard_normal(n_cur)
        h_polish = project_onto_constraint(h_polish, n_cur / 2.0)
        h_opt, c5_opt = slsqp_refine(h_polish, 0.0001, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    n_final = len(best_h)
    final_h = project_onto_constraint(best_h, n_final / 2.0)
    if np.allclose(final_h, np.round(final_h), atol=0.01):
        final_h = np.round(final_h)
        n_ones = int(final_h.sum())
        target = n_final // 2
        if n_ones > target:
            ones_f = np.where(final_h == 1)[0]
            final_h[rng.choice(ones_f, n_ones - target, replace=False)] = 0
        elif n_ones < target:
            zeros_f = np.where(final_h == 0)[0]
            final_h[rng.choice(zeros_f, target - n_ones, replace=False)] = 1

    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_final
