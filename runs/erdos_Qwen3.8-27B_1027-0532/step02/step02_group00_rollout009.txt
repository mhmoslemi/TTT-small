    """Compute autocorrelation R[k] = sum_i h[i]*h[i+k] for k=0..n-1"""

def project_sum_box(h, target_sum):
    """Project h to {x: 0<=x[i]<=1, sum(x)=target_sum}"""

        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def compute_autocorr(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    return np.fft.irfft(H * np.conj(H), N)[:n]


def weighted_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    max_lag = min(n - 1, 200)
    for lag in range(1, max_lag + 1):
        w = weights[lag - 1]
        if w < 1e-8:
            continue
        grad[:n - lag] += w * h[lag:]
        grad[lag:] += w * h[:n - lag]
    return grad


def subgradient_phase(h, target, time_limit, alpha_init=20.0, eta_init=0.003):
    n = len(h)
    start = time.time()
    alpha = alpha_init
    eta = eta_init
    no_improve = 0
    best_local = h.copy()
    best_local_score = compute_c5(h)
    while time.time() - start < time_limit:
        A = compute_autocorr(h)
        neg_A = -A[1:]
        max_neg = np.max(neg_A)
        weights = np.exp(alpha * (neg_A - max_neg))
        sw = np.sum(weights)
        if sw < 1e-15:
            break
        weights = weights / sw
        grad = weighted_gradient(h, weights)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        h_new = h + eta * grad / gn
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_local_score:
            best_local_score = score
            best_local = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.03, 0.02)
        else:
            no_improve += 1
            if no_improve > 80:
                eta *= 0.75
                alpha *= 1.2
                no_improve = 0
        h = h_new
    return best_local, best_local_score


def coordinate_descent_phase(h, target, time_limit, delta_values=None, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    if delta_values is None:
        delta_values = [0.002, -0.002, 0.005, -0.005, 0.01, -0.01, 0.015, -0.015]
    best = h.copy()
    best_score = compute_c5(h)
    no_improve_sweeps = 0
    while time.time() - start < time_limit:
        improved_this_sweep = False
        order = rng.permutation(n)
        for i in order:
            for d in delta_values:
                h_trial = best.copy()
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = h_trial
                    improved_this_sweep = True
                    break
        no_improve_sweeps = 0 if improved_this_sweep else no_improve_sweeps + 1
        if not improved_this_sweep:
            delta_values = [d * 0.5 for d in delta_values]
            if max(abs(d) for d in delta_values) < 1e-5:
                break
    return best, best_score


def power_mean_obj(x, p, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    n = len(x)
    dx = 2.0 / n
    return float(dx * np.sum(corr ** p) ** (1.0 / p))


def slsqp_refine(h, target, time_limit, use_logsumexp=True):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    if use_logsumexp:
        alphas = [5, 15, 50, 150]
    else:
        alphas = [10, 50, 100, 200]

    n_alphas = len(alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        time_per = remaining / (n_alphas - ai)
        maxiter = int(max(100, time_per * 3))

        if use_logsumexp:
            def obj(x, a=alpha):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        else:
            def obj(x, p=alpha):
                return power_mean_obj(x, p, target)

        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    return best, best_score


def randomized_local_search(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    T = 0.02
    while time.time() - start < time_limit:
        T *= 0.995
        i = rng.randint(n)
        delta_scale = T * rng.uniform(0.5, 2.0)
        d = rng.uniform(-delta_scale, delta_scale)
        h_trial = best.copy()
        val = best[i] + d
        if val < 0.0 or val > 1.0:
            continue
        h_trial[i] = val
        h_trial = project_feasible(h_trial, target)
        s = compute_c5(h_trial)
        if s < best_score:
            best_score = s
            best = h_trial
        elif rng.rand() < np.exp(-(s - best_score) / max(T * 0.1, 1e-10)):
            best = h_trial
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    if tleft() > 200:
        h1, s1 = subgradient_phase(h, target, time_limit=200, alpha_init=15.0, eta_init=0.004)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    if tleft() > 250:
        h2, s2 = coordinate_descent_phase(best_h, target, time_limit=250, seed=seed)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    if tleft() > 200:
        h3, s3 = slsqp_refine(best_h, target, time_limit=200, use_logsumexp=True)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    if tleft() > 250:
        h4, s4 = coordinate_descent_phase(best_h, target, time_limit=250,
                                          delta_values=[0.001, -0.001, 0.003, -0.003,
                                                        0.005, -0.005, 0.01, -0.01],
                                          seed=seed + 1)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    if tleft() > 150:
        h5, s5 = slsqp_refine(best_h, target, time_limit=150, use_logsumexp=False)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    if tleft() > 200:
        h6, s6 = randomized_local_search(best_h, target, time_limit=200, seed=seed + 2)
        if s6 < best_score:
            best_score = s5 if s6 >= best_score else s6
            if s6 < best_score:
                best_score = s6
                best_h = h6

    if tleft() > 150:
        h7, s7 = coordinate_descent_phase(best_h, target, time_limit=150,
                                          delta_values=[0.0005, -0.0005, 0.001, -0.001,
                                                        0.002, -0.002, 0.004, -0.004],
                                          seed=seed + 3)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    if tleft() > 100:
        h8, s8 = subgradient_phase(best_h, target, time_limit=100, alpha_init=60.0, eta_init=0.001)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    if tleft() > 60:
        h9, s9 = coordinate_descent_phase(best_h, target, time_limit=max(30, tleft() - 10),
                                          delta_values=[0.0003, -0.0003, 0.0008, -0.0008,
                                                        0.001, -0.001, 0.003, -0.003],
                                          seed=seed + 4)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final
    else:
        h_final = best_h
    return h_final, best_score, n
