import time

def lse_val(overlaps, alpha):

def lse_gradient(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)

    threshold = 1e-8
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(2 * n - 1)  # This is wrong, grad should be length n

        w = weights[idx]

            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def projected_gradient_step(h, grad, lr, target):
    h_new = h - lr * grad
    return project_box_sum(h_new, target)


def nesterov_optimize(h0, alpha, n_iters, lr_init, target, momentum=0.9):
    """Nesterov accelerated projected gradient for LSE objective."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = lse_val(compute_overlaps(h), alpha)
    
    for it in range(n_iters):
        lr = lr_init / (1.0 + 0.01 * it)
        # Nesterov lookahead
        z = h + momentum * v
        z_clipped = np.clip(z, 0.0, 1.0)
        # Compute gradient at z (projected)
        grad_z = lse_gradient(z_clipped, alpha)
        v_new = momentum * v - lr * grad_z
        h_new = project_box_sum(h + v_new, target)
        v = h_new - h
        h = h_new
        
        val = lse_val(compute_overlaps(h), alpha)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val


def backtrack_line_search(h, grad, alpha, target, lr_init, max_iter=50):
    """Find a good step size using backtracking."""
    n = len(h)
    c1 = 1e-4
    rho = 0.5
    f0 = lse_val(compute_overlaps(h), alpha)
    lr = lr_init
    for _ in range(max_iter):
        h_new = project_box_sum(h - lr * grad, target)
        f_new = lse_val(compute_overlaps(h_new), alpha)
        if f_new <= f0 + c1 * lr * np.dot(grad, h_new - h):
            return h_new, lr
        lr *= rho
    return project_box_sum(h - lr * grad, target), lr


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_gradient(h, alpha)


def sum_eq_fun(h):
    return np.sum(h) - len(h) / 2.0


def sum_eq_jac(h):
    return np.ones(len(h))


def run_slq(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def penalty_fun(h, alpha, penalty):
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    s = float(np.sum(h)) - len(h) / 2.0
    obj += penalty * s * s
    grad = lse_gradient(h, alpha) + 2.0 * penalty * s * np.ones(len(h))
    return obj, grad


def run_lbfgs(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_fun, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None

def upscale(h, new_n):

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):

    hi = 2.0

        else:

def lse_val(overlaps, alpha):
    m = np.max(overlaps)
    return float(m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha)


def lse_gradient(h, alpha, top_k=None):
    n = len(h)
    overlaps = compute_overlaps(h)
    if top_k is not None and top_k < len(overlaps):
        top_indices = np.argsort(-overlaps)[:top_k]
        m = np.max(overlaps)
        exp_vals = np.zeros(len(overlaps))
        exp_vals[top_indices] = np.exp(alpha * (overlaps[top_indices] - m))
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
    else:
        m = np.max(overlaps)
        exp_vals = np.exp(alpha * (overlaps - m))
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
    threshold = 1e-8
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)

        else:
