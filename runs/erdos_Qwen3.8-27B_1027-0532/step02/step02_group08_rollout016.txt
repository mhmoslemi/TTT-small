                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16, 'iprint': 0}
            )
            h_new = project_to_constraints(np.clip(res.x, 0.0, 1.0), n_try)
            s = eval_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
                h_curr = h_new.copy()
                best_n = n_try

    # Phase 5: Random restarts with diverse strategies
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    max_restarts = 800
    k_arr = np.arange(n_best)

    for i in range(max_restarts):
        if time_left() < 40:
            break
        strategy = i % 10

        if strategy == 0:
            pert = best_h + rng.randn(n_best) * 0.04
        elif strategy == 1:
            pert = best_h + rng.randn(n_best) * 0.1
        elif strategy == 2:
            freq = rng.randint(1, max(2, n_best // 5))
            phase = rng.uniform(0.0, 2.0 * np.pi)
            amp = rng.uniform(0.05, 0.2)
            pert = best_h + amp * np.sin(2.0 * np.pi * freq * k_arr / n_best + phase)
        elif strategy == 3:
            L = rng.randint(max(5, n_best // 15), max(10, n_best // 4))
            start = rng.randint(0, max(1, n_best - L))
            pert = best_h.copy()
            pert[start:start + L] = 1.0 - pert[start:start + L]
        elif strategy == 4:
            L = max(5, n_best // 5)
            start = rng.randint(0, max(1, n_best - L))
            pert = best_h.copy()
            pert[start:start + L] = best_h[start:start + L][::-1]
        elif strategy == 5:
            k_val = max(3, n_best // 25)
            idx_top = np.argsort(-best_h)[:k_val]
            idx_bot = np.argsort(best_h)[:k_val]
            pert = best_h.copy()
            pert[idx_top] = 1.0 - pert[idx_top]
            pert[idx_bot] = 1.0 - pert[idx_bot]
        elif strategy == 6:
            center = rng.randint(0, n_best)
            sigma = rng.uniform(n_best / 25.0, n_best / 4.0)
            w = np.exp(-0.5 * ((k_arr - center) / sigma) ** 2)
            pert = best_h + w * rng.randn(n_best) * 0.25
        elif strategy == 7:
            corr = compute_corr(best_h)
            k_star = np.argmax(corr)
            r = k_star - (n_best - 1)
            grad = np.zeros(n_best)
            for j in range(n_best):
                if 0 <= j - r < n_best:
                    grad[j] += 1.0 - best_h[j - r]
                if 0 <= j + r < n_best:
                    grad[j] -= best_h[j + r]
            pert = best_h - 0.005 * grad
        elif strategy == 8:
           pert = best_h + rng.randn(n_best) * 0.03
            freq2 = rng.randint(2, max(3, n_best // 8))
            pert += 0.08 * np.cos(2.0 * np.pi * freq2 * k_arr / n_best)
        else:
            pert = best_h.copy()
            mask = rng.rand(n_best) < 0.15
            pert[mask] = np.clip(pert[mask] + rng.randn(int(np.sum(mask))) * 0.5, 0.0, 1.0)

        pert = project_to_constraints(pert, n_best)
        alpha_r = 150 + 600 * rng.randint(6)
        maxiter_r = max(40, min(250, 120000 // n_best))
        res = minimize(
            smooth_max_obj_grad, pert, method='L-BFGS-B',
            jac=True, args=(alpha_r, 10000.0),
            bounds=bounds_best,
            options={'maxiter': maxiter_r, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0}
        )
        h_new = project_to_constraints(np.clip(res.x, 0.0, 1.0), n_best)
        s = eval_score(h_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_new.copy()

    # Phase 6: Final high-alpha polish
    bounds_final = [(0.0, 1.0)] * len(best_h)
    for alpha in [5000, 10000, 20000, 40000, 80000]:
        if time_left() < 15:
            break
        res = minimize(
            smooth_max_obj_grad, best_h, method='L-BFGS-B',
            jac=True, args=(alpha, 50000.0),
            bounds=bounds_final,
            options={'maxiter': 200, 'ftol': 1e-20, 'gtol': 1e-16, 'iprint': 0}
        )
        h_new = project_to_constraints(np.clip(res.x, 0.0, 1.0), len(best_h))
        s = eval_score(h_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_new.copy()

    # Phase 7: Subgradient final polish
    h_sg2, s_sg2 = subgradient_polish(best_h, 2000)
    if s_sg2 < best_score:
        best_score = s_sg2
        best_h = h_sg2.copy()

    # Phase 8: Adjacent resolution check
    if time_left() > 30:
        for n_adj in [len(best_h) + 10, len(best_h) - 10, len(best_h) + 25]:
            if time_left() < 15:
                break
            if n_adj < 100 or n_adj > 3000:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_adj, endpoint=False)
            h_adj = np.interp(x_new, x_old, best_h)
            h_adj = project_to_constraints(h_adj, n_adj)
            bounds_adj = [(0.0, 1.0)] * n_adj
            res = minimize(
                smooth_max_obj_grad, h_adj, method='L-BFGS-B',
                jac=True, args=(3000, 10000.0),
                bounds=bounds_adj,
                options={'maxiter': 150, 'ftol': 1e-20, 'gtol': 1e-16, 'iprint': 0}
            )
            h_adj = project_to_constraints(np.clip(res.x, 0.0, 1.0), n_adj)
            s = eval_score(h_adj)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_adj.copy()

    best_h = project_to_constraints(best_h, len(best_h))
    best_score = eval_score(best_h)
    return (best_h, best_score, len(best_h))
