import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = float(np.sum(exp_vals))
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def _lse_obj(x, alpha):
    return lse_val(compute_overlaps(x), alpha)


def _lse_obj_grad(x, alpha):
    return lse_val(compute_overlaps(x), alpha), lse_grad(x, alpha)


def _sum_con(x):
    return np.sum(x) - len(x) / 2.0


def _sum_con_jac(x):
    return np.ones(len(x))


def lse_slsqp(h0, alpha, maxiter, bounds):
    try:
        res = minimize(_lse_obj, h0, args=(alpha,), method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha), bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return res.x
    except Exception:
        return None


def _pen_obj_grad(x, alpha, penalty):
    n = len(x)
    overlaps = compute_overlaps(x)
    obj = lse_val(overlaps, alpha)
    s = float(np.sum(x)) - n / 2.0
    obj += penalty * s * s
    g = lse_grad(x, alpha) + 2.0 * penalty * s
    return obj, g


def lbfgs_lse(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(_pen_obj_grad, h0, args=(alpha, penalty),
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16})
        return res.x
    except Exception:
        return None


def subgrad_momentum(h0, n_iters, lr_init, target, top_k, momentum):

    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(-overlaps)[:top_k]
