import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def threshold_to_binary(h, target_ones):
    # Threshold h to binary with exactly target_ones ones
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(len(h))
    binary[sorted_idx[:target_ones]] = 1.0
    return binary

def binary_correlation_scores(h_binary):
    # For binary h, compute corr[k] * dx for all k
    n = len(h_binary)
    dx = 2.0 / n
    corr = compute_correlations(h_binary)
    return corr * dx

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha

def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s
    
    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

def swap_delta(h, a, b, n):
    """
    Compute change in correlation when h[a] goes 1->0 and h[b] goes 0->1.
    Returns delta array of length 2n-1.
    """
    delta = np.zeros(2*n - 1)
    offset = n - 1  # index offset for lag
    
    # Part 1: h[a] 1->0, affects terms where i=a (1st factor)
    # corr[lag] includes h[a]*(1-h[a+lag]) for valid a+lag
    # After: 0*(1-h[a+lag]) = 0
    # Delta: -(1-h[a+lag]) = h[a+lag] - 1
    # Valid: 0 <= a < n (yes), 0 <= a+lag < n => -a <= lag <= n-1-a
    lo = max(-a, -(n-1))
    hi = min(n-1-a, n-1)
    lags = np.arange(lo, hi+1)
    idx = lags + offset
    delta[idx] += h[a + lags] - 1.0

    # Part 2: h[a] 1->0, affects terms where i+lag=a (2nd factor)
    # corr[lag] includes h[i]*(1-h[i+lag]) where i+lag=a => i=a-lag
    # Before: h[a-lag]*(1-h[a]) = h[a-lag]*0 = 0
    # After: h[a-lag]*(1-0) = h[a-lag]
    # Delta: +h[a-lag]
    # Valid: 0 <= a-lag < n => a-n < lag <= a => a-n+1 <= lag <= a
    lo = max(a-n+1, -(n-1))
    hi = min(a, n-1)
    lags = np.arange(lo, hi+1)
    idx = lags + offset
    delta[idx] += h[a - lags]

    # Part 3: h[b] 0->1, affects terms where i=b (1st factor)
    # Before: h[b]*(1-h[b+lag]) = 0
    # After: 1*(1-h[b+lag]) = 1-h[b+lag]
    # Delta: +(1-h[b+lag])
    # Valid: 0 <= b+lag < n => -b <= lag <= n-1-b
    lo = max(-b, -(n-1))
    hi = min(n-1-b, n-1)
    lags = np.arange(lo, hi+1)
    idx = lags + offset
    delta[idx] += 1.0 - h[b + lags]

    # Part 4: h[b] 0->1, affects terms where i+lag=b (2nd factor)
    # Before: h[i]*(1-h[b]) = h[i]*1 = h[i] where i=b-lag
    # After: h[i]*(1-1) = 0
    # Delta: -h[b-lag]
    # Valid: 0 <= b-lag < n => b-n < lag <= b => b-n+1 <= lag <= b
    lo = max(b-n+1, -(n-1))
    hi = min(b, n-1)
    lags = np.arange(lo, hi+1)
    idx = lags + offset
    delta[idx] -= h[b - lags]

    return delta

        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s
    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x
