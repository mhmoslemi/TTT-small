                best_h = h_curr.copy()
                n_best = n_try

    # Phase 3: Diversified restarts
    n_best = len(best_h)
    target_best = n_best / 2.0
    n_restarts = 300
    for i in range(n_restarts):
        if tl() < 120:
            break

        strategy = i % 8
        if strategy == 0:
            h_init = best_h + rng.randn(n_best) * 0.04
        elif strategy == 1:
            h_init = best_h + rng.randn(n_best) * 0.1
        elif strategy == 2:
            h_init = best_h.copy()
            n_flip = max(5, n_best // 20)
            idx = rng.choice(n_best, size=n_flip, replace=False)
            h_init[idx] = 1.0 - h_init[idx]
        elif strategy == 3:
            freq = rng.randint(3, 20)
            phase = rng.uniform(0, 1.0)
            duty = 0.4 + 0.2 * rng.rand()
            h_init = make_periodic_init(n_best, freq, phase, duty)
        elif strategy == 4:
            h_init = best_h.copy()
            n_seg = rng.randint(3, 10)
            boundaries = np.sort(rng.choice(np.arange(1, n_best), size=n_seg - 1, replace=False))
            boundaries = np.concatenate([[0], boundaries, [n_best]])
            for s in range(n_seg):
                val = rng.rand()
                h_init[boundaries[s]:boundaries[s + 1]] = val
        elif strategy == 5:
            center = rng.uniform(0.2, 1.8)
            width = rng.uniform(0.1, 0.5)
            h_init = 0.3 * best_h + 0.7 * make_gaussian_bump_init(n_best, center, width)
        elif strategy == 6:
            h_init = best_h.copy()
            mask = rng.rand(n_best) < 0.15
            h_init[mask] = rng.rand(int(np.sum(mask)))
        else:
            h_init = best_h.copy()
            noise = np.zeros(n_best)
            noise[0] = rng.randn() * 0.1
            for j in range(1, n_best):
                noise[j] = 0.85 * noise[j - 1] + rng.randn() * 0.08
            h_init += noise

        h_init = project(h_init, target_best)
        h_curr = h_init.copy()

        alpha_r = 100 + 200 * rng.randint(5)
        base_step_r = 0.01 * (n0 / float(n_best))
        max_it_r = max(60, min(300, 150000 // n_best))
        h_curr, s_local = optimize_smooth_max(h_curr, target_best, [alpha_r], max_it_r, base_step_r, 5)

        if s_local < best_score:
            best_score = s_local
            best_h = h_curr.copy()

    # Phase 4: SLSQP final polish
    if tl() > 60:
        bounds = [(0.0, 1.0)] * n_best
        for alpha in [5000, 10000, 20000]:
            if tl() < 25:
                break
            try:
                res_x = slsqp_optimize(best_h, alpha, 150, bounds)
                if res_x is not None:
                    h_new = project(np.clip(res_x, 0, 1), n_best / 2.0)
                    s = eval_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new.copy()
            except Exception:
                pass

    # Phase 5: Very fine topk polish
    if tl() > 30:
        for k_f in [2, 3, 5]:
            if tl() < 10:
                break
            h_curr, s_local = optimize_topk(best_h, n_best / 2.0, k_f, 200, 0.005)
            if s_local < best_score:
                best_score = s_local
                best_h = h_curr.copy()

    # Final: try a couple of nearby resolutions
    for n_adj in [n_best + 10, n_best - 10]:
        if tl() < 15:
            break
        if n_adj < 100 or n_adj > 2500:
            continue
        x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_adj, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project(h_init, n_adj / 2.0)
        h_curr, s_local = optimize_smooth_max(h_init, n_adj / 2.0, [2000], 100, 0.008, 5)
        if s_local < best_score:
            best_score = s_local
            best_h = h_curr.copy()
            n_best = n_adj

    best_h = project(best_h, len(best_h) / 2.0)
    best_score = eval_score(best_h)
    return (best_h, best_score, len(best_h))
