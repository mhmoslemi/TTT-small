from scipy.optimize import minimize, differential_evolution
import time

_t_start = 0.0
_t_budget = 1300.0

_gp = 4.0
_gl = 200.0
_gt = 200.0

_de_target = 0.0

def _remaining_time():
    return _t_budget - 30.0 - (time.time() - _t_start)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def _wgrad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        m = s + (n-1)
        w = weights[m]
        if w < 1e-14:
            continue
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            a = -s
            grad[:n-a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n-a]
    return grad

def _pm_objective(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_pos = np.maximum(corr, 0.0)
    n = len(h)
    dx = 2.0 / n
    if _gp <= 1.5:
        pm = np.sum(corr_pos) * dx
    else:
        total = np.sum(corr_pos ** _gp)
        pm = total ** (1.0 / _gp) * dx if total > 1e-30 else 0.0
    return pm + _gl * (np.sum(h) - _gt) ** 2

def _pm_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_pos = np.maximum(corr, 0.0)
    if _gp <= 1.5:
        total = np.sum(corr_pos)
        if total < 1e-30:
            pm_grad = np.zeros(n)
        else:
            weights = np.full(2*n-1, 1.0/total)
            pm_grad = _wgrad(h, weights) * dx
    else:
        total = np.sum(corr_pos ** _gp)
        if total < 1e-30:
            pm_grad = np.zeros(n)
        else:
            weights = corr_pos ** (_gp - 1) * (total ** ((1.0 - _gp) / _gp))
            pm_grad = _wgrad(h, weights) * dx
    return pm_grad + 2.0 * _gl * (np.sum(h) - _gt) * np.ones(n)

def _de_objective(x):
    h = project_feasible(x, _de_target)
    return compute_c5(h)

def _interpolate(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    if n_src == n_dst:
        return project_feasible(h_src.copy
