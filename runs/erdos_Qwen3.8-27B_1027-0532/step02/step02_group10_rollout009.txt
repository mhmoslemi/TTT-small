    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    # Binary search for the right offset
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def single_overlap_grad(h, m, n, dx):
    """Gradient of the overlap at correlate index m"""
    grad = np.zeros(n)
    offset = m - (n - 1)
    i_lo = max(0, -offset)
    i_hi = min(n, n - offset)
    if i_lo < i_hi:
        grad[i_lo:i_hi] += (1.0 - h[i_lo + offset:i_hi + offset])
    i_lo2 = max(0, offset)
    i_hi2 = min(n, n + offset)
    if i_lo2 < i_hi2:
        grad[i_lo2:i_hi2] -= h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def full_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    # Lipschitz constant estimate
    step = 1.0 / (alpha * dx * 2.0 * n + 1.0)
    step = min(step, 0.5)
    prev_val = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_val is not None:
            if sm < prev_val - 1e-14:
                stall = 0
            else:
                stall += 1
            if stall > 100:
                break
        prev_val = sm
        grad = full_gradient(h, weights, dx)
        # Backtracking line search
        s = step
        improved = False
        for _ in range(40):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-14:
                break
        if not improved:
            step = s
            break
        step = min(s * 1.2, 1.0)
    return h

def worst_lag_optimize(h, target, max_iters, deadline):
    """Optimize by focusing on the worst lag at each step"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = 0.3
    prev_score = evaluate_score(h)

            stall += 1
            step *= 0.8
        else:
