            j = rng.choice(list(ones))
            delta = 0.0
            for k in range(-n + 1, n):
                k_val = k
                term_i = 0.0
                if 0 <= i - k_val < n:
                    term_i += (1.0 - h[i - k_val])
                if 0 <= i + k_val < n:
                    term_i -= h[i + k_val]
                term_j = 0.0
                if 0 <= j - k_val < n:
                    term_j += (1.0 - h[j - k_val])
                if 0 <= j + k_val < n:
                    term_j -= h[j + k_val]
                delta += term_i - term_j
            delta *= dx
            candidates.append((delta + cur_score, i, j))
        
        if not candidates:
            break
        candidates.sort()
        best_delta, bi, bj = candidates[0]
        new_score = cur_score + best_delta
        
        accept = False
        if new_score < cur_score - 1e-15:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
            accept = True
        
        if accept:
            h[bi] = 1.0
            h[bj] = 0.0
            ones.discard(bi)
            zeros.add(bi)
            ones.discard(bj)
            zeros.add(bj)
            cur_score = new_score
            if cur_score < best - 1e-14:
                best = cur_score
                best_h = h.copy()
        
        if temp > 0:
            temp *= 0.9995
    return best_h, best

def fast_binary_scoring(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    return np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')) * dx

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    
    best_h = project_onto_constraint(init.copy(), target)
    best_score = evaluate_score(best_h)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Adam with increasing alpha
    alphas = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 16000]
    for alpha in alphas:
        if time_left() < 150:
            break
        dl = min(time.time() + 60, t0 + budget_s - 140)
        h, sc = adam_opt(best_h, alpha, target, 5000, dl, lr=0.03)
        h = project_onto_constraint(h, target)
        sc = evaluate_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    # Phase 2: Subgradient refinement
    if time_left() > 120:
        dl = time.time() + min(80, time_left() - 115)
        h = subgrad_refine(best_h, 8000, dl, lr=0.003)
        h = project_onto_constraint(h, target)
        sc = evaluate_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    # Phase 3: Binary local search
    if time_left() > 200:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target)
        if abs(h_bin.sum() - target) < 1.0:
            n_ones = int(round(h_bin.sum()))
            h_bin2 = np.zeros(n)
            top_idx = np.argsort(-best_h)[:n_ones]
            h_bin2[top_idx] = 1.0
            h_bin2 = project_onto_constraint(h_bin2, target)
            
            dl = time.time() + min(100, time_left() - 190)
            h_b, sc_b = binary_local_search(h_bin2, target, 20000, dl, rng, temp=0.001)
            sc_b = fast_binary_scoring(h_b)
            if sc_b < best_score:
                best_score = sc_b
                best_h = h_b.copy()
            
            # Try a few more rounds with different random seeds
            for trial in range(3):
                if time_left() < 180:
                    break
                h_b2 = (best_h > 0.5).astype(float)
                h_b2 = project_onto_constraint(h_b2, target)
                dl2 = time.time() + min(40, time_left() - 175)
                h_b3, sc_b3 = binary_local_search(h_b2, target, 10000, dl2, rng, temp=0.005)
                sc_b3 = fast_binary_scoring(h_b3)
                if sc_b3 < best_score:
                    best_score = sc_b3
                    best_h = h_b3.copy()
    
    # Phase 4: Continuous refinement from best found
    if time_left() > 100:
        for alpha in [5000, 10000, 20000, 50000]:
            if time_left() < 90:
                break
            dl = min(time.time() + 30, t0 + budget_s - 85)
            h, sc = adam_opt(best_h, alpha, target, 3000, dl, lr=0.01)
            h = project_onto_constraint(h, target)
            sc = evaluate_score(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    
    # Phase 5: Subgradient polish
    if time_left() > 50:
        dl = time.time() + min(35, time_left() - 45)
        h = subgrad_refine(best_h, 5000, dl, lr=0.001)
        h = project_onto_constraint(h, target)
        sc = evaluate_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    # Phase 6: Try higher resolution
    if time_left() > 80:
        for n_try in [n + 100, 2 * n]:
            if n_try > 2000 or time_left() < 70:
                continue
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_onto_constraint(h_interp, n_try / 2.0)
            for alpha in [200, 1000, 5000]:
                if time_left() < 60:

                h_hi, sc_hi = adam_opt(h_interp, alpha, n_try / 2.0, 3000, dl, lr=0.02)

                sc_hi = evaluate_score(h_hi)
                if sc_hi < best_score:
                    best_score = sc_hi
                    best_h = h_hi.copy()
            
            # Binary search at higher res
            if time_left() > 65:
                h_bin_hi = (best_h > 0.5).astype(float)
                h_bin_hi = project_onto_constraint(h_bin_hi, len(h_bin_hi) / 2.0)
                dl = time.time() + min(25, time_left() - 60)
                h_bh, sc_bh = binary_local_search(h_bin_hi, len(h_bin_hi) / 2.0, 15000, dl, rng, temp=0.002)
                sc_bh = fast_binary_scoring(h_bh)
                if sc_bh < best_score:
                    best_score = sc_bh
                    best_h = h_bh.copy()
    
    # Phase 7: Final Adam polish
    if time_left() > 20:
        dl = time.time() + min(12, time_left() - 15)
        nf = len(best_h)
        h, sc = adam_opt(best_h, 30000, nf / 2.0, 2000, dl, lr=0.005)
        h = project_onto_constraint(h, nf / 2.0)
        sc = evaluate_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
