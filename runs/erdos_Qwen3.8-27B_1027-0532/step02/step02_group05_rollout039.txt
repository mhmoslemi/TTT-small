import numpy as np
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlations(h)) * dx)


def project_feasible(h, target_sum):
    N = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def compute_A(h, m):
    N = len(h)
    Nc = 2 * N - 1
    A = np.zeros(Nc)
    j1 = N - 1 - m
    j2 = 2 * N - 2 - m
    if 0 <= j1 and j2 < Nc and j1 <= j2:
        ks = np.arange(j1, j2 + 1) - (N - 1)
        A[j1:j2 + 1] += (1.0 - h[m + ks])
    j3 = m
    j4 = m + N - 1
    if 0 <= j3 and j4 < Nc and j3 <= j4:
        idx = m + N - 1 - np.arange(j3, j4 + 1)
        A[j3:j4 + 1] += (-h[idx])
    return A


def solve_1d_minmax(corr, A_vec, lo, hi, n_iter=50):
    if lo >= hi:
        return lo
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        vals = corr + A_vec * mid
        max_val = vals.max()
        tol_act = 1e-10 * max(1.0, abs(max_val))
        active = vals >= max_val - tol_act
        if not np.any(active):
            break
        min_s = A_vec[active].min()
        max_s = A_vec[active].max()
        if min_s <= 0.0 and max_s >= 0.0:
            break
        elif max_s < 0.0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0


def ap_cd_sweep(h, corr, pairs, rng, n_bisect=50):
    N = len(h)
    for m in pairs:
        p = N - 1 - m
        if m >= p:
            continue
        old = h[m]
        A_m = compute_A(h, m)
        A_p = compute_A(h, p)
        A_eff = A_m - A_p
        new = solve_1d_minmax(corr, A_eff, 0.0, 1.0, n_bisect)
        delta = new - old
        if abs(delta) > 1e-14:
            corr += delta * A_eff
            h[m] = new
            h[p] = 1.0 - new
