import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    N = len(h)
    if N == 0:
        return h
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def h_to_g(h):
    n = len(h)
    m = (n - 1) // 2
    partner_idx = 2 * m - np.arange(m)
    g = (h[:m] + 1.0 - h[partner_idx]) / 2.0
    return np.clip(g, 0.0, 1.0)

def smooth_max_topk_val(corr, alpha, topk):
    vals = corr[topk]
    t = np.max(vals)
    return t + np.log(np.sum(np.exp(alpha * (vals - t)))) / alpha

def grad_smooth_max_topk(h, alpha, topk):
    N = len(h)
    corr = compute_correlations(h)
    vals = corr[topk]
    t = np.max(vals)
    ex = np.exp(alpha * (vals - t))
    sw = float(np.sum(ex))
    if sw < 1e-300:
        # All weights are zero, just use the max
        top_val = np.argmax(vals)
        weights = np.zeros(len(vals))
        weights[top_val] = 1.0
    else:
        weights = ex / sw
    grad = np.zeros(N)
    for idx in range(len(topk)):
        w = weights[idx]
        if w < 1e-15:
            continue
        j = int(topk[idx])
        k = j - (N - 1)
        # First term: (1-h[m+k]) for valid m where 0<=m<N and 0<=m+k<N
        m1 = max(0, -k)
        m2 = min(N, N - k)
        if m2 > m1:
            grad[m1:m2] += w * (1.0 - h[m1+k:m2+k])
        # Second term: -h[m-k] for valid m where 0<=m<N and 0<=m-k<N
        m3 = max(0, k)
        m4 = min(N, N + k)
        if m4 > m3:
            grad[m3:m4] -= w * h[m3-k:m4-k]
    return grad

def _obj_g(g, alpha, topk):
    h = antipal_h(g)
    corr = compute_correlations(h)
    return smooth_max_topk_val(corr, alpha, topk)

def _jac_g(g, alpha, topk):
    h = antipal_h(g)
    grad_h = grad_smooth_max_topk(h, alpha, topk)
    m = len(g)
    idx1 = np.arange(m)
    idx2 = 2 * m - np.arange(m)
    return grad_h[idx1] - grad_h[idx2]

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def optimize_anti_lbfgsb(g0, K, alpha_list, maxiter, time_limit_s):
    m = len(g0)
    bounds = [(0.0, 1.0)] * m
    best_g = g0.copy()
    best_score = evaluate_score(antipal_h(best_g))
    t_start = time.time()
    
    for alpha in alpha_list:
        if time.time() - t_start > time_limit_s:
            break
        h_cur = antipal_h(best_g)
        corr = compute_correlations(h_cur)
        topk = np.argsort(corr)[::-1][:K].astype(int)
        
        res = minimize(
            _obj_g, best_g,
            method='L-BFGS-B',
            bounds=bounds,
            jac=_jac_g,
            args=(alpha, topk),
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40}
        )
        if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
            g_new = np.clip(res.x, 0.0, 1.0)
            s = evaluate_score(antipal_h(g_new))
            if s < best_score:
                best_score = s
                best_g = g_new.copy()
    
    return best_g, best_score

    odd_resolutions = []

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    N = len(h)
    if N == 0:
        return h
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def h_to_g(h):
    n = len(h)
    m = (n - 1) // 2
    partner_idx = 2 * m - np.arange(m)
    g = (h[:m] + 1.0 - h[partner_idx]) / 2.0
    return np.clip(g, 0.0, 1.0)


def smooth_max_topk_val(corr, alpha, topk):
    vals = corr[topk]
    t = np.max(vals)
    return t + np.log(np.sum(np.exp(alpha * (vals - t)))) / alpha


def grad_smooth_max_topk(h, alpha, topk):
    N = len(h)
    corr = compute_correlations(h)
    vals = corr[topk]
    t = np.max(vals)
    ex = np.exp(alpha * (vals - t))
    sw = float(np.sum(ex))
    if sw < 1e-300:
        weights = np.zeros(len(vals))
        weights[int(np.argmax(vals))] = 1.0
    else:
        weights = ex / sw
    grad = np.zeros(N)
    for idx in range(len(topk)):
        w = weights[idx]
        if w < 1e-15:
            continue
        j = int(topk[idx])
        k = j - (N - 1)
        m1 = max(0, -k)
        m2 = min(N, N - k)
        if m2 > m1:
            grad[m1:m2] += w * (1.0 - h[m1 + k:m2 + k])
        m3 = max(0, k)
        m4 = min(N, N + k)
        if m4 > m3:
            grad[m3:m4] -= w * h[m3 - k:m4 - k]
    return grad


def _obj_g(g, alpha, topk):
    h = antipal_h(g)
    corr = compute_correlations(h)
    return smooth_max_topk_val(corr, alpha, topk)


def _jac_g(g, alpha, topk):
    h = antipal_h(g)
    grad_h = grad_smooth_max_topk(h, alpha, topk)
    m = len(g)
    idx1 = np.arange(m)
    idx2 = 2 * m - np.arange(m)
    return grad_h[idx1] - grad_h[idx2]


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def optimize_anti_lbfgsb(g0, K, alpha_list, maxiter, time_limit_s):
    m = len(g0)
    bounds = [(0.0, 1.0)] * m
    best_g = g0.copy()
    best_score = evaluate_score(antipal_h(best_g))
    t_start = time.time()

    for alpha in alpha_list:
        if time.time() - t_start > time_limit_s:
            break
        h_cur = antipal_h(best_g)
        corr = compute_correlations(h_cur)
        topk = np.argsort(corr)[::-1][:K].astype(np.int64)

        res = minimize(
            _obj_g, best_g,
            method='L-BFGS-B',
            bounds=bounds,
            jac=_jac_g,
            args=(alpha, topk),
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40}
        )
        if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
            g_new = np.clip(res.x, 0.0, 1.0)
            s = evaluate_score(antipal_h(g_new))
            if s < best_score:
                best_score = s
                best_g = g_new.copy()

    return best_g, best_score

    m = len(g0)
    bounds = [(0.0, 1.0)] * m

    t_start = time.time()

        if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
