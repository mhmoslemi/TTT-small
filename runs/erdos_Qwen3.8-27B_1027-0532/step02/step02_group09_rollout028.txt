    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0

        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:

        else:

    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(x, alpha):

    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha, top_k=None):

    If top_k is given, only use the top_k overlaps (approximation)."""

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):

        else:

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad_topk(h, alpha, top_k):
    n = len(h)
    overlaps = compute_overlaps(h)
    idx = np.argsort(-overlaps)[:top_k]
    m = overlaps[idx[0]]
    exp_vals = np.exp(alpha * (overlaps[idx] - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    
    grad = np.zeros(n)

        w = weights[j]
        L = idx_val - (n - 1)

            grad[L:] -= w * h[:n - L]
        else:
