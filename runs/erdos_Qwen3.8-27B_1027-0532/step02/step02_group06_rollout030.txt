import numpy as np

import time

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def grad_lse(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    es = np.exp(mu * (corr - mc))
    w = es / np.sum(es)
    grad = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0*h)
    for k in range(1, n):
        wp = w[n-1+k]
        wm = w[n-1-k]
        if wp < 1e-12 and wm < 1e-12:
            continue
        if wp > 1e-12:
            grad[:n-k] += wp * (1.0 - h[k:])
            grad[k:] -= wp * h[:n-k]
        if wm > 1e-12:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] -= wm * h[k:]
    grad *= dx
    return grad

def obj_lse(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(mu * (corr - mc)))) / mu) * dx

def gradient_descent_momentum(h0, mu_list, iters_per_mu, beta, t_end, seed):
    n = len(h0)
    target = n / 2.0
    h = project_to_feasible(h0.copy(), target)
    best_h = h.copy()
    best_s = eval_score(h)
    rng = np.random.RandomState(seed)
    
    for mu in mu_list:
        if time.time() > t_end:
            break
        v = np.zeros(n)
        alpha = max(0.01, 0.5 / (mu * 0.01))  # heuristic initial step
        alpha = min(alpha, 0.5)
        stale = 0
        for it in range(iters_per_mu):
            if time.time() > t_end:
                break
            g = grad_lse(h, mu)
            gn = np.sqrt(np.sum(g * g))
            if gn < 1e-14:
                break
            v = beta * v - alpha * g
            h_cand = project_to_feasible(h + v, target)
            s_cand = eval_score(h_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_h = h_cand.copy()
            # Accept or reject
            cur_s = eval_score(h)
            if s_cand <= cur_s + 1e-13:
                h = h_cand
                stale = 0
            else:
                alpha *= 0.6
                stale += 1
            if stale > 80:
                h = best_h + rng.randn(n) * (0.02 + 0.05 * min(stale, 200) / 200.0) / np.sqrt(n)
                h = project_to_feasible(h, target)
                v = np.zeros(n)
                alpha = 0.1
                stale = 0
    return best_h, best_s
