def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Add to positions with most room
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            room[i] = 0.0
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

∂L/∂λ = Σ h[i] - target_sum

We can alternate:
- Fix λ, minimize over h (with bounds [0,1]): this is a smooth bounded problem, use L-BFGS-B
- Update λ: λ += η * (Σ h[i] - target_sum)

This is a saddle-point optimization. It might work but convergence can be slow.

Alternatively, I'll stick with the projection approach but make it more robust:

```
for each iteration:
    compute gradient of smoothed obj
    adjust gradient to account for sum constraint (project onto tangent space)
    take step with backtracking
    project to [0,1]
    fix sum
```

Projecting gradient onto tangent space: the sum constraint Σh = S has normal vector (1,1,...,1). So we subtract the mean of the gradient:
grad_tangent = grad - mean(grad) * 1

This ensures the step doesn't change the sum (to first order).

```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        idx = np.where(room > 0)[0]
        for i in idx:
            if diff <= 1e-14:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
    else:
        idx = np.where(h > 0)[0]
        for i in idx:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smoothed_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    s = exp_c.sum()
    p = exp_c / s
    obj = (max_c + np.log(s) / alpha) * dx
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-18:
            continue
        L = idx - (n - 1)
        if L >= 0:
            if n > L:
                grad[:n-L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n-L]
        else:
            M = -L
            if n > M:
                grad[M:] += w * (1.0 - h[:n-M])
                grad[:n-M] -= w * h[M:]
    return obj, grad
```

```python
def _project_sum(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    h = h.copy()
    # First clip
    h = np.clip(h, 0.0, 1.0)
    # Now fix sum
    diff = target_sum - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to add mass. Add to elements with most room.
        # Uniform addition to free elements, then clip.
        free = h < 1.0
        n_free = free.sum()
        if n_free == 0:
            return h
        add_per = min(diff / n_free, 1.0 - h[free].max() if n_free > 0 else 0)
        # Actually, let's just do it iteratively for correctness
        # Uniform add to all free, clip, repeat
        for _ in range(10):
            free = h < 1.0
            n_free = free.sum()
            if n_free == 0 or diff <= 0:
                break
            add = diff / n_free
            h[free] += add
            h = np.clip(h, 0, 1)
            diff = target_sum - h.sum()
    else:
        # Need to remove mass
        for _ in range(10):
            free = h > 0.0
            n_free = free.sum()
            if n_free == 0 or diff >= 0:
                break
            sub = -diff / n_free
            h[free] -= sub
            h = np.clip(h, 0, 1)
            diff = target_sum - h.sum()
    return h
```

        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-15})
        h_out = _project(res.x, target_sum)
        return h_out
    except Exception:
        return h_init.copy()


def _make_starts(n, seed, target_sum):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    starts = []
    starts.append(np.full(n, 0.5))
    starts.append(0.5 + 0.2 * np.sin(2 * np.pi * x))
    starts.append(0.5 + 0.2 * np.sin(4 * np.pi * x))
    starts.append(0.5 + 0.15 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x))
    starts.append(0.5 + 0.2 * np.cos(3 * np.pi * x))
    starts.append(np.clip(0.5 + 0.3 * rng.randn(n), 0.0, 1.0))
    starts.append(np.clip(0.5 + 0.2 * rng.randn(n), 0.0, 1.0))
    starts.append(0.5 + 0.1 * rng.randn(n))
    h = np.zeros(n)
    period = max(2, n // 5)
    for i in range(n):
        h[i] = 1.0 if (i % (2 * period)) < period else 0.0
    h += 0.15 * rng.randn(n)
    starts.append(h)
    starts = [_project(s, target_sum) for s in starts]
    return starts


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30.0
    best_h = None
    best_c5 = np.inf
    best_n = 0

    n_list = [100, 140, 180]
    alloc = {100: 0.35, 140: 0.40, 180: 0.25}

    for n in n_list:
        n_deadline = t_start + budget_s * alloc[n]
        n_deadline = min(n_deadline, deadline)
        if time.time() > n_deadline - 30:
            continue
        target_sum = n / 2.0
        h_starts = _make_starts(n, seed, target_sum)
        if n == 100 and 'initial_h_values' in globals() and initial_h_values is not None:
            h0 = np.array(initial_h_values, dtype=float).copy()
            h0 = _project(h0, target_sum)
            h_starts.insert(0, h0)

        for si, h0 in enumerate(h_starts):
            if time.time() > n_deadline - 15:
                break
            h_opt, c5 = _optimize_anneal(h0, n, target_sum, n_deadline)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n

    if best_h is not None and time.time() < deadline - 40:
        hp = _slsqp_polish(best_h, best_n, deadline)
        c5p = _c5_of(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp

    if best_h is not None and time.time() < deadline - 60:
        n2 = best_n
        target_sum = n2 / 2.0
        rng2 = np.random.RandomState(seed + 7777)
        for trial in range(5):
            if time.time() > deadline - 20:
                break
            h_p = _project(best_h + 0.05 * rng2.randn(n2), target_sum)
            h_o, c5_o = _optimize_anneal(h_p, n2, target_sum, deadline)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _project(best_h, n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
