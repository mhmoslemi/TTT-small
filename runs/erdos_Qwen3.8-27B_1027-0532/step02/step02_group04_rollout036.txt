def fast_local_search(h, n, target_sum, rng, n_iterations, t_start=0.01, t_end=0.0001):
    """Fast local search with O(n) per step."""
    curr = h.copy()
    curr_corr = compute_correlations(curr)
    curr_max = np.max(curr_corr)
    best = curr.copy()
    best_max = curr_max
    
    # Precompute lag for each correlation index
    lags = np.arange(2*n-1) - (n-1)  # lags range from -(n-1) to (n-1)
    
    for it in range(n_iterations):
        t = t_start * (t_end / t_start) ** (it / max(1, n_iterations-1))
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Choose step
        step = rng.choice([0.01, 0.005, 0.002, 0.001, 0.0005])
        if rng.rand() < 0.5:
            step = -step
        
        new_i = curr[i] + step
        new_j = curr[j] - step
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        
        # Compute change in correlations
        # For each lag l, the change in corr at index (l + n-1) is:
        # delta * [(1-h[i-l]) * 1_{0<=i-l<n} - h[i+l] * 1_{0<=i+l<n}]
        #       - delta * [(1-h[j-l]) * 1_{0<=j-l<n} - h[j+l] * 1_{0<=j+l<n}]
        
        # Vectorized computation
        l = lags  # array of shape (2n-1,)
        
        # For position i:
        valid_i_minus = (i - l >= 0) & (i - l < n)
        valid_i_plus = (i + l >= 0) & (i + l < n)
        
        h_i_minus = np.where(valid_i_minus, curr[np.clip(i - l, 0, n-1)], 0.0)
        h_i_plus = np.where(valid_i_plus, curr[np.clip(i + l, 0, n-1)], 0.0)
        
        sens_i = np.where(valid_i_minus, 1.0 - h_i_minus, 0.0) - np.where(valid_i_plus, h_i_plus, 0.0)
        
        # For position j:
        valid_j_minus = (j - l >= 0) & (j - l < n)
        valid_j_plus = (j + l >= 0) & (j + l < n)

        sens_j = np.where(valid_j_minus, 1.0 - h_j_minus, 0.0) - np.where(valid_j_plus, h_j_plus, 0.0)
        
        delta_corr = step * (sens_i - sens_j)

            curr_max = new_max

                best = curr.copy()
    
    return best, best_max
