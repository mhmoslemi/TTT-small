    corr = compute_corr(h)
    n = len(h)
    
    if n_active is not None and n_active < len(corr):

        top_idx = np.argsort(corr)[-n_active:]

        weights /= np.sum(weights)

        w_full = np.zeros_like(corr)
        w_full[top_idx] = weights

        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        weights /= np.sum(weights)

            grad[mm:n] -= w * h[0:n-mm]

    y = h - step_size * grad

- Compute gradient at "lookahead" point: y = h + (β_{t-1} - 1)/β_t * (h - h_prev)
- h_new = project(y - (1/β_t) * step * grad(y))
- β_t = (1 + sqrt(1 + 4*β_{t-1}²))/2

This should give O(1/t²) convergence rate (for convex problems) vs O(1/t) for plain gradient. Even for non-convex, it often helps in practice.

Let me also think about whether there's a way to use the problem structure more effectively.

**Key insight**: The correlation vector corr has a specific structure. For h with values in [0,1] and sum = n/2:
- corr at lag 0: Σ h[i]*(1-h[i]) = Σ h[i] - Σ h[i]² = n/2 - Σ h[i]²
  - This is maximized when h[i] = 1/2 for all i (giving n/4)
  - This is minimized when h is as "binary" as possible (h[i] ∈ {0,1}, giving n/4 - 0 = ... wait)
  - Actually Σ h[i](1-h[i]) = Σ h[i] - Σ h[i]² = n/2 - Σ h[i]²
  - To minimize this, maximize Σ h[i]², which happens when h is binary
  - For binary h with n/2 ones: Σ h[i]² = n/2, so corr[0] = n/2 - n/2 = 0
  - Wait that's not right. If h is binary with n/2 ones and n/2 zeros:
    Σ h[i] = n/2, Σ h[i]² = n/2 (since h[i]² = h[i] for binary)
    corr[0] = Σ h[i](1-h[i]) = Σ h[i] - Σ h[i]² = n/2 - n/2 = 0
  
  Hmm, so for binary h, the correlation at lag 0 is 0. That's interesting!
  
  But wait, that can't be right for the min-max problem. If lag 0 has value 0, then the max must come from other lags.

Let me reconsider. For binary h with n/2 ones:
- corr at lag 0: Σ h[i](1-h[i]) = 0 (as computed)
- corr at lag m > 0: Σ_{i=0}^{n-1-m} h[i+m](1-h[i]) = number of positions where h[i+m]=1 and h[i]=0

So the problem for binary h is: minimize max_m (number of 1s in h shifted by m that land on 0s of h).

This is equivalent to: minimize max_m [Σ h[i+m] - Σ h[i+m]h[i]] = minimize max_m [(n/2 - boundary_correction) - |A ∩ (A-m)|]

where A is the set of positions where h=1.

For a "nice" set A (like an interval), the overlaps |A ∩ (A-m)| would be large for small m, giving small values of the expression. But for large m, the overlap drops, giving larger values.

This is related to the **additive combinatorics** of the set A. Sets with large additive energy (like intervals or arithmetic progressions) have large overlaps at many shifts.

For the continuous (non-binary) case, we can do better by having h take intermediate values, which smooths out the correlation.

One more optimization: instead of using `np.argsort` to find the top-k lags every iteration (which is O(n log n)), I can maintain a running estimate. But for n=400, argsort is fast enough.

Let me also add a **coordinate descent** step as an alternative to full gradient:
- Pick a random subset of coordinates
- Optimize over just those coordinates (keeping others fixed)
- This is cheap and can escape some local minima

Actually, for the bilinear objective, coordinate descent has nice properties. If we fix all of h except h[j], the objective becomes:
f(h) = max_k Σ_i h[i](1-h[i+k])
For a fixed k, the term involving h[j] is:
h[j](1-h[j-k]) + (1-h[j])h[j+k] = h[j] - h[j]h[j-k] + h[j+k] - h[j]h[j+k]
= h[j] * (1 - h[j-k] - h[j+k]) + h[j+k]

This is linear in h[j]! So for a fixed active lag k, the optimal h[j] is:
- If coefficient (1 - h[j-k] - h[j+k]) > 0: h[j] = 0
- If coefficient < 0: h[j] = 1
- If coefficient = 0: any value

But we have the max over k, so it's more complex. And we have the sum constraint.

This suggests a **block coordinate descent** approach where we alternate between:
1. For each j, set h[j] to minimize the (smoothed) max correlation, given the sum constraint
2. This is like a "water-filling" problem

But implementing this correctly with the sum constraint is tricky. Let me stick with the projected gradient approach.

import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0
_g_n = 0
_g_alpha = 50.0
_g_k_active = 30
_g_t_start = 0.0
_g_budget = 1300.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(y, target_sum):
    lo = y.min() - 1.0
    hi = 1.0 - y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad(h, alpha, k_active):
    n = len(h)
    corr = compute_corr(h)
    if k_active < len(corr):
        top_idx = np.argsort(corr)[-k_active:]
    else:
        top_idx = np.arange(len(corr))
    corr_sub = corr[top_idx]
    shifted = alpha * corr_sub
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        m = top_idx[i] - (n - 1)
        if m >= 0:
            grad[m:n] += w * (1.0 - h[0:n - m])
            grad[0:n - m] -= w * h[m:n]
        else:
            p = -m
            grad[0:n - p] += w * (1.0 - h[p:n])
            grad[p:n] -= w * h[0:n - p]
    return grad, corr


def smooth_max_obj(h, alpha, k_active):
    corr = compute_corr(h)
    if k_active < len(corr):
        top = np.sort(corr)[-k_active:]
    else:
        top = corr
    mx = top[-1]
    shifted = alpha * (top - mx)
    return mx + np.log(np.sum(np.exp(shifted))) / alpha


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_obj(x):
    return smooth_max_obj(x, _g_alpha, _g_k_active)


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_refine(h, n, target, alphas, maxiter=300):
    global _g_target, _g_n, _g_alpha, _g_k_active
    _g_target = target
    _g_n = n
    _g_k_active = 30
    h_cur = h.copy()
    for a in alphas:
        _g_alpha = a
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def project_grad_descent(h, n, target, t_limit, rng):
    h_cur = h.copy()
    f_cur = compute_c5(h_cur)
    best_f = f_cur
    h_best = h_cur.copy()
    momentum = np.zeros(n)
    beta_mom = 0.82
    alpha = 25.0
    k_act = 15
    step_size = 0.25
    t0 = time.time()
    stall = 0

    while time.time() - t0 < t_limit:
        grad, corr = compute_grad(h_cur, alpha, k_act)
        momentum = beta_mom * momentum + grad
        h_new = project_feasible(h_cur - step_size * momentum, target)
        f_new = compute_c5(h_new)

        if f_new < f_cur - 1e-15:
            h_cur = h_new
            f_cur = f_new
            stall = 0
            if f_cur < best_f - 1e-10:
                best_f = f_cur
                h_best = h_cur.copy()
        else:
            stall += 1
            if stall > 40:
                noise_scale = 0.02 + 0.01 * min(stall / 50.0, 2.0)
                h_cur = project_feasible(h_cur + rng.randn(n) * noise_scale, target)
                f_cur = compute_c5(h_cur)
                momentum = np.zeros(n)
                step_size = 0.2
                stall = 0
            elif stall > 15:
                step_size *= 0.6
                momentum = beta_mom * momentum
                h_new = project_feasible(h_cur - step_size * momentum, target)
                f_new = compute_c5(h_new)
                if f_new < f_cur:
                    h_cur = h_new
                    f_cur = f_new
            else:
                step_size *= 0.85

        alpha = min(400.0, alpha * 1.004)
        step_size = min(0.35, step_size * 1.015)

    return h_best, best_f


def accelerated_pg(h, n, target, t_limit, rng):
    h_cur = h.copy()
    h_prev = h.copy()
    f_cur = compute_c5(h_cur)
    best_f = f_cur
    h_best = h_cur.copy()
    beta_t = 1.0
    t0 = time.time()
    alpha = 40.0
    k_act = 10
    base_step = 0.15
    stall = 0

    while time.time() - t0 < t_limit:
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_t * beta_t)) / 2.0
        gamma = (beta_t - 1.0) / beta_new
        y = h_cur + gamma * (h_cur - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        grad, corr = compute_grad(y, alpha, k_act)
        step = base_step / beta_new
        h_new = project_feasible(y - step * grad, target)
        f_new = compute_c5(h_new)

        if f_new < f_cur:
            h_prev = h_cur
            h_cur = h_new
            f_cur = f_new
            beta_t = beta_new
            stall = 0
            if f_cur < best_f:
                best_f = f_cur
                h_best = h_cur.copy()
        else:
            stall += 1
            if stall > 25:
                h_cur = project_feasible(h_cur + rng.randn(n) * 0.03, target)
                f_cur = compute_c5(h_cur)
                h_prev = h_cur.copy()
                beta_t = 1.0
                stall = 0
            else:
                beta_t = beta_new

        alpha = min(300.0, alpha * 1.003)

    return h_best, best_f


def multi_lag_descent(h, n, target, t_limit, rng):
    h_cur = h.copy()
    f_cur = compute_c5(h_cur)
    best_f = f_cur
    h_best = h_cur.copy()
    t0 = time.time()
    alpha = 80.0
    k_act = 8
    step = 0.1

    while time.time() - t0 < t_limit:
        grad, corr = compute_grad(h_cur, alpha, k_act)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-8:
            break
        direction = -grad / g_norm
        h_new = project_feasible(h_cur + step * direction * g_norm, target)
        f_new = compute_c5(h_new)
        if f_new < f_cur:
            h_cur = h_new
            f_cur = f_new
            step = min(0.3, step * 1.02)
            if f_cur < best_f:
                best_f = f_cur
                h_best = h_cur.copy()
        else:
            step *= 0.7
            if step < 0.003:
                h_cur = project_feasible(h_cur + rng.randn(n) * 0.02, target)
                f_cur = compute_c5(h_cur)
                step = 0.1

        alpha = min(500.0, alpha * 1.002)

    return h_best, best_f
