    corr = np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0

    n = len(h)

    else:

        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _nesterov_weighted(h0, n, target_sum, n_iters, alpha0, n_lags, temp):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = np.max(corr)
        weights = np.exp(temp * (corr - max_c))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        top_idx = np.argsort(weights)[::-1][:n_lags]
        w = weights[top_idx].copy()
        w = w / w.sum()
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            g = g + w[idx] * _grad_corr_term(y, int(top_idx[idx]), dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _exact_coord_descent(h0, target_sum, max_rounds, n_grid):

    h = h0.copy()

        for j in range(n):
            v = h[j]

                    best_t = t_val
            h[j] = best_t
            if best_t != v:
                improved = True
        h = _fix_sum(h, target_sum)
        h = np.clip(h, 0.0, 1.0)
        if not improved:
            break
    return h

                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx


def _nesterov_weighted(h0, n, target_sum, n_iters, alpha0, n_lags, temp):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = np.max(corr)
        weights = np.exp(temp * (corr - max_c))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights = weights / total_w
        top_idx = np.argsort(weights)[::-1][:n_lags]
        w = weights[top_idx].copy()
        w = w / w.sum()
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            g = g + w[idx] * _grad_corr_term(y, int(top_idx[idx]), dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val


def _exact_coord_descent(h0, target_sum, max_rounds, n_grid):
    n = len(h0)
    h = h0.copy()
    grid = np.linspace(0.0, 1.0, n_grid)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            v = h[j]
            best_t = v
            best_c = _c5_of(h)
            for t_val in grid:
                if abs(t_val - v) < 1e-10:
                    continue
                h[j] = t_val
                c = _c5_of(h)
                if c < best_c:
                    best_c = c
                    best_t = t_val
            h[j] = best_t
            if best_t != v:
                improved = True
        h = _fix_sum(h, target_sum)
        h = np.clip(h, 0.0, 1.0)
        if not improved:
            break
    return h


def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _slsqp_target, _slsqp_n
    _slsqp_target = target_sum
    _slsqp_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _slsqp_con_fun(x):
    return x.sum() - _slsqp_target


def _slsqp_con_jac(x):
    return np.ones(_slsqp_n)


def _slsqp_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)


def _smooth_coord_descent(h0, target_sum, max_rounds):
    n = len(h0)
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            v = h[j]
            best_t = v
            best_c = _c5_of(h)
            candidates = [0.0, 1.0]
            for frac in range(1, 10):
                candidates.append(frac / 10.0)
            for t_val in candidates:
                if abs(t_val - v) < 1e-10:

                h[j] = t_val
                c = _c5_of(h)
                if c < best_c:
                    best_c = c
                    best_t = t_val

                improved = True
        h = _fix_sum(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n = n_init
    target = n / 2.0
    h = _fix_sum(h_init.copy(), target)
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n

    if time.time() < deadline - 200:
        h, c5 = _nesterov_weighted(h, n, target, 250, 0.35, 10, 30)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

    if time.time() < deadline - 180:
        h = _exact_coord_descent(h, target, 80, 11)
        c5 = _c5_of(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

    if time.time() < deadline - 160:
        h, c5 = _nesterov_weighted(h, n, target, 250, 0.15, 25, 100)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

    if time.time() < deadline - 140:
        h = _exact_coord_descent(h, target, 100, 21)
        c5 = _c5_of(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

    n_candidates = sorted(set([n_init - 8, n_init - 6, n_init - 4, n_init - 2,
                               n_init + 2, n_init + 4, n_init + 6, n_init + 10,
                               n_init + 20]))
    for n_try in n_candidates:
        if n_try < 20 or n_try > 400 or n_try % 2 != 0:
            continue
        if time.time() > deadline - 120:
            break
        target_try = n_try / 2.0
        x_old = np.linspace(0.0, 1.0, n_init)
        x_new = np.linspace(0.0, 1.0, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = _fix_sum(h_try, target_try)

        h_try, c5_try = _nesterov_weighted(h_try, n_try, target_try, 200, 0.3, 12, 40)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try.copy(), n_try

        if c5_try < best_c5 + 0.003 and time.time() < deadline - 60:
            h_try = _exact_coord_descent(h_try, target_try, 80, 21)
            c5_try = _c5_of(h_try)
            if c5_try < best_c5:
                best_c5, best_h, best_n = c5_try, h_try.copy(), n_try

    n_cur = best_n
    target_cur = n_cur / 2.0
    perturb_sizes = [8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04]
    for size in perturb_sizes:
        for _ in range(4):
            if time.time() > deadline - 60:
                break
            h_pert = best_h + rng.uniform(-size, size, n_cur)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = _fix_sum(h_pert, target_cur)
            h_pert, c5_p = _nesterov_weighted(h_pert, n_cur, target_cur, 150, 0.25, 12, 40)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_pert.copy()
            if c5_p < best_c5 + 0.002 and time.time() < deadline - 40:
                h_pert = _exact_coord_descent(h_pert, target_cur, 60, 15)
                c5_p = _c5_of(h_pert)
                if c5_p < best_c5:
                    best_c5, best_h = c5_p, h_pert.copy()

    rng2 = np.random.RandomState(seed + 1)
    for _ in range(6):
        if time.time() > deadline - 50:
            break
        mask = rng2.rand(n_cur) < 0.3
        h_pert = best_h.copy()
        h_pert[mask] = rng2.rand(int(mask.sum()))
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = _fix_sum(h_pert, target_cur)
        h_pert, c5_p = _nesterov_weighted(h_pert, n_cur, target_cur, 120, 0.3, 10, 30)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_pert.copy()
        if c5_p < best_c5 + 0.002 and time.time() < deadline - 30:
            h_pert = _exact_coord_descent(h_pert, target_cur, 50, 15)
            c5_p = _c5_of(h_pert)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_pert.copy()

    n_final = best_n
    target_final = n_final / 2.0

    if time.time() < deadline - 40:
        best_h = _exact_coord_descent(best_h, target_final, 100, 31)
        best_c5 = _c5_of(best_h)

    if time.time() < deadline - 25:
        best_h = _exact_coord_descent(best_h, target_final, 80, 61)
        best_c5 = _c5_of(best_h)

    if time.time() < deadline - 15:
        best_h = _exact_coord_descent(best_h, target_final, 60, 101)
        best_c5 = _c5_of(best_h)

    if time.time() < deadline - 10:
        best_h = _exact_coord_descent(best_h, target_final, 40, 201)
        best_c5 = _c5_of(best_h)

    if time.time() < deadline - 8:
        best_h = _exact_coord_descent(best_h, target_final, 30, 501)
        best_c5 = _c5_of(best_h)

    best_h = _fix_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n_final)
