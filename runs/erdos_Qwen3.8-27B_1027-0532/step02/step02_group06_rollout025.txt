        d[lag:] += (1.0 - h[:-lag])
        d[:n - lag] -= h[lag:]
    else:

        d[:n - la] += (1.0 - h[la:])

    s = alpha * corr * dx
    s_max = np.max(s)
    w = np.exp(s - s_max)
    
    # Get top indices
    order = np.argsort(-w)
    top = order[:n_top]
    w_top = w[top]
    w_top = w_top / w_top.sum()
    
    grad = np.zeros(n)
    for i in range(len(top)):
        if w_top[i] < 1e-14:
            break
        lag = int(top[i] - (n - 1))
        grad += w_top[i] * dx * gradient_for_lag(h, lag, n)
    return grad

            break

        alpha = 5.0 + 200.0 * (it / total_iters) ** 2

            t_mom = t_new

        grad_norm = np.linalg.norm(grad)

                h_prev = h

            # Try from best_h directly
            step = max(step, 0.01)
            h_prev = h
            # Reset momentum
            t_mom = 1.0

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def gradient_for_lag(h, lag, n):
    d = np.zeros(n)
    if lag == 0:
        d[:] = 1.0 - 2.0 * h
    elif lag > 0:
        d[lag:] += (1.0 - h[:-lag])
        d[:n - lag] -= h[lag:]
    else:
        la = -lag
        d[:n - la] += (1.0 - h[la:])

def softmax_gradient(h, alpha, n_top=40):

def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        scaled = alpha * (corr - m)
        return m + np.log(np.sum(np.exp(scaled))) / alpha

    def con_fun(hv):
        return np.sum(hv) - target_sum

    def con_jac(hv):
        return np.ones(n)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        pass
    return None
