def _slsqp_with_grad(h_init, n, alpha, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m))
        weights = exp_vals / np.sum(exp_vals)
        
        grad = np.zeros(npts)
        for k in range(2 * npts - 1):
            if weights[k] < 1e-15:
                continue
            lag = k - (npts - 1)
            w = weights[k]
            if lag >= 0:
                if lag == 0:
                    grad += w * (1.0 - 2.0 * x)
                else:
                    grad[:npts-lag] += w * (1.0 - x[lag:])
                    grad[lag:] -= w * x[:npts-lag]
            else:
                L = -lag
                grad[L:] += w * (1.0 - x[:npts-L])
                grad[:npts-L] -= w * x[L:]
        
        return grad * dx
    
    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(npts)}]
    
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(x, dx, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0, np.zeros(len(x))
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    if max_shifted > 700:
        shifted = shifted - max_shifted + 1
        m += max_shifted / alpha
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return (m + np.log(s) / alpha) * dx, exp_vals / s

def _smooth_grad(x, dx, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    if max_shifted > 700:
        shifted = shifted - max_shifted + 1
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += w * (1.0 - 2.0 * x)
            else:
                grad[:n - lag] += w * (1.0 - x[lag:])
                grad[lag:] -= w * x[:n - lag]
        else:
            L = -lag
            grad[L:] += w * (1.0 - x[:n - L])
            grad[:n - L] -= w * x[L:]
    return grad * dx

def _slsqp_solve(h_init, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    target_sum = n / 2.0
    npts = n

    def obj(x):
        val, _ = _smooth_obj(x, dx, alpha)
        return val

    def obj_grad(x):
        return _smooth_grad(x, dx, alpha)

    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(npts)}]
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        try:
            res = minimize(obj, h_init, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            return res.x
        except Exception:
            return h_init.copy()

def _curriculum(h_init, n, alphas, deadline):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        h_new = _slsqp_solve(h, n, alpha, maxiter=300, ftol=1e-14)
        h_new = _fix_sum(h_new, n / 2.0)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _adam_optimize(h_init, n, n_iter, lr0, target_sum, deadline, top_k=3, check_every=300):
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = _c5_of(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    dx = 2.0 / n

    for i in range(n_iter):
        if time.time() > deadline - 5:
            break
        t += 1
        lr = lr0 / (1.0 + 0.0005 * (t - 1))
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:

            w = np.exp(alpha_shift(corr[k], np.max(corr)))
            if lag >= 0:

                else:
                    grad[:n - lag] += (1.0 - h[lag:])
                    grad[lag:] -= h[:n - lag]
            else:
                L = -lag
                grad[L:] += (1.0 - h[:n - L])
                grad[:n - L] -= h[L:]
        grad /= top_k
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad

def alpha_shift(val, max_val):
    d = max_val - val
    if d > 50:
        return 1.0
    return np.exp(-0.1 * d)

def _perturb_refine(h_best, n, target_sum, seed, deadline, n_attempts=10):

    best = h_best.copy()
    best_c5 = _c5_of(h_best)
