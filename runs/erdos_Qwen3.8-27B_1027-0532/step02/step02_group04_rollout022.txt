        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid

    # Lag 0 (idx = N-1)
    w0 = weights[N-1]
    grad += w0 * (1.0 - 2.0 * h)
    
    # Positive and negative lags
    for k in range(1, N):
        # +k lag (idx = N-1+k)
        wp = weights[N-1+k]
        # -k lag (idx = N-1-k)
        wn = weights[N-1-k]
        
        for m in range(N):
            g = 0.0
            # +k contribution
            if wp > 1e-15:
                if m >= k:
                    g += wp * (1.0 - h[m-k])
                if m <= N-1-k:
                    g -= wp * h[m+k]
            # -k contribution
            if wn > 1e-15:
                if m <= N-1-k:
                    g += wn * (1.0 - h[m+k])
                if m >= k:
                    g -= wn * h[m-k]
            grad[m] += g
    
    return grad
```

This is O(N²) but with a lot of Python overhead. For N=425, that's 425*425 = 180K iterations of the inner loop, each with a few operations. In pure Python this would be slow, but we can vectorize:

```python
def smooth_max_gradient(h, alpha):

    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    
    # Lag 0
    grad += weights[N-1] * (1.0 - 2.0 * h)
    
    # For each |lag| = k, combine +k and -k contributions
    for k in range(1, N):
        wp = weights[N-1+k]
        wn = weights[N-1-k]
        
        # Upper part: m = k to N-1
        # +k: wp * (1-h[m-k])
        # -k: -wn * h[m-k]
        # Combined: wp - (wp + wn) * h[m-k]  for m = k..N-1
        if wp + wn > 1e-15:
            grad[k:] += wp - (wp + wn) * h[:N-k]
        
        # Lower part: m = 0 to N-1-k
        # +k: -wp * h[m+k]
        # -k: wn * (1-h[m+k])
        # Combined: wn - (wp + wn) * h[m+k]  for m = 0..N-1-k
        if wp + wn > 1e-15:
            grad[:N-k] += wn - (wp + wn) * h[k:]
    
    return grad

- Total: wp*(1-h[m-k]) - wn*h[m-k] = wp - (wp+wn)*h[m-k]  ✓

And the "lower part" (m from 0 to N-1-k):

```python
def compute_Ph(h, p, fft_size=None):
    """Compute P*h where P = sum_k p_k * S_k (shift matrices)."""
    N = len(h)
    p_ext = np.zeros(3*N - 2)
    p_ext[N-1 : 2*N-1] = p  # p has length 2N-1, indexed from lag -(N-1) to N-1
    # Wait, p has length 2N-1 where p[idx] corresponds to the idx-th element of the correlation
    # The idx-th element corresponds to lag = idx - (N-1)
    # So p_ext[N-1+k] should be p[k + (N-1)] for k = -(N-1),...,N-1
    # i.e., p_ext[j] = p[j] for j = 0,...,2N-2 (just copy p into p_ext at the right offset)
    
    # Actually: p_ext[N-1+k] = p_k where p_k is the weight for lag k
    # And in the correlation output, the element at index idx = N-1+k has weight p[idx]
    # So p_ext[N-1+k] = p[N-1+k] = p[idx]
    # This means p_ext[0:2N-1] = p (just copy p into the first 2N-1 positions of p_ext)
    
    p_ext[:2*N-1] = p
    
    if fft_size is None:
        # Use direct convolution for small sizes
        result = np.convolve(p_ext, h, mode='full')
    else:
        # Use FFT
        f_p = np.fft.rfft(p_ext, fft_size)
        f_h = np.fft.rfft(h, fft_size)
        result = np.fft.irfft(f_p * f_h, fft_size)
    
    return result[N-1 : 2*N-1]

Wait, but we also need to think about what "project" means here. The projection onto {0≤h≤1, ∑h=N/2} is:

2. Adjust by a constant to make ∑h = N/2 (using bisection)

But there's a subtlety: after the gradient step, the values might be far from [0,1]. The projection will clip them, which is fine for the box constraint but might not be the optimal projection for the sum constraint. The bisection handles this correctly.

Let me also think about convergence. The alternating method is:
1. Fix h, optimize p (trivial: p = argmax or softmax)
2. Fix p, optimize h (projected gradient on quadratic)

This is block coordinate descent, which converges for the inner problem (convex maximization of a concave function... wait, we're maximizing h^T P h which is NOT concave in general!).

If P is positive semidefinite, then h^T P h is convex, and we're doing convex maximization, which is hard. But in practice, P is likely indefinite (since it's a sum of shift matrices with mixed signs in the spectrum), so the problem is non-convex.

This means the projected gradient might not converge to the global optimum of the inner problem, but it should still make progress. The key is that we're using it as part of an outer loop, so even if we don't fully solve the inner problem, we make progress.

Actually, I just realized something. The quadratic h^T P h can be written as:
h^T P h = Σ_k p_k * Σ_i h[i] * h[i+k]

For a binary h (h[i] ∈ {0,1}), this counts the number of "1-1" pairs at each lag. For a fractional h, it's a weighted count.

The overlap O(k) = Σ_i h[i] * (1-h[i+k]) = Σ_i h[i] - Σ_i h[i]*h[i+k]

So minimizing O(k) is equivalent to maximizing Σ_i h[i]*h[i+k] for that specific k.

The max over k of O(k) is minimized when all the lags are "balanced" - no single lag has too many "1-1" pairs.

This gives intuition: the optimal h should be as "delocalized" as possible, spreading the 1s evenly so that no shift aligns too many 1s with other 1s.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    grad += weights[N-1] * (1.0 - 2.0 * h)
    
    for k in range(1, N):
        wp = weights[N-1+k]
        wn = weights[N-1-k]
        sw = wp + wn
        if sw > 1e-15:
            grad[k:] += wp - sw * h[:N-k]
            grad[:N-k] += wn - sw * h[k:]
    
    return grad

def compute_Ph(h, p):
    """Compute P*h where P = sum_k p_k S_k, using convolution."""
    N = len(h)
    p_ext = np.zeros(3*N - 2)
    p_ext[:2*N-1] = p
    result = np.convolve(p_ext, h, mode='full')
    return result[N-1:2*N-1]

def alternating_method(h, n_iters, alpha_start, alpha_end, step_size, mu, n_pts, rng):
    """Alternating projected gradient method."""
    best_h = h.copy()
    best_score = evaluate_score(h)
    N = len(h)
    
    for it in range(n_iters):
        # Animate alpha
        frac = (it + 1) / float(n_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Compute overlaps
        corr = compute_correlations(best_h)
        dx = 2.0 / N
        overlaps = corr * dx
        
        # Compute attention weights (softmax)
        max_ov = np.max(overlaps)
        exp_ov = np.exp(alpha * (overlaps - max_ov))
        p = exp_ov / np.sum(exp_ov)
        
        # Projected gradient steps
        h_cur = best_h.copy()
        n_pg = 5 + 10 * frac  # more steps as alpha increases
        for pg_i in range(n_pg):
            grad = 2.0 * compute_Ph(h_cur, p)
            h_cur = h_cur + step_size * grad
            h_cur = project_feasible(h_cur, N / 2.0)
        
        # Evaluate
        score = evaluate_score(h_cur)
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
        
    return best_h, best_score

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        sm = m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

def targeted_local_search(h, n_local_iters, rng):
    """Local search focused on the worst lags."""
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(best)
    dx = 2.0 / N
    
    for _ in range(n_local_iters):
        # Find worst lags
        corr = compute_correlations(best)
        # Focus on top 5 lags
        top_k = np.argsort(corr)[-5:]
        k = top_k[rng.randint(5)]
        lag = k - (N - 1)
        
        # Try to reduce overlap at this lag
        if abs(lag) == 0:
            # Try to make h more "balanced" (closer to 0.5)
            i = rng.randint(N)
            if best[i] > 0.5:
                h_try = best.copy()
                h_try[i] -= 0.01
                h_try[rng.randint(N)] += 0.01
            else:
                h_try = best.copy()
                h_try[i] += 0.01
                h_try[rng.randint(N)] -= 0.01
        else:
            # Find a pair (i, i+lag) where we can reduce overlap
            # Overlap at lag: sum h[i]*(1-h[i+lag]) for valid i
            # To reduce: either decrease h[i] where h[i+lag] is low, or increase h[i+lag] where h[i] is high
            if lag > 0 and lag < N:
                i = rng.randint(N - lag)
                # Try transferring from i to i+lag or vice versa
                for delta in [0.005, 0.01, 0.02, -0.005, -0.01, -0.02]:
                    h_try = best.copy()
                    h_try[i] += delta
                    h_try[i + lag] -= delta
                    if 0 <= h_try[i] <= 1 and 0 <= h_try[i+lag] <= 1:
                        s = evaluate_score(h_try)
                        if s < best_score - 1e-12:
                            best = h_try
                            best_score = s
                            break
            elif lag < 0 and -lag < N:
                lk = -lag
                i = rng.randint(N - lk)
                for delta in [0.005, 0.01, 0.02, -0.005, -0.01, -0.02]:
                    h_try = best.copy()
                    h_try[i + lk] += delta
                    h_try[i] -= delta
                    if 0 <= h_try[i+lk] <= 1 and 0 <= h_try[i] <= 1:
                        s = evaluate_score(h_try)
                        if s < best_score - 1e-12:
                            best = h_try
                            best_score = s
                            break
    
    return best, best_score

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    best_h = project_feasible(init, n / 2.0)
    best_score = evaluate_score(best_h)
    
    # Phase 1: Alternating projected gradient (new method)
    # Use multiple alpha sweeps
    for alpha_end in [10, 50, 200, 500, 1000, 2000]:
        if time.time() - t0 > budget_s - 500:
            break
        n_iters = 50
        step_size = 0.05
        h_new, score_new = alternating_method(
            best_h, n_iters, alpha_start=5, alpha_end=alpha_end,
            step_size=step_size, mu=50000, n_pts=n, rng=rng
        )
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
    
    # Phase 2: Smooth-max L-BFGS-B with increasing alpha
    alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    for alpha in alphas:
        if time.time() - t0 > budget_s - 400:
            break
        maxiter = max(100, 80000 // n)
        x = ljbfgs_call(best_h, alpha, n, 50000.0, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
    
    # Phase 3: Targeted local search
    if time.time() - t0 < budget_s - 300:
        n_ls = min(3000, int((budget_s - 300 - (time.time() - t0)) * 100))
        best_h, best_score = targeted_local_search(best_h, n_ls, rng)
        best_h = project_feasible(best_h, n / 2.0)
        best_score = evaluate_score(best_h)
    
    # Phase 4: Resolution increase
    for n_try in [2 * n, 3 * n]:
        if n_try > 2500:
            continue
        if time.time() - t0 > budget_s - 200:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    grad += weights[N - 1] * (1.0 - 2.0 * h)
    for k in range(1, N):
        wp = weights[N - 1 + k]
        wn = weights[N - 1 - k]
        sw = wp + wn
        if sw > 1e-15:
            grad[k:] += wp - sw * h[:N - k]
            grad[:N - k] += wn - sw * h[k:]
    return grad


def compute_Ph(h, p):
    N = len(h)
    p_ext = np.zeros(3 * N - 2)
    p_ext[:2 * N - 1] = p
    result = np.convolve(p_ext, h, mode='full')
    return result[N - 1:2 * N - 1]


def alternating_method(h, n_iters, alpha_start, alpha_end, step_size, n_pts):
    best_h = h.copy()
    best_score = evaluate_score(h)
    N = len(h)
    for it in range(n_iters):
        frac = (it + 1) / float(n_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        corr = compute_correlations(best_h)
        dx = 2.0 / N
        overlaps = corr * dx
        max_ov = np.max(overlaps)
        exp_ov = np.exp(alpha * (overlaps - max_ov))
        p = exp_ov / np.sum(exp_ov)
        h_cur = best_h.copy()
        n_pg = 5 + int(15 * frac)
        for pg_i in range(n_pg):
            grad = 2.0 * compute_Ph(h_cur, p)
            h_cur = h_cur + step_size * grad
            h_cur = project_feasible(h_cur, N / 2.0)
        score = evaluate_score(h_cur)
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
    return best_h, best_score


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        sm = m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def targeted_local_search(h, n_local_iters, rng):
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(best)
    for _ in range(n_local_iters):
        corr = compute_correlations(best)
        top_idx = np.argsort(corr)[-8:]
        k = top_idx[rng.randint(len(top_idx))]
        lag = k - (N - 1)
        if abs(lag) == 0:
            i = rng.randint(N)
            h_try = best.copy()
            delta = 0.005 * (1.0 if best[i] > 0.5 else -1.0)
            j = rng.randint(N)
            h_try[i] += delta
            h_try[j] -= delta
            if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                s = evaluate_score(h_try)
                if s < best_score - 1e-13:
                    best = h_try
                    best_score = s
        elif 0 < abs(lag) < N:
            lk = abs(lag)
            valid_range = N - lk
            if valid_range <= 0:
                continue
            i = rng.randint(valid_range)
            if lag > 0:
                a, b = i, i + lag
            else:
                a, b = i + lk, i
            for delta in [0.003, 0.005, 0.01, 0.02, 0.03, -0.003, -0.005, -0.01, -0.02, -0.03]:
                h_try = best.copy()
                h_try[a] += delta
                h_try[b] -= delta
                if 0 <= h_try[a] <= 1 and 0 <= h_try[b] <= 1 and abs(np.sum(h_try) - N / 2.0) < 1e-6:
                    s = evaluate_score(h_try)
                    if s < best_score - 1e-13:
                        best = h_try
                        best_score = s
                        break
    return best, best_score


def pair_local_search(h, n_iters, rng):
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(best)
    for _ in range(n_iters):
        i = rng.randint(N)
        j = rng.randint(N)

            continue
