```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
from itertools import combinations

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(overlaps, alpha):
    m = np.max(overlaps)
    return float(m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad

def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad
```

Actually, I want to think about one more thing: the coordinate descent on max might be slow because minimize_scalar with 'bounded' method is not super fast for piecewise linear functions. A better approach for piecewise linear convex functions is to use the subgradient/bisection method directly.

For a convex piecewise linear function f(t) on [0,1], the minimum is at a point where the subgradient contains 0. We can find this by:
1. Evaluate f(0) and f(1)
2. Compute the subgradient at 0+ and 1-
3. If f is decreasing at 0 and increasing at 1, there's an interior minimum
4. Use bisection on the subgradient to find where it crosses 0

The subgradient of max of linear functions is the convex hull of the slopes of the active (maximizing) functions. At a generic point, only one function is active, so the subgradient is just the slope of that function.

For our problem, the slope of c[n-1+d] with respect to h[i] is:
- (1-h[i+d]) if 0 <= i+d < n (first factor)
- -h[i-d] if 0 <= i-d < n (second factor)
- Sum of both if both conditions hold

So the slope at position k=n-1+d is:
slope_d = (1-h[i+d]) * I(0<=i+d<n) - h[i-d] * I(0<=i-d<n)

And the function value at position k is base_k + (t-old)*slope_d.

The max function is: f(t) = max_d (base_d + (t-old)*slope_d)

This is a max of linear functions, hence convex and piecewise linear. The minimum occurs where the median of the active slopes is 0 (for a continuous function) or at a breakpoint.

For a practical approach, I'll just use minimize_scalar with a tight tolerance. The 'bounded' method should handle this well.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
from itertools import combinations


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(overlaps, alpha):
    m = np.max(overlaps)
    return float(m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def coord_max_obj_fast(h, i, t):
    old = h[i]
    delta = t - old
    if abs(delta) < 1e-14:
        return float(np.max(compute_overlaps(h)))
    n = len(h)
    base = compute_overlaps(h)
    new_base = base.copy()
    d_lo1 = max(-(n - 1), -i)
    d_hi1 = min(n - 1, n - 1 - i)
    if d_lo1 <= d_hi1:
        ds = np.arange(d_lo1, d_hi1 + 1)
        positions = n - 1 + ds
        new_base[positions] += delta * (1.0 - h[i + ds])
    d_lo2 = max(-(n - 1), i - n + 1)
    d_hi2 = min(n - 1, i)
    if d_lo2 <= d_hi2:
        ds = np.arange(d_lo2, d_hi2 + 1)
        positions = n - 1 + ds
        new_base[positions] -= delta * h[i - ds]
    return float(np.max(new_base))


def coordinate_descent_max(h, n_sweeps, rng=None, random_order=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    h = project_box_sum(h, target)
    best_raw = float(np.max(compute_overlaps(h)))
    best_h = h.copy()
    for sweep in range(n_sweeps):
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for i in order:
            res = minimize_scalar(
                coord_max_obj_fast, args=(h, i),
                bounds=(0.0, 1.0), method='bounded',
                options={'xatol': 1e-9}
            )
            h[i] = res.x
        h = project_box_sum(h, target)
        raw = float(np.max(compute_overlaps(h)))
        if raw < best_raw:
            best_raw = raw
            best_h = h.copy()
    return best_h, best_raw


def lbfgsb_optimize(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0


def sum_constraint_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def upscale(h_coarse, n_fine):
    x_c = np.linspace(0.0, 2.0, len(h_coarse), endpoint=False)
    x_f = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    return np.interp(x_f, x_c, h_coarse)

def subgrad_direction(h, top_k):
    n = len(h)
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(-overlaps)[:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
        else:
            M = -L
            grad[M:] += (1.0 - h[:n - M])
            grad[:n - M] -= h[M:]
    grad /= float(top_k)
    return grad


def subgrad_optimize(h0, n_iters, lr_init, target, top_k):
    h = h0.copy()
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    for it in range(n_iters):
        lr = lr_init / (1.0 + 0.03 * float(it))
        grad = subgrad_direction(h, top_k)
        h_new = project_box_sum(h - lr * grad, target)
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
    return best_h, best_raw
```
