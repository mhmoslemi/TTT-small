```python
                grad[l:] += w * (1.0 - h[:N - l])
                grad[:N - l] -= w * h[l:]
    return grad

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5

    for _ in range(80):

        if s > target_sum:

            lo = mid

        g += 2.0 * mu * s

def subgradient_optimize(h0, n, budget, mu=100.0):
    """Subgradient method with momentum for the min-max problem."""
    t0 = time.time()
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    # Initial step size
    alpha0 = 0.1
    decay = 0.999
    
    for t in range(200000):
        if time.time() - t0 > budget:
            break
        
        corr = compute_correlations(h)
        max_val = np.max(corr) * dx
        if max_val < best_score - 1e-10:
            best_score = max_val
            best_h = h.copy()
        
        # Active lags
        corr_scaled = corr * dx
        tol = 0.001
        active = np.where(corr_scaled >= max_val - tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            if lag == 0:
                g = 1.0 - 2.0 * h
            elif 0 < lag < n:
                g = np.zeros(n)
                g[:n - lag] = 1.0 - h[lag:]
                g[lag:] = -h[:n - lag]
            elif -n < lag < 0:
                l = -lag
                g = np.zeros(n)
                g[l:] = 1.0 - h[:n - l]
                g[:n - l] = -h[l:]
            else:
                continue
            grad += g
        grad /= len(active)
        
        # Constraint gradient
        s = np.sum(h) - n / 2.0
        grad += mu * s
        
        # Step size (decay)
        alpha_t = alpha0 * (decay ** t)
        
        # Momentum update
        v = beta * v - alpha_t * grad
        h_new = h + v
        h_new = project_feasible(h_new, n / 2.0)
        
        if not np.all(np.isfinite(h_new)):
            break
        h = h_new
    
    return best_h, best_score
```

Also, I realize that the subgradient method converges to a neighborhood of the optimum, not to the optimum itself. The size of the neighborhood depends on the step size. So for the final polish, I should use very small step sizes.

Let me reconsider. Maybe a better approach for the final polish is:

**Active set method**:
1. Identify active lags (those achieving the max)
2. Solve: minimize max_{k ∈ active} f_k(h) subject to 0 ≤ h ≤ 1, ∑h = n/2
3. This can be done by: minimize t subject to f_k(h) ≤ t for all k ∈ active
4. Use an SLP (successive linear programming) or similar approach

Actually, the simplest effective approach might be:

1. Identify active lags
2. Minimize the sum of active lags (which is a smooth function)
3. Re-identify active lags
4. Repeat

This is the "deterministic" version of the multiplicative weights approach.

Let me implement this:

```python
def active_set_cycle(h, n, max_cycles=100, tol=1e-6):
    dx = 2.0 / n
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for cycle in range(max_cycles):
        corr = compute_correlations(best_h) * dx
        max_val = np.max(corr)
        # Active lags: within tol of max
        active = np.where(corr >= max_val - tol)[0]
        
        # Minimize sum of active lags using L-BFGS-B
        def obj(h):
            corr_local = compute_correlations(h) * dx
            return np.sum(corr_local[active])
        
        def grad(h):
            g = np.zeros(n)
            for idx in active:
                lag = idx - (n - 1)
                if lag == 0:
                    g += 1.0 - 2.0 * h
                elif 0 < lag < n:
                    g[:n - lag] += 1.0 - h[lag:]
                    g[lag:] -= h[:n - lag]
                elif -n < lag < 0:
                    l = -lag
                    g[l:] += 1.0 - h[:n - l]
                    g[:n - l] -= h[l:]
            return g
        
        res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': 200, 'ftol': 1e-14})
        
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n / 2.0)
            s_new = evaluate_score(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new.copy()
    
    return best_h, best_score
```

This is clean and should work well. The key insight is that minimizing the sum of active lags is a smooth optimization problem that we can solve efficiently with L-BFGS-B. By cycling between identifying active lags and optimizing their sum, we gradually reduce the maximum.

```python
def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha

def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += w * (1.0 - h[:N - l])
            grad[:N - l] -= w * h[l:]
    return grad

        return g

def active_set_refine(h, n, max_cycles, tol=0.002, maxiter_per_cycle=300):
    """Iteratively identify active lags and minimize their sum."""
    dx = 2.0 / n
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for cycle in range(max_cycles):
        corr = compute_correlations(best_h) * dx
        max_val = np.max(corr)
        active = np.where(corr >= max_val - tol)[0]
        
        if len(active) == 0:
            break
        
        def obj_active(h_act):
            corr_local = compute_correlations(h_act) * dx
            return np.sum(corr_local[active])
        
        def grad_active(h_act):
            g = np.zeros(n)
            for idx in active:
                lag = idx - (n - 1)
                if lag == 0:
                    g += 1.0 - 2.0 * h_act
                elif 0 < lag < n:
                    g[:n - lag] += 1.0 - h_act[lag:]
                    g[lag:] -= h_act[:n - lag]
                elif -n < lag < 0:
                    l = -lag
                    g[l:] += 1.0 - h_act[:n - l]
                    g[:n - l] -= h_act[l:]
            s = np.sum(h_act) - n / 2.0
            g += 100.0 * s
            return g
        
        res = minimize(obj_active, best_h, jac=grad_active, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter_per_cycle, 'ftol': 1e-14})
        
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n / 2.0)
            s_new = evaluate_score(h_new)
            if s_new < best_score - 1e-12:
                best_score = s_new
                best_h = h_new.copy()
    
    return best_h, best_score

def local_search_pair(h, n_pts, n_iters, rng, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    for _ in range(n_iters):
        i = rng.randint(n_pts)

            if max_try < best_max - 1e-12:
                best = h_try
                best_corr = corr_try
                best_max = max_try
                break
    return best, best_max
```
