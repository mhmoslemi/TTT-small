def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)

    return np.sum(h) - len(h) / 2.0

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = h.copy()

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)

    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x if res.x is not None else None

                res_x = slsqp_optimize(h_cd, alpha, maxiter, bounds_try)
                if res_x is not None:
                    h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s_try = evaluate_score(h_new)
                        if s_try < s_cur:
                            s_cur = s_try
                            h_cd = h_new

        if time_left() > 15:
            h_cd = coordinate_descent(h_cd, n_sweeps=200, time_budget=min(12, time_left() - 8))
            s_cur = evaluate_score(h_cd)

        if s_cur < best_score:
            best_score = s_cur
            best_h = h_cd

    # Phase 7: More perturbations from new best
    n_cur = len(best_h)
    for i in range(30):
        if time_left() < 25:
            break
        ptype = i % 3
        if ptype == 0:
            scale = 0.01 + 0.08 * (i / 30.0)
            h_pert = perturbation_gaussian(best_h, rng, scale)
        elif ptype == 1:
            h_pert = perturbation_block(best_h, rng)
        else:
            h_pert = perturbation_local(best_h, rng)

        tb = min(3.5, (time_left() - 15) / max(30 - i, 1))
        h_cd = coordinate_descent(h_pert, n_sweeps=150, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 8: Final SLSQP with very high alpha
    if time_left() > 20:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [100000, 300000, 800000]:
            if time_left() < 8:
                break
            res_x = slsqp_optimize(best_h, alpha, 1000, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 9: Final intensive CD
    if time_left() > 15:
        h_cd = coordinate_descent(best_h, n_sweeps=1000, time_budget=min(40, time_left() - 5))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

        h_cd = coordinate_descent_reversed(best_h, n_sweeps=500, time_budget=min(20, time_left() - 3))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
