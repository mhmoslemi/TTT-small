```python
# Helper functions (top-level, no closures)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:
            continue
        offset = m - center
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    # Initial step size
    step = 1.0 / (alpha + 1.0) * 1.0 / (2.0 * n)
    step = max(step, 1e-6)
    step = min(step, 0.1)
    prev_val = None
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        ov = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(ov, alpha)
        
        if prev_val is not None:
            if sm < prev_val - 1e-12:
                stall = 0
            else:
                stall += 1
            if stall > 500:
                break
        prev_val = sm
        
        grad = compute_gradient(h, weights, dx)
        
        # Line search
        s = step
        improved = False
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm:
                h = h_trial
                improved = True
                break
            s *= 0.5
        if not improved:
            break
        
        # Increase step size
        step = min(s * 1.5, 0.5)
    
    return h

def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.1, 1.0 / (alpha * 2.0))
    prev_val = None
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        
        x_new = project_onto_constraint(y - step * grad_y, target)
        
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_and_weights(ov_x, alpha)
        
        if prev_val is not None and sm_x > prev_val + 1e-8:
            t_k = 1.0
        prev_val = sm_x
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    
    return x
```

```python
def compute_flip_delta(h, j_add, j_rem, n, dx):
    # Returns the change in correlation vector when flipping h[j_add] 0→1 and h[j_rem] 1→0
    offsets = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # For j_add (change = +1):
    idx_add_pos = j_add + offsets  # where h[j_add] contributes to (1-h[.]) term
    idx_add_neg = j_add - offsets  # where h[.] contributes to (1-h[j_add]) term
    
    # For j_rem (change = -1):
    idx_rem_pos = j_rem + offsets
    idx_rem_neg = j_rem - offsets
    
    delta = np.zeros(2*n-1)
    
    # Valid indices for j_add positive term: idx_add_pos in [0, n)
    valid = (idx_add_pos >= 0) & (idx_add_pos < n)
    delta[valid] += (1.0 - h[idx_add_pos[valid]])
    # j_add at offset: h[j_add] was 0, so (1-h[j_add]) = 1, after flip (1-h[j_add]) = 0
    # The negative term: h[j_add - offset] * (change in (1-h[j_add]))
    # = h[j_add - offset] * (-1)  [since (1-h[j_add]) goes from 1 to 0]
    valid2 = (idx_add_neg >= 0) & (idx_add_neg < n)
    delta[valid2] -= h[idx_add_neg[valid2]]
    
    # For j_rem (change = -1):
    valid3 = (idx_rem_pos >= 0) & (idx_rem_pos < n)
    delta[valid3] -= (1.0 - h[idx_rem_pos[valid3]])
    valid4 = (idx_rem_neg >= 0) & (idx_rem_neg < n)
    delta[valid4] += h[idx_rem_neg[valid4]]
    
    return delta * dx
```

```python
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha

    n = len(h)

    center = n - 1
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:

        i_hi = min(n, n - offset)
```

```python
        prev_val = sm_x
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x


def binary_local_search(h_init, n, deadline, rng, max_iters=200000):
    dx = 2.0 / n
    target = n // 2
    h = np.zeros(n)
    sorted_idx = np.argsort(-h_init)
    h[sorted_idx[:target]] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    current_score = float(np.max(c))
    best_score = current_score
    best_h = h.copy()
    offsets = np.arange(2 * n - 1) - (n - 1)
    delta = np.zeros(2 * n - 1)
    new_c = np.zeros(2 * n - 1)
    T0 = 0.004
    T_end = 1e-8
    accepted = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / float(max_iters))
        j_add = rng.randint(n)
        j_rem = rng.randint(n)
        if j_add == j_rem or h[j_add] != 0.0 or h[j_rem] != 1.0:
            continue
        idx_ap = j_add + offsets
        idx_an = j_add - offsets
        idx_rp = j_rem + offsets
        idx_rn = j_rem - offsets
        delta.fill(0.0)
        v = (idx_ap >= 0) & (idx_ap < n)
        delta[v] += 1.0 - h[idx_ap[v]]
        v = (idx_an >= 0) & (idx_an < n)
        delta[v] -= h[idx_an[v]]
        v = (idx_rp >= 0) & (idx_rp < n)
        delta[v] -= 1.0 - h[idx_rp[v]]
        v = (idx_rn >= 0) & (idx_rn < n)
        delta[v] += h[idx_rn[v]]
        delta *= dx
        np.add(c, delta, out=new_c)
        new_score = float(np.max(new_c))
        ds = new_score - current_score
        if ds < 0.0 or rng.random() < np.exp(-ds / max(T, 1e-12)):
            h[j_add] = 1.0
            h[j_rem] = 0.0
            c, new_c = new_c, c
            current_score = new_score
            accepted += 1
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    target0 = n0 / 2.0
    h = project_onto_constraint(np.array(initial_h_values, dtype=float), target0)
    best_h = h.copy()
    best_score = evaluate_score(best_h)

    # Phase 1: Refine at original resolution
    dl1 = t0 + min(180, budget_s * 0.25)
    alphas_p1 = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for ai, alpha in enumerate(alphas_p1):
        if time.time() > dl1 - 15:
            break
        remaining_alphas = len(alphas_p1) - ai
        time_per = max(10.0, (dl1 - 15 - time.time()) / remaining_alphas)
        h = pgd_optimize(h, alpha, target0, 15000, time.time() + time_per)
        h = project_onto_constraint(h, target0)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # FISTA at original resolution
    if time.time() < dl1:
        for alpha in [200, 1000, 5000]:
            if time.time() > dl1 - 5:
                break
            h_f = fista_optimize(best_h, alpha, target0, 8000, time.time() + 20)
            h_f = project_onto_constraint(h_f, target0)
            s = evaluate_score(h_f)
            if s < best_score:
                best_score = s
                best_h = h_f.copy()

    # Phase 2: Upscale to n=850 and binary local search
    dl2 = t0 + min(620, budget_s * 0.55)
    n_up = 850
    x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(x_new, x_old, best_h)
    target_up = n_up / 2.0
    h_up = project_onto_constraint(h_up, target_up)
    s_up = evaluate_score(h_up)
    if s_up < best_score:
        best_score = s_up
        best_h = h_up.copy()

    # Binary local search at n=850
    if time.time() < dl2 - 30:
        h_bin, score_bin = binary_local_search(h_up, n_up, time.time() + min(200, dl2 - 30 - time.time()), rng, max_iters=300000)
        s_bin = evaluate_score(h_bin)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()

    # Continuous refinement from binary solution
    if time.time() < dl2:
        h_cont = h_bin + rng.randn(n_up) * 0.05
        h_cont = project_onto_constraint(h_cont, target_up)
        for alpha in [50, 200, 800, 3000]:
            if time.time() > dl2 - 10:
                break
            h_cont = pgd_optimize(h_cont, alpha, target_up, 10000, time.time() + 25)
            h_cont = project_onto_constraint(h_cont, target_up)
            s = evaluate_score(h_cont)
            if s < best_score:
                best_score = s
                best_h = h_cont.copy()

    # Phase 3: More continuous refinement at n=850
    dl3 = t0 + min(900, budget_s * 0.75)
    for alpha in [500, 2000, 5000, 15000, 30000]:
        if time.time() > dl3 - 15:
            break
        h_r = pgd_optimize(best_h, alpha, target_up, 10000, time.time() + 30)
        h_r = project_onto_constraint(h_r, target_up)
        s = evaluate_score(h_r)
        if s < best_score:
            best_score = s
            best_h = h_r.copy()

    # FISTA refinement
    if time.time() < dl3:
        for alpha in [500, 2000, 10000]:
            if time.time() > dl3 - 10:
                break
            h_f = fista_optimize(best_h, alpha, target_up, 8000, time.time() + 25)
            h_f = project_onto_constraint(h_f, target_up)
            s = evaluate_score(h_f)
            if s < best_score:
                best_score = s
                best_h = h_f.copy()

    # Phase 4: Perturbation-based local search
    dl4 = t0 + budget_s - 200
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    for i in range(200):
        if time.time() > dl4:
            break
        if i % 40 == 0:
            sorted_idx = np.argsort(-best_h)
            binary = np.zeros(n_cur)
            binary[sorted_idx[:n_cur // 2]] = 1.0
            pert = binary + rng.randn(n_cur) * 0.15
        elif i % 40 == 20:
            ov = compute_overlaps(best_h)
            thr = ov.max() - 0.008
            crit = ov > thr
            weights = np.zeros(len(ov))
            weights[crit] = 1.0 / max(1.0, np.sum(crit))
            grad = compute_gradient(best_h, weights, 2.0 / n_cur)
            gmax = np.max(np.abs(grad))
            if gmax > 1e-12:
                pert = best_h - 0.03 * grad / gmax
            else:
                pert = best_h + rng.randn(n_cur) * 0.01
        else:
            scale = 0.002 + 0.02 * (i / 200.0)
            pert = best_h + rng.randn(n_cur) * scale
        pert = project_onto_constraint(pert, target_cur)
        alpha = 200 + 400 * rng.randint(6)
        h_new = pgd_optimize(pert, alpha, target_cur, 3000, time.time() + 12)
        h_new = project_onto_constraint(h_new, target_cur)
        s = evaluate_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()

    # Phase 5: Try higher resolution
    if time.time() < t0 + budget_s - 150:
        n_h = 1700
        x_old2 = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new2 = np.linspace(0.0, 2.0, n_h, endpoint=False)
        h_h = np.interp(x_new2, x_old2, best_h)
        target_h = n_h / 2.0
        h_h = project_onto_constraint(h_h, target_h)
        s_h = evaluate_score(h_h)
        if s_h < best_score:
            best_score = s_h
            best_h = h_h.copy()
        for alpha in [200, 1000, 5000, 20000]:
            if time.time() > t0 + budget_s - 60:
                break
            h_h = pgd_optimize(h_h, alpha, target_h, 8000, time.time() + 25)
            h_h = project_onto_constraint(h_h, target_h)
            s = evaluate_score(h_h)
            if s < best_score:
                best_score = s
                best_h = h_h.copy()

    # Phase 6: Final high-alpha refinement
    if time.time() < t0 + budget_s - 30:
        nf = len(best_h)
        tf = nf / 2.0
        for alpha in [20000, 50000, 100000, 200000]:
            if time.time() > t0 + budget_s - 15:
                break
            h_fin = pgd_optimize(best_h, alpha, tf, 5000, time.time() + 10)
            h_fin = project_onto_constraint(h_fin, tf)
            s = evaluate_score(h_fin)
            if s < best_score:
                best_score = s
                best_h = h_fin.copy()

    # Final projection
    nf = len(best_h)
    best_h = project_onto_constraint(best_h, nf / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, nf)
```
