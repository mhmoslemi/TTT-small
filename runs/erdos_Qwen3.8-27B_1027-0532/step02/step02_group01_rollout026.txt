import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def make_binary(h, rng):
    n = len(h)
    target = n // 2
    h_sorted = h.copy()
    idx = np.argsort(h_sorted)[::-1][:target]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def swap_perturb(h, rng, n_swaps=1):
    n = len(h)
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
    return h

def block_swap_perturb(h, rng, max_len=25):
    n = len(h)
    h = h.copy()
    blen = int(rng.integers(2, min(max_len, n // 6) + 1))
    a = int(rng.integers(0, n - blen + 1))
    b = int(rng.integers(0, n - blen + 1))
    if abs(a - b) < blen:
        return h
    i1, i2 = min(a, b), max(a, b)
    temp = h[i1:i1+blen].copy()
    h[i1:i1+blen] = h[i2:i2+blen]
    h[i2:i2+blen] = temp
    return h

def targeted_swap(h, rng, n_samples=100):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, c5
    best_c5 = c5
    best_h = h.copy()
    n_samp = min(n_samples, len(ones) * len(zeros))
    for _ in range(n_samp):
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        h[a] = 1.0
        h[b] = 0.0
    return best_h, best_c5

    n = len(h_init)
    pop = []
    scores = []
    h0_bin = make_binary(h_init, rng)
    pop.append(h0_bin)
    scores.append(compute_c5(h0_bin))
    for p in range(1, n_pop):
        h_rand = np.zeros(n)
        idx = rng.choice(n, size=n//2, replace=False)
        h_rand[idx] = 1.0
        pop.append(h_rand)
        scores.append(compute_c5(h_rand))
    best_h = pop[np.argmin(scores)].copy()
    best_c5 = min(scores)
    no_improve = 0
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        if no_improve > n_steps // 4:
            t = max(t, t_start * 0.3)
        improved_any = False
        for p in range(n_pop):
            h = pop[p]
            r = rng.random()

            elif r < 0.85:

            scores[worst_idx] = best_c5
    return best_h, best_c5

def smooth_c5_obj(h, eps):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_c5_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def continuous_optimize(h_init, eps_list, time_limit, start_time):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    h = np.clip(h, 0.0, 1.0)
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for eps in eps_list:
        if time.time() - start_time > time_limit:
            break
        def obj(x):
            base = smooth_c5_obj(x, eps)
            penalty = 1000.0 * (np.sum(x) - target) ** 2
            return base + penalty
        def gradient(x):
            base_g = smooth_c5_grad(x, eps)
            penalty_g = 2000.0 * (np.sum(x) - target) * np.ones(n)
            return base_g + penalty_g
        try:
            res = minimize(obj, h, jac=gradient, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = (h_new - h_new.mean() + target / n)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = h_new * (target / h_new.sum()) if h_new.sum() > 0 else h
            h_new = np.clip(h_new, 0.0, 1.0)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h = h_new
        except Exception:
            pass
    return h, best_c5

def upscale(h, new_n):
    n = len(h)
    out = np.zeros(new_n)
    for i in range(new_n):
        src = min(int(i * n / new_n), n - 1)
        out[i] = h[src]
    return out

def project_to_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    if abs(current_sum - target) < 1e-10:
        return h
    scale = target / current_sum if current_sum > 0 else 1.0
    h2 = h * scale
    if h2.max() <= 1.0 + 1e-6:
        h2 = np.clip(h2, 0.0, 1.0)
        return h2
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
