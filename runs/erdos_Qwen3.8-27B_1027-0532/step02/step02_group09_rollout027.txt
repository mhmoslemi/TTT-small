Let me think more carefully. np.correlate(h, 1-h, mode='full') gives:
c[m] = sum_i h[i+m] * (1-h[i]) for valid i

Actually, np.correlate(a, v) computes the cross-correlation:
c[k] = sum_n a[n+k] * conj(v[n])

For mode='full', the output has length 2n-1, and:
c[k] = sum_n a[n+k] * v[n] for k = 0 to n-1 (positive lags)
c[k] = sum_n a[n] * v[n-k] for k = -(n-1) to -1 (negative lags)

In numpy's convention for mode='full', the output is indexed 0 to 2n-2, where:
- Index 0 corresponds to lag -(n-1)
- Index n-1 corresponds to lag 0
- Index 2n-2 corresponds to lag n-1

So c[n-1 + lag] = sum_i a[i + lag] * v[i] (with appropriate bounds)

For our problem, a = h, v = 1-h.
C₅ = max_k sum_i h[i] * (1-h[i+k]) * dx

This is the correlation of h with (1-h) at various lags.

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def lse_wrapper(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def lse_grad_wrapper(h, alpha):
    return lse_grad(h, alpha)


def sum_con(h):
    return np.sum(h) - len(h) / 2.0


def sum_con_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            lse_wrapper, h0, args=(alpha,),
            method='SLSQP', jac=lse_grad_wrapper, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def lbfgsb_optimize(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None


def subgrad_direction(h, top_k):
    n = len(h)
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(-overlaps)[:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
        else:
            M = -L
            grad[M:] += (1.0 - h[:n - M])
            grad[:n - M] -= h[M:]
    grad /= float(top_k)
    return grad


def subgrad_optimize(h0, n_iters, lr_init, target, top_k):
    h = h0.copy()
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    momentum = np.zeros_like(h)
    beta = 0.9
    for it in range(n_iters):
        lr = lr_init / (1.0 + 0.01 * float(it))
        grad = subgrad_direction(h, top_k)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = project_box_sum(h - lr * momentum, target)
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
    return best_h, best_raw


def block_swap_local_search(h0, n_iters, rng, max_block_size=6):
    """Local search using block swaps for binary solutions."""
    n = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    
    if n_ones != n_target:
        if n_ones > n_target:
            idxs = np.where(h_bin == 1.0)[0]
            remove = rng.choice(idxs, size=n_ones - n_target, replace=False)
            h_bin[remove] = 0.0
        else:
            idxs = np.where(h_bin == 0.0)[0]
            add = rng.choice(idxs, size=n_target - n_ones, replace=False)
            h_bin[add] = 1.0
    
    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    temp = 0.003
    cooling = 0.995
    no_improve = 0
    for it in range(n_iters):
        if it % 50 == 49:
            temp *= cooling
        if temp < 1e-10:
            break
        
        # Choose move type: single swap (30%) or block swap (70%)
        if rng.random() < 0.3:
            # Single swap
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_bin.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        else:
            # Block swap: find consecutive blocks
            block_len = rng.randint(2, max_block_size + 1)
            # Find a block of 1s
            ones = np.where(h_bin == 1.0)[0]
            if len(ones) < block_len:
                continue
            # Find starts of consecutive blocks of 1s of length >= block_len
            one_starts = []
            for si in ones:
                end = min(si + block_len, n)
                if si + block_len <= n and np.all(h_bin[si:si+block_len] == 1.0):
                    one_starts.append(si)
            if len(one_starts) == 0:
                continue
            
            zeros = np.where(h_bin == 0.0)[0]
            if len(zeros) < block_len:
                continue
            zero_starts = []
            for si in zeros:
                end = min(si + block_len, n)
                if si + block_len <= n and np.all(h_bin[si:si+block_len] == 0.0):
                    zero_starts.append(si)
            if len(zero_starts) == 0:
                continue
            
            i = one_starts[rng.randint(len(one_starts))]
            j = zero_starts[rng.randint(len(zero_starts))]
            # Make sure blocks don't overlap
            if i + block_len > j and j + block_len > i:
                continue
            
            h_bin[i:i+block_len] = 0.0
            h_bin[j:j+block_len] = 1.0
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_bin.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h_bin[i:i+block_len] = 1.0
                h_bin[j:j+block_len] = 0.0
    
    return best_h, best_score


def fast_block_search(h0, n_iters, rng, max_block_size=8):
    """Faster block search that precomputes consecutive runs."""
    n = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    
    if n_ones != n_target:

            idxs = np.where(h_bin == 1.0)[0]

            h_bin[add] = 1.0
    
    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    temp = 0.004
    cooling = 0.997
    
    def get_runs(arr, val):
        """Get all (start, length) pairs of consecutive runs of val."""
        runs = []
        positions = np.where(arr == val)[0]
        if len(positions) == 0:
            return runs
        start = positions[0]
        prev = positions[0]
        for p in positions[1:]:
            if p == prev + 1:
                prev = p
            else:
                runs.append((int(start), int(prev - start + 1)))
                start = p
                prev = p
        runs.append((int(start), int(prev - start + 1)))
        return runs
    
    for it in range(n_iters):
        if it % 100 == 99:
            temp *= cooling
        if temp < 1e-10:
            break
        
        block_len = rng.randint(1, max_block_size + 1)
        
        if block_len == 1:
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score

                if s_new < best_score:
                    best_score = s_new
                    best_h = h_bin.copy()
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        else:
            one_runs = get_runs(h_bin, 1.0)
            zero_runs = get_runs(h_bin, 0.0)
            
            one_candidates = [(s, l) for s, l in one_runs if l >= block_len]
            zero_candidates = [(s, l) for s, l in zero_runs if l >= block_len]
            
            if len(one_candidates) == 0 or len(zero_candidates) == 0:
                continue
            
            si, li = one_candidates[rng.randint(len(one_candidates))]
            sj, lj = zero_candidates[rng.randint(len(zero_candidates))]
            
            # Pick a random offset within the run
            oi = si + rng.randint(0, li - block_len + 1)
            oj = sj + rng.randint(0, lj - block_len + 1)
            
            if oi + block_len > oj and oj + block_len > oi:
                continue
            
            h_bin[oi:oi+block_len] = 0.0
            h_bin[oj:oj+block_len] = 1.0
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                if s_new < best_score:

                    best_h = h_bin.copy()
            else:

def multi_start_binary(seed_base, n, n_starts, budget_per_start, rng):
    """Try multiple random binary initializations."""
    best_h = None
    best_score = float('inf')

        strategy = s % 4
        if strategy == 0:

            h_init = np.zeros(n)

def fourier_init(n, rng, n_harmonics=5):
    """Create initial solution using low-frequency Fourier modes."""
    h = np.ones(n) * 0.5
    for k in range(1, n_harmonics + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(2 * np.pi * k * np.arange(n) / n)
        h += b * np.sin(2 * np.pi * k * np.arange(n) / n)
    return np.clip(h, 0.0, 1.0)

def lse_val(x, alpha):
    m = np.max(x)

    m = np.max(overlaps)

def m_sequence_init(n, rng):

    while (period & (period + 1)) != 0:

    m = int(np.log2(period))
    if m < 3:
        h = (rng.randint(0, 2, size=n)).astype(float)
        return h
    state = rng.randint(1, 2 ** m)
    taps = [0] + [i for i in range(1, m) if ((period + 1) >> (m - i)) & 1 == 0]
    mseq = []
    for _ in range(n):

            st = state >> t & 1
            bit ^= st
        mseq.append(state & 1)

                        h_init = h_fine.copy()
            if rem() > 35 and len(best_h) == n_try:
                h_b, s_b = advanced_local_search(best_h, min(6000, int(rem() * 2)), rng, max_block=6, time_fn=rem)
                if s_b < best_score:
                    best_score = s_b
                    best_h = h_b.copy()

    # Phase 7: Multi-start with diverse inits
    if rem() > 100:
        n_cur = len(best_h)
        for trial in range(3):
            if rem() < 40:
                break
            if trial == 0:
                h_init = m_sequence_init(n_cur, rng)
            elif trial == 1:
                h_init = (rng.randint(0, 2, size=n_cur)).astype(float)
            else:
                h_init = 1.0 - best_h
            h_init = project_box_sum(h_init, n_cur / 2.0)
            per_iter = min(4000, int(rem() * 0.25))
            h_t, s_t = advanced_local_search(h_init, per_iter, rng, max_block=8, time_fn=rem)
            if s_t < best_score:
                best_score = s_t
                best_h = h_t.copy()

    # Phase 8: Final polish
    if rem() > 25:
        n_final = len(best_h)
        for alpha in [5000, 12000]:
            if rem() < 15:
                break
            penalty = alpha * 80.0
            bounds = [(0.0, 1.0)] * n_final
            res_x = lbfgsb_optimize(best_h, alpha, penalty, 400, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
        if rem() > 15:
            h_b, s_b = advanced_local_search(best_h, min(4000, int(rem() * 2)), rng, max_block=10, time_fn=rem)
            if s_b < best_score:
                best_score = s_b
                best_h = h_b.copy()

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
