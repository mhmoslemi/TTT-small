def compute_correlations(h):

def evaluate_score(h):

    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

        if abs(diff) < 1e-14:

def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    
    a = corr - h_j * slopes
    
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    
    for _ in range(40):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    
    V_star = (lo_V + hi_V) * 0.5
    
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    
    return float(np.clip(t_star, 0.0, 1.0)), slopes

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    h = project_sum(h, n / 2.0)
    return h

def subgradient_step(h, step_size, top_k_count=10, momentum=0.9):

def subgradient_minimax(h, n_steps=500, time_budget=100.0):

def time_remaining(t0, budget):

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):

def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0, t0=None):

    h = h.copy()
    corr = compute_correlations(h)

            break

                h[j] = t_opt
                corr = corr + change * slopes

        if sweep % 15 == 14:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def compute_corr_gradient(h, k_idx):
    n = len(h)
    shift = k_idx - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        mask1 = j >= shift
        idx1 = j[mask1] - shift
        grad[mask1] += (1.0 - h[idx1])
        mask2 = (j + shift) < n
        idx2 = j[mask2] + shift
        grad[mask2] -= h[idx2]
    else:
        s = -shift
        mask1 = j >= s
        idx1 = j[mask1] - s
        grad[mask1] += (1.0 - h[idx1])
        mask2 = (j + s) < n
        idx2 = j[mask2] + s
        grad[mask2] -= h[idx2]
    return grad
