def grad_for_shift(h, k, dx):
    n = len(h)
    if abs(k) >= n:
        return np.zeros(n)
    grad = np.zeros(n)

        i_s = np.arange(k, n)
        j_s = np.arange(0, n - k)

        j_s = np.arange(-k, n)
    grad[i_s] += dx * (1.0 - h[j_s])
    grad[j_s] -= dx * h[i_s]
    return grad
