import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_smooth_grad(h, corr, alpha):
    n = len(h)
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    p = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag >= n:
                continue
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            if la >= n:
                continue
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]
    return grad


def smooth_objective(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m))))) / alpha


def subgradient_optimize(h0, alpha_start, alpha_end, n_iters, step_init, time_limit, seed, t0):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)

    best_h = h.copy()
    best_score = evaluate_score(h)
    step = step_init
    v = np.zeros(n)
    momentum = 0.85
    stall_count = 0
    no_improve = 0

    for it in range(n_iters):
        if time.time() - t0 > t0 + time_limit:
            break

        frac = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        corr = compute_correlations(h)
        grad = compute_smooth_grad(h, corr, alpha)
        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-15:
            break

        v = momentum * v - step * grad
        h_new = project_to_feasible(h + v, target_sum)

        score = evaluate_score(h_new)
        if score < best_score:
            if score < best_score - 1e-10:
                no_improve = 0
            else:
                no_improve += 1
            best_score = score
            best_h = h_new.copy()
            h = h_new
            stall_count = 0
        else:
            no_improve += 1
            stall_count += 1
            step *= 0.9

        if no_improve > 200:
            pert_scale = 0.02 + 0.08 * min(no_improve / 500.0, 1.0)
            h = best_h.copy()
            for trial in range(10):
                if time.time() - t0 > t0 + time_limit:
                    break
                noise = rng.randn(n) * pert_scale / np.sqrt(n)
                h_try = project_to_feasible(h + noise, target_sum)
                s_try = evaluate_score(h_try)
                if s_try < best_score:
                    h = h_try
                    best_h = h_try.copy()
                    best_score = s_try
                    no_improve = 0
                    break
                # Try smooth structured perturbation
                freq = rng.randint(1, n // 10 + 1)
                phase = rng.uniform(0, 2 * np.pi)
                smooth_noise = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                smooth_noise -= np.mean(smooth_noise)
                h_try = project_to_feasible(h + smooth_noise * pert_scale / np.sqrt(n), target_sum)
                s_try = evaluate_score(h_try)

                    best_h = h_try.copy()
                    best_score = s_try
                    no_improve = 0
                    break
            v = np.zeros(n)
            step = step_init * (0.5 + 0.5 * rng.rand())
            momentum = 0.85

        if it > 0 and it % 2000 == 0:
            step = max(step_init * 0.92 ** (it // 2000), step_init * 0.05)

def slsqp_polish(h, alpha, maxiter):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def con_fun(hv):
        return np.sum(hv) - target_sum

    def con_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})

    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def binary_rounding(h):
    n = len(h)
    target_sum = n / 2.0
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    diff = target_sum - np.sum(h_bin)
    if diff > 0:
        order = np.argsort(-h)
        for i in range(int(abs(diff))):
            if i < len(order):
                h_bin[order[i]] = 1.0
    elif diff < 0:
        order = np.argsort(h)
        for i in range(int(abs(diff))):
            if i < len(order):
                h_bin[order[i]] = 0.0
    return h_bin


def coordinate_descent_polish(h, max_passes=5, t0=None, time_limit=None):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_score = evaluate_score(h)
    dx = 2.0 / n

    for p in range(max_passes):
        if t0 is not None and time.time() - t0 > t0 + time_limit:
            break
        improved = False
        idx_order = np.random.permutation(n)
        for idx in idx_order:
            if t0 is not None and (p * n + idx) % 1000 == 0 and time.time() - t0 > t0 + time_limit:
                break
            old_val = h[idx]
            # Try a few candidate values
            candidates = [0.0, 1.0, 0.1, 0.9, old_val * 0.5, old_val + (1.0 - old_val) * 0.5]
            best_candidate = old_val
            best_candidate_score = best_score + 1.0
            for cv in candidates:
                cv = np.clip(cv, 0.0, 1.0)
                adjustment = target_sum - (np.sum(h) - old_val + cv)
                if abs(adjustment) > 0.01:
                    continue
                h[idx] = cv
                h = project_to_feasible(h, target_sum)
                if np.abs(h[idx] - cv) > 0.3:
                    h[idx] = old_val
                    continue
                s = evaluate_score(h)
                if s < best_candidate_score:
                    best_candidate_score = s
                    best_candidate = h.copy()
            if best_candidate_score < best_score - 1e-10:
                h = best_candidate
                best_score = best_candidate_score
                improved = True
            else:
                h[idx] = old_val
                h = project_to_feasible(h, target_sum)
        if not improved:
            break
    return h, best_score


def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_feasible(h_new, n_new / 2.0)
