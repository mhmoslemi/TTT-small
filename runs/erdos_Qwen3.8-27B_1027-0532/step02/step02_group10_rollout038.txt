    return corr * dx

def evaluate_score(h):
    ov = compute_overlaps(h)
    return np.max(ov)

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    # Bisection to find the right shift
    lo = h.max() - target_sum / n  # ensure at least some room
    hi = -h.min() + (n - target_sum) / n  # ensure feasibility
    # Actually, simpler: binary search on the threshold
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        # Distribute equally among free variables
        free = (h > 1e-14) & (h < 1 - 1e-14)
        if free.sum() == 0:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        # h[i] appears in corr[m] as h[i+offset]*(1-h[i]) for valid i
        # So d/dh[i] of corr[m] = (1-h[i]) if i+offset valid (from h[i+offset]*(1-h[i]))
        #                              - h[i+offset] if i+offset valid (from h[i+offset]*(1-h[i])... wait)
        # 
        # corr[m] = sum_i h[i] * (1-h[i+offset]) for valid i, offset = m-(n-1)
        # d/dh[j] = (1-h[j+offset]) if j+offset valid
        #          - h[j-offset] if j-offset valid (from the term h[j-offset+offset]*(1-h[j]) = h[j]*(1-h[j])... no)
        #
        # Wait, let me be more careful.
        # corr[m] = sum_{i: valid} h[i] * (1 - h[i + offset])
        # where valid means 0 <= i < n and 0 <= i+offset < n
        #
        # d/dh[j] corr[m]:
        # - If j is in the sum as the "first" factor: term is h[j]*(1-h[j+offset]), derivative is (1-h[j+offset])
        #   Valid if 0 <= j < n and 0 <= j+offset < n
        # - If j is in the sum as the "second" factor: there exists i such that i+offset = j, i.e., i = j-offset
        #   Term is h[j-offset]*(1-h[j]), derivative is -h[j-offset]
        #   Valid if 0 <= j-offset < n and 0 <= j < n
        #
        # So:
        # d/dh[j] = (1-h[j+offset]) [if 0<=j+offset<n] - h[j-offset] [if 0<=j-offset<n]
        #
        # For all j at once:
        # First term: for j where 0<=j+offset<n: (1-h[j+offset])
        #   j ranges from max(0,-offset) to min(n-1, n-1-offset)
        #   The values are (1-h[max(-offset,n-1):min(n, n-offset)])... 
        #   Actually: j+offset ranges from max(offset, 0) to min(n-1, n-1+offset)
        #   So the first term vector is: (1-h[max(0,offset):min(n, n+offset)]) placed at positions j = max(0,-offset) to min(n-1, n-1-offset)
        #
        # Second term: for j where 0<=j-offset<n: -h[j-offset]
        #   j-offset ranges from max(-offset,0) to min(n-1-offset, n-1)
        #   j ranges from max(offset, 0) to min(n-1, n-1+offset)
        #   The values are -h[max(0,-offset):min(n, n-offset)]... 
        #   Wait: j-offset, with j from max(offset,0) to min(n-1+offset, n-1)
        #   j-offset from max(0, -offset) to min(n-1-offset, n-1)
        #   = max(0,-offset) to min(n-1, n-1-offset)
        #   So the second term is -h[max(0,-offset):min(n, n-offset)] placed at positions max(offset,0) to min(n-1, n-1+offset)
    
    # Let me just implement it directly
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        # First term: (1 - h[j+offset]) for j where 0 <= j < n and 0 <= j+offset < n
        if offset >= 0:
            j_lo, j_hi = 0, n - offset
        else:
            j_lo, j_hi = -offset, n
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo + offset:j_hi + offset])
        # Second term: -h[j-offset] for j where 0 <= j < n and 0 <= j-offset < n
        if offset >= 0:
            j_lo2, j_hi2 = offset, n
        else:
            j_lo2, j_hi2 = 0, n + offset
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 - offset:j_hi2 - offset]
    grad *= dx
    return grad

def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    # Lipschitz bound: ||grad||_Linf <= 2 (each h appears in at most 2 terms per overlap, each bounded by 1)
    # Actually, for the smooth max, the Lipschitz constant is at most 2*alpha (from the exp)
    # But in practice, 1/(alpha*dx*something) works
    step = 1.0 / (2.0 * n * dx)  # Conservative
    # Actually the gradient of each overlap w.r.t. h is bounded by 2 in L2 norm (each h[j] appears in at most 2 correlations, each with coefficient at most 1)
    # So L <= 2*dx*2 = 4*dx for a single overlap
    # For smooth max with alpha, L <= alpha * 4 * dx... no that's not right either.
    # Let me just use a safe step
    step = 0.8 / (2.0 * np.sqrt(n) * dx)
    prev_sm = None
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-13:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 300:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        # Backtracking line search
        improved = False
        s = step
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = s
            if step < 1e-14:
                break
            continue
        step = min(s * 1.3, 1.0)
    return h

def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = 0.5 / (2.0 * np.sqrt(n) * dx)
    prev_sm = None
    prev_x = x.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_and_weights(ov_x, alpha)
        
        if prev_sm is not None and sm_x > prev_sm + 1e-8:
            # Restart
            t_k = 1.0
            y = x_new.copy()
        prev_sm = sm_x
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x

def coordinate_descent_step(h, target, deadline, max_sweeps=1):
    """One or more sweeps of exact coordinate descent."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            # Compute D[m] and r0[m] for coordinate j
            # r[m] = sum_i h[i]*(1-h[i+offset]) for valid i, offset = m-(n-1)
            # 
            # D[m] = coefficient of h[j] in r[m]
            # Case 1: j is the "first" factor: h[j]*(1-h[j+offset])
            #   offset = m-(n-1), need 0 <= j-offset < n (i.e., j-offset valid as i)
            #   wait no: need 0 <= j < n (always true) and 0 <= j+offset < n
            #   D contribution: (1 - h[j+offset]) if 0 <= j+offset < n
            # Case 2: j is the "second" factor: h[i]*(1-h[j]) where i+offset = j, i = j-offset
            #   need 0 <= j-offset < n
            #   D contribution: -h[j-offset] if 0 <= j-offset < n
            #
            # r0[m] = r[m] - D[m]*h[j]
            
            # Instead of computing for all m, let's compute the effect directly
            # r[m] with h[j] set to x:
            # For m where the first case applies (j+offset valid):
            #   term becomes x*(1-h[j+offset]) instead of h[j]*(1-h[j+offset])
            #   change: (x - h[j])*(1-h[j+offset])
            # For m where the second case applies (j-offset valid):
            #   term becomes h[j-offset]*(1-x) instead of h[j-offset]*(1-h[j])
            #   change: -h[j-offset]*(x - h[j])
            
            # So r[m] = r0[m] + (x - h[j]) * D[m] where:
            # D[m] = (1-h[j+offset]) [if valid] - h[j-offset] [if valid]
            # r0[m] = r[m] - (h[j]-h[j])*D[m] = r[m] (when x = h[j] currently)
            # Wait, that's circular. Let me just compute r[m] at x=0 and the slope.
            
            # r[m] at x = h[j] is what we currently have. Let's call it r_cur[m].
            # r[m] at x = 0: r0[m] = r_cur[m] - h[j]*D[m]
            # r[m] at x: r0[m] + x*D[m]
            
            # Current overlaps
            # Actually, let me just compute r_cur once outside the loop and update incrementally.
            # For now, let's compute it each time (slower but correct).
            
            r_cur = np.correlate(h, 1.0 - h, mode='full') * dx
            
            # Compute D for all m
            # For m = 0 to 2n-2, offset = m - (n-1)
            D = np.zeros(2 * n - 1)
            # Case 1: j+offset in [0, n-1], i.e., offset in [-j, n-1-j]
            #   m in [n-1-j, n-1+n-1-j] = [n-1-j, 2n-2-j]
            #   D[m] += (1 - h[j+offset]) = (1 - h[j + m - n + 1])
            idx_1 = np.arange(max(0, n - 1 - j), min(2 * n - 1, 2 * n - 1 - j + 1))
            if len(idx_1) > 0:
                h_idx = j + idx_1 - (n - 1)  # = j + offset
                D[idx_1] += (1.0 - h[h_idx])
            
            # Case 2: j-offset in [0, n-1], i.e., offset in [j-(n-1), j]
            #   m in [n-1+j-(n-1), n-1+j] = [j, n-1+j]
            #   D[m] += -h[j-offset] = -h[j - m + n - 1]
            idx_2 = np.arange(max(0, j), min(2 * n - 1, n - 1 + j + 1))
            if len(idx_2) > 0:
                h_idx2 = j - (idx_2 - (n - 1))  # = j - offset
                D[idx_2] -= h[h_idx2]
            
            D *= dx
            r0 = r_cur - h[j] * D
            
            # Now minimize max(r0 + x*D) over x in [0,1]
            # This is a 1D convex piecewise linear minimization
            # Upper bound: max(r0 + 0*D) = max(r0), Lower: max(r0 + 1*D) = max(r0+D)
            
            # Find active line at x=0: m0 = argmax r0
            # Find active line at x=1: m1 = argmax (r0 + D)
            m0 = np.argmax(r0)
            m1 = np.argmax(r0 + D)
            
            if D[m0] >= 0:
                x_opt = 0.0
            elif D[m1] <= 0:
                x_opt = 1.0
            else:
                # Bisection: find x where 0 is in the subdifferential
                lo, hi = 0.0, 1.0
                for _ in range(40):
                    mid = (lo + hi) / 2.0
                    vals = r0 + mid * D
                    active = np.where(vals == vals.max())[0]
                    min_slope = D[active].min()
                    max_slope = D[active].max()
                    if min_slope > 0:
                        hi = mid
                    elif max_slope < 0:
                        lo = mid
                    else:
                        break
                x_opt = (lo + hi) / 2.0
            
            # Update h[j]
            h[j] = x_opt
    
    # Project onto constraint
    h = project_onto_constraint(h, target)
    return h

def run(seed=42, budget_s=1300, **kwargs):

Wait, I have a problem with the coordinate_descent_step function. Computing r_cur = np.correlate(h, 1.0-h, mode='full') for each coordinate j is O(n²) per coordinate, making the total O(n³) per sweep. For n=425, that's about 77M operations per coordinate and 32B per sweep. That's way too slow.

I need to make this more efficient. Let me precompute r_cur once and update D efficiently.

Actually, let me reconsider. For each j:
- D has non-zero entries only for m in a range of length at most 2*n (actually at most 2*min(j+1, n-j) for the first case and similar for the second)
- r0 = r_cur - h[j]*D

The key insight is that D is sparse! For coordinate j:
- Case 1 affects m from max(0, n-1-j) to min(2n-2, 2n-2-j), which is at most n entries
- Case 2 affects m from max(0, j) to min(2n-2, n-1+j), which is at most n entries

So D has at most 2n non-zero entries (and in practice about n on average).

But the issue is that computing r_cur = correlate(h, 1-h) is O(n²) (actually O(n * 2n) = O(n²) since the full correlation has 2n-1 entries and each requires O(n) work... actually numpy.correlate is O(n²) for full mode).

Wait, actually np.correlate in full mode is O(n * m) where n and m are the lengths. For same-length arrays, it's O(n²). But for n=425, that's about 180K multiplications, which is fast.

If I compute r_cur once per sweep (not per coordinate), that's O(n²) per sweep. Then for each coordinate, I just need to compute D (O(n)) and solve the 1D problem (O(n * log(1/eps)) with bisection, or O(n) with the argmax approach).

But this is still O(n³) per sweep because we compute r_cur for each j. For n=425, that's 425 * 425 * 850 ≈ 154M operations per sweep. Each "operation" is a FLOP, so at ~1 GFLOP/s, that's about 0.15 seconds per sweep. Actually that's fine!

Wait, np.correlate for two arrays of length 425 in full mode computes 849 values, each requiring up to 425 multiply-adds. So it's about 425 * 425 = 180K multiply-adds ≈ 360K FLOPs. For 425 iterations, that's about 153M FLOPs. At 1 GFLOP/s, that's 0.15 seconds. Very fast!

Actually, numpy.correlate is implemented in C and is quite fast. Let me just go with this approach.

But wait, I realize there might be an even better approach. Instead of recomputing r_cur from scratch each time, I can update it incrementally:

When I change h[j] from old_val to new_val (delta = new_val - old_val):
- For each m, r[m] changes by delta * D_j[m] where D_j[m] is the coefficient we computed above.

So I can maintain r_cur and update it in O(n) per coordinate change!

This makes the total O(n²) per sweep instead of O(n³).

def coordinate_descent_sweep_fast(h, target, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    # Precompute current overlaps
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Precompute D for all coordinates (this is the big win)
    # D_j[m] = coefficient of h[j] in r[m]
    # Case 1: (1-h[j+offset]) where offset = m-(n-1), valid if 0<=j+offset<n
    # Case 2: -h[j-offset] where offset = m-(n-1), valid if 0<=j-offset<n
    
    for j in range(n):
        if time.time() > deadline:
            break
        
        # Compute D_j
        D = np.zeros(2 * n - 1)
        # Case 1: j + (m - n + 1) in [0, n-1], m in [n-1-j, 2n-2-j]
        m_lo1 = max(0, n - 1 - j)
        m_hi1 = min(2 * n - 1, 2 * n - 2 - j + 1)
        if m_lo1 < m_hi1:
            h_idx = j + np.arange(m_lo1, m_hi1) - (n - 1)
            D[m_lo1:m_hi1] += (1.0 - h[h_idx])
        
        # Case 2: j - (m - n + 1) in [0, n-1], m in [j, n-1+j]
        m_lo2 = max(0, j)
        m_hi2 = min(2 * n - 1, n + j)
        if m_lo2 < m_hi2:
            h_idx2 = j - (np.arange(m_lo2, m_hi2) - (n - 1))
            D[m_lo2:m_hi2] -= h[h_idx2]
        
        D *= dx
        r0 = r - h[j] * D
        
        # Solve 1D minimax: min_x max_m (r0[m] + x*D[m])
        m0 = np.argmax(r0)
        m1 = np.argmax(r0 + D)
        
        if D[m0] >= 0:
            x_opt = 0.0
        elif D[m1] <= 0:
            x_opt = 1.0
        else:
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = r0 + mid * D
                max_val = vals.max()
                active = np.where(vals >= max_val - 1e-15)[0]
                min_slope = D[active].min()
                max_slope = D[active].max()
                if min_slope > 1e-15:
                    hi = mid
                elif max_slope < -1e-15:
                    lo = mid
                else:

        delta = x_opt - h[j]
        if abs(delta) > 1e-15:
            r += delta * D
            h[j] = x_opt
    
    h = project_onto_constraint(h, target)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    ov = compute_overlaps(h)
    return np.max(ov)


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = free.sum()
        if nf == 0:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[free] += diff / nf
            np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = np.clip(alpha * (x - m), -50, 50)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            j_lo1, j_hi1 = 0, n - offset
        else:
            j_lo1, j_hi1 = -offset, n
        if j_lo1 < j_hi1:
            grad[j_lo1:j_hi1] += w * (1.0 - h[j_lo1 + offset:j_hi1 + offset])
        if offset >= 0:
            j_lo2, j_hi2 = offset, n
        else:
            j_lo2, j_hi2 = 0, n + offset
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 - offset:j_hi2 - offset]
    grad *= dx
    return grad


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = 0.5 / (2.0 * np.sqrt(n) * dx)
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 400:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        improved = False
        s = step
        for _bt in range(25):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-13:
                break
        if not improved:
            step = s
            if step < 1e-15:
                break
            continue
        step = min(s * 1.35, 1.0)
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = 0.4 / (2.0 * np.sqrt(n) * dx)
    prev_sm = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_and_weights(ov_x, alpha)
        if prev_sm is not None and sm_x > prev_sm + 1e-9:
            t_k = 1.0
            y = x_new.copy()
        prev_sm = sm_x
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x


def coordinate_descent(h, target, deadline, n_sweeps=1):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    for _sweep in range(n_sweeps):
        for j in range(n):
            if time.time() > deadline:
                return h
            D = np.zeros(2 * n - 1)
            m_lo1 = max(0, n - 1 - j)
            m_hi1 = min(2 * n - 1, 2 * n - 1 - j)
            if m_lo1 <= m_hi1:
                h_idx = j + np.arange(m_lo1, m_hi1 + 1) - (n - 1)
                D[m_lo1:m_hi1 + 1] += (1.0 - h[h_idx])
            m_lo2 = max(0, j)
            m_hi2 = min(2 * n - 1, n - 1 + j)
            if m_lo2 <= m_hi2:
                h_idx2 = j - (np.arange(m_lo2, m_hi2 + 1) - (n - 1))
                D[m_lo2:m_hi2 + 1] -= h[h_idx2]
            D *= dx
            r0 = r - h[j] * D
            m0 = np.argmax(r0)
            m1 = np.argmax(r0 + D)
            if D[m0] >= 0.0:
                x_opt = 0.0
            elif D[m1] <= 0.0:
                x_opt = 1.0
            else:
                lo = 0.0
                hi = 1.0

    h = project_onto_constraint(h, target)
