        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Nesterov subgradient with decreasing top_k
    for top_k in [1, 2, 3, 5]:
        if elapsed() > phase_end_2:
            break
        def mu_sched(t):
            return 2.0 / (t + 10.0)
        h_new, c5_new = nesterov_subgrad(best_h, n, 3000, top_k, mu_sched)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new = slsqp_smooth(best_h, n, 100.0, 300)
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 3: Midpoint coordinate descent (pair moves on worst shift)
    if elapsed() < phase_end_3:
        alloc = time_remaining(phase_end_3 * 0.7)
        h_new, c5_new = midpoint_coord_descent(best_h, n, alloc, 5000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new = slsqp_smooth(best_h, n, 50.0, 400)
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 4: Fine coordinate descent (single coordinate)
    if elapsed() < phase_end_4:
        alloc = time_remaining(phase_end_4 * 0.7)
        h_new, c5_new = fine_coord_descent(best_h, n, alloc, 30000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 5: Additional SLSQP refinement + final coord descent
    if elapsed() < phase_end_5:
        for eps_val in [30.0, 20.0, 15.0, 10.0]:
            if elapsed() > phase_end_5 * 0.8:
                break
            h_new = slsqp_smooth(best_h, n, eps_val, 500)
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        
        if elapsed() < budget_s * 0.95:
            alloc = time_remaining(budget_s * 0.95)
            h_new, c5_new = fine_coord_descent(best_h, n, alloc, 50000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 6: Try n=200
    if elapsed() < budget_s * 0.97:
        n2 = 200
        h_up = project_feasible(upsample_h(best_h, n, n2), n2)
        c5_up = evaluate_c5(h_up)
        
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up
            best_n = n2
        else:
            best_n = n
        
        if best_n == n2:
            if elapsed() < budget_s * 0.99:
                h_new = slsqp_smooth(best_h, n2, 50.0, 300)
                h_new = project_feasible(h_new, n2)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                h_new, c5_new = fine_coord_descent(best_h, n2, 10.0, 10000)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
