            best_max = mv
    
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        h, mv = sa_run(best_h, 100000, 2.0, 0.001, rng, n, directed_prob=0.8, top_k=6)
        c5v = float((2.0 / n) * mv)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
            best_max = mv
    
    final_h = binary_from_scores(best_h, n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    else:
        final_h = best_h
    
    return final_h, float(final_c5), n
