**Alternative: Projected Subgradient with Nesterov Acceleration**

The idea:
- C(h) = max_k c_k(h) * dx
- We want to minimize C(h) subject to sum(h) = n/2, 0 ≤ h ≤ 1
- The subgradient of max_k c_k is a convex combination of the gradients of the active c_k's
- At each step, compute the active set, compute a subgradient, take a step, project

The advantage of this over coordinate descent is that it moves all coordinates simultaneously, which can be faster in practice.

Let me implement this:

```python
def subgradient_step(h, step_size, top_k=10):
    corr = compute_correlations(h)
    n = len(h)
    max_val = np.max(corr)
    
    # Find top-k correlations
    top_indices = np.argsort(corr)[-top_k:]
    
    # Compute subgradient as average of gradients of top-k correlations
    grad = np.zeros(n)
    for k_idx in top_indices:
        # corr[k_idx] = sum_i h[i+k]*(1-h[i]) where k = k_idx - (n-1)
        k = k_idx - (n - 1)
        # d corr[k_idx] / d h[j]:
        # h[j] appears in corr[k_idx] when:
        #   j = i + k for some valid i (gives contribution (1-h[i]) = (1-h[j-k]))
        #   j = i for some valid i where i+k is valid (gives contribution -h[i+k] = -h[j+k])
        
        # For the first term: j = i+k means i = j-k, valid when 0 <= j-k < n
        # Contribution: 1 - h[j-k]
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo < hi:
            grad[lo:hi] += (1.0 - h[lo-k:hi-k])
        
        # For the second term: j = i means i = j, valid when 0 <= j+k < n
        # Contribution: -h[j+k]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+k:hi2+k]
    
    # Normalize subgradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    # Take step
    h_new = h - step_size * grad
    
    # Project onto feasible set
    h_new = project_sum(h_new, n / 2.0)
    return h_new, corr
```

```python
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    return smooth_max(compute_correlations(h), alpha)

    return np.sum(h) - len(h) / 2.0

        options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
    )
    return res.x if res.x is not None else None

                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
```
