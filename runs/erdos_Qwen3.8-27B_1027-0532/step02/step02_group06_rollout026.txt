def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_lag_gradient(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def subgradient_run(h0, alpha, n_top, max_iters, t0, time_limit, seed):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)

    best_score = evaluate_score(h)
    best_h = h.copy()
    step = 1.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break

        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]

        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * compute_lag_gradient(h, lag)

        momentum = min(0.95, 0.5 + 0.5 * it / 5000.0)
        v = momentum * v + (1.0 - momentum) * grad
        effective_step = step / (np.sqrt(np.sum(v * v)) + 1e-12)

        h_new = project_to_feasible(h - effective_step * v, target_sum)
        score = evaluate_score(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.7
                v *= 0.5
            if stall > 500:
                pert = 0.02 + 0.08 * min(stall, 2000) / 2000.0
                h = best_h + rng.randn(n) * pert / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                v = np.zeros(n)
                step = 1.0 / np.sqrt(n)
                stall = 0

        if it % 2000 == 0 and it > 0:
            step = max(1.0 / np.sqrt(n) * 0.5, step * 0.98)

    return best_h, best_score


def smooth_obj_and_grad(x, alpha, mu):
    n = len(x)
    target_sum = n / 2.0
    corr = compute_correlations(x)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = m + log_sum / alpha
    obj += mu * (np.sum(x) - target_sum) ** 2

    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-12:
            continue
        w = weights[idx]
        lag = int(idx - (n - 1))
        if lag > 0:
            grad[lag:] += w * (1.0 - x[:-lag])
            grad[:n - lag] -= w * x[lag:]
        elif lag < 0:
            ga = -lag
            grad[:n - ga] += w * (1.0 - x[ga:])
            grad[ga:] -= w * x[:n - ga]
        else:
            grad[:] += w * (1.0 - 2.0 * x)
    grad += 2.0 * mu * (np.sum(x) - target_sum)
    return obj, grad


def lbfgs_smooth(h0, alpha, mu, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj_and_grad(x, alpha, mu)[0]

    def grad(x):
        return smooth_obj_and_grad(x, alpha, mu)[1]

    res = minimize(obj, h0, jac=grad, bounds=bounds,
                   method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), n / 2.0)
    return None


def fourier_perturb(h, n_modes, amplitude, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    x = np.arange(n)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        a = rng.randn() * amplitude / np.sqrt(k)
        b = rng.randn() * amplitude / np.sqrt(k)
        pert += a * np.cos(2.0 * np.pi * k * x / n) + b * np.sin(2.0 * np.pi * k * x / n)
    h_p = h + pert
    return project_to_feasible(h_p, target_sum)


def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_to_feasible(init, n_init / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)
    best_n = n_init
    rng = np.random.RandomState(seed)

    alpha_list = [5, 10, 20, 50, 100, 200, 500]
    n_top_list = [5, 10, 20, 40]

    # Phase 1: Subgradient at initial resolution
    time_phase1 = budget_s * 0.25
    time_phase2 = budget_s * 0.35
    time_phase3 = budget_s * 0.30
    time_phase4 = budget_s * 0.10

    # Phase 1: Subgradient
    t1_start = time.time()
    configs_p1 = [(5, 5, 1), (10, 10, 1), (20, 10, 1), (20, 20, 1),
                  (50, 10, 1), (50, 40, 1), (100, 20, 1)]
    for i, (alpha, n_top, _) in enumerate(configs_p1):
        if time.time() - t1_start > time_phase1:
            break
        t_allow = min(time_phase1 - (time.time() - t1_start), 60)
        if t_allow < 5:
            break
        start_h = best_h
        if i > 0 and i % 3 == 0:
            start_h = fourier_perturb(best_h, 5, 0.05, seed + i * 100)
        h, score = subgradient_run(start_h, alpha, n_top, 30000, t0, t_allow, seed + i)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    # Phase 2: L-BFGS-B refinement with increasing alpha
    t2_start = time.time()
    alphas_p2 = [5, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    mus_p2 = [500, 2000, 5000, 20000, 50000]
    for i, alpha in enumerate(alphas_p2):
        if time.time() - t2_start > time_phase2:
            break
        mu = mus_p2[min(i, len(mus_p2) - 1)]
        maxiter = 500 if alpha < 100 else 300
        h_ref = lbfgs_smooth(best_h, alpha, mu, maxiter)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    # Multi-start with Fourier perturbations
    if time.time() - t2_start < time_phase2 * 0.7:
        for p in range(4):
            if time.time() - t2_start > time_phase2 * 0.95:
                break
            start_h = fourier_perturb(best_h, 10, 0.1, seed + 1000 + p)
            h, score = subgradient_run(start_h, 20, 10, 10000, t0, 30, seed + 5000 + p)
            if score < best_score:
                best_score = score
                best_h = h.copy()
            h_ref = lbfgs_smooth(best_h, 100, 10000, 300)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    # Phase 3: Upscale and refine
    t3_start = time.time()
    n_targets = [n_init * 2, n_init * 3, n_init * 4, n_init * 6]
    n_targets = [x for x in n_targets if x <= 5000]
    n_targets = sorted(set(n_targets))

    for n_try in n_targets:
        if time.time() - t3_start > time_phase3:
            break
        if n_try <= best_n:
            continue
        t_allow = min(remaining() - time_phase4 - 10, time_phase3 * 0.4)
        if t_allow < 20:
            break

        h_up = upscale(best_h, n_try)
        score_up = evaluate_score(h_up)
        if score_up < best_score:
            best_score = score_up
            best_h = h_up.copy()
            best_n = n_try
        else:
            best_h = h_up.copy()
            best_n = n_try

        # Subgradient at new resolution
        t_sub = min(t_allow * 0.4, 40)
        h, score = subgradient_run(best_h, 20, 10, 20000, t0, t_sub, seed + 20000 + n_try)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        # L-BFGS-B at new resolution
        if time.time() - t3_start < time_phase3 * 0.9:
            for alpha in [20, 100, 500]:
                if time.time() - t3_start > time_phase3 * 0.95:
                    break
                h_ref = lbfgs_smooth(best_h, alpha, 20000, 200)
                if h_ref is not None:
                    score = evaluate_score(h_ref)
                    if score < best_score:
                        best_score = score
                        best_h = h_ref.copy()

    # Phase 4: Final polish
    t4_start = time.time()
    alphas_final = [500, 2000, 10000, 50000]
    for alpha in alphas_final:
        if time.time() - t4_start > time_phase4:
            break
        t_allow = min(remaining() - 5, time_phase4 - (time.time() - t4_start))
        if t_allow < 5:
            break
        h_ref = lbfgs_smooth(best_h, alpha, 50000, 200)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    # Final subgradient squeeze
    if remaining() > 15:
        h, score = subgradient_run(best_h, 100, 20, 50000, t0, remaining() - 10, seed + 99999)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if remaining() > 15:
            h_ref = lbfgs_smooth(best_h, 10000, 100000, 100)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
