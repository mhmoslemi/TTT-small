import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, eps):
    corr = compute_corr_vector(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    # Gradient: for each position j, sum over shifts k of w[k] * dcorr[k]/dh[j]
    # dcorr[k]/dh[j] = dx * ((1-h[j+s]) - h[j-s]) where s = k-(n-1)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s

**Theoretical insight**: The problem is related to the "minimum overlap" of two sets. If A = {x : h(x) = 1} and B = {x : h(x) = 0}, then the overlap at shift k is the measure of A ∩ (B-k) = A \ (A+k). We want to minimize the maximum of this over all k.

For a "random" binary sequence, the overlap at any given shift is approximately Binomial(n-|k|, 1/4), so the mean is (n-|k|)/4 and the std is sqrt((n-|k|)/16). For n=180, at shift 0 the overlap is 0, at shift 1 it's about 44.75 ± 3.36. The max over 359 shifts would be around 44.75 + 3.36*3 ≈ 55 (rough). So random gives about C₅ ≈ 0.6, which is much worse than 0.38.

A good construction needs to be much more structured than random. The previous code's solution at 0.38 means max correlation ≈ 34.27, which is (34.27 - 44.75)/3.36 ≈ -3.1 standard deviations below the random mean. So it's a very non-trivial construction.

def compute_corr(h):

def compute_flip_effects(h):
    n = len(h)
    k = np.arange(2 * n - 1)
    j_idx = np.arange(n)
    r = (k[None, :] - (n - 1) + j_idx[:, None]).astype(int)

    p = (j_idx[:, None] + (n - 1) - k[None, :]).astype(int)
    B = np.zeros((n, 2 * n - 1))
    valid_b = (p >= 0) & (p < n)
    B[valid_b] = h[p[valid_b]]
    return A - B
