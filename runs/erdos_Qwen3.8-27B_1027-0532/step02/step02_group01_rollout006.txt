    """Returns (smooth_max_value, weights) where weights sum to 1 and peak at max."""
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / ex.sum()
    val = cmax + eps * np.log(ex.sum())
    return val, w

    n = len(h)
    dx = 2.0 / n
    h = np.round(h).astype(float)

    if h.sum() != n/2:
        # Adjust
        diffs = np.abs(h - 0.5)
        order = np.argsort(diffs)
        h = np.zeros(n)
        h[order[n//2:]] = 1.0

    I = np.arange(n)[:, None]

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def precompute_effects(h):
    n = len(h)
    s_vals = np.arange(-(n - 1), n)
    I = np.arange(n)[:, None]
    S = s_vals[None, :]
    idx_r = I - S
    valid_r = (idx_r >= 0) & (idx_r < n)
    R1 = np.where(valid_r, 1.0 - h[np.clip(idx_r, 0, n - 1)], 0.0)
    idx_f = I + S
    valid_f = (idx_f >= 0) & (idx_f < n)
    F1 = np.where(valid_f, h[np.clip(idx_f, 0, n - 1)], 0.0)
    return R1, F1

def objective_fun(h, eps):
    n = len(h)
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def gradient_fun(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_step(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(objective_fun, h0, args=(eps,),
                          jac=gradient_fun, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def binary_steepest(h_init, time_limit, start_time, rng, allow_anneal=True):
    n = len(h_init)
    h = np.round(h_init).astype(float)
    target = n // 2
    ones_count = int(np.sum(h == 1.0))
    if ones_count > target:
        ones = np.where(h == 1.0)[0]
        flip_idx = rng.choice(ones, size=ones_count - target, replace=False)
        h[flip_idx] = 0.0
    elif ones_count < target:
        zeros = np.where(h == 0.0)[0]
        flip_idx = rng.choice(zeros, size=target - ones_count, replace=False)
        h[flip_idx] = 1.0

    corr = compute_corr(h)
    current_c5 = np.max(corr)
    best_c5 = current_c5
    best_h = h.copy()
    R1, F1 = precompute_effects(h)
    step = 0
    stuck = 0
    temp = 0.01
    while step < 200000:
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        delta_a = -R1[ones, :] + F1[ones, :]
        delta_b = R1[zeros, :] - F1[zeros, :]
        if delta_a.shape[0] * delta_b.shape[0] * corr.shape[0] < 20000000:
            delta_all = delta_a[:, None, :] + delta_b[None, :, :]
            new_corr_mat = corr[None, None, :] + delta_all
            new_c5_mat = np.max(new_corr_mat, axis=2)
        else:
            n_o = delta_a.shape[0]
            n_z = delta_b.shape[0]
            new_c5_mat = np.full((n_o, n_z), -np.inf)
            for s in range(corr.shape[0]):
                vals = corr[s] + delta_a[:, s][:, None] + delta_b[:, s][None, :]
                np.maximum(new_c5_mat, vals, out=new_c5_mat)
        flat_idx = int(np.argmin(new_c5_mat))
        best_row, best_col = np.unravel_index(flat_idx, new_c5_mat.shape)
        best_val = new_c5_mat[best_row, best_col]
        if best_val < current_c5 - 1e-12:
            a = ones[best_row]
            b = zeros[best_col]
            h[a] = 0.0
            h[b] = 1.0
            corr = corr + delta_all[best_row, best_col, :] if delta_a.shape[0] * delta_b.shape[0] * corr.shape[0] < 20000000 else compute_corr(h)
            current_c5 = best_val
            R1, F1 = precompute_effects(h)
            stuck = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            if allow_anneal and stuck < 200 and rng.random() < 0.4:
                temp = max(0.001, 0.01 * (0.95 ** stuck))
                a = ones[rng.integers(len(ones))]
                b = zeros[rng.integers(len(zeros))]
                delta = -R1[a, :] + F1[a, :] + R1[b, :] - F1[b, :]
                new_c5_r = np.max(corr + delta)
                if new_c5_r < current_c5 or rng.random() < np.exp(-(new_c5_r - current_c5) / temp):
                    h[a] = 0.0
                    h[b] = 1.0
                    corr = corr + delta
                    current_c5 = new_c5_r
                    R1, F1 = precompute_effects(h)
                    stuck = 0
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                else:
                    stuck += 1
            else:
                stuck += 1
                if stuck > 150:
                    break
        step += 1
    return best_h, best_c5

def continuous_polyak(h_init, eps, n_steps, lr0, time_limit, start_time):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)
    beta = 0.85
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for i in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        lr = lr0 / (1.0 + 0.01 * i)
        grad = gradient_fun(h, eps)
        v = beta * v - lr * grad
        h_new = project_onto_constraint(h + v, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def sequential_slsqp(h_init, eps_list, maxiter_per_eps, time_limit, start_time):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    for eps in eps_list:

        h, c5 = slsqp_step(h, eps, maxiter_per_eps)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5
