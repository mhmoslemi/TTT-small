import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if m < -100:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if m < -100:
        return np.zeros(n)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag > 0:
            grad[:n - lag] += w[k] * (1.0 - x[lag:])
            grad[lag:] -= w[k] * x[:n - lag]
        elif lag < 0:
            L = -lag
            grad[L:] += w[k] * (1.0 - x[:n - L])
            grad[:n - L] -= w[k] * x[L:]
        else:
            grad[:] += w[k] * (1.0 - 2.0 * x)
    return grad * dx

def _slsqp_smooth(x0, alpha, n, maxiter, ftol):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smooth_obj(x, alpha)

    def grad(x):
        return _smooth_grad(x, alpha)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, x0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return x0.copy()

def _active_set_polish(h, n, n_steps, lr, deadline):
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    for step in range(n_steps):
        if time.time() > deadline - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for k_idx in top_k:
            lag = k_idx - (n - 1)
            if lag > 0:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n - lag]
            elif lag < 0:
                L = -lag
                grad[L:] += (1.0 - h[:n - L])
                grad[:n - L] -= h[L:]
            else:
                grad[:] += (1.0 - 2.0 * h)
        grad /= len(top_k)
        grad *= (2.0 / n)
        h = h - lr * grad
        h = _fix_sum(h, target_sum)
        c5 = _c5_of(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.95
    return best_h, best_c5

def _make_initial(n, seed, variant):
    rng = np.random.RandomState(seed + variant * 104729)
    x = np.linspace(0, 2, n, endpoint=False)
    target_sum = n / 2.0
    if variant == 0:
        h = np.full(n, 0.5)
    elif variant == 1:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    elif variant == 2:
        h = 0.5 + 0.4 * np.sin(4 * np.pi * x)
    elif variant == 3:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x)
    elif variant == 4:
        h = 0.5 + 0.35 * np.sin(3 * np.pi * x) + 0.1 * np.sin(6 * np.pi * x)
    elif variant == 5:
        h = rng.rand(n)
    elif variant == 6:
        h = 0.5 + 0.4 * np.cos(2 * np.pi * x * 1.5) + 0.1 * rng.randn(n)
    elif variant == 7:
        period = max(2, n // (4 + rng.randint(0, 3)))
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % (2 * period)) < period else 0.0
        h += 0.08 * rng.randn(n)
    else:
        h = 0.5 + 0.3 * rng.randn(n)
    return _fix_sum(h, target_sum)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return h_new

def _fourier_initial(n, seed, n_modes):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_modes + 1):
        amp = 0.3 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
        h += amp * 0.5 * np.sin(np.pi * k * x + phase)
    return _fix_sum(h, n / 2.0)
