def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(v, S):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:

def _grad_corr(h, lag, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:lag+m]) * dx
            g[lag:lag+m] += (-h[:m]) * dx
    else:
        j = -lag
        m = n - j
        if m > 0:
            g[j:] += (1.0 - h[:m]) * dx
            g[:m] += (-h[j:]) * dx
    return g

def _soft_max_grad(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    total = weights.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= total
    g = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k_idx in active:
        lag = k_idx - (n - 1)
        g += weights[k_idx] * _grad_corr(h, lag, n)
    return g

def _nesterov_adaptive(h0, n, target_sum, n_iters, alpha0, lse_start, lse_end):

    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        frac = (it + 1) / n_iters
        la = lse_start + (lse_end - lse_start) * frac
        g = _soft_max_grad(y, n, la)
        step = alpha0 / np.sqrt(float(it + 1))

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        y = y_new
        t = t_new

            best = x.copy()
    return best, best_val

            for alpha in [200, 800]:
                if time.time() > deadline - 8:
                    break
                h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=250)
                c5_sl = _c5_of(h_sl)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()
                    best_n = n
                h_curr = h_sl

    if best_h is not None:
        n_best = best_n
        target = n_best / 2.0
        h_final = best_h.copy()

        for alpha in [300, 1000, 3000]:
            if time.time() > deadline - 8:
                break
            h_final = _slsqp_opt(h_final, n_best, target, alpha, maxiter=400)
            c5_f = _c5_of(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

        if time.time() <= deadline - 8:
            h_cd = _coord_descent_analytic(best_h, n_best, target, max_rounds=10)
            c5_cd = _c5_of(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time.time() <= deadline - 8:
            h_cb = _coord_descent_binary(best_h, n_best, target, max_rounds=8)
            c5_cb = _c5_of(h_cb)
            if c5_cb < best_c5:
                best_c5 = c5_cb
                best_h = h_cb.copy()

        if time.time() <= deadline - 8:
            h_pol = _slsqp_opt(best_h, n_best, target, 5000, maxiter=500)
            c5_pol = _c5_of(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time.time() <= deadline - 5:
            h_cd2 = _coord_descent_analytic(best_h, n_best, target, max_rounds=15)
            c5_cd2 = _c5_of(h_cd2)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

        if time.time() <= deadline - 5:
            h_sl2 = _slsqp_opt(best_h, n_best, target, 8000, maxiter=500)
            c5_sl2 = _c5_of(h_sl2)
            if c5_sl2 < best_c5:
                best_c5 = c5_sl2
                best_h = h_sl2.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
