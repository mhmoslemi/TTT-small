            grad[:n - lag] -= wm * h[lag:]
        else:
            lag_abs = -lag
            grad[:n - lag_abs] += wm * (1.0 - h[lag_abs:])
            grad[lag_abs:] -= wm * h[:n - lag_abs]

import numpy as np
from scipy.optimize import differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(y, target_sum):
    y = np.asarray(y, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h_test = np.clip(y - mid, 0.0, 1.0)
        s_test = np.sum(h_test)
        if s_test > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(y - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h_proj < 1.0)[0]
        else:
            idx = np.where(h_proj > 0.0)[0]
        if len(idx) > 0:
            h_proj[idx] += diff / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m] * dx
        if wm < 1e-13:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += wm * (1.0 - h[:n - lag])
            grad[:n - lag] -= wm * h[lag:]
        else:
            la = -lag
            grad[:n - la] += wm * (1.0 - h[la:])
            grad[la:] -= wm * h[:n - la]
    return grad


def pgd_optimize(h0, alphas, iters_per_alpha, step_scale=0.5):
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    for alpha in alphas:
        step = step_scale / max(alpha, 1.0)
        for it in range(iters_per_alpha):
            grad = smooth_grad(h, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            f_curr = compute_c5(h)
            improved = False
            for ls in range(45):
                h_try = project_feasible(h - step * grad, target)
                f_try = compute_c5(h_try)
                if f_try < f_curr - 1e-16:
                    h = h_try
                    improved = True
                    break
                step *= 0.5
            if not improved:
                if it > 5:
                    break
                h_try = project_feasible(h - step * grad, target)
                f_try = compute_c5(h_try)
                if f_try < f_curr:
                    h = h_try
                else:
                    break
            f_new = compute_c5(h)
            if f_new < best_score:
                best_score = f_new
                best_h = h.copy()
            if improved:
                step = min(step * 1.15, step_scale / max(alpha, 1.0))
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def make_initial_structured(n, seed, mode):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    if mode == 0:

    elif mode == 1:

        phase = rng.uniform(0, 2.0 * np.pi)

    elif mode == 2:
        f = rng.randint(2, 20)
        h = (f * x / 2.0) % 1.0
    elif mode == 3:
        f = rng.randint(2, 15)
        h = 0.5 + 0.4 * np.cos(np.pi * f * np.sin(2.0 * np.pi * x / 2.0))
    else:
        h = 0.5 + 0.3 * rng.randn(n) * np.exp(-x ** 2)
    return project_feasible(h, n / 2.0)
