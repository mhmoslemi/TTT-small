    """Compute delta for all lags k from -(n-1) to n-1."""
    delta = np.zeros(2*n-1)
    # A(k): for k such that 0<=i+k<n, i.e., -i <= k <= n-1-i
    lo = max(-i, -(n-1))
    hi = min(n-1-i, n-1)
    for k in range(lo, hi+1):
        delta[k + (n-1)] -= (1 - h[i+k])
    # B(k)
    # k=0: B=0 (skip)
    # k=i-j: B=1

    # k where 0<=j+k<n, k!=0, k!=i-j: B=1-h[j+k]
    lo = max(-j, -(n-1))

        delta[k + (n-1)] += (1 - h[j+k])
    # C(k): 0<=i-k<n, k!=0, k!=i-j

def compute_c5(h):

def project_binary(h, target_sum):
    """Project to binary with exactly target_sum ones."""
    n = len(h)
    order = np.argsort(-h)
    result = np.zeros(n)
    result[order[:target_sum]] = 1.0
    return result

def project_onto_constraint(h, target_sum):
    """Project to [0,1]^n with sum = target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)

        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
