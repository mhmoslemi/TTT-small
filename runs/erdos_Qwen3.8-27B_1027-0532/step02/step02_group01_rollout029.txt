import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:

def compute_c5(h):
    n = len(h)

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx

    x -= np.max(x)

        wk = w[k]

        s = k - (n - 1)

            if delta < 0 or (t > 1e-9 and rng.random() < np.exp(-delta / t)):
                scores[p] = new_score
                if new_score < best_bin_c5:
                    best_bin_c5 = new_score
                    best_bin_h = pop[p].copy()
                    steps_since_improve = 0
                else:
                    steps_since_improve += 1
            else:
                for i in range(nswap):
                    pop[p][chosen_ones[i]] = 1.0
                    pop[p][chosen_zeros[i]] = 0.0
                steps_since_improve += 1
        else:
            blen = int(rng.integers(2, max(3, n0 // 5) + 1))
            a = int(rng.integers(0, n0 - blen + 1))
            b = int(rng.integers(0, n0 - blen + 1))
            if abs(a - b) < blen:
                step += 1
                continue
            block_a = pop[p][a:a + blen].copy()
            block_b = pop[p][b:b + blen].copy()
            pop[p][a:a + blen] = 1.0 - block_b
            pop[p][b:b + blen] = 1.0 - block_a
            new_score = compute_c5(pop[p])
            delta = new_score - scores[p]
            if delta < 0 or (t > 1e-9 and rng.random() < np.exp(-delta / t)):
                scores[p] = new_score
                if new_score < best_bin_c5:
                    best_bin_c5 = new_score
                    best_bin_h = pop[p].copy()
                    steps_since_improve = 0
                else:
                    steps_since_improve += 1
            else:
                pop[p][a:a + blen] = block_a
                pop[p][b:b + blen] = block_b
                steps_since_improve += 1

        if steps_since_improve > 30000:
            for pp in range(n_pop):
                if scores[pp] > best_bin_c5 + 0.02:
                    pop[pp] = random_binary(n0, rng)
                    scores[pp] = compute_c5(pop[pp])
            steps_since_improve = 0

        step += 1

    best_bin_h, best_bin_c5 = fast_steepest_binary(best_bin_h, 50, budget_s - 200, start_time)

    h_refine = best_bin_h.astype(float) + 0.01 * rng.standard_normal(n0)
    h_refine = project_onto_constraint(h_refine, target_sum)
    remaining = budget_s - (time.time() - start_time)

    eps_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for eps in eps_list:
        if time.time() - start_time > budget_s - 150:
            break
        h_opt, c5_opt = slsqp_optimize(h_refine, eps, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_refine = h_opt.copy()

    remaining = budget_s - (time.time() - start_time) - 50
    restart_count = 0
    while remaining > 50 and restart_count < 25:
        strategy = restart_count % 6
        if strategy == 0:
            h_pert = best_h + 0.02 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = best_bin_h.astype(float) + 0.04 * rng.standard_normal(n0)
        elif strategy == 2:
            h_tmp = best_h + 0.2 * rng.standard_normal(n0)
            h_pert = make_binary_from_continuous(h_tmp, rng).astype(float)
            h_pert = h_pert + 0.03 * rng.standard_normal(n0)
        elif strategy == 3:
            h_pert = random_binary(n0, rng).astype(float) + 0.05 * rng.standard_normal(n0)
        elif strategy == 4:
            alpha = rng.uniform(0.4, 0.7)
            h_rand = random_binary(n0, rng).astype(float)
            h_pert = alpha * best_h + (1.0 - alpha) * h_rand
        else:
            h_pert = best_bin_h.astype(float)
            nflip = max(2, n0 // 12)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = h_pert + 0.02 * rng.standard_normal(n0)

        h_pert = project_onto_constraint(h_pert, target_sum)
        tl = min(remaining - 20, 50)
        if tl > 25:
            h_opt, c5_opt = slsqp_optimize(h_pert, 0.001, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_opt2, c5_opt2 = slsqp_optimize(h_opt, 0.0001, 300)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()

        restart_count += 1
        remaining = budget_s - (time.time() - start_time) - 50

    remaining = budget_s - (time.time() - start_time) - 30
    if remaining > 180:
        n1 = 2 * n0
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[i // 2]
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.01 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        eps_up = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        for eps in eps_up:
            if time.time() - start_time > budget_s - 30:
                break
            h_opt_up, c5_opt_up = slsqp_optimize(h_up, eps, 200)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
            h_up = h_opt_up.copy()

    remaining = budget_s - (time.time() - start_time) - 10
    if remaining > 30:
        n_cur = len(best_h)
        for eps in [0.00005, 0.00002, 0.00001]:
            if time.time() - start_time > budget_s - 5:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, eps, 400)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
