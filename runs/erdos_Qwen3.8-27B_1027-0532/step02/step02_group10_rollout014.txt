import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    h = h.copy()
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    value = m + np.log(total) / alpha
    weights = exp_s / total
    return value, weights


def grad_single_lag(h, lag, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    v1 = (j + lag >= 0) & (j + lag < n)
    grad[v1] += (1.0 - h[j[v1] + lag]) * dx
    v2 = (j - lag >= 0) & (j - lag < n)
    grad[v2] += (-h[j[v2] - lag]) * dx
    return grad


def smooth_max_grad(h, alpha, n, active_indices=None):
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    sm_val, weights = smooth_max(overlaps, alpha)
    if active_indices is not None:
        w_sel = weights[active_indices]
        idx_sel = active_indices
    else:
        w_sel = weights
        idx_sel = np.arange(len(weights))
    grad = np.zeros(n)
    for m in range(len(w_sel)):
        w = w_sel[m]
        if w < 1e-12:
            continue
        idx = idx_sel[m]
        lag = idx - (n - 1)
        grad += w * grad_single_lag(h, lag, n)
    return sm_val, grad, overlaps, weights


def pgd_optimize(h, alpha, target, max_iters, deadline, top_k=None):
    n = len(h)
    h = h.copy()
    step = min(0.4, 1.0 / (alpha * (2.0 / n) * 3.0 + 1.0))
    prev_val = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        val, grad, overlaps, _ = smooth_max_grad(h, alpha, n)
        if top_k is not None:
            active = np.argsort(overlaps)[-top_k:]
            val2, grad2, _, _ = smooth_max_grad(h, alpha, n, active)
            grad = grad2
        if prev_val is not None:
            if val < prev_val - 1e-14:
                stall = 0
            else:
                stall += 1
            if stall > 150:
                break
        prev_val = val
        s = step
        improved = False
        for _ in range(25):
            h_t = project_onto_constraint(h - s * grad, target)
            val_t, _, _, _ = smooth_max_grad(h_t, alpha, n)
            if val_t < val - 1e-15:
                h = h_t
                improved = True
                break
            s *= 0.5
            if s < 1e-14:
                break
        if not improved:
            break
        step = min(s * 1.35, 1.0)
    return h


def coordinate_descent(h, target, n_passes, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        for i in order:
            fixed = h.copy()
            fixed[i] = 0.0
            overlaps_base = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                lag = m - (n - 1)
                j = np.arange(n)
                v1 = (j + lag >= 0) & (j + lag < n)
                c_val = np.sum(fixed[j[v1]] * (1.0 - fixed[j[v1] + lag]))
                overlaps_base[m] = c_val * dx
            best_val = np.max(overlaps_base)
            best_hi = 0.0
            candidate_values = np.array([0.0, 1.0])
            for m in range(2 * n - 1):
                lag = m - (n - 1)
                v1 = (i + lag >= 0) & (i + lag < n)
                v2 = (i - lag >= 0) & (i - lag < n)
                coeff = 0.0
                if v1:
                    coeff += (1.0 - fixed[i + lag]) * dx
                if v2:
                    coeff += (-fixed[i - lag]) * dx
                if abs(coeff) > 1e-15:
                    intercept = overlaps_base[m]
                    cross = -intercept / coeff if coeff != 0 else -1.0
                    if 0.0 <= cross <= 1.0:
                        candidate_values = np.append(candidate_values, cross)
            for cv in candidate_values:
                cv = np.clip(cv, 0.0, 1.0)
                h_test = fixed.copy()
                h_test[i] = cv
                ov = np.zeros(2 * n - 1)
                for m in range(2 * n - 1):
                    lag = m - (n - 1)
                    j = np.arange(n)
                    v1 = (j + lag >= 0) & (j + lag < n)
                    ov[m] = np.sum(h_test[j[v1]] * (1.0 - h_test[j[v1] + lag])) * dx
                if np.max(ov) < best_val - 1e-15:
                    best_val = np.max(ov)
                    best_hi = cv
            h[i] = best_hi
        h = project_onto_constraint(h, target)
    return h


def active_set_lp(h, top_m, deadline):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    top_indices = np.argsort(overlaps)[-top_m:]
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_rows = []
    b_vals = []
    for idx in top_indices:
        lag = idx - (n - 1)
        g = grad_single_lag(h, lag, n)
        c_val = overlaps[idx]
        rhs = dx * (c_val - np.dot(g * dx, h / dx))
        row = np.zeros(n_vars)
        row[:n] = g * dx
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(dx * c_val - np.dot(g, h) * dx)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            h_new = np.clip(res.x[:n], 0, 1)
            h_new = project_onto_constraint(h_new, n / 2.0)
            return h_new
    except Exception:
        pass
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.25, 1.0 / (alpha * dx * 5.0 + 1.0))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        val, grad, _, _ = smooth_max_grad(y, alpha, n)
        x_new = project_onto_constraint(y - step * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x
