from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corrs)
    s = alpha * (corrs - m)
    s = np.clip(s, -60.0, 60.0)
    e = np.exp(s)
    se = np.sum(e)
    w = e / se
    obj = m + np.log(se) / alpha
    wr = w[::-1]
    c1 = np.convolve(1.0 - h, wr, mode='full')
    c2 = np.convolve(h, w, mode='full')
    idx = np.arange(n) + n - 1
    grad = c1[idx] - c2[idx]
    return obj, grad

    def fun(x):
        o, g = smooth_max_and_grad(x, alpha)
        v = np.sum(x) - n * 0.5
        o += 0.5 * mu * v * v
        g = g + mu * v
        return o, g
    return fun

def lbfgs_optimize(h0, alpha, mu, maxiter):
    n = len(h0)
    b = [(0.0, 1.0)] * n
    fun = make_lbfgs_fun(alpha, mu, n)
    res = minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=b,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'iprint': 0, 'maxls': 50})
    return res.x

def binary_hill_climb(h, n, max_iters, rng):
    """First-improvement hill climbing on binary vectors."""
    h = h.copy()
    cur = np.max(compute_correlations(h))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best = cur
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if no_improve > 5000:
            break
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        h[a] = 0.0
        h[b] = 1.0
        new = np.max(compute_correlations(h))
        if new < cur:
            cur = new
            no_improve = 0
            if new < best:
                best = new
                best_h = h.copy()
            ones = np.delete(ones, ia)
            ones = np.append(ones, b)
            zeros = np.delete(zeros, ib)
            zeros = np.append(zeros, a)
        else:
            h[b] = 0.0
            h[a] = 1.0

        b = zeros[ib]
        h[a] = 0.0

            ones = np.append(ones, b)
            zeros = np.delete(zeros, ib)
            zeros = np.append(zeros, a)
        else:
            h[b] = 0.0
            h[a] = 1.0
    return best_h, best

def subgradient_refine(h, n, steps, lr_init, lr_decay, rng):
    """Subgradient method focusing on active lag."""
    h = h.copy()
    for s in range(steps):
        corrs = compute_correlations(h)
        active = np.argmax(corrs)
        r = active - (n - 1)
        grad = np.zeros(n)
        if r >= 0:
            grad[:n - r] += (1.0 - h[r:])
            grad[r:] -= h[:n - r]
        else:
            s2 = -r
            grad[s2:] += (1.0 - h[:n - s2])
            grad[:n - s2] -= h[s2:]
        lr = lr_init * (lr_decay ** s)
        h = project_feasible(h - lr * grad, n * 0.5)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Initialize from parent
    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)
    best_h = project_feasible(init, n0 * 0.5)
    best_score = evaluate_score(best_h)
    
    # Phase 1: Binary local search at even resolutions
    # Interpolate to even sizes and do binary optimization
    even_sizes = [424, 426, 428, 430, 432, 436, 440, 444, 448]
    
    for n_try in even_sizes:
        if time_left() < 200:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        
        # Binary projection
        h_bin = np.zeros(n_try)
        top_idx = np.argsort(-h_interp)[:n_try // 2]
        h_bin[top_idx] = 1.0
        
        # Hill climbing
        h_bin, _ = binary_hill_climb(h_bin, n_try, 50000, rng)
        score = evaluate_score(h_bin)
        if score < best_score:
            best_score = score
            best_h = h_bin.copy()
        
        # Also try continuous optimization from this binary
        if time_left() < 100:
            break
        h_cont = h_bin.astype(float)
        for alpha in [500, 2000]:
            if time_left() < 50:
                break
            h_new = lbfgs_optimize(h_cont, alpha, 5000.0, 300)
            h_new = project_feasible(h_new, n_try * 0.5)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                h_cont = h_new.copy()
    
    # Phase 2: Random restarts with binary hill climbing
    n_current = len(best_h)
    # Make sure we work with an even size for binary operations

            flip_to = rng.choice(zeros, size=min(n_flip, len(zeros)), replace=False)
            h_pert[flip_from] = 0.0
