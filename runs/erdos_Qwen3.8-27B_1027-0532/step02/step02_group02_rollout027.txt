def _c5_of(h):

def _proj(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        d = ts - h.sum()
        if abs(d) < 1e-15:
            return h
        fr = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = fr.sum()
        if nf == 0:
            if d > 0:
                idx = np.argsort(1.0 - h)
            else:
                idx = np.argsort(h)
            for i in idx:
                if abs(d) < 1e-15:
                    break
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a
                    d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a
                    d += a
            return h
