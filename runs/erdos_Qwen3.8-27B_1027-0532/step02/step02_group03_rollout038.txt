import time
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    # Project onto {0 ≤ h ≤ 1, sum(h) = target_sum}
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    # Project y onto {0 ≤ x ≤ 1, sum(x) = S}
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _fftc5(h):
    # Fast C5 using FFT
    n = len(h)
    dh = 1.0 - h
    # Cross-correlation via FFT
    size = 2 * n
    F_h = fft(h, size)
    F_dh = fft(dh, size)
    corr = np.real(ifft(F_h * np.conj(F_dh)))
    # The 'full' correlation is shifted
    # np.correlate(a, v, 'full') has lag from -(n-1) to (n-1)
    # FFT gives circular correlation, need to rearrange
    # Actually, let me just use np.correlate since n is small
    return float(np.max(corr) * (2.0 / n))

def _de_optimize(h_init, n, target_sum, pop_size, max_gen, 
                 F_start, F_end, CR, budget_s, deadline):
    # Differential Evolution optimizer
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = _project(h_init, target_sum)
    for i in range(1, pop_size):
        # Mix of random and perturbed initial
        if i % 3 == 1:
            pop[i] = _project(rng.rand(n), target_sum)
        elif i % 3 == 2:
            pop[i] = _project(h_init + 0.3 * rng.randn(n), target_sum)
        else:
            pop[i] = _project((rng.rand(n) < 0.5).astype(float), target_sum)
    
    # Evaluate initial population
    fitness = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_val = fitness[best_idx]
    best_h = pop[best_idx].copy()
    
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        
        # Adaptive F
        progress = gen / max_gen
        F = F_start + (F_end - F_start) * progress
        
        for i in range(pop_size):
            if time.time() > deadline:
                    break
            
            # Select three distinct random indices
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            
            # Mutation: DE/rand/1
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            
            # Crossover
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            
            # Project
            trial = _project(trial, target_sum)
            
            # Selection
            trial_fit = _c5_of(trial)
            if trial_fit <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_fit
                if trial_fit < best_val:
                    best_val = trial_fit
                    best_h = trial.copy()
        
        # Periodic diversification
        if gen % 50 == 49:
            # Replace worst 20% with random
            worst_indices = np.argsort(fitness)[-pop_size // 5:]
            for idx in worst_indices:
                pop[idx] = _project(rng.rand(n) + 0.1 * rng.randn(n), target_sum)
                fitness[idx] = _c5_of(pop[idx])
                if fitness[idx] < best_val:
                    best_val = fitness[idx]
                    best_h = pop[idx].copy()
    
    return best_h, best_val

    stall_count = 0

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    # Smooth SLSQP optimization using soft-max
    def _smooth_obj(h, alpha, n):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        if m < -500:
            return m
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    def _eq_con_fun(x):
        return x.sum() - target_sum
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent(h0, n, target_sum, max_rounds=30, values=None):
    if values is None:
        values = [0.0, 0.25, 0.5, 0.75, 1.0]
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in values:
                if v == old:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

import time
import numpy as np
from scipy.optimize import minimize


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _fix_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def _project(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def _slsqp_smooth_obj(x, args):
    alpha, n, target_sum = args
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return m
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)


def _slsqp_eq_con(x):
    return x.sum() - x['_ts'] if isinstance(x, dict) else x.sum()


class _SLSQPState:
    pass


_slq_state = _SLSQPState()


def _eq_fun(x):
    return x.sum() - _slq_state.target_sum


def _eq_jac(x):
    return np.ones(_slq_state.n)


def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    _slq_state.target_sum = float(target_sum)
    _slq_state.n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_slsqp_smooth_obj, h0, args=(alpha, n, target_sum),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0


def _de_best1(h_init, n, target_sum, pop_size, max_gen,
              F_start, F_end, CR, deadline, seed):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = _project(h_init, target_sum)
    for i in range(1, pop_size):
        r = rng.rand()
        if r < 0.3:
            pop[i] = _project(rng.rand(n), target_sum)
        elif r < 0.6:
            pop[i] = _project(h_init + 0.4 * rng.randn(n), target_sum)
        else:
            block = np.zeros(n)
            pos = rng.randint(0, max(1, n - 10))
            ln = rng.randint(n // 4, n // 2 + 1)
            block[pos:pos + ln] = 1.0
            pop[i] = _project(block + 0.15 * rng.randn(n), target_sum)

    fitness = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_val = float(fitness.min())
    best_h = pop[int(np.argmin(fitness))].copy()
    stagnation = 0

    for gen in range(max_gen):
        if time.time() > deadline:
            break
        progress = gen / max(float(max_gen), 1.0)
        F = F_start * (1.0 - progress) + F_end * progress
        CR_eff = CR * (1.0 - 0.3 * progress)
        best_idx = int(np.argmin(fitness))

        for i in range(pop_size):
            if time.time() > deadline:
                break
            indices = rng.choice(pop_size, 3, replace=False)
            r1, r2, r3 = int(indices[0]), int(indices[1]), int(indices[2])
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            mask = rng.rand(n) < CR_eff
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            trial_fit = _c5_of(trial)
            if trial_fit <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_fit
                if trial_fit < best_val - 1e-12:
                    best_val = trial_fit
                    best_h = trial.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1

        if stagnation > pop_size * 2:
            stagnation = 0
            n_div = max(1, pop_size // 5)
            for _ in range(n_div):
                idx = rng.randint(pop_size)
                pop[idx] = _project(rng.rand(n) + 0.2 * rng.randn(n), target_sum)
                fitness[idx] = _c5_of(pop[idx])
                if fitness[idx] < best_val:
                    best_val = fitness[idx]
                    best_h = pop[idx].copy()

    return best_h, best_val

def _coord_descent(h0, n, target_sum, max_rounds, values):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in values:
                if abs(v - old) < 1e-10:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h


def _multi_start_de(n, h_parent, target_sum, deadline, seed, phase_budget):
    best_c = float('inf')
    best_h = None
    rng = np.random.RandomState(seed)

    if h_parent is not None:
        h_guess = _fix_sum(h_parent.copy(), target_sum)
    else:
        h_guess = _fix_sum(rng.rand(n), target_sum)

    c5_g = _c5_of(h_guess)
    if c5_g < best_c:
        best_c = c5_g
        best_h = h_guess.copy()

    configs = [
        (0.9, 0.35, 0.7, 40, 400),
        (0.8, 0.25, 0.8, 35, 350),
        (1.0, 0.5, 0.9, 30, 300),
        (0.7, 0.2, 0.6, 45, 350),
    ]

    for ci, (F_s, F_e, CR, pop_sz, max_g) in enumerate(configs):
        if time.time() > deadline - 30:
            break
        sub_deadline = min(deadline - 30, time.time() + phase_budget)
        h_de, c5_de = _de_best1(h_guess, n, target_sum, pop_sz, max_g,
                                F_s, F_e, CR, sub_deadline, seed + ci * 7)
        if c5_de < best_c:
            best_c = c5_de
            best_h = h_de.copy()

        if time.time() < deadline - 60:
            h_rand = _fix_sum(rng.rand(n), target_sum)
            sub_dl2 = min(deadline - 30, time.time() + phase_budget * 0.6)
            h_de2, c5_de2 = _de_rand1(h_rand, n, target_sum, pop_sz - 5,
                                       max_g - 50, F_s, F_e, CR,
                                       sub_dl2, seed + ci * 7 + 500)
            if c5_de2 < best_c:
                best_c = c5_de2
                best_h = h_de2.copy()

    return best_h, best_c
