```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0/eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    n_corr = 2*n - 1
    for k in range(n_corr):
        l = k - (n-1)
        wk = w[k]
        if wk < 1e-14:
            continue
        # grad of f_k = sum_i h[i](1-h[i+k]) w.r.t. h[j]
        # d/dh[j] sum_i h[i](1-h[i+k])
        # = sum_i [delta(i,j)(1-h[i+k]) + h[i)*(-delta(i+k,j))]
        # = (1-h[j+k]) if j+k < n  (from the first term, i=j)
        #   - h[j-k] if j-k >= 0    (from the second term, i+k=j, so i=j-k)
        j_range = np.arange(n)
        # First term: i=j, contributes (1-h[j+k]) for all j where j+k in [0,n)
        valid1 = (j_range + l >= 0) & (j_range + l < n)

        valid2 = (j_range - l >= 0) & (j_range - l < n)
        grad[valid2] += wk * (-h[j_range[valid2] - l])
    grad *= dx
    return grad

    l = k_star - (n-1)
    grad = np.zeros(n)
    j_range = np.arange(n)
    valid1 = (j_range + l >= 0) & (j_range + l < n)
    grad[valid1] += (1.0 - h[j_range[valid1] + l])
    valid2 = (j_range - l >= 0) & (j_range - l < n)
    grad[valid2] += (-h[j_range[valid2] - l])
    grad *= dx
    return grad

def block_nelder_mead(h, block_indices, n, maxiter=200):
    """Optimize a block of coordinates using Nelder-Mead on the exact objective."""
    def block_obj(x_block):
        h_temp = h.copy()
        h_temp[block_indices] = x_block
        h_temp = np.clip(h_temp, 0.0, 1.0)
        h_temp = project_feasible(h_temp, n)
        # Re-extract block (might have changed due to projection)
        return evaluate_c5(h_temp)
    
    x0 = h[block_indices].copy()
    result = minimize(block_obj, x0, method='Nelder-Mead', 
                      options={'maxiter': maxiter, 'fatol': 1e-14})
    h_new = h.copy()
    h_new[block_indices] = result.x
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, evaluate_c5(h_new)

def nesterov_smooth(h_start, n, bounds, eps0, eps1, n_iters, time_limit):
    """Nesterov accelerated gradient on smooth max with decreasing eps."""
    h = h_start.copy()
    m = h_start.copy()  # momentum
    best_h = h_start.copy()
    best_c5 = evaluate_c5(h_start)
    
    for t in range(1, n_iters+1):
        if time_limit():
            break
        # Interpolate eps
        eps = eps0 * (eps1/eps0) ** (t / n_iters)
        # Learning rate
        lr = 2.0 / (t + 10.0)
        
        # Nesterov step
        m_new = m - lr * smooth_max_grad(m, eps, n)
        # Project
        m_new = project_feasible(m_new, n)
        # Update momentum
        beta = (1 + np.sqrt(1 + 4*(t-1)**2)) / (1 + np.sqrt(1 + 4*t**2))
        h_new = m_new + beta * (m_new - h)
        h_new = project_feasible(h_new, n)
        m = h_new
        h = h_new
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

Actually, let me think about this differently. The projection problem is:
minimize ∑ (y_i - x_i)^2 subject to 0 ≤ y_i ≤ 1, ∑ y_i = target

This is a convex problem. The KKT conditions give:
y_i = clip(x_i - λ, 0, 1) for some λ.

To find λ, note that g(λ) = ∑ clip(x_i - λ, 0, 1) is a piecewise linear decreasing function of λ. We need g(λ) = target.

The breakpoints are at λ = x_i and λ = x_i - 1. Between consecutive breakpoints, g is linear. So we can:
1. Sort all breakpoints: x_1, x_2, ..., x_n, x_1-1, x_2-1, ..., x_n-1
2. Find the interval containing the solution
3. Solve the linear equation

```python
def topk_subgradient(h, n, k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Find top-k shifts
    top_k_indices = np.argsort(corr)[-k:][::-1]
    # Weight by softmax of top-k
    top_vals = corr[top_k_indices]
    weights = np.exp(100 * (top_vals - top_vals[0]))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    j_range = np.arange(n)
    for idx, ki in enumerate(top_k_indices):
        l = ki - (n-1)
        valid1 = (j_range + l >= 0) & (j_range + l < n)
        grad[valid1] += weights[idx] * (1.0 - h[j_range[valid1] + l])
        valid2 = (j_range - l >= 0) & (j_range - l < n)
        grad[valid2] += weights[idx] * (-h[j_range[valid2] - l])
    grad *= dx
    return grad
```

```python
def smooth_max_grad(h, eps, n):

        ht[block_idx] = xb
        ht = np.clip(ht, 0.0, 1.0)

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return project_feasible(out, n_to)

def downsamples(h, n_from, n_to):
```
