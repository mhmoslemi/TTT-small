The overlap at lag s is: c(s) = Σ_i h[i] * (1-h[i+s]) (over valid i)

But this is not a standard convolution because of the (1-h) term and the variable domain. Let me think...

c(s) = Σ_i h[i] - Σ_i h[i]*h[i+s] = (part of sum of h) - (autocorrelation of h at lag s)

The first term depends on s (it's the sum of h over a range that depends on s). The second term is the autocorrelation, which can be computed via FFT.

So: c(s) = S(s) - R(s) where S(s) = sum of h over the valid range, and R(s) = autocorrelation at lag s.

The gradient of S(s) w.r.t. h[j] is 1 if j is in the valid range, else 0.
The gradient of R(s) w.r.t. h[j] is h[j+s] + h[j-s] (with boundary effects).

So: ∂c(s)/∂h[j] = I_valid(j,s) - h[j+s] - h[j-s] (with appropriate boundary handling)

This is exactly what we had before. The FFT approach would help compute R(s) for all s at once, but for the gradient, we still need the per-lag contributions.

Actually, I realize that the full overlap array can be computed as:
correlate(h, 1-h, 'full') = correlate(h, 1, 'full') - correlate(h, h, 'full')

The first term is just a "boxcar" convolution (sum of h over a window), and the second is the autocorrelation. Both can be computed via FFT in O(n log n).

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_w(overlaps, alpha):
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    return m + np.log(total) / alpha, exp_s / total


def compute_grad(h, weights, dx, n):
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            ve = n - offset
            if ve > 0:
                grad[:ve] += w * (1.0 - h[offset:offset + ve])
                grad[offset:offset + ve] -= w * h[:ve]
        else:
            sh = -offset
            ve = n - sh
            if ve > 0:
                grad[sh:sh + ve] += w * (1.0 - h[:ve])
                grad[:ve] -= w[sh:sh + ve]
    return grad * dx


def lbfgs_solve(h_start, alpha, pen, max_iter, nv, tx, dx):
    def obj(x):
        ov = compute_overlaps(x)
        sm, _ = smooth_max_w(ov, alpha)
        return sm + pen * (x.sum() - tx) ** 2

    def grd(x):
        ov = compute_overlaps(x)
        _, w = smooth_max_w(ov, alpha)
        g = compute_grad(x, w, dx, nv)
        g += 2.0 * pen * (x.sum() - tx)
        return g

    bounds = [(0.0, 1.0)] * nv
    res = minimize(obj, h_start, method='L-BFGS-B', jac=grd, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                            'maxls': 50, 'maxfun': 200000})
    return res.x if res.x is not None else h_start


def pgd_refine(h0, alpha, target, max_iters, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    step = 1.0 / (alpha * dx * 2.0 + 1.0)
    step = min(step, 0.3)
    prev_f = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        sm, w = smooth_max_w(ov, alpha)
        if prev_f is not None:
            if sm < prev_f - 1e-14:
                stall = 0
            else:
                stall += 1
            if stall > 150:
                break
        prev_f = sm
        g = compute_grad(h, w, dx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        s = step
        for _ in range(30):
            ht = project_onto_constraint(h - s * g, target)
            ov_t = compute_overlaps(ht)
            sm_t, _ = smooth_max_w(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = ht
                improved = True
                break
            s *= 0.5
            if s < 1e-14:
                break
        if not improved:
            break
        step = min(s * 1.5, 0.5)
    return h
