        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)

    m = np.max(x)
    shifted = x - m
    max_shifted = np.max(shifted)
    if max_shifted > 0.5:
        shifted = shifted - max_shifted
        m += max_shifted
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def smooth_obj(h, alpha):
    corr = compute_correlations(h)
    return smooth_max(corr, alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds, ftol=1e-15):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def project_sum(h, target):

        m += max_sh
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
