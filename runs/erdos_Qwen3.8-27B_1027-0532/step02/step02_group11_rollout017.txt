    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for t in range(1, max_iters + 1):
        if time.time() > time_budget:
            break
        frac = t / max_iters
        eps = start_eps * (end_eps / start_eps) ** frac
        
        corr = compute_corr(h)
        m_val = np.max(corr)
        shifted = (corr - m_val) / eps
        w = np.exp(shifted)
        w_sum = w.sum()
        w = w / w_sum
        obj_val = (m_val + eps * np.log(w_sum)) * (2.0 / n)
        
        # Gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        j = np.arange(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-10:
                continue
            mask1 = (j - k >= 0) & (j - k < n)
            mask2 = (j + k >= 0) & (j + k < n)
            if mask1.any():
                grad[mask1] += wk * (1.0 - h[j[mask1] - k])
            if mask2.any():
                grad[mask2] += wk * (-h[j[mask2] + k])
        grad *= dx
        
        # Adam update
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad * grad
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_a)
        h_new = project_feasible(h_new, n)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def binary_sa(h_init, n, max_iters, time_budget, T0=1.0, T1=0.01):
    """Simulated annealing on binary sequences using pair swaps."""
    h = h_init.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    cur_corr = compute_corr(h)
    cur_max = np.max(cur_corr)
    
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    
    for t in range(max_iters):
        if time.time() > time_budget:
            break
        T = T0 * (T1 / T0) ** (t / max_iters)
        
        # Random pair swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Try the swap
        h_swap = h.copy()
        h_swap[i], h_swap[j] = 0.0, 1.0
        new_corr = compute_corr(h_swap)
        new_max = np.max(new_corr)
        new_c5 = new_max * dx
        
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            h = h_swap
            # Update ones/zeros sets
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            cur_c5 = new_c5
            cur_corr = new_corr
            cur_max = new_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

Hmm, the issue with the binary SA is that recomputing the full correlation for each swap is O(n) which is fine, but the `np.where` calls to update ones/zeros are also O(n). For n=170, this is all very fast.

But I realize there's a much faster way to do binary SA: instead of recomputing the full correlation, I can compute the DELTA in the correlation for a given swap. Let me derive this properly.

For the full correlation: corr[k] = sum_l h[l+k]*(1-h[l]) for valid l (where valid means 0<=l<n and 0<=l+k<n).

When we swap positions i (1→0) and j (0→1):

The change in corr[k] is:
- Position i: was contributing h[i]*(1-h[i-k]) = 1*(1-h[i-k]) at l=i-k (if valid), and h[i+k]*(1-h[i]) = h[i+k]*0 = 0 at l=i.
  After swap (h[i]=0): contributes 0 at l=i-k, and h[i+k]*(1-0) = h[i+k] at l=i.
  Net change from i: +h[i+k]*[0<=i+k<n] - (1-h[i-k])*[0<=i-k<n]

Wait, I need to be more careful. Let me denote the correlation value at "shift index" k (where k=0 corresponds to shift -(n-1) and k=2n-2 corresponds to shift +(n-1)).

corr[k] = sum_{l: 0<=l<n, 0<=l+k<n} h[l+k] * (1-h[l])

For position i (h[i]=1 before, 0 after):
- Before: i contributes to corr[k] when:
  (a) l+k = i, i.e., l = i-k: valid if 0<=i-k<n. Contribution: h[i]*(1-h[i-k]) = 1-h[i-k]
  (b) l = i: valid if 0<=i+k<n. Contribution: h[i+k]*(1-h[i]) = h[i+k]*0 = 0
  Total before: (1-h[i-k])*[0<=i-k<n]

- After (h[i]=0): 
  (a) l+k = i, l = i-k: contribution: 0*(1-h[i-k]) = 0
  (b) l = i: contribution: h[i+k]*(1-0) = h[i+k]
  Total after: h[i+k]*[0<=i+k<n]

- Change from i: Δ_i[k] = h[i+k]*[0<=i+k<n] - (1-h[i-k])*[0<=i-k<n]

For position j (h[j]=0 before, 1 after):
- Before: 
  (a) l+k = j, l = j-k: contribution: 0*(1-h[j-k]) = 0
  (b) l = j: contribution: h[j+k]*(1-0) = h[j+k]
  Total before: h[j+k]*[0<=j+k<n]

- After (h[j]=1):
  (a) l+k = j, l = j-k: contribution: 1*(1-h[j-k]) = 1-h[j-k]
  (b) l = j: contribution: h[j+k]*(1-1) = 0
  Total after: (1-h[j-k])*[0<=j-k<n]

- Change from j: Δ_j[k] = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]

Total change: Δ[k] = Δ_i[k] + Δ_j[k]

This can be computed in O(n) for all k simultaneously using the vectorized approach.

Let me implement this:

```python
def binary_swap_correlation_delta(h, i, j, n):
    """Compute delta in correlation array when swapping h[i]=1 to 0 and h[j]=0 to 1."""
    delta = np.zeros(2 * n - 1)
    
    # For i going 1->0:
    # Δ_i[k] = h[i+k]*[0<=i+k<n] - (1-h[i-k])*[0<=i-k<n]
    # k ranges [0, 2n-2]
    # i+k in [0, n-1] => k in [-i, n-1-i] => k in [max(0,-i), min(2n-2, n-1-i)]
    # i-k in [0, n-1] => k in [i-(n-1), i] => k in [max(0, i-(n-1)), min(2n-2, i)]
    
    # First part: h[i+k] for k where 0<=i+k<n
    k_start = max(0, -i)
    k_end = min(2*n-2, n-1-i)
    if k_start <= k_end:
        ks = np.arange(k_start, k_end+1)
        delta[ks] += h[i + ks]
    
    # Second part: -(1-h[i-k]) for k where 0<=i-k<n
    k_start2 = max(0, i - (n-1))
    k_end2 = min(2*n-2, i)
    if k_start2 <= k_end2:
        ks2 = np.arange(k_start2, k_end2+1)
        delta[ks2] -= (1.0 - h[i - ks2])
    
    # For j going 0->1:
    # Δ_j[k] = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]
    
    # First part: (1-h[j-k]) for k where 0<=j-k<n
    k_start3 = max(0, j - (n-1))
    k_end3 = min(2*n-2, j)
    if k_start3 <= k_end3:
        ks3 = np.arange(k_start3, k_end3+1)
        delta[ks3] += (1.0 - h[j - ks3])
    
    # Second part: -h[j+k] for k where 0<=j+k<n
    k_start4 = max(0, -j)
    k_end4 = min(2*n-2, n-1-j)
    if k_start4 <= k_end4:
        ks4 = np.arange(k_start4, k_end4+1)
        delta[ks4] -= h[j + ks4]
    
    return delta
```

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_and_weights(h, eps):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    return m, w, corr

def compute_grad_vec(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-11:
            continue
        mask1 = (j - k >= 0) & (j - k < n)
        mask2 = (j + k >= 0) & (j + k < n)
        if np.any(mask1):
            grad[mask1] += wk * (1.0 - h[j[mask1] - k])
        if np.any(mask2):
            grad[mask2] += wk * (-h[j[mask2] + k])
    grad *= dx
    return grad

def adam_refine(h_init, n, deadline, start_eps=30.0, end_eps=0.3, lr=0.08, max_iters=200000):
    h = h_init.astype(float).copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2 = 0.8, 0.999
    eps_a = 1e-8
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    t = 0
    while t < max_iters:
        t += 1
        if time.time() > deadline - 2.0:
            break
        frac = min(1.0, t / max_iters)
        eps = start_eps * (end_eps / start_eps) ** frac
        m_val, weights, corr = smooth_max_and_weights(h, eps)
        grad = compute_grad_vec(h, weights, n)
        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        m_hat = m_v / (1.0 - b1 ** t)
        v_hat = v_v / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_a)
        h_new = project_feasible(h_new, n)

            best_h = h_new.copy()
        h = h_new

def binary_sa(h_init, n, deadline, T0=0.5, T1=0.005, max_iters=5000000):

def slsqp_topk(h_init, n, deadline, k_top=8, eps=100.0, maxiter=300):
    h = h_init.astype(float).copy()
    best_c5 = evaluate_c5(h)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        k_idx = np.argsort(corr)[-k_top:]
        top_vals = corr[k_idx]
        m = np.max(top_vals)
        return (m + eps * np.log(np.sum(np.exp((top_vals - m) / eps)))) * (2.0 / n) + 1e6 * (np.sum(x) - n / 2.0) ** 2
    
    def constr(x):
        return np.sum(x) - n / 2.0
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constr}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                return h_new, c5_new
    except Exception:
        pass
    return h_init, best_c5

            out[i] = h[idx]

            h_s2, c5_s2 = slsqp_topk(best_h2, n2, deadline - 15, k_top=8, eps=50.0, maxiter=400)
            if c5_s2 < best_c5_2:
                best_c5_2 = c5_s2
                best_h2 = h_s2
            h_a2b, c5_a2b = adam_refine(best_h2, n2, deadline - 15,
                                         start_eps=5.0, end_eps=0.1, lr=0.02,
                                         max_iters=20000)
            if c5_a2b < best_c5_2:
                best_c5_2 = c5_a2b
                best_h2 = h_a2b
        
        if best_c5_2 < best_c5:
            best_c5 = best_c5_2
            best_h = best_h2
            n = n2

    # Final cleanup
    best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
