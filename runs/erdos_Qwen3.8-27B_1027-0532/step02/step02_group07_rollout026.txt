def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def get_slopes(h, j):
    n = len(h)
    m = 2 * n - 1
    p = np.arange(m)
    slopes = np.zeros(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    return slopes

def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    
    a = corr - h_j * slopes
    
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    
    for _ in range(38):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    
    V_star = (lo_V + hi_V) * 0.5
    
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    
    return float(np.clip(t_star, 0.0, 1.0)), slopes

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    h = project_sum(h, n / 2.0)
    return h

def beatty_init(n, alpha=None):
    if alpha is None:
        alpha = (1 + np.sqrt(5)) / 2.0  # golden ratio
    indices = np.arange(n)
    # Beatty sequence: positions where {i * alpha} < 0.5
    frac = (indices * alpha) % 1.0
    h = (frac < 0.5).astype(float)
    return h

def thue_morse_init(n):
    # Thue-Morse sequence
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i // 2] ^ (i & 1)
    h = tm.astype(float)
    return h

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    zeros = n // 2
    h = np.zeros(n)
    ones = rng.choice(n, size=zeros, replace=False)
    h[ones] = 1.0
    return h

def smooth_random_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    # Sum of random sinusoids
    h_raw = np.zeros(n)
    for k in range(1, 8):
        h_raw += rng.randn() * np.sin(k * np.pi * x + rng.rand() * 2 * np.pi)
    # Normalize to [0, 1]
    h = (h_raw - h_raw.min()) / (h_raw.max() - h_raw.min() + 1e-10)
    return h

def simulated_annealing_cd(h, n_sweeps, time_budget, seed=0, T_start=0.02, T_end=0.0001):
    n = len(h)
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    corr = compute_correlations(h_cur)
    best_h = h_cur.copy()
    best_score = float(np.max(corr))
    t_start = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t_start > time_budget:
            break
        frac = sweep / max(n_sweeps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h_cur, j, corr)
            current_max = float(np.max(corr))
            
            if rng.random() < T * 5:
                # Random perturbation
                delta = rng.uniform(-0.3, 0.3)
                t_new = np.clip(h_cur[j] + delta, 0, 1)
            else:
                t_new = t_opt
            
            change = t_new - h_cur[j]
            if abs(change) < 1e-14:
                continue
            
            new_corr = corr + change * slopes
            new_max = float(np.max(new_corr))
            
            delta_f = new_max - current_max
            if delta_f <= 0 or (T > 1e-12 and rng.random() < np.exp(-delta_f / T)):
                h_cur[j] = t_new
                corr = new_corr
                if new_max < best_score:
                    best_score = new_max
                    best_h = h_cur.copy()
    
    return best_h

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, alpha, maxiter, bounds):
    def smooth_obj(h):
        corr = compute_correlations(h)
        return smooth_max(corr, alpha)
    
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def pair_swap_search(h, n_trials, time_budget, seed=0):
    n = len(h)
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    corr = compute_correlations(h_cur)
    best_h = h_cur.copy()
    best_score = float(np.max(corr))
    t_start = time.time()
    
    for trial in range(n_trials):
        if time.time() - t_start > time_budget:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        if abs(h_cur[i] - h_cur[j]) < 1e-10:
            continue
        
        # Try swap
        h_test = h_cur.copy()
        h_test[i], h_test[j] = h_cur[j], h_cur[i]
        new_corr = compute_correlations(h_test)
        new_max = float(np.max(new_corr))
        
        if new_max < best_score - 1e-14:
            h_cur = h_test
            corr = new_corr
            best_score = new_max
            best_h = h_cur.copy()
    
    return best_h

def block_swap_search(h, block_size, n_trials, time_budget, seed=0):
    n = len(h)
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    corr = compute_correlations(h_cur)
    best_h = h_cur.copy()
    best_score = float(np.max(corr))
    t_start = time.time()
    
    for trial in range(n_trials):
        if time.time() - t_start > time_budget:
            break
        
        i = rng.randint(0, n - block_size + 1)
        j = rng.randint(0, n - block_size + 1)
        if abs(i - j) < block_size:
            continue
        
        # Swap blocks [i, i+bs) and [j, j+bs)
        h_test = h_cur.copy()
        h_test[i:i+block_size], h_test[j:j+block_size] = h_test[j:j+block_size].copy(), h_test[i:i+block_size].copy()
        
        new_corr = compute_correlations(h_test)
        new_max = float(np.max(new_corr))
        
        if new_max < best_score - 1e-14:
            h_cur = h_test
            corr = new_corr
            best_score = new_max
            best_h = h_cur.copy()
    
    return best_h

    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    ones = rng.choice(n, size=n_ones, replace=False)
    h[ones] = 1.0
    return h


def smooth_random_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h_raw = np.zeros(n)
    for k in range(1, 10):
        h_raw += rng.randn() * np.sin(k * np.pi * x + rng.rand() * 2 * np.pi)
    h = (h_raw - h_raw.min()) / (h_raw.max() - h_raw.min() + 1e-10)
    return h


def simulated_annealing_cd(h, n_sweeps, time_budget, seed=0, T_start=2.0, T_end=0.005):
    n = len(h)
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    corr = compute_correlations(h_cur)
    best_h = h_cur.copy()
    best_score = float(np.max(corr))
    t_start = time.time()

    for sweep in range(n_sweeps):
        if time.time() - t_start > time_budget:
            break
        frac = sweep / max(n_sweeps - 1, 1)
        T = T_start * (T_end / T_start) ** frac

        for j in range(n):
            t_opt, slopes = optimize_coordinate(h_cur, j, corr)
            current_max = float(np.max(corr))

            if rng.random() < T * 0.5:
                delta = rng.uniform(-0.4, 0.4)
                t_new = np.clip(h_cur[j] + delta, 0, 1)
            elif rng.random() < T * 0.2:
                t_new = 1.0 - t_opt
            else:
                t_new = t_opt

            change = t_new - h_cur[j]
            if abs(change) < 1e-14:
                continue

            new_corr = corr + change * slopes
            new_max = float(np.max(new_corr))

            delta_f = new_max - current_max
            if delta_f <= 0 or (T > 1e-12 and rng.random() < np.exp(-delta_f / T)):
                h_cur[j] = t_new
                corr = new_corr
                if new_max < best_score:
                    best_score = new_max
                    best_h = h_cur.copy()

    return best_h


def pair_swap_search(h, n_trials, time_budget, seed=0):
    n = len(h)
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = float(np.max(compute_correlations(h_cur)))
    t_start = time.time()

    for trial in range(n_trials):
        if time.time() - t_start > time_budget:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or abs(h_cur[i] - h_cur[j]) < 1e-10:
            continue

        h_test = h_cur.copy()
        h_test[i], h_test[j] = h_cur[j], h_cur[i]
        new_max = float(np.max(compute_correlations(h_test)))

        if new_max < best_score - 1e-12:
            h_cur = h_test
            best_score = new_max
            best_h = h_cur.copy()

    return best_h


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj_func, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    if res.x is not None:
        return res.x
    return None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_sum(init, n / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)

    if time_left() > 80:
        h_cd = coordinate_descent(best_h, n_sweeps=20000, time_budget=min(70, time_left() - 20))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score, best_h = s, h_cd

    diverse_inits = []
    alphas = [(1.0 + np.sqrt(5.0)) / 2.0, (1.0 + np.sqrt(3.0)) / 2.0, np.e, np.sqrt(2.0), np.pi / 2.0]
    for alpha in alphas:
        h_b = beatty_init(n, alpha)
        h_b = project_sum(h_b, n / 2.0)
        diverse_inits.append(h_b)
    h_tm = thue_morse_init(n)
    h_tm = project_sum(h_tm, n / 2.0)
    diverse_inits.append(h_tm)
    for s_id in range(6):
        h_rb = random_binary_init(n, seed + 100 + s_id)
        h_rb = project_sum(h_rb, n / 2.0)
        diverse_inits.append(h_rb)
    for s_id in range(4):
        h_sr = smooth_random_init(n, seed + 300 + s_id)
        h_sr = project_sum(h_sr, n / 2.0)
        diverse_inits.append(h_sr)

    n_div = len(diverse_inits)
    tb_each = min(12.0, (time_left() - 80) / max(n_div, 1))
    for h_d in diverse_inits:
        if time_left() < 80:
            break
        h_cd = coordinate_descent(h_d, n_sweeps=5000, time_budget=tb_each)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score, best_h = s, h_cd

    if time_left() > 120:
        h_sa = simulated_annealing_cd(best_h, n_sweeps=300, time_budget=min(80, time_left() - 30), seed=seed)
        s = evaluate_score(h_sa)
        if s < best_score:
            best_score, best_h = s, h_sa
        if time_left() > 30:
            h_cd = coordinate_descent(best_h, n_sweeps=10000, time_budget=min(25, time_left() - 15))
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score, best_h = s, h_cd

    if time_left() > 60:
        h_ps = pair_swap_search(best_h, n_trials=80000, time_budget=min(35, time_left() - 20), seed=seed + 50)
        s = evaluate_score(h_ps)
        if s < best_score:
            best_score, best_h = s, h_ps
        if time_left() > 20:
            h_cd = coordinate_descent(best_h, n_sweeps=5000, time_budget=min(15, time_left() - 10))
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score, best_h = s, h_cd

    if time_left() > 100:
        n2 = 2 * n
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n2 / 2.0)
        tb_hi = min(50, time_left() - 40)
        if tb_hi > 10:
            h_cd2 = coordinate_descent(h_interp, n_sweeps=10000, time_budget=tb_hi)
            s2 = evaluate_score(h_cd2)
            if s2 < best_score:
                best_score = s2
                best_h = h_cd2

    if time_left() > 60:
        for sa_seed in [seed + 10, seed + 20, seed + 30]:
            if time_left() < 30:
                break
            h_sa = simulated_annealing_cd(best_h, n_sweeps=200, time_budget=min(20, time_left() - 10), seed=sa_seed)
            s = evaluate_score(h_sa)
            if s < best_score:
                best_score, best_h = s, h_cd = s, h_sa
                best_score, best_h = s, h_sa
            if s < best_score:
                best_score, best_h = s, h_sa
                if time_left() > 15:
                    h_cd = coordinate_descent(best_h, n_sweeps=5000, time_budget=min(10, time_left() - 5))
                    s = evaluate_score(h_cd)
                    if s < best_score:
                        best_score, best_h = s, h_cd

    if time_left() > 40:
        n_cur = len(best_h)
        bounds = [(0.0, 1.0)] * n_cur
        for alpha in [200, 1000, 5000, 20000]:
            if time_left() < 12:
                break
            maxiter = max(50, 150000 // n_cur)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score, best_h = s, h_new

    if time_left() > 20:
        h_cd = coordinate_descent(best_h, n_sweeps=50000, time_budget=max(10, time_left() - 5))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score, best_h = s, h_cd

    if time_left() > 15:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [50000, 200000]:
            if time_left() < 8:
                break
            res_x = slsqp_optimize(best_h, alpha, 300, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score, best_h = s, h_new

    if time_left() > 10:
        h_cd = coordinate_descent(best_h, n_sweeps=100000, time_budget=max(5, time_left() - 2))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score, best_h = s, h_cd

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
