def _projected_gradient(h, n, target_sum, alpha, deadline, max_iter=5000):
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Compute corr and weights
        corr = np.correlate(h, 1-h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Compute gradient (weighted sum of lag gradients)
        grad = np.zeros(n)
        for j in range(len(corr)):
            if weights[j] < 1e-8:
                continue
            lag = j - (n - 1)
            # Add weighted gradient for this lag
            if lag >= 0:
                # d c_lag / d x[i] = (1-x[i-lag]) * [lag <= i <= n-1] - x[i+lag] * [0 <= i <= n-1-lag]
                if lag == 0:
                    grad[:] += weights[j] * (1 - 2*h)
                else:
                    grad[lag:] += weights[j] * (1 - h[:n-lag])
                    grad[:n-lag] -= weights[j] * h[lag:]
            else:
                L = -lag
                # d c_{-L} / d x[i] = (1-x[i+L]) * [0 <= i <= n-1-L] - x[i-L] * [L <= i <= n-1]
                grad[:n-L] += weights[j] * (1 - h[L:])
                grad[L:] -= weights[j] * h[:n-L]
        
        grad *= (2.0 / n)  # multiply by dx
        
        # Line search / adaptive step
        lr = 0.1  # Could use backtracking
        new_h = _fix_sum(h - lr * grad, target_sum)
        
        # Check improvement
        new_val = _smoothed_value(new_h, alpha)
        old_val = _smoothed_value(h, alpha)

            lr = min(lr * 1.1, 1.0)  # increase step

            lr *= 0.5
            if lr < 1e-8:
                break
            new_h = _fix_sum(h - lr * grad, target_sum)
            h = new_h
        
    return h

def _run_slsqp(h0, n, target_sum, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 0)
        e = np.exp(shifted)
        return (m + np.log(e.sum()) / alpha) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 0)
        e = np.exp(shifted)
        w = e / e.sum()
        
        g = np.zeros(n)
        for j in range(len(corr)):
            if w[j] < 1e-10:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g[:] += w[j] * (1 - 2*x)
            elif lag > 0:
                g[lag:] += w[j] * (1 - x[:n-lag])
                g[:n-lag] -= w[j] * x[lag:]
            else:
                L = -lag
                g[:n-L] += w[j] * (1 - x[L:])
                g[L:] -= w[j] * x[:n-L]
        g *= dx
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except:
        return h0.copy()

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smoothed_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.minimum(shifted, 0.0)
    shifted = np.maximum(shifted, -500.0)
    e = np.exp(shifted)
    return (m + np.log(e.sum()) / alpha) * (2.0 / len(x))

def _smoothed_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.minimum(shifted, 0.0)
    shifted = np.maximum(shifted, -500.0)
    e = np.exp(shifted)
    w = e / e.sum()
    
    g = np.zeros(n)
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g[:] += wj * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wj * (1.0 - x[:n-lag])
            g[:n-lag] -= wj * x[lag:]
        else:
            L = -lag
            g[:n-L] += wj * (1.0 - x[L:])
            g[L:] -= wj * x[:n-L]
    g *= dx
    return g

    dx = 2.0 / n
    
    def obj(x):
        return _smoothed_obj(x, alpha)
    
    def grad(x):
        return _smoothed_grad(x, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

        h = _fix_sum(h, target_sum)

Let me also think about whether there's a smarter way to compute the weighted gradient.

The gradient is:
g[i] = dx * sum_j w_j * (d c_j / d x[i])

For lag k (where j = k + n - 1):
- If k > 0: d c_k / d x[i] = (1-x[i-k]) * [k ≤ i ≤ n-1] - x[i+k] * [0 ≤ i ≤ n-1-k]
- If k = 0: d c_0 / d x[i] = 1 - 2*x[i]
- If k < 0 (L = -k): d c_{-L} / d x[i] = (1-x[i+L]) * [0 ≤ i ≤ n-1-L] - x[i-L] * [L ≤ i ≤ n-1]

We can rewrite this as:
g[i] = dx * (sum_{k>0, k≤i} w_k*(1-x[i-k]) + sum_{k>0, i≤n-1-k} (-w_k*x[i+k]) 
           + w_0*(1-2*x[i])
           + sum_{L>0, i≤n-1-L} w_{-L}*(1-x[i+L]) + sum_{L>0, i≥L} (-w_{-L}*x[i-L]))

This can be computed using convolutions! Specifically:
- sum_{k>0, k≤i} w_k * (1-x[i-k]) = (w_pos * (1-x))[i] where w_pos[k] = w_{n-1+k} for k=1..n-1
- sum_{k>0, i≤n-1-k} w_k * x[i+k] = (w_pos * x)[i] where w_pos is reversed

**Alternative: Direct "minimax" approach**

Instead of smoothing the max, we could:
1. Find the current worst lag k*
2. Try to minimize just that lag's contribution
3. Check if other lags get worse
4. If so, use a "balancing" approach

This is like the "cutting plane" method or the "Gilbert algorithm" for convex optimization. But our problem isn't convex (the constraint sum(h) = n/2 makes it linear, but the objective is non-convex due to the product h(x)*(1-h(x+k))).

Actually, is the problem convex? Let's see:
- The feasible set {h: 0≤h≤1, sum(h)=n/2} is convex.
- The objective is max_k f_k(h) where f_k(h) = sum_i h[i]*(1-h[i+k]) (for valid i).

Each f_k is a quadratic function of h. Is it convex or concave?
f_k(h) = sum_i h[i] - sum_i h[i]*h[i+k]

The first term is linear. The second term is -sum_i h[i]*h[i+k], which is a quadratic form. The Hessian of -h[i]*h[i+k] has off-diagonal entries at (i, i+k) and (i+k, i). This is not positive semidefinite in general, so f_k is not convex.

So the problem is non-convex, and we're looking for a local minimum. The smoothing approach is reasonable.

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smoothed_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return m * (2.0 / len(x))
    return (m + np.log(s) / alpha) * (2.0 / len(x))

def _smoothed_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)

        # Degenerate case: just use the argmax

        lag = k_best - (n - 1)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * x
        elif lag > 0:
            g[lag:] = (1.0 - x[:n-lag])
            g[:n-lag] = -x[lag:]
        else:
            L = -lag
            g[:n-L] = (1.0 - x[L:])
            g[L:] = -x[:n-L]

        lag = j - (n - 1)
        if lag == 0:
            g[:] += wj * (1.0 - 2.0 * x)
        elif lag > 0:

            g[:n-lag] -= wj * x[lag:]
        else:
            L = -lag
            g[:n-L] += wj * (1.0 - x[L:])
            g[L:] -= wj * x[:n-L]
    return g * dx

def _slsqp_refine(h0, n, target_sum, alpha, maxiter, ftol):

             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smoothed_obj, h0, args=(alpha,), jac=_smoothed_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()
