```python
def compute_corr_and_grad(h):
    n = len(h)
    # Compute all correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute gradient
    # corr[m] = sum_j h[j] * (1-h[j+m-n+1])
    # d corr[m] / d h[i] = (1-h[i+m-n+1]) - h[i-(m+1-n+1)] = (1-h[i+s]) - h[i-s]
    # where s = m - n + 1
    
    n2 = len(corr)  # 2n - 1
    grad = np.zeros((n2, n))
    for m in range(n2):
        s = m - n + 1
        # d corr[m] / d h[i] = (1-h[i+s]) - h[i-s] for valid indices
        # h[i+s] is valid when 0 <= i+s < n, i.e., -s <= i < n-s
        # h[i-s] is valid when 0 <= i-s < n, i.e., s <= i < n+s
        
        # First term: (1 - h[i+s]) for i in [max(0, -s), min(n, n-s))
        # Second term: -h[i-s] for i in [max(0, s), min(n, n+s))
        
        i_start1 = max(0, -s)
        i_end1 = min(n, n - s)
        if i_start1 < i_end1:
            idx = np.arange(i_start1, i_end1)
            grad[m, idx] += (1.0 - h[idx + s])
        
        i_start2 = max(0, s)
        i_end2 = min(n, n + s)
        if i_start2 < i_end2:
            idx = np.arange(i_start2, i_end2)
            grad[m, idx] -= h[idx - s]
    
    return corr, grad
```

```python
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init.copy(), target_init)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP refinement at initial resolution
    _g_n = n_init
    _g_target = target_init
    if tleft() > 120:
        h_refined = run_slsqp_round(best_h, n_init, target_init,
                                     [10, 30, 80, 200, 500], 800, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 2: Grid refinement to n=600
    if tleft() > 200:
        n_try = 600
        target_try = n_try / 2.0
        _g_n = n_try
        _g_target = target_try
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_refined = run_slsqp_round(h_interp, n_try, target_try,
                                     [20, 50, 150, 400], 500, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 3: Grid refinement to n=800
    if tleft() > 200:
        n_try = 800
        target_try = n_try / 2.0
        _g_n = n_try
        _g_target = target_try
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_refined = run_slsqp_round(h_interp, n_try, target_try,
                                     [20, 60, 200, 500], 400, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 4: Nesterov subgradient at best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 150:
        n_iter_ne = min(5000, int(tleft() * 50))
        h_ne, sc_ne = nesterov_subgrad(best_h, n_iter_ne, target_use,
                                        lr_init=0.5 * n_use, mu=0.01, top_k=60)
        if sc_ne < best_score:
            best_score = sc_ne
            best_h = h_ne

    # Phase 5: SLSQP polish after Nesterov
    _g_n = n_use
    _g_target = target_use
    if tleft() > 80:
        h_refined = run_slsqp_round(best_h, n_use, target_use,
                                     [200, 500, 1000], 400, 1e-16)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 6: Try n=1000 if time permits
    if tleft() > 200:
        n_big = 1000
        target_big = n_big / 2.0
        _g_n = n_big
        _g_target = target_big
        h_interp = interpolate_to(best_h, n_big, target_big)
        h_refined = run_slsqp_round(h_interp, n_big, target_big,
                                     [20, 80, 300], 300, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined
        if tleft() > 80:
            n_use = len(best_h)
            target_use = n_use / 2.0
            if tleft() > 100:
                n_iter_ne2 = min(3000, int(tleft() * 40))
                h_ne2, sc_ne2 = nesterov_subgrad(best_h, n_iter_ne2, target_use,
                                                  lr_init=0.5 * n_use, mu=0.01, top_k=60)
                if sc_ne2 < best_score:
                    best_score = sc_ne2
                    best_h = h_ne2

    # Phase 7: Final SLSQP polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    _g_n = n_final
    _g_target = target_final
    if tleft() > 30:
        h_final_try = run_slsqp_round(best_h, n_final, target_final,
                                       [500, 1000, 2000], 300, 1e-16)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 8: Multi-start fine perturbation
    n_use = len(best_h)
    target_use = n_use / 2.0
    _g_n = n_use
    _g_target = target_use
    bounds_use = [(0.0, 1.0)] * n_use
    cons_use = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    attempt = 0
    while tleft() > 30 and attempt < 15:
        attempt += 1
        noise = rng.choice([0.005, 0.01, 0.015, 0.02])
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        alpha = rng.choice([300, 600, 1000])
        try:
            res = minimize(smooth_obj, h_pert, args=(alpha,),
                           method='SLSQP', bounds=bounds_use, constraints=cons_use,
                           options={'maxiter': 300, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_use)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
