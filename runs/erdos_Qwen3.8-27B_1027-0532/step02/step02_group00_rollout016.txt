```python
    # where s_k = k - (n-1)

        for k in range(2*n-1):

            # Term 1: (1-h[i+s]) if 0 <= i+s < n

            # Term 2: -h[i-s] if 0 <= i-s < n
            if 0 <= i-s < n:
                val -= w[k] * h[i-s]
        grad[i] = val
    
    return F, grad
```

Actually, let me think about this differently. I can compute the gradient using convolutions:

The term `sum_k w[k] * (1 - h[i + k - (n-1)]) * [0 <= i+k-(n-1) < n]`

Let me define `w_ext` as a length-2n-1 array (which w already is). Then:
- For fixed i, sum over k where 0 <= i+k-(n-1) < n, i.e., n-1-i <= k <= 2n-2-i
  - `= sum_{k=n-1-i}^{2n-2-i} w[k] - sum_{k=n-1-i}^{2n-2-i} w[k] * h[i+k-(n-1)]`
  - The first part is a partial sum of w.
  - The second part: let j = i+k-(n-1), so k = j-i+n-1. As k goes from n-1-i to 2n-2-i, j goes from 0 to n-1.
  - `= sum_{j=0}^{n-1} w[j-i+n-1] * h[j]`
  - This is `(w_shifted * h)[i]` where `w_shifted` is w shifted by n-1. Actually it's a cross-correlation.

Let me define `w2[j] = w[j + n - 1]` for j in [-n+1, n-1] (zero-padded). Then:
`sum_j h[j] * w[j-i+n-1] = sum_j h[j] * w2[j-i]` which is the cross-correlation of h and w2 evaluated at i.

Similarly for the third term:
`sum_k w[k] * h[i-k+n-1] * [0 <= i-k+n-1 < n]`
Let j = i-k+n-1, so k = i+n-1-j.
`= sum_j h[j] * w[i+n-1-j] * [0<=j<n]`
`= sum_j h[j] * w2[n-1-i+j]` ... hmm this is getting confusing.

OK let me now design the full algorithm:

1. Start from initial_h_values (n=400)
2. Do a quick DE on small grids (20-50 points) for diversity - this gives a few alternative starting points
3. For each starting point:
   a. Interpolate to n=400
   b. Run L-BFGS-B with smooth objective, alpha schedule [5, 15, 50, 150, 500]
   c. Keep the best
4. Try to escape local optima:
   a. Perturb and re-optimize
   b. Try swapping parts of the solution
5. Grid refinement: interpolate to n=600, 800 and optimize
6. Return best

Let me also think about an alternative: instead of log-sum-exp, use the "max trick" more directly. At the optimum, the max is achieved at (potentially) multiple k values. We could try to identify which k values are "active" and optimize only those.

Actually, here's another idea: **projected subgradient method**. The function f(h) = max_k c_k * dx is convex in h (since each c_k is linear in h... wait, is it? c_k = sum_j h[j](1-h[j+s]) = sum_j h[j] - sum_j h[j]h[j+s]. The second term is quadratic, so c_k is NOT linear in h. It's quadratic (concave in h since it has -h*h terms). So f is a max of concave functions, which is NOT convex.

So the problem is NOT convex. This makes it harder. The log-sum-exp smoothing still helps by making it differentiable, but we can't guarantee global optimality.

Given the non-convexity, multi-start is essential. Let me focus on:
1. Good diverse starting points
2. Efficient local optimization
3. Smart perturbations that can escape local optima

For perturbations, instead of random Gaussian noise, I could try:
- Swapping segments
- Shifting the function by a small amount
- Smoothing edges
- Adding/removing "bumps"

Let me write the final implementation:

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n-1-i)
        khi = min(2*n-2, 2*n-2-i)
        term_a = np.sum(w[klo:khi+1])
        j_idx = np.arange(klo, khi+1) - (n-1) + i
        term_b = np.sum(w[klo:khi+1] * h[j_idx])
        klo_c = max(0, i)
        khi_c = min(2*n-2, i+n-1)
        if klo_c <= khi_c:
            j_idx_c = i - np.arange(klo_c, khi_c+1) + (n-1)
            term_c = np.sum(w[klo_c:khi_c+1] * h[j_idx_c])
        else:
            term_c = 0.0
        grad[i] = term_a - term_b - term_c
    return F, grad

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    
    best_h = project_feasible(h_init.copy(), target_init)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick DE on small grids for diverse starting points
    starting_points = [best_h.copy()]
    for n_coarse in [20, 30, 40, 55]:
        if tleft() < 60:
            break
        target_c = n_coarse / 2.0
        bounds_de = [(0.0, 1.0)] * n_coarse
        def de_obj(x):
            h = project_feasible(x, target_c)
            return compute_c5(h)
        popsize = max(8, min(15, 300 // n_coarse))
        maxiter_de = max(50, min(200, 4000 // n_coarse))
        try:
            result = differential_evolution(
                de_obj, bounds_de,
                maxiter=maxiter_de, popsize=popsize,
                tol=1e-8, seed=seed + n_coarse,
                mutation=(0.5, 1.0), recombination=0.7,
                polish=True, init='sobol'
            )
            h_coarse = project_feasible(result.x, target_c)
            h_interp = interpolate_to(h_coarse, n_init, target_init)
            starting_points.append(h_interp)
        except Exception:
            pass
    
    # Phase 2: Local optimization from each starting point
    def lbfgs_refine(h_start, n, target, alpha_schedule, maxiter_per_stage):
        h_cur = project_feasible(h_start.copy(), target)
        bounds = [(0.0, 1.0)] * n
        for alpha in alpha_schedule:
            if tleft() < 20:
                break
            def obj(x):
                F, _ = smooth_obj_and_grad(x, alpha)
                return F
            def jac(x):
                _, g = smooth_obj_and_grad(x, alpha)
                return g
            try:
                res = minimize(obj, h_cur, jac=jac, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': maxiter_per_stage,
                                                    'ftol': 1e-15, 'gtol': 1e-8})
                h_new = project_feasible(res.x, target)
                if compute_c5(h_new) <= compute_c5(h_cur):
                    h_cur = h_new
            except Exception:
                pass
        return h_cur

        if tleft() < 60:
            break
        h_refined = lbfgs_refine(sp, n_init, target_init, [5, 15, 50, 150, 500, 1000], 200)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    pert_types = {
        'gauss': lambda h, rng, amp: project_feasible(h + rng.randn(len(h)) * amp, target_init),
        'segment_swap': None,  # will implement below
        'shift': None,
    }
    
    for attempt in range(20):
        if tleft() < 40:
            break
        ptype = attempt % 3
        if ptype == 0:  # Gaussian noise
            amp = 0.02 * (1 + attempt * 0.3)
            h_pert = project_feasible(best_h + rng.randn(n_init) * amp, target_init)
        elif ptype == 1:  # Segment swap
            h_pert = best_h.copy()
            a = rng.randint(0, n_init - 40)
            b = a + rng.randint(10, 40)
            c = rng.randint(0, n_init - (b - a))
            if c + (b-a) <= n_init or c < a:  # avoid overlap issues
                h_pert[a:b] = best_h[c:c+(b-a)]
                h_pert = project_feasible(h_pert, target_init)
            else:
                h_pert = project_feasible(best_h + rng.randn(n_init) * 0.05, target_init)
        else:  # Circular shift
            shift = rng.randint(1, n_init // 4)
            h_pert = np.roll(best_h, shift)
            h_pert = project_feasible(h_pert, target_init)
        
        h_refined = lbfgs_refine(h_pert, n_init, target_init, [15, 50, 150, 500], 150)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined
    
    # Phase 4: Grid refinement
    for n_try in [500, 600, 700, 800]:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_refined = lbfgs_refine(h_interp, n_try, target_try, [10, 30, 100, 300, 800], 200)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined
    
    # Phase 5: Final SLSQP polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    
    def slsqp_polish(h_start, n, target, alphas, maxiter):
        h_cur = project_feasible(h_start.copy(), target)
        bounds_s = [(0.0, 1.0)] * n
        for alpha in alphas:
            if tleft() < 20:
                break
            def obj_s(x):
                F, _ = smooth_obj_and_grad(x, alpha)
                return F
            def jac_s(x):
                _, g = smooth_obj_and_grad(x, alpha)
                return g
            def cons_fun(x):
                return np.sum(x) - target
            def cons_jac(x):
                return np.ones(n)
            try:
                res = minimize(obj_s, h_cur, jac=jac_s, method='SLSQP',
                             bounds=bounds_s,
                             constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                             options={'maxiter': maxiter, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                if compute_c5(h_new) <= compute_c5(h_cur):
                    h_cur = h_new
            except Exception:
                pass
        return h_cur
    
    if tleft() > 60:
        h_final = slsqp_polish(best_h, n_use, target_use, [500, 1500], 300)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final
    
    # Phase 6: Final fine L-BFGS-B with very high alpha
    if tleft() > 40:
        h_final2 = lbfgs_refine(best_h, n_use, target_use, [1000, 3000], 300)
        sc = compute_c5(h_final2)
        if sc < best_score:
            best_score = sc
```

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        term_a = np.sum(w[klo:khi + 1])
        j_idx = np.arange(klo, khi + 1) - (n - 1) + i
        term_b = np.sum(w[klo:khi + 1] * h[j_idx])
        klo_c = max(0, i)
        khi_c = min(2 * n - 2, i + n - 1)
        if klo_c <= khi_c:
            j_idx_c = i - np.arange(klo_c, khi_c + 1) + (n - 1)
            term_c = np.sum(w[klo_c:khi_c + 1] * h[j_idx_c])
        else:
            term_c = 0.0
        grad[i] = term_a - term_b - term_c
    return F, grad


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lbfgs_stage(h_start, n, target, alpha, maxiter, ftol):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                               'gtol': 1e-7})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur


def run_full_refine(h_start, n, target, alpha_schedule, maxiter_per_stage, ftol, time_limit_fn):
    h_cur = project_feasible(h_start.copy(), target)
    current_score = compute_c5(h_cur)
    for alpha in alpha_schedule:
        if time_limit_fn() < 25:
            break
        h_new = lbfgs_stage(h_cur, n, target, alpha, maxiter_per_stage, ftol)
        sc = compute_c5(h_new)
        if sc <= current_score:
            h_cur = h_new
            current_score = sc
    return h_cur, current_score


def slsqp_stage(h_start, n, target, alpha, maxiter):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur


def de_objective_factory(target_sum):
    def _obj(x):
        h = project_feasible(x, target_sum)
        return compute_c5(h)
    return _obj
```
