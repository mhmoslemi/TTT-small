```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr)) * dx

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def project_clip(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-14:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_corr_full(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    ex = np.exp(alpha * (corr - m))
    p = ex / float(np.sum(ex))
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[:N - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:N - k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:N - kp])
            grad[:N - kp] -= w * h[kp:]
    return grad

def antipal_odd(g):
    half = len(g)
    h = np.empty(2 * half + 1)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def antipal_even(g):
    half = len(g)
    h = np.empty(2 * half)
    h[:half] = g
    h[half:] = 1.0 - g[::-1]
    return h

def make_anti_odd_g(h):
    n = len(h)
    half = n // 2
    g = (h[:half] + 1.0 - h[half + 1:][::-1]) / 2.0
    return np.clip(g, 0.0, 1.0)

def make_anti_even_g(h):
    n = len(h)
    half = n // 2
    g = (h[:half] + 1.0 - h[half:][::-1]) / 2.0
    return np.clip(g, 0.0, 1.0)

def _ap_fun_odd(g, alpha):
    h = antipal_odd(g)
    return smooth_max(compute_correlations(h), alpha)

def _ap_jac_odd(g, alpha):
    h = antipal_odd(g)
    gh = grad_corr_full(h, alpha)
    half = len(g)
    gg = gh[:half].copy()
    partners = 2 * half - np.arange(half)
    gg -= gh[partners]
    return gg

def _ap_fun_even(g, alpha):
    h = antipal_even(g)
    return smooth_max(compute_correlations(h), alpha)

def _ap_jac_even(g, alpha):
    h = antipal_even(g)
    gh = grad_corr_full(h, alpha)
    half = len(g)
    gg = gh[:half].copy()
    partners = 2 * half - 1 - np.arange(half)
    gg -= gh[partners]
    return gg

def lbfgsb_opt(g, is_odd, alpha, maxiter=10000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * len(g)
    if is_odd:
        res = minimize(_ap_fun_odd, g, jac=_ap_jac_odd, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'iprint': 0})
    else:
        res = minimize(_ap_fun_even, g, jac=_ap_jac_even, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'iprint': 0})
    return res.x

def topk_grad(g, top_indices, N_full, is_odd, eps=1e-8):
    """Compute value and gradient of max over top_indices of correlations"""
    if is_odd:
        h = antipal_odd(g)
    else:
        h = antipal_even(g)
    corr = compute_correlations(h)
    csub = corr[top_indices]
    max_val = float(np.max(csub))
    active = csub >= max_val - eps
    n_act = int(np.sum(active))
    
    grad_h = np.zeros(len(h))
    for idx in range(len(top_indices)):
        if not active[idx]:
            continue
        j = int(top_indices[idx])
        k = j - (len(h) - 1)
        if k >= 0:
            grad_h[:len(h) - k] += (1.0 - h[k:]) / n_act
            grad_h[k:] -= h[:len(h) - k] / n_act
        else:
            kp = -k
            grad_h[kp:] += (1.0 - h[:len(h) - kp]) / n_act
            grad_h[:len(h) - kp] -= h[kp:] / n_act
    
    half = len(g)
    gg = grad_h[:half].copy()
    if is_odd:
        partners = 2 * half - np.arange(half)
    else:
        partners = 2 * half - 1 - np.arange(half)
    gg -= grad_h[partners]
    return max_val, gg

def _topk_fun(g, top_indices, N_full, is_odd):
    val, _ = topk_grad(g, top_indices, N_full, is_odd)
    return val

def _topk_jac(g, top_indices, N_full, is_odd):
    _, grad = topk_grad(g, top_indices, N_full, is_odd)
    return grad

def interpolate_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
```

```python
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_clip(init, n_init / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)

    def _elapsed():
        return time.time() - t0

    # Phase 1: Anti-palindrome L-BFGS-B at multiple resolutions
    for n_res in [425, 625, 825]:
        if _elapsed() > budget_s - 250:
            break
        if n_res % 2 == 0:
            n_res += 1

        h_res = interp_to_n(best_h, n_res)
        g = make_anti_odd_g(h_res)
        bounds = [(0.0, 1.0)] * len(g)

        for alpha in [200, 500, 2000, 8000, 30000]:
            if _elapsed() > budget_s - 200:
                break
            maxiter = max(500, min(20000, 4000000 // n_res))
            res = minimize(_ap_fun, g, jac=_ap_jac, args=(alpha,),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                g = np.clip(res.x, 0.0, 1.0)
                h_new = antipal_odd(g)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Ensure odd length for anti-palindrome
    if len(best_h) % 2 == 0:
        best_h = interp_to_n(best_h, len(best_h) + 1)
        best_score = evaluate_score(best_h)

    g_best = make_anti_odd_g(best_h)
    bounds_g = [(0.0, 1.0)] * len(g_best)

    # Phase 2: Top-k iterative focused optimization
    for iteration in range(15):
        if _elapsed() > budget_s - 150:
            break
        h_curr = antipal_odd(g_best)
        corr_curr = compute_correlations(h_curr)
        top_k = 8
        top_idx = np.sort(np.argsort(corr_curr)[-top_k:])

        res = minimize(_topk_val, g_best, jac=_topk_grad, args=(top_idx,),
                       method='L-BFGS-B', bounds=bounds_g,
                       options={'maxiter': 8000, 'ftol': 1e-18,
                                'gtol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            g_best = np.clip(res.x, 0.0, 1.0)
            h_new = antipal_odd(g_best)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 3: Stochastic subgradient with top-shift selection
    if _elapsed() < budget_s - 150:
        n_sub = 500
        g_curr = g_best.copy()
        best_g_sub = g_best.copy()
        best_s_sub = best_score
        lr = 0.005
        h_t = antipal_odd(g_curr)
        c_t = compute_correlations(h_t)
        top_shifts = np.argsort(c_t)[-25:]

        for step in range(n_sub):
            if _elapsed() > budget_s - 100:
                break
            if step % 100 == 0:
                h_t = antipal_odd(g_curr)
                c_t = compute_correlations(h_t)
                top_shifts = np.argsort(c_t)[-25:]
                lr = max(0.001, 0.005 * (1.0 - step / float(n_sub)))

            j = top_shifts[rng.randint(len(top_shifts))]
            N_cur = len(h_t)
            k = int(j) - (N_cur - 1)
            h_t = antipal_odd(g_curr)

            gh = np.zeros(N_cur)
            if k >= 0:
                gh[:N_cur - k] += (1.0 - h_t[k:])
                gh[k:] -= h_t[:N_cur - k]
            else:
                kp = -k
                gh[kp:] += (1.0 - h_t[:N_cur - kp])
                gh[:N_cur - kp] -= h_t[kp:]

            half = len(g_curr)
            gg = gh[:half].copy()
            partners = 2 * half - np.arange(half)
            gg -= gh[partners]

            g_curr = np.clip(g_curr - lr * gg, 0.0, 1.0)

            if step % 50 == 0:
                s_t = evaluate_score(antipal_odd(g_curr))
                if s_t < best_s_sub:
                    best_s_sub = s_t
                    best_g_sub = g_curr.copy()

        if best_s_sub < best_score:
            best_score = best_s_sub
            best_h = antipal_odd(best_g_sub)
            g_best = best_g_sub.copy()
            bounds_g = [(0.0, 1.0)] * len(g_best)

    # Phase 4: Random restarts
    n_restarts = 30
    for r in range(n_restarts):
        if _elapsed() > budget_s - 100:
            break
        scale = 0.01 + 0.04 * (r / float(n_restarts))
        g_pert = g_best + rng.randn(len(g_best)) * scale
        g_pert = np.clip(g_pert, 0.0, 1.0)

        alpha_r = int(rng.choice([500, 2000, 8000, 30000]))
        maxiter_r = max(500, 3000000 // len(g_pert))
        bounds_r = [(0.0, 1.0)] * len(g_pert)
        res = minimize(_ap_fun, g_pert, jac=_ap_jac, args=(alpha_r,),
                       method='L-BFGS-B', bounds=bounds_r,
                       options={'maxiter': maxiter_r, 'ftol': 1e-18,
                                'gtol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            g_new = np.clip(res.x, 0.0, 1.0)
            h_new = antipal_odd(g_new)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                g_best = g_new.copy()
                bounds_g = [(0.0, 1.0)] * len(g_best)

    # Phase 5: Very high alpha final polish
    if _elapsed() < budget_s - 30:
        g_pol = make_anti_odd_g(best_h)
        bounds_pol = [(0.0, 1.0)] * len(g_pol)
        for alpha in [50000, 200000]:
            if _elapsed() > budget_s - 10:
                break
            maxiter = max(3000, 8000000 // len(g_pol))
            res = minimize(_ap_fun, g_pol, jac=_ap_jac, args=(alpha,),
                           method='L-BFGS-B', bounds=bounds_pol,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-15, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                g_pol = np.clip(res.x, 0.0, 1.0)
                h_new = antipal_odd(g_pol)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 6: Full-space SLSQP from best (remove anti-palindrome symmetry)
    if _elapsed() < budget_s - 40:
        h_try = best_h.copy()
        n_try = len(h_try)
        target = n_try / 2.0
        h_try = project_clip(h_try, target)
        for alpha in [5000, 20000]:
            if _elapsed() > budget_s - 15:
                break
            h_opt, s = _slsqp_full(h_try, alpha, target, 3000)
            if s < best_score:
                best_score = s
                best_h = h_opt
                h_try = h_opt.copy()

    best_h = project_clip(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
```
