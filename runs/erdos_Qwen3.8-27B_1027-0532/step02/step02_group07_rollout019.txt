            current_max = new_max

                h[j] = t_opt
                corr += change * slopes
                if abs(change) > max_change:
                    max_change = abs(change)
        if sweep % recompute_every == recompute_every - 1:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    h = project_sum(h, n / 2.0)
    return h

def swap_local_search(h, n_trials=200000, time_budget=30.0, seed=42):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    current_max = np.max(corr)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    for _ in range(n_trials):
        if time.time() - t0 > time_budget:
            break
        i, j = rng.choice(n, 2, replace=False)
        delta = 0.005 + 0.04 * rng.random()
        if h[i] < delta:
            continue
        new_j = h[j] + delta
        if new_j > 1.0:
            if h[j] >= 1.0 - 1e-12:
                continue
            delta = 1.0 - h[j]
            if delta < 1e-6:
                continue
        
        slopes_i = get_slopes(h, i)
        slopes_j = get_slopes(h, j)
        new_corr = corr - delta * slopes_i + delta * slopes_j
        new_max = np.max(new_corr)
        
        if new_max < current_max - 1e-14:
            h[i] -= delta
            h[j] = h[j] + delta
            corr = new_corr
            current_max = new_max
    
    return h

def shift_search(h, shifts, n_cd_sweeps=100, time_per_shift=15.0):
    n = len(h)
    best_h = h.copy()
    best_score = evaluate_score(h)
    t0 = time.time()
    
    for s in shifts:
        if time.time() - t0 > time_per_shift * len(shifts):
            break
        h_shifted = np.roll(h, s)
        h_cd = coordinate_descent(h_shifted, n_sweeps=n_cd_sweeps, time_budget=2.0)
        s_new = evaluate_score(h_cd)
        if s_new < best_score:
            best_score = s_new
            best_h = h_cd
    
    return best_h, best_score

    n = len(h)
    best = h.copy()
    best_score = evaluate_score(h)
    
    for alpha in alphas:
        def obj(x):
            corr = compute_correlations(x)
            return smooth_max(corr, alpha)
        
        def obj_jac(x):

            # Skip Jacobian, let SLSQP use finite differences
            return None
        
        def con_eq(x):
            return np.sum(x) - n / 2.0
        
        def con_jac(x):
            return np.ones(n)
        
        maxiter = max(50, maxiter_base // n)
        res = minimize(
            obj, best, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
        )
        if res.x is not None:
            h_new = project_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best = h_new
    
    return best, best_score
