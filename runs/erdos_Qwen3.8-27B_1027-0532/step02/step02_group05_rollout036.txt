Wait, let me be more careful.

corr_k = Σ_i h[i](1-h[i+k])

The terms that change:
- Term i=j: h[j](1-h[j+k]) changes to x(1-h[j+k])
  Δ = (x - g[j])(1-h[j+k])
- Term i=2m-j: h[2m-j](1-h[2m-j+k]) changes to (1-x)(1-h[2m-j+k])
  Δ = (1-x - (1-g[j]))(1-h[2m-j+k]) = (g[j]-x)(1-h[2m-j+k]) = -(x-g[j])(1-h[2m-j+k])
- Term i+k=j (i.e., i=j-k): h[j-k](1-h[j]) changes to h[j-k](1-x)
  Δ = h[j-k](1-x - (1-g[j])) = h[j-k](g[j]-x) = -h[j-k](x-g[j])
- Term i+k=2m-j (i.e., i=2m-j-k): h[2m-j-k](1-h[2m-j]) changes to h[2m-j-k](1-(1-x)) = h[2m-j-k]*x
  Δ = h[2m-j-k](x - g[j])

So total change in corr_k when g[j] changes by δ = x - g[j]:
Δcorr_k = δ * [(1-h[j+k]) - (1-h[2m-j+k]) - h[j-k] + h[2m-j-k]]
= δ * [h[2m-j+k] - h[j+k] - h[j-k] + h[2m-j-k]]

(All indices are valid; if an index is out of bounds, that term doesn't contribute.)

So a_k = h[2m-j+k] - h[j+k] - h[j-k] + h[2m-j-k] (with appropriate boundary handling)
b_k = corr_k(current) - a_k * g[j]

Then find x* = argmin_{x∈[0,1]} max_k (a_k * x + b_k)

This is the same 1D convex minimization as before!

  Δcorr_k = (1-h[j+k])*δ - h[j-k]*δ = (1-h[j+k]-h[j-k])*δ
  (with bounds checking)
  
- Find h[j]* = argmin_{x∈[0,1]} max_k (a_k * x + b_k)
- After full sweep, project onto sum(h) = N/2

The projection is:
- Find λ such that sum(clip(h - λ, 0, 1)) = N/2
- Use bisection

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return float(np.max(corr) * dx)


def build_h_from_g(g):
    m = len(g)
    N = 2 * m + 1
    h = np.empty(N)
    h[:m] = g
    h[m] = 0.5
    h[m+1:] = 1.0 - g[::-1]
    return h


def find_k_min(a, b):
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) <= np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_corr_full(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def _fun_smooth(x, alpha):
    return smooth_max_val(compute_corr_full(x), alpha)


def _jac_smooth(x, alpha):
    return grad_obj(x, alpha)


def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0


def _cons_jac(x):
    return np.ones(len(x))


def slsqp_polish(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        _fun_smooth, h0, method='SLSQP', bounds=bounds,
        jac=_jac_smooth, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x


def coord_descent_anti(g, n_rounds, rng):
    """Coordinate descent with anti-palindromic constraint."""
    m = len(g)
    N = 2 * m + 1
    Nc = 2 * N - 1
    h = build_h_from_g(g)
    corr = compute_corr_full(h)
    best_score = float(np.max(corr) * (2.0 / N))
    best_g = g.copy()
    
    k_vals = np.arange(-(N-1), N)
    
    for r in range(n_rounds):
        improved = False
        order = rng.permutation(m)
        for j in order:
            jk = j + k_vals
            jmk = j - k_vals
            j2k = (N-1-j) + k_vals
            j2mk = (N-1-j) - k_vals
            
            a = np.zeros(Nc)
            valid = (jk >= 0) & (jk < N)
            if np.any(valid):
                a[valid] += (1.0 - h[jk[valid]])
            valid = (j2k >= 0) & (j2k < N)
            if np.any(valid):
                a[valid] -= (1.0 - h[j2k[valid]])
            valid = (jmk >= 0) & (jmk < N)
            if np.any(valid):
                a[valid] -= h[jmk[valid]]
            valid = (j2mk >= 0) & (j2mk < N)
            if np.any(valid):
                a[valid] += h[j2mk[valid]]
            
            b = corr - a * g[j]
            x_new = find_k_min(a, b)
            delta = x_new - g[j]
            if abs(delta) > 1e-12:
                g[j] = x_new
                h[j] = x_new
                h[N-1-j] = 1.0 - x_new
                corr += delta * a
                improved = True
        
        score = float(np.max(corr) * (2.0 / N))
        if score < best_score:
            best_score = score
            best_g = g.copy()
        
        if not improved:
            break
    
    return best_g, best_score


def coord_descent_general(h, n_rounds, rng):
    """General coordinate descent with projection."""
    N = len(h)
    Nc = 2 * N - 1
    target = N / 2.0
    corr = compute_corr_full(h)
    best_score = float(np.max(corr) * (2.0 / N))
    best_h = h.copy()
    
    k_vals = np.arange(-(N-1), N)
    
    for r in range(n_rounds):
        improved = False
        order = rng.permutation(N)
        for j in order:
            jk = j + k_vals
            jmk = j - k_vals
            
            a = np.zeros(Nc)
            valid = (jk >= 0) & (jk < N)
            if np.any(valid):
                a[valid] += (1.0 - h[jk[valid]])
            valid = (jmk >= 0) & (jmk < N)
            if np.any(valid):
                a[valid] -= h[jmk[valid]]
            
            b = corr - a * h[j]
            x_new = find_k_min(a, b)
            delta = x_new - h[j]
            if abs(delta) > 1e-12:
                h[j] = x_new
                corr += delta * a
                improved = True
        
        # Project onto sum constraint
        h = project_feasible(h, target)
        corr = compute_corr_full(h)
        score = float(np.max(corr) * (2.0 / N))
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
    
    # Get initial values
    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    target = n_init / 2.0
    init = project_feasible(init, target)
    
    best_h = init.copy()
    best_score = eval_c5(init)

    # Extract g from init (assuming it's close to anti-palindromic)
    n = n_init
    if n % 2 == 0:
        n = n + 1  # make odd
        x_old = np.linspace(0.0, 2.0, n_init, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n / 2.0)
        best_h = h_interp
        best_score = eval_c5(h_interp)

        h_new = build_h_from_g(g_best)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return float(np.max(corr) * dx)


def build_h_from_g(g):
    m = len(g)
    N = 2 * m + 1
    h = np.empty(N)
    h[:m] = g
    h[m] = 0.5
    h[m+1:] = 1.0 - g[::-1]
    return h


def find_k_min(a, b):
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) <= np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_corr_full(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def _fun_smooth(x, alpha):
    return smooth_max_val(compute_corr_full(x), alpha)


def _jac_smooth(x, alpha):
    return grad_obj(x, alpha)


def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0


def _cons_jac(x):
    return np.ones(len(x))


def slsqp_polish(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        _fun_smooth, h0, method='SLSQP', bounds=bounds,
        jac=_jac_smooth, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x


def coord_descent_anti(g, n_rounds, rng, time_limit=None, t_start=None):
    m = len(g)
    N = 2 * m + 1
    Nc = 2 * N - 1
    h = build_h_from_g(g)
    corr = compute_corr_full(h).copy()
    best_score = float(np.max(corr) * (2.0 / N))
    best_g = g.copy()
    k_vals = np.arange(-(N - 1), N)

    for r in range(n_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        order = rng.permutation(m)
        for j in order:
            jk = j + k_vals
            jmk = j - k_vals
            j2k = (N - 1 - j) + k_vals
            j2mk = (N - 1 - j) - k_vals

            a = np.zeros(Nc)
            v = (jk >= 0) & (jk < N)
            if np.any(v):
                a[v] += (1.0 - h[jk[v]])
            v = (j2k >= 0) & (j2k < N)
            if np.any(v):
                a[v] -= (1.0 - h[j2k[v]])
            v = (jmk >= 0) & (jmk < N)
            if np.any(v):
                a[v] -= h[jmk[v]]
            v = (j2mk >= 0) & (j2mk < N)
            if np.any(v):
                a[v] += h[j2mk[v]]

            b = corr - a * g[j]
            x_new = find_k_min(a, b)
            delta = x_new - g[j]
            if abs(delta) > 1e-12:
                g[j] = x_new
                h[j] = x_new
                h[N - 1 - j] = 1.0 - x_new
                corr += delta * a
                improved = True

        score = float(np.max(corr) * (2.0 / N))
        if score < best_score - 1e-15:
            best_score = score
            best_g = g.copy()

    return best_g, best_score

            v = (j2k >= 0) & (j2k < N)
            if np.any(v):
                a[v] -= (1.0 - h[j2k[v]])
            v = (jmk >= 0) & (jmk < N)
            if np.any(v):
                a[v] -= h[jmk[v]]
            v = (j2mk >= 0) & (j2mk < N)
            if np.any(v):
                a[v] += h[j2mk[v]]

            b = corr - a * g[j]
            val0 = float(np.max(b))
            val1 = float(np.max(a + b))
            if val0 < val1:
                x_new = 0.0
            else:
                x_new = 1.0
            delta = x_new - g[j]
            if abs(delta) > 1e-12:
                g[j] = x_new
                h[j] = x_new
                h[N - 1 - j] = 1.0 - x_new
                corr += delta * a
                improved = True

        score = float(np.max(corr) * (2.0 / N))
        if score < best_score - 1e-15:
            best_score = score
            best_g = g.copy()
        if not improved:
            break

    return best_g, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining_budget():
        return budget_s - elapsed()

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    target_init = n_init / 2.0
    init = project_feasible(init, target_init)

    best_h = init.copy()
    best_score = eval_c5(init)

    n = n_init
    if n % 2 == 0:
        n = n + 1
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        best_h = np.interp(x_new, x_old, best_h)
        best_h = project_feasible(best_h, n / 2.0)
        best_score = eval_c5(best_h)

    m = n // 2
    g_anti = np.zeros(m)
    for j in range(m):
        g_anti[j] = (best_h[j] + 1.0 - best_h[n - 1 - j]) / 2.0
    g_anti = np.clip(g_anti, 0.0, 1.0)

    # Phase 1: Anti-palindromic coordinate descent (main)
    if remaining_budget() > 300:
        alloc = min(400, remaining_budget() - 300)
        rounds = max(200, int(800000 / max(m, 1)))
        g_best, score_best = coord_descent_anti(g_anti, rounds, rng,
                                                 time_limit=alloc, t_start=t0)
        if score_best < best_score:
            best_score = score_best
            best_h = build_h_from_g(g_best)

    # Phase 2: 0-1 local search then continuous CD
    if remaining_budget() > 250:
        g01 = np.where(g_anti > 0.5, 1.0, 0.0)
        g01, s01 = bit_local_search(g01, 200, rng)
        g_cont, s_cont = coord_descent_anti(g01, 300, rng,
                                            time_limit=min(120, remaining_budget() - 250), t_start=t0)
        score_cand = s_cont
        if score_cand < best_score:
            best_score = score_cand
            best_h = build_h_from_g(g_cont)

    # Phase 3: Multiple restarts with perturbations
    if remaining_budget() > 200:
        n_restarts = 8
        for r in range(n_restarts):
            if remaining_budget() < 180:
                break
            scale = 0.05 + 0.15 * (r / float(n_restarts))
            g_pert = np.clip(best_h[:m] + rng.randn(m) * scale, 0.0, 1.0)
            alloc_r = min(100, remaining_budget() - 180)
            g_new, score_new = coord_descent_anti(g_pert, 300, rng,
                                                  time_limit=alloc_r, t_start=t0)
            if score_new < best_score:
                best_score = score_new
                best_h = build_h_from_g(g_new)

    # Phase 4: General (non-anti-palindromic) coordinate descent
    if remaining_budget() > 150:
        h_work = best_h.copy()
        h_new, score_new = coord_descent_general(h_work, 300, rng,
                                                 time_limit=min(150, remaining_budget() - 150), t_start=t0)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new

    # Phase 5: SLSQP polish
    if remaining_budget() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for alpha in [500, 2000, 5000, 10000]:
            if remaining_budget() < 40:
                break
            maxiter = max(100, min(500, 150000 // max(n_cur, 1)))
            try:
                res_x = slsqp_polish(best_h, alpha, maxiter)
                if res_x is not None and np.all(np.isfinite(res_x)):
                    h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target_cur)
                    s = eval_c5(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new.copy()
            except Exception:
                pass

    # Phase 6: Resolution increase
    for factor in [2, 3]:
        n_try = len(best_h) * factor
        if n_try > 2500:
            break
        if remaining_budget() < 100:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new_arr = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_hires = np.interp(x_new_arr, x_old, best_h)
        h_hires = project_feasible(h_hires, n_try / 2.0)

        if n_try % 2 == 0:
            n_try += 1
            x_old2 = np.linspace(0.0, 2.0, len(h_hires), endpoint=False)
            x_new2 = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hires = np.interp(x_new2, x_old2, h_hires)
            h_hires = project_feasible(h_hires, n_try / 2.0)

        m_try = n_try // 2
        g_hires = np.zeros(m_try)
        for j in range(m_try):
            g_hires[j] = (h_hires[j] + 1.0 - h_hires[n_try - 1 - j]) / 2.0
        g_hires = np.clip(g_hires, 0.0, 1.0)

        alloc_hr = min(200, remaining_budget() - 100)
        g_new, score_new = coord_descent_anti(g_hires, 400, rng,
                                              time_limit=alloc_hr, t_start=t0)
        if score_new < best_score:
            best_score = score_new
            best_h = build_h_from_g(g_new)

        if remaining_budget() > 80:
            h_work2 = best_h.copy()
            h_new2, score_new2 = coord_descent_general(h_work2, 200, rng,
                                                       time_limit=min(100, remaining_budget() - 80), t_start=t0)
            if score_new2 < best_score:
                best_score = score_new2
                best_h = h_new2

    # Phase 7: Final SLSQP polish
    if remaining_budget() > 15:
        n_final = len(best_h)
        target_final = n_final / 2.0
        try:
            res_x = slsqp_polish(best_h, 20000, max(200, 300000 // max(n_final, 1)))
            if res_x is not None and np.all(np.isfinite(res_x)):
                h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target_final)
                s = eval_c5(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = eval_c5(best_h)

    return (best_h, best_score, len(best_h))
```
