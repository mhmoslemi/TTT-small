import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    result = np.zeros(n)
    result[idx] = 1.0
    return result

def compute_swap_contributions(h, n):
    n_corr = 2 * n - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    A = np.zeros((n_ones, n_corr))
    for idx in range(n_ones):
        i = ones[idx]
        lo = max(0, n - 1 - i)
        hi = min(n_corr - 1, n - 1 + i)
        if lo <= hi:
            lags = np.arange(lo, hi + 1) - (n - 1)
            positions = i + lags
            A[idx, lo:hi + 1] = -(1.0 - h[positions])
        lo2 = max(0, i)
        hi2 = min(n_corr - 1, i + n - 1)
        if lo2 <= hi2:
            lags = np.arange(lo2, hi2 + 1) - (n - 1)
            positions = i - lags
            A[idx, lo2:hi2 + 1] += h[positions]
    B = np.zeros((n_zeros, n_corr))
    for idx in range(n_zeros):
        j = zeros[idx]
        lo = max(0, n - 1 - j)
        hi = min(n_corr - 1, n - 1 + j)
        if lo <= hi:
            lags = np.arange(lo, hi + 1) - (n - 1)
            positions = j + lags
            B[idx, lo:hi + 1] = (1.0 - h[positions])
        lo2 = max(0, j)
        hi2 = min(n_corr - 1, j + n - 1)
        if lo2 <= hi2:
            lags = np.arange(lo2, hi2 + 1) - (n - 1)
            positions = j - lags
            B[idx, lo2:hi2 + 1] -= h[positions]
    return A, B, ones, zeros

def best_improvement_sweep(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    A, B, ones, zeros = compute_swap_contributions(h, n)
    n_ones = len(ones)
    n_zeros = len(zeros)
    if n_ones == 0 or n_zeros == 0:
        return None, None
    diffs = corr[None, None, :] + A[:, None, :] + B[None, :, :]
    new_c5s = np.max(diffs, axis=2) * dx
    best_flat = np.argmin(new_c5s)
    best_i, best_j = np.unravel_index(best_flat, new_c5s.shape)
    best_c5 = new_c5s[best_i, best_j]
    return (ones[best_i], zeros[best_j]), best_c5

def binary_local_search(h, n, time_limit, max_iters=50000):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for _ in range(max_iters):
        if time.time() > time_limit:
            break
        swap, new_c5 = best_improvement_sweep(best_h, n)
        if swap is None or new_c5 >= best_c5 - 1e-14:
            break
        i, j = swap
        best_h[i] = 0.0
        best_h[j] = 1.0
        best_c5 = new_c5
    return best_h, best_c5

def random_kick(h, n, n_kicks, rng):
    h_new = h.copy()
    ones = np.where(h_new == 1)[0]
    zeros = np.where(h_new == 0)[0]
    for _ in range(n_kicks):
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h_new[i], h_new[j] = h_new[j], h_new[i]
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
    return h_new

def smooth_max_correlation(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - (n - 1)
        for j in range(n):
            term = 0.0
            jml = j - lag
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + lag
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

def gradient_refine(h_init, n, time_limit):
    h = h_init.astype(float).copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    bounds = [(0.0, 1.0)] * n
    for eps in [500.0, 2000.0, 5000.0, 10000.0]:
        if time.time() > time_limit:
            break
        def obj(x):
            return smooth_max_correlation(x, eps)
        def grad(x):
            return smooth_max_grad(x, eps)
        def constr_eq(x):
            return np.sum(x) - n / 2.0
        try:
            res = minimize(obj, best_h.copy(), jac=grad, method='SLSQP',
                           bounds=bounds, constraints=[{'type': 'eq', 'fun': constr_eq}],
                           options={'maxiter': 300, 'ftol': 1e-16})
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

    h_bin2 = to_binary(h_up, n2)
    h_bin2, c5_bin2 = binary_local_search(h_bin2, n2, t4)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2
        n = n2

    # Final: quick gradient polish
    t5 = time_left(0.99)
    h_grad2, c5_grad2 = gradient_refine(best_h, n, t5)
    if c5_grad2 < best_c5:
        best_c5 = c5_grad2
        best_h = h_grad2

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
