    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def ljbfgs(h0, n, alpha, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s
    
    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def greedy_binary_search(h_bin, n, max_passes=50):
    """Greedy: repeatedly find the best swap and apply it."""
    best = h_bin.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    ones = np.where(best >= 1.0 - 1e-9)[0]
    zeros = np.where(best <= 1e-9)[0]
    
    for _ in range(max_passes):
        improved = False
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h_try = best.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                c = np.max(compute_correlations(h_try))
                delta = c - best_max
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            best[i] = 0.0
            best[j] = 1.0
            best_corr = compute_correlations(best)
            best_max = np.max(best_corr)
            ones = np.where(best >= 1.0 - 1e-9)[0]
            zeros = np.where(best <= 1e-9)[0]
            improved = True
        if not improved:
            break
    return best, best_max


def sa_binary_search(h_bin, n, n_iters, rng, t0_temp=1.0):
    """Simulated annealing on binary structure."""
    best = h_bin.copy()
    curr_max = np.max(compute_correlations(best))
    best_max = curr_max
    ones = np.where(best >= 1.0 - 1e-9)[0]
    zeros = np.where(best <= 1e-9)[0]
    temp = t0_temp
    
    for it in range(n_iters):
        if it % 5000 == 0:
            temp *= 0.95
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_try = best.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        
        new_max = np.max(compute_correlations(h_try))
        delta = new_max - curr_max
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            best = h_try
            curr_max = new_max
            ones = np.where(best >= 1.0 - 1e-9)[0]
            zeros = np.where(best <= 1e-9)[0]
            if curr_max < best_max:
                best_max = curr_max
    
    return best, best_max
