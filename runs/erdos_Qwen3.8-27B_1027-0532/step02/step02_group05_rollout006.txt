            # Now find δ that minimizes max_k (corr[k] + δ * dcorr[k])
            # subject to 0 <= g[j] + δ <= 1, i.e., -g[j] <= δ <= 1-g[j]
            
            # This is a 1D convex minimax problem
            lo_d = -g[j]
            hi_d = 1.0 - g[j]
            
            # Binary search on the "slope" to find the optimum
            # The function f(δ) = max_k (corr[k] + δ*dcorr[k]) is convex piecewise linear
            # The minimum is where the slope changes from negative to positive
            
            def f_of_delta(d):
                vals = corr + d * dcorr
                return np.max(vals)
            
            # Use ternary search or binary search on subgradient
            # f'(δ) = dcorr[argmax_k (corr[k] + δ*dcorr[k])]
            # At optimum, 0 should be in the subdifferential
            
            # Binary search: if f'(δ) < 0, increase δ; if f'(δ) > 0, decrease δ
            lo_b, hi_b = lo_d, hi_d
            for _ in range(50):
                mid = (lo_b + hi_b) / 2.0
                vals = corr + mid * dcorr
                imax = np.argmax(vals)
                slope = dcorr[imax]
                if slope < -1e-12:
                    lo_b = mid
                elif slope > 1e-12:
                    hi_b = mid
                else:
                    break
            best_d = (lo_b + hi_b) / 2.0
            
            if abs(best_d) > 1e-14:
                g[j] += best_d
                g[j] = np.clip(g[j], 0.0, 1.0)
                # Update h
                h[pos] = g[j]
                h[neg] = 1.0 - g[j]
                new_score = evaluate_score(h)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    improved = True
        
        if not improved:
            break
    
    return h, best_score

import numpy as np

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def project_feasible(h, target_sum):
    N = len(h)
    lo = -3.0
    hi = 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def grad_full(h, alpha, top_k=None):
    N = len(h)
    corr = compute_correlations(h)
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        m = float(np.max(corr[idx]))
        ex = np.exp(alpha * (corr[idx] - m))
        p = ex / np.sum(ex)
        grad = np.zeros(N)
        for ii in range(len(idx)):
            j = idx[ii]
            w = p[ii]
            if w < 1e-14:
                continue
            k = j - (N - 1)
            if k >= 0:
                grad[k:] += w * (1.0 - h[:N - k])
                grad[:N - k] -= w * h[k:]
            else:
                kp = -k
                grad[:N - kp] += w * (1.0 - h[kp:])
                grad[kp:] -= w * h[:N - kp]
    else:
        m = float(np.max(corr))

            k = j - (N - 1)

            else:
                kp = -k
                grad[:N - kp] += w * (1.0 - h[kp:])
                grad[kp:] -= w * h[:N - kp]
    return grad

    return grad_full(x, alpha, top_k)

    return np.sum(x) - len(x) / 2.0


def _cons_jac(x):
    return np.ones(len(x))

    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        _slsqp_fun, h0, method='SLSQP', bounds=bounds,
        jac=_slsqp_jac, args=(alpha, top_k),
        constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        return res.x
    return None
