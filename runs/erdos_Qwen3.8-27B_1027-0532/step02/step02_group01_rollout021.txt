import time

def compute_c5_from_corr(corr, n):
    dx = 2.0 / n
    return float(dx * np.max(corr))

    return np.correlate(h, 1.0 - h, mode='full')

def delta_corr_for_swap(h, i, j, n):
    """Compute the change in correlation vector when swapping positions i (1->0) and j (0->1)."""
    s_range = np.arange(-(n-1), n)  # s values, length 2n-1
    # General formula: for each s, delta = -(1-h[i+s]) + h[i-s] + (1-h[j+s]) - h[j-s]
    # with appropriate bounds checking
    
    # i+s valid: 0 <= i+s <= n-1, i.e., -i <= s <= n-1-i
    # i-s valid: 0 <= i-s <= n-1, i.e., i-n+1 <= s <= i, and s != 0
    # j+s valid: 0 <= j+s <= n-1, i.e., -j <= s <= n-1-j
    # j-s valid: 0 <= j-s <= n-1, i.e., j-n+1 <= s <= j, and s != 0

    s1_lo = max(-(n-1), -i)
    s1_hi = min(n-1, n-1-i)
    s1 = np.arange(s1_lo, s1_hi+1)
    delta[s1 + (n-1)] -= (1.0 - h[i + s1])

    s2_lo = max(-(n-1), i-n+1)
    s2_hi = min(n-1, i)
    s2 = np.arange(s2_lo, s2_hi+1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def to_binary(h, target_ones):
    n = len(h)
    idx = np.argsort(-h, kind='stable')
    result = np.zeros(n)
    result[idx[:target_ones]] = 1.0
    return result

def make_delta(h, i, j, n):
    off = n - 1
    delta = np.zeros(2 * n - 1)
    lo = max(-i, -(n-1))
    hi = min(n-1-i, n-1)
    if lo <= hi:
        sv = np.arange(lo, hi+1)
        delta[sv + off] -= (1.0 - h[i + sv])
    lo = max(i-(n-1), -(n-1))
    hi = min(i, n-1)
    if lo <= hi:
        sv = np.arange(lo, hi+1)
        sv = sv[sv != 0]
        if len(sv) > 0:
            delta[sv + off] += h[i - sv]
    lo = max(-j, -(n-1))
    hi = min(n-1-j, n-1)
    if lo <= hi:
        sv = np.arange(lo, hi+1)
        delta[sv + off] += (1.0 - h[j + sv])
    lo = max(j-(n-1), -(n-1))
    hi = min(j, n-1)
    if lo <= hi:
        sv = np.arange(lo, hi+1)
        sv = sv[sv != 0]
        if len(sv) > 0:
            delta[sv + off] -= h[j - sv]
    s1 = j - i
    if -(n-1) <= s1 <= n-1:
        delta[s1 + off] += 1.0
    s2 = i - j
    if -(n-1) <= s2 <= n-1:
        delta[s2 + off] += 1.0
    delta[off] = 0.0
    return delta

def hill_climb_binary(h, corr, dx, max_iters, start_time, budget_s):

    off = n - 1
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    curr_mx = float(corr.max())
    for it in range(max_iters):
        if budget_s - (time.time() - start_time) < 20:
            break
        best_imp = 1e-15
        best_i = -1
        best_j = -1
        best_nc = None
        for i in ones:
            base = np.zeros(2*n-1)
            lo = max(-i, -(n-1))
            hi = min(n-1-i, n-1)
            if lo <= hi:
                sv = np.arange(lo, hi+1)
                base[sv + off] -= (1.0 - h[i + sv])
            lo = max(i-(n-1), -(n-1))
            hi = min(i, n-1)
            if lo <= hi:
                sv = np.arange(lo, hi+1)
                sv = sv[sv != 0]
                if len(sv) > 0:
                    base[sv + off] += h[i - sv]
            base[off] = 0.0
            for j in zeros:
                jd = np.zeros(2*n-1)
                lo = max(-j, -(n-1))
                hi = min(n-1-j, n-1)
                if lo <= hi:
                    sv = np.arange(lo, hi+1)
                    jd[sv + off] += (1.0 - h[j + sv])
                lo = max(j-(n-1), -(n-1))
                hi = min(j, n-1)
                if lo <= hi:
                    sv = np.arange(lo, hi+1)
                    sv = sv[sv != 0]
                    if len(sv) > 0:
                        jd[sv + off] -= h[j - sv]
                jd[off] = 0.0
                s1 = j - i
                if -(n-1) <= s1 <= n-1:
                    jd[s1 + off] += 1.0
                s2 = i - j
                if -(n-1) <= s2 <= n-1:
                    jd[s2 + off] += 1.0
                nc = corr + base + jd
                nm = float(nc.max())
                imp = curr_mx - nm
                if imp > best_imp:

        ones.remove(best_i)
        ones.append(best_j)
        zeros.remove(best_j)
        zeros.append(best_i)
    return h, corr, float(curr_mx * dx)

def sa_binary(h, corr, dx, n_steps, T0, T1, rng, start_time, budget_s):
    n = len(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    curr_mx = float(corr.max())
    best_c5 = curr_mx * dx
    best_h = h.copy()
    best_corr = corr.copy()
    for step in range(n_steps):
        if budget_s - (time.time() - start_time) < 15:
            break
        if step % 100000 == 0 and step > 0:
            pass
        T = T0 * (T1 / T0) ** (step / max(n_steps - 1, 1))
        oi = rng.integers(len(ones))
        zi = rng.integers(len(zeros))
        i = ones[oi]
        j = zeros[zi]
        delta = make_delta(h, i, j, n)
        nc = corr + delta
        nm = float(nc.max())
        dc = dx * (nm - curr_mx)
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
            h[i] = 0.0
            h[j] = 1.0
            corr = nc
            curr_mx = nm
            ones[oi] = j
            zeros[zi] = i
            c5_val = curr_mx * dx
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                best_corr = corr.copy()
    return best_h, best_corr, best_c5

def perturb_for_restart(h, rng, n_perturb):
    n = len(h)
    h_new = h.copy()
    ones = list(np.where(h_new == 1)[0])
    zeros = list(np.where(h_new == 0)[0])
    for _ in range(n_perturb):
        oi = rng.integers(len(ones))
        zi = rng.integers(len(zeros))
        i = ones[oi]
        j = zeros[zi]
        h_new[i] = 0.0
        h_new[j] = 1.0
        ones[oi] = j
        zeros[zi] = i
    return h_new

def block_perturb(h, rng, max_block_len):
    n = len(h)
    h_new = h.copy()
    blen = rng.integers(2, min(max_block_len, n // 4) + 1)
    a = rng.integers(0, n - blen + 1)
    segment = h_new[a:a+blen].copy()
    h_new[a:a+blen] = 0.0
    h_new[a:a+blen] = segment[::-1]
    h_new = to_binary(h_new, n // 2)
    return h_new
