    """Solve the 1D subproblem for coordinate i."""
    n = len(h)
    if corr is None:
        corr = compute_corr(h)
    
    # Compute m_k for all k
    # m_k = (1-h[i-r])*[0<=i-r<n] - h[i+r]*[0<=i+r<n] where r = k-(n-1)
    # For k in [0, 2n-2], r = k-(n-1)
    r = np.arange(2*n-1) - (n-1)  # effective lags
    # m_k for coordinate i:
    valid_lo = (i - r) >= 0
    valid_hi = (i - r) < n
    h_ir = np.where(valid_lo & valid_hi, h[i - r], 0.0)
    
    valid_lo2 = (i + r) >= 0
    valid_hi2 = (i + r) < n
    h_ir2 = np.where(valid_lo2 & valid_hi2, h[i + r], 0.0)
    
    m = (1.0 - h_ir) * (valid_lo & valid_hi) - h_ir2 * (valid_lo2 & valid_hi2)
    
    # c_k = corr[k] - h[i] * m_k
    c = corr - h[i] * m
    
    # Solve: min_t max_k (c_k + m_k * t) for t in [0,1]
    # Bisection

        vals = c + m * mid
        k_star = np.argmax(vals)
        slope = m[k_star]
        if slope < -1e-10:
            lo = mid
        elif slope > 1e-10:
            hi = mid
        else:
            break

    # Check boundaries
    f0 = np.max(c)  # f(0) = max(c_k + 0)
    f1 = np.max(c + m)  # f(1)
    f_mid = np.max(c + m * t_star)
    
    if f0 < f_mid:
        t_star = 0.0
    elif f1 < f_mid:
        t_star = 1.0
    
    return t_star, f_mid

    corr = compute_corr(h)

    n = len(h)
    viol = np.sum(h) - n / 2.0
    obj_pen = obj + 0.5 * mu * viol * viol
    grad_pen = grad + mu * viol
    return obj_pen, grad_pen


def penalty_jac(h, alpha, mu):
    obj, grad = smooth_max_obj_grad(h, alpha)
    n = len(h)
    viol = np.sum(h) - n / 2.0
    grad_pen = grad + mu * viol
    return grad_pen


def coord_solve_1d(c, m):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = c + m * mid
        k_star = np.argmax(vals)
        slope = m[k_star]
        if slope < -1e-12:
            lo = mid
        elif slope > 1e-12:
            hi = mid
        else:
            return mid
    t_star = (lo + hi) / 2.0
    f0 = np.max(c)
    f1 = np.max(c + m)
    f_mid = np.max(c + m * t_star)
    if f0 <= f_mid and f0 <= f1:
        return 0.0
    elif f1 <= f_mid and f1 <= f0:
        return 1.0
    return t_star


def coord_descent_pass(h, corr, rng):
    n = len(h)
    r_all = np.arange(2 * n - 1) - (n - 1)
    order = rng.permutation(n)
    for i in order:
        valid_lo = (i - r_all) >= 0
        valid_hi = (i - r_all) < n
        h_ir = np.where(valid_lo & valid_hi, h[i - r_all], 0.0)
        valid_lo2 = (i + r_all) >= 0
        valid_hi2 = (i + r_all) < n
        h_ir2 = np.where(valid_lo2 & valid_hi2, h[i + r_all], 0.0)
        m_vec = (1.0 - h_ir) * (valid_lo & valid_hi).astype(float) - h_ir2 * (valid_lo2 & valid_hi2).astype(float)
        c_vec = corr - h[i] * m_vec
        t_star = coord_solve_1d(c_vec, m_vec)
        h[i] = t_star
    return h

    n = len(h)

def lbfgsb_optimize(h0, alpha, mu, maxiter, bounds):
    res = minimize(
        penalty_obj, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu), jac=penalty_jac,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0,
                 'maxls': 50, 'maxfun': maxiter * 30}
    )
    return res.x

            best_h = h_final.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
