corr[k] = sum_{j} h[j] * (1-h[j-(k-(n-1))])

Let s = k - (n-1). Then:
corr[n-1+s] = sum_{j: 0≤j<n, 0≤j-s<n} h[j] * (1-h[j-s])

If we change h[i] from 1 to 0 and h[j] from 0 to 1:

For corr[n-1+s] (shift s):
- Old contribution from h[i]: h[i] * (1-h[i-s]) = 1*(1-h[i-s]) if i-s valid, else 0
- New contribution from h[i]: 0
- Change: -(1-h[i-s]) if 0 ≤ i-s < n, else 0

- Old contribution from h[j]: h[j] * (1-h[j-s]) = 0
- New contribution from h[j]: 1 * (1-h[j-s]) if j-s valid, else 0
- Change: +(1-h[j-s]) if 0 ≤ j-s < n, else 0

- The term (1-h[j']) where j' = m-s for some m:
  - If j' = i: m = i+s. Old: h[i+s]*(1-h[i]) = h[i+s]*0 = 0. New: h[i+s]*(1-0) = h[i+s].
    Change: +h[i+s] if 0 ≤ i+s < n, else 0.
  - If j' = j: m = j+s. Old: h[j+s]*(1-h[j]) = h[j+s]*1 = h[j+s]. New: h[j+s]*(1-1) = 0.
    Change: -h[j+s] if 0 ≤ j+s < n, else 0.

So the total change in corr at shift s:
Δcorr(s) = -(1-h[i-s])·[0≤i-s<n] + (1-h[j-s])·[0≤j-s<n] + h[i+s]·[0≤i+s<n] - h[j+s]·[0≤j+s<n]

The average of |S ∩ (S-s)| over all s is approximately (n/2) * (n/2) / n = n/4 (by a double counting argument). So the average overlap is n/4, and we want all overlaps to be above n/4.

For n=180: average overlap ≈ 45. We want min overlap > 45 to get C5 < 0.5.
Wait, C5 = (2/n) * (n/2 - min_overlap) = 1 - 2*min_overlap/n.

For C5 = 0.38: min_overlap = (1 - 0.38) * n/2 = 0.62 * 90 = 55.8.

So we need min_s |S ∩ (S-s)| ≥ 56 (approximately). The average is about 45, so we need all shifts to be above average by about 25%. That's a significant requirement.

This means the optimal set S has a very "uniform" autocorrelation, significantly above the average. This is a strong constraint that suggests the optimal S has special structure.

Hmm, but wait. I've been ignoring the boundary effects. The "full" correlation only counts valid positions, so for large |s|, the count is naturally smaller (fewer valid positions). The worst shift is likely in the middle range.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return float(dx * np.max(corr))

def project_to_binary(h, target_ones):
    n = len(h)
    if target_ones <= 0:
        return np.zeros(n)
    if target_ones >= n:
        return np.ones(n)
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:target_ones]] = 1.0
    return result

def project_onto_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def eval_swap_delta(h, i, j):
    n = len(h)
    full_n = 2 * n - 1
    delta = np.zeros(full_n)
    s_lo1 = max(-(n - 1), i - n + 1)
    s_hi1 = min(n - 1, i)
    for s in range(s_lo1, s_hi1 + 1):
        idx = i - s
        if 0 <= idx < n:
            delta[n - 1 + s] -= (1.0 - h[idx])
    s_lo2 = max(-(n - 1), j - n + 1)
    s_hi2 = min(n - 1, j)
    for s in range(s_lo2, s_hi2 + 1):
        idx = j - s
        if 0 <= idx < n:
            delta[n - 1 + s] += (1.0 - h[idx])
    s_lo3 = max(-(n - 1), -i)
    s_hi3 = min(n - 1, n - 1 - i)
    for s in range(s_lo3, s_hi3 + 1):
        idx = i + s
        if 0 <= idx < n:
            delta[n - 1 + s] += h[idx]
    s_lo4 = max(-(n - 1), -j)
    s_hi4 = min(n - 1, n - 1 - j)
    for s in range(s_lo4, s_hi4 + 1):
        idx = j + s
        if 0 <= idx < n:
            delta[n - 1 + s] -= h[idx]
    return delta

def binary_local_search_fast(h_init, time_limit, start_time, rng, n_candidates=200):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    corr = compute_corr_full(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_corr = corr.copy()
    best_max = current_max
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    stall = 0
    patience = 200
    min_delta = -1e-12
    while time.time() - start_time < time_limit:
        best_new_max = current_max
        best_i = -1
        best_j = -1
        best_delta_arr = None
        rng_cands = min(n_candidates, len(ones_list) * len(zeros_list))
        for _ in range(rng_cands):
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            delta = eval_swap_delta(h, i, j)
            new_vals = corr + delta
            nm = float(np.max(new_vals))
            if nm < best_new_max - min_delta:
                best_new_max = nm
                best_i = i
                best_j = j
                best_delta_arr = delta
        if best_i >= 0:
            corr = corr + best_delta_arr
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones_set.remove(best_i)
            zeros_set.remove(best_j)
            ones_set.add(best_j)
            zeros_set.add(best_i)
            ones_list = list(ones_set)
            zeros_list = list(zeros_set)
            current_max = best_new_max
            if current_max < best_max - 1e-10:
                best_max = current_max
                best_h = h.copy()
                best_corr = corr.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                patience = max(int(patience * 0.8), 50)
                if patience <= 51:
                    break
            continue
    return best_h, best_corr, best_max * dx

def smooth_logsumexp_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_logsumexp_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_refine(h_init, eps, maxiter, target_sum):
    n = len(h_init)
    h0 = project_onto_box_sum(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        result = minimize(smooth_logsumexp_objective, h0, args=(eps,),
                          jac=smooth_logsumexp_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_onto_box_sum(result.x, target_sum)
        return h_opt
    except Exception:
        return h0

def greedy_construct(n, rng):
    h = np.zeros(n)
    corr = compute_corr_full(h)
    current_max = 0.0
    candidates = list(range(n))
    rng.shuffle(candidates)
    target = n // 2
    for step in range(target):
        best_pos = candidates[0]
        best_max = np.inf
        for pos in candidates:
            h[pos] = 1.0
            new_corr = compute_corr_full(h)
            nm = float(np.max(new_corr))
            h[pos] = 0.0
            if nm < best_max:
                best_max = nm
                best_pos = pos
        h[best_pos] = 1.0
        candidates.remove(best_pos)
    return h

def upsample(h_coarse, factor=2):
    n = len(h_coarse)
    h_fine = np.repeat(h_coarse, factor)
    return h_fine

def random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h
