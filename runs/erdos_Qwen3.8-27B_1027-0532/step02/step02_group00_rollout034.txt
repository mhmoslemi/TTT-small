    return obj, grad

    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target = n_init / 2.0
    
    best_h = project_feasible_v2(h_init.copy(), target)
    best_score = compute_c5(best_h)
    
    # Phase 1: Generate diverse starting points
    starting_points = [best_h.copy()]
    
    # Random binary sequences
    for _ in range(10):
        if tleft() < 500:
            break
        h_bin = np.zeros(n_init)
        ones_pos = rng.choice(n_init, size=n_init//2, replace=False)
        h_bin[ones_pos] = 1.0
        starting_points.append(h_bin)

    for n_coarse in [40, 60, 80]:

            h_c = project_feasible_v2(x, target_c)
            return compute_c5(h_c)

            # Interpolate to n_init
            h_interp = np.interp(np.linspace(0,1,n_init), np.linspace(0,1,n_coarse), h_c)
            h_interp = project_feasible_v2(h_interp, target)
            starting_points.append(h_interp)

                best_score = compute_c5(h_interp)
                best_h = h_interp
        except:
            pass

                             options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10,
                                      'maxfun': 10000})
                h_new = project_feasible_v2(res.x, target)
                sc = compute_c5(h_new)
                if sc < compute_c5(h_cur):
                    h_cur = h_new
            except:
                pass
