    # h = sigmoid(z + c) where c is chosen so sum(h) = n/2
    # Use bisection to find c
    n = len(z)
    target = n / 2.0
    
    def sum_h(c):
        return np.sum(1.0 / (1.0 + np.exp(-(z + c))))
    
    # Bisection
    lo, hi = -20.0, 20.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if sum_h(mid) < target:
            lo = mid
        else:
            hi = mid
    c = (lo + hi) / 2.0
    return 1.0 / (1.0 + np.exp(-(z + c)))

This automatically satisfies the sum constraint and the box constraint (since sigmoid is in (0,1)). The only "issue" is that h can never be exactly 0 or 1, but for large |z|, it's arbitrarily close.

The gradient of h with respect to z:
dh/dz_i = h_i * (1-h_i) (from the sigmoid derivative)

And the gradient of c with respect to z_i:
d(sum_h)/dz_i = h_i(1-h_i)
dc/dz_i = -h_i(1-h_i) / (sum_j h_j(1-h_j))

So the full gradient:
d h_k / d z_i = h_k(1-h_k) * [δ(ik) + dc/dz_i]
= h_k(1-h_k) * δ(ik) - h_k(1-h_k) * h_i(1-h_i) / S
where S = sum_j h_j(1-h_j)

This is a rank-1 correction to the diagonal, which is easy to compute.

Then the gradient of the objective with respect to z:
∂f/∂z_i = sum_k (∂f/∂h_k) * (∂h_k/∂z_i)
= sum_k grad_h_k * h_k(1-h_k) * [δ(ik) - h_i(1-h_i)/S]
= grad_h_i * h_i(1-h_i) - h_i(1-h_i) * (sum_k grad_h_k * h_k(1-h_k)) / S

This is elegant and the optimizer only works with the unconstrained variable z.

This is a significant improvement because:
1. No constraint handling needed (no penalty, no SLSQP)
2. The box constraint is automatically satisfied
3. The sum constraint is automatically satisfied
4. We can use standard L-BFGS-B without any constraint complications

The tradeoff is that h can never be exactly 0 or 1, but if the optimal solution is close to binary, we can use very large |z| values to get arbitrarily close.

import numpy as np
from scipy.optimize import minimize

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_and_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def sm_obj(h, alpha):
    return smooth_max_and_grad(h, alpha)[0]


def sm_jac(h, alpha):
    return smooth_max_and_grad(h, alpha)[1]


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

    n = len(z)

        s = np.sum(1.0 / (1.0 + np.exp(-(z + mid))))
        if s < target:
            lo = mid
        else:
            hi = mid
    c = (lo + hi) / 2.0
    return 1.0 / (1.0 + np.exp(-(z + c))), c

    h, c = sigmoid_h(z)
    return smooth_max_and_grad(h, alpha)[0]

    h, c = sigmoid_h(z)
    obj, grad_h = smooth_max_and_grad(h, alpha)
    s = h * (1.0 - h)
    S = np.sum(s)
    if S < 1e-30:
        return np.zeros_like(z)
    inner = np.dot(grad_h, s)
    grad_z = grad_h * s - s * inner / S
    return grad_z

                 'maxls': 80, 'maxfun': maxiter * 30}
    )
    return res.x

    x = np.linspace(0, 2, n, endpoint=False)
