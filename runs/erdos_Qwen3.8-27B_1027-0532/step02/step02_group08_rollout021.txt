   def evaluate_score(h, n):
       dx = 2.0 / n
       corrs = np.correlate(h, 1.0 - h, mode='full')

   def project_feasible(h, n):
       # Project to {0<=h<=1, sum(h)=n/2}
       h = h.copy()
       target = n / 2.0
       lo, hi = -2.0, 2.0
       for _ in range(60):
           tau = (lo + hi) / 2.0
           h_p = np.clip(h - tau, 0.0, 1.0)
           s = np.sum(h_p)
           if s > target:
               lo = tau
           else:
               hi = tau
       return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

   def cd_sweep(h, n, n_t=200, golden_iters=30):
