        if s < best_local:

            h = h_trial.copy()

def evaluate_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        if offset == 0:
            grad += w * (1.0 - 2.0 * h)
        elif offset > 0:
            grad[:n - offset] += w * (1.0 - h[offset:])
            grad[offset:] -= w * h[:n - offset]
        else:
            k = -offset
            grad[k:] += w * (1.0 - h[:-k])
            grad[:n - k] -= w * h[k:]
    grad *= dx
    return grad

def adam_optimize(h0, target, alphas, lr_base, max_time, deadline):
    """Adam optimizer with increasing alpha (annealing)."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_score = evaluate_score(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for alpha in alphas:
        alpha_time = min(max_time / len(alphas), deadline - time.time())
        if alpha_time <= 0:
            break
        
        # Reset momentum for new alpha
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        
        while time.time() < time.time() + alpha_time and time.time() < deadline:
            if time.time() > deadline:
                break
            t += 1
            decays = (1 - lr_base * 0.9995 ** t)
            
            overlaps = compute_overlaps(h)
            sm, weights = smooth_max(overlaps, alpha)
            grad = compute_gradient(h, weights, dx)
            
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            
            lr = lr_base * decays
            h_trial = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_trial = project_onto_constraint(h_trial, target)
            
            s = evaluate_score(h_trial)
            if s < best_score:
                best_score = s
                h = h_trial
    
    return h, best_score

def block_swap_search(h, target, n_iters, deadline, rng):

    n = len(h)

        h_trial[i1*b:(i1+1)*b], h_trial[i2*b:(i2+1)*b] = h_trial[i2*b:(i2+1)*b].copy(), h_trial[i1*b:(i1+1)*b].copy()
        
        s = evaluate_score(h_trial)
        if s < best_score - 1e-9:

def targeted_swap(h, target, n_iters, deadline, rng):

        offset = k_star - (n - 1)

            # O(0) = sum h[i](1-h[i]) - minimize by making h more binary
            # Find i where h is most "mixed"
            mixedness = h * (1 - h)
            i = np.argmax(mixedness)
            # Make it more extreme
            h_trial = h.copy()
            if h[i] < 0.5:
                h_trial[i] = min(1.0, h[i] + 0.1)
            else:
                h_trial[i] = max(0.0, h[i] - 0.1)
        else:
            k = abs(offset)
            if offset > 0:
                # O(k) = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
                # Find the pair that contributes most
                contrib = h[:n-k] * (1 - h[k:])
                idx = np.argmax(contrib)
                i = idx
                j = idx + k
            else:
                # O(-k) = sum_{i=k}^{n-1} h[i](1-h[i-k])
                contrib = h[k:] * (1 - h[:-k])
                idx = np.argmax(contrib)
                i = idx + k
                j = idx
                
            # Try swapping or blending
            h_trial = h.copy()
            # Blend: move h toward making h[i] and h[j] more similar
            h_trial[i] = h[i] + 0.05 * (h[j] - h[i])
            h_trial[j] = h[j] + 0.05 * (h[i] - h[j])
