def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_onto_constraint(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1} using bisection on lambda."""
    n = len(h)
    h = h.copy()
    # Find lambda such that sum(clamp(h + lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0

        if s > target:
            hi = mid
        else:

    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def objective_with_augmented_lagrangian(h, alpha, mu, lam):
    obj, grad = smooth_max_h_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += lam * viol + 0.5 * mu * viol * viol
    grad = grad + (lam + mu * viol) * np.ones(len(h))
    return obj, grad

def lbfgsb_optimize(h0, alpha, mu, lam, maxiter, bounds):

        objective_with_augmented_lagrangian, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu, lam),
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'ipoint': 0,
                 'maxls': 50, 'maxfun': maxiter * 20}
    )
    return res.x

def active_subgradient_step(h, n_active, step_size, n_iter):
    """Subgradient step focusing on the top k_active overlaps."""
    n = len(h)
    target = n / 2.0
    for _ in range(n_iter):
        corrs = compute_correlations(h)
        # Get top n_active lags
        top_idx = np.argsort(-corrs)[:n_active]
        grad = np.zeros(n)
        for ki in top_idx:
            r = ki - (n - 1)
            for m in range(n):
                val = 0.0
                if 0 <= m + r < n:
                    val += (1.0 - h[m + r])
                if 0 <= m - r < n:
                    val -= h[m - r]
                grad[m] += val
        grad /= len(top_idx)
        h = h - step_size * grad
        h = project_onto_constraint(h, target)
    return h

from scipy.optimize import minimize

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_onto_constraint(h, target):
    n = len(h)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_h_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def aug_lang_obj_grad(h, alpha, mu, lam):
    obj, grad = smooth_max_h_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += lam * viol + 0.5 * mu * viol * viol
    grad = grad + (lam + mu * viol) * np.ones(len(h))
    return obj, grad


def slsqp_obj(h, alpha):
    return smooth_max_h_grad(h, alpha)[0]


def slsqp_jac(h, alpha):
    return smooth_max_h_grad(h, alpha)[1]


def eq_constraint(h):
    return np.sum(h) - len(h) / 2.0


def eq_constraint_jac(h):
    return np.ones(len(h))


def lbfgsb_solve(h0, alpha, mu, lam, maxiter, bounds):
    res = minimize(

        bounds=bounds, args=(alpha, mu, lam),
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                 'iprint': 0, 'maxls': 50, 'maxfun': maxiter * 20}
    )
    return res.x


def slsqp_solve(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': eq_constraint, 'jac': eq_constraint_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x


def active_subgradient(h, n_active, step_size, n_iter):
    n = len(h)
    target = n / 2.0
    for _ in range(n_iter):
        corrs = compute_correlations(h)
        top_idx = np.argsort(-corrs)[:n_active]
        grad = np.zeros(n)
        for ki in top_idx:
            r = ki - (n - 1)
            for m in range(n):
                v = 0.0
                if 0 <= m + r < n:
                    v += (1.0 - h[m + r])
                if 0 <= m - r < n:
                    v -= h[m - r]
                grad[m] += v
        grad /= float(n_active)
        h = h - step_size * grad
        h = project_onto_constraint(h, target)
    return h

    n = len(h)
