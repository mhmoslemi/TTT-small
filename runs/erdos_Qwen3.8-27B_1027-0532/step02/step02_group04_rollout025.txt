import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def ljbfgs_smooth_max(h0, alpha, n_pts, mu, maxiter):
    n = n_pts

    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        corr = compute_correlations(h)
        N = len(corr)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)

        grad = np.zeros(n)
        for idx in range(N):
            w = weights[idx]
            if w < 1e-12:
                continue
            lag = N - 1 - idx
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                if lag < n:
                    grad[:n - lag] += w
                    grad[:n - lag] -= w * h[lag:]
                    grad[lag:] -= w * h[:n - lag]
            else:
                l = -lag
                if l < n:
                    grad[l:] += w
                    grad[l:] -= w * h[:n - l]
                    grad[:n - l] -= w * h[l:]

        s = np.sum(h) - n_pts / 2.0
        grad += 2.0 * mu * s
        return grad

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def lag_weighted_minimize(h0, weights, n_pts, mu, maxiter):
    n = n_pts

    def obj(h):
        corr = compute_correlations(h)
        val = np.dot(weights, corr)
        s = np.sum(h) - n_pts / 2.0
        return val + mu * s * s

    def grad(h):
        corr = compute_correlations(h)
        N = len(corr)
        grad = np.zeros(n)
        for idx in range(N):
            w = weights[idx]
            if w < 1e-14:
                continue
            lag = N - 1 - idx
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                if lag < n:

                    grad[:n - lag] -= w * h[lag:]
                    grad[lag:] -= w * h[:n - lag]
            else:
                l = -lag
                if l < n:
                    grad[l:] += w
                    grad[l:] -= w * h[:n - l]
                    grad[:n - l] -= w * h[l:]
        s = np.sum(h) - n_pts / 2.0
        grad += 2.0 * mu * s
        return grad

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def multiplicative_weights_rounds(h0, n_pts, mu, maxiter, n_rounds, top_m, beta):
    best = h0.copy()
    best_score = evaluate_score(best)
    n_corr = 2 * n_pts - 1

    for r in range(n_rounds):
        corr = compute_correlations(best)
        top_indices = np.argsort(corr)[-top_m:]
        weights = np.zeros(n_corr)
        top_vals = corr[top_indices]
        w_vals = np.exp(beta * top_vals)
        weights[top_indices] = w_vals
        weights /= np.sum(weights)

        x = lag_weighted_minimize(best, weights, n_pts, mu, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_pts / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best = h_new.copy()

    return best, best_score


def coordinate_descent_pass(h, alpha, n_pts, mu, n_sweeps):
    target_sum = n_pts / 2.0
    best = h.copy()

    for sweep in range(n_sweeps):
        order = np.random.permutation(n_pts)

            other_sum = np.sum(best) - best[i]

            def obj_1d(val):
                corr = compute_correlations(best)
                lag_pos = n_pts - 1 + i
                lag_neg = i
                delta = val - best[i]

                    if 0 <= lag_idx < len(corr):

                h_try = best.copy()
                h_try[i] = val
                c = compute_correlations(h_try)
                sm = smooth_max_value(c, alpha)

                return sm + mu * s * s

            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 50, 'xatol': 1e-8})
            if res.x is not None and 0.0 <= res.x <= 1.0:
                best[i] = res.x

def pair_transfer_search(h, n_iters, rng, steps):
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    n = len(h)

    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for step in steps:
            h_try = best.copy()
            h_try[i] = best[i] + step
            h_try[j] = best[j] - step
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            corr_try = compute_correlations(h_try)
            max_try = np.max(corr_try)
            if max_try < best_max - 1e-13:
                best = h_try
                best_corr = corr_try
                best_max = max_try
                break

    return best, best_max
