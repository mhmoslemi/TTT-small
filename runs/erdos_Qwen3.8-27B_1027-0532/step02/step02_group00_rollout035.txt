```python
def subgradient_step(h, target, lr, top_m=5):
    """One step of subgradient descent on max_k g_k"""
    n = len(h)
    corr = compute_corr(h)
    # Find top_m lags
    top_idx = np.argsort(corr)[-top_m:][::-1]
    # Use softmax weighting
    vals = corr[top_idx]
    exps = np.exp(vals - vals[0])
    weights = exps / np.sum(exps)
    
    # Compute combined gradient

        m = int(idx) - (n - 1)
        grad += w * grad_lag(h, m)
    
    # Step
```

```python
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    val = c_max + logsum / alpha
    penalty = mu * (np.sum(h) - target)**2
    obj = dx * val + penalty
    
    exps = np.exp(shifted)
    w_sum = np.sum(exps)
    
    n_sig = min(40, len(corr))
    top_idx = np.argsort(exps)[-n_sig:][::-1]
    
    grad = np.zeros(n)
    for idx in top_idx:
        w = exps[idx] / (alpha * w_sum)
        m = int(idx) - (n - 1)
        p = abs(m)
        if p == 0:
            grad += w * (1.0 - 2.0 * h)
        elif p < n:
            grad[:n-p] += w
            grad[:n-p] -= w * h[p:n]
            grad[p:] -= w * h[:n-p]
    
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - target)
    
    return obj, grad
```

def compute_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def grad_lag_p(h, p):
    """Gradient of g_p = sum_{i=0}^{n-p-1} h[i]*(1-h[i+p])"""
    n = len(h)
    g = np.zeros(n)
    if p == 0:
        return 1.0 - 2.0 * h
    if p >= n:
        return g
    g[:n - p] += 1.0
    g[:n - p] -= h[p:n]
    g[p:] -= h[:n - p]

    h_best = h.copy()

            break

        corr = compute_corr(h)

        vals = corr[top_idx] - corr[top_idx[0]]
        
        # Softmax with temperature
        temperature = 1.0
        exps = np.exp(vals / temperature)
        weights = exps / np.sum(exps)
        
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):

                grad[:n-p] -= w * h[p:n]
                grad[p:] -= w * h[:n-p]
        
        v = beta * v - lr * grad
        h = project_feasible(h + v, target)
