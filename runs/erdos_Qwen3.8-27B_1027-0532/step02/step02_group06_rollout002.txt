import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def compute_entropic_gradient(h, alpha, n_top):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h) * dx
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return None, corr
    weights = exp_vals / total
    
    top_idx = np.argsort(-corr)[:n_top]
    top_w = weights[top_idx]
    w_sum = top_w.sum()
    if w_sum < 1e-300:
        return None, corr
    top_w = top_w / w_sum
    
    grad = np.zeros(n)
    for wi, idx in zip(top_w, top_idx):
        lag = int(idx - (n - 1))
        grad += wi * compute_gradient_for_lag(h, lag)
    
    return grad, corr

    t_param = 1.0
    
    if step_init <= 0:
        step = 1.0 / (2.0 + 2.0 * n / alpha)
    else:
        step = step_init
    
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        grad, corr = compute_entropic_gradient(y, alpha, n_top)
        if grad is None:
            break
        
        # FISTA step
        z = y - step * grad
        h_new = project_to_feasible(z, target_sum)
        
        # Momentum update with periodic restart
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = h_new + (t_param / t_new - 1.0) * (h_new - h)
        h = h_new
        
        if it % restart_interval == 0:
            t_param = 1.0
            y = h.copy()
        else:
            t_param = t_new

def fista_anneal(h0, alpha_start, alpha_end, n_top, t0, time_limit, iters_per_alpha=200):

def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n
    
    def obj_hv(hv):
        corr = compute_correlations(hv) * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def constraint_fun_hv(hv):
        return np.sum(hv) - target_sum
    
    def constraint_jac_hv(hv):
        return np.ones(n)
    
    res = minimize(obj_hv, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun_hv, 'jac': constraint_jac_hv}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

    n = len(h)
    return h + rng.randn(n) * scale / np.sqrt(n)

        return None

    top_idx = np.argsort(-corr)[:n_top]
    top_w = weights[top_idx]
    w_sum = top_w.sum()
    if w_sum < 1e-300:
        return None
    top_w = top_w / w_sum
    grad = np.zeros(n)
    for wi, idx in zip(top_w, top_idx):
        lag = int(idx - (n - 1))
        grad += wi * compute_gradient_for_lag(h, lag)
    return grad


def fista_run(h0, alpha, max_iters, n_top, deadline, step_init=0.0, restart_interval=80):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    y = h.copy()
    t_param = 1.0
    if step_init <= 0.0:
        step = 1.0 / (2.5 + 2.0 * n / alpha)
    else:
        step = step_init
    best_h = h.copy()
    best_score = evaluate_score(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = compute_entropic_gradient(y, alpha, n_top)
        if grad is None:
            break
        z = y - step * grad
        h_new = project_to_feasible(z, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = h_new + (t_param / t_new - 1.0) * (h_new - h)
        h = h_new
        if it % restart_interval == 0:
            t_param = 1.0
            y = h.copy()
        else:
            t_param = t_new
        if it % 10 == 0:
            score = evaluate_score(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
    score = evaluate_score(h)
    if score < best_score:
        best_score = score
        best_h = h.copy()
    return best_h, best_score


def fista_anneal(h0, alpha_start, alpha_end, n_top, deadline, iters_per_phase=300, step_init=0.0):
    best_h = h0.copy()
    best_score = evaluate_score(h0)
    ratio = alpha_end / alpha_start
    if ratio <= 1.0:
        return fista_run(h0, alpha_start, iters_per_phase * 3, n_top, deadline, step_init)
    n_phases = int(np.ceil(np.log(ratio) / np.log(1.6)))
    n_phases = max(1, min(n_phases, 20))
    for p in range(n_phases):
        if time.time() > deadline:
            break
        alpha = alpha_start * (1.6 ** p)
        if alpha > alpha_end * 1.1:
            alpha = alpha_end
        h_new, s_new = fista_run(best_h, alpha, iters_per_phase, n_top, deadline, step_init)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
    return best_h, best_score


def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n

    def obj_fn(hv):
        corr = compute_correlations(hv) * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def c_fun(hv):
        return np.sum(hv) - target_sum

    def c_jac(hv):
        return np.ones(n)

    res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': c_fun, 'jac': c_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def perturb_h(h, scale, rng):
    n = len(h)
    return h + rng.randn(n) * scale / np.sqrt(n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def deadline(frac):
        return t0 + budget_s * frac

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    h_init = project_to_feasible(h_init, n / 2.0)
    best_h = h_init.copy()
    best_score = evaluate_score(h_init)
    rng = np.random.RandomState(seed)

    phase1_deadline = deadline(0.35)
    if time.time() < phase1_deadline:
        h1, s1 = fista_anneal(h_init, 5.0, 200.0, 40, phase1_deadline, iters_per_phase=250)
        if s1 < best_score:
            best_score = s1
            best_h = h1.copy()

    phase2_deadline = deadline(0.60)
    n_restarts = 6
    if time.time() < phase2_deadline:
        per_restart = (phase2_deadline - time.time()) / n_restarts
        for r in range(n_restarts):
            if time.time() > phase2_deadline:
                break
            scale = 0.02 + 0.05 * (r + 1) / n_restarts
            h_pert = perturb_h(best_h, scale, rng)
            h_pert = project_to_feasible(h_pert, n / 2.0)
            rd = min(phase2_deadline, time.time() + per_restart)
            hr, sr = fista_anneal(h_pert, 10.0, 300.0, 40, rd, iters_per_phase=200)
            if sr < best_score:
                best_score = sr
                best_h = hr.copy()

    phase3_deadline = deadline(0.82)
    if time.time() < phase3_deadline:
        for alpha in [30, 100, 300, 1000]:
            if time.time() > phase3_deadline:
                break
            if time.time() > phase3_deadline - 15:
                break
            h_ref = slsqp_refine(best_h, alpha, maxiter=400)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    phase4_deadline = deadline(0.93)
    if time.time() < phase4_deadline:
        h4, s4 = fista_run(best_h, 500.0, 100000, 50, phase4_deadline, step_init=0.01, restart_interval=50)
        if s4 < best_score:
            best_score = s4
            best_h = h4.copy()

    if time.time() < deadline(0.97):
        for alpha in [2000, 5000]:
            if time.time() > deadline(0.97):
                break
            h_ref = slsqp_refine(best_h, alpha, maxiter=200)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, n / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, n)
