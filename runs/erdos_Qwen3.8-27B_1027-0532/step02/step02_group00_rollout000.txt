import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h = h.copy()
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def eq_con(h):
    return np.sum(h) - _g_target


def eq_con_jac(h):
    return np.ones(_g_n)


def interp_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_round(h_start, n, target, alphas, maxiter, ftol):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_val, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def worst_lag_refine(h, n, target, n_iter=50, alpha=100.0):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}]
    h_cur = h.copy()
    for _ in range(n_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr))
        tol = 0.05 * abs(corr[k_star])
        active = np.where(corr > corr[k_star] - tol)[0]
        if len(active) < 1:
            active = np.array([k_star])
        active_corr = corr[active]
        c_max_a = np.max(active_corr)
        shifted = np.clip(alpha * (active_corr - c_max_a), -500.0, 0.0)
        obj_val = c_max_a + np.log(np.sum(np.exp(shifted))) / alpha

        def active_obj(h_tmp):
            c = np.correlate(h_tmp, 1.0 - h_tmp, mode='full')
            ca = c[active]
            cm = np.max(ca)
            sh = np.clip(alpha * (ca - cm), -500.0, 0.0)
            return float(cm + np.log(np.sum(np.exp(sh))) / alpha)

        try:
            res = minimize(active_obj, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-13, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def random_harmonic_h(n, rng, freqs_max=6):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    for k in range(1, freqs_max + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h = h + a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n / 2.0)


def two_level_h(n, rng, density):
    h = np.zeros(n)
    n_ones = int(round(n * density))
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n / 2.0)
