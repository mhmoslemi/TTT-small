def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[0:n-L] += w * (1.0 - h[L:n])
            if L > 0:
                grad[L:n] -= w * h[0:n-L]
        else:
            M = -L
            grad[M:n] += w * (1.0 - h[0:n-M])
            grad[0:n-M] -= w * h[M:n]
    return grad

def slsqp_grad(h, alpha):

    for step in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        delta_all = compute_delta_all(h)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_max = current_max
        best_a, best_b = -1, -1
        for a in ones:
            base = current_overlaps + delta_all[a]
            new_maxes = np.max(base[None, :] - delta_all[zeros], axis=1)
            min_idx = int(np.argmin(new_maxes))
            if new_maxes[min_idx] < best_max:
                best_max = float(new_maxes[min_idx])
                best_a = a
                best_b = zeros[min_idx]
        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        current_overlaps += delta_all[best_a] - delta_all[best_b]
        current_max = best_max
    return h, current_max

def sa_binary_search(h0, n_iters, rng, time_limit=300.0):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h))
    n_target = n // 2
    if n_ones > n_target:
        idxs = np.where(h == 1.0)[0]
        remove = idxs[rng.randint(len(idxs), size=n_ones - n_target)]
        h[remove] = 0.0
    elif n_ones < n_target:
        idxs = np.where(h == 0.0)[0]
        add = idxs[rng.randint(len(idxs), size=n_target - n_ones)]
        h[add] = 1.0

    current_overlaps = compute_overlaps(h)
    current_max = float(np.max(current_overlaps))
    best_h = h.copy()
    best_max = current_max
    t_start = time.time()
    temp = 0.003
    no_improve = 0

    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        if it % 300 == 299:
            temp *= 0.95
        if temp < 1e-12:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        if r < 0.6:
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            new_overlaps = compute_overlaps(h)
            new_max = float(np.max(new_overlaps))
        elif r < 0.85:
            if len(ones) < 2 or len(zeros) < 2:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a] = 0.0
                h[b] = 1.0
                new_overlaps = compute_overlaps(h)
                new_max = float(np.max(new_overlaps))
            else:
                a1, a2 = ones[rng.randint(len(ones), size=2)]
                b1, b2 = zeros[rng.randint(len(zeros), size=2)]
                h[a1] = 0.0
                h[a2] = 0.0
                h[b1] = 1.0
                h[b2] = 1.0
                new_overlaps = compute_overlaps(h)
                new_max = float(np.max(new_overlaps))
        else:
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            new_overlaps = compute_overlaps(h)
            new_max = float(np.max(new_overlaps))
            if len(ones) >= 2 and len(zeros) >= 2:
                a1, a2 = ones[rng.randint(len(ones), size=2)]
                b1, b2 = zeros[rng.randint(len(zeros), size=2)]
                h[a1] = 0.0
                h[a2] = 0.0
                h[b1] = 1.0
                h[b2] = 1.0
                new_overlaps = compute_overlaps(h)
                new_max = float(np.max(new_overlaps))

        delta = new_max - best_max
        if new_max < current_max or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            current_max = new_max
            current_overlaps = new_overlaps
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h = best_h.copy() if False else h
            current_overlaps = compute_overlaps(h)
            current_max = float(np.max(current_overlaps))
            no_improve += 1

        if no_improve > 2000:
            for _ in range(3):
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    h[ones[rng.randint(len(ones))]] = 0.0
                    h[zeros[rng.randint(len(zeros))]] = 1.0
            current_overlaps = compute_overlaps(h)
            current_max = float(np.max(current_overlaps))
            no_improve = 0

    return best_h, best_max

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_score = evaluate_score(h)

    def time_left():
        return budget_s - 20.0 - (time.time() - t0)

    # Phase 1: Quick continuous optimization
    for alpha in [100, 400, 1000]:
        if time_left() < 90:
            break
        penalty = alpha * 30.0
        bounds = [(0.0, 1.0)] * n
        res_x = lbfgsb_optimize(h, alpha, penalty, 500, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    # Phase 2: Steepest descent binary local search
    if time_left() > 120:
        h_bin = (h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_target = n // 2
        if n_ones > n_target:
            idxs = np.where(h_bin == 1.0)[0]
            remove = idxs[rng.randint(len(idxs), size=n_ones - n_target)]
            h_bin[remove] = 0.0
        elif n_ones < n_target:
            idxs = np.where(h_bin == 0.0)[0]
            add = idxs[rng.randint(len(idxs), size=n_target - n_ones)]
            h_bin[add] = 1.0

        h_sd, raw_sd = steepest_descent_binary(h_bin, max_steps=5000, time_limit=min(80.0, time_left() * 0.25))
        h_sd_proj = project_box_sum(h_sd, n / 2.0)
        s_sd = evaluate_score(h_sd_proj)
        if s_sd < best_score:
            best_score = s_sd
            best_h = h_sd_proj.copy()
        h = h_sd

        # Continue with more steps if still improving
        if time_left() > 100:
            h_sd2, raw_sd2 = steepest_descent_binary(h_sd, max_steps=3000, time_limit=min(60.0, time_left() * 0.2))
            h_sd2_proj = project_box_sum(h_sd2, n / 2.0)
            s_sd2 = evaluate_score(h_sd2_proj)
            if s_sd2 < best_score:
                best_score = s_sd2
                best_h = h_sd2_proj.copy()
            h = h_sd2

    # Phase 3: Simulated annealing with multi-swap
    if time_left() > 150:
        h_sa, raw_sa = sa_binary_search(h, min(20000, int(time_left() * 3)), rng, time_limit=min(250.0, time_left() * 0.5))
        h_sa_proj = project_box_sum(h_sa, n / 2.0)
        s_sa = evaluate_score(h_sa_proj)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa_proj.copy()
            h = h_sa
        else:
            h = best_h

        # Second SA round with different temperature
        if time_left() > 100:
            h2 = best_h + rng.randn(n) * 0.1
            h2 = project_box_sum(h2, n / 2.0)
            h_sa2, raw_sa2 = sa_binary_search(h2, min(15000, int(time_left() * 2)), rng, time_limit=min(120.0, time_left() * 0.5))
            h_sa2_proj = project_box_sum(h_sa2, n / 2.0)
            s_sa2 = evaluate_score(h_sa2_proj)
            if s_sa2 < best_score:
                best_score = s_sa2
                best_h = h_sa2_proj.copy()

    # Phase 4: Continuous polishing of best binary
    if time_left() > 60:
        h_polish = project_box_sum(best_h, n / 2.0)
        for alpha in [500, 2000, 5000]:
            if time_left() < 30:
                break
            penalty = alpha * 50.0
            bounds = [(0.0, 1.0)] * n
            res_x = lbfgsb_optimize(h_polish, alpha, penalty, 800, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_polish = h_new

        if time_left() > 30:
            bounds = [(0.0, 1.0)] * n
            res_x = slsqp_optimize(h_polish, 3000, 300, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 5: Try higher resolution
    if time_left() > 150:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_box_sum(h_fine, n2 / 2.0)

        for alpha in [200, 1000]:
            if time_left() < 40:
                break
            penalty = alpha * 30.0
            bounds_fine = [(0.0, 1.0)] * n2
            res_x = lbfgsb_optimize(h_fine, alpha, penalty, 300, bounds_fine)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n2 / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_fine = h_new

        if time_left() > 60 and len(best_h) == n2:
            bounds_fine = [(0.0, 1.0)] * n2
            res_x = slsqp_optimize(h_fine, 2000, 200, bounds_fine)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n2 / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 6: Final steepest descent + polish cycle
    if time_left() > 40:
        h_final = (best_h > 0.3).astype(float)
        nf = len(h_final)
        n_ones = int(np.sum(h_final))
        n_target = nf // 2
        if n_ones > n_target:
            idxs = np.where(h_final == 1.0)[0]
            remove = idxs[rng.randint(len(idxs), size=n_ones - n_target)]
            h_final[remove] = 0.0
        elif n_ones < n_target:
            idxs = np.where(h_final == 0.0)[0]
            add = idxs[rng.randint(len(idxs), size=n_target - n_ones)]
            h_final[add] = 1.0

        h_final, _ = steepest_descent_binary(h_final, max_steps=2000, time_limit=min(30.0, time_left() * 0.4))
        h_final_proj = project_box_sum(h_final, nf / 2.0)
        s_final = evaluate_score(h_final_proj)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final_proj.copy()

        if time_left() > 20:
            bounds_fin = [(0.0, 1.0)] * nf
            res_x = lbfgsb_optimize(best_h, 3000, 150000.0, 500, bounds_fin)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), nf / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Final projection
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
