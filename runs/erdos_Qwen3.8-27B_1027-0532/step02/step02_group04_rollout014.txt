    N = len(h)
    corr = compute_correlations(h)
    max_val = np.max(corr)
    
    # Find active set (within epsilon of max)
    epsilon = 1e-6 * max_val
    active = np.where(corr >= max_val - epsilon)[0]
    
    # Compute subgradient (average of gradients for active lags)
    grad = np.zeros(N)
    for idx in active:
        lag = idx - (N - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            k = lag
            grad[:N-k] -= h[k:]
            grad[k:] += (1.0 - h[:N-k])
        else:

            grad[:N-l] += (1.0 - h[l:])
            grad[l:] -= h[:N-l]
    
    grad /= len(active)
    
    # Take step
    h_new = h - step_size * grad
    # Project

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)

    grad = np.zeros(N)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (N - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            k = lag
            if k < N:
                grad[:N - k] -= w * h[k:]
                grad[k:] += w * (1.0 - h[:N - k])
        else:
            l = -lag
            if l < N:
                grad[:N - l] += w * (1.0 - h[l:])
                grad[l:] -= w * h[:N - l]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def subgradient_refine(h, n_steps, base_step, rng):
    N = len(h)
    best = h.copy()
    best_score = np.max(compute_correlations(best))
    for t in range(n_steps):
        corr = compute_correlations(best)
        max_val = np.max(corr)
        if max_val < best_score:
            best_score = max_val

        epsilon = 1e-4 * max_val + 1e-10
        active = np.where(corr >= max_val - epsilon)[0]

        grad = np.zeros(N)
        for idx in active:
            lag = idx - (N - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * best)
            elif lag > 0:
                k = lag
                if k < N:
                    grad[:N - k] -= best[k:]
                    grad[k:] += (1.0 - best[:N - k])
            else:
                l = -lag
                if l < N:
                    grad[:N - l] += (1.0 - best[l:])
                    grad[l:] -= best[:N - l]

        if len(active) > 0:
            grad /= len(active)

        step = base_step / (1.0 + 0.01 * t)
        h_new = best - step * grad
        h_new = project_feasible(h_new, N / 2.0)

        new_corr = compute_correlations(h_new)
        new_max = np.max(new_corr)
        if new_max < max_val - 1e-14:
            best = h_new
            if new_max < best_score:
                best_score = new_max
        else:
            step_damp = 0.5
            h_new = best - step * step_damp * grad
            h_new = project_feasible(h_new, N / 2.0)
            new_corr = compute_correlations(h_new)
            new_max = np.max(new_corr)
            if new_max < max_val - 1e-14:
                best = h_new
                if new_max < best_score:
                    best_score = new_max
    return best


def targeted_pair_search(h, n_iters, rng):
    N = len(h)
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)

    for it in range(n_iters):
        k_max = np.argmax(best_corr)
        lag = k_max - (N - 1)
        l = abs(lag)

        if l == 0:
            probs = best * (1.0 - best)
            i = np.argmax(probs)
            if probs[i] < 1e-6:
                continue
            delta = 0.02 * (2.0 * best[i] - 1.0)
            j = rng.randint(N)
            if j != i:
                h_try = best.copy()
                h_try[i] -= delta
                h_try[j] += delta
                if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                    c_try = compute_correlations(h_try)
                    if np.max(c_try) < best_max - 1e-13:
                        best = h_try
                        best_corr = c_try
                        best_max = np.max(c_try)
        elif l >= N:
            continue
        else:
            if lag > 0:
                contrib = best[l:] * (1.0 - best[:N - l])
            else:
                contrib = best[:N - l] * (1.0 - best[l:])

            if np.max(contrib) < 1e-10:
                continue

            n_top = max(3, int(0.05 * len(contrib)))
            top_idx = np.argsort(contrib)[-n_top:]
            idx = top_idx[rng.randint(n_top)]

            if rng.rand() < 0.5:
                move_pos = idx + l
                if move_pos >= N:
                    move_pos = idx
                direction = -1.0
            else:
                move_pos = idx
                if lag > 0:
                    direction = 1.0
                else:
                    direction = 1.0

            step = rng.uniform(0.005, 0.05)
            j = rng.randint(N)
            if j == (idx + l) % N or j == idx:
                continue

            h_try = best.copy()
            h_try[move_pos] += direction * step
            h_try[j] -= direction * step

            if 0 <= h_try[move_pos] <= 1 and 0 <= h_try[j] <= 1:
                c_try = compute_correlations(h_try)
                mx = np.max(c_try)
                if mx < best_max - 1e-13:
                    best = h_try
                    best_corr = c_try
                    best_max = mx

    return best
