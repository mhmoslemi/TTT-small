    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def subgradient_nesterov(h0, max_iters, alpha_schedule, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    v = np.zeros(n)
    rng = np.random.RandomState(seed)
    mu = 0.87
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break

        alpha = alpha_schedule[min(it // max(1, max_iters // len(alpha_schedule)), len(alpha_schedule) - 1)]

        k_t = 1.0 + np.sqrt(1.0 + 4.0 * mu * mu * (it + 1))
        beta = (1.0 - k_t / (k_t + 1.0)) / mu if it > 0 else 0.0
        y = h + beta * v

        corr = compute_correlations(y)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha * shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum

        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * gradient_at_lag(y, lag)

        v = mu * v - step * grad
        h_new = h + v
        h_new = project_to_feasible(h_new, target_sum)

        score = evaluate_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1

        h = h_new

        if stall > 80:
            step *= 0.5
        if stall > 400:
            h = best_h + rng.randn(n) * (0.015 + 0.01 * stall / 500.0) / np.sqrt(n)
            h = project_to_feasible(h, target_sum)
            v = np.zeros(n)
            step = step_init * (0.5 + 0.5 * rng.rand())
            stall = 0

        if it > 0 and it % 3000 == 0:
            step = max(step * 0.92, step_init * 0.008)

    return best_h, best_score


def chase_top_lags(h0, max_iters, step_init, n_chase, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break

        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_chase]
        top_vals = corr[top_idx]

        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx - (n - 1))
            grad += gradient_at_lag(h, lag)
        grad /= n_chase

        improved = False
        s = step
        for trial in range(12):
            h_new = h - s * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                step = min(s * 1.08, 8.0)
                stall = 0
                improved = True
                break
            s *= 0.45

        if not improved:
            stall += 1
            step *= 0.7
            if stall > 60:
                h = best_h + rng.randn(n) * 0.012 / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init * (0.6 + 0.4 * rng.rand())
                stall = 0

    return best_h, best_score


def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        if alpha < 1e5:
            return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        return m

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def local_search_bits(h0, max_flips, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()
    flip_count = 0

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]

    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_score

    for it in range(max_flips):
        if time.time() - t0 > t0 + time_limit:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        h = project_to_feasible(h, target_sum)
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
        flip_count += 1

    return best_h, best_score


def binarize_and_refine(h):
    n = len(h)
    target_sum = n / 2.0
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    curr_sum = int(np.sum(h_bin))
    diff = curr_sum - target_sum
    if diff > 0:
        ones = np.where(h_bin == 1.0)[0]
        sorted_ones = ones[np.argsort(h[ones])]
        h_bin[sorted_ones[:diff]] = 0.0
    elif diff < 0:
        zeros = np.where(h_bin == 0.0)[0]
        sorted_zeros = zeros[np.argsort(-h[zeros])]
        h_bin[sorted_zeros[:(-diff)]] = 1.0
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_to_feasible(init, n_init / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)
    best_n = n_init

    rng = np.random.RandomState(seed)

    alpha_schedules = [
        [1, 3, 10, 30],
        [3, 10, 50, 100],
        [10, 30, 100, 300],
        [30, 100, 300, 500],
        [100, 300, 500, 1000],
    ]
    n_tops = [3, 5, 8, 12]
    step_inits = [0.4, 0.8, 0.3, 1.2]

    run_count = 0
    for sched in alpha_schedules:
        for nt in n_tops:
            if remaining() < budget_s * 0.22:
                break
            step = step_inits[run_count % len(step_inits)]
            t_allow = min(remaining() * 0.35 - 15, budget_s * 0.055)
            if t_allow < 4:
                break
            start_h = best_h
            if run_count > 0 and run_count % 4 == 0:
                start_h = best_h + rng.randn(len(best_h)) * 0.012 / np.sqrt(len(best_h))
                start_h = project_to_feasible(start_h, len(start_h) / 2.0)
            h, score = subgradient_nesterov(start_h, 15000, sched, nt, step,
                                             seed + run_count * 73, t0, t_allow)
            if score < best_score:
                best_score = score
                best_h = h.copy()
            run_count += 1

    if remaining() > budget_s * 0.12:
        for i in range(5):
            if remaining() < budget_s * 0.04:
                break
            t_allow = min(remaining() * 0.12, budget_s * 0.045)
            if t_allow < 4:
                break
            nc = [1, 2, 3, 4, 5][i % 5]
            start_h = best_h
            if i > 0:
                start_h = best_h + rng.randn(len(best_h)) * 0.008 / np.sqrt(len(best_h))
                start_h = project_to_feasible(start_h, len(start_h) / 2.0)
            h, score = chase_top_lags(start_h, 40000, 0.4, nc, seed + 6000 + i * 111, t0, t_allow)
            if score < best_score:
                best_score = score
                best_h = h.copy()

    if remaining() > 20:
        alphas_slsqp = [3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
        for alpha in alphas_slsqp:
            if remaining() < 8:
                break
            h_ref = slsqp_refine(best_h, alpha, maxiter=250)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    if remaining() > 15:
        h_bin = binarize_and_refine(best_h)
        score_bin = evaluate_score(h_bin)
        if score_bin < best_score:
            best_score = score_bin
            best_h = h_bin.copy()
        if remaining() > 20:
            h_ref = slsqp_refine(h_bin, 200, maxiter=150)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    if remaining() > budget_s * 0.06:
        for mult in [2, 3]:
            if remaining() < budget_s * 0.025:
                break
            n_try = best_n * mult
            if n_try > 3000 or n_try == best_n:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_to_feasible(h_init, n_try / 2.0)
            t_allow = min(remaining() * 0.12, 45)
            if t_allow < 4:
                break
            h, score = subgradient_nesterov(h_init, 15000, [20, 50, 100, 200], 8, 0.4,
                                             seed + 8000 + n_try, t0, t_allow)
            if score < best_score:
                best_score = score
                best_h = h.copy()
                best_n = n_try

    if remaining() > 12:
        h, score = subgradient_nesterov(best_h, 20000, [50, 100, 300, 500], 5, 0.3,
                                         seed + 99999, t0, remaining() - 8)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    if remaining() > 10:
        for alpha in [3000, 10000, 50000]:
            if remaining() < 6:
                break
            h_ref = slsqp_refine(best_h, alpha, maxiter=150)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
