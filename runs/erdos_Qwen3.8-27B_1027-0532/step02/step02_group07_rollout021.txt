        else:
            order = range(n)
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum_exact(h, n / 2.0)
    return h


def gradient_step(h, alpha, lr):
    corr = compute_correlations(h)
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= weights.sum()
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        p = np.arange(2 * n - 1)
        idx1 = j + p - (n - 1)
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j - p + (n - 1)
        valid2 = (idx2 >= 0) & (idx2 < n)
        slopes = np.zeros(2 * n - 1)
        if np.any(valid1):
            slopes[valid1] += (1.0 - h[idx1[valid1]])
        if np.any(valid2):
            slopes[valid2] -= h[idx2[valid2]]
        grad[j] = np.dot(weights, slopes)
    h_new = h - lr * grad
    h_new = project_sum_exact(h_new, n / 2.0)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_sum_exact(init, n / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    if time_left() > 50:
        h_cd = coordinate_descent(best_h, n_sweeps=300, tol=1e-13, time_budget=min(40, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 50:
        n_cur = len(best_h)
        bounds = [(0.0, 1.0)] * n_cur
        for alpha in [500, 2000, 10000]:
            if time_left() < 15:
                break
            maxiter = max(100, 150000 // n_cur)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None and np.all(np.isfinite(res_x)):
                h_new = project_sum_exact(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    if time_left() > 50:
        h_cd = coordinate_descent(best_h, n_sweeps=400, tol=1e-13, time_budget=min(40, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    n_cur = len(best_h)
    n_perturb = 180
    for i in range(n_perturb):
        if time_left() < 30:
            break
        frac = i / max(n_perturb - 1, 1)
        scale = 0.015 + 0.12 * frac
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_sum_exact(h_pert, n_cur / 2.0)
        tb = min(4.0, (time_left() - 20) / max(n_perturb - i, 1))
        use_rand = (i % 3 != 0)
        h_cd = coordinate_descent(h_pert, n_sweeps=80, tol=1e-11, time_budget=tb, random_order=use_rand, rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    base_n = len(best_h)
    res_candidates = []
    for mult in [1.5, 2.0]:
        n_try = int(base_n * mult)
        if 100 <= n_try <= 3000:
            res_candidates.append(n_try)
    for n_try in res_candidates:
        if time_left() < 70:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum_exact(h_interp, n_try / 2.0)
        tb_cd = min(35, time_left() - 30)
        if tb_cd > 10:
            h_cd = coordinate_descent(h_interp, n_sweeps=200, tol=1e-12, time_budget=tb_cd)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    n_cur = len(best_h)
    for i in range(25):
        if time_left() < 25:
            break
        h_pert = best_h.copy()
        n_flip = rng.randint(3, max(4, n_cur // 5))
        flip_idx = rng.choice(n_cur, size=n_flip, replace=False)
        h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_sum_exact(h_pert, n_cur / 2.0)
        tb = min(4.0, (time_left() - 15) / max(25 - i, 1))
        h_cd = coordinate_descent(h_pert, n_sweeps=100, tol=1e-11, time_budget=tb, random_order=True, rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    if time_left() > 30:
        n_cur = len(best_h)
        for i in range(15):
            if time_left() < 15:
                break
            h_round = (best_h > 0.5).astype(float)
            h_round = project_sum_exact(h_round, n_cur / 2.0)
            h_pert = h_round + rng.randn(n_cur) * 0.05
            h_pert = project_sum_exact(h_pert, n_cur / 2.0)
            tb = min(3.0, (time_left() - 10) / max(15 - i, 1))
            h_cd = coordinate_descent(h_pert, n_sweeps=80, tol=1e-11, time_budget=tb, random_order=True, rng=rng)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    if time_left() > 50:
        h_cd = coordinate_descent(best_h, n_sweeps=600, tol=1e-14, time_budget=min(50, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 25:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [20000, 100000]:
            if time_left() < 10:
                break
            res_x = slsqp_optimize(best_h, alpha, 400, bounds)
            if res_x is not None and np.all(np.isfinite(res_x)):
                h_new = project_sum_exact(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    if time_left() > 20:
        h_cd = coordinate_descent(best_h, n_sweeps=400, tol=1e-15, time_budget=min(18, time_left() - 3))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    best_h = project_sum_exact(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
