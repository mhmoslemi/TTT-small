```python
def compute_delta_corr(h, a, b, n):
    # a: index of 1 being flipped to 0
    # b: index of 0 being flipped to 1
    L = np.arange(-(n-1), n)  # all shifts
    delta = np.zeros(2*n-1)
    # Term 1a: h[a+L] - 1 where 0 <= a+L <= n-1
    valid = (a + L >= 0) & (a + L < n)
    delta[valid] += h[a + L[valid]] - 1
    # Term 1b: 1 - h[b+L] where 0 <= b+L <= n-1
    valid = (b + L >= 0) & (b + L < n)
    delta[valid] += 1 - h[b + L[valid]]
    # Term 2a: h[a-L] where 0 <= a-L <= n-1
    valid = (a - L >= 0) & (a - L < n)
    delta[valid] += h[a - L[valid]]
    # Term 2b: -h[b-L] where 0 <= b-L <= n-1
    valid = (b - L >= 0) & (b - L < n)
    delta[valid] -= h[b - L[valid]]
    # Term 3: +1 if |L| == |a-b|
    d = abs(a - b)
    if d < n:
        delta[L == d] += 1
        delta[L == -d] += 1
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def make_binary_half(h, n, target, rng):
    h_bin = np.zeros(n)
    n_ones_target = int(target - 0.5)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones_target]] = 1.0
    half_candidates = sorted_idx[n_ones_target:n]
    half_pos = half_candidates[0]
    h_bin[half_pos] = 0.5
    return h_bin


def make_binary_half_v2(h, n, target, rng):
    n_ones_target = int(target - 0.5)
    h_bin = np.zeros(n)
    perm = rng.permutation(n)
    h_bin[perm[:n_ones_target - 1]] = 1.0
    h_bin[perm[n_ones_target - 1]] = 0.5
    h_bin[perm[n_ones_target]] = 1.0
    return h_bin


def compute_delta_vec(h, a, b, L_arr, n):
    dv = h[b] - h[a]
    if abs(dv) < 1e-15:
        return np.zeros(2 * n - 1)
    result = np.zeros(2 * n - 1)
    ap = a + L_arr
    bp = b + L_arr
    mask = (ap >= 0) & (ap < n)
    result[mask] -= dv * h[ap[mask]]
    mask = (bp >= 0) & (bp < n)
    result[mask] += dv * h[bp[mask]]
    am = a - L_arr
    bm = b - L_arr
    mask = (am >= 0) & (am < n)
    result[mask] -= dv * h[am[mask]]
    mask = (bm >= 0) & (bm < n)
    result[mask] += dv * h[bm[mask]]
    d = abs(a - b)
    if 0 < d < n:
        result[L_arr == d] += dv * dv
        result[L_arr == -d] += dv * dv
    return result


def sa_phase(h, corr, L_arr, n, time_budget, rng, seed_phase=0):
    best_h = h.copy()
    best_max = float(np.max(corr))
    current_max = best_max
    t_start = time.time()
    temp = 0.015
    step = 0
    improvement_steps = 0

    while (time.time() - t_start) < time_budget * 0.95:
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        va = h[a]
        vb = h[b]
        if abs(va - vb) < 1e-10:
            continue
        dv = vb - va
        ap = a + L_arr
        bp = b + L_arr
        am = a - L_arr
        bm = b - L_arr
        delta_corr = np.zeros(2 * n - 1)
        mask = (ap >= 0) & (ap < n)
        delta_corr[mask] -= dv * h[ap[mask]]
        mask = (bp >= 0) & (bp < n)
        delta_corr[mask] += dv * h[bp[mask]]
        mask = (am >= 0) & (am < n)
        delta_corr[mask] -= dv * h[am[mask]]
        mask = (bm >= 0) & (bm < n)
        delta_corr[mask] += dv * h[bm[mask]]
        d = abs(a - b)
        if 0 < d < n:
            delta_corr[L_arr == d] += dv * dv
            delta_corr[L_arr == -d] += dv * dv
        new_corr = corr + delta_corr
        new_max = float(np.max(new_corr))
        delta = new_max - current_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            h[a] = vb
            h[b] = va
            corr = new_corr
            current_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_h = h.copy()
                improvement_steps = 0
            else:
                improvement_steps += 1
        else:
            improvement_steps += 1
        step += 1
        if step % 20000 == 0:
            temp *= 0.97

            temp = 5e-7

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h_flat, alpha, penalty):
    n = len(h_flat)
    obj = lse_val(compute_overlaps(h_flat), alpha)
    grad = lse_grad(h_flat, alpha).copy()
    s = float(np.sum(h_flat)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0


def sum_constraint_jac(h):
    return np.ones(len(h))


def fractional_refine(h0, time_budget, alpha_list, penalty_mult):
    best_h = h0.copy()
    best_score = evaluate_score(h0)
    t_start = time.time()
    n = len(h0)
    for alpha in alpha_list:
        if (time.time() - t_start) > time_budget * 0.8:
            break
        penalty = alpha * penalty_mult
        bounds = [(0.0, 1.0)] * n
        maxiter = max(50, 200000 // n)
        try:
            res = minimize(
                penalty_obj_and_grad, h0, args=(alpha, penalty),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
            )
            h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
                h0 = h_new.copy()
        except Exception:
            pass
    return best_h, best_score

    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    L_arr = np.arange(-(n - 1), n)

    h_proj = project_box_sum(h0, target)
    h = make_binary_half(h_proj, n, target, rng)
    corr = compute_overlaps(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_raw_max = current_max

    elapsed = time.time() - t0
    sa_budget = budget_s * 0.65 - elapsed
    if sa_budget > 10:
        best_h, best_raw_max = sa_phase(h.copy(), corr.copy(), L_arr, n, sa_budget, rng)
        h = best_h.copy()
        corr = compute_overlaps(h)

    elapsed = time.time() - t0
    hc_budget = budget_s * 0.15 - elapsed
    if hc_budget > 5:
        h_hc, raw_hc = fast_hill_climb(h.copy(), corr.copy(), L_arr, n, hc_budget, rng)
        if raw_hc < best_raw_max:
            best_raw_max = raw_hc
            best_h = h_hc.copy()

    elapsed = time.time() - t0
    fr_budget = budget_s * 0.18 - elapsed
    if fr_budget > 10:
        alpha_list = [500, 1500, 3000, 6000]
        h_fr, score_fr = fractional_refine(best_h, fr_budget, alpha_list, 30.0)
        c5_fr = score_fr
        c5_best = best_raw_max * (2.0 / n)
        if c5_fr < c5_best:
            best_h = h_fr.copy()
            best_raw_max = c5_fr * (n / 2.0)

    elapsed = time.time() - t0
    if time.time() - t0 < budget_s * 0.9:
        h_rounded = make_binary_half(h_proj, n, target, rng)
        for i in range(n):
            if h_rounded[i] != best_h[i] and abs(h_rounded[i] - 0.5) > 0.1:
                pass
        h2 = project_box_sum(best_h, target)
        h2 = make_binary_half_v2(h2, n, target, rng)
        corr2 = compute_overlaps(h2)
        max2 = float(np.max(corr2))
        remaining = budget_s - (time.time() - t0) - 5.0
        if remaining > 20:
            h2, max2 = sa_phase(h2, corr2, L_arr, n, remaining * 0.7, rng)
            if max2 < best_raw_max:
                best_raw_max = max2
                best_h = h2.copy()

    best_h = project_box_sum(best_h, target)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, n)
```
