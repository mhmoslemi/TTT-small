import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_and_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        sum_exp = 1.0
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def penalty_obj_grad(h, alpha, mu):
    obj, grad = smooth_max_and_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * np.ones_like(h) * viol
    return obj, grad


def lbfgsb_run(h0, alpha, mu, maxiter, bounds):
    res = minimize(
        penalty_obj_grad, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu),
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                 'iprint': 0, 'maxls': 30, 'maxfun': maxiter * 30}
    )
    return res.x


def slsqp_fun(h, alpha):
    return smooth_max_and_grad(h, alpha)[0]


def slsqp_jac(h, alpha):
    return smooth_max_and_grad(h, alpha)[1]


def constraint_eq_h(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac_h(h):
    return np.ones(len(h))


def slsqp_run(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_fun, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq_h, 'jac': constraint_eq_jac_h}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x


def dc_step(h, k_star):
    n = len(h)
    r = k_star - (n - 1)
    valid = np.zeros(n, dtype=float)
    if r >= 0:
        valid[:n - r] = 1.0
    else:
        valid[-r:] = 1.0
    c = valid.copy()
    if r > 0:
        for m in range(n - r):
            c[m] -= 1.0
            c[m + r] -= 1.0
    elif r < 0:
        for m in range(0, -r):
            c[m] -= 1.0
            c[m - r] -= 1.0
    sort_idx = np.argsort(c)
    h_new = np.zeros(n)
    h_new[sort_idx[:n // 2]] = 1.0
    return h_new


def dc_algorithm(h_init, n_iters):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    for it in range(n_iters):
        corrs = compute_correlations(h)
        k_star = np.argmax(corrs)
        h_dc = dc_step(h, k_star)
        h_dc = project_sum(h_dc, n / 2.0)
        s = evaluate_score(h_dc)
        if s < best_score:
            best_score = s
            best_h = h_dc.copy()
        h = h_dc
    return best_h, best_score


def dc_soft_step(h, k_star, temperature):
    n = len(h)
    r = k_star - (n - 1)
    valid = np.zeros(n, dtype=float)

            c[m] -= 1.0
            c[m - r] -= 1.0
    h_soft = 1.0 / (1.0 + np.exp((c - np.median(c)) / temperature))
    h_soft = project_sum(h_soft, n / 2.0)
    return h_soft

        if len(high_idx) == 0 or len(low_idx) == 0:

        n_try = min(80, max(20, len(high_idx) * len(low_idx) // 10))

            h_trial = h.copy()
            h_trial[i] = max(0.0, h[i] - 1.0)

                high_idx = np.where(h > 0.6)[0]
                low_idx = np.where(h < 0.4)[0]
        if not improved:
            break
    return h, best_score

    x = np.linspace(0.0, 2.0, n, endpoint=False)

    h = (np.sin(2.0 * np.pi * x * freq * phi + phase) > 0.0).astype(float)
    h = project_sum(h, n / 2.0)
    return h

            res_x = lbfgsb_run(pert, alpha_r, mu_r, maxiter_r, bounds_best)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
        except Exception:
            pass

    # Phase 7: Coordinate descent polishing
    if time_left() > 40:
        h_cd, score_cd = coordinate_descent_polish(best_h, 300, rng)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd.copy()

    # Phase 8: Final SLSQP + high alpha L-BFGS-B
    n_final = len(best_h)
    bounds_final = [(0.0, 1.0)] * n_final
    for alpha in [5000, 10000, 20000]:
        if time_left() < 25:
            break
        try:
            res_x = slsqp_run(best_h, alpha, 150, bounds_final)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
        except Exception:
            pass

    for alpha in [30000, 60000]:
        if time_left() < 15:
            break
        try:
            res_x = lbfgsb_run(best_h, alpha, 50000.0, 200, bounds_final)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
        except Exception:
            pass

    # Final coordinate descent
    if time_left() > 10:
        h_cd, score_cd = coordinate_descent_polish(best_h, 100, rng)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
