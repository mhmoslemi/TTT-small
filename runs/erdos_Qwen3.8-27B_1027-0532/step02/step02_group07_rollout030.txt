                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0))

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                # Update corr
                m = 2*n-1
                p = np.arange(m)
                idx1 = j + p - (n-1)
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = j - p + (n-1)
                valid2 = (idx2 >= 0) & (idx2 < n)
                slopes = np.zeros(m)
                if np.any(valid1):
                    slopes[valid1] = (1.0 - h[idx1[valid1]])
                if np.any(valid2):
                    slopes[valid2] -= h[idx2[valid2]]
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h

    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def softmax_gradient_step(h, beta, lr):
    """Take a gradient step of the softmax approximation to max."""
    n = len(h)
    corr = compute_correlations(h)
    # Softmax weights
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    # Compute weighted gradient
    grad = np.zeros(n)
    for j in range(n):
        slopes = compute_slopes(h, j)
        grad[j] = np.dot(w, slopes)
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    return h_new

def softmax_cd_loop(h, n_outer=50, n_inner=50, beta_start=10, beta_end=10000, 
                     lr_start=0.05, lr_end=0.001, time_budget=200):
    """Alternating softmax gradient + CD."""
    t0 = time.time()
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for outer in range(n_outer):
        if time.time() - t0 > time_budget:
            break
        frac = outer / max(n_outer - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Softmax gradient steps
        for inner in range(n_inner):
            if time.time() - t0 > time_budget:
                break
            h = softmax_gradient_step(h, beta, lr)
        
        # CD to convergence
        tb = min(5.0, time_budget - (time.time() - t0))
        if tb > 2.0:
            h = coordinate_descent(h, n_sweeps=50, time_budget=tb)
        
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    
    return best_h, best_score

Wait, the `softmax_gradient_step` function is O(n * (2n-1)) per call because it computes slopes for all n coordinates. For n=425, that's about 360,000 operations per gradient step. With 50 inner iterations per outer, and 50 outer iterations, that's 50*50*360000 = 900,000,000 operations. That might be too slow.

Let me optimize: instead of computing slopes for each j separately, I can compute the full gradient more efficiently.

The gradient of f_k(h) with respect to h[j] is:
∂f_k/∂h[j] = (1-h[j+s]) * [j valid as first term] - h[j-s] * [j valid as second term]

where s = k - (n-1).

For a fixed k (shift s), the gradient vector is:
grad_k[j] = (1-h[j+s]) for j = 0 to n-1-s (j appears as first term)
           - h[j-s] for j = s to n-1 (j appears as second term)

So grad_k = (1 - h[s:]) - h[:-s] (padded with zeros) for s > 0
Or: grad_k[j] = (1-h[j+s]) - h[j-s] where both terms are defined, with boundary handling.

Actually, let me think about this differently. For the full correlate:
corr[p] = ∑_i h[i] * (1-h[i+p-(n-1)])

Let s = p - (n-1). Then:
corr[p] = ∑_i h[i] * (1-h[i+s]) for valid i (where 0 ≤ i < n and 0 ≤ i+s < n)

For s ≥ 0: valid i is 0 ≤ i ≤ n-1-s
For s < 0: valid i is -s ≤ i ≤ n-1

∂corr[p]/∂h[j]:
- j appears as h[i] when i = j: contributes (1-h[j+s]) if j is valid (0 ≤ j < n and 0 ≤ j+s < n)
- j appears as h[i+s] when i = j-s: contributes -h[j-s] if j-s is valid (0 ≤ j-s < n and 0 ≤ (j-s)+s < n = 0 ≤ j < n)

So:
∂corr[p]/∂h[j] = (1-h[j+s]) * [0 ≤ j < n and 0 ≤ j+s < n] - h[j-s] * [0 ≤ j-s < n and 0 ≤ j < n]

The second condition simplifies to 0 ≤ j < n and 0 ≤ j-s < n, which is 0 ≤ j < n and s ≤ j < n+s.

For s ≥ 0: second term valid when s ≤ j < n
For s < 0: second term valid when 0 ≤ j < n+s = n-|s|

First term valid when 0 ≤ j < n and 0 ≤ j+s < n:
For s ≥ 0: 0 ≤ j ≤ n-1-s
For s < 0: -s ≤ j ≤ n-1

So the full gradient of ∑_p w_p corr[p] with respect to h[j] is:
∑_p w_p [(1-h[j+s_p]) * valid1_p - h[j-s_p] * valid2_p]

where s_p = p - (n-1).

This can be computed efficiently using convolution:
grad[j] = ∑_p w_p (1-h[j+s_p]) * [valid1] - ∑_p w_p h[j-s_p] * [valid2]

The first term is like a convolution of w with (1-h), and the second is like a convolution of w with h.

Actually, let me think about it as:
grad = conv(w, 1-h) - conv(w, h)  (with appropriate boundary handling)

But the boundary handling is tricky because the "valid" conditions depend on s.

Let me just use a loop over p (shifts) instead:

```python
def compute_full_gradient(h, w):
    """Compute ∑_p w[p] * ∂corr[p]/∂h[j] for all j."""
    n = len(h)
    m = 2*n - 1
    grad = np.zeros(n)
    for p in range(m):
        if w[p] < 1e-15:
            continue
        s = p - (n-1)
        # First term: (1-h[j+s]) for valid j
        # j+s must be in [0, n-1], j in [0, n-1]
        # So j in [max(0, -s), min(n-1, n-1-s)]
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w[p] * (1.0 - h[j_lo+s:j_hi+s+1])
        # Second term: -h[j-s] for valid j
        # j-s must be in [0, n-1], j in [0, n-1]
        # So j in [max(0, s), min(n-1, n-1+s)]
        j_lo2 = max(0, s)
        j_hi2 = min(n-1, n-1+s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= w[p] * h[j_lo2-s:j_hi2-s+1]
    return grad
```

This is O(n * m) in the worst case (when all w[p] > 0), which is O(n²). For n=425, that's about 180,000 operations. Much better than the O(n * m) per coordinate approach.

But wait, in the softmax, all w[p] > 0 (just very small for most). So we'd need to iterate over all m = 2n-1 = 849 shifts. That's 849 * 425 ≈ 360,000 operations per gradient evaluation. That's fine.

Actually, I can optimize further: for the softmax with large β, only a few w[p] are significant. I can truncate:

```python
def compute_full_gradient_truncated(h, w, threshold=1e-6):
    n = len(h)
    m = 2*n - 1
    grad = np.zeros(n)
    significant = np.where(w > threshold * w.max())[0]
    for p in significant:
        s = p - (n-1)
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w[p] * (1.0 - h[j_lo+s:j_hi+s+1])
        j_lo2 = max(0, s)
        j_hi2 = min(n-1, n-1+s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= w[p] * h[j_lo2-s:j_hi2-s+1]
    return grad
```

This is much faster when β is large (only a few significant w[p]).

OK, let me now put together the full algorithm. I'll focus on making it different from the previous approach in a meaningful way.

**Final Algorithm Design:**

The key differences from the previous code:
1. **Softmax gradient descent** (new): Instead of SLSQP, use a custom gradient-based optimizer with softmax approximation to the max
2. **Active-set sub-gradient** (new): Direct sub-gradient steps using the active constraints
3. **Structured perturbations** (improved): Instead of pure random, use "swap" and "block" perturbations
4. **Adaptive resolution** (improved): More systematically try different resolutions
5. **Two-scale approach** (new): First optimize at low resolution (fast), then refine at high resolution

def project_sum(h, target):

    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])

def optimize_coordinate(h, j, corr):

    h = h.copy()

    t0 = time.time()

def compute_weighted_gradient(h, weights):
    """Compute ∑_p weights[p] * ∂corr[p]/∂h for all h[j]."""
    n = len(h)
    m = 2*n - 1
    grad = np.zeros(n)
    for p in range(m):
        wp = weights[p]
        if wp < 1e-10:
            continue
        s = p - (n-1)
        # First term: (1-h[j+s]) where j and j+s both in [0,n-1]
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wp * (1.0 - h[j_lo+s:j_hi+s+1])
        # Second term: -h[j-s] where j and j-s both in [0,n-1]
        j_lo2 = max(0, s)
        j_hi2 = min(n-1, n-1+s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wp * h[j_lo2-s:j_hi2-s+1]
    return grad

def subgradient_step(h, step_size, active_frac=0.1):
    """Take a sub-gradient step using the top active constraints."""
    n = len(h)
    m = 2*n - 1
    corr = compute_correlations(h)
    C5 = np.max(corr)
    # Find active constraints (top fraction)
    threshold = C5 * (1.0 - active_frac)
    active = np.where(corr >= threshold)[0]
    # Uniform weights on active set
    w = np.zeros(m)
    w[active] = 1.0 / len(active)
    # Compute gradient
    grad = compute_weighted_gradient(h, w)
    # Step
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    return h_new

def softmax_optimizer(h, n_steps=200, time_budget=100.0):
    """Optimize using softmax gradient descent with adaptive parameters."""
    t0 = time.time()
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    beta = 50.0
    lr = 0.02
    
    for step in range(n_steps):
        if time.time() - t0 > time_budget:
            break
        corr = compute_correlations(h)
        # Softmax weights
        shifted = corr - corr.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        # Gradient
        grad = compute_weighted_gradient(h, w)
        # Gradient step with backtracking
        for trial in range(10):
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum(h_new, n / 2.0)
            s = evaluate_score(h_new)
            if s <= best_score + 0.001:  # Allow small worsening
                h = h_new
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-6)
        
        # Periodic CD
        if step % 20 == 19:
            tb = min(3.0, time_budget - (time.time() - t0))
            if tb > 1.0:
                h = coordinate_descent(h, n_sweeps=30, time_budget=tb)
        
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
        
        # Adapt beta (increase over time for sharper max approximation)
        beta *= 1.02
        # Adapt lr (decrease over time)
        lr = max(0.02 * (1.0 - step / n_steps), 0.001)
    
    return best_h, best_score
