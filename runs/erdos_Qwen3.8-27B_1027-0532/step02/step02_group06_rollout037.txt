```python
    """Solve min grad·h s.t. 0<=h_i<=1, sum(h)=target_sum"""
    sorted_idx = np.argsort(grad)
    n = len(grad)
    target = target_sum
    # Greedy: take the 'target' smallest gradient values
    # But we need exactly 'target' units of mass
    # Since h_i ∈ [0,1], we take full 1 for the smallest (target) values
    # If target is not integer, we need a fractional one
    h_new = np.zeros(len(grad))
    # Sort indices by gradient value
    sorted_idx = np.argsort(grad)
    n_full = int(target)
    frac = target - n_full
    h_new[sorted_idx[:n_full]] = 1.0
    if frac > 0:
        h_new[sorted_idx[n_full]] = frac
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_of_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] -= h[:-lag]
        grad[:n-lag] += (1.0 - h[lag:])
    else:
        a = -lag
        grad[:n-a] -= h[a:]
        grad[a:] += (1.0 - h[:n-a])
    return grad

def subgradient_optimize(h, alpha, n_top, step_init, seed, t0, time_limit):
    n = len(h)
    target_sum = n / 2.0
    h = project_to_feasible(h.copy(), target_sum)
    rng = np.random.RandomState(seed)
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    alpha_current = alpha
    
    while time.time() - t0 < t0 + time_limit:
        corr = compute_correlations(h)
        m = np.max(corr)
        # Soft-max weights
        shifted = corr - m
        weights = np.exp(alpha_current * shifted)
        w_sum = np.sum(weights)
        weights /= w_sum
        
        # Only use top n_top
        top_idx = np.argsort(-corr)[:n_top]
        weights_full = np.zeros_like(corr)
        top_w = weights[top_idx]
        top_w /= top_w.sum()
        weights_full[top_idx] = top_w
        
        # Compute gradient

        grad = np.zeros(n)

            grad += top_w[np.searchsorted(top_idx, idx)] * gradient_of_lag(h, lag)
```

```python
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 0.1:
            lam += 0.5 * (current_sum - target_sum) / (target_sum + 1)
            lam = max(lam, 0.0)
        
        # Project
        h = project_to_feasible(h, target_sum)
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        if not changed:
            break
    
    return best_h, best_score
```

But the `_eval_objective_at` function would need to compute the full correlation each time, which is O(n). With 20 ternary steps per coordinate and n coordinates, that's O(20 * n²) per pass. For n=425, that's 20 * 425² ≈ 3.6M per pass. With 20 passes: 72M. That's reasonable.

But actually, I can optimize the evaluation. When I change h[i] to a new value t (keeping others fixed), I only need to update the correlations that involve h[i]. Let me precompute the "base" correlations (with h[i] at its current value) and then adjust.

Actually, for simplicity and given the time budget, let me just use the full evaluation. 72M operations should take maybe 1-2 seconds.

Hmm wait, each "full evaluation" is:
1. Change h[i] to t
2. Compute np.correlate(h, 1-h, 'full'): O(n) (actually it's a loop of n multiplications, so O(n) in practice but with numpy overhead)
3. Take max: O(n)

So each evaluation is O(n) with a constant factor. With 20 * n = 8500 evaluations per pass, each O(n=425), that's 8500 * 425 ≈ 3.6M multiply-adds per pass. Very fast.

But np.correlate has overhead (function call, memory allocation). Let me estimate: maybe 10μs per call? Then 8500 * 20 = 170K calls * 10μs = 1.7 seconds per pass. With 20 passes: 34 seconds. That's fine for a 1300s budget.

1. Subgradient (improved from before)
2. Coordinate descent with Lagrangian

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_of_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] -= h[:-lag]
        grad[:n-lag] += (1.0 - h[lag:])
    else:
        a = -lag
        grad[:n-a] -= h[a:]
        grad[a:] += (1.0 - h[:n-a])
    return grad

def subgradient_optimize(h0, alpha, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    alpha_cur = alpha
    
    while time.time() - t0 < t0 + time_limit:
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = corr - m
        weights = np.exp(alpha_cur * shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(corr)
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= wsum
        
        n_top_actual = min(n_top, len(corr))
        top_idx = np.argsort(-corr)[:n_top_actual]
        top_w = weights[top_idx].copy()
        tw = top_w.sum()
        if tw < 1e-300:
            top_w = np.zeros(n_top_actual)
            top_w[0] = 1.0
        else:
            top_w /= tw
        
        grad = np.zeros(n)
        for j in range(n_top_actual):
            lag = int(top_idx[j] - (n - 1))
            grad += top_w[j] * gradient_of_lag(h, lag)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        improved = False
        for trial in range(30):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.35
        
        if not improved:
            stall += 1
            if stall > 30:
                h = best_h + rng.randn(n) * 0.015 / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init * (0.3 + 0.7 * rng.rand())
                stall = 0
            elif stall > 150:
                h = best_h + rng.randn(n) * 0.04 / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init
                stall = 0
        
        alpha_cur = min(alpha_cur * 1.003, alpha * 200.0)
    
    return best_h, best_score

def coordinate_descent(h0, t0, time_limit, max_passes=10):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for path in range(max_passes):
        if time.time() - t0 > t0 + time_limit:
            break

            if time.time() - t0 > t0 + time_limit:

            lo_t, hi_t = 0.0, 1.0
            
            # Evaluate at endpoints

        h = project_to_feasible(h, target_sum)

            best_score = score
            best_h = h.copy()
        else:
            # Revert if worse
            h = best_h.copy()
    
    return best_h, best_score
