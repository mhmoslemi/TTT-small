def binary_local_search(h, max_rounds, rng, tabu_len=100, time_limit=300, start_time=0):
    n = len(h)

    current_max = np.max(corr)

            tabu_until[i] = round + tabu_len
            current_max = best_new_max

                best_h = h.copy()
        else:

            all_positions = np.concatenate([ones, zeros])
            i = all_positions[rng.integers(len(all_positions))]
            delta = -1.0 if h[i] == 1.0 else 1.0
