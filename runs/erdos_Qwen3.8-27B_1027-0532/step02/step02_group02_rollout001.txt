            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-15:
                break
    return h

def _grad_softmax(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    offset = n - 1
    for idx in range(len(corr)):
        if w[idx] < 1e-10:
            continue
        lag = idx - offset
        wt = w[idx]
        if lag >= 0:
            if lag < n:
                grad[lag:] += wt * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= wt * h[lag:]
        else:
            L = -lag
            if L < n:
                grad[:n - L] += wt * (1.0 - h[L:])
                grad[L:] -= wt * h[:n - L]
    return grad, corr

def _constraint_jac(x):
    return np.ones_like(x)

def _optimize_phase(h_init, target_sum, n_iter, alpha, lr, deadline, check_every=30):
    n = len(h_init)
    h = h_init.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0

    for t in range(n_iter):
        if time.time() > deadline - 5:
            break

        grad, corr = _grad_softmax(h, alpha)
        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        m_hat = m_v / (1.0 - b1 ** (t + 1))
        v_hat = v_v / (1.0 - b2 ** (t + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = _project(h - step, target_sum)

        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > check_every * 25:
                break

    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_polish(h_init, target_sum, alpha, maxiter, deadline):
    n = len(h_init)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

    def obj_grad(x):
        g, _ = _grad_softmax(x, alpha)
        return g * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _make_sum_func(target_sum), 'jac': _constraint_jac}]
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h_init.copy()

def _make_sum_func(target_sum):
    def f(x):
        return x.sum() - target_sum
    return f

def _make_inits(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    inits = []
    inits.append(np.full(n, 0.5))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x + rng.rand() * 6.2832))
    f1 = rng.uniform(1.5, 4.0)
    f2 = rng.uniform(5.0, 12.0)
    inits.append(0.5 + 0.2 * np.sin(2.0 * np.pi * f1 * x) + 0.1 * np.sin(2.0 * np.pi * f2 * x))
    inits.append(0.5 + 0.25 * (rng.rand(n) - 0.5))
    period = max(2, n // rng.randint(4, 9))
    hb = np.zeros(n)
    for i in range(n):
        hb[i] = 1.0 if (i % (2 * period)) < period else 0.0
    inits.append(0.5 + 0.4 * (hb - 0.5))
    inits.append(0.5 + 0.12 * rng.randn(n))
    return inits

def _multi_phase_optimize(h0, target_sum, deadline):
    h1, c5_1 = _optimize_phase(h0, target_sum, 2500, alpha=25, lr=0.05, deadline=deadline)
    if time.time() > deadline - 20:
        return h1, c5_1
    h2, c5_2 = _optimize_phase(h1, target_sum, 2500, alpha=80, lr=0.02, deadline=deadline)
    if time.time() > deadline - 15:
        return h2, c5_2
    h3, c5_3 = _optimize_phase(h2, target_sum, 3000, alpha=300, lr=0.007, deadline=deadline)
    if time.time() > deadline - 10:
        return h3, c5_3
    h4, c5_4 = _optimize_phase(h3, target_sum, 3000, alpha=1000, lr=0.003, deadline=deadline)
    return h4, c5_4

def _perturb_restart(h_best, target_sum, seed, deadline):
    rng = np.random.RandomState(seed)
    best = h_best.copy()
    best_c5 = _c5_of(h_best)
    for attempt in range(6):
        if time.time() > deadline - 15:
            break
        scale = 0.03 + 0.03 * rng.rand()
        h_p = _project(h_best + scale * rng.randn(len(h_best)), target_sum)
        h_r, c5_r = _multi_phase_optimize(h_p, target_sum, deadline)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best = h_r.copy()
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30

    best_h = None
    best_c5 = np.inf
    best_n = 0

    n_list = [98, 100, 102, 106, 110]

    for n in n_list:
        if time.time() > deadline - 150:
            break
        target_sum = n / 2.0

        h0_list = []

        if n == 100 and 'initial_h_values' in globals() and initial_h_values is not None:
            h0 = np.array(initial_h_values, dtype=float).copy()
            h0 = _project(h0, target_sum)
            h0_list.append(h0)

        init_list = _make_inits(n, seed)
        if n > 100:
            init_list = init_list[:4]

        for h0_raw in init_list:
            if time.time() > deadline - 80:
                break
            h0 = _project(h0_raw, target_sum)
            h_res, c5_res = _multi_phase_optimize(h0, target_sum, deadline)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n

        if best_n == n and best_c5 < np.inf and time.time() < deadline - 40:
            h_pol = _slsqp_polish(best_h, best_n / 2.0, alpha=1500, maxiter=200, deadline=deadline)
            h_pol = _project(h_pol, best_n / 2.0)
            c5_pol = _c5_of(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if best_h is not None and time.time() < deadline - 60:
        best_h, best_c5 = _perturb_restart(best_h, best_n / 2.0, seed + 7777, deadline)

    if best_h is not None and time.time() < deadline - 25:
        for alpha in [3000, 8000]:
            if time.time() > deadline - 15:
                break
            h_pol = _slsqp_polish(best_h, best_n / 2.0, alpha, maxiter=100, deadline=deadline)
            h_pol = _project(h_pol, best_n / 2.0)
            c5_pol = _c5_of(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _project(best_h, n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
