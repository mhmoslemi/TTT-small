import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_val(x, alpha):
    m = np.max(x)
    shifted = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def project_feasible(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_h_sm(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[:N-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:N-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:N-kp])
            grad[:N-kp] -= w * h[kp:]
    return grad

def obj_ap(g, alpha):
    h = antipal_h(g)
    return smooth_max_val(compute_correlations(h), alpha)

def jac_ap(g, alpha):
    h = antipal_h(g)
    gh = grad_h_sm(h, alpha)
    half = len(g)
    N = 2 * half + 1
    out = gh[:half].copy()
    for i in range(half):
        out[i] -= gh[N - 1 - i]
    return out

    bounds = [(0.0, 1.0)] * half
    try:
        res = minimize(obj_ap, g0, method='L-BFGS-B', bounds=bounds,
                       jac=jac_ap, args=(alpha,),
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except:
        return np.clip(g0, 0.0, 1.0)

def obj_full(h, alpha, mu):
    corr = compute_correlations(h)
    s = smooth_max_val(corr, alpha)
    dev = float(np.sum(h)) - len(h) / 2.0
    return s + mu * dev * dev

def jac_full(h, alpha, mu):
    gh = grad_h_sm(h, alpha)
    dev = float(np.sum(h)) - len(h) / 2.0
    return gh + 2.0 * mu * dev * np.ones(len(h))

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_val(x, alpha):
    m = np.max(x)
    shifted = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def project_feasible(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def grad_h_sm(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[:N - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:N - k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:N - kp])
            grad[:N - kp] -= w * h[kp:]
    return grad


def obj_ap(g, alpha):
    h = antipal_h(g)
    return smooth_max_val(compute_correlations(h), alpha)


def jac_ap(g, alpha):
    h = antipal_h(g)
    gh = grad_h_sm(h, alpha)
    half = len(g)
    N = 2 * half + 1
    out = gh[:half].copy()
    for i in range(half):
        out[i] -= gh[N - 1 - i]
    return out


def lfbgs_ap(g0, alpha, maxiter=300):
    half = len(g0)
    bounds = [(0.0, 1.0)] * half
    try:
        res = minimize(obj_ap, g0, method='L-BFGS-B', bounds=bounds,
                       jac=jac_ap, args=(alpha,),
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return np.clip(g0, 0.0, 1.0)


def obj_full_pen(h, alpha, mu):
    corr = compute_correlations(h)
    s = smooth_max_val(corr, alpha)
    dev = float(np.sum(h)) - len(h) / 2.0
    return s + mu * dev * dev


def jac_full_pen(h, alpha, mu):
    gh = grad_h_sm(h, alpha)
    dev = float(np.sum(h)) - len(h) / 2.0
    return gh + 2.0 * mu * dev * np.ones(len(h))


def lfbgs_full(h0, alpha, mu, maxiter=200):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    try:
        res = minimize(obj_full_pen, h0, method='L-BFGS-B', bounds=bounds,
                       jac=jac_full_pen, args=(alpha, mu),
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return np.clip(h0, 0.0, 1.0)


def coord_descent(h, n_sweeps=5):
    N = len(h)
    target = N / 2.0
    h = h.copy()
    for sweep in range(n_sweeps):
        for i in range(N):
            old_val = h[i]
            h[i] = 0.0
            corr_0 = compute_correlations(h)
            h[i] = 1.0
            corr_1 = compute_correlations(h)
            h[i] = old_val
            a = corr_1 - corr_0
            b = corr_0
            best_t = old_val
            best_score = float(np.max(a * old_val + b))
            for t_try in np.array([0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3,
                                   0.35, 0.4, 0.45, 0.5, 0.55, 0.6,
                                   0.65, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0]):
                sc = float(np.max(a * t_try + b))
                if sc < best_score:
                    best_score = sc
                    best_t = t_try
            h[i] = best_t
    h = project_feasible(h, target)
    return h
