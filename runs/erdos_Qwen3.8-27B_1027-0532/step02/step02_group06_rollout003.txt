def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    ew = np.exp(shifted)
    weights = ew / ew.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        m = k - (n - 1)
        p = abs(m)
        if p == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    
    return grad


def smooth_max_value(h, alpha):
    corr = compute_correlations(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha


def lbbfgs_run(h0, alpha, lam, maxiter, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        val = smooth_max_value(x, alpha)
        cv = (np.sum(x) - target_sum) ** 2
        return val + lam * cv
    
    def gradient(x):
        g = gradient_smooth_max(x, alpha)
        g += 2.0 * lam * (np.sum(x) - target_sum)
        return g
    
    try:
        res = minimize(objective, h0, method='L-BFGS-B', jac=gradient,
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-6, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        pass
    return h0
