def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    sm = np.max(shifted)
    exp_s = np.exp(shifted - sm)
    s = np.sum(exp_s)
    return m, exp_s / s

def weighted_grad(h, w):
    n = len(h)
    t1 = np.convolve(w[:n], 1.0 - h, mode='full')[:n]
    t2 = np.convolve(w[:n], h[::-1], mode='full')[n-1::-1]
    return t1 - t2

        h = hn

    def obj(v):

        mx = np.max(c)
        sh = alpha * (c - mx)
        return mx + np.log(np.sum(np.exp(sh - np.max(sh)))) / alpha

        return np.sum(v) - ts

        return np.ones(n)

                   constraints=[{'type': 'eq', 'fun': con, 'jac': coni}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

    return None
