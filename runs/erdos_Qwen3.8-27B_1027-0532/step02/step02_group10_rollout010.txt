        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        # Only adjust non-boundary elements
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx, n, top_k=100):
    """Compute gradient using only top-K shifts by weight."""
    grad = np.zeros(n)
    # Get top K shifts
    if len(weights) > top_k:
        idx = np.argpartition(-weights, top_k)[:top_k]
        idx = idx[weights[idx] > 1e-12]
    else:
        idx = np.arange(len(weights))
    
    center = n - 1
    for m_idx in idx:
        w = weights[m_idx]
        offset = m_idx - center
        # d/dh[i] corr[offset] = (1-h[i-offset]) - h[i+offset]
        # where i-offset and i+offset must be in [0, n-1]
        # i ranges from max(0, -offset) to min(n-1, n-1+offset) for the first term
        # and from max(0, offset) to min(n-1, n-1-offset) for the second term
        
        # First term: (1-h[i-offset]) for valid i-offset
        # i-offset in [0, n-1] => i in [offset, offset+n-1] => i in [max(0,offset), min(n-1, offset+n-1)] = [max(0,offset), n-1]
        lo1 = max(0, offset)
        hi1 = n  # exclusive
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - offset:hi1 - offset])
        
        # Second term: -h[i+offset] for valid i+offset
        # i+offset in [0, n-1] => i in [-offset, n-1-offset] => i in [max(0,-offset), min(n-1, n-1-offset)]
        lo2 = max(0, -offset)
        hi2 = min(n, n - offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + offset:hi2 + offset]
    
    grad *= dx
    return grad

def adam_optimize(h, alpha, target, max_iters, deadline, top_k=100, lr=0.01):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    prev_sm = None
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        overlaps = compute_overlaps(x)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        
        if prev_sm is not None:
            if sm < prev_sm - 1e-12:
                stall = 0
            else:
                stall += 1
            if stall > 100:
                break
        prev_sm = sm
        
        grad = compute_gradient(x, weights, dx, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_onto_constraint(x_new, target)
        
        # Check if it improved
        ov_new = compute_overlaps(x_new)
        sm_new, _ = smooth_max_and_weights(ov_new, alpha)
        
        if sm_new < sm:
            x = x_new
        else:
            # Line search: reduce step
            reduced = False
            for factor in [0.5, 0.25, 0.1]:
                x_trial = x - factor * lr * m_hat / (np.sqrt(v_hat) + eps)
                x_trial = project_onto_constraint(x_trial, target)
                ov_t = compute_overlaps(x_trial)
                sm_t, _ = smooth_max_and_weights(ov_t, alpha)
                if sm_t < sm:
                    x = x_trial
                    reduced = True
                    break
            if not reduced:
                break
    
    return x

def pgd_optimize(h, alpha, target, max_iters, deadline, top_k=100):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    step = min(0.5, 1.0 / (alpha * dx * 3.0 + 1.0))
    prev_sm = None
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(x)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        
        if prev_sm is not None:
            if sm < prev_sm - 1e-12:
                stall = 0
            else:
                stall += 1
            if stall > 150:
                break
        prev_sm = sm
        
        grad = compute_gradient(x, weights, dx, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        improved = False
        s = step
        for bt in range(30):
            x_trial = project_onto_constraint(x - s * grad, target)
            ov_t = compute_overlaps(x_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-14:
                x = x_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = s
            break
        step = min(s * 1.3, 1.0)
    return x

def binary_push(h, target, rng, frac=0.3):
    """Push h values toward 0 or 1 with some probability."""
    n = len(h)
    h_new = h.copy()
    mask = (h > 0.1) & (h < 0.9)
    if mask.sum() > 0:
        # Push some values toward 0 or 1
        n_push = max(1, int(frac * mask.sum()))
        idx = np.where(mask)[0]
        rng.shuffle(idx)
        for i in idx[:n_push]:
            if h[i] > 0.5:
                h_new[i] = min(1.0, h[i] + 0.3)
            else:
                h_new[i] = max(0.0, h[i] - 0.3)
    return project_onto_constraint(h_new, target)
