import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13 * n:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def active_set_weights(x, K=10, soft_alpha=50):
    """Get weights for top-K active set with soft weighting"""
    n = len(x)
    top_k = np.argpartition(x, -K)[-K:]
    active = x[top_k]
    a_max = np.max(active)
    shifted = soft_alpha * (active - a_max)
    shifted = np.clip(shifted, -50, 0)
    exps = np.exp(shifted)
    weights_active = exps / np.sum(exps)
    weights = np.zeros(n)
    weights[top_k] = weights_active
    return a_max + np.log(weights_active.sum() / K) / soft_alpha, weights

    """Compute sum_k weights[k] * d(overlap_k)/d(h)"""
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        if offset == 0:

            # d/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
            grad[:] += w * (1.0 - 2.0 * h)
        elif offset > 0:
            # overlap_k = sum_i h[i] * (1-h[i+k]) for i where i+k < n
            # d/dh[j] = (1-h[j+offset]) [from first term, i=j] 
            #          - h[j-offset] [from second term, i+k=j, so i=j-offset]
            i_lo = max(0, offset)

import numpy as np
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        adj = diff / mask.sum()
        h[mask] += adj
        np.clip(h, 0.0, 1.0, out=h)
    return h


def get_weights_smooth(overlaps, alpha):
    m = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - m), -60.0, 0.0)
    exps = np.exp(shifted)
    total = exps.sum()
    if total < 1e-30:
        w = np.zeros(len(overlaps))
        w[np.argmax(overlaps)] = 1.0
        return m, w
    w = exps / total
    return m, w


def get_weights_active(overlaps, K, soft_alpha=50.0):
    n = len(overlaps)
    K = min(K, n)
    if K == 1:
        idx = np.argmax(overlaps)
        w = np.zeros(n)
        w[idx] = 1.0
        return overlaps[idx], w
    top_idx = np.argpartition(overlaps, -K)[-K:]
    active_vals = overlaps[top_idx]
    a_max = np.max(active_vals)
    shifted = np.clip(soft_alpha * (active_vals - a_max), -60.0, 0.0)
    exps = np.exp(shifted)
    total = exps.sum()
    w = np.zeros(n)
    w[top_idx] = exps / total
    return a_max, w

            if r1_end > 0:
                grad[0:r1_end] += w * (1.0 - h[o:o + r1_end])
            r2_start = o
            r2_end = n - o
            if r2_end > r2_start:
                grad[r2_start:r2_end] += w * ((1.0 - h[r2_start + o:r2_end + o]) - h[r2_start - o:r2_end - o])
            r3_start = n - o
            if r3_start < n:
                grad[r3_start:] += w * (-h[r3_start - o:])
        else:
            k = -offset
            if k >= n:
                continue
            r1_end = min(k, n - k)
            if r1_end > 0:
                grad[0:r1_end] += w * (-h[k:k + r1_end])
            r2_start = k

def optimize_phase(h, alpha, target, deadline, use_active, K, max_iters, step_init):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = step_init
    prev_obj = np.inf
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        obj = np.max(overlaps)
        if use_active:
            _, weights = get_weights_active(overlaps, K, soft_alpha=alpha)
        else:
            _, weights = get_weights_smooth(overlaps, alpha)
        if obj < prev_obj - 1e-13:
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
        prev_obj = obj
        grad = compute_grad(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        improved = False
        s = step
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            if np.max(ov_t) < obj - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-15:
                break
        if not improved:
            step = s * 0.5
            if step < 1e-16:
                break
        else:
            step = min(step * 1.25, 2.0 / (n * dx + 1))
    return h


def upward_interpolate(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
