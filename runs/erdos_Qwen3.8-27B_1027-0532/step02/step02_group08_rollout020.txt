import numpy as np
from scipy.optimize import minimize
import time

def project_sum(h, target):
    """Project h to have sum = target, keeping in [0,1]."""
    n = len(h)
    h = h.copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad

def penalty_obj_grad(h, alpha, mu):
    obj, grad = smooth_max_obj_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad

def coord_descent_sweep(h, lam, order=None):
    """One sweep of coordinate descent with Lagrange multiplier lam for sum constraint."""
    n = len(h)
    h = h.copy()
    if order is None:
        order = np.arange(n)
    
    # Compute initial correlations
    c = np.correlate(h, 1.0 - h, mode='full')
    
    for j in order:
        # Compute coefficients a[idx] for all idx
        a = np.zeros(2 * n - 1)
        
        # For k >= 0, idx = k + n - 1
        # a[k+n-1] = (1-h[j-k]) if k <= j, plus -h[j+k] if j+k < n
        ks = np.arange(n)
        # First term: (1-h[j-k]) for k <= j
        mask1 = ks <= j
        a[mask1 + n - 1] += (1.0 - h[j - ks[mask1]])
        # Second term: -h[j+k] for j+k < n
        mask2 = (j + ks) < n
        a[mask2 + n - 1] += -h[j + ks[mask2]]
        
        # For k < 0, m = -k >= 1, idx = n - 1 - m
        ms = np.arange(1, n)
        # First term: (1-h[j+m]) for j+m < n
        mask3 = (j + ms) < n
        a[n - 1 - ms[mask3]] += (1.0 - h[j + ms[mask3]])
        # Second term: -h[j-m] for j >= m
        mask4 = j >= ms
        a[n - 1 - ms[mask4]] += -h[j - ms[mask4]]
        
        # b[idx] = c[idx] - a[idx] * h[j]
        b = c - a * h[j]
        
        # Minimize max_k (a_k * x + b_k) + lam * x = max_k ((a_k + lam) * x + b_k) over [0,1]
        d = a + lam
        
        # Check boundary x=0
        f0 = np.max(b)
        active0 = b >= f0 - 1e-14
        right_deriv0 = np.max(d[active0])
        
        # Check boundary x=1
        f1 = np.max(d + b)
        active1 = (d + b) >= f1 - 1e-14
        left_deriv1 = np.min(d[active1])
        
        if right_deriv0 >= 0:
            x_star = 0.0
        elif left_deriv1 <= 0:
            x_star = 1.0
        else:
            # Bisection
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = d * mid + b
                fmid = np.max(vals)
                active = vals >= fmid - 1e-12
                deriv_max = np.max(d[active])
                deriv_min = np.min(d[active])
                if deriv_min >= 0:
                    hi = mid
                elif deriv_max <= 0:
                    lo = mid
                else:
                    # We're at a break point, this is the minimum
                    lo = hi = mid
                    break
            x_star = (lo + hi) / 2.0
        
        # Update
        delta = x_star - h[j]
        if abs(delta) > 1e-15:
            c += a * delta
            h[j] = x_star
            # Clip to [0,1]
        h[j] = np.clip(x_star, 0.0, 1.0)
    
    return h

import numpy as np
from scipy.optimize import minimize
import time


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def penalty_obj_grad(h, alpha, mu):
    obj, grad = smooth_max_obj_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad


def coord_descent_sweep(h, lam, order):
    n = len(h)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    ks = np.arange(n)
    ms = np.arange(1, n)
    for j in order:
        a = np.zeros(2 * n - 1)
        mask1 = ks <= j
        a[mask1 + n - 1] += (1.0 - h[j - ks[mask1]])
        mask2 = (j + ks) < n
        a[mask2 + n - 1] += -h[j + ks[mask2]]
        mask3 = (j + ms) < n
        a[n - 1 - ms[mask3]] += (1.0 - h[j + ms[mask3]])
        mask4 = j >= ms
        a[n - 1 - ms[mask4]] += -h[j - ms[mask4]]
        b = c - a * h[j]
        d = a + lam
        f0 = np.max(b)
        active0 = b >= f0 - 1e-14
        right_deriv0 = np.max(d[active0])
        f1 = np.max(d + b)
        active1 = (d + b) >= f1 - 1e-14
        left_deriv1 = np.min(d[active1])
        if right_deriv0 >= 0:
            x_star = 0.0
        elif left_deriv1 <= 0:
            x_star = 1.0
        else:
            lo = 0.0
            hi = 1.0
            for _ in range(35):
                mid = (lo + hi) / 2.0
                vals = d * mid + b
                fmid = np.max(vals)
                active = vals >= fmid - 1e-12
                dmax = np.max(d[active])
                dmin = np.min(d[active])
                if dmin >= 0:
                    hi = mid
                elif dmax <= 0:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            x_star = (lo + hi) / 2.0
        delta = x_star - h[j]
        if abs(delta) > 1e-15:
            c += a * delta
        h[j] = np.clip(x_star, 0.0, 1.0)
    return h

def slsqp_fun(h, alpha):
    return smooth_max_obj_grad(h, alpha)[0]


def slsqp_jac(h, alpha):
    return smooth_max_obj_grad(h, alpha)[1]


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_fun, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x


def lbfgsb_optimize(h0, alpha, mu, maxiter, bounds):
    res = minimize(
        penalty_obj_grad, h0, method='L-BFGS-B',

        jac=True,

                 'maxls': 50, 'maxfun': maxiter * 20}
    )
    return res.x
