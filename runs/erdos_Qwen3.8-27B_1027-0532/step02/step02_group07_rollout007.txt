def subgradient_optimize(h, time_budget, rng, n_iters=10000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum parameter
    t0 = time.time()
    
    # Find the set of maximizing lags
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        
        corr = compute_correlations(h)
        max_val = np.max(corr)
        # Find all lag positions where corr is close to max
        eps = 1e-8
        max_mask = corr >= max_val - eps
        k_star = np.where(max_mask)[0]
        
        if len(k_star) == 0:
            k_star = np.array([np.argmax(corr)])
        
        # Compute subgradient (mixture over maximizing lags)
        subgrad = np.zeros(n)
        for k in k_star:
            m = k - (n - 1)
            # ∂corr[k]/∂h[j] = (1-h[j+m]) * I[valid1] - h[j-m] * I[valid2]
            j = np.arange(n)
            valid1 = (j + m >= 0) & (j + m < n)
            valid2 = (j - m >= 0) & (j - m < n)
            grad = np.zeros(n)
            grad[valid1] += (1.0 - h[j[valid1] + m])
            if np.any(valid2):
                grad[valid2] -= h[j[valid2] - m]
            subgrad += grad
        subgrad /= len(k_star)
        subgrad *= dx  # Scale for the actual objective
        
        # Update with momentum
        v = beta * v + subgrad
        # Adaptive step size
        step = 1.0 / (it + 1) ** 0.5 * 0.5  # or some other schedule
        
        h_new = h - step * v
        h_new = project_sum(h_new, n / 2.0)
        
        # Check if improvement
        if evaluate_score(h_new) < evaluate_score(h):
            h = h_new
        # else keep old h (or accept with some probability for exploration)
    
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + log_sum / alpha


def smooth_obj(h, alpha):
    corr = compute_correlations(h)
    return smooth_max(corr, alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    for _ in range(50):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0, temperature=0.0, rng=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            if temperature > 0.0 and rng is not None and rng.random() < 0.05:
                t_rand = t_opt + rng.randn() * temperature
                t_rand = float(np.clip(t_rand, 0.0, 1.0))
                change = t_rand - h[j]
                new_corr = corr + change * slopes
                new_obj = float(np.max(new_corr))
                old_obj = float(np.max(corr))
                delta = new_obj - old_obj
                if delta <= 0 or rng.random() < np.exp(-delta / (temperature + 1e-15)):
                    h[j] = t_rand
                    corr = new_corr
                    max_change = max(max_change, abs(change))
                continue
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def subgradient_step(h, k_star, dx):
    n = len(h)
    subgrad = np.zeros(n)
    for k in k_star:
        m = k - (n - 1)
        j = np.arange(n)
        valid1 = (j + m >= 0) & (j + m < n)
        valid2 = (j - m >= 0) & (j - m < n)
        if np.any(valid1):
            subgrad[valid1] += (1.0 - h[j[valid1] + m])
        if np.any(valid2):
            subgrad[valid2] -= h[j[valid2] - m]
    subgrad /= len(k_star)
    subgrad *= dx
    return subgrad


def subgradient_optimize(h, time_budget, n_iters=5000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    beta = 0.7
    t0 = time.time()
    best_h = h.copy()
    best_score = evaluate_score(h)
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        corr = compute_correlations(h)
        max_val = np.max(corr)
        eps = 1e-10
        max_mask = corr >= max_val - eps
        k_star = np.where(max_mask)[0]
        if len(k_star) == 0:
            k_star = np.array([np.argmax(corr)])
        if len(k_star) > 20:
            top_idx = np.argsort(corr)[::-1][:20]
            k_star = top_idx
        subgrad = subgradient_step(h, k_star, dx)
        v = beta * v + subgrad
        step = 2.0 / (np.sqrt(it + 1) + 1.0)
        h_new = h - step * v
        h_new = project_sum(h_new, n / 2.0)
        s_new = evaluate_score(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
