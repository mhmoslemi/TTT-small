```python
def update_corr(corr, h_old, i, j, n):
    """Update correlation after h[i]: 1->0, h[j]: 0->1.
    h_old is the state BEFORE the change.
    Returns new correlation vector."""
    new_corr = corr.copy()
    for l in range(-(n-1), n):
        delta = 0.0
        # Terms from i (was 1, now 0)
        im = i - l
        ip = i + l
        if 0 <= im < n:
            delta += h_old[im] - 1.0  # h(i-l) - 1
        if 0 <= ip < n:
            delta += h_old[ip]  # h(i+l)
        # Terms from j (was 0, now 1)
        jm = j - l
        jp = j + l
        if 0 <= jm < n:
            delta -= h_old[jm]  # -h(j-l)
            delta += 1.0  # +1
        if 0 <= jp < n:
            delta -= h_old[jp]  # -h(j+l)
        new_corr[n-1+l] += delta
    return new_corr
```

```python
    h is the state BEFORE the change.
    Returns new correlation vector."""
    lags = np.arange(-(n-1), n)

    # Δ from i as first factor: h(i-l) - 1, valid if 0 ≤ i-l < n
    # Δ from i as second factor: h(i+l), valid if 0 ≤ i+l < n
    
    delta = np.zeros(2*n-1)
    
    # i as first factor: i-l in [0,n-1] means l in [i-(n-1), i]

    hi_i1 = min(n-1, i)
    l_range_i1 = np.arange(lo_i1, hi_i1+1)
    # For these l values, contribution is h[i-l] - 1
    delta[l_range_i1 + (n-1)] += h[i - l_range_i1] - 1.0
    
    # i as second factor: i+l in [0,n-1] means l in [-i, n-1-i]
    lo_i2 = max(-(n-1), -i)
    hi_i2 = min(n-1, n-1-i)
    l_range_i2 = np.arange(lo_i2, hi_i2+1)
    delta[l_range_i2 + (n-1)] += h[i + l_range_i2]
    
    # j as first factor (0->1): contribution is 1 - h(j-l), valid if 0 ≤ j-l < n
    lo_j1 = max(0, j-(n-1))
    hi_j1 = min(n-1, j)
    l_range_j1 = np.arange(lo_j1, hi_j1+1)
    delta[l_range_j1 + (n-1)] += 1.0 - h[j - l_range_j1]
    
    # j as second factor: contribution is -h(j+l), valid if 0 ≤ j+l < n
    lo_j2 = max(-(n-1), -j)
    hi_j2 = min(n-1, n-1-j)
    l_range_j2 = np.arange(lo_j2, hi_j2+1)
    delta[l_range_j2 + (n-1)] -= h[j + l_range_j2]
    
    return corr + delta
```

```python
    current_c5 = corr.max() * dx
    best_c5 = current_c5

        frac = elapsed / total_time
        T = T_start * (T_end / T_start) ** frac

            l2 = np.arange(l2_lo, l2_hi+1)
            delta[l2 + (n-1)] += h[i + l2]

        l3_lo = max(0, j-(n-1))
```

Where Δ(l) comes from:
1. m=i term: h(i)*(1-h(i-l)) was (1-h(i-l)), becomes 0. Δ₁ = -(1-h(i-l)) [if valid]
2. m=i+l term (h(m-l)=h(i)): h(i+l)*(1-h(i)) was 0, becomes h(i+l). Δ₂ = +h(i+l) [if valid]
3. m=j term: h(j)*(1-h(j-l)) was 0, becomes (1-h(j-l)). Δ₃ = +(1-h(j-l)) [if valid]
4. m=j+l term (h(m-l)=h(j)): h(j+l)*(1-h(j)) was h(j+l), becomes 0. Δ₄ = -h(j+l) [if valid]

Total: Δ(l) = -(1-h(i-l)) + h(i+l) + (1-h(j-l)) - h(j+l) [with validity checks]
           = h(i-l) - 1 + h(i+l) + 1 - h(j-l) - h(j+l) [if all valid]
           = h(i-l) + h(i+l) - h(j-l) - h(j+l) [if all valid]

The -1 and +1 cancel only if both i-l and j-l are valid. If only i-l is valid, we get an extra -1. If only j-l is valid, we get an extra +1.

In my vectorized code:
- For the "i as first factor" part: I add h[i-l1] - 1.0 for valid l1. This gives h(i-l) - 1. ✓
- For the "i as second factor" part: I add h[i+l2] for valid l2. This gives h(i+l). ✓
- For the "j as first factor" part: I add 1.0 - h[j-l3] for valid l3. This gives 1 - h(j-l). ✓
- For the "j as second factor" part: I subtract h[j+l4] for valid l4. This gives -h(j+l). ✓

Total: (h(i-l) - 1) + h(i+l) + (1 - h(j-l)) - h(j+l) = h(i-l) + h(i+l) - h(j-l) - h(j+l) + [boundary corrections]

This looks correct! The boundary corrections are automatically handled by only adding terms for valid indices.

frac = elapsed / total_time
T = T_start * (T_end / T_start) ** frac

This gives T = T_start at frac=0 and T = T_end at frac=1. The rate of cooling is:
dT/dfrac = T_start * (T_end/T_start) ** frac * ln(T_end/T_start)
= T * ln(T_end/T_start)
= T * ln(1e-6/0.02)

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def make_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))[:ones - target]
        h[idx[perm]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))[:target - ones]
        h[idx[perm]] = 1
    return h

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def delta_corr_for_swap(h, i, j, n):
    lags_n1 = n - 1
    delta = np.zeros(2 * n - 1)
    l1_lo = max(0, i - lags_n1)
    l1_hi = min(lags_n1, i)
    if l1_lo <= l1_hi:
        l1 = np.arange(l1_lo, l1_hi + 1)
        delta[l1 + lags_n1] += h[i - l1] - 1.0
    l2_lo = max(-lags_n1, -i)
    l2_hi = min(lags_n1, lags_n1 - i)
    if l2_lo <= l2_hi:
        l2 = np.arange(l2_lo, l2_hi + 1)
        delta[l2 + lags_n1] += h[i + l2]
    l3_lo = max(0, j - lags_n1)
    l3_hi = min(lags_n1, j)
    if l3_lo <= l3_hi:
        l3 = np.arange(l3_lo, l3_hi + 1)
        delta[l3 + lags_n1] += 1.0 - h[j - l3]
    l4_lo = max(-lags_n1, -j)
    l4_hi = min(lags_n1, lags_n1 - j)
    if l4_lo <= l4_hi:
        l4 = np.arange(l4_lo, l4_hi + 1)
        delta[l4 + lags_n1] -= h[j + l4]
    return delta

def fast_sa(h_init, n, duration, rng, T_start=0.015, T_end=1e-7):
    dx = 2.0 / n
    h = h_init.copy()
    corr = compute_corr(h)
    current_c5 = corr.max() * dx
    best_c5 = current_c5
    best_h = h.copy()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    start = time.time()
    last_improve = start
    no_improve_reheat = 0.0
    while True:
        elapsed = time.time() - start
        if elapsed >= duration:
            break
        frac = elapsed / duration
        T = T_start * (T_end / T_start) ** frac
        oi = rng.randint(len(ones_list))
        zi = rng.randint(len(zeros_list))
        i = ones_list[oi]
        j = zeros_list[zi]
        delta = delta_corr_for_swap(h, i, j, n)
        new_max = (corr + delta).max()
        new_c5 = new_max * dx
        new_delta = new_c5 - current_c5
        if new_delta < -1e-15 or rng.rand() < np.exp(-new_delta / T):
            h[i] = 0
            h[j] = 1
            ones_list[oi] = j
            zeros_list[zi] = i
            corr = corr + delta
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improve = time.time()
    return best_h, best_c5

def local_search_targeted(h, n, max_rounds, time_budget, rng):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = evaluate_c5(h)
    corr = compute_corr(h)
    start = time.time()
    for rnd in range(max_rounds):
        if time.time() - start > time_budget:
            break
        k_star = np.argmax(corr)
        l_star = k_star - (n - 1)
        best_delta = 0.0
        best_i = -1
        best_j = -1
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        n_try_ones = min(40, len(ones_idx))
        n_try_zeros = min(40, len(zeros_idx))
        s_ones = ones_idx[rng.choice(len(ones_idx), n_try_ones, replace=False)]
        s_zeros = zeros_idx[rng.choice(len(zeros_idx), n_try_zeros, replace=False)]
        for i in s_ones:
            for j in s_zeros:
                delta_vec = delta_corr_for_swap(h, i, j, n)
                new_max = (corr + delta_vec).max() * dx
                d = new_max - best_c5
                if d < best_delta:
                    best_delta = d
                    best_i = i
                    best_j = j
                    best_delta_vec = delta_vec
        if best_delta < -1e-14:
            h[best_i] = 0
            h[best_j] = 1
            corr = corr + best_delta_vec
            best_c5 += best_delta
        else:
            break
    return h, best_c5

def upsample(h_from, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h_from[idx] + frac * h_from[idx + 1]
        else:
            out[i] = h_from[idx]
    return out

def thue_morse_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    positions = np.arange(n)
    counts = np.zeros(n, dtype=int)
    x = positions.copy()
    while np.any(x > 0):
        counts += x & 1
        x >>= 1
    tm = (counts % 2 == 0).astype(float)
    ones_count = int(tm.sum())
    if ones_count > half:
        idx = np.where(tm == 1)[0]
        perm = rng.permutation(len(idx))[:ones_count - half]
        tm[idx[perm]] = 0
    elif ones_count < half:
        idx = np.where(tm == 0)[0]
        perm = rng.permutation(len(idx))[:half - ones_count]
        tm[idx[perm]] = 1
    return tm

def beatty_binary(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    half = n // 2
    h = np.zeros(n)
    beatty = np.floor(np.arange(1, 2 * half + 1) * phi).astype(int)
    beatty = beatty[beatty < n]
    beatty = beatty[:half]
    h[beatty] = 1.0
    return h
```
