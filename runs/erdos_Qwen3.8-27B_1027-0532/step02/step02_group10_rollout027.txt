                pert = project_onto_constraint(best_h + rng.randn(n) * 0.02, target)
        pert = project_onto_constraint(pert, target)
        alpha_s = rng.choice([100, 200, 500, 1000])
        alpha_e = alpha_s * rng.choice([3, 5, 10])
        dl = time.time() + min(10, max(2, (phase2_end - time.time()) / max(1, n_trials - i)))
        h = pgd_adaptive(pert, alpha_s, alpha_e, target, 3000, dl)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    remain = budget_s - (time.time() - t0)
    phase3_end = time.time() + min(500, 0.40 * budget_s)
    for alpha_pair in [(5000, 15000), (15000, 40000), (40000, 80000)]:
        if time.time() > phase3_end:
            break
        dl = min(time.time() + 60, phase3_end)
        h = pgd_adaptive(best_h, alpha_pair[0], alpha_pair[1], target, 20000, dl)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    for alpha in [2000, 5000, 10000]:
        if time.time() > phase3_end:
            break
        dl = min(time.time() + 30, phase3_end)
        h = fista_optimize(best_h, alpha, target, 8000, dl)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    if time.time() - t0 < budget_s - 80:
        for alpha in [20000, 50000, 100000]:
            if time.time() - t0 > budget_s - 50:
                break
            try:
                h = lbfgsb_refine(best_h, alpha, target, time.time() + 20)
                h = project_onto_constraint(h, target)
                s = evaluate_score(h)
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
            except Exception:
                pass

    if time.time() - t0 < budget_s - 30:
        for alpha in [50000, 100000, 200000]:
            if time.time() - t0 > budget_s - 15:
                break
            dl = time.time() + 10
            h = pgd_adaptive(best_h, alpha, alpha * 2, target, 5000, dl)
            h = project_onto_constraint(h, target)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

    best_h = project_onto_constraint(best_h, target)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, n)
