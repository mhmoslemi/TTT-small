```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = (h > 0) & (h < 1)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad


def adam_optimize(h, alpha, target, max_iters, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    prev_sm = None
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        overlaps = compute_overlaps(x)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        
        if prev_sm is not None:
            if sm < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 300:
                break
        prev_sm = sm
        
        grad = compute_gradient(x, weights, dx)
        
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        
        x_new = x - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        x = project_onto_constraint(x_new, target)
        
        # Check bounds
        if not np.all((x >= -1e-10) & (x <= 1.0 + 1e-10)):
            x = np.clip(x, 0.0, 1.0)
            x = project_onto_constraint(x, target)
    
    return x

def coordinate_descent(h, target, max_rounds, deadline, tol=1e-10):
    """Optimize one coordinate at a time."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try to find best value for h[i] given rest fixed
            # The objective is max_k O(k), and O(k) is linear in h[i]
            # So for each k, O(k) = a_k * h[i] + b_k where a_k depends on whether i is in the overlap
            # We want to minimize max_k (a_k * h[i] + b_k)
            # subject to: h[i] new such that sum constraint still holds
            
            # Actually, if we change h[i] by delta, we need to adjust another coordinate
            # for the sum constraint. This makes it more complex.
            # Let's just do a simple 1D search: try h[i] in {0, 0.25, 0.5, 0.75, 1, old_val ± small}
            # and check if the score improves (with sum constraint approximately satisfied)
            
            # Simpler approach: try a few candidate values
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0, old_val]
            # Also try values between neighbors
            if i > 0:
                candidates.append((h[i-1] + old_val) / 2)
            if i < n - 1:
                candidates.append((old_val + h[i+1]) / 2)
            
            best_cand = old_val
            best_sm = None
            for c in candidates:
                h[i] = c
                # Need to fix sum constraint - adjust a random other coordinate
                s = h.sum()
                diff = target - s
                if abs(diff) > 1e-10:
                    # Adjust coordinate i itself (simplest)
                    h[i] = c + diff
                    if h[i] < 0 or h[i] > 1:
                        h[i] = old_val
                        continue
                sm = evaluate_score(h)
                if best_sm is None or sm < best_sm:
                    best_sm = sm
                    best_cand = h[i].copy()
                    best_cand = c + diff
            h[i] = best_cand
            if abs(h[i] - old_val) > tol:
                improved = True
        
        h = project_onto_constraint(h, target)
        if not improved:
            break
    
    return h


def lbfgsb_refine(h, alpha, target, deadline):
    n = len(h)
    dx = 2.0 / n
    n_iter = [0]
    
    def obj(hv):
        n_iter[0] += 1
        ov = compute_overlaps(hv)
        sm, _ = smooth_max_and_weights(ov, alpha)
        pen = 500.0 * (hv.sum() - n / 2.0) ** 2
        return sm + pen
    
    def obj_grad(hv):
        ov = compute_overlaps(hv)
        sm, w = smooth_max_and_weights(ov, alpha)
        g = compute_gradient(hv, w, dx) + 1000.0 * (hv.sum() - n / 2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', jac=obj_grad, bounds=bounds,
                   options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
    return res.x if res.x is not None else h
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = (h > 0) & (h < 1)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad


def adam_optimize(h, alpha, target, max_iters, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        overlaps = compute_overlaps(x)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 300:
                break
        prev_sm = sm
        grad = compute_gradient(x, weights, dx)
        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)
        m_hat_hat = m_hat / (1.0 - beta1 ** t)
        v_hat_hat = v_hat / (1.0 - beta2 ** t)
        x_new = x - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        x = project_onto_constraint(x_new, target)
    return x


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = min(0.5, 1.0 / (alpha * dx * 3.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-13:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 200:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        improved = False
        s = step
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-14:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = s
            break
        step = min(s * 1.3, 1.0)
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.3, 1.0 / (alpha * dx * 5.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_and_weights(ov_x, alpha)
        if prev_sm is not None:
            if sm_x < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
        prev_sm = sm_x
        if stall_count > 300:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x


def coordinate_descent_polish(h, target, max_rounds, deadline):
    n = len(h)
    h = h.copy()
    current_score = evaluate_score(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                return h
            old_val = h[i]
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
            if i > 0:
                candidates.append((h[i - 1] + old_val) / 2.0)
            if i < n - 1:
                candidates.append((old_val + h[i + 1]) / 2.0)
            best_cand = old_val
            best_s = current_score
            for c in candidates:
                h[i] = c
                s = h.sum()
                diff = target - s
                if abs(diff) > 1e-12:
                    h[i] = c + diff
                    if h[i] < -1e-10 or h[i] > 1.0 + 1e-10:
                        h[i] = old_val
                        continue
                    np.clip(h, 0.0, 1.0, out=h)
                sc = evaluate_score(h)
                if sc < best_s:
                    best_s = sc
                    best_cand = h[i]
            h[i] = best_cand
            current_score = best_s
        h = project_onto_constraint(h, target)
        current_score = evaluate_score(h)
    return h


def lbfgsb_obj(x, alpha, n, dx):
    ov = compute_overlaps(x)
    sm, _ = smooth_max_and_weights(ov, alpha)
    pen = 500.0 * (x.sum() - n / 2.0) ** 2
    return sm + pen


def lbfgsb_grad(x, alpha, n, dx):
    ov = compute_overlaps(x)
    sm, w = smooth_max_and_weights(ov, alpha)
    g = compute_gradient(x, w, dx) + 1000.0 * (x.sum() - n / 2.0)
```
