import numpy as np
from scipy.optimize import minimize

def project_to_feasible(h, target_sum):
    # References: [0,1)
    lo, hi = 0.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:-n + lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def compute_score_for_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def subgradient_run(h0, max_iters, alpha, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    best_score = compute_score_for_h(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    np.random.seed(seed)
    for i in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        # Compute correlations and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        weights = np.exp(alpha * (top_vals - top_vals[0])) / np.sum(weights)
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * compute_gradient_for_lag(h, lag)
        # Gradient adjustment
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        # Line search for improvement
        improved = False
        for trial in range(20):
            t = step * 0.1 * (trial + 1)
            h_new = h - t * grad
            h_new = project_to_feasible(h_new, target_sum)
            # Regularized fitness
            if np.random.rand() < (1 - trial/40) and grad_norm > 1e-6:
                noise = np.random.randn(len(h)) * (grad_norm / 20)
                h_new += noise
                h_new = project_to_feasible(h_new, target_sum)
            score = compute_score_for_h(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                improved = True

                break
        if improved:

            step *= 0.8
        else:
            stall += 1
            if stall > 300:

                step = step_init * (0.5 + 0.5 * np.random.rand())
                stall = 0
    h = best_h
    return h, best_score

def refine_with_slqp(h, target_sum, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        max_corr = np.max(corr)
        if np.isinf(max_corr) or np.isnan(max_corr):
            raise ValueError("Infinite value in correlation")
        return max_corr
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    cons = {'type': 'eq', 'fun': constraint_fun}
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': maxiter})
    if res.success and np.all(np.isfinite(res.x)):
        return project_to_feasible(res.x, target_sum)
    return None
