One more thought: for the coordinate descent, instead of just trying 0 and 1, I should also try the "median" value that equalizes the top 2-3 correlation terms. This is because the optimum of a max of linear functions occurs where two (or more) of the linear functions cross.

For coordinate j, the objective as a function of h[j]=v is:
g(v) = max_k (a_k * v + b_k)

where a_k and b_k are computed from the current h (with h[j] removed). The minimum of g(v) on [0,1] occurs at v=0, v=1, or at a crossing point of two active linear functions.

To find crossing points: solve a_k1 * v + b_k1 = a_k2 * v + b_k2 for v, which gives v = (b_k2 - b_k1) / (a_k1 - a_k2).

For the top 2 correlation terms, I can compute this crossing point and try it.

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            count = int(np.sum(mask))
            if count == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / count)
        else:
            mask = h > 1e-14
            count = int(np.sum(mask))
            if count == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / count)
    return h

def _smooth_obj(x):
    alpha = _state['alpha']
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pen_obj(x):
    alpha = _state['alpha']
    mu = _state['mu']
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sv = np.sum(x) - _state['target']
    return float(smooth_max + mu * sv * sv)

def _con_fun(x):
    return np.sum(x) - _state['target']

def _con_jac(x):
    return np.ones(_state['n'])

def _de_obj(x):
    return compute_c5(project_feasible(x, _state['target']))

def run_slsqp(h0, n, target, alphas, maxiter, ftol):
    _state['n'] = n
    _state['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = project_feasible(h0.copy(), target)
    for alpha in alphas:
        _state['alpha'] = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def run_lbfgs(h0, n, target, alphas, mus, maxiter):
    _state['n'] = n
    _state['target'] = target
    bounds = [(0.0, 1.0)] * n
    h_cur = project_feasible(h0.copy(), target)
    for alpha in alphas:
        for mu in mus:
            _state['alpha'] = alpha
            _state['mu'] = mu
            try:
                res = minimize(_pen_obj, h_cur, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
                h_new = project_feasible(res.x, target)
                if compute_c5(h_new) < compute_c5(h_cur):
                    h_cur = h_new
            except Exception:
                pass
    return h_cur

def coord_descent_polish(h, n, target, rng, max_sweeps=50):
    h = project_feasible(h.copy(), target)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            for v in (0.0, 1.0):
                if abs(h[j] - v) < 1e-10:
                    continue
                h_trial = h.copy()
                h_trial[j] = v
                h_fixed = project_feasible(h_trial, target)
                c5_val = compute_c5(h_fixed)
                if c5_val < best_c5 - 1e-13:
                    best_c5 = c5_val
                    h = h_fixed
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def interpolate_to(h_src, n_dst):
    target_dst = n_dst / 2.0
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
