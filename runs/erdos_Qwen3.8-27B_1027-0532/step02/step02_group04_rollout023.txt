import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:N - lag] += w
            grad[:N - lag] -= w * h[lag:]
            grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            grad[l:] += w
            grad[l:] -= w * h[:N - l]
            grad[:N - l] -= w * h[l:]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = (t & 1).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:n_ones - target]] = 0.0
    return h


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

        else:

    return best, best_max


def active_subgrad(h0, n_iters, n_top, cooling):
    N = len(h0)
