def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad

def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

    return lse_val(compute_overlaps(h), alpha)

def slsqp_grad(h, alpha):

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,

            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def coord_descent_1d(h, i, current_overlaps):
    n = len(h)
    l_eff_range = np.arange(-(n - 1), n)
    valid1 = (i - l_eff_range >= 0) & (i - l_eff_range < n)
    valid2 = (i + l_eff_range >= 0) & (i + l_eff_range < n)

    a = np.zeros(2 * n - 1)
    a[valid1] += (1.0 - h[i - l_eff_range[valid1]])
    a[valid2] -= h[i + l_eff_range[valid2]]

    b = current_overlaps - a * h[i]

    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a * m1 + b))
        f2 = float(np.max(a * m2 + b))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5


def coord_descent_pass(h, order):
    n = len(h)
    h = h.copy()
    overlaps = compute_overlaps(h)
    for i in order:
        opt = coord_descent_1d(h, i, overlaps)
        h[i] = opt
        l_eff_range = np.arange(-(n - 1), n)
        valid1 = (i - l_eff_range >= 0) & (i - l_eff_range < n)
        valid2 = (i + l_eff_range >= 0) & (i + l_eff_range < n)
        a = np.zeros(2 * n - 1)
        a[valid1] += (1.0 - h[i - l_eff_range[valid1]])
        a[valid2] -= h[i + l_eff_range[valid2]]
        b = overlaps - a * 0.0
        overlaps = a * opt + b
    return h, float(np.max(overlaps))


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0


def sum_constraint_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(

            method='SLSQP', jac=slsqp_grad, bounds=bounds,

            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None

            h_sub, raw_sub = subgrad_optimize(best_h, n_sub, 0.4, n / 2.0, top_k)
            s_sub = raw_sub * (2.0 / n)
            if s_sub < best_score:
                best_score = s_sub
                best_h = h_sub.copy()
            # Polish with coord descent
            if time_left() > 40:
                for it in range(50):
                    if time_left() < 15:
                        break
                    order = list(range(n))
                    rng.shuffle(order)
                    h_cd, _ = coord_descent_pass(best_h, order)
                    h_cd = project_box_sum(h_cd, n / 2.0)
                    s = evaluate_score(h_cd)
                    if s < best_score:
                        best_score = s
                        best_h = h_cd.copy()

    # Phase 5: Multi-start from diverse initializations
    if time_left() > 80:
        # Random binary start
        h_bin = np.zeros(n)
        ones_pos = rng.choice(n, size=n // 2, replace=False)
        h_bin[ones_pos] = 1.0
        h_bin = project_box_sum(h_bin, n / 2.0)
        # Quick LSE from this start
        alpha = 500
        penalty = alpha * 40.0
        bounds = [(0.0, 1.0)] * n
        res_x = lbfgsb_optimize(h_bin, alpha, penalty, 300, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
        # Coord descent polish
        for it in range(50):
            if time_left() < 30:
                break
            h_cand = h_new if res_x is not None else h_bin
            order = list(range(n))
            rng.shuffle(order)
            h_cd, _ = coord_descent_pass(h_cand, order)
            h_cd = project_box_sum(h_cd, n / 2.0)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd.copy()
                h_cand = h_cd.copy()

    # Phase 6: Perturbation + coordinate descent
    if time_left() > 60:
        for pert_sigma in [0.03, 0.08, 0.15]:
            if time_left() < 30:
                break
            h_pert = best_h + rng.randn(n) * pert_sigma
            h_pert = project_box_sum(h_pert, n / 2.0)
            # LSE polish
            alpha = 1000
            penalty = alpha * 50.0
            bounds = [(0.0, 1.0)] * n
            res_x = lbfgsb_optimize(h_pert, alpha, penalty, 400, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            else:
                h_new = h_pert
            # Coord descent
            for it in range(80):
                if time_left() < 20:
                    break
                order = list(range(n))
                rng.shuffle(order)
                h_cd, _ = coord_descent_pass(h_new, order)
                h_cd = project_box_sum(h_cd, n / 2.0)
                s = evaluate_score(h_cd)
                if s < best_score:
                    best_score = s
                    best_h = h_cd.copy()
                h_new = h_cd

    # Phase 7: SLSQP for constraint-accurate finishing
    if time_left() > 45:
        for alpha in [2000, 5000, 10000]:
            if time_left() < 20:
                break
            bounds = [(0.0, 1.0)] * n
            res_x = slsqp_optimize(best_h, alpha, 200, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 8: Final intensive coordinate descent
    if time_left() > 20:
        max_rounds = min(500, int(time_left() * 8))
        for it in range(max_rounds):
            if time_left() < 5:
                break
            order = list(range(n))
            rng.shuffle(order)
            h_cd, _ = coord_descent_pass(best_h, order)
            h_cd = project_box_sum(h_cd, n / 2.0)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd.copy()

    # Final projection and evaluation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
