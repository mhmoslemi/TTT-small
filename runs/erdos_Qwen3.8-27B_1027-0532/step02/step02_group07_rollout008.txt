import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(40):
        mid_V = (lo_V + hi_V) * 0.5

        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0, rng=None, ordered=False):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()

    if rng is None:
        rng = np.random.RandomState(0)

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        if ordered:
            order = list(range(n))
        else:
            order = list(range(n))
            rng.shuffle(order)
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 15 == 14:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def compute_subgradient(h, eps=1e-6):
    n = len(h)
    corr = compute_correlations(h)
    V = np.max(corr)
    active = corr >= V - eps
    n_active = np.sum(active)
    if n_active == 0:
        return np.zeros(n)

    offset = n - 1
    grad = np.zeros(n)

    active_indices = np.where(active)[0]
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for k in active_indices:
            idx1 = j + k - offset
            if 0 <= idx1 < n:
                s1 += (1.0 - h[idx1])
            idx2 = j - k + offset
            if 0 <= idx2 < n:
                s2 += h[idx2]
        grad[j] = (s1 - s2) / n_active

    return grad


def subgradient_descent(h, n_iter=200, lr0=0.01, time_budget=30.0):
    n = len(h)
    h = h.copy()
    t0 = time.time()
    best_h = h.copy()
    best_score = evaluate_score(h)

    for it in range(n_iter):
        if time.time() - t0 > time_budget:
            break
        lr = lr0 / (1.0 + it * 0.05)
        grad = compute_subgradient(h)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, n / 2.0)
        s = evaluate_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def fourier_perturb(h, rng, max_mode=5, amp_scale=1.0):
    n = len(h)
    h_pert = h.copy()
    for k in range(1, max_mode + 1):
        amp = rng.uniform(0.02, 0.10) * amp_scale
        phase = rng.uniform(0, 2 * np.pi)
        h_pert += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    h_pert = np.clip(h_pert, 0.0, 1.0)
    h_pert = project_sum(h_pert, n / 2.0)
    return h_pert


def simulated_annealing_search(h, rng, n_iter=50, T0=0.005, T_end=0.001, time_budget=60.0):
    n = len(h)
    h_cur = h.copy()
    s_cur = evaluate_score(h_cur)
    h_best = h_cur.copy()
    s_best = s_cur
    t0 = time.time()

    for it in range(n_iter):
        if time.time() - t0 > time_budget:
            break
        T = T0 * (T_end / T0) ** (it / max(n_iter - 1, 1))

        h_pert = fourier_perturb(h_cur, rng, max_mode=3, amp_scale=1.0)
        tb = min(3.0, (time_budget + t0 - time.time()) * 0.3)
        h_cand = coordinate_descent(h_pert, n_sweeps=80, time_budget=tb, rng=rng)
        s_cand = evaluate_score(h_cand)

        if s_cand < s_best:
            s_best = s_cand
            h_best = h_cand.copy()
            h_cur = h_cand.copy()
            s_cur = s_cand
        elif rng.random() < np.exp(-(s_cand - s_cur) / T):
            h_cur = h_cand.copy()
            s_cur = s_cand

    return h_best, s_best


def make_block_init(n, rng, frac=0.5):
    h = np.zeros(n)
    n_ones = int(n * frac)
    positions = rng.choice(n, size=n_ones, replace=False)
    h[positions] = 1.0
    return project_sum(h, n / 2.0)

        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
