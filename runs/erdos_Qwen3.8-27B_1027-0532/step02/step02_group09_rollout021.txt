def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad

    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    cur_score = best_score
    temp = temp_init
    for it in range(n_iters):
        if it % 300 == 299:
            temp *= 0.96
        if temp < 1e-16:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < n_flips or len(zeros) < n_flips:
            break
        flip_from = ones[rng.choice(len(ones), size=n_flips, replace=False)]
        flip_to = zeros[rng.choice(len(zeros), size=n_flips, replace=False)]
        old_f = h_bin[flip_from].copy()
        old_t = h_bin[flip_to].copy()
        h_bin[flip_from] = 0.0
        h_bin[flip_to] = 1.0
        s_new = evaluate_score(h_bin)
        delta = s_new - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-18)):
            cur_score = s_new
            if s_new < best_score:
                best_score = s_new
                best_h = h_bin.copy()
        else:
            h_bin[flip_from] = old_f
            h_bin[flip_to] = old_t
    return best_h, best_score


def downsample(h, n_new):
    n_old = len(h)
    if n_new >= n_old:
        return h.copy()
    result = np.zeros(n_new)
    for i in range(n_new):
        start = int(round(i * n_old / n_new))
        end = int(round((i + 1) * n_old / n_new))
        if end > start:
            result[i] = np.mean(h[start:end])
        else:
            result[i] = h[start]
    return result


def upsample(h, n_new):
    n_old = len(h)
    if n_new <= n_old:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def incremental_eval(h, i, j, old_overlaps):
    n = len(h)
    h_new = h.copy()
    h_new[i] = 1.0 - h_new[i]
    h_new[j] = 1.0 - h_new[j]
    new_overlaps = compute_overlaps(h_new)
    return float(np.max(new_overlaps)), new_overlaps


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 25.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n_parent = len(h_init)

    best_h = h_init.copy()
    best_score = evaluate_score(h_init)

    # Phase 1: Multi-resolution coarse-to-fine
    coarse_sizes = [16, 32, 64, 128]
    for n_c in coarse_sizes:
        if time_left() < 250:
            break
        h_c = downsample(h_init, n_c)
        h_c = project_box_sum(h_c, n_c / 2.0)
        for alpha in [100, 400]:
            if time_left() < 220:
                break
            penalty = alpha * 25.0
            bounds = [(0.0, 1.0)] * n_c
            maxiter = max(200, 800000 // n_c)
            res_x = lbfgsb_optimize(h_c, alpha, penalty, maxiter, bounds)
            if res_x is not None:
                h_c = project_box_sum(np.clip(res_x, 0.0, 1.0), n_c / 2.0)
        if time_left() > 200:
            n_sa = min(3000, max(500, int(time_left() * 1.5)))
            h_c_bin, _ = sa_with_restart(h_c, n_sa, rng, 0.05, 0.95, 100, 800, 5)
            h_up = upsample(h_c_bin, n_parent)
            h_up = project_box_sum(h_up, n_parent / 2.0)
            s_up = evaluate_score(h_up)
            if s_up < best_score:
                best_score = s_up
                best_h = h_up.copy()

    # Phase 2: LSE optimization at parent resolution (escalating alpha)
    alpha_schedule = [200, 500, 1000, 2500, 5000]
    for alpha in alpha_schedule:
        if time_left() < 180:
            break
        penalty = alpha * 35.0
        bounds = [(0.0, 1.0)] * n_parent
        maxiter = max(300, 500000 // n_parent)
        res_x = lbfgsb_optimize(best_h, alpha, penalty, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_parent / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 3: SA with restarts at parent resolution
    if time_left() > 200:
        n_iters = min(50000, int(time_left() * 6))
        h_sa, s_sa = sa_with_restart(best_h, n_iters, rng, 0.02, 0.97, 150, 5000, 8)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa.copy()

    # Phase 4: Multi-flip SA
    if time_left() > 150:
        n_iters = min(20000, int(time_left() * 4))
        h_mp, s_mp = sa_multi_flip(best_h, n_iters, rng, 0.03, 2)
        if s_mp < best_score:
            best_score = s_mp
            best_h = h_mp.copy()

    # Phase 5: Try nearby even-n grids (purely binary possible)
    if time_left() > 180:
        for n_try in [n_parent - 1, n_parent + 1, n_parent + 3, n_parent - 3]:
            if n_try < 100 or n_try % 2 != 0 or time_left() < 150:
                continue
            h_e = upsample(best_h, n_try)
            h_e_bin = to_binary(h_e, rng)
            s_e = evaluate_score(h_e_bin)
            candidate_h = h_e_bin.copy()
            if s_e < best_score:
                best_score = s_e
                best_h = h_e_bin.copy()
            if time_left() > 100:
                n_iters = min(20000, int(time_left() * 3))
                h_e2, s_e2 = sa_with_restart(h_e_bin, n_iters, rng, 0.015, 0.97, 150, 4000, 6)
                if s_e2 < best_score:
                    best_score = s_e2
                    best_h = h_e2.copy()

    # Phase 6: Second round of LSE polish on best found so far
    if time_left() > 80:
        n_cur = len(best_h)
        for alpha in [5000, 10000, 20000]:
            if time_left() < 30:
                break
            penalty = alpha * 50.0
            bounds = [(0.0, 1.0)] * n_cur
            res_x = lbfgsb_optimize(best_h, alpha, penalty, 600, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 7: Final focused SA
    if time_left() > 40:
        n_iters = min(15000, int(time_left() * 3))
        h_f, s_f = sa_with_restart(best_h, n_iters, rng, 0.008, 0.98, 200, 3000, 5)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f.copy()

    # Final projection and evaluation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
