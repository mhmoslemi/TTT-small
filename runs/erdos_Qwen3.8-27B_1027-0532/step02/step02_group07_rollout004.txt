    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(40):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_to_feasible(h, n / 2.0)
    return h


def smooth_gradient_descent(h, alphas, time_budget=60.0, momentum_decay=0.9):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    t0 = time.time()

    for alpha in alphas:
        if time.time() - t0 > time_budget:
            break
        current_obj = smooth_max_value(compute_correlations(h), alpha)
        step_scale = 0.01

        while time.time() - t0 < time_budget:
            grad = compute_smooth_gradient(h, alpha)
            v = momentum_decay * v - step_scale * grad
            h_new = project_to_feasible(h + v, n / 2.0)
            new_obj = smooth_max_value(compute_correlations(h_new), alpha)

            if new_obj < current_obj + 1e-15:
                h = h_new
                if new_obj < current_obj:
                    current_obj = new_obj
                    step_scale = min(step_scale * 1.1, 0.1)
                else:
                    step_scale *= 0.7
            else:
                step_scale *= 0.5
            if step_scale < 1e-8:
                break

    return h


def shift_perturb(h, rng, max_shift=5):
    n = len(h)
    s = rng.randint(1, max_shift + 1)
    if rng.random() < 0.5:
        h_new = np.roll(h, s)
    else:
        h_new = np.roll(h, -s)
    return project_to_feasible(h_new, n / 2.0)


def sinusoidal_perturb(h, rng):
    n = len(h)
    freq = rng.randint(1, max(2, n // 20))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.15)
    pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    h_new = h + pert
    return project_to_feasible(h_new, n / 2.0)


def block_perturb(h, rng):
    n = len(h)
    block_size = rng.randint(n // 20, n // 5)
    start = rng.randint(0, n - block_size)
    h_new = h.copy()
    shift_amt = rng.uniform(-0.3, 0.3)
    h_new[start:start + block_size] += shift_amt
    h_new[:start] -= shift_amt * block_size / max(start, 1)
    if n - start - block_size > 0:
        h_new[start + block_size:] -= shift_amt * block_size / (n - start - block_size)
    return project_to_feasible(h_new, n / 2.0)


def multiresolve(h_best, n_coarse, time_budget=40.0):
    n_full = len(h_best)
    x_full = np.linspace(0.0, 2.0, n_full, endpoint=False)
    x_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    h_coarse = np.interp(x_coarse, x_full, h_best)
    h_coarse = project_to_feasible(h_coarse, n_coarse / 2.0)

    t0 = time.time()
    budget_cd = (time_budget - 10) * 0.5
    budget_sgd = (time_budget - 10) * 0.3
    budget_perturb = (time_budget - 10) * 0.2

    if time_budget - 10 > 5:
        h_coarse = coordinate_descent(h_coarse, n_sweeps=200, time_budget=budget_cd)

    if time.time() - t0 < time_budget - 10:
        h_coarse = smooth_gradient_descent(h_coarse, [100, 500, 2000], time_budget=max(0, time_budget - (time.time() - t0) - 10))

    if time.time() - t0 < time_budget - 5:
        h_coarse = coordinate_descent(h_coarse, n_sweeps=100, time_budget=max(0, time_budget - (time.time() - t0) - 5))

    h_up = np.interp(x_full, x_coarse, h_coarse)
    h_up = project_to_feasible(h_up, n_full / 2.0)
    return h_up


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_to_feasible(init, n / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: Initial CD
    if time_left() > 100:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(80, time_left() - 50))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: Smooth gradient descent with increasing alpha
    if time_left() > 120:
        h_sgd = smooth_gradient_descent(best_h, [50, 200, 1000, 5000, 20000],
                                        time_budget=min(80, time_left() - 60))
        s = evaluate_score(h_sgd)
        if s < best_score:
            best_score = s
            best_h = h_sgd

    # Phase 3: CD polish after SGD
    if time_left() > 60:
        h_cd = coordinate_descent(best_h, n_sweeps=300, time_budget=min(50, time_left() - 40))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 4: Structured perturbations (shifts, sinusoidal, blocks)
    n_perturb = 200
    for i in range(n_perturb):
        if time_left() < 30:
            break
        frac = i / max(n_perturb - 1, 1)

        r = rng.random()
        if r < 0.3:
            h_pert = shift_perturb(best_h, rng, max_shift=3 + int(7 * frac))
        elif r < 0.6:
            h_pert = sinusoidal_perturb(best_h, rng)
        elif r < 0.8:
            h_pert = block_perturb(best_h, rng)
        else:
            scale = 0.05 + 0.35 * frac
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_to_feasible(h_pert, n / 2.0)

        tb = min(5.0, (time_left() - 20) / max(n_perturb - i, 1))
        h_cd = coordinate_descent(h_pert, n_sweeps=150, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 5: Multi-resolution
    if time_left() > 100:
        for n_coarse in [n // 2, n // 3, n // 4]:
            if time_left() < 50:
                break
            if n_coarse < 50:
                continue
            h_up = multiresolve(best_h, n_coarse, time_budget=min(50, time_left() - 40))
            h_cd = coordinate_descent(h_up, n_sweeps=200, time_budget=min(30, time_left() - 30))
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 6: High-resolution attempt
    if time_left() > 100:
        n_hi = min(n * 2, 2000)
        if n_hi > n:
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_hi, endpoint=False)
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = project_to_feasible(h_hi, n_hi / 2.0)
            h_hi = coordinate_descent(h_hi, n_sweeps=200, time_budget=min(40, time_left() - 50))
            s_hi = evaluate_score(h_hi)
            if s_hi < best_score:
                best_score = s_hi
                # Downsample back
                best_h = np.interp(x_old, x_new, h_hi)
                best_h = project_to_feasible(best_h, n / 2.0)
                best_score = evaluate_score(best_h)

    # Phase 7: Second round of perturbations with smaller scale
    n_perturb2 = 80
    for i in range(n_perturb2):
        if time_left() < 25:
            break
        frac = i / max(n_perturb2 - 1, 1)
        r = rng.random()
        if r < 0.4:
            h_pert = shift_perturb(best_h, rng, max_shift=2 + int(3 * frac))
        elif r < 0.7:
            scale = 0.02 + 0.12 * frac
            h_pert = best_h + rng.randn(n) * scale
        else:
            h_pert = sinusoidal_perturb(best_h, rng)
        h_pert = project_to_feasible(h_pert, n / 2.0)

        tb = min(4.0, (time_left() - 15) / max(n_perturb2 - i, 1))
        h_cd = coordinate_descent(h_pert, n_sweeps=120, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 8: Final smooth gradient descent
    if time_left() > 40:
        h_sgd = smooth_gradient_descent(best_h, [5000, 20000, 80000],
                                        time_budget=min(30, time_left() - 15))
        s = evaluate_score(h_sgd)
        if s < best_score:
            best_score = s
            best_h = h_sgd

    # Phase 9: Final CD polish
    if time_left() > 20:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(15, time_left() - 5))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Final cleanup
    best_h = project_to_feasible(best_h, n / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
