2. The subgradient at t is the set of D_j for all j that achieve the max
3. If the subgradient contains 0, we're at the minimum
4. Otherwise, move in the direction of the negative of a subgradient element

At any t, the active set is {j: C_j + D_j * t = max_k (C_k + D_k * t)}. The slope is the range of D_j over the active set. If min D_j < 0 < max D_j, we're at the minimum (or on a flat region).

1. Start at t=0. Find active set. If all D_j >= 0 for active j, move right. If all D_j <= 0, we're at minimum (since we started at 0 and can't go left).
2. Move to the next breakpoint (where a currently-inactive line becomes active or an active line becomes inactive).
3. Repeat.

def _c5_of(h):

    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Add to coords with most room, prioritizing those where addition hurts least
        # Simple approach: add to coords with highest (1-h[i])
        room = 1.0 - h
        order = np.argsort(-room)  # most room first
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff -= add
    else:
        order = np.argsort(-h)  # largest h first
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff += sub
    return h

def _coord_optimal_t(h, i, n):
    """Find optimal t for h[i] minimizing max_j out_j (ignoring sum constraint)."""
    out = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute D: coefficient of h[i] in each out[j]
    D = np.zeros(2*n - 1)
    
    # Type 1: h[i] appears as h[m] with m=i
    # out[j] includes h[i]*(1-h[i+j-(n-1)]) if 0 <= i+j-(n-1) < n
    j_range_1 = range(max(0, n-1-i), min(2*n-1, 2*n-1-i+1))
    for j in j_range_1:
        idx = i + j - (n-1)

            D[j] += (1.0 - h[idx])
    
    # Type 2: h[i] appears as (1-h[m+j-(n-1)]) with m+j-(n-1)=i
    # m = i-j+(n-1), need 0 <= m < n
    j_range_2 = range(max(0, i), min(2*n-1, i+n))
    for j in j_range_2:
        m2 = i - j + (n-1)
        if 0 <= m2 < n:

def _slsqp_refine(h_init, n, alpha, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)
    
    def obj(x):

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    
    def obj_grad(x):

        m = np.max(corr)
        e = np.exp(alpha * (corr - m))

        weights = e / s  # softmax weights
        # grad of max smoothed = weighted average of grads of each corr[j]
        grad = np.zeros(npts)

            if w < 1e-12:

                grad[:] += w * (1.0 - 2.0 * x)
        return grad * dx

        if res.success or res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h_init.copy()

        grad /= len(top_idx)
        v = momentum * v - lr * grad
        h = _fix_sum(h + v, target_sum)
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(100, 0.5)

    n = len(h0)
    target_sum = n / 2.0
    h0 = _fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_of(h0)

    # Phase 1: Coordinate descent from parent
    if time.time() < deadline - 120:
        h_cd, c5_cd = _coord_descent(h0, n, target_sum, max_passes=500)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: SLSQP with increasing alpha
    h_cur = best_h.copy()
    for alpha in [30, 80, 200, 500, 1500, 4000]:
        if time.time() > deadline - 40:
            break
        h_s = _slsqp_with_grad(h_cur, n, alpha, maxiter=400, ftol=1e-15)
        h_s = _fix_sum(h_s, target_sum)
        c5_s = _c5_of(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        h_cur = h_s
        if time.time() < deadline - 30:
            h_cd, c5_cd = _coord_descent(h_cur, n, target_sum, max_passes=100)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h_cur = h_cd

    # Phase 3: Subgradient from parent with different params
    if time.time() < deadline - 60:
        h_sg, c5_sg = _subgrad_step(h0, n, top_k=1, n_iter=20000,
                                     lr0=0.3, decay=0.0005, momentum=0.9,
                                     check_every=500, deadline=deadline)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        if time.time() < deadline - 30:
            h_cd, c5_cd = _coord_descent(h_sg, n, target_sum, max_passes=150)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 4: Perturbation restarts
    rng = np.random.RandomState(seed + 7777)
    n_perturb = 8
    for attempt in range(n_perturb):
        if time.time() > deadline - 50:
            break
        scale = 0.005 + 0.04 * rng.rand()
        h_p = _fix_sum(best_h + scale * rng.randn(n), target_sum)
        h_p, c5_p = _coord_descent(h_p, n, target_sum, max_passes=80)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
        if time.time() < deadline - 25 and c5_p < best_c5 + 0.002:
            h_sr = _slsqp_with_grad(h_p, n, 300, maxiter=200, ftol=1e-14)
            h_sr = _fix_sum(h_sr, target_sum)
            c5_sr = _c5_of(h_sr)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr.copy()
            if time.time() < deadline - 15:
                h_cd, c5_cd = _coord_descent(h_sr, n, target_sum, max_passes=80)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 5: Final polish
    if time.time() < deadline - 20:
        h_f, c5_f = _coord_descent(best_h, n, target_sum, max_passes=200)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < deadline - 10:
        h_f = _slsqp_with_grad(best_h, n, 1000, maxiter=300, ftol=1e-15)
        h_f = _fix_sum(h_f, target_sum)
        c5_f = _c5_of(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n)
