def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _grad_corr_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:n])
        grad[lag:n] -= h[:n - lag]
    else:
        L = -lag
        grad[L:n] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:n]
    return grad

    if wsum < 1e-300:

        n = len(h)

                lag = k_idx - (n - 1)

import time
import numpy as np
from scipy.optimize import minimize


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def _grad_corr_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:n])
        grad[lag:n] -= h[:n - lag]
    else:
        L = -lag
        grad[L:n] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:n]
    return grad


def _soft_max_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx


def _soft_max_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    thresh = 1.0 / (len(corr) * 20)
    active = np.where(weights > thresh)[0]
    for k_idx in active:
        lag = k_idx - (n - 1)
        grad += weights[k_idx] * _grad_corr_lag(h, lag)
    return grad * dx


def _softmax_gd(h_init, n, target_sum, alpha_start, alpha_end, n_iter,
                lr0, momentum_start, momentum_end, deadline, check_every=200):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    v = np.zeros(n)
    no_improve = 0
    patience = check_every * 50
    dx = 2.0 / n

    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        frac = t / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        mom = momentum_start + (momentum_end - momentum_start) * frac

        grad = _soft_max_grad(h, alpha, dx)
        lr = lr0 / (1.0 + t * 0.0001)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-8:
            lr = min(lr, 0.05 / gnorm)

        v = mom * v - lr * grad
        h = _fix_sum(h + v, target_sum)

        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break

    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def _slsqp_refine(h_init, n, alpha, maxiter=600, ftol=1e-15):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

    def obj_grad(x):
        return _soft_max_grad(x, alpha, dx)

    bounds = [(0.0, 1.0)] * npts

    def cons_fun(x):
        return x.sum() - target_sum

    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()


def _coordinate_descent(h_init, n, target_sum, n_sweep=3):
    h = h_init.copy()
    best_c5 = _c5_of(h)
    dx = 2.0 / n
    for sweep in range(n_sweep):
        improved = False
        for i in range(n):
            for delta in [0.002, -0.002, 0.005, -0.005, 0.01, -0.01]:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                h_try = _fix_sum(h_try, target_sum)
                c5_try = _c5_of(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def _make_sine_init(n, freq, phase, amp, target_sum):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    return _fix_sum(h, target_sum)
