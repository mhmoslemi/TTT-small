```python
def compute_swap_delta(h, i, j, n):
    """Compute the change in full correlation when h[i]:1->0 and h[j]:0->1"""
    delta = np.zeros(2*n-1)
    # term 1: -(1-h[i-d]) for d such that 0<=i-d<n, d = m-(n-1), m in [0, 2n-2]
    # i-d valid means 0<=i-d<n, d = m-(n-1), so m = i-d+(n-1)
    # d ranges from i-(n-1) to i (for 0<=i-d<n)
    # m ranges from i-(n-1)+(n-1) to i+(n-1), i.e., from i to i+n-1
    # But m must be in [0, 2n-2]
    m_start1 = max(0, i)
    m_end1 = min(2*n-2, i+n-1)
    if m_start1 <= m_end1:
        d_vals = np.arange(m_start1, m_end1+1) - (n-1)  # d = m-(n-1)
        idx = i - d_vals  # i-d
        delta[m_start1:m_end1+1] -= (1.0 - h[idx])
    
    # term 2: +(1-h[j-d]) similarly
    m_start2 = max(0, j)
    m_end2 = min(2*n-2, j+n-1)
    if m_start2 <= m_end2:
        d_vals = np.arange(m_start2, m_end2+1) - (n-1)
        idx = j - d_vals
        delta[m_start2:m_end2+1] += (1.0 - h[idx])
    
    # term 3: +h[i+d] for d such that 0<=i+d<n
    # d = m-(n-1), so m = i+d+(n-1)
    # d ranges from -i to n-1-i
    # m ranges from i+(-i)+(n-1) to i+(n-1-i)+(n-1) = n-1 to 2n-2
    m_start3 = max(0, n-1-i)
    m_end3 = min(2*n-2, 2*n-2-i)
    if m_start3 <= m_end3:
        d_vals = np.arange(m_start3, m_end3+1) - (n-1)
        idx = i + d_vals
        delta[m_start3:m_end3+1] += h[idx]
    
    # term 4: -h[j+d]
    m_start4 = max(0, n-1-j)
    m_end4 = min(2*n-2, 2*n-2-j)
    if m_start4 <= m_end4:
        d_vals = np.arange(m_start4, m_end4+1) - (n-1)
        idx = j + d_vals
        delta[m_start4:m_end4+1] -= h[idx]
    
    return delta
```

```python
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):

    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid

def to_binary(h, n):

    h_sorted = np.sort(h)[::-1]
    # Find threshold: top n/2 values become 1, rest become 0
    # But we need sum = n/2 exactly
    # If n is even: n/2 ones, n/2 zeros
    # If n is odd: (n-1)/2 ones, (n-1)/2 zeros, one 0.5
    if n % 2 == 0:

        binary = np.zeros(n)
        binary[np.argsort(h)[::-1][:k]] = 1.0
    else:

        binary[np.argsort(h)[::-1][:k]] = 1.0

        remaining = np.where(binary == 0)[0]
        if len(remaining) > 0:
            idx = remaining[np.argmax(h[remaining])]
            binary[idx] = 0.5
    return binary
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def to_binary(h, n):
    if n % 2 == 0:
        k = n // 2
        binary = np.zeros(n)
        idx = np.argsort(h)[::-1][:k]
        binary[idx] = 1.0
    else:
        k = (n - 1) // 2
        binary = np.zeros(n)
        idx = np.argsort(h)[::-1][:k]
        binary[idx] = 1.0
        remaining = np.where(binary == 0)[0]
        if len(remaining) > 0:
            best_rem = remaining[np.argmax(h[remaining])]
            binary[best_rem] = 0.5
    return binary


def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = (np.binary_repr(i).count('1')) % 2
    return tm


def swap_delta(h, i, j, n):
    m = 2 * n - 1
    delta = np.zeros(m)
    lo_n = n - 1
    ms = i
    me = min(m - 1, i + n - 1)
    if ms <= me:
        d = np.arange(ms, me + 1) - lo_n
        idx = i - d
        delta[ms:me + 1] -= (1.0 - h[idx])
    ms = j
    me = min(m - 1, j + n - 1)
    if ms <= me:
        d = np.arange(ms, me + 1) - lo_n
        idx = j - d
        delta[ms:me + 1] += (1.0 - h[idx])
    ms = max(0, lo_n - i)
    me = min(m - 1, 2 * lo_n - i)
    if ms <= me:
        d = np.arange(ms, me + 1) - lo_n
        idx = i + d
        delta[ms:me + 1] += h[idx]
    ms = max(0, lo_n - j)
    me = min(m - 1, 2 * lo_n - j)
    if ms <= me:
        d = np.arange(ms, me + 1) - lo_n
        idx = j + d
        delta[ms:me + 1] -= h[idx]
    return delta


def binary_anneal(h, n, n_iters, rng, t0, time_budget):
    corr = compute_correlations(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    dx = 2.0 / n
    T_start = 5.0
    T_end = 0.05
    ones = np.where(h > 0.4)[0]
    zeros = np.where(h < 0.6)[0]
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        frac = it / max(1, n_iters)
        T = T_start * (T_end / T_start) ** frac
        if it % 50 == 0 and it > 0:
            worst_idx = np.argmax(corr)
            lag = worst_idx - (n - 1)
            if lag > 0 and lag < n:
                found = False
                for _ in range(200):
                    n_pos = rng.randint(n - lag)
                    i_pos = n_pos + lag
                    j_pos = n_pos
                    if h[i_pos] > 0.5 and h[j_pos] < 0.5:
                        delta = swap_delta(h, i_pos, j_pos, n)
                        new_max = float(np.max(corr + delta))
                        if new_max < current_max or (rng.random() < np.exp((current_max - new_max) / T)):
                            h[i_pos] = 1.0 - h[i_pos]
                            h[j_pos] = 1.0 - h[j_pos]
                            corr += delta
                            current_max = float(np.max(corr))
                            if current_max < best_max:
                                best_max = current_max
                                best_h = h.copy()
                            ones = np.where(h > 0.4)[0]
                            zeros = np.where(h < 0.6)[0]
                            found = True
                        if found:
                            break
            if not found:

                j = zeros[rng.randint(len(zeros))]

                    continue
                delta = swap_delta(h, i, j, n)
                new_max = float(np.max(corr + delta))
                if new_max < current_max or (rng.random() < np.exp((current_max - new_max) / T)):
                    h[i] = 1.0 - h[i]
                    h[j] = 1.0 - h[j]
                    corr += delta
                    current_max = float(np.max(corr))
                    if current_max < best_max:
                        best_max = current_max
                        best_h = h.copy()
                    ones = np.where(h > 0.4)[0]

        else:

def block_swap_search(h, n, n_iters, rng, t0, time_budget):
    corr = compute_correlations(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        blen = rng.randint(2, min(15, n // 10))
        start = rng.randint(0, n - blen)
        end = start + blen
        block = h[start:end]
        n_ones_in_block = int(np.sum(block > 0.5))
        if n_ones_in_block == 0 or n_ones_in_block == blen:
            continue
        comp_start = rng.randint(0, n - blen)
        if abs(comp_start - start) < blen:
            continue
        comp_end = comp_start + blen
        comp_block = h[comp_start:comp_end]
        n_ones_in_comp = int(np.sum(comp_block > 0.5))
        if n_ones_in_comp == n_ones_in_block:
            continue
        h_try = h.copy()
        h_try[start:end] = comp_block
        h_try[comp_start:comp_end] = block
        if np.abs(np.sum(h_try) - n / 2.0) > 1e-6:
            continue
        corr_try = compute_correlations(h_try)
        max_try = float(np.max(corr_try))
        if max_try < current_max - 1e-14:
            h = h_try
            corr = corr_try
            current_max = max_try
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    return best_h, best_max


def continuous_refine(h0, n, alpha, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_shifted = np.max(shifted)
        sm = m + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s

    def grad(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_shifted = np.max(shifted)
        exp_sh = np.exp(shifted - max_shifted)
        weights = exp_sh / np.sum(exp_sh)
        N = len(h)
        g = np.zeros(N)
        for idx in range(len(weights)):
            w = weights[idx]
            if w < 1e-12:
                continue
            lag = N - 1 - idx
            if lag == 0:
                g += w * (1.0 - 2.0 * h)
            elif lag > 0 and lag < N:
                g[:N - lag] += w
                g[:N - lag] -= w * h[lag:]
                g[lag:] -= w * h[:N - lag]
            elif -lag < N:
                l = -lag
                g[l:] += w
                g[l:] -= w * h[:N - l]
                g[:N - l] -= w * h[l:]
        s = np.sum(h) - N / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def greedy_hill_climb(h, n, rng, t0, time_budget, n_rounds=50):
    corr = compute_correlations(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    for rnd in range(n_rounds):
        if time.time() - t0 > time_budget:
            break
        improved = False
        ones = np.where(h > 0.4)[0]
        zeros = np.where(h < 0.6)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if time.time() - t0 > time_budget:
                break
            for j in zeros:
                if time.time() - t0 > time_budget:
                    break
                if i == j:
                    continue
                delta = swap_delta(h, i, j, n)
                new_max = float(np.max(corr + delta))
                if new_max < current_max - 1e-14:
                    h[i] = 1.0 - h[i]
                    h[j] = 1.0 - h[j]
                    corr += delta
                    current_max = new_max
                    improved = True
                    if current_max < best_max:
                        best_max = current_max
                        best_h = h.copy()
                    break
    return best_h, best_max
```
