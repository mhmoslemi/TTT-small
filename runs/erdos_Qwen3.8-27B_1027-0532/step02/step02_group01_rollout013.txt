import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_and_max_lag(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    kmax = np.argmax(corr)
    return float(corr[kmax]), kmax, corr

def binary_from_continuous(h, target_sum):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    h_bin = np.zeros(n)
    h_bin[order[n//2:]] = 1.0
    return h_bin

    h = np.round(h_init).astype(float)
    # Ensure exactly n/2 ones
    if h.sum() != n/2:
        h = binary_from_continuous(h_init, n/2)
    
    current_c5, _, _ = compute_c5_and_max_lag(h)
    best_h = h.copy()
    best_c5 = current_c5
    best_iter = 0
    no_improve = 0
    
    tabu_moves = []
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Generate candidate moves
        moves = []
        
        # Single swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(5):
            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
            if (oi, zi) in tabu_moves or (zi, oi) in tabu_moves:
                continue
            moves.append(('swap', oi, zi))
        
        # Block reversal
        blen = rng.integers(2, min(6, n//10 + 1))
        a = rng.integers(0, n - blen + 1)
        if not any(m[0] == 'rev' and m[1] == a for m in tabu_moves):
            moves.append(('rev', a, blen))
        
        # Double swap
        if len(ones) > 2 and len(zeros) > 2:
            o1 = ones[rng.integers(len(ones))]
            o2 = ones[rng.integers(len(ones))]
            z1 = zeros[rng.integers(len(zeros))]
            z2 = zeros[rng.integers(len(zeros))]
            if o1 != o2 and z1 != z2:
                moves.append(('dswap', o1, o2, z1, z2))
        
        # Evaluate moves and pick best
        best_move = None
        best_move_c5 = current_c5 + 0.1  # threshold
        best_move_delta = 0.1
        
        for move in moves:
            h_new = h.copy()
            if move[0] == 'swap':
                _, oi, zi = move
                h_new[oi] = 0.0
                h_new[zi] = 1.0
            elif move[0] == 'rev':
                _, a, blen = move
                h_new[a:a+blen] = h_new[a:a+blen][::-1]
            elif move[0] == 'dswap':
                _, o1, o2, z1, z2 = move
                h_new[o1] = 0.0
                h_new[o2] = 0.0
                h_new[z1] = 1.0
                h_new[z2] = 1.0

            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
            h[oi] = 0.0
            h[zi] = 1.0
            current_c5 = compute_c5(h)
            tabu_moves.append(('swap', oi, zi))
            if len(tabu_moves) > tabu_len:
                tabu_moves.pop(0)
            continue
        
        # Apply best move
        if best_move[0] == 'swap':
            _, oi, zi = best_move
            h[oi] = 0.0
            h[zi] = 1.0
        elif best_move[0] == 'rev':
            _, a, blen = best_move
            h[a:a+blen] = h[a:a+blen][::-1]

            h[o1] = 0.0

            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

For a swap at positions a (1→0) and b (0→1):
The correlation at lag s (where s = k - (n-1) for the k-th element of the "full" output) changes by:

Let me define corr[s] = sum_i h[i] * (1-h[i+s]) where the sum is over valid i.

After the swap:
- h[a] changes from 1 to 0
- h[b] changes from 0 to 1

- From the term where i=a: was h[a]*(1-h[a+s]) = 1-h[a+s], now 0. Change: -(1-h[a+s]) = h[a+s]-1
- From the term where i+s=a, i=a-s: was h[a-s]*(1-h[a]) = h[a-s]*0 = 0, now h[a-s]*1 = h[a-s]. Change: +h[a-s]
  (valid if 0 ≤ a-s ≤ n-1 and a-s ≠ a and a-s ≠ b)
- From the term where i=b: was 0, now 1-h[b+s]. Change: +(1-h[b+s])
- From the term where i+s=b, i=b-s: was h[b-s]*(1-h[b]) = h[b-s]*1 = h[b-s], now h[b-s]*0 = 0. Change: -h[b-s]
  (valid if 0 ≤ b-s ≤ n-1 and b-s ≠ a and b-s ≠ b)

But we need to be careful about overlaps: if a+s = b, then the change at i=a involves h[b], which is also changing. Similarly for other special cases.

This incremental update is O(1) per lag, so O(n) total per swap. That's much faster than O(n²) for the full correlation.

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def make_binary(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    h_bin = np.zeros(n)
    h_bin[order[n // 2:]] = 1.0
    return h_bin

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def upsample_block(h, factor):
    n = len(h)
    n_new = n * factor
    h_new = np.zeros(n_new)
    for i in range(n):
        h_new[i * factor:(i + 1) * factor] = h[i]
    return h_new

def local_search_sampled(h, n_trials, time_limit, start_time, rng, tabu_size=60):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    tabu = []
    tabu_set = set()
    stuck = 0

    while time.time() - start_time < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        candidates = []
        for _ in range(n_trials):
            oi = int(ones[rng.integers(len(ones))])
            zi = int(zeros[rng.integers(len(zeros))])
            if (oi, zi) not in tabu_set:
                candidates.append(('S', oi, zi))

        for blen in [2, 3, 4, 5]:
            a = int(rng.integers(0, n - blen + 1))
            candidates.append(('R', a, blen))

        if rng.random() < 0.3:

            a = int(rng.integers(0, n - blen + 1))

        best_cand = None

                h[oi] = 0.0
                h[zi] = 1.0

            oi = int(ones[rng.integers(len(ones))])

def steepest_descent_binary(h, time_limit, start_time):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    while time.time() - start_time < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        best_c5_step = current_c5
        best_oi = -1
        best_zi = -1

        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                c5 = compute_c5(h)
                h[oi] = 1.0
                h[zi] = 0.0
                if c5 < best_c5_step - 1e-14:
                    best_c5_step = c5
                    best_oi = oi
                    best_zi = zi

        if best_oi < 0:
            break

        h[best_oi] = 0.0
        h[best_zi] = 1.0
        current_c5 = best_c5_step

        if current_c5 < best_c5 - 1e-14:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def continuous_optimize(h_init, eps, maxiter, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)}

    def obj(x):
        return smooth_objective(x, eps)

    try:
        result = minimize(obj, h0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

        h_ls, c5_ls = local_search_sampled(best_h, 300, tl, start_time, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    remaining = budget_s - (time.time() - start_time)

    # Phase 3: Multiple restarts at n0
    restart_count = 0
    while remaining > 100 and restart_count < 15:
        strategy = restart_count % 5
        if strategy == 0:
            h_pert = best_h.copy()
            nflip = max(2, n0 // 10)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = make_binary(h_pert)
        elif strategy == 1:
            h_pert = best_h.copy()
            blen = int(rng.integers(3, max(4, n0 // 6)))
            a = int(rng.integers(0, n0 - blen + 1))
            h_pert[a:a + blen] = h_pert[a:a + blen][::-1]
        elif strategy == 2:
            h_cont = best_h + 0.15 * rng.standard_normal(n0)
            h_pert = make_binary(project_onto_constraint(h_cont, target_sum))
        elif strategy == 3:
            h_rand = (rng.random(n0) < 0.5).astype(float)
            alpha = 0.6
            h_mix = alpha * best_h + (1 - alpha) * h_rand
            h_pert = make_binary(h_mix)
        else:
            h_pert = best_h.copy()
            nswaps = int(rng.integers(3, 8))
            for _ in range(nswaps):
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
                oi = int(ones[rng.integers(len(ones))])
                zi = int(zeros[rng.integers(len(zeros))])
                h_pert[oi] = 0.0
                h_pert[zi] = 1.0

        tl = min(remaining * 0.15, 60)
        h_ls, c5_ls = local_search_sampled(h_pert, 200, tl, start_time, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        restart_count += 1
        remaining = budget_s - (time.time() - start_time)

    remaining = budget_s - (time.time() - start_time)

    # Phase 4: Upsample to 2*n0 and search
    if remaining > 200:
        n1 = 2 * n0
        h_up = upsample_block(best_h, 2)
        h_up_bin = make_binary(h_up)
        c5_up = compute_c5(h_up_bin)

        if c5_up <= best_c5 + 0.005:
            h_work = h_up_bin.copy()
            tl_up = min(remaining * 0.4, 400)
            h_ls_up, c5_ls_up = local_search_sampled(h_work, 300, tl_up, start_time, rng)
            if c5_ls_up < best_c5:
                best_c5 = c5_ls_up
                best_h = h_ls_up.copy()

            remaining = budget_s - (time.time() - start_time)
            if remaining > 100 and c5_ls_up < best_c5 + 1e-6:
                restarts_up = 0
                while remaining > 60 and restarts_up < 8:
                    h_pert = best_h.copy()
                    nflip = max(3, len(best_h) // 12)
                    idx = rng.choice(len(best_h), size=nflip, replace=False)
                    h_pert[idx] = 1.0 - h_pert[idx]
                    tl = min(remaining * 0.12, 50)
                    h_ls, c5_ls = local_search_sampled(h_pert, 250, tl, start_time, rng)
                    if c5_ls < best_c5:
                        best_c5 = c5_ls
                        best_h = h_ls.copy()
                    restarts_up += 1
                    remaining = budget_s - (time.time() - start_time)

    remaining = budget_s - (time.time() - start_time)

    # Phase 5: Continuous polishing
    n_cur = len(best_h)
    if remaining > 60:
        eps_list = [0.01, 0.005, 0.001, 0.0005, 0.0001]
        for eps in eps_list:
            if time.time() - start_time > budget_s - 20:
                break
            tl = min(remaining * 0.15, 40)
            h_cont = best_h + 0.01 * rng.standard_normal(n_cur)
            h_cont = project_onto_constraint(h_cont, n_cur / 2.0)
            h_opt, c5_opt = continuous_optimize(h_cont, eps, 200, tl, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break

    remaining = budget_s - (time.time() - start_time)

    # Phase 6: Final binary polish
    if remaining > 30:
        h_final_bin = make_binary(best_h)
        c5_final_bin = compute_c5(h_final_bin)
        if c5_final_bin < best_c5:
            best_h = h_final_bin
            best_c5 = c5_final_bin
        tl = min(remaining - 10, 30)
        h_ls, c5_ls = local_search_sampled(best_h, 500, tl, start_time, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final: try continuous one more time if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        h_final_cont = best_h + 0.005 * rng.standard_normal(len(best_h))
        h_final_cont = project_onto_constraint(h_final_cont, len(best_h) / 2.0)
        h_opt, c5_opt = continuous_optimize(h_final_cont, 0.0001, 300, min(remaining - 5, 15), start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final projection and verification
    n_final = len(best_h)
    final_h = project_onto_constraint(best_h, n_final / 2.0)
    final_c5 = compute_c5(final_h)

    # If binary is slightly better, use it
    h_bin_final = make_binary(final_h)
    c5_bin_final = compute_c5(h_bin_final)
    if c5_bin_final < final_c5:
        final_h = h_bin_final
        final_c5 = c5_bin_final

    return final_h, final_c5, len(final_h)
