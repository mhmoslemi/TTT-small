        return grad_lse(hv, alpha)

        return np.ones(n)

Actually, you know what, let me try a completely different approach: **Alternating minimization / Frank-Wolfe (conditional gradient) method**.

The Frank-Wolfe method works as follows for minimizing f(x) over a convex set C:
1. At each iteration, compute g = ∇f(x)
2. Find s = argmin_{y ∈ C} g·y (the "linear oracle")
3. Update: x ← x + γ*(s - x) where γ is the step size

For our problem, C = {x : 0≤x_i≤1, Σx_i = n/2}. The linear oracle is:
s_j = 0 if g_j > 0 (and we need to add mass elsewhere)
s_j = 1 if g_j < 0 (and we need to remove mass from elsewhere)
With the constraint Σs_j = n/2, we sort g and pick the threshold.

This is actually very efficient! The linear oracle is O(n log n) (just sorting).

Let me implement this:

Wait, that's not quite right. We need sum(s) = target_sum = n/2, and s ∈ [0,1]^n. The linear oracle for this set gives:
- Sort g in increasing order
- Set s_j = 1 for the smallest g values
- Set s_j = 0 for the largest g values
- One element is fractional to hit the exact sum

But actually, for the Frank-Wolfe method, we just need a minimizer of g·s over C. Since C is a polytope (box ∩ hyperplane), the minimum is achieved at a vertex, which has exactly n/2 ones and n/2 zeros (assuming n/2 is integer).

So the linear oracle is: sort g, set the n/2 smallest to 1, the rest to 0.

The Frank-Wolfe update is: x ← (1-γ)x + γ*s

This is nice because:
- Each iteration is O(n log n) for sorting + O(n) for the update
- No projection needed (the convex combination stays in C)
- Converges at O(1/T) rate (or O(1/T²) with strong step sizes)

def project_to_feasible(h, target_sum):

def lse_value(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def grad_lse_topk(h, alpha, n_top):

    best_score = evaluate_score(h_init)
    best_n = n
    rng = np.random.RandomState(seed)

    # Phase 1: FW optimization at original resolution (25% budget)
    if remaining() > budget_s * 0.25:
        t1 = budget_s * 0.22
        for alpha_try in [30, 100, 300]:
            if remaining() < budget_s * 0.15:
                break
            h_fw, s_fw = fw_optimize(best_h, alpha_try, 50, 500000, t0, min(remaining() * 0.6, t1 * 0.4), use_away=True)
            if s_fw < best_score:
                best_score = s_fw
                best_h = h_fw.copy()

    # Phase 2: Subgradient refinement (20% budget)
    if remaining() > budget_s * 0.18:
        configs = [
            (50, 20, 0.5), (100, 30, 0.3), (200, 20, 0.2),
            (300, 50, 0.15), (80, 10, 0.4),
        ]
        for i, (a, kt, si) in enumerate(configs):
            if remaining() < budget_s * 0.12:
                break
            t_allow = min(remaining() * 0.25, budget_s * 0.04)
            if t_allow < 3:
                break
            h, s = subgradient_run(best_h, a, kt, si, 50000, seed + 100 + i, t0, t_allow)
            if s < best_score:
                best_score = s
                best_h = h.copy()

    # Phase 3: FISTA at original resolution (15% budget)
    if remaining() > budget_s * 0.12:
        t3 = min(remaining() * 0.5, budget_s * 0.12)
        if t3 > 10:
            h_f, s_f = fista_optimize(best_h, 200, 40, 100000, t0, t3)
            if s_f < best_score:
                best_score = s_f
                best_h = h_f.copy()

    # Phase 4: Upscale to 2x resolution (25% budget)
    if remaining() > budget_s * 0.20:
        n2 = 2 * n
        h2 = upscale_h(best_h, n, n2)
        h2 = project_to_feasible(h2, n2 / 2.0)

        t4_total = budget_s * 0.22
        if remaining() > t4_total:
            # FW at higher resolution
            h_fw2, s_fw2 = fw_optimize(h2, 100, 50, 300000, t0, min(remaining() * 0.4, t4_total * 0.5), use_away=True)
            if s_fw2 < best_score:
                best_score = s_fw2
                best_h = h_fw2.copy()
                best_n = n2

            # Subgradient at higher resolution
            if remaining() > budget_s * 0.15:
                for i in range(3):
                    if remaining() < budget_s * 0.12:
                        break
                    t_allow = min(remaining() * 0.3, budget_s * 0.04)
                    if t_allow < 3:
                        break
                    h, s = subgradient_run(best_h, 150, 30, 0.2, 50000, seed + 5000 + i, t0, t_allow)
                    if s < best_score:
                        best_score = s
                        best_h = h.copy()

            if remaining() > budget_s * 0.12:
                h_f2, s_f2 = fista_optimize(best_h, 300, 50, 50000, t0, min(remaining() * 0.3, 60))
                if s_f2 < best_score:
                    best_score = s_f2
                    best_h = h_f2.copy()

    # Phase 5: SLSQP refinement (10% budget)
    if remaining() > 50:
        alphas = [50, 200, 1000, 3000]
        for a in alphas:
            if remaining() < 25:
                break
            h_ref = slsqp_refine(best_h, a, maxiter=150)
            if h_ref is not None:
                s = evaluate_score(h_ref)
                if s < best_score:
                    best_score = s
                    best_h = h_ref.copy()

    # Phase 6: Final fine-tuning
    if remaining() > 30:
        t6 = remaining() * 0.5
        h, s = subgradient_run(best_h, 500, 20, 0.1, 50000, seed + 99999, t0, t6)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    if remaining() > 20:
        for a in [500, 2000, 5000]:
            if remaining() < 10:
                break
            h_ref = slsqp_refine(best_h, a, maxiter=100)
            if h_ref is not None:
                s = evaluate_score(h_ref)
                if s < best_score:
                    best_score = s
                    best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
