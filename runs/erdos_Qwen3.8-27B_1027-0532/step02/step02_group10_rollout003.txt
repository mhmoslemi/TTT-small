- ∇_h [sum_k w_k O_k(h) + λ sum(h)] = 0 (stationarity)
- w_k (O_k(h) - t) = 0 (complementary slackness)

2. Update h by gradient descent on the weighted sum
3. Update w_k by projected gradient ascent on the dual

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_weights(x, alpha):
    m = np.max(x)
    sh = np.clip(alpha * (x - m), -70, 0)
    e = np.exp(sh)
    s = e.sum()
    if s < 1e-30:
        w = np.full(len(x), 1.0 / len(x))
        return m, w
    return m + np.log(s) / alpha, e / s

def grad_smooth_max(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        a, b = max(0, -off), min(n, n - off)
        if a < b:
            g[a:b] += wk * (1.0 - h[a + off:b + off])
        a2, b2 = max(0, off), min(n, n + off)
        if a2 < b2:
            g[a2:b2] -= wk * h[a2 - off:b2 - off]
    return g * dx
