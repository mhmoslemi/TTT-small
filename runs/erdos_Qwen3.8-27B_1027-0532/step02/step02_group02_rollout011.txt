        # Second term: m in [0, n-1-L]: grad[m] -= h[m+L]
        grad[:n-L] -= h[L:]
    return grad

        for i in order:
            if diff <= 1e-12:

            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad_at(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-lag] += 1.0 - h[lag:]
        grad[lag:] -= h[:n-lag]
    else:
        L = -lag
        grad[L:] += 1.0 - h[:n-L]
        grad[:n-L] -= h[L:]
    return grad

def _sm_oag(h, alpha):
    """Smooth max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    ew = np.exp(shifted)
    sw = np.sum(ew)
    obj = (m + np.log(sw) / alpha) * dx
    weights = ew / sw
    # Active lags
    idx = np.where(weights > 1e-12)[0]
    grad = np.zeros(n)
    for k in idx:
        lag = k - (n - 1)
        grad += weights[k] * _corr_grad_at(h, lag)
    grad *= dx
    return obj, grad

def _proj_grad(h0, n, target_sum, alpha, n_iter, lr0, lr_decay, momentum, top_k, deadline):
    """Projected gradient descent with momentum."""
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_obj = _sm_oag(h, alpha)[0]
    best_c5 = _c5_of(h)
    
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        lr = lr0 / (1.0 + t * lr_decay)
        obj, grad = _sm_oag(h, alpha)
        v = momentum * v - lr * grad
        h = _fix_sum(h + v, target_sum)
        if (t + 1) % 500 == 0:
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_refine(h0, n, alpha, maxiter):
    """SLSQP with analytical gradient."""
    target_sum = n / 2.0
    
    def objfun(x):
        return _sm_oag(x, alpha)[0]
    
    def objgrad(x):
        return _sm_oag(x, alpha)[1]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(objfun, h0, jac=objgrad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except Exception:
        return h0.copy()

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad_at(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] += 1.0 - h[lag:]
        grad[lag:] -= h[:n - lag]
    else:
        L = -lag
        grad[L:] += 1.0 - h[:n - L]
        grad[:n - L] -= h[L:]
    return grad

def _sm_oag(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    ew = np.exp(shifted)
    sw = np.sum(ew)
    obj = (m + np.log(sw) / alpha) * dx
    weights = ew / sw
    idx = np.where(weights > 1e-10)[0]
    grad = np.zeros(n)
    for k in idx:
        lag = k - (n - 1)
        grad += weights[k] * _corr_grad_at(h, lag)
    grad *= dx
    return obj, grad

def _proj_grad(h0, n, target_sum, alpha, n_iter, lr0, lr_decay, momentum, deadline):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    check = 200
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        lr = lr0 / (1.0 + t * lr_decay)
        obj, grad = _sm_oag(h, alpha)
        v = momentum * v - lr * grad
        h = _fix_sum(h + v, target_sum)
        if (t + 1) % check == 0:
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_refine(h0, n, alpha, maxiter):
    target_sum = n / 2.0

    def objfun(x):
        return _sm_oag(x, alpha)[0]

    def objgrad(x):
        return _sm_oag(x, alpha)[1]

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objfun, h0, jac=objgrad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h0.copy()

def _multi_alpha_refine(h0, n, target_sum, deadline, alphas, lrs, n_iter_per=15000):
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_c5 = _c5_of(h_cur)
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 20:
            break
        lr = lrs[i] if i < len(lrs) else 0.01
        h_cur, c5_cur = _proj_grad(h_cur, n, target_sum, alpha,
                                    n_iter=n_iter_per, lr0=lr,
                                    lr_decay=0.0005, momentum=0.9,
                                    deadline=deadline)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    return best_h, best_c5

def _try_perturb(h_best, n, target_sum, seed, deadline, n_trials=6):
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = _c5_of(h_best)

                                deadline=deadline)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5
