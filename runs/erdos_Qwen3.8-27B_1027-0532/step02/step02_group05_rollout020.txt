def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    N = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(60):

        elif sg_max < 1e-10:
            d_lo = d_mid
        else:
            break
    return np.clip(cur_val + (d_lo + d_hi) * 0.5, 0.0, 1.0)

def coord_descent(h, time_limit):
    """Full coordinate descent with sum constraint projection."""
    N = len(h)
    target = N / 2.0
    corr = compute_corr(h)
    best_h = h.copy()

    t0 = time.time()
    sweep = 0
    
    while time.time() - t0 < time_limit:

            new_val = bisection_1d(corr, A, h[i])
            delta = new_val - h[i]

                h[i] = new_val
        # Project and recompute
        h = project_sum(h, target)
        corr = compute_corr(h)
        score = np.max(corr) * (2.0/N)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        sweep += 1
        # Check for convergence
        if sweep > 10 and score > best_score - 1e-10:
            pass  # Still keep going, might break through

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_sum(h, target):
    N = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def sens_vec(h, i):
    N = len(h)
    A = np.zeros(2 * N - 1)
    j_lo = max(0, i - 1)
    j_hi = min(2 * N - 2, i + N - 2)
    if j_lo <= j_hi:
        k_vals = np.arange(j_lo, j_hi + 1) - (N - 1)
        idx = i - k_vals
        A[j_lo:j_hi + 1] += (1.0 - h[idx])
    j_lo2 = max(0, N - 1 - i)
    j_hi2 = min(2 * N - 2, 2 * N - 2 - i)
    if j_lo2 <= j_hi2:
        k_vals = np.arange(j_lo2, j_hi2 + 1) - (N - 1)
        idx = i + k_vals
        A[j_lo2:j_hi2 + 1] -= h[idx]
    return A


def bisection_1d(corr, S, cur_val, n_bisect=45):
    d_lo = -cur_val
    d_hi = 1.0 - cur_val
    for _ in range(n_bisect):
        d_mid = (d_lo + d_hi) * 0.5
        vals = corr + d_mid * S
        m_val = np.max(vals)
        active = vals >= m_val - 1e-9
        sg_min = float(np.min(S[active]))
        sg_max = float(np.max(S[active]))
        if sg_min > -1e-10:
            d_hi = d_mid
        elif sg_max < 1e-10:
            d_lo = d_mid
        else:
            break
    return float(np.clip(cur_val + (d_lo + d_hi) * 0.5, 0.0, 1.0))


def coord_descent_full(h, time_limit):
    N = len(h)
    target = N / 2.0
    corr = compute_corr(h)
    best_h = h.copy()
    best_score = float(np.max(corr)) * (2.0 / N)
    t0 = time.time()
    no_improve = 0
    while time.time() - t0 < time_limit:
        for i in range(N):
            A = sens_vec(h, i)
            new_val = bisection_1d(corr, A, h[i])
            delta = new_val - h[i]
            if abs(delta) > 1e-13:
                corr += delta * A
                h[i] = new_val
        h = project_sum(h, target)
        corr = compute_corr(h)
        score = float(np.max(corr)) * (2.0 / N)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            break
    return best_h, best_score

def smooth_max_val(x, alpha):
    m = float(np.max(x))
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_corr(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def slsqp_opt(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    target = N / 2.0

    def obj_func(x):
        return smooth_max_val(compute_corr(x), alpha)

    def jac_func(x):
        return grad_obj(x, alpha)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(len(x))

    res = minimize(
        obj_func, h0, method='SLSQP', bounds=bounds,
        jac=jac_func, args=(),
        constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None
