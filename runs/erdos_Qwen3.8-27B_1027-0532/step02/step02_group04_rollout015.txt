import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(v, target_sum):
    v = np.asarray(v, dtype=float).copy()
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def solve_lp_alt(q, n, A_bound, A_eq, b_eq, c_obj):
    """Solve the alternating LP."""
    n_lags = 2 * n - 1
    A_lag = np.zeros((n_lags, n + 1))
    A_lag[:, 0] = 1.0
    
    for m in range(n_lags):
        k = m - (n - 1)
        if k >= 0:
            A_lag[m, 1:(n-k)+1] = -q[k:]
        else:
            l = -k
            A_lag[m, 1+l:] = -q[:n-l]
    
    A_ub = np.vstack([A_lag, A_bound])
    b_ub = np.concatenate([np.zeros(n_lags), np.ones(n), np.zeros(n)])
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if result.success:
        return result.x[1:]
    return None

    for it in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for step in steps:
            h_try = best.copy()
            h_try[i] += step
            h_try[j] -= step
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            max_try = np.max(corr_try)
            if max_try < best_max - 1e-14:
                best = h_try
                best_corr = corr_try
                best_max = max_try
                break
    return best
