def solve_1d_minimax(a, b, n_bisect=60):
    """Solve min_{t in [0,1]} max_k (a[k] + b[k]*t) using subgradient bisection."""
    lo, hi = 0.0, 1.0
    for _ in range(n_bisect):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        f_mid = np.max(vals)
        # Active set: indices where vals is within tolerance of max
        tol = 1e-12 * max(1.0, abs(f_mid))
        active = np.where(vals >= f_mid - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        sub_min = np.min(b[active])
        sub_max = np.max(b[active])
        if sub_min > 1e-14:
            # All active slopes positive → f is increasing → go left
            hi = mid
        elif sub_max < -1e-14:
            # All active slopes negative → f is decreasing → go right
            lo = mid
        else:
            # 0 is in subgradient → optimal (or in flat region)
            break
    return (lo + hi) / 2.0

            h_cur = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.85
            h_cur = best_h.copy()
        if lr < 1e-6:
            break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP refinement on initial grid
    if tleft() > 250:
        alphas = [5, 15, 40, 100, 300, 800]
        best_h, best_score = slsqp_refine(best_h, n_init, target_init, alphas, 1000, 1e-14)

    # Phase 2: Subgradient polish
    if tleft() > 200:
        best_h, best_score = subgrad_polish(best_h, target_init, n_steps=300, lr_init=0.03)

    # Phase 3: Coordinate descent on initial grid
    if tleft() > 180:
        best_h, best_score = coord_descent(best_h, target_init, n_passes=8, grid_size=500)

    # Phase 4: Resolution increase with refinement
    n_targets = [500, 640, 800, 1000, 1200]
    n_targets = [nt for nt in n_targets if nt != n_init and nt % 2 == 0]
    for n_try in n_targets:
        if tleft() < 350:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [10, 40, 150, 400]
        h_new, sc = slsqp_refine(h_interp, n_try, target_try, alphas_try, 600, 1e-14)
        if tleft() > 250:
            h_new, sc = subgrad_polish(h_new, target_try, n_steps=150, lr_init=0.02)
        if tleft() > 200:
            h_new, sc = coord_descent(h_new, target_try, n_passes=4, grid_size=400)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Iterated local search on best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    for attempt in range(10):
        if tleft() < 100:
            break
        n_pert = rng.randint(max(3, n_use // 20), n_use // 5)
        indices = rng.choice(n_use, size=n_pert, replace=False)
        h_pert = best_h.copy()
        h_pert[indices] = rng.uniform(0.0, 1.0, size=n_pert)
        h_pert = project_feasible(h_pert, target_use)
        alphas_r = [20, 80, 300]
        h_new, sc = slsqp_refine(h_pert, n_use, target_use, alphas_r, 400, 1e-14)
        if tleft() > 80:
            h_new, sc = coord_descent(h_new, target_use, n_passes=2, grid_size=300)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Final high-alpha polish
    if tleft() > 80:
        alphas_final = [200, 500, 1000, 2000]
        best_h, best_score = slsqp_refine(best_h, n_use, target_use, alphas_final, 300, 1e-15)

    # Phase 7: Final coordinate descent
    if tleft() > 50:
        best_h, best_score = coord_descent(best_h, target_use, n_passes=6, grid_size=600)

    # Phase 8: Final subgradient
    if tleft() > 30:
        best_h, best_score = subgrad_polish(best_h, target_use, n_steps=100, lr_init=0.01)

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
