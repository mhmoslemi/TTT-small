    return np.apply_along_axis(hilbert_map, 1, h, alpha)

def multi_scale_mutation(h, scales=[(0.33, 0.05), (0.66, 0.02), (1.0, 0.01)]):
    """
    Scales: (scale_factor, mutation_stddev)
    """
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for s, std in scales:
        h_scaled = hilbert_projection(best_h, s)
        h_mutated = h_scaled + std * np.random.normal(size=len(h))
        h_clipped = np.clip(h_mutated, 0, 1)
        h_constrained = project_onto_constraint(h_clipped, n/2)
        
        score = evaluate_score(h_constrained)
        if score < best_score:
            best_score = score
            best_h = h_constrained.copy()
