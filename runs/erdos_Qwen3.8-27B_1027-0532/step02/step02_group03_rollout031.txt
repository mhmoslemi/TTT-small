    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add

def _de_step(pop, target_sum, f, cr, n):

    best = pop[best_idx].copy()

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5):

    x = h0.copy()
    y = h0.copy()

            max_corr = np.max(corr)
            weights = np.exp(50.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()

        y = y_new
        t = t_new

        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.4 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.9 - 0.5 * t_frac
        cr = 0.4 + 0.3 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < cr:
                    trial[j] = mutant[j]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 15:
            break
    return best_h, best_c5

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(50.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_descent(h0, n, target_sum, max_rounds=30):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=10):
    h = h0.copy()
    vals = [0.0, 0.2, 0.25, 1.0/3.0, 0.5, 2.0/3.0, 0.75, 0.8, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _perturb_refine(h0, n, target_sum, deadline, rng, n_perturbs=20):
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for _ in range(n_perturbs):
        if time.time() > deadline:
            break
        h_pert = h0.copy()
        n_change = rng.randint(max(1, n // 20), max(2, n // 5) + 1)
        idx = rng.choice(n, size=n_change, replace=False)
        for i in idx:
            h_pert[i] = np.clip(h_pert[i] + rng.normal(0, 0.05), 0.0, 1.0)
        h_pert = _fix_sum(h_pert, target_sum)
        c5_pert = _c5_of(h_pert)
        if c5_pert < best_c5:
            h_n, c5_n = _nesterov(h_pert, n, target_sum, n_iters=100, alpha0=0.3,
                                    use_smooth_sg=True, sg_m=3, deadline=deadline)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
            h_sl = _slsqp_opt(best_h, n, target_sum, 300, maxiter=200)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start0 = time.time()
    deadline = t_start0 + budget_s - 25

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_candidates = [n_init]
    for delta in [10, -10, 20, -20, 30, -30, 40, -40]:
        n_try = n_init + delta
        if n_try % 2 == 0 and 50 <= n_try <= 200:
            n_candidates.append(n_try)
    seen = set()
    n_candidates_u = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_candidates_u.append(nn)
    n_candidates = n_candidates_u

    for n in n_candidates:
        if time.time() > deadline - 100:
            break
        target_sum = n / 2.0

        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)

        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        # DE for global exploration
        pop_size = min(15, max(8, n // 7))
        n_gen = min(100, max(40, int((deadline - time.time() - 80) / max(1, pop_size * 0.02))))
        h_de, c5_de = _de_optimize(h_start, n, target_sum, pop_size, n_gen,
                                     deadline - 80, rng)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n

        # Nesterov with single max grad
        if time.time() < deadline - 50:
            h_n1, c5_n1 = _nesterov(h_de, n, target_sum, n_iters=300, alpha0=0.5,
                                     deadline=deadline - 50)
            if c5_n1 < best_c5:
                best_c5 = c5_n1
                best_h = h_n1.copy()
                best_n = n

        # Nesterov with smooth top-m gradient
        if time.time() < deadline - 40:
            h_n2, c5_n2 = _nesterov(h_n1, n, target_sum, n_iters=250, alpha0=0.35,
                                     use_smooth_sg=True, sg_m=5, deadline=deadline - 40)
            if c5_n2 < best_c5:
                best_c5 = c5_n2
                best_h = h_n2.copy()
                best_n = n

        # SLSQP refinement with increasing alpha
        h_curr = h_n2
        for alpha in [30, 100, 300, 800, 1500]:
            if time.time() > deadline - 30:
                break
            h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=350)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_curr = h_sl

        # Coordinate descent (binary)
        if time.time() < deadline - 20:
            h_cd = _coord_descent(h_curr, n, target_sum, max_rounds=25)
            c5_cd = _c5_of(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

        # Fractional coordinate descent
        if time.time() < deadline - 15:
            h_cf = _coord_descent_frac(h_cd, n, target_sum, max_rounds=10)
            c5_cf = _c5_of(h_cf)
            if c5_cf < best_c5:
                best_c5 = c5_cf
                best_h = h_cf.copy()
                best_n = n

        # Perturbation + refine
        if time.time() < deadline - 20:
            h_pr, c5_pr = _perturb_refine(best_h, best_n, best_n / 2.0,
                                           deadline - 15, rng, n_perturbs=15)
            if c5_pr < best_c5:
                best_c5 = c5_pr
                best_h = h_pr.copy()

    # Final aggressive refinement
    n_best = best_n
    target = n_best / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 15:
        for alpha in [200, 500, 1000, 2000, 5000]:
            if time.time() > deadline - 12:
                break
            h_final = _slsqp_opt(h_final, n_best, target, alpha, maxiter=500)
            c5_f = _c5_of(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    if time.time() < deadline - 10:
        h_cd_f = _coord_descent(best_h, n_best, target, max_rounds=35)
        c5_cd_f = _c5_of(h_cd_f)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 8:
        h_cf_f = _coord_descent_frac(best_h, n_best, target, max_rounds=12)
        c5_cf_f = _c5_of(h_cf_f)
        if c5_cf_f < best_c5:
            best_c5 = c5_cf_f
            best_h = h_cf_f.copy()

    if time.time() < deadline - 5:
        h_pol = _slsqp_opt(best_h, n_best, target, 8000, maxiter=600)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # One more SLSQP + CD cycle if time permits
    if time.time() < deadline - 3:
        h_pol2 = _slsqp_opt(best_h, n_best, target, 5000, maxiter=400)
        c5_pol2 = _c5_of(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
        if time.time() < deadline - 2:
            h_cd_final = _coord_descent(best_h, n_best, target, max_rounds=20)
            c5_cd_final = _c5_of(h_cd_final)
            if c5_cd_final < best_c5:
                best_c5 = c5_cd_final
                best_h = h_cd_final.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
