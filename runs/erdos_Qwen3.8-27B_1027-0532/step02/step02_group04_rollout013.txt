```python
def gradient_smooth_max(h, alpha):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(N)
    # w[0] corresponds to lag = -(N-1), w[2N-2] corresponds to lag = N-1
    # w[N-1] corresponds to lag = 0
    for k_idx in range(2*N - 1):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        lag = k_idx - (N - 1)  # lag ranges from -(N-1) to N-1
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            # corr = sum_{i=0}^{N-1-lag} h[i] * (1-h[i+lag])
            # d/dh[i] for i in [0, N-1-lag]: (1-h[i+lag])
            # d/dh[i] for i in [lag, N-1]: -h[i-lag]
            grad[:N-lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:N-lag]
        else:
            l = -lag
            # corr = sum_{i=0}^{N-1-l} h[i+l] * (1-h[i])
            # d/dh[i] for i in [0, N-1-l]: -h[i+l]
            # d/dh[i] for i in [l, N-1]: (1-h[i-l])
            grad[:N-l] -= wk * h[l:]
            grad[l:] += wk * (1.0 - h[:N-l])
    
    return grad
```

```python
    return (best_h, best_score, len(best_h))
```
