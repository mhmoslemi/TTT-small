import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-10:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def lse_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def lse_func_grad(h, alpha):
    return lse_val(compute_overlaps(h), alpha), lse_grad(h, alpha)


def sum_eq(h):
    return np.sum(h) - len(h) / 2.0


def sum_eq_jac(h):
    return np.ones(len(h))

    # Phase 4: Final optimal 0.5 placement on best binary set
    best_m_final, best_max_final = best_half_placement(best_bin)
    best_h = best_bin.copy()
    best_h[best_m_final] = 0.5
    best_score = best_max_final * (2.0 / n)

    if best_score < best_overall_score:
        best_overall = best_h.copy()
        best_overall_score = best_score

    # Phase 5: Fractional refinement
    if time.time() - t0 < budget_s - 60:
        h_frac = best_overall.copy()
        for alpha in [1000, 3000, 8000]:
            if time.time() - t0 > budget_s - 30:
                break
            try:
                res = minimize(
                    lse_obj, h_frac, args=(alpha,),
                    method='L-BFGS-B', jac=lambda x, a=alpha: lse_grad(x, a),
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12}
                )
                h_frac = project_box_sum(np.clip(res.x, 0, 1), target)
                s = evaluate_score(h_frac)
                if s < best_overall_score:
                    best_overall_score = s
                    best_overall = h_frac.copy()
            except Exception:
                pass

    # Phase 6: SLSQP polish
    if time.time() - t0 < budget_s - 20:
        try:
            def _sum_eq(x):
                return np.sum(x) - len(x) / 2.0

            def _sum_jac(x):
                return np.ones(len(x))

            res = minimize(
                lse_obj, best_overall, args=(5000,),
                method='SLSQP', jac=lambda x: lse_grad(x, 5000),
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}],
                options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0}
            )
            h_slsqp = project_box_sum(np.clip(res.x, 0, 1), target)
            s = evaluate_score(h_slsqp)
            if s < best_overall_score:
                best_overall_score = s
                best_overall = h_slsqp.copy()
        except Exception:
            pass

    # Phase 7: Try n=424 (even, allows pure binary)
    if time.time() - t0 < budget_s - 80:
        n2 = 424
        dx2 = 2.0 / n2
        # Create initial binary for n=424 from current best
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_overall)
        h2 = (h_interp > 0.5).astype(float)
        n2_target = n2 // 2
        n2_ones = int(np.sum(h2))
        if n2_ones > n2_target:
            ones2 = np.where(h2 == 1)[0]
            h2[rng.choice(ones2, size=n2_ones - n2_target, replace=False)] = 0
        elif n2_ones < n2_target:
            zeros2 = np.where(h2 == 0)[0]
            h2[rng.choice(zeros2, size=n2_target - n2_ones, replace=False)] = 1

        # Quick local search on n=424
        best2_score = evaluate_score(h2)
        temp2 = 0.1
        for it2 in range(50000):
            if time.time() - t0 > budget_s - 40:
                break
            if it2 % 2000 == 1999:
                temp2 *= 0.95
            ovs2 = compute_overlaps(h2)
            cur2 = float(np.max(ovs2))
            ones2 = np.where(h2 == 1)[0]
            zeros2 = np.where(h2 == 0)[0]
            if len(ones2) == 0 or len(zeros2) == 0:
                break
            i2 = ones2[rng.randint(len(ones2))]
            j2 = zeros2[rng.randint(len(zeros2))]
            h2[i2] = 0
            h2[j2] = 1
            new2 = float(np.max(compute_overlaps(h2)))
            d2 = new2 - cur2
            if d2 < 0 or rng.random() < np.exp(-d2 / max(temp2, 1e-10)):
                if new2 < best2_score:
                    best2_score = new2
                    h2_best = h2.copy()
            else:
                h2[i2] = 1
                h2[j2] = 0

        if 'h2_best' in dir():
            score2 = best2_score * dx2
            if score2 < best_overall_score:
                best_overall = h2_best.copy()
                best_overall_score = score2
                n = n2

    # Final
    best_overall = project_box_sum(best_overall, len(best_overall) / 2.0)
    final_score = evaluate_score(best_overall)

    return (best_overall, final_score, len(best_overall))
