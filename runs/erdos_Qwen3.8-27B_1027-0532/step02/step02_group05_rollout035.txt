    """Try random swaps to improve the binary solution."""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = np.correlate(h, 1-h, mode='full')
    best_corr = np.max(corr)
    
    for step in range(n_steps):
        if temperature <= 0:
            # Pure hill climbing: only accept improvements
            for _ in range(100):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                if i == j:
                    continue
                h[i], h[j] = 0, 1
                new_corr = np.correlate(h, 1-h, mode='full')
                new_max = np.max(new_corr)
                if new_max <= best_corr:
                    best_corr = new_max
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    corr = new_corr
                else:
                    h[i], h[j] = 1, 0
        else:
            # Simulated annealing
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0, 1
            new_corr = np.correlate(h, 1-h, mode='full')
            new_max = np.max(new_corr)
            delta = new_max - best_corr
            if delta < 0 or np.random.random() < np.exp(-delta / temperature):
                best_corr = new_max if new_max < best_corr else best_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                corr = new_corr
            else:
                h[i], h[j] = 1, 0
    
    return h, best_corr

def evaluate_score(h):

    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:

    n = len(h)

            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]

            g[i - s_pos] -= h[i]
    
    h_new = project_feasible(h - lr * g, target)

    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def grad_obj(h, alpha):

    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)

        if w < 1e-14:
            continue

            grad[idx1 + k] -= w * h[idx1]

            idx1 = np.arange(kp, N)
            grad[idx1] += w * (1.0 - h[idx1 - kp])
            grad[idx1 - kp] -= w * h[idx1]
    return grad

def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    diff = target_sum - float(np.sum(proj))

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)

        if w < 1e-14:
