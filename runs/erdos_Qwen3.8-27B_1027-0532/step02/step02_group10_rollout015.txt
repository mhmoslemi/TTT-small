- If j-lag ≥ 0 and j-lag < n-lag: ∂/∂h[j] of -(h[j-lag]*h[j]) = -h[j-lag]

    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - center
        if offset >= 0:
            # c[m] = sum_{i=0}^{n-1-offset} h[i] * (1-h[i+offset])
            # ∂c[m]/∂h[j] = (1-h[j+offset]) for j < n-offset
            #               - h[j-offset] for j >= offset (and j-offset < n-offset, i.e., j < n)
            i_lo = 0
            i_hi = n - offset
            # First term: (1-h[j+offset]) for j in [0, n-offset)
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo+offset:i_hi+offset])
            # Second term: -h[j-offset] for j in [offset, n)
            j_lo = offset
            j_hi = n
            grad[j_lo:j_hi] -= w * h[j_lo-offset:j_hi-offset]
        else:

            # c[m] = sum_{i=L}^{n-1} h[i] * (1-h[i-L])
