def fft_cross_correlate(h):
    n = len(h)
    N = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    Nfft = 1
    while Nfft < N:
        Nfft *= 2
    a = np.zeros(Nfft, dtype=np.float64)
    a[:n] = h
    b = np.zeros(Nfft, dtype=np.float64)
    b[:n] = 1.0 - h
    # Cross-correlation via FFT
    result = np.real(np.fft.ifft(np.fft.fft(a) * np.conj(np.fft.fft(b))))
    return result[:N]

import numpy as np
from scipy.optimize import minimize
import time


def fft_cross_correlate(h):
    n = len(h)
    N = 2 * n - 1
    Nfft = 1
    while Nfft < N:
        Nfft *= 2
    a = np.zeros(Nfft, dtype=np.float64)
    a[:n] = h
    b = np.zeros(Nfft, dtype=np.float64)
    b[:n] = 1.0 - h
    Fa = np.fft.rfft(a)
    Fb = np.fft.rfft(b)
    result = np.fft.irfft(Fa * np.conj(Fb), Nfft)
    return result[:N]


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_cross_correlate(h)
    return float(np.max(corr)) * dx


def antipal_even_from_g(g):
    m = len(g)
    h = np.empty(2 * m)
    h[:m] = g
    h[m:] = 1.0 - g[::-1]
    return h


def project_to_binary_anti(h, n):
    m = n // 2

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def grad_smooth_max_h(h, alpha):
    n = len(h)
    corr = fft_cross_correlate(h)
    m = np.max(corr)
    ex = np.exp(alpha * (corr - m))
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        k = j - (n - 1)
        if k >= 0:
            xmax = n - k
            grad[:xmax] += wj * (1.0 - h[k:xmax + k])
            grad[k:] -= wj * h[:xmax]
        else:
            kp = -k
            grad[kp:] += wj * (1.0 - h[:n - kp])
            grad[:n - kp] -= wj * h[kp:]
    return grad

    return grad_g
