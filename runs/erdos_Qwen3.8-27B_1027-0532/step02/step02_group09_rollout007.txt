import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target_sum):
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-12
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    return grad

def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)

def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def sum_constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None

def lbfgsb_optimize(h0, alpha, penalty, maxiter, bounds):

            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18}
        )

    except Exception:

def binary_local_search_2swap(h0, n_iters, rng, max_temp=0.01):
    n = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n_target = n // 2
    n_ones = int(np.sum(h_bin))
    
    if n_ones > n_target:
        idxs = np.where(h_bin == 1.0)[0]
        remove = idxs[rng.randint(len(idxs), size=n_ones - n_target)]
        h_bin[remove] = 0.0
    elif n_ones < n_target:
        idxs = np.where(h_bin == 0.0)[0]
        add = idxs[rng.randint(len(idxs), size=n_target - n_ones)]
        h_bin[add] = 1.0
    
    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    temp = max_temp
    
    for it in range(n_iters):
        if it % 100 == 99:
            temp *= 0.97
        if temp < 1e-12:
            break
        
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose move type: 1-swap or 2-swap
        if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.5:
            # 2-swap
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                continue
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                continue
            
            h_bin[i1] = 0.0
            h_bin[i2] = 0.0
            h_bin[j1] = 1.0
            h_bin[j2] = 1.0
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_bin.copy()
            else:
                h_bin[i1] = 1.0
                h_bin[i2] = 1.0
                h_bin[j1] = 0.0
                h_bin[j2] = 0.0
        else:
            # 1-swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0

                    best_h = h_bin.copy()

                h_bin[j] = 0.0
    
    return best_h, best_score

    # Now compute b[idx] = c[idx] - a[idx] * h[j]
    c = compute_overlaps(h)
    b = c - a * h[j]
    
    # Bisection to find t in [0,1] such that F'(t) = 0
    # F'(t) = sum(a * exp(alpha*(a*t + b))) / sum(exp(alpha*(a*t + b)))
    
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        exponents = alpha * (a * mid + b)
        exponents -= np.max(exponents)  # for numerical stability
        weights = np.exp(exponents)
        Fprime = np.sum(a * weights) / np.sum(weights)
        if Fprime > 0:
            lo = mid
        else:
            hi = mid
    
    return (lo + hi) * 0.5

Wait, I have the bisection direction wrong. If F'(mid) > 0, the function is increasing at mid, so the minimum is to the left, meaning I should set hi = mid. Let me fix:

If F'(mid) > 0: minimum is to the left, so hi = mid
If F'(mid) < 0: minimum is to the right, so lo = mid

Let me also handle the edge case where a is all zeros (h[j] doesn't appear in any c[idx], which can't happen but let's be safe).

This coordinate descent approach is a genuinely different method from the previous code (which used L-BFGS-B and SLSQP). It should be:
- More robust (no need for line search, trust region, etc.)
- Faster per iteration (each step is O(n) for the bisection)
- Better for the non-convex problem (it naturally handles the bilinear structure)

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target_sum):
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-12
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    return grad

def obj_grad_lbfgs(h, alpha, penalty):
    n = len(h)
    obj = lse_val(compute_overlaps(h), alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n * 0.5
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

def slsqp_f(h, alpha):
    return lse_val(compute_overlaps(h), alpha)

def slsqp_g(h, alpha):
    return lse_grad(h, alpha)

def eq_con(h):
    return np.sum(h) - len(h) * 0.5

    return np.ones(len(h))

def coord_descent_optimize(h0, alpha, n_sweeps):
    """Coordinate descent for the soft-max objective."""
    h = h0.copy()
    n = len(h)
    
    for sweep in range(n_sweeps):
        for j in range(n):
            # Compute coefficient a[idx] for h[j] in each correlate value
            a = np.zeros(2 * n - 1)
            
            # For idx >= n-1: L = idx - (n-1), L in [0, n-1]
            Ls = np.arange(0, n)
            idxs = n - 1 + Ls
            
            mask1 = (Ls <= j)
            a[idxs[mask1]] += (1.0 - h[j - Ls[mask1]])
            
            mask2 = (Ls <= n - 1 - j)
            a[idxs[mask2]] -= h[j + Ls[mask2]]
            
            # For idx < n-1: M = (n-1) - idx, M in [1, n-1]
            Ms = np.arange(1, n)
            idxs2 = n - 1 - Ms
