import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_and_grad(h):
    """Compute correlation and gradient of each lag's correlation w.r.t. h."""
    n = len(h)
    corr = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            L = lag
            corr[k] = np.sum(h[:n-L] * (1.0 - h[L:]))
        else:
            L = -lag
            corr[k] = np.sum(h[L:] * (1.0 - h[:n-L]))
    return corr

def _soft_max_obj_grad(h, alpha, n, var_lam=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    dc = alpha * (corr - m)
    exp_dc = np.exp(dc)
    sum_exp = np.sum(exp_dc)
    if sum_exp < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exp_dc / sum_exp
    
    if var_lam > 0:
        var_term = var_lam * np.var(corr)
        obj_val = (m + np.log(sum_exp) / alpha) * dx + var_term * dx
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n - lag])
        else:
            L = -lag
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] += w * (-h[L:])
    
    grad *= dx
    
    if var_lam > 0:
        var_grad = _var_gradient(h, n, var_lam)
        grad += var_grad
    
    if var_lam > 0:
        return obj_val, grad
    return (m + np.log(sum_exp) / alpha) * dx, grad

def _var_gradient(h, n, lam):
    """Gradient of lam * var(corr) w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mean_c = np.mean(corr)
    # d(var)/d(corr[k]) = 2*(corr[k] - mean_c) / len(corr)
    # d(correlation[k])/d(h[i]) as before
    dx = 2.0 / n
    npts = n
    grad = np.zeros(npts)
    for k in range(2 * npts - 1):
        coeff = 2.0 * lam * (corr[k] - mean_c) / (2 * npts - 1) * dx
        if abs(coeff) < 1e-15:
            continue
        lag = k - (npts - 1)
        if lag == 0:
            grad[:] += coeff * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:npts - lag] += coeff * (1.0 - h[lag:])
            grad[lag:] += coeff * (-h[:npts - lag])
        else:
            L = -lag
            grad[L:] += coeff * (1.0 - h[:npts - L])
            grad[:npts - L] += coeff * (-h[L:])
    return grad

def _slsqp_opt(h_init, alpha, n, maxiter, ftol, var_lam=0.0):
    target_sum = n / 2.0
    h_init = np.clip(h_init.copy(), 0, 1)
    
    def obj(x):
        val, _ = _soft_max_obj_grad(x, alpha, n, var_lam)
        return val
    
    def obj_grad(x):
        _, g = _soft_max_obj_grad(x, alpha, n, var_lam)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = np.clip(res.x, 0, 1)
        if _c5_of(h_out) _c5_of(h_init):
            return h_out
    except Exception:
        pass
    return h_init

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _sm_obj(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

def _sm_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    dc = alpha * (corr - m)
    exp_dc = np.exp(dc)
    s = np.sum(exp_dc)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_dc / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] += wk * (-h[:n - lag])
        else:
            L = -lag
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] += wk * (-h[L:])
    return grad * dx

def _sm_obj_var(h, alpha, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    var = np.var(corr)
    return (sm + beta * var) * dx

def _sm_grad_var(h, alpha, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    dc = alpha * (corr - m)
    exp_dc = np.exp(dc)
    s = np.sum(exp_dc)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_dc / s
    nc = len(corr)
    mean_c = np.mean(corr)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        coeff = w[k] + 2.0 * beta * (corr[k] - mean_c) / nc
        if abs(coeff) < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += coeff * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += coeff * (1.0 - h[lag:])
            grad[lag:] += coeff * (-h[:n - lag])
        else:
            L = -lag
            grad[L:] += coeff * (1.0 - h[:n - L])
            grad[:n - L] += coeff * (-h[L:])
    return grad * dx

def _slsqp(h0, alpha, n, maxit, ftol, use_var=False, beta=0.0):
    ts = n / 2.0
    h0 = np.clip(h0.copy(), 0, 1)
    if use_var:
        def f(x):
            return _sm_obj_var(x, alpha, n, beta)
        def g(x):
            return _sm_grad_var(x, alpha, n, beta)
    else:
        def f(x):
            return _sm_obj(x, alpha, n)
        def g(x):
            return _sm_grad(x, alpha, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(f, h0, method='SLSQP', bounds=bnds, constraints=cons,
                     jac=g, options={'maxiter': maxit, 'ftol': ftol})
        h_out = np.clip(r.x, 0, 1)
        if _c5_of(h_out) < _c5_of(h0) - 1e-14:
            return h_out
    except Exception:
        pass
    return h0.copy()

def _coord_polish(h, n, ts, deadline, max_sweeps):
    best = h.copy()
    best_c = _c5_of(best)
    deltas = [0.008, 0.004, 0.002, 0.001, 0.0005]
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > deadline - 5:
            break
        improved = False
        idx = np.random.default_rng(sweep * 31 + 7).permutation(n)
        for i in idx:
            for delta in deltas:
                for sgn in [1, -1]:
                    h_try = best.copy()
                    h_try[i] += sgn * delta
                    if h_try[i] < 0 or h_try[i] > 1:
                        continue
                    diff = ts - h_try.sum()
                    if abs(diff) < 1e-10:
                        continue
                    if diff > 0:
                        cand = np.where(best < 1.0 - 1e-6)[0]
                    else:
                        cand = np.where(best > 1e-6)[0]
                    if len(cand) == 0:
                        continue
                    j = cand[np.argmin(np.abs(best[cand] - 0.5))]
                    h_try[j] += diff
                    if h_try[j] < 0 or h_try[j] > 1:
                        continue
                    c = _c5_of(h_try)
                    if c < best_c - 1e-12:
                        best = h_try.copy()
                        best_c = c
                        improved = True
        sweep += 1
        if not improved:
            break
    return best, best_c

def _interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_ext = np.concatenate([h, [h[0]]])
    return np.interp(x_new, x_old, h_ext)

def _subgrad_push(h, n, ts, n_iter, lr0, lr_decay, top_k, deadline):
    best_h = h.copy()
    best_c = _c5_of(h)
    h_cur = h.copy()
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        lr = lr0 / (1.0 + t * lr_decay)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            if lag == 0:
                grad[:] += (1.0 - 2.0 * h_cur)
            elif lag > 0:
                grad[:n - lag] += (1.0 - h_cur[lag:])
                grad[lag:] += (-h_cur[:n - lag])
            else:
                L = -lag
                grad[L:] += (1.0 - h_cur[:n - L])
                grad[:n - L] += (-h_cur[L:])
        grad /= len(top_idx)
        h_cur = _fix_sum(h_cur - lr * grad, ts)
        if (t + 1) % 200 == 0:
            c = _c5_of(h_cur)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h_cur.copy()
    c = _c5_of(h_cur)
    if c < best_c:
        best_c = c
        best_h = h_cur.copy()
    return best_h, best_c
