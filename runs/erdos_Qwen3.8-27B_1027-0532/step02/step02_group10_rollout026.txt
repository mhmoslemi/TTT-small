    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):

        if s > target_sum:

        else:

    n = len(h)

        if w < 1e-12:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            length = n - offset
            if length > 0:
                grad[:length] += w * (1.0 - h[offset:offset+length])
                grad[offset:] -= w * h[:length]
        else:
            length = n + offset
            if length > 0:
                grad[-offset:] += w * (1.0 - h[:length])
                grad[:-offset] -= w * h[-offset:]
