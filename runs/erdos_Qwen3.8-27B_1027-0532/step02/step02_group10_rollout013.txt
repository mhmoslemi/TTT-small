    m = np.max(x)
    s = alpha * (x - m)
    e = np.exp(np.clip(s, -40.0, 40.0))
    tot = e.sum()
    val = m + np.log(tot) / alpha
    w = e / tot
    return val, w

    n = len(h)
    g = np.zeros(n)
    for m in range(len(w)):
        wm = w[m]
        if wm < 1e-14:

        k = m - (n - 1)

        else:

            e = n - kk
            if e > 0:
                g[kk:] += wm * (1.0 - h[:e])
                g[:e] -= wm * h[kk:kk+e]

    dx = 2.0 / n
    h = h.copy()

    # More carefully: the gradient of the smoothed max has norm bounded by alpha * L_corr
    # where L_corr is the Lipschitz constant of the correlation map
    # L_corr ≈ 2 * dx * max_k |sum_i (derivatives)| ≈ 4 * dx
    # So step ≈ 1 / (alpha * 4 * dx)
    step = 1.0 / (alpha * 4.0 * dx + 1.0)
    step = min(step, 1.0)
    prev_val = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        sm, w = smooth_max_lse(ov, alpha)
        if prev_val is not None:
            if sm < prev_val - 1e-12:
                stall = 0
            else:
                stall += 1
            if stall > 150:
                break
        prev_val = sm
        g = grad_weighted(h, w, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        # Backtracking
        improved = False
        s = step
        for bt in range(30):

                step = min(s * 1.5, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    return np.max(compute_overlaps(h))


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 3.0
    hi = h.max() + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[free] += diff / nf
            np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_lse(x, alpha):
    m = np.max(x)
    s = alpha * (x - m)
    e = np.exp(np.clip(s, -40.0, 40.0))
    tot = e.sum()
    val = m + np.log(tot) / alpha
    w = e / tot
    return val, w


def grad_weighted(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for m in range(len(w)):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            e = n - k
            if e > 0:
                g[:e] += wm * (1.0 - h[k:k + e])
                g[k:] -= wm * h[:e]
        else:
            kk = -k
            e = n - kk
            if e > 0:
                g[kk:] += wm * (1.0 - h[:e])
                g[:e] -= wm * h[kk:kk + e]
    g *= dx
    return g


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = min(0.8, 1.0 / (alpha * 3.0 * dx + 1.0))
    prev_val = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        sm, w = smooth_max_lse(ov, alpha)
        if prev_val is not None:
            if sm < prev_val - 1e-12:
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break
        prev_val = sm
        g = grad_weighted(h, w, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        s = step
        for bt in range(30):
            h_t = project_onto_constraint(h - s * g, target)
            ov_t = compute_overlaps(h_t)
            sm_t, _ = smooth_max_lse(ov_t, alpha)
            if sm_t < sm - 1e-13:
                h = h_t
                improved = True
                step = min(s * 1.4, 1.0)
                break
            s *= 0.5
            if s < 1e-14:
                break
        if not improved:
            step = max(s, 1e-15)
            if s < 1e-14:
                break
    return h


def active_set_polish(h, target, deadline, top_k=10):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_max = evaluate_score(h)
    for outer in range(300):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        m = np.max(ov)
        threshold = m - 0.008
        active = np.where(ov >= threshold)[0]
        if len(active) > top_k:
            idx = np.argsort(ov, kind='stable')
            active = idx[-top_k:]
        w = np.zeros(len(ov))
        w[active] = 1.0 / len(active)
        g = grad_weighted(h, w, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        s = 0.3
        for bt in range(35):
            h_t = project_onto_constraint(h - s * g, target)
            s_val = evaluate_score(h_t)
            if s_val < cur_max - 1e-12:
                h = h_t
                cur_max = s_val
                improved = True
                break
            s *= 0.5
        if not improved:
            s = 0.05
            for bt in range(20):
                h_t = project_onto_constraint(h - s * g, target)
                s_val = evaluate_score(h_t)
                if s_val < cur_max - 1e-12:

def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.4, 1.0 / (alpha * 4.0 * dx + 1.0))
    prev_val = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_lse(ov_y, alpha)
        g_y = grad_weighted(y, w_y, dx)
        x_new = project_onto_constraint(y - step * g_y, target)
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_lse(ov_x, alpha)
        if prev_val is not None:
            if sm_x < prev_val - 1e-12:
                stall = 0
            else:
                stall += 1
            if stall > 150:
                break
        prev_val = sm_x
        if sm_x > prev_val + 1e-8 if prev_val is not None else False:
            t_k = 1.0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x
