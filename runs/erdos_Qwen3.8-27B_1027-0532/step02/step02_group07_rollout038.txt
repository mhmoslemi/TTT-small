import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None

    else:

    return float(np.clip(t_star, 0.0, 1.0)), slopes

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h

def binary_simulated_annealing(h_init, n, time_budget=300.0, seed=42):
    """SA on binary vectors with exactly n//2 ones."""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    m_target = n // 2  # target number of ones
    
    # Ensure correct number of ones
    ones_count = int(np.sum(h))
    if ones_count > m_target:
        excess = ones_count - m_target
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, size=excess, replace=False)
        h[to_remove] = 0
    elif ones_count < m_target:
        deficit = m_target - ones_count
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, size=deficit, replace=False)
        h[to_add] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_energy = float(np.max(corr) * dx)
    
    best_h = h.copy()
    best_energy = current_energy
    
    # SA parameters
    T0 = 0.02  # initial temperature (in C5 units)
    T_final = 0.001
    n_iters = int(time_budget / 0.0001)  # estimate: ~100us per iteration
    # Actually let's just run until time budget
    T = T0
    cooling = 0.99995  # per iteration
    
    t0 = time.time()
    iters = 0
    no_improve = 0
    
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    
    while time.time() - t0 < time_budget:
        # Make a move: swap a random 1 with a random 0
        ia = rng.randint(len(ones_list))
        ib = rng.randint(len(zeros_list))
        a = ones_list[ia]
        b = zeros_list[ib]
        
        # Compute new energy
        h[a] = 0
        h[b] = 1
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_energy = float(np.max(corr_new) * dx)
        delta_E = new_energy - current_energy
        
        # Metropolis acceptance
        if delta_E < 0 or rng.random() < np.exp(-delta_E / T):
            current_energy = new_energy
            corr = corr_new
            # Update lists
            ones_list[ia] = b
            zeros_list[ib] = a
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            h[a] = 1
            h[b] = 0
            no_improve += 1
        
        iters += 1
        T *= cooling
        
        # Periodic restart cooling if stuck
        if no_improve > 50000:
            T = max(T, T0 * 0.1)
            no_improve = 0
    
    return best_h, best_energy
