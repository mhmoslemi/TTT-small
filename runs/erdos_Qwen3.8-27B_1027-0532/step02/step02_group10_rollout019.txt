from scipy.optimize import minimize
import time

def compute_overlaps(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

def compute_overlaps(h):

def evaluate_score(h):

def project_onto_constraint(h, target_sum):

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):

        offset = m - (n - 1)

        i_hi = min(n, n - offset)
        if i_lo < i_hi:

        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]

    return grad
