def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad_vec(h, eps):
    """Vectorized smooth max gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp  # shape (2n-1,)
    
    onemh = 1.0 - h
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        l = k - (n - 1)
        if l >= 0:
            # First term: (1-h)[j-l] for j in [l, n-1]
            grad[l:] += w * onemh[:n - l]
            # Second term: -h[j+l] for j in [0, n-1-l]
            grad[:n - l] -= w * h[l:]
        else:
            # First term: (1-h)[j-l] = (1-h)[j+|l|] for j in [0, n-1+|l|-1] = [0, n-l-1]

            grad[m_idx:] -= w * h[:n - m_idx]

        h_new, c5_new = slsqp_smooth(best_h, n, eps_val, 500, 1e-15)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: Higher resolution attempt
    n2 = 200
    h_up = project_feasible(upsample(best_h, n, n2), n2)
    best_h2 = h_up.copy()
    best_c52 = evaluate_c5(h_up)

    if best_c52 <= best_c5 + 0.005:
        for eps_val in [200.0, 1000.0, 5000.0, 20000.0, 50000.0]:
            if elapsed() > budget_s * 0.92:
                break
            h_new2, c5_new2 = slsqp_smooth(best_h2, n2, eps_val, 400, 1e-14)
            if c5_new2 < best_c52 - 1e-13:
                best_c52 = c5_new2
                best_h2 = h_new2

        if elapsed() < budget_s * 0.97:
            best_h2, best_c52 = pair_swap_descent(best_h2, n2, 8000, time_limit(0.98))

    # Final comparison and return
    if best_c52 < best_c5:
        final_h = best_h2
        final_c5 = best_c52
        final_n = n2
    else:
        final_h = best_h
        final_c5 = best_c5
        final_n = n

    final_h = project_feasible(final_h, final_n)
    final_c5 = evaluate_c5(final_h)
    return final_h, final_c5, final_n
