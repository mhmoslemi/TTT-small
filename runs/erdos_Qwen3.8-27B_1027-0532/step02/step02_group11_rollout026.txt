def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def get_correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_and_grad(h, eps):
    corr = get_correlation(h)
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    val = (m + np.log(sum_exp) / eps)
    weights = exp_vals / sum_exp
    
    # Gradient: sum_k w_k * grad_corr_k
    n = len(h)
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        # corr_k = sum_i h[i] * (1-h[i-lag]) for valid i
        # d/dh[j]: if j appears as h[i], contributes (1-h[j-lag])
        #           if j appears as h[i-lag], contributes -h[j+lag]
        # For lag >= 0: i ranges from lag to n-1
        #   h[i] contributes (1-h[i-lag])
        #   h[i-lag] contributes -h[i]
        # For lag < 0: i ranges from 0 to n-1+lag
        #   Same formula with i-lag = i+|lag|
        
        # General formula: for each j, 
        #   term1: j is in position i (h[j] * (1-h[j-lag])), needs 0<=j<n and 0<=j-lag<n
        #   term2: j is in position i-lag (h[j+lag] * (1-h[j])), needs 0<=j+lag<n and 0<=j<n
        #   contribution = (1-h[j-lag]) + (-h[j+lag]) where valid
        
        # Vectorized:
        # For term1: j ranges over intersection of [0,n-1] and [lag, n-1+lag]
        j1_lo = max(0, lag)
        j1_hi = min(n-1, n-1+lag)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += w * (1.0 - h[j1_lo-lag:j1_hi+1-lag])
        
        # For term2: j+lag ranges over [0,n-1], so j ranges over [-lag, n-1-lag]
        j2_lo = max(0, -lag)
        j2_hi = min(n-1, n-1-lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo+lag:j2_hi+1+lag])
    
    return val, grad

def slsqp_smooth(h0, n, eps, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        val, _ = smooth_max_and_grad(x, eps)
        return val
    
    def gradient(x):
        _, g = smooth_max_and_grad(x, eps)
        return g
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    res = minimize(objective, h0.copy(), jac=gradient, method='SLSQP',
                   bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': 0})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h0, evaluate_c5(h0)

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

**New idea: Use the structure of the correlation to guide optimization.**

The correlation at lag k is:
corr(k) = Σ_i h(i) · (1-h(i+k))

If h is close to binary (which it should be at the optimum), then corr(k) counts (approximately) the number of positions where h transitions from 1 to 0 at lag k.

For a "good" sequence, we want this to be as uniform as possible across all lags.

**Key insight**: If we think of the positions of the 1's as a set S, then corr(k) ≈ |S| - |S ∩ (S-k)| = |S| - autocorr_S(k).

So minimizing max corr(k) is equivalent to maximizing min autocorr_S(k) (for k ≠ 0).

This is the "maximin autocorrelation" problem for binary sequences, which is well-studied!
