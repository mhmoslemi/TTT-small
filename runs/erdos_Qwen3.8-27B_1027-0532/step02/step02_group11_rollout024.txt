def compute_delta_corr(h, i0, j0, delta, corr):
    """Compute the change in corr when h[i0] decreases by delta and h[j0] increases by delta."""
    n = len(h)
    d = np.zeros(2*n-1)
    l_vals = np.arange(2*n-1) - (n-1)  # shift values
    
    # Generic formula: delta * (h[i0+l] + h[j0-l] - h[j0+l] - h[i0-l])
    # with out-of-bounds = 0
    
    for k, l in enumerate(l_vals):
        val = 0.0
        # h[i0+l]
        idx = i0 + l
        if 0 <= idx < n:
            val += h[idx]
        # h[j0-l]
        idx = j0 - l
        if 0 <= idx < n:
            val += h[idx]
        # -h[j0+l]
        idx = j0 + l
        if 0 <= idx < n:
            val -= h[idx]
        # -h[i0-l]
        idx = i0 - l
        if 0 <= idx < n:
            val -= h[idx]
        d[k] = delta * val
    
    # Special cases
