import cvxpy as cp

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_lag_gradient(h, r):
    """Compute gradient of overlap at lag r with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if r >= 0:
        # overlap = sum_{i=0}^{n-1-r} h[i+r] * (1-h[i])
        # grad[j] from h[j] in h[i+r]: (1-h[i]) where i=j-r, valid for j >= r
        # grad[j] from h[j] in (1-h[i]): -h[i+r] where i=j, valid for j < n-r
        if r == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[r:] += (1.0 - h[:n-r])  # (1-h[j-r]) for j = r,...,n-1
            grad[:n-r] -= h[r:]  # -h[j+r] for j = 0,...,n-r-1
    else:

        # overlap = sum_{i=0}^{n-1-M} h[i] * (1-h[i+M])
        # grad[j] from h[j] in h[i]: (1-h[i+M]) where i=j, valid for j < n-M
        # grad[j] from h[j] in (1-h[i+M]): -h[i] where i+M=j, i=j-M, valid for j >= M
        grad[:n-M] += (1.0 - h[M:])  # (1-h[j+M]) for j = 0,...,n-M-1
        grad[M:] -= h[:n-M]  # -h[j-M] for j = M,...,n-1
    return grad

def solve_cutting_plane(h0, n, dx, max_iters, top_k_init, solver):
    """Iterative linearization (cutting-plane) for the Erdős problem."""
    h = h0.copy()
    cuts = []  # list of (grad, c) pairs
    
    for it in range(max_iters):
        # Compute overlaps
        overlaps = compute_overlaps(h)
        max_overlap = float(np.max(overlaps))
        
        # Select top-k lags
        k = min(top_k_init, len(overlaps))
        top_idx = np.argsort(-overlaps)[:k]
        
        # Build and solve LP

        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]

            constraints.append(grad @ h_var - t_var / dx <= c_r)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n-L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    return grad

def pnorm_val(h, p):
    overlaps = compute_overlaps(h)
    return float(np.sum(overlaps**p) ** (1.0/p))

    f_val = pnorm_val(h, p)

        L = idx - (n - 1)

            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    return grad

def obj_lse(h, alpha, penalty):
    n = len(h)
    s = float(np.sum(h)) - n / 2.0
    return lse_val(compute_overlaps(h), alpha) + penalty * s * s

def grad_lse(h, alpha, penalty):
    n = len(h)
    s = float(np.sum(h)) - n / 2.0
    g = lse_grad(h, alpha) + 2.0 * penalty * s * np.ones(n)
    return g

def obj_pnorm(h, p, penalty):
    n = len(h)
    s = float(np.sum(h)) - n / 2.0
    return pnorm_val(h, p) + penalty * s * s

def grad_pnorm(h, p, penalty):
    n = len(h)
    s = float(np.sum(h)) - n / 2.0
    g = pnorm_grad(h, p) + 2.0 * penalty * s * np.ones(n)
    return g

def solve_lp_step(h0, n, dx, top_k, solver='GLOP'):
    """Solve one LP step of the cutting-plane method."""
    overlaps = compute_overlaps(h0)
    max_idx = np.argsort(-overlaps)[:top_k]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    for idx in max_idx:
        r = idx - (n - 1)
        # Compute gradient of O_r at h0
        grad = np.zeros(n)
        if r >= 0:
            if r == 0:
                grad[:] = 1.0 - 2.0 * h0
            else:
                grad[r:] += (1.0 - h0[:n-r])
                grad[:n-r] -= h0[r:]
        else:
            M = -r
            grad[:n-M] += (1.0 - h0[M:])
            grad[M:] -= h0[:n-M]
        
        o_r = overlaps[idx]
        c_r = o_r - float(np.dot(grad, h0))
        constraints.append(grad @ h_var - t_var / dx <= c_r)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='GLPK')
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    h_sol = np.array(h_var.value, dtype=float)
    return h_sol, float(t_var.value)

def cutting_plane_optimize(h0, n, dx, max_iters, top_k, solver='GLOP'):
    """Iterative LP linearization."""
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for it in range(max_iters):
        h_sol, t_val = solve_lp_step(h, n, dx, top_k, solver)
        if h_sol is None:
            break
        h_new = project_box_sum(np.clip(h_sol, 0.0, 1.0), n / 2.0)
        s = evaluate_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()
            h = h_new
        else:
            # Try convex combination
            improved = False
            for alpha in [0.5, 0.3, 0.1]:
                h_mix = (1 - alpha) * h + alpha * h_new
                h_mix = project_box_sum(h_mix, n / 2.0)
                s_mix = evaluate_score(h_mix)
                if s_mix < best_score:
                    best_score = s_mix
                    best_h = h_mix.copy()
                    h = h_mix.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_score

def subgrad_step(h, top_k, lr, target):
    n = len(h)
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(-overlaps)[:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        L = idx - (n - 1)
        if L >= 0:
            grad[:n-L] += (1.0 - h[L:])
            grad[L:] -= h[:n-L]
        else:
            M = -L
            grad[M:] += (1.0 - h[:n-M])
            grad[:n-M] -= h[M:]
    grad /= float(top_k)
    h_new = project_box_sum(h - lr * grad, target)
    return h_new

def subgrad_optimize(h0, n_iters, lr_init, target, top_k=5, decay=0.02):
    h = h0.copy()
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    for it in range(n_iters):
        lr = lr_init / (1.0 + decay * float(it))
        h_new = subgrad_step(h, top_k, lr, target)
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
    return best_h, best_raw

def binary_local_search_feasible(h0, n_iters, rng, target):
    """Binary-ish local search that maintains feasibility."""
    n = len(h0)
    # Create a nearly-binary solution
    # Sort h0 and threshold
    sorted_h = np.sort(h0)
    # Find threshold where sum = target
    # For nearly binary: most values are 0 or 1, few in between
    h_bin = np.zeros(n)
    # Put 1s at the top n//2 positions of the sorted original
    order = np.argsort(-h0)
    h_bin[order[:int(target)]] = 1.0
    # For the fractional part
    frac = target - int(target)
    if frac > 0:
        # Put frac in the next position
        h_bin[order[int(target)]] = frac
    
    best_h = project_box_sum(h_bin, target)
    best_score = evaluate_score(best_h)
    
    # Local search: swap operations
    ones = np.where(h_bin >= 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    
    temp = 0.01
    for it in range(n_iters):
        if it % 500 == 499:
            temp *= 0.95
        if temp < 1e-8:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        old_i, old_j = h_bin[i], h_bin[j]
        h_bin[i] = old_j
        h_bin[j] = old_i
        
        s_new = evaluate_score(h_bin)
        delta = s_new - best_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            if s_new < best_score:
                best_score = s_new
                # Update tracked indices
                ones = np.where(h_bin >= 0.5)[0]
                zeros = np.where(h_bin < 0.5)[0]
        else:
            h_bin[i] = old_i
            h_bin[j] = old_j
    
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    def time_left():
        return budget_s - 30.0 - (time.time() - t0)
    
    # Phase 1: Cutting-plane LP method
    if time_left() > 60:
        for top_k in [10, 30, 50]:
            if time_left() < 30:
                break
            n_iters = min(30, int(time_left() // 3))
            h_cp, s_cp = cutting_plane_optimize(h, n, dx, n_iters, top_k)
            if s_cp < best_score:
                best_score = s_cp
                best_h = h_cp.copy()
                h = h_cp.copy()
    
    # Phase 2: p-norm optimization (smooth intermediate)
    for p in [10, 20, 50]:
        if time_left() < 60:
            break
        penalty = 50.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                obj_pnorm, h, args=(p, penalty),
                method='L-BFGS-B', jac=grad_pnorm, bounds=bounds,
                options={'maxiter': 500, 'gtol': 1e-14, 'ftol': 1e-16}
            )
            h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    # Phase 3: LSE optimization (high alpha for tight max approximation)
    for alpha in [200, 500, 1000, 2000, 5000]:
        if time_left() < 60:
            break
        penalty = alpha * 20.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                obj_lse, h, args=(alpha, penalty),
                method='L-BFGS-B', jac=grad_lse, bounds=bounds,
                options={'maxiter': 600, 'gtol': 1e-14, 'ftol': 1e-16}
            )
            h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    # Phase 4: Subgradient on true max
    if time_left() > 90:
        n_sub = min(20000, int(time_left() * 10))
        h_sub, raw_sub = subgrad_optimize(best_h, n_sub, 0.5, target, top_k=8, decay=0.01)
        s_sub = raw_sub * dx
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()
        # Polish
        if time_left() > 60:
            alpha_p = 3000
            penalty_p = alpha_p * 30.0
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(
                    obj_lse, best_h, args=(alpha_p, penalty_p),
                    method='L-BFGS-B', jac=grad_lse, bounds=bounds,
                    options={'maxiter': 800, 'gtol': 1e-15, 'ftol': 1e-16}
                )
                h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), target)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Phase 5: Multiple random restarts + local search
    if time_left() > 90:
        for restart in range(min(5, int(time_left() // 30))):
            h_pert = best_h + rng.randn(n) * 0.1
            h_pert = project_box_sum(h_pert, target)
            # Quick LSE optimization
            if time_left() < 30:
                break
            alpha_r = 1000
            penalty_r = alpha_r * 30.0
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(
                    obj_lse, h_pert, args=(alpha_r, penalty_r),
                    method='L-BFGS-B', jac=grad_lse, bounds=bounds,
                    options={'maxiter': 300, 'gtol': 1e-13, 'ftol': 1e-15}
                )
                h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), target)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Phase 6: Binary local search with feasibility
    if time_left() > 60:
        n_iters = min(50000, int(time_left() * 20))
        h_bin, s_bin = binary_local_search_feasible(best_h, n_iters, rng, target)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()
    
    # Phase 7: Multi-resolution refinement
    if time_left() > 150:
        for n_try in [n * 2, int(n * 1.5)]:
            if n_try > 2000 or time_left() < 60:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_box_sum(h_init, n_try / 2.0)
            for alpha in [500, 2000]:
                if time_left() < 30:
                    break
                penalty = alpha * 30.0
                bounds = [(0.0, 1.0)] * n_try
                maxiter = max(50, 200000 // n_try)
                try:
                    res = minimize(
                        obj_lse, h_init, args=(alpha, penalty),
                        method='L-BFGS-B', jac=grad_lse, bounds=bounds,
                        options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
                    )
                    h_fine = project_box_sum(np.clip(res.x, 0.0, 1.0), n_try / 2.0)
                    s = evaluate_score(h_fine)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_fine.copy()
                        h_init = h_fine.copy()
                except Exception:
                    pass
    
    # Phase 8: Final high-precision refinement
    if time_left() > 20:
        n_final = len(best_h)
        target_final = n_final / 2.0
        for alpha in [5000, 10000, 20000]:
            if time_left() < 15:
                break
            penalty = alpha * 50.0
            bounds = [(0.0, 1.0)] * n_final
            try:
                res = minimize(
                    obj_lse, best_h, args=(alpha, penalty),
                    method='L-BFGS-B', jac=grad_lse, bounds=bounds,
                    options={'maxiter': 500, 'gtol': 1e-16, 'ftol': 1e-18}
                )
                h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), target_final)
                s = evaluate_score(h_new)
                if
