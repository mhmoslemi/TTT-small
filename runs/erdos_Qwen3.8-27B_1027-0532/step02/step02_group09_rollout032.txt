import numpy as np
from scipy.optimize import minimize
import time


def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlation(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def solve_1d_max_line(a, b):
    n = len(a)
    val0 = float(np.max(b))
    val1 = float(np.max(a + b))
    if val0 <= val1:
        upper_bound = val0
    else:
        upper_bound = val1
    lb = np.where(a >= 0, b, a + b)
    lower_bound = float(np.max(lb))
    if lower_bound >= upper_bound - 1e-14:
        return 0.0 if val0 <= val1 else 1.0
    lo = lower_bound - 0.01
    hi = upper_bound + 0.01
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        pos = a > 1e-14
        neg = a < -1e-14
        t_upper = 1.0
        t_lower = 0.0
        if np.any(pos):
            t_upper = min(t_upper, float(np.min((mid - b[pos]) / a[pos])))
        if np.any(neg):
            t_lower = max(t_lower, float(np.max((mid - b[neg]) / a[neg])))
        zero = ~(pos | neg)
        if np.any(zero) and float(np.max(b[zero])) > mid:
            lo = mid
            continue
        if t_lower <= t_upper + 1e-12:
            hi = mid
        else:
            lo = mid
    T_star = 0.5 * (lo + hi)
    pos = a > 1e-14
    neg = a < -1e-14
    t_upper = 1.0
    t_lower = 0.0
    if np.any(pos):
        t_upper = min(t_upper, float(np.min((T_star - b[pos]) / a[pos])))
    if np.any(neg):

    t_opt = 0.5 * (t_lower + t_upper)
    return float(np.clip(t_opt, 0.0, 1.0))


def coordinate_descent(h, n_sweeps, seed=42):
    n = len(h)
    h = h.copy()
    rng = np.random.RandomState(seed)
    corr = compute_correlation(h)
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for j in order:
            k_min = max(-j, j - n + 1)
            k_max = min(n - 1 - j, j)
            if k_min > k_max:
                continue
            ks = np.arange(k_min, k_max + 1)
            jp = j + ks
            jm = j - ks
            a = (1.0 - h[jp]) - h[jm]
            b = corr[ks + n - 1] - a * h[j]
            t_new = solve_1d_max_line(a, b)
            delta = t_new - h[j]
            if abs(delta) > 1e-12:
                h[j] = t_new
                corr[ks + n - 1] += a * delta
    return h


def lse_computation(h, alpha):
    corr = compute_correlation(h)
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    s = np.sum(exps)
    weights = exps / s
    val = m + np.log(s) / alpha
    return corr, weights, val


def lse_gradient(corr, weights, h, alpha):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h][:n - M]"  # placeholder
    return grad


def _obj_h(h, alpha, penalty):
    corr, weights, val = lse_computation(h, alpha)
    obj = val
    s_sum = float(np.sum(h)) - len(h) / 2.0
    obj += penalty * s_sum * s_sum
    return obj

    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(corr)):

        if w < 1e-12:

            grad[:n - M] -= w * h[M:]
    s_sum = float(np.sum(h)) - n / 2.0
    grad += 2.0 * penalty * s_sum
    return grad

    n_ones = int(np.sum(h_bin))

        remove = idxs[rng.randint(len(idxs), size=n_ones - n_target)]
        h_bin[remove] = 0.0

        h_bin[add] = 1.0

    for it in range(n_iters):
        if it % 300 == 299:

        ones = np.where(h_bin == 1.0)[0]

            break
        i = ones[rng.randint(len(ones))]

                best_h = h_bin.copy()
        else:
