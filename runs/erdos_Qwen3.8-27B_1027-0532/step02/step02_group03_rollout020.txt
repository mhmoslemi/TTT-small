        return h

            grad[:m] += -h[lag:lag+m]
    else:

        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad

    dx = 2.0 / n

    m = np.max(corr)
    exp_terms = np.exp(alpha * (corr - m))
    Z = np.sum(exp_terms)
    w = exp_terms / Z
    obj = (m + np.log(Z) / alpha) * dx
    # Gradient: dx * sum_k w[k] * grad_corr[k]
    g = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] > 1e-10:
            g += w[k_idx] * _grad_at_lag(h, k_idx)
    g *= dx
    return obj, g

        return val
    def obj_grad(x):
        _, g = _smooth_obj_with_grad(x, alpha, n)
        return g

Actually, let me think about this differently. The function f(v) = max_k (a_k + v * b_k) is piecewise linear and convex. The minimum of a convex piecewise linear function on an interval occurs at a "kink" point or at the boundary. The kink points are where two lines cross.

So I can:
1. Compute a_k and b_k for all k (where corr[k] = a_k + v * b_k)
2. Find all kink points in [lo, hi]
3. Evaluate f at each kink point and at the boundaries
4. Return the best

This is exact! And it's O(n^2) to find all kink points (for each pair of lines, compute the crossing point and check if it's in [lo, hi]).

For n=100, there are 199 correlation values, so O(199^2) = ~40000 kink points to check. That's fast.

def _find_best_pair_value(h, i, j, s_pair):
    """Find the best h[i] = v (with h[j] = s_pair - v) to minimize C5.
    Returns the optimal v."""
    n = len(h)
    lo = max(0.0, s_pair - 1.0)
    hi = min(1.0, s_pair)
    if hi - lo < 1e-12:
        return h[i]
    
    # Save and set h[i] = 0, h[j] = s_pair to compute base correlation
    old_i = h[i]
    old_j = h[j]
    h[i] = 0.0
    h[j] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    
    # Now compute the effect of setting h[i] = v and h[j] = s_pair - v
    # corr[k] = corr_base[k] + v * delta_i[k] + (s_pair - v) * delta_j[k]
    # = corr_base[k] + s_pair * delta_j[k] + v * (delta_i[k] - delta_j[k])
    
    h[i] = 1.0
    h[j] = 0.0
    corr_i = np.correlate(h, 1.0 - h, mode='full')
    delta_i = corr_i - corr_base
    
    h[i] = 0.0
    h[j] = 1.0
    corr_j = np.correlate(h, 1.0 - h, mode='full')
    delta_j = corr_j - corr_base
    
    h[i] = old_i
    h[j] = old_j
    
    # corr[k] as function of v:
    # a_k = corr_base[k] + s_pair * delta_j[k]
    # b_k = delta_i[k] - delta_j[k]
    a = corr_base + s_pair * delta_j
    b = delta_i - delta_j
    
    # f(v) = max_k (a_k + v * b_k)
    # Find kink points: for each pair (p, q) where b_p != b_q:
    # a_p + v*b_p = a_q + v*b_q => v = (a_q - a_p) / (b_p - b_q)
    
    # But this is O(n^2) pairs. For n=199 correlation values, that's ~20000 pairs.
    # Let me just use a simpler approach: evaluate at a fine grid plus the kink points near the minimum.
    
    # Actually, for convex piecewise linear functions, the minimum is where the subgradient contains 0.
    # The subgradient at v is conv({b_k : k in active set at v}).
    # So I need to find v where min_active(b) <= 0 <= max_active(b).
    
    # Simple approach: binary search on v
    # f is convex, so f'(v) is monotone (in the sense that the subgradient shifts right)
    
    # Even simpler: just evaluate at the boundaries and use the fact that the minimum
    # of a convex function is where the slope changes sign.
    
    # Let me just do a fine grid search + local refinement
    n_eval = 200
    v_grid = np.linspace(lo, hi, n_eval)
    f_vals = np.max(a[None, :] + v_grid[:, None] * b[None, :], axis=1)
    idx_min = np.argmin(f_vals)
    
    # Refine around the minimum
    if 0 < idx_min < n_eval - 1:
        v_coarse = v_grid[idx_min]
        # The kink points near v_coarse are where a[k1] + v_coarse*b[k1] = a[k2] + v_coarse*b[k2]
        # and both are active (in the max)
        active = np.where(np.abs(a + v_coarse * b - np.max(a + v_coarse * b)) < 1e-10)[0]
        kinks = set()
        for p_idx in range(len(active)):
            for q_idx in range(p_idx+1, len(active)):
                p = active[p_idx]
                q = active[q_idx]
                if abs(b[p] - b[q]) > 1e-15:
                    v_kink = (a[q] - a[p]) / (b[p] - b[q])
                    if lo <= v_kink <= hi:
                        kinks.add(v_kink)
        # Evaluate at kinks
        best_v = v_grid[idx_min]
        best_f = f_vals[idx_min]
        for vk in kinks:
            fv = np.max(a + vk * b)
            if fv < best_f:
                best_f = fv
                best_v = vk
        # Also check boundaries
        if np.max(a + lo * b) < best_f:
            best_f = np.max(a + lo * b)
            best_v = lo
        if np.max(a + hi * b) < best_f:
            best_f = np.max(a + hi * b)
            best_v = hi
        return best_v
    else:
        return v_grid[idx_min]

Actually, even simpler: since f(v) is convex, I can use bisection on the subgradient. The subgradient at v is the set {b_k : k is active at v} (more precisely, the convex hull of those b_k values). The minimum is where 0 ∈ subgradient, i.e., min_{active} b_k <= 0 <= max_{active} b_k.

Algorithm:
1. Start with v_lo = lo, v_hi = hi
2. At v = (v_lo + v_hi) / 2:
   - Find active set: all k where a_k + v*b_k = max
   - If min_active(b) >= 0: the function is non-decreasing at v, so minimum is to the left → v_hi = v
   - If max_active(b) <= 0: the function is non-increasing at v, so minimum is to the right → v_lo = v
   - Otherwise: 0 is in the subgradient, so v is optimal (or very close)
3. Repeat until v_hi - v_lo < tol

This is O(n log(1/tol)) which is very fast.

Let me implement this:

```python
def _bisection_subgrad(a, b, lo, hi, tol=1e-10):
    """Find v in [lo, hi] that minimizes max_k (a_k + v * b_k) using subgradient bisection."""
    for _ in range(60):
        v = (lo + hi) / 2.0
        vals = a + v * b
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        min_b = np.min(b[active])
        max_b = np.max(b[active])
        if min_b >= 0:
            # Function is non-decreasing, go left
            hi = v
        elif max_b <= 0:
            # Function is non-increasing, go right
            lo = v
        else:
            # 0 is in subgradient, v is optimal
            break
    return (lo + hi) / 2.0
```

from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

        if s > S:
            lo = mid
        else:

    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:

            for i in order:
                if diff <= 1e-14:

                add = min(diff, 1.0 - x[i])
                x[i] += add

                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub

def _grad_lag(h, k_idx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]

def _slsqp_smooth(h0, n, target_sum, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)

        return x.sum() - target_sum

                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:

        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _pair_cd(h, target_sum, max_rounds, pair_mode):
    """Pair coordinate descent with exact 1D optimization via subgradient bisection."""
    n = len(h)
    h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            if pair_mode == 0:
                j = (i + 1) % n
            elif pair_mode == 1:
                j = (i + n // 2) % n
            else:
                j = (i * 7 + 3) % n  # Some mixing
            if i == j:
                continue
            
            s_pair = h[i] + h[j]
            lo = max(0.0, s_pair - 1.0)
            hi = min(1.0, s_pair)
            if hi - lo < 1e-12:
                continue
            
            # Compute a and b for the 1D problem
            old_i = h[i]
            old_j = h[j]
            h[i] = 0.0
            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            
            h[i] = 1.0
            h[j] = 0.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            
            h[i] = 0.0
            h[j] = 1.0
            corr2 = np.correlate(h, 1.0 - h, mode='full')
            
            h[i] = old_i
            h[j] = old_j
            
            # corr(v) = corr0 + v*(corr1-corr0) + (s_pair-v)*(corr2-corr0)
            #         = corr0 + s_pair*(corr2-corr0) + v*((corr1-corr0)-(corr2-corr0))
            a = corr0 + s_pair * (corr2 - corr0)
            b = (corr1 - corr0) - (corr2 - corr0)
            
            # Minimize max_k (a_k + v * b_k) for v in [lo, hi]
            # Using subgradient bisection
            lo_b = lo
            hi_b = hi
            for _iter in range(50):
                v_mid = (lo_b + hi_b) / 2.0
                vals = a + v_mid * b
                max_val = np.max(vals)
                active_tol = 1e-8
                active = np.where(np.abs(vals - max_val) <= active_tol)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(vals)])
                min_b = np.min(b[active])
                max_b = np.max(b[active])
                if min_b >= 0:
                    hi_b = v_mid
                elif max_b <= 0:
                    lo_b = v_mid
                else:
                    lo_b = v_mid
                    hi_b = v_mid
                    break
            
            v_opt = (lo_b + hi_b) / 2.0
            new_c5 = np.max(a + v_opt * b) * (2.0 / n)
            old_c5 = _c5_of(h)
            
            if new_c5 < old_c5 - 1e-14:
                h[i] = v_opt
                h[j] = s_pair - v_opt
                improved = True
        
        if not improved:
            break
    
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(100, 0.5)
    
    n = len(h_init)
    target_sum = n / 2.0
    h_init = _fix_sum(h_init, target_sum)
    
    best_h = h_init.copy()
    best_c5 = _c5_of(h_init)
    
    # Phase 1: SLSQP polishing
    for alpha in [100, 300, 1000, 3000]:
        if time.time() > deadline - 250:
            break
        h_pol = _slsqp_smooth(best_h, n, target_sum, alpha, maxiter=300)
        c5 = _c5_of(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol
    
    # Phase 2: Nesterov
    for ss in [0.3, 0.5, 0.8]:
        if time.time() > deadline - 200:
            break
        h_nest, c5_nest = _nesterov_sub(best_h, n, target_sum, 500, ss, 5)
        if c5_nest < best_c5:
            best_c5 = c5_nest
            best_h = h_nest
    
    # Phase 3: Pair CD
    for pm in [0, 1]:
        if time.time() > deadline - 150:
            break
        h_cd = _pair_cd(best_h, target_sum, 20, pm)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 4: SLSQP polish
    for alpha in [1000, 5000]:
        if time.time() > deadline - 50:
            break
        h_pol = _slsqp_smooth(best_h, n, target_sum, alpha, maxiter=500)
        c5 = _c5_of(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol
    
    # Phase 5: More pair CD
    if time.time() < deadline - 50:
        h_cd = _pair_cd(best_h, target_sum, 20, 2)
        c5 = _c5_of(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd
    
    # Phase 6: Try different n
    if time.time() < deadline - 100:
        for n_try in [n+2, n-2, n+4, n-4]:
            if n_try < 50 or n_try % 2 != 0 or time.time() > deadline - 30:
                continue
            h_try = np.interp(np.linspace(0, n-1, n_try), np.arange(n), best_h)
            h_try = _fix_sum(np.clip(h_try, 0, 1), n_try/2.0)
            h_try = _slsqp_smooth(h_try, n_try, n_try/2.0, 500, maxiter=200)
            h_try, c5_try = _nesterov_sub(h_try, n_try, n_try/2.0, 200, 0.4, 3)
            c5_try = _c5_of(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                n = n_try
                target_sum = n / 2.0
    
    # Final polish
    if time.time() < deadline - 10:
        h_pol = _slsqp_smooth(best_h, n, target_sum, 5000, maxiter=500)
        c5 = _c5_of(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol
    
    if time.time() < deadline - 5:
        h_cd = _pair_cd(best_h, target_sum, 10, 0)
        c5 = _c5_of(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd
    
    best_h = _fix_sum(best_h, n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n)
