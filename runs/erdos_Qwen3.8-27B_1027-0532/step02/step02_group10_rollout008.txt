For k=0: O(0) = 0.5 - A²
For k=1: O(1) = 0.5 - A² cos(2π) = 0.5 - A² (same as k=0!)
For k=0.5: O(0.5) = 0.5 - A² cos(π) = 0.5 + A²

One more thought: the "targeted moves" idea. For the worst lag k*, I want to maximize ∑ h[i]h[i+k*]. The gradient of this w.r.t. h is:
d/dh[j] ∑ h[i]h[i+k*] = h[j-k*] + h[j+k*] (where valid)

So the "anti-gradient" of O(k*) w.r.t. h[j] is:
-(h[j-k*] + h[j+k*]) * dx

We want to move h in the direction that decreases O(k*), which means increasing h[j] where h[j-k*] + h[j+k*] is large... wait, that doesn't sound right.

Let me re-derive. O(k*) = ∑_i h[i](1-h[i+k*]) = ∑ h[i] - ∑ h[i]h[i+k*] = n/2 - ∑ h[i]h[i+k*]

To minimize O(k*), we maximize ∑ h[i]h[i+k*].
d/dh[j] (∑ h[i]h[i+k*]) = h[j-k*] (from the term where i=j-k*) + h[j+k*] (from the term where i=j, h[i+k*]=h[j+k*])

Wait: ∑_i h[i]h[i+k*]. The terms involving h[j] are:
- i = j: h[j] * h[j+k*]
- i = j - k*: h[j-k*] * h[j]

So d/dh[j] = h[j+k*] + h[j-k*] (where the indices are valid)

To maximize ∑h[i]h[i+k*], we should increase h[j] where h[j+k*] + h[j-k*] is large.
But we also need to decrease some other h values to maintain the sum constraint.

So the move is: transfer mass from h[j] where h[j+k*]+h[j-k*] is small to h[i] where h[i+k*]+h[i-k*] is large.

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_s = np.exp(np.clip(shifted, -500, 500))
    total = np.sum(exp_s)
    value = m + np.log(total) / alpha
    weights = exp_s / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
            grad[i_lo + offset:i_hi + offset] -= w * h[i_lo:i_hi]
    grad *= dx
    return grad


def targeted_moves(h, target, n_moves, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for _ in range(n_moves):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        worst_idx = np.argmax(ov)
        k_star = worst_idx - (n - 1)
        if k_star == 0:
            break
        scores = np.zeros(n)
        i_lo = max(0, -k_star)
        i_hi = min(n, n - k_star)
        scores[i_lo:i_hi] += h[i_lo + k_star:i_hi + k_star]
        i_lo2 = max(0, k_star)
        i_hi2 = min(n, n + k_star)
        scores[i_lo2:i_hi2] += h[i_lo2 - k_star:i_hi2 - k_star]
        source_candidates = np.where(scores < np.median(scores))[0]
        dest_candidates = np.where(scores > np.median(scores))[0]
        if len(source_candidates) == 0 or len(dest_candidates) == 0:
            break
        src = source_candidates[np.random.randint(len(source_candidates))]
        dst = dest_candidates[np.random.randint(len(dest_candidates))]
        if src == dst:
            continue
        delta = min(h[src], 0.1) * 0.5
        h[src] -= delta
        h[dst] += delta
        np.clip(h, 0.0, 1.0, out=h)
        h = project_onto_constraint(h, target)
    return h


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = min(0.5, 1.0 / (alpha * dx * 3.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 300:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        improved = False
        s = step
        for bt in range(40):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-13:
                break
        if not improved:
            step = s
            break
        step = min(s * 1.5, 1.0)
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.3, 1.0 / (alpha * dx * 5.0 + 1.0))
    prev_sm = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_and_weights(ov_x, alpha)
        if prev_sm is not None and sm_x > prev_sm + 1e-10:
            t_k = 1.0
        prev_sm = sm_x
        x = x_new
        t_k = t_new
    return x


def spectral_equalize(h, target, n_iters, deadline):
    n = len(h)
    h = h.copy()
    H = np.fft.rfft(h)
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        peak = np.max(ov)
        H_new = np.fft.rfft(h)
        ac_mask = np.ones(len(H_new), dtype=bool)
        ac_mask[0] = False
        magnitudes = np.abs(H_new[ac_mask])
        if magnitudes.max() > 0:
            target_mag = np.mean(magnitudes) * 1.1
            scale = np.minimum(1.0, target_mag / (magnitudes + 1e-30))
            angles = np.angle(H_new[ac_mask])
            H_new[ac_mask] = target_mag * scale * np.exp(1j * angles)
            h_new = np.fft.irfft(H_new, n=n)
            h_new = project_onto_constraint(h_new, target)
            sc_new = evaluate_score(h_new)
            if sc_new < peak:
                h = h_new
    return h


def upsample(h, new_n):
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def diverse_init(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    pattern = rng.randint(2, n_half + 1)
    for i in range(n):
        h[i] = 1.0 if (i % pattern) < pattern // 2 else 0.0
    h += rng.randn(n) * 0.1
    return project_onto_constraint(h, n / 2.0)
