```python
def binary_local_search(h, n, rng, time_budget):
    # h is binary (0s and 1s), sum = n/2
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    
    corr = np.correlate(h, 1-h, mode='full')
    best_max = np.max(corr)
    best_h = h.copy()
    
    t_start = time.time()
    while time.time() - t_start < time_budget:
        # Pick a random 1 and a random 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h_try = best_h.copy()
        h_try[i] = 0
        h_try[j] = 1
        
        corr_try = np.correlate(h_try, 1-h_try, mode='full')
        max_try = np.max(corr_try)
        
        if max_try < best_max - 1e-12:
            best_h = h_try
            best_max = max_try
            ones = np.flatnonzero(best_h == 1)
            zeros = np.flatnonzero(best_h == 0)
    
    return best_h, best_max * (2.0/n)
```

```python
    # Returns delta corr vector for swap (i:1→0, j:0→1)
    delta = np.zeros(2*n-1)
    # For each lag k (represented as index idx = k + n - 1 in the full corr):
    # delta[idx] = -(1-h[i+k])*[0≤i+k<n] + h[i-k]*[0≤i-k<n] + (1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]
    
    # This is equivalent to: -impact(i, .) + impact(j, .)
    # where impact(p, k) = (1-h[p+k])*[0≤p+k<n] - h[p-k]*[0≤p-k<n]

    # -(1-h[i+k]) for k in range(-i, n-i): i.e., 0 ≤ i+k < n
    # +h[i-k] for k in range(i-(n-1), i+1): i.e., 0 ≤ i-k < n

    # Term from h[i] going to 0:
    #   -(1-h[i+k]) for k where 0 ≤ i+k < n, i.e., k ∈ [-i, n-1-i]
    #   +h[i-k] for k where 0 ≤ i-k < n, i.e., k ∈ [i-(n-1), i]
    
    # In terms of correlation index idx = k + n - 1:
    # k = idx - (n-1)
    
    # Let me just vectorize it
    # For the -(1-h[i+k]) term:
    # Valid k: max(-(n-1), -i) ≤ k ≤ min(n-1, n-1-i)
    # i.e., k from -i to n-1-i (clamped to [-(n-1), n-1])
    
    lo_k1 = max(-(n-1), -i)
    hi_k1 = min(n-1, n-1-i)
    if lo_k1 <= hi_k1:
        ks = np.arange(lo_k1, hi_k1+1)
        idxs = ks + n - 1
        delta[idxs] -= (1.0 - h[i+ks])
    
    # For the +h[i-k] term:
    # Valid k: max(-(n-1), i-(n-1)) ≤ k ≤ min(n-1, i)
    lo_k2 = max(-(n-1), i-(n-1))
    hi_k2 = min(n-1, i)
    if lo_k2 <= hi_k2:
        ks = np.arange(lo_k2, hi_k2+1)
        idxs = ks + n - 1
        delta[idxs] += h[i-ks]
    
    # For the +(1-h[j+k]) term:
    lo_k3 = max(-(n-1), -j)
    hi_k3 = min(n-1, n-1-j)
    if lo_k3 <= hi_k3:
        ks = np.arange(lo_k3, hi_k3+1)
        idxs = ks + n - 1
        delta[idxs] += (1.0 - h[j+ks])

    lo_k4 = max(-(n-1), j-(n-1))

    return delta
```

```python
def compute_correlations(h):

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    sm_val = m + np.log(sum_exp) / alpha
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = n - 1 - idx
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < n:
                grad[:n - lag] += w
                grad[:n - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:n - lag]
        else:
            l = -lag
            if l < n:
                grad[l:] += w
                grad[l:] -= w * h[:n - l]
                grad[:n - l] -= w * h[l:]
    return sm_val, grad


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        sm = m + np.log(np.sum(np.exp(shifted))) / alpha
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        _, g = smooth_max_and_grad(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return res.x

        remaining_ls = budget_s - 45 - (time.time() - t0)
        if remaining_ls > 15:
            h_bin2 = make_binary(best_h, len(best_h))
            h_bin2, score_bin2 = binary_local_search(h_bin2, len(h_bin2), rng, remaining_ls)
            if score_bin2 < best_score:
                best_score = score_bin2
                best_h = h_bin2.copy()
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
```
