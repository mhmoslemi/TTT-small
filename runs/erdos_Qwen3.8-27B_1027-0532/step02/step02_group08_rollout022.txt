**Key innovation: "Lag-tracking" optimization**

Instead of using a smooth max that approximates all lags simultaneously:
1. Track the current maximum lag k*
2. Directly minimize corr[k*] (which is a smooth function of h)
3. Periodically re-evaluate all lags to check if k* has changed
4. When k* changes, switch to the new lag

This is essentially a "branch and bound" on the lags, but much more efficient because we're always optimizing a smooth objective.

The gradient of corr[k] with respect to h[i] is:
∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with appropriate boundary handling)

This is simple and cheap to compute!

So the algorithm becomes:
1. Compute all correlations, find k*
2. Compute gradient of corr[k*]
3. Take a projected gradient step
4. Every T iterations, recompute all correlations and update k*
5. Repeat

This is much simpler and potentially more effective than the smooth max approach.

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def corr_gradient_at_k(h, k):
    """Gradient of corr[k] with respect to h."""
    n = len(h)
    r = k - (n - 1)  # actual shift
    grad = np.zeros(n)
    for i in range(n):
        # corr[k] = sum_i h[i] * (1-h[i+r]) for valid i
        # d/dh[j] corr[k] = (1-h[j+r]) * delta(j==i) * ... 
        # Actually: corr[k] = sum_{i: 0<=i<n, 0<=i+r<n} h[i]*(1-h[i+r])
        # d/dh[j] = (1-h[j+r]) if 0<=j+r<n  [from the h[i] term with i=j]
        #           - h[j-r] if 0<=j-r<n    [from the (1-h[i+r]) term with i+r=j, i=j-r]
        val = 0.0
        if 0 <= j + r < n:
            val += (1.0 - h[j + r])
        if 0 <= j - r < n:
            val -= h[j - r]
        grad[j] = val
    return grad

    """Project h to have sum = target, staying in [0,1]."""
    n = len(h)
    h = h.copy()
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lag_tracking_optimize(h0, n_iter, recompute_freq, step_size_init, momentum=0.9):
    """Optimize by tracking the maximum lag."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)  # momentum
    step = step_size_init
    
    corr = compute_correlations(h)
    k_star = np.argmax(corr)
    
    for it in range(n_iter):
        if it % recompute_freq == 0:
            corr = compute_correlations(h)
            k_star = np.argmax(corr)
        
        grad = corr_gradient_at_k(h, k_star)

import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-16:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    # Gradient
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad

def penalty_wrapper(h, alpha, mu):
    obj, grad = smooth_max_obj_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad

        h[-1] = 0.0

            h[i] = 1.0
            h[j] = 0.0

Let me optimize: For a single swap (i: 1→0, j: 0→1), the change in correlation at lag r is:
- If we're looking at corr at shift r (where r = k - (n-1)):
  - The h[i] term: removes h[i]·(1-h[i+r]) and adds 0
  - The h[j] term: adds h[j]·(1-h[j+r]) = 1·(1-h[j+r])
  - The (1-h[i]) term in other correlates: for shift r' where i = i' + r', the (1-h[i]) changes from 0 to 1
  - Similarly for j

```
def active_lag_optimize(h, max_iters, recompute_every):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    k_star = np.argmax(corr)
    step = 0.1
    v = np.zeros(n)  # momentum
    best = np.max(corr)
    
    for it in range(max_iters):
        if it % recompute_every == 0:
            corr = compute_correlations(h)
            k_star = np.argmax(corr)
            if np.max(corr) < best:
                best = np.max(corr)
                step = min(step * 1.1, 0.5)  # increase step slightly
            else:
                step *= 0.7
        
        # Gradient of corr[k_star] w.r.t. h
        r = k_star - (n - 1)
        grad = np.zeros(n)
        # For each i: d corr / d h[i] = (1-h[i+r]) - h[i-r] (with bounds)
        if r >= 0:
            grad[:n-r] += (1.0 - h[r:n])  # (1-h[i+r]) for i=0..n-1-r
            grad[r:] -= h[:n-r]            # -h[i-r] for i=r..n-1
        else:
            grad[-r:] += (1.0 - h[:n+r])   # (1-h[i+r]) for i=-r..n-1
            grad[:-r] -= h[-r:]            # -h[i-r] for i=0..n-1+r
        
        # Momentum + projection
        v = 0.9 * v - step * grad
        h_new = project_sum(h + v, n / 2.0)
        h = h_new
    
    return h
```

This is O(n) per gradient computation and O(n) per step, so O(n) per iteration. Much faster than the smooth max gradient which is O(n²) due to the convolutions.

Let me verify the gradient formula:
corr[k] = ∑_{i: 0≤i<n, 0≤i+r<n} h[i] · (1-h[i+r]) where r = k-(n-1)

If r ≥ 0:
corr[k] = ∑_{i=0}^{n-1-r} h[i] · (1-h[i+r])

∂/∂h[j]:
- From h[j] term (j as the first index): (1-h[j+r]) if 0 ≤ j ≤ n-1-r
- From (1-h[j-r]) term (j as the second index, i.e., i+r=j → i=j-r): -h[j-r] if 0 ≤ j-r ≤ n-1-r, i.e., r ≤ j ≤ n-1

So:
grad[j] = (1-h[j+r]) · [0≤j≤n-1-r] - h[j-r] · [r≤j≤n-1]

If r < 0, let r' = -r > 0:
corr[k] = ∑_{i=r'}^{n-1} h[i] · (1-h[i-r'])

∂/∂h[j]:
- From h[j] term: (1-h[j-r']) if r' ≤ j ≤ n-1
- From (1-h[j+r']) term: -h[j+r'] if 0 ≤ j+r' ≤ n-1, i.e., 0 ≤ j ≤ n-1-r'

So:
grad[j] = (1-h[j-r']) · [r'≤j≤n-1] - h[j+r'] · [0≤j≤n-1-r']

        grad[rp:] += (1.0 - h[:n-rp])
        # -h[i+rp] for i=0..n-1-rp

    return smooth_max_obj_grad(h, alpha)[1]


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def corr_grad_at_lag(h, k):
    n = len(h)
    r = k - (n - 1)
    grad = np.zeros(n)
    if r >= 0:
        if r > 0:
            grad[:n - r] += (1.0 - h[r:n])
            grad[r:] -= h[:n - r]
        else:
            grad[:] += (1.0 - h)
    else:
        rp = -r
        if rp > 0:
            grad[rp:] += (1.0 - h[:n - rp])
            grad[:n - rp] -= h[rp:]

def active_lag_optimize(h0, max_iters, recompute_every, step_init, momentum, time_check):
    n = len(h0)

    v = np.zeros(n)

    h = h0.copy()

    corr = compute_correlations(h)

        # Active-lag at this resolution
        h_al2 = active_lag_optimize(h_curr, max_iters=20000, recompute_every=100,
                                    step_init=0.04, momentum=0.7, time_check=time_left)
        h_al2 = project_sum(h_al2, n_try / 2.0)
        s_al2 = evaluate_score(h_al2)
        if s_al2 < best_score:
            best_score = s_al2
            best_h = h_al2.copy()
            h_curr = h_al2.copy()
            n_best = n_try

        for alpha in [200, 500, 1000, 3000]:
            if time_left() < 60:
                break
            maxiter = max(80, min(400, 200000 // n_try))
            res_x = lbfgsb_with_penalty(h_curr, alpha, mu_try, maxiter, bounds_try)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_curr = h_new.copy()
                        n_best = n_try

    # Phase 7: Random restarts with diverse strategies
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    max_restarts = 500
    for i in range(max_restarts):
        if time_left() < 60:
            break
        strategy = i % 8
        if strategy == 0:
            pert = best_h + rng.randn(n_best) * 0.04
        elif strategy == 1:
            pert = best_h + rng.randn(n_best) * 0.12
        elif strategy == 2:
            pert = best_h.copy()
            n_swap = max(3, n_best // 20)
            idx = rng.choice(n_best, size=n_swap, replace=False)
            pert[idx] = 1.0 - pert[idx]
        elif strategy == 3:
            sorted_idx = np.argsort(-best_h)
            pert = np.zeros(n_best)
            pert[sorted_idx[:n_best // 2]] = 1.0
            pert += rng.randn(n_best) * 0.1
        elif strategy == 4:
            pert = best_h.copy()
            mask = rng.rand(n_best) < 0.15
            pert[mask] = np.clip(pert[mask] + rng.randn(int(np.sum(mask))) * 0.5, 0, 1)
        elif strategy == 5:
            # Shift-based perturbation
            shift = rng.randint(1, max(2, n_best // 50))
            pert = np.roll(best_h, shift)
            pert = project_sum(pert, n_best / 2.0)
        elif strategy == 6:
            # Blending with binary
            h_bin = np.zeros(n_best)
            h_bin[:n_best // 2] = 1.0
            t = rng.uniform(0.3, 0.7)
            pert = t * best_h + (1 - t) * h_bin
        else:
            # Frequency-domain perturbation
            h_fft = np.fft.rfft(best_h)
            h_fft[1:-1] *= (1.0 + rng.randn(len(h_fft) - 2) * 0.05)
            pert = np.fft.irfft(h_fft, n_best)
            pert = project_sum(pert, n_best / 2.0)

        pert = project_sum(pert, n_best / 2.0)
        # Quick active-lag optimization
        h_quick = active_lag_optimize(pert, max_iters=5000, recompute_every=100,
                                      step_init=0.03, momentum=0.7, time_check=time_left)
        h_quick = project_sum(h_quick, n_best / 2.0)
        if i % 3 == 0 and time_left() > 30:
            alpha_r = 200 + 400 * rng.randint(8)
            maxiter_r = max(50, min(200, 100000 // n_best))
            res_x = lbfgsb_with_penalty(h_quick, alpha_r, 8000.0, maxiter_r, bounds_best)
            if res_x is not None:
                h_quick = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)

        s = evaluate_score(h_quick)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_quick.copy()

    # Phase 8: Final SLSQP polish
    n_best = len(best_h)
    bounds_final = [(0.0, 1.0)] * n_best
    for alpha in [5000, 10000, 20000]:
        if time_left() < 20:
            break
        res_x = slsqp_solve(best_h, alpha, 150, bounds_final)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 9: Very high alpha L-BFGS-B
    for alpha in [30000, 60000]:
        if time_left() < 10:
            break
        res_x = lbfgsb_with_penalty(best_h, alpha, 30000.0, 200, bounds_final)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 10: Neighboring resolutions
    n_best = len(best_h)
    for n_adj in [n_best + 3, n_best - 3, n_best + 10, n_best - 10, n_best + 20]:
        if time_left() < 15:
            break
        if n_adj < 100 or n_adj > 2000:
            continue
        x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_adj, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_sum(h_init, n_adj / 2.0)
        h_adj = active_lag_optimize(h_init, max_iters=5000, recompute_every=50,
                                    step_init=0.03, momentum=0.7, time_check=time_left)
        h_adj = project_sum(h_adj, n_adj / 2.0)
        bounds_adj = [(0.0, 1.0)] * n_adj
        if time_left() > 10:
            res_x = lbfgsb_with_penalty(h_adj, 3000, 10000.0, 150, bounds_adj)
            if res_x is not None:
                h_adj = project_sum(np.clip(res_x, 0.0, 1.0), n_adj / 2.0)
        s = evaluate_score(h_adj)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_adj.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
