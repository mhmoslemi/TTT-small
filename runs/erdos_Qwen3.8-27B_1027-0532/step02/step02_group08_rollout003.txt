from scipy.optimize import minimize

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_h_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

    obj, grad = smooth_max_h_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0

                    need = to_flip[rng.randint(len(to_flip), size=target_ones - cur_ones)]
                    h_sa[need] = 1.0
            
            max_iters_sa = max(8000, min(50000, 30000000 // max(n_target, 1)))
            h_sa_best, cost_sa = simulated_annealing_binary(
                h_sa, 8.0, 0.005, 0.999, max_iters_sa, rng
            )
            dx_t = 2.0 / n_target
            score_sa = cost_sa * dx_t
            if score_sa < best_score:
                best_score = score_sa
                best_h = h_sa_best.copy()

    # Phase 6: Final gradient polish
    n_final = len(best_h)
    bounds_final = [(0.0, 1.0)] * n_final
    best_h = project_sum(np.clip(best_h, 0.0, 1.0), n_final / 2.0)
    
    for alpha in [2000, 5000, 10000, 20000]:
        if time_left() < 20:
            break
        h_new = lbfgsb_optimize(best_h, alpha, 20000.0, 300, bounds_final)
        if h_new is not None and np.all(np.isfinite(h_new)):
            h_new = project_sum(np.clip(h_new, 0.0, 1.0), n_final / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 7: SLSQP final refinement
    if time_left() > 30:
        n_f = len(best_h)
        bounds_f = [(0.0, 1.0)] * n_f
        
        def slsqp_fun(h_x, a):
            return smooth_max_h_grad(h_x, a)[0]
        
        def slsqp_jac(h_x, a):
            return smooth_max_h_grad(h_x, a)[1]
        
        def cons_eq(h_x):
            return np.sum(h_x) - len(h_x) / 2.0
        
        def cons_eq_jac(h_x):
            return np.ones(len(h_x))
        
        res = minimize(
            slsqp_fun, best_h, method='SLSQP', bounds=bounds_f, args=(5000,),
            jac=slsqp_jac,
            constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}],
            options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_sum(np.clip(res.x, 0.0, 1.0), len(res.x) / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
