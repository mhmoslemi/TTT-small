def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad_at_lag(h, lag):

        grad[lag:] += (1.0 - h[:n - lag])

        grad[:n - L] += (1.0 - h[L:])

def _corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _proj_grad_descent(h_init, target_sum, n_iter, lr0, lr_decay, top_k, momentum, deadline, check_every=200):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 20
    
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        lr = lr0 / (1.0 + t * lr_decay)
        corr = _corr_values(h)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            grad += _corr_grad_at_lag(h, lag)
        grad /= len(top_idx)
        
        v = momentum * v - lr * grad
        h = _fix_sum(h + v, target_sum)
        
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_refine(h_init, alpha, maxiter=200, ftol=1e-13):
    npts = len(h_init)
    target_sum = npts / 2.0
    dx = 2.0 / npts
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    
    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

            c5_r = _c5_of(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    if best_h is not None and time.time() < deadline - 10:
        h_r = _slsqp_direct_max(best_h, maxiter=200, ftol=1e-14)
        h_r = _fix_sum(h_r, best_n / 2.0)
        c5_r = _c5_of(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _fix_sum(best_h, n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
