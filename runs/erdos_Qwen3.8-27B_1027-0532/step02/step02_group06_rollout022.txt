```python
def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_smooth(h0, alpha, target_sum, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = compute_correlations(x)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def obj_grad(x):
        corr = compute_correlations(x)
        m = np.max(corr)
        exp_alpha = np.exp(alpha * (corr - m))
        w_sum = np.sum(exp_alpha)
        w = exp_alpha / w_sum
        grad = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-12:
                continue
            L = k - (n - 1)
            if L == 0:
                grad[:] += wk * (1.0 - 2.0 * x)
            elif L > 0:
                grad[L:] += wk * (1.0 - x[:-L])
                grad[:n-L] -= wk * x[L:]
            else:
                la = -L
                grad[:n-la] += wk * (1.0 - x[la:])
                grad[la:] -= wk * x[:n-la]
        return grad
    
    def cfun(x):
        return np.sum(x) - target_sum
    
    def cjac(x):
        return np.ones(n)
    
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def binary_refine(h_cont, n_rounds=50):
    """Convert to binary and do local search."""
    n = len(h_cont)
    target = n // 2  # Should be n/2 since sum(h)*dx = 1 means sum(h) = n/2
    
    # Round to binary
    h = np.zeros(n, dtype=np.float64)
    # Sort by value, take top n/2
    order = np.argsort(-h_cont)
    h[order[:target]] = 1.0
    
    # But we need exactly n/2 ones
    current_sum = np.sum(h)
    if current_sum != target:
        # Adjust
        order2 = np.argsort(-h_cont)
        h = np.zeros(n)
        h[order2[:target]] = 1.0
    
    corr = compute_correlations(h)
    score = np.max(corr) * (2.0 / n)
    
    for _ in range(n_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Try moving 1 from i to j
                h[i] = 0.0
                h[j] = 1.0
                corr_new = compute_correlations(h)
                score_new = np.max(corr_new) * (2.0 / n)
                if score_new < score - 1e-15:
                    score = score_new
                    corr = corr_new
                    improved = True
                    # Continue trying with this new state
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        
        if not improved:
            break
    
    return h, score
```

```python
        zeros = np.where(h == 0.0)[0]
        
        best_delta = 0
        best_pair = None
        
        for _ in range(n_candidates):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Compute delta for swap (i, j)
            delta = np.zeros(len(corr))
            # Effect of h[i]: 1 -> 0
            L_range = np.arange(-(n-1), n)  # all lags
            # i as first index: valid when 0 <= i + L < n, i.e., -i <= L < n-i
            valid1 = (0 <= i + L_range) & (i + L_range < n)
            delta[valid1] -= (1 - h[i + L_range[valid1]])
            # i as second index: valid when 0 <= i - L < n, i.e., i-n < L <= i
            valid2 = (0 <= i - L_range) & (i - L_range < n)
            delta[valid2] += h[i - L_range[valid2]]
            # Effect of h[j]: 0 -> 1
            valid3 = (0 <= j + L_range) & (j + L_range < n)
            delta[valid3] += (1 - h[j + L_range[valid3]])
            valid4 = (0 <= j - L_range) & (j - L_range < n)
            delta[valid4] -= h[j - L_range[valid4]]
            
            new_scores = corr + delta
            new_max = np.max(new_scores)
            
            if new_max < score + best_delta - 1e-12:
                best_delta = new_max - score
                best_pair = (i, j, delta)
        
        if best_pair is not None:
            i, j, delta = best_pair
            h[i] = 0.0
            h[j] = 1.0

            improved = True
```

```python
        if L == 0:
            grad[:] += wk * (1.0 - 2.0 * x)
        elif L > 0:
            grad[L:] += wk * (1.0 - x[:-L])
            grad[:N-L] -= wk * x[L:]
        else:
            la = -L
            grad[:N-la] += wk * (1.0 - x[la:])
            grad[la:] -= wk * x[:N-la]
    if _g_use_penalty:
        grad += 2.0 * _g_gamma * (np.sum(x) - _g_target)
    return grad
```

This way, I can use the same `_obj` and `_grad` for both SLSQP and L-BFGS-B, just by toggling `_g_use_penalty`.

For the binary local search, let me write a clean version:

```python
def binary_search(h_cont, n_trials=100000, seed=42):
    N = len(h_cont)
    dx = 2.0 / N
    target = N // 2
    rng = np.random.RandomState(seed)
    
    # Round to binary: take top N/2 values
    h = np.zeros(N)
    order = np.argsort(-h_cont)
    h[order[:target]] = 1.0
    
    corr = compute_correlations(h)
    score = np.max(corr)
    
    for trial in range(n_trials):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_correlations(h)
        new_score = np.max(new_corr)
        if new_score < score - 1e-14:
            score = new_score
            corr = new_corr
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    final_score = score * dx
    return h, final_score
```

This is O(n_trials * N^2) for the correlation computations. For N=425 and 100K trials: 100K * 425^2 ≈ 18B ops. At numpy speed, each correlation takes about 50-100 µs (it's a C function). So 100K * 100µs = 10 seconds. That's fine!

```python
                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
    
    # Phase 3: Binary local search
    if time_left() > 60:
        h_bin, score_bin = binary_search(best_h, n_trials=50000, seed=seed+1)
        if score_bin < best_score:
            best_score = score_bin
            best_h = h_bin.copy()
    
    # Phase 4: More SLSQP with very high alpha
    if time_left() > 100:
        for alpha in [5000, 10000, 20000, 50000]:
            if time_left() < 50:

            h_new = slsqp_run(best_h, alpha, target, maxiter=200)
            if h_new is not None:
                score = evaluate_score(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
    
    # Phase 5: Multiple restarts
    if time_left() > 200:
        for rk in range(5):
            if time_left() < 60:
                break
            h_pert = best_h + rng.randn(N) * 0.05 / np.sqrt(N)

            h_new = slsqp_run(h_pert, 1000, target, maxiter=200)

                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
            # Also try binary from this perturbation
            if time_left() > 30:

                if s_b < best_score:
                    best_score = s_b
                    best_h = h_b.copy()
    
    # Phase 6: Fine grid (if time allows)

        for alpha in [100, 1000, 5000]:

                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()

    best_h = project_to_feasible(best_h, N_final / 2.0)
    best_score = evaluate_score(best_h)
    
    return (best_h, best_score, N_final)
```
