import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_binary(h, n, target):
    """Round h to binary and repair to have exactly target ones."""
    h = (h >= 0.5).astype(float)
    current_sum = int(np.sum(h))
    if current_sum == target:
        return h
    elif current_sum > target:
        # Need to flip (current_sum - target) ones to zeros
        excess = current_sum - target
        ones = np.where(h == 1.0)[0]
        np.random.shuffle(ones)
        for i in range(excess):
            h[ones[i]] = 0.0
    else:
        # Need to flip (target - current_sum) zeros to ones
        deficit = target - current_sum
        zeros = np.where(h == 0.0)[0]
        np.random.shuffle(zeros)
        for i in range(deficit):
            h[zeros[i]] = 1.0
    return h

def local_search_binary(h, n, time_limit_fn):
    """Greedy local search: accept any move that decreases C5."""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    target = n // 2

    #   Add position i as a "0": gain h[i]*(1-h[i+l]) = 0 and h[i-l]*(1-h[i]) = h[i-l]*1 = h[i-l]
    #   Remove position j as a "0": lose h[j]*(1-h[j+l]) = 0 and h[j-l]*(1-h[j]) = h[j-l]*1 = h[j-l]
    #   Add position j as a "1": gain h[j]*(1-h[j+l]) = 1*(1-h[j+l]) and h[j-l]*(1-h[j]) = 0

        if 0 <= jm < n:

    # Update h

    h[j] = 1.0
    return corr_full

- m=j-l: old contribution = h_old[j-l]*(1-h_old[j]) = h_old[j-l]*1 = h_old[j-l]. New = h_new[j-l]*(1-h_new[j]) = h_new[j-l]*0 = 0. Change: -h_new[j-l].
  But h_new[j-l] might be h_new[i] if j-l=i, in which case h_new[i]=0.

Now, if i and j are distinct (which they are, since h[i]=1 and h[j]=0), the only issue is when the "neighbor" of one is the other:
- i+l = j: then in the m=i term, h_old[i+l] = h_old[j] = 0.
- i-l = j: then in the m=i-l term, h_new[i-l] = h_new[j] = 1.
- j+l = i: then in the m=j term, h_new[j+l] = h_new[i] = 0.
- j-l = i: then in the m=j-l term, h_new[j-l] = h_new[i] = 0.

Also, there could be overlap between the affected terms:
- m=i and m=j are the same if i=j (impossible).
- m=i and m=j-l are the same if i = j-l, i.e., l = j-i.
- m=i-l and m=j are the same if i-l = j, i.e., l = i-j.
- m=i-l and m=j-l are the same if i=j (impossible).
- m=i and m=i-l are the same if l=0.
- m=j and m=j-l are the same if l=0.

So the overlaps are:
1. l=0: m=i term and m=i-l=i term overlap. Also m=j and m=j-l=j overlap.
2. l=j-i: m=i term and m=j-l=j-(j-i)=i term overlap.
3. l=i-j: m=i-l=i-(i-j)=j term and m=j term overlap.

In cases 2 and 3, two of the four "affected terms" are actually the same term in the sum. This means I should only count it once.

    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")

    target = n // 2
