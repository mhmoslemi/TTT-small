import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def adcoord_descent(h_start, n, time_limit_frac, start_time, budget_s):
    """Adaptive coordinate descent with geometric step sizes."""
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / n
    
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in step_sizes:
        improved_any = True
        while improved_any:
            if time.time() - start_time > budget_s * time_limit_frac:
                return h, best_c5
            improved_any = False

                        break

        s = np.sum(np.clip(x - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid

def compute_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr)
    l = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        term = 0.0
        jml = j - l
        if 0 <= jml < n:
            term += (1.0 - h[jml])
        jpl = j + l
        if 0 <= jpl < n:
            term += (-h[jpl])
        grad[j] = term
    grad *= dx
    return grad

                        h_try[i] = vi
                        h_try[j] = vj
                        c5 = evaluate_c5(h_try)
                        if c5 < best_c5:
                            h = h_try
                            best_c5 = c5
                            improved = True
                            break
            if not improved and step < 0.001:
                break
        if time.time() > deadline:
            break
    return h, best_c5

def upsample(h, n_from, n_to):
    x_old = np.linspace(0.0, 2.0, n_from)
    x_new = np.linspace(0.0, 2.0, n_to)
    out = np.interp(x_new, x_old, h)
    return out

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def deadline(frac):
        return start_time + budget_s * frac

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h = project_feasible(h0, n)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    d1 = deadline(0.08)
    h, c5 = coord_descent(h, n, [0.1, 0.05, 0.02, 0.01], 30, d1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    d2 = deadline(0.25)
    for restart in range(6):
        if time.time() > d2:
            break
        if restart == 0:
            h_pert = best_h.copy()
        else:
            scale = 0.01 + 0.02 * restart / 6.0
            h_pert = best_h + scale * np.random.randn(n)
            h_pert = project_feasible(h_pert, n)
        h_sg, c5_sg = subgrad_descent(h_pert, n, 3000, 0.3, d2)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        h_cd, c5_cd = coord_descent(best_h, n, [0.01, 0.005, 0.002], 20, d2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    d3 = deadline(0.45)
    h, c5 = coord_descent(best_h, n, [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], 50, d3)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    d4 = deadline(0.60)
    h, c5 = pair_coord_descent(best_h, n, [0.005, 0.002, 0.001], 30, d4)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    d5 = deadline(0.75)
    h_sa, c5_sa = simulated_annealing(best_h, n, 300000, 0.005, 1e-5, d5)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    h, c5 = coord_descent(best_h, n, [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], 40, d5)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    d6 = deadline(0.92)
    n2 = 2 * n
    h_up = project_feasible(upsample(best_h, n, n2), n2)
    c5_up = evaluate_c5(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
        n = n2
    h, c5 = coord_descent(best_h, n, [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005], 30, d6)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    d7 = deadline(0.97)
    h, c5 = coord_descent(best_h, n, [0.001, 0.0005, 0.0002, 0.0001], 40, d7)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
