import time

    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} using bisection on Lagrange multiplier"""
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) * 0.5
    return np.clip(h - mu, 0.0, 1.0)

    max_c = np.max(corr)
    w = np.exp(alpha * (corr - max_c))
    w /= w.sum()
    return w

def compute_full_gradient(h, alpha):
    """Compute gradient of smooth max of correlations w.r.t. h"""
    n = len(h)
    m = 2 * n - 1
    corr = compute_correlations(h)
    weights = smooth_max_weights(corr, alpha)

    grad = np.zeros(n)

            grad[:n - L] -= w * h[L:]
        else:
            La = -L
            grad[La:] += w * one_minus_h[:n - La]
            grad[:n - La] -= w * h[La:]
    return grad

def subgradient_descent(h, alpha, eta, n_steps, momentum=0.9, time_budget=None, t0=None):
    """Momentum subgradient descent with projection"""
    h = h.copy()
    n = len(h)
    target = n / 2.0
    velocity = np.zeros(n)
    
    for step in range(n_steps):
        if time_budget and t0 and time.time() - t0 > time_budget:
            break
        grad = compute_full_gradient(h, alpha)
        velocity = momentum * velocity - eta * grad
        h_new = project_sum(h + velocity, target)
        h = h_new
    
    return h

    n = len(h)
    h_j = h[j]
