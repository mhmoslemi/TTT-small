import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def compute_grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def weighted_gradient(h, corr, alpha, n_top):
    n = len(h)
    top_idx = np.argsort(-corr)[:n_top]
    vals = corr[top_idx]
    m = vals[0]
    shifted = np.clip(alpha * (vals - m), -700, 700)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(n_top)
        w[0] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi > 1e-12:
            lag = int(idx - (n - 1))
            grad += wi * compute_grad_lag(h, int(lag))
    return grad

def subgradient_optimize(h0, max_iters, alpha, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    momentum = np.zeros(n)
    beta = 0.7

    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        corr = compute_correlations(h)
        grad = weighted_gradient(h, corr, alpha, n_top)
        momentum = beta * momentum + (1.0 - beta) * grad
        grad_norm_sq = np.dot(momentum, momentum)
        if grad_norm_sq < 1e-20:
            break

        improved = False
        c1 = 1e-4
        rho = 0.5
        f_curr = smooth_max_val(corr, alpha)
        for trial in range(20):
            h_cand = h - step * momentum
            h_cand = project_to_feasible(h_cand, target_sum)
            corr_cand = compute_correlations(h_cand)
            f_cand = smooth_max_val(corr_cand, alpha)
            if f_cand <= f_curr - c1 * step * grad_norm_sq:
                h = h_cand
                corr = corr_cand
                improved = True
                break
            step *= rho

        if not improved:
            h = h - step * momentum
            h = project_to_feasible(h, target_sum)
            corr = compute_correlations(h)
            step = step_init * (0.5 + 0.5 * rng.rand())

        score = np.max(corr) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            stall = 0
            if improved and trial < 15:
                step = min(step * 1.05, step_init * 3.0)
        else:
            stall += 1
            if stall > 300:
                scale = 0.02 + 0.04 * min(stall, 1000) / 1000.0
                h = best_h + rng.randn(n) * scale / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init
                momentum = np.zeros(n)
                stall = 0

        if it > 0 and it % 10000 == 0:
            step = max(step_init * 0.99 ** (it // 10000), step_init * 0.01)
            beta = max(0.3, beta * 0.95)

    return best_h, best_score

def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700, 700)
        return m + np.log(np.sum(np.exp(shifted))) / alpha

    def obj_grad(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700, 700)
        w = np.exp(shifted)
        w /= np.sum(w)
        grad = np.zeros(n)
        for idx in range(len(corr)):
            if w[idx] > 1e-13:
                lag = int(idx - (n - 1))
                grad += w[idx] * compute_grad_lag(hv, int(lag))
        return grad

    def con_fun(hv):
        return np.sum(hv) - target_sum

    def con_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def slsqp_multi_restart(h, alpha, n_restarts, maxiter, seed, t0, time_limit):
    n = len(h)
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_score = evaluate_score(h)
    for r in range(n_restarts):
        if time.time() - t0 > t0 + time_limit:
            break
        if r == 0:
            start = h.copy()
        else:
            scale = 0.005 + 0.02 * r / n_restarts
            start = h + rng.randn(n) * scale / np.sqrt(n)
            start = project_to_feasible(start, target_sum)
        h_ref = slsqp_refine(start, alpha, maxiter=maxiter)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref
    return best_h, best_score
