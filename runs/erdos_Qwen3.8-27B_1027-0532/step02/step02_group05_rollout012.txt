    return np.correlate(h, 1.0 - h, mode='full')

    corr = compute_correlations(h)
    return float(np.max(corr)) * dx

    m = float(np.max(x))
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)

        if w < 1e-14:

            grad[:N - k] -= w * h[k:]

            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]

    p = ex / np.sum(ex)

        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
