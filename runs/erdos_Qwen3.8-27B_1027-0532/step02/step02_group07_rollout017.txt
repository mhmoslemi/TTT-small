```python
    best = evaluate_score(h)
    best_h = h.copy()

        max_val = np.max(corr)
```

- Second term (from idx2): valid for p in [max(0, j), min(2n-2, n-1+j)]

```python
def compute_slopes_fast(h, j):
    n = len(h)
    m = 2 * n - 1
    slopes = np.zeros(m)
    # First term: p in [n-1-j, n-1+j], contribution (1-h[j+p-(n-1)])
    p_start1 = max(0, n - 1 - j)
    p_end1 = min(m - 1, n - 1 + j)
    if p_start1 <= p_end1:
        idx = j + np.arange(p_start1, p_end1 + 1) - (n - 1)
        slopes[p_start1:p_end1+1] += (1.0 - h[idx])
    # Second term: p in [j, n-1+j], contribution -h[j-p+(n-1)]
    p_start2 = max(0, j)
    p_end2 = min(m - 1, n - 1 + j)
    if p_start2 <= p_end2:
        idx = j - np.arange(p_start2, p_end2 + 1) + (n - 1)
        slopes[p_start2:p_end2+1] -= h[idx]
    return slopes
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
            constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
        )
        return res.x if res.x is not None else None
    except Exception:
        return None


def compute_slopes_fast(h, j):
    n = len(h)
    m = 2 * n - 1
    slopes = np.zeros(m)
    p_start1 = max(0, n - 1 - j)
    p_end1 = min(m - 1, n - 1 + j)
    if p_start1 <= p_end1:
        idx = j + np.arange(p_start1, p_end1 + 1) - (n - 1)
        slopes[p_start1:p_end1 + 1] += (1.0 - h[idx])
    p_start2 = max(0, j)
    p_end2 = min(m - 1, n - 1 + j)
    if p_start2 <= p_end2:
        idx = j - np.arange(p_start2, p_end2 + 1) + (n - 1)
        slopes[p_start2:p_end2 + 1] -= h[idx]
    return slopes


def optimize_coordinate(h, j, corr):
    n = len(h)
    slopes = compute_slopes_fast(h, j)
    h_j = h[j]
    m = 2 * n - 1
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    for _ in range(40):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def sa_optimize(h, rng, time_budget=150.0, t_start=0.3, t_end=0.001):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    t0 = time.time()
    n_steps = 0
    accept_count = 0
    while time.time() - t0 < time_budget:
        frac = (time.time() - t0) / time_budget
        T = t_start * (t_end / t_start) ** frac
        j1 = rng.randint(n)
        j2 = rng.randint(n - 1)
        if j2 >= j1:
            j2 += 1
        delta = rng.uniform(-0.3, 0.3)

        new_h2 = h[j2] - delta
        if new_h1 < 0.0 or new_h1 > 1.0 or new_h2 < 0.0 or new_h2 > 1.0:
            continue
        slopes1 = compute_slopes_fast(h, j1)
        slopes2 = compute_slopes_fast(h, j2)
        delta_corr = delta * (slopes1 - slopes2)
        new_max_row = corr + delta_corr
        new_max = float(np.max(new_max_row))
        dE = new_max - current_max
        if dE < 0 or rng.random() < np.exp(-dE / T):
            h[j1] = new_h1
            h[j2] = new_h2
            corr = new_max_row
            current_max = new_max
            accept_count += 1
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        n_steps += 1
    h = best_h
    h = project_sum(h, n / 2.0)

def subgradient_method(h, time_budget=80.0, eta0=0.005, momentum=0.85):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    velocity = np.zeros(n)
    best = float(np.max(corr))
    best_h = h.copy()
    t0 = time.time()
    step = 0
    while time.time() - t0 < time_budget:
        step += 1
        max_val = float(np.max(corr))
        threshold = max_val - 1e-4
        S = np.where(corr >= threshold)[0]
        n_S = len(S)
        if n_S == 0:
            S = np.array([int(np.argmax(corr))])
            n_S = 1
        g = np.zeros(n)
        for j in range(n):
            sl = compute_slopes_fast(h, j)
            g[j] = np.sum(sl[S]) / n_S
        velocity = momentum * velocity - eta0 * g
        h_new = h + velocity
        h_new = project_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)
        corr = compute_correlations(h_new)
        cur_max = float(np.max(corr))
        if cur_max < best:
            best = cur_max
            best_h = h_new.copy()
        h = h_new
        eta0 *= 0.9995
    return best_h


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    h_interp = project_sum(h_interp, n_new / 2.0)
    return h_interp
```
