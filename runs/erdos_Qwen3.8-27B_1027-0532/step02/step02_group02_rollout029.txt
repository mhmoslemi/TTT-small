import time
import numpy as np
from scipy.optimize import minimize

def _corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_box_sum(y, target_sum):
    n = len(y)
    lo = y.min() - 0.5
    hi = y.max() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _smooth_max_step(h, alpha, top_m, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    half = corr[n - 1:]
    if top_m is not None and top_m < len(half):
        idx = np.argpartition(half, -top_m)[-top_m:]
        c_sub = half[idx]
    else:
        idx = np.arange(len(half))
        c_sub = half
    m = c_sub.max()
    exp_vals = np.exp(alpha * (c_sub - m))
    w = exp_vals / exp_vals.sum()
    obj = m + np.log(exp_vals.sum()) / alpha
    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-14:
            continue
        lag = int(idx[i])
        grad += w[i] * _corr_grad(h, lag)
    return obj, grad

def _momentum_optimize(h_init, n, alpha, n_iter, lr, momentum, top_m, deadline, check_every=500):
    target_sum = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 15
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        cur_lr = lr / (1.0 + 0.001 * t)
        _, grad = _smooth_max_step(h, alpha, top_m, n)
        v = momentum * v - cur_lr * grad
        h = _project_box_sum(h + v, target_sum)
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_polish(h_init, n, alpha, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)
    bounds = [(0.0, 1.0)] * npts

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

    def _obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        w = np.exp(alpha * (corr - m))
        w /= w.sum()
        g = np.zeros(npts)
        for k in range(len(corr)):
            if w[k] < 1e-14:
                continue
            lag = k - (npts - 1)
            g += w[k] * _corr_grad(x, lag)
        return g * dx

    def _con_fun(x):
        return x.sum() - target_sum

    def _con_jac(x):
        return np.ones(npts)

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h_init, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def _make_starts(n, seed, include_initial):
    rng = np.random.RandomState(seed)
    starts = []
    target_sum = n / 2.0
    if include_initial and 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        if len(h_init) == n:
            starts.append(_project_box_sum(h_init, target_sum))
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(_project_box_sum(0.5 + 0.3 * np.sin(2 * np.pi * x), target_sum))
    starts.append(_project_box_sum(0.5 + 0.3 * np.sin(4 * np.pi * x), target_sum))
    starts.append(_project_box_sum(0.5 + 0.2 * np.sin(2 * np.pi * x * 3) + 0.1 * np.sin(4 * np.pi * x), target_sum))
    starts.append(_project_box_sum(rng.rand(n), target_sum))
    period = max(2, n // 5)
    saw = np.zeros(n)
    for i in range(n):
        saw[i] = 1.0 if (i % (2 * period)) < period else 0.0
    saw += 0.05 * rng.randn(n)
    starts.append(_project_box_sum(saw, target_sum))
    return starts

def _multi_alpha_optimize(h0, n, deadline):
    target_sum = n / 2.0
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    alphas = [30, 80, 150, 300, 600]
    top_ms = [30, 30, 20, 15, 10]
    lrs = [0.05, 0.03, 0.015, 0.008, 0.004]
    n_iters = [80000, 60000, 50000, 40000, 30000]
    for i in range(len(alphas)):
        if time.time() > deadline - 30:
            break
        h0, c5 = _momentum_optimize(best_h, n, alphas[i], n_iters[i],
                                     lrs[i], 0.9, top_ms[i], deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30

    best_h = None
    best_c5 = np.inf
    best_n = 0

    n_candidates = [100, 120, 86, 150, 200]

    for n in n_candidates:
        t_rem = deadline - time.time()
        if t_rem < 120:
            break

        include_init = (n == 100)
        starts = _make_starts(n, seed, include_init)

        for si, h0 in enumerate(starts):
            t_rem = deadline - time.time()
            if t_rem < 60:
                break
            h_opt, c5_opt = _multi_alpha_optimize(h0, n, deadline)
            if time.time() > deadline - 30:
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                continue
            for alpha in [100, 400]:
                if time.time() > deadline - 20:
                    break
                h_pol = _slsqp_polish(h_opt, n, alpha, maxiter=200, ftol=1e-14)
                h_pol = _project_box_sum(h_pol, n / 2.0)
                c5_pol = _c5_of(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n
                h_opt = h_pol

    if best_h is not None and time.time() < deadline - 40:
        t_rem = deadline - time.time()
        n_best = best_n
        target_sum = n_best / 2.0
        rng = np.random.RandomState(seed + 12345)
        for attempt in range(6):
            if time.time() > deadline - 20:
                break
            scale = 0.01 + 0.03 * rng.rand()
            h_p = _project_box_sum(best_h + scale * rng.randn(n_best), target_sum)
            h_p, c5_p = _momentum_optimize(h_p, n_best, 200, 20000, 0.01, 0.9, 15, deadline)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _project_box_sum(best_h, n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _project_box_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
