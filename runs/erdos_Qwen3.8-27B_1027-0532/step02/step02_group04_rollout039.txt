Let me also add a key new idea: **Lag-specific optimization**.

For a specific lag k, corr[k] = sum_j h[j]*(1-h[j+k]) (for k > 0).
To minimize this specific lag, we want h[j] and h[j+k] to be "aligned" - when h[j] is high, h[j+k] should also be high.

This suggests: for each active lag k, we can do a "smoothing" step that encourages h[j] ≈ h[j+k] for all j. But we need to do this for all active lags simultaneously.

This is like a multi-scale smoothing problem. We can solve it iteratively:
- For each active lag k: update h to minimize corr[k] (holding other constraints)
- This is a quadratic subproblem for each lag

The subproblem for lag k (positive):
minimize sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
subject to sum(h) = n/2, 0 ≤ h ≤ 1

This is a QP. We can solve it with a simple projected gradient method or even a few steps of conjugate gradient on the QP.

The objective is: sum_j h[j] - sum_j h[j]*h[j+k] = sum_j h[j] - h^T S_k h (where S_k is the shift matrix)
The constraint is linear: 1^T h = n/2
Bounds: 0 ≤ h ≤ 1

The unconstrained minimum (ignoring bounds) of sum_j h[j] - h^T S_k h subject to 1^T h = n/2:
Lagrangian: sum_j h[j] - h^T S_k h - λ(1^T h - n/2)
dL/dh = 1 - 2 S_k h - λ * 1 = 0
So (2 S_k) h = (1 - λ) * 1
h = (1-λ)/(2) * S_k^{-1} * 1

But S_k might not be invertible... This is getting complicated. Let me just use projected gradient for this subproblem.

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -1.5
    hi = 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_obj(h, alpha, n, mu):
    corr = compute_correlations(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -400, 400)
    sm = max_c + np.log(np.sum(np.exp(shifted))) / alpha
    s = np.sum(h) - n / 2.0
    return sm + mu * s * s


def smooth_max_grad(h, alpha, n, mu):
    nt = len(h)
    corr = compute_correlations(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -400, 400)
    exp_s = np.exp(shifted - np.max(shifted))
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(nt)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-8:
            continue
        lag = idx - (nt - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:nt - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:nt - lag]
        else:
            k = -lag
            grad[k:] += w * (1.0 - h[:nt - k])
            grad[:nt - k] -= w * h[k:]
    s = np.sum(h) - n / 2.0
    grad += 2.0 * mu * s
    return grad


def active_obj(h, active, n, mu):
    corr = compute_correlations(h)
    s = np.sum(h) - n / 2.0
    return np.sum(corr[active]) + mu * s * s


def active_grad(h, active, n, mu):
    nt = len(h)
    s = np.sum(h) - n / 2.0
    grad = 2.0 * mu * s * np.ones(nt)
    for idx in active:
        lag = idx - (nt - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:nt - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:nt - lag]
        else:
            k = -lag
            grad[k:] += (1.0 - h[:nt - k])
            grad[:nt - k] -= h[k:]
    return grad


def weighted_active_obj(h, active, weights, n, mu):
    corr = compute_correlations(h)
    s = np.sum(h) - n / 2.0
    return np.dot(weights, corr[active]) + mu * s * s


def weighted_active_grad(h, active, weights, n, mu):
    nt = len(h)
    s = np.sum(h) - n / 2.0
    grad = 2.0 * mu * s * np.ones(nt)
    for ii in range(len(active)):
        w = weights[ii]
        if w < 1e-8:
            continue
        idx = active[ii]
        lag = idx - (nt - 1)
        if lag == 0:

        else:
            k = -lag
            grad[k:] += w * (1.0 - h[:nt - k])

    return grad

                best_score = s
                best_h = h_new.copy()

    # Phase 2: Active-set focused optimization with weighted lags
    for m_active in [2, 3, 4, 5, 7, 10, 15, 20, 30]:
        if elapsed() > budget_s - 300:
            break
        for _ in range(5):
            corr = compute_correlations(best_h)
            top_idx = np.argsort(corr)[::-1][:m_active]
            top_vals = corr[top_idx]
            max_v = top_vals[0]
            w = np.exp(-10.0 * (max_v - top_vals))
            w = w / np.sum(w)
            maxiter = max(80, 50000 // n)
            res = minimize(weighted_active_obj, best_h, jac=weighted_active_grad,
                           method='L-BFGS-B', bounds=bounds,
                           args=(top_idx, w, n, MU),
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 3: Diverse restarts
    n_restarts = 250
    for i in range(n_restarts):
        if elapsed() > budget_s - 300:
            break
        strategy = i % 7
        if strategy == 0:
            scale = 0.02 + 0.08 * (i / float(n_restarts))
            pert = best_h + rng.randn(n) * scale
        elif strategy == 1:
            pert = best_h.copy()
            blen = rng.randint(4, min(30, n // 5))
            bstart = rng.randint(0, n - blen)
            pert[bstart:bstart + blen] = 1.0 - pert[bstart:bstart + blen]
        elif strategy == 2:
            pert = best_h.copy()
            n_freq = rng.randint(1, 6)
            x_idx = np.arange(n, dtype=float)
            for kk in range(n_freq):
                f = rng.randint(1, 20)
                amp = rng.uniform(0.02, 0.12)
                phase = rng.rand() * 2.0 * np.pi
                pert += amp * np.sin(2.0 * np.pi * f * x_idx / n + phase)
        elif strategy == 3:
            pert = best_h.copy()
            nb = rng.randint(2, 10)
            bs = max(1, n // nb)
            for b in range(0, n, bs):
                s1 = min(b + bs, n)
                if rng.rand() < 0.5:
                    pert[b:s1] = 1.0 - pert[b:s1]
                else:
                    pert[b:s1] += rng.randn(s1 - b) * 0.08
        elif strategy == 4:
            n_c = max(40, n // 5)
            x_f = np.linspace(0, 2, n, endpoint=False)
            x_c = np.linspace(0, 2, n_c, endpoint=False)
            coarse = np.interp(x_c, x_f, best_h)
            coarse = project_feasible(coarse, n_c / 2.0)
            coarse += rng.randn(n_c) * 0.08
            coarse = project_feasible(coarse, n_c / 2.0)
            pert = np.interp(x_f, x_c, coarse)
        elif strategy == 5:
            pert = best_h.copy()
            n_swap = rng.randint(3, 15)
            for _ in range(n_swap):
                a = rng.randint(n)
                b = rng.randint(n)
                pert[a], pert[b] = pert[b], pert[a]
        else:
            pert = (best_h >= 0.5).astype(float)
            n_flip = rng.randint(5, min(40, n // 10))
            flip_idx = rng.choice(n, n_flip, replace=False)
            pert[flip_idx] = 1.0 - pert[flip_idx]

        pert = project_feasible(pert, n / 2.0)
        alpha_pick = [30, 100, 300, 800, 2000, 5000][rng.randint(6)]
        maxiter_r = max(60, 50000 // n)
        res = minimize(smooth_max_obj, pert, jac=smooth_max_grad,
                       method='L-BFGS-B', bounds=bounds,
                       args=(alpha_pick, n, MU),
                       options={'maxiter': maxiter_r, 'ftol': 1e-16, 'gtol': 1e-10})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

        if (i + 1) % 25 == 0 and i % 35 == 0:
            corr_now = compute_correlations(best_h)
            top = np.argsort(corr_now)[::-1][:3]
            maxiter_as = max(50, 30000 // n)
            res_as = minimize(active_obj, best_h, jac=active_grad,
                              method='L-BFGS-B', bounds=bounds,
                              args=(top, n, MU),
                              options={'maxiter': maxiter_as, 'ftol': 1e-16, 'gtol': 1e-10})
            if np.all(np.isfinite(res_as.x)):
                h_new = project_feasible(res_as.x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 4: Multi-resolution refinement
    for mult in [2, 3, 4, 5]:
        n_new = n * mult
        if n_new > 3000:
            break
        if elapsed() > budget_s - 250:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_new / 2.0)
        bounds_new = [(0.0, 1.0)] * n_new

        for alpha in [20, 100, 500, 2000, 8000]:
            if elapsed() > budget_s - 120:
                break
            maxiter_n = max(50, 400000 // n_new)
            res = minimize(smooth_max_obj, h_new, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds_new,
                           args=(alpha, n_new, MU),
                           options={'maxiter': maxiter_n, 'ftol': 1e-16, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                h_proj = project_feasible(res.x, n_new / 2.0)
                s = evaluate_score(h_proj)
                if s < best_score:
                    best_score = s
                    best_h = h_proj.copy()
                    h_new = h_proj.copy()
                    bounds_new = [(0.0, 1.0)] * n_new

    # Phase 5: Final high-alpha polish
    n_fin = len(best_h)
    bounds_fin = [(0.0, 1.0)] * n_fin
    for alpha in [5000, 20000, 80000]:
        if elapsed() > budget_s - 60:
            break
        maxiter_f = max(200, 300000 // n_fin)
        res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                       method='L-BFGS-B', bounds=bounds_fin,
                       args=(alpha, n_fin, MU),
                       options={'maxiter': maxiter_f, 'ftol': 1e-16, 'gtol': 1e-10})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n_fin / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 6: Targeted worst-lag optimization
    if elapsed() < budget_s - 40:
        for _ in range(8):
            if elapsed() > budget_s - 30:
                break
            corr = compute_correlations(best_h)
            worst = int(np.argmax(corr))
            maxiter_t = max(100, 50000 // n_fin)
            res = minimize(active_obj, best_h, jac=active_grad,
                           method='L-BFGS-B', bounds=bounds_fin,
                           args=(np.array([worst]), n_fin, MU),
                           options={'maxiter': maxiter_t, 'ftol': 1e-16, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n_fin / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 7: Local search with adaptive steps
    if elapsed() < budget_s - 15:
        n_ls = len(best_h)
        steps = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
        n_iters = min(30000, int(max(0, budget_s - 20 - elapsed()) * 60))
        if n_iters > 0:
            best_max = np.max(compute_correlations(best_h))
            for _ in range(n_iters):
                i = rng.randint(n_ls)
                j = rng.randint(n_ls)
                if i == j:
                    continue
                for step in steps:
                    h_try = best_h.copy()
                    new_i = best_h[i] + step
                    new_j = best_h[j] - step
                    if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
                        continue
                    h_try[i] = new_i
                    h_try[j] = new_j
                    max_try = np.max(compute_correlations(h_try))
                    if max_try < best_max - 1e-13:
                        best_h = h_try
                        best_max = max_try
                        break

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
