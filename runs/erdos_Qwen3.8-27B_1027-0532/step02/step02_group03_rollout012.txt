from scipy.optimize import minimize, fminbounded

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_simplex(y, S):
    """Project y onto {x: 0<=x<=1, sum(x)=S} using bisection on the water level."""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding issues
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

    h = h0.copy()
    for rnd in range(max_rounds):
        max_change = 0.0
        for j in range(n):
            old_v = h[j]
            # Coarse grid
            grid = np.linspace(0.0, 1.0, grid_size)
            best_val = float('inf')
            best_v = old_v
            for v in grid:
                h[j] = v
                c5 = _c5_of(h)
                if c5 < best_val:
                    best_val = c5
                    best_v = v
            # Fine grid around best
            step = 1.0 / (grid_size - 1)
            lo_r = max(0.0, best_v - 2 * step)
            hi_r = min(1.0, best_v + 2 * step)
            if hi_r > lo_r:
                fine = np.linspace(lo_r, hi_r, refine_size)
                for v in fine:
                    h[j] = v
                    c5 = _c5_of(h)
                    if c5 < best_val:
                        best_val = c5
                        best_v = v
            h[j] = best_v
            max_change = max(max_change, abs(best_v - old_v))
        h = _fix_sum(h, target_sum)
        if max_change < 1e-9:
            break
    return h

def _de_optimize(h0, n, target_sum, npop=40, n_gen=500, F=0.8, CR=0.7, seed=42):
    """Differential Evolution for the Erdős problem."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    
    # Initialize population
    pop = np.zeros((npop, n))
    pop[0] = h0
    for i in range(1, npop):
        if i == 1:
            pop[i] = np.full(n, 0.5)
        else:
            pop[i] = _project(rng.rand(n), target_sum)
    
    # Evaluate
    fitness = np.array([_c5_of(pop[i]) for i in range(npop)])
    best_idx = np.argmin(fitness)
    global_best = pop[best_idx].copy()
    global_best_val = fitness[best_idx]
    
    for gen in range(n_gen):
        # Sort by fitness
        order = np.argsort(fitness)
        pop = pop[order]
        fitness = fitness[order]
        
        for i in range(npop):
            if i == 0:
                continue  # Keep best
            # Select three distinct individuals
            candidates = list(range(npop))
            candidates.remove(i)
            a, b, c = rng.choice(candidates, size=3, replace=False)
            
            # Mutation
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Crossover
            trial = pop[i].copy()
            cross_idx = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == cross_idx:
                    trial[j] = mutant[j]
            trial = _project(trial, target_sum)
            
            # Selection
            f_trial = _c5_of(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < global_best_val:
                    global_best_val = f_trial
                    global_best = trial.copy()
    
    return global_best, global_best_val

def _adam_subgrad(h0, n, target_sum, n_iters=1000, lr=0.1, seed=42):
    """Adam optimizer with subgradient of the max."""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_h = h.copy()
    best_val = _c5_of(h)
    
    for it in range(n_iters):
        h = _project(h, target_sum)  # Ensure feasibility
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Subgradient of max: gradient of the argmax term
        k_star = np.argmax(corr)
        dx = 2.0 / n
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m_len = n - lag
            g[lag:lag+m_len] += (1.0 - h[:m_len])
            g[:m_len] += -h[lag:lag+m_len]
        else:
            j_shift = -lag
            m_len = n - j_shift
            g[:m_len] += (1.0 - h[j_shift:])
            g[j_shift:] += -h[:m_len]
        g *= dx
        
        # Adam update
        lr_t = lr / (1.0 + 0.01 * it)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _project(h_new, target_sum)
        
        val = _c5_of(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val

def _compute_coefficients(h, j, n):

    For k=0: L_0(v) = (L_0 - old_v + old_v²) + v - v²  [the quadratic part]
    Actually let me define it as:

def _solve_1d_subproblem(L, A, center, j, n, current_v):
    """Find v that minimizes max_k L_k(v)."""
    # L_k(v) = L[k] + A[k]*(v - current_v) for k ≠ center
    # L_center(v) = (L[center] - current_v + current_v**2) + v - v**2
    C_0 = L[center] - current_v + current_v ** 2
    
    # Candidate values
    candidates = [0.0, 1.0, 0.5, current_v]
    
    # Add intersections of L_center with each line
    for k in range(len(L)):
        if k == center:
            continue
        # C_0 + v - v² = L[k] + A[k]*(v - current_v)
        # C_0 + v - v² = L[k] + A[k]*v - A[k]*current_v
        # -v² + v - A[k]*v + C_0 - L[k] + A[k]*current_v = 0
        # v² + (A[k] - 1)*v + (L[k] - A[k]*current_v - C_0) = 0
        a = 1.0
        b = A[k] - 1.0
        c = L[k] - A[k] * current_v - C_0
        disc = b*b - 4*a*c
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            v1 = (-b + sqrt_disc) / (2*a)
            v2 = (-b - sqrt_disc) / (2*a)
            if 0 <= v1 <= 1:
                candidates.append(v1)
            if 0 <= v2 <= 1:
                candidates.append(v2)
    
    # Add some grid points for safety
    candidates.extend(np.linspace(0, 1, 30).tolist())
    
    # Evaluate f(v) = max_k L_k(v) for each candidate
    best_val = float('inf')
    best_v = current_v
    for v in candidates:
        if v < 0 or v > 1:
            continue
        # Compute max_k L_k(v)
        max_val = C_0 + v - v*v  # center term
        for k in range(len(L)):
            if k == center:
                continue
            val_k = L[k] + A[k] * (v - current_v)
            if val_k > max_val:
                max_val = val_k
        if max_val < best_val:
            best_val = max_val
            best_v = v
    
    return best_v

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                h[i] -= sub
                diff += sub
    return x

def _compute_A_and_C0(h, j, n):
    n_corr = 2 * n - 1
    center = n - 1
    A = np.zeros(n_corr)
    curr = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    C0 = corr[center] - curr + curr * curr
    for m in range(n_corr):
        lag = m - center
        if lag == 0:
            continue
        A_m = 0.0
        jp = j + lag
        jm = j - lag
        if 0 <= jp < n:
            A_m += (1.0 - h[jp])
        if 0 <= jm < n:
            A_m += (-h[jm])
        A[m] = A_m
    return A, C0, corr

def _solve_1d(A, C0, corr, center, n, current_v):
    n_corr = len(corr)
    candidates = [0.0, 1.0, 0.5, current_v]
    for k in range(n_corr):
        if k == center:
            continue
        a_coeff = 1.0
        b_coeff = A[k] - 1.0
        c_coeff = corr[k] - A[k] * current_v - C0
        disc = b_coeff * b_coeff - 4.0 * a_coeff * c_coeff
        if disc >= 0:
            sq = np.sqrt(disc)
            v1 = (-b_coeff + sq) / 2.0
            v2 = (-b_coeff - sq) / 2.0
            if 0.0 <= v1 <= 1.0:
                candidates.append(v1)
            if 0.0 <= v2 <= 1.0:
                candidates.append(v2)
    grid = np.linspace(0.0, 1.0, 21)
    candidates.extend(grid.tolist())
    best_val = float('inf')
    best_v = current_v
    base = corr.copy()
    for v in candidates:
        if v < -1e-12 or v > 1.0 + 1e-12:
            continue
        v = max(0.0, min(1.0, v))
        max_val = C0 + v - v * v
        for k in range(n_corr):
            if k == center:
                continue
            val_k = base[k] + A[k] * (v - current_v)
            if val_k > max_val:
                max_val = val_k
        if max_val < best_val:
            best_val = max_val
            best_v = v
    return best_v

def _coord_descent_analytical(h0, n, target_sum, max_rounds=200, tol=1e-10):
    h = h0.copy()
    h = _fix_sum(h, target_sum)
    for rnd in range(max_rounds):
        max_change = 0.0
        for j in range(n):
            curr = h[j]
            A, C0, corr = _compute_A_and_C0(h, j, n)
            center = n - 1
            best_v = _solve_1d(A, C0, corr, center, n, curr)
            h[j] = best_v
            max_change = max(max_change, abs(best_v - curr))
        h = _fix_sum(h, target_sum)
        if max_change < tol:
            break
    return h

def _adam_subgrad(h0, n, target_sum, n_iters=3000, lr=0.2, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _c5_of(h)
    for it in range(n_iters):
        h = _fix_sum(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m_len = n - lag
            if m_len > 0:
                g[lag:lag + m_len] += (1.0 - h[:m_len])
                g[:m_len] += -h[lag:lag + m_len]
        else:
            js = -lag
            m_len = n - js
            if m_len > 0:
                g[:m_len] += (1.0 - h[js:])
                g[js:] += -h[:m_len]
        g *= dx
        lr_t = lr * 0.995 ** (it // 100)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** (it + 1))
        v_hat = v_vec / (1.0 - beta2 ** (it + 1))
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _project(h_new, target_sum)
        val = _c5_of(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

def _de_optimize(h0, n, target_sum, npop=35, n_gen=800, F=0.85, CR=0.75, seed=42):
    rng = np.random.RandomState(seed)
    pop = np.zeros((npop, n))
    pop[0] = h0.copy()
    for i in range(1, npop):
        if i == 1:
            pop[i] = np.full(n, 0.5)
        elif i == 2:
            pop[i] = _fix_sum((rng.rand(n) < 0.5).astype(float), target_sum)
        else:
            pop[i] = _project(rng.rand(n), target_sum)
    fitness = np.array([_c5_of(pop[i]) for i in range(npop)])
    order = np.argsort(fitness)
    pop = pop[order]
    fitness = fitness[order]
    global_best = pop[0].copy()
    global_best_val = fitness[0]
    for gen in range(n_gen):
        for i in range(1, npop):
            cands = [c for c in range(npop) if c != i]
            a, b, c = rng.choice(cands, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            cross_pt = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[cross_pt] = True
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            f_t = _c5_of(trial)
            if f_t < fitness[i]:
                pop[i] = trial
                fitness[i] = f_t
            if f_t < global_best_val:
                global_best_val = f_t
                global_best = trial.copy()
        order = np.argsort(fitness)
        pop = pop[order]
        fitness = fitness[order]
    return global_best, global_best_val

                noise_scale = 0.03
            h_pert = best_h + rng.normal(0, noise_scale, best_n)
            h_pert = _fix_sum(h_pert, best_n / 2.0)
            h_pert_cd = _coord_descent_analytical(h_pert, best_n, best_n / 2.0, max_rounds=150)
            update(h_pert_cd, best_n)

    # Phase 4: DE on multiple n values
    if time_ok(400):
        n_trials = []
        for d in [0, 4, -4, 8, -8, 12, -12, 16, -16, 20, -20]:
            nt = n_init + d
            if nt % 2 == 0 and 40 <= nt <= 200 and nt not in n_trials:
                n_trials.append(nt)
        for nt in n_trials:
            if not time_ok(80):
                break
            target = nt / 2.0
            if nt == n_init:
                h_seed = h_init.copy()
            else:
                h_seed = np.interp(np.linspace(0, n_init - 1, nt), np.arange(n_init), h_init)
                h_seed = _fix_sum(np.clip(h_seed, 0, 1), target)
            h_de, c5_de = _de_optimize(h_seed, nt, target, npop=30, n_gen=600,
                                        F=0.85, CR=0.7, seed=seed + nt)
            update(h_de, nt)
            h_de_cd = _coord_descent_analytical(h_de, nt, target, max_rounds=200)
            update(h_de_cd, nt)

    # Phase 5: Second round of perturbation on best
    if time_ok(150):
        for restart in range(20):
            if not time_ok(30):
                break
            h_pert = best_h + rng.normal(0, 0.08, best_n)
            h_pert = _fix_sum(h_pert, best_n / 2.0)
            h_pert_cd = _coord_descent_analytical(h_pert, best_n, best_n / 2.0, max_rounds=200)
            update(h_pert_cd, best_n)

    # Phase 6: Final polish
    if time_ok(40):
        h_final = _coord_descent_analytical(best_h, best_n, best_n / 2.0, max_rounds=500, tol=1e-12)
        update(h_final, best_n)

    if time_ok(30):
        h_adam2, _ = _adam_subgrad(best_h, best_n, best_n / 2.0, n_iters=1000, lr=0.1, seed=seed + 999)
        update(h_adam2, best_n)
        h_final2 = _coord_descent_analytical(h_adam2, best_n, best_n / 2.0, max_rounds=300)
        update(h_final2, best_n)

    if time_ok(20):
        h_final3 = _coord_descent_analytical(best_h, best_n, best_n / 2.0, max_rounds=500, tol=1e-13)
        update(h_final3, best_n)

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
