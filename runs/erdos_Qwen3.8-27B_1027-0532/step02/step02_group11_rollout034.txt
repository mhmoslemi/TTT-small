import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def to_binary_feasible(h, n):
    """Convert h to binary with exactly n/2 ones."""
    target = n // 2
    # Sort indices by h value descending, take top target
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_eval(h, i, j, n):
    """Evaluate C5 after swapping h[i] and h[j] (assuming h[i]=1, h[j]=0)."""
    # After swap: h[i]=0, h[j]=1
    # We need to recompute correlation
    # This is O(n) for a full recompute
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return evaluate_c5(h_new)

def simulated_annealing(h_init, n, t_max, t_min, alpha, max_iters, time_check):
    """Simulated annealing with 2-bit swap moves."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    curr_c5 = best_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time_check():
            break
        if it % 100 == 0:
            t = t_max * (t_min / t_max) ** (it / max_iters)
        
        # Pick random one and zero to swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        
        delta = new_c5 - curr_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
            curr_c5 = new_c5
            # Update ones/zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    # Use argsort for stability
    order = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def local_search(h, n, time_limit_fn, max_sweeps=10000):
    """Steepest descent local search with 2-bit swaps."""
    h = h.copy()
    c5 = evaluate_c5(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time_limit_fn():
            break
        best_delta = 0
        best_i = None
        best_j = None
        
        # Try all swaps
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = evaluate_c5(h)
                delta = new_c5 - c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0
        
        if best_i is None:
            break  # Local optimum
        
        # Apply best swap
        i, j = best_i, best_j
        h[i] = 0.0
        h[j] = 1.0
        c5 += best_delta
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
    
    return h, c5

def compute_effects(h, n):
    """Compute effect vectors for all positions.
    E[i, p] = effect on corr[p] when h[i] changes from 1 to 0.
    corr has length 2n-1, indexed 0 to 2n-2.
    corr[p] = sum_m h[m + (p - (n-1))] * (1-h[m])
    """

    for i in range(n):
        # Effect of h[i]: 1 -> 0
        # Term 1: m + (p-(n-1)) = i => m = i - p + (n-1)
        #   Contribution: -(1 - h[m]) where m = i - p + (n-1), valid if 0 <= m < n
        #   p from (n-1-i) to (n-1-i+n-1) = 2(n-1)-i
        #   But p must be in [0, 2n-2]
        p_lo = max(0, n - 1 - i)
        p_hi = min(ncorr - 1, n - 1 - i + n - 1)
        if p_lo <= p_hi:
            m_vals = i - np.arange(p_lo, p_hi + 1) + (n - 1)
            effects[i, p_lo:p_hi + 1] -= (1.0 - h[m_vals])
        
        # Term 2: m = i => h[i + (p-(n-1))] * 1
        #   Contribution: +h[i + p - (n-1)] where valid if 0 <= i + p - (n-1) < n
        #   p from (n-1-i) to (n-1-i+n-1) = 2(n-1)-i
        #   Same range!
        if p_lo <= p_hi:
            idx_vals = i + np.arange(p_lo, p_hi + 1) - (n - 1)
            effects[i, p_lo:p_hi + 1] += h[idx_vals]
    
    return effects
