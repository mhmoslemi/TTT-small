import time

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(50.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent(h0, n, target_sum, max_rounds=30):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=8):
    h = h0.copy()
    vals = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-10:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _multi_alpha_slsqp(h0, n, target_sum, alphas, maxiter_each, deadline):
