```python
def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    weights = np.exp(alpha * (corr - M))
    weights /= np.sum(weights)
    
    # grad[i] = dx * sum_j weights[j] * d(corr[j])/dh[i]
    grad = np.zeros(n)
    for j in range(2*n - 1):
        w = weights[j]
        if w < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            # d(corr[j])/dh[i] for i in [lag, n-1]: 1 - h[i-lag]
            # d(corr[j])/dh[i] for i in [0, n-1-lag]: -h[i+lag]
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
    
    grad *= dx
    return grad
```

```python
def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    exp_vals = np.exp(alpha * (corr - M))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        k = j - (n - 1)
        if k >= 0:
            # d(corr_k)/dh[p] = (k <= p <= n-1) * (1-h[p-k]) - (0 <= p <= n-1-k) * h[p+k]
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            m = -k
            # d(corr_k)/dh[p] = (0 <= p <= n-1-m) * (1-h[p+m]) - (m <= p <= n-1) * h[p-m]
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
    
    grad *= dx
    return grad, weights
```

```python
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, target_sum):
    """Project y onto {h: 0 <= h <= 1, sum(h) = target_sum}"""
    n = len(y)
    # Binary search for t such that sum(clip(y - t, 0, 1)) = target_sum
    lo = y.min() - target_sum  # at this t, all clipped to 1, sum = n
    hi = y.max()               # at this t, all clipped to 0, sum = 0
    # Actually: t should be in [y.min() - 1, y.max()]
    # Let me think again. We want sum(clip(y[i] - t, 0, 1)) = target_sum.
    # As t increases, the sum decreases.
    # At t = y.max(), all y[i] - t <= 0, sum = 0.
    # At t = y.min() - 1, all y[i] - t >= 1, sum = n.
    # So t is in [y.min() - 1, y.max()]
    
    lo = y.min() - 1.0  # sum = n at this point (or close)
    hi = y.max()        # sum = 0 at this point
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(y - t, 0.0, 1.0)

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    if alpha < 1e6:
        exp_vals = np.exp(alpha * (corr - M))
        log_sum = np.log(np.sum(exp_vals))
        weights = exp_vals / np.sum(exp_vals)
    else:
        # Just use the max
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
        log_sum = 0.0
    
    obj = (M + log_sum / alpha) * dx if alpha < 1e6 else M * dx
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            m = -k
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
    
    grad *= dx
    return obj, grad

def _adam_smooth(h_init, n, alpha, n_iter, lr, deadline, check_every=50):
    target_sum = n / 2.0
    m_ = np.zeros(n)  # first moment
    v_ = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    
    for t in range(1, n_iter + 1):
        if time.time() > deadline - 5:
            break
        obj, grad = _smooth_obj_and_grad(h, alpha)
        m_ = beta1 * m_ + (1 - beta1) * grad
        v_ = beta2 * v_ + (1 - beta2) * grad**2
        m_hat = m_ / (1 - beta1**t)
        v_hat = v_ / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = _project_box_sum(h_new, target_sum)
        
        if t % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    if alpha > 1e4:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
        obj = M * dx
    else:
        exp_vals = np.exp(alpha * (corr - M))
        s = np.sum(exp_vals)
        weights = exp_vals / s
        obj = (M + np.log(s) / alpha) * dx
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            m = -k
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    grad *= dx
    return obj, grad

        best_h = h.copy()
```
