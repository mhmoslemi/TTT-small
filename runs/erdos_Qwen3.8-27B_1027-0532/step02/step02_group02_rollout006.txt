from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _smoothed_obj_and_grad(h, alpha, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -200:
        lse_val = 0.0
        weights = np.zeros(len(corr))
    else:
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700, 0)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        if sum_exp < 1e-300:
            lse_val = m
            weights = np.zeros(len(corr))
        else:
            lse_val = m + np.log(sum_exp) / alpha
            weights = exp_shifted / sum_exp
    pen_val = mu * (h.sum() - n / 2.0) ** 2
    val = lse_val * dx + pen_val
    grad = np.zeros(n)
    for j in range(len(weights)):

        lag = j - (n - 1)
        grad += weights[j] * _corr_grad(h, lag)
    grad *= dx
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return val, grad

        return v

        _, g = _smoothed_obj_and_grad(x, alpha, mu, n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                                'gtol': 1e-12, 'maxls': 50})
        return res.x
    except Exception:
        return h_init.copy()

def _slsqp_refine(h_init, n, alpha, maxiter, ftol):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -200:
            return 0.0
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700, 0)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]

        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()
