    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

            sum_minus[:n-j] += w * h[j:]

def _optimal_coordinate(h, j, n):
    """Find optimal value of h[j] to minimize max corr"""
    dx = 2.0 / n
    a = h[j]
    
    # For each lag τ, compute the slope of corr[τ] w.r.t. h[j]
    # slope[τ] = (1-h[j+τ]) - h[j-τ] (with h[invalid]=0)
    # corr[τ](v) = corr[τ](a) + (v-a)*slope[τ]
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute slopes
    slopes = np.zeros(len(corr))
    for k_idx in range(len(corr)):
        tau = k_idx - (n - 1)
        s = 0.0
        if 0 <= j+tau < n:
            s += (1.0 - h[j+tau])
        if 0 <= j-tau < n:
            s -= h[j-tau]
        slopes[k_idx] = s
    
    # We want to minimize max_τ (corr[τ] + (v-a)*slopes[τ]) over v ∈ [0,1]
    # This is: minimize max_τ (A_τ + B_τ * v) where A_τ = corr[τ] - a*slopes[τ], B_τ = slopes[τ]
    
    A = corr - a * slopes
    B = slopes
    
    # The function f(v) = max_τ (A_τ + B_τ * v) is convex piecewise linear in v
    # Find minimum on [0, 1]
    
    # Check endpoints
    f0 = np.max(A)  # v=0
    f1 = np.max(A + B)  # v=1
    
    # Check where f(v) = 0 (if the minimum is at an interior point)
    # At the minimum, there exist τ1, τ2 such that A_τ1 + B_τ1*v = A_τ2 + B_τ2*v
    # and this value is ≤ all other A_τ + B_τ*v
    
    # Simple approach: try v=0, v=1, and for each pair of "active" constraints,
    # find the intersection point.
    
    # More efficient: binary search for the minimum
    # f(v) is convex, so we can use ternary search or binary search on the derivative
    
    # The "derivative" (subgradient) at v is: max(B_τ) - min(B_τ) for active τ
    # At the minimum, 0 is in the subgradient, meaning there are active constraints
    # with positive and negative slopes (or all zero slopes).
    
    # Practical approach: try v = 0, v = 1, and the "equalization point"
    # The equalization point is where the max of the "rising" lines equals the max of the "falling" lines.
    
    # Simplest: try a grid of values
    best_v = a
    best_f = f0 if f0 < f1 else f1
    for v in np.linspace(0, 1, 21):
        f = np.max(A + v * B)
        if f < best_f:
            best_f = f
            best_v = v
    
    # Refine around best_v
    lo = max(0, best_v - 0.05)
    hi = min(1, best_v + 0.05)
    for v in np.linspace(lo, hi, 21):
        f = np.max(A + v * B)
        if f < best_f:
            best_f = f
            best_v = v
    
    return best_v

def _project_sum(y, S):
    n = len(y)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):

        if s > S:
            lo = mid
        else:
            hi = mid

                diff -= add

def _project_sum(y, S):
    n = len(y)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0

        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:

**Multiplicative Weights Algorithm (MWA):**
1. Initialize weights w_k = 1/(2n-1) for all k
2. For t = 1, 2, ...:
   a. Solve min_h ∑_k w_k * corr_k(h) s.t. h ∈ [0,1], sum(h) = n/2
   b. Let h_t be the solution
   c. Update: w_k ← w_k * exp(-η * corr_k(h_t) / (2n-1)) (or similar)
3. Return the average of h_t

The inner problem (step a) is a QP: minimize ∑_k w_k * h^T A_k h + b_k^T h s.t. box and sum constraints.

This can be solved with standard QP solvers (like scipy.optimize.minimize with method='SLSQP' or a dedicated QP solver).

= h^T 1 - h^T A_k h
