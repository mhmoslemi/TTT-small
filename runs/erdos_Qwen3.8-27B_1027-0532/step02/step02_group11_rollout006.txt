import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_s = np.max(shifted)
    return (m + max_s + (1.0/eps) * np.log(np.sum(np.exp(shifted - max_s)))) * dx

def smooth_max_grad(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - nm1
        # term1: (1-h)[j-l] for j where 0 <= j-l < n
        if l >= 0:
            j_lo, j_hi = l, n
        else:
            j_lo, j_hi = 0, n + l
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi)
            grad[idx] += w * (1.0 - h[idx - l])
        # term2: -h[j+l] for j where 0 <= j+l < n
        if l >= 0:
            j_lo2, j_hi2 = 0, n - l
        else:
            j_lo2, j_hi2 = -l, n
        if j_lo2 < j_hi2:
            idx2 = np.arange(j_lo2, j_hi2)
            grad[idx2] += w * (-h[idx2 + l])
    
    grad *= dx
    return grad

def slsqp_step(h_start, n, eps, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_val(x, eps, n)
    
    def grad(x):
        return smooth_max_grad(x, eps, n)
    
    def constr(x):
        return np.sum(x) - n / 2.0
    
    try:
        res = minimize(obj, h_start.copy(), jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constr}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    except Exception:
        return h_start, evaluate_c5(h_start)

def l_bfgsb_step(h_start, n, eps, penalty, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_val(x, eps, n) + penalty * (np.sum(x) - n/2.0)**2
    
    def grad(x):
        g = smooth_max_grad(x, eps, n)
        g += 2 * penalty * (np.sum(x) - n/2.0)
        return g
    
    res = minimize(obj, h_start.copy(), jac=grad, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    h_new = project_feasible(res.x, n)
    return h_new, evaluate_c5(h_new)
