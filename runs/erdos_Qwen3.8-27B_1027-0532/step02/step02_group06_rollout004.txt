        if it % 100 == 0:
            step = max(step * 0.97, 0.1 / alpha)

    return best_h, best_score


def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        c = compute_correlations(hv)
        mm = np.max(c)
        shifted = alpha * (c - mm)
        shifted = np.clip(shifted, -500, 500)
        return mm + np.log(np.sum(np.exp(shifted))) / alpha

    def con_fun(hv):
        return np.sum(hv) - target_sum

    def con_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def local_search_binary_round(h, target_sum, n_trials=200, t0=None, time_limit=30.0):
    n = len(h)
    best = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.RandomState(42)

    for _ in range(n_trials):
        if t0 is not None and time.time() - t0 > t0 + time_limit:
            break
        mask = rng.rand(n) < 0.3
        h_try = h.copy()
        h_try[mask] = np.where(h_try[mask] > 0.5, 1.0, 0.0)
        h_try = project_to_feasible(h_try, target_sum)
        score = evaluate_score(h_try)
        if score < best_score:
            best_score = score
            best = h_try.copy()
    return best, best_score


def coordinate_polish(h, target_sum, n_rounds=200, t0=None, time_limit=30.0):
    n = len(h)
    h = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.RandomState(99)

    for rd in range(n_rounds):
        if t0 is not None and time.time() - t0 > t0 + time_limit:
            break
        improved = False
        for _ in range(30):
            i = rng.randint(n)
            for delta in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h_try = project_to_feasible(h_try, target_sum)
                score = evaluate_score(h_try)
                if score < best_score - 1e-13:
                    best_score = score
                    h = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_to_feasible(init, n / 2.0)
    target_sum = n / 2.0

    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: Smoothed PGD with increasing alpha
    alphas_pgd = [5, 10, 30, 80, 200, 500]
    for alpha in alphas_pgd:
        if remaining() < 70:
            break
        t_allow = min(remaining() * 0.1, 100)
        if t_allow < 10:
            break
        h_new, score = pgd_with_backtracking(best_h, alpha, target_sum,
                                             3000, t0, t_allow)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    # Phase 2: SLSQP with multiple alpha values
    slsqp_alphas = [10, 30, 100, 300, 1000, 3000]
    for alpha in slsqp_alphas:
        if remaining() < 15:
            break
        h_ref = slsqp_refine(best_h, alpha, maxiter=200)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    # Phase 3: Random restarts with PGD + SLSQP
    rng = np.random.RandomState(seed)
    for r in range(30):
        if remaining() < 50:
            break
        pert = 0.01 + 0.06 * rng.rand()
        start = best_h + rng.randn(n) * pert / np.sqrt(n)
        start = project_to_feasible(start, target_sum)
        alpha = 30 + int(300 * rng.rand())
        t_allow = min(remaining() * 0.04, 40)
        if t_allow < 5:
            break
        h_new, score = pgd_with_backtracking(start, alpha, target_sum,
                                             2000, t0, t_allow)
        if remaining() > 15:
            h_ref = slsqp_refine(h_new, 200, maxiter=100)
            if h_ref is not None:
                s2 = evaluate_score(h_ref)
                if s2 < score:
                    score = s2
                    h_new = h_ref
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    # Phase 4: Coordinate polish and binary rounding
    if remaining() > 30:
        h_cp, score_cp = coordinate_polish(best_h, target_sum,
                                           n_rounds=200, t0=t0, time_limit=20)
        if score_cp < best_score:
            best_score = score_cp
            best_h = h_cp

    if remaining() > 25:
        h_br, score_br = local_search_binary_round(best_h, target_sum,
                                                   n_trials=100, t0=t0, time_limit=15)
        if score_br < best_score:
            best_score = score_br
            best_h = h_br
            if remaining() > 15:
                h_ref = slsqp_refine(best_h, 300, maxiter=200)
                if h_ref is not None:
                    s2 = evaluate_score(h_ref)
                    if s2 < best_score:
                        best_score = s2
                        best_h = h_ref

    # Phase 5: Grid refinement
    for factor in [2, 3]:
        if remaining() < 60:
            break
        n_try = n * factor
        if n_try > 2500:
            continue
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_feasible(h_interp, n_try / 2.0)
        ts_new = n_try / 2.0
        t_allow = min(remaining() * 0.08, 60)
        if t_allow < 10:
            break
        h_new, score = pgd_with_backtracking(h_interp, 100, ts_new,
                                             2000, t0, t_allow)
        if remaining() > 15:
            h_ref = slsqp_refine(h_new, 300, maxiter=150)
            if h_ref is not None:
                s2 = evaluate_score(h_ref)
                if s2 < score:
                    score = s2
                    h_new = h_ref
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            n = n_try
            target_sum = ts_new

    # Phase 6: Final intensive SLSQP
    if remaining() > 15:
        for alpha in [500, 2000, 5000]:
            if remaining() < 10:
                break
            h_ref = slsqp_refine(best_h, alpha, maxiter=300)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    # Phase 7: Final PGD with very high alpha
    if remaining() > 15:
        h_new, score = pgd_with_backtracking(best_h, 2000, target_sum,
                                             5000, t0, remaining() - 10)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
