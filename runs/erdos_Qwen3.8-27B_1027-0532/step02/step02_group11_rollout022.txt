```python
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps):
    dx = 2.0 / len(h)
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

def slsqp_refine(h_start, n, eps, maxiter):

    def obj(x):
        return smooth_max_val(x, eps)
    def jac(x):

                   bounds=bounds,

                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def block_local_search(h_start, n, block_sizes, max_iters, time_check):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    n_iter = 0
    while n_iter < max_iters:
        if time_check():
            break
        n_iter += 1

        bs = np.random.choice(block_sizes)
        i = np.random.randint(0, n - bs)

        delta = np.random.uniform(-0.1, 0.1, bs)
        h_try = h.copy()
        h_try[i:i+bs] = np.clip(h_try[i:i+bs] + delta, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5 = evaluate_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()
            h = h_try
    return best_h, best_c5

def simulated_annealing(h_start, n, t_max, t_min, cooling, max_iters, time_check):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps):
    dx = 2.0 / len(h)
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        # For this shift l: contribution to gradient
        # d/dh[j] of sum_i h[i+l]*(1-h[i]) where i+l and i are valid
        # = sum over valid pairs
        # For h[j]: appears as h[j+l]*(1-h[j]) when i=j, and as h[i+l]*(1-h[j]) when i+l=j i.e. i=j-l
        # Also (1-h[j]) factor: -h[j+l] when i=j, and -h[j] when i+l=j (i.e., the h[i+l] term)
        # Let me be more careful:
        # corr_k = sum_{i: 0<=i<n, 0<=i+l<n} h[i+l] * (1-h[i])
        # d/dh[j]: 
        #   If 0<=j-l<n: h[j] appears as (1-h[j-l+h-l-l])... 
        #   Let me just do it directly.
        # corr_k = sum_i h[i+l]*(1-h[i]) for valid i
        # d/dh[j] corr_k = sum_i [d(h[i+l])/dh[j] * (1-h[i]) + h[i+l] * d(1-h[i])/dh[j]]
        # = sum_i [delta(i+l,j) * (1-h[i]) - delta(i,j) * h[i+l]]
        # = (1-h[j-l]) * [0<=j-l<n and 0<=j-l+l<n] - h[j+l] * [0<=j<n and 0<=j+l<n]
        # = (1-h[j-l]) * [0<=j-l<n and 0<=j<n] - h[j+l] * [0<=j<n and 0<=j+l<n]
        # Since 0<=j<n always:
        # = (1-h[j-l]) * [0<=j-l<n] - h[j+l] * [0<=j+l<n]
        jl = j - l
        jp = j + l
        term = 0.0
        if 0 <= jl < n:
            term += (1.0 - h[jl])
        if 0 <= jp < n:
            term += (-h[jp])
        grad[j] += w * term
    grad *= dx
    return grad

def slsqp_refine(h_start, n, eps, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val(x, eps)
    def jac(x):
        return smooth_max_grad(x, eps)
    def cons(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h_start.copy(), jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h_start, evaluate_c5(h_start)

def nm_local(h_start, n, maxiter, time_check):
    # Nelder-Mead with penalty for constraint violation
    def obj(x):
        c5 = evaluate_c5(x)
        penalty = 100.0 * (np.sum(x) - n / 2.0) ** 2
        return c5 + penalty
    x0 = h_start.copy()
    # Scale initial simplex
    step = 0.01
    from scipy.optimize import minimize
    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def targeted_coord_descent(h_start, n, time_check, max_sweeps=100):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    # Find critical shifts
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    critical_indices = np.where(corr > 0.95 * np.max(corr))[0]
    
    deltas = [0.001, 0.0005, 0.0002, -0.001, -0.0005, -0.0002]
    
    for sweep in range(max_sweeps):
        if time_check():
            break
        improved = False
        # Focus on coordinates that affect critical shifts
        for i in range(n):
            for d in deltas:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5 = evaluate_c5(h_try)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def multi_scale_local_search(h_start, n, time_check):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Multi-scale: try perturbations at different scales
    scales = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for scale in scales:
        if time_check():
            break
        for it in range(2000):
            if time_check():
                break
            # Random perturbation
            n_pert = np.random.randint(1, 6)
            indices = np.random.choice(n, n_pert, replace=False)
            h_try = h.copy()
            h_try[indices] = np.clip(h_try[indices] + np.random.uniform(-scale, scale, n_pert), 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c5 = evaluate_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
                h = h_try
    return best_h, best_c5

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def elapsed():
        return time.time() - start_time
    
    def time_frac(frac):
        return elapsed() > budget_s * frac

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    best_h = project_feasible(h0, n)
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: SLSQP with increasing epsilon (smooth to sharp)
    eps_schedule = [50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0,
                    10000.0, 20000.0, 50000.0, 100000.0, 200000.0, 500000.0]
    for eps_val in eps_schedule:
        if time_frac(0.25):
            break
        h_new, c5_new = slsqp_refine(best_h, n, eps_val, 1000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Multi-scale local search
    best_h, best_c5 = multi_scale_local_search(best_h, n, lambda: time_frac(0.40))
    
    # Phase 3: SLSQP with very high epsilon
    for eps_val in [500000.0, 1000000.0, 2000000.0, 5000000.0]:
        if time_frac(0.55):
            break
        h_new, c5_new = slsqp_refine(best_h, n, eps_val, 1500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 4: Nelder-Mead
    if not time_frac(0.65):
        h_new, c5_new = nm_local(best_h, n, 5000, lambda: time_frac(0.65))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 5: More SLSQP
    for eps_val in [1000000.0, 5000000.0, 10000000.0]:
        if time_frac(0.75):
            break
        h_new, c5_new = slsqp_refine(best_h, n, eps_val, 2000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 6: Fine coordinate descent
    best_h, best_c5 = targeted_coord_descent(best_h, n, lambda: time_frac(0.88))
    
    # Phase 7: Final SLSQP
    for eps_val in [5000000.0, 20000000.0]:

    best_h = project_feasible(best_h, n)
```

```python
def subgradient_polish(h_start, n, max_iters, time_check):
    h = h_start.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    alpha0 = 0.01
    for t in range(1, max_iters + 1):
        if time_check():
            break
        alpha = alpha0 / (t ** 0.5)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        k_star = np.argmax(corr)
        l = k_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            jl = j - l
            jp = j + l
            if 0 <= jl < n:
                grad[j] += (1.0 - h[jl])
            if 0 <= jp < n:
                grad[j] += (-h[jp])
        grad *= dx
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
```

```python
def smooth_max_grad_fast(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    # Numerical stability
    shifted -= np.max(shifted)  # This is already 0 since shifted ≤ 0
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        gl = np.zeros(n)
        if l >= 0:
            gl[l:n] += (1.0 - h[0:n-l])
            gl[0:n-l] += -h[l:n]
        else:
            lp = -l
            gl[0:n-lp] += (1.0 - h[lp:n])
            gl[lp:n] += -h[0:n-lp]
        grad += w * gl
    grad *= dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad_fast(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        gl = np.zeros(n)
        if l >= 0:
            gl[l:n] += (1.0 - h[0:n - l])
            gl[0:n - l] += -h[l:n]
        else:
            lp = -l
            gl[0:n - lp] += (1.0 - h[lp:n])
            gl[lp:n] += -h[0:n - lp]
        grad += w * gl
    grad *= dx
    return grad

def slsqp_refine(h_start, n, eps, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val(x, eps)
    def jac(x):
        return smooth_max_grad_fast(x, eps)
    def cons(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h_start.copy(), jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h_start, evaluate_c5(h_start)

def multi_scale_local_search(h_start, n, time_check, max_iters=30000):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    scales = [0.05, 0.03, 0.01, 0.005, 0.002, 0.001]
    iters_per_scale = max_iters // len(scales)
    for scale in scales:
        for _ in range(iters_per_scale):
            if time_check():
                return best_h, best_c5
            n_pert = np.random.randint(1, 5)
            indices = np.random.choice(n, n_pert, replace=False)
            h_try = h.copy()
            h_try[indices] = np.clip(h_try[indices] + np.random.uniform(-scale, scale, n_pert), 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c5 = evaluate_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
                h = h_try
    return best_h, best_c5

def restart_loop(h_start, n, time_check, n_restarts=10, perturb_scale=0.02):
    best_h = h_start.copy()
    best_c5 = evaluate_c5(h_start)
    for r in range(n_restarts):
        if time_check():
            break
        scale = perturb_scale * (0.5 + 0.5 * r / n_restarts)
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        for eps_val in [20000.0, 100000.0, 500000.0]:

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_pert = best_h
    return best_h, best_c5

def fine_coord_descent(h_start, n, time_check, max_sweeps=300):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, -0.001, -0.0005, -0.0002, -0.0001]
    for sweep in range(max_sweeps):
        if time_check():
            break
        improved = False
        for i in range(n):
            for d in deltas:
                h_try = h.copy()
                val = h[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_try[i] = val
                h_try = project_feasible(h_try, n)
                c5 = evaluate_c5(h_try)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out
```
