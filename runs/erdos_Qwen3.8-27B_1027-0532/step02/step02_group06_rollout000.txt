def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        la = -lag
        grad[la:] += (1.0 - h[:n - la])
        grad[:n - la] -= h[la:]
    return grad


def softmax_gradient_full(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:

def smooth_refine_slsqp(h, alpha, maxiter=100):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        return np.sum(np.exp(shifted)) / alpha + m

    def obj_grad(hv):
        return softmax_gradient_full(hv, alpha)

    def con_fun(hv):
        return np.sum(hv) - target_sum

    def con_jac(hv):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        pass
    return None

            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
