import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def project_feasible(h_new, target):
    n = len(h_new)
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h_new - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    h_proj = np.clip(h_new - mid, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        h_proj = h_proj + diff / n
        np.clip(h_proj, 0.0, 1.0, out=h_proj)
    return h_proj


def smooth_max(x, alpha):
    m = np.max(x)
    if alpha <= 0:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    ex = np.exp(alpha * (corr - m))
    total = np.sum(ex)
    w = ex / total
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wk * (1.0 - h[j_lo + s:j_hi + s + 1])
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, n - 1 + s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wk * h[j_lo2 - s:j_hi2 - s + 1]
    return grad


def cons_sum(x):
    return np.sum(x) - len(x) / 2.0


def cons_sum_jac(x):
    return np.ones(len(x))


def slsqp_opt(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, jac=smooth_grad, args=(alpha,),
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons_sum, 'jac': cons_sum_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    for _ in range(50):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0, rng=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        if rng is not None and sweep > 0:
            order = rng.permutation(n)
        else:
            order = range(n)
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
            h = project_sum(h, n / 2.0)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def projected_gradient_method(h0, alpha, n_iters, time_budget):
    n = len(h0)
    h = h0.copy()
    t0 = time.time()
    best_h = h.copy()
    best_score = evaluate_score(h)
    lr = 0.05
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        grad = smooth_grad(h, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        accepted = False
        for bt in range(15):
            lr_try = lr * (0.5 ** bt)
            h_try = project_feasible(h - lr_try * grad, n / 2.0)
            score_try = smooth_max(compute_correlations(h_try), alpha)
            score_cur = smooth_max(compute_correlations(h), alpha)
            if score_try <= score_cur - 1e-4 * lr_try * grad_norm * grad_norm:
                h = h_try
                lr = min(lr * 1.3, 0.5)
                accepted = True
                break
        if not accepted:
            lr = max(lr * 0.5, 1e-6)
            h = project_feasible(h - lr * grad, n / 2.0)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    best_h = project_sum(best_h, n / 2.0)
    return best_h, evaluate_score(best_h)


def gauss_southwell_cd(h, n_steps, time_budget, rng=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for step in range(n_steps):
        if time.time() - t0 > time_budget:
            break
        best_j = -1
        best_new_max = float(np.max(corr))
        best_t_opt = None
        best_slopes = None
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            if abs(t_opt - h[j]) < 1e-14:
                continue
            new_corr = corr + (t_opt - h[j]) * slopes
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_j = j
                best_t_opt = t_opt
                best_slopes = slopes
        if best_j < 0:
            break
        h[best_j] = best_t_opt
        corr = corr + (best_t_opt - h[best_j] + h[best_j] - h[best_j]) * best_slopes
        corr = compute_correlations(h)
        if step % 50 == 49:
            h = project_sum(h, n / 2.0)
            corr = compute_correlations(h)
    h = project_sum(h, n / 2.0)
    return h


def random_binary_init(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    if target == int(target):
        ones = rng.choice(n, size=int(target), replace=False)
        h[ones] = 1.0
    else:
        int_part = int(target)
        frac_part = target - int_part
        ones = rng.choice(n, size=int_part, replace=False)
        h[ones] = 1.0
        remaining = np.setdiff1d(np.arange(n), ones)
        h[remaining[0]] = frac_part
    return h
