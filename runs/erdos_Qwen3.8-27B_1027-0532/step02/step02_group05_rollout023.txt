    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    ex = np.exp(alpha * (corr - mx))
    p = ex / np.sum(ex)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for kp in range(N):
        if kp == 0:
            w = p[N-1]
        else:
            w = p[N-1+kp] + p[N-1-kp]
        if w < 1e-14:
            continue
        if kp > 0:
            grad[kp:] += w * one_minus_h[:N-kp]
            grad[:N-kp] -= w * h[kp:]
        else:
            grad += w * one_minus_h
            grad -= w * h
    return grad

def _obj_fun(x, alpha):
    return smooth_max_val(compute_correlations(x), alpha)

def _obj_jac(x, alpha):
    return grad_smooth(x, alpha)

def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        _obj_fun, h0, method='SLSQP', bounds=bounds,
        jac=_obj_jac, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x

def fista_optimize(h0, alpha, target_sum, t0, time_limit):
    N = len(h0)
    x = h0.copy().astype(float)
    z = h0.copy().astype(float)
    t_k = 1.0
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    lr = 1.0 / (alpha * N)
    
    for it in range(100000):
        if time.time() - t0 > time_limit:
            break
        g = grad_smooth(z, alpha)
        # Backtracking
        step = lr
        fg = smooth_max_val(compute_correlations(z), alpha)
        for _ in range(30):
            x_trial = project_feasible(z - step * g, target_sum)
            f_trial = smooth_max_val(compute_correlations(x_trial), alpha)
            if f_trial <= fg + 0.5 * step * step * (np.dot(g, g)):
                break
            step *= 0.5
        x_new = project_feasible(z - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z_new = project_feasible(z_new, target_sum)
        x = x_new
        z = z_new
        t_k = t_new
        lr = step
        
        score = evaluate_score(x)
        if score < best_score:
            best_score = score
            best_x = x.copy()
    return best_x, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def grad_smooth(h, alpha):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    ex = np.exp(alpha * (corr - mx))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            n_valid = N - k
            grad[k:] += w * (1.0 - h[:n_valid])
            grad[:n_valid] -= w * h[k:]
        else:
            kp = -k
            n_valid = N - kp
            grad[:n_valid] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n_valid]
    return grad


def obj_fun(x, alpha):
    return smooth_max_val(compute_correlations(x), alpha)


def obj_jac(x, alpha):
    return grad_smooth(x, alpha)


def cons_fun(x):
    return np.sum(x) - len(x) / 2.0


def cons_jac(x):
    return np.ones(len(x))


def slsqp_optimize(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        obj_fun, h0, method='SLSQP', bounds=bounds,
        jac=obj_jac, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x
