        if x_new is not None:
            h_new = euclidean_project(np.clip(x_new, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 8: Final polish - very high alpha
    for alpha in [8000, 15000, 30000]:
        if time_left() < 20:
            break
        x_new = run_lbfgsb(best_h, alpha, 50000.0, 300, bounds_best)
        if x_new is not None:
            h_new = euclidean_project(np.clip(x_new, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 9: Final subgradient + L-BFGS hybrid
    if time_left() > 30:
        h_hybrid = hybrid_refine(best_h, 15, 100, 0.5, 2000, 20000.0, bounds_best, n_best / 2.0)
        h_hybrid = euclidean_project(np.clip(h_hybrid, 0.0, 1.0), n_best / 2.0)
        s = evaluate_score(h_hybrid)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_hybrid.copy()

    # Phase 10: Adjacent resolution check
    for n_adj in [n_best + 7, n_best - 7, n_best + 23, n_best - 23]:
        if time_left() < 15:
            break
        if n_adj < 100 or n_adj > 2500:
            continue
        h_adj = interpolate_h(best_h, n_best, n_adj)
        h_adj = euclidean_project(h_adj, n_adj / 2.0)
        bounds_adj = [(0.0, 1.0)] * n_adj
        x_new = run_lbfgsb(h_adj, 2000, 15000.0, 200, bounds_adj)
        if x_new is not None:
            h_new = euclidean_project(np.clip(x_new, 0.0, 1.0), n_adj / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    best_h = euclidean_project(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
