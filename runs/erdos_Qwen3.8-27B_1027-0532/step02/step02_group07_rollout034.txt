import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / float(np.sum(free))
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + shift:hi2 + shift]
    return grad

    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()

def slsqp_constraint(x):
    return np.sum(x) - len(x) / 2.0


def slsqp_constraint_jac(x):
    return np.ones(len(x))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_obj_grad, h0, method='SLSQP', bounds=bounds,
        args=(alpha,), jac=True,
        constraints=[{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_constraint_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:

        if feasible:

            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0

def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0, random_order=False, rng=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    order = np.arange(n)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        max_change = 0.0
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h

def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / float(np.sum(free))
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + shift:hi2 + shift]
    return grad


def slsqp_obj_grad(x, alpha):
    corr = compute_correlations(x)
    val = smooth_max_val(corr, alpha)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * x[lo2 + shift:hi2 + shift]
    return val, grad


def slsqp_constraint(x):
    return np.sum(x) - len(x) / 2.0


def slsqp_constraint_jac(x):
    return np.ones(len(x))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_obj_grad, h0, method='SLSQP', bounds=bounds,
        args=(alpha,), jac=True,
        constraints=[{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_constraint_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    for _ in range(44):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0, random_order=False, rng=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    order = np.arange(n)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        max_change = 0.0
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def gradient_exploration(h, alpha, lr0, momentum, n_steps, time_budget):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = lr0
    t0 = time.time()
    curr_val = smooth_max_val(compute_correlations(h), alpha)
    for step in range(n_steps):
        if time.time() - t0 > time_budget:
            break
        g = compute_smooth_grad(h, alpha)
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            v = momentum * v - lr * g / g_norm
        else:
            break
        step_size = np.linalg.norm(v)
        if step_size > 0.5:
            v = v * 0.5 / step_size
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, n / 2.0)
        new_val = smooth_max_val(compute_correlations(h_new), alpha)
        if new_val < curr_val + 1e-12:
            h = h_new
            curr_val = new_val
            lr = min(lr * 1.02, 0.5)
        else:
            lr = max(lr * 0.4, 1e-6)
            v = v * 0.3
            h_new = h + v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum(h_new, n / 2.0)
            h = h_new
            curr_val = smooth_max_val(compute_correlations(h_new), alpha)
    return h


def make_block_init(n, rng):
    h = np.zeros(n)
    start = rng.randint(0, n)
    h[start:start + n // 2] = 1.0
    return project_sum(h, n / 2.0)


def make_sine_init(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    freq = rng.choice([1, 2, 3, 4, 5])
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.5 * np.sin(np.pi * freq * x + phase)
    return project_sum(h, n / 2.0)


def make_gaussian_init(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    center = rng.uniform(0.5, 1.5)
    width = rng.uniform(0.3, 1.0)
    h = 0.5 + 0.5 * np.exp(-0.5 * ((x - center) / width) ** 2)
    return project_sum(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_sum(init, n / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h = coordinate_descent(best_h, n_sweeps=400, tol=1e-13, time_budget=min(60, rem - 40))
        s = evaluate_score(h)
        if s < best_score:
            best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 100:
        for alpha in [30, 80, 200, 500]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            h = gradient_exploration(best_h, alpha, 0.02, 0.85, 3000, min(15, rem - 15))
            h = project_sum(h, len(h) / 2.0)
            s = evaluate_score(h)
            if s < best_score:
                best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 70:
        h = coordinate_descent(best_h, n_sweeps=500, tol=1e-14, time_budget=min(50, rem - 30))
        s = evaluate_score(h)
        if s < best_score:
            best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 120:
        bounds = [(0.0, 1.0)] * len(best_h)
        for alpha in [200, 1000, 5000, 20000]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            maxiter = max(100, 300000 // len(best_h))
            res = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res is not None:
                h_new = project_sum(np.clip(res, 0.0, 1.0), len(best_h) / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score, best_h = s, h_new

    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h = coordinate_descent(best_h, n_sweeps=500, tol=1e-15, time_budget=min(45, rem - 25))
        s = evaluate_score(h)
        if s < best_score:
            best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 80:
        for trial in range(4):
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            h = coordinate_descent(best_h, n_sweeps=300, tol=1e-14,
                                   time_budget=min(18, rem - 15),
                                   random_order=True, rng=rng)
            s = evaluate_score(h)
            if s < best_score:
                best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    n_cur = len(best_h)
    n_perturb = 25
    for i in range(n_perturb):
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        frac = i / max(n_perturb - 1, 1)
        scale = 0.01 + 0.06 * frac
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_sum(h_pert, n_cur / 2.0)
        tb = min(4.0, (rem - 15) / max(n_perturb - i, 1))
        h = coordinate_descent(h_pert, n_sweeps=150, tol=1e-13, time_budget=tb)
        s = evaluate_score(h)
        if s < best_score:
            best_score, best_h = s, h
            n_cur = len(best_h)

    rem = budget_s - (time.time() - t0)
    init_types = [make_block_init, make_sine_init, make_gaussian_init]
    for _ in range(6):
        rem = budget_s - (time.time() - t0)
        if rem < 30:
            break
        init_fn = rng.choice(len(init_types))
        h_alt = init_types[init_fn](n_cur, rng)
        tb = min(8.0, (rem - 20) / 3.0)
        h = coordinate_descent(h_alt, n_sweeps=200, tol=1e-13, time_budget=tb)
        s = evaluate_score(h)
        if s < best_score:
            best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 250:
        n_new = 2 * len(best_h)
        if n_new <= 4000:
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_sum(h_interp, n_new / 2.0)
            h = coordinate_descent(h_interp, n_sweeps=400, tol=1e-14,
                                   time_budget=min(70, rem - 120))
            s = evaluate_score(h)
            rem = budget_s - (time.time() - t0)
            if rem > 60:
                h = gradient_exploration(h, 300, 0.01, 0.85, 2000, min(30, rem - 40))
                h = project_sum(h, len(h) / 2.0)
                s = evaluate_score(h)
            rem = budget_s - (time.time() - t0)
            if rem > 40:
                h = coordinate_descent(h, n_sweeps=300, tol=1e-15, time_budget=min(35, rem - 20))
                s = evaluate_score(h)
            if s < best_score:
                best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h = coordinate_descent(best_h, n_sweeps=600, tol=1e-15, time_budget=min(40, rem - 10))
        s = evaluate_score(h)
        if s < best_score:
            best_score, best_h = s, h

    rem = budget_s - (time.time() - t0)
    if rem > 30 and len(best_h) <= 600:
        bounds = [(0.0, 1.0)] * len(best_h)
        for alpha in [50000, 200000]:
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            res = slsqp_optimize(best_h, alpha, 400, bounds)
            if res is not None:
                h_new = project_sum(np.clip(res, 0.0, 1.0), len(best_h) / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score, best_h = s, h_new

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
