def paired_coord_step(h, i, j, n):
    """Find optimal d for changing h[i]+d, h[j]-d"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    
    # Compute slopes for each correlation
    n_corr = len(corr)
    slopes = np.zeros(n_corr)
    for k in range(n_corr):
        l = k - (n - 1)
        # grad at i for shift l
        gi = 0.0
        iml = i - l
        if 0 <= iml < n:
            gi += (1.0 - h[iml])
        il = i + l
        if 0 <= il < n:
            gi += (-h[il])
        # grad at j for shift l
        gj = 0.0
        jml = j - l
        if 0 <= jml < n:
            gj += (1.0 - h[jml])
        jl = j + l
        if 0 <= jl < n:
            gj += (-h[jl])
        slopes[k] = (gi - gj) * dx
    
    # Feasible range for d
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi:
        return 0.0
    
    # Binary search for minimum of g(d) = max_k (corr[k] + d * slopes[k])
    # g is convex, so binary search on subgradient
    lo, hi = d_lo, d_hi
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = corr + mid * slopes
        k_star = np.argmax(vals)
        if slopes[k_star] < 0:
            hi = mid  # derivative negative, minimum is to the right
        else:
            lo = mid  # derivative positive (or zero), minimum is to the left

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    return (m + eps * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    n_corr = len(corr)
    grad = np.zeros(n)
    for k in range(n_corr):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        l = k - (n - 1)
        if l >= 0:
            grad[l:] += w * (1 - x[:n-l])
            grad[:n-l] += w * (-x[l:])
        else:
            ll = -l
            grad[:n-ll] += w * (1 - x[ll:])
            grad[ll:] += w * (-x[:n-ll])
    return grad

def slsqp_optimize(h_start, n, eps, maxiter=500):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, eps, n)
    def grad(x):
        return smooth_max_grad(x, eps, n)
    def constr(x):
        return np.sum(x) - n / 2.0

                   bounds=bounds, constraints=[{'type': 'eq', 'fun': constr}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})

def find_optimal_d(h, i, j, n):

Actually, here's an even simpler observation: the optimal d for a pair (i,j) is the one that equalizes the top two correlations (if possible within the feasible range). So:
1. Find the current maximum correlation k*
2. The slope s_{k*} tells us which direction to go
3. If s_{k*} < 0, increase d (go right); if > 0, decrease d (go left)
4. Keep going until two correlations are tied at the maximum

This is essentially what the binary search does, so let me just stick with it.

Let me simplify my approach and focus on what's most likely to give a quick improvement:

1. Start with initial_h_values (n=170)
2. SLSQP with eps=10, 5, 3 (very sharp)
3. Paired CD with random pairs, 100000 iterations
4. SLSQP again
5. Try n=340
6. Return best

Let me also add a key trick: after SLSQP, the solution might not be exactly feasible. The projection step handles this, but I should make sure the projection doesn't significantly degrade the solution.

One more thought: the problem has a symmetry. If h is a solution, then 1-h (on [0,2]) is not necessarily a solution because the constraint is ∫h = 1, not ∫h = 1 (which would give ∫(1-h) = 1 as well since ∫1 dx = 2 and ∫h = 1, so ∫(1-h) = 1). Wait, ∫_0^2 (1-h) dx = 2 - 1 = 1. So 1-h is also feasible!

And C_5(1-h) = max_k ∫ (1-h)(x) * h(x+k) dx = max_k ∫ h(x+k)(1-h(x)) dx = C_5(h) (by substitution).

So h and 1-h have the same C_5. This means the feasible set has a symmetry, and the objective is symmetric. This doesn't directly help but is good to know.

When h changes at positions i and j, only the rows of G that are affected change. But since every row is affected (every correlation involves most coordinates), this doesn't help much. Let me just recompute G every few iterations.

Actually, here's a better idea: compute G once at the start of a batch, do several CD steps using the same G (which introduces small errors), then recompute G. This amortizes the cost.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    return (m + eps * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    n_corr = len(corr)
    grad = np.zeros(n)
    for k in range(n_corr):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        l = k - (n - 1)
        if l >= 0:
            grad[l:] += w * (1.0 - x[:n - l])
            grad[:n - l] += w * (-x[l:])
        else:
            ll = -l
            grad[:n - ll] += w * (1.0 - x[ll:])
            grad[ll:] += w * (-x[:n - ll])
    return grad

def slsqp_optimize(h_start, n, eps, maxiter=600):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_max_obj, h_start.copy(), jac=smooth_max_grad,
        args=(eps, n), method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-16})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def compute_grad_matrix(h, n):
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))
    for k in range(n_corr):
        l = k - (n - 1)
        if l >= 0:
            G[k, l:] += (1.0 - h[:n - l])
            G[k, :n - l] += (-h[l:])
        else:
            ll = -l
            G[k, :n - ll] += (1.0 - h[ll:])
            G[k, ll:] += (-h[:n - ll])
    return G

def find_optimal_d(h, i, j, n, G, corr, dx):
    slopes = (G[:, i] - G[:, j]) * dx
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-14:
        return 0.0
    lo, hi = d_lo, d_hi
    for _ in range(35):
        mid = (lo + hi) / 2.0
        vals = corr + mid * slopes
        k_star = np.argmax(vals)
        if slopes[k_star] < -1e-14:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def paired_cd_random(h, n, n_iters, rng, time_check):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    for it in range(n_iters):
        if time_check():
            break
        if it % 50 == 0:
            G = compute_grad_matrix(h, n)
            dx = 2.0 / n
            corr = np.correlate(h, 1 - h, mode="full")
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = find_optimal_d(h, i, j, n, G, corr, dx)
        if abs(d) < 1e-12:
            no_improve += 1
            continue
        old_i = h[i]
        old_j = h[j]
        h[i] += d
        h[j] -= d
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = old_i
            h[j] = old_j
            no_improve += 1
        if no_improve > 8000:
            no_improve = 0
            h = best_h.copy()
    return best_h, best_c5

def paired_cd_systematic(h, n, n_sweeps, rng, time_check):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for sweep in range(n_sweeps):
        if time_check():
            break
        G = compute_grad_matrix(h, n)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        best_d = 0.0
        best_impr = 1e-15
        best_i, best_j = -1, -1
        n_pairs = min(n * (n - 1) // 2, 8000)
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = find_optimal_d(h, i, j, n, G, corr, dx)
            if abs(d) < 1e-12:
                continue
            vals = corr + d * ((G[:, i] - G[:, j]) * dx)
            new_max = np.max(vals) * dx
            impr = best_c5 - new_max
            if impr > best_impr:
                best_impr = impr
                best_d = d
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] += best_d
            h[best_j] -= best_d
            best_c5 = evaluate_c5(h)
            best_h = h.copy()
        else:
            break
    return best_h, best_c5

def upsample_linear(h, n_from, n_to):
    ratio = n_from / n_to
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = (i + 0.5) * ratio - 0.5
        idx = int(np.floor(pos))
        idx = max(0, min(idx, n_from - 1))
        frac = pos - idx
        if idx + 1 < n_from and frac > 0:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return project_feasible(out, n_to)

def diverse_init(n, rng, parent=None):
    candidates = []
    if parent is not None:
        scale = 0.03
        h_p = parent + scale * rng.randn(n)
        h_p = project_feasible(h_p, n)
        candidates.append(h_p)
    h_bin = np.zeros(n)
    h_bin[:n // 2] = 1.0
    rng.shuffle(h_bin)
    candidates.append(project_feasible(h_bin, n))
    for t in range(3):
        freq = rng.randint(3, 10)
        phase = rng.uniform(0, 2 * np.pi)
        h_wave = np.zeros(n)
        for i in range(n):
            x = 2.0 * i / n
            v = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
            h_wave[i] = np.clip(v, 0, 1)
        candidates.append(project_feasible(h_wave, n))
    best_h = None
    best_c5 = 1e9
    for h_c in candidates:
        c5 = evaluate_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def make_time_limit(frac):
        lim = budget_s * frac

    h0 = np.array(initial_h_values, dtype=float)
