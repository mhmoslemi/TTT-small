def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):

    dx = 2.0 / n
    corr = compute_correlations(h)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    # Use bisection on the shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    # Distribute among non-boundary elements

    if np.sum(mask) > 0:
        projected[mask] += diff / np.sum(mask)

    return projected

    n = len(h)

    idx = np.argsort(-corr)[:n_active]

    # = sum_j grad[m,j] * h[j] + (corr[m] - sum_j grad[m,j] * h_old[j])
    
    # grad[m, j] = (1 - h[j+r_m]) - h[j-r_m] where r_m = m - (n-1)
    # For m in idx:
    #   r = m - (n-1)
    #   grad[j] = (1 - h[j+r]) * [0<=j+r<n] - h[j-r] * [0<=j-r<n]
    
    # Build A matrix for LP:
    # Variables: h[0], ..., h[n-1], t
    # Constraints:
    #   For each active m: sum_j grad[m,j] * h[j] - t <= -const[m]
    #   sum(h) = target_sum
    #   0 <= h[j] <= 1
    #   t >= 0

    # s.t. sum_j grad[m,j] * h[j] + const[m] <= t  for active m
    #      sum(h) = target_sum
    #      0 <= h[j] <= 1
    #      t unrestricted (but bounded by problem)
    
    # In standard LP form: min c^T x
    # x = [h, t]
    # c = [0,...,0, 1]
    # A_ub @ x <= b_ub: for each active m, [grad[m], -1] @ x <= -const[m]
    # A_eq @ x == b_eq: [ones, 0] @ x == target_sum
    # bounds: 0 <= h[j] <= 1, t free (but we can bound it)
    
    n_active_actual = len(idx)
    
    # Compute gradients for active lags
    A = np.zeros((n_active_actual, n + 1))
    b = np.zeros(n_active_actual)
    
    for i, m in enumerate(idx):
        r = m - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j + r < n:
                val += (1.0 - h[j + r])
            if 0 <= j - r < n:
                val -= h[j - r]
            grad[j] = val
        const = corr[m] - np.dot(grad, h)
        A[i, :n] = grad
        A[i, n] = -1.0
        b[i] = -const
    
    # Equality constraint: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    
    # Objective: min t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds_lp = [(0, 1)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds_lp, method='highs')
    
    if result.success:
        h_new = result.x[:n]
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        return h_new
    return h


def slp_optimize(h0, n_active, max_iter, tol=1e-8):
    """Run SLP for a fixed number of iterations."""
    h = h0.copy()
    best_h = h0.copy()
    best_score = evaluate_score(h0)
    
    for it in range(max_iter):
        h_new = slp_step(h, n_active)
        s_new = evaluate_score(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
        if abs(s_new - best_score) < tol:
            break
        h = h_new
    
    return best_h, best_score

            break
        h_s, s_s = slp_optimize(best_h, n_iters=50, n_active=n_act)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()
    if time_left() > 60:
        h_s, s_s = slp_optimize(best_h, n_iters=40, n_active=10, use_all=True)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()

    # Phase 5: Try higher resolutions
    n_best = len(best_h)
    target_ns = [500, 600, 700, 800, 900, 1000, 1200, 1500]
    for n_try in target_ns:
        if time_left() < 200:
            break
        if n_try >= n_best:
            continue
        x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_feasible(h_init, n_try / 2.0)
        bounds_try = [(0.0, 1.0)] * n_try

        h_s, s_s = slp_optimize(h_init, n_iters=40, n_active=30)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()
            n_best = n_try

        for alpha in [200, 500, 2000]:
            if time_left() < 80:
                break
            x_new = lfgs_smooth(best_h, alpha, 300, bounds_try)
            h_new = project_feasible(np.clip(x_new, 0, 1), n_try / 2.0)
            s_new = evaluate_score(h_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_h = h_new.copy()

        if time_left() > 40:
            h_s, s_s = slp_optimize(best_h, n_iters=30, n_active=50)
            if s_s < best_score:
                best_score = s_s
                best_h = h_s.copy()

    # Phase 6: Alternative initializations at best resolution
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    inits = make_initial_constructions(n_best, rng)
    for h_init in inits:
        if time_left() < 60:
            break
        h_s, s_s = slp_optimize(h_init, n_iters=30, n_active=30)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()

    # Phase 7: Random restarts
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    max_restarts = 300
    for i in range(max_restarts):
        if time_left() < 30:
            break
        strategy = i % 5
        if strategy == 0:
            pert = best_h + rng.randn(n_best) * 0.04
        elif strategy == 1:
            pert = best_h + rng.randn(n_best) * 0.1
        elif strategy == 2:
            pert = best_h.copy()
            n_swap = max(3, n_best // 20)
            idx = rng.choice(n_best, size=n_swap, replace=False)
            pert[idx] = 1.0 - pert[idx]
            pert = project_feasible(pert, n_best / 2.0)
        elif strategy == 3:
            shift = rng.randint(1, max(2, n_best // 10))
            pert = np.roll(best_h, shift)
            pert = project_feasible(pert, n_best / 2.0)
        else:
            pert = 1.0 - best_h
            pert = project_feasible(pert, n_best / 2.0)

        pert = project_feasible(pert, n_best / 2.0)
        h_s, s_s = slp_optimize(pert, n_iters=25, n_active=20)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()

    # Phase 8: Final intensive SLP polish
    if time_left() > 30:
        for n_act in [50, 100, 200, 500]:
            if time_left() < 15:
                break
            h_s, s_s = slp_optimize(best_h, n_iters=50, n_active=n_act)
            if s_s < best_score:
                best_score = s_s
                best_h = h_s.copy()

    if time_left() > 20:
        h_s, s_s = slp_optimize(best_h, n_iters=40, n_active=10, use_all=True)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()

    # Final smooth max polish
    if time_left() > 15:
        n_best = len(best_h)
        bounds_best = [(0.0, 1.0)] * n_best
        for alpha in [5000, 20000, 50000]:
            if time_left() < 8:
                break
            x_new = lfgs_smooth(best_h, alpha, 200, bounds_best)
            h_new = project_feasible(np.clip(x_new, 0, 1), n_best / 2.0)
            s_new = evaluate_score(h_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_h = h_new.copy()

    # Final SLP
    if time_left() > 5:
        h_s, s_s = slp_optimize(best_h, n_iters=30, n_active=100)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
