import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def binary_round(h, target_ones):
    # Round to binary, keeping exactly target_ones ones
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    h_bin = np.zeros(n)
    h_bin[order[n - target_ones:]] = 1.0
    return h_bin

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def sa_move_single(h, rng):
    """Swap one 1 with one 0."""
    n = len(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    oi = ones[rng.integers(len(ones))]
    zi = zeros[rng.integers(len(zeros))]
    h_new = h.copy()
    h_new[oi] = 0.0
    h_new[zi] = 1.0
    return h_new

def sa_move_block_swap(h, rng):
    """Swap two blocks of the same length."""
    n = len(h)
    h_new = h.copy()
    max_len = max(2, n // 4)
    blen = int(rng.integers(1, max_len + 1))
    a = int(rng.integers(0, n - blen + 1))
    b = int(rng.integers(0, n - blen + 1))
    i1, i2 = min(a, b), max(a, b)
    if i2 - i1 < blen:
        return h.copy()
    block1 = h_new[i1:i1+blen].copy()
    h_new[i1:i1+blen] = h_new[i2:i2+blen].copy()
    h_new[i2:i2+blen] = block1
    return h_new

def sa_move_reverse(h, rng):
    """Reverse a segment."""
    n = len(h)
    h_new = h.copy()
    a = int(rng.integers(0, n - 1))
    b = int(rng.integers(a + 1, n))
    h_new[a:b] = h_new[a:b][::-1]
    return h_new

def sa_move_multi_swap(h, rng, k=2):
    """Swap k ones with k zeros."""
    n = len(h)
    h_new = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) < k or len(zeros) < k:
        return h.copy()
    oi = rng.choice(ones, size=k, replace=False)
    zi = rng.choice(zeros, size=k, replace=False)
    for i, z in zip(oi, zi):
        h_new[i] = 0.0
        h_new[z] = 1.0
    return h_new

def population_sa(h_init, n_steps, t_start, t_end, rng, n_pop=5):
    """Population-based simulated annealing with diverse moves."""
    n = len(h_init)
    target_ones = int(n // 2)
    
    # Initialize population
    pop = [h_init.copy() for _ in range(n_pop)]
    pop[0] = binary_round(h_init, target_ones)
    for p in range(1, n_pop):
        h_rand = (rng.random(n) < 0.5).astype(float)
        # Ensure exactly target_ones ones
        ones_count = int(np.sum(h_rand))
        if ones_count > target_ones:
            excess = rng.choice(np.where(h_rand == 1.0)[0], size=ones_count - target_ones, replace=False)
            h_rand[excess] = 0.0
        elif ones_count < target_ones:
            deficit = rng.choice(np.where(h_rand == 0.0)[0], size=target_ones - ones_count, replace=False)
            h_rand[deficit] = 1.0
        pop[p] = h_rand
    
    scores = [compute_c5(p) for p in pop]
    best = pop[0].copy()
    best_score = min(scores)
    
    move_funcs = [sa_move_single, sa_move_block_swap, sa_move_reverse, sa_move_multi_swap]
    
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Select population member (weighted by quality)
        p = int(rng.integers(n_pop))
        
        # Choose move
        move = move_funcs[int(rng.integers(len(move_funcs)))]
        if move == sa_move_multi_swap:
            h_new = move(pop[p], rng, k=int(rng.integers(2, 4)))
        else:
            h_new = move(pop[p], rng)
        
        if np.array_equal(h_new, pop[p]):
            continue
        
        # Verify sum constraint
        if int(np.sum(h_new)) != target_ones:
            continue
        
        new_score = compute_c5(h_new)
        delta = new_score - scores[p]
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            pop[p] = h_new
            scores[p] = new_score
            if new_score < best_score:
                best_score = new_score
                best = h_new.copy()
    
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def binary_round(h, target_ones):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    h_bin = np.zeros(n)
    h_bin[order[n - target_ones:]] = 1.0
    return h_bin

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def sa_move_single(h, rng):
    n = len(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    oi = ones[rng.integers(len(ones))]
    zi = zeros[rng.integers(len(zeros))]
    h_new = h.copy()
    h_new[oi] = 0.0
    h_new[zi] = 1.0
    return h_new

def sa_move_block_swap(h, rng):
    n = len(h)
    h_new = h.copy()
    max_len = max(2, n // 4)
    blen = int(rng.integers(1, max_len + 1))
    a = int(rng.integers(0, n - blen + 1))
    b = int(rng.integers(0, n - blen + 1))
    i1, i2 = min(a, b), max(a, b)
    if i2 - i1 < blen:
        return h.copy()
    block1 = h_new[i1:i1 + blen].copy()
    h_new[i1:i1 + blen] = h_new[i2:i2 + blen]
    h_new[i2:i2 + blen] = block1
    return h_new

def sa_move_reverse(h, rng):
    n = len(h)
    h_new = h.copy()
    a = int(rng.integers(0, n - 1))
    b = int(rng.integers(a + 1, n))
    h_new[a:b] = h_new[a:b][::-1]
    return h_new

def sa_move_multi_swap(h, rng, k=2):
    n = len(h)
    h_new = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) < k or len(zeros) < k:
        return h.copy()
    oi = rng.choice(ones, size=k, replace=False)
    zi = rng.choice(zeros, size=k, replace=False)
    for i, z in zip(oi, zi):
        h_new[i] = 0.0
        h_new[z] = 1.0
    return h_new

def make_random_binary(n, target_ones, rng):
    h = (rng.random(n) < 0.5).astype(float)
    cnt = int(np.sum(h))
    if cnt > target_ones:
        excess = rng.choice(np.where(h == 1.0)[0], size=cnt - target_ones, replace=False)
        h[excess] = 0.0
    elif cnt < target_ones:
        deficit = rng.choice(np.where(h == 0.0)[0], size=target_ones - cnt, replace=False)
        h[deficit] = 1.0
    return h

def population_sa(h_init, n_steps, t_start, t_end, rng, n_pop=5, use_diverse=True):

    target_ones = int(n // 2)

    pop = [h_init.copy() for _ in range(n_pop)]
    pop[0] = binary_round(h_init, target_ones)
    for p in range(1, n_pop):
        pop[p] = make_random_binary(n, target_ones, rng)

    scores = [compute_c5(p) for p in pop]
    best = pop[int(np.argmin(scores))].copy()
    best_score = min(scores)

    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        p = int(rng.integers(n_pop))

        r = rng.random()
        if use_diverse:
            if r < 0.4:
                h_new = sa_move_single(pop[p], rng)
            elif r < 0.6:
                h_new = sa_move_block_swap(pop[p], rng)
            elif r < 0.75:
                h_new = sa_move_reverse(pop[p], rng)
            else:
                h_new = sa_move_multi_swap(pop[p], rng, k=int(rng.integers(2, 4)))
        else:
            h_new = sa_move_single(pop[p], rng)

        if np.array_equal(h_new, pop[p]):
            continue
        if int(np.sum(h_new)) != target_ones:
            continue

        new_score = compute_c5(h_new)

            scores[p] = new_score
            if new_score < best_score:

        remaining = budget_s - (time.time() - start_time)

    remaining = budget_s - (time.time() - start_time) - 60
    if remaining > 300:
        n1 = 2 * n0
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[i // 2]
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up_bin = binary_round(h_up, n1 // 2)
        c5_up = compute_c5(h_up_bin)

        n_steps = min(150000, int((remaining - 150) * 30))
        h_sa_up, c5_sa_up = population_sa(h_up_bin, n_steps, 0.4, 0.01, rng, n_pop=4, use_diverse=True)
        if c5_sa_up < best_c5:
            best_c5 = c5_sa_up
            best_h = h_sa_up.copy()

        remaining = budget_s - (time.time() - start_time) - 60
        if remaining > 100:
            h_soft = best_h + 0.03 * rng.standard_normal(len(best_h))
            h_soft = project_onto_constraint(h_soft, len(best_h) / 2.0)
            h_opt, c5_opt = slsqp_optimize(h_soft, 0.005, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            h_opt, c5_opt = slsqp_optimize(h_opt, 0.001, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    remaining = budget_s - (time.time() - start_time) - 30
    if remaining > 100:
        n_cur = len(best_h)
        h_final_bin = binary_round(best_h, n_cur // 2)
        h_t, c5_t = targeted_binary_search(h_final_bin, rng, n_iters=min(200000, int(remaining * 50)))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    remaining = budget_s - (time.time() - start_time) - 20
    if remaining > 50:
        n_cur = len(best_h)
        h_polish = best_h + 0.005 * rng.standard_normal(n_cur)
        h_polish = project_onto_constraint(h_polish, n_cur / 2.0)
        for eps in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            if time.time() - start_time > budget_s - 15:
                break
            h_opt, c5_opt = slsqp_optimize(h_polish, eps, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            h_polish = h_opt

    n_final = len(best_h)
    final_h = project_onto_constraint(best_h, n_final / 2.0)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_c5 = compute_c5(final_h)

    return final_h, final_c5, n_final
