from scipy.optimize import minimize
import time

_g_eps = 1000.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        l = k - (n - 1)
        if l > 0:
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
        elif l < 0:
            m2 = -l
            grad[m2:] += w * (1.0 - h[:n - m2])
            grad[:n - m2] -= w * h[m2:]
        else:
            grad += w * (1.0 - 2.0 * h)
    grad *= dx
    return grad

def _obj(x):
    return smooth_max_val(x, _g_eps)

def _grad(x):
    return smooth_max_grad(x, _g_eps)

def _cons_eq(x):
    return np.sum(x) - len(x) / 2.0

def _cons_eq_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h_start, eps, maxiter):
    global _g_eps
    _g_eps = eps
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h_start.copy(), jac=_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h_start, evaluate_c5(h_start)

def coord_descent(h_start, n, max_sweeps, start_time, budget_s, time_frac):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    no_improve = 0
    step = 0.005
    for sweep in range(max_sweeps):
        if (time.time() - start_time) >= budget_s * time_frac:
            break
        improved = False
        for i in range(n):
            cur = h[i]
            if cur < 0.005:
                targets = [0.0, 0.01, 0.02]
            elif cur > 0.995:
                targets = [1.0, 0.99, 0.98]
            else:
                targets = [cur - step, cur + step, cur - step * 0.3, cur + step * 0.3]
            for t in targets:
                if t < 0.0 or t > 1.0:
                    continue
                h_try = h.copy()
                h_try[i] = t
                h_try = project_feasible(h_try, n)
                c5 = evaluate_c5(h_try)
                if c5 < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5
                    improved = True
                    break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            step *= 0.7
            if step < 5e-5 or no_improve > 8:
                break
    return h, best_c5

def upsample_nn(h, n_from, n_to):
    scale = float(n_to) / float(n_from)
    idx = (np.arange(n_to) / scale).astype(int)
    idx = np.clip(idx, 0, n_from - 1)
    return h[idx].copy()

def upsample_interp(h, n_from, n_to):
    pos = np.arange(n_to) * (n_from - 1) / float(n_to - 1)
    idx = np.floor(pos).astype(int)
    idx = np.clip(idx, 0, n_from - 2)
    frac = pos - idx
    out = (1.0 - frac) * h[idx] + frac * h[idx + 1]
    return out

def binary_local_search(h, n, n_iters, start_time, budget_s, time_frac):
