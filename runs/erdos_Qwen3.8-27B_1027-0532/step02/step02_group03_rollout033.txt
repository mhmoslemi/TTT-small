    best_n = n_init

    # Phase 1: Adaptive Nesterov from parent (broad to sharp)
    if time.time() < deadline - 200:
        for (al, ah, iters) in [(10, 100, 400), (30, 300, 350), (100, 800, 300)]:
            if time.time() > deadline - 180:
                break
            h_res, c_res = _nesterov_adaptive(best_h, best_n, target, iters, al, ah)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = h_res.copy()

    # Phase 2: Top-k Nesterov
    if time.time() < deadline - 180:
        for tk in [3, 5, 10]:
            if time.time() > deadline - 160:
                break
            h_res, c_res = _topk_nesterov(best_h, best_n, target, 250, tk, 0.35)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = h_res.copy()

    # Phase 3: DE exploration
    if time.time() < deadline - 150:
        h_res, c_res = _de_optimize(best_h, best_n, target, 30, 60, deadline - 120, seed + 100)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res.copy()

    # Phase 4: Stagnation restarts
    if time.time() < deadline - 120:
        h_res, c_res = _stagnation_restart(best_h, best_n, target, rng, 5, deadline - 100)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res.copy()

    # Phase 5: SLSQP refinement
    if time.time() < deadline - 80:
        for alpha in [50, 200, 500]:
            if time.time() > deadline - 70:
                break
            h_sl = _slsqp_opt(best_h, best_n, target, alpha, maxiter=300)
            c_sl = _c5_of(h_sl)
            if c_sl < best_c5:
                best_c5 = c_sl
                best_h = h_sl.copy()

    # Phase 6: Coordinate descent
    if time.time() < deadline - 50:
        h_cd = _coord_line_search(best_h, best_n, target, max_rounds=20)
        c_cd = _c5_of(h_cd)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 7: Try expanded dimension
    if time.time() < deadline - 100:
        n_expand = best_n + 20
        if n_expand <= 240:
            x_old = np.arange(float(best_n))
            x_new = np.linspace(0, float(best_n - 1), n_expand)
            h_exp = np.interp(x_new, x_old, best_h)
            h_exp = _fix_sum(np.clip(h_exp, 0, 1), float(n_expand) / 2.0)
            c_exp = _c5_of(h_exp)
            if c_exp < best_c5:
                best_c5 = c_exp
                best_h = h_exp.copy()
                best_n = n_expand
                target = float(n_expand) / 2.0
                if time.time() < deadline - 60:
                    h_res, c_res = _nesterov_adaptive(best_h, best_n, target, 150, 50, 500)
                    if c_res < best_c5:
                        best_c5 = c_res
                        best_h = h_res.copy()
                    if time.time() < deadline - 40:
                        h_sl = _slsqp_opt(best_h, best_n, target, 300, maxiter=200)
                        c_sl = _c5_of(h_sl)
                        if c_sl < best_c5:
                            best_c5 = c_sl
                            best_h = h_sl.copy()

    # Phase 8: Final SLSQP polish with high alpha
    if time.time() < deadline - 30:
        h_sl = _slsqp_opt(best_h, best_n, target, 1000, maxiter=500)
        c_sl = _c5_of(h_sl)
        if c_sl < best_c5:
            best_c5 = c_sl
            best_h = h_sl.copy()

    # Phase 9: Final coordinate descent
    if time.time() < deadline - 10:
        h_cd = _coord_line_search(best_h, best_n, target, max_rounds=25)
        c_cd = _c5_of(h_cd)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Final validation
    best_h = _fix_sum(best_h, float(best_n) / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
