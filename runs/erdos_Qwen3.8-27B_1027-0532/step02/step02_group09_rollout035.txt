def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlap_vector(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for idx in range(len(overlaps)):
        w = weights[idx]
        if w < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            if L < n:
                grad[L:] += w * (1.0 - h[:n - L])
                grad[:n - L] += -w * h[L:]
        else:
            M = -L
            if M < n:
                grad[M:] += w * (1.0 - h[:n - M])
                grad[:n - M] += -w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    overlaps = compute_overlap_vector(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - len(h) / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None

def to_binary(h, n_target):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])
        remove = ones_idx[order[:n_ones - n_target]]
        h_bin[remove] = 0.0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])
        add = zeros_idx[order[:n_target - n_ones]]
        h_bin[add] = 1.0
    return h_bin

            break
    return best_h, best_max * dx


def stochastic_local_search(h0, n_restarts, perturb_scale, target, rng, top_k=5, lr=0.05):
    n = len(h0)
    best_h = h0.copy()
    best_score = evaluate_score(h0)
    for r in range(n_restarts):
        h = h0 + rng.randn(n) * perturb_scale
        h = project_box_sum(h, target)
        for _ in range(200):
            grad = active_set_gradient(h, top_k)
            h_new = project_box_sum(h - lr * grad, target)
            s = evaluate_score(h_new)
            if s < evaluate_score(h):
                h = h_new
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = evaluate_score(h)

    def time_left():
        return budget_s - 30.0 - (time.time() - t0)

    # Phase 1: Active-set gradient (fast, focused)
    for top_k in [3, 5, 8, 10]:
        if time_left() < 60:
            break
        n_iters = min(20000, int(time_left() * 5))
        lr = 0.3
        h_new, s_new = active_set_optimize(best_h, n_iters, top_k, lr, target)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 2: LSE-based L-BFGS-B polishing
    for alpha in [200, 500, 1000, 2000]:
        if time_left() < 60:
            break
        penalty = alpha * 20.0
        bounds = [(0.0, 1.0)] * n
        res_x = lbfgsb_optimize(best_h, alpha, penalty, 500, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 3: SLSQP with high alpha
    for alpha in [500, 1000, 2000]:
        if time_left() < 60:
            break
        bounds = [(0.0, 1.0)] * n
        res_x = slsqp_optimize(best_h, alpha, 300, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 4: Binary local search + simulated annealing
    if time_left() > 90:
        h_bin = to_binary(best_h, int(target))
        # Greedy improvement
        h_g, s_g = binary_greedy(h_bin, 50, rng, target)
        if s_g < best_score:
            best_score = s_g
            best_h = h_g.copy()
        h_bin = to_binary(best_h, int(target))
        # Simulated annealing
        n_sa = min(50000, int(time_left() * 3))
        t_init = max(0.5, best_score * n)
        t_final = 1e-8
        h_sa, s_sa = binary_sa(h_bin, n_sa, t_init, t_final, rng, target, bias_frac=0.5)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa.copy()
        # Polish binary result with continuous optimization
        if time_left() > 60:
            for alpha in [1000, 3000]:
                if time_left() < 30:
                    break
                penalty = alpha * 30.0
                bounds = [(0.0, 1.0)] * n
                res_x = lbfgsb_optimize(best_h, alpha, penalty, 400, bounds)
                if res_x is not None:
                    h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), target)
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()

    # Phase 5: Stochastic local search with perturbations
    if time_left() > 90:
        for scale in [0.05, 0.1, 0.15, 0.2]:
            if time_left() < 60:
                break
            h_sl, s_sl = stochastic_local_search(best_h, 5, scale, target, rng, top_k=5, lr=0.08)
            if s_sl < best_score:
                best_score = s_sl
                best_h = h_sl.copy()
                # Polish
                if time_left() > 60:
                    alpha = 2000
                    penalty = alpha * 30.0
                    bounds = [(0.0, 1.0)] * n
                    res_x = lbfgsb_optimize(best_h, alpha, penalty, 500, bounds)
                    if res_x is not None:
                        h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), target)
                        s = evaluate_score(h_new)
                        if np.isfinite(s) and s < best_score:
                            best_score = s
                            best_h = h_new.copy()

    # Phase 6: Symmetry-based restarts
    if time_left() > 90:
        # Complement
        h_sym = 1.0 - best_h
        h_sym = project_box_sum(h_sym, target)
        for alpha in [500, 1500]:
            if time_left() < 40:
                break
            penalty = alpha * 30.0
            bounds = [(0.0, 1.0)] * n
            res_x = lbfgsb_optimize(h_sym, alpha, penalty, 400, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), target)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                h_sym = h_new
        # Reversal
        h_rev = best_h[::-1].copy()
        h_rev = project_box_sum(h_rev, target)
        for alpha in [500, 1500]:
            if time_left() < 40:
                break
            penalty = alpha * 30.0
            bounds = [(0.0, 1.0)] * n
            res_x = lbfgsb_optimize(h_rev, alpha, penalty, 400, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), target)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 7: Multi-resolution (upscale and re-optimize)
    if time_left() > 120:
        for n_try in [n * 2, n * 3]:
            if n_try > 2000 or time_left() < 80:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_box_sum(h_init, n_try / 2.0)
            for alpha in [200, 800, 2000]:
                if time_left() < 40:
                    break
                penalty = alpha * 25.0
                bounds = [(0.0, 1.0)] * n_try
                maxiter = max(50, 200000 // n_try)
                res_x = lbfgsb_optimize(h_init, alpha, penalty, maxiter, bounds)
                if res_x is not None:
                    h_fine = project_box_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    s = evaluate_score(h_fine)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_fine.copy()
                        h_init = h_fine.copy()

    # Phase 8: Final high-precision refinement
    if time_left() > 40:
        n_final = len(best_h)
        for alpha in [3000, 5000, 10000]:
            if time_left() < 20:
                break
            penalty = alpha * 50.0
            bounds = [(0.0, 1.0)] * n_final
            res_x = lbfgsb_optimize(best_h, alpha, penalty, 800, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
        # Final SLSQP
        if time_left() > 20:
            bounds = [(0.0, 1.0)] * n_final
            res_x = slsqp_optimize(best_h, 5000, 200, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
        # One more active-set pass
        if time_left() > 15:
            h_new, s_new = active_set_optimize(best_h, 5000, 5, 0.2, n_final / 2.0)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
