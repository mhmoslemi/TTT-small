```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    logsumexp = m + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))
    return logsumexp

def smooth_gradient(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    p = np.exp(alpha * (corr - m))
    p /= p.sum()
    
    n = len(h)
    grad = np.zeros(n)
    n_full = len(corr)
    center = n - 1
    for k in range(n_full):

        if lag == 0:

        else:

def slsqp_smooth(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        return smooth_objective(hv, alpha)
    
    def obj_grad(hv):
        return smooth_gradient(hv, alpha)
    
    def con_fun(hv):
        return np.sum(hv) - target_sum
    
    def con_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
```

Each SLSQP iteration:

- Gradient evaluation: O(n·k) where k is number of significant softmax weights
- SLSQP internal: O(n²) for the QP subproblem

For n=425, the QP subproblem is 425×425 which is about 180K. With 300 iterations, that's 54M operations. Should be a few seconds at most.

Actually, SLSQP in scipy uses a finite-difference approximation if no gradient is provided, which would be much slower (n extra function evaluations per iteration). Since I'm providing the gradient analytically, it should be fast.

Let me also consider: the smooth_gradient function loops over all 2n-1 = 849 correlation lags. For each lag, it does O(n) work. So the total gradient computation is O(n²) = O(425²) = 180K. This is fine.

But wait, I can optimize this! The gradient can be computed more efficiently using the structure:

∇g_α[h][t] = ∑_j p_j · (1 - h[t+j]·1_{valid} - h[t-j]·1_{valid})

This is a convolution-like operation! I can compute it as:
∇g_α = ∑_j p_j - ∑_j p_j · h shifted by j - ∑_j p_j · h shifted by -j
     = sum(p) - (p * h) - (p[::-1] * h)  [in convolution notation]

Where p is the softmax vector of length 2n-1 and h is of length n.

Actually, this IS a convolution! Let me define:
p_centered[j] = p[j + (n-1)] for j from -(n-1) to (n-1)

Then ∇g_α[h][t] = sum(p) - ∑_j p_centered[j] * h[t-j] (with boundary handling)

This is a correlation of h with p_centered, which can be done via FFT in O(n log n)!

Let me verify:
∇g_α[h][t] = ∑_j p_j · (1 - h[t+j]·1_{0≤t+j<n} - h[t-j]·1_{0≤t-j<n})

= n·sum(p) - ∑_j p_j · h[t+j]·1_{0≤t+j<n} - ∑_j p_j · h[t-j]·1_{0≤t-j<n}

p has length 2n-1, indexed from 0 to 2n-2.
lag_j = j - (n-1) for j in [0, 2n-2]

Wait, that's not right either. Let me go back to basics.

C(j) = ∑_i h[i](1-h[i+j]) where the sum is over valid i (0≤i<n and 0≤i+j<n).

∂C(j)/∂h[t]:
- h[t] appears as h[i] when i=t: need 0≤t<n and 0≤t+j<n. Contribution: (1-h[t+j])
- h[t] appears as h[i+j] when i=t-j: need 0≤t-j<n and 0≤t<n. Contribution: -h[t-j]

So ∂C(j)/∂h[t] = (1-h[t+j])·1_{0≤t+j<n} - h[t-j]·1_{0≤t-j<n}

And:
∇g_α[h][t] = ∑_j p_j · [(1-h[t+j])·1_{0≤t+j<n} - h[t-j]·1_{0≤t-j<n}]

First term: ∑_j p_j · 1_{0≤t+j<n} = ∑_{j: -t ≤ j < n-t} p_j

```python
def smooth_gradient(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    p = np.exp(alpha * (corr - m))
    p /= p.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for j in range(len(p)):
        w = p[j]
        if w < 1e-15:
            continue
        lag = j - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n-la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n-la]
    return grad
```

Let me also add a key insight: since we're minimizing the max of correlations, and the correlations are smooth functions of h, the optimal solution likely has all the "active" correlations (those near the max) equalized. This is like a water-filling problem. I can use this to design a better algorithm.

**Water-filling inspired algorithm:**
1. Compute all correlations C(k)
2. The active set is {k : C(k) ≥ C_max - ε}
3. We want to decrease all active correlations simultaneously
4. The direction that most decreases the max is the normalized sum of gradients of active correlations
5. Take a step in that direction, project to feasible set
6. Repeat

This is essentially what the previous subgradient method does, but I can make it more principled with:
- Proper step size (line search)

```python
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_gradient(h, corr, p):
    """Compute gradient of sum_j p_j * C(j)"""
    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for j in range(len(p)):
        w = p[j]
        if w < 1e-15:
            continue
        lag = j - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n-la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n-la]
    return grad

def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))
    
    def obj_grad(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        p = np.exp(alpha * (corr - m))
        p /= p.sum()
        return get_gradient(hv, corr, p)
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda v: np.sum(v) - target_sum,
                                 'jac': lambda v: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def lbfgsb_refine(h, alpha, maxiter=2000):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    mu = 50.0
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        base = m + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))
        penalty = mu * (np.sum(hv) - target_sum)**2
        return base + penalty
    
    def obj_grad(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        p = np.exp(alpha * (corr - m))
        p /= p.sum()
        base_grad = get_gradient(hv, corr, p)
        penalty_grad = 2.0 * mu * (np.sum(hv) - target_sum)
        return base_grad + penalty_grad * np.ones(n)
    
    res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def nesterov_projected_gradient(h, alpha, n_iter, step, momentum, t0, time_limit):
    """Nesterov accelerated projected gradient on smooth objective"""
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    h_prev = h.copy()
    velocity = np.zeros(n)
    
    best_score = evaluate_score(h_curr)
    best_h = h_curr.copy()
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        
        # Nesterov step
        h_nest = h_curr + momentum * velocity
        
        # Compute gradient at h_nest (projected)
        h_nest_p = project_to_feasible(h_nest, target_sum)
        corr = compute_correlations(h_nest_p)
        m = np.max(corr)
        p = np.exp(alpha * (corr - m))
        p /= p.sum()
        grad = get_gradient(h_nest_p, corr, p)
        
        # Update
        velocity = momentum * velocity - step * grad
        h_new = h_curr + velocity
        h_new = project_to_feasible(h_new, target_sum)
        
        # Line search (simple backtracking)
        score_new = evaluate_score(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        
        h_curr = h_new
        h_prev = h_curr
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def get_gradient(h, p):
    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    n_full = len(p)
    for j in range(n_full):
        w = p[j]
        if w < 1e-14:
            continue
        lag = j - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]
    return grad


def smooth_obj(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))


def smooth_grad(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    p = np.exp(alpha * (corr - m))
    p /= p.sum()
    return get_gradient(h, p)


def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        return smooth_obj(hv, alpha)

    def obj_grad(hv):
        return smooth_grad(hv, alpha)

    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda v: np.sum(v) - target_sum,
                                 'jac': lambda v: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def lbfgsb_refine(h, alpha, maxiter=2000):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    mu = 100.0

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        base = m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))
        penalty = mu * (np.sum(hv) - target_sum) ** 2
        return base + penalty

    def obj_grad(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        p = np.exp(alpha * (corr - m))
        p /= p.sum()
        base_grad = get_gradient(hv, p)
        pen_val = 2.0 * mu * (np.sum(hv) - target_sum)
        return base_grad + pen_val * np.ones(n)

    res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
```
