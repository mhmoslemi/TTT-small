import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def pnorm_value(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.sum(corr**p) ** (1.0/p)

def pnorm_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Guard against zero/negative correlations (shouldn't happen but be safe)

from scipy.optimize import minimize
import time

_gn = 0
_gt = 0.0
_gp = 4.0
_gpen = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-15)[0]
    else:
        idx = np.where(h > 1e-15)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def pnorm_val(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr**p) ** (1.0/p))

def pnorm_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(corr, 1e-30)
    w = c**(p-1)
    nf = np.sum(c**p)**(1.0/p - 1.0)
    A = np.correlate(1.0-h, w, mode='full')[3*n-3 : 2*n-2 : -1]
    B = np.convolve(h, w, mode='full')[n-1 : 2*n-1]
    return nf * (A - B)

    return pnorm_val(x, _gp) + _gpen * (np.sum(x) - _gt)**2

def grad_pen(x):
    return pnorm_grad(x, _gp) + 2.0 * _gpen * (np.sum(x) - _gt)

def obj_np(x):
    return pnorm_val(x, _gp)

def grad_np(x):
    return pnorm_grad(x, _gp)

def cons_fun(x):
    return np.sum(x) - _gt

def cons_jac(x):
    return np.ones(_gn)

def interp(h_src, n_dst, t_dst):
    ns = len(h_src)
    if ns == n_dst:
        return project_feasible(h_src.copy(), t_dst)
    xn = np.linspace(0.0, 1.0, n_dst)
    xo = np.linspace(0.0, 1.0, ns)
    ho = np.interp(xn, xo, h_src)
    return project_feasible(ho, t_dst)

def lbfgsb_opt(h0, n, target, p_list, iters=200, pen=1000.0):
    global _gp, _gpen, _gt, _gn
    _gn = n
    _gt = target
    _gpen = pen
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_sc = compute_c5(h)
    for p in p_list:
        _gp = float(p)
        try:
            res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': iters, 'gtol': 1e-12, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                h = h_new
                best_sc = sc
            else:
                h = h_new  # still use as starting point for next p
        except Exception:
            pass
    return h, best_sc

def trust_opt(h0, n, target, p_list, iters=200):
    global _gp, _gt, _gn
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h = h0.copy()
    best_sc = compute_c5(h)
    for p in p_list:
        _gp = float(p)

            if sc < best_sc:
                h = h_new
                best_sc = sc
            else:
                h = h_new
        except Exception:
            pass
    return h, best_sc

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-15)[0]
    else:
        idx = np.where(h > 1e-15)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def pnorm_val(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr**p) ** (1.0 / p))


def pnorm_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(corr, 1e-30)
    w = c**(p - 1)
    nf = np.sum(c**p)**(1.0 / p - 1.0)
    A = np.correlate(1.0 - h, w, mode='full')[3 * n - 3:2 * n - 2:-1]
    B = np.convolve(h, w, mode='full')[n - 1:2 * n - 1]
    return nf * (A - B)


def obj_pen(x):
    return pnorm_val(x, _gp) + _gpen * (np.sum(x) - _gt)**2


def grad_pen(x):
    return pnorm_grad(x, _gp) + 2.0 * _gpen * (np.sum(x) - _gt)


def obj_direct(x):
    return pnorm_val(x, _gp)


def grad_direct(x):
    return pnorm_grad(x, _gp)


def cons_fun(x):
    return np.sum(x) - _gt


def cons_jac(x):
    return np.ones(_gn)


def interp(h_src, n_dst, t_dst):
    ns = len(h_src)
    if ns == n_dst:
        return project_feasible(h_src.copy(), t_dst)
    xn = np.linspace(0.0, 1.0, n_dst)
    xo = np.linspace(0.0, 1.0, ns)
    ho = np.interp(xn, xo, h_src)
    return project_feasible(ho, t_dst)


def lbfgsb_round(h0, n, target, p_list, iters=300, pen=100.0):
    global _gn, _gt, _gp, _gpen
    _gn = n
    _gt = target
    _gpen = pen
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_sc = compute_c5(h)
    for p in p_list:
        _gp = float(p)
        try:
            res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': iters, 'gtol': 1e-12, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
            h = h_new
        except Exception:
            pass
    return h, best_sc


def trust_round(h0, n, target, p_list, iters=200):
    global _gn, _gt, _gp
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h = h0.copy()
    best_sc = compute_c5(h)
    for p in p_list:
        _gp = float(p)
        try:
            res = minimize(obj_direct, h, jac=grad_direct, method='trust-constr',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': iters, 'gtol': 1e-10, 'verbose': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
            h = h_new
        except Exception:
            pass
    return h, best_sc


def slsqp_round(h0, n, target, p_list, iters=500, pen=100.0):
    global _gn, _gt, _gp, _gpen
    _gn = n
    _gt = target
    _gpen = pen
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_sc = compute_c5(h)
    for p in p_list:
        _gp = float(p)
        try:
            res = minimize(obj_pen, h, jac=grad_pen, method='SLSQP',
                           bounds=bounds,
                           options={'maxiter': iters, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
            h = h_new
        except Exception:
            pass
    return h, best_sc


def run(seed=42, budget_s=1300, **kwargs):
    global _gn, _gt, _gp, _gpen
    start = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start)

    h_raw = np.array(initial_h_values, dtype=float)
    n0 = len(h_raw)
    t0 = n0 / 2.0
    h_best = project_feasible(h_raw, t0)
    sc_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B on n0 with p-annealing
    if tleft() > 100:
        p_sched = [4, 8, 16, 32, 64, 128, 256]
        h_r, sc_r = lbfgsb_round(h_best, n0, t0, p_sched, iters=400, pen=100.0)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    # Phase 2: Trust-constr with exact constraint
    if tleft() > 120:
        p_sched2 = [16, 32, 64, 128, 256]
        h_r, sc_r = trust_round(h_best, n0, t0, p_sched2, iters=300)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    # Phase 3: Higher resolutions
    n_candidates = [500, 600, 800, 1000, 1200]
    for n_try in n_candidates:
        if tleft() < 150:
            break
        t_try = n_try / 2.0
        h_interp = interp(h_best, n_try, t_try)
        p_sched3 = [8, 16, 32, 64, 128, 256]
        h_r, sc_r = lbfgsb_round(h_interp, n_try, t_try, p_sched3, iters=500, pen=100.0)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    # Phase 4: Multi-start perturbations on best resolution
    n_cur = len(h_best)
    t_cur = n_cur / 2.0
    noise_levels = [0.005, 0.01, 0.02, 0.03, 0.05]
    p_fine = [32, 64, 128, 256]
    attempt = 0
    while tleft() > 80 and attempt < 40:
        attempt += 1
        noise = rng.choice(noise_levels)
        h_pert = h_best + rng.randn(n_cur) * noise
        h_pert = project_feasible(h_pert, t_cur)
        h_r, sc_r = lbfgsb_round(h_pert, n_cur, t_cur, p_fine, iters=300, pen=100.0)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    # Phase 5: SLSQP refinement (different optimizer may find different basin)
    if tleft() > 100:
        p_sls = [32, 64, 128]
        h_r, sc_r = slsqp_round(h_best, n_cur, t_cur, p_sls, iters=800, pen=500.0)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    # Phase 6: Final high-p polish
    if tleft() > 40:
        p_final = [128, 256]
        h_r, sc_r = lbfgsb_round(h_best, n_cur, t_cur, p_final, iters=1000, pen=500.0)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    # Phase 7: One more trust-constr polish
    if tleft() > 40:
        p_tc = [128, 256]
        h_r, sc_r = trust_round(h_best, n_cur, t_cur, p_tc, iters=500)
        if sc_r < sc_best:
            h_best, sc_best = h_r, sc_r

    h_final = project_feasible(h_best, len(h_best) / 2.0)
    sc_final = compute_c5(h_final)
    return h_final, sc_final, len(h_final)
