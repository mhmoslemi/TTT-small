from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_corr_objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    obj = (m + np.log(w_sum)) / alpha * dx
    
    # Compute gradient
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        L = j - (n - 1)
        if L == 0:
            grad += w[j] * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w[j] * (1.0 - h[:n-L])
            grad[:n-L] -= w[j] * h[L:]
        else:
            La = -L
            grad[:n-La] += w[j] * (1.0 - h[La:])
            grad[La:] -= w[j] * h[:n-La]
    grad *= dx
    return obj, grad

def fista_run(h0, alpha, max_iters, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    x = project_to_feasible(h0.copy(), target_sum)
    y = x.copy()
    t = 1.0
    step = 1.0 / (alpha * n)  # initial step estimate
    best_score = evaluate_score(x)
    best_x = x.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        obj, grad = smooth_corr_objective_and_grad(y, alpha)
        
        # Backtracking line search for step size
        while True:
            z = y - step * grad
            # Project z
            z_proj = project_to_feasible(z, target_sum)
            # Check sufficient decrease
            obj_check, _ = smooth_corr_objective_and_grad(z_proj, alpha)
            if obj_check <= obj + np.dot(grad, z_proj - y) + (2.0 / (2.0 * step)) * np.sum((z_proj - y)**2):
                break
            step *= 0.5
            if step < 1e-15:
                break
        
        x_new = z_proj
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        x = x_new
        t = t_new
        
        # Adaptive step increase
        if it % 100 == 0:
            step = min(step * 1.1, 1.0 / (alpha * 10))
        
        score = evaluate_score(x)
        if score < best_score:
            best_score = score
            best_x = x.copy()
    
    return best_x, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shift = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shift)
    w = exp_shifted / exp_shifted.sum()
    obj = (m + (max_shift + np.log(exp_shifted.sum())) / alpha) * dx
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-13:
            continue
        L = j - (n - 1)
        if L == 0:
            grad += w[j] * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w[j] * (1.0 - h[:n-L])
            grad[:n-L] -= w[j] * h[L:]
        else:
            La = -L
            grad[:n-La] += w[j] * (1.0 - h[La:])
            grad[La:] -= w[j] * h[:n-La]
    grad *= dx
    return obj, grad

def fista_optimize(h0, alpha, max_iters, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    x = project_to_feasible(h0.copy(), target_sum)
    x_prev = x.copy()
    y = x.copy()
    t_param = 1.0
    
    # Estimate initial step size
    step = 1.0 / (2.0 * alpha)
    
    best_score = evaluate_score(x)
    best_x = x.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        obj_y, grad_y = smooth_obj_and_grad(y, alpha)
        
        # Backtracking line search
        z = y - step * grad_y
        z_proj = project_to_feasible(z, target_sum)
        obj_z, grad_z = smooth_obj_and_grad(z_proj, alpha)
        
        count = 0
        while obj_z > obj_y + np.dot(grad_y, z_proj - y) + 1.5 * step * np.dot(z_proj - y, z_proj - y):
            step *= 0.5
            z = y - step * grad_y
            z_proj = project_to_feasible(z, target_sum)
            obj_z, grad_z = smooth_obj_and_grad(z_proj, alpha)
            count += 1
            if count > 30:
                break
        
        x_new = z_proj
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = x_new + (t_param - 1.0) / t_new * (x_new - x)
        x = x_new
        x_prev = x
        t_param = t_new
        
        # Occasionally try to increase step
        if count == 0 and it % 50 == 0:
            step = min(step * 1.25, 2.0 / (2.0 * alpha))
        
        score = evaluate_score(x)
        if score < best_score:
            best_score = score
            best_x = x.copy()
    
    return best_x, best_score

def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    
    def constraint_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def run(seed=42, budget_s=1300, **kwargs):

        return budget_s - (time.time() - t0)

    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted - np.max(shifted))
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum
    obj = (m + (np.max(shifted) + np.log(w_sum)) / alpha) * dx
    
    grad = np.zeros(n)

    grad *= dx
    return obj, grad

        pass
    return None

        h_new = project_to_feasible(h_new, target_sum)
        score = evaluate_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_refine(h, alpha, maxiter=500):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_fn(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def grad_fn(hv):
        _, g = smooth_obj_and_grad(hv, alpha)
        return g

    def cfun(hv):
        return np.sum(hv) - target_sum

    def cjac(hv):
        return np.ones(n)

    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds, jac=grad_fn,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_to_feasible(init, n_init / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)
    best_n = n_init
    rng = np.random.RandomState(seed)

    phase1_time = budget_s * 0.38
    phase2_time = budget_s * 0.15
    phase3_time = budget_s * 0.22
    phase4_time = budget_s * 0.12
    phase5_time = budget_s * 0.10

    p1_end = t0 + phase1_time
    p2_end = p1_end + phase2_time
    p3_end = p2_end + phase3_time
    p4_end = p3_end + phase4_time
    p5_end = p4_end + phase5_time

    for alpha in [20, 50, 100, 300, 1000]:
        if time.time() > p1_end:
            break
        t_allow = min(p1_end - time.time() - 10, budget_s * 0.08)
        if t_allow < 15:
            break
        h, score = fista_optimize(best_h, alpha, 200000, t0, t_allow)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    for alpha in [50, 150, 500]:
        if time.time() > p2_end:
            break
        t_allow = min(p2_end - time.time() - 10, budget_s * 0.05)
        if t_allow < 15:
            break
        h, score = adam_optimize(best_h, alpha, 200000, t0, t_allow, lr=3e-4)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    slsqp_alphas = [100, 500, 2000, 10000, 50000]
    for alpha in slsqp_alphas:
        if time.time() > p3_end:
            break
        if p3_end - time.time() < 20:
            break
        h_ref = slsqp_refine(best_h, alpha, maxiter=2000)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    n_restarts = 10
    for r in range(n_restarts):
        if time.time() > p4_end:
            break
        t_allow = min(p4_end - time.time() - 5, budget_s * 0.015)
        if t_allow < 8:
            break
        h_p = best_h + rng.randn(best_n) * 0.008 / np.sqrt(best_n)
        h_p = project_to_feasible(h_p, best_n / 2.0)
        alpha = 200 + 200 * (r % 5)
        h, score = fista_optimize(h_p, alpha, 30000, t0, t_allow)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if p4_end - time.time() > 20:
            h_ref = slsqp_refine(h, 5000, maxiter=1000)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    for n_mult in [2, 3, 4]:
        if time.time() > p5_end:
            break
        n_new = best_n * n_mult
        if n_new > 3000:
            continue
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new_arr = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_init = np.interp(x_new_arr, x_old, best_h)
        h_init = project_to_feasible(h_init, n_new / 2.0)
        t_allow = min(p5_end - time.time() - 15, budget_s * 0.04)
        if t_allow < 10:
            break
        h, score = fista_optimize(h_init, 300, 50000, t0, t_allow)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            best_n = n_new
        if p5_end - time.time() > 25:
            h_ref = slsqp_refine(best_h, 5000, maxiter=1000)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    if remaining() > 30:
        for alpha in [10000, 50000, 200000]:
            if remaining() < 20:
                break
            h_ref = slsqp_refine(best_h, alpha, maxiter=1000)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
