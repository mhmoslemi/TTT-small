import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def logsumexp_gradient(h, corr, alpha):
    """Compute gradient of (1/alpha)*log(sum(exp(alpha*corr))) w.r.t. h"""
    n = len(h)
    shifted = alpha * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, range(len(corr))):
        lag = int(idx - (n - 1))
        grad += w * compute_gradient_for_lag(h, lag)
    return grad

def nesterov_optimize(h0, target_sum, alpha, max_iters, t0, time_limit, n_top=None):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    beta = 0.9
    step = 0.1
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        # Compute gradient at y (momentum point)
        y = project_to_feasible(h + beta * v, target_sum)
        corr = compute_correlations(y)
        
        if n_top is not None:
            # Focus on top lags
            idx = np.argsort(-corr)[:n_top]
            corr_top = corr[idx]
            m = np.max(corr_top)
            shifted = alpha * (corr_top - m)
            weights = np.exp(shifted)
            weights /= weights.sum()
            grad = np.zeros(n)
            for w, i in zip(weights, idx):
                lag = int(i - (n - 1))
                grad += w * compute_gradient_for_lag(y, lag)
        else:
            grad = logsumexp_gradient(y, corr, alpha)
        
        # Nesterov update with backtracking
        improved = False
        for trial in range(30):
            h_new = y - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                v = h_new - h
                h = h_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            step *= 1.5  # try larger step
            if step > 1.0:
                step = 1.0
                v *= 0.5
                beta = min(beta + 0.01, 0.95)
        
        # Periodically update momentum
        if it % 50 == 0:
            beta = 0.9 + 0.05 * (it % 100) / 100.0
    
    return best_h, best_score

    # coeff[m] = d corr[m] / d h[i]
    # rest[m] = corr[m] - h[i] * coeff[m] (approximately, for linear terms)

    coeff = np.zeros(2*n - 1)
    for m in range(2*n - 1):
        lag = m - (n - 1)
        c = 0.0
        if lag == 0:
            # C(0) = sum h[j]*(1-h[j]), term for j=i is h[i] - h[i]^2
            # dc/dh[i] = 1 - 2*h[i]
            c = 1.0 - 2.0 * h[i]
        elif lag > 0:
            # C(lag) = sum_{j=0}^{n-1-lag} h[j]*(1-h[j+lag])
            if i <= n - 1 - lag:
                c += (1.0 - h[i + lag])  # from j=i
            if i >= lag:
                c += (-h[i - lag])  # from j=i-lag, term is h[i-lag]*(1-h[i])
        else:
            la = -lag
            # C(-la) = sum_{j=la}^{n-1} h[j]*(1-h[j-la])
            # = sum_{k=0}^{n-1-la} h[k+la]*(1-h[k])
            if i >= la:
                c += (1.0 - h[i - la])  # from k+la=i, k=i-la, term is h[i]*(1-h[i-la])
            if i + la < n:
                c += (-h[i + la])  # from k=i, term is h[la+i]*(1-h[i]) = h[i+la]*(1-h[i])
        coeff[m] = c
    
    # Now the objective as function of t = h[i]:
    # f(t) = max_m (corr[m] + coeff[m]*(t - h[i])) * dx  [for linear terms]
    # But for m = n-1 (lag=0), it's quadratic: corr[n-1] + (t - h[i]) - (t² - h[i]²)
    # Actually let me be more careful.
    
    # For lag = 0: C(0) = A + t - t² where A = sum_{j≠i} h[j](1-h[j])
    # For other lags: C(k) = B_k + a_k * t where B_k = C(k) - a_k * h[i]
    
    # So f(t) = max(A + t - t², max_{k≠0} (B_k + a_k * t))
    
    # Let me compute A and the (B_k, a_k) pairs
    m0 = n - 1  # index for lag 0
    A = corr[m0] - h[i] * (1 - h[i])  # constant part
    # Actually: C(0) = sum_j h[j](1-h[j]) = A + h[i](1-h[i])
    # So A = C(0) - h[i](1-h[i])
    
    # For other lags, C(k) is linear in h[i]: C(k) = B_k + a_k * h[i]
    # So B_k = C(k) - a_k * h[i]
    
    # Candidate points for optimal t:
    candidates = [0.0, 1.0, 0.5]
    
    # Intersection of quadratic with each line:
    # A + t - t² = B_k + a_k*t
    # -t² + (1-a_k)*t + (A - B_k) = 0
    # t² - (1-a_k)*t - (A-B_k) = 0
    # t = [(1-a_k) ± sqrt((1-a_k)² + 4(A-B_k))] / 2
    
    for m in range(2*n - 1):
        if m == m0:
            continue
        a_k = coeff[m]
        B_k = corr[m] - a_k * h[i]
        discriminant = (1.0 - a_k)**2 + 4.0 * (A - B_k)
        if discriminant >= 0:
            sqrt_d = np.sqrt(discriminant)
            t1 = ((1.0 - a_k) + sqrt_d) / 2.0
            t2 = ((1.0 - a_k) - sqrt_d) / 2.0
            if 0.0 <= t1 <= 1.0:
                candidates.append(t1)
            if 0.0 <= t2 <= 1.0:
                candidates.append(t2)
    
    # Evaluate f(t) at all candidates and pick the best
    best_t = h[i]
    best_val = np.inf
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        # Compute f(t)
        c0 = A + t - t*t
        # For other lags: B_k + a_k*t = corr[m] - a_k*h[i] + a_k*t = corr[m] + a_k*(t-h[i])
        vals = corr + coeff * (t - h[i])
        vals[m0] = c0  # override with exact quadratic
        val = np.max(vals)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

def evaluate_score(h):

    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def coord_descent_sweep(h, n, order=None):
    """Perform one full coordinate descent sweep."""
    target_sum = n / 2.0
    if order is None:
        order = np.arange(n)
    
    corr = compute_correlations(h)
    
    for i in order:
        t_opt = coord_optimize(n, h, i, corr)
        if t_opt != h[i]:

            h[i] = t_opt
            # Recompute corr (simplest approach)
            corr = compute_correlations(h)
    
    return h

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def coord_optimize(n, h, i, corr):
    """Find optimal h[i] given all else fixed. Returns optimal t."""
    m0 = n - 1
    A = corr[m0] - h[i] * (1.0 - h[i])
    
    # Compute coefficients a[m] for all m
    a = np.zeros(2 * n - 1)
    if n > 1:
        k_vals = np.arange(1, n)
        idx_p = i + k_vals
        idx_m = i - k_vals
        mask_p = idx_p < n
        mask_m = idx_m >= 0
        
        # a[m0-k] (lag = -k): (1-h[i-k])*[i>=k] - h[i+k]*[i+k<n]
        a[:0]  # placeholder
        neg_lags = np.where(mask_m, 1.0 - h[idx_m], 0.0) - np.where(mask_p, h[idx_p], 0.0)
        pos_lags = np.where(mask_p, 1.0 - h[idx_p], 0.0) - np.where(mask_m, h[idx_m], 0.0)
        
        a[m0 + k_vals] = pos_lags
        a[m0 - k_vals] = neg_lags
    
    a[m0] = 1.0 - 2.0 * h[i]
    
    # B[m] = corr[m] - a[m]*h[i]
    B = corr - a * h[i]
    B[m0] = A
    
    # Generate candidates: intersections of quadratic with lines
    # A + t - t² = B[m] + a[m]*t
    # t² - (1-a[m])*t + (B[m]-A) = 0
    p = 1.0 - a
    q = B - A
    disc = p * p - 4.0 * q
    
    nc = 2 * n - 1
    m_indices = np.arange(nc)
    valid = (disc >= 0.0) & (m_indices != m0)
    
    candidates = np.array([0.0, 1.0, 0.5])
    
    if np.any(valid):
        sd = np.sqrt(np.maximum(disc, 0.0))
        t1 = (p + sd) / 2.0
        t2 = (p - sd) / 2.0
        
        valid1 = valid & (t1 >= 0.0) & (t1 <= 1.0)
        valid2 = valid & (t2 >= 0.0) & (t2 <= 1.0)
        
        if np.any(valid1):
            candidates = np.concatenate([candidates, t1[valid1]])
        if np.any(valid2):
            candidates = np.concatenate([candidates, t2[valid2]])
    
    # Evaluate f(t) at all candidates
    # f(t) = max(A + t - t², max_{m≠m0}(B[m] + a[m]*t))
    best_t = h[i]
    best_val = np.inf
    
    # Vectorized evaluation
    candidates = np.unique(np.clip(candidates, 0.0, 1.0))
    quad_vals = A + candidates - candidates * candidates  # quadratic value at each candidate
    line_vals = B[None, :] + a[None, :] * candidates[:, None]  # (n_cand, n_corr)
    line_vals[:, m0] = -np.inf  # exclude m0
    line_max = np.max(line_vals, axis=1)
    f_vals = np.maximum(quad_vals, line_max)
    
    best_idx = np.argmin(f_vals)
    best_t = candidates[best_idx]
    best_val = f_vals[best_idx]
    
    return best_t

        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]

        weights /= weights.sum()
        
        grad = np.zeros(n)

            lag = int(idx - (n - 1))
            grad += w * compute_gradient_for_lag(h, lag)

def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    
    def constraint_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def coord_optimize(n, h, i, corr):
    m0 = n - 1
    A = corr[m0] - h[i] * (1.0 - h[i])
    nc = 2 * n - 1
    a = np.zeros(nc)
    if n > 1:
        k_vals = np.arange(1, n)
        idx_p = i + k_vals
        idx_m = i - k_vals
        mask_p = idx_p < n
        mask_m = idx_m >= 0
        np_ = np.where(mask_p, 1.0 - h[idx_p], 0.0)
        nm_ = np.where(mask_m, 1.0 - h[idx_m], 0.0)
        hp_ = np.where(mask_p, h[idx_p], 0.0)
        hm_ = np.where(mask_m, h[idx_m], 0.0)
        a[m0 + k_vals] = np_ - hm_
        a[m0 - k_vals] = nm_ - hp_
    a[m0] = 1.0 - 2.0 * h[i]
    B = corr - a * h[i]
    B[m0] = A
    p = 1.0 - a
    q = B - A
    disc = p * p - 4.0 * q
    m_indices = np.arange(nc)
    valid = (disc >= 0.0) & (m_indices != m0)
    candidates = [0.0, 1.0, 0.5]
    if np.any(valid):
        sd = np.sqrt(np.maximum(disc, 0.0))
        t1 = (p + sd) / 2.0
        t2 = (p - sd) / 2.0
        v1 = valid & (t1 >= 0.0) & (t1 <= 1.0)
        v2 = valid & (t2 >= 0.0) & (t2 <= 1.0)
        if np.any(v1):
            candidates.extend(t1[v1].tolist())
        if np.any(v2):
            candidates.extend(t2[v2].tolist())
    candidates = np.array(candidates)
    candidates = np.unique(np.clip(candidates, 0.0, 1.0))
    quad_vals = A + candidates - candidates * candidates
    line_vals = B[None, :] + a[None, :] * candidates[:, None]
    line_vals[:, m0] = -np.inf
    line_max = np.max(line_vals, axis=1)
    f_vals = np.maximum(quad_vals, line_max)
    best_idx = np.argmin(f_vals)
    return candidates[best_idx]


def coord_descent(h, n, target_sum, order, max_sweeps, t0, deadline):
    corr = compute_correlations(h).copy()
    best_score = evaluate_score(h)
    best_h = h.copy()
    stall_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        changed = 0
        for i in order:
            t_opt = coord_optimize(n, h, i, corr)
            if abs(t_opt - h[i]) > 1e-10:
                h[i] = t_opt
                changed += 1
        if changed == 0:
            break
        corr = compute_correlations(h)
        h = project_to_feasible(h, target_sum)
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
    return best_h, best_score


def subgradient_run(h0, max_iters, alpha, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * compute_gradient_for_lag(h, lag)
        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-12:
            break
        improved = False
        for trial in range(20):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.4
        if not improved:
            stall += 1
            if stall > 200:
                h = best_h + rng.randn(n) * (0.02 + 0.05 * stall / 300.0) / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init * (0.5 + 0.5 * rng.rand())
                stall = 0
        if it > 0 and it % 500 == 0:
            step = max(step_init * 0.93 ** (it // 500), step_init * 0.02)
    return best_h, best_score


def slsqp_refine(h, alpha, maxiter=150):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
