def slsqp_refine(h, alpha, maxiter=500):

    target_sum = n / 2.0

    def obj(hv):
        corr = compute_correlations(hv)
        m = corr.max()
        return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m)))) + m
    
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun}],

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def slsqp_refine(h, alpha, maxiter=400):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def slsqp_refine_multi(h, alphas, maxiter_base=400):
    best = h.copy()
    best_score = evaluate_score(h)

        if result is not None:
            score = evaluate_score(result)
            if score < best_score:

                best = result.copy()
    return best, best_score


def fourier_perturb(h, rng, n_modes=4, scale=0.03):
    n = len(h)
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 3))
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = scale * rng.uniform(0.5, 1.5)
        pert += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return h + pert


def block_perturb(h, rng, n_blocks=3, block_frac=0.1):
    n = len(h)

        start = rng.randint(0, n - block_size)
        shift = (rng.rand() - 0.5) * 0.3
        h_new[start:start + block_size] += shift
    return h_new
