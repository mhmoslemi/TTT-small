    target = n // 2
    dx = 2.0 / n

    h = np.zeros(n)

    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = np.max(corr)
    current_c5 = current_max * dx
    best_h = h.copy()
    best_c5 = current_c5
    best_max = current_max
    
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        
        # Compute delta patterns for all positions
        # For position p with h[p]=1 (going 1→0): delta = h[?] - (1-h[?])
        # For position p with h[p]=0 (going 0→1): delta = (1-h[?]) - h[?]
        
        # Build delta matrix: deltas[p, k] = change in corr[k] when h[p] flips

            g = 1.0 - h

            k_hi = min(corr_len - 1, corr_len - 1 - p)
            if k_lo <= k_hi:

                deltas[p, k_lo:k_hi+1] += d * g[idx]
