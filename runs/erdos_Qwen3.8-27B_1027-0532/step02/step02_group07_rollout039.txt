def active_set_qp(h, alpha=100):
    """Solve: min sum_{k in A} corr[k] / |A| using SLSQP
    where A = {k: corr[k] > max_corr - 1/alpha}"""
    n = len(h)
    corr = compute_correlations(h)
    max_corr = np.max(corr)
    A = np.where(corr > max_corr - 1.0/alpha)[0]
    
    # Objective: sum of corr[k] for k in A
    # corr[k] = sum_i h[i]*(1-h[i+k]) for valid i
    # This is a quadratic function of h
    
    def obj(h):
        c = compute_correlations(h)
        return np.sum(c[A]) / len(A)
    
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
                   options={'maxiter': 500, 'ftol': 1e-15})
    return res.x

def active_set_method(h, max_iter=20, alpha_start=50, alpha_end=10000):
    """Iteratively solve the active set QP"""
    h = h.copy()
    n = len(h)
    for it in range(max_iter):
        log_alpha = np.log(np.linspace(alpha_start, alpha_end, max_iter))
        alpha = float(np.exp(log_alpha[it]))
        h = active_set_qp(h, alpha)
        h = project_sum(h, n/2)
    return h

    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x if res.x is not None else None
