```python
    dx = 2.0 / n
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_local = evaluate_score(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        grad = compute_gradient_efficient(h, weights, dx)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = project_onto_constraint(h_new, target)
        
        # Check improvement
        if it % 50 == 0 or it == max_iters - 1:
            sc = evaluate_score(h)
            if sc < best_local:
                best_local = sc
    
    return h, best_local
```

```python
def minmax_alternation(h, target, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        # Find worst overlap
        overlaps = compute_overlaps(h)
        k_star = np.argmax(overlaps)
        offset = k_star - (n - 1)
        
        # Minimize this specific overlap using L-BFGS
        def obj(hv):
            if offset == 0:
                return -np.sum(hv * (1.0 - hv)) * dx + 100.0 * (hv.sum() - target)**2
            elif offset > 0:
                return -np.sum(hv[:-offset] * (1.0 - hv[offset:])) * dx + 100.0 * (hv.sum() - target)**2
            else:
                off = -offset
                return -np.sum(hv[off:] * (1.0 - hv[:n-off])) * dx + 100.0 * (hv.sum() - target)**2
        
        def obj_grad(hv):
            if offset == 0:
                g = -(1.0 - 2.0 * hv)
            elif offset > 0:
                g = np.zeros(n)
                g[:-offset] -= (1.0 - hv[offset:])
                g[offset:] += hv[:-offset]
            else:
                off = -offset
                g = np.zeros(n)
                g[off:] -= (1.0 - hv[:n-off])
                g[:n-off] += hv[off:]
            g *= dx
            g += 200.0 * (hv.sum() - target)
            return g
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='L-BFGS-B', jac=obj_grad, bounds=bounds,
                       options={'maxiter': 50, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 20})
        h_new = res.x if res.x is not None else h
        h = project_onto_constraint(np.clip(h_new, 0, 1), target)
    
    return h
```

```python
    # Phase 5: Sharpen + refine
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_sharp = sharpen_binary(best_h, n_target / 2.0)
        s = evaluate_score(h_sharp)
        if s < best_score:
            best_score = s
            best_h = h_sharp.copy()
        h = adam_optimize(best_h, 2000, n_target / 2.0, time.time() + min(25, remaining - 40), lr=0.01, max_iters=20000, top_k=50)
        h = project_onto_constraint(h, n_target / 2.0)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 6: Final min-max
    remaining = budget_s - (time.time()
    if remaining > 50:
        h = minmax_lbfgs(best_h, n_target / 2.0, time.time() + min(35, remaining - 20), max_rounds=30)
        h = project_onto_constraint(h, n_target / 2.0)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 7: Very high alpha Adam
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        for alpha in [5000, 15000]:
            if time.time() - t0 > budget_s - 15:
                break
            h = adam_optimize(best_h, alpha, n_target / 2.0, time.time() + min(10, (budget_s - 15 - (time.time() - t0))), lr=0.005, max_iters=10000, top_k=30)
            h = project_onto_constraint(h, n_target / 2.0)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
```
