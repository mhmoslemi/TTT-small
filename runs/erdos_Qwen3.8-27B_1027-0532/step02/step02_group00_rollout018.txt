import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.exp(alpha * (corr - c_max))
    return c_max + np.log(np.sum(shifted)) / alpha

def compute_gradient_topk(h, active_lags, weights=None):
    """Compute gradient of weighted sum of overlaps at active lags."""
    n = len(h)
    grad = np.zeros(n)
    if weights is None:
        weights = np.ones(len(active_lags)) / len(active_lags)
    one_minus_h = 1.0 - h
    for w, lag_idx in zip(weights, active_lags):
        lag = lag_idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n-lag] += w * one_minus_h[lag:]
                grad[lag:] += w * (-h[:n-lag])
        else:
            k = -lag
            if k < n:
                grad[k:] += w * one_minus_h[:n-k]
                grad[:n-k] += w * (-h[k:])
    return grad

def project_and_check(h, target_sum):
    h = project_feasible(h, target_sum)
    return h

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha, 
                     noise_prob=0.0, noise_scale=0.05, patience=100):
    """Subgradient method with adaptive learning rate and noise."""
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        
        # Find top-k lags
        top_indices = np.argsort(corr)[::-1][:topk]
        top_values = corr[top_indices]
        
        # Compute weights (softmax over top-k)
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w /= np.sum(w)
        
        # Compute gradient
        grad = compute_gradient_topk(h, top_indices, w)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Optional noise
        if noise_prob > 0 and np.random.random() < noise_prob:
            h_new = h_new + np.random.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        
        # Evaluate
        score = compute_c5(h_new)
        
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.05, 10.0)  # increase lr slightly
        else:
            stagnation += 1
            lr *= 0.95  # decrease lr
        
        h = h_new
        
        if stagnation > patience:
            break
    
    return best_h, best_score

            weights = shifted / np.sum(shifted)
            # Gradient of smooth obj = sum_k weights[k] * grad_corr[k]
            return compute_gradient_topk(x, np.arange(2*n-1), weights)

        try:

import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_lag(h, lag):
    """Compute gradient of overlap at a single lag."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    if lag >= 0:
        if lag < n:
            grad[lag:] += one_minus_h[:n-lag]
            grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n-m] += one_minus_h[m:]
            grad[m:] -= h[:n-m]
    return grad

def compute_grad_weighted(h, lags, weights):
    """Compute weighted gradient over multiple lags."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for w, lag_idx in zip(weights, lags):
        if w < 1e-12:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n-lag]
                grad[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n-m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n-m]
    return grad

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2*n-1), weights)

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
