import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_softmax_gradient(h, dx, alpha, n_top):
    """Compute gradient of soft-max approximation using top lags."""
    n = len(h)
    corr = compute_correlations(h)
    g = corr * dx  # individual overlap values
    
    # Find top lags
    top_idx = np.argsort(-g)[:n_top]
    top_g = g[top_idx]
    
    # Soft-max weights
    shifted = top_g - top_g[0]
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂g_lag/∂h[j] = dx*(1-h[j+lag]) for j in [0, n-1-lag]
            #              - dx*h[j-lag] for j in [lag, n-1]
            grad[:n-lag] += w * dx * (1.0 - h[lag:])
            grad[lag:] += w * dx * (-h[:n-lag])
        else:
            m = -lag
            # ∂g_{-m}/∂h[j] = dx*(1-h[j-m]) for j in [m, n-1]
            #                - dx*h[j+m] for j in [0, n-1-m]
            grad[m:] += w * dx * (1.0 - h[:n-m])
            grad[:n-m] += w * dx * (-h[m:])
    
    return grad, g, top_idx

def bb_step_and_project(h, grad, prev_h, prev_grad, step_init, target_sum):
    """Barzili-Borwein step size with projection."""
    s = h - prev_h
    y = grad - prev_grad
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    ss = np.dot(s, y)  # Wait, BB uses s^T y / y^T y for the "B1" version
    # Actually BB step: α = (s^T * s) / (s^T * y) for the "B2" version
    # or α = (s^T * y) / (y^T * y) for the "B1" version
    
    if np.abs(yy) > 1e-12:
        alpha_bb = sy / yy if np.abs(sy) > 1e-12 else step_init
    else:
        alpha_bb = step_init
    
    alpha_bb = np.clip(alpha_bb, step_init * 0.01, step_init * 10)
    
    h_new = h - alpha_bb * grad
    h_new = project_to_feasible(h_new, target_sum)
    return h_new, alpha_bb

def subgradient_bb(h0, alpha, n_top, max_iters, step_init, seed, t0, time_limit):
    """Subgradient method with BB step size and soft-max smoothing."""
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    stall = 0
    
    prev_grad = np.zeros(n)
    prev_h = h.copy()
    step = step_init
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        grad, g, top_idx = compute_softmax_gradient(h, 2.0/n, alpha, n_top)
        
        # BB step
        s = h - prev_h
        y = grad - prev_grad
        sy = np.dot(s, y)
        yy = np.dot(y, y)
        
        if yy > 1e-12 and abs(sy) > 1e-12 and sy > 0:
            step = np.clip(sy / yy, step * 0.1, step * 10)
        
        h_new = h - step * grad
        h_new = project_to_feasible(h_new, target_sum)
        
        score = evaluate_score(h_new)
        if score < best_score - 1e-10:
            best_score = score
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
        prev_h = h_new.copy()
        prev_grad = grad.copy()
        
        if stall > 100:
            # Perturb and restart
            h = best_h + rng.randn(n) * 0.05 / np.sqrt(n)
            h = project_to_feasible(h, target_sum)
            step = step_init
            prev_h = h.copy()
            prev_grad = np.zeros(n)
            stall = 0
    
    return best_h, best_score

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, dx, alpha, n_top):
    """Compute soft-max gradient using top n_top lags."""
    n = len(h)
    corr = compute_correlations(h)
    g = corr * dx
    
    # Soft-max weights for top lags
    top_idx = np.argsort(-g)[:n_top]
    top_g = g[top_idx]
    shifted = top_g - top_g.max()
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):

        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂g_lag/∂h[j] = dx*(1-h[j-lag])*I(lag<=j<=n-1) - dx*h[j+lag]*I(0<=j<=n-1-lag)

            grad[:n-lag] += w * dx * (-h[lag:])

            # ∂g_{-m}/∂h[j] = dx*(1-h[j+m])*I(0<=j<=n-1-m) - dx*h[j-m]*I(m<=j<=n-1)
            grad[:n-m] += w * dx * (1.0 - h[m:])
            grad[m:] += w * dx * (-h[:n-m])

    return None

    target_sum = n / 2.0

    best_score = evaluate_score(best)

def lbfgs_refine(h, alpha, beta, maxiter):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        g = corr * dx
        m = g.max()
        shifted = g - m
        soft = m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
        penalty = beta * (np.sum(hv) - target_sum) ** 2
        return soft + penalty

    def jac(hv):
        grad, _ = softmax_grad(hv, dx, alpha, min(30, 2 * n - 1))
        grad += beta * 2.0 * (np.sum(hv) - target_sum)
        return grad

    res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

        prev_h = h.copy()
        prev_grad = grad.copy()

        if stall > 80:
            noise_scale = 0.02 + 0.04 * min(stall / 200.0, 1.0)
            h = best_h + rng.randn(n) * noise_scale / np.sqrt(n)
            h = project_to_feasible(h, target_sum)
            prev_h = h.copy()
            prev_grad = np.zeros(n)
            step = step_init
            stall = 0

    return best_h, best_score


def coordinate_wise_refine(h, n_passes=2):
    n = len(h)
    target_sum = n / 2.0
    best = h.copy()
    best_score = evaluate_score(best)
    rng = np.random.RandomState(12345)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.2, -0.01, -0.02, -0.05, -0.1, -0.2]
    for _ in range(n_passes):
        perm = rng.permutation(n)
        for j in perm:
            for d in deltas:
                trial = best.copy()
                val = trial[j] + d
                if val < 0 or val > 1:
                    continue
                trial[j] = val
                trial = project_to_feasible(trial, target_sum)
                s = evaluate_score(trial)
                if s < best_score - 1e-12:
                    best = trial
                    best_score = s
    return best, best_score


def lbfgs_refine(h, alpha, beta, maxiter):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        g = corr * dx
        m = g.max()
        shifted = g - m
        soft = m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
        penalty = beta * (np.sum(hv) - target_sum) ** 2
        return soft + penalty

    def jac(hv):
        grad, _ = softmax_grad(hv, dx, alpha, min(30, 2 * n - 1))
        grad += beta * 2.0 * (np.sum(hv) - target_sum)
        return grad

    res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_to_feasible(init, n_init / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)
    best_n = n_init

    rng = np.random.RandomState(seed)

    configs = [
        (5, 300, 10, 0.05),
        (10, 100, 10, 0.04),
        (5, 500, 8, 0.03),
        (10, 50, 15, 0.05),
        (20, 30, 10, 0.08),
        (50, 15, 10, 0.1),
        (20, 200, 20, 0.03),
        (100, 10, 8, 0.15),
        (200, 5, 10, 0.08),
        (500, 3, 5, 0.05),
        (200, 20, 15, 0.04),
        (1000, 5, 10, 0.03),
    ]

    for i, (alpha_end, max_iter_equiv, n_top, step_init) in enumerate(configs):
        if remaining() < budget_s * 0.35:
            break
        t_allow = min(remaining() * 0.12, 100)
        if t_allow < 10:
            break

        start_h = best_h.copy()
        if i > 0 and i % 4 == 0:
            start_h = best_h + rng.randn(len(best_h)) * 0.04 / np.sqrt(len(best_h))
            start_h = project_to_feasible(start_h, len(start_h) / 2.0)
        elif i > 0:
            start_h = best_h + rng.randn(len(best_h)) * 0.015 / np.sqrt(len(best_h))
            start_h = project_to_feasible(start_h, len(start_h) / 2.0)

        alpha_start = alpha_end * 0.1
        h, score = project_gradient_descent(start_h, t0, t_allow, alpha_start, alpha_end,
                                            n_top, step_init, seed + i * 777)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    if remaining() > budget_s * 0.28:
        for mult in [2, 3]:
            n_fine = best_n * mult
            if n_fine > 2500 or remaining() < 80:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_fine, endpoint=False)
            h_fine_init = np.interp(x_new, x_old, best_h)
            h_fine_init = project_to_feasible(h_fine_init, n_fine / 2.0)
            t_allow = min(remaining() * 0.3, 250)
            if t_allow < 15:
                break
            h_fine, score = project_gradient_descent(h_fine_init, t0, t_allow, 5, 200,
                                                     15, 0.03, seed + 50000 + mult)
            if score < best_score:
                best_score = score
                best_h = h_fine.copy()
                best_n = n_fine

    if remaining() > 50:
        h_ref, score = coordinate_wise_refine(best_h, n_passes=2)
        if score < best_score:
            best_score = score
            best_h = h_ref.copy()

    if remaining() > 40:
        for alpha in [100, 500]:
            if remaining() < 20:
                break
            h_ref = lbfgs_refine(best_h, alpha, 1000, 50)
            if h_ref is not None:
                score = evaluate_score(h_ref)
                if score < best_score:
                    best_score = score
                    best_h = h_ref.copy()

    if remaining() > 30:
        h, score = project_gradient_descent(best_h, t0, remaining() - 5, 50, 2000,
                                            20, 0.02, seed + 99999)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    if remaining() > 20:
        h_ref, score = coordinate_wise_refine(best_h, n_passes=3)
        if score < best_score:
            best_score = score
            best_h = h_ref.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
