```python
    n = len(h)

    coeff_i = get_coeff(i)
    coeff_j = get_coeff(j)
    d = coeff_j - coeff_i  # change in c_m per unit δ
    
    # c_m(δ) = overlaps[m] + δ * d[m]
    # Maximize... wait, we want to minimize max_m (overlaps[m] + δ * d[m])
    # Constraints: -h[i] ≤ δ ≤ 1 - h[j]
    
    low = -h[i]
    high = 1.0 - h[j]
    
    # Find the minimum of max_m (overlaps[m] + δ * d[m]) over δ ∈ [low, high]
    # This is a convex piecewise-linear function.
    # The minimum is where the subgradient contains 0, or at a breakpoint, or at a boundary.
    
    # Breakpoints: where overlaps[m1] + δ*d[m1] = overlaps[m2] + δ*d[m2]
    # δ = (overlaps[m2] - overlaps[m1]) / (d[m1] - d[m2])
    
    # Instead of computing all breakpoints (O(n²)), use a simpler approach:
    # Evaluate at a grid + refine
    
    # Actually, for efficiency, let's use the "upper envelope" approach:
    # The minimum of the upper envelope of lines occurs at a vertex.
    # We can find it by: the optimal δ* satisfies:
    #   min_m d[m] ≤ 0 ≤ max_m d[m] (if interior)
    #   and the active set at δ* has d values straddling 0
    
    # Simpler: use ternary search or binary search on the subgradient.
    # The subgradient at δ is the convex hull of {d[m] : m is active at δ}
    # where active means c_m(δ) = max.
    
    # Let's just use a fine grid for simplicity (n is only 425)
    grid = np.linspace(low, high, 200)
    vals = overlaps[None, :] + grid[:, None] * d[None, :]
    obj = np.max(vals, axis=1)
    best_idx = np.argmin(obj)
    best_delta = grid[best_idx]
    
    # Refine around the best point
    if 0 < best_idx < len(grid) - 1:
        local_grid = np.linspace(grid[best_idx-1], grid[best_idx+1], 100)
        vals = overlaps[None, :] + local_grid[:, None] * d[None, :]
        obj = np.max(vals, axis=1)
        best_delta = local_grid[np.argmin(obj)]
    
    h[i] -= best_delta
    h[j] += best_delta
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0


def sum_constraint_jac(h):
    return np.ones(len(h))


def lbfgsb_optimize(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def compute_pair_delta(h, i, j, overlaps):
    """Compute the change vector d[m] for a pair swap i->j."""
    n = len(h)
    L = np.arange(2 * n - 1) - (n - 1)
    
    # Coefficient of h[pos] in each overlap
    # For overlap m: c_m = sum over valid i of h[i]*(1-h[i+L])
    # h[pos] appears as:
    #   h[pos]*(1-h[pos+L]) if pos+L valid -> coeff += (1-h[pos+L])
    #   h[pos-L]*(1-h[pos]) -> coeff -= h[pos-L] if pos-L valid
    
    pos_i_L = i + L
    pos_i_mL = i - L
    valid_i_plus = (pos_i_L >= 0) & (pos_i_L < n)
    valid_i_minus = (pos_i_mL >= 0) & (pos_i_mL < n)
    
    pos_j_L = j + L
    pos_j_mL = j - L
    valid_j_plus = (pos_j_L >= 0) & (pos_j_L < n)
    valid_j_minus = (pos_j_mL >= 0) & (pos_j_mL < n)
    
    d = np.zeros(2 * n - 1)
    d[valid_i_plus] -= (1.0 - h[pos_i_L[valid_i_plus]])
    d[valid_i_minus] += h[pos_i_mL[valid_i_minus]]
    d[valid_j_plus] += (1.0 - h[pos_j_L[valid_j_plus]])
    d[valid_j_minus] -= h[pos_j_mL[valid_j_minus]]
    
    return d


def pair_swap_optimize(h, i, j, overlaps, n_grid=100):
    """Find optimal delta for pair swap i->j, update h and overlaps."""
    n = len(h)
    d = compute_pair_delta(h, i, j, overlaps)
    
    low = -h[i]
    high = 1.0 - h[j]
    if low >= high:
        return h, overlaps, 0.0
    
    grid = np.linspace(low, high, n_grid)
    vals = overlaps[None, :] + grid[:, None] * d[None, :]
    obj = np.max(vals, axis=1)
    best_idx = np.argmin(obj)
    best_delta = grid[best_idx]
    
    # Refine
    if 0 < best_idx < n_grid - 1:
        local = np.linspace(grid[best_idx - 1], grid[best_idx + 1], 50)
        vals = overlaps[None, :] + local[:, None] * d[None, :]
        obj = np.max(vals, axis=1)
        best_delta = local[np.argmin(obj)]
    
    h[i] -= best_delta
    h[j] += best_delta
    overlaps += best_delta * d
    
    return h, overlaps, best_delta


def pair_swap_search(h, n_steps, rng, n_grid=100):
    """Perform random pair swap local search."""
    n = len(h)
    h = h.copy()
    overlaps = compute_overlaps(h)
    best_h = h.copy()
    best_obj = float(np.max(overlaps))
    
    for step in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h, overlaps, delta = pair_swap_optimize(h, i, j, overlaps, n_grid)
        if abs(delta) < 1e-12:
            continue
        obj = float(np.max(overlaps))
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    
    return best_h, best_obj


def coord_descent_with_penalty(h, penalty, n_passes=1, n_grid=50, rng=None):
    """Coordinate descent with soft penalty for sum constraint."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    for _pass in range(n_passes):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            s_j = float(np.sum(h)) - h[j]
            overlaps = compute_overlaps(h)
            
            # Compute coefficient of h[j] in each overlap
            L = np.arange(2 * n - 1) - (n - 1)
            j_L = j + L
            j_mL = j - L
            valid_plus = (j_L >= 0) & (j_L < n)
            valid_minus = (j_mL >= 0) & (j_mL < n)
            
            coeff = np.zeros(2 * n - 1)
            coeff[valid_plus] = 1.0 - h[j_L[valid_plus]]
            coeff[valid_minus] -= h[j_mL[valid_minus]]
            
            b = overlaps - h[j] * coeff
            
            # Minimize: max_m(coeff[m]*t + b[m]) + penalty*(s_j + t - target)^2
            # over t in [0, 1]
            
            candidates = np.array([0.0, 1.0])
            # Add critical points
            t_cands = target - s_j - coeff / (2.0 * penalty)
            t_cands = t_cands[(t_cands > 0.0) & (t_cands < 1.0)]
            if len(t_cands) > 0:
                candidates = np.concatenate([candidates, t_cands])
            
            # Also add a few grid points for robustness
            grid = np.linspace(0.0, 1.0, 20)
            candidates = np.concatenate([candidates, grid])
            candidates = np.unique(np.sort(candidates))
            
            vals = coeff[None, :] * candidates[:, None] + b[None, :]
            max_vals = np.max(vals, axis=1)
            pen_vals = penalty * (s_j + candidates - target) ** 2
            obj = max_vals + pen_vals
            
            best_idx = np.argmin(obj)
            h[j] = candidates[best_idx]
    
    return h
```

Δc_m = -δ * [coefficient of h[i] in c_m] + δ * [coefficient of h[j] in c_m]

The coefficient of h[pos] in c_m:
- If pos+L is a valid index (0 ≤ pos+L < n): h[pos] appears as the first factor, contributing (1-h[pos+L])
- If pos-L is a valid index (0 ≤ pos-L < n): h[pos] appears as the second factor in h[pos-L]*(1-h[pos]), contributing -h[pos-L]

So coeff_m(pos) = (1-h[pos+L]) * I[0≤pos+L<n] - h[pos-L] * I[0≤pos-L<n]

This is what I have in `compute_pair_delta`. Let me verify with a small example.

n=3, h = [a, b, c]
Lags m=0,1,2,3,4 correspond to L=-2,-1,0,1,2

For m=1 (L=-1):
c_1 = Σ_i h[i]*(1-h[i-1]) where i-1 must be valid
= h[1]*(1-h[0]) + h[2]*(1-h[1])
= b*(1-a) + c*(1-b)

Coefficient of h[0]=a in c_1: from b*(1-a), the coefficient is -b
So coeff_1(0) = -h[1] = -b ✓ (using formula: pos-L = 0-(-1) = 1, valid, so -h[1] = -b)

Coefficient of h[1]=b in c_1: from b*(1-a) → (1-a); from c*(1-b) → -c
So coeff_1(1) = (1-a) - c
Using formula: pos+L = 1+(-1) = 0, valid → (1-h[0]) = (1-a); pos-L = 1-(-1) = 2, valid → -h[2] = -c
So coeff_1(1) = (1-a) - c ✓

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 3.0
    hi = float(np.max(h)) + 3.0
    for _ in range(300):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0


def sum_constraint_jac(h):
    return np.ones(len(h))


def lbfgsb_optimize(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None


def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def compute_pair_delta(h, i, j):
    n = len(h)
    L = np.arange(2 * n - 1) - (n - 1)
    pos_i_L = i + L
    pos_i_mL = i - L
    vi_p = (pos_i_L >= 0) & (pos_i_L < n)
    vi_m = (pos_i_mL >= 0) & (pos_i_mL < n)
    pos_j_L = j + L
    pos_j_mL = j - L
    vj_p = (pos_j_L >= 0) & (pos_j_L < n)
    vj_m = (pos_j_mL >= 0) & (pos_j_mL < n)
    d = np.zeros(2 * n - 1)
    d[vi_p] -= (1.0 - h[pos_i_L[vi_p]])
    d[vi_m] += h[pos_i_mL[vi_m]]
    d[vj_p] += (1.0 - h[pos_j_L[vj_p]])
    d[vj_m] -= h[pos_j_mL[vj_m]]
    return d


def find_best_delta(overlaps, d, low, high, n_grid):
    if low >= high:
        return 0.0
    grid = np.linspace(low, high, n_grid)
    vals = overlaps[None, :] + grid[:, None] * d[None, :]
    obj = np.max(vals, axis=1)
    best_idx = int(np.argmin(obj))
    best_delta = grid[best_idx]
    if 0 < best_idx < n_grid - 1:
        local = np.linspace(grid[best_idx - 1], grid[best_idx + 1], 60)
        vals2 = overlaps[None, :] + local[:, None] * d[None, :]
        obj2 = np.max(vals2, axis=1)
        best_delta = local[int(np.argmin(obj2))]
        if 0 < int(np.argmin(obj2)) < 59:
            ii = int(np.argmin(obj2))
            local2 = np.linspace(local[ii - 1], local[ii + 1], 30)
            vals3 = overlaps[None, :] + local2[:, None] * d[None, :]
            obj3 = np.max(vals3, axis=1)
            best_delta = local2[int(np.argmin(obj3))]
    return best_delta


def pair_swap_deterministic(h, n_steps, rng, n_grid=80):
    n = len(h)
    h = h.copy()
    overlaps = compute_overlaps(h)
    best_h = h.copy()
    best_obj = float(np.max(overlaps))
    stuck_count = 0
    for step in range(n_steps):
        if step % 500 == 499:
            overlaps = compute_overlaps(h)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = compute_pair_delta(h, i, j)
        low = -h[i]
        high = 1.0 - h[j]
        if low >= high:
            continue
        best_delta = find_best_delta(overlaps, d, low, high, n_grid)
        if abs(best_delta) < 1e-14:
            stuck_count += 1
            continue
        h[i] -= best_delta
        h[j] += best_delta
        overlaps += best_delta * d
        obj = float(np.max(overlaps))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
    return best_h, best_obj


def pair_swap_sa(h, n_steps, rng, n_grid=80, temp_init=0.0005):
    n = len(h)
    h = h.copy()
    overlaps = compute_overlaps(h)
    best_h = h.copy()
    best_obj = float(np.max(overlaps))
    temp = temp_init
    decay_per = max(1, n_steps // 200)
    for step in range(n_steps):
        if step % decay_per == decay_per - 1:
            temp *= 0.95
        if step % 800 == 799:
            overlaps = compute_overlaps(h)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = compute_pair_delta(h, i, j)
        low = -h[i]
        high = 1.0 - h[j]
        if low >= high:
            continue
        old_obj = float(np.max(overlaps))
        save_i = h[i]
        save_j = h[j]
        save_overlaps = overlaps.copy()
        best_delta = find_best_delta(overlaps, d, low, high, n_grid)
        if abs(best_delta) < 1e-14:
            continue
        h[i] -= best_delta
        h[j] += best_delta
        overlaps += best_delta * d
        new_obj = float(np.max(overlaps))
        delta_obj = new_obj - old_obj
        if delta_obj < 0:
            if new_obj < best_obj:
                best_obj = new_obj
                best_h = h.copy()
        else:
            if temp > 1e-14 and rng.random() < np.exp(-delta_obj / temp):
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_h = h.copy()
            else:
                h[i] = save_i
                h[j] = save_j
                overlaps = save_overlaps
    return best_h, best_obj


def coord_descent_penalty(h, penalty, n_passes, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for _p in range(n_passes):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for j in order:
            s_j = float(np.sum(h)) - h[j]
            overlaps = compute_overlaps(h)
            L = np.arange(2 * n - 1) - (n - 1)
            j_L = j + L
            j_mL = j - L
            vp = (j_L >= 0) & (j_L < n)
            vm = (j_mL >= 0) & (j_mL < n)
            coeff = np.zeros(2 * n - 1)
            coeff[vp] = 1.0 - h[j_L[vp]]
            coeff[vm] -= h[j_mL[vm]]
            b = overlaps - h[j] * coeff
            t_cands = target - s_j - coeff / (2.0 * penalty)
            t_cands = t_cands[(t_cands > 0.001) & (t_cands < 0.999)]
            candidates = np.concatenate([[0.0, 1.0], t_cands])
            if len(candidates) > 200:
                idxs = rng.choice(len(candidates), 200, replace=False) if rng is not None else np.linspace(0, len(candidates)-1, 200).astype(int)
                candidates = candidates[idxs]
            grid = np.linspace(0.0, 1.0, 30)
            candidates = np.concatenate([candidates, grid])
            candidates = np.unique(np.sort(candidates))
            vals = coeff[None, :] * candidates[:, None] + b[None, :]
            max_vals = np.max(vals, axis=1)
            pen_vals = penalty * (s_j + candidates - target) ** 2
            obj = max_vals + pen_vals
            h[j] = candidates[int(np.argmin(obj))]
    return h
```
