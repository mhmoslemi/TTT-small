def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project(y, S):
    """Project y onto {x: 0 <= x <= 1, sum(x) = S}"""
    n = len(y)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = y.min() - 0.5
    hi = y.max() + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune to fix sum exactly
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            # Add to elements with most room
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-12:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    """SLSQP optimization with smooth surrogate."""
    # Use a soft-max approximation
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        scaled = alpha * (corr - m)
        return (m + np.log(np.sum(np.exp(scaled))) / alpha) * (2.0 / n)
    
    def gradient(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        scaled = alpha * (corr - m)
        weights = np.exp(scaled) / np.sum(np.exp(scaled))
        # Gradient of smooth max of corr
        g = np.zeros(n)
        dx = 2.0 / n
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-10:
                continue
            lag = k_idx - (n - 1)
            if lag >= 0:
                m2 = n - lag
                if m2 > 0:
                    g[lag:lag+m2] += weights[k_idx] * (1.0 - x[:m2])
                    g[:m2] += -weights[k_idx] * x[lag:lag+m2]
            else:
                j = -lag
                m2 = n - j
                if m2 > 0:
                    g[:m2] += weights[k_idx] * (1.0 - x[j:])
                    g[j:] += -weights[k_idx] * x[:m2]
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(objective, h0, jac=gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return _project(res.x, target_sum)
    except Exception:
        return h0

def _projected_gradient(h0, n, target_sum, n_steps, step_size):
    """Projected subgradient method focusing on max correlation lag."""
    h = h0.copy()
    dx = 2.0 / n
    t = 1.0
    y = h.copy()
    for step in range(n_steps):
        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[lag:lag+m] += (1.0 - y[:m])
                g[:m] += -y[lag:lag+m]
        else:
            j = -lag
            m = n - j
            if m > 0:
                g[:m] += (1.0 - y[j:])
                g[j:] += -y[:m]
        g *= dx
        alpha = step_size / np.sqrt(step + 1)
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - h)
        y_new = _project(y_new, target_sum)
        h = x_new
        y = y_new
        t = t_new
    return h

def _coord_descent_binary(h0, n, target_sum, max_rounds=20):
    """Coordinate descent trying binary values."""
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):

            h[j] = 0.0

            co = _c5_of(h)
            if min(c0, c1) < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

                m2 = n - j
                if m2 > 0:
                    g[:m2] += w[k_idx] * (1.0 - x[j:])
                    g[j:] += -w[k_idx] * x[:m2]
        return g * dx

    cons_fun = lambda x: x.sum() - target_sum
    cons_jac = lambda x: np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(objective, h0, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(res.x, target_sum)
    except Exception:
        return h0

def _coord_binary(h0, n, target_sum, max_rounds=20):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if min(c0, c1) < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_frac(h0, n, target_sum, max_rounds=8):
    h = h0.copy()
    vals = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if abs(best_v - old) > 1e-10:
                improved = True
        if not improved:
            break
    return h

def _gen_pop(n, target_sum, h_parent, n_parent, seed):
    rng = np.random.RandomState(seed)
    x_grid = np.linspace(0, 2, n, endpoint=False)
    pop = []
    if n_parent > 0 and h_parent is not None:
        if n == n_parent:
            pop.append(h_parent.copy())
        else:
            h_int = np.interp(np.linspace(0, n_parent - 1, n),
                               np.arange(n_parent), h_parent)
            pop.append(_project(np.clip(h_int, 0.0, 1.0), target_sum))
    for _ in range(4):
        pop.append(_project(rng.rand(n), target_sum))
    pop.append(_project((rng.rand(n) < 0.5).astype(float), target_sum))
    pop.append(0.5 + 0.4 * np.cos(np.pi * x_grid))
    pop.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x_grid))
    pop.append(0.5 + 0.3 * np.sin(4.0 * np.pi * x_grid + 0.5))
    h_blk = np.zeros(n)
    h_blk[::2] = 1.0
    pop.append(h_blk)
    h_blk2 = np.zeros(n)
    q = n // 3
    h_blk2[q:2*q] = 1.0
    pop.append(h_blk2)
    h_blk3 = np.zeros(n)
    q2 = n // 4
    h_blk3[q2:3*q2] = 1.0
    pop.append(h_blk3)
    h_saw = np.zeros(n)
    for i in range(n):
        phase = (i * 3.0 / n) % 1.0
        h_saw[i] = 1.0 if phase < 0.5 else 0.0
    pop.append(h_saw)
    pop = [_project(np.clip(p, 0.0, 1.0), target_sum) for p in pop]
    return pop

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 25

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 100
        h_parent = np.full(n_parent, 0.5)

    best_c5 = _c5_of(h_parent)
    best_h = h_parent.copy()
    best_n = n_parent

    n_candidates = []
    for delta in [0, 2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -14, 16, -18, 20]:
        n_try = n_parent + delta
        if n_try % 2 == 0 and 50 <= n_try <= 180:
            if n_try not in n_candidates:
                n_candidates.append(n_try)
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    total_time = budget_s - 80
    time_per_n = total_time / len(n_candidates)

    for n_idx, n in enumerate(n_candidates):
        if time.time() > deadline - 30:
            break
        time_left = deadline - time.time()
        alloc = min(time_per_n * 1.3, time_left - 20)
        if alloc < 25:
            break
        target_sum = n / 2.0

        pop = _gen_pop(n, target_sum, h_parent, n_parent, seed + n_idx * 7)
        pop_size = min(len(pop), 35)
        pop = pop[:pop_size]

        best_de_h, best_de_c = _de_evolve(
            pop, target_sum, max_gen=300, F_init=0.85, CR=0.6,
            time_limit=t_start + alloc * 0.45, seed=seed + 100 + n)

        if best_de_c < best_c5:
            best_c5 = best_de_c
            best_h = best_de_h.copy()
            best_n = n

        h_curr = best_de_h.copy()

        if time.time() < deadline - 30:
            h_nest, c5_nest = _nesterov_multi(h_curr, n, target_sum,
                                              n_iters=400, alpha0=0.35, top_m=3)
            if c5_nest < best_c5:
                best_c5 = c5_nest
                best_h = h_nest.copy()
                best_n = n
            h_curr = h_nest

        if time.time() < deadline - 25:
            h_nest2, c5_nest2 = _nesterov_multi(h_curr, n, target_sum,
                                                 n_iters=300, alpha0=0.2, top_m=5)
            if c5_nest2 < best_c5:
                best_c5 = c5_nest2
                best_h = h_nest2.copy()
                best_n = n
            h_curr = h_nest2

        if time.time() < deadline - 20:
            for alpha in [30, 100, 300]:
                if time.time() > deadline - 15:
                    break
                h_sl = _slsqp_smooth(h_curr, n, target_sum, alpha, maxiter=300)
                c5_sl = _c5_of(h_sl)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()
                    best_n = n
                h_curr = h_sl

        if time.time() < deadline - 15:
            h_bd = _coord_binary(h_curr, n, target_sum, max_rounds=20)
            c5_bd = _c5_of(h_bd)
            if c5_bd < best_c5:
                best_c5 = c5_bd
                best_h = h_bd.copy()
                best_n = n
            h_curr = h_bd

        if time.time() < deadline - 15:
            h_cf = _coord_frac(h_curr, n, target_sum, max_rounds=6)
            c5_cf = _c5_of(h_cf)
            if c5_cf < best_c5:
                best_c5 = c5_cf
                best_h = h_cf.copy()
                best_n = n

        if time.time() < deadline - 10:
            h_sl2 = _slsqp_smooth(h_curr, n, target_sum, 500, maxiter=400)
            c5_sl2 = _c5_of(h_sl2)
            if c5_sl2 < best_c5:
                best_c5 = c5_sl2
                best_h = h_sl2.copy()
                best_n = n

    if best_h is not None and time.time() < deadline - 10:
        n_best = best_n
        target = n_best / 2.0
        for alpha in [100, 500, 2000, 5000]:
            if time.time() > deadline - 5:
                break
            h_fin = _slsqp_smooth(best_h, n_best, target, alpha, maxiter=500)
            c5_fin = _c5_of(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin

        if time.time() < deadline - 5:
            h_cd = _coord_binary(best_h, n_best, target, max_rounds=25)
            c5_cd = _c5_of(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

        if time.time() < deadline - 5:
            h_cf = _coord_frac(best_h, n_best, target, max_rounds=10)
            c5_cf = _c5_of(h_cf)
            if c5_cf < best_c5:
                best_c5 = c5_cf
                best_h = h_cf

        if time.time() < deadline - 3:
            h_pol = _slsqp_smooth(best_h, n_best, target, 3000, maxiter=500)
            c5_pol = _c5_of(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
