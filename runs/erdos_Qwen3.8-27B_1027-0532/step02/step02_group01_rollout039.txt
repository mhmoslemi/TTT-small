We want the autocorrelation to be as "flat" (high) as possible at all lags. This is the opposite of the usual "low autocorrelation" goal in signal processing.

The maximum possible autocorrelation at lag 0 is n/2 (since sum h[m] = n/2). At lag k ≠ 0, the autocorrelation is at most n-|k| (if all overlapping entries are 1), but realistically much less.

For a "random" binary sequence with n/2 ones, the expected autocorrelation at lag k (for k ≠ 0) is approximately (n/2) * (1/2) = n/4 (since each product h[m]*h[m+k] is 1 with probability 1/4, and there are n-|k| terms).

def sa_binary(h_start, n_steps, t_start, t_end, rng, time_limit=None, start_time=None):
    n = len(h_start)
    target_sum = n // 2
    h = project_binary_deterministic(h_start, target_sum)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for step in range(n_steps):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Random perturbation: swap a 1 with a 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    
    return best_h, best_c5

def compute_delta(h, i, j, n):
    """Compute the change in the full correlation array when swapping h[i]:1->0 and h[j]:0->1."""
    delta = np.zeros(2*n-1)
    # Term 1: m=i contributes
    p_start = n-1-i
    p_end = 2*n-1-i  # exclusive
    q = np.arange(p_start, p_end)
    idx = (q - (n-1) + i)  # = i+k where k = q-(n-1), so idx = i + q - (n-1)
    # idx ranges from 0 to n-1
    delta[p_start:p_end] += h[i:n] - 1  # h[i+k] - 1 for k=0..n-1-i... 
    
    # Hmm, this isn't quite right. Let me think again.
    # p ranges from n-1-i to 2n-2-i (inclusive)
    # k = p - (n-1) ranges from -i to n-1-i
    # i+k ranges from i-i=0 to i+(n-1-i)=n-1
    # So h[i+k] = h[0..n-1] = h (the full array!)
    # delta[p_start:p_end] += h - 1
    
    delta[p_start:p_end] += (h - 1)
    
    # Term 2: m=j contributes
    p_start2 = n-1-j
    p_end2 = 2*n-1-j
    # k ranges from -j to n-1-j
    # j+k ranges from 0 to n-1
    # 1-h[j+k] = 1-h[0..n-1] = 1-h
    delta[p_start2:p_end2] += (1 - h)
    
    # But we need to fix the special case: when j+k = i, the formula gives 1-h[i] = 0
    # but the correct value is 1 (since h'[i] = 0 after swap)
    # j+k = i means k = i-j, so p = n-1+i-j
    if 0 <= i-j and n-1+j <= 2*n-2 and (n-1+i-j) >= p_start2 and (n-1+i-j) < p_end2:
        # Actually, j+k = i means k = i-j. We need k in [-j, n-1-j], i.e., i-j >= -j (always, since i>=0) and i-j <= n-1-j (i.e., i <= n-1, always true)
        # So the special case applies when 0 <= i-j, i.e., i >= j
        # p = n-1 + (i-j)
        # We added 1-h[j+k] = 1-h[i] = 0 at this position
        # Correct value is 1
        # So we need to add 1 to delta at position n-1+i-j
        p_special = n-1 + i - j
        if 0 <= p_special < 2*n-1:
            delta[p_special] += 1
    
    # Similarly, for term 1: when i+k = j, formula gives h[j]-1 = -1, correct value is -1. ✓ No fix needed.
    
    # Term 3: (1-h[i]) contribution
    # p ranges such that k = p-(n-1), i-k = i-(p-(n-1)) = i-p+n-1, need 0 <= i-p+n-1 < n
    # i-p+n-1 >= 0 → p <= i+n-1
    # i-p+n-1 < n → p > i-1 → p >= i
    # Wait: i-p+n-1 < n → i-p < 1 → p > i-1 → p >= i (since p is integer)
    # And i-p+n-1 >= 0 → p <= i+n-1
    # So p ranges from i to i+n-1
    # But also need k ≠ 0, i.e., p ≠ n-1
    # And i-k ≠ j, i.e., i-(p-(n-1)) ≠ j → i-p+n-1 ≠ j → p ≠ i+n-1-j
    # And p must be in [0, 2n-2]
    p_lo3 = max(0, i)
    p_hi3 = min(2*n-2, i+n-1)
    for p in range(p_lo3, p_hi3+1):
        if p == n-1:
            continue
        r = i + n - 1 - p  # = i-k
        if r == j:
            continue
        delta[p] += h[r]
    
    # Term 4: (1-h[j]) contribution
    # j-k = j-(p-(n-1)) = j-p+n-1, need 0 <= j-p+n-1 < n
    # j-p+n-1 >= 0 → p <= j+n-1
    # j-p+n-1 < n → p >= j
    # p ≠ n-1 (k≠0)
    # j-k ≠ i → j-p+n-1 ≠ i → p ≠ j+n-1-i
    p_lo4 = max(0, j)
    p_hi4 = min(2*n-2, j+n-1)

            continue
        r = j + n - 1 - p
        if r == i:
            continue
        delta[p] += -h[r]
