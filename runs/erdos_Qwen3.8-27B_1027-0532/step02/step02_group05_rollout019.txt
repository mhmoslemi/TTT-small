Now, to find min of max_j (a_j * t + b_j) for t ∈ [0,1]:
This is a convex piecewise-linear function. The minimum is at a breakpoint or at t=0 or t=1.

Approach:

2. Find all breakpoints in (0,1): for each pair (j,k), the breakpoint is t = (b_k - b_j)/(a_j - a_k) if a_j ≠ a_k and 0 < t < 1.
   But there are O(N²) pairs, which is too many.

Better approach: Use the fact that we only need to find the minimum. Since f is convex:
- If f'(0+) >= 0, minimum is at t=0
- If f'(1-) <= 0, minimum is at t=1
- Otherwise, minimum is at a point where 0 ∈ subdifferential, i.e., where there exist j with a_j < 0 and k with a_k > 0 both achieving the max.

The subgradient at t is the convex hull of {a_j : j ∈ argmax f(t)}. So f has a minimum at t* iff 0 is in the convex hull of {a_j : j ∈ argmax f(t*)}, i.e., min_{j∈argmax} a_j ≤ 0 ≤ max_{j∈argmax} a_j.

Algorithm:
1. Start at t=0.5 (or any point)
2. Compute f(t) = max_j (a_j * t + b_j)
3. Find the argmax set
4. If all a_j > 0 for j in argmax: f is increasing here, move left
5. If all a_j < 0 for j in argmax: f is decreasing here, move right
6. If mixed: found the minimum

To "move left" or "right", we can binary search for the next breakpoint.

Key observation: the minimum of a convex piecewise-linear function on [0,1] is at a point where the subgradient contains 0, i.e., where the "active" set (argmax) has both positive and negative slopes.

Algorithm:
1. Evaluate at t=0: find argmax set, check if all slopes positive → min is at 0
2. Evaluate at t=1: find argmax set, check if all slopes negative → min is at 1
3. Otherwise, binary search for the crossing point:
   - At t, if all active slopes > 0, the minimum is to the left
   - If all active slopes < 0, the minimum is to the right
   - If mixed, we've found the minimum (or very close to it)

This takes O(log(1/ε) * N) evaluations where ε is the desired accuracy. With ε = 1e-6, that's about 20 * N operations per coordinate.

Actually, for a convex function, if at point t, the subdifferential (set of subgradients) doesn't contain 0, then we can move in the direction of the negative subgradient. The subdifferential of f at t is the convex hull of {A_j : j ∈ argmax f(t)}.

So: if min_{j∈argmax} A_j > 0, move left. If max_{j∈argmax} A_j < 0, move right. Otherwise, t is optimal.

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlations(h)) * dx)


def antipal_h(g):
    m = len(g)
    N = 2 * m + 1
    h = np.empty(N)
    h[:m] = g
    h[m] = 0.5
    h[m+1:] = 1.0 - g[::-1]
    return h


def antipal_project(h):
    N = len(h)
    m = N // 2
    g = 0.5 * (h[:m] + (1.0 - h[m+1:][::-1]))
    return np.clip(g, 0.0, 1.0)


def compute_sens_col(h, p):
    N = len(h)
    S = np.zeros(2 * N - 1)
    for k in range(0, N):
        j = N - 1 + k
        if p >= k:
            S[j] += 1.0 - h[p - k]
        if p + k <= N - 1:
            S[j] -= h[p + k]
    for kp in range(1, N):
        j = N - 1 - kp
        if p >= kp:
            S[j] += 1.0 - h[p - kp]
        if p + kp <= N - 1:
            S[j] -= h[p + kp]
    return S


def find_1d_min_bisect(A, B):
    lo_v = float(np.min(B))
    hi_v = float(np.max(np.maximum(B, A + B)))
    t_opt = 0.5
    for _ in range(100):
        v = 0.5 * (lo_v + hi_v)
        pos = A > 1e-15
        neg = A < -1e-15
        zero = ~(pos | neg)
        t_hi = 1.0
        t_lo = 0.0
        feasible = True
        if np.any(pos):
            ub = (v - B[pos]) / A[pos]
            t_hi = min(t_hi, float(np.min(ub)))
        if np.any(neg):
            lb = (v - B[neg]) / A[neg]
            t_lo = max(t_lo, float(np.max(lb)))
        if np.any(zero):
            if float(np.max(B[zero])) > v + 1e-15:
                feasible = False
        if feasible and t_lo <= t_hi and t_hi >= 0.0 and t_lo <= 1.0:
            hi_v = v
            t_opt = 0.5 * (max(t_lo, 0.0) + min(t_hi, 1.0))
        else:
            lo_v = v
    return float(np.clip(t_opt, 0.0, 1.0))


def coord_descent_antipal(g, max_sweeps=5000, use_linear_update=True):
    m = len(g)
    N = 2 * m + 1
    dx = 2.0 / N
    g = g.copy()
    best_g = g.copy()
    h = antipal_h(g)
    corr = compute_correlations(h).copy()
    best_score = float(np.max(corr)) * dx

    for sweep in range(max_sweeps):
        if use_linear_update:
            h = antipal_h(g)
            corr = compute_correlations(h).copy()
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-14:
            best_score = score
            best_g = g.copy()

        any_improved = False
        for i in range(m):
            Si = compute_sens_col(h, i)
            Sib = compute_sens_col(h, 2 * m - i)
            A = Si - Sib
            B = corr
            t_opt = find_1d_min_bisect(A, B)
            delta = t_opt - g[i]
            if abs(delta) > 1e-10:
                g[i] = t_opt
                any_improved = True
                if use_linear_update:
                    h[i] += delta
                    h[2 * m - i] -= delta
                    corr += delta * A
                    np.clip(h, 0.0, 1.0, out=h)

        if not any_improved:
            break

    h_final = antipal_h(g)
    score_final = evaluate_score(h_final)
    if score_final < best_score:
        best_score = score_final
        best_g = g.copy()
    return best_g, best_score


def smooth_max(x, alpha):
    m = float(np.max(x))
    return m + float(np.log(np.sum(np.exp(alpha * (x - m))))) / alpha


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / float(np.sum(ex))
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def _ap_obj(g, alpha):
    h = antipal_h(g)
    return smooth_max(compute_correlations(h), alpha)


def _ap_jac(g, alpha):
    h = antipal_h(g)
    grad_h = grad_obj(h, alpha)
    half = len(g)
    grad_g = grad_h[:half].copy()
    for i in range(half):
        j = 2 * half - i
        grad_g[i] += -grad_h[j]
    return grad_g


def _ap_cons_fun(g):
    return np.zeros(1)


def slsqp_antipal(g0, alpha, maxiter):
    half = len(g0)
    bounds = [(0.0, 1.0)] * half
    try:
        res = minimize(
            _ap_obj, g0, method='SLSQP', bounds=bounds,
            jac=_ap_jac, args=(alpha,),
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return g0


def project_feasible(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def upscale(h_from, n_to):
    n_from = len(h_from)
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(x_new, x_old, h_from)
