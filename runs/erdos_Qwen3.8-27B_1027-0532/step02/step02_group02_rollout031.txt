            h_pert = _fix_sum(best_h + scale * rng.randn(n_b), tgt_b)
            h_new = _lbfgs_run(h_pert, n_b, 800, 1e5, 6, 150, deadline)
            h_new = _fix_sum(h_new, tgt_b)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_b
    
    if best_h is not None and time.time() < deadline - 30:
        n_b = best_n
        tgt_b = n_b / 2.0
        h_final = _lbfgs_run(best_h, n_b, 5000, 1e6, 3, 300, deadline)
        h_final = _fix_sum(h_final, tgt_b)
        c5_final = _c5_of(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    if best_h is None:
        best_h = np.full(n100, 0.5)
        best_n = n100
        best_c5 = _c5_of(best_h)
    
    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
