One more thought: the **coordinate descent** approach I described earlier (where for each coordinate j, the objective is a max of linear functions) is actually very powerful for this problem. Let me implement it properly.

For coordinate j, given all other h[i] fixed:
- For each lag m (0 to 2N-2), corr[m] is an affine function of h[j]:
  corr[m] = A_m * h[j] + B_m

where A_m and B_m can be computed from the current h (with h[j] excluded).

The objective is max_m (A_m * h[j] + B_m) (times dx, but that's a constant).

To minimize this over h[j] in [0,1]:
- This is a 1D convex optimization (max of affine functions is convex)
- The minimum occurs where the "kink" is, or at a boundary

We can find it by:
1. Computing all (A_m, B_m) pairs: O(N)
2. Finding the lower envelope of the max: this is equivalent to finding the minimum of a convex piecewise-linear function
3. The minimum of a convex PL function over [0,1] is at a breakpoint or at 0 or 1

The breakpoints occur where two lines cross: A_m1 * t + B_m1 = A_m2 * t + B_m2, i.e., t = (B_m1 - B_m2) / (A_m2 - A_m1)

But checking all O(N²) pairs is too expensive. Instead, we can:
- Sample the function at a few points
- Or use a bisection on the subgradient

Actually, the subgradient of f(t) = max_m (A_m * t + B_m) at t is:
∂f(t) = conv{A_m : f(t) = A_m * t + B_m}

At the minimum, 0 ∈ ∂f(t*), which means there exist active m's with A_m values on both sides of 0 (or A_m = 0 for some active m).

We can use bisection:
- At t=0: compute f(0) and the subgradient [min active A, max active A]
- At t=1: same
- If min active A at t=0 > 0, the minimum is at t=0
- If max active A at t=1 < 0, the minimum is at t=1
- Otherwise, bisect

Each evaluation is O(N) (compute all A_m * t + B_m and find the max). So bisection takes O(N log(1/eps)) time.

Total per coordinate: O(N) to compute A, B + O(N log(1/eps)) for bisection = O(N log(1/eps))
Total per sweep over all N coordinates: O(N² log(1/eps))

For N=425 and eps=1e-6: O(425² * 20) ≈ 3.6M operations per sweep. That's fast!

def grad_smooth_max(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / float(np.sum(ex))
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (N - 1)
        if k >= 0:
            if k < N:
                grad[k:N] += w * (1.0 - h[:N - k])
                grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            if kp < N:
                grad[:N - kp] += w * (1.0 - h[kp:])
                grad[kp:] -= w * h[:N - kp]
    return grad


def grad_p_norm(h, p):
    N = len(h)
    corr = compute_correlations(h)
    pn = float(np.sum(corr ** p))
    if pn < 1e-30:
        return np.zeros(N)
    coef = pn ** (1.0 / p - 1.0)
    w = corr ** (p - 1) * coef
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        wt = w[j]
        if wt < 1e-18:
            continue
        k = j - (N - 1)
        if k >= 0:
            if k < N:
                grad[k:N] += wt * (1.0 - h[:N - k])
                grad[:N - k] -= wt * h[k:]
        else:
            kp = -k
            if kp < N:
                grad[:N - kp] += wt * (1.0 - h[kp:])
                grad[kp:] -= wt * h[:N - kp]
    return grad


def fista(h0, target, grad_fn, max_iter, lr, use_accel=True):
    N = len(h0)
    x = h0.copy()
    z = h0.copy()
    t_k = 1.0
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    for it in range(max_iter):
        g = grad_fn(z)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-16:
            break
        z_new = project_feasible(z - lr * g, target)
        if use_accel:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            x_new = z_new + ((t_k - 1.0) / t_new) * (z_new - x)
            x_new = project_feasible(x_new, target)
            t_k = t_new
        else:
            x_new = z_new
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
        z = z_new
    return best_x, best_score


def fista_adaptive(h0, target, alpha, time_limit, p_mode=False, p_val=50):
    N = len(h0)
    if p_mode:
        grad_fn = lambda z: grad_p_norm(z, p_val)
        obj_fn = lambda z: p_norm_val(compute_correlations(z), p_val)
    else:
        grad_fn = lambda z: grad_smooth_max(z, alpha)
        obj_fn = lambda z: smooth_max_val(compute_correlations(z), alpha)

    lr = 1.0 / (alpha * N) if not p_mode else 1.0 / (p_val * N)
    t_start = time.time()
    x = h0.copy()
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    no_improve = 0

    while time.time() - t_start < time_limit:
        batch = 200
        x, _ = fista(x, target, grad_fn, batch, lr)
        s = evaluate_score(x)
        if s < best_score - 1e-12:
            best_score = s
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                lr *= 0.7
                no_improve = 0
            elif no_improve > 10:
                break

    return best_x, best_score


def slsqp_smooth(h0, alpha, target, maxiter):
    N = len(h0)

    def fun(x):
        return smooth_max_val(compute_correlations(x), alpha)

    def jac(x):
        return grad_smooth_max(x, alpha)

    def cons_fun(x):
        return float(np.sum(x)) - target

    def cons_jac(x):
        return np.ones(len(x))

    bounds = [(0.0, 1.0)] * N
    res = minimize(
        fun, h0, method='SLSQP', bounds=bounds, jac=jac,
        constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def antipal_grad(g, alpha, n_full):
    h = antipal_h(g)
    grad_h = grad_smooth_max(h, alpha)
    half = len(g)
    grad_g = grad_h[:half].copy()
    for i in range(half):
        j = n_full - 1 - i
        grad_g[i] += -grad_h[j]
    return grad_g


def antipal_fun(g, alpha):
    h = antipal_h(g)
    return smooth_max_val(compute_correlations(h), alpha)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    init = project_feasible(init, target)

    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: FISTA with smooth max, geometric alpha schedule
    alphas_f1 = [20, 50, 100, 300, 1000, 3000, 10000, 30000]
    for a in alphas_f1:
        if remaining() < 60:
            break
        alloc = min(120.0, remaining() * 0.15)
        h_new, s_new = fista_adaptive(best_h, target, a, alloc, p_mode=False)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 2: p-norm optimization (different smoothness)
    for p_val in [20, 50, 100]:
        if remaining() < 50:
            break
        alloc = min(80.0, remaining() * 0.1)
        h_new, s_new = fista_adaptive(best_h, target, 1.0, alloc, p_mode=True, p_val=p_val)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 3: Random restarts with FISTA
    n_restarts = 40
    for r in range(n_restarts):
        if remaining() < 80:
            break
        scale = 0.02 + 0.15 * rng.rand()
        pert = best_h + rng.randn(n) * scale
        pert = project_feasible(pert, target)
        a = [100, 500, 2000, 5000][rng.randint(4)]
        alloc = min(60.0, remaining() * 0.05)
        h_new, s_new = fista_adaptive(pert, target, a, alloc, p_mode=False)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 4: SLSQP polish on current best
    if remaining() > 80:
        for a in [1000, 5000, 20000]:
            if remaining() < 40:
                break
            maxiter = max(200, min(800, 200000 // n))
            try:
                res_x = slsqp_smooth(best_h, a, target, maxiter)
                if res_x is not None and np.all(np.isfinite(res_x)):
                    h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target)
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new.copy()
            except Exception:
                pass

    # Phase 5: Anti-palindrome optimization (different structure)
    if remaining() > 100:
        n_ap = n if n % 2 == 1 else n + 1
        half = n_ap // 2
        target_ap = n_ap / 2.0
        # Initialize anti-pal from current best
        if n == n_ap:
            g0 = best_h[:half].copy()
        else:
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new_arr = np.linspace(0.0, 2.0, n_ap, endpoint=False)
            h_up = np.interp(x_new_arr, x_old, best_h)
            g0 = h_up[:half].copy()
        g0 = np.clip(g0, 0.0, 1.0)
        # Check sum constraint for anti-pal: sum(g) + 0.5 + sum(1-g) = sum(g) + 0.5 + half - sum(g) = half + 0.5 = n_ap/2 ✓

        bounds_ap = [(0.0, 1.0)] * half
        for a in [500, 2000, 10000]:
            if remaining() < 40:
                break
            maxiter = max(100, min(400, 100000 // half))
            try:
                res_ap = minimize(
                    antipal_fun, g0.copy(), method='SLSQP', bounds=bounds_ap,
                    jac=lambda g, aa=a: antipal_grad(g, aa, n_ap),
                    options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
                )
                if res_ap.x is not None and np.all(np.isfinite(res_ap.x)):
                    h_ap = antipal_h(np.clip(res_ap.x, 0.0, 1.0))
                    h_ap = project_feasible(h_ap, target_ap)
                    s_ap = evaluate_score(h_ap)
                    if s_ap < best_score:
                        best_score = s_ap
                        best_h = h_ap.copy()
            except Exception:
                pass

    # Phase 6: Multi-resolution - try smaller n then upsample
    for n_small in [200, 150, 100]:
        if remaining() < 120:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
        h_small = np.interp(x_small, x_old, best_h)
        h_small = project_feasible(h_small, n_small / 2.0)

        for a in [200, 1000, 5000]:
            if remaining() < 60:
                break
            alloc = min(40.0, remaining() * 0.08)
            h_sm, s_sm = fista_adaptive(h_small, n_small / 2.0, a, alloc, p_mode=False)

        # Upsample back
        x_up = np.linspace(0.0, 2.0, n, endpoint=False)
        h_up = np.interp(x_up, x_small, h_sm)
        h_up = project_feasible(h_up, target)
        s_up = evaluate_score(h_up)
        if s_up < best_score:
            best_score = s_up
            best_h = h_up.copy()

    # Phase 7: Multi-resolution - try larger n
    for n_large in [n * 2, n * 3]:
        if n_large > 3000 or remaining() < 100:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_large = np.linspace(0.0, 2.0, n_large, endpoint=False)
        h_large = np.interp(x_large, x_old, best_h)
        h_large = project_feasible(h_large, n_large / 2.0)

        for a in [100, 500, 2000]:
            if remaining() < 50:
                break
            alloc = min(50.0, remaining() * 0.08)
            h_lg, s_lg = fista_adaptive(h_large, n_large / 2.0, a, alloc, p_mode=False)
            h_large = h_lg if s_lg < evaluate_score(h_large) else h_large

        s_lg = evaluate_score(h_large)
        if s_lg < best_score:
            best_score = s_lg
            best_h = h_large.copy()

    # Phase 8: Final high-alpha polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    for a in [10000, 30000, 100000]:
        if remaining() < 20:
            break
        alloc = min(60.0, remaining() * 0.3)
        h_new, s_new = fista_adaptive(best_h, target_final, a, alloc, p_mode=False)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP
    if remaining() > 30:
        maxiter = max(300, min(1000, 500000 // n_final))
        try:
            res_x = slsqp_smooth(best_h, 50000, target_final, maxiter)
            if res_x is not None and np.all(np.isfinite(res_x)):
                h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target_final)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
        except Exception:
            pass

    # Final cleanup
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
