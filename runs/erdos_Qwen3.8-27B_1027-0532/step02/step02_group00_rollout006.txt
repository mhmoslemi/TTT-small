                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def make_weights_topk(h, k_top, base_weight):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_cc = len(corr)
    sorted_idx = np.argsort(corr)[::-1]
    weights = np.full(n_cc, base_weight)
    weights[sorted_idx[:k_top]] = 1.0 / k_top
    return weights


def single_slsqp(h_start, target, alpha, maxiter, ftol=1e-15):
    global _target, _n_pts_global
    _target = target
    _n_pts_global = len(h_start)
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_jac}]
    try:
        res = minimize(smooth_obj_full, h_start, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start


def power_sum_obj(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    scaled = (corr * dx) ** p
    return float(np.sum(scaled) ** (1.0 / p))


def run(seed=42, budget_s=1300, **kwargs):
    global _target, _n_pts_global, _weights_global, _start_time, _budget_s
    _start_time = time.time()
    _budget_s = budget_s
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Progressively refine with increasing alpha
    if tleft() > 250:
        alphas_1 = [8, 20, 50, 120, 300, 700, 1500, 3000]
        h_refined = slsqp_optimize(best_h, target_init, smooth_obj_full, alphas_1, 1200, 1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 2: Top-k focused optimization
    for k_top in [5, 10, 15, 30, 50]:
        if tleft() < 60:
            break
        _weights_global = make_weights_topk(best_h, k_top, 0.003)
        target_cur = len(best_h) / 2.0
        alphas_2 = [50, 150, 400, 1000, 2000]
        h_refined = slsqp_optimize(best_h, target_cur, smooth_obj_weighted, alphas_2, 800, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 3: Power-sum objective with L-BFGS-B (penalty for sum constraint)
    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0

        def penalty_obj(x):
            h = np.clip(x, 0.0, 1.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            dx = 2.0 / n_use
            scaled = (corr * dx) ** 20
            obj = float(np.sum(scaled) ** (1.0 / 20.0))
            penalty = 1000.0 * (np.sum(h) - target_use) ** 2
            return obj + penalty

        try:
            res = minimize(penalty_obj, best_h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n_use,
                           options={'maxiter': 2000, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target_use)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    # Phase 4: Multi-resolution exploration
    n_candidates = [300, 350, 450, 500, 550, 600, 700, 800]
    for n_try in n_candidates:
        if tleft() < 130:
            break
        if n_try == len(best_h):
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_3 = [15, 60, 200, 600, 1500]
        h_refined = slsqp_optimize(h_interp, target_try, smooth_obj_full, alphas_3, 600, 1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 5: Re-refine on best resolution with top-k
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 80:
        for k_top in [5, 10, 20]:
            _weights_global = make_weights_topk(best_h, k_top, 0.002)
            h_refined = slsqp_optimize(best_h, target_use, smooth_obj_weighted,
                                        [100, 500, 1500, 3000], 800, 1e-15)
            sc = compute_c5(h_refined)
            if sc < best_score:
                best_score = sc
                best_h = h_refined

    # Phase 6: Aggressive fine-tuning with perturbations
    fine_noise = [0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08]
    fine_alphas = [500, 1000, 2000, 4000, 6000]

    attempt = 0
    while tleft() > 40 and attempt < 60:
        attempt += 1
        noise = rng.choice(fine_noise)
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        alpha = rng.choice(fine_alphas)
        h_new = single_slsqp(h_pert, target_use, alpha, 500, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Nelder-Mead polish on small neighborhood
    if tleft() > 50:
        h_pert = best_h + rng.randn(n_use) * 0.005
        h_pert = project_feasible(h_pert, target_use)
        try:
            res = minimize(compute_c5, h_pert, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            h_new = project_feasible(res.x, target_use)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    # Phase 8: Final SLSQP polish
    if tleft() > 25:
        h_final = slsqp_optimize(best_h, target_use, smooth_obj_full,
                                  [1000, 2000, 5000, 8000], 600, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
