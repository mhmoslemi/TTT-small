def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, target_sum):
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_c5(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        if l == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif l > 0:
            grad[0:n - l] += w * (1.0 - h[l:n])
            grad[l:n] += w * (-h[0:n - l])
        else:
            l2 = -l
            grad[l2:n] += w * (1.0 - h[0:n - l2])
            grad[0:n - l2] += w * (-h[l2:n])

        if elapsed() > budget_s * 0.94:
            break
        scale = 0.003 + 0.03 * restart / 20.0
        h_pert = best_h + scale * np.random.randn(best_n)
        h_pert = project_feasible(h_pert, best_n / 2.0)

        for eps in [1000.0, 100.0, 10.0]:
            if elapsed() > budget_s * 0.94:
                break
            h_new, c5_new = slsqp_refine(h_pert, best_n, eps, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_pert = h_new.copy()

    # Phase 5: Final polish
    if elapsed() < budget_s * 0.97:
        deltas_final = [0.001, 0.0005, 0.0002, 0.0001, -0.0001, -0.0002, -0.0005, -0.001]
        best_h, best_c5 = coord_descent(best_h, best_n, tlimit(0.97), 500, deltas_final)

    if elapsed() < budget_s * 0.99:
        h_new, c5_new = slsqp_refine(best_h, best_n, 5.0, 200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
