def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_simplex_dir(h, target_sum, step):

    best_n = 0
    rng = np.random.RandomState(seed)

    n_base = 100
    target_sum = n_base / 2.0

    candidates = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        if len(h_parent) == n_base:
            candidates.append(h_parent)

    bin_cands = _make_binary_init(n_base, rng, n_cands=16)
    for bc in bin_cands[:8]:
        for eps in [0.05, 0.12, 0.2]:
            candidates.append(_soften_binary(bc, eps))

    x = np.arange(n_base, dtype=float) / n_base * 2.0
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.2, 0.35]:
            h = 0.5 + amp * np.sin(np.pi * freq * x)
            candidates.append(_fix_sum(h, target_sum))

    for i in range(4):
        h = rng.rand(n_base)
        h = _fix_sum(h, target_sum)
        candidates.append(h)

    scored = []
    for c in candidates:
        c5 = _c5_of(c)
        scored.append((c5, c))
    scored.sort(key=lambda x: x[0])

    t_coarse_end = t_start + 150
    top_cands = [c for _, c in scored[:10]]
    for h_c in top_cands:
        if time.time() > t_coarse_end:
            break
        h_opt, c5_opt = _projected_gradient(h_c, n_base, target_sum,
                                             n_iter=6000, lr0=0.12,
                                             lr_decay=0.0004,
                                             deadline=min(t_coarse_end, deadline),
                                             top_k=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_base

    if best_h is None:
        best_h = scored[0][1].copy()
        best_c5 = scored[0][0]
        best_n = n_base

    t_slsqp_end = t_start + 480
    refine_pool = [best_h.copy()]
    for c5_s, h_s in scored[:6]:
        if h_s is not best_h:
            refine_pool.append(h_s.copy())

    for h_r in refine_pool[:6]:
        if time.time() > t_slsqp_end - 40:
            break
        h_cur = h_r.copy()
        for alpha in [20, 60, 150, 400, 1000]:
            if time.time() > t_slsqp_end - 15:
                break
            h_new = _slsqp_smooth(h_cur, alpha, maxiter=200, ftol=1e-13)
            h_new = _fix_sum(h_new, target_sum)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_base
            h_cur = h_new
        h_direct = _slsqp_direct(h_cur, maxiter=80)
        h_direct = _fix_sum(h_direct, target_sum)
        c5_d = _c5_of(h_direct)
        if c5_d < best_c5 - 1e-12:
            best_c5 = c5_d
            best_h = h_direct.copy()
            best_n = n_base

    t_up_end = t_start + 900
    if best_h is not None and time.time() < t_up_end - 80:
        for n_fine in [200, 300]:
            if time.time() > t_up_end - 50:
                break
            target_f = n_fine / 2.0
            h_up = _upsample(best_h, n_fine)
            h_opt, c5_opt = _projected_gradient(h_up, n_fine, target_f,
                                                 n_iter=4000, lr0=0.08,
                                                 lr_decay=0.0008,
                                                 deadline=min(t_up_end - 50, deadline),
                                                 top_k=3)
            h_cur = h_opt
            for alpha in [30, 100, 300]:
                if time.time() > t_up_end - 20:
                    break
                h_new = _slsqp_smooth(h_cur, alpha, maxiter=150, ftol=1e-13)
                h_new = _fix_sum(h_new, target_f)
                c5_new = _c5_of(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_fine
                h_cur = h_new

    if best_h is not None and time.time() < deadline - 50:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        for _ in range(8):
            if time.time() > deadline - 25:
                break
            for scale in [0.008, 0.02, 0.04]:
                if time.time() > deadline - 15:
                    break
                h_p = _fix_sum(best_h + scale * rng.randn(n_fin), t_fin)
                h_new = _slsqp_smooth(h_p, 500, maxiter=120, ftol=1e-14)
                h_new = _fix_sum(h_new, t_fin)
                c5_new = _c5_of(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_fin

    if best_h is None:
        best_h = np.full(n_base, 0.5)
        best_h = _fix_sum(best_h, n_base / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n_base

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
