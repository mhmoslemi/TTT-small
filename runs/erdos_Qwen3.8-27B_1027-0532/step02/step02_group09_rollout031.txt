    n = len(h)
    overlaps = compute_overlaps(h)
    m = float(np.max(overlaps))
    shifted = alpha * (overlaps - m)
    exp_vals = np.exp(shifted)
    wsum = float(np.sum(exp_vals))
    log_sum = m + np.log(wsum) / alpha

            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16, 'maxfun': maxiter * 5}
        )
        return res.x
    except Exception:
        return None

def slsqp_optimize(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            lse_jac_combined, h0, args=(alpha, penalty),
            method='SLSQP', jac=True, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None

def nesterov_subgrad(h0, n_iters, alpha, target):
    n = len(h0)
    x = h0.copy()
    x_prev = h0.copy()
    best_x = h0.copy()
    best_raw = float(np.max(compute_overlaps(h0)))
    for it in range(n_iters):
        t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * float(it) * float(it)))
        if it > 0:
            y = x + ((t_k - 1.0) / t_k) * (x - x_prev)
        else:
            y = x.copy()
        x_prev = x.copy()
        gamma = 0.5 / (0.5 + 0.08 * np.sqrt(float(it)))
        overlaps = compute_overlaps(y)
        m = float(np.max(overlaps))
        shifted = alpha * (overlaps - m)
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        wsum = float(np.sum(exp_vals))
        weights = exp_vals / wsum
        grad = np.zeros(n)
        sig_idx = np.where(weights > 1e-10)[0]
        for idx in sig_idx:
            w = weights[idx]
            L = idx - (n - 1)
            if L >= 0:
                grad[:n - L] -= w * y[L:]
                grad[L:] += w * (1.0 - y[:n - L])
            else:
                F = -L
                grad[F:] += w * (1.0 - y[:n - F])
                grad[:n - F] -= w * y[F:]
        x = project_box_sum(y - gamma * grad, target)
        raw = float(np.max(compute_overlaps(x)))
        if raw < best_raw:
            best_raw = raw
            best_x = x.copy()
    return best_x, best_raw

def block_swap_search(h0, n_swaps, rng, block_sizes):
    n = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n_target = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1.0)[0]
        remove = rng.choice(ones_idx, size=n_ones - n_target, replace=False)
        h_bin[remove] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h_bin == 0.0)[0]
        add = rng.choice(zero_idx, size=n_target - n_ones, replace=False)
        h_bin[add] = 1.0
    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    temp = 0.003
    for it in range(n_swaps):
        if it % 300 == 299:
            temp *= 0.92
        if temp < 1e-10:
            break
        bs = block_sizes[rng.randint(len(block_sizes))]
        if bs > n // 2:
            bs = 1
        start_a = rng.randint(0, n - bs + 1)
        start_b = rng.randint(0, n - bs + 1)
        if np.any(h_bin[start_a:start_a + bs] != h_bin[start_b:start_b + bs]):
            continue
        a_mask = h_bin[start_a:start_a + bs]
        b_mask = h_bin[start_b:start_b + bs]
        h_bin[start_a:start_a + bs] = b_mask
        h_bin[start_b:start_b + bs] = a_mask
        s_new = evaluate_score(h_bin)
        delta = s_new - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            if s_new < best_score:
                best_score = s_new
                best_h = h_bin.copy()
        else:
            h_bin[start_a:start_a + bs] = a_mask
            h_bin[start_b:start_b + bs] = b_mask
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 25.0 - (time.time() - t0)

    h_parent = np.array(initial_h_values, dtype=float)
    n_p = len(h_parent)
    h0 = project_box_sum(h_parent, n_p / 2.0)
    best_h = h0.copy()
    best_score = evaluate_score(h0)

    # Phase 1: Quick optimization at parent resolution
    n = n_p
    bounds = [(0.0, 1.0)] * n
    h_curr = h0.copy()
    for alpha in [200, 800, 2000, 5000]:
        if time_left() < 50:
            break
        penalty = alpha * 20.0
        maxiter = max(100, 300000 // n)
        res_x = lbfgsb_optimize(h_curr.copy(), alpha, penalty, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h_curr = h_new

    # Phase 2: SLSQP polish at parent resolution
    for alpha in [1000, 3000]:
        if time_left() < 30:
            break
        penalty = alpha * 15.0
        res_x = slsqp_optimize(h_curr.copy(), alpha, penalty, 400, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h_curr = h_new

    # Phase 3: Upscale to 2x and optimize
    if time_left() > 200:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_box_sum(h2, n2 / 2.0)
        bounds2 = [(0.0, 1.0)] * n2
        for alpha in [100, 400, 1200, 3000, 6000]:
            if time_left() < 50:
                break
            penalty = alpha * 20.0
            maxiter = max(80, 250000 // n2)
            res_x = lbfgsb_optimize(h2.copy(), alpha, penalty, maxiter, bounds2)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n2 / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                h2 = h_new

    # Phase 4: SLSQP at higher resolution
    if time_left() > 100:
        n_c = len(h2)
        bounds_c = [(0.0, 1.0)] * n_c
        for alpha in [2000, 5000]:
            if time_left() < 40:
                break
            penalty = alpha * 15.0
            res_x = slsqp_optimize(h2.copy(), alpha, penalty, 300, bounds_c)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_c / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                h2 = h_new

    # Phase 5: Nesterov subgradient at current best resolution
    if time_left() > 80:
        n_s = len(best_h)
        n_iters = min(4000, int(time_left() * 12))
        h_sub, raw_sub = nesterov_subgrad(best_h.copy(), n_iters, 300, n_s / 2.0)
        s_sub = raw_sub * (2.0 / n_s)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()
        # Polish
        if time_left() > 40:
            bounds_s = [(0.0, 1.0)] * n_s
            res_x = lbfgsb_optimize(h_sub.copy(), 5000, 5000.0 * 30.0, 500, bounds_s)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_s / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 6: Try 3x resolution if time allows
    if time_left() > 200:
        n3 = n_p * 3
        x_old3 = np.linspace(0.0, 2.0, n_p, endpoint=False)
        x_new3 = np.linspace(0.0, 2.0, n3, endpoint=False)
        h3 = np.interp(x_new3, x_old3, best_h)
        h3 = project_box_sum(h3, n3 / 2.0)
        bounds3 = [(0.0, 1.0)] * n3
        for alpha in [200, 800, 3000]:
            if time_left() < 50:
                break
            penalty = alpha * 20.0
            maxiter = max(60, 200000 // n3)
            res_x = lbfgsb_optimize(h3.copy(), alpha, penalty, maxiter, bounds3)
            if res_x is not None:
                h_new3 = project_box_sum(np.clip(res_x, 0.0, 1.0), n3 / 2.0)
                s = evaluate_score(h_new3)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new3.copy()
                h3 = h_new3
        if time_left() > 60:
            n3_c = len(h3)
            bounds3_c = [(0.0, 1.0)] * n3_c
            res_x = slsqp_optimize(h3.copy(), 3000, 3000.0 * 15.0, 250, bounds3_c)
            if res_x is not None:
                h_new3 = project_box_sum(np.clip(res_x, 0.0, 1.0), n3_c / 2.0)
                s = evaluate_score(h_new3)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new3.copy()

    # Phase 7: Binary block-swap local search
    if time_left() > 50:
        n_b = len(best_h)
        n_swaps = min(6000, int(time_left() * 8))
        h_bin, s_bin = block_swap_search(best_h, n_swaps, rng, [1, 2, 3, 4])
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()

    # Phase 8: Final high-precision polish
    if time_left() > 30:
        n_f = len(best_h)
        bounds_f = [(0.0, 1.0)] * n_f
        for alpha in [8000, 15000]:
            if time_left() < 15:
                break
            penalty = alpha * 40.0
            maxiter = max(150, 300000 // n_f)
            res_x = lbfgsb_optimize(best_h.copy(), alpha, penalty, maxiter, bounds_f)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_f / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Final exact projection and evaluation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
