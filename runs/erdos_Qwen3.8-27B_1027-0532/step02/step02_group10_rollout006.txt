from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    h = h.copy()
    # Binary search for the threshold
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(np.clip(shifted, -500, 0))
    total = np.sum(exp_shifted)
    value = m - np.log(total) / alpha  # Note: this is m + log(sum(exp(shifted)))/alpha = m + log(exp(0) + ...)/alpha
    # Actually: logsumexp(alpha*(x-m)) / alpha + m
    # = (log(sum(exp(alpha*(x-m)))) + 0) / alpha + m
    # = log(sum(exp(alpha*(x-m)))) / alpha + m
    # sum(exp(alpha*(x-m))) >= 1 (since exp(0)=1 is one term)
    # so log(sum(...)) >= 0
    # so value = m + log(sum(...))/alpha >= m
    # This is correct - smooth max >= true max
    weights = exp_shifted / total
    return value, weights

def top_k_sum(x, k):
    """Sum of the top k values."""
    idx = np.argpartition(x, -k)[-k:]
    return np.sum(x[idx])

    """Gradient of sum of top-k overlaps."""
    n = len(h)
    overlaps = compute_overlaps(h)
    idx = np.argpartition(overlaps, -k)[-k:]
    # For each active shift, compute gradient contribution
    grad = np.zeros(n)
    for m in idx:
        offset = int(m - (n - 1))

        # where indices are clamped
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:

        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= dx * h[i_lo2 - offset:i_hi2 - offset]
    return grad

def topk_pgd(h, k, target, max_iters, deadline, lr=None):
    """PGD with top-k objective."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    if lr is None:
        lr = 2.0 / (k * dx * 4.0 + 1.0)
    best_score = evaluate_score(h)
    best_h = h.copy()

            best_h = h_new.copy()
            stall = 0
            h = h_new
            # Increase learning rate slightly
            lr = min(lr * 1.05, 5.0)
        else:

            lr *= 0.7
            if stall > 100:

    return best_h

    """FISTA with top-k objective."""
    n = len(h)

    x = h.copy()
    y = h.copy()
    t_k = 1.0
    lr = 1.5 / (k * dx * 3.0 + 1.0)

        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return best_h

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_val_weights(overlaps, alpha):
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    exp_s = np.exp(np.clip(shifted, -500, 0))
    total = np.sum(exp_s)
    value = m + np.log(total) / alpha
    weights = exp_s / total
    return value, weights


def compute_gradient_weighted(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-13:
            continue
        offset = idx - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h.copy(), 0.0, 1.0)
    step = min(0.8, 1.0 / (alpha * dx * 2.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_val_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 300:
                break
        prev_sm = sm
        grad = compute_gradient_weighted(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        improved = False
        s = step
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_val_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-14:
                break
        if not improved:
            step = s
            if s < 1e-10:
                break
            continue
        step = min(s * 1.5, 5.0)
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = project_onto_constraint(h.copy(), target)
    y = x.copy()
    t_k = 1.0
    step = min(0.5, 1.0 / (alpha * dx * 4.0 + 1.0))
    best_x = x.copy()
    best_sm = evaluate_score(x)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_val_weights(ov_y, alpha)
        grad_y = compute_gradient_weighted(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        score = evaluate_score(x_new)
        if score < best_sm:
            best_sm = score
            best_x = x_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return best_x


def topk_pgd(h, k, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = project_onto_constraint(h.copy(), target)
    step = 1.0 / (k * dx * 2.0 + 1.0)
    best_score = evaluate_score(h)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        idx = np.argpartition(overlaps, -k)[-k:]
        grad = compute_gradient_weighted(h, np.ones(len(idx)) / k, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        s = step
        improved = False
        for bt in range(25):
            h_trial = project_onto_constraint(h - s * grad, target)
            score = evaluate_score(h_trial)
            if score < best_score - 1e-15:
                h = h_trial
                best_score = score
                best_h = h_trial.copy()
                stall = 0
                improved = True
                step = min(s * 1.3, 10.0)
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = s
            stall += 1
            if stall > 150:
                break
    return best_h


def block_coordinate_optimize(h, target, deadline, block_size=8):
    n = len(h)
    h = project_onto_constraint(h.copy(), target)
    best_score = evaluate_score(h)
    best_h = h.copy()
    n_blocks = (n + block_size - 1) // block_size
    improvement = True
    while improvement and time.time() < deadline:
        improvement = False
        for b in range(n_blocks):
            if time.time() > deadline:
                break
            start = b * block_size
            end = min(n, start + block_size)
            h_block = h[start:end].copy()
            fixed = h.copy()
            for trial in range(50):
                h_block_new = h_block + np.random.randn(end - start) * 0.05 * (1.0 - trial / 50.0)
                h_trial = fixed.copy()
                h_trial[start:end] = np.clip(h_block_new, 0, 1)
                h_trial = project_onto_constraint(h_trial, target)
                score = evaluate_score(h_trial)
                if score < best_score - 1e-15:
                    h = h_trial
                    best_score = score
                    best_h = h_trial.copy()
                    h_block = h[start:end].copy()
                    improvement = True
                    break
    return best_h


def lbfgsb_refine(h, alpha, target, deadline):
    n = len(h)
    dx = 2.0 / n

    def obj(hv):
        ov = compute_overlaps(hv)
        sm, _ = smooth_max_val_weights(ov, alpha)
        pen = 1000.0 * (hv.sum() - target) ** 2
        return sm + pen

    def obj_grad(hv):
        ov = compute_overlaps(hv)
        sm, w = smooth_max_val_weights(ov, alpha)
        g = compute_gradient_weighted(hv, w, dx) + 2000.0 * (hv.sum() - target)
        return g

    bounds = [(0.0, 1.0)] * n
    max_iter = 500 if n <= 600 else 200
    res = minimize(obj, h, method='L-BFGS-B', jac=obj_grad, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
    return res.x if res.x is not None else h


def upsample_h(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
