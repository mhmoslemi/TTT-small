import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val_and_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = float(m + np.log(wsum) / alpha)
    grad = np.zeros(n)
    sig_idx = np.where(weights > 1e-12)[0]
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return val, grad

def fista_optimize(h0, alpha, n_iters, target, lr0=None):
    n = len(h0)
    h = h0.copy()
    x = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_val = evaluate_score(h0)
    if lr0 is None:
        # Estimate Lipschitz constant
        grad0 = lse_val_and_grad(h, alpha)[1]
        L = np.max(np.abs(grad0)) + 1.0
        if L < 1e-10:
            L = 1.0
        lr0 = 1.0 / L
    for it in range(n_iters):
        val, grad = lse_val_and_grad(x, alpha)
        # Backtracking
        lr = lr0
        while True:
            h_new = project_box_sum(x - lr * grad, target)
            val_new = lse_val_and_grad(h_new, alpha)[0]
            if val_new <= val + np.dot(grad, h_new - x) + (1.0/(2*lr)) * np.dot(h_new-x, h_new-x):
                break
            lr *= 0.5
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
        x = h_new + ((t-1.0)/t_new) * (h_new - h)
        h = h_new
        t = t_new
        lr0 = lr
        score = evaluate_score(h)
        if score < best_val:
            best_val = score
            best_h = h.copy()
    return best_h, best_val

def coordinate_descent(h, alpha, penalty, n_sweeps):

    target = n / 2.0
    h = h.copy()
    c = compute_overlaps(h)
    sum_h = float(np.sum(h))
    
    for sweep in range(n_sweeps):
        for i in range(n):
            old_vi = h[i]
            # Compute A_k = dc[k]/dh[i]
            m = np.arange(2*n-1) - (n-1)  # m_k for each k
            # A_k = (1-h[i+m_k]) * 1_{0<=i+m_k<=n-1} - h[i-m_k] * 1_{0<=i-m_k<=n-1}
            A = np.zeros(2*n-1)
            valid1 = (i + m >= 0) & (i + m <= n-1)
            valid2 = (i - m >= 0) & (i - m <= n-1)
            A[valid1] = 1.0 - h[i + m[valid1]]
            A[valid2] -= h[i - m[valid2]]
            # B_k = c[k] - A_k * old_vi
            B = c - A * old_vi
            
            # 1D minimization: minimize (1/alpha)*log(sum exp(alpha*(A*v + B))) + penalty*(sum_h - old_vi + v - target)^2
            # over v in [0,1]

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val_and_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = float(m + np.log(wsum) / alpha)
    grad = np.zeros(n)
    sig_idx = np.where(weights > 1e-12)[0]
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return val, grad


def fista_optimize(h0, alpha, n_iters, target, t0):
    n = len(h0)
    h = h0.copy()
    x = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_score = evaluate_score(h0)
    grad0 = lse_val_and_grad(h, alpha)[1]
    L_est = float(np.max(np.abs(grad0))) * alpha + 1.0
    if L_est < 1e-8:
        L_est = 1.0
    lr = 1.0 / L_est
    for it in range(n_iters):
        if time.time() - t0 > 1200:
            break
        val, grad = lse_val_and_grad(x, alpha)
        h_new = project_box_sum(x - lr * grad, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        x = h_new + ((t - 1.0) / t_new) * (h_new - h)
        h = h_new
        t = t_new
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if it % 200 == 199:
            lr *= 0.95
    return best_h, best_score


def coord_desc_sweep(h, alpha, penalty, n_sweeps, t0, rng):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c = compute_overlaps(h)
    sum_h = float(np.sum(h))
    m = np.arange(2 * n - 1) - (n - 1)
    best_score = evaluate_score(h)
    for sweep in range(n_sweeps):
        if time.time() - t0 > 1250:
            break
        perm = rng.permutation(n)
        for i in perm:
            vi = h[i]
            A = np.zeros(2 * n - 1)
            valid1 = (i + m >= 0) & (i + m <= n - 1)
            valid2 = (i - m >= 0) & (i - m <= n - 1)
            A[valid1] = 1.0 - h[i + m[valid1]]
            A[valid2] -= h[i - m[valid2]]
            B = c - A * vi
            alpha_A = alpha * A
            alpha_B = alpha * B
            rest = sum_h - vi - target
            pen_c = 2.0 * penalty
            lo_v, hi_v = 0.0, 1.0
            for _ in range(30):
                mid_v = (lo_v + hi_v) * 0.5
                xv = alpha_A * mid_v + alpha_B
                xv_shift = xv - np.max(xv)
                exp_xv = np.exp(xv_shift)
                w = exp_xv / np.sum(exp_xv)
                g = float(np.dot(A, w)) + pen_c * (rest + mid_v)
                if g > 0.0:
                    hi_v = mid_v
                else:
                    lo_v = mid_v
            new_vi = (lo_v + hi_v) * 0.5
            h[i] = new_vi
            delta = new_vi - vi
            if abs(delta) > 1e-15:
                sum_h += delta
                c += A * delta
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
    return h, best_score


def active_set_optimize(h, top_k, alpha, n_iters, t0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_score = evaluate_score(h)
    c = compute_overlaps(h)
    for it in range(n_iters):
        if time.time() - t0 > 1270:
            break
        top_idx = np.argsort(-c)[:top_k]
        sub_c = c[top_idx].copy()
        m_val = np.max(sub_c)
        exp_sub = np.exp(alpha * (sub_c - m_val))
        w = exp_sub / np.sum(exp_sub)
        grad = np.zeros(n)
        for idx in top_idx:
            wi = w[np.where(top_idx == idx)[0][0]]
            L = idx - (n - 1)
            if L >= 0:
                grad[:n - L] += wi * (1.0 - h[L:])
                grad[L:] -= wi * h[:n - L]
            else:
                M = -L
                grad[M:] += wi * (1.0 - h[:n - M])
                grad[:n - M] -= wi * h[M:]
        lr = 1.0 / (alpha * n + 1.0)
        h_new = project_box_sum(h - lr * grad, target)
        c_new = compute_overlaps(h_new)
        score = float(np.max(c_new)) * (2.0 / n)
        if score < best_score:
            best_score = score
            h = h_new
            c = c_new
        else:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, target)
            c_new = compute_overlaps(h_new)
            score = float(np.max(c_new)) * (2.0 / n)
            if score < best_score:
                best_score = score
                h = h_new
                c = c_new
    return h, best_score


def bilocal_search(h0, n_moves, t0, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    c = compute_overlaps(h)
    for it in range(n_moves):
        if time.time() - t0 > 1260:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        c_new = compute_overlaps(h)
        score = float(np.max(c_new)) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            c = c_new
        else:
            h[i], h[j] = old_i, old_j
    return best_h, best_score
