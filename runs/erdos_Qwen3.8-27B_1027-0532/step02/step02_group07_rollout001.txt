import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    def obj(h):
        return smooth_max(compute_correlations(h), alpha)
    res = minimize(
        obj, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def cd_single_update(h, j, corr, bisection_iters=25):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    for _ in range(bisection_iters):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    t_star = float(np.clip(t_star, 0.0, 1.0))
    return t_star, slopes


def coordinate_descent(h, n_sweeps=200, time_budget=60.0, bisection_iters=25, tol=1e-12):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = cd_single_update(h, j, corr, bisection_iters)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h

def periodic_optimize(h_flat, n, time_budget=30.0, n_sweeps=100):
    h = h_flat
    for period in [2, 3, 4, 5]:
        if n % period != 0:
            continue
        n_base = n // period
        h_base = np.zeros(n_base)
        for p in range(period):
            idx = np.arange(p * n_base, (p + 1) * n_base)
            h_base += h[idx]
        h_base /= period
        h_base = project_sum(h_base, n_base / 2.0)

        h_rep = project_sum(h_rep, n / 2.0)
        s = evaluate_score(h_rep)
        if s < evaluate_score(h):
            h = h_rep
    return h

    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_sum(h_new, n_new / 2.0)
    return h_new
