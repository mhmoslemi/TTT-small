def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_full(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]

def constraint_jac(h):

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,

        return h_opt, compute_c5(h_opt)

def continuous_refine(h_init, eps_list, maxiter_per_eps, time_limit, start_time):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    for eps in eps_list:
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        h, c5 = slsqp_optimize(h, eps, maxiter_per_eps)
        if c5 < best_c5:

def targeted_local_search(h_init, n_iterations, rng):
    """Targeted local search that focuses on reducing the worst lag."""
    n = len(h_init)

    h = h_init.copy()
    target_sum = n / 2.0

    best_corr = np.correlate(h, 1.0 - h, mode='full') * dx

        # Find worst lag
        worst_k = int(np.argmax(best_corr))
        s = worst_k - (n - 1)

        if s >= 0:

            # To reduce: find i where h[i+s]=1 and h[i]=0, try to flip

                h[candidate_idx + s] = 0.0
                new_c5 = compute_c5(h)
                # Need to maintain sum constraint - find a 0 to make 1

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)

    x -= np.max(x)
