Let me verify: corr_k = sum_i h[i](1-h[i+k]). If I change h[j] by δ and h[j'] by -δ (where j' = N-1-j), then:

                h[j] = v_new
                h[jp] = 1.0 - v_new
                corr += delta * a
                cur = np.max(corr) * dx
                if cur < best_score:
                    best_score = cur
                    improved = True

        if not improved and sweep > 3:
            break

    best_score = evaluate_score(h)
    return h, best_score


def coord_descent_general(h, n_sweeps=50, grid_size=201, rng=None, time_limit=None, t0=None):
    N = len(h)
    n_corr = 2 * N - 1
    dx = 2.0 / N
    k_vals = np.arange(n_corr) - (N - 1)
    target = N / 2.0

    h = h.copy()
    corr = compute_correlations(h)
    best_score = np.max(corr) * dx

    for sweep in range(n_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        if rng is not None:
            order = rng.permutation(N)
        else:
            order = np.arange(N)

        improved = False
        for j in order:
            a = 1.0 - safe_get(h, j + k_vals) - safe_get(h, j - k_vals)
            b = corr - a * h[j]
            grid = np.linspace(0.0, 1.0, grid_size)
            vals = np.max(b[:, None] + a[:, None] * grid[None, :], axis=0)
            best_idx = np.argmin(vals)
            v_new = grid[best_idx]

            if 0 < best_idx < grid_size - 1:
                lo = grid[best_idx - 1]
                hi = grid[best_idx + 1]
                fg = np.linspace(lo, hi, 501)
                vals2 = np.max(b[:, None] + a[:, None] * fg[None, :], axis=0)
                v_new = fg[np.argmin(vals2)]

            if abs(v_new - h[j]) > 1e-12:
                delta = v_new - h[j]
                h[j] = v_new
                corr += delta * a
                cur = np.max(corr) * dx
                if cur < best_score:
                    best_score = cur
                    improved = True

        h = project_feasible(h, target)
        corr = compute_correlations(h)
        cur = np.max(corr) * dx
        if cur < best_score:
            best_score = cur

        if not improved and sweep > 3:
            break

    best_score = evaluate_score(h)
    return h, best_score


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def slsqp_opt(h0, alpha, maxiter, n_points):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N

    def fun(x):
        return smooth_max_val(compute_correlations(x), alpha)

    def cons_fun(x):
        return np.sum(x) - n_points / 2.0

    res = minimize(
        fun, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(len(x))}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    init = project_feasible(init, target)

    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: Anti-palindromic coordinate descent from initial
    h_ap = to_antipalindromic(init)
    h_ap, s_ap = coord_descent_antipal(h_ap, n_sweeps=300, grid_size=201,
                                        rng=rng, time_limit=budget_s * 0.15, t0=t0)
    if s_ap < best_score:
        best_score = s_ap
        best_h = h_ap.copy()

    # Phase 2: SLSQP polish on antipalindromic solution
    if elapsed() < budget_s * 0.35:
        for alpha in [200, 1000, 5000]:
            if elapsed() > budget_s * 0.35:
                break
            maxiter = max(100, min(500, 100000 // n))
            res_x = slsqp_opt(best_h, alpha, maxiter, n)
            if res_x is not None and np.all(np.isfinite(res_x)):
                h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 3: Random restarts with antipalindromic coordinate descent
    n_restarts = 30
    restart_budget = budget_s * 0.35
    per_restart = max(1.0, restart_budget / n_restarts)
    for r in range(n_restarts):
        if elapsed() > budget_s * 0.55:
            break
        scale = 0.02 + 0.08 * (r / float(n_restarts))
        pert = best_h + rng.randn(n) * scale
        pert = to_antipalindromic(pert)
        pert = np.clip(pert, 0.0, 1.0)
        h_new, s_new = coord_descent_antipal(pert, n_sweeps=150, grid_size=201,
                                              rng=rng, time_limit=per_restart, t0=t0)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 4: Try different resolutions (downsample then optimize)
    if elapsed() < budget_s * 0.75:
        base_x = np.linspace(0.0, 2.0, n, endpoint=False)
        for n_try in [101, 201, 301]:
            if elapsed() > budget_s * 0.75:
                break
            x_new_arr = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init_low = np.interp(x_new_arr, base_x, best_h)
            if n_try % 2 == 0:
                n_try = n_try + 1
                x_new_arr = np.linspace(0.0, 2.0, n_try, endpoint=False)
                h_init_low = np.interp(x_new_arr, base_x, best_h)
            h_low = to_antipalindromic(h_init_low)
            h_low = np.clip(h_low, 0.0, 1.0)
            h_low, s_low = coord_descent_antipal(h_low, n_sweeps=200, grid_size=201,
                                                  rng=rng, time_limit=30, t0=t0)
            # Upsample back
            x_orig = np.linspace(0.0, 2.0, n, endpoint=False)
            x_high = np.linspace(0.0, 2.0, len(h_low), endpoint=False)
            h_up = np.interp(x_orig, x_high, h_low)
            h_up = to_antipalindromic(h_up)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up, s_up = coord_descent_antipal(h_up, n_sweeps=200, grid_size=201,
                                                rng=rng, time_limit=40, t0=t0)
            if s_up < best_score:
                best_score = s_up
                best_h = h_up.copy()

    # Phase 5: General (non-antipalindromic) coordinate descent
    if elapsed() < budget_s * 0.85:
        h_gen = best_h.copy()
        h_gen, s_gen = coord_descent_general(h_gen, n_sweeps=100, grid_size=201,
                                             rng=rng, time_limit=budget_s * 0.1, t0=t0)
        if s_gen < best_score:
            best_score = s_gen
            best_h = h_gen.copy()

    # Phase 6: More random restarts at the best resolution with mixed strategies
    if elapsed() < budget_s * 0.92:
        n_more = 20
        for r in range(n_more):
            if elapsed() > budget_s * 0.92:
                break
            strat = r % 3
            if strat == 0:
                pert = best_h + rng.randn(n) * 0.05
                pert = to_antipalindromic(pert)
                pert = np.clip(pert, 0.0, 1.0)
                h_new, s_new = coord_descent_antipal(pert, n_sweeps=150, grid_size=201,
                                                      rng=rng, time_limit=10, t0=t0)
            elif strat == 1:
                pert = best_h.copy()
                idx = rng.choice(n, size=n // 5, replace=False)
                pert[idx] = rng.uniform(0, 1, size=len(idx))
                pert = to_antipalindromic(pert)
                pert = np.clip(pert, 0.0, 1.0)
                h_new, s_new = coord_descent_antipal(pert, n_sweeps=150, grid_size=201,
                                                      rng=rng, time_limit=10, t0=t0)
            else:
                pert = best_h.copy()
                h_new, s_new = coord_descent_general(pert, n_sweeps=80, grid_size=201,
                                                      rng=rng, time_limit=10, t0=t0)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new.copy()

    # Phase 7: Final high-resolution antipalindromic coordinate descent
    if elapsed() < budget_s * 0.97:
        h_final = to_antipalindromic(best_h)
        h_final = np.clip(h_final, 0.0, 1.0)
        h_final, s_final = coord_descent_antipal(h_final, n_sweeps=500, grid_size=401,
                                                  rng=rng, time_limit=budget_s * 0.03, t0=t0)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
