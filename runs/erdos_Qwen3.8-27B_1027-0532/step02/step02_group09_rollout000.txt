Let me also add the "worst-lag targeted" binary move:
- Find k* = argmax overlap
- Find the positions (i, i+k*) where h[i]=1, h[i+k*]=0 (these contribute to the overlap)
- Try flipping h[i] to 0 (which reduces the overlap at k* by 1 but might affect other lags)
- Or try flipping h[i+k*] to 1 (same effect)
- Choose the flip that minimizes the new max

This is more targeted than random swaps.

Or simpler: **iterated local search (ILS)**:
- Do a local search (steepest descent: try all single swaps, take the best)
- Perturb the solution (randomly flip k bits)
- Repeat

ILS is often very effective for combinatorial optimization.

Let me implement ILS for the binary case:

```
ils_binary(h_init, n_restarts, n_il_iters):

            best_move = None
            best_new_score = evaluate(h)
            ones = where(h==1)
            zeros = where(h==0)
            # Try a random subset of swaps (too expensive to try all)
            for _ in range(100):
                i = random(ones)
                j = random(zeros)
                h[i], h[j] = h[j], h[i]
                s = evaluate(h)
                if s < best_new_score:
                    best_new_score = s
                    best_move = (i, j)
                h[i], h[j] = h[j], h[i]
            if best_move is None:
                break
            i, j = best_move
            h[i], h[j] = h[j], h[i]
        
        # Update best
        if evaluate(h) < best_score:
            best_h = h.copy()
            best_score = evaluate(h)
        
        # Perturb
        k_perturb = rng.randint(3, 8)
        ones = where(h==1)
        zeros = where(h==0)
        for _ in range(k_perturb):
            i = random(ones)
            j = random(zeros)
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    lo = -2.0
    hi = 3.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad_full(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = float(np.sum(exp_vals))
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for idx in range(len(overlaps)):
        w = weights[idx]
        if w < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def cont_func(h, alpha, penalty):
    n = len(h)
    obj = lse_val(compute_overlaps(h), alpha)
    grad = lse_grad_full(h, alpha)
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad = grad + 2.0 * penalty * s
    return obj, grad


def lbfgsb_run(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            cont_func, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18}
        )
        return res.x
    except Exception:
        return None


def slsqp_func(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_jac(h, alpha):
    return lse_grad_full(h, alpha)


def slsqp_cons_fun(h):
    return np.sum(h) - len(h) / 2.0


def slsqp_cons_jac(h):
    return np.ones(len(h))


def slsqp_run(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_func, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_jac, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def ils_binary(h_init, n_restarts, il_trials, rng, time_fn):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for restart in range(n_restarts):
        if time_fn() < 60:
            break
        cur_score = evaluate_score(h)
        for il in range(200):
            if time_fn() < 60:
                break
            best_move = None
            best_new_score = cur_score
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            for t in range(il_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                s = evaluate_score(h)
                if s < best_new_score:
                    best_new_score = s
                    best_move = (i, j)
                h[i] = 1.0
                h[j] = 0.0
            if best_move is None:
                break
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            cur_score = best_new_score
        
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        
        k_pert = rng.randint(3, 9)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(k_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    
    return best_h, best_score


def targeted_binary_move(h, rng, max_attempts=50):
    n = len(h)
    overlaps = compute_overlaps(h)
    k_star = int(np.argmax(overlaps))
    L = k_star - (n - 1)
    candidates = []
    if L >= 0:
        for i in range(n - L):
            if h[i] == 1.0 and h[i + L] == 0.0:
                candidates.append(('flip_h_i', i))
                if i + L != i:
                    candidates.append(('flip_h_j', i + L))
    else:
        M = -L
        for i in range(n - M):
            if h[i] == 0.0 and h[i + M] == 1.0:
                candidates.append(('flip_h_i', i))
                if i + M != i:
                    candidates.append(('flip_h_j', i + M))
    
    if len(candidates) == 0:
        return h.copy()
    
    cur_score = evaluate_score(h)
    best_move = None
    best_new_score = cur_score
    
    step = max(1, len(candidates) // max_attempts)
    for ci in range(0, len(candidates), step):
        ctype, cpos = candidates[ci]
        h[cpos] = 1.0 - h[cpos]
        s = evaluate_score(h)
        h[cpos] = 1.0 - h[cpos]
        if s < best_new_score:
            best_new_score = s
            best_move = cpos
    
    if best_move is not None:
        h[best_move] = 1.0 - h[best_move]
    return h


def binary_refine_targeted(h_init, n_rounds, rng, time_fn):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for r in range(n_rounds):
        if time_fn() < 30:
            break
        h = targeted_binary_move(h, rng)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    return best_h, best_score


def multi_start_ils(h_base, n_starts, il_trials, rng, time_fn):

            h_pert[i] = 0.0
            h_pert[j] = 1.0

            best_h = h_res.copy()
```
