```python
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _adam_optimize(h_init, n, target_sum, n_iter, lr, top_k, deadline, check_every=200):
    h = h_init.copy()
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 15
    
    for t in range(1, n_iter + 1):
        if time.time() > deadline - 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            grad += _corr_grad(h, lag)
        grad /= len(top_idx)
        
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad**2
        m_hat = m_state / (1 - beta1**t)
        v_hat = v_state / (1 - beta2**t)
        
        h = np.clip(h - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0, 1.0)
        h = _fix_sum(h, target_sum)
        
        if t % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_refine(h_init, n, alpha, maxiter, ftol, deadline):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        weights = np.exp(alpha * (corr - m))
        weights /= np.sum(weights)
        grad = np.zeros(npts)
        for k_idx in range(len(weights)):
            if weights[k_idx] > 1e-6:
                lag = k_idx - (npts - 1)
                grad += weights[k_idx] * _corr_grad(x, lag)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(npts)}]
    
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def _coordinate_descent_polish(h_init, n, target_sum, max_sweeps, deadline):
    h = h_init.copy()
    best_c5 = _c5_of(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline - 10:
            break
        indices = rng.permutation(n)
        improved = False
        
        for i in indices:
            if time.time() > deadline - 10:
                break
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_max = np.max(corr)
            
            # Compute gradient of each correlation w.r.t. h[i]
            # Use the formula: for lag k, d(C[k])/d(h[i])
            grad_i = np.zeros(2 * n - 1)
            for k in range(-(n-1), n):
                m = k + (n - 1)
                if k == 0:
                    grad_i[m] = 1.0 - 2.0 * h[i]
                elif k > 0:
                    if k <= i < n:
                        grad_i[m] += (1.0 - h[i - k])
                    if i < n - k:
                        grad_i[m] -= h[i + k]
                else:
                    L = -k
                    if L <= i <= n - 1:
                        grad_i[m] += (1.0 - h[i - L])
                    if i <= n - 1 - L:
                        grad_i[m] -= h[i + L]
            
            # Find v in [0,1] minimizing max_m (corr[m] + (v-h[i])*grad_i[m])
            active_mask = corr >= current_max - 0.015
            active_idx = np.where(active_mask)[0]
            
            if len(active_idx) < 2:
                active_idx = np.argsort(corr)[-5:]
            
            a_vals = corr[active_idx] - h[i] * grad_i[active_idx]
            b_vals = grad_i[active_idx]
            
            candidates = [0.0, 1.0, h[i]]
            for p in range(len(active_idx)):
                for q in range(p + 1, len(active_idx)):
                    if abs(b_vals[p] - b_vals[q]) > 1e-12:
                        v_int = (a_vals[q] - a_vals[p]) / (b_vals[p] - b_vals[q])
                        if -0.01 <= v_int <= 1.01:
                            candidates.append(np.clip(v_int, 0.0, 1.0))
            
            best_v = h[i]
            best_obj = np.max(a_vals + b_vals * h[i])
            for v in candidates:
                obj_val = np.max(a_vals + b_vals * v)
                if obj_val < best_obj - 1e-14:
                    best_obj = obj_val
                    best_v = v
            
            if abs(best_v - h[i]) > 1e-10:
                h[i] = best_v
                h = _fix_sum(h, target_sum)
                improved = True
        
        c5 = _c5_of(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c5
```

```python
import time
import numpy as np
from scipy.optimize import minimize

_G_n = 100
_G_target_sum = 50.0
_G_alpha = 200.0
_G_dx = 0.02

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _slsqp_con_fun(x):
    return x.sum() - _G_target_sum

def _slsqp_con_jac(x):
    return np.ones(_G_n)

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full') * _G_dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(_G_alpha * (corr - m)))) / _G_alpha

def _slsqp_grad(x):
    npts = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * _G_dx
    m = np.max(corr)
    weights = np.exp(_G_alpha * (corr - m))
    weights /= np.sum(weights)
    grad = np.zeros(npts)
    for k_idx in range(2 * npts - 1):
        w = weights[k_idx]
        if w > 1e-7:
            lag = k_idx - (npts - 1)
            grad += w * _corr_grad(x, lag)
    return grad * _G_dx

def _slsqp_refine(h_init, n, alpha, maxiter, ftol):
    global _G_n, _G_target_sum, _G_alpha, _G_dx
    _G_n = n
    _G_target_sum = n / 2.0
    _G_alpha = float(alpha)
    _G_dx = 2.0 / n
    npts = n
    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h_init, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def _adam_optimize(h_init, n, target_sum, n_iter, lr, top_k, deadline, check_every=200):
    h = h_init.copy()
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 15
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, n_iter + 1):
        if time.time() > deadline - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            grad += _corr_grad(h, lag)
        grad /= len(top_idx)
        m_s = beta1 * m_s + (1.0 - beta1) * grad
        v_s = beta2 * v_s + (1.0 - beta2) * grad * grad
        m_hat = m_s / (1.0 - beta1 ** t)
        v_hat = v_s / (1.0 - beta2 ** t)
        h = np.clip(h - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0, 1.0)
        h = _fix_sum(h, target_sum)
        if t % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _coord_grad_at_i(h, i):
    n = len(h)
    grad = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        m = k + (n - 1)
        if k == 0:
            grad[m] = 1.0 - 2.0 * h[i]
        elif k > 0:
            if k <= i:
                grad[m] += (1.0 - h[i - k])
            if i <= n - 1 - k:
                grad[m] -= h[i + k]
        else:
            L = -k
            if i <= n - 1 - L:
                grad[m] += (1.0 - h[i + L])
            if i >= L:
                grad[m] -= h[i - L]
    return grad

def _coord_descent_polish(h_init, n, target_sum, max_sweeps, deadline, seed=42):
    h = h_init.copy()
    best_c5 = _c5_of(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline - 5:
            break
        indices = rng.permutation(n)
        improved = False
        for i in indices:
            if time.time() > deadline - 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_max = np.max(corr)
            grad_i = _coord_grad_at_i(h, i)
            # Active set: correlations within tolerance of max
            tol_frac = 0.02
            active_mask = corr >= current_max - tol_frac
            active_idx = np.where(active_mask)[0]
            if len(active_idx) < 3:
                active_idx = np.argsort(corr)[-8:]
            a_vals = corr[active_idx] - h[i] * grad_i[active_idx]
            b_vals = grad_i[active_idx]
            # Find v in [0,1] minimizing max(a + b*v)
            candidates = [0.0, 1.0, h[i]]
            for p in range(len(active_idx)):
                for qidx in range(p + 1, len(active_idx)):
                    db = b_vals[p] - b_vals[qidx]
                    if abs(db) > 1e-14:
                        vi = (a_vals[qidx] - a_vals[p]) / db
                        if 0.0 <= vi <= 1.0:
                            candidates.append(vi)
            best_v = h[i]
            best_obj = np.max(a_vals + b_vals * h[i])
            for v in candidates:
                ov = np.max(a_vals + b_vals * v)
                if ov < best_obj - 1e-15:
                    best_obj = ov
                    best_v = v
            if abs(best_v - h[i]) > 1e-10:
                h[i] = best_v
                h = _fix_sum(h, target_sum)
                improved = True
        c5 = _c5_of(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

        if c5_new < best_c5:

        h_cur = h_new.copy()

            c5_new = _c5_of(h_new)

        x = np.linspace(0, 2.0, n, endpoint=False)
```
