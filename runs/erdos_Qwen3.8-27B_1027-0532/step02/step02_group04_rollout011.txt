    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = shifted - np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_topk_value(corr, alpha, k):
    # Focus on top-k correlations
    m = np.max(corr)
    idx = np.argsort(corr)[-k:]
    topk = corr[idx]
    shifted = alpha * (topk - m)
    shifted = shifted - np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_topk_gradient(h, alpha, k):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    idx = np.argsort(corr)[-k:]
    topk = corr[idx]

    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)

        lag = N - 1 - index

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def smooth_max(corr, alpha):
    m = np.max(corr)
    t = alpha * (corr - m) - alpha * np.max(corr - m)
    return m + np.log(np.sum(np.exp(t))) / alpha

def smooth_max_topk(corr, alpha, k):
    m = np.max(corr)
    idx = np.argsort(corr)[-k:]
    t = alpha * (corr[idx] - m)
    t = t - np.max(t)
    return m + np.log(np.sum(np.exp(t))) / alpha

def grad_corr_at_lag(h, lag):
    """Gradient of correlation at given lag w.r.t. h"""
    N = len(h)
    g = np.zeros(N)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        if lag < N:
            g[:N-lag] += 1.0
            g[:N-lag] -= h[lag:]
            g[lag:] -= h[:N-lag]
    else:
        l = -lag
        if l < N:
            g[l:] += 1.0
            g[l:] -= h[:N-l]
            g[:N-l] -= h[l:]
    return g

def smooth_max_grad(h, alpha, topk=None):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    if topk is not None:
        idx = np.argsort(corr)[-topk:]
        t = alpha * (corr[idx] - m)
        t = t - np.max(t)
        weights = np.exp(t) / np.sum(np.exp(t))
    else:
        t = alpha * (corr - m)
        t = t - np.max(t)
        weights = np.exp(t) / np.sum(np.exp(t))
        idx = np.arange(len(corr))
    
    g = np.zeros(N)
    for i, cidx in enumerate(idx):
        w = weights[i]
        if w < 1e-12:
            continue
        lag = N - 1 - cidx
        g += w * grad_corr_at_lag(h, lag)
    return g

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def optimize_lbfgs(h0, alpha, n_pts, mu, maxiter, topk=None):
    N = n_pts
    
    def obj(h):
        corr = compute_correlations(h)
        if topk:
            sm = smooth_max_topk(corr, alpha, topk)
        else:
            sm = smooth_max(corr, alpha)
        s = np.sum(h) - N / 2.0
        return sm + mu * s * s
    
    def grad(h):
        g = smooth_max_grad(h, alpha, topk)
        s = np.sum(h) - N / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def _grad_single_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        if lag < N:
            g[:N - lag] += 1.0 - h[lag:]
            g[lag:] -= h[:N - lag]
    else:
        l = -lag
        if l < N:
            g[l:] += 1.0 - h[:N - l]
            g[:N - l] -= h[l:]
    return g


def _smooth_max_obj_grad(h, alpha, topk):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    if topk is not None and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
        t = alpha * (corr[idx] - m)
        t = t - np.max(t)
        w = np.exp(t)
        w = w / w.sum()
        active_idx = idx
        active_w = w
    else:
        t = alpha * (corr - m)
        t = t - np.max(t)
        w = np.exp(t)
        w = w / w.sum()
        active_idx = np.arange(len(corr))
        active_w = w
    obj_val = m + np.log(np.sum(np.exp(alpha * (corr[active_idx] - m) - np.max(alpha * (corr[active_idx] - m))))) / alpha
    g = np.zeros(N)
    for i in range(len(active_idx)):
        wi = active_w[i]
        if wi < 1e-12:
            continue
        lag = N - 1 - active_idx[i]
        g += wi * _grad_single_lag(h, lag)
    return obj_val, g


def _lbfgs_phase(h0, alpha, n_pts, mu, maxiter, topk=None):
    N = n_pts

    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        if topk is not None and topk < len(corr):
            idx = np.argsort(corr)[-topk:]
            t = alpha * (corr[idx] - m)
            t = t - np.max(t)
            sm = m + np.log(np.sum(np.exp(t))) / alpha
        else:
            t = alpha * (corr - m)
            t = t - np.max(t)
            sm = m + np.log(np.sum(np.exp(t))) / alpha
        s = np.sum(h) - N / 2.0
        return sm + mu * s * s

    def grad(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        if topk is not None and topk < len(corr):
            idx = np.argsort(corr)[-topk:]
            t = alpha * (corr[idx] - m)
            t = t - np.max(t)
            active_w = np.exp(t)
            active_w = active_w / np.sum(active_w)
            active_idx = idx
        else:
            t = alpha * (corr - m)
            t = t - np.max(t)
            active_w = np.exp(t)
            active_w = active_w / np.sum(active_w)
            active_idx = np.arange(len(corr))
        g = np.zeros(N)
        for i in range(len(active_idx)):
            wi = active_w[i]
            if wi < 1e-12:
                continue
            lag = N - 1 - active_idx[i]
            g += wi * _grad_single_lag(h, lag)
        s = np.sum(h) - N / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x
