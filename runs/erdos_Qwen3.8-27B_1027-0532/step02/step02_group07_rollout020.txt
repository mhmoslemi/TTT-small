import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def compute_gradient_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    c = corr_full * dx
    max_c = np.max(c)
    exps = np.exp(alpha * (c - max_c))
    p = exps / np.sum(exps)
    p_n = p[:n]
    one_minus_h = 1.0 - h
    conv1 = np.convolve(p_n, one_minus_h, mode='full')[:n]
    second = np.correlate(h, p_n, mode='full')[:n]
    grad = dx * (conv1 - second)
    return grad


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return smooth_max(corr * dx, alpha)


def smooth_obj_full(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    c = corr * dx
    return smooth_max(c, alpha), compute_gradient_smooth_max(h, alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    def obj(x):
        return smooth_obj(x, alpha)

    def obj_grad(x):
        val, grad = smooth_obj_full(x, alpha)
        return val, grad

    res = minimize(
        obj_grad, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(40):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-11, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    order = np.arange(n)

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        np.random.shuffle(order)
        max_change = 0.0
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def structured_perturbation(h, rng, mode):
    n = len(h)
    h_p = h.copy()
    if mode == 0:
        # Block flip
        block_size = rng.randint(max(2, n // 20), max(3, n // 5))
        start = rng.randint(0, n - block_size)
        h_p[start:start+block_size] = 1.0 - h_p[start:start+block_size]
    elif mode == 1:
        # Gaussian perturbation
        scale = rng.uniform(0.05, 0.25)
        h_p += rng.randn(n) * scale
    elif mode == 2:
        # Circular shift
        shift = rng.randint(1, n // 2)
        h_p = np.roll(h_p, shift)
    elif mode == 3:
        # Smooth perturbation
        kernel_size = rng.randint(3, min(20, n // 10))
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4.0))
        kernel /= kernel.sum()
        h_p = np.convolve(h_p, kernel, mode='same')
    elif mode == 4:
        # Binary snap with noise
        threshold = rng.uniform(0.3, 0.7)
        h_bin = (h_p > threshold).astype(float)
        h_p = h_bin + rng.randn(n) * 0.1
    else:
        # Random block move
        block_size = rng.randint(max(2, n // 30), max(3, n // 8))
        start1 = rng.randint(0, n - block_size)
        block = h_p[start1:start1+block_size].copy()
        h_p[start1:start1+block_size] = h_p[start1:block_size] if start1 >= block_size else 0.5
        start2 = rng.randint(0, n - block_size)
        h_p[start2:start2+block_size] = block
    return project_sum(np.clip(h_p, 0.0, 1.0), n / 2.0)
