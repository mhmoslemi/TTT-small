def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

            if h_try[i] == 1.0:

            if c5 < best_c5 - 1e-15:

    # term2[j] = sum_{l: 0<=j-l<n} w_l * h[j-l]

    # l ranges from -(n-1) to (n-1)

    # w_ext has length 2n-1, centered at n-1
    # term1[j] = sum_l w_l * (1-h[j+l]) for valid l,j+l
    # = sum_l w_l - sum_l w_l * h[j+l] (for valid ranges)

    grad = np.zeros(n)
    # For each shift l, the contribution to grad is:
    # For j where both j and j+l are in [0,n): grad[j] += w_l * (1-h[j+l])
    # For j where both j and j-l are in [0,n): grad[j] -= w_l * h[j-l]

        w = weights[l + (n-1)]
        if w < 1e-14:

        # First term: j ranges such that 0<=j<n and 0<=j+l<n
        j_min = max(0, -l)
        j_max = min(n, n - l)  # exclusive
        if j_max > j_min:

        j_max2 = min(n, n + l)
        if j_max2 > j_min2:
            grad[j_min2:j_max2] -= w * h[j_min2 - l : j_max2 - l]
    
    grad *= dx
    return grad

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def to_binary(h, n):
    n_target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_target]] = 1.0
    return h_bin

def smooth_max_val(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad_vec(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        w = weights[l + (n - 1)]
        if w < 1e-14:
            continue
        j_min = max(0, -l)
        j_max = min(n, n - l)
        if j_max > j_min:
            grad[j_min:j_max] += w * (1.0 - h[j_min + l:j_max + l])
        j_min2 = max(0, l)
        j_max2 = min(n, n + l)
        if j_max2 > j_min2:
            grad[j_min2:j_max2] -= w * h[j_min2 - l:j_max2 - l]
    grad *= dx
    return grad

def obj_func(x, eps, n):
    return smooth_max_val(x, eps, n)

def grad_func(x, eps, n):
    return smooth_max_grad_vec(x, eps, n)

def constr_func(x, n):
    return np.sum(x) - n / 2.0

def constr_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h_start, n, eps, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_max_val(x, eps, n),
            h_start.copy(),
            jac=lambda x: smooth_max_grad_vec(x, eps, n),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-18}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h_start, evaluate_c5(h_start)

def lbfgs_smooth(h_start, n, eps, maxiter):
    bounds = [(0.0, 1.0)] * n
    penalty = 5000.0
    try:
        res = minimize(
            lambda x: smooth_max_val(x, eps, n) + penalty * (np.sum(x) - n / 2.0) ** 2,
            h_start.copy(),
            jac=lambda x: smooth_max_grad_vec(x, eps, n) + 2.0 * penalty * (np.sum(x) - n / 2.0) * np.ones(n),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h_start, evaluate_c5(h_start)

def sa_binary(h_bin, n, start_temp, cooling, n_iters, time_start, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5

    for it in range(n_iters):
        if time.time() - time_start > time_limit:
            break
        temp = start_temp * (cooling ** it)
        r = rng.random()
        h_new = h.copy()
        valid_move = False

        if r < 0.35:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0.0
                h_new[j] = 1.0
                valid_move = True
        elif r < 0.55:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(ones, size=2, replace=False)
                j1, j2 = rng.choice(zeros, size=2, replace=False)
                h_new[i1] = 0.0
                h_new[i2] = 0.0
                h_new[j1] = 1.0
                h_new[j2] = 1.0
                valid_move = True
        elif r < 0.70:
            if n > 4:
                i = rng.randint(0, n - 2)
                j = rng.randint(i + 1, n)
                h_new[i:j + 1] = h[i:j + 1][::-1]
                valid_move = True
        elif r < 0.85:
            i = rng.randint(0, n - 1)
            if h[i] != h[i + 1]:
                h_new[i] = h[i + 1]
                h_new[i + 1] = h[i]
                valid_move = True
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= 3 and len(zeros) >= 3:
                i1, i2, i3 = rng.choice(ones, size=3, replace=False)
                j1, j2, j3 = rng.choice(zeros, size=3, replace=False)
                h_new[i1] = 0.0
                h_new[i2] = 0.0
                h_new[i3] = 0.0
                h_new[j1] = 1.0
                h_new[j2] = 1.0
                h_new[j3] = 1.0
                valid_move = True

        if not valid_move:
            continue

        c5_new = evaluate_c5(h_new)
        delta = c5_new - current_c5
        if delta < 0 or (temp > 1e-12 and rng.random() < np.exp(-delta / temp)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    return best_h, best_c5

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def continuation_refine(h_start, n, eps_list, maxiter_per_eps, time_start, time_limit):
    h = h_start.copy()
    current_c5 = evaluate_c5(h)
    for eps in eps_list:
        if time.time() - time_start > time_limit:
            break
        h_new, c5_new = slsqp_smooth(h, n, eps, maxiter_per_eps)
        if c5_new <= current_c5 + 1e-12:
            h = h_new
            current_c5 = c5_new
    return h, current_c5

def coord_descent_binary(h_bin, n, max_sweeps, time_start, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_c5 = evaluate_c5(h)

    for sweep in range(max_sweeps):
        if time.time() - time_start > time_limit:
            break
        improved = False
        indices = list(range(n))
        rng.shuffle(indices)
        for i in indices:
            if time.time() - time_start > time_limit:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if h[i] == 1.0:
                if len(zeros) > 0:
                    j = zeros[rng.randint(len(zeros))]
                    h_try = h.copy()
                    h_try[i] = 0.0
                    h_try[j] = 1.0
                    c5 = evaluate_c5(h_try)
                    if c5 < best_c5 - 1e-15:
                        h = h_try
                        best_c5 = c5
                        improved = True
            else:
                if len(ones) > 0:
                    j = ones[rng.randint(len(ones))]
                    h_try = h.copy()
                    h_try[i] = 1.0
                    h_try[j] = 0.0
                    c5 = evaluate_c5(h_try)
                    if c5 < best_c5 - 1e-15:
                        h = h_try
                        best_c5 = c5
                        improved = True
        if not improved:
            break
    return h, best_c5

def kswap_local_search(h_bin, n, max_iters, time_start, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5

    for it in range(max_iters):
        if time.time() - time_start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        k = rng.randint(1, 4)
        if len(ones) < k or len(zeros) < k:
            k = min(k, len(ones), len(zeros))
        if k < 1:
            break
        i_idx = rng.choice(ones, size=k, replace=False)
        j_idx = rng.choice(zeros, size=k, replace=False)
        h_try = h.copy()
        h_try[i_idx] = 0.0
        h_try[j_idx] = 1.0
        c5_new = evaluate_c5(h_try)
        if c5_new < current_c5 - 1e-15:
            h = h_try
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new

    return best_h if False else h, best_c5
