def fft_full_correlate(a, v):
    """Compute np.correlate(a, v, 'full') using FFT."""
    n_a, n_v = len(a), len(v)
    N = 1
    while N < n_a + n_v - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v, N)
    full = np.fft.irfft(A * np.conj(V), N)
    return np.concatenate([full[N - (n_v - 1):], full[:n_a]])

def get_correlations(h):
    """Cross-correlation of h with 1-h."""
    return fft_full_correlate(h, 1.0 - h)

def binary_local_search(h, n_iter=1000):
    """Local search on binary-valued h (0 or 1, exactly n/2 ones)."""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Flip some 1s to 0s
        ones_idx = np.where(h_bin == 1)[0]
        flip = np.random.choice(ones_idx, size=n_ones - target, replace=False)
        h_bin[flip] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        flip = np.random.choice(zeros_idx, size=target - n_ones, replace=False)
        h_bin[flip] = 1

    current = h_bin.copy()

        current[j] = 1

            best_score = s
            h_bin = current.copy()

            current[i] = 1
            current[j] = 0
