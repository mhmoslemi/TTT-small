from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = h.copy()
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad

        k_star = np.argmax(corrs)
        r = k_star - (n - 1)
        # Gradient of corr[k_star] w.r.t. h
        grad = np.zeros(n)
        # For each m, grad[m] = (1-h[m+r]) - h[m-r] with boundary handling
        # h[m+r] is valid when 0 <= m+r < n
        # h[m-r] is valid when 0 <= m-r < n
        mr = np.arange(n) + r

        h = project_to_feasible(h + v, n)

def slp_step(h, n, dx):
    """One SLP iteration: linearize and solve LP"""
    from scipy.optimize import linprog
    # n variables h[0..n-1], plus M
    # Total: n+1 variables
    
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k (there are 2n-1 shifts):
    # corr_k(h_0) + sum_m grad_k[m] * (h[m] - h_0[m]) <= M
    # => sum_m grad_k[m] * h[m] - M <= M - corr_k(h_0) - sum_m grad_k[m] * h_0[m] + corr_k(h_0)
    # Wait, let me redo this.
    # corr_k_linear = corr_k(h_0) + sum_m grad_k[m] * (h[m] - h_0[m])
    # We want: corr_k_linear <= M
    # => corr_k(h_0) + sum_m grad_k[m] * h[m] - sum_m grad_k[m] * h_0[m] <= M
    # => sum_m grad_k[m] * h[m] - M <= -corr_k(h_0) + sum_m grad_k[m] * h_0[m]
    
    corrs = compute_correlations(h)
    n_shifts = len(corrs)
    
    # This creates a (2n-1) x (n+1) matrix, which for n=425 is 849 x 426
    # That's manageable for linprog
    
    A_ub = np.zeros((n_shifts, n + 1))
    b_ub = np.zeros(n_shifts)
    
    for k in range(n_shifts):
        r = k - (n - 1)
        # grad_k[m] = (1-h[m+r]) - h[m-r] with boundary handling
        grad_k = np.zeros(n)
        mr_plus = np.arange(n) + r
        valid_plus = (mr_plus >= 0) & (mr_plus < n)
        grad_k[valid_plus] += (1.0 - h[mr_plus[valid_plus]])
        mr_minus = np.arange(n) - r
        valid_minus = (mr_minus >= 0) & (mr_minus < n)
        grad_k[valid_minus] -= h[mr_minus[valid_minus]]
        
        A_ub[k, :n] = grad_k
        A_ub[k, n] = -1.0
        b_ub[k] = -corrs[k] + np.dot(grad_k, h)
    
    # Equality constraint: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[m] <= 1, M >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return h_new
    return None

def slp_trust_region(h, n, dx, delta):
    """One SLP iteration with trust region"""
    corrs = compute_correlations(h)
    n_shifts = len(corrs)
    
    # Build the linearization
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    
    for k in range(n_shifts):
        r = k - (n - 1)
        grad_k = np.zeros(n)
        mr_plus = np.arange(n) + r
        valid_plus = (mr_plus >= 0) & (mr_plus < n)
        grad_k[valid_plus] += (1.0 - h[mr_plus[valid_plus]])
        mr_minus = np.arange(n) - r
        valid_minus = (mr_minus >= 0) & (mr_minus < n)
        grad_k[valid_minus] -= h[mr_minus[valid_minus]]
        
        A[k] = grad_k
        b[k] = corrs[k] - np.dot(grad_k, h)
    
    # Solve: minimize t
    # subject to: A @ h_new + b <= t (for all k)
    #             sum(h_new) = n/2
    #             ||h_new - h||_inf <= delta
    #             0 <= h_new <= 1
    
    # The ||h_new - h||_inf <= delta gives: h[m] - delta <= h_new[m] <= h[m] + delta
    # Combined with 0 <= h_new[m] <= 1:
    # max(0, h[m]-delta) <= h_new[m] <= min(1, h[m]+delta)
    
    lo = np.maximum(0.0, h - delta)
    hi = np.minimum(1.0, h + delta)
    
    # LP: min t s.t. A@x - t <= -b, sum(x) = n/2, lo <= x <= hi
    from scipy.optimize import linprog
    
    n_vars = n + 1  # x[0..n-1], t
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    A_ub = np.zeros((n_shifts, n_vars))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -b
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = list(zip(lo, hi)) + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                 bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def slp_one_step(h, n, delta):
    corrs = compute_correlations(h)
    n_shifts = len(corrs)
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    for k in range(n_shifts):
        r = k - (n - 1)
        grad_k = np.zeros(n)
        mr_plus = np.arange(n) + r
        valid_plus = (mr_plus >= 0) & (mr_plus < n)
        grad_k[valid_plus] += (1.0 - h[mr_plus[valid_plus]])
        mr_minus = np.arange(n) - r
        valid_minus = (mr_minus >= 0) & (mr_minus < n)
        grad_k[valid_minus] -= h[mr_minus[valid_minus]]
        A[k] = grad_k
        b[k] = corrs[k] - np.dot(grad_k, h)
    lo = np.maximum(0.0, h - delta)
    hi = np.minimum(1.0, h + delta)
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    A_ub = np.zeros((n_shifts, n_vars))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -b
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = list(zip(lo, hi)) + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_to_feasible(h_new, n)
        return h_new
    return None


def slp_optimize(h0, n, delta_start, delta_end, n_iter, t0, budget):
    h = h0.copy()
    best_h = h0.copy()
    best_score = evaluate_score(h)
    deltas = np.geomspace(delta_start, delta_end, n_iter)
    for i in range(n_iter):
        if (time.time() - t0) > budget - 60:
            break
        d = deltas[i]
        h_new = slp_one_step(h, n, d)
        if h_new is not None:
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new
        else:
            h = slp_one_step(h, n, max(0.001, d * 0.5))
            if h is not None:
                s = evaluate_score(h)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h.copy()
                h = h_new if h_new is not None else h
    return best_h, best_score


def lbfgsb_smooth_max(h0, alpha, maxiter, bounds):
    def fun(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, alpha)[1]
    res = minimize(fun, h0, method='L-BFGS-B', bounds=bounds, jac=jac,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                            'iprint': 0, 'maxls': 50})
    return res.x


def slsqp_smooth_max(h0, alpha, maxiter, bounds, n):
    def fun(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, alpha)[1]
    def const_fun(x):
        return np.sum(x) - n / 2.0
    def const_jac(x):
        return np.ones(n)
    res = minimize(fun, h0, method='SLSQP', bounds=bounds, jac=jac,
                   constraints=[{'type': 'eq', 'fun': const_fun, 'jac': const_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    return res.x


def subgradient_momentum(h, n, n_iter, eta0, decay, beta, t0, budget):
    v = np.zeros(n)
    best_h = h.copy()
    best_score = evaluate_score(h)
    for it in range(n_iter):
        if (time.time() - t0) > budget - 30:
            break
        corrs = compute_correlations(h)
        k_star = np.argmax(corrs)
        r = k_star - (n - 1)
        grad = np.zeros(n)
        mr = np.arange(n) + r
        valid_plus = (mr >= 0) & (mr < n)
        grad[valid_plus] += (1.0 - h[mr[valid_plus]])
        mr2 = np.arange(n) - r
        valid_minus = (mr2 >= 0) & (mr2 < n)
        grad[valid_minus] -= h[mr2[valid_minus]]
        eta = eta0 / (1.0 + decay * it)
        v = beta * v - eta * grad
        h = project_to_feasible(h + v, n)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    return best_h, best_score


def binary_local_search(h, n, max_flips, t0, budget):
    target_ones = n // 2
    h_bin = (h > 0.5).astype(float)
    ones_idx = np.where(h_bin == 1.0)[0]
    zeros_idx = np.where(h_bin == 0.0)[0]
    if len(ones_idx) > target_ones:
        for i in range(len(ones_idx) - target_ones):
            h_bin[ones_idx[-1 - i]] = 0.0
    elif len(ones_idx) < target_ones:
        for i in range(target_ones - len(ones_idx)):
            h_bin[zeros_idx[i]] = 1.0
    best_bin = h_bin.copy()
    best_score = evaluate_score(h_bin)
    for _ in range(max_flips):
        if (time.time() - t0) > budget - 15:
            break
        corrs = compute_correlations(h_bin)
        k_star = np.argmax(corrs)
        r = k_star - (n - 1)
        ones = np.where(h_bin == 1.0)[0]
        best_gain = 0.0
        best_m = -1
        for m in ones:
            gain = 0.0
            if 0 <= m + r < n:
                gain += (1.0 - h_bin[m + r])
            if 0 <= m - r < n:
                gain += h_bin[m - r]
            if gain > best_gain:
                best_gain = gain
                best_m = m
        if best_m >= 0 and best_gain > 0:
            h_bin[best_m] = 0.0
            s = int(np.sum(h_bin))
            if s < target_ones:
                rem = np.where(h_bin == 0.0)[0]
                if len(rem) > 0:
                    h_bin[rem[0]] = 1.0
            s_new = evaluate_score(h_bin)
            if s_new < best_score:
                best_score = s_new
                best_bin = h_bin.copy()
            else:
                h_bin[best_m] = 1.0
                if s < target_ones:
                    rem = np.where(h_bin == 0.0)[0]
                    if len(rem) > 0:
                        h_bin[rem[0]] = 0.0
                break
        else:
            break
    return best_bin, best_score
