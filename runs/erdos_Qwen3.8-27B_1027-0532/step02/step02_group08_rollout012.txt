def compute_corr_d(h, d):

    """Momentum subgradient on active set."""
    n = len(h)

    beta = momentum

        m = np.max(corrs)

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr_d(h, d):
    n = len(h)
    if d == 0:
        return np.sum(h * (1.0 - h))
    return np.sum(h[:n - d] * (1.0 - h[d:]))


def grad_corr_d(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    return grad


def compute_corr_all(h):
    n = len(h)
    corrs = np.zeros(n)
    for d in range(n):
        corrs[d] = compute_corr_d(h, d)
    return corrs


def project_simplex_box(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_active_obj(x, d_values, alpha, S, mu):
    n = len(x)
    dx = 2.0 / n
    corrs = np.array([compute_corr_d(x, d) for d in d_values])
    m = np.max(corrs)
    shifted = np.clip(alpha * (corrs - m), -700, 700)
    exp_s = np.exp(shifted)
    sum_e = np.sum(exp_s)
    obj = (m + np.log(sum_e) / alpha) * dx
    viol = np.sum(x) - S
    obj += 0.5 * mu * viol * viol
    return obj


def smooth_active_grad(x, d_values, alpha, S, mu):
    n = len(x)
    dx = 2.0 / n
    corrs = np.array([compute_corr_d(x, d) for d in d_values])
    m = np.max(corrs)
    shifted = np.clip(alpha * (corrs - m), -700, 700)
    exp_s = np.exp(shifted)
    sum_e = np.sum(exp_s)
    weights = exp_s / sum_e

            grad += w * grad_corr_d(x, d)
    grad *= dx
    viol = np.sum(x) - S
    grad += mu * viol * np.ones(n)
    return grad


def solve_active(x0, d_values, alpha, S, mu, maxiter, bounds):
    res = minimize(smooth_active_obj, x0, method='L-BFGS-B',
                   jac=smooth_active_grad, args=(d_values, alpha, S, mu),
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                            'iprint': 0, 'maxls': 50, 'maxfun': maxiter * 10})
    return res.x


def full_smooth_obj(x, alpha, S, mu):
    n = len(x)
    dx = 2.0 / n
    gvec = 1.0 - x
    corrs = np.correlate(x, gvec, mode='full')
    m = np.max(corrs)
    shifted = np.clip(alpha * (corrs - m), -700, 700)
    exp_s = np.exp(shifted)
    sum_e = np.sum(exp_s)
    obj = (m + np.log(sum_e) / alpha) * dx
    viol = np.sum(x) - S
    obj += 0.5 * mu * viol * viol
    return obj


def full_smooth_grad(x, alpha, S, mu):
    n = len(x)
    dx = 2.0 / n
    gvec = 1.0 - x
    corrs = np.correlate(x, gvec, mode='full')
    m = np.max(corrs)
    shifted = np.clip(alpha * (corrs - m), -700, 700)
    exp_s = np.exp(shifted)
    sum_e = np.sum(exp_s)
    weights = exp_s / sum_e
    w_rev = weights[::-1]
    conv1 = np.convolve(gvec, w_rev, mode='full')
    conv2 = np.convolve(x, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    viol = np.sum(x) - S
    grad += mu * viol * np.ones(n)
    return grad * dx


def get_active_lags(h, k):
    n = len(h)
    corrs = compute_corr_all(h)
    return np.argsort(-corrs)[:k]


def momentum_subgrad(h, d_values, lr, beta, n_steps, S):
    n = len(h)
    v = np.zeros(n)
    for _ in range(n_steps):
        corrs = np.array([compute_corr_d(h, d) for d in d_values])
        m = np.max(corrs)
        tol = max(0.05 * m, 0.01)
        active = corrs > m - tol
        if not np.any(active):
            active = np.zeros(len(corrs), dtype=bool)
            active[np.argmax(corrs)] = True
        g = np.zeros(n)
        count = 0
        for i in range(len(d_values)):
            if active[i]:
                g += grad_corr_d(h, d_values[i])
                count += 1
        if count > 0:
            g /= count
        v = beta * v - lr * g
        h = project_simplex_box(h + v, S)
    return h


def try_solution(h_new, S, best_score, best_h):
    if h_new is None or not np.all(np.isfinite(h_new)):

    h_new = project_simplex_box(h_new, S)
    s = evaluate_score(h_new)
    if s < best_score:
        return s, h_new.copy(), True
    return best_score, best_h, False
