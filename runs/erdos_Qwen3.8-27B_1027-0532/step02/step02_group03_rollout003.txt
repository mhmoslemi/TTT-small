def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add

                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

    y = h0.copy()

                weights[np.argmax(corr)] = 1.0

            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)

            corr = np.correlate(y, 1.0 - y, mode='full')

        y = y_new
        t = t_new

            best_val = cv
            best = x.copy()
    return best, best_val

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(100.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_descent_multi(h0, n, target_sum, max_rounds=20, n_vals=11):
    """Coordinate descent trying many values per coordinate"""
    h = h0.copy()
    vals = np.linspace(0.0, 1.0, n_vals)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _perturb_and_refine(h0, n, target_sum, rng, n_perturb, deadline, alpha_sl=500):
    """Perturb some coordinates and re-optimize"""
    h = h0.copy()
    idx_perturb = rng.choice(n, size=n_perturb, replace=False)

                       options={'maxiter': 100, 'ftol': 1e-15})
        h = _fix_sum(res.x, target_sum)
    except Exception:
        pass
    return h

_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

            grad[:m] += -h[lag:lag + m]

        m = n - j

            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

            best = h_sl.copy()
        h_curr = h_sl
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_candidates = []
    for delta in range(-6, 7, 2):
        n_try = n_init + delta
        if n_try % 2 == 0 and 80 <= n_try <= 130:
            n_candidates.append(n_try)
    if n_init not in n_candidates:
        n_candidates.insert(0, n_init)

    phase1_deadline = t_start + 80
    for n in n_candidates:
        if time.time() > phase1_deadline:
            break
        target_sum = n / 2.0
        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            h_start = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), target_sum)
        c5 = _c5_of(h_start)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_start.copy()
            best_n = n
        if time.time() > phase1_deadline - 5:
            continue
        h_n1, c5_n1 = _nesterov(h_start, n, target_sum, n_iters=120, alpha0=0.3,
                                 sg_smooth=80.0, sg_m=5)
        if c5_n1 < best_c5:
            best_c5 = c5_n1
            best_h = h_n1.copy()
            best_n = n

    phase2_deadline = t_start + 400
    n_focus = best_n
    de_ns = [n_focus]
    for d in [2, -2, 4, -4]:
        n_try = n_focus + d
        if n_try % 2 == 0 and 80 <= n_try <= 130 and n_try not in de_ns:
            de_ns.append(n_try)

    for n in de_ns:
        if time.time() > phase2_deadline:
            break
        target_sum = n / 2.0
        if n == best_n:
            h_start = best_h.copy()
        else:
            h_start = np.interp(np.linspace(0, best_n - 1, n), np.arange(best_n), best_h)
            h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), target_sum)
        pop_size = min(35, max(20, n // 3))
        max_gens = 70
        h_de, c5_de = _de_optimize(h_start, n, target_sum, pop_size, max_gens, rng,
                                    min(phase2_deadline, time.time() + 250))
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n

    if best_h is not None:
        n_best = best_n
        target = n_best / 2.0
        h_curr = best_h.copy()

        for round_idx in range(4):
            if time.time() > deadline - 60:
                break

            alphas = [100, 300, 1000, 3000]
            if round_idx > 0:
                alphas = [500, 2000, 8000]
            h_cont, c5_cont = _slsqp_continuation(h_curr, n_best, target, alphas, 300)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
            h_curr = h_cont

            if time.time() > deadline - 40:
                continue
            h_n2, c5_n2 = _nesterov(h_curr, n_best, target, n_iters=200, alpha0=0.15,
                                     sg_smooth=120.0, sg_m=7)
            if c5_n2 < best_c5:
                best_c5 = c5_n2
                best_h = h_n2.copy()
            h_curr = h_n2

            if time.time() > deadline - 30:
                continue
            h_cd = _coord_descent_multi(h_curr, n_best, target, max_rounds=12, n_vals=7)
            c5_cd = _c5_of(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            h_curr = h_cd

        for n_pert in [4, 8, 12, 6]:
            if time.time() > deadline - 40:
                break
            h_pert = best_h.copy()
            idx_p = rng.choice(n_best, size=n_pert, replace=False)
            h_pert[idx_p] = rng.rand(n_pert)
            h_pert = _project(h_pert, target)
            alphas_short = [200, 1000, 4000]
            h_pert, c5_pert = _slsqp_continuation(h_pert, n_best, target, alphas_short, 200)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

        if time.time() <= deadline - 30:
            h_cd_f = _coord_descent_multi(best_h, n_best, target, max_rounds=20, n_vals=11)
            c5_cd_f = _c5_of(h_cd_f)
            if c5_cd_f < best_c5:
                best_c5 = c5_cd_f
                best_h = h_cd_f.copy()

        if time.time() <= deadline - 20:
            h_fin, c5_fin = _slsqp_continuation(best_h, n_best, target,
                                                 [2000, 8000, 20000], 400)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

        if time.time() <= deadline - 10:
            h_cd_fin = _coord_descent_multi(best_h, n_best, target, max_rounds=15, n_vals=13)
            c5_cd_fin = _c5_of(h_cd_fin)
            if c5_cd_fin < best_c5:
                best_c5 = c5_cd_fin
                best_h = h_cd_fin.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
