**Coordinate descent approach:**

            new_ones1 = int(np.sum(seg2)) - int(np.sum(seg1))
            if new_ones1 != 0:
                if new_ones1 > 0 and int(np.sum(h_bin)) + new_ones1 > target_ones:
                    continue
                if new_ones1 < 0 and int(np.sum(h_bin)) + new_ones1 < target_ones:
                    continue
            h_bin[start1:start1 + block_len] = seg2
            h_bin[start2:start2 + block_len] = seg1
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp * dx * 3.0, 1e-12)):
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_bin.copy()
            else:
                h_bin[start2:start2 + block_len] = seg1
                h_bin[start1:start1 + block_len] = seg1
        else:
            seg_len = int(rng.randint(3, 15))
            if n < seg_len + 2:
                continue
            start = int(rng.randint(0, n - seg_len))
            h_bin[start:start + seg_len] = h_bin[start:start + seg_len][::-1]
            s_new = evaluate_score(h_bin)
            delta = s_new - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp * dx * 2.0, 1e-12)):
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_bin.copy()
            else:
                h_bin[start:start + seg_len] = h_bin[start:start + seg_len][::-1]
    return best_h, best_score


def coordinate_descent_polish(h0, n, n_iters, t_start):
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    for it in range(n_iters):
        if time.time() - t_start > budget_global - 3:
            break
        overlaps = compute_overlaps(h)
        k_star = int(np.argmax(overlaps))
        L_shift = k_star - (n - 1)
        if L_shift == 0:
            direction = -(1.0 - 2.0 * h)
        elif L_shift > 0:
            direction = np.zeros(n)
            direction[L_shift:] = -(1.0 - h[:n - L_shift])
            direction[:n - L_shift] -= h[L_shift:]
        else:
            M = -L_shift
            direction = np.zeros(n)
            direction[:n - M] = -(1.0 - h[M:])
            direction[M:] -= h[:n - M]
        for step in [0.01, 0.005, 0.002, 0.001]:
            h_try = project_box_sum(h + step * direction, n / 2.0)
            s_try = evaluate_score(h_try)
            if s_try < best_score:
                best_score = s_try
                best_h = h_try.copy()
                h = h_try.copy()
                break
    return best_h, best_score


def multi_resolution_refine(h_best, n, t_start, rng):
    best_h = h_best.copy()
    best_score = evaluate_score(h_best)
    for n_coarse in [50, 100, 200]:
        if time.time() - t_start > budget_global - 60:
            break
        step = n // n_coarse
        h_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            h_coarse[i] = float(np.mean(h_best[i * step:(i + 1) * step]))
        h_coarse = project_box_sum(h_coarse, n_coarse / 2.0)
        for alpha in [50, 200, 1000, 3000]:
            if time.time() - t_start > budget_global - 30:
                break
            h_coarse = fista_single(h_coarse, alpha, 1500, n_coarse, t_start)
            h_coarse = project_box_sum(h_coarse, n_coarse / 2.0)
        x_old = np.arange(n_coarse) / float(n_coarse)
        x_new = np.arange(n) / float(n)
        h_upper = np.interp(x_new, x_old, h_coarse)
        h_upper = project_box_sum(h_upper, n / 2.0)
        h_upper = fista_single(h_upper, 1000, 1000, n, t_start)
        h_upper = project_box_sum(h_upper, n / 2.0)
        s = evaluate_score(h_upper)
        if s < best_score:
            best_score = s
            best_h = h_upper.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global budget_global
    budget_global = budget_s
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_score = evaluate_score(h)

    def time_left():
        return budget_s - 15.0 - (time.time() - t0)

    # Phase 1: Continuation proximal gradient with LSE
    alphas_p1 = [30, 80, 200, 500]
    iters_p1 = [4000, 3000, 2500, 2000]
    h_curr = h.copy()
    for i, alpha in enumerate(alphas_p1):
        if time_left() < 120:
            break
        h_curr = fista_single(h_curr, alpha, iters_p1[i], n, t0)
        h_curr = project_box_sum(h_curr, n / 2.0)
        s = evaluate_score(h_curr)
        if s < best_score:
            best_score = s
            best_h = h_curr.copy()

    # Phase 2: Higher alpha continuation
    alphas_p2 = [1000, 2000, 5000]
    iters_p2 = [2000, 1500, 1000]
    for i, alpha in enumerate(alphas_p2):
        if time_left() < 90:
            break
        h_curr = fista_single(h_curr, alpha, iters_p2[i], n, t0)
        h_curr = project_box_sum(h_curr, n / 2.0)
        s = evaluate_score(h_curr)
        if s < best_score:
            best_score = s
            best_h = h_curr.copy()

    # Phase 3: Subgradient top-k optimization
    if time_left() > 80:
        n_sub = min(8000, int(time_left() * 10))
        h_sub, s_sub = subgrad_topk_optimize(best_h, n, n_sub, 3, 0.5, t0)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()
        # Polish subgradient result
        if time_left() > 60:
            h_pol = fista_single(h_sub, 3000, 1000, n, t0)
            h_pol = project_box_sum(h_pol, n / 2.0)
            s_pol = evaluate_score(h_pol)
            if s_pol < best_score:
                best_score = s_pol
                best_h = h_pol.copy()

    # Phase 4: Symmetry h -> 1-h
    if time_left() > 80:
        h_sym = 1.0 - best_h
        h_sym = project_box_sum(h_sym, n / 2.0)
        h_sym = fista_single(h_sym, 500, 3000, n, t0)
        h_sym = fista_single(h_sym, 2000, 2000, n, t0)
        h_sym = project_box_sum(h_sym, n / 2.0)
        s_sym = evaluate_score(h_sym)
        if s_sym < best_score:
            best_score = s_sym
            best_h = h_sym.copy()

    # Phase 5: Binary local search with block moves
    if time_left() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = adjust_binary(h_bin, n // 2)
        n_bin_iters = min(20000, int(time_left() * 12))
        h_bin_best, s_bin = binary_block_search(h_bin, n // 2, rng, n_bin_iters, t0)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin_best.copy()
        # Try perturbed binary
        if time_left() > 40:
            h_pert = best_h + rng.randn(n) * 0.1
            h_pert = project_box_sum(h_pert, n / 2.0)
            h_bin2 = (h_pert > 0.5).astype(float)
            h_bin2 = adjust_binary(h_bin2, n // 2)
            h_bin2_best, s_bin2 = binary_block_search(h_bin2, n // 2, rng,
                                                      min(10000, int(time_left() * 8)), t0)
            if s_bin2 < best_score:
                best_score = s_bin2
                best_h = h_bin2_best.copy()

    # Phase 6: Multiple perturbations with quick FISTA
    if time_left() > 90:
        for trial in range(4):
            if time_left() < 30:
                break
            noise_scale = 0.05 + 0.03 * trial
            h_try = best_h + rng.randn(n) * noise_scale
            h_try = project_box_sum(h_try, n / 2.0)
            h_try = fista_single(h_try, 500, 1500, n, t0)
            h_try = fista_single(h_try, 2000, 1000, n, t0)
            h_try = project_box_sum(h_try, n / 2.0)
            s_try = evaluate_score(h_try)
            if s_try < best_score:
                best_score = s_try
                best_h = h_try.copy()

    # Phase 7: Multi-resolution refinement
    if time_left() > 100:
        h_mr, s_mr = multi_resolution_refine(best_h, n, t0, rng)
        if s_mr < best_score:
            best_score = s_mr
            best_h = h_mr.copy()

    # Phase 8: Coordinate descent polish
    if time_left() > 40:
        h_cd, s_cd = coordinate_descent_polish(best_h, n,
                                               min(3000, int(time_left() * 5)), t0)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd.copy()

    # Phase 9: Final high-precision FISTA
    if time_left() > 30:
        for alpha in [5000, 10000, 20000]:
            if time_left() < 15:
                break
            h_final = fista_single(best_h, alpha, 1500, n, t0)
            h_final = project_box_sum(h_final, n / 2.0)
            s_final = evaluate_score(h_final)
            if s_final < best_score:
                best_score = s_final
                best_h = h_final.copy()

    # Final projection and evaluation
    best_h = project_box_sum(best_h, n / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, n)
