import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _full_gradient(h, alpha):
    """Compute gradient of smooth max of correlation w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    dx = 2.0 / n
    
    # Compute gradient
    g = np.zeros(n)
    # Only process lags with significant weight
    significant = w > 1e-12
    for idx in np.where(significant)[0]:
        wi = w[idx]
        lag = idx - (n - 1)
        if lag > 0:
            g[:n-lag] -= wi * h[lag:]
            g[lag:] += wi * (1.0 - h[:n-lag])
        elif lag < 0:
            L = -lag
            g[:n-L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:n-L]
        else:
            g[:] += wi * (1.0 - 2.0 * h)
    
    return g * dx, corr, np.max(corr)

def _gradient_descent_phase(h, n, alpha, target_sum, n_steps, lr, momentum, deadline, check_every=200):
    """Run gradient descent with momentum for a given alpha."""
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = _c5_of(h_cur)
    
    for t in range(n_steps):
        if time.time() > deadline - 10:
            break
        
        g, corr, max_c = _full_gradient(h_cur, alpha)
        
        # Project gradient onto constraint (sum = target_sum)
        # The constraint is sum(h) = target_sum, so we remove the mean component
        g_mean = np.mean(g)
        g_proj = g - g_mean  # This projects onto the sum=0 subspace
        
        v = momentum * v - lr * g_proj
        h_new = h_cur + v
        h_cur = _fix_sum(h_new, target_sum)
        
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    
    c5 = _c5_of(h_cur)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_cur.copy()
    
    return best_h, best_c5

def _slsqp_refine(h_init, n, alpha, maxiter, ftol=1e-14):
    """Refine using SLSQP with smooth max objective."""
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500, 500)
        smooth = (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
        return smooth
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500, 500)
        w = np.exp(shifted)
        w /= np.sum(w)
        
        g = np.zeros(n)
        significant = w > 1e-12
        for idx in np.where(significant)[0]:
            wi = w[idx]
            lag = idx - (n - 1)
            if lag > 0:
                g[:n-lag] -= wi * x[lag:]
                g[lag:] += wi * (1.0 - x[:n-lag])
            elif lag < 0:
                L = -lag
                g[:n-L] += wi * (1.0 - x[L:])
                g[L:] -= wi * x[:n-L]
            else:
                g[:] += wi * (1.0 - 2.0 * x)
        
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def _coordinate_descent(h_init, n, target_sum, deadline, n_rounds=5):
    """Simple coordinate descent: try small perturbations to individual coordinates."""
    h_cur = h_init.copy()
    best_c5 = _c5_of(h_cur)
    target_sum = n / 2.0
    rng = np.random.RandomState(12345)
    
    for round_idx in range(n_rounds):
        if time.time() > deadline - 15:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 15:
                break
            for delta in [-0.05, -0.02, 0.02, 0.05]:
                h_try = h_cur.copy()
                h_try[i] += delta
                # Compensate sum
                # Find where to add/subtract the rest
                # Simple: spread the compensation over all other coordinates
                comp = -delta / (n - 1)
                h_try[(np.arange(n) != i)] += comp
                h_try = _fix_sum(h_try, target_sum)
                c5_try = _c5_of(h_try)
                if c5_try < best_c5 - 1e-12:
                    h_cur = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    
    return h_cur, best_c5

def _make_initializations(n, seed, target_sum):
    """Create diverse initializations."""
    inits = []
    rng = np.random.RandomState(seed)
    
    # Constant
    inits.append(np.full(n, 0.5))
    
    # Sine wave
    x = np.linspace(0, 2, n, endpoint=False)
    inits.append(0.5 + 0.25 * np.sin(np.pi * x))
    inits.append(0.5 + 0.25 * np.sin(2 * np.pi * x))
    inits.append(0.5 + 0.2 * np.sin(3 * np.pi * x))
    
    # Block function
    h_block = np.zeros(n)
    period = max(2, n // 4)
    for i in range(n):
        h_block[i] = 1.0 if (i % (2 * period)) < period else 0.0
    inits.append(h_block + 0.1 * rng.randn(n))
    
    # Random
    inits.append(rng.rand(n))
    
    # Multi-scale
    inits.append(0.5 + 0.2*np.sin(np.pi*x) + 0.1*np.sin(2*np.pi*x) + 0.05*np.sin(4*np.pi*x))
    
    # Fix all
    fixed = []
    for h in inits:
        fixed.append(_fix_sum(h, target_sum))
    return fixed

**Alternative: Minimax via bisection**

def _c5_of(h):

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700, 700)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = exps / s
    g = np.zeros(n)
    sig = np.where(w > 1e-10)[0]
    for idx in sig:
        wi = w[idx]
        lag = idx - (n - 1)
        if lag > 0:
            g[:n - lag] -= wi * h[lag:]
            g[lag:] += wi * (1.0 - h[:n - lag])
        elif lag < 0:
            L = -lag
            g[:n - L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:n - L]
        else:
            g[:] += wi * (1.0 - 2.0 * h)
    return g * dx

def _adam_phase(h_init, n, alpha, target_sum, n_steps, lr, beta1, beta2, deadline):
    h_cur = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = _c5_of(h_cur)
    stale = 0
    max_stale = max(200, n_steps // 4)
    for t in range(n_steps):
        if time.time() > deadline - 10:
            break
        g = _grad_smooth(h_cur, alpha)
        g -= np.mean(g)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        h_new = h_cur - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_cur = _fix_sum(h_new, target_sum)
        if (t + 1) % 150 == 0:
            c5 = _c5_of(h_cur)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h_cur.copy()
                stale = 0
            else:
                stale += 150
            if stale > max_stale:
                break
    c5 = _c5_of(h_cur)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_cur.copy()
    return best_h, best_c5

def _momentum_phase(h_init, n, alpha, target_sum, n_steps, lr, momentum, deadline):
    h_cur = h_init.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = _c5_of(h_cur)
    stale = 0
    max_stale = max(200, n_steps // 3)

        if time.time() > deadline - 10:

        g = _grad_smooth(h_cur, alpha)
        g -= np.mean(g)
        v = momentum * v - lr * g
        h_cur = _fix_sum(h_cur + v, target_sum)

    c5 = _c5_of(h_cur)

    return best_h, best_c5

def _slsqp_refine(h_init, n, alpha, maxiter):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700, 700)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    def grad(x):
        n_ = len(x)
        dx_ = 2.0 / n_
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700, 700)
        exps = np.exp(shifted)
        s = np.sum(exps)
        if s < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w = exps / s
        g = np.zeros(n_)
        sig = np.where(w > 1e-10)[0]
        for idx in sig:
            wi = w[idx]
            lag = idx - (n_ - 1)
            if lag > 0:
                g[:n_ - lag] -= wi * x[lag:]
                g[lag:] += wi * (1.0 - x[:n_ - lag])
            elif lag < 0:
                L = -lag
                g[:n_ - L] += wi * (1.0 - x[L:])
                g[L:] -= wi * x[:n_ - L]
            else:
                g[:] += wi * (1.0 - 2.0 * x)
        return g * dx_

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h_init.copy()
