    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)

        if w < 1e-10:

        lag = N - 1 - index
        if lag == 0:

            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]

def binary_local_search(h, n, max_iters, rng, top_k=20):
    best_h = h.copy()
    corr = compute_correlations(best_h)
    best_max = np.max(corr)
    
    for iteration in range(max_iters):
        # Identify hot lags
        hot_lags = np.argsort(corr)[-top_k:]
        hot_set = set(hot_lags.tolist())
        
        # Try random bit flips
        i = rng.randint(n)
        old_val = best_h[i]
        new_val = 1.0 - old_val
        if new_val < 0 or new_val > 1:
            continue
        
        best_h[i] = new_val
        new_corr = compute_correlations(best_h)
        new_max = np.max(new_corr)
        
        # Score: weighted by how much hot lags changed
        hot_change = np.sum(new_corr[hot_lags] - corr[hot_lags])
        
        if new_max < best_max - 1e-12 or (new_max <= best_max + 1e-10 and hot_change < 0):
            best_max = new_max
            corr = new_corr
        else:
            best_h[i] = old_val
    
    return best_h, best_max

    """Compute the change in all correlations when h[i] changes by flip."""
    n = len(h)
    old = h[i]
    new = old + flip
    if new < -1e-10 or new > 1.0 + 1e-10:
        return None
    
    # corr has 2n-1 entries, indexed 0 to 2n-2
    # corr[j] corresponds to lag = j - (n-1)
    delta = np.zeros(2*n-1)
