import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad_calc(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

def slsqp_local(h_start, n, bounds, eps, maxiter, ftol):
    def obj(x):
        return smooth_max_val(x, eps, n)
    def grad(x):
        return smooth_max_grad_calc(x, eps, n)
    def constr(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h_start.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=[{'type': 'eq', 'fun': constr}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def mss_local_search(h_start, n, max_sweeps, time_check, rng):
    """Multiple swap search local search.
    At each sweep, try all pairs (i, j) where h[i]=1, h[j]=0,
    and take the best improving swap."""
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    
    for sweep in range(max_sweeps):
        if time_check():
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_delta = 0.0
        best_pair = None
        cur_c5 = evaluate_c5(h)
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = evaluate_c5(h)
                delta = new_c5 - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            best_c5 = cur_c5 + best_delta
        else:
            break
    
    return h, best_c5

def sa_binary_swaps(h_start, n, max_iters, t0, t1, time_check, rng):
    """Simulated annealing on binary vectors using bit-pair swaps."""
    h = h_start.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if time_check():
            break
        T = t0 * (t1 / t0) ** ((it + 1) / max_iters)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        new_c5 = evaluate_c5(h)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if new_c5 <= best_c5:
                best_c5 = new_c5
                best = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
    
    return best, best_c5

def ilsa(h_start, n, max_rounds, n_sweeps, n_perturb, time_check, rng):
    """Iterated local search with perturbations."""
    h = h_start.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    
    for rnd in range(max_rounds):
        if time_check():
            break
        # Local search
        h, c5 = mss_local_search(h, n, n_sweeps, time_check, rng)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        
        # Perturb
        for _ in range(n_perturb):
            ones = np.where(best >= 0.5)[0]
            zeros = np.where(best < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            best[i] = 1.0 - best[i]
            best[j] = 1.0 - best[j]
    
    return best, best_c5

- From case 1: 0 * (1-h[a-l]) = 0
- From case 2: h[a+l] * (1-0) = h[a+l]

So the change in f(l) from changing h[a]: 1→0 is:
Δf(l) = [0 + h[a+l]*[0≤a+l<n]] - [1*(1-h[a-l])*[0≤a-l<n] + 0]
= h[a+l]*[0≤a+l<n] - (1-h[a-l])*[0≤a-l<n]

Where [P] is 1 if P is true, 0 otherwise.

Similarly, changing h[b]: 0→1:
Old contribution:
- Case 1: 0 * (1-h[b-l]) = 0
- Case 2: h[b+l] * (1-0) = h[b+l] (if valid)

New contribution:
- Case 1: 1 * (1-h[b-l]) = (1-h[b-l]) (if valid)
- Case 2: h[b+l] * (1-1) = 0

Change: (1-h[b-l])*[0≤b-l<n] - h[b+l]*[0≤b+l<n]

So total change in f(l) from swapping a(1→0) and b(0→1):
Δf(l) = h[a+l]*[0≤a+l<n] - (1-h[a-l])*[0≤a-l<n] + (1-h[b-l])*[0≤b-l<n] - h[b+l]*[0≤b+l<n]

This is O(1) per shift l! So I can compute the change in all 2n-1 shifts in O(n) time per swap evaluation.

But wait, I need the max over all shifts, not just a specific shift. So I need to compute Δf(l) for all l and find the new max. That's O(n) per swap.

    lo = -2.0
    hi = 2.0

def smooth_max_val(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad_calc(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)

        l = k - (n - 1)
        w = weights[k]

            jml = j - l

                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

def slsqp_local(h_start, n, bounds, eps, maxiter, ftol):
    def obj(x):
        return smooth_max_val(x, eps, n)
    def grad(x):
        return smooth_max_grad_calc(x, eps, n)
    def constr(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h_start.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=[{'type': 'eq', 'fun': constr}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def sa_binary_swaps(h_start, n, max_iters, t0, t1, time_check, rng):
    h = h_start.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    for it in range(max_iters):
        if time_check():
            break
        T = t0 * max(t1 / t0, 1e-12) ** ((it + 1) / max_iters)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        use_2swap = (len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.25)
        if use_2swap:
            i1 = ones[rng.integers(len(ones))]
            i2 = i1
            for _ in range(10):
                i2 = ones[rng.integers(len(ones))]
                if i2 != i1:
                    break
            j1 = zeros[rng.integers(len(zeros))]
            j2 = j1
            for _ in range(10):
                j2 = zeros[rng.integers(len(zeros))]
                if j2 != j1 and j2 != i1 and j2 != i2:
                    break
            if j2 == j1 or j2 == i1 or j2 == i2 or j1 == i1 or j1 == i2:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = evaluate_c5(h)
                delta = new_c5 - cur_c5
                if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                    cur_c5 = new_c5
                    if new_c5 < best_c5 - 1e-15:
                        best_c5 = new_c5
                        best = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = evaluate_c5(h)
            delta = new_c5 - cur_c5
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best = h.copy()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        else:
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = evaluate_c5(h)
            delta = new_c5 - cur_c5
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
    return best, best_c5

def mss_local_search(h_start, n, max_sweeps, time_check):
    h = h_start.copy()
    cur_c5 = evaluate_c5(h)
    for sweep in range(max_sweeps):
        if time_check():
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_delta = 0.0
        best_pair = None
        n_ones = len(ones)
        n_zeros = len(zeros)
        for ii in range(n_ones):
            i = ones[ii]
            for jj in range(n_zeros):
                j = zeros[jj]
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = evaluate_c5(h)
                delta = new_c5 - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is not None and best_delta < -1e-15:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            cur_c5 += best_delta
        else:
            break
    return h, cur_c5

def binary_from_continuous(h_cont, n, rng):
    h = (h_cont >= 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1.0)[0].copy()
        rng.shuffle(ones_idx)
        for idx in ones_idx:
            if n_ones == target:
                break
            h[idx] = 0.0
            n_ones -= 1
    elif n_ones < target:
        zeros_idx = np.where(h == 0.0)[0].copy()
        rng.shuffle(zeros_idx)
        for idx in zeros_idx:
            if n_ones == target:
                break
            h[idx] = 1.0
            n_ones += 1
    return h

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def time_limit(frac):
        return lambda: elapsed() > budget_s * frac

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)

    h_bin = binary_from_continuous(h0, n, rng)
    best_h = h_bin.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: SA on binary n (0-25% time)
    sa_configs = [
        (500000, 0.05, 0.0005),
        (400000, 0.03, 0.0003),
        (300000, 0.015, 0.0002),
    ]
    for max_it, t0, t1 in sa_configs:
        if elapsed() > budget_s * 0.25:
            break
        h_sa, c5_sa = sa_binary_swaps(best_h, n, max_it, t0, t1, time_limit(0.25), rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 2: MSS polish (25-40% time)
    if elapsed() < budget_s * 0.40:
        h_mss, c5_mss = mss_local_search(best_h, n, 500, time_limit(0.40))
        if c5_mss < best_c5:
            best_c5 = c5_mss
            best_h = h_mss

    # Phase 3: ILS - perturb + SA + MSS (40-65% time)
    for restart in range(10):
        if elapsed() > budget_s * 0.65:
            break
        n_pert = 4 + restart * 2
        h_pert = best_h.copy()
        ones = np.where(h_pert >= 0.5)[0]
        zeros = np.where(h_pert < 0.5)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert >= 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
        t0 = 0.03 - 0.002 * restart
        t1 = 0.0003 - 0.00002 * restart
        h_sa, c5_sa = sa_binary_swaps(h_pert, n, 200000, max(t0, 0.005), max(t1, 0.0001), time_limit(0.63), rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        if elapsed() < budget_s * 0.65:
            h_mss, c5_mss = mss_local_search(best_h, n, 300, time_limit(0.65))
            if c5_mss < best_c5:
                best_c5 = c5_mss
                best_h = h_mss

    # Phase 4: SLSQP continuous refinement (65-78% time)
    bounds_n = [(0.0, 1.0)] * n
    for eps_val in [200.0, 1000.0, 5000.0, 20000.0]:
        if elapsed() > budget_s * 0.78:
            break
        h_new, c5_new = slsqp_local(best_h, n, bounds_n, eps_val, 400, 1e-15)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_mss, c5_mss = mss_local_search(binary_from_continuous(best_h, n, rng), n, 200, time_limit(0.78))
        if c5_mss < best_c5:
            best_c5 = c5_mss
            best_h = h_mss

    # Phase 5: Try n=200 (78-95% time)
    n2 = 200
    h_up = upsample(best_h, n, n2)
    h_bin2 = binary_from_continuous(h_up, n2, rng)
    best_h2 = h_bin2.copy()
    best_c52 = evaluate_c5(best_h2)

    if elapsed() < budget_s * 0.85:
        h_sa2, c5_sa2 = sa_binary_swaps(best_h2, n2, 300000, 0.04, 0.0003, time_limit(0.85), rng)
        if c5_sa2 < best_c52:
            best_c52 = c5_sa2
            best_h2 = h_sa2

    if elapsed() < budget_s * 0.90:
        h_mss2, c5_mss2 = mss_local_search(best_h2, n2, 300, time_limit(0.90))
        if c5_mss2 < best_c52:
            best_c52 = c5_mss2
            best_h2 = h_mss2

    bounds_n2 = [(0.0, 1.0)] * n2
    for eps_val in [1000.0, 5000.0, 20000.0]:
        if elapsed() > budget_s * 0.95:
            break
        h_new2, c5_new2 = slsqp_local(best_h2, n2, bounds_n2, eps_val, 300, 1e-15)
        if c5_new2 < best_c52:
            best_c52 = c5_new2
            best_h2 = h_new2

    # Phase 6: Try n=256 if time permits (95-99% time)
    n3 = 256
    if elapsed() < budget_s * 0.96:
        h_up3 = upsample(best_h, n, n3)
        h_bin3 = binary_from_continuous(h_up3, n3, rng)
        best_h3 = h_bin3.copy()
        best_c53 = evaluate_c5(best_h3)
        h_sa3, c5_sa3 = sa_binary_swaps(best_h3, n3, 100000, 0.03, 0.0002, time_limit(0.97), rng)
        if c5_sa3 < best_c53:
            best_c53 = c5_sa3
            best_h3 = h_sa3
        bounds_n3 = [(0.0, 1.0)] * n3
        if elapsed() < budget_s * 0.98:
            h_new3, c5_new3 = slsqp_local(best_h3, n3, bounds_n3, 5000.0, 200, 1e-15)
            if c5_new3 < best_c53:
                best_c53 = c5_new3
                best_h3 = h_new3
        if best_c53 < best_c52:
            best_c52 = best_c53
            best_h2 = best_h3

    # Final comparison
    if best_c52 < best_c5:
        final_h = best_h2
        final_c5 = best_c52
        final_n = len(best_h2)
    else:
        final_h = best_h
        final_c5 = best_c5
        final_n = n

    final_h = project_feasible(final_h, final_n)
    final_c5 = evaluate_c5(final_h)
    return final_h, final_c5, final_n
