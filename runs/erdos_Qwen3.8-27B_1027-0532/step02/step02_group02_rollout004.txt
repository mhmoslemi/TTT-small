    diff = target_sum - h.sum()

    corr = _correlations(h)
    m = np.max(corr)
    # shifted for numerical stability
    shifted = corr - m
    lse = m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return lse * (2.0 / len(h))

    n = len(h)
    corr = _correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    weights = np.exp(shifted)

    # Gradient of log-sum-exp: sum_k weights_k * grad_k

        lag = k - (n - 1)
        grad += weights[k] * _grad_at_lag(h, lag, n)
    return grad * (2.0 / n)

        best_h = _fix_sum(np.full(n, 0.5), n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
