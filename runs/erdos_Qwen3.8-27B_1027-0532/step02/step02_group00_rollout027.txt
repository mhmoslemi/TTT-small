    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    """Project to binary (0-1) solution with exact weight target_sum."""
    n = len(h)
    h = np.round(h)
    h = np.clip(h, 0, 1).astype(int)
    current_sum = h.sum()
    if current_sum == target_sum:
        return h.astype(float)
    
    # Adjust: flip lowest-scoring 1s to 0 or highest-scoring 0s to 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if current_sum > target_sum:
        need = current_sum - target_sum
        # Flip the 'worst' 1s to 0
        # Score by original h value (lower = more likely to flip)
        order = np.argsort(h[ones])[:need]
        h[ones[order]] = 0
    else:
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h[zeros[order]] = 1
    
    return h.astype(float)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

        max_it = int(max(2000, tleft() * 1500))
        h_sa_f, c5_f = sa_optimize(h_bin_final, target_use, max_it, 0.005, 0.00005,
                                   rng, time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 8:
            h_ref, sc = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                     [1000, 3000], 200, 1e-15,
                                     time_limit=max(3.0, tleft() - 5), start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
