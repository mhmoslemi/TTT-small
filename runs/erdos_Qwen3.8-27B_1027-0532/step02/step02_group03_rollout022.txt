def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_eq(y, S):
    n = len(y)
    if S < 0 or S > n:
        return np.clip(y, 0, 1)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break

                x[i] += add
                diff -= add
        else:

                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _smoothed_obj(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < -500:
        return m
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _smoothed_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < -500:
        return np.zeros(n)
    shifted = alpha * (corr - m)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            mgt = n - lag
            if mgt > 0:
                grad[lag:lag+mgt] += (1.0 - h[:mgt]) * weights[k]
                grad[:mgt] += -h[lag:lag+mgt] * weights[k]
        else:
            j = -lag
            mgt = n - j
            if mgt > 0:
                grad[:mgt] += (1.0 - h[j:]) * weights[k]
                grad[j:] += -h[:mgt] * weights[k]
    return grad * dx

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha0, alpha_ls, m_top=3):
    """Nesterov accelerated gradient with smoothed subgradient (top-m weighted)."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        max_corr = np.max(corr)
        
        # Compute weights for top-m lags
        if m_top >= len(corr):
            m_top = len(corr)
        
        # Use softmax over all, but focus on top
        shifted = alpha_ls * (corr - max_corr)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        
        # Optionally zero out very small weights
        threshold = 1e-8
        weights[weights < threshold] = 0
        if weights.sum() > 0:
            weights = weights / weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < threshold:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                mgt = n - lag
                if mgt > 0:
                    grad[lag:lag+mgt] += (1.0 - y[:mgt]) * weights[k]
                    grad[:mgt] += -y[lag:lag+mgt] * weights[k]
            else:
                j2 = -lag
                mgt = n - j2
                if mgt > 0:
                    grad[:mgt] += (1.0 - y[j2:]) * weights[k]
                    grad[j2:] += -y[:mgt] * weights[k]
        grad *= dx
        
        # Nesterov step
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_eq(y - alpha * grad, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_eq(y_new, target_sum)
        
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _proj_grad_ls(h0, n, target_sum, n_iters, alpha0, alpha_ls):
    """Projected gradient with backtracking line search on smoothed objective."""
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    best_smooth = _smoothed_obj(x, alpha_ls, n)
    
    for it in range(n_iters):
        g = _smoothed_grad(x, alpha_ls, n)
        
        # Backtracking line search
        alpha = alpha0 / (1.0 + 0.1 * it)
        f_curr = _smoothed_obj(x, alpha_ls, n)
        c1 = 1e-4  # Armijo condition
        
        for _ in range(30):
            x_trial = _project_box_eq(x - alpha * g, target_sum)
            f_trial = _smoothed_obj(x_trial, alpha_ls, n)
            if f_trial <= f_curr + c1 * alpha * np.dot(g, x_trial - x):
                break
            alpha *= 0.5
        
        x = x_trial
        f_new = _smoothed_obj(x, alpha_ls, n)
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
            best_smooth = f_new
    
    return best, best_val

For binary h, the correlation at lag k is:
$$c_k = \sum_i h[i] \cdot (1-h[i+k]) = \sum_i h[i] - \sum_i h[i]h[i+k]$$

For lag 0: c_0 = sum(h) - sum(h²) = n/2 - n/2 = 0... wait that can't be right.

So for a binary h, the lag-0 correlation is always 0! That means the max is achieved at some other lag.

This is a key insight. For binary h, we need to minimize the maximum of c_k for k ≠ 0.

c_k for lag k (k > 0): sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
= sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i]*h[i+k]

The first term is the number of 1s in positions 0 to n-k-1.
The second term is the number of positions where both h[i] and h[i+k] are 1.

So c_k = (number of 1s in first n-k positions) - (number of 1s at distance k apart, both being 1).

This is like a "bounded aperiodic" problem. We want the binary sequence to have as few "close pairs" as possible at every distance.

This is related to the concept of "Costas arrays" and "Golomb rulers" but for a circular/linear binary sequence.

For the binary case, the problem becomes: find a binary sequence of length n with exactly n/2 ones, such that for every lag k, the number of "1 followed by 0 at distance k" pairs is minimized (maximized over k).

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_eq(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _smoothed_obj_for_slsqp(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < -500:
        return m
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _smoothed_grad_slsqp(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < -500:
        return np.zeros(n)
    shifted = alpha * (corr - m)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            mgt = n - lag
            if mgt > 0:
                grad[lag:lag+mgt] += (1.0 - h[:mgt]) * w
                grad[:mgt] += -h[lag:lag+mgt] * w
        else:
            j2 = -lag
            mgt = n - j2
            if mgt > 0:
                grad[:mgt] += (1.0 - h[j2:]) * w
                grad[j2:] += -h[:mgt] * w
    return grad * dx

def _slsqp_run(h0, n, target_sum, alpha, maxiter=500):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smoothed_obj_for_slsqp(x, alpha, n)
    
    def grad(x):
        return _smoothed_grad_slsqp(x, alpha, n)
    
    def eq_fun(x):
        return x.sum() - target_sum
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov(h0, n, target_sum, n_iters, alpha0):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        max_corr = np.max(corr)
        
        # Softmax weights
        shifted = 30.0 * (corr - max_corr)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        weights[weights < 1e-6] = 0
        if weights.sum() > 0:
            weights /= weights.sum()
        
        grad = np.zeros(n)
        n_corr = len(corr)
        for k in range(n_corr):
            w = weights[k]
            if w < 1e-8:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                mgt = n - lag
                if mgt > 0:
                    grad[lag:lag+mgt] += (1.0 - y[:mgt]) * w
                    grad[:mgt] += -y[lag:lag+mgt] * w
            else:
                j2 = -lag
                mgt = n - j2
                if mgt > 0:
                    grad[:mgt] += (1.0 - y[j2:]) * w
                    grad[j2:] += -y[:mgt] * w
        grad *= dx
        
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_eq(y - alpha * grad, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_eq(y_new, target_sum)
        
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

    """Find optimal v for h[j1]=v, h[j2]=s-v. Returns (best_v, best_c5)."""
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)

        return lo, _c5_of(h)
    
    # Ternary search
    a, b = lo, hi
    for _ in range(40):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[j1] = m1; h[j2] = s - m1

        else:
            a = m1
