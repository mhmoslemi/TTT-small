def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _grad_at_lag(h, lag_idx, dx):
    """Gradient of corr[lag_idx] with respect to h."""
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] -= h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad * dx

    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    shifted = alpha * (corr - mx)
    s = np.sum(np.exp(shifted))
    return mx + np.log(s) / alpha

def _smooth_max_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    shifted = alpha * (corr - mx)
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    # Top-k for efficiency
    k = min(20, len(corr))
    top_k = np.argsort(weights)[::-1][:k]
    w_k = weights[top_k]
    w_k = w_k / w_k.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for idx in range(len(top_k)):
        g += w_k[idx] * _grad_at_lag(x, top_k[idx], dx)
    return g

    n = len(h)

                improved = True

**Idea: Use the fact that the problem is equivalent to finding a set A with large minimum autocorrelation.**

The autocorrelation of a set A (indicator function) at lag k is:
R(k) = measure(A ∩ (A-k))

We want to maximize min_k R(k) subject to measure(A) = 1.

This is related to "difference sets" in additive combinatorics. A (v, k, λ) difference set has the property that every nonzero element of the group is represented exactly λ times as a difference of two elements of the set.

For the continuous case, if we can find a set A ⊂ [0,2] of measure 1 such that R(k) = c (constant) for all k, that would be ideal. This is like a "continuous difference set" or a "Boltyanskii-type" construction.

However, such perfect objects don't exist in general. The best we can do is make R(k) as close to constant as possible, which is what the optimization is doing.

**Another idea: Use the Fourier transform.**

The autocorrelation R(k) = ∫ h(x)h(x+k) dx is the inverse Fourier transform of |ĥ(ω)|². So:
R(k) = ∫ |ĥ(ω)|² e^{iωk} dω

We want to minimize max_k ∫ h(x)(1-h(x+k)) dx = max_k (1 - R(k)) = 1 - min_k R(k).

So we want to maximize min_k R(k) = min_k ∫ |ĥ(ω)|² e^{iωk} dω.

The constraint is ∫ h = 1, which means ĥ(0) = 1.
And h ∈ [0,1] means the inverse Fourier transform is in [0,1].

This is a non-trivial optimization in Fourier space. Not sure it's more practical than the direct approach.

import time
import numpy as np
from scipy.optimize import minimize

_g_target = 0.0
_g_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _grad_at_lag(h, lag_idx, dx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] -= h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad * dx

def _slsqp_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    shifted = alpha * (corr - mx)
    s = np.sum(np.exp(shifted))
    return mx + np.log(s) / alpha

def _slsqp_obj_jac(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    shifted = alpha * (corr - mx)
    weights = np.exp(shifted)
    k = min(30, len(corr))
    top_k = np.argsort(weights)[::-1][:k]
    w_k = weights[top_k].copy()
    w_k /= w_k.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for idx in range(len(top_k)):
        g += w_k[idx] * _grad_at_lag(x, top_k[idx], dx)
    return g

def _eq_con_fun(x, target, n):
    return x.sum() - target

def _eq_con_jac_fun(x, target, n):
    return np.ones(n)

def _slsqp_opt(h0, n, target, alpha, maxiter=300):
    global _g_target, _g_n
    _g_target = target
    _g_n = n
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac_fun, 'args': (target, n)}]
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP', args=(alpha, n),
                       jac=_slsqp_obj_jac, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = _fix_sum(res.x, target)
        if _c5_of(h_out) <= _c5_of(h0) + 1e-8:
            return h_out
    except Exception:
        pass
    return h0.copy()

def _coord_descent(h, target, max_rounds=20, iters_per_coord=30):
    n = len(h)
    h = h.copy()
    prev_c5 = _c5_of(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            old_v = h[j]
            lo, hi = 0.0, 1.0
            for _ in range(iters_per_coord):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[j] = m1
                h1 = _fix_sum(h, target)
                c1 = _c5_of(h1)
                h[j] = m2
                h2 = _fix_sum(h, target)
                c2 = _c5_of(h2)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            best_v = (lo + hi) / 2.0
            h[j] = best_v
            h = _fix_sum(h, target)
            c_new = _c5_of(h)
            if c_new < prev_c5 - 1e-14:
                prev_c5 = c_new
                improved = True
        if not improved:
            break
    return h, prev_c5

def _nesterov_subgrad(h0, n, target, n_iters=300, alpha0=0.35, top_k=8):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = _corr_full(y)
        mx = np.max(corr)
        k = min(top_k, len(corr))
        top_k_idx = np.argsort(corr)[::-1][:k]
        shifted = 25.0 * (corr[top_k_idx] - mx)
        weights = np.exp(shifted)
        weights /= weights.sum()
        g = np.zeros(n)
        for idx in range(len(top_k_idx)):
            g += weights[idx] * _grad_at_lag(y, top_k_idx[idx], dx)
        step = alpha0 / np.sqrt(it + 1)
        x_new = _fix_sum(y - step * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _fix_sum(y_new, target)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _perturb(h, target, rng, scale=0.05):
    n = len(h)
    h_p = h + scale * (rng.rand(n) - 0.5)

def _make_starts(h_init, n_init, n, target, rng):

    if n == n_init:

    h_blk = np.zeros(n)
    h_blk[::2] = 1.0
    starts.append(_fix_sum(h_blk, target))
