The parent uses log-sum-exp smooth max with increasing alpha. The issue is that for large alpha, the smooth max becomes very sharp and the gradient becomes concentrated on the argmax lags, which can cause numerical issues.

An alternative is to use the **power mean** (p-norm):
f_p(h) = (1/M * sum_k g_k(h)^p)^(1/p) for large p

This is differentiable and for p→∞ approaches the max. The gradient is:
∂f_p/∂h[i] = (1/p) * (sum g_k^p)^(-1/p) * p * sum_k g_k^(p-1) * ∂g_k/∂h[i]
= (sum g_k^p)^(1/p - 1) * sum_k g_k^(p-1) * ∂g_k/∂h[i]

For large p, this weights the lags by g_k^(p-1) / (sum g_k^p)^(1/p), which is heavily concentrated on the max.

The advantage of p-norm over log-sum-exp: it's more numerically stable for large p (no overflow in exp), and the gradient is smoother.

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def obj_func(h, alpha, mu, target_sum):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha
    s = np.sum(h) - target_sum
    return val + mu * s * s


def grad_func(h, alpha, mu, target_sum):
    N = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)

    grad = np.zeros(N)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-13:
            continue
        lag = idx - (N - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w * (1.0 - h[:N - l])
                grad[:N - l] -= w * h[l:]

    s = np.sum(h) - target_sum
    grad += 2.0 * mu * s
    return grad


def ljbfgs(h0, alpha, n_pts, mu, maxiter):
    target_sum = n_pts / 2.0
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj_func, h0, jac=lambda h: grad_func(h, alpha, mu, target_sum),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def to_binary(h, n):

    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

    current_max = np.max(g)

            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        else:
            g = g_new
            current_max = new_max
    return h_bin, current_max * dx

    current_max = np.max(g)

def thue_morse(n):
    h = np.zeros(n)
