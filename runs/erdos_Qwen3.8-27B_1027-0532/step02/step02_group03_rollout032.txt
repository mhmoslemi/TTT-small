def _c5_of(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    """Project y onto {x: 0<=x<=1, sum(x)=S}"""
    n = len(y)

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _de_optimize(h0, n, target_sum, pop_size, max_gens, F, CR, seed, deadline):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = h0
    for i in range(1, pop_size):
        pop[i] = _project_box_sum(rng.rand(n) * 1.8 - 0.1, target_sum)
    fitness = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    global_best = pop[best_idx].copy()
    global_best_f = fitness[best_idx]
    no_improve = 0
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        perm = rng.permutation(pop_size)
        new_pop = pop.copy()
        new_fit = fitness.copy()
        gen_improvement = False
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = perm[i % pop_size]
            r2 = perm[(i + 1) % pop_size]
            r3 = perm[(i + 2) % pop_size]

                r2 = rng.randint(pop_size)
                r3 = rng.randint(pop_size)
            if gen % 3 == 0:

            else:

            mutant = _project_box_sum(mutant, target_sum)
            trial = pop[i].copy()

            trial = _project_box_sum(trial, target_sum)
            f_trial = _c5_of(trial)
            if f_trial <= fitness[i]:
                new_pop[i] = trial
                new_fit[i] = f_trial
                if f_trial < global_best_f:
                    global_best_f = f_trial
                    global_best = trial.copy()
                    best_idx = i
                    gen_improvement = True
        pop = new_pop
        fitness = new_fit

            if no_improve > 30:

    cooling = 0.9997

    while time.time() < deadline:

def _continuous_refine(h0, n, target_sum, n_iters, deadline):
    dx = 2.0 / n
    x = h0.copy().astype(float)
    best = x.copy()
    best_val = _c5_of(x)
    step = 0.05
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_k = min(5, len(corr))
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        gap = corr[sorted_idx[0]] - corr[sorted_idx]
        weights = np.exp(-gap * 50.0)
        weights = weights / weights.sum()
        g = np.zeros(n)
        for idx in range(top_k):
            g += weights[idx] * _grad_corr_term(x, sorted_idx[idx], dx)
        x_new = _project_box_sum(x - step * g, target_sum)
        f_new = _c5_of(x_new)
        if f_new < best_val:
            best_val = f_new
            best = x_new.copy()
            x = x_new
            step = min(step * 1.1, 0.1)
        else:
            x = x_new
            step *= 0.7
            if step < 1e-5:
                step = 0.01
    return best, best_val

def _slsqp_polish(h0, n, target_sum, alpha, maxiter, deadline):
    dx = 2.0 / n
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return m * dx
        return (m + np.log(np.sum(np.exp((corr - m) * alpha)) / alpha)) * dx
    def _obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        w = np.exp((corr - m) * alpha)
        w = w / w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-10:
                g += w[k] * _grad_corr_term(x, k, dx)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return _fix_sum(res.x, target_sum)
    except Exception:
        pass
    return h0

                                pop_size=pop_size, max_gens=300,
                                F=0.8, CR=0.7, seed=seed, deadline=min(deadline, time.time() + 120))
    h_round = (h_de > 0.5).astype(float)
    h_round = _fix_sum(h_round, target_sum)
    c5_round = _c5_of(h_round)
    best_local = h_de.copy()
    best_f = c5_de
    if c5_round < best_f:
        best_f = c5_round
        best_local = h_round.copy()
    h_sa, c5_sa = _sa_01(best_local, n, target_sum, deadline=min(deadline, time.time() + 100), seed=seed+100)
    if c5_sa < best_f:
        best_f = c5_sa
        best_local = h_sa.copy()
    h_cd = _coord_descent_fast(best_local, n, target_sum, max_rounds=10, deadline=min(deadline, time.time() + 60))
    c5_cd = _c5_of(h_cd)
    if c5_cd < best_f:
        best_f = c5_cd
        best_local = h_cd.copy()
    h_cont, c5_cont = _continuous_refine(best_local, n, target_sum, n_iters=500, deadline=min(deadline, time.time() + 80))
    if c5_cont < best_f:
        best_f = c5_cont
        best_local = h_cont.copy()
    return best_local, best_f

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_candidates = [n_init]
    for delta in [2, -2, 4, -4, 6, -6, 8, -8, 10, -10]:
        n_try = n_init + delta
        if n_try % 2 == 0 and 60 <= n_try <= 160:
            n_candidates.append(n_try)
    seen = set()
    n_list = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_list.append(nn)

    time_per_n = max(80, min(250, (deadline - t_start - 60) // len(n_list)))

    for n in n_list:
        t_phase_start = time.time()
        phase_deadline = min(deadline - 50, t_phase_start + time_per_n)
        if phase_deadline < t_start + 30:
            break
        target_sum = n / 2.0

        if n == n_init:
            h0 = _fix_sum(h_init.copy(), target_sum)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h0 = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)

        c0 = _c5_of(h0)
        if c0 < best_c5:
            best_c5 = c0
            best_h = h0.copy()
            best_n = n

        structured = _try_structured(n, rng)
        for h_s in structured:
            c5_s = _c5_of(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n

        n_starts = 3
        time_per_start = max(30, (phase_deadline - 20) // n_starts)
        for start_idx in range(n_starts):
            t_sub_start = time.time()
            sub_deadline = min(phase_deadline - 10, t_sub_start + time_per_start)
            if sub_deadline < t_start + 10:
                break
            if start_idx == 0:
                h_start = h0
            elif start_idx == 1:
                h_start = _fix_sum(rng.rand(n), target_sum)
            else:
                h_start = structured[0] if len(structured) > 0 else h0

            h_pipe, c5_pipe = _pipeline_start(h_start, n, target_sum, sub_deadline, seed + start_idx * 1000 + n)
            if c5_pipe < best_c5:
                best_c5 = c5_pipe
                best_h = h_pipe.copy()
                best_n = n

    if best_h is not None:
        n_best = best_n
        target = n_best / 2.0

        h_r = (best_h > 0.5).astype(float)
        h_r = _fix_sum(h_r, target)
        c5_r = _c5_of(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

        h_cd_f = _coord_descent_fast(best_h, n_best, target, max_rounds=15, deadline=deadline - 30)
        c5_cd_f = _c5_of(h_cd_f)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

        h_sa_f, c5_sa_f = _sa_01(best_h, n_best, target, deadline=deadline - 20, seed=seed + 9999)
        if c5_sa_f < best_c5:
            best_c5 = c5_sa_f
            best_h = h_sa_f.copy()

        h_cont_f, c5_cont_f = _continuous_refine(best_h, n_best, target, n_iters=800, deadline=deadline - 10)
        if c5_cont_f < best_c5:
            best_c5 = c5_cont_f
            best_h = h_cont_f.copy()

        if time.time() < deadline - 10:
            h_sl = _slsqp_polish(best_h, n_best, target, alpha=200, maxiter=200, deadline=deadline)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
