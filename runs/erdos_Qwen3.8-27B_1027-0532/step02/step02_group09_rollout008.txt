    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-8
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad

def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)

def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)

def sum_eq(h):
    return np.sum(h) - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def lbfgsb_optimize(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None

def slsqp_optimize(h0, alpha, maxiter, bounds):
    try:
        res = minimize(
            slsqp_obj, h0, args=(alpha,),
            method='SLSQP', jac=slsqp_grad, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None

def multi_flip_binary_search(h0, n_iters, rng, flip_size=2):
    n = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    if n_ones > n_target:
        idxs = np.where(h_bin == 1.0)[0]
        rng.shuffle(idxs)
        for i in range(n_ones - n_target):
            h_bin[idxs[i]] = 0.0
    elif n_ones < n_target:
        idxs = np.where(h_bin == 0.0)[0]
        rng.shuffle(idxs)
        for i in range(n_target - n_ones):
            h_bin[idxs[i]] = 1.0
    
    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    temp = 0.003
    
    for it in range(n_iters):
        if it % 500 == 499:
            temp *= 0.95
        if temp < 1e-10:
            break
        
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < flip_size or len(zeros) < flip_size:
            break
        
        # Pick flip_size ones to turn off and flip_size zeros to turn on
        pick_ones = rng.choice(ones, size=flip_size, replace=False)
        pick_zeros = rng.choice(zeros, size=flip_size, replace=False)
        
        h_bin[pick_ones] = 0.0
        h_bin[pick_zeros] = 1.0
        
        s_new = evaluate_score(h_bin)
        delta = s_new - best_score
        
        accept = False
        if s_new < best_score:
            accept = True
            best_score = s_new
            best_h = h_bin.copy()
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                accept = True
        
        if not accept:
            h_bin[pick_ones] = 1.0
            h_bin[pick_zeros] = 0.0
    
    return best_h, best_score

def coordinate_descent_swap(h, n_iters, rng, target):
    n = len(h)
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for it in range(n_iters):
        # Pick a random pair (i, j) and try to swap part of their values
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Try moving mass from i to j or j to i
        # h[i] += delta, h[j] -= delta (or vice versa)
        for direction in [+1, -1]:
            # h[i] += direction*delta, h[j] -= direction*delta
            # Find best delta in [-h[i], 1-h[i]] ∩ [h[j]-1, -h[j]] (adjusted for direction)
            
            if direction > 0:
                # h[i] increases, h[j] decreases
                # delta in [0, 1-h[i]] and [0, h[j]]
                delta_max = min(1.0 - h[i], h[j])
                if delta_max <= 0:
                    continue
                # Try a few delta values
                for frac in [0.1, 0.25, 0.5, 0.75, 1.0]:
                    delta = delta_max * frac
                    h_new = h.copy()
                    h_new[i] += delta
                    h_new[j] -= delta
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h = h_new.copy()
            
    return best_h, best_score

def subgrad_optimize(h0, n_iters, lr_init, target, top_k):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    for it in range(n_iters):
        lr = lr_init / (1.0 + 0.03 * float(it))
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(-overlaps)[:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            L = idx - (n - 1)
            if L >= 0:
                grad[:n - L] += (1.0 - h[L:])
                grad[L:] -= h[:n - L]
            else:
                M = -L
                grad[M:] += (1.0 - h[:n - M])
                grad[:n - M] -= h[M:]
        grad /= float(top_k)
        h_new = project_box_sum(h - lr * grad, target)
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
    return best_h, best_raw

    n_target = n // 2
    if n_ones > n_target:
        idxs = np.where(h_bin == 1.0)[0]
        order = rng.permutation(len(idxs))
        for i in range(n_ones - n_target):
            h_bin[idxs[order[i]]] = 0.0
    elif n_ones < n_target:
        idxs = np.where(h_bin == 0.0)[0]
        order = rng.permutation(len(idxs))
        for i in range(n_target - n_ones):
            h_bin[idxs[order[i]]] = 1.0

    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    temp = 0.004

    for it in range(n_iters):
        if it % 400 == 399:
            temp *= 0.93
        if temp < 1e-10:
            break

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < flip_size or len(zeros) < flip_size:
            break

        pick_ones = rng.choice(ones, size=flip_size, replace=False)
        pick_zeros = rng.choice(zeros, size=flip_size, replace=False)

        h_bin[pick_ones] = 0.0
        h_bin[pick_zeros] = 1.0

        s_new = evaluate_score(h_bin)
        delta = s_new - best_score

        accept = False
        if s_new < best_score:
            accept = True
            best_score = s_new
            best_h = h_bin.copy()
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                accept = True

        if not accept:
            h_bin[pick_ones] = 1.0
            h_bin[pick_zeros] = 0.0

    return best_h, best_score


def subgrad_optimize(h0, n_iters, lr_init, target, top_k):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    for it in range(n_iters):
        lr = lr_init / (1.0 + 0.03 * float(it))
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(-overlaps)[:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            L = idx - (n - 1)
            if L >= 0:
                grad[:n - L] += (1.0 - h[L:])
                grad[L:] -= h[:n - L]
            else:
                M = -L
                grad[M:] += (1.0 - h[:n - M])
                grad[:n - M] -= h[M:]
        grad /= float(top_k)
        h_new = project_box_sum(h - lr * grad, target)
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
    return best_h, best_raw


def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_score = evaluate_score(h)

    # Phase 1: L-BFGS-B at original resolution, alpha from low to high
    alphas1 = [80, 200, 500, 1500, 4000]
    for alpha in alphas1:
        if time.time() - t0 > budget_s - 120:
            break
        penalty = alpha * 40.0
        bounds = [(0.0, 1.0)] * n
        maxiter = max(200, 600000 // n)
        res_x = lbfgsb_optimize(h, alpha, penalty, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    # Phase 2: SLSQP for exact constraint handling
    for alpha in [300, 1500]:
        if time.time() - t0 > budget_s - 150:
            break
        bounds = [(0.0, 1.0)] * n
        maxiter = max(100, 400000 // n)
        res_x = slsqp_optimize(h, alpha, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    # Phase 3: Upscale to 2x resolution
    if time.time() - t0 < budget_s - 250:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_box_sum(h2, n2 / 2.0)
        for alpha in [80, 300, 1000, 3000, 8000]:
            if time.time() - t0 > budget_s - 120:
                break
            penalty = alpha * 35.0
            bounds = [(0.0, 1.0)] * n2
            maxiter = max(100, 300000 // n2)
            res_x = lbfgsb_optimize(h2, alpha, penalty, maxiter, bounds)
            if res_x is not None:
                h2 = project_box_sum(np.clip(res_x, 0.0, 1.0), n2 / 2.0)
                s = evaluate_score(h2)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h2.copy()

    # Phase 4: Upscale to 3x resolution
    if time.time() - t0 < budget_s - 180:
        n3 = n * 3
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n3, endpoint=False)
        h3 = np.interp(x_new, x_old, best_h)
        h3 = project_box_sum(h3, n3 / 2.0)
        for alpha in [100, 400, 1500, 5000]:
            if time.time() - t0 > budget_s - 80:
                break
            penalty = alpha * 30.0
            bounds = [(0.0, 1.0)] * n3
            maxiter = max(80, 200000 // n3)
            res_x = lbfgsb_optimize(h3, alpha, penalty, maxiter, bounds)
            if res_x is not None:
                h3 = project_box_sum(np.clip(res_x, 0.0, 1.0), n3 / 2.0)
                s = evaluate_score(h3)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h3.copy()

    # Phase 5: Multi-flip binary search at best resolution
    n_cur = len(best_h)
    if time.time() - t0 < budget_s - 100:
        n_bf = min(20000, int((budget_s - 100 - (time.time() - t0)) * 8))
        h_bin, s_bin = multi_flip_binary_search(best_h, n_bf, rng, 2)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()

    # Phase 6: Multi-flip with size 3
    if time.time() - t0 < budget_s - 60:
        n_cur = len(best_h)
        n_bf3 = min(12000, int((budget_s - 60 - (time.time() - t0)) * 5))
        h_bin3, s_bin3 = multi_flip_binary_search(best_h, n_bf3, rng, 3)
        if s_bin3 < best_score:
            best_score = s_bin3
            best_h = h_bin3.copy()

    # Phase 7: Subgradient at best resolution
    if time.time() - t0 < budget_s - 50:
        n_cur = len(best_h)
        n_sub = min(5000, int((budget_s - 50 - (time.time() - t0)) * 6))
        h_sub, raw_sub = subgrad_optimize(best_h, n_sub, 0.5, n_cur / 2.0, 4)
        s_sub = raw_sub * (2.0 / n_cur)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()

    # Phase 8: Random restart + optimize (try a different basin)
    if time.time() - t0 < budget_s - 100:
        n_restart = len(best_h)
        h_rand = random_binary_init(n_restart, rng)
        h_rand = project_box_sum(h_rand, n_restart / 2.0)
        for alpha in [200, 1000, 3000]:
            if time.time() - t0 > budget_s - 40:
                break
            penalty = alpha * 40.0
            bounds = [(0.0, 1.0)] * n_restart
            maxiter = max(100, 200000 // n_restart)
            res_x = lbfgsb_optimize(h_rand, alpha, penalty, maxiter, bounds)
            if res_x is not None:
                h_rand = project_box_sum(np.clip(res_x, 0.0, 1.0), n_restart / 2.0)
                s = evaluate_score(h_rand)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_rand.copy()

    # Phase 9: Final high-precision L-BFGS-B polishing
    if time.time() - t0 < budget_s - 20:
        n_final = len(best_h)
        for alpha in [8000, 15000]:
            if time.time() - t0 > budget_s - 10:
                break
            penalty = alpha * 50.0
            bounds = [(0.0, 1.0)] * n_final
            res_x = lbfgsb_optimize(best_h, alpha, penalty, 400, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
