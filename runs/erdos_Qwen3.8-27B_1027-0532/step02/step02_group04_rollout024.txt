import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha

def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()

        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    """Compute subgradient of correlation at a specific shift."""
    n = len(h)
    grad = np.zeros(n)
    n_min = max(0, -shift)
    n_max = min(n, n - shift)

    np.add.at(grad, idx + shift, 1.0 - h[idx])
    np.add.at(grad, idx, -h[idx + shift])
    return grad

def subgradient_optimize(h0, n_subgrad_iters, eta0, eta_min, rng):
    """Optimize using subgradient method targeting the worst lag."""
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = np.max(compute_correlations(h))
    
    for it in range(n_subgrad_iters):
        eta = eta0 * (eta_min / eta0) ** (it / max(1, n_subgrad_iters - 1))
        
        corr = compute_correlations(h)
        max_idx = np.argmax(corr)
        shift = n - 1 - max_idx  # convert index to shift
        
        grad = subgradient_for_lag(h, shift)
        h_new = h - eta * grad
        h = project_feasible(h_new, n / 2.0)
        
        current_max = np.max(compute_correlations(h))
        if current_max < best_score:
            best_score = current_max
            best_h = h.copy()
    
    return best_h, best_score

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

Let me also add one more idea: **iterative best response**. 

The problem can be written as:
min_h max_k sum_{n} h[n+k](1-h[n])

Fix all of h except for a block of positions. The optimal values for that block (given the rest) can be found by optimization. This is block coordinate descent.

For a single position h[m], given all other h[i] (i≠m):
The objective becomes: max_k (a_k + b_k * h[m])
where a_k = sum_{n in S_k, n≠m, n+k≠m} h[n+k](1-h[n]) (the part not involving h[m])
and b_k = coefficient of h[m] in c_k.

We want to choose h[m] in [0,1] to minimize max_k (a_k + b_k * h[m]).

This is a 1D optimization problem! The function max_k (a_k + b_k * h[m]) is the max of linear functions of h[m], which is convex. We can minimize it in closed form.

The minimum of max_k (a_k + b_k * t) over t in [0,1] is achieved at:
- t = 0 if the function is decreasing at t=0
- t = 1 if the function is increasing at t=1
- Otherwise, at the point where the two "active" linear functions cross

More precisely, let f(t) = max_k (a_k + b_k * t).
f(0) = max_k a_k
f(1) = max_k (a_k + b_k)

If f is decreasing on [0,1] (i.e., all b_k < 0 for the active k's at t=1), then t*=1.
If f is increasing on [0,1] (i.e., all b_k > 0 for the active k's at t=0), then t*=0.
Otherwise, t* is where the max changes from one function to another.

This gives us an exact coordinate descent step!

def coordinate_descent_sweep(h, grid_size=101, n_sweeps=1):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full')
    best_max = np.max(corr)
    
    for sweep in range(n_sweeps):
        for m in range(n):
            h_old = h[m]
            
            # Compute b_k for all k (the coefficient of h[m] in c_k)
            # b_k = (1-h[m-k])*I(m-k valid) - h[m+k]*I(m valid as n)
            # where "valid" depends on the lag
            b = np.zeros(2*n-1)
            for k_idx in range(2*n-1):
                k = k_idx - (n-1)  # actual shift
                bc = 0.0
                # m appears as n+k (= m) when n = m-k
                n_val = m - k
                if 0 <= n_val < n and 0 <= m < n:
                    bc += (1.0 - h[n_val])
                # m appears as n (= m) when n+k >= 0
                if 0 <= m < n and 0 <= m + k < n:
                    bc += (-h[m + k])
                b[k_idx] = bc
            
            # a_k = corr_k - h[m] * b_k
            a = corr - h_old * b
            
            # 1D optimization: min_t max_k (a[k] + t * b[k]) for t in [0,1]
            t_grid = np.linspace(0, 1, grid_size)
            vals = np.max(a[None, :] + t_grid[:, None] * b[None, :], axis=1)
            best_t = t_grid[np.argmin(vals)]
            
            h[m] = best_t
            
            # Update corr
            delta = best_t - h_old
            if abs(delta) > 1e-15:
                corr += delta * b
        
        new_max = np.max(corr)
        if new_max < best_max:
            best_max = new_max
        else:
            break
    
    return h, best_max

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def coordinate_descent(h, n_sweeps=10, grid_size=201):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    best_max = np.max(corr)
    k_values = np.arange(2 * n - 1) - (n - 1)
    t_grid = np.linspace(0.0, 1.0, grid_size)
    for sweep in range(n_sweeps):
        improved = False
        for m in range(n):
            h_old = h[m]
            valid1 = (m - k_values >= 0) & (m - k_values < n)
            valid2 = (m + k_values >= 0) & (m + k_values < n)
            b = np.zeros(2 * n - 1)
            b[valid1] = 1.0 - h[m - k_values[valid1]]
            b[valid2] -= h[m + k_values[valid2]]
            a = corr - h_old * b
            vals = np.max(a[None, :] + t_grid[:, None] * b[None, :], axis=1)
            best_t = t_grid[np.argmin(vals)]
            if abs(best_t - h_old) > 1e-10:
                h[m] = best_t
                corr += (best_t - h_old) * b
                improved = True
        current_max = np.max(corr)
        if current_max < best_max:
            best_max = current_max
        if not improved:
            break
    return h, best_max


def gradient_guided_search(h, n_moves, rng):
    n = len(h)
    best = h.copy()
    corr = compute_correlations(best)
    best_max = np.max(corr)
    alpha = 20000
    for _ in range(n_moves):
        grad = smooth_max_gradient(best, alpha)
        neg_idx = np.argsort(grad)
        pos_idx = np.argsort(grad)[::-1]
        improved = False
        for i in neg_idx[:15]:
            if improved:
                break
            for j in pos_idx[:15]:
                if i == j:
                    continue
                for step in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
                    h_try = best.copy()
                    h_try[i] = best[i] + step
                    h_try[j] = best[j] - step
                    if h_try[i] > 1.0 or h_try[i] < 0.0 or h_try[j] > 1.0 or h_try[j] < 0.0:
                        continue
                    c_try = compute_correlations(h_try)
                    m_try = np.max(c_try)
                    if m_try < best_max - 1e-12:
                        best = h_try
                        corr = c_try
                        best_max = m_try
                        improved = True

        if not improved:
            for _ in range(30):

                if i == j:

                h_try = best.copy()
                h_try[i] = best[i] + step

def block_local_search(h, n_iters, rng):
    n = len(h)
    best = h.copy()
    corr = compute_correlations(best)
    best_max = np.max(corr)
    for _ in range(n_iters):
        move_type = rng.randint(4)
        improved = False
        if move_type == 0:
            bsize = rng.randint(2, min(20, n // 4))
            s1 = rng.randint(0, n - bsize)
            s2 = rng.randint(0, n - bsize)

            h_try = best.copy()
            h_try[s1:s1+bsize], h_try[s2:s2+bsize] = best[s2:s2+bsize].copy(), best[s1:s1+bsize].copy()
            c_try = compute_correlations(h_try)
            m_try = np.max(c_try)
            if m_try < best_max - 1e-12:
                best = h_try
                corr = c_try
                best_max = m_try
        elif move_type == 1:
            bsize = rng.randint(3, min(25, n // 3))
            s1 = rng.randint(0, n - bsize)
            h_try = best.copy()
            h_try[s1:s1+bsize] = 1.0 - best[s1:s1+bsize]
            c_try = compute_correlations(h_try)
            m_try = np.max(c_try)
            if m_try < best_max - 1e-12:
                best = h_try
                corr = c_try
                best_max = m_try
        elif move_type == 2:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for step in [0.01, 0.02, 0.05, 0.1, 0.2]:
                h_try = best.copy()
                h_try[i] = best[i] + step
                h_try[j] = best[j] - step
                if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                    c_try = compute_correlations(h_try)
                    m_try = np.max(c_try)
                    if m_try < best_max - 1e-12:
                        best = h_try
                        corr = c_try
                        best_max = m_try
                        improved = True
                        break
        else:
            n_pts = rng.randint(2, 6)
            src = rng.choice(n, n_pts, replace=False)
            dst = rng.choice(n, n_pts, replace=False)
            step = rng.choice([0.02, 0.05, 0.1])
            h_try = best.copy()
            valid = True
            for s, d in zip(src, dst):
                if s == d:
                    valid = False
                    break
                h_try[s] -= step
                h_try[d] += step
            if valid and np.all(h_try >= -1e-9) and np.all(h_try <= 1 + 1e-9):
                h_try = np.clip(h_try, 0, 1)
                c_try = compute_correlations(h_try)
                m_try = np.max(c_try)
                if m_try < best_max - 1e-12:
                    best = h_try
                    corr = c_try
                    best_max = m_try
    return best, best_max


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    best_h = project_feasible(init, n / 2.0)
    best_score = evaluate_score(best_h)

    MU = 100000.0

    # Phase 1: L-BFGS-B alpha sweep
    alphas_p1 = [50, 100, 200, 500, 1000, 2000, 5000]
    for alpha in alphas_p1:
        if time.time() - t0 > budget_s - 400:
            break
        maxiter = max(200, max(50000 // n, 300))
        x = ljbfgs_call(best_h, alpha, n, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 2: Coordinate descent
    if time.time() - t0 < budget_s - 300:
        h_cd, _ = coordinate_descent(best_h, n_sweeps=20, grid_size=201)
        h_cd = project_feasible(h_cd, n / 2.0)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd.copy()

    # Phase 3: Higher alpha L-BFGS-B
    alphas_p3 = [5000, 10000, 20000, 50000]
    for alpha in alphas_p3:
        if time.time() - t0 > budget_s - 300:
            break
        maxiter = max(300, 100000 // n)
        x = ljbfgs_call(best_h, alpha, n, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 4: More coordinate descent
    if time.time() - t0 < budget_s - 250:
        h_cd, _ = coordinate_descent(best_h, n_sweeps=30, grid_size=301)
        h_cd = project_feasible(h_cd, n / 2.0)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd.copy()

    # Phase 5: Diverse restarts
    n_restarts = 150
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 200:
            break
        strategy = i % 5
        if strategy == 0:
            pert = best_h + rng.randn(n) * 0.04
        elif strategy == 1:
            pert = best_h.copy()
            blen = rng.randint(5, min(30, n // 5))
            bstart = rng.randint(0, n - blen)
            pert[bstart:bstart+blen] = 1.0 - pert[bstart:bstart+blen]
        elif strategy == 2:
            pert = best_h.copy()
            nf = rng.randint(1, 6)
            x_idx = np.arange(n, dtype=float)
            for _ in range(nf):
                f = rng.randint(1, 12)
                amp = rng.uniform(0.02, 0.08)
                ph = rng.rand() * 2.0 * np.pi
                pert += amp * np.sin(2.0 * np.pi * f * x_idx / n + ph)
        elif strategy == 3:
            pert = best_h + rng.randn(n) * 0.12
            pert = project_feasible(pert, n / 2.0)
            nc = max(50, n // 5)
            xc = np.linspace(0, 2, nc, endpoint=False)
            xf = np.linspace(0, 2, n, endpoint=False)
            coarse = np.interp(xc, xf, pert)
            coarse = project_feasible(coarse, nc / 2.0)
            pert = np.interp(xf, xc, coarse)
        else:
            pert = best_h.copy()
            nb = rng.randint(2, 5)
            bs = n // nb
            for b in range(nb):
                s0 = b * bs
                s1 = min(s0 + bs, n)
                if rng.rand() < 0.5:
                    pert[s0:s1] = 1.0 - pert[s0:s1]
                else:
                    pert[s0:s1] += rng.randn(s1 - s0) * 0.08

        pert = project_feasible(pert, n / 2.0)
        alpha_pick = [200, 500, 2000, 5000, 10000, 20000][rng.randint(6)]
        maxiter_r = max(100, 40000 // n)
        x = ljbfgs_call(pert, alpha_pick, n, MU, maxiter_r)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                h_new_cd, _ = coordinate_descent(h_new, n_sweeps=10, grid_size=151)
                h_new_cd = project_feasible(h_new_cd, n / 2.0)
                s2 = evaluate_score(h_new_cd)
                if s2 < best_score:
                    best_score = s2
                    best_h = h_new_cd.copy()

    # Phase 6: Gradient-guided local search
    if time.time() - t0 < budget_s - 100:
        remaining = budget_s - 100 - (time.time() - t0)
        n_moves = max(500, int(remaining * 50))
        best_h, _ = gradient_guided_search(best_h, n_moves, rng)
        best_h = project_feasible(best_h, n / 2.0)
        best_score = evaluate_score(best_h)

    # Phase 7: Block local search
    if time.time() - t0 < budget_s - 60:
        remaining = budget_s - 60 - (time.time() - t0)
        n_ls = max(2000, int(remaining * 60))
        best_h, _ = block_local_search(best_h, n_ls, rng)
        best_h = project_feasible(best_h, n / 2.0)
        best_score = evaluate_score(best_h)

    # Phase 8: Higher resolution
    for n_try in [2 * n, 3 * n]:
        if n_try > 2000:
            continue
        if time.time() - t0 > budget_s - 100:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_feasible(h_init, n_try / 2.0)

        for alpha in [200, 1000, 5000, 20000]:
            if time.time() - t0 > budget_s - 50:
                break
            maxiter_n = max(50, 200000 // n_try)
            x = ljbfgs_call(h_init, alpha, n_try, MU, maxiter_n)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_try / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_init = h_new.copy()
                    h_cd, _ = coordinate_descent(h_new, n_sweeps=5, grid_size=101)
                    h_cd = project_feasible(h_cd, n_try / 2.0)
                    s2 = evaluate_score(h_cd)
                    if s2 < best_score:
                        best_score = s2
                        best_h = h_cd.copy()
                        h_init = h_cd.copy()

    # Phase 9: Final polish
    n_final = len(best_h)
    for alpha in [50000, 100000, 200000]:
        if time.time() - t0 > budget_s - 20:
            break
        maxiter_f = max(100, 150000 // n_final)
        x = ljbfgs_call(best_h, alpha, n_final, MU, maxiter_f)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_final / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Final coordinate descent polish
    if time.time() - t0 < budget_s - 10:
        h_cd, _ = coordinate_descent(best_h, n_sweeps=20, grid_size=301)
        h_cd = project_feasible(h_cd, len(h_cd) / 2.0)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
