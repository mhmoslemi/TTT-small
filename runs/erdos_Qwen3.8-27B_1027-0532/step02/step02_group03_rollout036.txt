def _de_optimize(h_start, n, target_sum, n_pop, n_gens, F=0.7, CR=0.8, 
                 deadline=None):
    """Differential evolution with feasibility projection"""
    # Initialize population
    pop = np.zeros((n_pop, n))
    pop[0] = _fix_sum(h_start.copy(), target_sum)
    for i in range(1, n_pop):
        if i % 3 == 1:
            pop[i] = _fix_sum(np.random.rand(n), target_sum)
        elif i % 3 == 2:
            pop[i] = _fix_sum(np.clip(h_start + 0.3*np.random.randn(n), 0, 1), target_sum)
        else:
            pop[i] = _fix_sum(np.clip(h_start[:n] + 0.1*np.random.randn(n), 0, 1), target_sum)
    
    fitness = np.array([_c5_of(pop[i]) for i in range(n_pop)])
    best_val = fitness.min()
    best_idx = fitness.argmin()
    best_h = pop[best_idx].copy()
    
    for gen in range(n_gens):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n_pop):
            if deadline is not None and time.time() > deadline:
                break
            # Select three distinct individuals (not i)
            while True:
                a, b, c = np.random.choice(n_pop, 3, replace=False)
                if a != b and b != c and a != c and a != i and b != i and c != i:
                    break
            mutant = np.clip(pop[a] + F * (pop[b] - pop[c]), 0, 1)
            mutant = _project_sum(mutant, target_sum)
            
            # Crossover
            trial = pop[i].copy()
            j_rand = np.random.randint(n)
            for j in range(n):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
            trial = _fix_sum(trial, target_sum)
            
            f_trial = _c5_of(trial)
            f_victim = fitness[i]
            
            if f_trial <= f_victim:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_val:
                    best_val = f_trial
                    best_h = trial.copy()
                    best_idx = i
            # Elite preservation
            if fitness[best_idx] > best_val:
                fitness[best_idx] = best_val
                pop[best_idx] = best_h.copy()
    
    return best_h, best_val

def _anneal_local(h0, n, target_sum, n_steps, temp0, deadline=None):
    """Simulated annealing on the discrete-like space"""
    h = h0.copy()
    best = h.copy()
    best_val = _c5_of(h)
    temp = temp0
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        temp *= 0.995
        # Random perturbation
        n_perturb = max(1, n // 20)
        indices = np.random.choice(n, n_perturb, replace=False)
        h_trial = h.copy()
        h_trial[indices] = np.clip(h_trial[indices] + np.random.randn(n_perturb) * 0.1 * temp * 10, 0, 1)
        h_trial = _fix_sum(h_trial, target_sum)
        
        try_val = _c5_of(h_trial)
        delta = try_val - _c5_of(h)
        
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-10)):
            h = h_trial
            if try_val < best_val:
                best_val = try_val
                best = h.copy()
    
    return best, best_val

            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        
        m = min(top_m, len(weights))
        top_idx = np.argsort(weights)[::-1][:m]
        w_m = weights[top_idx].copy()
        w_m = w_m / w_m.sum()
        
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            g += w_m[idx] * _grad_corr_term(y, top_idx[idx], dx)
        
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_sum(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _smooth_obj_and_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -500.0, 0.0)
    e = np.exp(alpha * shifted)
    w = e / e.sum()
    lse = (m + np.log(e.sum()) / alpha) * (2.0 / n)
    
    # Gradient
    g = np.zeros(n)
    # Only compute gradient for significant weights
    thresh = 1e-10
    for k_idx in np.where(w > thresh)[0]:
        g += w[k_idx] * _grad_corr_term(x, k_idx, 2.0/n)
    
    return lse, g

def _slsqp_opt(h0, n, target_sum, alpha, maxiter=300):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    
    def obj(x):
        v, _ = _smooth_obj_and_grad(x, alpha, n)
        return v
    
    def obj_grad(x):
        _, g = _smooth_obj_and_grad(x, alpha, n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent_binary(h0, n, target_sum, max_rounds=20):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            if c0 <= c1:
                h[j] = 0.0
            else:
                h[j] = 1.0
            c_new = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if c_new < co - 1e-12:
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=8, n_vals=7):
    h = h0.copy()
    vals = np.linspace(0.0, 1.0, n_vals)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _fine_coord_descent(h0, n, target_sum, max_rounds=5, n_vals=21):
    """Finer grid coordinate descent"""
    h = h0.copy()
    vals = np.linspace(0.0, 1.0, n_vals)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            # Only search in a neighborhood of the current value
            lo = max(0.0, old - 0.3)
            hi = min(1.0, old + 0.3)
            local_vals = np.linspace(lo, hi, n_vals)
            best_c = _c5_of(h)
            best_v = old
            for v in local_vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _de_optimize(h_start, n, target_sum, n_pop, n_gens, F=0.6, CR=0.8, deadline=None):
    pop = np.zeros((n_pop, n))
    pop[0] = _fix_sum(h_start.copy(), target_sum)
    for i in range(1, n_pop):
        r = np.random.rand()
        if r < 0.3:
            pop[i] = _fix_sum(np.random.rand(n), target_sum)
        elif r < 0.6:
            pop[i] = _fix_sum(np.clip(h_start + 0.3*np.random.randn(n), 0, 1), target_sum)
        else:
            pop[i] = _fix_sum(np.clip(h_start + 0.1*np.random.randn(n), 0, 1), target_sum)
    
    fitness = np.array([_c5_of(pop[i]) for i in range(n_pop)])
    best_val = fitness.min()
    best_h = pop[fitness.argmin()].copy()
    
    for gen in range(n_gens):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n_pop):
            if deadline is not None and time.time() > deadline:
                break
            a, b, c = np.random.choice(n_pop, 3, replace=False)
            while a == i or b == i or c == i:
                a, b, c = np.random.choice(n_pop, 3, replace=False)
            mutant = np.clip(pop[a] + F * (pop[b] - pop[c]), 0, 1)
            mutant = _project_sum(mutant, target_sum)
            
            trial = pop[i].copy()
            j_rand = np.random.randint(n)
            mask = (np.random.rand(n) < CR) | (np.arange(n) == j_rand)
            trial[mask] = mutant[mask]
            trial = _fix_sum(trial, target_sum)
            
            f_trial = _c5_of(trial)
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_val:
                    best_val = f_trial
                    best_h = trial.copy()
    
    return best_h, best_val

def _interp(h_src, n_src, n_dst):
    x_src = np.arange(n_src, dtype=float)
    x_dst = np.linspace(0, n_src - 1, n_dst)
    return np.clip(np.interp(x_dst, x_src, h_src), 0.0, 1.0)

    if time.time() < deadline - 90:

        if c5 < best_c5:
