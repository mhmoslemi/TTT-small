import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha_start, alpha_end, lr):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        frac = float(it) / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac if alpha_end > 0 else alpha_start
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(corr))
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= w_sum
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-6:
                g += w[k_idx] * _grad_corr_term(y, k_idx, dx)
        step = lr / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _top_k_subgrad(h0, n, target_sum, n_iters, lr, top_k):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_indices = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for k_idx in top_indices:
            g += _grad_corr_term(y, k_idx, dx)
        g /= float(top_k)
        step = lr / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent_01(h0, n, target_sum, max_rounds=15):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            co = _c5_of(h)
            if old <= 0.001 or old >= 0.999:
                continue
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            if min(c0, c1) < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h

def _ternary_coord(h0, n, target_sum, max_rounds=8):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            co = _c5_of(h)
            lo = 0.0
            hi = 1.0
            best_c = co
            best_v = old
            for _ in range(8):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[j] = m1
                c1 = _c5_of(h)
                h[j] = m2
                c2 = _c5_of(h)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
                for mid in [m1, m2]:
                    h[j] = mid
                    cm = _c5_of(h)
                    if cm < best_c:
                        best_c = cm
                        best_v = mid
            h[j] = best_v
            if best_c < co - 1e-10:
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h

def _quantize_entropy(h0, n, target_sum, levels):
    h = h0.copy()
    for j in range(n):
        best_c = _c5_of(h)
        best_v = h[j]
        for lv in levels:
            h[j] = lv
            c = _c5_of(h)
            if c < best_c - 1e-12:
                best_c = c
                best_v = lv
        h[j] = best_v
    return h

def _de_optimize(h_parent, n, n_pop, n_gens, rng, time_limit):
    target_sum = n / 2.0
    pop = np.zeros((n_pop, n))
    if h_parent is not None:
        h_p = h_parent.copy()
        if len(h_p) != n:
            h_p = np.interp(np.linspace(0, n-1, n), np.arange(len(h_parent)), h_parent)
        h_p = np.clip(h_p, 0, 1)
        pop[0] = _fix_sum(h_p, target_sum)
        for i in range(1, n_pop):
            noise = rng.randn(n) * (0.1 + 0.05 * i / max(n_pop-1, 1))
            pop[i] = _fix_sum(h_p + noise, target_sum)
    else:
        for i in range(n_pop):
            pop[i] = _fix_sum(rng.rand(n), target_sum)
    fitness = np.array([_c5_of(pop[i]) for i in range(n_pop)])
    best_h = pop[int(np.argmin(fitness))].copy()
    best_c = float(np.min(fitness))
    F = 0.8
    stall = 0
    t_start = time.time()
    for gen in range(n_gens):
        if time.time() - t_start > time_limit:
            break
        order = np.argsort(fitness)
        for i in range(n_pop):
            if time.time() - t_start > time_limit:
                break
            a = order[0]
            b = order[rng.randint(1, n_pop)]
            c = order[rng.randint(1, n_pop)]
            while c == b:
                c = order[rng.randint(1, n_pop)]
            v = pop[a] + F * (pop[b] - pop[c])
            trial = np.where(rng.rand(n) < 0.7, v, pop[i])
            trial = _fix_sum(np.clip(trial, 0, 1), target_sum)
            ct = _c5_of(trial)
            if ct <= fitness[i]:
                pop[i] = trial
                fitness[i] = ct
                if ct < best_c:
                    best_c = ct
                    best_h = trial.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > n_pop * 2:
            F = max(0.4, F * 0.85)
            ri = int(rng.randint(n_pop))
            pop[ri] = _fix_sum(rng.rand(n), target_sum)
            fitness[ri] = _c5_of(pop[ri])
            stall = 0
    return best_h, best_c

def _restart_optimize(h_best, n, target_sum, rng, n_restarts, time_limit):
    best_h = h_best.copy()
    best_c = _c5_of(h_best)
    t_start = time.time()
    for r in range(n_restarts):
        if time.time() - t_start > time_limit:
            break
        noise_scale = 0.05 + 0.1 * (r / max(n_restarts - 1, 1))
        h_pert = _fix_sum(h_best + rng.randn(n) * noise_scale, target_sum)
        h_sm, c5_sm = _nesterov_smooth(h_pert, n, target_sum, n_iters=80,
                                        alpha_start=200, alpha_end=800, lr=0.2)
        h_sl = _slsqp_opt(h_sm, n, target_sum, 500, maxiter=150)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c:
            best_c = c5_sl
            best_h = h_sl.copy()
    return best_h, best_c
