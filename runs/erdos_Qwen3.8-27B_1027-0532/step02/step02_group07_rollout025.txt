  - If off >= 0: j in [0, n-off), term = (1-h[j+off]) = (1-h[off : n])*[(j-off)]

    So grad[-off:n] += w_k * (1-h[0:n+off])

- Second term: sum over j where j-off in [0,n), i.e., j in [off, n+off) intersect [0,n)
  - If off >= 0: j in [off, n), term = h[j-off] = h[0:n-off]*[(j-off)]
    So grad[off:n] -= w_k * h[0:n-off]
  - If off < 0: j in [0, n+off), term = h[j-off] = h[-off:n]*[(j-off)]
    So grad[0:n+off] -= w_k * h[-off:n]

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x


def optimize_coordinate(h, j, corr):
    n = len(h)
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h[j] * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h[j], slopes

    for _ in range(42):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h[j]
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-11, time_budget=60.0, reverse=False, relax=1.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    order = list(range(n-1, -1, -1)) if reverse else list(range(n))

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = (t_opt - h[j]) * relax
            if abs(change) > 1e-14:
                new_val = h[j] + change
                new_val = np.clip(new_val, 0.0, 1.0)
                actual_change = new_val - h[j]
                if abs(actual_change) > 1e-14:
                    h[j] = new_val
                    corr = corr + actual_change * slopes
                    max_change = max(max_change, abs(actual_change))
        if sweep % 25 == 24:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def gradient_step(h, alpha, step_size):
    n = len(h)
    corr = compute_correlations(h)
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    weights /= np.sum(weights)

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-8:
            continue
        off = k - (n - 1)
        w = weights[k]
        if off >= 0:
            length = n - off
            if length > 0:
                grad[:length] += w * (1.0 - h[off:n])
                grad[off:] -= w * h[:length]
        else:
            length = n + off
            if length > 0:
                grad[-off:] += w * (1.0 - h[:length])
                grad[:length] -= w * h[-off:]

    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    return h_new


def multiresolution_refine(h, n_target, time_budget=30.0):
    n_old = len(h)
    if n_target == n_old:
        return h
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    h_interp = project_sum(h_interp, n_target / 2.0)
    h_cd = coordinate_descent(h_interp, n_sweeps=200, time_budget=time_budget)
    return h_cd


def try_period1(h, time_budget=20.0):
    n = len(h)
    if n % 2 != 0:
        m = n // 2
        half = h[:m].copy()
    else:
        m = n // 2
        half = (h[:m] + h[m:2*m]) / 2.0

    target = n / 4.0
    for _ in range(500):
        s = np.sum(half)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        half += diff / m
        np.clip(half, 0.0, 1.0, out=half)

    h_sym = np.zeros(n)
    h_sym[:m] = half
    if n % 2 == 0:
        h_sym[m:n] = half
    else:
        h_sym[m:n] = np.resize(half, n - m)
    h_sym = project_sum(h_sym, n / 2.0)
    h_cd = coordinate_descent(h_sym, n_sweeps=150, time_budget=time_budget)
    return h_cd
