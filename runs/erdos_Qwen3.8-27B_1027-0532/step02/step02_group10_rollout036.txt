```python
def fw_linear_oracle(grad, n, target):
    """Find vertex of polytope {h: 0<=h<=1, sum(h)=target} minimizing grad·h"""
    # Sort by gradient value
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    # Assign 1 to the floor(target) smallest gradient values
    int_part = int(target)
    frac_part = target - int_part
    v[sorted_idx[:int_part]] = 1.0
    if frac_part > 1e-10:
        v[sorted_idx[int_part]] = frac_part
    return v
```

```python
    dx = 2.0 / n

        h_new = (1 - t) * h + t * v

    return best_h
```

```python
    ov = compute_overlap_vector(h)  # length n, ov[k] for k=0..n-1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_h = h[i]
            # Compute delta: how overlap_k changes per unit change in h[i]
            # overlap_k = sum_j h[j+k]*(1-h[j]) for j=0..n-1-k
            # d(overlap_k)/d(h[i]):
            #   (1-h[i-k]) if i-k is in [0, n-1-k] and i is in the "h[j+k]" position
            #   -h[i+k] if i is in the "h[j]" position (j=i, so i in [0, n-1-k])
            # More precisely:
            #   If i >= k: h[i] appears as h[j+k] with j=i-k, contributing (1-h[j]) = (1-h[i-k])
            #   If i <= n-1-k: h[i] appears as h[j] with j=i, contributing -h[j+k] = -h[i+k]
            
            # But we also need to account for the sum constraint!
            # If we change h[i] by dt, we need to compensate elsewhere.
            # For simplicity, let's project after each update.
            
            delta = np.zeros(n)
            # First part: i in [k, n-1] contributes (1-h[i-k]) to overlap_k
            for k in range(min(i, n-1) + 1):
                delta[k] += (1.0 - h[i-k])
            # Second part: i in [0, n-1-k] contributes -h[i+k] to overlap_k
            for k in range(min(n-1-i, n-1) + 1):
                delta[k] -= h[i+k]
            
            # Now: new_ov[k] = ov[k] + (t - old_h) * delta[k]
            # Minimize max_k new_ov[k] over t in [0,1]
            # Also need to maintain sum constraint; for now, project after
            
            # Find optimal t using golden section
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
            m1 = a + (1 - gr) * (b - a)
            m2 = a + gr * (b - a)
            f1 = np.max(ov + (m1 - old_h) * delta)
            f2 = np.max(ov + (m2 - old_h) * delta)
            for _ in range(40):
                if f1 < f2:
                    b = m2
                    m2 = m1
                    f2 = f1
                    m1 = a + (1 - gr) * (b - a)
                    f1 = np.max(ov + (m1 - old_h) * delta)
                else:
                    a = m1
                    m1 = m2
                    f1 = f2
                    m2 = a + gr * (b - a)
                    f2 = np.max(ov + (m2 - old_h) * delta)
            
            best_t = (a + b) / 2.0
            best_val = np.max(ov + (best_t - old_h) * delta)
            old_val = np.max(ov)  # current max overlap
            
            if best_val < old_val - 1e-14:
                dt = best_t - old_h
                h[i] = best_t
                ov += dt * delta
                improved = True
        
        if not improved:
            break
    
    h = project_onto_constraint(h, target)
    return h
```

Wait, there's a problem. The golden section search assumes f is unimodal, which it is (convex). But the issue is that after we update h[i], the overlap vector changes, and we need to update it. But the sum constraint might be violated.

For the sum constraint: if we change h[i] from old_h to best_t, the sum changes by (best_t - old_h). To maintain the sum, we'd need to adjust other coordinates too. This makes pure coordinate descent tricky.

**Alternative**: Use the constraint differently. Instead of hard constraint, use a penalty:
Objective: max_k overlap_k + λ * (sum(h) - target)²

Then coordinate descent works without the coupling issue.

Or: Use the "water-filling" projection after each coordinate update (which is what project_onto_constraint does).

Actually, the cleanest approach is:
1. Do coordinate descent on the unconstrained problem (just 0 ≤ h ≤ 1)
2. Project onto the sum constraint at the end (or periodically)

Since the sum constraint is a single linear constraint, the projection is well-defined and the objective change due to projection is small (it's a small perturbation).

Let me also add a "reset" of the overlap vector periodically to avoid numerical drift.

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m_idx in range(len(weights)):
        w = weights[m_idx]
        if w < 1e-14:
            continue
        offset = m_idx - (n - 1)
        k_abs = abs(offset)
        if k_abs >= n:
            continue
        grad[k_abs:n] += w * (1.0 - h[0:n - k_abs])
        grad[0:n - k_abs] -= w * h[k_abs:n]
    grad *= dx
    return grad


def fw_linear_oracle(grad, n, target):
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    int_part = int(target)
    frac_part = target - int_part
    if int_part > 0:
        v[sorted_idx[:int_part]] = 1.0
    if frac_part > 1e-10 and int_part < n:
        v[sorted_idx[int_part]] = frac_part
    return v


def frank_wolfe_optimize(h_init, alpha, target, max_iters, deadline):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_f = np.inf
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(ov, alpha)
        if sm < best_f - 1e-15:
            best_f = sm
            best_h = h.copy()
        grad = compute_gradient(h, weights, dx)
        v = fw_linear_oracle(grad, n, target)
        t = 2.0 / (it + 2)
        h = (1.0 - t) * h + t * v
        np.clip(h, 0.0, 1.0, out=h)
        h = project_onto_constraint(h, target)
    return best_h


def pgd_optimize(h_init, alpha, target, max_iters, deadline):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_f = np.inf
    step = 1.0 / (alpha * dx * 2.0 + 1.0)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(ov, alpha)
        if sm < best_f - 1e-15:
            best_f = sm
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        grad = compute_gradient(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        s = step
        improved = False
        for bt in range(40):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-13:
                break
        if not improved:
            step = s
            if step < 1e-14:
                break
        else:
            step = min(s * 1.3, 1.0)
    return best_h


def coord_descent_sweep(h_init, target, max_sweeps, deadline):
    n = len(h_init)
    h = h_init.copy()
    ov = compute_overlaps(h)[n - 1:]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_h = h[i]
            delta = np.zeros(n)
            max_k1 = min(i, n - 1)
            for k in range(max_k1 + 1):
                delta[k] += (1.0 - h[i - k])
            max_k2 = min(n - 1 - i, n - 1)
            for k in range(max_k2 + 1):
                delta[k] -= h[i + k]
            a_lo, a_hi = 0.0, 1.0
            m1 = a_lo + (1.0 - gr) * (a_hi - a_lo)
            m2 = a_lo + gr * (a_hi - a_lo)
            f1 = np.max(ov + (m1 - old_h) * delta)
            f2 = np.max(ov + (m2 - old_h) * delta)
            for _ in range(35):
                if f1 < f2:
                    a_hi = m2
                    m2 = m1
                    f2 = f1
                    m1 = a_lo + (1.0 - gr) * (a_hi - a_lo)
                    f1 = np.max(ov + (m1 - old_h) * delta)
                else:
                    a_lo = m1
                    m1 = m2
                    f1 = f2
                    m2 = a_lo + gr * (a_hi - a_lo)
                    f2 = np.max(ov + (m2 - old_h) * delta)
            best_t = (a_lo + a_hi) / 2.0
            best_val = np.max(ov + (best_t - old_h) * delta)
            cur_val = np.max(ov)
            if best_val < cur_val - 1e-14:
                dt = best_t - old_h
                h[i] = best_t
                ov += dt * delta
                improved = True
        if not improved:
            break
    h = project_onto_constraint(h, target)
    return h


def targeted_boost(h, target, deadline):
    n = len(h)
    dx = 2.0 / n
    ov_full = compute_overlaps(h)
    top_k = np.argsort(-ov_full)[:5]
    for k_idx in top_k:
        if time.time() > deadline:
            break
        offset = k_idx - (n - 1)
        k_abs = abs(offset)
        if k_abs == 0:
            continue
        pairs = np.arange(n - k_abs)
        contrib = h[k_abs:n] * (1.0 - h[0:n - k_abs])
        top_pairs = np.argsort(-contrib)[:max(5, n // 10)]
        for p in top_pairs:
            if h[p + k_abs] > 0.5:
                h[p + k_abs] = np.clip(h[p + k_abs] - 0.02, 0, 1)
            if h[p] < 0.5:
                h[p] = np.clip(h[p] + 0.02, 0, 1)
    h = project_onto_constraint(h, target)
    return h
```
