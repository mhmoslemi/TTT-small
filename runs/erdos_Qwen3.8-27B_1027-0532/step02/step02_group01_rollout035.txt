import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_binary(h, target):
    """Project h to binary with exactly target ones."""
    h_bin = (h >= 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        # Flip least confident ones
        probs = h[ones]  # higher means more confident as 1
        to_flip = ones[np.argsort(probs)[:excess]]
        h_bin[to_flip] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        probs = 1.0 - h[zeros]  # higher means more confident as 0, so less confident as 1
        to_flip = zeros[np.argsort(probs)[:deficit]]
        h_bin[to_flip] = 1
    return h_bin

def make_binary(n, target, rng):
    """Create random binary vector with exactly target ones."""
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def sa_single_swap(h, rng):
    """Single swap: exchange a random 1 and a random 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    i = ones[rng.integers(len(ones))]
    j = zeros[rng.integers(len(zeros))]
    h[i], h[j] = h[j], h[i]
    return i, j

def sa_double_swap(h, rng):
    """Double swap: exchange two pairs."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    i1 = ones[rng.integers(len(ones))]
    i2 = ones[rng.integers(len(ones))]
    j1 = zeros[rng.integers(len(zeros))]
    j2 = zeros[rng.integers(len(zeros))]
    if i1 == i2 or j1 == j2:
        i2 = ones[rng.integers(len(ones))]
        if i1 == i2:
            return sa_single_swap(h, rng)
    h[i1], h[j1] = h[j1], h[i1]
    h[i2], h[j2] = h[j2], h[i2]
    return i1, i2, j1, j2

def sa_block_swap(h, rng):
    """Swap two blocks of the same length."""
    n = len(h)
    max_len = max(2, n // 8)
    blen = rng.integers(2, max_len + 1)
    a = rng.integers(0, n - blen + 1)
    b = rng.integers(0, n - blen + 1)
    if b == a:
        b = (b + blen) % (n - blen + 1)
    if abs(a - b) < blen:
        # Overlapping, skip or handle
        return None
    h[a:a+blen], h[b:b+blen] = h[b:b+blen].copy(), h[a:a+blen].copy()
    return (a, b, blen)

def sa_block_reverse(h, rng):
    """Reverse a random block."""
    n = len(h)
    max_len = max(3, n // 6)
    blen = rng.integers(3, max_len + 1)
    a = rng.integers(0, n - blen + 1)
    h[a:a+blen] = h[a:a+blen][::-1]
    return (a, blen)

def sa_move(h, move_type, rng):
    """Apply a random move and return undo info."""
    if move_type == 0:
        return sa_single_swap(h, rng)
    elif move_type == 1:
        return sa_double_swap(h, rng)
    elif move_type == 2:
        return sa_block_swap(h, rng)
    elif move_type == 3:
        return sa_block_reverse(h, rng)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(n, target, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def project_onto_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def _c_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def _c_jac(h):
    return np.ones(len(h))

def slsqp_refine(h_init, eps_list, maxiter, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_box_sum(h_init.astype(float), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    for eps in eps_list:
        if time.time() - start_time > time_limit:
            break
        try:
            result = minimize(smooth_objective, h, args=(eps,),
                            jac=smooth_gradient, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': maxiter, 'ftol': 1e-14})
            h = project_onto_box_sum(result.x, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c5

def apply_informed_move(h, rng):
    n = len(h)
    corr = compute_corr(h)
    k_star = int(np.argmax(corr)) - (n - 1)
    if k_star >= 0:
        cands = []
        for i in range(n - k_star):
            if h[i + k_star] > 0.99 and h[i] < 0.01:

    idx = int(rng.integers(len(cands)))

    h[j] = 0.0
    return True

def apply_random_swap(h, rng):
    ones = np.where(h > 0.99)[0]
