Wait, there's a subtlety. When I do `corr = corr + delta * g`, I'm updating the correction for the change in h[i]. But this assumes that the correlation changes linearly with h[i], which is true! Because corr[k] = sum_j h[j+k](1-h[j]) is linear in each h[i] when the others are fixed.

Let me verify: if h[i] changes by δ, then:
- In corr[k] = sum_j h[j+k](1-h[j]):
  - The term j+k=i (j=i-k) changes: h[i](1-h[i-k]) → (h[i]+δ)(1-h[i-k]), change = δ(1-h[i-k])
  - The term j=i changes: h[i+k](1-h[i]) → h[i+k](1-h[i]-δ), change = -δ*h[i+k]
  - Total change = δ(1-h[i-k]) - δ*h[i+k] = δ * g_k

Yes! So the incremental update is exact.

Now, after projection, the sum constraint is satisfied but all h values may have changed (due to the clip), so we need to recompute corr from scratch. That's what `corr = fast_correlate(h)` does.

Let me think about efficiency. Per sweep:
- n coordinate updates, each O(n) for compute_g and O(n) for optimize_1d (with 60 iterations of O(n) each = O(60n))
  Actually, optimize_1d does 60 iterations, each computing `corr + mid * g` (O(n)) and `np.max` (O(n)) and the active mask (O(n)). So it's O(60n) per coordinate.
- Total per sweep: n * (n + 60n) = n * 61n ≈ 61n²
- For n=425: 61 * 425² ≈ 11M operations per sweep
- Plus the FFT recompute after projection: O(N log N) ≈ 10K

So each sweep takes maybe 0.01-0.05 seconds. We can do hundreds of sweeps easily within the time budget.

Let me also think about whether the coordinate descent will converge to a better solution than L-BFGS. The key advantage is:
1. It directly optimizes the true objective (max of correlations), not a smooth approximation
2. Each step is exact (for the 1D subproblem)
3. It's very fast, allowing many iterations

The disadvantage:
1. Coordinate descent can be slow to converge (zig-zagging)
2. It might get stuck in different local minima

I think combining both approaches (coordinate descent + L-BFGS + coordinate descent) will give the best results.

Let me also think about the "pair" coordinate descent. For pairs (i, j), the 2D subproblem is:
min_{δ_i, δ_j} max_k (c_k + δ_i * g_k^(i) + δ_j * g_k^(j))

This can be solved by:
- For each candidate δ_j (on a grid), find the optimal δ_i (1D convex)
- The resulting function of δ_j is convex
- Do bisection on δ_j

This is O(n * grid_size * n) per pair = O(n² * grid_size). With n=849 and grid_size=50, that's about 36M per pair. If we try 100 random pairs per sweep, that's 3.6B... too slow.

```python
import numpy as np
from scipy.optimize import minimize
import time


def fast_correlate(h):
    n = len(h)
    g = 1.0 - h
    N = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([corr[N - (n - 1):], corr[:n - 1]])


def compute_g(i, h, n):
    lags = np.arange(-(n - 1), n)
    j1 = i - lags
    mask1 = (j1 >= 0) & (j1 < n)
    j2 = i + lags
    mask2 = (j2 >= 0) & (j2 < n)
    g = np.zeros(2 * n - 1)
    g[mask1] += 1.0 - h[j1[mask1]]
    g[mask2] -= h[j2[mask2]]
    return g


def optimize_1d(corr, g, lo, hi, n_iter=80):
    if lo >= hi - 1e-15:
        return 0.0
    for _ in range(n_iter):
        mid = (lo + hi) * 0.5
        val = corr + mid * g
        max_val = np.max(val)
        active = val >= max_val - 1e-14
        g_active = g[active]
        if g_active.max() < -1e-14:
            lo = mid
        elif g_active.min() > 1e-14:
            hi = mid
        else:
            break
    return (lo + hi) * 0.5


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_correlate(h)
    return np.max(corr) * dx


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = fast_correlate(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += w
            grad[:N - lag] -= w * h[lag:]
            grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if 0 < l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad


def ljbfgs_objective(h, alpha, n_pts, mu):
    corr = fast_correlate(h)
    sm = smooth_max_value(corr, alpha)
    s = np.sum(h) - n_pts / 2.0
    return sm + mu * s * s


def ljbfgs_gradient(h, alpha, n_pts, mu):
    g = smooth_max_gradient(h, alpha)
    s = np.sum(h) - n_pts / 2.0
    g += 2.0 * mu * s
    return g


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(ljbfgs_objective, h0, jac=ljbfgs_gradient, method='L-BFGS-B',
                   bounds=bounds, args=(alpha, n_pts, mu),
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def coordinate_descent(h, n, max_sweeps, time_limit, rng):
    t0 = time.time()
    corr = fast_correlate(h)
    best_max = np.max(corr)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        # Random order for this sweep
        order = rng.permutation(n)
        for i in order:
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-12:
                continue
            g = compute_g(i, h, n)
            delta = optimize_1d(corr, g, lo, hi)
            if abs(delta) > 1e-14:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                corr = corr + delta * g
        # Project
        h = project_feasible(h, n / 2.0)
        corr = fast_correlate(h)
        current_max = np.max(corr)
        if current_max < best_max:
            best_max = current_max
    
    return h


def run(seed=42, budget_s=1300, **kwargs):

    init = np.array(initial_h_values, dtype=float)

        if s < best_score:
            best_score = s
            best_h = h_hires.copy()
    
    # Final
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    
    return (best_h, best_score, len(best_h))
```
