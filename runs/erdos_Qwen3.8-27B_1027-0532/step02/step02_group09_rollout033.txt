from scipy.optimize import minimize

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def to_binary_smart(h, rng):
    """Convert continuous h to binary with exactly n//2 ones."""
    n = len(h)
    n_target = n // 2
    # Use the values themselves as "confidence scores"
    # Sort by h value, take top n_target
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[n_target:]] = 1.0  # top n_target values become 1
    return h_bin

def lse_value(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_gradient(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    # Threshold to avoid computing gradients for negligible weights
    threshold = 1e-8
    grad = np.zeros(n)
    sig_idx = np.where(weights > threshold)[0]
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad

def subgradient_momentum(h0, n_iters, alpha, lr_init, momentum, target):
    """Subgradient descent with momentum for the LSE approximation."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    
    for it in range(n_iters):
        if it % 1000 == 0:
            lr = lr_init / (1.0 + 0.01 * float(it))
        else:
            lr = lr_init / (1.0 + 0.01 * float(it))
        
        grad = lse_gradient(h, alpha)
        v = momentum * v - lr * grad
        h_new = project_box_sum(h + v, target)
        
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_raw

def binary_sa_parallel(h_init, n_iters_total, rng, n_replicas=5):
    """Parallel tempering simulated annealing for binary optimization."""
    n = len(h_init)
    n_target = n // 2
    
    # Initialize replicas
    replicas = [h_init.copy() for _ in range(n_replicas)]
    # Vary the temperatures
    temps = np.array([0.05, 0.03, 0.02, 0.01, 0.005])
    
    best_overall = h_init.copy()
    best_score = evaluate_score(h_init)
    
    # Precompute initial scores
    scores = [evaluate_score(r) for r in replicas]
    
    for it in range(n_iters_total):
        # Every 500 iterations, try to swap adjacent replicas
        if it % 500 == 499:
            for r in range(n_replicas - 1):
                # Metropolis swap
                delta = (scores[r] - scores[r+1]) * (1.0/temps[r+1] - 1.0/temps[r])
                if delta < 0 or rng.random() < np.exp(-delta):
                    replicas[r], replicas[r+1] = replicas[r+1].copy(), replicas[r].copy()
                    scores[r], scores[r+1] = scores[r+1], scores[r]
        
        # For each replica, do a move
        for r in range(n_replicas):
            h = replicas[r]
            # Pick a random 1 and a random 0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Make the move
            h[i] = 0.0
            h[j] = 1.0
            new_score = evaluate_score(h)
            delta = new_score - scores[r]
            
            # Metropolis criterion
            if delta < 0 or rng.random() < np.exp(-delta / temps[r]):
                scores[r] = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_overall = h.copy()
            else:
                # Revert
                h[i] = 1.0
                h[j] = 0.0
        
        # Slowly cool
        if it % 5000 == 4999 and it > 0:
            temps *= 0.95
    
    return best_overall, best_score

def binary_local_search(h_bin, max_rounds=3):
    """Try all single swaps to see if any improves."""
    n = len(h_bin)
    best_h = h_bin.copy()
    best_score = evaluate_score(h_bin)
    
    for round in range(max_rounds):
        improved = False
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        
        for i in ones:
            for j in zeros:
                best_h[i] = 0.0
                best_h[j] = 1.0

                    improved = True
                    # Don't revert, continue from here

                    zeros = np.where(best_h == 0.0)[0]
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_score

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):

def project_box_sum(h, target):
    n = len(h)
    lo = -2.0
    hi = 2.0

        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))

        else:

def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-8
    grad = np.zeros(n)
    sig_idx = np.where(weights > threshold)[0]
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad
