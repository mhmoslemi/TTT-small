    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    S = np.arange(-n + 1, n)
    
    js = J[:, None] + S[None, :]  # (n, 2n-1)
    jsm = J[:, None] - S[None, :]  # (n, 2n-1)
    
    valid_p = (js >= 0) & (js < n)  # (n, 2n-1)
    valid_m = (jsm >= 0) & (jsm < n)  # (n, 2n-1)
    
    # For valid_p entries: contribution is (1-h[js])
    # For valid_m entries: contribution is -h[jsm]
    
    h_pad = np.zeros(2 * n - 1)
    h_pad[n-1:2*n-1] = h  # center h
    
    # js + (n-1) gives the index into h_pad
    js_pad = js + (n - 1)  # (n, 2n-1), values in [0, 2n-2]
    jsm_pad = jsm + (n - 1)  # (n, 2n-1), values in [0, 2n-2]
    
    h_js = h_pad[js_pad]  # (n, 2n-1)
    h_jsm = h_pad[jsm_pad]  # (n, 2n-1)
    
    # term1[j] = sum_s p[s] * (1 - h[j+s]) * valid_p[j,s]
    term1 = np.sum(p[None, :] * valid_p * (1.0 - h_js), axis=1)
    # term2[j] = sum_s p[s] * h[j-s] * valid_m[j,s]
    term2 = np.sum(p[None, :] * valid_m * h_jsm, axis=1)
    
    return dx * (term1 - term2)

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))
