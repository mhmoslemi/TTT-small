4. **Coordinate descent**:

   - For each coordinate j, the objective is a piecewise quadratic function (since each overlap is bilinear)

Let me implement the coordinate descent more carefully. Fix all h[i] for i ≠ j. The overlap at shift idx is:
overlap_idx = sum_i h[i](1-h[i+idx]) (with appropriate indexing)

The terms involving h[j] are:
- If j is in range as first factor: h[j] * (1-h[j+shift]) for shifts where j is the first index
- If j is in range as second factor: h[j-shift] * (1-h[j]) for shifts where j is the second index

So for a fixed j, the overlap at shift k depends on h[j] linearly (not quadratically!) because h[j] appears at most once in each term.

Wait, that's right! In the overlap sum, h[j] appears at most once per term. So for a fixed j, each overlap is LINEAR in h[j]. Therefore the max of overlaps is a convex piecewise-linear function of h[j].

This means we can find the optimal h[j] by:
1. Evaluating the max-overlap function at h[j] = 0 and h[j] = 1
2. Finding where the piecewise-linear function achieves its minimum
3. The minimum is at a breakpoint (where two or more overlaps are tied for the maximum) or at a boundary (0 or 1)

This is a 1D optimization that can be solved exactly!

Let me formalize: For fixed j, let f_j(a) = max_k overlap_k(h with h[j]=a) for a in [0,1].

Each overlap_k is linear in a: overlap_k = c_k + d_k * a (where c_k and d_k depend on j and the other h values).

So f_j(a) = max_k (c_k + d_k * a), which is the upper envelope of lines. The minimum of this piecewise-linear convex function over [0,1] is at:
- a = 0
- a = 1
- A point where two lines cross (a = (c_k - c_m) / (d_m - d_k) for some k, m)

We can find all breakpoints and evaluate f_j at each to find the minimum.

The number of breakpoints is O(number of shifts) = O(n). So coordinate descent is O(n^2) per sweep, and we do n sweeps, giving O(n^3) total. For n=425, that's about 76 million operations, which is feasible.

Actually, we don't need all n sweeps. We can do a few passes and it should converge.

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_val_and_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = m + np.log(wsum) / alpha
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n-L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    return val, grad

def lse_obj(h, alpha):
    return lse_val_and_grad(h, alpha)[0]

    return lse_val_and_grad(h, alpha)[1]

    val += penalty * s * s

def solve_lp_trust(h0, top_k=None, solver=None):
    """Solve LP: min t s.t. linearized overlaps <= t, sum h = n/2, 0<=h<=1"""
    n = len(h0)
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0, t >= 0]
    
    overlaps = compute_overlaps(h0)
    total_shifts = len(overlaps)
    
    if top_k is not None and top_k < total_shifts:
        # Use only top-k shifts
        top_idx = np.argsort(-overlaps)[:top_k]
    else:
        top_idx = range(total_shifts)
    
    for idx in top_idx:
        g = np.zeros(n)
        L = idx - (n - 1)
        if L >= 0:
            g[:n-L] = (1.0 - h0[L:])
            g[L:] -= h0[:n-L]
        else:
            M = -L
            g[M:] = (1.0 - h0[:n-M])
            g[:n-M] -= h0[M:]
        
        lin_overlap = overlaps[idx] + g @ h - float(g @ h0)
        constraints.append(lin_overlap <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers = [solver] if solver else [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PYPDPP]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if h.value is not None and np.isfinite(t.value):
                return np.clip(h.value, 0, 1), t.value
        except:
            continue
    return None, None

def coordinate_descent(h, n_passes=5):
    """Exact coordinate descent for the max-overlap problem."""
    n = len(h)
    h = h.copy()
    for _pass in range(n_passes):
        improved = False
        for j in range(n):
            old_val = h[j]
            h[j] = 0.0
            overlaps_0 = compute_overlaps(h)
            h[j] = 1.0
            overlaps_1 = compute_overlaps(h)

            d = overlaps_1 - overlaps_0
            
            # Find min of max_k (c[k] + d[k]*a) over a in [0,1]

            f0 = np.max(c)
            f1 = np.max(overlaps_1)
            best_a = 0.0 if f0 <= f1 else 1.0
            best_f = min(f0, f1)
            
            # Check breakpoints: for the active constraint at best_a
            active_k = np.argmax(c + d * best_a)
            for m in range(len(c)):
                if m == active_k:

                denom = d[active_k] - d[m]
                if abs(denom) < 1e-12:

                a_bp = (c[m] - c[active_k]) / denom

                improved = True
        
        # Project to fix sum
        h = project_box_sum(h, n / 2.0)
        if not improved:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val_and_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = m + np.log(wsum) / alpha
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return val, grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    val, grad = lse_val_and_grad(h, alpha)
    s = float(np.sum(h)) - n / 2.0
    val += penalty * s * s
    grad = grad + 2.0 * penalty * s
    return val, grad


def build_lp_constrains(h0, overlaps, top_k):
    n = len(h0)
    total = len(overlaps)
    if top_k >= total:
        idxs = list(range(total))
    else:
        idxs = list(np.argsort(-overlaps)[:top_k])
    return idxs


def solve_lp(h0, top_k=None):
    n = len(h0)
    overlaps = compute_overlaps(h0)
    total = len(overlaps)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0, t_var >= 0]
    
    if top_k is None or top_k >= total:
        idxs = range(total)
    else:
        idxs = np.argsort(-overlaps)[:top_k]
    
    for idx in idxs:
        g = np.zeros(n)
        L = idx - (n - 1)
        if L >= 0:
            g[:n - L] = (1.0 - h0[L:])
            g[L:] -= h0[:n - L]
        else:
            M = -L
            g[M:] = (1.0 - h0[:n - M])
            g[:n - M] -= h0[M:]
        
        base = float(overlaps[idx] - g @ h0)
        constraints.append(base + g @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.PDEL1]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None and np.all(np.isfinite(h_var.value)):
                return np.clip(h_var.value, 0, 1)
        except Exception:
            continue
    return None

def lse_lbfgsb(h0, alpha, penalty, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18}
        )
        if res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def coordinate_descent(h, n_passes=3):
    n = len(h)
    h = h.copy()
    for _ in range(n_passes):
        any_change = False
        for j in range(n):
            h[j] = 0.0
            ov0 = compute_overlaps(h)
            h[j] = 1.0
            ov1 = compute_overlaps(h)
            c = ov0
            d = ov1 - ov0
            
            f0 = float(np.max(c))
            f1 = float(np.max(ov1))
            best_a = 0.0 if f0 <= f1 else 1.0
            best_f = min(f0, f1)
            
            # Check breakpoints
            ak = int(np.argmax(c + d * best_a))
            for m in range(len(c)):
                if m == ak:
                    continue
                denom = float(d[ak] - d[m])
                if abs(denom) < 1e-12:
                    continue
                a_bp = float((c[m] - c[ak]) / denom)
                if 0.0 <= a_bp <= 1.0:
                    f_bp = float(np.max(c + d * a_bp))
                    if f_bp < best_f - 1e-12:
                        best_f = f_bp
                        best_a = a_bp
                        ak = int(np.argmax(c + d * a_bp))
            
            new_val = float(np.clip(best_a, 0.0, 1.0))
            if abs(new_val - h[j]) > 1e-10:
                h[j] = new_val
                any_change = True
        
        h = project_box_sum(h, n / 2.0)
        if not any_change:
            break
    return h
