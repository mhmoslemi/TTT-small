    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    
    # Compute weights
    shifted = alpha * (corr - np.max(corr))
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # grad[i] = sum_k weights[k] * d(corr[k])/d(h[i])
    # d(corr[k])/d(h[i]) = dx * (1 - h[i+s_k]) - dx * h[i-s_k]
    # where s_k = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        # Contribution from (1 - h[i+s])
        # i+s in [0, n) => i in [-s, n-s)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:

        # Contribution from -h[i-s]
        # i-s in [0, n) => i in [s, n+s)
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - s : i_hi2 - s]
    
    return grad

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def project_polytope(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        interior = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            proj[interior] += diff / np.sum(interior)
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def smooth_obj(h, alpha):
    corr = compute_correlations(h) * (2.0 / len(h))
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def smooth_obj_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h) * dx
    shifted = alpha * (corr - np.max(corr))
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + s:i_hi + s])
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - s:i_hi2 - s]
    return grad


def slsqp_optimize(h0, alpha, maxiter, bounds):
    def obj(h):
        return smooth_obj(h, alpha)
    def cons(h):
        return np.sum(h) - len(h) / 2.0
    def cons_jac(h):
        return np.ones(len(h))
    res = minimize(
        obj, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons, 'jac': cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    zero_mask = (np.abs(slopes) < 1e-14)

    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))

            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0, shuffle=True):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    rng = np.random.RandomState(12345)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        if shuffle:
            order = rng.permutation(n)
        else:
            order = range(n)
        max_change = 0.0
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h

        bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        if res.status != 0:
            step_size *= 0.5
            if step_size < 0.01:
                break
            continue
        h_lp = res.x[:n]
        for alpha_step in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            h_try = h + alpha_step * (h_lp - h)
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_sum(h_try, n / 2.0)
            s_try = evaluate_score(h_try)
            if s_try < best_score - 1e-12:
                best_score = s_try
                best_h = h_try.copy()
                h = h_try.copy()
                step_size = min(1.0, step_size * 1.1)
                break
        else:
            step_size *= 0.5
            if step_size < 0.01:
                break
    return best_h, best_score


def adam_optimize(h0, alpha, n_iters, lr, time_budget):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t0 = time.time()
    best_h = h.copy()
    best_score = evaluate_score(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for t in range(1, n_iters + 1):
        if time.time() - t0 > time_budget:
            break
        grad = smooth_obj_gradient(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_polytope(h_new, n / 2.0)
        s_new = evaluate_score(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def multi_scale_optimize(h0, target_n, time_budget):
    n_old = len(h0)
    if target_n == n_old:
        return h0, evaluate_score(h0)
    if target_n < n_old:
        x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
        h_down = np.zeros(target_n)
        for i in range(target_n):
            i_start = int(i * n_old / target_n)
            i_end = int((i + 1) * n_old / target_n)
            if i_end > i_start:
                h_down[i] = np.mean(h0[i_start:i_end])
        h_down = project_sum(h_down, target_n / 2.0)
        tb1 = time_budget * 0.4
        h_cd = coordinate_descent(h_down, n_sweeps=200, time_budget=tb1)
        tb2 = time_budget * 0.3
        h_lin, s_lin = linearization_optimize(h_cd, tb2, max_iters=200)
        best_h = h_lin if s_lin < evaluate_score(h_cd) else h_cd
        tb3 = time_budget * 0.25
        h_cd2 = coordinate_descent(best_h, n_sweeps=100, time_budget=tb3)
    else:
        x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, h0)
        h_up = project_sum(h_up, target_n / 2.0)
        tb1 = time_budget * 0.4
        h_cd = coordinate_descent(h_up, n_sweeps=200, time_budget=tb1)
        tb2 = time_budget * 0.3
        h_lin, s_lin = linearization_optimize(h_cd, tb2, max_iters=200)
        best_h = h_lin if s_lin < evaluate_score(h_cd) else h_cd
        tb3 = time_budget * 0.25
        h_cd2 = coordinate_descent(best_h, n_sweeps=100, time_budget=tb3)
    return h_cd2, evaluate_score(h_cd2)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    else:
        n_init = 200
        init = np.full(n_init, 0.5)
        init[:n_init // 2] = 1.0
        rng.shuffle(init)
        init = project_sum(init, n_init / 2.0)

    n = len(init)
    best_h = init.copy()
    best_score = evaluate_score(best_h)
    if best_score > 0.4:
        best_score = float('inf')

    # Phase 1: Initial CD
    if time_left() > 100:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(80, time_left() - 40), shuffle=True)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: Linearization
    if time_left() > 120:
        h_lin, s_lin = linearization_optimize(best_h, time_budget=min(200, time_left() - 40), max_iters=400)
        if s_lin < best_score:
            best_score = s_lin
            best_h = h_lin
        if time_left() > 60:
            h_cd = coordinate_descent(best_h, n_sweeps=300, time_budget=min(40, time_left() - 20), shuffle=True)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 3: Adam optimization
    if time_left() > 150:
        for alpha_val in [500, 2000, 8000]:
            if time_left() < 60:
                break
            h_adam, s_adam = adam_optimize(best_h, alpha_val, n_iters=3000, lr=0.5,
                                           time_budget=min(50, time_left() - 30))
            if s_adam < best_score:
                best_score = s_adam
                best_h = h_adam
        if time_left() > 60:
            h_cd = coordinate_descent(best_h, n_sweeps=300, time_budget=min(40, time_left() - 20), shuffle=True)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 4: More linearization
    if time_left() > 100:
        h_lin, s_lin = linearization_optimize(best_h, time_budget=min(150, time_left() - 30), max_iters=300)
        if s_lin < best_score:
            best_score = s_lin
            best_h = h_lin
        if time_left() > 40:
            h_cd = coordinate_descent(best_h, n_sweeps=200, time_budget=min(30, time_left() - 15), shuffle=True)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 5: Multi-scale
    n_cur = len(best_h)
    if time_left() > 200:
        for target_n in [min(1000, 2 * n_cur), max(150, n_cur // 2)]:
            if time_left() < 80:
                break
            h_ms, s_ms = multi_scale_optimize(best_h, target_n, time_budget=min(120, time_left() - 30))
            if target_n != n_cur:
                x_old = np.linspace(0.0, 2.0, len(h_ms), endpoint=False)
                x_new = np.linspace(0.0, 2.0, n_cur, endpoint=False)
                h_back = np.interp(x_new, x_old, h_ms)
                h_back = project_sum(h_back, n_cur / 2.0)
                h_cd = coordinate_descent(h_back, n_sweeps=200, time_budget=min(30, time_left() - 15), shuffle=True)
                s = evaluate_score(h_cd)
                if s < best_score:
                    best_score = s
                    best_h = h_cd
            else:
                if s_ms < best_score:
                    best_score = s_ms
                    best_h = h_ms

    # Phase 6: Random and structured perturbations
    n_cur = len(best_h)
    n_perturb = 30
    for i in range(n_perturb):
        if time_left() < 50:
            break
        frac = i / max(n_perturb - 1, 1)
        scale = 0.02 + 0.15 * frac
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_sum(h_pert, n_cur / 2.0)
        tb = min(5.0, (time_left() - 30) / max(n_perturb - i, 1))
        h_cd = coordinate_descent(h_pert, n_sweeps=150, time_budget=tb, shuffle=True)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    # Phase 7: Sinusoidal perturbations
    for freq in [3, 5, 7, 11, 13]:
        if time_left() < 40:
            break
        x = np.linspace(0, 2, n_cur, endpoint=False)
        for amp in [0.03, 0.08, 0.15]:
            if time_left() < 20:
                break
            h_pert = best_h + amp * np.sin(2 * np.pi * freq * x / 2.0)
            h_pert = project_sum(h_pert, n_cur / 2.0)
            h_cd = coordinate_descent(h_pert, n_sweeps=100, time_budget=min(8, time_left() - 10), shuffle=True)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 8: High resolution attempt
    n_cur = len(best_h)
    if time_left() > 150 and n_cur < 850:
        n_hi = min(850, 2 * n_cur)
        x_old = np.linspace(0.0, 2.0, n_cur, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_sum(h_hi, n_hi / 2.0)
        h_cd = coordinate_descent(h_hi, n_sweeps=300, time_budget=min(50, time_left() - 60), shuffle=True)
        h_lin, s_lin = linearization_optimize(h_cd, time_budget=min(40, time_left() - 30), max_iters=150)
        if s_lin < evaluate_score(h_cd):
            h_cd = h_lin
        if time_left() > 30:
            h_cd = coordinate_descent(h_cd, n_sweeps=200, time_budget=min(25, time_left() - 10), shuffle=True)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 9: Final SLSQP polish
    if time_left() > 30:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [2000, 10000, 50000]:
            if time_left() < 15:
                break
            maxiter = max(50, 200000 // n_final)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 10: Final CD polish
    if time_left() > 20:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(40, time_left() - 5), shuffle=True)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Final verification
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
