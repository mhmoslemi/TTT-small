def simulated_annealing(h_start, n, T_init, T_final, n_iters, time_limit_fn):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    T = T_init
    
    for i in range(n_iters):
        if time_limit_fn():
            break
        # Update temperature (geometric cooling)
        T = T_init * (T_final / T_init) ** (i / max(n_iters - 1, 1))
        
        # Generate perturbation
        # Mix of small and large perturbations
        if np.random.rand() < 0.3:
            # Large perturbation: flip a random block
            block_len = np.random.randint(3, 15)
            start = np.random.randint(0, n - block_len)
            h_try = h.copy()
            h_try[start:start+block_len] = np.random.rand(block_len)
        else:
            # Small perturbation: Gaussian noise
            noise_scale = 0.01 + 0.05 * T / T_init
            h_try = h + noise_scale * np.random.randn(n)
        
        h_try = project_feasible(h_try, n)
        c5_try = evaluate_c5(h_try)
        
        # Metropolis acceptance
        if c5_try < current_c5 or np.random.rand() < np.exp(-(c5_try - current_c5) / max(T, 1e-10)):
            h = h_try
            current_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - x[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-x[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def slsqp_refine(h_start, n, bounds, eps, maxiter):
    res = minimize(smooth_max_val, h_start.copy(), args=(eps, n),
                   jac=smooth_max_grad, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def de_obj(x, n):
    return evaluate_c5(project_feasible(x, n))

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def simulated_annealing(h_start, n, T_init, T_final, n_iters, time_limit_fn):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    for i in range(n_iters):
        if time_limit_fn():
            break
        T = T_init * (T_final / T_init) ** (float(i) / max(n_iters - 1, 1))
        if np.random.rand() < 0.3:
            block_len = np.random.randint(3, max(4, n // 30))
            start = np.random.randint(0, max(1, n - block_len))
            h_try = h.copy()
            h_try[start:start + block_len] = np.random.rand(block_len)
        else:
            noise_scale = 0.005 + 0.02 * T / T_init
            h_try = h + noise_scale * np.random.randn(n)
        h_try = project_feasible(h_try, n)
        c5_try = evaluate_c5(h_try)
        if c5_try < current_c5 or np.random.rand() < np.exp(-(c5_try - current_c5) / max(T, 1e-10)):
            h = h_try
            current_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
    return best_h, best_c5

def coord_descent(h_start, n, deltas, max_sweeps, time_limit_fn):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time_limit_fn():
            break
        improved = False
        for i in range(n):
            for d in deltas:
                h_try = h.copy()
                old_val = h_try[i]
                h_try[i] = old_val + d
                if h_try[i] < 0.0 or h_try[i] > 1.0:
                    continue
                # Adjust one other element to maintain sum constraint
                # Actually, let's just project
                h_try = project_feasible(h_try, n)
                c5 = evaluate_c5(h_try)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    h = h_try
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return h, best_c5
