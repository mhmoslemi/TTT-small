def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

                h[j] = old_val + best_delta
                corr = corr + best_delta * coeff
                n_improved += 1
        sum_val = h.sum()
        lam += eta * (sum_val - target_sum)
        lam = np.clip(lam, -5.0, 5.0)
        if n_improved == 0:
            break
    return _fix_sum(h, target_sum)

def _pairwise_local(h0, n, target_sum, max_iter=800, rng=None):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = np.max(corr)
    if rng is None:
        rng = np.random.RandomState(0)
    for it in range(max_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        coeff_i = _compute_coeff(h, i, n)
        coeff_j = _compute_coeff(h, j, n)
        diff_coeff = coeff_j - coeff_i
        lo_d = -h[j]
        hi_d = min(h[i], 1.0 - h[j])
        if hi_d - lo_d < 1e-12:
            continue
        best_delta = _golden_section(corr, diff_coeff, lo_d, hi_d, 35)
        if abs(best_delta) > 1e-10:
            new_max = np.max(corr + best_delta * diff_coeff)
            if new_max < current_c5 - 1e-12:
                h[i] -= best_delta
                h[j] += best_delta
                corr = corr + best_delta * diff_coeff
                current_c5 = new_max
    return h

def _coord_descent_simple(h0, n, target_sum, max_rounds=50, gs_iters=45):
    h = h0.copy()
    for rnd in range(max_rounds):
        n_improved = 0
        for j in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            coeff = _compute_coeff(h, j, n)
            old_val = h[j]
            best_delta = _golden_section(corr, coeff, -old_val, 1.0 - old_val, gs_iters)
            if abs(best_delta) > 1e-12:
                h[j] = old_val + best_delta
                n_improved += 1
        if n_improved == 0:
            break
        h = _fix_sum(h, target_sum)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(100, 0.5)
    n_parent = len(h_parent)

    best_h = h_parent.copy()
    best_c5 = _c5_of(h_parent)
    best_n = n_parent

    rng = np.random.RandomState(seed)

    n_list = sorted(set([n_parent] + [n for n in range(100, 520, 20) if n % 2 == 0]))

    for n in n_list:
        if time.time() > deadline - 100:
            break
        target_sum = n / 2.0

        starts = []
        if n == best_n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts.append(_fix_sum(np.clip(h_interp, 0, 1), target_sum))
        starts.append(_fix_sum(rng.rand(n), target_sum))

        for h_s in starts:
            if time.time() > deadline - 50:
                break
            candidates = []

            h1, c5_1 = _nesterov(h_s, n, target_sum, n_iters=200, alpha0=0.4)
            candidates.append((h1, c5_1))

            if time.time() > deadline - 30:
                break

            h2, c5_2 = _nesterov_smooth(h1, n, target_sum, n_iters=150, alpha0=0.25, top_m=5)
            candidates.append((h2, c5_2))

            h3 = _coord_descent_lagrangian(h2, n, target_sum, max_rounds=60, lam_init=0.0, eta=0.5, gs_iters=45)
            c5_3 = _c5_of(h3)
            candidates.append((h3, c5_3))

            if time.time() > deadline - 30:
                break

            h4 = _pairwise_local(h3, n, target_sum, max_iter=1000, rng=rng)
            c5_4 = _c5_of(h4)
            candidates.append((h4, c5_4))

            if time.time() > deadline - 25:
                break

            h5 = _coord_descent_lagrangian(h4, n, target_sum, max_rounds=60, lam_init=0.0, eta=0.3, gs_iters=50)
            c5_5 = _c5_of(h5)
            candidates.append((h5, c5_5))

            if time.time() > deadline - 20:
                break

            h6 = _pairwise_local(h5, n, target_sum, max_iter=500, rng=rng)
            c5_6 = _c5_of(h6)
            candidates.append((h6, c5_6))

            best_local = min(candidates, key=lambda x: x[1])
            if best_local[1] < best_c5:
                best_h = best_local[0].copy()
                best_c5 = best_local[1]
                best_n = n

    if time.time() <= deadline - 20:
        target_sum = best_n / 2.0
        h_pol = _coord_descent_lagrangian(best_h, best_n, target_sum, max_rounds=100, lam_init=0.0, eta=0.5, gs_iters=50)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    if time.time() <= deadline - 15:
        target_sum = best_n / 2.0
        h_pw = _pairwise_local(best_h, best_n, target_sum, max_iter=2000, rng=rng)
        c5_pw = _c5_of(h_pw)
        if c5_pw < best_c5:
            best_h, best_c5 = h_pw, c5_pw

    if time.time() <= deadline - 10:
        target_sum = best_n / 2.0
        h_fin = _coord_descent_simple(best_h, best_n, target_sum, max_rounds=30, gs_iters=50)
        c5_fin = _c5_of(h_fin)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
