import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(n):
    N = 1
    while N < n:
        N *= 2
    return N


def fft_cross_corr(h, g):
    n = len(h)
    N = next_pow2(2 * n)
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    C = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([C[N - n + 1:], C[:n]])


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return fft_cross_corr(h, 1.0 - h) * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_cross_corr(h, 1.0 - h)
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):

    total = exp_s.sum()
    value = m + np.log(total) / alpha
    weights = exp_s / total

def subgradient_optimize(h, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_score = evaluate_score(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        k_star = np.argmax(ov)
        offset = k_star - (n - 1)
        grad = np.zeros(n)
        if offset >= 0:
            i_lo = offset
            i_hi = n
            grad[i_lo:i_hi] += (1.0 - h[i_lo - offset:i_hi - offset])
        else:
            i_lo = 0
            i_hi = n + offset
            grad[i_lo:i_hi] += (1.0 - h[i_lo - offset:i_hi - offset])
        if offset >= 0:
            i_lo2 = 0
            i_hi2 = n - offset
            grad[i_lo2:i_hi2] -= h[i_lo2 + offset:i_hi2 + offset]
        else:
            i_lo2 = -offset
            i_hi2 = n
            grad[i_lo2:i_hi2] -= h[i_lo2 + offset:i_hi2 + offset]
        grad *= dx
        # Step size
        step = 0.1 / np.sqrt(it + 1.0)
        h_trial = h - step * grad
        h = project_onto_constraint(h_trial, target)
        s = evaluate_score(h)

```python
def compute_gradient_fft(h, w, dx):
    n = len(h)
    f = 1.0 - h
    W = w  # length 2n-1
    W_rev = W[::-1]
    N = next_pow2(3 * n)
    F = np.fft.rfft(f, N)
    WF = np.fft.rfft(W, N)
    WRF = np.fft.rfft(W_rev, N)
    HF = np.fft.rfft(h, N)
    conv1 = np.fft.irfft(F * WF, N)
    conv2 = np.fft.irfft(HF * WRF, N)
    A = conv1[n-1 : 2*n-1]  # n values
    B = conv2[n-1 : 2*n-1][::-1]  # n values
    return dx * (A - B)
```
