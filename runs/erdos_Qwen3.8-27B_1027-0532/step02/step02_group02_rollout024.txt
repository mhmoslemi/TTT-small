```python
def _softmax_weighted_gradient(h, alpha, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Shift for numerical stability
    shift = np.max(corr)
    weights = np.exp(alpha * (corr - shift))
    weights /= np.sum(weights)
    
    if top_m is not None:
        top_idx = np.argsort(-weights)[:top_m]
        weights[top_idx] = weights[top_idx] / np.sum(weights[top_idx])  # renormalize
        # or just zero out the rest
        mask = np.zeros_like(weights)
        mask[top_idx] = 1.0
        weights *= mask
        weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-15:
            continue
        lag = k - (n - 1)
        grad += weights[k] * _corr_grad(h, lag)
    
    return grad, weights, corr
```

def _corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        if lag < n:
            grad[:n - lag] -= h[lag:]
            grad[lag:] += (1.0 - h[:n - lag])
    else:
        L = -lag
        if L < n:
            grad[:n - L] -= h[L:]
            grad[L:] += (1.0 - h[:n - L])
    return grad

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_to_polytope(y, S):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(y - tau, 0.0, 1.0)

def _corr_pos(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n - 1:]

def _corr_grad_pos(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L < n:
        grad[:n - L] -= h[L:]
        grad[L:] += (1.0 - h[:n - L])
    return grad

def _softmax_grad(h, alpha):
    n = len(h)
    corr = _corr_pos(h)
    max_c = np.max(corr)
    w = np.exp(alpha * (corr - max_c))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(n)
        w[np.argmax(corr)] = 1.0
        w_sum = 1.0
    w /= w_sum
    
    grad = np.zeros(n)
    threshold = 0.5 / n
    for k in range(n):
        if w[k] < threshold:
            continue
        grad += w[k] * _corr_grad_pos(h, k)
    
    return grad, corr

def _run_gradient(h0, n_iter, alpha_lo, alpha_hi, lr0, lr_dec, mom, deadline):
    n = len(h0)
    S = n / 2.0
    h = _project_to_polytope(h0, S)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    stale = 0
    patience = 4000
    
    for t in range(n_iter):
        if time.time() > deadline:
            break
        frac = t / max(1, n_iter - 1)
        alpha = alpha_lo * (alpha_hi / alpha_lo) ** frac
        grad, _ = _softmax_grad(h, alpha)
        lr = lr0 / (1.0 + t * lr_dec)
        v = mom * v - lr * grad
        h = _project_to_polytope(h + v, S)
        
        if (t + 1) % 300 == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                stale = 0
            else:
                stale += 300
            if stale > patience:
                break
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp(h0, alpha, maxiter):
    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = _corr_pos(x)
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    
    def obj_grad(x):
        corr = _corr_pos(x)
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        s = np.sum(w)
        if s < 1e-300:
            return np.zeros(n)
        w /= s
        g = np.zeros(n)
        for k in range(n):
            if w[k] < 1e-12:
                continue
            g += w[k] * _corr_grad_pos(x, k)
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h0.copy()

def _structured_inits(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    inits = []
    
    for pf in [0.25, 0.33, 0.4, 0.5, 0.6, 0.7, 0.8]:
        p = max(2, int(n * pf))
        d = max(1, p // 2)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % p) < d else 0.0
        h += 0.05 * rng.randn(n)
        inits.append(h)
    
    for f in [1, 2, 3, 4, 6, 8]:
        h = 0.5 + 0.3 * np.sin(np.pi * f * x)
        h += 0.03 * rng.randn(n)
        inits.append(h)
    
    inits.append(rng.rand(n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 30
    
    best_h = None
    best_c5 = np.inf
    best_n = 0
    
    def _update(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = _c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
    
    # Build candidate list
    cands = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        iv = np.array(initial_h_values, dtype=float)
        cands.append((len(iv), iv.copy()))
    
    # Add structured inits for several n
    for n in [86, 100, 106, 120, 140, 160]:
        if len(cands) == 1 and cands[0][0] == n:
            continue  # already have initial for this n
        inits = _structured_inits(n, seed)
        for h0 in inits[:3]:  # limit per n
            cands.append((n, h0))
    
    # Sort by n (prefer n matching initial)
    cands.sort(key=lambda x: 0 if (cands and cands[0] and x[0] == cands[0][0]) else 1)
    
    # Phased optimization
    for idx, (n, h0) in enumerate(cands):
        t_rem = deadline - time.time()
        if t_rem < 45:
            break
        
        S = n / 2.0
        h0 = _project_to_polytope(h0, S)
        
        # Phase A: coarse (low alpha, high lr)
        h_a, c5_a = _run_gradient(h0, 6000, 5, 100, 2.0, 0.0005, 0.85, deadline)
        _update(h_a, n)
        
        if time.time() > deadline - 40:
            continue
        
        # Phase B: fine (high alpha, lower lr)
        h_b, c5_b = _run_gradient(h_a, 5000, 100, 1500, 0.8, 0.001, 0.8, deadline)
        _update(h_b, n)
        
        if time.time() > deadline - 35:
            continue
        
        # Phase C: SLSQP polish
        h_c = _slsqp(h_b, 300, 250)
        h_c = _project_to_polytope(h_c, S)
        h_c = _slsqp(h_c, 2000, 150)
        h_c = _project_to_polytope(h_c, S)
        _update(h_c, n)
    
    # Final: perturbation refinement on best
    if best_h is not None and time.time() < deadline - 25:
        n = best_n
        S = n / 2.0
        rng = np.random.RandomState(seed + 7)
        for attempt in range(10):
            if time.time() > deadline - 15:
                break
            scale = 0.01 + 0.03 * rng.rand()
            h_p = _project_to_polytope(best_h + scale * rng.randn(n), S)
            h_p = _slsqp(h_p, 1000, 100)
            h_p = _project_to_polytope(h_p, S)
            _update(h_p, n)
    
    if best_h is None:
        n = 100
        best_h = _project_to_polytope(np.full(n, 0.5), n / 2.0)
        best_n = n
        best_c5 = _c5_of(best_h)
    
    best_h = _project_to_polytope(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)

import time

from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_to_polytope(y, S):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(y - tau, 0.0, 1.0)

def _corr_pos(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n - 1:]

def _corr_grad_pos(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L < n:
        grad[:n - L] -= h[L:]
        grad[L:] += (1.0 - h[:n - L])
    return grad

def _softmax_grad(h, alpha):
    n = len(h)
    corr = _corr_pos(h)
    max_c = float(np.max(corr))
    w = np.exp(alpha * (corr - max_c))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(n)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    thr = 0.5 / n
    for k in range(n):
        if w[k] < thr:
            continue
        grad += w[k] * _corr_grad_pos(h, k)
    return grad, corr

def _slsqp_constraint(x, S):
    return x.sum() - S

def _slsqp_obj(x, alpha, n):
    corr = _corr_pos(x)
    m = float(np.max(corr))
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _slsqp_jac(x, alpha, n):
    corr = _corr_pos(x)
    m = float(np.max(corr))
    w = np.exp(alpha * (corr - m))
    s = float(np.sum(w))
    if s < 1e-300:
        return np.zeros(n)
    w = w / s
    g = np.zeros(n)
    for k in range(n):
        if w[k] < 1e-12:
            continue
        g += w[k] * _corr_grad_pos(x, k)
    return g * (2.0 / n)

def _slsqp(h0, alpha, maxiter):
    n = len(h0)
    S = n / 2.0
    bounds = [(0.0, 1.0)] * n

    try:

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h0.copy()
