Hmm wait, there's an issue with the corr update. When I do `corr += (t_best - h_old) * s`, this updates ALL entries including k=n-1. But the exact update for k=n-1 should be `(t-h_old)*s[n-1] - (t-h_old)²`. So after the linear update, I need to subtract `(t-h_old)²` from corr[n-1]. That's what I have.

But wait, there's another issue. When I change h[j], it also affects the slopes for other coordinates. But since I'm recomputing the slopes for each coordinate from scratch (using the current h), this is handled correctly in the Gauss-Seidel style.

However, there's a subtle issue with the corr array. When I update h[j] and do `corr += delta * s`, this is the exact change in corr due to changing h[j]. But the slope s was computed using the current h (before the update). Is this correct?

Let me verify: corr[k] = sum_i h[i] * (1-h[i+offset_k])
The change when h[j] changes from h_old to t:
= (t-h_old) * [(1-h[j+offset_k]) - h[j-offset_k]]

This is exact (we derived it earlier). The value (1-h[j+offset_k]) uses the current h (which hasn't been updated yet for position j+offset_k), and h[j-offset_k] also uses the current h. So the slope s is correct at the time of computation, and the update is exact.

After updating h[j], the slopes for other coordinates will be recomputed from the new h, so they'll be correct too.

Let me also think about whether to use the anti-palindromic constraint. If I enforce it, I can't independently update h[j] and h[n-1-j]. Instead, I'd need to update them together. This makes the subproblem 2D (or more complex).

Alternative: Do coordinate descent on the full h, and periodically project to anti-palindromic form. The projection is:

This maintains the sum constraint and [0,1] bounds. After projection, I can continue the coordinate descent from the projected point.

Actually, I think the simplest and most effective approach is:

2. Check if anti-palindromic projection improves things
3. If so, use the projected version as the new starting point

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_obj(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad

def slsqp_func(h, alpha):
    return smooth_max_val(compute_correlations(h), alpha)

def slsqp_jac(h, alpha):
    return grad_obj(h, alpha)

def slsqp_cons(h):
    return np.sum(h) - len(h) / 2.0

def slsqp_cons_jac(h):
    return np.ones(len(h))

def slsqp_opt(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        slsqp_func, h0, method='SLSQP', bounds=bounds,
        jac=slsqp_jac, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def coord_descent(h_init, max_sweeps=2000, tol=1e-12, n_grid=100, time_limit=None, t0=None):

    n = len(h)
    n_full = 2 * n - 1
    dx = 2.0 / n
    k_mid = n - 1  # index of self-correlation
    
    corr = compute_correlations(h).astype(float)
    best_score = np.max(corr) * dx
    best_h = h.copy()
    
    offsets = np.arange(n_full) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        max_change = 0.0
        for j in range(n):
            h_old = h[j]
            
            # Compute slopes
            a_idx = j + offsets
            b_idx = j - offsets
            mask_a = (a_idx >= 0) & (a_idx < n)
            mask_b = (b_idx >= 0) & (b_idx < n)
            
            s = np.zeros(n_full)
            s[mask_a] += 1.0 - h[a_idx[mask_a]]
            s[mask_b] -= h[b_idx[mask_b]]
            
            c = corr - h_old * s
            
            # 1D subproblem: min_t max(max_{k≠k_mid}(c[k]+s[k]*t), c[k_mid]+s[k_mid]*t-(t-h_old)²)
            # Exclude k_mid from linear part
            c_lin = np.concatenate([c[:k_mid], c[k_mid+1:]])
            s_lin = np.concatenate([s[:k_mid], s[k_mid+1:]])
            
            # Coarse grid
            t_grid = np.linspace(0.0, 1.0, n_grid)
            L = np.max(c_lin + np.outer(t_grid, s_lin), axis=1)
            Q = c[k_mid] + s[k_mid] * t_grid - (t_grid - h_old)**2
            f_vals = np.maximum(L, Q)
            
            best_idx = np.argmin(f_vals)
            
            # Refine
            if 0 < best_idx < n_grid - 1:
                t_lo = t_grid[best_idx - 1]
                t_hi = t_grid[best_idx + 1]
            else:
                t_lo = max(0.0, 1.0 / n_grid * (best_idx - 1))
                t_hi = min(1.0, 1.0 / n_grid * (best_idx + 1))
            
            t_fine = np.linspace(t_lo, t_hi, 50)
            L_fine = np.max(c_lin + np.outer(t_fine, s_lin), axis=1)

            t_best = t_fine[np.argmin(f_fine)]
            
            if abs(t_best - h_old) > 1e-12:
                delta = t_best - h_old
                h[j] = t_best
                corr += delta * s
                corr[k_mid] -= delta * delta
                max_change = max(max_change, abs(delta))
        
        current_score = np.max(corr) * dx
        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
        if max_change < tol:
            break
    
    return best_h, best_score

def antipal_projection(h):
    n = len(h)
    h_ap = np.zeros(n)
    for i in range(n):
        h_ap[i] = (h[i] + 1.0 - h[n - 1 - i]) / 2.0
    return h_ap
