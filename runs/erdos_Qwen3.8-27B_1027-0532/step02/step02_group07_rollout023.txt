    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15 * target:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def optimize_coordinate(h, j, corr):
    n = len(h)
    m_len = 2 * n - 1
    p = np.arange(m_len)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m_len)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    hj = h[j]
    a = corr - hj * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-13:
        return hj, slopes

    for _ in range(50):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-13:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-13:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = hj
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=500, tol=1e-12, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def precompute_structure(n):
    m_len = 2 * n - 1
    m1_list = []
    h1_list = []
    m2_list = []
    h2_list = []
    for k in range(n):
        m_lo1 = max(0, n - 1 - k)
        m_hi1 = min(m_len - 1, 2 * n - 2 - k)
        ms1 = np.arange(m_lo1, m_hi1 + 1)
        hs1 = k + ms1 - (n - 1)
        m1_list.append(ms1)
        h1_list.append(hs1)
        m_lo2 = max(0, k)
        m_hi2 = min(m_len - 1, k + n - 1)
        ms2 = np.arange(m_lo2, m_hi2 + 1)
        hs2 = k - ms2 + (n - 1)
        m2_list.append(ms2)
        h2_list.append(hs2)
    return m1_list, h1_list, m2_list, h2_list


def compute_slopes_matrix(h, structure):
    n = len(h)
    m_len = 2 * n - 1
    m1_list, h1_list, m2_list, h2_list = structure
    S = np.zeros((n, m_len))
    for k in range(n):
        if len(m1_list[k]) > 0:
            S[k, m1_list[k]] += (1.0 - h[h1_list[k]])
        if len(m2_list[k]) > 0:
            S[k, m2_list[k]] -= h[h2_list[k]]
    return S


def slsqp_optimize(h0, alpha, maxiter, bounds, structure):
    n = len(h0)

    def obj_and_grad(x):
        corr = compute_correlations(x)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        w = np.exp(shifted)
        w /= np.sum(w)
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
        S = compute_slopes_matrix(x, structure)
        grad = S.T.dot(w)
        return val, grad

    def con_eq(x):
        return np.sum(x) - n / 2.0

    def con_eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(
            obj_and_grad, h0, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}],
            jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None


def lbfgsb_step(h0, alpha, maxiter, structure):
    n = len(h0)
    mu = 5000.0 * alpha

    def obj_and_grad(x):
        corr = compute_correlations(x)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        w = np.exp(shifted)
        w /= np.sum(w)
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
        S = compute_slopes_matrix(x, structure)
        grad_sm = S.T.dot(w)
        sum_diff = np.sum(x) - n / 2.0
        val += mu * sum_diff * sum_diff
        grad = grad_sm + 2.0 * mu * sum_diff
        return val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj_and_grad, h0, method='L-BFGS-B', bounds=bounds,
            jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None
