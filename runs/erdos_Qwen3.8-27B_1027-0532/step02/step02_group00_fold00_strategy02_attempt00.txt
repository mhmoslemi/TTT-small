                h[i] = np.clip(h[i] + adjust_amount, 0, 1)
            return project_feasible(h, target_sum)
    else:  # current_sum > target_sum
        # Too high, need to decrease values
        idx = np.where(h == 1)[0]
        spread = target_sum - current_sum
        if len(idx) > 0:
            adjust_amount = spread / len(idx)
            for i in idx:
                h[i] = np.clip(h[i] + adjust_amount, 0, 1)
            return project_feasible(h, target_sum)
    
    # Fallback - use uniform distribution
    if abs(current_sum - target_sum) > 1e-5:
        factor = target_sum / current_sum
        h_scaled = h * factor
        h = np.clip(h_scaled, 0, 1)
    return h

def smooth_obj(h, alpha):
    """Smooth convex outer approximation of C5"""
    if np.issubdtype(h.dtype, int):
        h = h.astype(float)
    h_feas = np.clip(h, 0, 1)
    
    if np.defined(corr_full):  # Cache previous correlation value
        base_corr = np.max(corr_full)
    else:
        corr_full = np.correlate(np.clip(h_feas, 0, 1), 1.0 - np.clip(h_feas, 0, 1), mode='full')
        base_corr = np.max(corr_full)
    
    # Try to approximate without recomputing full convolution twice
    if hasattr(c_and_1_h, 'corr_value'):
        c_max = c_and_1_h.corr_value
    else:
        c_max = base_corr
        c_and_1_h.corr_value = c_max
    
    log_sum = np.log(np.sum(np.exp(alpha * (c_max - np.roll(c_max, 1)))))
    return float(base_corr + log_sum / alpha)

def interpolate_to(h_src, n_dst, target_dst):
    """Interpolate sample h_src between 0 and 1 to grid with n_dst points"""
    if np.issubdtype(h_src.dtype, np.integer):
        h_src = h_src.astype(float)
    x_orig = np.linspace(0.0, 2.0, len(h_src))
    t_new = np.linspace(0.0, 2.0, n_dst)
    y_new = np.interp(t_new, x_orig, h_src)
    return project_feasible(y_new, target_dst)

def update_feasibility_cache(h):
    """Update caching of function feasibility"""
    global c_and_1_h, corr_full
    if np.defined(c_and_1_h):
        c_and_1_h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(c_and_1_h, 1.0 - c_and_4int.clip(h,0,1), mode="full")
    return np.max(corr), corr, c_and_1_h

def run_slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    """Local refinement around h_start with SLSQP algorithm"""
    global c_and_1_h_cand, best_score_cand
    bounds = [(0.0, 1.0)] * n
    constants = [{'type': 'eq', 'fun': lambda h_start: np.sum(h_start) - target}, {'type': 'ineq', 'fun': lambda h_start: np.clip(h_start, 0, 1).sum()}]
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    c_and_1_h_cand = None
    default_obj = None
    trend_value = 0
    
    # Check initial candidate feasibility without resetting
    if calculate_overlap:
        corr_full, info = np.correlate(np.clip(best_h,0,1),1.0-np.clip(best_h,0,1),mode='full',method='auto')
        best_score = np.max(corr_full)
    
    for alpha in alphas:
        try:
            # Update the trend value based on economic evaluation
            trend_value *= 0.99  # Exploration decay
            
            def wrapper(x):
                global c_and_1_h_cand
                if len([x for x in c_and_1_h_cand]) == 0 or alpha != alpha_valtrap:
                    # Single point evaluation is okay
                    evaluated = False
                    candidate = np.clip(x, 0.0, 1.0)
                    smooth_val = smooth_obj(candidate, alpha)
                    return smooth_val + trend_value * np.ones(len(candidate))
                return smooth_obj(x, alpha)
            
            alpha_valtrap = alpha
            options = {'maxiter': int(maxiter_per_alpha), 'ftol': ftol, 'disp': False,
                       'bounds': bounds, 'constraints': constants}
            res = minimize(x0=best_h, fun=wrapper, method='SLSQP', **options)
            h_new = res.x
            h_new_clip = np.clip(h_new, 0.0, 1.0)
            try:
                kc_value = compute_c5(h_new_clip)
                if kc_value < best_score:
                    best_score = kc_value
                    best_h = h_new_clip.copy()
            except Exception as e:
                pass
                
        except Exception as e:
            pass
    
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    # Initialize global state
    global c_and_1_h_cand, c_and_4int, corr_full, best_score, timer_started
    timer_started = time.time()
    best_score = 0.381
    
    # Load initial condition if available
    if hasattr(globals(), 'initial_h_values'):
        h0 = initial_h_values.copy()
        n_init = len(h0)
        target_init = n_init / 2.0
    else:
        h0 = np.zeros(400) + 0.3
        n_init = 400
        target_init = 200.0
    
    # Multi-resolution candidate grid
    n_candidates_stages = [400, 500, 600, 700, 800]
    best_h = h0
    best_score = compute_c5(best_h)
    early_stop_count = 0
    
    # Hybrid cache management for state recovery
    state_cache = {
        'current': None,
        'prev': None,
        'stdev': None,
        'significant_improvement': 0.0,
        'consecutive_failures': 0
    }
    
    # Iteration loop
    for n_candidate in n_candidates_stages:
        if budget_s - (time.time() - timer_started) < 350:  # Safety margin
            break
        
        # Adjust dimensionality
        if n_candidate != len(best_h):
            h_interpolated = interpolate_to(best_h, n_candidate, n_init/2)
            best_score_inter = compute_c5(h_interpolated)
            if best_score_inter < best_score:
                best_score = best_score_inter
                best_h = h_interpolated.copy()
        
        # Local search phase 1: focused refinement with dynamic penalty
        try:
            perturbed_starts = []
            if state_cache.get('consecutive_failures') < 5:
                # Enhanced perturbation - magnitude related to time left
                perturb_scale = 0.1 / np.sqrt(n_candidate)
                for i in range(0, min(3, max(1, int(budget_s/30)))):
                    perturb = np.random.randn(n_candidate) * perturb_scale
                    h_pert = best_h + perturb
                    h_pert = project_feasible(h_pert, n_candidate/2.0)
                    perturbed_starts.append(h_pert)
            
            # Main local search
            h_local, score_local = run_slsqp_round(best_h, n_candidate, target_init, alphas=[5, 15, 40, 100], maxiter_per_alpha=500, ftol=1e-6)
        except:
            # Resort to adaptive random search if local methods fail
            h_local = np.clip(best_h + np.random.randn(n_candidate)*0.05, 0, 1)
            h_local = project_feasible(h_local, target_init)
            score_local = compute_c5(h_local)
            best_score = min(best_score, score_local)
        
        if score_local < best_score:
            best_score = score_local
            best_h = h_local.copy()
            state_cache.update({
                'significant_improvement': best_score * 0.99,
                'consecutive_failures': 0
            })
            # Quantize the hourglass timer for future tuning
            state_cache['stdev'] = np.std(np.random.randn(100)) * 0.8   # Reduced variance for faster processing
            
        # Time budget monitoring
        elapsed = time.time() - timer_started
        if elapsed > budget_s - 3600/state_cache['iterations_perturb']:
            break
        
    # Ensure grid size alignment
    best_h = interpolate_to(best_h, 400, 200)
    final_score = compute_c5(best_h)
    
    # Finalization
    post_processed = project_feasible(best_h, 200)
    post_score = compute_c5(post_processed)
    return post_processed, post_score, len(post_processed)
