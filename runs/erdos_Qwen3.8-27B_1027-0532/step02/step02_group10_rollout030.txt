from scipy.optimize import minimize

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    num_shifts = len(weights)
    for m_idx in range(num_shifts):
        w = weights[m_idx]
        if w < 1e-13:
            continue
        offset = m_idx - (n - 1)
        if offset >= 0:
            # overlap at positive shift: sum_i h[i] * (1-h[i+offset])
            # d/dh[i] = (1-h[i+offset]) for i < n-offset
            #           - h[i-offset] for i >= offset
            valid = n - offset
            grad[:valid] += w * (1.0 - h[offset:])
            grad[offset:] -= w * h[:valid]
        else:
            # offset is negative, let offset_abs = -offset
            offset_abs = -offset
            valid = n - offset_abs
            # overlap at negative shift: sum_i h[i+offset_abs] * (1-h[i])
            # which is sum_j h[j] * (1-h[j-offset_abs]) where j = i+offset_abs
            # d/dh[j] = (1-h[j-offset_abs]) for j >= offset_abs
            #           - h[j+offset_abs] for j < n-offset_abs
            grad[offset_abs:] += w * (1.0 - h[:valid])
            grad[:valid] -= w * h[offset_abs:]
    grad *= dx
    return grad

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Compute overlap change
            h[j] = 1
            h[i] = 0

            h[i] = 1
            h[j] = 0

            h[best_i] = 0
            h[best_j] = 1
            current_score -= best_improvement
            ones = np.where(h == 1)[0]

        else:

    n = len(h_binary)
    h = h_binary.copy().astype(int)
    current_score = evaluate_score(h.astype(float))
    best_h = h.copy()

    T_end = 0.001
    n_ones = h.sum()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_score = evaluate_score(h.astype(float))
        delta = new_score - current_score

            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_score
