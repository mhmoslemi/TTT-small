def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_box_sum(x, target_sum):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(x - tau, 0.0, 1.0)

def _smoothed_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_vals = np.exp(alpha * (corr - m))
    exp_sum = exp_vals.sum()
    obj = (m + np.log(exp_sum) / alpha) * dx
    w = exp_vals / exp_sum
    grad = np.zeros(n)
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-14:
            continue
        lag = j - (n - 1)
        if lag > 0:
            grad[:n - lag] -= wj * h[lag:]
            grad[lag:] += wj * (1.0 - h[:n - lag])
        elif lag < 0:
            L = -lag
            grad[:n - L] += wj * (1.0 - h[L:])
            grad[L:] -= wj * h[:n - L]
        else:
            grad[:] += wj * (1.0 - 2.0 * h)
    grad *= dx
    return obj, grad
