def m_sequence(m, n):
    """Generate m-sequence of length 2^m - 1, truncated to n"""
    length = 2**m - 1
    # Use a primitive polynomial for m-bit LFSR
    # For m=7: x^7 + x^3 + 1 (primitive)
    # For m=8: x^8 + x^4 + x^3 + x^2 + 1 (primitive)
    
    # General approach: use a known primitive polynomial
    # Tap positions (0-indexed, LSB is position 0)
    taps = {
        4: [4, 3],  # x^4 + x^3 + 1
        5: [5, 3, 1],  # x^5 + x^3 + x + 1
        6: [6, 5, 3, 2],  # x^6 + x^5 + x^3 + x^2 + 1
        7: [7, 3, 1],  # x^7 + x^3 + 1
        8: [8, 4, 3, 2, 1],  # x^8 + x^4 + x^3 + x^2 + x + 1
    }
    
    if m not in taps:
        return None
    
    tap_pos = taps[m]
    # State is m bits
    state = 1  # Start with all 1s
    seq = np.zeros(length)
    
    for i in range(length):
        seq[i] = 1.0  # Output the LSB
        # Feedback
        feedback = 0
        for t in tap_pos:
            feedback ^= (state >> (t - 1)) & 1
        state = (state >> 1) | (feedback << (m - 1))
    
    return seq[:n]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

def legendre_sequence(p):
    h = np.zeros(p)
    h[0] = 1.0
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    return h

def make_binary_balanced(h, n):
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:target]] = 1.0
    return result

def smooth_max_obj(x, eps):
    h = x
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    return (m + eps * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps):
    h = x
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / eps
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        offset = (n - 1) - k
        if offset >= 0:
            vn = n - offset
            grad[:vn] += wk * (1 - h[offset:offset + vn])
            grad[offset:offset + vn] -= wk * h[:vn]
        else:
            d = -offset
            vn = n - d
