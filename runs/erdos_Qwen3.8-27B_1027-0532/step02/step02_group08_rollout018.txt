```python
    # m-(n-1) <= i-(n-1) <= m  =>  m <= i <= m+n-1
    
    # For term2, valid lags where 0<=m+lags<n:
    # -m <= lags < n-m  =>  n-1-m <= i < n+m-1+1 = n+m
    
    # Let me just compute it with a loop over m (n iterations, each O(n)):
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for m in range(n):
        # term1: sum over i where m <= i <= m+n-1 (clipped to [0, 2n-2])
        i_lo = max(0, m)
        i_hi = min(2*n-2, m+n-1)
        if i_lo <= i_hi:
            l_valid = lags[i_lo:i_hi+1]
            u = m - l_valid  # = m - (i - (n-1)) = m - i + n - 1
            # u ranges from m - l_valid[i_hi] to m - l_valid[i_lo]
            # = m - (m+n-1-(n-1)) to m - (m-(n-1)) = 0 to n-1 (approximately)
            term1 = np.dot(weights[i_lo:i_hi+1], one_minus_h[u])
        
        # term2: sum over i where n-1-m <= i <= n+m-1 (clipped)
        i_lo2 = max(0, n-1-m)
        i_hi2 = min(2*n-2, n+m-1)
        if i_lo2 <= i_hi2:
            l_valid2 = lags[i_lo2:i_hi2+1]
            v = m + l_valid2  # = m + i - (n-1)
            term2 = np.dot(weights[i_lo2:i_hi2+1], h[v])
        
        grad[m] = term1 - term2
    
    return obj, grad
```

Hmm, this loop over m is O(n²) total but with a Python loop of n iterations, each doing O(n) numpy operations. For n=2000, that's 2000 iterations with small numpy ops, which might take 100-500ms. That's too slow for 500 optimization iterations.

Let me use the FFT approach for the gradient instead. Let me work out the correct formula.

The gradient is:
grad[m] = sum_l w[l] * (1-h[m-l]) * 1_{0≤m-l<n} - sum_l w[l] * h[m+l] * 1_{0≤m+l<n}

First term: Let u = m-l, so l = m-u.
= sum_u w[m-u] * (1-h)[u] * 1_{0≤u<n}
= sum_{u=0}^{n-1} w[m-u+n-1] * (1-h)[u]  (using w indexed by 0..2n-2, with w[k] corresponding to lag k-(n-1))

Wait, w is indexed by the output index k (0 to 2n-2), and the lag is l = k-(n-1).
So w[m-u+n-1] is the weight for lag l = (m-u+n-1)-(n-1) = m-u. ✓

This is a convolution: (w * (1-h))[m] where w is appropriately zero-padded.

Specifically, if I define W of length N with W[k] = w[k] for k=0,...,2n-2 and 0 otherwise:
Then the first term at position m is: sum_u W[m-u] * (1-h)[u] for u=0,...,n-1
= (W * (1-h))[m] (linear convolution, if N is large enough)

But wait, the convolution (W * (1-h))[m] = sum_u W[m-u] * (1-h)[u] for ALL u.
Since (1-h)[u] = 0 for u≥n (if we zero-pad), this is fine.
And W[m-u] is nonzero only when 0≤m-u≤2n-2, i.e., m-(2n-2)≤u≤m.

So the first term = ifft(fft(W, N) * fft(1-h_padded, N)).real[:n]

Second term: Let v = m+l, so l = v-m.
= sum_v w[v-m] * h[v] * 1_{0≤v<n}
= sum_{v=0}^{n-1} w[v-m+n-1] * h[v]

This is: sum_v W[v-m] * h[v] (where W is zero-padded)
= (W_rev * h)[m] where W_rev[k] = W[-k] ... 

Hmm, this is a cross-correlation. Let me think...
sum_v W[v-m] * h[v] = sum_v h[v] * W[v-m]

If I define W_rev[j] = W[-j] (with appropriate zero-padding), then:
sum_v h[v] * W[v-m] = sum_v h[v] * W_rev[m-v] = (h * W_rev)[m] (convolution)

So second term = ifft(fft(h_padded, N) * fft(W_rev_padded, N)).real[:n]

And W_rev[k] = W[-k mod N] = W_padded[(-k) mod N]

In FFT: fft(W_rev) = conj(fft(W)) (if W is real)... no, that's not right either.

Actually, for real signals: if W_rev[k] = W[-k mod N], then FFT(W_rev)[j] = conj(FFT(W)[j]) * ... 

Let me just use: FFT(W_rev)[j] = sum_k W[-k] * exp(-2πijk/N) = sum_k W[k'] * exp(2πijk'/N) (with k'=-k)
= conj(FFT(W)[j]) (if W is real)

Wait: FFT(W)[j] = sum_k W[k] * exp(-2πijk/N)
conj(FFT(W)[j]) = sum_k W[k] * exp(+2πijk/N)

And FFT(W_rev)[j] = sum_k W_rev[k] * exp(-2πijk/N) = sum_k W[-k] * exp(-2πijk/N)
Let k' = -k: = sum_{k'} W[k'] * exp(+2πijk'/N) = conj(FFT(W)[j])

Yes! So FFT(W_rev) = conj(FFT(W)) for real W.

Therefore:
First term = ifft(FFT(W) * FFT(1-h_padded)).real[:n]
Second term = ifft(FFT(h_padded) * FFT(W_rev)).real[:n] = ifft(FFT(h_padded) * conj(FFT(W))).real[:n]

And grad = first_term - second_term

This is all O(N log N)!

Let me verify with a small example mentally... actually, let me just trust the math and implement it.

So the smooth max objective and gradient can both be computed in O(N log N) where N is the FFT size (next power of 2 ≥ 3n-1 or so).

For n=2000, N=4096 or 8192, and the FFT takes about 100μs. So each gradient evaluation is about 0.5ms, and 500 iterations take about 0.25 seconds.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def fft_corr(h):
    """Compute full correlation using FFT."""
    n = len(h)
    N = 1
    while N < 2*n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    c = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([c[N-n+1:], c[:n]])

    """Compute smooth max of correlations and its gradient using FFT."""
    n = len(h)
    N = 1
    while N < 4*n:

    # Compute correlations via FFT
    corr = fft_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    
    # Compute gradient via FFT
    # W[k] = weights[k] for k=0,...,2n-2
    W = np.zeros(N)
    W[:2*n-1] = weights
    
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    Wf = np.fft.rfft(W, N)

    term1 = np.fft.irfft(Wf * G, N)[:n]

    term2 = np.fft.irfft(H * np.conj(Wf), N)[:n]

    """Compute delta in correlation when flipping h[p]=0→1 and h[q]=1→0."""
    l = np.arange(-(n-1), n)
    p_plus = p + l
    p_minus = p - l
    q_plus = q + l
    q_minus = q - l
    
    vp_p = (p_plus >= 0) & (p_plus < n)
    vm_p = (p_minus >= 0) & (p_minus < n)
    vp_q = (q_plus >= 0) & (q_plus < n)
    vm_q = (q_minus >= 0) & (q_minus < n)
    
    delta = np.zeros(2*n - 1)
    delta[vp_p] += (1.0 - h[p_plus[vp_p]])
    delta[vm_p] -= h[p_minus[vm_p]]
    delta[vp_q] -= (1.0 - h[q_plus[vp_q]])
    delta[vm_q] += h[q_minus[vm_q]]
    
    return delta
