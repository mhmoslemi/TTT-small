For the full correlation r = np.correlate(h, 1-h, mode='full'), the index mapping is:
r[k] = Σ_{i} h[i] * (1-h)[i + k - (n-1)]

where the sum is over all valid i such that 0 ≤ i < n and 0 ≤ i+k-(n-1) < n.

Let m = k - (n-1), so k = m + (n-1), and m ranges from -(n-1) to n-1.

r[m + (n-1)] = Σ_{i: 0≤i<n, 0≤i+m<n} h[i] * (1-h[i+m])

For m ≥ 0: = Σ_{i=0}^{n-1-m} h[i] * (1-h[i+m])
For m < 0: = Σ_{i=|m|}^{n-1} h[i] * (1-h[i+m]) = Σ_{i=0}^{n-1-|m|} h[i+|m|] * (1-h[i])

Now, when h[i] changes by Δh_i (from 1 to 0, so Δh_i = -1):
- For m ≥ 0:
  - h[i] appears as h[p] when p=i, valid if i ≤ n-1-m, i.e., m ≤ n-1-i
    - Change: Δh_i * (1-h[i+m]) = -1 * (1-h[i+m])

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj_and_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    obj = lse_val(overlaps, alpha)
    grad = lse_grad(h, alpha).copy()
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

        res = minimize(
            penalty_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None

                    h_kick[zeros2[rng.randint(len(zeros2))]] = 0.5

            # Greedy re-optimize
            h_k_result, _ = greedy_1swap(h_kick, max_rounds=15, time_limit_s=budget_s - 60, t0=t0)
            h_k_result = project_box_sum(h_k_result, n / 2.0)
            try_update(h_k_result, n)

    # Phase 5: Continuous refinement with L-BFGS-B
    if time_left() > 60:
        start_h = best_h.copy()
        for alpha in [500, 1000, 2000, 4000]:
            if time_left() < 20:
                break
            penalty = alpha * 50.0
            maxiter = max(100, min(1000, int(200000 / n)))
            bounds = [(0.0, 1.0)] * n
            res_x = lbf_optimize(start_h, alpha, penalty, maxiter, bounds)
            if res_x is not None:
                start_h = try_update(res_x, n)
                if start_h is None:
                    break

    # Phase 6: Try even n for clean binary, then interpolate
    if time_left() > 90:
        for n_try in [n - 1, n + 1, n + 4, n - 4]:
            if n_try < 200 or n_try > 600 or time_left() < 40:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_box_sum(h_interp, n_try / 2.0)
            h_b = (h_interp > 0.5).astype(float)
            # Fix sum for even n
            ones = np.where(h_b == 1)[0]
            if len(ones) > n_try // 2:
                for idx in ones[:len(ones) - n_try // 2]:
                    h_b[idx] = 0.0
            elif len(ones) < n_try // 2:
                zeros = np.where(h_b == 0)[0]
                for idx in zeros[:n_try // 2 - len(ones)]:
                    h_b[idx] = 1.0
            h_g2, _ = greedy_1swap(h_b, max_rounds=20, time_limit_s=60, t0=t0)
            h_g2 = project_box_sum(h_g2, n_try / 2.0)
            try_update(h_g2, n_try)

    # Phase 7: Final SLSQP polish
    if time_left() > 30:
        nf = len(best_h)
        for alpha in [3000, 8000]:
            if time_left() < 15:
                break
            bounds = [(0.0, 1.0)] * nf
            res_x = slsqp_optimize(best_h, alpha, 300, bounds)
            if res_x is not None:
                try_update(res_x, nf)

    # Final: also try a fresh continuous optimization from a different start
    if time_left() > 30:
        h_fresh = best_h.copy()
        h_fresh = 1.0 - h_fresh
        h_fresh = project_box_sum(h_fresh, len(h_fresh) / 2.0)
        for alpha in [1000, 3000]:
            if time_left() < 15:
                break
            penalty = alpha * 40.0
            nf = len(h_fresh)
            bounds = [(0.0, 1.0)] * nf
            res_x = lbf_optimize(h_fresh, alpha, penalty, 500, bounds)
            if res_x is not None:
                try_update(res_x, nf)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
