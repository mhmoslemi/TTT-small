import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def _smooth_obj(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def _smooth_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    idx = np.arange(n)
    lags = np.arange(-(n - 1), n)
    ii = idx[:, None]
    ll = lags[None, :]
    jl = ii + ll
    jr = ii - ll
    vl = (jl >= 0) & (jl < n)
    vr = (jr >= 0) & (jr < n)
    jl_c = np.clip(jl, 0, n - 1)
    jr_c = np.clip(jr, 0, n - 1)
    t1 = vl * (1.0 - x[jl_c])
    t2 = vr * x[jr_c]
    return np.sum(w[None, :] * (t1 - t2), axis=1) * dx

        else:
