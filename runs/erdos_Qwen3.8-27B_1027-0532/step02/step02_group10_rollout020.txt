    value = m + np.log(total) / alpha
    weights = exp_vals / total
    return value, weights

def smooth_min_and_weights(x, alpha):
    m = np.min(x)
    shifted = -alpha * (x - m)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    value = m - np.log(total) / alpha
    weights = exp_vals / total
    return value, weights

def grad_smooth_min_autocorr(h, ac, weights, dx):
    """Gradient of -smooth_min(A_k) w.r.t. h.
    We want to MAXIMIZE smooth_min, so we minimize -smooth_min.
    d(-smooth_min)/dh[j] = -sum_k w_k * dA_k/dh[j]
    where dA_k/dh[j] = dx*(h[j+k] if j+k<n) + dx*(h[j-k] if j-k>=0)
    """
    n = len(h)
    neg_grad = np.zeros(n)  # This will be the negative gradient (direction to increase smooth_min)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        neg_grad[:n-k] += w * h[k:]
        neg_grad[k:] += w * h[:n-k]
    return -dx * neg_grad  # negative of gradient of smooth_min = direction to decrease -smooth_min

def pgd_smooth_min(h, alpha, target, max_iters, deadline):

    Equivalent to minimizing max of overlaps (approximately)."""
    n = len(h)

    h = h.copy()

    step = min(step, 0.5)
    prev_val = None
    stall = 0

        ac = autocorr_truncated(h)
        sm, weights = smooth_min_and_weights(ac, alpha)
        if prev_val is not None:
            if sm > prev_val + 1e-14:
                stall = 0
            else:
                stall += 1
            if stall > 300:
                break
        prev_val = sm
        # Gradient of smooth_min w.r.t. h (we want to go in +gradient direction)
        # d(smooth_min)/dh[j] = sum_k w_k * dx * (h[j+k] + h[j-k])
        # We want to maximize, so we do h += step * gradient
        grad = np.zeros(n)
        for k in range(len(weights)):
            w = weights[k]
            if w < 1e-15:
                continue
            grad[:n-k] += w * h[k:]
            grad[k:] += w * h[:n-k]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

def simulated_annealing(h, target, n_iters, deadline, rng, alpha=500):
    """Simulated annealing on individual coordinates."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    T = 0.02
    T_min = 0.001
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pick a random coordinate
        j = rng.randint(n)
        # Propose a change
        delta = rng.randn() * (0.01 + 0.1 * T)
        h_new = h.copy()
        h_new[j] = np.clip(h[j] + delta, 0.0, 1.0)
        # Maintain sum constraint: adjust another coordinate
        sum_diff = target - h_new.sum()
        if abs(sum_diff) > 1e-10:
            # Pick another coordinate to adjust
            j2 = rng.randint(n)
            if j2 != j:
                h_new[j2] = np.clip(h[j2] + sum_diff, 0.0, 1.0)
            else:
                h_new = h  # reject
        score_new = evaluate_score(h_new)
        score_cur = evaluate_score(h)
        delta_f = score_new - score_cur
        if delta_f < 0 or rng.rand() < np.exp(-delta_f / T):
            h = h_new
            if score_new < best_score:
                best_score = score_new
                best_h = h.copy()
        # Cool down
        T = max(T_min, T * 0.999)
    return best_h, best_score

def block_perturb(h, target, rng, n_blocks=5):
    """Perturb by flipping blocks."""
    n = len(h)
    h_new = h.copy()
    block_size = n // n_blocks
    for _ in range(rng.randint(1, 4)):
        start = rng.randint(0, n - block_size)
        end = min(start + block_size, n)
        h_new[start:end] = np.clip(h_new[start:end] + rng.randn(end - start) * 0.2, 0, 1)
    return project_onto_constraint(h_new, target)

def compute_overlaps_fast(h):
    """Compute all O_s values using FFT. Returns array of length 2n-1 in correlate order."""
    n = len(h)
    dx = 2.0 / n
    h_pad = np.zeros(2 * n)
    h_pad[:n] = h
    H = np.fft.fft(h_pad)
    ac_raw = np.fft.ifft(H * np.conj(H)).real[:n]
    ac = ac_raw * dx  # ac[s] = dx * sum_{i=0}^{n-1-s} h[i+s]*h[i]
    
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(h)
    total = prefix[n]
    
    # For shift s >= 0 (output index n-1+s):
    # O_s = dx * (sum_{j=s}^{n-1} h[j] - ac_raw[s])
    # = dx * (total - prefix[s] - ac_raw[s])
    # For shift -s (s > 0) (output index n-1-s):
    # O_{-s} = dx * (sum_{j=0}^{n-1-s} h[j] - ac_raw[s])
    # = dx * (prefix[n-s] - ac_raw[s])
    
    # Return in the same order as np.correlate(h, 1-h, 'full')
    result = np.empty(2 * n - 1)
    # s = n-1, n-2, ..., 1, 0, 1, ..., n-1
    # index 0 corresponds to shift -(n-1)
    # index n-1 corresponds to shift 0
    # index 2n-2 corresponds to shift n-1
    
    # For shift s (s >= 0), output index = n-1+s
    for s in range(n):
        result[n-1+s] = total - prefix[s] - ac_raw[s]
    # For shift -s (s > 0), output index = n-1-s
    for s in range(1, n):
        result[n-1-s] = prefix[n-s] - ac_raw[s]
    
    return result * dx

def gradient_overlaps(h, weights, dx):
    """Compute gradient of smooth max of overlaps.
    weights[s] corresponds to O_{s-(n-1)} (the shift value).
    """
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-14:
            continue
        s = idx - (n - 1)  # shift value
        if s >= 0:
            # O_s = sum_{j=s}^{n-1} h[j] - A_s (times dx)
            # dO_s/dh[j] = 1 (if j >= s) - h[j+s] (if j+s<n) - h[j-s] (if j-s>=0)
            # The "1" term: contributes 1 to grad[j] for j >= s
            grad[s:] += w
            # The -h[j+s] term: for j=0..n-1-s, contributes -h[j+s]
            grad[:n-s] -= w * h[s:]
            # The -h[j-s] term: for j=s..n-1, contributes -h[j-s]
            grad[s:] -= w * h[:n-s]
        else:
            # O_{-s'} where s' = -s > 0
            # O_{-s'} = sum_{j=0}^{n-1-s'} h[j] - A_{s'} (times dx)
            sp = -s
            # dO_{-s'}/dh[j] = 1 (if j <= n-1-s') - h[j+s'] (if j+s'<n) - h[j-s'] (if j-s'>=0)
            grad[:n-sp] += w
            grad[:n-sp] -= w * h[sp:n]  # for j=0..n-1-sp: -h[j+sp]
            grad[sp:] -= w * h[:n-sp]  # for j=sp..n-1: -h[j-sp]
    return dx * grad
