                        else:
                            best[di] = best[di] - d1
                            best[dj] = best[dj] - d2
                    if found:
                        break
                if found:
                    break
            if found:
                break
        if not found:
            no_improve += 1
    return best, best_max * dx


def random_local_search(h, n_iters, rng):
    N = len(h)
    dx = 2.0 / N
    best = h.copy()
    best_max = np.max(compute_corr(best))
    steps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]
    no_improve = 0
    for it in range(n_iters):
        if no_improve > 2000:
            break
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            no_improve += 1
            continue
        found = False
        for step in steps:
            for (d1, d2) in [(step, -step), (-step, step)]:
                ni = best[i] + d1
                nj = best[j] + d2
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                best[i] = ni
                best[j] = nj
                new_max = np.max(compute_corr(best))
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    no_improve = 0
                    found = True
                    break
                else:
                    best[i] = ni - d1
                    best[j] = nj - d2
            if found:
                break
        if not found:
            no_improve += 1
    return best, best_max * dx


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    best_h = project_feasible(init, n / 2.0)
    best_score = evaluate_score(best_h)

    MU = 50000.0

    # Phase 1: L-BFGS with increasing alpha
    alphas = [100, 300, 1000, 3000, 10000, 30000]
    for alpha in alphas:
        if time.time() - t0 > budget_s - 350:
            break
        maxiter = max(100, 60000 // n)
        x = lbfgs_run(best_h, alpha, n, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score - 1e-14:
                best_score = s
                best_h = h_new.copy()

    # Phase 2: Subgradient refinement
    if time.time() - t0 < budget_s - 250:
        for eta0 in [0.01, 0.005, 0.002]:
            if time.time() - t0 > budget_s - 200:
                break
            best_h, sg_max = subgrad_refine(best_h, 3000, eta0, 0.002)
            best_h = project_feasible(best_h, n / 2.0)
            s = evaluate_score(best_h)
            if s < best_score:
                best_score = s

    # Phase 3: Guided local search
    remaining = budget_s - 200 - (time.time() - t0)
    if remaining > 30:
        n_ls = int(remaining * 1.5)
        best_h, ls_score = guided_local_search(best_h, n_ls, rng)
        best_h = project_feasible(best_h, n / 2.0)
        s = evaluate_score(best_h)
        if s < best_score:
            best_score = s

    # Phase 4: Random local search from different perturbations
    if time.time() - t0 < budget_s - 100:
        for pert_scale in [0.05, 0.1, 0.15]:
            if time.time() - t0 > budget_s - 60:
                break
            pert = best_h + rng.randn(n) * pert_scale
            pert = project_feasible(pert, n / 2.0)
            alpha_p = [200, 500, 1000, 3000][rng.randint(4)]
            maxiter_p = max(50, 30000 // n)
            x = lbfgs_run(pert, alpha_p, n, MU, maxiter_p)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score - 1e-14:
                    best_score = s
                    best_h = h_new.copy()

        n_rs = int((budget_s - 60 - (time.time() - t0)) * 1.5)
        if n_rs > 100:
            best_h, rs_score = random_local_search(best_h, n_rs, rng)
            best_h = project_feasible(best_h, n / 2.0)
            s = evaluate_score(best_h)
            if s < best_score:
                best_score = s

    # Phase 5: Higher resolution
    for n_try in [2 * n, 3 * n]:
        if n_try > 2500:
            continue
        if time.time() - t0 > budget_s - 80:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_try / 2.0)
        for alpha in [200, 1000, 5000, 15000]:
            if time.time() - t0 > budget_s - 40:
                break
            maxiter_hi = max(40, 80000 // n_try)
            x = lbfgs_run(h_hi, alpha, n_try, MU * (float(n_try) / n), maxiter_hi)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_try / 2.0)
                s = evaluate_score(h_new)
                if s < best_score - 1e-14:
                    best_score = s
                    best_h = h_new.copy()
                    h_hi = h_new.copy()
        # Local search at higher res
        rem = budget_s - 40 - (time.time() - t0)
        if rem > 20 and len(best_h) == n_try:
            n_ls2 = int(rem * 1.0)
            best_h, _ = guided_local_search(best_h, n_ls2, rng)
            best_h = project_feasible(best_h, n_try / 2.0)
            s = evaluate_score(best_h)
            if s < best_score:
                best_score = s

    # Final high-alpha polish
    n_final = len(best_h)
    for alpha in [50000, 100000]:
        if time.time() - t0 > budget_s - 15:
            break
        maxiter_f = max(200, 100000 // n_final)
        x = lbfgs_run(best_h, alpha, n_final, MU, maxiter_f)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_final / 2.0)
            s = evaluate_score(h_new)
            if s < best_score - 1e-14:
                best_score = s
                best_h = h_new.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
