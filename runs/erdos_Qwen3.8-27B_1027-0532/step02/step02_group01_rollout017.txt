import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, eps):
    corr = compute_corr_vector(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += wk * dx * omh[s:s+m]
            grad[s:] -= wk * dx * h[:m]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:r+m]
            grad[r:] -= wk * dx * h[:m]
    return grad

def grad_single_shift(h, s, dx):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[:m] += dx * (1.0 - h[s:s+m])
        g[s:] -= dx * h[:m]
    else:
        r = -s
        m = n - r
        g[r:] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[r:r+m]
    return g

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def cutting_plane_refine(h, max_iters=200, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_onto_constraint(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = compute_corr_vector(h)
        cmax = np.max(corr)
        
        if cmax < best_c5:
            best_c5 = cmax
            best_h = h.copy()
        
        active_thresh = cmax - 0.02
        active_idx = np.where(corr >= active_thresh)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        if len(active_idx) > 60:
            order = np.argsort(corr[active_idx])[::-1]
            active_idx = active_idx[order[:60]]
        
        c_obj = np.zeros(n + 1)
        c_obj[0] = 1.0
        A_ub_list = []
        b_ub_list = []
        
        for k_idx in active_idx:
            s = k_idx - (n - 1)
            gs = grad_single_shift(h, s, dx)
            row = np.zeros(n + 1)
            row[0] = -1.0
            row[1:] = gs
            A_ub_list.append(row)
            b_ub_list.append(-corr[k_idx] + 1e-12)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([0.0])
        
        bounds = [(0.0, None)]
        for j in range(n):
            bounds.append((-h[j], 1.0 - h[j]))
        
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break
        
        if not result.success:
            break
        
        delta = result.x[1:]
        step_norm = np.max(np.abs(delta))
        if step_norm < tol:
            break
        
        best_alpha = 0
        best_val = cmax
        for alpha in [1.0, 0.9, 0.7, 0.5, 0.3, 0.1, 0.05]:
            h_trial = project_onto_constraint(h + alpha * delta, target_sum)
            val = compute_c5(h_trial)

                best_val = val
                best_alpha = alpha
                best_h_trial = h_trial
        
        if best_alpha == 0:
            break
        
        h = best_h_trial
    
    return h, best_c5

def binary_sa_fast(h_start, n_swaps, t_start, t_end, rng):

    h = np.round(h_start).astype(float)

    target = n // 2
    if one_count > target:
        ones = np.where(h == 1.0)[0]
        excess = one_count - target
        to_flip = ones[:excess]
        h[to_flip] = 0.0
    elif one_count < target:
        zeros = np.where(h == 0.0)[0]
        deficit = target - one_count
        to_flip = zeros[:deficit]
        h[to_flip] = 1.0
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for i in range(n_swaps):
        t = t_start * (t_end / t_start) ** (i / max(n_swaps - 1, 1))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5

            if current_c5 < best_c5:

                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    
    return best_h, best_c5

def pauli_jaccard_perturb(h, rng):
    n = len(h)
    h_new = h.copy()
    for _ in range(5):
        blen = rng.integers(3, max(4, n // 5) + 1)
        a = rng.integers(0, n - blen + 1)
        b = rng.integers(0, n - blen + 1)
        if a == b:
            continue
        i1, i2 = min(a, b), max(a, b)
        if i2 - i1 < blen:
            continue
        block = h_new[i1:i1+blen].copy()
        h_new[i1:i1+blen] = h_new[i2:i2+blen]
        h_new[i2:i2+blen] = block
    return h_new

def random_block_flip(h, rng):
    n = len(h)
    h_new = h.copy()
    n_blocks = rng.integers(2, 5)
    for _ in range(n_blocks):
        blen = rng.integers(2, max(3, n // 6) + 1)
        a = rng.integers(0, n - blen + 1)
        h_new[a:a+blen] = 1.0 - h_new[a:a+blen]
    return h_new

def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_new = np.zeros(n_new)
    for i in range(n_new):
        pos = i * n / n_new
        lo = int(pos)
        hi = min(lo + 1, n - 1)
        frac = pos - lo
        h_new[i] = (1 - frac) * h[lo] + frac * h[hi]
    return h_new
