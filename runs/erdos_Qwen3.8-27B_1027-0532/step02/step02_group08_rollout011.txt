        penalty_obj_grad, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu),
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15, 'iprint': 0,
                 'maxls': 200, 'maxfun': maxiter * 80}
    )
    return res.x


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_fun, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x


def trust_constr_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_fun, h0, method='trust-constr', bounds=bounds, args=(alpha,),
        jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'gtol': 1e-12, 'xtol': 1e-14,
                 'verbose': 0, 'bar_maxiter': 500}
    )
    return res.x


def binary_projection(h, n):
    target = n / 2.0
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:int(target)]] = 1.0
    remainder = target - int(target)
    if remainder > 1e-10:
        h_b[idx[int(target)]] = remainder
    return h_b


def corr_guided_perturb(h, rng, scale):
    n = len(h)
    corrs = compute_correlations(h)
    m = np.max(corrs)
    active_mask = corrs > m - 0.05 * m
    k_star = np.argmax(corrs)
    r = k_star - (n - 1)
    grad_k = np.zeros(n)
    for i in range(n):
        if 0 <= i + r < n:
            grad_k[i] += (1.0 - h[i + r])
        if 0 <= i - r < n:
            grad_k[i] -= h[i - r]
    grad_k = grad_k - np.mean(grad_k)
    norm = np.linalg.norm(grad_k)
    if norm > 1e-10:
        grad_k = grad_k / norm
    pert = h - scale * grad_k
    noise = rng.randn(n) * scale * 0.3
    pert = pert + noise
    return pert


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)
    best_h = project_sum(init, n0 / 2.0)
    best_score = evaluate_score(best_h)

    bounds0 = [(0.0, 1.0)] * n0

    # Phase 1: Quick refinement at original resolution with moderate alpha
    for alpha in [100, 300, 800, 2000]:
        if time_left() < 200:
            break
        res_x = lbfgsb_optimize(best_h, alpha, 5000.0, 600, bounds0)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n0 / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 2: Multi-resolution exploration
    n_best = len(best_h)
    resolutions = [60, 100, 150, 250, 425, 550, 700, 900, 1100]
    for n_try in resolutions:
        if time_left() < 250:
            break
        if n_try == n_best:
            continue
        h_init = interpolate_h(best_h, n_best, n_try)
        h_init = project_sum(h_init, n_try / 2.0)
        bounds_try = [(0.0, 1.0)] * n_try
        mu_try = 8000.0
        h_curr = h_init.copy()
        alphas_seq = [50, 150, 400, 1000, 3000, 8000]
        for alpha in alphas_seq:
            if time_left() < 100:
                break
            maxiter = max(80, min(800, 300000 // n_try))
            res_x = lbfgsb_optimize(h_curr, alpha, mu_try, maxiter, bounds_try)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_curr = h_new.copy()
                        n_best = n_try
                    else:
                        h_curr = h_new.copy()

    # Phase 3: Try binary projection at best resolution
    n_best = len(best_h)
    h_bin = binary_projection(best_h, n_best)
    h_bin_opt = project_sum(h_bin, n_best / 2.0)
    s_bin = evaluate_score(h_bin_opt)
    if s_bin < best_score:
        best_score = s_bin
        best_h = h_bin_opt.copy()

    # If binary was better, optimize from binary starting point
    if time_left() > 300:
        h_curr = project_sum(best_h, n_best / 2.0)
        bounds_best = [(0.0, 1.0)] * n_best
        for alpha in [200, 600, 1500, 4000, 10000]:
            if time_left() < 150:
                break
            res_x = lbfgsb_optimize(h_curr, alpha, 10000.0, 500, bounds_best)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_curr = h_new.copy()

    # Phase 4: SLSQP refinement
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    for alpha in [3000, 8000, 15000, 30000]:
        if time_left() < 80:
            break
        res_x = slsqp_optimize(best_h, alpha, 250, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 5: Trust-constr refinement
    if time_left() > 120:
        for alpha in [5000, 12000]:
            if time_left() < 60:
                break
            res_x = trust_constr_optimize(best_h, alpha, 300, bounds_best)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()

    # Phase 6: Extensive random restarts at best resolution
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    max_restarts = 800
    for i in range(max_restarts):
        if time_left() < 50:
            break
        strategy = i % 8
        if strategy == 0:
            scale = 0.02 + 0.06 * (i / float(max_restarts))
            pert = best_h + rng.randn(n_best) * scale
        elif strategy == 1:
            scale = 0.08 + 0.12 * (i / float(max_restarts))
            pert = best_h + rng.randn(n_best) * scale
        elif strategy == 2:
            pert = best_h.copy()
            n_swap = max(3, n_best // 20)
            idx = rng.choice(n_best, size=n_swap, replace=False)
            pert[idx] = 1.0 - pert[idx]
        elif strategy == 3:
            sorted_idx = np.argsort(-best_h)
            pert = np.zeros(n_best)
            pert[sorted_idx[:n_best // 2]] = 1.0
            pert += rng.randn(n_best) * 0.1
        elif strategy == 4:
            pert = corr_guided_perturb(best_h, rng, 0.05 + 0.1 * rng.rand())
        elif strategy == 5:
            pert = best_h.copy()
            mask = rng.rand(n_best) < 0.15
            n_mask = int(np.sum(mask))
            if n_mask > 0:
                pert[mask] = np.clip(pert[mask] + rng.randn(n_mask) * 0.5, 0, 1)
        elif strategy == 6:
            n_block = max(5, n_best // 10)
            start = rng.randint(0, max(1, n_best - n_block))
            pert = best_h.copy()
            pert[start:start + n_block] = rng.rand(n_block)
        else:
            pert = best_h + rng.randn(n_best) * (0.03 + 0.04 * rng.rand())

        pert = project_sum(np.clip(pert, 0.0, 1.0), n_best / 2.0)
        alpha = 200 + 800 * rng.randint(8)
        maxiter_r = max(60, min(400, 200000 // n_best))
        res_x = lbfgsb_optimize(pert, alpha, 10000.0, maxiter_r, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    if i % 100 == 0:
                        h_bin2 = binary_projection(best_h, n_best)
                        h_bin2 = project_sum(h_bin2, n_best / 2.0)
                        s_b2 = evaluate_score(h_bin2)
                        if s_b2 < best_score:
                            best_score = s_b2
                            best_h = h_bin2.copy()

    # Phase 7: High-alpha polish
    for alpha in [20000, 50000, 100000]:
        if time_left() < 20:
            break
        res_x = lbfgsb_optimize(best_h, alpha, 50000.0, 500, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 8: Try nearby resolutions
    for n_adj in [n_best + 10, n_best - 10, n_best + 25, n_best - 25, n_best + 50]:
        if time_left() < 25:
            break
        if n_adj < 100 or n_adj > 2500:
            continue
        h_init = interpolate_h(best_h, n_best, n_adj)
        h_init = project_sum(h_init, n_adj / 2.0)
        bounds_adj = [(0.0, 1.0)] * n_adj
        for alpha in [2000, 5000, 12000]:
            if time_left() < 15:
                break
            res_x = lbfgsb_optimize(h_init, alpha, 15000.0, 300, bounds_adj)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_adj / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_init = h_new.copy()

    # Phase 9: Final SLSQP polish
    if time_left() > 40:
        n_final = len(best_h)
        bounds_final = [(0.0, 1.0)] * n_final
        for alpha in [20000, 50000]:
            if time_left() < 15:
                break
            res_x = slsqp_optimize(best_h, alpha, 150, bounds_final)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()

    # Phase 10: Try binary projection one more time
    n_final = len(best_h)
    h_bin = binary_projection(best_h, n_final)
    h_bin = project_sum(h_bin, n_final / 2.0)
    s_bin = evaluate_score(h_bin)
    if s_bin < best_score:
        best_score = s_bin
        best_h = h_bin.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
