    """SLSQP with log-sum-exp smooth max."""
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -100:
            return 0.0
        shifted = alpha * (corr - m)
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 50)
        log_sum = m + np.log(np.sum(np.exp(shifted))) / alpha
        return log_sum * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -50, 50)
        exps = np.exp(shifted)
        weights = exps / np.sum(exps)
        
        # Compute gradient
        grad = np.zeros(n)
        # Only process lags with significant weight
        threshold = 1e-6
        sig_indices = np.where(weights > threshold)[0]

            lag = idx - (n - 1)

        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -50, 50)
        exps = np.exp(shifted)
        total = np.sum(exps)
        weights = exps / total
        
        grad = np.zeros(n)
        # Only process significant lags
        for idx in range(ncorr):

            if w < 1e-5:

            if lag == 0:
                grad += w * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[:n-lag] += w * (1.0 - x[lag:])
                grad[lag:] -= w * x[:n-lag]
            else:
                L = -lag
                grad[:n-L] += w * (1.0 - x[L:])
                grad[L:] -= x[:n-L] * w
        
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
