import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} via bisection on threshold."""
    n = len(h)
    lo = h.min() - target/n - 2.0
    hi = h.max() - target/n + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    return x

def smooth_max_obj_grad(h, alpha):
    """Smooth max of correlations and its gradient."""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = alpha * (corrs - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = exp_v.sum()
    weights = exp_v / sum_exp
    obj = m + np.log(sum_exp) / alpha
    
    # Gradient via convolutions
    g = 1.0 - h
    conv1 = np.convolve(weights, g, mode='full')  # length (2n-1)+n-1 = 3n-2
    conv2 = np.convolve(h, weights[::-1], mode='full')  # same length
    idx = slice(n - 1, 2 * n - 1)
    grad = conv1[idx] - conv2[idx]
    return obj, grad

def direct_max_grad(h):
    """Exact gradient of the max correlation (subgradient)."""
    n = len(h)
    corrs = compute_correlations(h)
    k_star = corrs.argmax()
    s = k_star - (n - 1)  # actual shift
    g = 1.0 - h
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= i - s < n:
            val += g[i - s]
        if 0 <= i + s < n:
            val -= h[i + s]
        grad[i] = val
    return corrs[k_star], grad

def coordinate_descent_step(h, alpha_smooth, n_cd_steps):
    """Do several coordinate descent steps on the smooth max objective."""
    n = len(h)
    h_curr = h.copy()
    for _ in range(n_cd_steps):
        obj, grad = smooth_max_obj_grad(h_curr, alpha_smooth)
        # Find the coordinate with the most negative gradient (most improvement)
        # But we need to respect the sum constraint and box constraints
        # Simple approach: take a step in the negative gradient direction, then project
        # Use a backtracking line search
        best_h = h_curr.copy()
        best_obj = obj
        lr = 1.0 / (alpha_smooth * n)
        for ls in range(30):
            trial = h_curr - lr * grad
            trial_proj = project_box_sum(trial, n / 2.0)
            trial_obj, _ = smooth_max_obj_grad(trial_proj, alpha_smooth)

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = -3.0
    hi = 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = np.clip(alpha * (corrs - m), -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = exp_v.sum()
    weights = exp_v / sum_exp
    obj = m + np.log(sum_exp) / alpha
    g = 1.0 - h
    conv1 = np.convolve(weights, g, mode='full')
    conv2 = np.convolve(h, weights[::-1], mode='full')
    idx = slice(n - 1, 2 * n - 1)
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def topk_smooth_obj_grad(h, alpha, k):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corrs)[-k:]
    tc = corrs[top_idx]
    m = tc.max()
    shifted = np.clip(alpha * (tc - m), -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = exp_v.sum()
    w_top = exp_v / sum_exp
    obj = m + np.log(sum_exp) / alpha
    weights = np.zeros(len(corrs))
    weights[top_idx] = w_top
    g = 1.0 - h
    conv1 = np.convolve(weights, g, mode='full')
    conv2 = np.convolve(h, weights[::-1], mode='full')
    idx = slice(n - 1, 2 * n - 1)
    grad = conv1[idx] - conv2[idx]
    return obj, grad

    obj, grad = smooth_max_obj_grad(h, alpha)
    viol = np.sum(h) - n / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad

    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad


def direct_max_polish(h, n_steps=500):
    n = len(h)
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_score = evaluate_score(h_curr)
    lr = 5e-5
    for _ in range(n_steps):
        corrs = compute_correlations(h_curr)
        k_star = corrs.argmax()
        s = k_star - (n - 1)
        g = 1.0 - h_curr
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            if 0 <= i - s < n:
                v += g[i - s]
            if 0 <= i + s < n:
                v -= h_curr[i + s]
            grad[i] = v
        trial = project_box_sum(h_curr - lr * grad, n / 2.0)
        sc = evaluate_score(trial)
        if sc < best_score:
            best_score = sc
            best_h = trial.copy()
            h_curr = trial.copy()
            lr = min(lr * 1.3, 2e-3)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return best_h, best_score
