Similarly, this is linear in h[j] for all k ≠ 0.

So the objective as a function of h[j] (with all other coordinates fixed) is:
max( max_{k≠0} (a_k * h[j] + b_k), h[j] - h[j]^2 + b_0 ) * dx

where a_k and b_k are constants (determined by the other coordinates), and b_0 is a constant.

This is the max of a set of affine functions and one concave quadratic. The max of affine functions is convex. The max of a convex function and a concave function is... not necessarily convex or concave.

However, we can still use a 1D optimizer. Since the function is piecewise (determined by which term achieves the max), we can find the optimal h[j] by:
1. Computing the value of the objective at h[j] = 0 and h[j] = 1
2. Finding the "breakpoints" where the active term changes
3. Optimizing within each piece

This is complex. A simpler approach: use a fine grid search over h[j] ∈ [0,1] (say 100 points) and pick the best. This is O(n * 100 * (n + n)) = O(n^2 * 100) per round, which for n=100 is about 1e6 operations per round. That's fast.

Actually, even simpler: for each coordinate j, evaluate the objective at h[j] = 0, 0.1, 0.2, ..., 1.0 and pick the best. This is n * 11 evaluations of _c5_of, each of which is O(n) for the correlate + O(n) for the max. So total is n * 11 * O(n) = O(n^2 * 11). For n=100, that's about 110,000 operations per round. import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _cord_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smo_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def _smo_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w = w / w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    lags = np.arange(len(corr)) - (n - 1)
    for idx in range(len(corr)):
        weight = w[idx]
        if weight < 1e-10:
            continue
        lag = lags[idx]
        if lag >= 0:
            mlen = n - lag
            if mlen > 0:
                g[lag:lag+mlen] += (1.0 - h[:mlen])
                g[:mlen] += -h[lag:lag+mlen]
        else:
            j = -lag
            mlen = n - j
            if mlen > 0:
                g[:mlen] += (1.0 - h[j:])
                g[j:] += -h[:mlen]
    return g * dx * alpha * w.sum() / (alpha * (2.0/n)) * (2.0/n) / w.sum()

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n

    def obj(x):
        return _smo_obj(x, alpha, n)

    def cons(x):
        return x.sum() - target_sum

    bounds = [(0.0, 1.0)] * n
    cons_dict = [{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons_dict,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.success or res.fun < float('inf'):
            return _fix_sum(res.x, target_sum)
    except Exception:
        pass
    return h0

def _fine_coord_descent(h0, n, target_sum, max_rounds=30, n_levels=20):
    h = h0.copy()
    grid = np.linspace(0.0, 1.0, n_levels)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in grid:
                if abs(v - old) < 1e-12:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _binary_round(h, target_sum):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    h_bin = _fix_sum(h_bin, target_sum)
    return h_bin

def _de_optimize(h_init, n, target_sum, pop_size=30, n_gens=200, f_mut=0.7, f_cross=0.8, seed=42):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = h_init.copy()
    for i in range(1, pop_size):
        if i < pop_size // 3:
            pert = h_init + 0.3 * rng.randn(n)
        elif i < 2 * pop_size // 3:
            pert = 0.5 * h_init + 0.5 * rng.rand(n)
        else:
            pert = rng.rand(n)
        pop[i] = _fix_sum(pert, target_sum)

    pop_obj = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(pop_obj)
    best_h = pop[best_idx].copy()
    best_c = pop_obj[best_idx]

    for gen in range(n_gens):
        for i in range(pop_size):
            a, b, c = rng.choice(pop_size, 3, replace=False)
            mutant = pop[a] + f_mut * (pop[b] - pop[c])
            mask = rng.rand(n) < f_cross
            mutant[mask] = h_init[mask] if False else pop[i][mask]
            mutant = _fix_sum(mutant, target_sum)
            obj_m = _c5_of(mutant)
            obj_v = pop_obj[i]
            if obj_m <= obj_v:
                pop[i] = mutant
                pop_obj[i] = obj_m
                if obj_m < best_c:
                    best_c = obj_m
                    best_h = mutant.copy()
                    best_idx = i
            else:
                pop[i] = np.clip(pop[i] + 0.1 * rng.randn(n), 0, 1)
                pop[i] = _fix_sum(pop[i], target_sum)
                pop_obj[i] = _c5_of(pop[i])
                if pop_obj[i] < best_c:
                    best_c = pop_obj[i]
                    best_h = pop[i].copy()
                    best_idx = i
    return best_h, best_c

def _sim_anneal(h0, n, target_sum, t_max=0.5, t_min=0.01, n_iters=500, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    curr_c = _c5_of(h)
    best_h = h.copy()
    best_c = curr_c
    for it in range(n_iters):
        temp = t_max * (t_min / t_max) ** (float(it) / float(n_iters))
        j = rng.randint(n)
        old = h[j]
        h[j] = np.clip(h[j] + rng.randn() * 0.3, 0.0, 1.0)
        h = _fix_sum(h, target_sum)
        new_c = _c5_of(h)
        delta = new_c - curr_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h = best_h.copy() if False else h
            h[j] = old
            h = _fix_sum(h, target_sum)
    return best_h, best_c

def _interpolate(h_src, n_src, n_dst):
    x_src = np.arange(n_src, dtype=float)
    x_dst = np.linspace(0, n_src - 1, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return np.clip(h_dst, 0.0, 1.0)

def _nesterov(h0, n, target_sum, n_iters=200, alpha0=0.3):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[lag:lag+m] += (1.0 - y[:m])
                g[:m] += -y[lag:lag+m]
        else:
            j = -lag
            m = n - j
            if m > 0:
                g[:m] += (1.0 - y[j:])
                g[j:] += -y[:m]
        g *= dx
        lr = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - lr * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _smooth_sg_nesterov(h0, n, target_sum, n_iters=200, alpha0=0.25, top_m=8):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        shifted = 80.0 * (corr - max_corr)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        top_idx = np.argsort(weights)[::-1][:top_m]
        w_m = weights[top_idx].copy()
        w_m = w_m / w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            k = top_idx[idx]
            lag = k - (n - 1)
            if lag >= 0:
                m = n - lag
                if m > 0:
                    g[lag:lag+m] += (1.0 - y[:m])
                    g[:m] += -y[lag:lag+m]
            else:
                j = -lag
                m = n - j
                if m > 0:
                    g[:m] += (1.0 - y[j:])
                    g[j:] += -y[:m]
        g *= dx * w_m.sum()
        g = np.average([w_m[idx] * (g / w_m.sum()) for idx in range(len(top_idx))], axis=0, weights=w_m)
        lr = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - lr * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _full_pipeline(h0, n, target_sum, deadline):
    c5 = _c5_of(h0)
    h_curr = h0.copy()

    h_n, c5_n = _nesterov(h_curr, n, target_sum, n_iters=150, alpha0=0.3)
    if c5_n < c5:
        h_curr, c5 = h_n, c5_n

    if time.time() > deadline - 5:
        return h_curr, c5

    h_s, c5_s = _smooth_sg_nesterov(h_curr, n, target_sum, n_iters=150, alpha0=0.2, top_m=8)
    if c5_s < c5:
        h_curr, c5 = h_s, c5_s

    alphas = [200, 800, 3000, 10000, 30000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=400)
        c5_sl = _c5_of(h_sl)
        if c5_sl < c5:
            h_curr, c5 = h_sl, c5_sl

    if time.time() > deadline - 5:
        return h_curr, c5

    h_cd = _fine_coord_descent(h_curr, n, target_sum, max_rounds=15, n_levels=15)
    c5_cd = _c5_of(h_cd)
    if c5_cd < c5:
        h_curr, c5 = h_cd, c5_cd

    if time.time() > deadline - 5:
        return h_curr, c5

    for alpha in [5000, 20000]:
        if time.time() > deadline - 5:

        h_sl2 = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=300)
