    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad_calc(h, eps, n):

    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)

            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term

    return grad

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    count = int(np.sum(h_bin))
    target = n // 2
    if count > target:
        ones_idx = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(ones_idx))
        for k in range(count - target):
            h_bin[ones_idx[perm[k]]] = 0
    elif count < target:
        zeros_idx = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(zeros_idx))
        for k in range(target - count):
            h_bin[zeros_idx[perm[k]]] = 1
    return h_bin

def exhaustive_single_swap(h, n, max_rounds, time_check):
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_val = float(np.max(corr))
    dx = 2.0 / n
    best_c5 = best_val * dx
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for r in range(max_rounds):
        if time_check():
            break
        best_swap = None
        best_new_val = best_val
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_corr = np.correlate(h, 1 - h, mode="full")
                new_max = float(np.max(new_corr))
                if new_max < best_new_val:
                    best_new_val = new_max
                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        corr = np.correlate(h, 1 - h, mode="full")
        best_val = float(np.max(corr))
        best_c5 = best_val * dx
    return h, best_c5

def simulated_annealing_binary(h, n, time_check, t_start, t_end, n_steps):
    h = h.copy()
    best_h = h.copy()
    curr_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
    best_val = curr_val
    dx = 2.0 / n
    t = t_start
    cool = (t_end / t_start) ** (1.0 / max(n_steps, 1))
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for step in range(n_steps):
        if time_check():
            break
        t *= cool
        # Random single swap
        i = ones_list[np.random.randint(len(ones_list))]
        j = zeros_list[np.random.randint(len(zeros_list))]
        h[i] = 0
        h[j] = 1
        new_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
        delta = new_val - curr_val
        if delta < 0 or np.random.rand() < np.exp(-delta / (t * dx)):
            curr_val = new_val
            if curr_val < best_val:
                best_val = curr_val
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_val * dx

        h_rand_bs, c5_rand_bs = exhaustive_single_swap(h_rand, n, 30, lambda: time_limit(0.60))
        h_rand_sa, c5_rand_sa = sa_single_swap(
            h_rand_bs, n,
            time_check=lambda: time_limit(0.63),
            t_start=0.05, t_end=0.0005, n_steps=40000
        )
        h_rand_bs2, c5_rand_bs2 = exhaustive_single_swap(h_rand_sa, n, 30, lambda: time_limit(0.65))
        if c5_rand_bs2 < best_c5:
            best_c5 = c5_rand_bs2
            best_h_bin = h_rand_bs2

    if not time_limit(0.70):
        h_bs, c5_bs = exhaustive_single_swap(best_h_bin, n, 80, lambda: time_limit(0.70))
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h_bin = h_bs

    n2 = 2 * n
    if not time_limit(0.92):
        h_up = upsample(best_h_bin, n, n2)
        h2_init = make_binary(h_up, n2, np.random)
        best_c5_2 = evaluate_c5(h2_init)
        best_h2 = h2_init.copy()

        if not time_limit(0.78):
            h2_bs, c5_2_bs = exhaustive_single_swap(best_h2, n2, 40, lambda: time_limit(0.78))
            if c5_2_bs < best_c5_2:
                best_c5_2 = c5_2_bs
                best_h2 = h2_bs

        if not time_limit(0.85):
            h2_sa, c5_2_sa = sa_single_swap(
                best_h2, n2,
                time_check=lambda: time_limit(0.85),
                t_start=0.03, t_end=0.0005, n_steps=50000
            )
            h2_bs, c5_2_bs = exhaustive_single_swap(h2_sa, n2, 30, lambda: time_limit(0.85))
            if c5_2_bs < best_c5_2:
                best_c5_2 = c5_2_bs
                best_h2 = h2_bs

        if not time_limit(0.90):
            h2_dsa, c5_2_dsa = sa_double_swap(
                best_h2, n2,
                time_check=lambda: time_limit(0.90),
                t_start=0.02, t_end=0.0005, n_steps=20000
            )
            h2_bs, c5_2_bs = exhaustive_single_swap(h2_dsa, n2, 30, lambda: time_limit(0.90))
            if c5_2_bs < best_c5_2:
                best_c5_2 = c5_2_bs
                best_h2 = h2_bs

        if best_c5_2 < best_c5:
            best_c5 = best_c5_2
            best_h_bin = best_h2
            n = n2

    if not time_limit(0.98):
        bounds_n = [(0.0, 1.0)] * n
        h_cont = best_h_bin.copy()
        for eps_val in [5000.0, 10000.0]:
            if time_limit(0.98):
                break
            h_new, c5_new = slsqp_local(h_cont, n, bounds_n, eps_val, 300, 1e-15)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cont = h_new
        if best_c5 > evaluate_c5(best_h_bin):
            h_cont = best_h_bin.copy()
            best_c5 = evaluate_c5(h_cont)

    n_final = len(best_h_bin)
    h_final = project_feasible(best_h_bin, n_final)
    c5_final = evaluate_c5(h_final)
    return h_final, c5_final, n_final
