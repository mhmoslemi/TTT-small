import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _soft_grad(h, alpha):
    """Compute the gradient of the log-sum-exp approximation of C5."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Normalize for numerical stability
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= weights.sum()
    
    # Now compute weighted sum of gradients
    # For each lag k (index in corr array), the actual lag is k - (n-1)
    grad = np.zeros(n)
    max_lag = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # corr[k] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j] = (1-h[j+lag]) for j < n-lag, minus h[j-lag] for j >= lag
            l = lag
            # First part: j from 0 to n-1-l
            grad[:n-l] += w * (1.0 - h[l:])
            # Second part: j from l to n-1
            grad[l:] -= w * h[:n-l]
        else:
            l = -lag
            # corr[k] = sum_{i=0}^{n-1-l} h[i+l]*(1-h[i])
            # d/dh[j] = (1-h[j-l]) for j >= l, minus h[j+l] for j < n-l
            grad[l:] += w * (1.0 - h[:n-l])
            grad[:n-l] -= w * h[l:]
    
    return grad * (2.0 / n)  # include the dx factor

def _lse_objective(h, alpha):
    """Log-sum-exp approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

def _soft_grad_projected(h, alpha, target_sum, lr, n_iter, deadline, check_every):
    """Projected gradient descent with soft (softmax-weighted) gradient."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 15
    
    # Use momentum
    v = np.zeros(n)
    momentum = 0.7
    
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        
        g = _soft_grad(h, alpha)
        
        # Adjust for the equality constraint using projection
        # The gradient of the constraint is uniform, so we subtract the mean
        # to stay on the sum = target_sum manifold
        g = g - np.mean(g)  # This keeps sum(h) approximately constant
        
        lr_t = lr / (1.0 + 0.001 * t)
        v = momentum * v - lr_t * g
        h_new = h + v
        h = _fix_sum(h_new, target_sum)
        
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _lse_obj_and_grad(h, alpha):
    """Compute log-sum-exp objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    logsumexp = m + np.log(exp_vals.sum()) / alpha
    obj = logsumexp * dx
    
    # Softmax weights
    w = exp_vals / exp_vals.sum()
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += w[k] * (1.0 - 2.0 * h)

            grad[l:] += w[k] * (1.0 - h[:n-l])

def _slsqp_polish(h_init, alpha, maxiter=300, ftol=1e-14):

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:

            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:

            sub = min(-diff, h[i])

            diff += sub
    return h
