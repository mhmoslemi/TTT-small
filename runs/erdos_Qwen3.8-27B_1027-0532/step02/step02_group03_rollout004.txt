import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    """Project y onto {x: 0<=x<=1, sum(x)=S}"""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_at_lag(h, lag, dx):
    """Gradient of c[lag_index] with respect to h, where lag_index = n-1+lag"""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            # c = sum_{i=0}^{m-1} h[i+lag] * (1-h[i])
            # d/dh[j] for j in [lag, n): contributes (1-h[j-lag]) from h[j]*(1-h[j-lag])
            # d/dh[j] for j in [0, m): contributes -h[j+lag] from h[j+lag]*(1-h[j])
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            # c = sum_{i=j}^{n-1} h[i] * (1-h[i-j])
            grad[j:] += (1.0 - h[:m])
            grad[:m] += -h[j:]
    return grad * dx

def _subgrad_topK(h, dx, K=5, alpha=50.0):
    """Subgradient using top-K lags with softmax weighting"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    weights = np.exp(alpha * (corr - c_max))
    total = weights.sum()
    if total < 1e-300:
        # All mass on argmax
        k_star = np.argmax(corr)
        return _grad_at_lag(h, k_star - (n-1), dx)
    weights = weights / total
    # Take top K
    top_K = np.argsort(weights)[::-1][:K]
    w_top = weights[top_K].copy()
    w_top = w_top / w_top.sum()
    g = np.zeros(n)
    for idx in range(K):
        lag = top_K[idx] - (n - 1)
        g += w_top[idx] * _grad_at_lag(h, lag, dx)
    return g

def _optimize_subgrad(h0, n, target_sum, max_iters, alpha_start=0.5, alpha_decay=0.999, topK=5, smooth_alpha=50.0):
    """Subgradient method with adaptive step size and restart"""
    dx = 2.0 / n
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    alpha = alpha_start
    momentum = np.zeros(n)
    beta = 0.9

        momentum = beta * momentum + (1 - beta) * g
        x_new = _project_box_sum(x - alpha * momentum, target_sum)
        x = x_new
        val = _c5_of(x)
        if val < best_val:
            best_val = val
            best = x.copy()
        alpha *= alpha_decay
        if alpha < 0.001:
            alpha = 0.01  # restart
    
    return best, best_val

def _optimize_nesterov(h0, n, target_sum, max_iters, alpha0=0.3, topK=5, smooth_alpha=50.0):
    """Nesterov-accelerated subgradient method"""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(max_iters):
        g = _subgrad_topK(y, dx, K=topK, alpha=smooth_alpha)
        step = alpha0 / np.sqrt(it + 1)
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _project_box_sum(y, target_sum)
        x = x_new
        t = t_new
        val = _c5_of(x)
        if val < best_val:
            best_val = val
            best = x.copy()
    
    return best, best_val

def _coordinate_descent(h0, n, target_sum, max_rounds=20, mode='binary'):
    """Coordinate descent: try setting each coordinate to 0 or 1"""
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            c_old = _c5_of(h)
            best_c = c_old
            best_v = old
            if mode == 'binary':
                for v in [0.0, 1.0]:
                    h[j] = v
                    c = _c5_of(h)
                    if c < best_c:
                        best_c = c
                        best_v = v

                    h[j] = v
                    c = _c5_of(h)
                    if c < best_c:
                        best_c = c
                        best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

    v = np.zeros(n)
    best = x.copy()
    best_v = _c5_of(x)
    for it in range(iters):
        g = _subgrad(x, dx, topK, sa)
        v = beta * v + (1.0 - beta) * g
        step = alpha0 / (1.0 + 0.03 * it)
        x_new = _project(x - step * v, target)
        x = x_new
        val = _c5_of(x)
        if val < best_v:
            best_v = val
            best = x.copy()
    return best, best_v

def _coord_binary(h0, n, target, rounds=20):
    h = h0.copy()
    for _ in range(rounds):
        imp = False
        for j in range(n):
            old = h[j]
            c_old = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = old
            if c0 < c_old - 1e-15:
                h[j] = 0.0
                imp = True
            elif c1 < c_old - 1e-15:
                h[j] = 1.0
                imp = True
        if not imp:
            break
    return h

def _coord_frac(h0, n, target, rounds=5):
    h = h0.copy()
    vals = [0.0, 0.25, 0.5, 0.75, 1.0]
    for _ in range(rounds):
        imp = False
        for j in range(n):
            old = h[j]
            c_old = _c5_of(h)
            best_c = c_old
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-12:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                imp = True
        if not imp:
            break
    return h

def _continuation(h0, n, target, total_iters, alpha_list, topK_list):
    dx = 2.0 / n
    x = h0.copy()
    best = x.copy()
    best_v = _c5_of(x)
    n_phases = len(alpha_list)
    iters_per = max(1, total_iters // n_phases)
    for p in range(n_phases):
        sa = alpha_list[p]
        tk = topK_list[p]
        a0 = 0.35 / (1.0 + 0.3 * p)
        h_r, c5_r = _nesterov(x, n, target, iters_per, a0, tk, sa)
        x = h_r
        if c5_r < best_v:
            best_v = c5_r
            best = x.copy()
    return best, best_v

def _perturb_reopt(h0, n, target, rng, n_perturb=5, iters_each=200):
    best = h0.copy()
    best_v = _c5_of(h0)
    for _ in range(n_perturb):
        h_p = h0.copy()
        n_flip = rng.randint(3, 8)
        indices = rng.choice(n, size=n_flip, replace=False)
        for i in indices:
            h_p[i] = 1.0 - h_p[i]
        h_p = _fix_sum(h_p, target)
        h_r, c5_r = _nesterov(h_p, n, target, iters_each, 0.2, 5, 50.0)
        if c5_r < best_v:
            best_v = c5_r
            best = h_r.copy()
    return best, best_v

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    n = n_init
    target = n / 2.0
    h_curr = _fix_sum(h_init, target)

    # Phase 1: Continuation method (adaptive alpha)
    if time.time() < deadline - 200:
        alpha_seq = [10.0, 25.0, 50.0, 100.0, 200.0]
        topK_seq = [15, 10, 8, 5, 3]
        h_c, c5_c = _continuation(h_curr, n, target, 1500, alpha_seq, topK_seq)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
        h_curr = h_c

    # Phase 2: Nesterov with fine tuning
    if time.time() < deadline - 180:
        h_r, c5_r = _nesterov(h_curr, n, target, 400, 0.15, 3, 200.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        h_curr = h_r

    # Phase 3: Momentum GD for different basin
    if time.time() < deadline - 160:
        h_m, c5_m = _momentum_gd(h_curr, n, target, 400, 0.2, 5, 80.0, beta=0.9)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()
        h_curr = h_m

    # Phase 4: Coordinate descent binary
    if time.time() < deadline - 100:
        h_cd = _coord_binary(h_curr, n, target, rounds=25)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        h_curr = h_cd

    # Phase 5: Nesterov from CD result
    if time.time() < deadline - 100:
        h_r, c5_r = _nesterov(h_curr, n, target, 300, 0.12, 5, 150.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        h_curr = h_r

    # Phase 6: Coord binary again
    if time.time() < deadline - 60:
        h_cd2 = _coord_binary(h_curr, n, target, rounds=20)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
        h_curr = h_cd2

    # Phase 7: Perturbation to escape local optima
    if time.time() < deadline - 80:
        h_p, c5_p = _perturb_reopt(h_curr, n, target, rng, n_perturb=6, iters_each=200)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
        h_curr = h_p

    # Phase 8: Final Nesterov + CD
    if time.time() < deadline - 50:
        h_r, c5_r = _nesterov(h_curr, n, target, 300, 0.1, 3, 300.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        h_curr = h_r

    if time.time() < deadline - 30:
        h_cd3 = _coord_binary(h_curr, n, target, rounds=20)
        c5_cd3 = _c5_of(h_cd3)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Phase 9: Try nearby resolutions
    n_opts = [n_init + d for d in [-10, -5, 5, 10, 15, -15]
              if (n_init + d) % 2 == 0 and 60 <= (n_init + d) <= 180]
    seen = set()
    n_opts = [x for x in n_opts if x not in seen and not seen.add(x)]

    for nt in n_opts:
        if time.time() > deadline - 30:
            break
        tgt = nt / 2.0
        h_int = np.interp(np.linspace(0, n_init - 1, nt), np.arange(n_init), best_h)
        h_int = _fix_sum(np.clip(h_int, 0.0, 1.0), tgt)
        h_r, c5_r = _nesterov(h_int, nt, tgt, 250, 0.2, 5, 80.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt

    # Phase 10: Final polish on best solution
    nf = best_n
    tf = nf / 2.0

    if time.time() < deadline - 10:
        h_cd_f = _coord_binary(best_h, nf, tf, rounds=15)
        c5_f = _c5_of(h_cd_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 5:
        h_n_f = _nesterov(best_h, nf, tf, 150, 0.08, 5, 500.0)
        c5_n = _c5_of(h_n_f)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n_f.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
