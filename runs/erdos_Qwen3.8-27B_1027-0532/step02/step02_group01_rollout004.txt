import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_binary(h, target_ones):
    """Project to binary with exactly target_ones ones."""
    n = len(h)
    h = h.copy()
    # Sort by value descending, take top target_ones
    order = np.argsort(-h)
    result = np.zeros(n)
    result[order[:target_ones]] = 1.0
    return result

def project_continuous(h, target_sum):
    """Project to [0,1] with sum = target_sum using bisection on Lagrange multiplier."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-12):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_continuous(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_continuous(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def multi_swap_local_search(h_bin, n_iterations, t_start, t_end, rng, max_swaps=3):

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_continuous(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_ones):
    order = np.argsort(-h)
    result = np.zeros(len(h))
    result[order[:target_ones]] = 1.0
    return result

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def smooth_objective(h, eps, top_k=20):
    """Log-sum-exp on top-k overlaps for better conditioning."""
    overlaps = get_overlaps(h)
    # Use top-k for stability
    if top_k is not None and len(overlaps) > top_k:
        idx = np.argpartition(-overlaps, top_k)[:top_k]
        vals = overlaps[idx]
    else:
        vals = overlaps
    cmax = np.max(vals)
    x = (vals - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

    overlaps = get_overlaps(h)
    cmax = np.max(overlaps)
    x = (overlaps - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient_full(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)

        s = k - (n - 1)

        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

min_{h,t} t + λ * (sum(h) - n/2)
subject to f_k(h) ≤ t for all k in active, 0 ≤ h ≤ 1

This is a QP (quadratic objective and constraints) if f_k is quadratic. But f_k(h) = sum h[i]*(1-h[i+s]) is not a standard quadratic form...

Actually, f_k(h) = sum h[i] - sum h[i]*h[i+s] for valid i. The term sum h[i]*h[i+s] is a quadratic form but not necessarily positive/negative definite.

So the QP might not be convex. But we can still try to solve it numerically.

Alternatively, we can linearize: for a given h, f_k(h) ≤ t is equivalent to sum h[i] - t ≤ sum h[i]*h[i+s]. The RHS is the "agreement" at shift s.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_continuous(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_ones):
    order = np.argsort(-h, kind='stable')
    result = np.zeros(len(h))
    result[order[:target_ones]] = 1.0
    return result

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def logsumexp_objective(h, eps):
    ov = get_overlaps(h)
    cmax = np.max(ov)
    x = (ov - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def logsumexp_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def cons_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def cons_jac(h):
    return np.ones(len(h))

def slsqp(h_init, eps, maxiter, ftol=1e-13):
    n = len(h_init)
    ts = n / 2.0
    h0 = project_continuous(h_init.astype(float).copy(), ts)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(logsumexp_objective, h0, args=(eps,),
                       jac=logsumexp_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_continuous(res.x, ts)
        return ho, compute_c5(ho)
    except Exception:
        return h0, compute_c5(h0)

def binary_sa(h_bin, n_iter, t0, t1, rng, move_weights=None):
    n = len(h_bin)
    h = h_bin.copy()
    cur = compute_c5(h)
    bh = h.copy()
    bc = cur
    if move_weights is None:
        move_weights = [0.35, 0.25, 0.20, 0.10, 0.10]
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(n_iter - 1, 1))
        mt = rng.random()
        cum = 0.0
        chosen = 0
        for m in range(5):
            cum += move_weights[m]
            if mt < cum:
                chosen = m
                break
        new_c = cur
        acc = False
        if chosen == 0:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            new_c = compute_c5(h)
            if new_c < cur or rng.random() < np.exp(-(new_c - cur) / max(t, 1e-12)):
                acc = True
            else:
                h[a] = 1.0
                h[b] = 0.0
        elif chosen == 1:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            a1 = ones[rng.integers(len(ones))]
            a2 = ones[rng.integers(len(ones))]
            b1 = zeros[rng.integers(len(zeros))]
            b2 = zeros[rng.integers(len(zeros))]
            if a1 == a2 or b1 == b2:
                continue
            h[a1] = 0.0
            h[a2] = 0.0
            h[b1] = 1.0
            h[b2] = 1.0
            new_c = compute_c5(h)
            if new_c < cur or rng.random() < np.exp(-(new_c - cur) / max(t, 1e-12)):
                acc = True
            else:
                h[a1] = 1.0
                h[a2] = 1.0
                h[b1] = 0.0
                h[b2] = 0.0
        elif chosen == 2:
            mx = max(3, n // 10)
            bl = rng.integers(2, mx + 1)
            a = rng.integers(0, n - bl + 1)
            b = rng.integers(0, n - bl + 1)
            if abs(a - b) < bl:
                continue
            ba = h[a:a+bl].copy()
            bb = h[b:b+bl].copy()
            h[a:a+bl] = bb
            h[b:b+bl] = ba
            new_c = compute_c5(h)
            if new_c < cur or rng.random() < np.exp(-(new_c - cur) / max(t, 1e-12)):
                acc = True
            else:
                h[a:a+bl] = ba
                h[b:b+bl] = bb
        elif chosen == 3:
            mx = max(4, n // 8)
            bl = rng.integers(3, mx + 1)
            a = rng.integers(0, n - bl + 1)
            rot = rng.integers(1, bl)
            blk = h[a:a+bl].copy()
            h[a:a+bl] = np.roll(blk, rot)
            new_c = compute_c5(h)
            if new_c < cur or rng.random() < np.exp(-(new_c - cur) / max(t, 1e-12)):
                acc = True
            else:
                h[a:a+bl] = blk
        else:
            ones = np.where(h == 1.0)[0]
            if len(ones) == 0:
                continue
            a = ones[rng.integers(len(ones))]
            d = rng.integers(1, n // 5 + 1)
            ddir = rng.choice([-1, 1])
            b = a + ddir * d
            if 0 <= b < n and h[b] == 0.0:
                h[a] = 0.0
                h[b] = 1.0
                new_c = compute_c5(h)
                if new_c < cur or rng.random() < np.exp(-(new_c - cur) / max(t, 1e-12)):
                    acc = True
                else:
                    h[a] = 1.0
                    h[b] = 0.0
        if acc:
            cur = new_c
            if cur < bc:
                bc = cur
                bh = h.copy()
    return bh, bc

def multi_start_search(h_best, n_starts, time_per_start, rng, n0):
    best_h = h_best.copy()
    best_c = compute_c5(h_best)
    ts = n0 / 2.0
    for s in range(n_starts):
        if time.time() > time_per_start:
            break
        strat = s % 6
        if strat == 0:
            hs = best_h + 0.03 * rng.standard_normal(n0)
            hs = project_continuous(hs, ts)
        elif strat == 1:
            hs = (rng.random(n0) < 0.5).astype(float)
            hs = project_binary(hs, n0 // 2)
        elif strat == 2:
            hs = best_h.copy()
            bl = rng.integers(5, n0 // 4)
            a = rng.integers(0, n0 - bl)
            hs[a:a+bl] = 1.0 - hs[a:a+bl]
            hs = project_continuous(hs, ts)
        elif strat == 3:
            al = rng.uniform(0.3, 0.6)
            hr = (rng.random(n0) < 0.5).astype(float)
            hr = project_binary(hr, n0 // 2)
            hs = al * best_h + (1.0 - al) * hr
            hs = project_continuous(hs, ts)
        elif strat == 4:
            hs = project_binary(best_h, n0 // 2)
            nf = rng.integers(3, 12)
            idx = rng.choice(n0, size=nf, replace=False)
            hs[idx] = 1.0 - hs[idx]
            hs = project_binary(hs, n0 // 2)
        else:
            hs = best_h.copy()
            nb = rng.integers(2, 4)
            for _ in range(nb):
                bl = rng.integers(3, n0 // 6)
                a = rng.integers(0, n0 - bl)
                b = rng.integers(0, n0 - bl)
                if abs(a - b) >= bl:
                    hs[a:a+bl], hs[b:b+bl] = hs[b:b+bl].copy(), hs[a:a+bl].copy()
            hs = project_continuous(hs, ts)

        hb = project_binary(hs, n0 // 2)
        nb_iter = 15000
        hb, bc = binary_sa(hb, nb_iter, 0.3, 0.003, rng)
        if bc < best_c:
            best_c = bc
            best_h = hb.copy()

        hc = hs + 0.02 * rng.standard_normal(n0)
        hc = project_continuous(hc, ts)
        try:
            hc, cc = slsqp(hc, 0.005, 150)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()
        except Exception:
            pass
    return best_h, best_c
