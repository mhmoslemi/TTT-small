    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) * dx
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    logsum = c_max * dx + np.log(np.sum(exp_vals)) / alpha
    p = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n + 1
        i_min = max(0, -d)
        i_max = min(n - 1, n - 1 - d)
        if i_min > i_max:

        grad[i_min:i_max+1] += -pk * h[i_min+d:i_max+d+1]

            mask = h < 1.0 - 1e-15
        else:

        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_descent(h, target_sum, alpha_schedule, eta0, decay, n_iter_per_alpha):

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) * dx
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    logsum = c_max * dx + np.log(s) / alpha
    p = exp_vals / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        d = k - n + 1
        i_min = max(0, -d)
        i_max = min(n - 1, n - 1 - d)
        if i_min > i_max:
            continue
        m1_s = i_min + d
        m1_e = i_max + d
        if m1_s <= m1_e:
            grad[m1_s:m1_e + 1] += pk * (1.0 - h[m1_s - d:m1_e - d + 1])
        grad[i_min:i_max + 1] -= pk * h[i_min + d:i_max + d + 1]
    grad *= dx
    return logsum, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_smooth_step(h, target, alpha, maxiter):
    n = len(h)

    def obj(x):
        v, _ = smooth_obj_and_grad(x, alpha)
        return v

    def grad(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h


def projected_gradient(h, target, alpha, n_iter, eta0):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = eta0
    for _ in range(n_iter):
        _, grad = smooth_obj_and_grad(h, alpha)
        g_central = grad - np.mean(grad)
        h_try = project_feasible(h - eta * g_central, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        if c5_try <= compute_c5(h):
            h = h_try
            eta = min(eta * 1.05, 0.1)
        else:
            eta *= 0.5
            if eta < 1e-8:
                break
    return best_h, best_c5


def nesterov_projected_gradient(h, target, alpha, n_iter, eta0):
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    eta = eta0
    for it in range(n_iter):
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = h_curr + momentum * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, grad = smooth_obj_and_grad(y, alpha)
        g_central = grad - np.mean(grad)
        h_new = project_feasible(y - eta * g_central, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= compute_c5(h_curr) + 1e-12:
            h_prev = h_curr
            h_curr = h_new
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.5
            if eta < 1e-9:
                break
        t = t_new
    return best_h, best_c5
