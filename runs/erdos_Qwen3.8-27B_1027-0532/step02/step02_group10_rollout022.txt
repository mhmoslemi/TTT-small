            if s_lb < best_score:
                best_score = s_lb
                best_h = h_lb.copy()
        except Exception:
            pass

    # Phase 6: Higher resolution (200s)
    for n_try in [n + 75, 2 * n, 3 * n]:
        if n_try > 2500 or remaining() < 60:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init_hr = np.interp(x_new, x_old, best_h)
        h_init_hr = project_onto_constraint(h_init_hr, n_try / 2.0)
        for alpha_hr in [100, 500, 2000]:
            if remaining() < 50:
                break
            dl7 = time.time() + 30
            h_hr = pgd_optimize(h_init_hr, alpha_hr, n_try / 2.0, 6000, dl7)
            h_hr = project_onto_constraint(h_hr, n_try / 2.0)
            s_hr = evaluate_score(h_hr)
            if s_hr < best_score:
                best_score = s_hr
                best_h = h_hr.copy()
                h_init_hr = h_hr.copy()

    # Phase 7: Final high-alpha polish (80s)
    for alpha in [10000, 20000, 50000]:
        if remaining() < 20:
            break
        nf = len(best_h)
        dl8 = time.time() + 15
        h_final = pgd_optimize(best_h, alpha, nf / 2.0, 5000, dl8)
        h_final = project_onto_constraint(h_final, nf / 2.0)
        s_final = evaluate_score(h_final)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final.copy()

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
