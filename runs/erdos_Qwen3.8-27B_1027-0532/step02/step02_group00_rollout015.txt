The problem is: minimize max_k sum_n h[n+s_k] * (1-h[n]) * dx subject to sum h = n/2, 0<=h<=1.

One observation: if h is binary (0 or 1), then h*(1-h[s]) is 1 when h=1 and h[s]=0, and 0 otherwise. So the overlap counts the number of 1s in h that have a 0 at the shifted position.

For a binary h with exactly n/2 ones, the overlap at shift s is:
number of i where h[i]=1 and h[i+s]=0 (for appropriate range)

This is related to the autocorrelation of the binary sequence.

If h is "random" binary (each position independently 0 or 1 with prob 1/2), the expected overlap at each shift is about n/4 (half the 1s have a 0 neighbor). The C5 would be about (n/4) * (2/n) = 1/2 = 0.5.

To minimize the maximum, we want the binary sequence to be as "balanced" as possible at all shifts. This is related to the concept of "Barker sequences" or "low autocorrelation sequences."

But we're not restricted to binary solutions - we can use fractional values. The optimal solution is likely to have values close to 0 or 1 for most positions, with some intermediate values.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        lo, hi = 0.0, 1.0
        for _ in range(64):
            mid = (lo + hi) * 0.5
            val = np.sum(np.minimum(h + mid, 1.0))
            if val < target_sum:
                lo = mid
            else:
                hi = mid
        h = np.minimum(h + (lo + hi) * 0.5, 1.0)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(64):
            mid = (lo + hi) * 0.5
            val = np.sum(np.maximum(h - mid, 0.0))
            if val > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.maximum(h - (lo + hi) * 0.5, 0.0)
    return np.clip(h, 0.0, 1.0)

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    shifts = np.arange(nc) - (n - 1)
    
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        s = shifts[k]
        # First term: (1-h[j-s]) for valid j

        lo = max(0, -s) + s

        if lo2 <= hi2:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / alpha

def pgd_optimize(h_start, n, target, alpha, n_iter, lr0, momentum=0.9):
    """Projected gradient descent with momentum."""
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = lr0
    
    for it in range(n_iter):
        grad = compute_smooth_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        # Backtracking if objective increased
        cur_val = smooth_value(h_new, alpha)
        old_val = smooth_value(h, alpha)
        if cur_val > old_val + 1e-10:
            lr = max(lr * 0.5, 1e-10)
        else:
            lr = min(lr * 1.05, lr0)
        
        h = h_new
    
    return best_h, best_val

def coordinate_descent(h, target, n_iter=100):
    """Coordinate descent: for each j, find optimal h[j]."""
    n = len(h)
    h = h.copy()
    best_val = compute_c5(h)

        improved = False
        for j in range(n):

            # where b_k = dcorr[k]/dh[j]

            #   0 <= h[j] <= 1

            # we need to adjust other elements. This makes it more complex.
            # Let's just try a 1D search: for each candidate h[j] value,
            # project and evaluate.
            
            old_val = compute_c5(h)
            
            # Try a few values

            best_v = old_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        lo, hi = 0.0, 1.0
        for _ in range(64):
            mid = (lo + hi) * 0.5
            if np.sum(np.minimum(h + mid, 1.0)) < target_sum:
                lo = mid
            else:
                hi = mid
        h = np.minimum(h + (lo + hi) * 0.5, 1.0)
    else:
        lo, hi = 0.0, 1.0

            mid = (lo + hi) * 0.5
            if np.sum(np.maximum(h - mid, 0.0)) > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.maximum(h - (lo + hi) * 0.5, 0.0)
    return h

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    shifts = np.arange(nc) - (n - 1)
    
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    total = np.sum(exp_s)
    w = exp_s / total
    
    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        s = shifts[k]
        # First term: (1-h[j-s]) where n=j-s must be valid
        # Valid n range: max(0,-s) to min(n-1, n-1-s)
        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)
        # j = n + s, so j ranges from n_lo+s to n_hi+s
        j_lo = max(0, n_lo + s)
        j_hi = min(n - 1, n_hi + s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wk * (1.0 - h[n_lo:n_hi+1])
        
        # Second term: -h[j+s] where n=j must be valid
        j_lo2 = max(0, n_lo)
        j_hi2 = min(n - 1, n_hi)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += wk * (-h[n_lo+s:n_hi+s+1])
    
    return grad

def smooth_value(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / alpha

def pgd_step(h, grad, momentum_vec, lr, mu, target):
    momentum_vec[:] = mu * momentum_vec - lr * grad
    h_new = h + momentum_vec
    return project_feasible(h_new, target)

def pgd_optimize(h_start, target, alpha, n_iter, lr_init, mu=0.9, time_limit=None, start_time=None):
    n = len(h_start)
    h = h_start.copy()
    mom = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stagnation = 0

                break
        
        grad = compute_smooth_grad(h, alpha)
        h = pgd_step(h, grad, mom, lr, mu, target)

            best_h = h.copy()
            stagnation = 0

            stagnation += 1

    return best_h, best_c5
