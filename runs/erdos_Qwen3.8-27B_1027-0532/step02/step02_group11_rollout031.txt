    x_new = x.copy()

    best_v = eval_c5(h)

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        l = k_star - (n - 1)
        # Compute subgradient
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            jml = j - l
            if 0 <= jml < n:
                g += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                g += (-h[jpl])
            grad[j] = g
        grad *= (2.0 / n)
        alpha = c_alpha / np.sqrt(float(it))
        h_new = h - alpha * grad
        h_new = proj(h_new, n)
        v = eval_c5(h_new)
        h = h_new
        if v < best_v:
            best_v, best = v, h.copy()
    return best, best_v

def proj(x, n):
    t = n * 0.5
    lo, hi = -1.0, 2.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(x - m, 0.0, 1.0)) > t:
            lo = m
        else:
            hi = m
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def binarize(h, n):
    t = n // 2
    b = (h >= 0.5).astype(float)

        order = idx[np.argsort(h[idx])]
        for i in range(c - t):
            b[order[i]] = 0.0
    elif c < t:
        idx = np.where(b == 0)[0]
        order = idx[np.argsort(-h[idx])]
        for i in range(t - c):
            b[order[i]] = 1.0
    return b
