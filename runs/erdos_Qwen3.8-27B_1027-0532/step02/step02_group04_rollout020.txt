def coordinate_descent_fast(h, n, n_passes=100, grid_size=11):
    """Fast coordinate descent with incremental correlation updates."""
    best = h.copy()
    corr = compute_correlations(best)
    best_score = np.max(corr)
    
    for p in range(n_passes):
        for j in range(n):
            # B[s] = h[j+s] + h[j-s] (out of range = 0)
            # delta_corr(s) = delta * (1 - B[s])
            
            old_val = best[j]
            # Compute B
            B = np.zeros(2*n - 1)
            # B[s_idx] where s = s_idx - (n-1)
            # B = h[j+s] + h[j-s]
            # For s_idx from 0 to 2n-2, s = s_idx - (n-1)
            s_range = np.arange(-(n-1), n)
            # j+s in range [0,n-1]'s: s in [-j, n-1-j]
            # j-s in range [0,n-1]'s: s in [j-(n-1), j]
            
            # h[j+s] for valid s:
            valid_js = (s_range >= -j) & (s_range <= n-1-j)
            valid_jms = (s_range >= j-(n-1)) & (s_range <= j)
            
            B[valid_js] += best[j + s_range[valid_js]]
            B[valid_jms] += best[j - s_range[valid_jms]]
            
            # A = corr + B * old_val
            A = corr + B * old_val
            
            # Minimize max(A - B*t) for t in [0,1]
            # The function is convex piecewise linear
            # Use grid search
            t_grid = np.linspace(0, 1, grid_size)
            vals = np.max(A[:, None] - B[:, None] * t_grid[None, :], axis=0)
            best_t = t_grid[np.argmin(vals)]
            
            new_val = best_t
            if new_val != old_val:
                delta = new_val - old_val
                corr += delta * (1.0 - B)
                best[j] = new_val
                curr_max = np.max(corr)
                # Track best score
                if curr_max < best_score:
                    best_score = curr_max
    
    best_score_final = np.max(compute_correlations(best))
    return best, best_score_final

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def coordinate_descent(h, n, n_passes=100, grid_size=21, mu=1000.0):
    """Fast coordinate descent with incremental correlation updates."""
    best = h.copy()
    corr = compute_correlations(best)
    current_sum = np.sum(best)
    
    for p in range(n_passes):
        for j in range(n):
            # Compute B[s] = h[j+s] + h[j-s]
            # Using the shift index convention: s_idx = 0 to 2n-2, shift s = s_idx - (n-1)
            B = np.zeros(2 * n - 1)
            s_vals = np.arange(-(n - 1), n)
            
            # h[j+s] valid where 0 <= j+s < n
            mask_js = (j + s_vals >= 0) & (j + s_vals < n)
            B[mask_js] += best[j + s_vals[mask_js]]
            
            # h[j-s] valid where 0 <= j-s < n
            mask_jms = (j - s_vals >= 0) & (j - s_vals < n)
            B[mask_jms] += best[j - s_vals[mask_jms]]
            
            # A = corr + B * best[j]
            A = corr + B * best[j]
            
            # 1D optimization: minimize max(A - B*t) + mu*(current_sum + t - best[j] - n/2)²
            sum_others = current_sum - best[j]
            t_target = n / 2.0 - sum_others  # target for t to satisfy sum constraint
            
            # Grid search
            t_grid = np.linspace(0.0, 1.0, grid_size)
            # Compute max(A - B*t) for each t
            obj_vals = np.zeros(grid_size)
            for idx in range(grid_size):
                t = t_grid[idx]
                obj_vals[idx] = np.max(A - B * t) + mu * (sum_others + t - n / 2.0) ** 2
            
            best_t = t_grid[np.argmin(obj_vals)]
            
            if best_t != best[j]:
                delta = best_t - best[j]
                corr += delta * (1.0 - B)
                current_sum += delta
                best[j] = best_t
    
    return best, np.max(compute_correlations(best))

def ljbfgs_optimize(h0, alpha, n, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_shifted = np.max(shifted)
        sm = m + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s
    
    def grad(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_shifted = np.max(shifted)
        exp_shifted = np.exp(shifted - max_shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        
        grad = np.zeros(n)
        N = n
        for index in range(len(weights)):
            w = weights[index]
            if w < 1e-12:
                continue
            lag = N - 1 - index
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                if lag < N:
                    grad[:N - lag] += w
                    grad[:N - lag] -= w * h[lag:]
                    grad[lag:] -= w * h[:N - lag]
            else:
                l = -lag
                if l < N:
                    grad[l:] += w
                    grad[l:] -= w * h[:N - l]
                    grad[:N - l] -= w * h[l:]
        
        s = np.sum(h) - n / 2.0
        grad += 2.0 * mu * s
        return grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -(np.max(h) if len(h) > 0 else 0.0) - 1.0
    hi = (1.0 - np.min(h) if len(h) > 0 else 0.0) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def coord_descent(h, n, n_passes, grid_size=15, mu=1000.0, s_vals=None):
    best = h.copy()
    corr = compute_correlations(best)
    current_sum = np.sum(best)
    t_grid = np.linspace(0.0, 1.0, grid_size)
    
    if s_vals is None:
        s_vals = np.arange(-(n - 1), n)
    
    for p in range(n_passes):
        for j in range(n):
            # Compute B = h[j+s] + h[j-s]
            B = np.zeros(2 * n - 1)
            js = j + s_vals
            mask = (js >= 0) & (js < n)
            if np.any(mask):
                B[mask] += best[js[mask]]
            jms = j - s_vals
            mask2 = (jms >= 0) & (jms < n)
            if np.any(mask2):
                B[mask2] += best[jms[mask2]]
            
            A = corr + B * best[j]
            sum_others = current_sum - best[j]
            
            # Grid search for optimal t
            best_val = np.inf
            best_t = best[j]
            for idx in range(grid_size):
                t = t_grid[idx]
                penalty = mu * (sum_others + t - n / 2.0) ** 2
                val = np.max(A - B * t) + penalty
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if best_t != best[j]:
                delta = best_t - best[j]
                corr += delta * (1.0 - B)
                current_sum += delta
                best[j] = best_t
    
    return best, np.max(corr)

def ljbfgs(h0, alpha, n, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        ms = np.max(shifted)
        sm = m + np.log(np.sum(np.exp(shifted - ms))) / alpha
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s
    
    def grad(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        ms = np.max(shifted)
        ev = np.exp(shifted - ms)
        w = ev / np.sum(ev)
        g = np.zeros(n)
        N = n
        for idx in range(len(w)):
            wi = w[idx]
            if wi < 1e-12:
                continue
            lag = N - 1 - idx
            if lag == 0:
                g += wi * (1.0 - 2.0 * h)
            elif lag > 0:
                if lag < N:
                    g[:N-lag] += wi
                    g[:N-lag] -= wi * h[lag:]
                    g[lag:] -= wi * h[:N-lag]
            else:
                l = -lag
                if l < N:
                    g[l:] += wi
                    g[l:] -= wi * h[:N-l]
                    g[:N-l] -= wi * h[l:]
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x
