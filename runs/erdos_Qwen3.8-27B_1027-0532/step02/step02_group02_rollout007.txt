import time
import numpy as np


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _project_box_sum(x, target_sum):
    n = len(x)
    clipped = np.clip(x, 0.0, 1.0)
    if abs(clipped.sum() - target_sum) < 1e-10:
        return clipped
    lo = x.min() - 1.5
    hi = x.max() + 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h


def _objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400, 500)
    exps = np.exp(shifted)
    w_sum = exps.sum()
    weights = exps / w_sum
    obj = (c_max + np.log(w_sum) / alpha) * dx
    grad = np.zeros(n)
    for i in range(len(corr)):
        w = weights[i]
        if w < 1e-12:
            continue
        lag = i - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] -= w * h[lag:]
            grad[lag:] += w * (1.0 - h[:n - lag])
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return obj, grad, corr


def _optimize_pgm(h0, deadline, alpha0=40.0, alpha1=500.0,
                  lr=0.12, momentum=0.82, n_iter=40000,
                  patience=6000, check_every=150):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    lr_decay = 0.00015

    for t in range(n_iter):
        if time.time() > deadline - 3:
            break
        frac = min(1.0, t / (n_iter * 0.35))
        alpha = alpha0 + (alpha1 - alpha0) * (frac * frac)
        curr_lr = lr / (1.0 + t * lr_decay)

        obj, grad, _ = _objective_and_grad(h, alpha)
        v = momentum * v - curr_lr * grad
        h = _project_box_sum(h + v, target_sum)

        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve >= patience:
                break

    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def _subgrad_active(h0, deadline, n_iter=15000, lr=0.04, top_k=3,
                    check_every=200, patience=3000):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0

    for t in range(n_iter):
        if time.time() > deadline - 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for i in idx:
            lag = i - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif lag > 0:
                grad[:n - lag] -= h[lag:]
                grad[lag:] += (1.0 - h[:n - lag])
            else:
                L = -lag
                grad[:n - L] += (1.0 - h[L:])
                grad[L:] -= h[:n - L]
        grad /= len(idx)
        h = _project_box_sum(h - lr * grad, target_sum)

        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve >= patience:
                break

    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = _project_box_sum(h_new, target)
    return h_new


def _downsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = _project_box_sum(h_new, target)
    return h_new


def _make_initializations(n, seed):
    inits = []
    rng = np.random.RandomState(seed)
    target = n / 2.0
    x = np.linspace(0, 2, n, endpoint=False)

    h1 = np.full(n, 0.5)
    inits.append(_project_box_sum(h1, target))

    h2 = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    inits.append(_project_box_sum(h2, target))

    h3 = 0.5 + 0.3 * np.sin(4 * np.pi * x) + 0.1 * np.sin(6 * np.pi * x)
    inits.append(_project_box_sum(h3, target))

    period = max(3, n // 5)
    h4 = np.zeros(n)
    for i in range(n):
        h4[i] = 1.0 if (i % (2 * period)) < period else 0.0
    h4 += 0.08 * rng.randn(n)
    inits.append(_project_box_sum(h4, target))

    h5 = rng.rand(n)
    inits.append(_project_box_sum(h5, target))

    h6 = 0.5 + 0.25 * np.sin(3 * np.pi * x) + 0.1 * rng.randn(n)
    inits.append(_project_box_sum(h6, target))

    return inits


def _perturbation_refine(h_best, n, deadline, seed, n_trials=6):
    target = n / 2.0
    rng = np.random.RandomState(seed)
    best = h_best.copy()
    best_c5 = _c5_of(h_best)

    for trial in range(n_trials):
        if time.time() > deadline - 15:
            break
        scale = 0.03 + 0.05 * rng.rand()
        h_p = _project_box_sum(h_best + scale * rng.randn(n), target)
        h_r, c5_r = _optimize_pgm(h_p, deadline, alpha0=200.0, alpha1=800.0,
                                  lr=0.06, momentum=0.85, n_iter=8000,
                                  patience=3000, check_every=100)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best = h_r.copy()

    return best, best_c5
