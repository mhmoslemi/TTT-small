    """Find optimal value of h[j] given all other coordinates fixed."""
    dx_unused = 2.0 / n  # not needed since we're maximizing corr, not C5
    
    # Compute current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # s = k - (n-1) for k in range(0, 2n-1)
    s = np.arange(2*n - 1) - (n - 1)  # shape (2n-1,)

        k_star = np.argmax(vals)

        elif grad < -1e-12:
            lo = mid  # function decreasing at mid, min is to the right
        else:
            break  # found the minimum

        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            noise_scale = 0.005 * min(1.0, no_improve / patience)
            h = best_h + rng.randn(n) * noise_scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0

    return best_h, best_score

def subgradient_descent(h, target_sum, max_time, rng):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = h.copy()
    h = project_feasible(h, target_sum)
    best_score = compute_c5(h)
    best_h = h.copy()
    lr = 0.002
    elapsed_start = time.time()
    stagnant = 0

    while time.time() - elapsed_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx

        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 2000:
                break

        active_mask = corr > c_max - 1e-6
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            active_mask = np.zeros(2 * n - 1)
            active_mask[np.argmax(corr)] = 1
            n_active = 1

        g = np.zeros(n)
        for idx in np.where(active_mask)[0]:
            s_val = s_arr[idx]
            js = np.arange(n) - s_val
            js_plus = np.arange(n) + s_val
            contrib = np.zeros(n)
            m1 = (js >= 0) & (js < n)
            m2 = (js_plus >= 0) & (js_plus < n)
            contrib[m1] += (1.0 - h[js[m1]])
            contrib[m2] -= h[js_plus[m2]]
            g += contrib / n_active

        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        score_new = compute_c5(h_new)

        if score_new < score:
            h = h_new
            lr = min(lr * 1.05, 0.01)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_h, best_score

def smooth_objective(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def cons_fun(h, target_sum):
    return float(np.sum(h) - target_sum)

def cons_jac(n):
    return np.ones(n)

def slsqp_refine(h, target_sum, alphas, maxiter_each, time_limit):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            def obj(x, a=alpha):
                return smooth_objective(x, a)

            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x: cons_fun(x, target_sum),
                                        'jac': lambda x: cons_jac(n)}],
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass

    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent at initial resolution
    if tleft() > 200:
        h_cd, sc_cd = coord_descend_fast(best_h, target_init, min(350, tleft() - 200), rng, patience=25)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 2: Subgradient descent for fine tuning
    if tleft() > 150:
        h_sg, sc_sg = subgradient_descent(best_h, target_init, min(100, tleft() - 150), rng)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 3: SLSQP refinement with high alpha
    if tleft() > 120:
        alphas = [100, 300, 800, 1500]
        h_sl, sc_sl = slsqp_refine(best_h, target_init, alphas, 400, max(30, tleft() - 120))
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 4: Multi-resolution
    n_attempts = [n_init + 100, n_init + 200, n_init - 100, n_init + 300]
    n_attempts = [n for n in n_attempts if n != n_init and n >= 100]
    for n_try in n_attempts:
        if tleft() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cd2, sc_cd2 = coord_descend_fast(h_interp, target_try, min(80, tleft() - 120), rng, patience=15)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 5: Final coord descent on best solution
    n_best = len(best_h)
    target_best = n_best / 2.0
    if tleft() > 60:
        h_final, sc_final = coord_descend_fast(best_h, target_best, min(80, tleft() - 60), rng, patience=30)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 6: Final SLSQP polish
    if tleft() > 40:
        h_pol, sc_pol = slsqp_refine(best_h, target_best, [500, 1000], 300, max(20, tleft() - 40))
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score

    return h_final, best_score, n_final
