        a, b = lo, hi
        best_delta = 0.0
        best_val = current_max
        
        for _ in range(n_bs_iters):

            val = np.max(corr + mid * B)
            if val < best_val - 1e-15:
                best_val = val
                best_delta = mid
            temp = corr + mid * B
            active = temp >= (np.max(temp) - 1e-14)
            min_s = np.min(B[active])

                a = mid

            corr += best_delta * B
            if best_val < current_max:

    hi = 1.0 - np.min(h) + 0.5

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index

                grad[l:] += w

                idx2 = rng.randint(n)
            pert[idx1], pert[idx2] = pert[idx2], pert[idx1]
            pert = project_feasible(pert, n / 2.0)

        pert = project_feasible(pert, n / 2.0)
        h_new = pert.copy()
        n_inner = 30
        for _ in range(n_inner):
            if not remaining(180):
                break
            rng.shuffle(order)
            h_new, _, mx = coord_desc_pass(h_new, order, 30)
            s = mx * (2.0 / n)
            if s < best_score - 1e-12:
                best_score = s
                best_h = h_new.copy()

    # Phase 4: More L-BFGS-B
    for alpha in [5000, 15000, 40000, 100000]:
        if not remaining(150):
            break
        maxiter = max(200, 150000 // n)
        res = ljbfgs_call(best_h, alpha, n, MU, maxiter)
        if res is not None and np.all(np.isfinite(res)):
            h_new = project_feasible(res, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 5: Resolution increase
    current_best = best_h.copy()
    for n_try in [2 * n, 3 * n, 4 * n]:
        if n_try > 2200:
            continue
        if not remaining(200):
            break
        x_old = np.arange(len(current_best)) * 2.0 / len(current_best)
        x_new = np.arange(n_try) * 2.0 / n_try
        h_init = np.interp(x_new, x_old, current_best)
        h_init = project_feasible(h_init, n_try / 2.0)
        s_init = evaluate_score(h_init)

        order_n = np.arange(n_try)

        # CD at new resolution
        n_cd_passes = 800
        for p in range(n_cd_passes):
            if not remaining(150):
                break
            rng.shuffle(order_n)
            h_new = h_init.copy()
            _, corr, mx = coord_desc_pass(h_new, order_n, 35)
            s = mx * (2.0 / n_try)
            if s < best_score - 1e-12:
                best_score = s
                best_h = h_new.copy()
                h_init = h_new.copy()
                current_best = h_new.copy()

        # L-BFGS-B at new resolution
        for alpha in [500, 2000, 8000, 20000]:
            if not remaining(100):
                break
            maxiter = max(100, 300000 // n_try)
            res = ljbfgs_call(h_init, alpha, n_try, MU, maxiter)
            if res is not None and np.all(np.isfinite(res)):
                h_new = project_feasible(res, n_try / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_init = h_new.copy()
                    current_best = h_new.copy()

        # More CD after L-BFGS-B
        for p in range(500):
            if not remaining(80):
                break
            rng.shuffle(order_n)
            h_new = h_init.copy()
            _, corr, mx = coord_desc_pass(h_new, order_n, 35)
            s = mx * (2.0 / n_try)
            if s < best_score - 1e-12:
                best_score = s
                best_h = h_new.copy()
                h_init = h_new.copy()
                current_best = h_new.copy()

    # Phase 6: Final coordinate descent
    n_final = len(best_h)
    order_f = np.arange(n_final)
    for p in range(3000):
        if not remaining(30):
            break
        rng.shuffle(order_f)
        h_new = best_h.copy()
        _, corr, mx = coord_desc_pass(h_new, order_f, 35)
        s = mx * (2.0 / n_final)
        if s < best_score - 1e-12:
            best_score = s
            best_h = h_new.copy()

    # Final projection
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
