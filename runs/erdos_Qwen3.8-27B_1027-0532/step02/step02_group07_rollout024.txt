def optimize_single_coord(h, j, corr):
    """Find optimal h[j] given all other h fixed, to minimize max corr."""
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    
    # Compute slopes: d(corr[k])/d(h[j]) for each k
    slopes = np.zeros(m)
    for k in range(m):
        lag = k - (n - 1)
        slope = 0.0
        if 0 <= j - lag < n:
            slope += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            slope -= h[j + lag]
        slopes[k] = slope
    
    # corr[k] as a function of t = h[j]: corr[k](t) = corr[k] - h_j * slopes[k] + t * slopes[k]
    # = (corr[k] - h_j * slopes[k]) + t * slopes[k]
    a = corr - h_j * slopes
    
    # We want to minimize max_k (a[k] + t * slopes[k]) subject to 0 <= t <= 1
    # This is a convex piecewise-linear function of t
    # Find the minimizer using bisection on the optimal value V
    
    # Lower bound: min over t of max_k (a[k] + t*slopes[k])
    # At the optimum, the max is minimized
    
    # For a given V, is there a t in [0,1] such that a[k] + t*slopes[k] <= V for all k?

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s2 = np.clip(h - mid, 0.0, 1.0).sum()
        if s2 > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    x = np.clip(h - t, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        if interior.sum() > 0:
            x[interior] += diff / interior.sum()
            np.clip(x, 0.0, 1.0, out=x)
    return x


def smooth_max_val(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    ms = np.max(shifted)
    return m + ms / alpha + np.log(np.sum(np.exp(shifted - ms))) / alpha


def compute_grad_h(corr, h, alpha):
    n = len(h)
    m = 2 * n - 1
    m_corr = np.max(corr)
    shifted = alpha * (corr - m_corr)
    ms = np.max(shifted)
    e = np.exp(shifted - ms)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(m):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                if n - 1 - lag >= 0:
                    grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            if L < n:
                grad[:n - L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:n - L]
    return grad


def lbfgs_objective(h, alpha, mu):
    corr = compute_correlations(h)
    val = smooth_max_val(corr, alpha)
    s = np.sum(h) - len(h) / 2.0
    val += 0.5 * mu * s * s
    return val


def lbfgs_gradient(h, alpha, mu):
    corr = compute_correlations(h)
    g = compute_grad_h(corr, h, alpha)
    s = np.sum(h) - len(h) / 2.0
    g = g + mu * s
    return g


def lbfgs_optimize(h0, alpha, mu, maxiter, time_limit):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    best_x = h0.copy()
    best_f = lbfgs_objective(h0, alpha, mu)
    try:
        res = minimize(
            lbfgs_objective, h0, jac=lbfgs_gradient,
            args=(alpha, mu),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'factr': 1e14, 'gtol': 1e-14,
                     'maxfun': maxiter * 3}
        )
        if res.x is not None:
            f_res = lbfgs_objective(res.x, alpha, mu)
            if f_res < best_f:
                best_x = res.x
    except Exception:
        pass
    return best_x


def optimize_single_coord(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    slopes = np.zeros(m)
    for k in range(m):
        lag = k - (n - 1)
        s = 0.0
        if 0 <= j - lag < n:
            s += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            s -= h[j + lag]
        slopes[k] = s
    a = corr - h_j * slopes
    zero_mask = np.abs(slopes) < 1e-14
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    if np.any(zero_mask):
        lo_V = float(np.max(np.concatenate([a[zero_mask], np.minimum(a[pos_mask], a[pos_mask] + slopes[pos_mask])]) if pos_mask.any() else [a[zero_mask]]))
        if neg_mask.any():
            lo_V = max(lo_V, float(np.max(np.minimum(a[neg_mask], a[neg_mask] + slopes[neg_mask]))))
    else:
        parts = []
        if pos_mask.any():
            parts.append(np.minimum(a[pos_mask], a[pos_mask] + slopes[pos_mask]))
        if neg_mask.any():
            parts.append(np.minimum(a[neg_mask], a[neg_mask] + slopes[neg_mask]))
        if parts:
            lo_V = float(np.max(np.concatenate(parts)))
        else:
            lo_V = 0.0
    hi_V = float(np.max(a + h_j * slopes))
    if lo_V >= hi_V - 1e-14:
        return h_j
    for _ in range(50):
        mid_V = 0.5 * (lo_V + hi_V)
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            lo_t = max(lo_t, 0.0)
            hi_t = min(hi_t, 1.0)
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = 0.5 * (lo_V + hi_V)
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = 0.5 * (lo_t + hi_t)
    return float(np.clip(t_star, 0.0, 1.0))


def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt = optimize_single_coord(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                m = 2 * n - 1
                slopes = np.zeros(m)
                for k in range(m):
                    lag = k - (n - 1)
                    s = 0.0
                    if 0 <= j - lag < n:
                        s += (1.0 - h[j - lag])
                    if 0 <= j + lag < n:
                        s -= h[j + lag]
                    slopes[k] = s
                corr = corr + change * slopes
                h[j] = t_opt
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def pair_coordinate_descent(h, n_sweeps=100, tol=1e-12, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    rng = np.random.RandomState(12345)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for _ in range(n):
            i, j = rng.randint(0, n), rng.randint(0, n)
            if i == j:
                continue
            m = 2 * n - 1
            slopes_i = np.zeros(m)
            slopes_j = np.zeros(m)
            for k in range(m):
                lag = k - (n - 1)
                si = 0.0
                sj = 0.0
                if 0 <= i - lag < n:
                    si += (1.0 - h[i - lag])
                if 0 <= i + lag < n:
                    si -= h[i + lag]
                if 0 <= j - lag < n:
                    sj += (1.0 - h[j - lag])
                if 0 <= j + lag < n:
                    sj -= h[j + lag]
                slopes_i[k] = si
                slopes_j[k] = sj
            diff_slope = slopes_i - slopes_j
            a = corr
            lo_delta = max(-h[i], h[j] - 1.0)
            hi_delta = min(1.0 - h[i], h[j])
            if lo_delta >= hi_delta - 1e-14:
                continue
            zero_mask = np.abs(diff_slope) < 1e-14
            pos_mask = diff_slope > 1e-14
            neg_mask = diff_slope < -1e-14
            if np.any(zero_mask):
                lo_V = float(np.max(a[zero_mask]))
            else:
                lo_V = 0.0
            if np.any(pos_mask):
                candidates = a[pos_mask] + hi_delta * diff_slope[pos_mask]
                lo_V = max(lo_V, float(np.max(np.minimum(a[pos_mask], candidates))))
            if np.any(neg_mask):
                candidates = a[neg_mask] + lo_delta * diff_slope[neg_mask]
                lo_V = max(lo_V, float(np.max(np.minimum(a[neg_mask], candidates))))
            hi_V = float(np.max(np.maximum(a + lo_delta * diff_slope, a + hi_delta * diff_slope)))
            if lo_V >= hi_V - 1e-14:
                continue
            for _ in range(45):
                mid_V = 0.5 * (lo_V + hi_V)
                feasible = True
                if np.any(zero_mask):
                    if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                        feasible = False
                if feasible:
                    lo_t = lo_delta
                    if np.any(neg_mask):
                        lo_t = max(lo_t, float(np.max((mid_V - a[neg_mask]) / diff_slope[neg_mask])))
                    hi_t = hi_delta
                    if np.any(pos_mask):
                        hi_t = min(hi_t, float(np.min((mid_V - a[pos_mask]) / diff_slope[pos_mask])))
                    if lo_t > hi_t + 1e-14:
                        feasible = False
                if feasible:
                    hi_V = mid_V
                else:
                    lo_V = mid_V
            V_star = 0.5 * (lo_V + hi_V)
            lo_t = lo_delta
            hi_t = hi_delta
            if np.any(neg_mask):
                lo_t = max(lo_t, float(np.max((V_star - a[neg_mask]) / diff_slope[neg_mask])))
            if np.any(pos_mask):
                hi_t = min(hi_t, float(np.min((V_star - a[pos_mask]) / diff_slope[pos_mask])))
            lo_t = max(lo_t, lo_delta)
            hi_t = min(hi_t, hi_delta)
            if lo_t > hi_t:
                continue
            delta_star = 0.5 * (lo_t + hi_t)
            if abs(delta_star) > 1e-14:
                h[i] = float(np.clip(h[i] + delta_star, 0.0, 1.0))
                h[j] = float(np.clip(h[j] - delta_star, 0.0, 1.0))
                corr = corr + delta_star * diff_slope
                max_change = max(max_change, abs(delta_star))
        if sweep % 30 == 29:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def make_sinusoidal_init(n, freq1, freq2, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(np.pi * freq1 * x + phase) + 0.15 * np.sin(np.pi * freq2 * x)
    return project_sum(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    best_h = project_sum(h_init, n / 2.0)
    best_score = evaluate_score(best_h)

    # Phase 1: CD from initial
    if time_left() > 80:
        h_cd = coordinate_descent(best_h, n_sweeps=400, time_budget=min(70, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: L-BFGS with increasing alpha
    for alpha in [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]:
        if time_left() < 25:
            break
        mu = 5000.0
        h_new = lbfgs_optimize(best_h, alpha, mu, maxiter=300, time_limit=min(20, time_left() - 5))
        if h_new is not None:
            h_new = project_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new

    # Phase 3: CD polish after L-BFGS
    if time_left() > 50:
        h_cd = coordinate_descent(best_h, n_sweeps=300, time_budget=min(40, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 4: Pair CD
    if time_left() > 80:
        h_pc = pair_coordinate_descent(best_h, n_sweeps=200, time_budget=min(60, time_left() - 20))
        s = evaluate_score(h_pc)
        if s < best_score:
            best_score = s
            best_h = h_pc
        h_cd = coordinate_descent(best_h, n_sweeps=200, time_budget=min(20, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 5: Random restarts with CD
    n_restarts = 300
    for i in range(n_restarts):
        if time_left() < 25:
            break
        frac = i / max(n_restarts - 1, 1)
        scale = 0.03 + 0.35 * frac
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_sum(h_pert, n / 2.0)
        tb = min(2.5, max(0.5, (time_left() - 20) / max(n_restarts - i, 1)))
        h_cd = coordinate_descent(h_pert, n_sweeps=150, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            if i < n_restarts - 50:
                if time_left() > 40:
                    h_pc = pair_coordinate_descent(best_h, n_sweeps=50, time_budget=min(15, time_left() - 30))
                    s2 = evaluate_score(h_pc)
                    if s2 < best_score:
                        best_score = s2
                        best_h = h_pc

    # Phase 6: Alternative initializations
    alt_inits = []
    for freq in [1, 2, 3, 4, 5, 6, 8, 10]:
        h_sin = make_sinusoidal_init(n, freq, freq + 1, 0.0)
        alt_inits.append(h_sin)
    for bi in range(3):
        h_bin = np.zeros(n)
        ones_pos = rng.choice(n, size=n // 2, replace=False)
        h_bin[ones_pos] = 1.0
        h_bin = project_sum(h_bin, n / 2.0)
        alt_inits.append(h_bin)

    for h_alt in alt_inits:
        if time_left() < 30:
            break
        tb = min(5.0, max(2.0, (time_left() - 20) / max(len(alt_inits), 1)))
        h_cd = coordinate_descent(h_alt, n_sweeps=150, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            if time_left() > 30:
                h_new = lbfgs_optimize(best_h, 5000, 5000.0, maxiter=200, time_limit=min(20, time_left() - 10))
                if h_new is not None:
                    h_new = project_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)
                    s2 = evaluate_score(h_new)
                    if s2 < best_score:
                        best_score = s2
                        best_h = h_new

    # Phase 7: Higher resolution
    n_base = len(best_h)
    for ratio in [2, 1.5, 3]:
        n_try = int(n_base * ratio)
        if n_try < 200 or n_try > 4000:
            continue
        if time_left() < 100:
            break
        x_old = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)

        tb_cd = min(50, time_left() - 60)
        if tb_cd > 10:
            h_cd = coordinate_descent(h_interp, n_sweeps=200, time_budget=tb_cd)
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd
                if time_left() > 40:
                    for alpha in [1000, 5000, 20000]:
                        if time_left() < 20:
                            break
                        h_new = lbfgs_optimize(best_h, alpha, 5000.0, maxiter=200, time_limit=min(25, time_left() - 10))
                        if h_new is not None:
                            h_new = project_sum(np.clip(h_new, 0.0, 1.0), len(h_new) / 2.0)
                            s2 = evaluate_score(h_new)
                            if s2 < best_score:
                                best_score = s2
                                best_h = h_new
                    if time_left() > 20:
                        h_cd2 = coordinate_descent(best_h, n_sweeps=200, time_budget=min(20, time_left() - 10))
                        s3 = evaluate_score(h_cd2)
                        if s3 < best_score:
                            best_score = s3
                            best_h = h_cd2

    # Phase 8: Final intensive polish
    n_final = len(best_h)
    if time_left() > 60:
        h_cd = coordinate_descent(best_h, n_sweeps=500, tol=1e-13, time_budget=min(50, time_left() - 20))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 40:
        h_pc = pair_coordinate_descent(best_h, n_sweeps=200, time_budget=min(30, time_left() - 15))
        s = evaluate_score(h_pc)
        if s < best_score:
            best_score = s
            best_h = h_pc

    if time_left() > 30:
        for alpha in [20000, 50000, 100000, 200000]:
            if time_left() < 15:
                break
            h_new = lbfgs_optimize(best_h, alpha, 10000.0, maxiter=400, time_limit=min(20, time_left() - 5))
            if h_new is not None:
                h_new = project_sum(np.clip(h_new, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    if time_left() > 15:
        h_cd = coordinate_descent(best_h, n_sweeps=500, tol=1e-14, time_budget=min(12, time_left() - 3))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Final projection
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
