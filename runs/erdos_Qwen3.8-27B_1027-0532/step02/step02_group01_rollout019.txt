```python
def efficient_local_search(h, max_iters=10000):
    n = len(h)
    # h is binary
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    current_max = corr.max()
    
    for _ in range(max_iters):
        best_delta = 0
        best_pair = None
        for a in ones:
            for b in zeros:
                # Compute delta for swap (a, b)
                s = np.arange(-(n-1), n)
                delta = np.zeros(2*n-1)
                # Effect of h[a]: 1->0
                mask_a = (a + s >= 0) & (a + s < n)
                delta[mask_a] += (h[a+s[mask_a]] - 1.0)
                # Effect of h[b]: 0->1 (as first factor)
                mask_b1 = (b + s >= 0) & (b + s < n)
                delta[mask_b1] += (1.0 - h[b+s[mask_b1]])
                # Effect of h[b]: 0->1 (as second factor)
                mask_b2 = (b - s >= 0) & (b - s < n)
                delta[mask_b2] += (-h[b-s[mask_b2]])
                
                new_corr = corr + delta
                new_max = new_corr.max()
                if new_max < current_max:
                    best_delta = new_max - current_max
                    best_pair = (a, b)
                    if best_delta < -0.001:  # early termination for big gains
                        break
            if best_pair and best_delta < -0.001:
                break
        
        if best_pair is None:
            break  # local optimum
        
        a, b = best_pair
        h[a] = 0
        h[b] = 1
        corr += delta  # update correlation
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        current_max = corr.max()
    
    return h, current_max
```

```python
def compute_delta(h, a, b, n):
    s = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Four main terms
    mask_a = (a + s >= 0) & (a + s < n)
    delta[mask_a] += h[a + s[mask_a]] - 1.0
    
    mask_b1 = (b + s >= 0) & (b + s < n)
    delta[mask_b1] += 1.0 - h[b + s[mask_b1]]
    
    mask_a2 = (a - s >= 0) & (a - s < n)
    delta[mask_a2] += h[a - s[mask_a2]]
    
    mask_b2 = (b - s >= 0) & (b - s < n)
    delta[mask_b2] += -h[b - s[mask_b2]]
    
    # Corrections
    if a != b:
        s_ab = a - b
        s_ba = b - a
        # Find indices in s array where s == s_ab
        idx_ab = s_ab + (n-1)  # index in delta array
        delta[idx_ab] += 1.0
        idx_ba = s_ba + (n-1)
        delta[idx_ba] += 1.0
    
    # s=0 correction
    delta[n-1] -= 2.0  # s=0 is at index n-1
    
    return delta
```

```python
import numpy as np

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection on the Lagrange multiplier."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def round_to_binary(h, n_ones):
    """Round h to binary with exactly n_ones ones."""
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(len(h))
    result[order[len(h)-n_ones:]] = 1.0
    return result

def local_search(h, corr, max_rounds=200):
    """Best-improvement local search using efficient delta computation."""
    n = len(h)
    s_arr = np.arange(-(n-1), n)
    
    for _ in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n1 = len(ones)
        n0 = len(zeros)
        current_max = corr.max()
        
        # Precompute A (effect of flipping each 1 to 0) and B (effect of flipping each 0 to 1)
        A = np.zeros((n1, 2*n-1))
        for i in range(n1):
            a = ones[i]
            m1 = (a + s_arr >= 0) & (a + s_arr < n)
            A[i, m1] = h[a + s_arr[m1]] - 1.0
            m2 = (a - s_arr >= 0) & (a - s_arr < n)
            A[i, m2] += h[a - s_arr[m2]]
        
        B = np.zeros((n0, 2*n-1))
        for i in range(n0):
            b = zeros[i]
            m1 = (b + s_arr >= 0) & (b + s_arr < n)
            B[i, m1] = 1.0 - h[b + s_arr[m1]]
            m2 = (b - s_arr >= 0) & (b - s_arr < n)
            B[i, m2] += -h[b - s_arr[m2]]
        
        best_new_max = current_max
        best_swap = None
        
        for i in range(n1):
            base = corr + A[i]
            # Vectorized: compute max for all j at once
            cand_max = (base[None, :] + B).max(axis=1)
            # Top-k

            top_idx = np.argpartition(cand_max, k)[:k]
            for j in top_idx:

                b = zeros[j]
                delta = base - corr  # = A[i]
                delta = A[i] + B[j]
                # Add corrections
                idx_ab = (a - b) + (n - 1)
                idx_ba = (b - a) + (n - 1)
                delta[idx_ab] += 1.0
                delta[idx_ba] += 1.0
                delta[n-1] -= 2.0
                
                new_max = (corr + delta).max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        a, b = ones[i], zeros[j]
        h[a] = 0.0
        h[b] = 1.0
        delta = A[i] + B[j]
        idx_ab = (a - b) + (n - 1)
        idx_ba = (b - a) + (n - 1)

def perturb_binary(h, n_swaps, rng):
    """Apply random swaps to a binary solution."""
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

        zi = zeros[rng.integers(len(zeros))]

        h[zi] = 1.0

        zeros = np.where(h == 0.0)[0]
    return h
```

```python
def minimize_max_lines(p, q, t_min=0.0, t_max=1.0):
    """Find t in [t_min, t_max] that minimizes max_i (p[i]*t + q[i])."""
    n_lines = len(p)
    # Handle edge cases
    if n_lines == 0:
        return t_min
    
    # Sort by slope
    order = np.argsort(p)
    p_sorted = p[order]
    q_sorted = q[order]
    
    # Compute upper hull
    # Hull is a list of (p, q, t_start) where t_start is where this line becomes active
    hull = []
    for i in range(n_lines):
        pi, qi = p_sorted[i], q_sorted[i]
        # Check if this line is above all existing hull lines
        # For lines with the same slope, keep the one with higher q
        if hull and pi == hull[-1][0]:
            if qi <= hull[-1][1]:
                continue
            else:
                hull.pop()
        
        # Find where this line becomes better than the last hull line
        if not hull:
            hull.append((pi, qi, -np.inf))
        else:
            # Intersection with last hull line
            p_last, q_last = hull[-1][0], hull[-1][1]
            if pi == p_last:
                continue  # Same slope, already handled
            t_int = (q_last - qi) / (pi - p_last)
            # Check if this intersection is before the last line's start
            if t_int <= hull[-1][2]:
                hull.pop()
                if not hull:
                    hull.append((pi, qi, -np.inf))
                else:
                    # Recheck with new last
                    p_last, q_last = hull[-1][0], hull[-1][1]
                    t_int = (q_last - qi) / (pi - p_last)
                    hull.append((pi, qi, t_int))
            else:
                hull.append((pi, qi, t_int))
    
    # Find minimum on [t_min, t_max]
    # The upper hull is convex, so the minimum is where the slope changes from negative to positive
    best_t = t_min
    best_val = np.inf
    for i in range(len(hull)):
        pi, qi, t_start = hull[i]
        t_end = hull[i+1][2] if i+1 < len(hull) else np.inf
        # Clip to [t_min, t_max]
        seg_start = max(t_start, t_min)
        seg_end = min(t_end, t_max)
        if seg_start > seg_end:
            continue
        # The line is active on [seg_start, seg_end]

        # Minimum is at an endpoint
        for t_candidate in [seg_start, seg_end]:
```
