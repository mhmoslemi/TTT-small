    active = np.where(corr > c_max - eps)[0]

    gradients = []
    for k_idx in active:
        g = _grad_corr_term(h, k_idx, 2.0/n)

    #         0 <= h + delta <= 1

            h_sl = _slsqp_opt(h_work, n, target_sum, p, maxiter=300)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_work = h_sl

        if time.time() > deadline - 15:
            break

        h_cd2 = _coord_descent_fine(h_work, n, target_sum, max_rounds=40, n_trials=80)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n
        h_work = h_cd2

        if time.time() > deadline - 20:
            break

        h_work, c5_pert = _perturb_and_refine(h_work, n, target_sum, rng, 8, 0.03, deadline - 20)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_work.copy()
            best_n = n

    if best_h is not None:
        n_best = best_n
        target = n_best / 2.0
        h_final = best_h.copy()

        if time.time() < deadline - 25:
            h_cd_final = _coord_descent_fine(h_final, n_best, target, max_rounds=60, n_trials=100)
            c5_f = _c5_of(h_cd_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_cd_final.copy()
                h_final = h_cd_final

        if time.time() < deadline - 15:
            h_eq_final = _equalize_top(h_final, n_best, target, top_k=10, n_rounds=40)
            c5_fe = _c5_of(h_eq_final)
            if c5_fe < best_c5:
                best_c5 = c5_fe
                best_h = h_eq_final.copy()
                h_final = h_eq_final

        if time.time() < deadline - 12:
            h_cd_f2 = _coord_descent_fine(h_final, n_best, target, max_rounds=50, n_trials=120)
            c5_f2 = _c5_of(h_cd_f2)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_cd_f2.copy()
                h_final = h_cd_f2

        if time.time() < deadline - 10:
            for p in [500, 2000, 10000]:
                if time.time() > deadline - 8:
                    break
                h_pol = _slsqp_opt(h_final, n_best, target, p, maxiter=600)
                c5_pol = _c5_of(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    h_final = h_pol

        if time.time() < deadline - 5:
            h_cd_last = _coord_descent_fine(h_final, n_best, target, max_rounds=80, n_trials=150)
            c5_last = _c5_of(h_cd_last)
            if c5_last < best_c5:
                best_c5 = c5_last
                best_h = h_cd_last

        if time.time() < deadline - 3:
            h_eq_last = _equalize_top(best_h, n_best, target, top_k=12, n_rounds=40)
            c5_el = _c5_of(h_eq_last)
            if c5_el < best_c5:
                best_c5 = c5_el
                best_h = h_eq_last

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
