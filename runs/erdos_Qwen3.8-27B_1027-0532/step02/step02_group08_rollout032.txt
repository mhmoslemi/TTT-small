```python
    """Find best response to the top n_active lags."""
    corr = np.correlate(h, 1-h, 'full')
    # Get the top n_active lags
    top_idx = np.argsort(-corr)[:n_active]
    
    # For each top lag, the coefficient for h[i] is (1 - h[i+k])
    # We want to minimize sum of h[i] * coeff[i]
    # So set h[i] = 1 where coeff is smallest
    
    # Average the coefficients over top lags
    coeffs = np.zeros(n)
    for k in top_idx:
        r = k - (n - 1)  # actual lag
        for i in range(n):
            j = i + r
            if 0 <= j < n:
                coeffs[i] += (1 - h[j])
            else:
                # h[i] doesn't appear in this correlation
                pass
    coeffs /= len(top_idx)
    
    # Set h = 1 where coeffs is smallest
    h_new = np.zeros(n)
    sorted_idx = np.argsort(coeffs)
    h_new[sorted_idx[:n//2]] = 1.0
    # Handle fractional
    remainder = n/2 - n//2
    if remainder > 0:
        h_new[sorted_idx[n//2]] = remainder
    return h_new
```

```python
def grad_at_shift(h, r):
    n = len(h)
    grad = np.zeros(n)
    if r >= 0:
        if r < n:
            grad[:n - r] += (1.0 - h[r:])
            grad[r:] -= h[:n - r]
    else:
        s = -r
        if s < n:
            grad[s:] += (1.0 - h[:n - s])

    return grad


def topk_subgradient_step(h, n_top, step_size, target_sum):

    corrs = compute_corrs(h)
    top_idx = np.argsort(-corrs)[:n_top]

        if abs(r) < n:
            grad += grad_at_shift(h, r)

        viol = np.sum(h) - n / 2.0

            for _ in range(3):
                pert = topk_subgradient_step(pert, 2, 0.003, n_best / 2.0)

        pert = project_feasible(pert, n_best / 2.0)
        alpha_r = 100 + 400 * rng.randint(8)
        maxiter_r = max(60, min(300, 150000 // n_best))
        res_x = lbfgsb_run(pert, alpha_r, 10000.0, maxiter_r, bounds_best)
        if res_x is not None:
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = eval_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 7: Final polish with very high alpha
    if time_left() > 30:
        for alpha in [10000, 30000, 50000]:
            if time_left() < 10:
                break
            res_x = lbfgsb_run(best_h, alpha, 50000.0, 200, bounds_best)
            if res_x is not None:
                h_new = project_feasible(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = eval_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()

    # Phase 8: SLSQP final
    if time_left() > 20:
        res_x = slsqp_run(best_h, 20000, 150, bounds_best)
        if res_x is not None:
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = eval_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 9: Try slightly different resolutions
    if time_left() > 30:
        for n_adj in [n_best + 10, n_best - 10, n_best + 30, n_best + 50]:
            if time_left() < 15:
                break
            if n_adj < 100 or n_adj > 2000:
                continue
            x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_adj, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_feasible(h_init, n_adj / 2.0)
            bounds_adj = [(0.0, 1.0)] * n_adj
            for alpha in [500, 2000, 5000]:
                if time_left() < 10:
                    break
                res_x = lbfgsb_run(h_init, alpha, 10000.0, 200, bounds_adj)
                if res_x is not None:
                    h_new = project_feasible(np.clip(res_x, 0.0, 1.0), n_adj / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s = eval_score(h_new)
                        if np.isfinite(s) and s < best_score:
                            best_score = s
                            best_h = h_new.copy()
                            h_init = h_new.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = eval_score(best_h)
    return (best_h, best_score, len(best_h))
```
