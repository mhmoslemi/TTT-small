    # Phase 1: Active-set projected gradient (focus on worst k)
    for round_idx in range(300):
        if _time_left() < 350:
            break
        base_lr = 0.015 * (1.0 - 0.5 * round_idx / 300.0)
        h_new = projected_gradient_active(best_h, 20, base_lr, n_top=2, decay=0.92)
        s = evaluate_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new

    # Phase 2: Smooth max L-BFGS-B with penalty
    mu = 5000.0
    alpha_list = [100, 300, 600, 1000, 2000, 4000]
    for alpha in alpha_list:
        if _time_left() < 250:
            break
        res_x = penalty_lbfgs(best_h, alpha, mu, 400, bounds)
        if res_x is not None:
            h_new = proj_sum_box(np.clip(res_x, 0.0, 1.0), len(best_h) / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    # Phase 3: Binary-continuous alternating
    n_curr = len(best_h)
    bounds_curr = [(0.0, 1.0)] * n_curr
    for it in range(60):
        if _time_left() < 250:
            break
        b = binary_projection(best_h, n_curr)
        noise_scale = 0.015 + 0.04 * (it / 60.0)
        pert = b + rng.randn(n_curr) * noise_scale
        pert = proj_sum_box(np.clip(pert, 0.0, 1.0), n_curr / 2.0)
        alpha = 400 + 300 * (it % 5)
        res_x = penalty_lbfgs(pert, alpha, 8000, 150, bounds_curr)
        if res_x is not None:
            h_new = proj_sum_box(np.clip(res_x, 0.0, 1.0), n_curr / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new
                    n_curr = len(best_h)
                    bounds_curr = [(0.0, 1.0)] * n_curr

    # Phase 4: Multi-resolution search
    n_best = len(best_h)
    res_list = [500, 600, 750, 900, 1100]
    for n_try in res_list:
        if _time_left() < 300:
            break
        x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = proj_sum_box(h_init, n_try / 2.0)
        bounds_try = [(0.0, 1.0)] * n_try
        h_curr = h_init.copy()
        for alpha in [200, 500, 1000, 2000, 4000]:
            if _time_left() < 100:
                break
            maxiter = max(60, min(350, 120000 // n_try))
            res_x = penalty_lbfgs(h_curr, alpha, 10000, maxiter, bounds_try)
            if res_x is not None:
                h_new = proj_sum_box(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new
                        h_curr = h_new.copy()
                        n_best = n_try

        # Active-set gradient at new resolution
        n_best = len(best_h)
        bounds_try = [(0.0, 1.0)] * n_best
        for _ in range(30):
            if _time_left() < 50:
                break
            h_new = projected_gradient_active(best_h, 15, 0.008, n_top=2, decay=0.93)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new

    # Phase 5: Diverse random restarts
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    for i in range(300):
        if _time_left() < 80:
            break
        strategy = i % 5
        if strategy == 0:
            b = binary_projection(best_h, n_best)
            pert = b + rng.randn(n_best) * (0.01 + 0.03 * i / 300.0)
        elif strategy == 1:
            pert = best_h + rng.randn(n_best) * 0.04
        elif strategy == 2:
            pert = best_h.copy()
            n_flip = max(3, n_best // 25)
            idx = rng.choice(n_best, size=n_flip, replace=False)
            pert[idx] = 1.0 - pert[idx]
        elif strategy == 3:
            sorted_idx = np.argsort(-best_h)
            pert = np.zeros(n_best)
            pert[sorted_idx[:n_best // 2]] = 1.0
            pert += rng.randn(n_best) * 0.08
        else:
            mask = rng.rand(n_best) < 0.15
            pert = best_h.copy()
            n_mask = int(np.sum(mask))
            if n_mask > 0:
                pert[mask] = np.clip(pert[mask] + rng.randn(n_mask) * 0.5, 0, 1)

        pert = proj_sum_box(np.clip(pert, 0.0, 1.0), n_best / 2.0)
        alpha = 200 + 500 * rng.randint(10)
        maxiter_r = max(60, min(250, 100000 // n_best))
        res_x = penalty_lbfgs(pert, alpha, 10000, maxiter_r, bounds_best)
        if res_x is not None:
            h_new = proj_sum_box(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    # Phase 6: SLSQP refinement
    for alpha in [2000, 5000, 10000]:
        if _time_left() < 60:
            break
        res_x = slsqp_optimize(best_h, alpha, 200, bounds_best)
        if res_x is not None:
            h_new = proj_sum_box(np.clip(res_x, 0.0, 1.0), len(best_h) / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    # Phase 7: Final high-alpha polish
    for alpha in [10000, 20000, 50000]:
        if _time_left() < 20:
            break
        res_x = penalty_lbfgs(best_h, alpha, 50000, 200, bounds_best)
        if res_x is not None:
            h_new = proj_sum_box(np.clip(res_x, 0.0, 1.0), len(best_h) / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    # Phase 8: Final active-set refinement
    for _ in range(50):
        if _time_left() < 10:
            break
        lr = 0.005 * (1.0 - _ / 50.0)
        if lr < 0.0005:
            lr = 0.0005
        h_new = projected_gradient_active(best_h, 10, lr, n_top=3, decay=0.95)
        s = evaluate_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new

    best_h = proj_sum_box(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
