```python
def project_binary(h, target_ones):
    """Project h to binary with exactly target_ones ones."""
    n = len(h)
    h = np.round(h)
    # If we have too many ones, flip the ones closest to 0.5 to 0
    # If too few, flip the zeros closest to 0.5 to 1

    return h

def make_binary_with_sum(h_real, target_ones):
    """Convert a real-valued vector to binary with exactly target_ones ones."""
    n = len(h_real)
    idx = np.argsort(-h_real)  # sort by value descending
    result = np.zeros(n)
    result[idx[:target_ones]] = 1.0
    return result

def perturb_block_swap(h, rng, max_block=20):
    """Swap two non-overlapping blocks."""
    n = len(h)
    h_new = h.copy()
    blen = rng.integers(2, min(max_block, n//4) + 1)
    a = rng.integers(0, n - blen + 1)
    b = rng.integers(0, n - blen + 1)
    if abs(a - b) < blen:
        return h_new
    i1, i2 = min(a, b), max(a, b)
    h_new[i1:i1+blen], h_new[i2:i2+blen] = h_new[i2:i2+blen], h_new[i1:i1+blen]
    return h_new

    h_new = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    oi = ones[rng.integers(len(ones))]
    zi = zeros[rng.integers(len(zeros))]
    h_new[oi] = 0.0
    h_new[zi] = 1.0
    return h_new

def perturb_adaptive(h, rng, step):
    """Adaptive perturbation that changes with annealing progress."""
    n = len(h)
    # At high temperature, use larger blocks; at low temperature, smaller
    max_block = max(2, int(n // 4 * (step ** 0.5)))  # decreases over time
    max_block = min(max_block, n // 3)
    r = rng.random()
    if r < 0.3:
        return perturb_single_swap(h, rng)
    elif r < 0.7:
        return perturb_block_swap(h, rng, max_block=max_block)
    else:
        # Reverse a block (flip it)
        blen = rng.integers(2, max_block + 1)
        a = rng.integers(0, n - blen + 1)
        h_new = h.copy()
        h_new[a:a+blen] = 1.0 - h_new[a:a+blen]
        return h_new

def population_sa(h_starts, target_ones, n_steps, t_start, t_end, rng, 
                  t_check_interval=1000, time_limit=None, start_time=None):
    """Population-based simulated annealing."""
    n = len(h_starts[0])
    pop_size = len(h_starts)
    dx = 2.0 / n
    
    pops = [s.copy() for s in h_starts]
    scores = [compute_c5(p) for p in pops]
    best_h = pops[int(np.argmin(scores))].copy()
    best_score = min(scores)
    
    t = t_start
    decay = (t_end / t_start) ** (1.0 / max(n_steps - 1, 1))
    
    for step in range(n_steps):
        t = t_start * decay ** step
        for i in range(pop_size):
            h_new = perturb_adaptive(pops[i], rng, step / n_steps)
            # Ensure sum constraint
            sum_new = np.sum(h_new)
            if sum_new != target_ones:
                # This shouldn't happen with our perturbations, but just in case
                h_new = make_binary_with_sum(h_new, target_ones)
            
            score_new = compute_c5(h_new)
            delta = score_new - scores[i]
            
            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
                pops[i] = h_new
                scores[i] = score_new
                if score_new < best_score:
                    best_score = score_new
                    best_h = h_new.copy()
        
        # Periodic: replace worst with crossover of two best
        if step > 0 and step % 500 == 0:
            order = np.argsort(scores)
            best_idx = order[0]
            second_best_idx = order[1]
            worst_idx = order[-1]
            # Crossover
            cross_point = rng.integers(n//4, 3*n//4)
            cx = np.concatenate([pops[best_idx][:cross_point], pops[second_best_idx][cross_point:]])
            cx = make_binary_with_sum(cx, target_ones)
            pops[worst_idx] = cx
            scores[worst_idx] = compute_c5(cx)
        
        # Time check
        if time_limit is not None and start_time is not None:
            if step % t_check_interval == 0:
                if time.time() - start_time > time_limit:
                    break
    
    return best_h, best_score

def slsqp_polish(h_init, target_ones, maxiter=200, eps=0.001):
    """Polish a binary solution using SLSQP."""
    n = len(h_init)
    h0 = h_init.astype(float).copy()
    
    def objective(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        x = (corr - cmax) / eps
        x -= np.max(x)
        return cmax + eps * np.log(np.sum(np.exp(x)))
    
    def objective_grad(h):
        dx = 2.0 / n
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full') * dx
        cmax = np.max(corr)
        x = (corr - cmax) / eps
        x -= np.max(x)
        ex = np.exp(x)
        sexp = np.sum(ex)
        w = ex / sexp
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * dx * omh[:m]
                grad[:m] -= wk * dx * h[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:m]
        return grad
    
    def constraint_fun(h):
        return float(np.sum(h) - target_ones)
    
    def constraint_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        result = minimize(objective, h0, jac=objective_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = result.x
        # Project to ensure constraint
        h_opt = np.clip(h_opt, 0, 1)
        # Adjust sum
        diff = target_ones - np.sum(h_opt)
        if abs(diff) > 1e-6:
            # Simple projection
            h_opt += diff / n
            h_opt = np.clip(h_opt, 0, 1)
        return h_opt, compute_c5(h_opt)
    except:
        return h0, compute_c5(h0)
```

Let me also think about the "fitness landscape" of this problem. For binary sequences, the landscape is very rugged (combinatorial). SA with block swaps is good because it can make large jumps.

One more idea: instead of random block swaps, we can use "informed" swaps. For example, identify the lag k where the correlation is currently maximum, and try to reduce it by swapping positions that affect that lag.

```python
def directed_swap(h, rng, target_ones):
    """Find the max lag and try to reduce it by swapping."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_lag_idx = np.argmax(corr)
    # max_lag_idx corresponds to lag s = max_lag_idx - (n-1)
    s = max_lag_idx - (n - 1)
    
    # Try to find a swap that reduces corr at this lag
    # If s > 0: corr at lag s = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
    # To reduce it, we want to find a position i where h[i+s]=1, h[i]=0
    # and either set h[i+s]=0 (swap with a 0) or h[i]=1 (swap with a 1)
    
    h_new = h.copy()
    if s > 0:
        # Find positions where h[i+s]=1 and h[i]=0
        candidates = np.where((h[s:] == 1) & (h[:n-s] == 0))[0]
        if len(candidates) > 0:
            # Also need a 1 in the first part to swap with, or a 0 in the second part
            # Swap h[i+s] with some h[j] where h[j]=0 and j is not in the affected range
            # Actually, simpler: just flip h[i+s] to 0 and h[i] to 1
            i = candidates[rng.integers(len(candidates))]
            h_new[i+s] = 0.0
            h_new[i] = 1.0
    elif s < 0:
        r = -s
        candidates = np.where((h[:n-r] == 1) & (h[r:] == 0))[0]
        if len(candidates) > 0:
            i = candidates[rng.integers(len(candidates))]
            h_new[i] = 0.0
            h_new[i+r] = 1.0
    # s == 0: all positions with h[i]=1 and (1-h[i])=1 are impossible for binary
    # So s=0 gives sum h[i]*(1-h[i]) = 0 for binary
    
    return h_new
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def make_binary_with_sum(h_real, target_ones):
    n = len(h_real)
    idx = np.argsort(-h_real, kind='stable')
    result = np.zeros(n)
    result[idx[:target_ones]] = 1.0
    return result

def perturb_block_swap(h, rng, max_block=None):
    n = len(h)
    if max_block is None:
        max_block = max(2, n // 5)
    h_new = h.copy()
    blen = rng.integers(2, min(max_block, n // 3) + 1)
    a = rng.integers(0, n - blen + 1)
    b = rng.integers(0, n - blen + 1)
    if abs(a - b) < blen:
        return h, False
    i1, i2 = min(a, b), max(a, b)
    h_new[i1:i1+blen], h_new[i2:i2+blen] = h_new[i2:i2+blen].copy(), h_new[i1:i1+blen].copy()
    return h_new, True

                best_h = h_poly.copy()
                h_cont = best_h + 0.01 * rng.standard_normal(n)
                h_cont = project_continuous(h_cont, float(target_ones))

    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n2 = 2 * n
        target2 = n2 // 2
        h_up = np.zeros(n2)
        for i in range(n2):
            h_up[i] = best_h[i // 2]
        h_up = h_up + 0.05 * rng.standard_normal(n2)
        h_up = project_continuous(h_up, float(target2))
        h_up_bin = make_binary_with_sum(h_up, target2)

        starts_up = [h_up_bin.copy(), make_binary_with_sum(h_up_bin + 0.1 * rng.standard_normal(n2), target2)]
        n_steps_up = 15000
        tl_up = min(remaining - 60, 400)
        h_sa_up, c5_sa_up = population_sa(starts_up, target2, n_steps_up, 0.3, 0.001,
                                           rng, tl_up, start_time, check_interval=2000)
        if c5_sa_up < best_c5:
            best_c5 = c5_sa_up
            best_h = h_sa_up.copy()
            n = n2
            target_ones = target2

    remaining = budget_s - (time.time() - start_time)
    if remaining > 50 and len(best_h) == n:
        n_cur = len(best_h)
        h_final_cont = best_h + 0.005 * rng.standard_normal(n_cur)
        h_final_cont = project_continuous(h_final_cont, float(n_cur // 2))
        h_poly, c5_poly = slsqp_polish(h_final_cont, n_cur // 2, 0.0005, maxiter=300)
        if c5_poly < best_c5:
            best_c5 = c5_poly
            best_h = h_poly.copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 100 and len(best_h) < 300:
        n_cur = len(best_h)
        starts3 = [best_h.copy()]
        for _ in range(2):
            starts3.append(make_binary_with_sum(best_h + 0.08 * rng.standard_normal(n_cur), n_cur // 2))
        n_steps3 = 10000
        h_sa3, c5_sa3 = population_sa(starts3, n_cur // 2, n_steps3, 0.1, 0.00005,
                                       rng, min(remaining - 20, 200), start_time, check_interval=1000)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3

    final_h = best_h.copy()
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
```
