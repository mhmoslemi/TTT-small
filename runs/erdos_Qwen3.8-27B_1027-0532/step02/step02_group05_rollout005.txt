        ex = np.exp(shifted - max_s)
        p = ex / np.sum(ex)
    
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            idx = np.arange(k, N)
            grad[idx] += w * (1.0 - h[idx - k])
            idx2 = np.arange(0, N - k)
            grad[idx2] -= w * h[idx2 + k]
        else:
            kp = -k
            idx = np.arange(kp, N)
            grad[idx] += w * (1.0 - h[idx - kp])
            idx2 = np.arange(0, N - kp)
            grad[idx2] -= w * h[idx2 + kp]
    return grad

def fun_sm(x, alpha):
    return smooth_max(compute_correlations(x), alpha)

def jac_sm(x, alpha):
    return grad_smooth_max(x, alpha)

def con_eq(x):
    return np.sum(x) - len(x) / 2.0

def con_eq_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    try:
        res = minimize(
            fun_sm, h0, method='SLSQP', bounds=bounds,
            jac=jac_sm, args=(alpha,),
            constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
        )
        if res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except:
        pass
    return None

def do_fista(h0, alpha, target, max_iter, lr):
    N = len(h0)
    x = h0.copy()
    z = h0.copy()
    t_k = 1.0
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    for it in range(max_iter):
        g = grad_smooth_max(z, alpha)
        z_new = project_feasible(z - lr * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = z_new + ((t_k - 1.0) / t_new) * (z_new - x)
        x_new = project_feasible(x_new, target)
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
        z = z_new
        t_k = t_new
    return best_x, best_score

def do_momentum(h0, alpha, target, max_iter, lr0, momentum=0.9):
    N = len(h0)
    x = h0.copy()
    v = np.zeros(N)
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    for it in range(max_iter):
        g = grad_smooth_max(x, alpha)
        lr = lr0 / (1.0 + 0.005 * it)
        v = momentum * v + lr * g
        x_new = project_feasible(x - v, target)
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
    return best_x, best_score

def do_active(h0, target, max_iter, n_active, lr0=0.05):
    N = len(h0)
    x = h0.copy()
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    for it in range(max_iter):
        corr = compute_correlations(x)
        idx = np.argsort(corr)[-n_active:]
        grad = np.zeros(N)
        for j in idx:
            k = j - (N - 1)
            if k >= 0:
                idx_arr = np.arange(k, N)
                grad[idx_arr] += (1.0 - x[idx_arr - k])
                idx2 = np.arange(0, N - k)
                grad[idx2] -= x[idx2 + k]
            else:
                kp = -k
                idx_arr = np.arange(kp, N)
                grad[idx_arr] += (1.0 - x[idx_arr - kp])
                idx2 = np.arange(0, N - kp)
                grad[idx2] -= x[idx2 + kp]
        grad /= float(len(idx))
        lr = lr0 / (1.0 + 0.002 * it)
        x_new = project_feasible(x - lr * grad, target)
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
    return best_x, best_score
