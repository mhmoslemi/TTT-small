import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add

            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _weighted_gradient(h, dx, top_M, sharpness=80.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp(sharpness * (corr - max_corr))
    total = weights.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / total
    top_idx = np.argsort(weights)[::-1][:top_M]
    w_top = weights[top_idx].copy()
    w_top = w_top / w_top.sum()
    g = np.zeros(len(h))
    for idx in range(len(top_idx)):
        g += w_top[idx] * _grad_corr_term(h, top_idx[idx], dx)
    return g

def _nesterov(h0, n, target_sum, n_iters, alpha0, top_M=1, sharpness=80.0):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if top_M == 1:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        else:
            g = _weighted_gradient(y, dx, top_M, sharpness)
        
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    dx = 2.0 / n
    target = target_sum
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        ev = np.exp(alpha * (corr - m))
        w = ev / ev.sum()
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-14:
                g += w[k_idx] * _grad_corr_term(x, k_idx, dx)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-14:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-14:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _weighted_gradient(h, dx, top_M, sharpness):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp(sharpness * (corr - max_corr))
    total = weights.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / total
    top_idx = np.argsort(weights)[::-1][:top_M]
    w_top = weights[top_idx].copy()
    w_top = w_top / w_top.sum()
    g = np.zeros(len(h))
    for idx in range(len(top_idx)):
        g += w_top[idx] * _grad_corr_term(h, top_idx[idx], dx)
    return g

def _nesterov(h0, n, target_sum, n_iters, alpha0, top_M, sharpness, stall_limit):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    stall = 0
    for it in range(n_iters):
        if top_M == 1:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        else:
            g = _weighted_gradient(y, dx, top_M, sharpness)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val - 1e-12:
            best_val = cv
            best = x.copy()
            stall = 0
        else:
            stall += 1
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    dx = 2.0 / n
    target = target_sum

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        ev = np.exp(alpha * (corr - m))
        return (m + np.log(ev.sum()) / alpha) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        ev = np.exp(alpha * (corr - m))
        w = ev / ev.sum()
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-14:
                g += w[k_idx] * _grad_corr_term(x, k_idx, dx)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _fix_sum(res.x, target)
    except Exception:
        return h0
