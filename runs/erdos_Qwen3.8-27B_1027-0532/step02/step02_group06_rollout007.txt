def slsqp_optimize(h, alpha, maxiter=200, reg=0.0):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        if reg > 0:
            d = np.diff(hv)
            val += reg * np.sum(d * d)
        return val
    
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    
    def constraint_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

        if result is not None:

def subgradient_refine(h, alpha, n_top, step_init, max_iters, seed, t0, time_limit):

    n = len(h)
    target_sum = n / 2.0
    h = project_to_feasible(h.copy(), target_sum)
    rng = np.random.RandomState(seed)
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        
        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            if lag >= 0:
                # f_lag = sum_{i=0}^{n-1-lag} h[i+lag](1-h[i])
                # df/dh[j] = (1-h[j-lag])*1[j>=lag] - h[j+lag]*1[j<=n-1-lag]
                if lag == 0:
                    grad[:] += w * (1.0 - 2.0 * h)
                else:
                    grad[lag:n] += w * (1.0 - h[:n-lag])
                    grad[:n-lag] -= w * h[lag:n]
            else:
                la = -lag
                # f_{-la} = sum_{i=0}^{n-1-la} h[i](1-h[i+la])
                # df/dh[j] = (1-h[j+la])*1[j<=n-1-la] - h[j-la]*1[j>=la]
                grad[:n-la] += w * (1.0 - h[la:n])
                grad[la:n] -= w * h[:n-la]
        
        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-12:
            break
        
        improved = False
        for trial in range(20):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.4
        
        if not improved:
            stall += 1
            if stall > 200:
                noise_scale = 0.02 + 0.05 * min(stall / 300.0, 1.0)
                h = best_h + rng.randn(n) * noise_scale / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init * (0.5 + 0.5 * rng.rand())
                stall = 0
        
        if it > 0 and it % 1000 == 0:
            step = max(step_init * 0.93 ** (it // 1000), step_init * 0.005)
    
    return best_h, best_score


def upsample(h, new_n):
    """Upsample h to new_n points."""
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)  # Wait, h is not periodic
    h_new = np.interp(x_new, x_old, h)
    return h_new

from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lse_objective(h, alpha, reg=0.0):
    corr = compute_correlations(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    if reg > 0:
        d = np.diff(h)
        val += reg * np.sum(d * d)
    return val


def lse_gradient(h, alpha, reg=0.0):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= weights.sum()

    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-15:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]

    if reg > 0:
        d = np.diff(h)
        grad[1:] += 2.0 * reg * (d - np.roll(d, 1))
        grad[0] += 2.0 * reg * d[0]
        grad[-1] -= 2.0 * reg * d[-1]

    return grad


def slsqp_analytical(h, alpha, maxiter=200, reg=0.0):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        return lse_objective(hv, alpha, reg)

    def grad(hv):
        return lse_gradient(hv, alpha, reg)

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        pass
    return None


def subgradient_run(h0, alpha, n_top, step_init, max_iters, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)

    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break

        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]

        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                la = -lag
                grad[:n - la] += w * (1.0 - h[la:])
                grad[la:] -= w * h[:n - la]

        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-14:
            break

        improved = False
        for trial in range(25):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.35

        if not improved:
            stall += 1
            if stall > 150:
                noise_scale = 0.02 + 0.06 * min(stall / 400.0, 1.0)
                h = best_h + rng.randn(n) * noise_scale / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init * (0.5 + 0.5 * rng.rand())
                stall = 0

        if it > 0 and it % 500 == 0:
            step = max(step_init * 0.94 ** (it // 500), step_init * 0.003)

    return best_h, best_score


def upsample(h, new_n):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_to_feasible(init, n_init / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)
    best_n = n_init

    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP with analytical gradient and continuation at original resolution
    alphas_phase1 = [5, 10, 20, 50, 100, 200, 400, 800, 1500]
    current = best_h.copy()
    for alpha in alphas_phase1:
        if remaining() < 45:
            break
        result = slsqp_analytical(current, alpha, maxiter=150, reg=0.0)
        if result is not None:
            score = evaluate_score(result)
            if score < best_score:
                best_score = score
                best_h = result.copy()
            current = result

    # Phase 2: Subgradient with various configs
    subgrad_configs = [
        (5, 5, 0.5), (10, 10, 0.3), (20, 10, 0.2), (20, 20, 0.3),
        (50, 10, 0.1), (50, 30, 0.15), (100, 20, 0.08), (100, 50, 0.1),
    ]
    for i, (alpha, n_top, step_init) in enumerate(subgrad_configs):
        if remaining() < 100:
            break
        t_allow = min(remaining() * 0.12, 100)
        if t_allow < 10:
            break
        if i % 3 == 1:
            start_h = best_h + rng.randn(len(best_h)) * 0.015 / np.sqrt(len(best_h))
            start_h = project_to_feasible(start_h, len(best_h) / 2.0)
        else:
            start_h = best_h
        h, score = subgradient_run(start_h, alpha, n_top, step_init,
                                   40000, seed + i * 77, t0, t_allow)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if remaining() > 60:
            result = slsqp_analytical(best_h, 300, maxiter=100, reg=0.0)
            if result is not None:
                score = evaluate_score(result)
                if score < best_score:
                    best_score = score
                    best_h = result.copy()

    # Phase 3: Up-sampling to higher resolution
    targets = [2 * best_n, 3 * best_n]
    for n_try in sorted(set(targets)):
        if remaining() < 150:
            break
        if n_try > 4000:
            continue
        h_up = upsample(best_h, n_try)
        h_up = project_to_feasible(h_up, n_try / 2.0)

        alphas_up = [10, 30, 100, 300, 600, 1200]
        for alpha in alphas_up:
            if remaining() < 60:
                break
            result = slsqp_analytical(h_up, alpha, maxiter=120, reg=0.0)
            if result is not None:
                h_up = result

        score = evaluate_score(h_up)
        if score < best_score:
            best_score = score
            best_h = h_up.copy()
            best_n = n_try

    # Phase 4: More subgradient at current best resolution
    if remaining() > 80:
        config_high = [
            (30, 20, 0.15), (50, 30, 0.1), (100, 20, 0.05),
            (200, 30, 0.03), (50, 50, 0.1),
        ]
        for i, (alpha, n_top, step_init) in enumerate(config_high):
            if remaining() < 50:
                break
            t_allow = min(remaining() * 0.1, 80)
            if t_allow < 8:
                break
            start_h = best_h
            if i % 2 == 1:
                start_h = best_h + rng.randn(len(best_h)) * 0.01 / np.sqrt(len(best_h))
                start_h = project_to_feasible(start_h, len(best_h) / 2.0)
            h, score = subgradient_run(start_h, alpha, n_top, step_init,
                                       50000, seed + 500 + i * 33, t0, t_allow)
            if score < best_score:
                best_score = score
                best_h = h.copy()
            if remaining() > 40:
                result = slsqp_analytical(best_h, 500, maxiter=100, reg=0.0)
                if result is not None:
                    score = evaluate_score(result)
                    if score < best_score:
                        best_score = score
                        best_h = result.copy()

    # Phase 5: Try even higher resolution if time allows
    if remaining() > 200 and best_n < 2000:
        n_try2 = min(4 * best_n, 3400)
        if n_try2 != best_n:
            h_up = upsample(best_h, n_try2)
            h_up = project_to_feasible(h_up, n_try2 / 2.0)
            for alpha in [20, 60, 200, 500, 1000]:
                if remaining() < 60:
                    break
                result = slsqp_analytical(h_up, alpha, maxiter=100, reg=0.0)
                if result is not None:
                    h_up = result
            score = evaluate_score(h_up)
            if score < best_score:
                best_score = score
                best_h = h_up.copy()
                best_n = n_try2

    # Phase 6: Final aggressive SLSQP refinement
    if remaining() > 30:
        final_alphas = [500, 1000, 2000, 5000]
        for alpha in final_alphas:
            if remaining() < 15:
                break
            result = slsqp_analytical(best_h, alpha, maxiter=200, reg=0.0)
            if result is not None:
                score = evaluate_score(result)
                if score < best_score:
                    best_score = score
                    best_h = result.copy()

    # Phase 7: Final subgradient push
    if remaining() > 25:
        h, score = subgradient_run(best_h, 100, 30, 0.1, 80000,
                                   seed + 99999, t0, remaining() - 10)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if remaining() > 20:
            result = slsqp_analytical(best_h, 2000, maxiter=150, reg=0.0)
            if result is not None:
                score = evaluate_score(result)
                if score < best_score:
                    best_score = score
                    best_h = result.copy()

    # Final projection
    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
