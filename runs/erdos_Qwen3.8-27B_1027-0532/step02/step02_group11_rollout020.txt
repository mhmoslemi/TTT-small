def to_binary(h, n):
    """Convert to binary vector with exactly n/2 ones."""
    target = n // 2
    # Threshold at 0.5
    binary = (h >= 0.5).astype(float)
    current_sum = int(binary.sum())
    if current_sum == target:
        return binary
    if current_sum > target:
        # Need to flip some 1s to 0s
        indices_1 = np.where(binary == 1.0)[0]
        np.random.shuffle(indices_1)
        for idx in indices_1[:current_sum - target]:
            binary[idx] = 0.0
    else:

        indices_0 = np.where(binary == 0.0)[0]
        np.random.shuffle(indices_0)

            binary[idx] = 1.0
    return binary

    h[j] = 1.0
    h[i] = 0.0
    c5 = evaluate_c5(h)
    h[i] = 1.0
    h[j] = 0.0

def simulated_annealing(h_start, n, start_temp, end_temp, max_iters, time_limit_fn):
    """SA with pairwise swaps on binary vector."""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    n_iters_done = 0
    for it in range(max_iters):
        if time_limit_fn():
            break
        n_iters_done += 1
        
        # Pick random 1-position and 0-position
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        
        delta = new_c5 - current_c5
        temp = start_temp * (end_temp / start_temp) ** (it / max_iters)
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            # Accept
            current_c5 = new_c5
            # Update ones and zeros
            ones = ones[ones != i]
            zeros = np.delete(zeros, np.where(zeros == j)[0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def local_search_binary(h, n, max_passes, time_limit_fn):
    """Hill climbing for binary vector with sum constraint."""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _pass in range(max_passes):
        if time_limit_fn():
            break
        
        corr = np.correlate(h, 1-h, mode="full")
        dx = 2.0 / n
        k_star = np.argmax(corr)
        max_val = corr[k_star]
        
        if max_val == 0:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_swap = None
        best_new_max = np.inf
        
        for i in ones:
            for j in zeros:
                # Try swapping i and j
                h[i] = 0.0
                h[j] = 1.0
                corr_new = np.correlate(h, 1-h, mode="full")
                new_max = corr_new[k_star]  # Focus on the bottleneck
                h[i] = 1.0
                h[j] = 0.0
                
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i, j, new_max)
        
        if best_swap is None:
            break
        
        i, j, new_max = best_swap
        if new_max >= max_val:
            break  # No improvement possible
        
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = evaluate_c5(h)
    
    return h, best_c5

def compute_swap_deltas(h, n, i, j):
    """Compute change in full correlation for swapping positions i (1→0) and j (0→1)."""
    # Full correlation has 2n-1 entries, indexed 0 to 2n-2
    # Shift l = k - (n-1) where k is the index in the full correlation
    # Position p is active for shift l if 0 <= p < n and 0 <= p+l < n
    # i.e., -p <= l <= n-1-p
    
    deltas = np.zeros(2*n-1)
    
    # Contribution from i (1→0): removes (1-h[i+l]) for valid l
    l_min_i = -i
    l_max_i = n-1-i
    l_range_i = np.arange(l_min_i, l_max_i+1)
    h_vals_i = 1.0 - h[i + l_range_i]
    k_range_i = l_range_i + (n-1)
    deltas[k_range_i] -= h_vals_i
    
    # Contribution from j (0→1): adds (1-h[j+l]) for valid l
    l_min_j = -j
    l_max_j = n-1-j
    l_range_j = np.arange(l_min_j, l_max_j+1)
    h_vals_j = 1.0 - h[j + l_range_j]
    k_range_j = l_range_j + (n-1)
    deltas[k_range_j] += h_vals_j
    
    return deltas

        best_c5 = c5_sa
        best_h = h_sa.copy()

    h_ls, c5_ls = ls_binary(best_h, n0, 200, tl(0.35), rng, n_trials=8000)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    h_t, c5_t = ls_binary_target(best_h, n0, target_mc, 300, tl(0.40), rng, n_trials=10000)
    if c5_t < best_c5:
        best_c5 = c5_t
        best_h = h_t.copy()

    best_h, best_c5 = ilss(best_h, n0, tl(0.65), rng, n_restarts=80)

    if elapsed() < budget_s * 0.70:
        h_ls, c5_ls = full_ls(best_h, n0, 100, tl(0.75), rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    n2 = n0 * 2
    if n2 <= 400 and elapsed() < budget_s * 0.78:
        target_mc2 = int(np.floor(0.3808 * n2 / 2.0))
        h_up = np.zeros(n2)
        for i in range(n2):
            pos = i * n0 / n2
            idx = min(int(pos), n0 - 1)
            frac = pos - idx
            if idx + 1 < n0:
                h_up[i] = (1 - frac) * best_h[idx] + frac * best_h[idx + 1]
            else:
                h_up[i] = best_h[idx]
        h_up_bin = to_binary(h_up, n2, rng)
        c5_up = evaluate_c5(h_up_bin)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_bin.copy()

        h_t2, c5_t2 = ls_binary_target(h_up_bin, n2, target_mc2, 200, tl(0.88), rng, n_trials=10000)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

        h_ls2, c5_ls2 = ls_binary(best_h, n2, 100, tl(0.92), rng, n_trials=10000)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    if elapsed() < budget_s * 0.95:
        target_mc_final = int(np.floor(0.3807 * len(best_h) / 2.0))
        h_tf, c5_tf = ls_binary_target(best_h, len(best_h), target_mc_final, 200, tl(0.98), rng, n_trials=10000)
        if c5_tf < best_c5:
            best_c5 = c5_tf
            best_h = h_tf.copy()

    if not np.all((best_h == 0) | (best_h == 1)):
        best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
