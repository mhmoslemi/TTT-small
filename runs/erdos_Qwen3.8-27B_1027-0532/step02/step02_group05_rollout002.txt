    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_val(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def grad_smooth_max(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    ex = np.exp(shifted)
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        shift = j - (N - 1)
        if 0 <= shift:
            grad[shift:] += w * (1.0 - h[:N - shift])
            grad[:N - shift] -= w * h[shift:]
        else:
            sp = -shift
            grad[:N - sp] += w * (1.0 - h[sp:])
            grad[sp:] -= w * h[:N - sp]
    return grad


def coordinate_descent_fast(h, target, n_sweeps, alpha):
    N = len(h)
    best = h.copy()
    corr = compute_correlations(best).copy()
    best_score = evaluate_score(best)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_bound = max(0.0, target - (N - 1))
    hi_bound = min(1.0, target)
    for sweep in range(n_sweeps):
        for i in range(N):
            old_val = best[i]
            c_i = np.zeros(2 * N - 1)
            for k in range(2 * N - 1):
                shift = k - (N - 1)
                coeff = 0.0
                if 0 <= i + shift <= N - 1:
                    coeff += (1.0 - best[i + shift])
                if 0 <= i - shift <= N - 1:
                    coeff -= best[i - shift]
                c_i[k] = coeff
            if np.all(np.abs(c_i) < 1e-15):
                continue
            base_corr = corr - old_val * c_i
            a, b = lo_bound, hi_bound
            for _ in range(35):
                m1 = b - gr * (b - a)
                m2 = a + gr * (b - a)
                vals1 = base_corr + m1 * c_i
                mx1 = float(np.max(vals1))
                sh1 = np.clip(alpha * (vals1 - mx1), -500.0, 500.0)
                f1 = mx1 + np.log(np.sum(np.exp(sh1))) / alpha
                vals2 = base_corr + m2 * c_i
                mx2 = float(np.max(vals2))
                sh2 = np.clip(alpha * (vals2 - mx2), -500.0, 500.0)
                f2 = mx2 + np.log(np.sum(np.exp(sh2))) / alpha
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            new_val = (a + b) / 2.0
            best[i] = new_val
            corr += (new_val - old_val) * c_i
        best = project_feasible(best, target)
        corr = compute_correlations(best).copy()
        s = evaluate_score(best)
        if s < best_score:
            best_score = s
    return best, best_score


def slsqp_active(h0, alpha, n_top, maxiter, target_sum):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N

    def fun(x):
        corr = compute_correlations(x)
        if n_top < 2 * N - 1:
            idx = np.argpartition(corr, -n_top)[-n_top:]
            sub = corr[idx]
        else:
            sub = corr
        m = float(np.max(sub))
        sh = np.clip(alpha * (sub - m), -500.0, 500.0)
        return m + np.log(np.sum(np.exp(sh))) / alpha

    def jac(x):
        corr = compute_correlations(x)
        if n_top < 2 * N - 1:
            idx = np.argpartition(corr, -n_top)[-n_top:]
            sub = corr[idx]
        else:
            idx = np.arange(2 * N - 1)
            sub = corr
        m = float(np.max(sub))
        sh = np.clip(alpha * (sub - m), -500.0, 500.0)
        ex = np.exp(sh)
        p = ex / np.sum(ex)
        grad = np.zeros(N)
        for ii in range(len(idx)):
            w = p[ii]
            if w < 1e-15:
                continue
            j = idx[ii]
            shift = j - (N - 1)
            if shift >= 0:
                grad[shift:] += w * (1.0 - x[:N - shift])
                grad[:N - shift] -= w * x[shift:]
            else:
                sp = -shift
                grad[:N - sp] += w * (1.0 - x[sp:])
                grad[sp:] -= w * x[:N - sp]
        return grad

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(N)

    res = minimize(fun, h0, method='SLSQP', bounds=bounds, jac=jac,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
    return res.x if res.x is not None else None


def discrete_refine(h, target, n_trials, rng):
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(h)
    deltas = [0.005, 0.01, 0.02, 0.05]
    for _ in range(n_trials):
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        sign = rng.choice([1, -1])
        trial = best.copy()
        trial[i] += sign * d
        trial[j] -= sign * d
        if trial[i] < -1e-9 or trial[i] > 1 + 1e-9 or trial[j] < -1e-9 or trial[j] > 1 + 1e-9:
            continue
        trial = np.clip(trial, 0.0, 1.0)
        s = evaluate_score(trial)
        if s < best_score - 1e-13:
            best_score = s
            best = trial.copy()
    return best, best_score


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    init = project_feasible(init, target)

    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: Coordinate descent with increasing alpha
    for alpha in [300, 500, 1000, 2000, 4000]:
        if elapsed() > budget_s - 400:
            break
        n_sweeps = 8
        best_h, best_score = coordinate_descent_fast(best_h, target, n_sweeps, alpha)

    # Phase 2: SLSQP active-set refinement
    for alpha in [1000, 3000, 8000]:
        if elapsed() > budget_s - 300:
            break
        n_top = max(5, (2 * n - 1) // 3)
        maxiter = max(200, min(600, 180000 // n))
        res_x = slsqp_active(best_h, alpha, n_top, maxiter, target)
        if res_x is not None and np.all(np.isfinite(res_x)):
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 3: More coordinate descent (finer)
    for alpha in [2000, 5000, 10000]:
        if elapsed() > budget_s - 250:
            break
        n_sweeps = 12
        best_h, best_score = coordinate_descent_fast(best_h, target, n_sweeps, alpha)

    # Phase 4: Discrete local search
    if elapsed() < budget_s - 200:
        h_disc, s_disc = discrete_refine(best_h, target, 40000, rng)
        if s_disc < best_score:
            best_score = s_disc
            best_h = h_disc.copy()

    # Phase 5: SLSQP after discrete
    for alpha in [5000, 15000]:
        if elapsed() > budget_s - 150:
            break
        n_top = max(3, (2 * n - 1) // 5)
        maxiter = max(300, min(800, 250000 // n))
        res_x = slsqp_active(best_h, alpha, n_top, maxiter, target)
        if res_x is not None and np.all(np.isfinite(res_x)):
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 6: Anti-palindromic SLSQP
    if elapsed() < budget_s - 150:
        half = (n - 1) // 2
        if half > 0:
            g0 = best_h[:half].copy()
            bounds_ap = [(0.0, 1.0)] * half

            def _ap_fun(g, al):
                h = antipal_h(g)
                corr = compute_correlations(h)
                return smooth_max_val(corr, al)

            def _ap_jac(g, al):
                h = antipal_h(g)
                grad_h = grad_smooth_max(h, al)
                grad_g = grad_h[:half].copy()
                for ii in range(half):
                    jj = n - 1 - ii
                    grad_g[ii] += -grad_h[jj]
                return grad_g

            for alpha in [3000, 10000]:
                if elapsed() > budget_s - 100:
                    break
                res_ap = minimize(_ap_fun, g0, method='SLSQP', bounds=bounds_ap,
                                  jac=_ap_jac, args=(alpha,),
                                  options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
                if res_ap.x is not None and np.all(np.isfinite(res_ap.x)):
                    h_ap = antipal_h(np.clip(res_ap.x, 0.0, 1.0))
                    h_ap = project_feasible(h_ap, target)
                    s_ap = evaluate_score(h_ap)
                    if s_ap < best_score:
                        best_score = s_ap
                        best_h = h_ap.copy()

    # Phase 7: Random restarts
    n_restarts = 15
    for r in range(n_restarts):
        if elapsed() > budget_s - 100:
            break
        scale = 0.02 + 0.06 * (r / float(n_restarts))
        pert = best_h + rng.randn(n) * scale
        pert = project_feasible(pert, target)
        alpha = 2000 + 4000 * rng.randint(3)
        best_h_r, best_score_r = coordinate_descent_fast(pert, target, 5, alpha)
        if best_score_r < best_score:
            best_score = best_score_r
            best_h = best_h_r.copy()

    # Phase 8: Final coordinate descent polish
    for alpha in [5000, 20000]:
        if elapsed() > budget_s - 30:
            break
        best_h, best_score = coordinate_descent_fast(best_h, target, 10, alpha)

    # Phase 9: Final SLSQP
    if elapsed() < budget_s - 15:
        n_top = max(2, (2 * n - 1) // 8)
        maxiter = max(200, 300000 // n)
        res_x = slsqp_active(best_h, 20000, n_top, maxiter, target)
        if res_x is not None and np.all(np.isfinite(res_x)):
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Final cleanup
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
