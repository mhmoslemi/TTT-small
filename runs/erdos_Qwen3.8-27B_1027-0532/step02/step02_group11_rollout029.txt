```python
    for d in range(-(n-1), n):
        w = weights[d + (n-1)]
        if w < 1e-12:
            continue
        if d >= 0:
            # (1-h[j-d]) for j in [d, n)
            if d < n:
                grad[d:] += w * (1 - h[:n-d])
            # -h[j+d] for j in [0, n-d)
            if d < n:
                grad[:n-d] += w * (-h[d:])
        else:
            dp = -d  # positive
            # (1-h[j-d]) = (1-h[j+dp]) for j in [0, n-dp)
            grad[:n-dp] += w * (1 - h[dp:])
            # -h[j+d] = -h[j-dp] for j in [dp, n)
            grad[dp:] += w * (-h[:n-dp])
    grad *= dx
    return grad
```

This is O(n) per lag, total O(n²) but with NumPy vectorized operations, it should be fast. For n=170, 339 lags * O(170) vector ops = ~57630 element ops total per gradient call. Should take < 1ms.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0/eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad_vec(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
