        # We want to maximize this (add the 1 that contributes most)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute contribution vectors
        # For each i: c_i = h[i+k_star] (if valid) + h[i-k_star] (if valid)
        c_ones = np.zeros(len(ones))
        for a, i in enumerate(ones):
            c = 0.0
            if 0 <= i + k_star < n:
                c += h[i + k_star]
            if 0 <= i - k_star < n:
                c += h[i - k_star]
            c_ones[a] = c
        
        c_zeros = np.zeros(len(zeros))
        for a, j in enumerate(zeros):
            c = 0.0
            if 0 <= j + k_star < n:
                c += h[j + k_star]
            if 0 <= j - k_star < n:
                c += h[j - k_star]
            c_zeros[a] = c
        
        # Best swap: minimize c_ones[i] - c_zeros[j]
        # = minimize c_ones[i] and maximize c_zeros[j]
        best_i = ones[np.argmin(c_ones)]
        best_j = zeros[np.argmax(c_zeros)]
        
        # Perform swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        new_overlaps = compute_overlaps(h)
        new_score = float(np.max(new_overlaps))
        
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            overlaps = new_overlaps
        else:
            # Revert
            h[best_i] = 1.0
            h[best_j] = 0.0
    
    return best_h, best_score * (2.0 / n)

import numpy as np
import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    obj = float(m + np.log(np.sum(exp_vals)) / alpha)
    grad = np.zeros(n)
    offset = n - 1
    n_lags = len(overlaps)
    for idx in range(n_lags):
        w = p[idx]
        if w < 1e-10:
            continue
        k = idx - offset
        if k >= 0:
            nk = n - k
            grad[k:] += w * (1.0 - h[:nk])
            grad[:nk] -= w * h[k:]
        else:
            M = -k
            nM = n - M
            grad[:nM] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:nM]
    return obj, grad

def adaptive_gradient(h0, alpha, n_iters, target):
    """Adaptive step size gradient descent with Nesterov momentum."""
    n = len(h0)
    h = h0.copy()
    z = h0.copy()
    t_mom = 1.0
    lr = 1.0 / (alpha * 4.0)
    best_h = h.copy()
    best_score = evaluate_score(h)
    momentum = 0.9
    
    for it in range(n_iters):
        obj_z, grad_z = lse_obj_and_grad(z, alpha)
        
        # Try step
        h_new = project_box_sum(z - lr * grad_z, target)
        score_new = np.max(compute_overlaps(h_new)) * (2.0 / n)
        
        if score_new < best_score:
            # Accept, maybe increase lr
            best_score = score_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 5.0 / alpha)
        else:
            # Reject, decrease lr
            lr *= 0.7
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        z_new = h_new + beta * (h_new - h)
        z_new = project_box_sum(z_new, target)
        h = h_new
        t_mom = t_new
    
    return best_h, best_score

def greedy_binary_local_search(h_bin, n_iters, rng, top_k=5):
    n = len(h_bin)
    h = h_bin.copy()
    # Ensure exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    target_n = n // 2
    if len(ones) > target_n:

        h[remove] = 0.0
    elif len(ones) < target_n:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(zeros, size=target_n-len(ones), replace=False)
        h[add] = 1.0
    
    overlaps = compute_overlaps(h)
    best_score = float(np.max(overlaps))
    best_h = h.copy()
    stuck_count = 0
    
    for it in range(n_iters):
        if stuck_count > 200:
            break
        
        top_indices = np.argsort(-overlaps)[:top_k]
        
        best_improve = 0
        best_i = -1
        best_j = -1
        
        for idx in top_indices:
            k_star = idx - (n - 1)
            c = np.zeros(n)
            if k_star >= 0:
                nk = n - k_star
                c[k_star:] += h[:nk]
                c[:nk] += h[k_star:]
            else:
                M = -k_star
                nM = n - M
                c[:nM] += h[M:]
                c[M:] += h[:nM]
            
            ones_mask = (h == 1.0)
            zeros_mask = (h == 0.0)
            
            c_o = c[ones_mask]
            c_z = c[zeros_mask]
            
            if len(c_o) == 0 or len(c_z) == 0:
                continue
            
            min_idx = np.argmin(c_o)
            max_idx = np.argmax(c_z)
            improve = c_o[min_idx] - c_z[max_idx]
            
            if improve > best_improve:
                best_improve = improve
                ones_pos = np.where(ones_mask)[0]
                zeros_pos = np.where(zeros_mask)[0]
                best_i = ones_pos[min_idx]
                best_j = zeros_pos[max_idx]
        
        if best_i < 0 or best_improve <= 0:

            continue
        
        h[best_i] = 0.0

        new_overlaps = compute_overlaps(h)

            best_score = new_score
            best_h = h.copy()
            overlaps = new_overlaps
            stuck_count = 0
        else:
            h[best_i] = 1.0
            h[best_j] = 0.0
            stuck_count += 1
    
    return best_h, best_score * (2.0 / n)

    """Simulated annealing for binary h."""
    n = len(h_bin)
    h = (h_bin > 0.5).astype(float)
    target_n = n // 2
    ones = np.where(h == 1.0)[0]
    if len(ones) > target_n:

        h[remove] = 0.0
    elif len(ones) < target_n:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(zeros, size=target_n-len(ones), replace=False)
        h[add] = 1.0
    
    overlaps = compute_overlaps(h)
    current_score = float(np.max(overlaps))
    best_score = current_score
    best_h = h.copy()
    temp = t_init
    
    for it in range(n_iters):

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_overlaps = compute_overlaps(h)
        new_score = float(np.max(new_overlaps))
        delta = new_score - current_score

            h[j] = 0.0

Now let me also reconsider the algorithm. I think the key issue is that the adaptive gradient method might not be converging well. Let me think about why.

The LSE objective with alpha=1000 is very close to the max. The gradient in this regime is dominated by the single maximum lag. So the gradient step is essentially the subgradient of the max, which is the gradient of a single linear (well, quadratic) function. This should work well!

But the step size is critical. If it's too large, we oscillate. If too small, we converge slowly.

Let me use a more principled step size. For the LSE with parameter alpha:
- The objective is (1/alpha) * log(sum exp(alpha * o_k))
- The gradient is sum p_k * grad(o_k) where p_k is the softmax
- The Hessian involves alpha * (covariance matrix of p) * [grad(o_k)] ^T [grad(o_k)]
- The maximum eigenvalue of the Hessian is bounded by alpha * max_k ||grad(o_k)||^2

For a single overlap o_k = sum h[n+k]*(1-h[n]), the gradient w.r.t. h is a vector with entries in [-1, 1]. The L2 norm is at most sqrt(n). So L <= alpha * n.

For alpha=1000, n=425: L <= 425000. Step size = 1/L = 2.3e-6. That's very small!

But this is a very conservative bound. In practice, the effective Lipschitz constant is much smaller because:
1. The weights p_k are concentrated on a few lags
2. The gradients of different lags are not aligned

A better approach: use **backtracking line search**. Start with a moderate step size and halve it if the objective increases.

Let me implement this properly:

import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):

    return float(np.max(compute_overlaps(h))) * (2.0 / n)

        if s > target:
            lo = mid
        else:
            hi = mid

    shifted = alpha * (overlaps - m)
    shifted = np.clip(shifted, -500.0, 500.0)

    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    shifted = np.clip(shifted, -500.0, 500.0)

            h_new = project_box_sum(z, target)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        z_new = h_new + beta * (h_new - h)
        z_new = project_box_sum(z_new, target)
        
        h = h_new
        t_mom = t_new
        lr = step * 1.3
        
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score

def greedy_binary_search(h_init, n_iters, rng, top_k):
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    target_ones = n // 2
    ones = np.where(h == 1.0)[0]
    if len(ones) > target_ones:
        rem = rng.choice(ones, size=len(ones) - target_ones, replace=False)
        h[rem] = 0.0
    elif len(ones) < target_ones:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(zeros, size=target_ones - len(ones), replace=False)
        h[add] = 1.0
    
    overlaps = compute_overlaps(h)
    best_score = float(np.max(overlaps))
    best_h = h.copy()
    stuck = 0
    
    for it in range(n_iters):
        if stuck > 500:
            break
        top_idx = np.argsort(-overlaps)[:top_k]
        best_improve = -1.0
        bi = -1
        bj = -1
        ones_mask = (h == 1.0)
        zeros_mask = (h == 0.0)
        ones_pos = np.where(ones_mask)[0]
        zeros_pos = np.where(zeros_mask)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        
        for idx in top_idx:
            k_s = idx - (n - 1)
            c = np.zeros(n)
            if k_s >= 0:
                nk = n - k_s
                c[k_s:] += h[:nk]
                c[:nk] += h[k_s:]
            else:
                M = -k_s
                nM = n - M
                c[:nM] += h[M:]
                c[M:] += h[:nM]
            c_o = c[ones_pos]
            c_z = c[zeros_pos]
            mi = np.argmin(c_o)
            ma = np.argmax(c_z)
            imp = float(c_o[mi] - c_z[ma])
            if imp > best_improve:
                best_improve = imp
                bi = ones_pos[mi]
                bj = zeros_pos[ma]
        
        if bi < 0 or best_improve <= 0:
            stuck += 1
            continue
        
        h[bi] = 0.0
        h[bj] = 1.0
        new_ov = compute_overlaps(h)
        new_sc = float(np.max(new_ov))
        if new_sc < best_score:
            best_score = new_sc
            best_h = h.copy()
            overlaps = new_ov
            stuck = 0
        else:
            h[bi] = 1.0
            h[bj] = 0.0
            stuck += 1
    
    return best_h, best_score * (2.0 / n)

def sa_binary(h_init, n_iters, rng, t_max):
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    target_ones = n // 2
    ones = np.where(h == 1.0)[0]
    if len(ones) > target_ones:
        rem = rng.choice(ones, size=len(ones) - target_ones, replace=False)
        h[rem] = 0.0
    elif len(ones) < target_ones:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(zeros, size=target_ones - len(ones), replace=False)
        h[add] = 1.0
    
    overlaps = compute_overlaps(h)
    cur = float(np.max(overlaps))
    best = cur
    best_h = h.copy()
    temp = t_max
    
    for it in range(n_iters):
        if it % 200 == 199:
            temp *= 0.95
            if temp < 1e-12:
                break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_ov = compute_overlaps(h)
        new_sc = float(np.max(new_ov))
        d = new_sc - cur
        if d < 0 or rng.random() < np.exp(-d / max(temp, 1e-15)):
            cur = new_sc
            overlaps = new_ov
            if new_sc < best:
                best = new_sc
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best * (2.0 / n)
