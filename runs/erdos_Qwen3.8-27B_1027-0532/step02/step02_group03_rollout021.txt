import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14: break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14: break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    """Project y onto {x: 0 <= x <= 1, sum(x) = S}."""
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0: break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0: break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr(h, k_idx, n, dx):
    """Gradient of corr[k_idx] * dx with respect to h."""
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] -= h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad * dx

def _nesterov(h0, n, target_sum, n_iters, alpha, use_topk=False, topk=5):
    """Nesterov accelerated gradient with projection."""
    dx = 2.0 / n
    x = h0.copy().astype(float)
    y = h0.copy().astype(float)
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        if use_topk:
            top_indices = np.argsort(corr)[::-1][:topk]
            # Soft weights based on gap to max
            max_c = corr[top_indices[0]]
            weights = np.exp(100.0 * (corr[top_indices] - max_c))
            weights /= weights.sum()
            g = np.zeros(n)
            for wi, ki in zip(weights, top_indices):
                g += wi * _grad_corr(y, ki, n, dx)
        else:
            k_star = int(np.argmax(corr))
            g = _grad_corr(y, k_star, n, dx)
        
        step = alpha / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _slsqp_smooth(h0, n, target_sum, alpha, maxiter=200):
    """SLSQP optimization of the smooth (log-sum-exp) max."""
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500: return m * dx
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            # Fallback to hard max gradient
            k_star = int(np.argmax(corr))
            return _grad_corr(x, k_star, n, dx)
        weights = np.exp(alpha * (corr - m))
        weights /= weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, range(len(corr))):
            if w < 1e-10: continue
            g += w * _grad_corr(x, k, n, dx)
        return g
    
    def eq_fun(x):
        return x.sum() - target_sum
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _binary_local_search(h, n, target_sum, time_limit=60.0, max_rounds=200):
    """Best-improvement swap local search on binary sequences."""
    t0 = time.time()
    hb = h.copy()
    hb[hb >= 0.5] = 1.0
    hb[hb < 0.5] = 0.0
    # Fix to exactly target_sum ones
    n_ones = int(hb.sum())
    if n_ones > int(target_sum):
        # Remove excess ones (from positions that are "least critical")
        excess = n_ones - int(target_sum)
        ones = np.where(hb == 1.0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        for idx in range(excess):
            hb[ones[idx]] = 0.0
            hb[ones[idx + excess]] = 1.0
    elif n_ones < int(target_sum):
        deficit = int(target_sum) - n_ones
        zeros = np.where(hb == 0.0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        for idx in range(deficit):
            hb[zeros[idx]] = 1.0
    
    best_c5 = _c5_of(hb)
    
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        
        best_improve = 0.0
        best_pair = None
        
        # Try all swaps (or a random subset if too many)
        if len(ones) * len(zeros) > 50000:
            # Sample
            sample_size = 50000
            pairs = []
            for _ in range(sample_size):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                if i != j:
                    pairs.append((i, j))
        else:
            pairs = [(i, j) for i in ones for j in zeros]
        
        for i, j in pairs:
            hb[i] = 0.0
            hb[j] = 1.0
            c = _c5_of(hb)
            if c < best_c5 - best_improve:
                best_improve = best_c5 - c
                best_pair = (i, j)
            hb[i] = 1.0
            hb[j] = 0.0
        
        if best_pair is not None:
            hb[best_pair[0]] = 0.0
            hb[best_pair[1]] = 1.0
            best_c5 = _c5_of(hb)
        else:
            break
    
    return hb, best_c5

def _coord_descent_binary(h, n, target_sum, max_rounds=20):
    """Coordinate descent: try setting each position to 0 or 1."""
    h = h.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            if c0 < c1 and old > 0.5:
                h[j] = 0.0
                improved = True
            elif c1 < c0 and old < 0.5:
                h[j] = 1.0
                improved = True
        if not improved:
            break
    # Fix sum
    h = _fix_sum(h, target_sum)
    return h

def _interpolate_to_n(h_from, n_from, n_to):
    """Interpolate h from n_from points to n_to points."""
    x_from = np.arange(n_from)
    x_to = np.linspace(0, n_from - 1, n_to)
    h_to = np.interp(x_to, x_from, h_from)
    return np.clip(h_to, 0.0, 1.0)

def _grad_corr_fast(h, k_idx, n, dx):
    """Fast gradient computation for a single lag."""
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:

    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad * dx

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0
_global_dx = 0.0
_global_alpha_lse = 0.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_fast(h, k_idx, n, dx):
    m = k_idx - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            mm = n - m
            grad[m:] += (1.0 - h[:mm])
            grad[:mm] -= h[m:]
    else:
        p = -m
        pp = n - p
        grad[:pp] += (1.0 - h[p:])
        grad[p:] -= h[:pp]
    return grad * dx

def _obj_lse(x):
    n = _global_n
    dx = _global_dx
    alpha = _global_alpha_lse
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * dx
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

def _obj_lse_grad(x):
    n = _global_n
    dx = _global_dx
    alpha = _global_alpha_lse
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        k_star = int(np.argmax(corr))
        return _grad_corr_fast(x, k_star, n, dx)
    weights = np.exp(alpha * (corr - m))
    weights /= weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, range(len(corr))):
        if w < 1e-8:
            continue
        g += w * _grad_corr_fast(x, k, n, dx)
    return g

def _eq_fun(x):
    return x.sum() - _global_target_sum

def _eq_jac(x):
    return np.ones(_global_n)

def _slsqp_smooth(h0, n, target_sum, alpha, maxiter=300):
    global _global_target_sum, _global_n, _global_dx, _global_alpha_lse
    _global_target_sum = target_sum
    _global_n = n
    _global_dx = 2.0 / n
    _global_alpha_lse = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_obj_lse, h0, jac=_obj_lse_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov(h0, n, target_sum, n_iters, alpha0, topk=3):
    dx = 2.0 / n
    x = h0.copy().astype(float)
    y = h0.copy().astype(float)
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_indices = np.argsort(corr)[::-1][:topk]
        max_c = corr[top_indices[0]]
        weights = np.exp(80.0 * (corr[top_indices] - max_c))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(top_indices))
            weights[0] = 1.0
        else:
            weights = weights / total_w
        g = np.zeros(n)
        for wi, ki in zip(weights, top_indices):
            g += wi * _grad_corr_fast(y, ki, n, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _binary_local_search(h, n, target_sum, time_limit=90.0, max_rounds=300, rng=None):
    t0 = time.time()
    hb = h.copy()
    hb[hb >= 0.5] = 1.0
    hb[hb < 0.5] = 0.0
    n_ones = int(hb.sum())
    target_int = int(round(target_sum))
    if n_ones > target_int:

        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]

            rng.shuffle(ones)
            rng.shuffle(zeros)
        else:
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
        for idx in range(excess):
            hb[ones[idx]] = 0.0
            hb[zeros[idx]] = 1.0
    elif n_ones < target_int:
        deficit = target_int - n_ones
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if rng is not None:
            rng.shuffle(zeros)
        else:
            np.random.shuffle(zeros)
        for idx in range(deficit):
            hb[zeros[idx]] = 1.0
    best_c5 = _c5_of(hb)
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(hb == 1.0)[0]

        best_improve = 0.0

        pairs_list = [(i, j) for i in ones for j in zeros]

def _interpolate_to_n(h_from, n_from, n_to):
    if n_from == n_to:
        return h_from.copy()
    x_from = np.arange(n_from, dtype=float)
    x_to = (np.linspace(0, n_from - 1, n_to)).astype(float)
    h_to = np.interp(x_to, x_from, h_from)
    return np.clip(h_to, 0.0, 1.0)

def _stochastic_binary_search(h, n, target_sum, time_limit=60.0, rng=None):
    t0 = time.time()
    if rng is None:
        rng = np.random.RandomState(42)
    hb = h.copy()
    hb[hb >= 0.5] = 1.0
    hb[hb < 0.5] = 0.0
    n_ones = int(hb.sum())
    target_int = int(round(target_sum))
    if n_ones > target_int:
        excess = n_ones - target_int
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for idx in range(excess):
            hb[ones[idx]] = 0.0
            hb[zeros[idx]] = 1.0
    elif n_ones < target_int:
        deficit = target_int - n_ones
        zeros = np.where(hb == 0.0)[0]
        rng.shuffle(zeros)
        for idx in range(deficit):
            hb[zeros[idx]] = 1.0
    best_c5 = _c5_of(hb)
    temperature = 0.02
    for _ in range(100000):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i] = 0.0
        hb[j] = 1.0
        c_new = _c5_of(hb)
        delta = c_new - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / temperature):
            best_c5 = min(best_c5, c_new)
            temperature = max(temperature * 0.999, 0.001)
        else:
            hb[i] = 1.0
            hb[j] = 0.0
    return hb, best_c5

        # Nesterov polish
        if time_remaining() > 10:
            h_nes, c5_nes = _nesterov(best_h, n_f, target_f,
                                      n_iters=400, alpha0=0.25, topk=8)
            if c5_nes < best_c5:
                best_c5 = c5_nes
                best_h = h_nes.copy()

        # Final SLSQP
        if time_remaining() > 8:
            h_fin = _slsqp_smooth(best_h, n_f, target_f, 8000, maxiter=400)
            c5_fin = _c5_of(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

        # One more binary search from the continuous solution
        if time_remaining() > 10:
            h_b2, c5_b2 = _binary_local_search(best_h, n_f, target_f,
                                               time_limit=min(time_remaining() * 0.4, 50),
                                               max_rounds=150, rng=rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
            if time_remaining() > 8:
                h_b2r = _nesterov(h_b2, n_f, target_f, n_iters=150, alpha0=0.3, topk=5)
                c5_b2r = _c5_of(h_b2r)
                if c5_b2r < best_c5:
                    best_c5 = c5_b2r
                    best_h = h_b2r.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
