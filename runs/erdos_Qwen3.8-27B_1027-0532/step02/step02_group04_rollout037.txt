    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)

                h_try[j] -= step

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted - max_s))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(N)
    for index in range(N):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N-lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:N-lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w * (1.0 - h[:N-l])
                grad[:N-l] -= w * h[l:]
    return grad

            hi = mid
        else:

    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n / 2.0

        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def swap_search(h, n, n_iters, rng, temperature=0.0):

    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)

    for it in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(best[i] - best[j]) < 1e-10:
            continue
        h_try = best.copy()
        h_try[i], h_try[j] = best[j], best[i]
        new_corr = compute_correlations(h_try)
        new_max = np.max(new_corr)
        delta = new_max - best_max
        if delta < -1e-14:
            best = h_try
            best_corr = new_corr
            best_max = new_max
        elif temperature > 0.0:
            accept_prob = np.exp(-delta / temperature) if delta < 1.0 else 0.0
            if rng.rand() < accept_prob:
                best = h_try
                best_corr = new_corr
                best_max = new_max

np.correlate(h, 1-h, mode='full') computes:
result[k] = sum_{t} h[t] * (1-h)[t + k - (N-1)]

where the valid range of t ensures both indices are in [0, N-1].

More precisely, for mode='full', the output has length 2N-1, and:
result[k] = sum_t h[t] * (1-h)[t + k - (N-1)]

The lag corresponding to output index k is: lag = k - (N-1)
- lag = 0 when k = N-1: result[N-1] = sum_t h[t] * (1-h[t]) = sum h[t] - sum h[t]²
- lag > 0 when k > N-1: result[k] = sum_t h[t] * (1-h[t + lag])
  where t ranges so that both t and t+lag are valid
- lag < 0 when k < N-1: result[k] = sum_t h[t] * (1-h[t + lag])
  where lag is negative, so t + lag < t

So for lag > 0: result[N-1+lag] = sum_{t=0}^{N-1-lag} h[t] * (1-h[t+lag])
For lag < 0: result[N-1+lag] = sum_{t=-lag}^{N-1} h[t] * (1-h[t+lag]) = sum_{s=0}^{N-1+lag} h[s+(-lag)] * (1-h[s]) = sum_{s=0}^{N-1-|lag|} h[s+|lag|] * (1-h[s])

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):

        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def _lbfgs_obj_grad(h, alpha, n, mu):
    corr = compute_correlations(h)
    sm = smooth_max_value(corr, alpha)
    s = np.sum(h) - n / 2.0
    obj_val = sm + mu * s * s
    g = smooth_max_gradient(h, alpha)
    g += 2.0 * mu * s
    return obj_val, g


def lbfgs_step(h0, alpha, n, mu, maxiter):
    def obj(h):
        v, _ = _lbfgs_obj_grad(h, alpha, n, mu)
        return v
    def grad_fn(h):
        _, g = _lbfgs_obj_grad(h, alpha, n, mu)
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad_fn, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return res.x


def swap_search(h, n, n_iters, rng, temperature=0.0):
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(best[i] - best[j]) < 1e-10:
            continue
        h_try = best.copy()
        h_try[i], h_try[j] = best[j], best[i]
        new_max = np.max(compute_correlations(h_try))
        delta = new_max - best_max
        if delta < -1e-14:
            best = h_try
            best_max = new_max
        elif temperature > 0.0 and delta < 0.5:
            if rng.rand() < np.exp(-delta / temperature):
                best = h_try
                best_max = new_max
    return best, best_max


def targeted_lag_swap(h, n, max_rounds=500):
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    for _ in range(max_rounds):
        k_star = int(np.argmax(best_corr))
        lag = n - 1 - k_star
        improved = False
        if 0 < lag < n:
            for i in range(n - lag):
                j = i + lag
                if best[i] > best[j] + 1e-10:
                    h_try = best.copy()
                    h_try[i], h_try[j] = best[j], best[i]
                    lc = 0.0
                    for a in range(n - lag):
                        lc += h_try[a] * (1.0 - h_try[a + lag])
                    if lc < best_corr[k_star] - 1e-10:
                        full = compute_correlations(h_try)
                        fm = np.max(full)
                        if fm < best_max - 1e-14:
                            best = h_try
                            best_corr = full
                            best_max = fm
                            improved = True
                            break
        if 0 < -lag < n:
            ll = -lag
            for i in range(n - ll):
                if ll > 0:
                    if best[i + ll] > best[i] + 1e-10:
                        h_try = best.copy()
                        h_try[i], h_try[i + ll] = best[i + ll], best[i]
                        lc = 0.0
                        for a in range(n - ll):
                            lc += h_try[a + ll] * (1.0 - h_try[a])
                        if lc < best_corr[k_star] - 1e-10:
                            full = compute_correlations(h_try)
                            fm = np.max(full)
                            if fm < best_max - 1e-14:
                                best = h_try
                                best_corr = full
                                best_max = fm
                                improved = True
                                break
        if not improved:
            break
    return best, best_max


def block_swap_search(h, n, n_iters, rng):
    best = h.copy()
    best_max = np.max(compute_correlations(best))
    for _ in range(n_iters):
        blen = rng.randint(2, max(3, n // 25))
        i = rng.randint(0, max(1, n - blen))
        j = rng.randint(0, max(1, n - blen))
        if abs(i - j) < blen:
            continue
        h_try = best.copy()
        b1 = h_try[i:i + blen].copy()
        h_try[i:i + blen] = best[j:j + blen]
        h_try[j:j + blen] = b1
        new_max = np.max(compute_correlations(h_try))
        if new_max < best_max - 1e-14:
            best = h_try
            best_max = new_max
    return best, best_max


def half_position_search(h, n):
    best = h.copy()
    best_max = np.max(compute_correlations(best))
    frac_mask = (best > 0.01) & (best < 0.99)
    frac_indices = np.where(frac_mask)[0]
    for fi in frac_indices:
        orig = best[fi]
        for pos in range(n):
            if pos == fi:
                continue
            h_try = best.copy()
            h_try[fi] = best[pos]
            h_try[pos] = orig
            nm = np.max(compute_correlations(h_try))
            if nm < best_max - 1e-14:
                best = h_try
                best_max = nm
                orig = best[fi]
    return best, best_max
