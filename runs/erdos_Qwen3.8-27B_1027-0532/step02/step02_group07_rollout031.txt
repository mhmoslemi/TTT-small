```python
    lr = 1.0  # learning rate, will be adapted
    
    h_curr = h.copy()
    h_prev = h.copy()
    t_mom = 1.0  # momentum parameter

        if time.time() - t0 > time_budget:
            break

        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')

        # d(corr[p])/d(h[j]) = (1-h[j+p-(n-1)]) - h[j-p+(n-1)] (with bounds)
```

```python
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, alpha, maxiter, bounds):

        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None
```

Let me focus on making the perturbation step smarter. After CD converges, I know:
- The current maximum correlation value
- Which lags are active (at or near the maximum)
- The "bottleneck" lags that prevent further improvement

A good perturbation would try to specifically reduce the active correlations. Here's an idea:
- For each active lag k, compute the gradient of corr_k with respect to h
- Find a direction that reduces all active correlations simultaneously
- Take a small step in that direction (projected to stay feasible)

This is essentially a subgradient step, but focused on the active set.

Let me implement this properly:

```python
def subgradient_step(h, alpha_smooth=100, step_size=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    
    # Soft weights
    weights = np.exp(alpha_smooth * (corr - m))
    weights /= np.sum(weights)
    
    # Compute gradient
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if weights[p] < 1e-10:
            continue
        k = p - (n - 1)
        # d(corr[p])/d(h[j]) = (1-h[j+k]) - h[j-k] (with bounds)
        # Positive part: for j where 0 <= j+k < n
        js = max(0, -k)
        je = min(n-1, n-1-k)
        if js <= je:
            grad[js:je+1] += weights[p] * (1.0 - h[js+k:je+1+k])
        # Negative part: for j where 0 <= j-k < n
        js2 = max(0, k)
        je2 = min(n-1, n-1+k)
        if js2 <= je2:
            grad[js2:je2+1] -= weights[p] * h[js2-k:je2+1-k]
    
    # Take step
    h_new = h - step_size * grad
    h_new = project_sum(h_new, n / 2.0)
    return h_new
```

```python
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(42):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    else:

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h

        je2 = min(n - 1, n - 1 + k)

                else:
                    best_h = h_cd

    if time_left() > 30:
        h_cd = coordinate_descent(best_h, n_sweeps=1000, time_budget=min(30, time_left() - 5))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 15:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [20000, 100000, 500000]:
            if time_left() < 8:
                break
            res_x = slsqp_optimize(best_h, alpha, 400, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
```
