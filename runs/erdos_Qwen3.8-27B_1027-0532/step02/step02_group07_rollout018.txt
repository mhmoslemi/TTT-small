def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        # Random ordering to avoid systematic bias
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            t_opt, slopes = optimize_coordinate_v2(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    h = project_sum(h, n / 2.0)
    return h


def annealed_cd(h, time_budget=100.0, n_restarts=20):
    """Coordinate descent with annealed perturbations"""
    n = len(h)
    best_h = h.copy()
    best_score = evaluate_score(best_h)
    t0 = time.time()
    rng = np.random.RandomState(42)
    
    for restart in range(n_restarts):
        if time.time() - t0 > time_budget:
            break
        frac = restart / max(n_restarts - 1, 1)
        # Decreasing perturbation scale
        scale = 0.3 * (1 - frac) + 0.01
        tb = min(5.0, (time_budget - (time.time() - t0) - 10) / max(n_restarts - restart, 1))
        if tb < 0.5:
            break
        
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_sum(h_pert, n / 2.0)
        
        h_cd = coordinate_descent(h_pert, n_sweeps=300, tol=1e-12, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
    
    return best_h, best_score


def smooth_gradient_step(h, alpha=100, lr=0.01):
    """One step of smoothed gradient descent on the max correlation"""
    corr = compute_correlations(h)
    weights = smooth_max_grad(corr, alpha)[1]
    
    # Compute gradient: for each h[i], d(corr[p])/d(h[i]) = slopes[i] (from coordinate analysis)
    # But we need the weighted sum: grad[i] = sum_p weights[p] * d(corr[p])/d(h[i])
    # This is O(n²) which is expensive.
    
    # Alternative: use the fact that the gradient of corr wrt h is related to the correlation of weights
    # Actually, let me think about this differently.
    # corr[p] = sum_i h[i+k_p] * (1-h[i]) where k_p = p-(n-1)
    # d(corr[p])/d(h[j]) = (1-h[j-k_p]) - h[j+k_p] (with boundary handling)
    # So grad[j] = sum_p weights[p] * ((1-h[j-k_p]) - h[j+k_p])
    #            = sum_p weights[p] * (1-h[j-k_p]) - sum_p weights[p] * h[j+k_p]
    
    n = len(h)
    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - (n-1)  # -n+1 to n-1
    
    for j in range(n):
        for k_idx, k in enumerate(k_vals):
            if k >= 0:
                # h[j] in h[i+k] position: i = j-k, valid if 0<=j-k<=n-1-k
                if 0 <= j-k and j-k <= n-1-k:
                    grad[j] += weights[k_idx] * (1 - h[j-k])
                # h[j] in (1-h[i]) position: i=j, valid if j<=n-1-k
                if j <= n-1-k:
                    grad[j] -= weights[k_idx] * h[j+k]
            else:
                k_abs = -k
                # h[j] as h[i]: i=j, valid if 0<=j<=n-1-k_abs
                if 0 <= j <= n-1-k_abs:
                    grad[j] += weights[k_idx] * (1 - h[j+k_abs])
                # h[j] in (1-h[i+k_abs]): i+k_abs=j, i=j-k_abs, valid if 0<=j-k_abs<=n-1-k_abs
                if 0 <= j-k_abs and j-k_abs <= n-1-k_abs:
                    grad[j] -= weights[k_idx] * h[j-k_abs]
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = project_sum(h_new, n / 2.0)
    return h_new

def targeted_perturbation(h, rng, epsilon=0.1):
    """Perturb h in a direction that reduces the worst correlations"""
    n = len(h)
    corr = compute_correlations(h)
    
    # Find the top 3 worst lags
    top_k = 3
    worst_indices = np.argsort(corr)[-top_k:]
    
    # Compute the subgradient (average gradient over worst lags)
    grad = np.zeros(n)
    for idx in worst_indices:
        k = idx - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            # d(corr[k])/d(h[i])
            if k >= 0:
                # h[i] in h[j+k] position: j = i-k, valid if 0<=i-k<=n-1-k

                # h[i] in (1-h[j]) position: j=i, valid if i<=n-1-k

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(40):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0, rng=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    if rng is None:
        rng = np.random.RandomState(0)

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        order = rng.permutation(n)
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def compute_max_gradient(h, corr, top_k=5):
    n = len(h)
    worst_indices = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    for idx in worst_indices:
        k = k_vals[idx]
        abs_k = abs(k)
        for j in range(n):
            val = 0.0
            if j >= abs_k:
                val += (1.0 - h[j - abs_k])
            if j + abs_k < n:
                val -= h[j + abs_k]
            grad[j] += val
    grad /= top_k
    return grad


def targeted_perturbation(h, rng, epsilon=0.1, top_k=5):
    n = len(h)
    corr = compute_correlations(h)
    grad = compute_max_gradient(h, corr, top_k=top_k)
    noise = rng.randn(n) * epsilon * 0.3
    h_new = h - epsilon * grad + noise
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    return h_new


def upsample_nn(h, new_n):
    n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        h_new[i] = h[min(i * n // new_n, n - 1)]
    return h_new


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_base = len(init)
    init = project_sum(init, n_base / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: CD at base resolution
    if time_left() > 40:
        h_cd = coordinate_descent(best_h, n_sweeps=1000, time_budget=30, rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: Targeted perturbation + CD at base resolution
    n_pert_base = 60
    for i in range(n_pert_base):
        if time_left() < 80:
            break
        frac = i / max(n_pert_base - 1, 1)
        eps = 0.15 * (1.0 - frac) + 0.02
        h_pert = targeted_perturbation(best_h, rng, epsilon=eps, top_k=5)
        tb = min(3.0, (time_left() - 70) / max(n_pert_base - i, 1))
        if tb < 0.3:
            tb = 0.3
        h_cd = coordinate_descent(h_pert, n_sweeps=200, time_budget=tb, rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 3: Upscale to 2x
    n2 = 2 * n_base
    h2 = upsample_nn(best_h, n2)
    h2 = project_sum(h2, n2 / 2.0)
    if time_left() > 100:
        h2 = coordinate_descent(h2, n_sweeps=500, time_budget=60, rng=rng)
        s = evaluate_score(h2)
        if s < best_score:
            best_score = s
            best_h = h2

    # Phase 4: Targeted perturbation + CD at 2x resolution
    n_cur = len(best_h)
    n_pert_2x = 40
    for i in range(n_pert_2x):
        if time_left() < 80:
            break
        frac = i / max(n_pert_2x - 1, 1)
        eps = 0.10 * (1.0 - frac) + 0.01
        h_pert = targeted_perturbation(best_h, rng, epsilon=eps, top_k=4)
        tb = min(3.0, (time_left() - 60) / max(n_pert_2x - i, 1))
        if tb < 0.3:
            tb = 0.3
        h_cd = coordinate_descent(h_pert, n_sweeps=150, time_budget=tb, rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    # Phase 5: SLSQP polish at current resolution
    if time_left() > 60:
        n_sl = len(best_h)
        bounds = [(0.0, 1.0)] * n_sl
        for alpha in [500, 2000, 10000]:
            if time_left() < 20:
                break
            maxiter = max(50, 200000 // n_sl)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_sl / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 6: CD after SLSQP
    if time_left() > 50:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(40, time_left() - 20), rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 7: Try 3x resolution if we have time
    n3 = 3 * n_base
    if time_left() > 200:
        x_old = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n3, endpoint=False)
        h3 = np.interp(x_new, x_old, best_h)
        h3 = project_sum(h3, n3 / 2.0)
        h3 = coordinate_descent(h3, n_sweeps=300, time_budget=min(60, time_left() - 80), rng=rng)
        s = evaluate_score(h3)
        if s < best_score:
            best_score = s
            best_h = h3
            # Quick perturb + CD at 3x
            for i in range(15):
                if time_left() < 60:
                    break
                eps = 0.05 * (1 - i / 15) + 0.01
                h_pert = targeted_perturbation(best_h, rng, epsilon=eps, top_k=4)
                tb = min(3.0, (time_left() - 40) / max(15 - i, 1))
                h_cd = coordinate_descent(h_pert, n_sweeps=100, time_budget=tb, rng=rng)
                s2 = evaluate_score(h_cd)
                if s2 < best_score:
                    best_score = s2
                    best_h = h_cd

    # Phase 8: Final aggressive CD
    if time_left() > 40:
        h_cd = coordinate_descent(best_h, n_sweeps=1000, time_budget=min(60, time_left() - 10), rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 9: Final SLSQP
    if time_left() > 15:
        n_fin = len(best_h)
        bounds_fin = [(0.0, 1.0)] * n_fin
        for alpha in [5000, 50000]:
            if time_left() < 8:
                break
            res_x = slsqp_optimize(best_h, alpha, 400, bounds_fin)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_fin / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Final CD
    if time_left() > 10:
        h_cd = coordinate_descent(best_h, n_sweeps=300, time_budget=min(20, time_left() - 2), rng=rng)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
