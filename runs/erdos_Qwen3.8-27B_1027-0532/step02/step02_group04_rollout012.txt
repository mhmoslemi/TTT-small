        if s > target_sum:
            hi = mid
        else:

    """Round h to binary with exactly n_ones ones."""
    idx = np.argsort(-h)
    result = np.zeros(len(h))
    result[idx[:n_ones]] = 1.0

def discrete_best_swap(h, max_passes=50):
    """Best-improvement single swap local search for binary (or near-binary) h."""
    n = len(h)
    h = h.copy().astype(float)
    corr = compute_correlations(h)
    current_max = np.max(corr)
    
    # Precompute the add_effect matrix
    # add_effect[j, k] = (1-h[j-d])*valid(j-d) - h[j+d]*valid(j+d)

    J = np.arange(n)[:, None].astype(float)
    D = np.arange(-(n-1), n)[None, :].astype(float)
    pos1 = J - D  # j - d
    pos2 = J + D  # j + d
    valid1 = (pos1 >= 0) & (pos1 < n)
    valid2 = (pos2 >= 0) & (pos2 < n)
    
    # For the current h, compute add_effect
    def compute_add_effect(h_cur):
        h_vis1 = np.where(valid1, h_cur[np.clip(pos1.astype(int), 0, n-1)], 0.0)
        h_vis2 = np.where(valid2, h_cur[np.clip(pos2.astype(int), 0, n-1)], 0.0)
        return (1.0 - h_vis1) * valid1.astype(float) - h_vis2 * valid2.astype(float)
    
    add_eff = compute_add_effect(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for pass_num in range(max_passes):
        best_improvement = 0
        best_i = -1
        best_j = -1
        best_new_max = current_max
        
        for i in ones:
            # Compute remove_effect for this i
            lag_vals = np.arange(-(n-1), n)
            # remove_effect[k] = -(1-h[i-d])*valid(i-d) + h[i+d]*valid(i+d)

            valid_rm1 = (pos_rm1 >= 0) & (pos_rm1 < n)
            valid_rm2 = (pos_rm2 >= 0) & (pos_rm2 < n)

            # We want to minimize max_k new_corr[j, k]
            # = min_j max_k (corr[k] + remove_eff[k] + add_eff[j, k])

            new_max_all = np.max(base[None, :] + add_eff, axis=1)  # shape (n,)
            
            # Mask out non-zero positions
            new_max_all[ones] = np.inf
            
            j = np.argmin(new_max_all)
            if new_max_all[j] < best_new_max:
                best_new_max = new_max_all[j]
                best_i = i
                best_j = j
                best_improvement = current_max - best_new_max
        
        if best_i >= 0 and best_improvement > 1e-12:
            # Apply swap
            h[best_i] = 0.0
            h[best_j] = 1.0

            zeros = np.where(h == 0.0)[0]
            add_eff = compute_add_effect(h)
        else:
            break
    
    return h, current_max
