from scipy.optimize import minimize
import time

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def power_sum_obj(h, p, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    ps = float(np.sum(overlaps**p)) ** (1.0/p)
    s = float(np.sum(h)) - n/2.0
    return ps + penalty * s * s

def power_sum_grad(h, p, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    ps = float(np.sum(overlaps**p)) ** (1.0/p)
    # d/dh of ps = (sum f^p)^(1/p - 1) * sum f^(p-1) * df/dh
    # = ps^(1-p) * sum f^(p-1) * df/dh
    weights = overlaps ** (p - 1) * (ps ** (1 - p))
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n-L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    s = float(np.sum(h)) - n/2.0
    grad += 2.0 * penalty * s
    return grad

import numpy as np

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def _overlap_grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-13:
            continue
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def lse_obj_grad(h, alpha, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = float(np.max(overlaps))
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = float(np.sum(exp_vals))
    weights = exp_vals / wsum
    val = m + float(np.log(wsum)) / alpha
    s = float(np.sum(h)) - n / 2.0
    val += penalty * s * s
    grad = _overlap_grad_weighted(h, weights)
    grad += 2.0 * penalty * s
    return val, grad


def slsqp_obj(h, alpha):
    overlaps = compute_overlaps(h)
    m = float(np.max(overlaps))
    return m + float(np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha)

    n = len(h)
    overlaps = compute_overlaps(h)
    m = float(np.max(overlaps))
    exp_vals = np.exp(alpha * (overlaps - m))
    weights = exp_vals / float(np.sum(exp_vals))
    return _overlap_grad_weighted(h, weights)


def sum_eq(h):
    return float(np.sum(h)) - len(h) / 2.0


def sum_eq_jac(h):
    return np.ones(len(h))


def optimize_slsqp(h0, alpha, maxiter, bounds):
    try:
        res = minimize(slsqp_obj, h0, args=(alpha,), method='SLSQP',
                       jac=slsqp_grad, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return res.x
    except Exception:
        return None


def optimize_lbfgsb(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(lse_obj_grad, h0, args=(alpha, penalty), method='L-BFGS-B',
                       jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-20,
                                'maxls': 50})
        return res.x
    except Exception:
        return None


def var_obj_grad(h, penalty):
    n = len(h)
    overlaps = compute_overlaps(h)
    dk = len(overlaps)
    mean = float(np.mean(overlaps))
    var = float(np.mean((overlaps - mean) ** 2))
    s = float(np.sum(h)) - n / 2.0
    val = var + penalty * s * s
    coeffs = 2.0 * (overlaps - mean) / dk
    grad = _overlap_grad_weighted(h, coeffs)
    grad += 2.0 * penalty * s
    return val, grad
