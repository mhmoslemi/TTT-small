def lse_grad(h, alpha):
    n = len(h)

    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))

from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = float(np.sum(exp_vals))
    weights = exp_vals / wsum
    threshold = 1e-12
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def make_lse_obj(alpha, penalty):
    def obj(h):
        n = len(h)
        val = lse_val(compute_overlaps(h), alpha)
        s = float(np.sum(h)) - n / 2.0
        val += penalty * s * s
        return val
    def grad(h):
        n = len(h)
        g = lse_grad(h, alpha).copy()
        s = float(np.sum(h)) - n / 2.0
        g += 2.0 * penalty * s
        return g
    return obj, grad


def lbfgsb_optimize(h0, alpha, penalty, maxiter):
    n = len(h0)
    obj, grad = make_lse_obj(alpha, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18, 'maxls': 50})
        return res.x
    except Exception:
        return None


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_fun(h):
    return np.sum(h) - len(h) / 2.0


def sum_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h0, args=(alpha,), method='SLSQP', jac=slsqp_grad,
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return res.x
    except Exception:
        return None

def generate_diverse_inits(n, rng):

    tm = np.zeros(n, dtype=float)

        v = i
        while v > 0:

    for block in [16, 32, 64]:
