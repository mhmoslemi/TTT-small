        return h0.copy()

    free_target = target_sum - fixed_sum

def _fix_sum(h, target_sum):

        for alpha in [80, 250, 600, 1500]:
            if time.time() > deadline - 25:
                break
            h_sl = _slsqp_smooth(h_curr, n, target_sum, alpha, maxiter=250)
            c_sl = _c5_of(h_sl)
            if c_sl < best_c5:
                best_c5 = c_sl
                best_h = h_sl.copy()
                best_n = n
            h_curr = h_sl

        # Phase 4: Subspace refinement
        if time.time() <= deadline - 20:
            h_sub = _subspace_refine(h_curr, n, target_sum, margin=0.02, maxiter=400)
            c_sub = _c5_of(h_sub)
            if c_sub < best_c5:
                best_c5 = c_sub
                best_h = h_sub.copy()
                best_n = n
            h_curr = h_sub

        # Phase 5: More SLSQP at higher alpha
        for alpha in [3000, 8000]:
            if time.time() > deadline - 20:
                break
            h_sl2 = _slsqp_smooth(h_curr, n, target_sum, alpha, maxiter=300)
            c_sl2 = _c5_of(h_sl2)
            if c_sl2 < best_c5:
                best_c5 = c_sl2
                best_h = h_sl2.copy()
                best_n = n
            h_curr = h_sl2

        # Phase 6: Subspace refinement again
        if time.time() <= deadline - 15:
            h_sub2 = _subspace_refine(h_curr, n, target_sum, margin=0.01, maxiter=500)
            c_sub2 = _c5_of(h_sub2)
            if c_sub2 < best_c5:
                best_c5 = c_sub2
                best_h = h_sub2.copy()
                best_n = n
            h_curr = h_sub2

        # Phase 7: Coordinate descent continuous
        if time.time() <= deadline - 15:
            h_cd = _coord_continuous(h_curr, n, target_sum, max_rounds=8, n_vals=7)
            c_cd = _c5_of(h_cd)
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()
                best_n = n
            h_curr = h_cd

        # Phase 8: Final SLSQP polish
        if time.time() <= deadline - 12:
            h_pol = _slsqp_smooth(h_curr, n, target_sum, 5000, maxiter=300)
            c_pol = _c5_of(h_pol)
            if c_pol < best_c5:
                best_c5 = c_pol
                best_h = h_pol.copy()
                best_n = n
            h_curr = h_pol

    # Try mirror descent from best
    if best_h is not None and time.time() <= deadline - 20:
        h_md, c_md = _mirror_descent(best_h, best_n, best_n/2.0, n_iters=300, eta0=0.3, target_c5=target_c5)
        if c_md < best_c5:
            best_c5 = c_md
            best_h = h_md.copy()

    # Final polish loop
    if best_h is not None:
        n_f = best_n
        t_f = n_f / 2.0
        for alpha in [2000, 5000, 12000]:
            if time.time() > deadline - 8:
                break
            h_f = _slsqp_smooth(best_h, n_f, t_f, alpha, maxiter=400)
            c_f = _c5_of(h_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f.copy()

        if time.time() <= deadline - 6:
            h_sr = _subspace_refine(best_h, n_f, t_f, margin=0.005, maxiter=500)
            c_sr = _c5_of(h_sr)
            if c_sr < best_c5:
                best_c5 = c_sr
                best_h = h_sr.copy()

        if time.time() <= deadline - 4:
            h_cd_f = _coord_continuous(best_h, n_f, t_f, max_rounds=5, n_vals=5)
            c_cd_f = _c5_of(h_cd_f)
            if c_cd_f < best_c5:
                best_c5 = c_cd_f
                best_h = h_cd_f.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
