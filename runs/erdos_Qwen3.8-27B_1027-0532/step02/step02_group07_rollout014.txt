    h_new = np.interp(x_new, x_old, h, period=None)
    return project_sum(h_new, n_new / 2.0)


def sine_init(n, freq, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.cos(np.pi * freq * x + phase)
    return project_sum(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_sum(init, n_init / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Initial CD
    if time_left() > 200:
        h_cd = coordinate_descent(best_h, n_sweeps=300, time_budget=80)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: Multi-resolution cycling
    for cycle in range(4):
        if time_left() < 120:
            break
        n_cur = len(best_h)
        n_up = min(3 * n_cur, 2500)
        if n_up <= n_cur:
            n_up = n_cur + 100

        h_up = upsample(best_h, n_up)
        h_up = coordinate_descent(h_up, n_sweeps=150, time_budget=min(50, time_left() - 60))

        if time_left() > 60:
            maxiter = max(50, 100000 // n_up)
            res_x = slsqp_smooth(h_up, 5000, maxiter, n_up)
            if res_x is not None:
                h_up = project_sum(np.clip(res_x, 0.0, 1.0), n_up / 2.0)
                h_up = coordinate_descent(h_up, n_sweeps=50, time_budget=min(15, time_left() - 30))

        h_down = downsample(h_up, n_cur)
        h_down = coordinate_descent(h_down, n_sweeps=100, time_budget=min(25, time_left() - 20))
        s = evaluate_score(h_down)
        if s < best_score:
            best_score = s
            best_h = h_down

    # Phase 3: SLSQP with increasing alpha on current resolution
    n_cur = len(best_h)
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time_left() < 40:
            break
        maxiter = max(100, 300000 // n_cur)
        res_x = slsqp_smooth(best_h, alpha, maxiter, n_cur)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new

    # Phase 4: CD after SLSQP
    if time_left() > 80:
        h_cd = coordinate_descent(best_h, n_sweeps=400, time_budget=min(60, time_left() - 20))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 5: Higher resolution attempt
    n_cur = len(best_h)
    for target_n in [2 * n_cur, 3 * n_cur]:
        if target_n > 3000 or time_left() < 100:
            break
        h_hi = upsample(best_h, target_n)
        h_hi = coordinate_descent(h_hi, n_sweeps=100, time_budget=min(40, time_left() - 50))
        if time_left() > 50:
            maxiter = max(50, 80000 // target_n)
            res_x = slsqp_smooth(h_hi, 10000, maxiter, target_n)
            if res_x is not None:
                h_hi = project_sum(np.clip(res_x, 0.0, 1.0), target_n / 2.0)
                h_hi = coordinate_descent(h_hi, n_sweeps=50, time_budget=min(15, time_left() - 30))
        h_back = downsample(h_hi, n_cur)
        h_back = coordinate_descent(h_back, n_sweeps=100, time_budget=min(20, time_left() - 20))
        s = evaluate_score(h_back)
        if s < best_score:
            best_score = s
            best_h = h_back

    # Phase 6: Diverse restarts
    n_cur = len(best_h)
    n_restarts = 40
    for i in range(n_restarts):
        if time_left() < 25:
            break
        frac = i / max(n_restarts - 1, 1)
        if i < 8:
            h_pert = best_h + rng.randn(n_cur) * 0.005
        elif i < 16:
            h_pert = best_h + rng.randn(n_cur) * (0.02 + 0.05 * frac)
        elif i < 24:
            h_pert = best_h + rng.randn(n_cur) * (0.05 + 0.10 * frac)
        elif i < 32:
            h_pert = best_h + rng.randn(n_cur) * (0.10 + 0.20 * frac)
        else:
            h_sine = sine_init(n_cur, rng.randint(1, 5), rng.uniform(0, 2 * np.pi))
            h_pert = 0.5 * best_h + 0.5 * h_sine
            h_pert = h_pert + rng.randn(n_cur) * 0.05

        h_pert = project_sum(h_pert, n_cur / 2.0)
        tb = min(4.0, max(1.0, (time_left() - 15) / max(n_restarts - i, 1)))
        h_cd = coordinate_descent(h_pert, n_sweeps=150, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 7: Final CD polish
    if time_left() > 50:
        h_cd = coordinate_descent(best_h, n_sweeps=600, time_budget=min(45, time_left() - 15))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 8: Final SLSQP
    if time_left() > 30:
        n_final = len(best_h)
        for alpha in [100000, 500000]:
            if time_left() < 15:
                break
            res_x = slsqp_smooth(best_h, alpha, 400, n_final)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
