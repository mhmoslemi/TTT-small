def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

    current_val = _c5_of(h)
    best = h.copy()
    best_val = current_val
    
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        # Random perturbation
        n_pert = rng.randint(1, max(2, n // 5) + 1)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + rng.uniform(-0.5, 0.5, n_pert), 0.0, 1.0)
        h_new = _fix_sum(h_new, target_sum)
        
        new_val = _c5_of(h_new)
        delta = new_val - current_val

            h = h_new
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
    
    return best, best_val

from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add

    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _subgrad_weighted(h, dx, m=5, alpha=100.0):
    """Compute weighted subgradient from top-m lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp(alpha * (corr - max_corr))
    total_w = weights.sum()
    if total_w < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / total_w
    top_m = np.argsort(weights)[::-1][:m]
    w_m = weights[top_m].copy()
    w_m = w_m / w_m.sum()
    g = np.zeros(n)
    for idx in range(len(top_m)):
        if w_m[idx] > 1e-10:
            g += w_m[idx] * _grad_corr_term(h, top_m[idx], dx)
    return g

def _nesterov_refine(h0, n, target_sum, n_iters, step0, m, w_alpha, 
                      decrease_rate=0.999, min_step=0.01,
                      patience=100, restart_factor=0.5):
    """Nesterov with adaptive step and restart."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    step = step0
    stall_count = 0
    
    for it in range(n_iters):
        g = _subgrad_weighted(y, dx, m=m, alpha=w_alpha)
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val - 1e-10:
            best_val = cv
            best = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > patience:
            # Restart: reduce step, reset momentum
            step *= restart_factor
            x = best.copy()
            y = best.copy()
            t = 1.0
            stall_count = 0
            if step < min_step:
                break
        
        step *= decrease_rate
    
    return best, best_val

def _balancing_search(h0, n, target_sum, n_iters, rng):

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):

    _global_target_sum = target_sum
    _global_n = n

        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h0

    return np.ones(_global_n)

def _coord_descent_01(h0, n, target_sum, max_rounds=30):
    """Binary coordinate descent: try setting each h[j] to 0 or 1."""
    h = h0.copy()
    current_val = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            # Try 0
            if old > 0:
                h[j] = 0.0
                c0 = _c5_of(h)
                if c0 < current_val - 1e-12:

                    improved = True
                else:
                    h[j] = old
            # Try 1

                c1 = _c5_of(h)
                if c1 < current_val - 1e-12:
                    current_val = c1
                    improved = True
                else:
                    h[j] = old
        if not improved:
            break
    return h
