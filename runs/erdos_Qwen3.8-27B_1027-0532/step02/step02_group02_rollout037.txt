                best_c5 = c5_try
                best = h_try.copy()
                h = h_try.copy()

Actually, let me reconsider the optimization strategy. The key challenge is that we're trying to minimize a max of many bilinear functions. This is a non-convex problem with many local minima.

A potentially better strategy:
1. **Fourier-based initialization**: Since we know the problem has a Fourier structure, initialize with a function whose Fourier coefficients are designed to spread the correlation energy.

2. **Alternating optimization**: Alternate between:
   a. Given the current h, find the lag k* that maximizes g_k(h)
   b. Optimize h to minimize g_{k*}(h) while keeping other g_k ≤ g_{k*}(h)

3. **Level method**: Set a target T and check if there exists h with g_k(h) ≤ T for all k. This is a feasibility problem that can be approached with semidefinite programming (SDP) relaxation.

    cons = {'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_cons_jac}

    best_x = h_init.copy()
    best_val = _c5_of(h_init)
    try:
        res = minimize(_slsqp_obj, h_init, jac=_slsqp_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_res = np.clip(res.x, 0.0, 1.0)
        c5_res = _c5_of(h_res)
        if c5_res < best_val:
            best_val = c5_res
            best_x = h_res
    except Exception:
        pass
    return best_x

def _upscale(h, new_n):
    old_n = len(h)
    x_old = np.arange(2 * old_n, dtype=float)
    h_ext = np.concatenate([h, h])
    x_new = np.linspace(0, 2 * old_n, new_n, endpoint=False)
    cs = CubicSpline(x_old, h_ext)
    h_new = cs(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _fix_sum(h_new, new_n / 2.0)

def _nm_refine(h, n_pert, scale_min, scale_max, deadline):
    n = len(h)
    target_sum = n / 2.0
    rng = np.random.RandomState(2024)
    best = h.copy()
    best_c5 = _c5_of(h)

    for _ in range(n_pert):
        if time.time() > deadline - 5:
            break
        d = rng.randn(n)
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm < 1e-12:
            continue
        d /= norm

        for s in [scale_min, -scale_min, scale_max, -scale_max,
                  (scale_min + scale_max) / 2, -(scale_min + scale_max) / 2]:
            if time.time() > deadline - 5:
                break
            h_try = _fix_sum(best + s * d, target_sum)
            c5_try = _c5_of(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best = h_try.copy()

    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 25

    best_h = None
    best_c5 = np.inf
    best_n = 0

    n1 = 100
    target_sum1 = n1 / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
        while len(h_start) < n1:
            h_start = np.concatenate([h_start, h_start[:n1 - len(h_start)]])
        h_start = h_start[:n1]
        h_start = _fix_sum(h_start, target_sum1)
    else:
        h_start = np.full(n1, 0.5)

    # Phase 1a: Aggressive subgradient at n=100
    t1 = deadline - budget_s * 0.28
    h_a, c5_a = _subgrad_optimize(h_start, n_iter=100000, lr0=0.45, decay=0.0001,
                                   top_k_schedule=[(0.0, 1), (0.4, 2), (0.7, 3)],
                                   momentum=0.95, check_every=400, deadline=t1)
    if c5_a < best_c5:
        best_c5, best_h, best_n = c5_a, h_a.copy(), n1

    # Phase 1b: Finer subgradient
    t1b = deadline - budget_s * 0.22
    h_b, c5_b = _subgrad_optimize(h_a, n_iter=80000, lr0=0.12, decay=0.0004,
                                   top_k_schedule=[(0.0, 3), (0.4, 5), (0.8, 10)],
                                   momentum=0.93, check_every=400, deadline=t1b)
    if c5_b < best_c5:
        best_c5, best_h, best_n = c5_b, h_b.copy(), n1
    h_cur = h_b if c5_b <= c5_a else h_a

    # Phase 1c: SLSQP polish at n=100
    for alpha in [40, 150, 500, 2000]:
        if time.time() > deadline - budget_s * 0.2:
            break
        h_pol = _slsqp_optimize(h_cur, alpha=alpha, maxiter=400, ftol=1e-15)
        h_pol = _fix_sum(h_pol, target_sum1)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5, best_h, best_n = c5_pol, h_pol.copy(), n1
        h_cur = h_pol

    # Phase 2: Upscale to n=300
    n2 = 300
    h2 = _upscale(best_h, n2)

    t2 = deadline - budget_s * 0.22
    h2_opt, c5_2 = _subgrad_optimize(h2, n_iter=120000, lr0=0.18, decay=0.00008,
                                      top_k_schedule=[(0.0, 1), (0.25, 3), (0.5, 5), (0.8, 10)],
                                      momentum=0.95, check_every=500, deadline=t2)
    if c5_2 < best_c5:
        best_c5, best_h, best_n = c5_2, h2_opt.copy(), n2

    # SLSQP at n=300
    h2_cur = h2_opt
    for alpha in [40, 200, 600]:
        if time.time() > deadline - budget_s * 0.18:
            break
        h2_pol = _slsqp_optimize(h2_cur, alpha=alpha, maxiter=300, ftol=1e-15)
        h2_pol = _fix_sum(h2_pol, n2 / 2.0)
        c5_2p = _c5_of(h2_pol)
        if c5_2p < best_c5:
            best_c5, best_h, best_n = c5_2p, h2_pol.copy(), n2
        h2_cur = h2_pol

    # Phase 3: Try n=200 from n=100 solution
    if time.time() < deadline - 80:
        n3 = 200
        h3 = _upscale(h_cur, n3)
        t3 = deadline - 60
        h3_opt, c5_3 = _subgrad_optimize(h3, n_iter=60000, lr0=0.2, decay=0.0002,
                                          top_k_schedule=[(0.0, 1), (0.3, 3), (0.7, 6)],
                                          momentum=0.94, check_every=400, deadline=t3)
        h3_cur = h3_opt
        for alpha in [50, 200]:
            if time.time() > deadline - 40:
                break
            h3_pol = _slsqp_optimize(h3_cur, alpha=alpha, maxiter=250, ftol=1e-15)
            h3_pol = _fix_sum(h3_pol, n3 / 2.0)
            c5_3p = _c5_of(h3_pol)
            if c5_3p < best_c5:
                best_c5, best_h, best_n = c5_3p, h3_pol.copy(), n3
            h3_cur = h3_pol

    # Phase 4: Try a different initialization at n=100
    if time.time() < deadline - 80:
        rng = np.random.RandomState(seed + 7)
        x = np.linspace(0, 2, n1, endpoint=False)
        h_alt = 0.5 + 0.15 * np.sin(np.pi * x) + 0.05 * np.sin(2 * np.pi * x) + 0.03 * rng.randn(n1)
        h_alt = _fix_sum(h_alt, target_sum1)
        t_alt = deadline - 50
        h_alt_opt, c5_alt = _subgrad_optimize(h_alt, n_iter=50000, lr0=0.3, decay=0.0002,
                                               top_k_schedule=[(0.0, 1), (0.5, 3), (0.8, 5)],
                                               momentum=0.94, check_every=400, deadline=t_alt)
        if time.time() < deadline - 30:
            h_alt_pol = _slsqp_optimize(h_alt_opt, alpha=200, maxiter=200, ftol=1e-15)
            h_alt_pol = _fix_sum(h_alt_pol, target_sum1)
            c5_alt = _c5_of(h_alt_pol)
        if c5_alt < best_c5:
            best_c5, best_h, best_n = c5_alt, h_alt_pol.copy(), n1

    # Phase 5: Final NM refinement
    if time.time() < deadline - 10 and best_h is not None:
        best_h, best_c5 = _nm_refine(best_h, n_pert=40, scale_min=0.02,
                                       scale_max=0.08, deadline=deadline)

    # Final: ensure sum constraint is exact
    if best_h is None:
        best_h = h_start
        best_c5 = _c5_of(best_h)
        best_n = n1

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)

    return (best_h, best_c5, best_n)
