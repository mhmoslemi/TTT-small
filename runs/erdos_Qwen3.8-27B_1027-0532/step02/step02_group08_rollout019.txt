import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1].copy()
    g = 1.0 - h
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad


def top_k_max_obj_grad(h, k_top, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    if k_top is None or k_top >= len(corrs):
        return smooth_max_obj_grad(h, alpha)
    top_idx = np.argpartition(corrs, -k_top)[-k_top:]
    top_corrs = corrs[top_idx]
    m = np.max(top_corrs)
    shifted = alpha * (top_corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_full = np.zeros(len(corrs))
    w_full[top_idx] = weights
    w_rev = w_full[::-1].copy()
    g = 1.0 - h
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, w_full, mode='full')
    idx_arr = np.arange(n) + n - 1
    grad = conv1[idx_arr] - conv2[idx_arr]
    return obj, grad


def penalty_obj(h, alpha, mu, k_top):
    obj, grad = top_k_max_obj_grad(h, k_top, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad


def opt_lbfgsb(h0, alpha, mu, k_top, maxiter, bounds):
    res = minimize(
        penalty_obj, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu, k_top), jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13,
                 'iprint': 0, 'maxls': 100, 'maxfun': maxiter * 50}
    )
    return res.x


def gradient_step_zeromean(h, alpha, step_size, n_steps):
    for _ in range(n_steps):
        _, grad = smooth_max_obj_grad(h, alpha)
        grad_zm = grad - np.mean(grad)
        h = h - step_size * grad_zm
        h = project_sum(h, len(h) / 2.0)
    return h


def binary_local_search(h, n_iter, rng, t_start=0.4):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    half = n // 2
    while np.sum(h_bin) > half:
        oi = np.where(h_bin == 1)[0]

        zi = np.where(h_bin == 0)[0]
        h_bin[zi[rng.randint(len(zi))]] = 1
    corr = compute_correlations(h_bin)

    best_max = curr_max

        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        new_corr = compute_correlations(h_bin)

                best_bin = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
