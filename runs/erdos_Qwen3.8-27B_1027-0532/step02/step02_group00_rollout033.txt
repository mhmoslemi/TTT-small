_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    vals = h[idx] + diff / len(idx)
    h[idx] = np.clip(vals, 0.0, 1.0)
    # If clipping changed the sum, redistribute
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        
        # For each lag k from -(n-1) to (n-1):
        # c_k = sum_n h[n] * g[n-k] * dx <= t

        for k in range(-(n-1), n):
            coeff = np.zeros(n)
            # n-k must be in [0, n), so n must be in [k, k+n)
            # and n must be in [0, n)
            lo = max(0, k)
            hi = min(n, k + n)

                coeff[lo:hi] = g[lo-k:hi-k] * dx
            constraints.append(coeff @ h_var <= t_var)

                break

            break

        h = project_feasible(h, target)

            best_c5 = c5_val
            best_h = h.copy()

            if damping <= 0.1 and it > 50:

def smooth_obj(h, alpha):

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    _global_target = target

            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr) - (n - 1)  # The lag with maximum correlation

            g2 = h[i + k]

            best_score = c5_lp
            best_h = h_lp

    # Phase 3: SLSQP refinement from best LP solution
    if tleft() > 120:
        h_refined = slsqp_round(best_h, n, target,
                                 [10, 50, 150, 400, 1000], 1500, 1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 4: SLSQP from perturbed versions
    for noise in [0.02, 0.05, 0.1]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_refined = slsqp_round(h_pert, n, target,
                                 [20, 100, 300, 800], 800, 1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 5: Try higher resolution grids with alternating LP
    if tleft() > 200:
        for n_try in [500, 700, 900]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(120, tleft() * 0.25)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try,
                                          max_iter=4000, time_limit=lp_time,
                                          damping_init=0.6, reg=1e-4)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try
            if tleft() < 80:
                break

    # Phase 6: Subgradient improvement on best solution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                           max_steps=5000, time_limit=min(80, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Final SLSQP polish with high alpha
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = slsqp_round(best_h, n_use, target_use,
                                   [500, 1000, 2000, 3000], 800, 1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Final feasibility check and return
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
