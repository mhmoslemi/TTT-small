        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    n_half = n - 1
    grad = np.zeros(n)
    threshold = 1e-12
    for k in range(len(corr)):
        w = weights[k]
        if w < threshold:
            continue
        lag = k - n_half
        grad += w * _corr_grad(h, lag)
    
    obj = (m + np.log(total) / alpha) * dx
    return obj, grad * dx

def _nesterov_optimize(h0, n_iter, alpha_start, alpha_end, lr0, lr_decay, deadline):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    momentum = 0.9
    no_improve = 0
    patience = n_iter // 3
    
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        frac = min(1.0, t / max(1, n_iter - 1))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr0 / (1.0 + t * lr_decay)
        
        obj_y, grad_y = _lse_grad(y, alpha)
        v = momentum * v + grad_y
        h = _fix_sum(y - lr * v, target_sum)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = h + (momentum / beta) * (h - y)
        y = np.clip(y, 0.0, 1.0)
        
        if (t + 1) % 200 == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 200
            if no_improve > patience:
                break
    
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    return best_h, best_c5

def _slsqp_polish(h0, n, alpha, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h0)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        return val * dx
    
    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < obj(h0):
            return res.x
    except Exception:
        pass
    return h0.copy()

def _make_init(n, seed, variant):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    if variant == 0:

    elif variant == 1:

    elif variant == 2:
        return np.clip(0.5 + 0.4 * np.sin(2 * np.pi * x * 2), 0, 1)
    elif variant == 3:
        return np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x * 3) + 0.2 * np.sin(2 * np.pi * x * 5), 0, 1)
    elif variant == 4:
        h = (np.sin(2 * np.pi * x * rng.randint(1, 8)) > 0).astype(float)
        return h + 0.1 * rng.randn(n)
    elif variant == 5:

        return h
    else:

            width = rng.randint(5, n // 3)
            for j in range(n):

                h[j] = max(h[j], np.exp(-d ** 2 / (2 * width ** 2)))
        return h + 0.05 * rng.randn(n)

    diff = target_sum - h.sum()

    else:

            h[i] -= sub
            diff += sub

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = alpha * (corr - m)

    for ell in range(1, n):
        wp = weights[nh + ell]
        wn = weights[nh - ell]

        bc = c

    dx = 2.0 / n
    def ob(x):

        m = np.max(c)
        return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * dx
