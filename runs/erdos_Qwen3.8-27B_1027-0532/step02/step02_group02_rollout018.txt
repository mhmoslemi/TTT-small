```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _softmax_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft-max weights
    max_corr = np.max(corr)
    p = np.exp(alpha * (corr - max_corr))
    p /= np.sum(p)
    
    # Gradient of soft-max objective w.r.t. h
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if p[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        # c[lag] = sum_n h[n+lag]*(1-h[n]) for lag > 0
        # dc/dh[i] = (1-h[i-lag]) - h[i+lag] (valid terms)
        if lag == 0:
            # c[0] = sum h[n]*(1-h[n]), dc/dh[i] = 1-2h[i]
            grad += p[k_idx] * (1.0 - 2.0 * h)
        elif lag > 0:
            # c[lag] = sum_{n=0}^{n-1-lag} h[n+lag]*(1-h[n])
            # dc/dh[i]: h[i] appears as h[n+lag] when n=i-lag, and as h[n] when n=i
            # As h[n+lag]: contributes (1-h[n]) = (1-h[i-lag]) at n=i-lag
            # As (1-h[n]): contributes -h[n+lag] = -h[i+lag] at n=i
            # So dc/dh[i] = (1-h[i-lag]) [if i-lag >= 0 and i-lag <= n-1-lag, i.e., lag <= i <= n-1]
            #              - h[i+lag] [if i+lag < n, i.e., 0 <= i <= n-1-lag]
            # For i in [0, lag-1]: only the -h[i+lag] term (since i-lag < 0)
            # For i in [lag, n-1-lag]: both terms
            # For i in [n-lag, n-1]: only the (1-h[i-lag]) term (since i+lag >= n)
            
            # Vectorized:
            # Term 1: (1-h[i-lag]) for i >= lag
            term1 = np.zeros(n)
            term1[lag:] = 1.0 - h[:n-lag]
            # Term 2: -h[i+lag] for i <= n-1-lag
            term2 = np.zeros(n)
            term2[:n-lag] = -h[lag:]
            grad += p[k_idx] * (term1 + term2)
        else:
            # lag < 0, let L = -lag
            # c[-L] = sum_{n=L}^{n-1} h[n-L]*(1-h[n])
            # = sum_{j=0}^{n-1-L} h[j]*(1-h[j+L])  [j = n-L]
            # dc/dh[i]: h[i] appears as h[j] when j=i (contributes (1-h[i+L]))
            #           and as h[j+L] when j+L=i, j=i-L (contributes -h[i-L])
            # For i in [0, L-1]: only (1-h[i+L]) term
            # For i in [L, n-1-L]: both terms  
            # For i in [n-L, n-1]: only -h[i-L] term
            L = -lag
            term1 = np.zeros(n)
            term1[:n-L] = 1.0 - h[L:]
            term2 = np.zeros(n)
            term2[L:] = -h[:n-L]
            grad += p[k_idx] * (term1 + term2)
    
    return grad
```

```python
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            if m * alpha > 500:
                # Numerical stability
                return np.sum(corr) / len(corr)
            return np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha + m

            m = np.max(corr)
            p = np.exp(alpha * (corr - m))
            p /= np.sum(p)
```

```python
def _compute_grad(h, p):
    n = len(h)
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 1e-8
    significant = np.where(p > threshold)[0]
    
    for k_idx in significant:
        w = p[k_idx]
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] += w * (-h[lag:])
        else:
            L = -lag
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] += w * (-h[:n-L])
    
    return grad
```

This is vectorized within each lag and only processes significant lags. For large alpha, only a few lags are significant, making this very fast.

```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _softmax_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = (log_sum + alpha * m) / alpha
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = p[k_idx]
        if w < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] += -w * h[:n - L]
    
    return obj, grad

def _slsqp_solve(h_init, n, alpha, maxiter, ftol, deadline):
    target_sum = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 500)
        return float(np.log(np.sum(np.exp(shifted))) / alpha + m)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p = exp_vals / np.sum(exp_vals)
        
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            w = p[k_idx]
            if w < 1e-10:
                continue
            lag = k_idx - (n - 1)
            if lag == 0:
                grad += w * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:] += w * (1.0 - x[:n - lag])
                grad[:n - lag] += -w * x[lag:]
            else:
                L = -lag
                grad[:n - L] += w * (1.0 - x[L:])
                grad[L:] += -w * x[:n - L]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

    h = h_init.copy()

    best_c5 = _c5_of(h)
    no_improve = 0

def _make_binary_h(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.randint(0, 2, size=n).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:n_ones - target]] = 0.0
    return h

def _make_structured_h(n, seed, variant):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    
    if variant == 0:
        return np.full(n, 0.5)
    elif variant == 1:
        return _make_binary_h(n, seed)
    elif variant == 2:
        # Two-scale pattern
        h = 0.5 + 0.3 * np.sin(2 * np.pi * n * x / 2 * rng.randint(1, 5))
        return np.clip(h, 0, 1)
    elif variant == 3:
        # Block pattern
        block_size = n // rng.randint(3, 8)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i // block_size) % 2 == 0 else 0.0
        return h
    elif variant == 4:

        h = rng.rand(n)
        # Smooth
        kernel = np.ones(5) / 5
        h = np.convolve(h, kernel, mode='same')

    else:
```

```python
def _compute_grad(h, p, n):
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        w = p[k_idx]
        if w < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂C_lag/∂h[i] = (1-h[i-lag]) * 1[i >= lag] + (-h[i+lag]) * 1[i <= n-1-lag]
            # Term 1: (1-h[i-lag]) for i = lag, ..., n-1
            grad[lag:] += w * (1.0 - h[:n-lag])
            # Term 2: -h[i+lag] for i = 0, ..., n-1-lag
            grad[:n-lag] += -w * h[lag:]
        else:
            L = -lag
            # ∂C_{-L}/∂h[i] = (1-h[i+L]) * 1[i <= n-1-L] + (-h[i-L]) * 1[i >= L]
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] += -w * h[:n-L]
    return grad
```
