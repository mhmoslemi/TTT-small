def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(500):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5

def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_grad_full(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    threshold = 1e-10
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)

    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

    m = np.max(overlaps)

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def lse_val(x, alpha):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)


def lse_grad(h, alpha):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    thresh = 1e-8
    sig = np.where(weights > thresh)[0]
    for idx in sig:
        w = weights[idx]
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad


def penalty_obj(h, alpha, penalty):
    n = len(h)
    ov = compute_overlaps(h)
    obj = lse_val(ov, alpha)
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    return obj


def penalty_grad(h, alpha, penalty):
    n = len(h)
    s = float(np.sum(h)) - n / 2.0
    g = lse_grad(h, alpha)
    g += 2.0 * penalty * s
    return g


def penalty_obj_grad(h, alpha, penalty):
    return (penalty_obj(h, alpha, penalty), penalty_grad(h, alpha, penalty))


def do_lbfgs(h0, alpha, penalty, maxiter, bounds):
    try:
        res = minimize(penalty_obj_grad, h0, args=(alpha, penalty),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18})
        return res.x
    except Exception:
        return None


def slsqp_obj(h, alpha):
    return lse_val(compute_overlaps(h), alpha)


def slsqp_grad(h, alpha):
    return lse_grad(h, alpha)


def sum_fun(h):
    return np.sum(h) - len(h) / 2.0


def sum_jac(h):
    return np.ones(len(h))


def do_slsqp(h0, alpha, maxiter, bounds):
    try:
        res = minimize(slsqp_obj, h0, args=(alpha,), method='SLSQP',
                       jac=slsqp_grad, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}],
                       options={'maxiter': maxiter, 'iprint': 0})
        return res.x
    except Exception:
        return None


def coord_deltas(h, i):
    n = len(h)
    deltas = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        L = idx - (n - 1)
        if L >= 0:
            d = 0.0
            if i <= n - 1 - L:
                d += 1.0 - h[i + L]
            if i >= L:
                d -= h[i - L]
            deltas[idx] = d
        else:
            M = -L
            d = 0.0
            if i >= M:
                d += 1.0 - h[i - M]
            if i <= n - 1 - M:
                d -= h[i + M]
            deltas[idx] = d
    return deltas


def coord_descent(h, n_sweeps, n_grid, time_fn, target):
    n = len(h)
    h = h.copy()
    best_score = evaluate_score(h)
    grid = np.linspace(0.0, 1.0, n_grid)
    for sweep in range(n_sweeps):
        if time_fn() < 15:
            break
        overlaps = compute_overlaps(h)
        for i in range(n):
            old = h[i]
            deltas = coord_deltas(h, i)
            base = overlaps - old * deltas
            vals = base[None, :] + grid[:, None] * deltas[None, :]
            max_vals = np.max(vals, axis=1)
            best_idx = np.argmin(max_vals)
            best_t = grid[best_idx]
            # Refine around best_t
            if 0 < best_idx < n_grid - 1:
                lo_t = grid[best_idx - 1]
                hi_t = grid[best_idx + 1]
                fine_grid = np.linspace(lo_t, hi_t, 20)
                vals2 = base[None, :] + fine_grid[:, None] * deltas[None, :]
                max_vals2 = np.max(vals2, axis=1)
                best_t = fine_grid[np.argmin(max_vals2)]
            h[i] = best_t
            overlaps += (best_t - old) * deltas
        h = project_box_sum(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
    return h, best_score

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 15.0 - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)

    # Phase 1: L-BFGS-B with increasing alpha
    for alpha in [50, 150, 500, 1500, 5000]:
        if time_left() < 80:
            break
        penalty = alpha * 25.0
        bounds = [(0.0, 1.0)] * n

        res_x = do_lbfgs(h, alpha, penalty, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)

            if np.isfinite(s) and s < best_score:

            h = h_new

    # Phase 3: Coordinate descent at original resolution

        h_cd, s_cd = coord_descent(best_h, 40, 100, time_left, n / 2.0)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd.copy()

    # Phase 4: Random perturbation restarts
    for restart in range(4):
        if time_left() < 80:
            break
        noise_scale = 0.02 + 0.01 * restart
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_box_sum(h_pert, n / 2.0)
        for alpha in [1000, 3000]:

            penalty = alpha * 30.0

                h_pert = h_new
        # Quick coordinate descent on perturbed result

        h_init = np.interp(x_new, x_old, best_h)

        for alpha in [500, 2000, 8000]:
            if time_left() < 50:

            penalty = alpha * 20.0

            if res_x is not None:
                h_fine = project_box_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)

        n_final = len(best_h)
        for alpha in [10000, 30000]:
