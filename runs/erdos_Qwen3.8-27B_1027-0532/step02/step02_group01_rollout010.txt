def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def make_binary(h, rng):
    n = len(h)
    target = n // 2
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[target:]] = 1.0
    return h_bin

def thue_morse(n):
    """Generate first n bits of Thue-Morse sequence (as 0/1)."""
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = tm[i >> 1] + (i & 1)
        tm[i] = tm[i] % 2
    return tm

def local_search(h, n_evals, rng, time_limit, start_time):

    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def make_binary(h, rng):
    n = len(h)
    target = n // 2
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[target:]] = 1.0
    return h_bin

def thue_morse(n):
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = (tm[i >> 1] + (i & 1)) % 2.0
    return tm

def batch_greedy(h, batch_size, max_iters, rng, deadline):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        best_delta = 0.0
        best_a = -1
        best_b = -1
        for _ in range(batch_size):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            h[a], h[b] = h[b], h[a]
            nc = compute_c5(h)
            d = nc - c5
            if d < best_delta:
                best_delta = d
                best_a = a
                best_b = b
            h[a], h[b] = h[b], h[a]
        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            ia = np.where(ones == best_a)[0][0]
            ib = np.where(zeros == best_b)[0][0]
            ones[ia] = best_b
            zeros[ib] = best_a
            ones.sort()

            stall = 0

    best_h, best_c5 = batch_greedy(hb, 150, 10**9, rng, d1)

    d2 = start + budget_s * 0.50
    best_h, best_c5 = batch_greedy(best_h, 300, 10**9, rng, d2)

    n_chains = 6
    t_chain_start = time.time()
    t_chain_end = start + budget_s * 0.75
    for c in range(n_chains):
        if time.time() > t_chain_end:
            break
        if c == 0:
            h_c = best_h.copy()
            t0 = 0.015
        elif c == 1:
            h_c = perturb_swap(best_h, int(rng.integers(5, n // 5)), rng)
            t0 = 0.02
        elif c == 2:
            h_c = perturb_block(best_h, rng)
            t0 = 0.025
        elif c == 3:
            h_c = thue_morse(n)
            h_c = make_binary(h_c + 0.3 * best_h, rng)
            t0 = 0.03
        elif c == 4:
            h_c = perturb_swap(best_h, int(rng.integers(10, n // 3)), rng)
            t0 = 0.02
        else:
            h_c = np.zeros(n)
            h_c[rng.choice(n, size=n // 2, replace=False)] = 1.0
            t0 = 0.04
        t_end_chain = min(t_chain_end, time.time() + (t_chain_end - t_chain_start) / (n_chains - c))
        h_c, c5_c = sa_chain(h_c, 10**8, t0, 0.0003, rng, t_end_chain)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    d3 = start + budget_s * 0.88
    best_h, best_c5 = batch_greedy(best_h, 500, 10**9, rng, d3)

    n_il
    = 3
    t_ils_start = time.time()
    t_ils_end = start + budget_s * 0.95
    for r in range(n_ils):
        if time.time() > t_ils_end:
            break
        t_ils_chunk = (t_ils_end - time.time()) / (n_ils - r)
        h_p = perturb_swap(best_h, int(rng.integers(3, max(4, n // 10))), rng)
        h_p, c5_p = batch_greedy(h_p, 200, 10**9, rng, time.time() + t_ils_chunk * 0.6)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
        h_p2, c5_p2 = sa_chain(h_p, 10**7, 0.003, 0.0001, rng, time.time() + t_ils_chunk * 0.4)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    d_final = deadline
    best_h, best_c5 = batch_greedy(best_h, 400, 10**9, rng, d_final)

    return best_h, best_c5, n
