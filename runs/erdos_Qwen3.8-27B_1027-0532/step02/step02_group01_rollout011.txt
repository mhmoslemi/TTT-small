```python
    """Evaluate the effect of swapping h[i]=1→0 and h[j]=0→1 on the max correlation.

    n = len(h)

        return np.max(corr)
```

```python
def evaluate_swap_max(h, i, j, corr):
    """Evaluate new max correlation after swapping h[i]=1→0, h[j]=0→1"""
    n = len(h)
    # s ranges from -(n-1) to n-1, corr index = s + n - 1
    # Valid s: 0 <= i+s < n and 0 <= j+s < n
    s_lo = max(0 - i, 0 - j, -(n-1))
    s_hi = min(n-1-i, n-1-j, n-1)
    
    if s_lo > s_hi:
        return np.max(corr)
    
    ci_lo = s_lo + n - 1
    ci_hi = s_hi + n - 1
    
    # Affected portion
    delta = h[i+s_lo : i+s_hi+1] - h[j+s_lo : j+s_hi+1]
    affected = corr[ci_lo : ci_hi+1] + delta
    new_max_affected = np.max(affected)
    
    # Unaffected portion
    if ci_lo > 0:
        left_max = np.max(corr[:ci_lo])
    else:
        left_max = -1e10
    if ci_hi < len(corr) - 1:
        right_max = np.max(corr[ci_hi+1:])
    else:
        right_max = -1e10
    
    return max(new_max_affected, left_max, right_max)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_binary(h, n_ones, rng):
    """Project h to binary with exactly n_ones ones"""
    h = np.round(h).astype(float)
    cur = int(h.sum())
    if cur == n_ones:
        return h
    if cur > n_ones:
        idx = np.where(h == 1.0)[0]
        flip = rng.choice(idx, size=cur - n_ones, replace=False)
        h[flip] = 0.0
    else:
        idx = np.where(h == 0.0)[0]
        flip = rng.choice(idx, size=n_ones - cur, replace=False)
        h[flip] = 1.0
    return h

def swap_eval(h, i, j, corr):
    """New max corr after swapping h[i]=1->0, h[j]=0->1"""
    n = len(h)
    s_lo = max(-i, -j, -(n-1))
    s_hi = min(n-1-i, n-1-j, n-1)
    if s_lo > s_hi:
        return np.max(corr)
    ci_lo = s_lo + n - 1
    ci_hi = s_hi + n - 1
    delta = h[i+s_lo:i+s_hi+1] - h[j+s_lo:j+s_hi+1]
    new_aff = corr[ci_lo:ci_hi+1] + delta
    m = np.max(new_aff)
    if ci_lo > 0:
        m = max(m, np.max(corr[:ci_lo]))
    if ci_hi < len(corr)-1:
        m = max(m, np.max(corr[ci_hi+1:]))
    return m

def local_search(h, rng, max_iter=3000, time_limit=60, start_time=0):
    """Subgradient-guided binary local search"""
    n = len(h)
    h = h.copy()
    corr = get_corr(h)
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 500:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Find critical shifts
        threshold = cur_max - 3  # shifts within 3 of max
        crit_idx = np.where(corr >= threshold)[0]
        crit_shifts = crit_idx - (n - 1)
        
        # Compute cost for each position
        cost_ones = np.zeros(n)
        cost_zeros = np.zeros(n)
        for s in crit_shifts:
            for i in ones:
                idx = i + s
                if 0 <= idx < n and h[idx] == 0.0:
                    cost_ones[i] += 1
            for j in zeros:
                idx = j + s
                if 0 <= idx < n and h[idx] == 1.0:
                    cost_zeros[j] += 1
        
        # Generate candidates: high-cost ones with high-cost zeros
        top_ones = ones[np.argsort(cost_ones[ones])[::-1][:30]]
        top_zeros = zeros[np.argsort(cost_zeros[zeros])[::-1][:30]]
        
        best_impr = 0
        best_sw = None
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                nm = swap_eval(h, i, j, corr)
                impr = cur_max - nm
                if impr > best_impr:
                    best_impr = impr
                    best_sw = (i, j)
        
        # Also try some random swaps
        for _ in range(50):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            nm = swap_eval(h, i, j, corr)
            impr = cur_max - nm
            if impr > best_impr:
                best_impr = impr
                best_sw = (i, j)
        
        if best_sw is None:
            no_improve += 1
            continue
        
        i, j = best_sw
        h[i] = 0.0
        h[j] = 1.0
        corr = get_corr(h)
        cur_max = np.max(corr)
        no_improve = 0
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max * 2.0 / n

def simulated_annealing(h, rng, n_steps=50000, t_start=1.0, t_end=0.01, time_limit=60, start_time=0):
    """Binary simulated annealing"""
    n = len(h)
    h = h.copy()
    corr = get_corr(h)
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = get_corr(h)
        new_max = np.max(new_corr)
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            cur_max = new_max
            corr = new_corr
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_max * 2.0 / n

def multi_swap_perturb(h, rng, n_swaps=2):
    """Perturb by swapping n_swaps pairs"""
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
    return h

def block_swap(h, rng):
    """Swap two blocks of random length"""
    n = len(h)
    blen = rng.integers(2, max(3, n // 10))
    a = rng.integers(0, n - blen)
    b = rng.integers(0, n - blen)
    if abs(a - b) < blen:
        return h
    h = h.copy()
    h[a:a+blen], h[b:b+blen] = h[b:b+blen].copy(), h[a:a+blen]
    return h
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_binary(h, n_ones, rng):
    h = np.round(h).astype(float).flatten()
    cur = int(h.sum())
    if cur == n_ones:
        return h
    if cur > n_ones:
        idx = np.where(h == 1.0)[0]
        flip = rng.choice(idx, size=cur - n_ones, replace=False)
        h[flip] = 0.0
    else:
        idx = np.where(h == 0.0)[0]
        flip = rng.choice(idx, size=n_ones - cur, replace=False)
        h[flip] = 1.0
    return h

def swap_new_max(h, i, j, corr):
    n = len(h)
    s_lo = max(-i, -j, -(n - 1))
    s_hi = min(n - 1 - i, n - 1 - j, n - 1)
    if s_lo > s_hi:
        return np.max(corr)
    ci_lo = s_lo + n - 1
    ci_hi = s_hi + n - 1
    delta = h[i + s_lo:i + s_hi + 1] - h[j + s_lo:j + s_hi + 1]
    new_aff = corr[ci_lo:ci_hi + 1] + delta
    m = float(np.max(new_aff))
    if ci_lo > 0:
        m = max(m, float(np.max(corr[:ci_lo])))
    if ci_hi < len(corr) - 1:
        m = max(m, float(np.max(corr[ci_hi + 1:])))
    return m

def local_search(h, rng, max_iter=5000, time_limit=60.0, start_time=0.0):

    best_h = h.copy()
    best_max = cur_max
    stuck = 0

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if stuck > 800:

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        crit_shifts = crit_idx - (n - 1)

def simulated_annealing(h, rng, n_steps=40000, t_start=0.8, t_end=0.01, time_limit=60.0, start_time=0.0):

def multi_swap_perturb(h, rng, n_swaps):
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
    return h

def block_swap_perturb(h, rng):
    n = len(h)
    blen = int(rng.integers(2, max(3, n // 8)))
    a = int(rng.integers(0, n - blen))
    b = int(rng.integers(0, n - blen))
    if abs(a - b) < blen:
        return h.copy()
    h = h.copy()
    tmp = h[a:a + blen].copy()
    h[a:a + blen] = h[b:b + blen]
    h[b:b + blen] = tmp
    return h

            best_h = h_ls.copy()

        elapsed = time.time() - start_time
        remaining = budget_s - 60 - elapsed
        if remaining < 20:
            break

        tl2 = min(tl_this * 0.35, remaining)
        h_sa, c5_sa = simulated_annealing(h_ls, rng, n_steps=40000,
                                          t_start=0.5, t_end=0.01,
                                          time_limit=tl2, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        elapsed = time.time() - start_time
        remaining = budget_s - 60 - elapsed
        if remaining < 20:
            break

        tl3 = min(tl_this * 0.25, remaining)
        h_ls2, c5_ls2 = local_search(best_h, rng, max_iter=4000, time_limit=tl3, start_time=start_time)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Final phase: intensive optimization on best solution
    elapsed = time.time() - start_time
    remaining = budget_s - 30 - elapsed
    if remaining > 20 and best_h is not None:
        n = len(best_h)
        n_restarts = min(8, int(remaining // 30))
        for r in range(n_restarts):
            elapsed = time.time() - start_time
            if time.time() - start_time > budget_s - 20:
                break
            tl = min(50, budget_s - 20 - (time.time() - start_time))
            if tl < 10:
                break
            if r % 3 == 0:
                h_pert = multi_swap_perturb(best_h, rng, 3)
            elif r % 3 == 1:
                h_pert = block_swap_perturb(best_h, rng)
            else:
                h_pert = multi_swap_perturb(best_h, rng, 5)

            h_sa, c5_sa = simulated_annealing(h_pert, rng, n_steps=25000,
                                              t_start=0.4, t_end=0.02,
                                              time_limit=tl * 0.6, start_time=start_time)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

            elapsed = time.time() - start_time
            tl_ls = min(30, budget_s - 20 - elapsed)
            if tl_ls > 5:
                h_ls, c5_ls = local_search(best_h, rng, max_iter=3000,
                                           time_limit=tl_ls, start_time=start_time)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Final local search polish
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10 and best_h is not None:
        tl_final = min(30, budget_s - 10 - elapsed)
        if tl_final > 3:
            h_final, c5_final = local_search(best_h, rng, max_iter=2000,
                                             time_limit=tl_final, start_time=start_time)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()

    if best_h is None:
        n = 180
        best_h = (rng.random(n) < 0.5).astype(float)
        best_h = project_binary(best_h, n // 2, rng)
        best_c5 = compute_c5(best_h)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
