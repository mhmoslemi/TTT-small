from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_and_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    obj = (max_corr + np.log(np.sum(exp_shifted)) / alpha) * dx
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += w * one_minus_h[lag:]
            if lag < n:
                grad[lag:] -= w * h[:n - lag]
        else:
            p = -lag
            if n - p > 0:
                grad[p:] += w * one_minus_h[:n - p]
            if p < n:
                grad[:n - p] -= w * h[p:]
    grad *= dx
    
    # Add penalty gradient
    sum_diff = np.sum(h) - n / 2.0
    obj += penalty * sum_diff ** 2
    grad += 2.0 * penalty * sum_diff * np.ones(n)
    
    return obj, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run_lbfgs(h_start, n, target, alpha, penalty, maxiter, ftol):
    def obj_grad(x):
        return smooth_obj_and_grad(x, alpha, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_and_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    obj = (max_corr + np.log(sum_exp) / alpha) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += w * one_minus_h[lag:]
            if lag < n:
                grad[lag:] -= w * h[:n - lag]
        else:
            p = -lag
            if n - p > 0:
                grad[p:] += w * one_minus_h[:n - p]
            if p < n:
                grad[:n - p] -= w * h[p:]
    grad *= dx
    
    sum_diff = np.sum(h) - n * 0.5
    obj += penalty * sum_diff * sum_diff
    grad += 2.0 * penalty * sum_diff
    
    return obj, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
