With 500K evaluations, that's 65 billion operations. At ~100M ops/sec for numpy, that's about 650 seconds. That might be too slow.

Let me optimize the SA. Instead of doing the full correlation each time, I can compute the change in the objective more efficiently.

When we swap positions o (1→0) and z (0→1), the change in corr[k] for each lag k is:
- The term h[i](1-h[i+k]) changes for:
  - i=o: was 1*(1-h[o+k]) = 1-h[o+k], now 0. Change: -(1-h[o+k])
  - i=o-k (if valid): was h[o-k]*(1-1) = 0, now h[o-k]*(1-0) = h[o-k]. Change: +h[o-k]
  - i=z: was 0*(1-h[z+k]) = 0, now 1*(1-h[z+k]) = 1-h[z+k]. Change: +(1-h[z+k])
  - i=z-k (if valid): was h[z-k]*(1-0) = h[z-k], now h[z-k]*(1-1) = 0. Change: -h[z-k]

Wait, I need to be more careful. The correlation at lag s (where s = k - (n-1) in the full output) is:
corr_full[n-1+s] = sum_i h[i] * (1-h[i+s]) for i where 0 ≤ i < n and 0 ≤ i+s < n

When h[o] changes from 1 to 0:
- Term i=o in sum for lag s: was 1*(1-h[o+s]), now 0. Change: -(1-h[o+s]) if 0≤o+s<n
- Term i=o-s in sum for lag s: was h[o-s]*(1-h[o]) = h[o-s]*0 = 0, now h[o-s]*1 = h[o-s]. Change: +h[o-s] if 0≤o-s<n

When h[z] changes from 0 to 1:
- Term i=z in sum for lag s: was 0*(1-h[z+s]) = 0, now 1*(1-h[z+s]). Change: +(1-h[z+s]) if 0≤z+s<n
- Term i=z-s in sum for lag s: was h[z-s]*(1-h[z]) = h[z-s]*1 = h[z-s], now h[z-s]*0 = 0. Change: -h[z-s] if 0≤z-s<n

So total change at lag s:
delta[s] = -(1-h[o+s]) + h[o-s] + (1-h[z+s]) - h[z-s]
         = h[o+s] - 1 + h[o-s] + 1 - h[z+s] - h[z-s]
         = h[o+s] + h[o-s] - h[z+s] - h[z-s]

(with appropriate boundary handling: if index out of range, term is 0)

This means I can compute the change in C5 in O(n) instead of O(n^2)! I just need to compute the new max of (corr + delta).

But wait, I still need to maintain the full corr array to know the new max. Let me maintain corr incrementally:
- corr_new[s] = corr_old[s] + delta[s]
- new_C5 = max(corr_new) * dx

This is O(n) per evaluation! Much faster.

def fast_binary_sa(h_start, n_moves, t_start, t_end, rng, start_time, time_limit):
    n = len(h_start)
    dx = 2.0 / n
    n_ones = int(round(h_start.sum()))
    h = np.zeros(n)
    order = np.argsort(-h_start)
    h[order[:n_ones]] = 1.0
    
    # Precompute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    current_c5 = float(dx * np.max(corr))
    best_h = h.copy()
    best_c5 = current_c5
    best_corr = corr.copy()
    
    for i in range(n_moves):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (i / max(n_moves - 1, 1))
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        o = ones[rng.integers(len(ones))]
        z = zeros[rng.integers(len(zeros))]
        
        # Compute delta for each lag
        # delta[s] = h[o+s] + h[o-s] - h[z+s] - h[z-s] (with bounds)
        # In the full correlation array (length 2n-1), lag s corresponds to index n-1+s
        # s ranges from -(n-1) to (n-1)
        
        # Instead of computing all 2n-1 deltas, just update corr incrementally
        corr_new = corr.copy()  # O(n)
        
        # For position o (1 -> 0):
        # lag s: -(1-h[o+s]) if 0<=o+s<n, +h[o-s] if 0<=o-s<n
        # For position z (0 -> 1):
        # lag s: +(1-h[z+s]) if 0<=z+s<n, -h[z-s] if 0<=z-s<n
        
        # Let me compute this vectorized
        # For o (1->0):
        #   affects corr at indices where i=o (forward) or i+s=o (backward)
        #   forward: corr[n-1+s] -= (1-h[o+s]) for s where 0<=o+s<n
        #     s ranges from -(o) to n-1-o, i.e., index from n-1 to ... 
        #     Actually, for index j in [0, 2n-2], s = j-(n-1)
        #     i=o is valid when 0<=o<n and 0<=o+s<n => -o <= s < n-o
        #     So j from n-1-o to n-1+n-1-o = 2n-2-o
        #     corr[j] -= (1-h[o+s]) where s=j-(n-1), so h[o+j-(n-1)]
        
        # This is getting complex. Let me just do it simply:
        # The change at lag s (correlation index n-1+s) is:
        #   from o (1->0): -(1-h[o+s]) * [0<=o+s<n] + h[o-s] * [0<=o-s<n]
        #   from z (0->1): +(1-h[z+s]) * [0<=z+s<n] - h[z-s] * [0<=z-s<n]
        
        # Vectorized computation:
        s = np.arange(-(n-1), n)  # all lags, length 2n-1
        # o+s valid where 0 <= o+s < n => -o <= s < n-o
        mask_os = (s >= -o) & (s < n - o)
        mask_oms = (s >= o - (n-1)) & (s <= o)  # o-s valid where 0 <= o-s < n => o-n+1 <= s <= o
        # Wait: 0 <= o-s < n => o-n < s <= o, i.e., o-n+1 <= s <= o (integer)
        # Hmm, let me be more careful: 0 <= o-s < n means o-n < s <= o
        # Since s is integer: o-n+1 <= s <= o
        
        mask_zs = (s >= -z) & (s < n - z)
        mask_zms = (s >= z - (n-1)) & (s <= z)
        
        # h[o+s] where valid
        delta = np.zeros(2*n-1)
        os_idx = o + s[mask_os]
        delta[mask_os] -= (1 - h[os_idx])
        
        oms_idx = o - s[mask_oms]
        delta[mask_oms] += h[oms_idx]
        
        zs_idx = z + s[mask_zs]
        delta[mask_zs] += (1 - h[zs_idx])
        
        zms_idx = z - s[mask_zms]
        delta[mask_zms] -= h[zms_idx]
        
        corr_test = corr + delta
        new_c5 = float(dx * np.max(corr_test))
        
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(t, 1e-12)):
            # Accept
            h[o] = 0.0
            h[z] = 1.0
            corr = corr_test
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_corr = corr.copy()
        else:
            # Reject (don't update)
            pass
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
            if time.time() - start_time < budget_s - 40:
                h_cont3 = best_h + 0.005 * rng.standard_normal(len(best_h))
                h_cont3 = project_onto_constraint(h_cont3, len(best_h) / 2.0)
                h_opt3, c5_opt3 = slsqp_optimize(h_cont3, 0.0003, 300)
                if c5_opt3 < best_c5:
                    best_c5 = c5_opt3
                    best_h = h_opt3
    remaining = budget_s - (time.time() - start_time) - 20
    if remaining > 30:
        h_final = best_h + 0.003 * rng.standard_normal(len(best_h))
        h_final = project_onto_constraint(h_final, len(best_h) / 2.0)
        h_opt, c5_opt = slsqp_optimize(h_final, 0.0001, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    remaining = budget_s - (time.time() - start_time) - 10
    if remaining > 15:
        h_polish = best_h.copy()
        h_opt, c5_opt = slsqp_optimize(h_polish, 0.00005, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
