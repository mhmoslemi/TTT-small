If h is the indicator of a set A, then h(x)(1-h(x+k)) = 1 if x ∈ A and x+k ∉ A. So the integral is |A ∩ (A-c_k)| where c_k is the complement of A shifted by k... no, it's the measure of x such that x ∈ A and x+k ∉ A = |A \ (A-k)|.

This is related to the "difference set" of A. We want a set A of measure 1 in [0,2] such that A \ (A-k) is as small as possible for all k.

For a "perfect" set, A \ (A-k) would be 0 for all k, meaning A = A-k for all k, which is impossible (it would mean A is periodic with period k for all k).

The best we can do is related to the additive combinatorics of the set A.

For binary sequences, this is related to "difference sets" in finite abelian groups. A (v, k, λ) difference set is a subset D of a group G of order v such that every non-identity element of G is represented exactly λ times as a difference of two elements of D.

If we could find a perfect difference set, the autocorrelation would be constant, which is ideal. But perfect difference sets only exist for certain parameters.

For our problem, we're looking at a "one-sided" correlation (h against 1-h, not h against h), so it's slightly different.

    """L-BFGS-B continuous optimization."""
    def obj(h):

        m = np.max(corr)
        # Smooth max
        shifted = alpha * (corr - m)
        sm = m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s
    
    def grad(h):
        corr = compute_corr(h)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_shifted = np.max(shifted)
        exp_shifted = np.exp(shifted - max_shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        
        g = np.zeros(n)
        for idx in range(len(weights)):
            w = weights[idx]
            if w < 1e-10:
                continue

                g[l:] -= w * h[:n-l]
                g[:n-l] -= w * h[l:]
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    lo = -float(np.max(h)) - 0.5
    hi = 1.0 - float(np.min(h)) + 0.5

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:

    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n, rng=None):

    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx[:target]] = 1.0
    return h_bin

def local_search(h, n, rng, time_limit, max_passes=30, sample_size=200):
    best_h = h.copy()
    best_max = float(np.max(compute_corr(best_h)))
    best_c5 = best_max * 2.0 / n
    current = best_h.copy()
    current_max = best_max
    t0 = time.time()
    
    for _ in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for i in ones:
            if time.time() - t0 > time_limit:
                break
            n_zeros = len(zeros)
            if n_zeros > sample_size:
                js = zeros[rng.choice(n_zeros, sample_size, replace=False)]
            else:
                js = zeros
            best_j = -1
            best_new_max = current_max
            for j in js:
                current[i] = 0.0
                current[j] = 1.0
                nm = float(np.max(compute_corr(current)))
                if nm < best_new_max:
                    best_new_max = nm
                    best_j = j
                current[i] = 1.0
                current[j] = 0.0
            if best_j >= 0:
                current[i] = 0.0
                current[best_j] = 1.0
                current_max = best_new_max
                if current_max < best_max:
                    best_max = current_max
                    best_c5 = best_max * 2.0 / n
                    best_h = current.copy()
                improved = True
                ones = np.where(current == 1)[0]
                zeros = np.where(current == 0)[0]
        if not improved:
            break
    return best_h, best_c5

def sim_anneal(h, n, rng, time_limit):
    best_h = h.copy()
    best_max = float(np.max(compute_corr(best_h)))
    best_c5 = best_max * 2.0 / n
    current = best_h.copy()
    current_max = best_max
    T = 3.0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        T = max(T * 0.99999, 0.05)
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        current[i] = 0.0
        current[j] = 1.0
        nm = float(np.max(compute_corr(current)))
        delta = (nm - current_max) * 2.0 / n
        if delta <= 0 or rng.rand() < np.exp(-delta / (T * 0.002)):
            current_max = nm
            if nm < best_max:
                best_max = nm
                best_c5 = nm * 2.0 / n
                best_h = current.copy()
        else:
            current[i] = 1.0
            current[j] = 0.0
    return best_h, best_c5

def cont_refine(h0, n, alpha, maxiter):
    mu = 50000.0
    def obj(h):
        corr = compute_corr(h)
        m = float(np.max(corr))
        shifted = alpha * (corr - m)
        ms = float(np.max(shifted))
        sm = m + float(np.log(np.sum(np.exp(shifted - ms)))) / alpha
        s = float(np.sum(h)) - n / 2.0
        return sm + mu * s * s
    def grad(h):
        corr = compute_corr(h)
        m = float(np.max(corr))
        shifted = alpha * (corr - m)
        ms = float(np.max(shifted))
        es = np.exp(shifted - ms)
        w = es / float(np.sum(es))
        g = np.zeros(n)
        for idx in range(len(w)):
            wi = w[idx]
            if wi < 1e-10:
                continue
            lag = (n - 1) - idx
            if lag == 0:
                g += wi * (1.0 - 2.0 * h)
            elif 0 < lag < n:
                g[:n-lag] += wi
                g[:n-lag] -= wi * h[lag:]
                g[lag:] -= wi * h[:n-lag]
            elif 0 < -lag < n:
                ll = -lag
                g[ll:] += wi
                g[ll:] -= wi * h[:n-ll]
                g[:n-ll] -= wi * h[ll:]
        s = float(np.sum(h)) - n / 2.0
        g += 2.0 * mu * s
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return res.x

- h[j] changes from 0 to 1: affects terms where j appears
