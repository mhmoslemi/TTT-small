def _c5_of(h):

def _project_to_simplex_box(y, S):
    """Project y onto {h : 0 <= h_i <= 1, sum(h) = S}"""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding issues
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(x - 0.5)))
        x[idx] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_topm(h, m, alpha):
    """Compute weighted gradient using top-m lags with soft weighting."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft-max weights
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    total = weights.sum()
    if total < 1e-300:
        k_star = np.argmax(corr)
        return _grad_single(h, k_star, dx)
    weights = weights / total
    # Use top m lags
    top_m = np.argsort(weights)[::-1][:m]
    w_m = weights[top_m]
    w_m = w_m / w_m.sum()
    g = np.zeros(n)
    for idx in range(len(top_m)):
        g += w_m[idx] * _grad_single(h, top_m[idx], dx)
    return g

def _grad_single(h, k_idx, dx):
    """Gradient of the k-th correlation term w.r.t. h."""
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _bb_step(h_prev, g_prev, h_curr, g_curr):
    """Barzibar-Borwein step size."""
    s = h_curr - h_prev
    y = g_curr - g_prev
    ys = s @ y
    yy = y @ y
    if yy < 1e-16 or ys <= 1e-16:
        return 0.1
    alpha = ys / yy
    return np.clip(alpha, 1e-6, 10.0)

        val = _c5_of(h)
        if val < best_val:

            best = h.copy()

This is the number of elements of A that, when shifted by k, land outside A. Equivalently, it's |A| - |A ∩ (A-k)| = n/2 - |A ∩ (A-k)|.

So c_k = dx * (n/2 - |A ∩ (A-k)|) for k ≥ 0 (with appropriate boundary handling).

To minimize max_k c_k, we want to maximize min_k |A ∩ (A-k)|.
