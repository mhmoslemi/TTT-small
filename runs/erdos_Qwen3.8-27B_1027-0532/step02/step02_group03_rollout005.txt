    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

def _simulated_annealing(h0, n, target_sum, max_iters, T_start, T_end, deadline):

        if time.time() > deadline:
            break
        
        # Cool
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Perturbation
        h_new = h.copy()
        n_pert = rng.randint(1, max(2, n // 10))
        indices = rng.choice(n, n_pert, replace=False)
        for idx in indices:
            if rng.random() < 0.5:
                h_new[idx] = 0.0
            elif rng.random() < 0.5:
                h_new[idx] = 1.0
            else:
                h_new[idx] = rng.random()
        h_new = _fix_sum(h_new, target_sum)

        # Accept

            h = h_new
            h_val = new_val
            if h_val < best_val:
                best_val = h_val
                best_h = h.copy()
    
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent(h0, n, target_sum, max_rounds=20):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=8):
    h = h0.copy()
    vals = [0.0, 0.25, 0.5, 0.75, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(50.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    """Differential evolution for global search."""
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        if i % 5 == 1:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif i % 5 == 2:
            pop[i] = _fix_sum(1.0 - h0 + 0.1 * (rng.rand(n) - 0.5), target_sum)
        elif i % 5 == 3:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1 - t) * rng.rand(n), target_sum)
        elif i % 5 == 4:
            # Permutation-based
            shuffled = h0[rng.permutation(n)]
            pop[i] = _fix_sum(shuffled, target_sum)
        else:
            # Block swap
            pop[i] = h0.copy()
            block_size = n // 4
            start = rng.randint(0, n - 2 * block_size)
            mid = start + block_size
            pop[i][start:mid] = h0[mid:mid+block_size].copy()
            pop[i][mid:mid+block_size] = h0[start:mid].copy()
            pop[i] = _fix_sum(pop[i], target_sum)
    
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    
    stall_count = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        
        F = F_start + (F_end - F_start) * (gen / max_gen)
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            
            # Crossover
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross_mask = (rng.random(n) < CR)
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            trial_val = _c5_of(trial)
            
            if trial_val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    best_idx = i
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1
        
        if stall_count > pop_size:
            worst_half = np.argsort(pop_vals)[pop_size//2:]
            for idx in worst_half:
                t = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(t * best_h + (1-t) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall_count = 0
    
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent(h0, n, target_sum, max_rounds=20):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=8):
    h = h0.copy()
    vals = [0.0, 0.25, 0.5, 0.75, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5, sg_alpha=50.0):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(sg_alpha * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        mod = i % 6
        if mod == 1:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif mod == 2:
            pop[i] = _fix_sum(1.0 - h0 + 0.05 * (rng.rand(n) - 0.5), target_sum)
        elif mod == 3:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif mod == 4:
            shuffled = h0[rng.permutation(n)]
            pop[i] = _fix_sum(shuffled, target_sum)
        elif mod == 5:
            pop[i] = h0.copy()
            bs = max(2, n // 6)
            s1 = rng.randint(0, n - 2 * bs)
            s2 = rng.randint(0, n - 2 * bs)
            pop[i][s1:s1+bs] = h0[s2:s2+bs].copy()
            pop[i][s2:s2+bs] = h0[s1:s1+bs].copy()
            pop[i] = _fix_sum(pop[i], target_sum)
        else:
            pop[i] = _fix_sum(h0 + 0.3 * (rng.rand(n) - 0.5), target_sum)

    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    stall_count = 0

    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = F_start + (F_end - F_start) * (gen / max_gen)
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross_mask = (rng.random(n) < CR)
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            trial_val = _c5_of(trial)
            if trial_val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    best_idx = i
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1
        if stall_count > pop_size:
            worst_half = np.argsort(pop_vals)[pop_size // 2:]
            for idx in worst_half:
                t_mix = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(t_mix * best_h + (1.0 - t_mix) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall_count = 0
    return best_h, best_val

def _sa_optimize(h0, n, target_sum, max_iters, T_start, T_end, deadline, rng):
    h = h0.copy()
    h_val = _c5_of(h)
    best_h = h.copy()
    best_val = h_val
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max_iters))
        h_new = h.copy()
        n_pert = rng.randint(1, max(2, n // 8))
        indices = rng.choice(n, n_pert, replace=False)
        for idx in indices:
            r = rng.random()
            if r < 0.3:
                h_new[idx] = 0.0
            elif r < 0.6:
                h_new[idx] = 1.0
            else:
                h_new[idx] = rng.random()
        h_new = _fix_sum(h_new, target_sum)
        new_val = _c5_of(h_new)
        acceptance = np.exp(-(new_val - h_val) / max(T, 1e-10))
        if new_val < h_val or rng.random() < acceptance:
            h = h_new
            h_val = new_val
            if h_val < best_val:
                best_val = h_val
                best_h = h.copy()
    return best_h, best_val

def _refine(h0, n, target_sum, deadline):
    h_best = h0.copy()
    c5_best = _c5_of(h0)
    if time.time() > deadline - 40:
        return h_best, c5_best
    h_n1, c5_n1 = _nesterov(h_best, n, target_sum, 150, 0.3, use_smooth_sg=True, sg_m=7, sg_alpha=80.0)
    if c5_n1 < c5_best:
        c5_best = c5_n1
        h_best = h_n1
    if time.time() > deadline - 30:
        return h_best, c5_best
    h_sl = _slsqp_opt(h_best, n, target_sum, 200, 250)
    c5_sl = _c5_of(h_sl)
    if c5_sl < c5_best:

    h_sl2 = _slsqp_opt(h_best, n, target_sum, 800, 200)

    if c5_sl2 < c5_best:
        c5_best = c5_sl2
        h_best = h_sl2
    if time.time() > deadline - 15:

    h_cd = _coord_descent(h_best, n, target_sum, 20)
    c5_cd = _c5_of(h_cd)
    if c5_cd < c5_best:
