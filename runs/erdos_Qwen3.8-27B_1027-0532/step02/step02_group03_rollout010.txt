    corr = np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)

            if diff <= 1e-12:

Let me optimize the projection. Instead of 80 bisection steps, I can use a smarter approach: sort the values and use the fact that the optimal threshold splits the sorted array.

Actually, the standard approach for projecting onto {x: 0<=x_i<=1, sum(x)=S} is:
1. Sort y in decreasing order: y_(1) >= y_(2) >= ... >= y_(n)
2. Find the largest k such that y_(k) - (sum_{i<=k} y_(i) - S) / k > 0
3. The threshold is (sum_{i<=k} y_(i) - S) / k
4. x_i = clip(y_i - threshold, 0, 1)

This is O(n log n) instead of O(n × bisection_steps).

```python
def _project_box_sum_fast(y, S):
    n = len(y)
    # Sort in decreasing order
    y_sorted = np.sort(y)[::-1]
    # Find the breakpoint
    cumsum = np.cumsum(y_sorted)
    # We need the largest k such that y_sorted[k-1] > (cumsum[k-1] - S) / k
    thresholds = (cumsum - S) / np.arange(1, n+1)
    valid = y_sorted > thresholds
    # Find the last valid index
    k = np.sum(valid)
    if k == 0:
        k = 1
    if k == n:
        # All should be 1, but sum might be too high
        threshold = (cumsum[n-1] - S) / n
        x = np.clip(y - threshold, 0, 1)
    else:
        threshold = (cumsum[k-1] - S) / k
        x = np.clip(y - threshold, 0, 1)
    # Fix any numerical issues
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

_g_target = 0.0
_g_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_fast(y, S):
    n = len(y)
    y_sorted = np.sort(y)[::-1]
    cumsum = np.cumsum(y_sorted)
    thresholds = (cumsum - S) / np.arange(1, n + 1)
    valid = y_sorted > thresholds
    k = int(np.sum(valid))
    if k < 1:
        k = 1
    if k >= n:
        threshold = (cumsum[-1] - S) / n
    else:
        threshold = (cumsum[k - 1] - S) / k
    x = np.clip(y - threshold, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _eq_fun(x):
    return x.sum() - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def _lse_obj(x, alpha):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * dx
    return (m + np.log(np.sum(np.exp(alpha * (corr - m))))) / alpha * dx

def _slsqp(h0, n, target, alpha, maxiter):
    global _g_target, _g_n
    _g_target = target
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_lse_obj, h0, args=(alpha,),
                       bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _nesterov_soft(h0, n, target, n_iters, alpha0, deadline, n_top, temp):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        max_c = corr[top_idx[0]]
        w = np.exp(temp * (corr[top_idx] - max_c))
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(n_top):
            if w[ii] < 1e-10:
                continue
            k_idx = top_idx[ii]
            lag = k_idx - (n - 1)
            if lag >= 0:
                m = n - lag
                if m > 0:
                    g[lag:lag + m] += w[ii] * (1.0 - y[:m])
                    g[:m] += w[ii] * (-y[lag:lag + m])
            else:
                j = -lag
                m = n - j
                if m > 0:
                    g[:m] += w[ii] * (1.0 - y[j:])
                    g[j:] += w[ii] * (-y[:m])
        g *= dx
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_fast(y - step * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y_new = _project_fast(y_new, target)
        x = x_new
        y = y_new
        t_k = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_opt(h0, n, target, pop_size, n_gens, f_init, f_end, cr, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = _fix_sum(h0.copy(), target)
    for i in range(1, pop_size):

            pop[i] = _fix_sum(rng.rand(n), target)
        else:

            mut = base + 0.5 * rng.randn(n)
            pop[i] = _fix_sum(mut, target)
    fit = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_i = int(np.argmin(fit))
    best_fit = fit[best_i]
    stall_gen = 0
    for gen in range(n_gens):
        if time.time() > deadline:
            break
        f = f_init + (f_end - f_init) * (gen / max(1, n_gens - 1))
        gen_improved = False
        for i in range(pop_size):
            if time.time() > deadline:
                break
            cands = [j for j in range(pop_size) if j != i and j != best_i]
            if len(cands) < 2:
                cands = [j for j in range(pop_size) if j != i]
            if len(cands) < 2:
                continue
            a = cands[rng.randint(len(cands))]
            b = cands[rng.randint(len(cands))]
            mutant = pop[best_i] + f * (pop[a] - pop[b])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.random() < cr or j == j_rand:
                    trial[j] = mutant[j]
            trial = _project_fast(trial, target)
            f_trial = _c5_of(trial)
            if f_trial <= fit[i]:
                pop[i] = trial
                fit[i] = f_trial
                if f_trial < best_fit:
                    best_fit = f_trial
                    best_i = i
                    gen_improved = True
        if gen_improved:
            stall_gen = 0
        else:
            stall_gen += 1
        if stall_gen > 200:
            for i in range(pop_size):
                if i != best_i:
                    mut = pop[best_i] + 0.6 * rng.randn(n)
                    pop[i] = _fix_sum(mut, target)
                    fit[i] = _c5_of(pop[i])
            stall_gen = 0
    return pop[best_i].copy(), best_fit

def _coord_bin(h0, n, target, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            c_best = _c5_of(h)
            v_best = old
            for v in (0.0, 1.0):
                h[j] = v
                c = _c5_of(h)
                if c < c_best - 1e-14:
                    c_best = c
                    v_best = v
            h[j] = v_best
            if v_best != old:
                improved = True
        if not improved:
            break
    return h

def _coord_frac(h0, n, target, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
