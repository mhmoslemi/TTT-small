import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _smooth_obj(x, target_sum, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    if m < -500:
        return m * (2.0 / n)
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def _de_optimize(h0, n, target_sum, pop_size, n_generations, rng):
    best = h0.copy()
    best_val = _c5_of(best)
    pop = np.zeros((pop_size, n))
    pop[0] = best
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.25:
            pop[i] = _project(best + rng.randn(n) * 0.08, target_sum)
        elif r < 0.45:
            pop[i] = _project(best + rng.randn(n) * 0.25, target_sum)
        elif r < 0.6:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif r < 0.75:
            pop[i] = _fix_sum((rng.rand(n) < 0.5).astype(float), target_sum)
        else:
            xg = np.linspace(0, 2, n, endpoint=False)
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.2, 0.5)
            pop[i] = _fix_sum(0.5 + amp * np.cos(freq * np.pi * xg + phase), target_sum)
    vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    stall_count = 0
    for gen in range(n_generations):
        order = np.argsort(vals)
        pop = pop[order]
        vals = vals[order]
        if vals[0] < best_val:
            best_val = vals[0]
            best = pop[0].copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 20 and gen % 20 == 19:
            n_repop = pop_size // 4
            for i in range(n_repop, pop_size):
                r = rng.random()
                if r < 0.5:
                    pop[i] = _fix_sum(rng.rand(n), target_sum)
                else:
                    pop[i] = _fix_sum((rng.rand(n) < 0.5).astype(float), target_sum)
                vals[i] = _c5_of(pop[i])
        progress = gen / max(n_generations, 1)
        F = 0.9 - 0.4 * progress
        CR = 0.4 + 0.3 * rng.random()
        for i in range(1, pop_size):
            r = rng.random()

def _coord_descent_pairs(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    best_val = _c5_of(h)
    steps = [0.05, 0.1, 0.2, 0.3, 0.5]
    for rd in range(max_rounds):
        improved = False
        n_pairs = rng.randint(20, 50)
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for di in steps:
                for dj in steps:
                    for si in [-1, 1]:
                        for sj in [-1, 1]:
                            trial = h.copy()
                            trial[i] = np.clip(h[i] + si * di, 0, 1)
                            trial[j] = np.clip(h[j] + sj * dj, 0, 1)
                            trial = _project(trial, target_sum)
                            cv = _c5_of(trial)
                            if cv < best_val - 1e-14:
                                h = trial
                                best_val = cv
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_val

def _anneal(h0, n, target_sum, t_start, t_end, n_iters, rng):
    h = h0.copy()
    best = h0.copy()
    best_val = _c5_of(h)
    cur_val = best_val
    t = t_start
    for it in range(n_iters):
        temp = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        trial = h.copy()
        n_pert = rng.randint(1, 4)
        indices = rng.choice(n, n_pert, replace=False)
        for idx in indices:
            delta = rng.randn() * 0.15
            trial[idx] = np.clip(h[idx] + delta, 0, 1)
        trial = _project(trial, target_sum)
        trial_val = _c5_of(trial)
        delta_e = trial_val - cur_val
        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(temp, 1e-10)):
            h = trial
            cur_val = trial_val
            if cur_val < best_val:
                best_val = cur_val
                best = h.copy()
    return best, best_val

def _slsqp_pass(h0, n, target_sum, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, target_sum, alpha, n)
    def con_fun(x):
        return x.sum() - target_sum
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0.copy()
