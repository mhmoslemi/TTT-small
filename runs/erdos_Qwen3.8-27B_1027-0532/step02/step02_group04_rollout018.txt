```python
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def lse_max(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha

def lse_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad

def optimize_lse(h0, alpha, n_pts, mu, maxiter, ftol=1e-18, gtol=1e-12):
    def obj(h):
        corr = compute_correlations(h)
        sm = lse_max(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s
    
    def grad(h):
        g = lse_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
    return res.x

def active_set_optimize(h, n_resteps, alpha, mu, n_top, maxiter_per_step):
    """Optimize focusing on the top n_top correlations."""
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(best)
    
    for _ in range(n_resteps):
        corr = compute_correlations(best)
        # Get the top indices
        top_idx = np.argsort(corr)[-n_top:]
        # Optimize the sum of top correlations
        def obj_top(h):
            c = compute_correlations(h)
            s = np.sum(h) - N / 2.0
            return np.sum(c[top_idx]) + mu * s * s
        
        def grad_top(h):
            g = np.zeros(N)
            # Gradient of sum of top correlations
            for idx in top_idx:
                lag = N - 1 - idx
                if lag == 0:
                    g += (1.0 - 2.0 * h)
                elif lag > 0:
                    if lag < N:
                        g[:N - lag] += 1.0
                        g[:N - lag] -= h[lag:]
                        g[lag:] -= h[:N - lag]
                else:
                    l = -lag
                    if l < N:
                        g[l:] += 1.0
                        g[l:] -= h[:N - l]
                        g[:N - l] -= h[l:]
            s = np.sum(h) - N / 2.0
            g += 2.0 * mu * s
            return g
        
        bounds = [(0.0, 1.0)] * N
        res = minimize(obj_top, best, jac=grad_top, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter_per_step, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, N / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best = h_new.copy()
    return best, best_score

def local_search_binary(h, n_iters, rng):
    """Local search on binarized version, then relax."""
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(best)
    threshold = np.median(best)
    binary = (best >= threshold).astype(float)
    binary = project_feasible(binary, N / 2.0)
    b_score = evaluate_score(binary)
    
    for _ in range(n_iters):
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j or binary[i] == binary[j]:
            continue
        binary[i] = 1.0 - binary[i]
        binary[j] = 1.0 - binary[j]
        s = evaluate_score(binary)
        if s < b_score - 1e-12:
            b_score = s
        else:
            binary[i] = 1.0 - binary[i]
            binary[j] = 1.0 - binary[j]
        if b_score < best_score:
            best = project_feasible(binary, N / 2.0)
            best_score = evaluate_score(best)
    return best, best_score

def block_search(h, rng, n_attempts):
    """Try flipping or shifting blocks."""
    N = len(h)
    best = h.copy()
    best_score = evaluate_score(best)
    
    for _ in range(n_attempts):
        block_len = rng.randint(3, max(4, N // 10))
        start = rng.randint(0, N - block_len)
        end = start + block_len
        
        action = rng.randint(3)
        trial = best.copy()
        
        if action == 0:

            trial[start:end] = 1.0 - trial[start:end]
        elif action == 1:
            # Shift block value up
            trial[start:end] = np.clip(trial[start:end] + 0.1, 0, 1)
        else:
            # Shift block value down
            trial[start:end] = np.clip(trial[start:end] - 0.1, 0, 1)
        
        trial = project_feasible(trial, N / 2.0)
        s = evaluate_score(trial)
        if s < best_score - 1e-12:
            best = trial
            best_score = s
    
    return best, best_score
```

```python
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def lse_objective(h, alpha, N, mu):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    val = m + np.log(np.sum(np.exp(shifted - max_s))) / alpha
    s = np.sum(h) - N / 2.0
    return val + mu * s * s

def lse_grad(h, alpha, N, mu):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(N)
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        lag = N - 1 - idx
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N-lag] += wi
            grad[:N-lag] -= wi * h[lag:]
            grad[lag:] -= wi * h[:N-lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += wi
            grad[l:] -= wi * h[:N-l]
            grad[:N-l] -= wi * h[l:]
    s = np.sum(h) - N / 2.0
    grad += 2.0 * mu * s
    return grad

def run_lbfgs(h0, alpha, N, mu, maxiter):
    bounds = [(0.0, 1.0)] * N
    res = minimize(lse_objective, h0, args=(alpha, N, mu),
                   jac=lambda h: lse_grad(h, alpha, N, mu),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x

def topk_objective(h, top_idx, N, mu):
    corr = compute_correlations(h)
    val = np.sum(corr[top_idx])
    s = np.sum(h) - N / 2.0
    return val + mu * s * s

def topk_grad(h, top_idx, N, mu):
    grad = np.zeros(N)
    for idx in top_idx:
        lag = N - 1 - idx
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N-lag] += 1.0
            grad[:N-lag] -= h[lag:]
            grad[lag:] -= h[:N-lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += 1.0
            grad[l:] -= h[:N-l]
            grad[:N-l] -= h[l:]
    s = np.sum(h) - N / 2.0
    grad += 2.0 * mu * s
    return grad

def run_topk(h0, n_top, N, mu, maxiter):
    corr = compute_correlations(h0)
    top_idx = np.argsort(corr)[-n_top:]
    bounds = [(0.0, 1.0)] * N
    res = minimize(topk_objective, h0, args=(top_idx, N, mu),
                   jac=lambda h: topk_grad(h, top_idx, N, mu),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x
```

Then:
grad[p] = w_0*(1-2h[p])
         + (Σ_{k=1}^{p} t_k^+) + (Σ_{k=1}^{m} t_k^-)
         - Σ_{k=1}^{p} s_k * h[p-k]
         - Σ_{k=1}^{m} s_k * h[p+k]

Hmm, this is still O(N) per p, giving O(N²) total. But the sums Σ_{k=1}^{p} t_k^+ and Σ_{k=1}^{m} t_k^- can be precomputed as prefix sums.

The remaining terms are:
- Σ_{k=1}^{p} s_k * h[p-k] = (s * h)[p] in some sense (a correlation)
- Σ_{k=1}^{m} s_k * h[p+k] = this is like a reverse correlation

These can be computed using np.correlate!

Let me define:
A[p] = Σ_{k=1}^{p} s_k * h[p-k] = correlation of s[1:N-1] with h, at position p
B[p] = Σ_{k=1}^{N-1-p} s_k * h[p+k] = this is correlation of s[1:N-1] with h, but reversed

Actually, let me think about this differently.

A[p] = Σ_{k=1}^{p} s[k] * h[p-k] (for k from 1 to min(p, N-1))

This is the (p-1)-th element of the full correlation of s[1:N] and h (or something like that).

Let me define s_idx = s[1:N] (length N-1, indices 0 to N-2 corresponding to lags 1 to N-1).

Then A[p] = Σ_{k=1}^{min(p, N-1)} s_idx[k-1] * h[p-k]

This is the cross-correlation of s_idx with h at lag p (in the "valid" sense for the overlapping part).

Similarly, B[p] = Σ_{k=1}^{min(N-1-p, N-1)} s[k] * h[p+k] = Σ_{k=1}^{N-1-p} s_idx[k-1] * h[p+k]

Let j = p+k, then k = j-p, and we sum over j from p+1 to N-1:
B[p] = Σ_{j=p+1}^{N-1} s_idx[j-p-1] * h[j]

This is like the cross-correlation but with a different alignment.

Alternatively, if I define s_rev = s_idx[::-1], then:
B[p] = Σ_{k=1}^{N-1-p} s_idx[k-1] * h[p+k]

Let me substitute: let's define a new array where we flip h: h_flip[j] = h[N-1-j].
Then B[p] = Σ_{k=1}^{N-1-p} s_idx[k-1] * h_flip[N-1-p-k]

Hmm, this is getting complicated. Let me just use the direct O(N) approach per p, but vectorize it using numpy.

Actually, the simplest approach that's still efficient:

For the gradient, I can compute it as:
grad = w_0 * (1 - 2*h)

Then for each lag k (from 1 to N-1):
  contrib_k = zeros(N)
  contrib_k[:N-k] -= s_k * h[k:]  (from -h[p+k] terms, where s_k = w_k^+ + w_k^-... wait no)

Let me just go back to the original formula and compute it directly but vectorized:

```python
def lse_grad_fast(h, alpha, N, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(N)
    
    # Zero lag
    w0 = w[N-1]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    
    # Positive lags (k = 1 to N-1)
    for k in range(1, N):
        wk_plus = w[N-1+k]
        if wk_plus < 1e-14:
            continue
        # g_k[p] = -h[p+k] (if p < N-k) + (1-h[p-k]) (if p >= k)
        grad[:N-k] -= wk_plus * h[k:]
        grad[k:] += wk_plus * (1.0 - h[:N-k])
    
    # Negative lags (k = 1 to N-1)
    for k in range(1, N):
        wk_minus = w[N-1-k]
        if wk_minus < 1e-14:
            continue
        # g_{-k}[p] = (1-h[p+k]) (if p < N-k) + (-h[p-k]) (if p >= k)
        grad[:N-k] += wk_minus * (1.0 - h[k:])
        grad[k:] -= wk_minus * h[:N-k]
    
    s = np.sum(h) - N / 2.0
    grad += 2.0 * mu * s
    return grad
```

This is O(N²) but with small constant factors (each inner operation is a numpy slice operation). For N=850, the loop runs 2*(N-1) = 1698 times, each doing O(N) work. Total: ~1.4M operations per gradient. With numpy overhead for each slice op, this might be ~10-50ms per gradient. With 200-500 L-BFGS iterations, that's 2-25s per alpha value. With 9 alpha values, 18-225s total. This should be manageable.

Actually, for the LSE with high alpha, most weights will be near zero (only a few lags have significant weight). So the `if wk_plus < 1e-14: continue` check will skip most iterations, making it much faster in practice.

Let me now think about whether there's a smarter approach.

**Key realization**: With high alpha, the LSE approximation to max becomes very sharp - only the top 1-3 lags have significant weight. So the gradient computation becomes O(N) in practice (only 1-3 lags contribute).

This means the bottleneck is the initial iterations when alpha is low and many lags contribute. But we can skip those by starting with a high alpha directly.

**Revised strategy:**
1. Start at n=850 (double the initial)
2. Immediately use high alpha (1000+) since the initial solution is already good
3. Focus on the final precision with very high alpha
4. Use top-k optimization as a complementary approach
5. Try some diverse restarts

Let me also consider: should I work at even higher resolution? The previous code went up to 4*n = 1700. Let me try n=1275 (3x) or n=1700 (4x) directly.

At n=1700, the gradient loop (if all lags contribute) would be 2*1699 = 3398 iterations, each O(1700). That's ~5.8M operations per gradient. With the sparsity of high-alpha LSE, it's much less in practice.

But the L-BFGS memory and computation also scale with N. For N=1700, each function evaluation involves computing the correlation (O(N²) = 2.9M) which is the main cost.

Let me estimate: np.correlate on arrays of length 1700 takes about 2.9M multiply-adds, which numpy can do in ~1-3ms. So each function evaluation is ~1-3ms, and each gradient (with high alpha, few active lags) is ~5-10ms. With 500 L-BFGS iterations, that's ~3-8 seconds per alpha value. Very manageable.

6. Final polish

- Find the argmax lag k*

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def lse_objective(h, alpha, N, mu):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    val = m + np.log(np.sum(np.exp(shifted - max_s))) / alpha
    s = np.sum(h) - N / 2.0
    return val + mu * s * s


def lse_gradient(h, alpha, N, mu):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / np.sum(exp_s)

    grad = np.zeros(N)
    w0 = w[N - 1]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    for k in range(1, N):
        wk_p = w[N - 1 + k]
        if wk_p > 1e-14:
            grad[:N - k] -= wk_p * h[k:]
            grad[k:] += wk_p * (1.0 - h[:N - k])
        wk_n = w[N - 1 - k]
        if wk_n > 1e-14:
            grad[:N - k] += wk_n * (1.0 - h[k:])
            grad[k:] -= wk_n * h[:N - k]
    s = np.sum(h) - N / 2.0
    grad += 2.0 * mu * s
    return grad


def run_lbfgs(h0, alpha, N, mu, maxiter):
    def obj(h):
        return lse_objective(h, alpha, N, mu)

    def jac(h):
        return lse_gradient(h, alpha, N, mu)

    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x


def topk_objective(h, top_idx, N, mu):
    corr = compute_correlations(h)
    val = np.max(corr[top_idx])
    s = np.sum(h) - N / 2.0
    return val + mu * s * s


def topk_gradient(h, top_idx, N, mu):
    corr = compute_correlations(h)
    mx = np.max(corr[top_idx])
    active = [idx for idx in top_idx if corr[idx] > mx - 1e-10]
    grad = np.zeros(N)
    for idx in active:
        lag = N - 1 - idx
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] -= h[lag:]
            grad[lag:] += (1.0 - h[:N - lag])
        else:
            l = -lag
            if l < N:
                grad[l:] += (1.0 - h[:N - l])
                grad[:N - l] -= h[l:]
    s = np.sum(h) - N / 2.0
    grad += 2.0 * mu * s
    return grad


def run_topk_lbfgs(h0, n_top, N, mu, maxiter):
    corr = compute_correlations(h0)
    top_idx = np.argsort(corr)[-n_top:]

    def obj(h):
        return topk_objective(h, top_idx, N, mu)

    def jac(h):
        return topk_gradient(h, top_idx, N, mu)

    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def subgradient_step(h, N, mu, step_size, n_steps):
    best = h.copy()
    best_score = evaluate_score(best)
    for _ in range(n_steps):
        corr = compute_correlations(best)
        k_star = np.argmax(corr)
        lag = N - 1 - k_star
        g = np.zeros(N)
        if lag == 0:
            g = 1.0 - 2.0 * best
        elif 0 < lag < N:
            g[:N - lag] -= best[lag:]
            g[lag:] += (1.0 - best[:N - lag])
        else:
            l = -lag
            if l < N:
                g[l:] += (1.0 - best[:N - l])
                g[:N - l] -= best[l:]
        s = np.sum(best) - N / 2.0
        g += 2.0 * mu * s
        trial = best - step_size * g
        trial = project_feasible(trial, N / 2.0)
        sc = evaluate_score(trial)
        if sc < best_score:
            best = trial
            best_score = sc
            step_size = min(step_size * 1.2, 0.5)
        else:
            step_size *= 0.5
        if step_size < 1e-8:
            break
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    MU = 50000.0

    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)

    n = n0 * 2
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_h = np.interp(x_new, x_old, init)
    best_h = project_feasible(best_h, n / 2.0)
    best_score = evaluate_score(best_h)

    alpha_schedule = [200, 500, 1000, 2000, 5000, 10000, 20000, 40000]
    for alpha in alpha_schedule:

        maxiter = max(200, 120000 // n)
        x = run_lbfgs(best_h, alpha, n, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):

        strategy = i % 6
        if strategy == 0:

        elif strategy == 2:

            n_freq = rng.randint(1, 4)
            for _ in range(n_freq):

                amp = rng.uniform(0.02, 0.10)
                phase = rng.rand() * 2.0 * np.pi
                xx = np.arange(n_cur, dtype=float)
                pert += amp * np.sin(2.0 * np.pi * f * xx / n_cur + phase)
        elif strategy == 3:
            pert = best_h.copy()
            pert = (pert > 0.5).astype(float)
            n_flip = rng.randint(3, max(4, n_cur // 15))
            flip_idx = rng.choice(n_cur, n_flip, replace=False)
            pert[flip_idx] = 1.0 - pert[flip_idx]
        elif strategy == 4:
            pert = best_h.copy()
            n_blocks = rng.randint(2, 5)

            coarse = np.interp(xc, xf, best_h)

            s = evaluate_score(h_new)

                best_h = h_new.copy()

        if n_try > 2500 or n_try <= len(best_h):
            continue
        if time.time() - t0 > budget_s - 100:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False)

    if time.time() - t0 < budget_s - 5:
```
