import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr)) * dx

def smooth_max_val(x, alpha):
    m = float(np.max(x))
    return m + float(np.log(np.sum(np.exp(alpha * (x - m))))) / alpha

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def ap_score(g):
    h = antipal_h(g)
    corr = compute_correlations(h)
    n = len(h)
    return float(np.max(corr)) * 2.0 / n

def project_feasible(h, target_sum):
    N = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def ap_smooth_fun(g, alpha):
    h = antipal_h(g)
    corr = compute_correlations(h)
    return smooth_max_val(corr, alpha)

def ap_grad(g, alpha):
    h = antipal_h(g)
    corr = compute_correlations(h)
    n = len(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / float(np.sum(ex))
    half = len(g)
    # Compute gradient w.r.t. h first
    grad_h = np.zeros(n)
    # corr[j] = sum_i h[i] * (1-h[i + j - (n-1)])
    # d(corr[j])/d(h[i]) = (1-h[i+j-(n-1)]) if i+j-(n-1) != i, else 1-2h[i]... 
    # Actually it's simpler: corr[j] = sum_{valid i} h[i] * (1-h[i+s]) where s = j-(n-1)
    # d/dh[i] = (1-h[i+s]) - h[i] * delta (if i == i+s, i.e., s=0)
    # For s != 0: d/dh[i] of h[i]*(1-h[i+s]) = (1-h[i+s]) for the h[i] term, -h[i] for the h[i+s] term
    # Let me just do it by looping over shifts
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad_h[i] += w * (1.0 - h[i + s])
                grad_h[i + s] -= w * h[i]
        else:
            t = -s
            for i in range(n - t):
                grad_h[i + t] += w * (1.0 - h[i])
                grad_h[i] -= w * h[i + t]
    # Now map grad_h to grad_g
    # h[i] = g[i] for i < half
    # h[half] = 0.5 (fixed)
    # h[i] = 1 - g[n-1-i] for i > half
    grad_g = np.zeros(half)
    for i in range(half):
        grad_g[i] = grad_h[i]
    for i in range(half + 1, n):
        idx = n - 1 - i
        grad_g[idx] -= grad_h[i]
    return grad_g

def slsqp_ap(g0, alpha, maxiter):
    half = len(g0)
    bounds = [(0.0, 1.0)] * half
    res = minimize(
        ap_smooth_fun, g0, method='SLSQP', bounds=bounds,
        jac=lambda x: ap_grad(x, alpha), args=(alpha,),
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def coord_descent_ap(g0, max_sweeps=50):
    g = g0.copy()
    best_score = ap_score(g)
    for sweep in range(max_sweeps):

                g[j] = res.x
                best_score = res.fun
                improved = True
            else:
                g[j] = saved
        if not improved:
            break
    return g, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr)) * dx


def smooth_max_val(x, alpha):
    m = float(np.max(x))
    return m + float(np.log(np.sum(np.exp(alpha * (x - m))))) / alpha


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def ap_score(g):
    h = antipal_h(g)
    corr = compute_correlations(h)
    n = len(h)
    return float(np.max(corr)) * 2.0 / n


def project_feasible(h, target_sum):
    N = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def ap_smooth_fun(g, alpha):
    h = antipal_h(g)
    corr = compute_correlations(h)
    return smooth_max_val(corr, alpha)


def ap_grad(g, alpha):
    h = antipal_h(g)
    corr = compute_correlations(h)
    n = len(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / float(np.sum(ex))
    half = len(g)
    grad_h = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad_h[i] += w * (1.0 - h[i + s])
                grad_h[i + s] -= w * h[i]
        else:
            t = -s
            for i in range(n - t):
                grad_h[i + t] += w * (1.0 - h[i])
                grad_h[i] -= w * h[i + t]
    grad_g = np.zeros(half)
    for i in range(half):
        grad_g[i] = grad_h[i]
    for i in range(half + 1, n):
        idx = n - 1 - i
        grad_g[idx] -= grad_h[i]
    return grad_g


def slsqp_ap(g0, alpha, maxiter):
    half = len(g0)
    bounds = [(0.0, 1.0)] * half
    res = minimize(
        ap_smooth_fun, g0, method='SLSQP', bounds=bounds,
        jac=ap_grad, args=(alpha,),
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    if res.x is not None:
        return np.clip(res.x, 0.0, 1.0)
    return None


def golden_section_1d(g, j, iters=30):
    a = 0.0
    b = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    g[j] = c
    fc = ap_score(g)
    g[j] = d
    fd = ap_score(g)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            g[j] = c
            fc = ap_score(g)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            g[j] = d
            fd = ap_score(g)
    g[j] = (a + b) / 2.0


def coord_descent_ap(g0, max_sweeps=30, time_limit=200.0):
    t_start = time.time()
    g = g0.copy()
    best_score = ap_score(g)
    n_vars = len(g)
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n_vars):
            saved = g[j]
            golden_section_1d(g, j, iters=28)
            g[j] = max(0.0, min(1.0, g[j]))
            f_new = ap_score(g)
            if f_new < best_score - 1e-14:
                best_score = f_new
                improved = True
            else:
                g[j] = saved
        if not improved:
            break
    return g, best_score
