    elif deficit < -1e-12:
        room = h.copy()
        idx = np.argsort(room)
        i = 0
        deficit = -deficit
        while deficit > 1e-12 and i < n:
            sub = min(deficit * 0.999, room[idx[i]] * 0.999)
            if sub > 0:
                h[idx[i]] -= sub
                deficit -= sub
            i += 1
    return h


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    np.clip(shifted, -700, 700, out=shifted)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = (conv1[idx] - conv2[idx]) / alpha
    return obj, grad


def active_set_obj_grad(h, alpha, top_k):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    k_indices = np.argsort(corrs)[-top_k:]
    active_weights = np.zeros(len(corrs))
    active_vals = corrs[k_indices]
    am = np.max(active_vals)
    shifted = alpha * (active_vals - am)
    np.clip(shifted, -700, 700, out=shifted)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w_active = exp_shifted / sum_exp
    active_weights[k_indices] = w_active
    obj = am + np.log(sum_exp) / alpha
    w_rev = active_weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, active_weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = (conv1[idx] - conv2[idx]) / alpha
    return obj, grad, k_indices


def penalty_obj_grad(h, alpha, mu):
    obj, grad = smooth_max_obj_grad(h, alpha)
    n = len(h)
    viol = np.sum(h) - n / 2.0
    obj += 0.5 * mu * viol * viol
    grad += mu * viol
    return obj, grad


def active_penalty_obj_grad(h, alpha, mu, top_k):
    obj, grad, _ = active_set_obj_grad(h, alpha, top_k)
    n = len(h)
    viol = np.sum(h) - n / 2.0
    obj += 0.5 * mu * viol * viol
    grad += mu * viol
    return obj, grad


def lbfgsb_optimize(h0, alpha, mu, maxiter, bounds, use_active=False, top_k=10):
    if use_active:
        res = minimize(
            active_penalty_obj_grad, h0, method='L-BFGS-B',
            bounds=bounds, args=(alpha, mu, top_k),
            jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'iprint': 0, 'maxls': 50, 'maxfun': maxiter * 30}
        )
    else:
        res = minimize(
            penalty_obj_grad, h0, method='L-BFGS-B',
            bounds=bounds, args=(alpha, mu),
            jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'iprint': 0, 'maxls': 50, 'maxfun': maxiter * 30}
        )
    return res.x


def slsqp_obj(h, alpha):
    return smooth_max_obj_grad(h, alpha)[0]


def slsqp_jac(h, alpha):
    return smooth_max_obj_grad(h, alpha)[1]


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        slsqp_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x


def freq_perturbation(h, n_points, rng, max_freq=8):
    x = np.linspace(0.0, 2.0, n_points, endpoint=False)
    pert = np.zeros(n_points)
    n_harmonics = rng.randint(1, 4)
    for _ in range(n_harmonics):
        f = rng.randint(1, max_freq + 1)
        amp = rng.uniform(0.02, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    return h + pert


def block_perturbation(h, rng, n_blocks):
    n = len(h)
    pert = h.copy()
    block_size = n // n_blocks
    for i in range(n_blocks):
        if rng.rand() < 0.4:
            start = i * block_size
            end = min(start + block_size, n)
            pert[start:end] = 1.0 - pert[start:end]
            if rng.rand() < 0.5:
                pert[start:end] += rng.randn(end - start) * 0.1
    return np.clip(pert, 0.0, 1.0)


def antisymmetric_perturbation(h, rng):
    n = len(h)
    h_flip = h[::-1].copy()
    h_anti = 0.5 * (h + (1.0 - h_flip))
    noise = rng.randn(n) * 0.05
    return np.clip(h_anti + noise, 0.0, 1.0)


def structured_init(n, rng, strategy):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if strategy == 0:
        f = rng.randint(2, 6)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
    elif strategy == 1:
        periods = rng.randint(3, 8)
        h = np.zeros(n)
        block_len = n // (2 * periods)
        for i in range(periods):
            start = 2 * i * block_len
            end = start + block_len
            if end <= n:
                h[start:end] = 1.0
    elif strategy == 2:
        h = np.full(n, 0.5)
        n_changes = rng.randint(n // 10, n // 3)
        change_points = np.sort(rng.choice(n, n_changes, replace=False))
        val = 0.5 + 0.3 * rng.uniform(-1, 1)
        h[change_points[0]:] = val
        for cp in change_points[1:]:
            val = np.clip(val + rng.uniform(-0.4, 0.4), 0.0, 1.0)
            h[cp:] = val
    elif strategy == 3:
        f1 = rng.randint(2, 5)
        f2 = rng.randint(5, 10)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f1 * x / 2.0) + 0.2 * np.sin(2 * np.pi * f2 * x / 2.0)
    elif strategy == 4:
        h = np.zeros(n)
        n_blocks = rng.randint(5, 15)
        block_size = n // n_blocks
        for i in range(n_blocks):
            start = i * block_size
            end = min(start + block_size, n)
            h[start:end] = rng.uniform(0.0, 1.0)
    else:
        h = rng.rand(n) * 0.5 + 0.25
    return np.clip(h, 0.0, 1.0)


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=None)
    return np.clip(h_new, 0.0, 1.0)


def sa_step(h, temp, n_points, rng):
    n = len(h)
    strategy = rng.randint(5)
    if strategy == 0:
        idx = rng.randint(n)
        delta = rng.uniform(-0.1, 0.1)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
    elif strategy == 1:
        block_size = rng.randint(2, max(3, n // 20))
        start = rng.randint(0, n - block_size)
        h_new = h.copy()
        h_new[start:start + block_size] = 1.0 - h_new[start:start + block_size]
    elif strategy == 2:
        idx = rng.choice(n, size=rng.randint(3, max(4, n // 10)), replace=False)
        h_new = h.copy()
        h_new[idx] += rng.randn(len(idx)) * 0.15
        np.clip(h_new, 0.0, 1.0, out=h_new)
    elif strategy == 3:
        h_new = h + freq_perturbation(np.zeros(n), n, rng, max_freq=6) - np.zeros(n)
        h_new = h + rng.randn(n) * 0.05
        np.clip(h_new, 0.0, 1.0, out=h_new)
    else:
        idx = rng.choice(n, size=rng.randint(1, max(2, n // 30)), replace=False)
        h_new = h.copy()
        h_new[idx] = rng.uniform(0.0, 1.0, len(idx))
    h_new = project_sum(h_new, n / 2.0)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)
    best_h = project_sum(init, n0 / 2.0)
    best_score = evaluate_score(best_h)
    best_n = n0

    def time_left():
        return budget_s - (time.time() - t0)

    def try_accept(h_candidate):
        nonlocal best_h, best_score, best_n
        h_c = project_sum(h_candidate, len(h_candidate) / 2.0)
        if not np.all(np.isfinite(h_c)):
            return False
        s = evaluate_score(h_c)
        if np.isfinite(s) and s < best_score:
            best_h = h_c.copy()
            best_score = s
            best_n = len(h_c)
            return True
        return False

    # Phase 1: L-BFGS-B refinement at original resolution with active set
    bounds0 = [(0.0, 1.0)] * n0
    alphas_p1 = [50, 100, 200, 400, 800, 1500, 3000, 5000, 8000]
    for alpha in alphas_p1:
        if time_left() < 200:
            break
        h_curr = best_h.copy()
        for top_k in [5, 8, 15, 25]:
            if time_left() < 100:
                break
            maxiter = max(100, min(500, 150000 // n0))
            res_x = lbfgsb_optimize(h_curr, alpha, 5000.0, maxiter, bounds0,
                                    use_active=True, top_k=top_k)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n0 / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_curr = h_new.copy()

    # Phase 2: Full smooth max L-BFGS-B
    for alpha in [1000, 3000, 6000, 10000, 20000]:
        if time_left() < 150:
            break
        maxiter = max(100, min(400, 120000 // best_n))
        bounds_best = [(0.0, 1.0)] * best_n
        res_x = lbfgsb_optimize(best_h, alpha, 8000.0, maxiter, bounds_best,
                                use_active=False)
        if res_x is not None:
            try_accept(np.clip(res_x, 0.0, 1.0))

    # Phase 3: Multi-resolution exploration
    resolutions = [300, 400, 450, 500, 550, 600, 700, 800, 900, 1000, 1200, 1500]
    for n_try in resolutions:
        if time_left() < 250:
            break
        h_init = interpolate_h(best_h, n_try)
        h_init = project_sum(h_init, n_try / 2.0)
        bounds_try = [(0.0, 1.0)] * n_try
        h_curr = h_init.copy()
        for alpha in [100, 300, 800, 2000, 5000]:
            if time_left() < 100:
                break
            maxiter = max(80, min(400, 200000 // n_try))
            for top_k in [10, 20]:
                if time_left() < 60:
                    break
                res_x = lbfgsb_optimize(h_curr, alpha, 10000.0, maxiter, bounds_try,
                                        use_active=True, top_k=top_k)
                if res_x is not None:
                    h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s = evaluate_score(h_new)
                        if np.isfinite(s) and s < best_score:
                            best_score = s
                            best_h = h_new.copy()
                            best_n = n_try
                            h_curr = h_new.copy()

    # Phase 4: SLSQP refinement
    bounds_best = [(0.0, 1.0)] * best_n
    for alpha in [2000, 5000, 10000, 20000]:
        if time_left() < 120:
            break
        res_x = slsqp_optimize(best_h, alpha, 200, bounds_best)
        if res_x is not None:
            try_accept(np.clip(res_x, 0.0, 1.0))

    # Phase 5: Structured random restarts
    n_best = best_n
    bounds_best = [(0.0, 1.0)] * n_best
    max_restarts = 300
    for i in range(max_restarts):
        if time_left() < 80:
            break
        strategy = i % 8
        if strategy == 0:
            pert = best_h + rng.randn(n_best) * 0.04
        elif strategy == 1:
            pert = best_h + rng.randn(n_best) * 0.08
        elif strategy == 2:
            pert = freq_perturbation(best_h, n_best, rng, max_freq=10)
        elif strategy == 3:
            pert = block_perturbation(best_h, rng, n_blocks=rng.randint(4, 12))
        elif strategy == 4:
            pert = antisymmetric_perturbation(best_h, rng)
        elif strategy == 5:
            pert = structured_init(n_best, rng, strategy=rng.randint(6))
        elif strategy == 6:
            pert = best_h.copy()
            n_flip = max(3, n_best // 25)
            idx = rng.choice(n_best, size=n_flip, replace=False)
            pert[idx] = 1.0 - pert[idx]
        else:
            pert = best_h + rng.randn(n_best) * 0.06
            pert = project_sum(pert, n_best / 2.0)
            idx = rng.randint(n_best)
            block_sz = rng.randint(2, max(3, n_best // 15))
            end = min(idx + block_sz, n_best)
            pert[idx:end] = 1.0 - pert[idx:end]

        pert = project_sum(pert, n_best / 2.0)
        alpha = 200 + 400 * rng.randint(8)
        maxiter_r = max(60, min(300, 100000 // n_best))
        use_act = (i % 3 == 0)
        tk = [5, 10, 15][i % 3]
        res_x = lbfgsb_optimize(pert, alpha, 8000.0, maxiter_r, bounds_best,
                                use_active=use_act, top_k=tk)
        if res_x is not None:
            try_accept(np.clip(res_x, 0.0, 1.0))

    # Phase 6: Simulated annealing with structured moves
    if time_left() > 120:
        n_sa = best_n
        h_sa = best_h.copy()
        s_sa = best_score
        temp = 0.002
        cooling = 0.995
        n_sa_steps = min(50000, int((time_left() - 120) * 80))
        for step in range(n_sa_steps):
            if time_left() < 100:
                break
            h_new = sa_step(h_sa, temp, n_sa, rng)
            s_new = evaluate_score(h_new)
            if s_new < s_sa or rng.rand() < np.exp(-(s_new - s_sa) / (temp * 0.01)):
                h_sa = h_new
                s_sa = s_new
                if s_sa < best_score:
                    best_score = s_sa
                    best_h = h_sa.copy()
            if step % 500 == 0:
                temp *= cooling

    # Phase 7: Final polish with very high alpha
    for alpha in [30000, 60000, 100000]:
        if time_left() < 30:
            break
        res_x = lbfgsb_optimize(best_h, alpha, 20000.0, 300, bounds_best,
                                use_active=False)
        if res_x is not None:
            try_accept(np.clip(res_x, 0.0, 1.0))

    # Phase 8: Try adjacent resolutions
    for delta_n in [-10, -5, 5, 10, 20, -20, 25, -25]:
        if time_left() < 20:
            break
        n_adj = best_n + delta_n
        if n_adj < 100 or n_adj > 3000:
            continue
        h_adj = interpolate_h(best_h, n_adj)
        h_adj = project_sum(h_adj, n_adj / 2.0)
        bounds_adj = [(0.0, 1.0)] * n_adj
        for alpha in [3000, 8000, 15000]:
            if time_left() < 10:
                break
            res_x = lbfgsb_optimize(h_adj, alpha, 15000.0, 150, bounds_adj,
                                    use_active=True, top_k=10)
            if res_x is not None:
                try_accept(np.clip(res_x, 0.0, 1.0))

    # Final projection and evaluation
    best_h = project_sum(best_h, best_n / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, best_n)
