def project_sum(h, target):
    """Project h to satisfy sum(h) = target and 0 <= h <= 1."""
    h = h.copy()
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def get_slopes(h, j):
    """Get dc_p/dh[j] for all p."""
    n = len(h)
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    return slopes

def cd_single_step(h, j, corr):
    """Perform one coordinate descent step, return new value and slopes."""
    n = len(h)
    slopes = get_slopes(h, j)
    a = corr - h[j] * slopes
    
    zero_mask = np.abs(slopes) < 1e-14
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    
    if lo_V >= hi_V - 1e-14:
        return h[j], slopes
    
    for _ in range(42):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):

                feasible = False

            if lo_t > hi_t + 1e-14:

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = cd_single_step(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr += change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    corr = compute_correlations(h)
    h = project_sum(h, n / 2.0)
    return h

def subgradient_step(h, corr, step_size, eps=0.01):
    """Take one projected subgradient step."""
    n = len(h)
    m = 2 * n - 1
    max_val = np.max(corr)
    active = np.where(corr > max_val - eps)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    # Compute gradient as weighted average of active gradients
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            idx1 = j + k - (n - 1)
            idx2 = j - k + (n - 1)
            g = 0.0
            if 0 <= idx1 < n:
                g += (1.0 - h[idx1])
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[j] += g
    
    grad /= len(active)
    
    h_new = h - step_size * grad
    h_new = project_sum(h_new, n / 2.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    return h_new

    return float(np.max(corr) * dx)

        diff = target - s
        if abs(diff) < 1e-15:

    p = np.arange(m)

def compute_swap_delta(h, i, j, n):
    """Compute change in all correlations when swapping h[i] (was 1) and h[j] (was 0)."""
    m = 2 * n - 1
    p = np.arange(m)

    idx_i = i + p - (n - 1)
    valid_i = (idx_i >= 0) & (idx_i < n)
    term1 = np.zeros(m)
    term1[valid_i] = -(1.0 - h[idx_i[valid_i]])
    
    # Term 2: +(1-h[j+k-(n-1)]) for valid k
    idx_j = j + p - (n-1)
    valid_j = (idx_j >= 0) & (idx_j < n)
    term2 = np.zeros(m)
    term2[valid_j] = (1.0 - h[idx_j[valid_j]])
    
    # Term 3: -h[j-k+(n-1)] for valid k
    idx_j2 = j - p + (n-1)

    term3 = np.zeros(m)
    term3[valid_j2] = -h[idx_j2[valid_j2]]

    term4 = np.zeros(m)
    term4[valid_i2] = h[idx_i2[valid_i2]]
    
    return term1 + term2 + term3 + term4

def compute_swap_delta_general(h, i, j, n):
    """Compute change in correlations when swapping h[i] and h[j]."""
    a = h[i]
    b = h[j]
    if a == b:
        return np.zeros(2*n-1)
    
    m = 2 * n - 1
    p = np.arange(m)
    d = b - a  # This is +1 if a=0,b=1, or -1 if a=1,b=0
    
    idx_i = i + p - (n - 1)

    idx_j = j + p - (n - 1)
    valid_j = (idx_j >= 0) & (idx_j < n)
    idx_i2 = i - p + (n - 1)

    valid_j2 = (idx_j2 >= 0) & (idx_j2 < n)

    if np.any(valid_j2):
        delta[valid_j2] += d * h[idx_j2[valid_j2]]
    
    return delta
