import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def upsample_h(h, n_from, n_to):
    pos = np.linspace(0.0, float(n_from - 1), n_to)
    idx = np.minimum(pos.astype(int), n_from - 1)
    frac = pos - idx
    idx2 = np.minimum(idx + 1, n_from - 1)
    return (1.0 - frac) * h[idx] + frac * h[idx2]

def make_binary(h, n):
    half = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:half]] = 1.0
    return out

def compute_S(h, n, k_lag):
    a = np.arange(n) - k_lag
    b = np.arange(n) + k_lag
    Sa = np.where((a >= 0) & (a < n), h[np.clip(a, 0, n - 1)], 0.0)
    Sb = np.where((b >= 0) & (b < n), h[np.clip(b, 0, n - 1)], 0.0)
    return Sa + Sb

def greedy_binary_swap(h, n, max_iters, time_fn, rng):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    no_improve = 0
    for it in range(max_iters):
        if time_fn():
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        max_idx = np.argmax(corr)
        k_lag = max_idx - (n - 1)
        S = compute_S(h, n, k_lag)
        s_ones = S[ones]
        s_zeros = S[zeros]
        i_best = ones[np.argmin(s_ones)]
        j_best = zeros[np.argmax(s_zeros)]
        delta = s_zeros.max() - s_ones.min()
        if delta <= 0:
            no_improve += 1
            if no_improve >= 30:
                break
            found = False
            ntry = min(30, len(ones) * len(zeros))
            for _ in range(ntry):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i] = 1.0 - h[i]
                h[j] = 1.0 - h[j]
                corr_new = np.correlate(h, 1.0 - h, mode="full")
                c5_new = np.max(corr_new) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    corr = corr_new
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    no_improve = 0
                    found = True
                    break
                else:
                    h[i] = 1.0 - h[i]
                    h[j] = 1.0 - h[j]
            if found:
                continue
        else:
            h[i_best] = 1.0 - h[i_best]
            h[j_best] = 1.0 - h[j_best]
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr) * dx

    return h, best_c5

def sa_binary_swap(h, n, max_iters, time_fn, rng, T_start=0.5, T_end=0.001):

    best_c5 = evaluate_c5(h)

def coord_descent_fast(h, n, max_sweeps, time_fn, deltas):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    no_improve_sweeps = 0
    for sweep in range(max_sweeps):
        if time_fn():
            break
        improved = False
        for i in range(n):
            for d in deltas:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5 = evaluate_c5(h_try)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    h = h_try
                    improved = True
                    break
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
        else:
            no_improve_sweeps = 0
    return h, best_c5

def slsqp_smooth(h_start, n, eps, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            l = k - (n - 1)
            wk = w[k]
            if wk < 1e-14:
                continue
            a = np.arange(n) - l
            b = np.arange(n) + l
            term1 = np.where((a >= 0) & (a < n), 1.0 - x[np.clip(a, 0, n - 1)], 0.0)
            term2 = np.where((b >= 0) & (b < n), -x[np.clip(b, 0, n - 1)], 0.0)
            g += wk * (term1 + term2)
        g *= dx
        return g
    def constr(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h_start.copy(), jac=grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': constr}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)
