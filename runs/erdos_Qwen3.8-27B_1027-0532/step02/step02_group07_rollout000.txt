This is a nice reformulation! We want to arrange n/2 ones in n positions such that the minimum number of 1-1 pairs at any distance is maximized.

This is related to the problem of finding "well-distributed" subsets.

For a random arrangement, pairs_k ≈ (n/2)^2 / n = n/4 for each k (by symmetry, approximately). So the expected overlap at each lag is n/2 - n/4 = n/4, times dx = 2/n, giving overlap ≈ 1/2. That's the random baseline.

To do better, we need a structured arrangement. The optimal arrangement would have pairs_k as equal as possible across all k.

This is essentially a "balanced incomplete block design" or "difference set" problem!

A (v, k, λ) difference set in a group G of order v is a k-element subset D such that every nonzero group element is represented exactly λ times as a difference of two elements of D.

If we can find such a structure, the overlap would be uniform across all lags.

For our problem: v = n, k = n/2. We want a difference set with parameters (n, n/2, λ).
The parameter relation is: k(k-1) = λ(v-1)
So: (n/2)(n/2 - 1) = λ(n-1)
λ = (n/2)(n/2-1)/(n-1) = n(n-2)/(4(n-1))

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j
    for _ in range(35):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0))


def get_coord_slopes(h, j, n):
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    return slopes


def coordinate_descent(h, n_sweeps=300, tol=1e-11, time_budget=60.0, recompute_every=25):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    order = np.arange(n)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in order:
            t_opt = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                slopes = get_coord_slopes(h, j, n)
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % recompute_every == recompute_every - 1:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def active_set_cd(h, n_sweeps=200, time_budget=30.0, top_frac=0.6):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        thresh = np.quantile(corr, top_frac)
        active = np.where(corr > thresh)[0]
        active_coords = set()
        for k in active:
            offset = k - (n - 1)
            if 0 <= offset < n:
                active_coords.add(offset)
            lo = max(0, -offset)
            hi = min(n, n - offset)
            active_coords.update(range(lo, hi))
        max_change = 0.0
        for j in sorted(active_coords):
            t_opt = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                slopes = get_coord_slopes(h, j, n)
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < 1e-12:
            break
    h = project_sum(h, n / 2.0)
    return h


def pair_local_search(h, n_iters=500, time_budget=30.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    current_max = float(np.max(corr))
    t0 = time.time()
    for it in range(n_iters):
        if time.time() - t0 > time_budget:
            break
        i = it % n
        j = (it * 7 + 3) % n
        if i == j:
            j = (j + 1) % n
        slope_i = get_coord_slopes(h, i, n)
        slope_j = get_coord_slopes(h, j, n)
        ti = h[i]
        tj = h[j]
        base = corr - ti * slope_i - tj * slope_j
        new_ti, new_tj = ti, tj
        lo_V = float(np.max(np.minimum(base, base + np.minimum(slope_i + slope_j, 0))))
        hi_V = current_max
        for _ in range(25):
            mid_V = (lo_V + hi_V) * 0.5
            feasible = True
            lo_ti = 0.0
            hi_ti = 1.0
            lo_tj = 0.0
            hi_tj = 1.0
            mixed_slope = slope_i + slope_j
            pos_m = mixed_slope > 1e-14
            neg_m = mixed_slope < -1e-14
            if np.any(neg_m):
                lo_ti = float(np.max((mid_V - base[neg_m]) / slope_i[neg_m]))
            if np.any(pos_m):
                hi_ti = float(np.min((mid_V - base[pos_m]) / slope_i[pos_m]))
            lo_ti = max(0.0, lo_ti)
            hi_ti = min(1.0, hi_ti)
            if lo_ti > hi_ti + 1e-10:
                feasible = False
            if feasible:
                test_ti = (lo_ti + hi_ti) * 0.5
                base2 = base + test_ti * slope_i
                lo_tj2 = 0.0
                hi_tj2 = 1.0
                pos_j = slope_j > 1e-14
                neg_j = slope_j < -1e-14
                zero_j = np.abs(slope_j) < 1e-14
                if np.any(zero_j):
                    if float(np.max(base2[zero_j])) > mid_V + 1e-10:
                        feasible = False
                if feasible:
                    if np.any(neg_j):
                        lo_tj2 = float(np.max((mid_V - base2[neg_j]) / slope_j[neg_j]))
                    if np.any(pos_j):
                        hi_tj2 = float(np.min((mid_V - base2[pos_j]) / slope_j[pos_j]))
                    lo_tj2 = max(0.0, lo_tj2)
                    hi_tj2 = min(1.0, hi_tj2)
                    if lo_tj2 > hi_tj2 + 1e-10:
                        feasible = False
            if feasible:
                hi_V = mid_V
            else:
                lo_V = mid_V
        V_star = (lo_V + hi_V) * 0.5
        test_ti = (0.0, 1.0)
        lo_ti = 0.0
        hi_ti = 1.0
        pos_mi = slope_i > 1e-14
        neg_mi = slope_i < -1e-14
        if np.any(neg_mi):
            lo_ti = float(np.max((V_star - base[neg_mi]) / slope_i[neg_mi]))
        if np.any(pos_mi):
            hi_ti = float(np.min((V_star - base[pos_mi]) / slope_i[pos_mi]))
        lo_ti = max(0.0, lo_ti)
        hi_ti = min(1.0, hi_ti)
        if lo_ti <= hi_ti:
            test_ti = (lo_ti + hi_ti) * 0.5
            base2 = base + test_ti * slope_i
            lo_tj = 0.0
            hi_tj = 1.0
            pos_j = slope_j > 1e-14
            neg_j = slope_j < -1e-14
            zero_j = np.abs(slope_j) < 1e-14
            if np.any(neg_j):
                lo_tj = float(np.max((V_star - base2[neg_j]) / slope_j[neg_j]))
            if np.any(pos_j):
                hi_tj = float(np.min((V_star - base2[pos_j]) / slope_j[pos_j]))
            lo_tj = max(0.0, lo_tj)
            hi_tj = min(1.0, hi_tj)
            if lo_tj <= hi_tj:
                new_ti = test_ti
                new_tj = (lo_tj + hi_tj) * 0.5
                h[i] = new_ti
                h[j] = new_tj
                corr = compute_correlations(h)
                new_max = float(np.max(corr))
                if new_max < current_max:
                    current_max = new_max
                else:
                    h[i] = ti
                    h[j] = tj
                    corr = corr + (ti - new_ti) * slope_i + (tj - new_tj) * slope_j
    h = project_sum(h, n / 2.0)
    return h


def smooth_max_obj(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_polish(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_max_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    used = set()
    positions = list(range(n))
    rng.shuffle(positions)
    for p in positions:
        if len(used) >= target:
            break
        h[p] = 1.0
        used.add(p)
    return h


def rational_init(n, alpha_num, alpha_den):
    h = np.zeros(n)
    alpha = alpha_num / alpha_den
    for i in range(n):
        frac = (i * alpha) % 1.0
        if frac < 0.5:
            h[i] = 1.0
    s = np.sum(h)
    if s != n // 2:
        flat_idx = np.argsort(h)
        if s < n // 2:
            for idx in flat_idx:
                if h[idx] == 0:
                    h[idx] = 1.0
                    s += 1
                    if s == n // 2:
                        break
        else:
            for idx in flat_idx[::-1]:
                if h[idx] == 1:
                    h[idx] = 0.0
                    s -= 1
                    if s == n // 2:
                        break
    return h
