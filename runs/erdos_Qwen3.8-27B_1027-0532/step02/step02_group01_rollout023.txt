            k_end = -1
            delta_add[i, k_start + n - 1 : k_end + n - 1 + 1] = -h[i+k_start : i+k_end+1]

def binary_local_search(h_init, max_sweeps=1000, time_limit=60, start_time=0):
    """Best-improvement binary local search with exact correlation updates."""
    n = len(h_init)
    n_lags = 2 * n - 1
    h = h_init.astype(float).copy()
    corr = compute_corr_array(h)  # This already includes dx
    current_c5 = np.max(corr)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        # Precompute deltas
        delta_remove, delta_add = precompute_deltas(h)
        
        # Find all ones and zeros
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_delta = 0  # We want to minimize, so we want the most negative change
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                # New corr = corr + delta_remove[i] + delta_add[j]
                # New C5 = max(new_corr)
                # = max(max of unaffected, max of affected)
                # For simplicity, just compute:
                new_corr = corr + delta_remove[i] + delta_add[j]
                new_max = np.max(new_corr)
                delta = new_max - current_c5
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = compute_corr_array(h)
            current_c5 = np.max(corr)
        else:
            break  # No improving move found
    
    return h, float(current_c5)

    # Shape: (n_ones, n_zeros, n_lags)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def slsqp_optimize(h_init, eps=0.001, maxiter=300):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    
    def smooth_obj(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        x = (corr - cmax) / eps
        x -= np.max(x)
        return cmax + eps * np.log(np.sum(np.exp(x)))
    
    def smooth_grad(h):
        dx = 2.0 / n
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full') * dx
        cmax = np.max(corr)
        x = (corr - cmax) / eps
        x -= np.max(x)
        ex = np.exp(x)
        sexp = np.sum(ex)
        w = ex / sexp
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * dx * omh[:m]
                grad[:m] -= wk * dx * h[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:m]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 
             'jac': lambda h: np.ones(n)}]
    
    try:
        result = minimize(smooth_obj, h0, jac=smooth_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

        h_up_bin = make_binary(h_up, n1)

        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_ls

    remaining = budget_s - (time.time() - start_time)
    
    # Phase 6: Final polish
    if remaining > 20:
        h_final = best_h.astype(float)
        h_ls_final, c5_final = binary_ls(h_final, max_sweeps=10000, time_limit=min(remaining-10, 60), start_time=start_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_ls_final

    n_final = len(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
