    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    m = len(g)
    N = 2 * m + 1
    h = np.empty(N)
    h[:m] = g
    h[m] = 0.5
    h[m+1:] = 1.0 - g[::-1]
    return h

    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:

            grad[:N-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:N-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:N-kp])
            grad[:N-kp] -= w * h[kp:]

    h = antipal_h(g)
    return smooth_max(compute_correlations(h), alpha)

    grad_h = grad_corr(h, alpha)
    m = len(g)
    grad_g = grad_h[:m] - grad_h[m+1:][::-1]
    return grad_g

    g = (init[:m] + 1.0 - init[N-1:m:-1]) / 2.0

    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)

    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    N = 2 * m + 1
    h = np.empty(N)
    h[:m] = g
    h[m] = 0.5
    h[m+1:] = 1.0 - g[::-1]
    return h

    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    s = np.sum(ex)
    if s < 1e-300:
        p = np.zeros(len(corr))
        p[np.argmax(corr)] = 1.0

        p = ex / s

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def antipal_h(g):
    m = len(g)
    N = 2 * m + 1
    h = np.empty(N)
    h[:m] = g
    h[m] = 0.5
    h[m+1:] = 1.0 - g[::-1]
    return h

def grad_corr(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    s = np.sum(ex)
    if s < 1e-300:
        p = np.zeros(len(corr))
        p[int(np.argmax(corr))] = 1.0
    else:
        p = ex / s
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[:N-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:N-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:N-kp])
            grad[:N-kp] -= w * h[kp:]
    return grad

def ap_fun(g, alpha):
    h = antipal_h(g)
    return smooth_max(compute_correlations(h), alpha)

def ap_jac(g, alpha):
    h = antipal_h(g)
    grad_h = grad_corr(h, alpha)
    m = len(g)
    grad_g = grad_h[:m].copy()
    grad_g -= grad_h[m+1:][::-1]
    return grad_g

def project_to_anti(init):
    N = len(init)
    m = (N - 1) // 2
    g = (init[:m] + 1.0 - init[m+1:][::-1]) / 2.0
    g = np.clip(g, 0.0, 1.0)
    return g

def interp_anti(g_old, n_new):
    N_old = 2 * len(g_old) + 1
    h_old = antipal_h(g_old)
    if n_new % 2 == 0:
        n_new += 1
    x_old = np.linspace(0.0, 2.0, N_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    m_new = (n_new - 1) // 2
    g_new = (h_new[:m_new] + 1.0 - h_new[m_new+1:][::-1]) / 2.0
    g_new = np.clip(g_new, 0.0, 1.0)
    return g_new

def do_lbfgsb(g0, alpha, maxiter):
    m = len(g0)
    bounds = [(0.0, 1.0)] * m
    res = minimize(ap_fun, g0.copy(), args=(alpha,), jac=ap_jac,
                   bounds=bounds, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return np.clip(res.x, 0.0, 1.0)
    return g0.copy()

def smooth_perturb(g, rng, n_modes, base_amp):
    m = len(g)
    x = np.arange(m, dtype=float)
    p = g.copy()
    for _ in range(n_modes):
        freq = rng.uniform(0.5, 4.0)
        amp = base_amp * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        p += amp * np.sin(freq * np.pi * x / m + phase)
    return np.clip(p, 0.0, 1.0)

def random_anti_g(m, rng):
    g = rng.rand(m)
    return g

def block_perturb(g, rng, n_blocks, amp):
    m = len(g)
    p = g.copy()
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, m - 20))
        length = rng.randint(3, min(25, m - start))
        p[start:start+length] += amp * rng.choice([-1, 1])
    return np.clip(p, 0.0, 1.0)
