        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_ref = refine_lbfgsb(h_interp, n_try, target_try, [20, 100, 400], maxiter=2500)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            if tleft() > 50:
                h_ref2 = refine_slsqp(h_ref, n_try, target_try, [400, 1000], maxiter=2000)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2
        elif tleft() > 50:
            h_ref2 = refine_slsqp(h_interp, n_try, target_try, [100, 400], maxiter=1500)
            sc2 = compute_c5(h_ref2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2

    # Phase 4: Go back to best resolution and polish
    n_cur = len(best_h)
    target_cur = float(n_cur) / 2.0

    if tleft() > 40:
        h_ref = refine_slsqp(best_h, n_cur, target_cur, [500, 1500, 5000], maxiter=3000)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Final micro-perturbations
    if tleft() > 30:
        for noise in [0.005, 0.01, 0.015, 0.02]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(n_cur) * noise
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = refine_slsqp(h_pert, n_cur, target_cur, [1000, 3000], maxiter=2000)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Final feasibility and score
    h_final = project_feasible(best_h, float(len(best_h)) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
