def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = corr.max()
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    
    def constraint_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def full_lse_gradient(h, T):
    n = len(h)
    corr = compute_correlations(h)
    max_c = corr.max()
    exp_vals = np.exp(T * (corr - max_c))
    total = exp_vals.sum()
    weights = exp_vals / total
    grad = np.zeros(n)
    w0 = weights[n - 1]
    grad[:] += w0 * (1.0 - 2.0 * h)
    for l in range(1, n):
        wp = weights[n - 1 + l]
        wn = weights[n - 1 - l]
        if wp < 1e-18 and wn < 1e-18:
            continue
        if wp > 1e-18:
            grad[l:] += wp * (1.0 - h[:n - l])
            grad[:n - l] += -wp * h[l:]
        if wn > 1e-18:
            grad[:n - l] += wn * (1.0 - h[l:])
            grad[l:] += -wn * h[:n - l]
    return grad, corr

def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = corr.max()
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
