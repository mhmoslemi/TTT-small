import numpy as np

from scipy.signal import fftconvolve

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_value(h, eps, n):
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = eps * (corr - m)
    val = m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))
    return val * dx

def smooth_max_gradient(h, eps, n):
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    nc = 2 * n - 1
    for k in range(nc):
        w = weights[k]
        if w < 1e-15:
            continue
        l = k - (n - 1)
        idx1 = np.arange(n) - l
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = np.arange(n) + l
        valid2 = (idx2 >= 0) & (idx2 < n)
        contrib = np.zeros(n)
        contrib[valid1] += 1.0 - h[idx1[valid1]]
        contrib[valid2] += -h[idx2[valid2]]
        grad += w * contrib
    grad *= dx
    return grad

def proximal_gradient(h_start, n, eps, alpha0, max_iters, time_check, min_alpha_ratio=0.01, max_tries=50):
    h = h_start.copy()
    best_h = h.copy()

    alpha = alpha0

        if time_check():
            break
        g = smooth_max_gradient(h, eps, n)

        improved = False

            c5_new = evaluate_c5(h_new)

                best_h = h.copy()

                improved = True
                break
            alpha *= 0.6
            tried += 1
        if not improved:
            if alpha < alpha0 * min_alpha_ratio:
                alpha = alpha0 * 0.5
            continue
        alpha = min(alpha * 1.1, alpha0)
    return best_h, best_c5

def slsqp_refine(h_start, n, bounds, eps, maxiter, ftol):
    def obj(x):
        return smooth_max_value(x, eps, n)
    def jac(x):
        return smooth_max_gradient(x, eps, n)
    def con(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h_start.copy(), jac=jac, method='SLSQP',
                   bounds=bounds, constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def coordinate_descent(h_start, n, deltas, max_sweeps, time_check):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    for sweep in range(max_sweeps):

                if around:
                    break
            if around:

        if not around:

            if no_improve >= 3:

        else:
            no_improve = 0
    return h, best_c5

def upsample_h(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out
