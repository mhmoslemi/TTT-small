        # Find a pair (j, j+shift) that contributes significantly
        if shift >= 0:
            valid_j = np.arange(n - shift) if shift < n else np.array([], dtype=int)
        else:
            valid_j = np.arange(-shift, n)
        if len(valid_j) == 0:
            continue
        # Pick a random j from valid positions
        j = valid_j[rng.randint(len(valid_j))]
        l = j + shift
        if l < 0 or l >= n:
            continue
        # Try flipping or adjusting this pair
        h_try = h_cur.copy()
        h_try[j] = 1.0 - h_cur[j]
        h_try[l] = 1.0 - h_cur[l]
        h_try = project_sum(h_try, n / 2.0)
        corr_try = compute_correlations(h_try)
        if np.max(corr_try) < best_score - 1e-12:
            best_score = np.max(corr_try)
            best_h = h_try
            h_cur = h_try
            corr_cur = corr_try
            active_lags = np.where(corr_cur > best_score - 1e-10 * best_score)[0]
    return best_h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_sum(init, n / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    # Phase 1: Initial CD
    if time_left() > 120:
        h_cd = coordinate_descent(best_h, n_sweeps=2000, tol=1e-14,
                                  time_budget=min(90, time_left() - 60), n_bisect=55)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: SLSQP with increasing alpha
    if time_left() > 80:
        n_cur = len(best_h)
        bounds = [(0.0, 1.0)] * n_cur
        for alpha in [2000, 10000, 50000, 200000]:
            if time_left() < 20:
                break
            maxiter = max(200, 600000 // n_cur)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 3: CD after SLSQP
    if time_left() > 60:
        h_cd = coordinate_descent(best_h, n_sweeps=3000, tol=1e-14,
                                  time_budget=min(50, time_left() - 40), n_bisect=60)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 4: Targeted pair perturbations
    if time_left() > 40:
        corr_best = compute_correlations(best_h)
        h_targeted = targeted_perturb(best_h, corr_best, rng, nAttempts=200)
        h_cd = coordinate_descent(h_targeted, n_sweeps=1000, tol=1e-14,
                                  time_budget=min(30, time_left() - 20), n_bisect=55)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 5: Diversified random perturbations + CD
    n_cur = len(best_h)
    n_perturb = 200
    for i in range(n_perturb):
        if time_left() < 25:
            break
        frac = i / max(n_perturb - 1, 1)
        strategy = i % 6
        if strategy == 0:
            scale = 0.02 + 0.12 * frac
            h_pert = best_h + rng.randn(n_cur) * scale
        elif strategy == 1:
            h_pert = best_h.copy()
            n_flips = rng.randint(n_cur // 30, n_cur // 8)
            flip_pos = rng.choice(n_cur, size=n_flips, replace=False)
            h_pert[flip_pos] = 1.0 - h_pert[flip_pos]
        elif strategy == 2:
            h_pert = best_h.copy()
            k = rng.randint(3, max(4, n_cur // 5))
            s_pos = rng.randint(0, n_cur - k)
            h_pert[s_pos:s_pos + k] = best_h[s_pos:s_pos + k][::-1]
        elif strategy == 3:
            h_pert = best_h.copy()
            n_sparse = rng.randint(2, 8)
            pos = rng.choice(n_cur, size=n_sparse, replace=False)
            h_pert[pos] = np.clip(h_pert[pos] + rng.randn(n_sparse) * 0.4, 0, 1)
        elif strategy == 4:
            h_pert = best_h.copy()
            n_bin = rng.randint(3, n_cur // 6)
            pos = rng.choice(n_cur, size=n_bin, replace=False)
            h_pert[pos] = rng.choice([0.0, 1.0], size=n_bin)
        else:
            # Structured: perturb a block with a smooth pattern
            h_pert = best_h.copy()
            k = rng.randint(10, n_cur // 3)
            s_pos = rng.randint(0, n_cur - k)
            x = np.linspace(0, 2 * np.pi, k)
            amp = rng.uniform(0.2, 0.5)
            h_pert[s_pos:s_pos + k] = np.clip(
                best_h[s_pos:s_pos + k] + amp * np.sin(x + rng.uniform(0, 2 * np.pi)), 0, 1)

        h_pert = project_sum(h_pert, n_cur / 2.0)
        tb = max(0.5, min(4.0, (time_left() - 20) / max(n_perturb - i, 1) * 0.6))
        h_cd = coordinate_descent(h_pert, n_sweeps=400, time_budget=tb, n_bisect=45)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    # Phase 6: Higher resolution attempt
    base_n = len(best_h)
    for n_try in [int(base_n * 1.5), 2 * base_n]:
        if time_left() < 50:
            break
        if n_try > 2000:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)
        tb_cd = min(30, time_left() - 30)
        if tb_cd > 5:
            h_cd = coordinate_descent(h_interp, n_sweeps=800, time_budget=tb_cd, n_bisect=45)
            if time_left() > 20:
                bounds_try = [(0.0, 1.0)] * n_try
                res_x = slsqp_optimize(h_cd, 50000, max(100, 300000 // n_try), bounds_try)
                if res_x is not None:
                    h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s = evaluate_score(h_new)
                        if s < best_score:
                            best_score = s
                            best_h = h_new
                            h_cd = h_new
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 7: Final intensive CD
    if time_left() > 25:
        h_cd = coordinate_descent(best_h, n_sweeps=5000, tol=1e-14,
                                  time_budget=min(35, time_left() - 10), n_bisect=60)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 8: Final SLSQP
    if time_left() > 12:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [500000, 2000000]:
            if time_left() < 6:
                break
            res_x = slsqp_optimize(best_h, alpha, 300, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
