import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _project(y, S):
    """Project y onto {x : 0 <= x_i <= 1, sum(x) = S}"""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix numerical errors
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    """Gradient of corr[k_idx] with respect to h."""
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _softmax_gradient(h, alpha, n_lags=None):
    """Full softmax-weighted gradient over all lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    total = weights.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / total
    
    grad = np.zeros(n)
    # For efficiency, only compute top terms if too many
    if n_lags is not None and len(corr) > n_lags:
        top = np.argsort(weights)[::-1][:n_lags]
        w = weights[top]
        w = w / w.sum()
        for idx in range(len(top)):
            grad += w[idx] * _grad_corr_term(h, top[idx], dx)
    else:
        # Full sum is expensive, use top terms
        top = np.argsort(weights)[::-1][:min(50, len(corr))]
        w = weights[top]
        w = w / w.sum()
        for idx in range(len(top)):
            grad += w[idx] * _grad_corr_term(h, top[idx], dx)
    return grad

def _nesterov_softmax(h0, n, target_sum, n_iters, alpha0, alpha_end, n_top=20):
    """Nesterov with adaptive softmax gradient."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        # Adaptive alpha: start low, increase
        progress = it / max(n_iters - 1, 1)
        alpha = alpha0 + (alpha_end - alpha0) * progress
        
        # Compute gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = np.max(corr)
        weights = np.exp(alpha * (corr - m))
        total = weights.sum()
        if total < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total
        
        top = np.argsort(weights)[::-1][:n_top]
        w = weights[top]
        w = w / w.sum()
        
        g = np.zeros(n)
        for idx in range(len(top)):
            g += w[idx] * _grad_corr_term(y, top[idx], dx)
        
        # Nesterov step
        alpha_step = 1.0 / np.sqrt(it + 1)
        x_new = _project(y - alpha_step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent(h0, n, target_sum, max_rounds=30):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=10):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            # Try several values around the current
            candidates = [min(max(old + d, 0.0), 1.0) for d in [-0.5, -0.25, -0.1, 0.1, 0.25, 0.5]]
            candidates += [0.0, 1.0]
            for v in candidates:
                if v == old:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if abs(best_v - old) > 1e-10:
                improved = True
        if not improved:
            break
    return h

def _multigrid_refine(h_coarse, n_fine, target_sum_fine):
    """Upsample from coarse to fine grid and project."""
    n_coarse = len(h_coarse)
    h_fine = np.interp(np.linspace(0, n_coarse - 1, n_fine), np.arange(n_coarse), h_coarse)
    h_fine = np.clip(h_fine, 0.0, 1.0)
    return _fix_sum(h_fine, target_sum_fine)

def _random_starts(n, target_sum, rng):

    starts = []
    
    # Uniform random
    starts.append(_fix_sum(rng.rand(n), target_sum))

    # Block: 1 on a contiguous interval of length n/2

    h_blk = np.zeros(n)

    q = n // 4

    h_alt = np.zeros(n)

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _nesterov_softmax(h0, n, target_sum, n_iters, alpha0, alpha_end, n_top, step_scale):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)

    for it in range(n_iters):
        progress = it / max(n_iters - 1, 1)
        alpha = alpha0 + (alpha_end - alpha0) * progress

        corr = np.correlate(y, 1.0 - y, mode='full')
        m = np.max(corr)
        weights = np.exp(alpha * (corr - m))
        total = weights.sum()
        if total < 1e-300:
            top = np.array([np.argmax(corr)])
            w = np.array([1.0])
        else:
            weights = weights / total
            top = np.argsort(weights)[::-1][:n_top]
            w = weights[top].copy()
            w = w / w.sum()

        g = np.zeros(n)
        for idx in range(len(top)):
            g += w[idx] * _grad_corr_term(y, top[idx], dx)

        step = step_scale / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new

        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()

    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):

    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h0

def _coord_descent_binary(h0, n, target_sum, max_rounds=30):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=10):
    h = h0.copy()
    fixed_vals = [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]
    for _ in range(max_rounds):

        for j in range(n):
            old = h[j]

            best_v = old
            candidates = list(fixed_vals)

                    best_v = v
            h[j] = best_v
            if abs(best_v - old) > 1e-9:
                improved = True
        if not improved:
            break
    return h

def _active_set_lsqqp(h0, n, target_sum, tol_frac=0.02, maxiter=200):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full')

        sub = corr_x[_active_idx]
        sm = np.max(sub)
        if sm < -500:
            return sm * dx
        return (sm + np.log(np.sum(np.exp(alpha * (sub - sm)))) / alpha) * dx

    def _active_grad(x, alpha, _active_idx):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        sub = corr_x[_active_idx]
        sm = np.max(sub)
        w = np.exp(alpha * (sub - sm))
        tw = w.sum()
        if tw < 1e-300:

            w[0] = 1.0
