import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def smooth_obj_grad(h, alpha, mu, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    weights = exp_shifted / s
    fval = c_max + np.log(s) / alpha
    sum_h = np.sum(h)
    fval += mu * (sum_h - target_sum) ** 2
    A = weights
    A_rev = A[::-1].copy()
    one_minus_h = 1.0 - h
    conv1 = np.convolve(A_rev, one_minus_h, mode='full')
    Term1 = conv1[n - 1:2 * n - 1]
    conv2 = np.convolve(h, A, mode='full')
    Term2 = conv2[n - 1:2 * n - 1]
    grad = Term1 - Term2
    grad += 2.0 * mu * (sum_h - target_sum) * np.ones(n)
    return fval, grad


def lbfgsb_single(h_start, alpha, mu, target, maxiter):
    n = len(h_start)
    h_cur = project_feasible(h_start.copy(), target)
    n2 = n

    def obj_fn(x):
        v, g = smooth_obj_grad(x, alpha, mu, target)
        return v, g

    try:
        res = minimize(obj_fn, h_cur, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n2,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 10000})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur


def lbfgsb_refine(h_start, alphas, mu, target, maxiter_per_alpha):
    h_cur = project_feasible(h_start.copy(), target)
    for alpha in alphas:
        h_cur = lbfgsb_single(h_cur, alpha, mu, target, maxiter_per_alpha)
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_polish(h, target, max_steps=200, step_init=0.005, seed_val=0):
    rng_sg = np.random.RandomState(seed_val)
    h = project_feasible(h.copy(), target)
    n = len(h)
    step = step_init
    best_local = compute_c5(h)
    for it in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        active = np.where(corr >= c_max - 1e-12)[0]
        k = active[rng_sg.randint(len(active))]
        offset = k - n + 1
        sg = np.zeros(n)
        i_start1 = max(0, -offset)
        i_end1 = min(n, n - offset)
        for i in range(i_start1, i_end1):
            sg[i] += (1.0 - h[i + offset])
        i_start2 = max(0, offset)
        i_end2 = min(n, n)
        for i in range(i_start2, i_end2):
            if 0 <= i - offset < n:
                sg[i] += -h[i - offset]
        norm = np.linalg.norm(sg)
        if norm < 1e-15:
            break
        h_try = project_feasible(h - step * sg / norm, target)
        c_try = compute_c5(h_try)
        if c_try < best_local - 1e-15:
            h = h_try
            best_local = c_try
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.8
            if step < 1e-10:
                break
    return h


def diverse_starts(base_h, n, target, rng, n_starts=5):
    starts = []
    for i in range(n_starts):
        noise_scale = 0.03 + 0.04 * i
        h_pert = base_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        starts.append(h_pert)
    h_inv = project_feasible(1.0 - base_h, target)
    starts.append(h_inv)
    h_shifted = np.roll(base_h, n // 4)
    h_shifted = project_feasible(h_shifted, target)
    starts.append(h_shifted)
    return starts
