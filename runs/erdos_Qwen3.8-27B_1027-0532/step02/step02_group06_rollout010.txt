For SLSQP, I need to specify the objective and constraints as functions. The objective is:
f(h) = max_k corr_k(h) * dx

But this is not differentiable. I can use the softmax approximation:
f_α(h) = (1/α) * log(sum_k exp(α * corr_k(h) * dx))

This is smooth and its gradient is:
∇f_α(h) = sum_k p_k * ∇corr_k(h) * dx
where p_k = exp(α * corr_k * dx) / sum_j exp(α * corr_j * dx)

For large α, this approximates the gradient of the max (subgradient at the argmax).

subject to: corr_k(h) * dx ≤ t for all k in active_set
            sum(h) = n/2
            0 ≤ h ≤ 1

def compute_correlations(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def softmax_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    scores = corr * dx
    m = scores.max()
    shifted = alpha * (scores - m)
    exp_s = np.exp(shifted)
    log_sum = m + np.log(np.sum(exp_s)) / alpha
    
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    
    # L = 0
    w0 = w[n-1]
    if w0 > 1e-14:
        grad[:] += w0 * (1.0 - 2.0 * h) * dx
    
    # L > 0: process +L and -L together
    for L in range(1, n):
        wp = w[n-1+L]
        wn = w[n-1-L]
        wsum = wp + wn
        if wsum < 1e-14:
            continue
        
        # For j in [0, n-1-L]: contribution from +L is wp*(1-h[j+L]), from -L is -wn*h[j+L]

        return np.ones(n)
