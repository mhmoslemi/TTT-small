    idx0 = new_frac.astype(int) % old_n
    idx1 = (idx0 + 1) % old_n

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx)

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_shifted = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    # Lag 0
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    # Positive lags
    for L in range(1, n):
        wk = w[n-1+L] * dx
        if abs(wk) < 1e-15:
            continue
        grad[L:] += wk * (1.0 - h[:n-L])
        grad[:n-L] -= wk * h[L:]
    # Negative lags
    for M in range(1, n):
        wk = w[n-1-M] * dx
        if abs(wk) < 1e-15:
            continue
        grad[:n-M] += wk * (1.0 - h[M:])
        grad[M:] -= wk * h[:n-M]
    
    return grad

    h = np.clip(h.copy(), 0.0, 1.0)

    if abs(diff) < 1e-12:
        return h
    if diff > 0:

    if abs(diff) < 1e-12:

    if diff > 0:
        room = 1.0 - h  # how much room each coordinate has

    total_room = room.sum()
    if total_room < 1e-12:
        return _fix_sum(h, target_sum)  # fallback
    h = h + diff * room / total_room
    return np.clip(h, 0.0, 1.0)

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room > 1e-12:
            h = h + diff * room / total_room
            h = np.clip(h, 0.0, 1.0)
        else:
            order = np.argsort(1.0 - h)
            for i in order:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
    else:
        room = h.copy()
        total_room = room.sum()
        if total_room > 1e-12:
            h = h + diff * room / total_room
            h = np.clip(h, 0.0, 1.0)
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= -1e-12:
                    break
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                if diff2 <= 1e-12:

                h[i] += add

        else:
            order = np.argsort(h)
            for i in order:
                if diff2 >= -1e-12:
                    break
                sub = min(-diff2, h[i])
                h[i] -= sub
                diff2 += sub
    return h

def _upsample_linear(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    new_frac = np.arange(new_n) * (old_n / float(new_n))
    idx0 = new_frac.astype(int) % old_n
    idx1 = (idx0 + 1) % old_n
    frac = new_frac - new_frac.astype(int)
    return h[idx0] * (1.0 - frac) + h[idx1] * frac

def _coord_descent(h, target_sum, n_passes, deadline, grid_n, rng, verbose=False):
    n = len(h)
    h = h.copy()
    h = _fix_sum(h, target_sum)
    best_c5 = _c5_of(h)
    best_h = h.copy()
    grid = np.linspace(0.0, 1.0, grid_n)
    
    for p in range(n_passes):
        if time.time() > deadline - 5:
            break
        order = rng.permutation(n)
        any_change = False
        for i in order:
            old_val = h[i]
            curr_c5 = _c5_of(h)
            best_trial = old_val
            best_local = curr_c5
            for cv in grid:
                if abs(cv - old_val) < 1e-8:
                    continue
                h[i] = cv
                c5_try = _c5_of(h)
                if c5_try < best_local - 1e-12:
                    best_local = c5_try
                    best_trial = cv
            if best_trial != old_val:
                h[i] = best_trial
                any_change = True
            else:
                h[i] = old_val
        
        h = _fix_sum(h, target_sum)
        c5_curr = _c5_of(h)
        if c5_curr < best_c5 - 1e-10:
            best_c5 = c5_curr
            best_h = h.copy()
        elif not any_change:
            break
    
    c5_final = _c5_of(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_smooth(h_init, n, alpha, maxiter, ftol, deadline):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        smax = m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
        penalty = 1000.0 * (x.sum() - target_sum) ** 2
        return float(smax * dx + penalty)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exp_s = np.exp(alpha * (corr - m))
        w = exp_s / np.sum(exp_s)
        grad = np.zeros(n)
        grad += w[n-1] * dx * (1.0 - 2.0 * x)
        for L in range(1, n):
            wk = w[n-1+L] * dx
            if abs(wk) < 1e-14:
                continue
            grad[L:] += wk * (1.0 - x[:n-L])
            grad[:n-L] -= wk * x[L:]
        for M in range(1, n):
            wk = w[n-1-M] * dx
            if abs(wk) < 1e-14:
                continue
            grad[:n-M] += wk * (1.0 - x[M:])
            grad[M:] -= wk * x[:n-M]
        grad += 2000.0 * (x.sum() - target_sum)
        return grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return res.x
    except Exception:
        try:
            res = minimize(obj, h_init, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
            return res.x
        except Exception:
            return h_init.copy()

def _run_slsqp_phase(h_init, n, alpha_list, maxiter_list, deadline):
