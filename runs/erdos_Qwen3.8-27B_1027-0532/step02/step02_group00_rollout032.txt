def active_set_descent(h, target_sum, max_iter=2000, max_active=10,

def smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def cons_fun(h):
    return np.sum(h) - _g_target


def cons_jac(h):
    return np.ones(_g_n)


def slsqp_refine(h, target_sum, alphas, maxiter=500, ftol=1e-14):
    global _g_n, _g_target, _g_alpha
    n = len(h)
    _g_n = n
    _g_target = target_sum
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h.copy()
    best_c = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def penalty_obj(x, target_sum, pen):

    c_max = np.max(corr)
    s = np.sum(x) - target_sum
    return float(c_max * (2.0 / len(x)) + pen * s * s)

            if c_new < best_c:

        h_hi = interpolate_to(best_h, n_hi, target_hi)
        h_try, c_try = active_set_descent(h_hi, target_hi, max_iter=2000,
                                           max_active=10, alpha0=0.05)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            h_hi = best_h.copy()
        if tleft() > 80:
            h_ref, c_ref = slsqp_refine(h_hi, target_hi, [100, 300], 800, 1e-14)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref

    # Phase 6: Final polish with very tight active set
    n_final = len(best_h)
    target_final = n_final / 2.0
    if tleft() > 30:
        h_try, c_try = active_set_descent(best_h.copy(), target_final,
                                           max_iter=5000, max_active=3,
                                           alpha0=0.01, epsilon_rel=0.003)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    if tleft() > 20:
        h_try, c_try = active_set_descent(best_h.copy(), target_final,
                                           max_iter=3000, max_active=7,
                                           alpha0=0.005, epsilon_rel=0.005)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

    h_final = project_feasible(best_h, target_final)
    final_c = compute_c5(h_final)
    return h_final, final_c, len(h_final)
