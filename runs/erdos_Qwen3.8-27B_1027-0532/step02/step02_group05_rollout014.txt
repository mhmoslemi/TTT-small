- From the term where i+k=m, i.e., i=m-k (if 0≤m-k≤N-1-k, i.e., k≤m≤N-1): 
  ∂/∂h[m] [h[m-k]*(1-h[m])] = h[m-k] * (-1) = -h[m-k]

So for k > 0:
∂f/∂h[m] = (1-h[m+k]) * 1[m≤N-1-k] + (-h[m-k]) * 1[k≤m≤N-1]

For k = 0:
f = sum_i h[i]*(1-h[i])
∂f/∂h[m] = (1-h[m]) - h[m] = 1 - 2h[m]

For k < 0, let k' = -k:
f = sum_{i=k'}^{N-1} h[i]*(1-h[i-k'])
∂f/∂h[m]:
- From i=m (if m ≥ k'): (1-h[m-k'])
- From i-k'=m, i=m+k' (if m+k'≤N-1): -h[m+k']

So:
∂f/∂h[m] = (1-h[m-k']) * 1[m≥k'] + (-h[m+k']) * 1[m≤N-1-k']

Now I can vectorize this. Let me define for each k:
- mask1[m] = 1 if m is valid as the first index
- mask2[m] = 1 if m is valid as the second index
- grad_k[m] = w * (value from mask1 - value from mask2)

For k > 0:
- mask1 = range(N-k) (indices 0 to N-1-k)
- mask2 = range(k, N) (indices k to N-1)
- grad_k[0:N-k] += w * (1 - h[k:N])
- grad_k[k:N] -= w * h[0:N-k]

For k < 0 (k' = -k):
- mask1 = range(k', N) (indices k' to N-1)  
- mask2 = range(0, N-k') (indices 0 to N-1-k')
- grad_k[k':N] += w * (1 - h[0:N-k'])
- grad_k[0:N-k'] -= w * h[k':N]

For k = 0:
- grad_0 = w * (1 - 2*h)

This is O(N) per lag, so O(N²) total. For N=425, that's about 180K operations per gradient evaluation. With scipy minimize doing maybe 1000 evaluations, that's 180M ops. Should be fine within the time budget.

    dx = 2.0 / n

    return np.max(corr) * dx

            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def grad_smooth_max(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)

        w = p[j]
        if w < 1e-15:

        elif k > 0:

            grad[k:] -= w * h[:N-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:N-kp])
            grad[:N-kp] -= w * h[kp:]
    return grad

def obj_smooth(x, alpha):
    return smooth_max(compute_correlations(x), alpha)

def jac_smooth(x, alpha):

def coord_descent(h, target_sum, max_sweeps=50, n_trials=21):
    """Coordinate descent: for each coordinate, try multiple values and pick best."""
    N = len(h)
    h = h.copy()
    best_score = evaluate_score(h)
    grid = np.linspace(0.0, 1.0, n_trials)
    for sweep in range(max_sweeps):
        improved = False
        # Random order
        order = np.random.permutation(N)
        for m in order:
            old_val = h[m]
            best_trial = old_val
            best_trial_score = best_score
            for t in grid:
                if abs(t - old_val) < 1e-10:
                    continue
                h[m] = t
                s = evaluate_score(h)
                if s < best_trial_score - 1e-12:
                    best_trial_score = s
                    best_trial = t
            h[m] = best_trial
            if best_trial != old_val:
                improved = True
        h = project_feasible(h, target_sum)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
        if not improved:
            break
    return h, best_score

def compute_correlations(h):

def project_feasible(h, target_sum):

        if float(np.sum(proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))

            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def smooth_max(x, alpha):
    m = float(np.max(x))
    return m + float(np.log(np.sum(np.exp(alpha * (x - m))))) / alpha

def _compute_influence(h, m):
    N = len(h)
    influence = np.zeros(2 * N - 1)
    for k in range(1, N):
        if m + k < N:
            influence[N - 1 + k] += (1.0 - h[m + k])
        if m >= k:
            influence[N - 1 + k] -= h[m - k]
        if m >= k:
            influence[N - 1 - k] += (1.0 - h[m - k])
        if m + k < N:
            influence[N - 1 - k] -= h[m + k]
    return influence
