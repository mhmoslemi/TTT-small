def slsqp_refine(h, alpha, maxiter=300):

    target_sum = n / 2.0

    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def precompute_indices(n):
    js = np.arange(2 * n - 1)
    idx1 = np.zeros((n, 2 * n - 1), dtype=np.int32)
    idx2 = np.zeros((n, 2 * n - 1), dtype=np.int32)
    mask1 = np.zeros((n, 2 * n - 1), dtype=bool)
    mask2 = np.zeros((n, 2 * n - 1), dtype=bool)
    for pos in range(n):
        idx1[pos] = pos - js + n - 1
        idx2[pos] = pos + js - n + 1
        mask1[pos] = (idx1[pos] >= 0) & (idx1[pos] < n)
        mask2[pos] = (idx2[pos] >= 0) & (idx2[pos] < n)
    return idx1, idx2, mask1, mask2


def find_optimal_t(corr, b, lo_t, hi_t):
    for _ in range(45):
        mid = (lo_t + hi_t) / 2.0
        vals = corr + mid * b
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        if np.all(b[active] >= 0):
            hi_t = mid
        elif np.all(b[active] <= 0):
            lo_t = mid
    return (lo_t + hi_t) / 2.0


def cd_sweep_fast(h, corr, order, idx1, idx2, mask1, mask2):

    for pos in order:
        old = h[pos]

        b = np.zeros(m)

        b[m2] -= h[idx2[pos][m2]]
        opt_t = find_optimal_t(corr, b, lo_t, hi_t)
        new_val = old + opt_t
        if new_val < 0.0:
            new_val = 0.0
        elif new_val > 1.0:
            new_val = 1.0
        delta = new_val - old
        if abs(delta) > 1e-15:
            corr = corr + delta * b
            h[pos] = new_val
    return h, corr

        order = rng.permutation(n)

    return h, score

def subgradient_step(h, alpha, n_top, step, t0, time_limit):
    n = len(h)
    target_sum = n / 2.0
    corr = compute_correlations(h)
    top_idx = np.argsort(-corr)[:n_top]
    top_vals = corr[top_idx]
    shifted = top_vals - top_vals[0]
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()

    grad = np.zeros(n)
    for w, j in zip(weights, top_idx):
        lag = int(j - (n - 1))
        if lag == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]

    h_new = h - step * grad
    h_new = project_to_feasible(h_new, target_sum)
    return h_new
