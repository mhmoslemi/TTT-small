    best_c5 = _c5_of(h_init)
    n_center = n - 1
    alpha_schedule = [3, 8, 15, 30, 60, 120, 250]
    for alpha in alpha_schedule:
        if time.time() > deadline - 10:
            break
        n_iter = max(80, min(600, 40000 // (alpha + 10)))
        lr_base = 0.8 / np.sqrt(alpha + 1.0)
        h = best_h.copy() if alpha > 8 else h_init.copy()
        v = np.zeros(n)
        momentum = 0.82
        prev_obj = _c5_of(h)
        stuck = 0
        for t in range(n_iter):
            if time.time() > deadline - 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            shifted = alpha * (corr - m)
            shifted = np.minimum(shifted, 0.0)
            exp_v = np.exp(shifted)
            total_exp = exp_v.sum()
            if total_exp < 1e-300:
                break
            weights = exp_v / total_exp
            grad = np.zeros(n)
            for k in range(len(corr)):
                if weights[k] > 1e-7:
                    lag = k - n_center
                    grad += weights[k] * _grad_for_lag(h, lag)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                grad /= gnorm
            lr_t = lr_base / (1.0 + 0.005 * t)
            v = momentum * v - lr_t * grad
            step = v
            h_try = _fix_sum(h + step, target_sum)
            c5_try = _c5_of(h_try)
            if c5_try < prev_obj - 1e-12:
                h = h_try
                prev_obj = c5_try
                stuck = 0
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
            else:
                h_try2 = _fix_sum(h + 0.3 * step, target_sum)
                c5_try2 = _c5_of(h_try2)
                if c5_try2 < prev_obj - 1e-12:
                    h = h_try2
                    prev_obj = c5_try2
                    stuck = 0
                    if c5_try2 < best_c5:
                        best_c5 = c5_try2
                        best_h = h.copy()
                else:
                    stuck += 1
                    v *= 0.5
            if stuck > 40:
                break
    return best_h, best_c5

def _coordinate_descent_minimax(h_init, deadline, max_iter=2000):
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    n_center = n - 1
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        worst_lag = max_idx - n_center
        grad_worst = _grad_for_lag(h, worst_lag)
        top_k_lags = np.argsort(corr)[-3:]
        grad_top = np.zeros(n)
        for ki in top_k_lags:
            lag = ki - n_center
            grad_top += _grad_for_lag(h, lag)
        grad_top /= len(top_k_lags)
        best_improve = 1e-10
        best_j = -1
        best_dir = 1.0
        best_delta = 0.0
        step_candidates = [0.05, 0.1, 0.15, 0.2]
        for j in range(n):
            if h[j] <= 0.001 and best_dir == 1.0:
                continue
            if h[j] >= 0.999 and best_dir == -1.0:
                continue
            for delta in step_candidates:
                for direction in [1.0, -1.0]:
                    if direction == 1.0 and h[j] + delta > 1.0:
                        continue
                    if direction == -1.0 and h[j] - delta < 0.0:
                        continue
                    eff = -direction * (0.7 * grad_worst[j] + 0.3 * grad_top[j])
                    if eff > best_improve:
                        best_improve = eff
                        best_j = j
                        best_dir = direction
                        best_delta = delta
        if best_j < 0:
            break
        partner_pool = np.delete(np.arange(n), best_j)
        partner_score = np.zeros(n - 1)
        for pi in range(n - 1):
            j2 = partner_pool[pi]
            if best_dir == 1.0:
                if h[j2] < 0.01:
                    partner_score[pi] = -1e6
                else:
                    partner_score[pi] = 0.7 * grad_worst[j2] + 0.3 * grad_top[j2]
            else:
                if h[j2] > 0.99:
                    partner_score[pi] = -1e6
                else:
                    partner_score[pi] = -(0.7 * grad_worst[j2] + 0.3 * grad_top[j2])
        best_partner_idx = np.argmax(partner_score)
        j2 = partner_pool[best_partner_idx]
        h[best_j] = np.clip(h[best_j] + best_dir * best_delta, 0.0, 1.0)
        h[j2] = np.clip(h[j2] - best_dir * best_delta, 0.0, 1.0)
        h = _fix_sum(h, target_sum)
        c5_now = _c5_of(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def _slsqp_polish(h_init, deadline, alpha=300.0, maxiter=300):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.minimum(shifted, 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    try:
        res = minimize(obj, h_init, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.fun < obj(h_init):
            return res.x
    except Exception:
        pass
    return h_init.copy()

def _interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, np.concatenate([h_src, [h_src[0]]]), period=2.0)
    return h_dst

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20
    best_h_global = None
    best_c5_global = np.inf
    best_n_global = 0

    n_primary = 100
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init_100 = np.array(initial_h_values, dtype=float).copy()
        if len(h_init_100) != n_primary:
            h_init_100 = _interpolate_to_n(h_init_100, n_primary)
        h_init_100 = _fix_sum(h_init_100, n_primary / 2.0)
    else:
        h_init_100 = np.full(n_primary, 0.5)

    # Phase 1: Adaptive gradient from initial
    h1, c5_1 = _adaptive_gradient(h_init_100, deadline, seed=seed)
    if c5_1 < best_c5_global:
        best_c5_global = c5_1
        best_h_global = h1.copy()
        best_n_global = n_primary

    # Phase 2: Coordinate descent refinement
    if time.time() < deadline - 30:
        h2, c5_2 = _coordinate_descent_minimax(h1, deadline - 60, max_iter=3000)
        if c5_2 < best_c5_global:
            best_c5_global = c5_2
            best_h_global = h2.copy()
            best_n_global = n_primary

    # Phase 3: SLSQP polish
    if time.time() < deadline - 25:
        h3 = _slsqp_polish(best_h_global, deadline, alpha=400.0, maxiter=200)
        h3 = _fix_sum(h3, best_n_global / 2.0)
        c5_3 = _c5_of(h3)
        if c5_3 < best_c5_global:
            best_c5_global = c5_3
            best_h_global = h3.copy()

    # Phase 4: Multiple restarts
    rng = np.random.RandomState(seed + 777)
    n_restarts = 5
    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        scale = 0.04 + 0.03 * r
        h_pert = _fix_sum(best_h_global + scale * rng.randn(best_n_global), best_n_global / 2.0)
        hr, cr = _adaptive_gradient(h_pert, deadline - 30, seed=seed + 1000 + r)
        if cr < best_c5_global:
            best_c5_global = cr
            best_h_global = hr.copy()
        if time.time() < deadline - 20:
            hcd, ccd = _coordinate_descent_minimax(hr, deadline - 15, max_iter=1000)
            if ccd < best_c5_global:
                best_c5_global = ccd
                best_h_global = hcd.copy()

    # Phase 5: Try n=120
    if time.time() < deadline - 60:
        n_alt = 120
        h_alt = _interpolate_to_n(best_h_global, n_alt)
        h_alt = _fix_sum(h_alt, n_alt / 2.0)
        h_alt_opt, c5_alt = _adaptive_gradient(h_alt, deadline - 30, seed=seed + 500)
        if c5_alt < best_c5_global:
            best_c5_global = c5_alt
            best_h_global = h_alt_opt.copy()
            best_n_global = n_alt
        if time.time() < deadline - 20:
            hcd_alt, ccd_alt = _coordinate_descent_minimax(h_alt_opt, deadline - 10, max_iter=1000)
            if ccd_alt < best_c5_global:
                best_c5_global = ccd_alt
                best_h_global = hcd_alt.copy()
                best_n_global = n_alt

    # Phase 6: Final polish
    if time.time() < deadline - 15:
        h_final = _slsqp_polish(best_h_global, deadline, alpha=500.0, maxiter=200)
        h_final = _fix_sum(h_final, best_n_global / 2.0)
        c5_final = _c5_of(h_final)
        if c5_final < best_c5_global:
            best_c5_global = c5_final
            best_h_global = h_final.copy()

    # Phase 7: One more coordinate descent pass
    if time.time() < deadline - 10:
        h_cd_final, c5_cd_final = _coordinate_descent_minimax(best_h_global, deadline, max_iter=2000)
        if c5_cd_final < best_c5_global:
            best_c5_global = c5_cd_final
            best_h_global = h_cd_final.copy()

    # Final fix
    best_h_global = _fix_sum(best_h_global, best_n_global / 2.0)
    best_c5_global = _c5_of(best_h_global)
    return (best_h_global, best_c5_global, best_n_global)
