        if np.sum(x) > target_sum:

        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_gradient(h, alpha):

    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    weights = np.exp(scaled - max_s)
    wsum = np.sum(weights)
    if wsum < 1e-15:

    weights = (weights / wsum) * dx
    
    grad = np.zeros(n)
    # Lag 0
    w0 = weights[n-1]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    
    # Non-zero lags

            continue
        ak = abs(idx - (n-1))
        grad[ak:] += w * (1.0 - h[:n-ak])
        grad[:n-ak] += w * (-h[ak:])
    
    return grad

```python
def binary_local_search(h_start, max_iters, t0, time_limit, seed):
    n = len(h_start)
    rng = np.random.RandomState(seed)
    
    # Start from thresholded version
    h = (h_start > 0.5).astype(int)
    ones_set = set(np.where(h == 1)[0])
    zeros_set = set(np.where(h == 0)[0])
    
    # Adjust to have exactly n//2 ones
    while len(ones_set) > n // 2:
        p = rng.choice(list(ones_set))
        ones_set.remove(p)
        zeros_set.add(p)
        h[p] = 0
    while len(ones_set) < n // 2:
        q = rng.choice(list(zeros_set))
        ones_set.add(q)
        zeros_set.remove(q)
        h[q] = 1
    
    best_h = h.copy()
    best_score = evaluate_score(h)
    current_score = best_score
    
    # Precompute for fast evaluation
    # We'll just recompute the score after each accepted move
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        if len(ones_set) == 0 or len(zeros_set) == 0:
            break
        
        p = rng.choice(list(ones_set))
        q = rng.choice(list(zeros_set))
        if p == q:
            continue
        
        # Try the swap
        h[p] = 0
        h[q] = 1
        score = evaluate_score(h)
        
        if score < current_score:
            # Accept
            ones_set.remove(p)
            zeros_set.add(p)
            zeros_set.remove(q)
            ones_set.add(q)
            current_score = score
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            # Reject (or accept with some probability for SA)
            h[p] = 1
            h[q] = 0
    
    return best_h, best_score
```

The issue with this is that `evaluate_score` is O(N^2) per call (due to np.correlate). For N=425, that's about 180K operations, which in numpy takes maybe 50-100μs. So we can do about 10K-20K evaluations per second. With a 100-second budget for this phase, that's 1-2 million evaluations. That should be enough for a decent local search.

But the `list(ones_set)` conversion is O(N) each time, and `rng.choice` on a list is also somewhat expensive. Let me optimize by using arrays instead:

```python
def binary_local_search(h_start, max_iters, t0, time_limit, seed):
    n = len(h_start)
    rng = np.random.RandomState(seed)
    
    # Start from thresholded version, ensure exactly n//2 ones
    h = (h_start > 0.5).astype(np.int8)
    # Force to have exactly n//2 ones
    diff = int(np.sum(h)) - n // 2
    if diff > 0:
        # Remove 'diff' ones (choose random ones to set to 0)
        ones_idx = np.where(h == 1)[0]
        rng.shuffle(ones_idx)
        h[ones_idx[:diff]] = 0
    elif diff < 0:
        zeros_idx = np.where(h == 0)[0]
        rng.shuffle(zeros_idx)
        h[zeros_idx[:(-diff)]] = 1
    
    best_h = h.copy().astype(float)
    best_score = evaluate_score(best_h)
    current_score = best_score
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        p = ones_idx[rng.randint(len(ones_idx))]
        q = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[p] = 0
        h[q] = 1
        score = evaluate_score(h.astype(float))
        
        if score < current_score:
            current_score = score
            if score < best_score:
                best_score = score
                best_h = h.copy().astype(float)
        else:
            h[p] = 1
            h[q] = 0
    
    return best_h, best_score
```

This is simpler but still O(N) per iteration for finding ones and zeros. That's fine.

Actually, I realize I can make this much faster by using the incremental update formula. Instead of recomputing the full correlation, I can compute the change in the maximum correlation directly.

But that's complex. Let me just use the full recomputation since N=425 is small enough.

Let me estimate the time: np.correlate on two arrays of length 425 takes about 20-50μs (it's implemented in C). evaluate_score adds a max and multiply, so maybe 30-60μs total. With 100s budget, that's about 2-3 million iterations. Each iteration also involves numpy operations for indexing (np.where, rng.randint), which add maybe another 20-50μs. So realistically about 1-2 million iterations.

That should be more than enough for a good local search on a binary space of this size.

Now let me put together the final implementation. I'll combine:
1. Smoothed gradient descent (main optimization)
2. Binary local search (exploration)
3. Final refinement

Let me also add a "hybrid" approach: take the best binary solution, and use it as a starting point for the continuous optimization. This might find solutions that are "near-binary" which could be good.

Here's my final code:

```python
import numpy as np
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_gradient(h, alpha, threshold=1e-10):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    weights = np.exp(scaled - max_s)
    wsum = np.sum(weights)
    if wsum < 1e-15:
        return np.zeros(n)
    weights = (weights / wsum) * dx
    grad = np.zeros(n)
    active = np.where(weights > threshold)[0]
    for idx in active:
        w = weights[idx]
        if idx == n - 1:
            grad += w * (1.0 - 2.0 * h)
        else:
            ak = abs(idx - (n - 1))
            grad[ak:] += w * (1.0 - h[:n-ak])
            grad[:n-ak] += w * (-h[ak:])
    return grad

def optimize_smoothed(h0, alpha, max_iters, step_init, t0, time_limit, seed):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_score = evaluate_score(h)
    step = step_init
    rng = np.random.RandomState(seed)
    stall = 0
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        grad = smooth_gradient(h, alpha)
        grad_norm = np.linalg.norm(grad)
        if
