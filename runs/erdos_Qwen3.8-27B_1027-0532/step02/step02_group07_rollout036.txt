            m_end = n + q if q < 0 else n

    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_value(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    g = 1.0 - h
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        q = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        # Term 1: (1-h[m+q]) for valid m
        lo1 = max(0, -q)
        hi1 = min(n - 1, n - 1 - q)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * g[lo1+q:hi1+1+q]
        # Term 2: -h[m-q] for valid m  
        lo2 = max(0, q)
        hi2 = min(n - 1, n - 1 + q)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * h[lo2-q:hi2+1-q]
    
    return (1.0 / alpha) * grad

def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            h_j = h[j]
            m = 2 * n - 1
            p = np.arange(m)
            
            idx1 = j + p - (n - 1)
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = j - p + (n - 1)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            slopes = np.zeros(m)
            if np.any(valid1):
                slopes[valid1] += (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                slopes[valid2] -= h[idx2[valid2]]
            
            a = corr - h_j * slopes
            
            zero_mask = (np.abs(slopes) < 1e-14)
            pos_mask = slopes > 1e-14
            neg_mask = slopes < -1e-14
            
            lo_V = float(np.max(np.minimum(a, a + slopes)))
            hi_V = float(np.max(corr))
            
            if lo_V >= hi_V - 1e-14:
                continue
            
            for _ in range(40):
                mid_V = (lo_V + hi_V) * 0.5
                feasible = True
                if np.any(zero_mask):
                    if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                        feasible = False
                if feasible:
                    lo_t = 0.0
                    if np.any(neg_mask):
                        lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
                    hi_t = 1.0
                    if np.any(pos_mask):
                        hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
                    if lo_t > hi_t + 1e-14:
                        feasible = False
                
                if feasible:
                    hi_V = mid_V
                else:
                    lo_V = mid_V
            
            V_star = (lo_V + hi_V) * 0.5
            lo_t = 0.0
            hi_t = 1.0
            if np.any(neg_mask):
                lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
            if np.any(pos_mask):
                hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
            
            lo_t = max(lo_t, 0.0)
            hi_t = min(hi_t, 1.0)
            if lo_t > hi_t:
                t_star = h_j
            else:
                t_star = (lo_t + hi_t) * 0.5
            
            t_star = float(np.clip(t_star, 0.0, 1.0))
            change = t_star - h_j
            if abs(change) > 1e-13:
                h[j] = t_star
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    
    h = project_sum(h, n / 2.0)
    return h

def adam_optimize(h0, alpha, lr, n_steps, time_budget):
    n = len(h0)
    h = h0.copy()
    target = n / 2.0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t0 = time.time()
    
    for t in range(1, n_steps + 1):
        if time.time() - t0 > time_budget:
            break
        
        grad = smooth_max_grad(h, alpha)
        # Project gradient to sum-zero subspace (tangent to constraint)
        grad -= np.mean(grad)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        
        # Check if we improved
        s_new = evaluate_score(h_new)
        s_old = evaluate_score(h)
        if s_new <= s_old + 1e-10:
            h = h_new
        else:
            # Rejected, halve learning rate
            lr *= 0.5
    
    return h
