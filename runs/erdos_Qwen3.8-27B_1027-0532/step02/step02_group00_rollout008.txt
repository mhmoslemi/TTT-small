```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5_detail(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n, mapped to [0,1]"""
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = (tm[i >> 1] + (i & 1)) % 2
    return tm

def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def interpolation(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
                g[lag:] -= x[:n-lag]
        else:
            if -lag < n:
                g[-lag:] -= x[:n+lag]
            if n + lag > 0:
                g[:n+lag] -= x[-lag:]
        grad += weights[idx] * g * dx_val
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_step(h_cur, n, target, alpha, active_lags, maxiter, ftol):
    global _g_n, _g_target, _g_active, _g_alpha
    _g_n = n
    _g_target = target
    _g_active = active_lags
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h_cur, jac=_jac,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur.copy()

def get_active_lags(h, top_frac=0.1, tol_frac=0.05):
    c = corr_vec(h)
    c_max = np.max(c)
    threshold = c_max * (1.0 - tol_frac)
    idx = np.where(c >= threshold)[0]
    n_top = max(1, int(len(c) * top_frac))
    top_idx = np.argsort(c)[-n_top:]
    return np.union1d(idx, top_idx)

def pair_optimize(h, j, k, n, target, n_iter=60):
    """Optimize h[j] and h[k] maintaining h[j]+h[k]=s"""
    s = h[j] + h[k]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h
    
    h_work = h.copy()
    curr = h_work[j]
    
    def f_1d(t):
        h_work[j] = t
        h_work[k] = s - t
        return compute_c5(h_work)
    
    # Golden section search
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    invphi2 = invphi * invphi
    a, b = lo, hi
    c_pt = b - invphi * (b - a)
    d_pt = a + invphi * (b - a)
    fc = f_1d(c_pt)
    fd = f_1d(d_pt)
    
    for _ in range(n_iter):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - invphi * (b - a)
            fc = f_1d(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + invphi * (b - a)
            fd = f_1d(d_pt)
    
    t_opt = (a + b) / 2.0
    # Check if this is actually better than current
    f_opt = f_1d(t_opt)
    f_curr = f_1d(curr)
    if f_opt < f_curr:
        h_work[j] = t_opt
        h_work[k] = s - t_opt
        # Verify constraint
        h_work = project_feasible(h_work, target)
    return h_work
```
