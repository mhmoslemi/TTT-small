        # Shift left

        state[0] = feedback
    return seq

    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def to_binary_from_fractional(h, n):
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def make_thue_morse_binary(n, seed):
    rng = np.random.RandomState(seed)
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    target = n // 2
    num_ones = int(tm.sum())
    if num_ones > target:
        one_pos = np.where(tm == 1)[0]
        to_remove = rng.choice(one_pos, num_ones - target, replace=False)
        tm[to_remove] = 0
    elif num_ones < target:
        zero_pos = np.where(tm == 0)[0]
        to_add = rng.choice(zero_pos, target - num_ones, replace=False)
        tm[to_add] = 1
    return tm

def slsqp_refine(h_start, n, eps, maxiter):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1 - x, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        exp_s = np.exp(shifted)
        sum_exp = np.sum(exp_s)
        weights = exp_s / sum_exp
        grad = np.zeros(n)
        n_corr = 2 * n - 1
        for k in range(n_corr):
            l = k - (n - 1)
            w = weights[k]
            if w < 1e-14:
                continue
            for jj in range(n):
                jml = jj - l
                if 0 <= jml < n:
                    grad[jj] += w * (1.0 - x[jml])
                jpl = jj + l
                if 0 <= jpl < n:
                    grad[jj] += w * (-x[jpl])
        grad *= dx
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h_start, evaluate_c5(h_start)

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out
