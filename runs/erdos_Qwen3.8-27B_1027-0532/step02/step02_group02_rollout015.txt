import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return grad

def _project_onto_set(z, target_sum):
    n = len(z)
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(z - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    result = np.clip(z - tau, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def _subgrad_optimize(h_init, n_iter, lr0, lr_decay, alpha0, alpha_growth, deadline, check_every=200):

    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 30
    
    for t in range(n_iter):
        if time.time() > deadline - 10:
            break
        alpha = alpha0 * (1.0 + t * alpha_growth)
        corr = _corr_full(h)
        max_c = corr.max()
        shifted = alpha * (corr - max_c)
        shifted = np.clip(shifted, -50.0, 0.0)

        n_lags = 2 * n - 1

            if w[k] < 1e-8:

            lag = k - (n - 1)
            grad += w[k] * _grad_lag(h, lag)
        
        lr = lr0 / (1.0 + t * lr_decay)

    return best_h, best_c5

def _slsqp_refine(h_init, alpha, maxiter=300, ftol=1e-14):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        if m > 0:
            log_sum = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
            return log_sum * dx
        return 0.0
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -50.0, 0.0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        grad = np.zeros(n)
        n_lags = 2 * n - 1
        for k in range(n_lags):
            if w[k] < 1e-8:
                continue
            lag = k - (n - 1)
            grad += w[k] * _grad_lag(x, lag)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol},
                       jac=obj_grad)
        return res.x
    except Exception:
        return h_init.copy()

def _upsample(h, n_new):
    n_old = len(h)

    return h_new

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return grad

def _project_onto_set(z, target_sum):
    n = len(z)
    lo = float(z.min()) - 2.0
    hi = float(z.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(z - mid, 0.0, 1.0)
        s = float(proj.sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    result = np.clip(z - tau, 0.0, 1.0)
    diff = target_sum - float(result.sum())
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def _softGrad(h, alpha):
    n = len(h)
    corr = _corr_full(h)
    max_c = float(corr.max())
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / float(exp_s.sum())
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        if w[k] < 1e-9:
            continue
        lag = k - (n - 1)
        grad += w[k] * _grad_lag(h, lag)
    return grad, max_c, corr

def _subgrad_optimize(h_init, n_iter, lr0, lr_decay, alpha0, alpha_growth, deadline, check_every=150):

    target = n / 2.0
    h = h_init.copy()

    no_improve = 0

            break

        grad, max_c, _ = _softGrad(h, alpha)
        lr = lr0 / (1.0 + t * lr_decay)
        z = h - lr * grad
        h = _project_onto_set(z, target)
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:

                best_h = h.copy()
                no_improve = 0
            else:

            if no_improve > patience:
                break
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_obj(x, alpha, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(corr.max())
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -50.0, 0.0)
    log_sum = float(np.sum(np.exp(shifted)))
    return (m + np.log(log_sum) / alpha) * dx

def _slsqp_grad(x, alpha, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(corr.max())
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / float(exp_s.sum())
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        if w[k] < 1e-9:
            continue
        lag = k - (n - 1)
        grad += w[k] * _grad_lag(x, lag)
    return grad * dx

def _slsqp_refine(h_init, alpha, maxiter=400, ftol=1e-14):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: float(x.sum()) - target, 'jac': lambda x: ones}]
    try:
        res = minimize(_slsqp_obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol},
                       jac=lambda x: _slsqp_grad(x, alpha, dx),
                       args=(alpha, dx))
        return res.x
    except Exception:
        return h_init.copy()

def _coord_descent(h_init, n_sweeps, n_grid, deadline):

    h = h_init.copy()

                    best_t = t
            h[i] = best_t
        h = _project_onto_set(h, target)
    return h, _c5_of(h)
