    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h = h.copy()
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def subgradient_for_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        n_valid = n - k
        if n_valid > 0:
            grad[:n_valid] += (1.0 - h[k:])
            grad[k:] -= h[:n_valid]
    else:
        m = -k
        n_valid = n - m
        if n_valid > 0:
            grad[m:] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[m:]
    return grad

def topk_subgradient(h, K=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for idx in top_indices:
        k = idx - (n - 1)
        grad += subgradient_for_k(h, k)
    return grad / K

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

_g_n = 0
_g_target = 0.0

def slsqp_polish(h0, n, target, alphas, maxiter, ftol):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h0.copy()
    best_local = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_local:
                best_local = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_local

def subgradient_descent(h0, n, target, max_iter, rng, start_time, budget_s, reserve_s):
    """Nesterov momentum subgradient with adaptive step size"""
    h = h0.copy()
    h_prev = h.copy()
    grad_prev = np.zeros(n)
    lr = 1.0 / np.sqrt(n)
    beta = 0.9
    best_h = h.copy()
    best_score = compute_c5(h)
    patience = 0
    max_patience = 200
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - reserve_s:
            break
        
        # Nesterov prediction
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute top-k subgradient at y
        grad = topk_subgradient(y, K=3)
        
        # Gradient momentum
        grad_mom = grad + beta * (grad - grad_prev)
        
        # Step
        h_new = y - lr * grad_mom
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive step size and momentum
        if patience == max_patience:
            lr *= 0.5
            beta = max(0.5, beta * 0.95)
            patience = 0
            if lr < 1e-6:
                break
        
        h_prev = h
        h = h_new
        grad_prev = grad
        
        # Periodic SLSQP polish
        if (it + 1) % 500 == 0 and time.time() - start_time < budget_s - reserve_s - 30:
            h_p, _ = slsqp_polish(h, n, target, [100, 300], 200, 1e-14)
            if compute_c5(h_p) < best_score:
                best_score = compute_c5(h_p)
                best_h = h_p
                h = h_p.copy()
                h_prev = h.copy()
                grad_prev = np.zeros(n)
                patience = 0
    
    return best_h, best_score
