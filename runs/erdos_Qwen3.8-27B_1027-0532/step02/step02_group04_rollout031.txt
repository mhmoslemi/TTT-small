import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = shifted - np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += w
            grad[:N - lag] -= w * h[lag:]
            grad[lag:] -= w * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += w
            grad[l:] -= w * h[:N - l]
            grad[:N - l] -= w * h[l:]
    return grad


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x

def local_search_greedy(h0, t_end, rng):
    n = len(h0)
    best_h = h0.copy()
    best_score = evaluate_score(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        found = False
        for s in step_sizes:
            for sign in (1.0, -1.0):
                di = s * sign
                ni = cur_h[i] + di
                nj = cur_h[j] - di
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                cur_h[i] = ni
                cur_h[j] = nj
                sc = evaluate_score(cur_h)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    found = True
                    break
                else:
                    cur_h[i] = ni - di
                    cur_h[j] = nj + di
            if found:
                break
        if found and cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()

    return best_h, best_score
