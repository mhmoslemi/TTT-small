def smooth_max_h_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    # Gradient
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def smooth_max_h_grad(h, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(1.0 - h, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = conv1[idx] - conv2[idx]
    return obj, grad

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def penalty_obj_grad(h, alpha, mu):
    obj, grad = smooth_max_h_grad(h, alpha)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad

def lbfgsb_optimize(h0, alpha, mu, maxiter, bounds):
    res = minimize(
        penalty_obj_grad, h0, method='L-BFGS-B',
        bounds=bounds, jac=True, args=(alpha, mu),
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0,
                 'maxls': 50, 'maxfun': maxiter * 20}
    )
    return res.x

def binary_local_search(h, rng, max_iters, time_limit, t0):
    """Efficient binary local search with swap operations."""
    n = len(h)
    h_arr = (h > 0.5).astype(np.float64)
    ones = np.where(h_arr > 0.5)[0].tolist()
    zeros = np.where(h_arr <= 0.5)[0].tolist()
    
    corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
    current_max = np.max(corr)
    best_max = current_max
    best_arr = h_arr.copy()
    
    r_vals = np.arange(-(n-1), n)
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        # Pick a swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        idx_a = i + r_vals
        idx_b = j + r_vals
        idx_c = i - r_vals
        idx_d = j - r_vals
        
        t_a = np.where((idx_a >= 0) & (idx_a < n), h_arr[idx_a], 0.0)
        t_b = np.where((idx_b >= 0) & (idx_b < n), h_arr[idx_b], 0.0)
        t_c = np.where((idx_c >= 0) & (idx_c < n), h_arr[idx_c], 0.0)
        t_d = np.where((idx_d >= 0) & (idx_d < n), h_arr[idx_d], 0.0)
        
        delta = t_a - t_b + t_c - t_d
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        if new_max < current_max - 1e-12:

            h_arr[i] = 0.0
            h_arr[j] = 1.0
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            corr = new_corr
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_arr = h_arr.copy()
    
    return best_arr, best_max

def iterative_max_optimize(h0, n_iters, bounds):
    """Optimize by iteratively focusing on the current max lag."""
    n = len(h0)
    h = h0.copy()
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        r_star = np.argmax(corr)
        r = r_star - (n - 1)
        
        # Gradient of c_{r_star} with respect to h
        grad = np.zeros(n)
        for m in range(n):
            val = 0.0
            if 0 <= m + r < n:
                val += (1.0 - h[m + r])
            if 0 <= m - r < n:
                val -= h[m - r]
            grad[m] = val
        
        # Line search or fixed step
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)  # projection onto sum-preserving subspace
        
        # Take a step
        step = 0.1 / (np.max(np.abs(grad)) + 1e-10)
        h_new = h - step * grad
        h_new = project_sum(np.clip(h_new, 0, 1), n / 2.0)
        
        # Accept if improves
        new_score = evaluate_score(h_new)
        old_score = evaluate_score(h)
        if new_score < old_score:
            h = h_new
        else:
            # Try smaller step
            step *= 0.5
            h_new = h - step * grad
            h_new = project_sum(np.clip(h_new, 0, 1), n / 2.0)
            if evaluate_score(h_new) < old_score:
                h = h_new
    
    return h

def guided_swap_search(h, rng, time_limit, t0):
    n = len(h)
    h_arr = (h > 0.5).astype(np.float64)
    ones = np.where(h_arr > 0.5)[0]
    zeros = np.where(h_arr <= 0.5)[0]
    
    corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
    current_score = np.max(corr) * 2.0 / n
    best_score = current_score
    best_arr = h_arr.copy()
    
    r_vals = np.arange(-(n-1), n)
    
    while time.time() - t0 < time_limit:
        corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
        r_star_idx = np.argmax(corr)
        r_star = int(r_star_idx) - (n - 1)
        
        # Guided: find 1s where h[i+r_star] = 0 (these contribute to c_{r_star})
        candidates_i = []
        for i in ones:
            if 0 <= i + r_star < n and h_arr[i + r_star] == 0:
                candidates_i.append(i)
        
        # Find 0s where h[j+r_star] = 1 (these would reduce c_{r_star} if we put 1 there)
        candidates_j = []
        for j in zeros:
            if 0 <= j + r_star < n and h_arr[j + r_star] == 1:
                candidates_j.append(j)
        
        if not candidates_i or not candidates_j:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = candidates_i[rng.randint(len(candidates_i))]
            j = candidates_j[rng.randint(len(candidates_j))]
        
        # Compute delta
        idx_a = i + r_vals
        idx_b = j + r_vals
        idx_c = i - r_vals
        idx_d = j - r_vals
        
        t_a = np.where((idx_a >= 0) & (idx_a < n), h_arr[idx_a], 0.0)
        t_b = np.where((idx_b >= 0) & (idx_b < n), h_arr[idx_b], 0.0)
        t_c = np.where((idx_c >= 0) & (idx_c < n), h_arr[idx_c], 0.0)
        t_d = np.where((idx_d >= 0) & (idx_d < n), h_arr[idx_d], 0.0)
        
        delta = t_a - t_b + t_c - t_d
        new_corr = corr + delta
        new_max = np.max(new_corr)

            ones = np.append(ones, j)
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            zeros = np.append(zeros, i)
            
            new_score = new_max * 2.0 / n
            if new_score < best_score:
                best_score = new_score
                best_arr = h_arr.copy()

    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return np.max(compute_correlations(h)) * (2.0 / n)

def smooth_max_h_grad(h, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m + np.log(sum_exp) / alpha
    w_rev = weights[::-1]
    conv1 = np.convolve(1.0 - h, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
