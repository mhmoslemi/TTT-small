def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n

def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)

    m = float(np.max(x))

    h = np.empty(N)

    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

    val = smooth_max_val(compute_correlations(x), alpha)
    viol = float(np.sum(x)) - target_sum
    return val + 0.5 * penalty * viol * viol

    gh = corr_grad_h(x, alpha)
    viol = float(np.sum(x)) - target_sum
    return gh + penalty * viol * np.ones(len(x))
