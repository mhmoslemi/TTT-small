    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def vectorized_greedy(h, max_iters, rng, time_limit=None, start_time=None, top_k=8):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 300
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        max_val = float(corr.max())
        
        # Try up to 50 random swaps
        found = False
        n_try = min(50, max(10, len(ones) * len(zeros)))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Check if this swap reduces the max
            # For swap (i: 1->0, j: 0->1):
            # corr[k] changes by: -(1-h[i+k]) + (1-h[j+k]) for each k
            # = h[i+k] - h[j+k]
            # We need the new max to be < old max
            
            # Quick check: find the max shift and see if it improves
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            ipk = i + k_star
            jpk = j + k_star
            h_ipk = h[ipk] if 0 <= ipk < n else 0.0
            h_jpk = h[jpk] if 0 <= jpk < n else 0.0
            delta_max = h_ipk - h_jpk
            if delta_max >= 0:
                continue  # This swap won't help the max shift
            
            # Do the swap and recompute
            h[i] = 0
            h[j] = 1
            corr_new = compute_corr(h)
            new_max = float(corr_new.max())
            
            if new_max < max_val:
                corr = corr_new
                new_c5 = new_max * (2.0 / n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stall = 0
                found = True
                break
            else:
                # Revert
                h[i] = 1
                h[j] = 0
        
        if not found:
            stall += 1
    
    return best_h, best_c5

def sa_optimize(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 5000
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Choose move
        move_type = rng.randint(4)
        h_new = h.copy()
        i_new, j_new = -1, -1
        
        if move_type <= 1:
            # Focused move: try to fix the worst shift
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            # Find ones that, if flipped, would help
            i_pool = [idx for idx in ones if 0 <= idx + k_star < n and h[idx + k_star] == 0]
            j_pool = [idx for idx in zeros if 0 <= idx + k_star < n and h[idx + k_star] == 1]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i_new = i_pool[rng.randint(len(i_pool))]
                j_new = j_pool[rng.randint(len(j_pool))]
                h_new[i_new] = 0
                h_new[j_new] = 1
            else:
                continue
        elif move_type == 2:
            # Random swap
            i_new = ones[rng.randint(len(ones))]
            j_new = zeros[rng.randint(len(zeros))]
            h_new[i_new] = 0
            h_new[j_new] = 1
        else:
            # 2-swap
            if len(ones) >= 2 and len(zeros) >= 2:
                perm = rng.permutation(4)
                i1, i2 = ones[perm[0]], ones[perm[1]]
                j1, j2 = zeros[perm[2]], zeros[perm[3]]
                h_new[i1] = 0
                h_new[i2] = 0
                h_new[j1] = 1
                h_new[j2] = 1
                i_new, j_new = i1, j1
        
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-15))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def subgradient_optimize(h_start, n, target, time_limit=None, start_time=None, n_iters=5000):
    """Subgradient method for the min-max problem with projection."""
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Adaptive step size
    step = 1.0 / np.sqrt(len(h))
    
    for it in range(n_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        corr = compute_corr(h)
        max_val = float(corr.max())
        
        # Find active set (shifts near the max)
        tol = max(0.5, 0.01 * max_val)
        active = np.where(corr >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        # Take top few
        if len(active) > 10:
            active = active[np.argsort(corr[active])[-10:]]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += (1.0 - h[k:])
                if k > 0:
                    grad[k:] -= h[:n-k]
            else:
                mk = -k
                if mk < n:
                    grad[mk:] += (1.0 - h[:n-mk])
                if mk > 0:
                    grad[:n-mk] -= h[mk:]
        grad /= len(active)
        
        # Step and project
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        # Check if binary projection also helps
        h_bin = project_binary(h_new, int(target))
        sc_bin = compute_c5(h_bin)
        sc_cont = compute_c5(h_new)
        
        if sc_cont < best_c5:
            best_c5 = sc_cont
            best_h = h_new.copy()
        if sc_bin < best_c5:
            best_c5 = sc_bin
            best_h = h_bin.copy()
        
        h = h_new if sc_cont <= sc_bin else h_bin
        
        # Decay step size
        step *= 0.999
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 500:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find good candidates
        i_cand = []
        for i in ones:
            ipk = i + k_star
            if 0 <= ipk < n and h[ipk] == 0:
                i_cand.append(i)
        j_cand = []
        for j in zeros:
            jpk = j + k_star
            if 0 <= jpk < n and h[jpk] == 1:
                j_cand.append(j)
        
        if len(i_cand) > 0 and len(j_cand) > 0:
            n_try = min(50, len(i_cand) * len(j_cand))
            found = False
            for _ in range(n_try):
                i = i_cand[rng.randint(len(i_cand))]
                j = j_cand[rng.randint(len(j_cand))]
                h[i] = 0
                h[j] = 1
                corr_new = compute_corr(h)
                new_max = float(corr_new.max())
                if new_max < max_val:
                    corr = corr_new
                    new_c5 = new_max * (2.0 / n)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    stall = 0
                    found = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            if not found:
                stall += 1
        else:
            stall += 1
            # Random move to escape
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            if float(corr.max()) * (2.0/n) < best_c5:
                best_c5 = float(corr.max()) * (2.0/n)
                best_h = h.copy()
                stall = 0
    
    return best_h, best_c5

def sa_long(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        h_new = h.copy()
        
        # Move selection: 60% focused, 30% random, 10% 2-swap
        r = rng.random()
        if r < 0.6:
            # Focused on worst shift
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            # Vectorized candidate finding
            i_mask = np.zeros(n, dtype=bool)
            j_mask = np.zeros(n, dtype=bool)
            for i in range(n):
                ipk = i + k_star
                if 0 <= ipk < n:
                    if h[i] == 1 and h[ipk] == 0:
                        i_mask[i] = True
                    if h[i] == 0 and h[ipk] == 1:
                        j_mask[i] = True
            
            i_pool = np.where(i_mask)[0]
            j_pool = np.where(j_mask)[0]
            
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        elif r < 0.9:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                h_new[i1] = 0
                h_new[i2] = 0
                h_new[j1] = 1
                h_new[j2] = 1
            else:
                continue
        
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
