The projection onto {0<=h<=1, sum(h)=n/2} is the tricky part. The standard approach is:

2. Iteratively: clip to [0,1], then adjust sum by adding/subtracting from non-saturated coordinates

def project_box_hyperplane(y, c):
    # Project y onto {0<=h<=1, sum(h)=c}
    # Find lambda such that sum(clip(y[i]-lambda, 0, 1)) = c
    lo = y.min() - c  # very negative
    hi = y.max() + c  # very positive
    for _ in range(100):
        mid = (lo + hi) / 2

        if proj_sum > c:
            lo = mid

    h_f2, sc_f2 = fista_optimize(best_h, n, target, 800, 60000, deadline4)
    if sc_f2 < best_c5:
        best_c5 = sc_f2
        best_h = h_f2
    if time.time() - t0 > budget_s - 80:
        best_h = project_box_hyperplane(best_h, float(target))
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, n

    deadline4b = min(deadline4 + 100, t0 + int(budget_s * 0.52))
    h_sg, sc_sg = subgradient_refine(best_h, n, target, 20000, 0.5, 0.01, deadline4b)
    if sc_sg < best_c5:
        best_c5 = sc_sg
        best_h = h_sg
    if time.time() - t0 > budget_s - 100:
        best_h = project_box_hyperplane(best_h, float(target))
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, n

    n2 = n * 2
    target2 = n2 // 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h_up = np.interp(x_new, x_old, best_h)
    h_up = project_box_hyperplane(h_up, float(target2))

    deadline5 = t0 + min(950, int(budget_s * 0.72))
    h_up_f, sc_up = fista_optimize(h_up, n2, target2, 200, 40000, deadline5)
    if sc_up < best_c5:
        best_c5 = sc_up
        best_h = h_up_f.copy()

    if time.time() - t0 > budget_s - 100:
        best_h = project_box_hyperplane(best_h, float(best_h.size // 2))
        best_c5 = compute_c5(best_h)
        n = best_h.size
        return best_h, best_c5, n

    n_best = len(best_h)
    target_best = n_best // 2
    deadline6 = min(t0 + int(budget_s * 0.82), t0 + 1050)
    if n_best == n2:
        for alpha in [300, 800]:
            if time.time() > deadline6:
                break
            h_s2, sc_s2 = slsqp_step(best_h, n_best, target_best, alpha, 2000, deadline6)
            if sc_s2 < best_c5:
                best_c5 = sc_s2
                best_h = h_s2

    if time.time() - t0 > budget_s - 80:
        best_h = project_box_hyperplane(best_h, float(target_best))
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, n_best

    deadline7 = t0 + budget_s - 80
    n_best = len(best_h)
    target_best = n_best // 2
    for attempt in range(15):
        if time.time() > deadline7:
            break
        block_size = rng.randint(8, max(10, min(40, n_best // 5)))
        start = rng.randint(0, n_best - block_size)
        h_pert = best_h.copy()
        block_sum = float(h_pert[start:start + block_size].sum())
        new_block = rng.dirichlet(np.ones(block_size) * 1.5) * block_sum
        h_pert[start:start + block_size] = np.minimum(new_block, 2.0)
        h_pert = project_box_hyperplane(h_pert, float(target_best))
        h_qf, sc_qf = fista_optimize(h_pert, n_best, target_best, 300, 8000, deadline7)
        if sc_qf < best_c5:
            best_c5 = sc_qf
            best_h = h_qf

    if time.time() - t0 > budget_s - 40:
        best_h = project_box_hyperplane(best_h, float(target_best))
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, n_best

    deadline8 = t0 + budget_s - 5
    n_best = len(best_h)
    target_best = n_best // 2
    for alpha in [2000, 5000, 10000]:
        if time.time() > deadline8:
            break
        h_fin, sc_fin = slsqp_step(best_h, n_best, target_best, alpha, 3000, deadline8)
        if sc_fin < best_c5:
            best_c5 = sc_fin
            best_h = h_fin

    best_h = project_box_hyperplane(best_h, float(target_best))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    n_best = len(best_h)
    return best_h, best_c5, n_best
