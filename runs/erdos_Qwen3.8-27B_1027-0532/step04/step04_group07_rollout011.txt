    b = np.zeros(m)

        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h - corr[ix]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        nfree = int(free.sum())
        if nfree == 0:
            break
        per = diff / nfree
        if diff > 0:
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def get_grad_single(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def get_grad_multi(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:

        lag = lag_indices[i] - (n - 1)

        else:
