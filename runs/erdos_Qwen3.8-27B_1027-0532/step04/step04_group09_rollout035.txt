_g_target = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])

            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
