def project_binary(h, target_sum):
    # Project to binary with exact sum
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum > target_sum:
        # Need to turn off some 1s
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        order = np.argsort(h[ones])  # Turn off those with smallest h value

        order = np.argsort(-h[zeros])  # Turn on those with largest h value

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):

             'jac': lambda x: np.ones(n)}]

            break
        try:
            res = minimize(smooth_objective, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})

        if b % 2 == 0:
            start = b * block_size
            end = min(start + block_size, n)
            h[start:end] = 1.0
        else:
            start = b * block_size
            end = min(start + block_size, n)
            frac = block_size // 2
            h[start:start + frac] = 1.0
    return project_binary(h, target_sum)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

_smooth_n = 0
_smooth_target = 0.0

def _cons_fun(h):
    return np.sum(h) - _smooth_target

def _cons_jac(h):
    return np.ones(_smooth_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _smooth_n, _smooth_target
    _smooth_n = n
    _smooth_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def smart_perturb(h, target_sum, rng, n_pert):
    h = h.copy()
    corr = compute_corr(h)
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        peak_idx = int(np.argmax(corr))
        k_peak = peak_idx - (n - 1)
        i_mask = (ones + k_peak >= 0) & (ones + k_peak < n)
        j_mask = (zeros + k_peak >= 0) & (zeros + k_peak < n)
        i_pool = ones[i_mask] if len(ones[i_mask]) > 0 else ones
        j_pool = zeros[j_mask] if len(zeros[j_mask]) > 0 else zeros
        i = i_pool[rng.randint(len(i_pool))]
        j = j_pool[rng.randint(len(j_pool))]
        h[i] = 0.0
        h[j] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        corr = compute_corr(h)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _smooth_n, _smooth_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick binary local search
    h_bin = project_binary(h_init, target_init)
    allowed = min(60, max(10, tleft() * 0.08))
    h_ls, c5_ls = focused_local_search(h_bin, target_init, allowed, rng, start_time)
    if c5_ls < best_score:
        best_score = c5_ls
        best_h = h_ls.copy()

    # Phase 2: SA with 2-swap
    allowed = min(120, max(20, tleft() * 0.12))
    h_sa, c5_sa = sa_with_2swap(h_ls, target_init, allowed, rng, start_time,
                                t_init=0.02, t_final=0.00003, max_iters=500000)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Multi-start with diverse initializations
    diverse_starts = [
        gen_random_start(n_init, target_init, rng),
        gen_modular_start(n_init, target_init, 3, rng),
        gen_modular_start(n_init, target_init, 5, rng),
        gen_golden_start(n_init, target_init, rng),
        gen_block_start(n_init, target_init, rng),
        gen_modular_start(n_init, target_init, 7, rng),
    ]
    for si, h_start in enumerate(diverse_starts):
        if tleft() < 40:
            break
        allowed_ls = min(25, max(5, tleft() * 0.03))
        h_perm_ls, c5_perm = focused_local_search(h_start, target_init, allowed_ls, rng, start_time)
        if tleft() < 25:
            break
        allowed_sa = min(35, max(5, tleft() * 0.03))
        h_perm_sa, c5_perm_sa = sa_with_2swap(h_perm_ls, target_init, allowed_sa, rng, start_time,
                                              t_init=0.015, t_final=0.0001, max_iters=200000)
        if c5_perm_sa < best_score:
            best_score = c5_perm_sa
            best_h = h_perm_sa.copy()

    # Phase 4: Iterated local search with smart perturbation
    for ils_iter in range(20):
        if tleft() < 25:
            break
        n_pert = max(4, 4 + ils_iter)
        h_pert = smart_perturb(best_h, target_init, rng, n_pert)
        allowed = min(20, max(3, tleft() * 0.02))
        h_ig, c5_ig = focused_local_search(h_pert, target_init, allowed, rng, start_time)
        if c5_ig < best_score:
            best_score = c5_ig
            best_h = h_ig.copy()
        if tleft() < 20:
            break
        allowed_sa = min(15, max(3, tleft() * 0.015))
        h_ig_sa, c5_ig_sa = sa_with_2swap(h_ig, target_init, allowed_sa, rng, start_time,
                                          t_init=0.008, t_final=0.0001, max_iters=100000)
        if c5_ig_sa < best_score:
            best_score = c5_ig_sa
            best_h = h_ig_sa.copy()

    # Phase 5: Try different resolutions
    n_candidates = []
    if n_init > 400:
        n_candidates.append(n_init - 100)
    n_candidates.append(n_init + 100)
    n_candidates.append(n_init + 200)
    n_candidates = [n for n in n_candidates if n >= 200 and n % 2 == 0 and n != n_init]

    for n_try in n_candidates:
        if tleft() < 45:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(25, max(5, tleft() * 0.03))
        h_g2, c5_g2 = focused_local_search(h_bin_try, target_try, allowed, rng, start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()
        if tleft() < 30:
            break
        allowed2 = min(30, max(5, tleft() * 0.025))
        h_sa2, c5_sa2 = sa_with_2swap(h_g2, target_try, allowed2, rng, start_time,
                                       t_init=0.01, t_final=0.0001, max_iters=200000)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Final SA polish on best binary
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_final = project_binary(best_h, target_use)
        allowed = max(10.0, min(80, tleft() * 0.3))
        h_sa_f, c5_f = sa_with_2swap(h_bin_final, target_use, allowed, rng, start_time,
                                       t_init=0.01, t_final=0.00002, max_iters=2000000)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_bin_for_slsqp = project_binary(best_h, int(n_use / 2))
        alphas = [20, 50, 150, 400, 1000, 3000]
        allowed = max(5.0, min(100, tleft() - 10))
        h_ref, sc = slsqp_refine(h_bin_for_slsqp, n_use, target_use, alphas, 2000, 1e-14,
                                  allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 8: Final binary + quick SLSQP
    if tleft() > 15:
        n_use = len(best_h)
        target_bin = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_bin)
        allowed_ls = max(3.0, min(10, tleft() * 0.15))
        h_final_ls, c5_final_ls = focused_local_search(h_bin_final, target_bin, allowed_ls, rng, start_time)
        if c5_final_ls < best_score:
            best_score = c5_final_ls
            best_h = h_final_ls.copy()
        if tleft() > 10:
            target_use = float(n_use) / 2.0
            alphas = [500, 1500, 3000]
            allowed = max(3.0, min(15, tleft() - 5))
            h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 1000, 1e-15,
                                      allowed, start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
