def grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    g = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        m = lag_indices[i] - (n - 1)
        if m >= 0:
            if m < n:
                g[m:] += w * (1.0 - h[:n - m])
                g[:n - m] -= w * h[m:]
        else:
            p = -m
            if p < n:
                g[:n - p] += w * (1.0 - h[p:])
                g[p:] -= w * h[:n - p]
    return g


def smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(nc)
        vals = corr
    c_ref = np.max(vals)
    shifted = np.clip(alpha * (vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(nc)
        vals = corr
    c_ref = np.max(vals)
    ex = np.exp(np.clip(alpha * (vals - c_ref), -500.0, 0.0))
    tot = np.sum(ex)
    w = ex / tot
    return grad_multi_lag(x, idx, w)


def slsqp_opt(h0, target, alpha_topk, maxiter):
    global _C_N, _C_TARGET
    _C_N = len(h0)
    _C_TARGET = target

    def obj_eq(x):
        return np.sum(x) - _C_TARGET

    def jac_eq(x):
        return np.ones(_C_N)

    bounds = [(0.0, 1.0)] * _C_N
    cons = [{'type': 'eq', 'fun': obj_eq, 'jac': jac_eq}]
    h = h0.copy()
    best = compute_c5(h)
    for alpha, topk in alpha_topk:
        try:
            res = minimize(smooth_obj, h, args=(alpha, topk),
                           jac=smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            hn = project_feasible(res.x, target)
            sc = compute_c5(hn)
            if sc < best:
                best = sc
                h = hn
        except Exception:
            pass
    return h, best


def lbfgs_opt(h0, target, alpha_topk, maxiter):
    global _C_N, _C_TARGET
    _C_N = len(h0)
    _C_TARGET = target

    def obj_con(x):
        return np.array([np.sum(x) - _C_TARGET])

    def jac_con(x):
        return np.array([np.ones(_C_N)])

    bounds = [(0.0, 1.0)] * _C_N
    cons = [{'type': 'eq', 'fun': obj_con, 'jac': jac_con}]
    h = h0.copy()
    best = compute_c5(h)
    for alpha, topk in alpha_topk:
        try:
            res = minimize(smooth_obj, h, args=(alpha, topk),
                           jac=smooth_grad, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxls': 20})
            hn = project_feasible(res.x, target)
            sc = compute_c5(hn)
            if sc < best:
                best = sc
                h = hn
        except Exception:
            pass
    return h, best


def lp_step(h, target, topk):
    n = len(h)
    corr = overlap_corrs(h)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(nc)
    m = len(idx)
    A = np.zeros((m, n))
    b = np.zeros(m)
    for i, j in enumerate(idx):
        g = grad_single_lag(h, j)
        A[i, :] = g
        b[i] = g @ h - corr[j]
    c = np.zeros(n)
    A_eq = np.ones((1, n))
    b_eq = np.array([target])
    bnd = [(0.0, 1.0)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnd, method='highs')
        if res.success:
            return project_feasible(res.x, target)
    except Exception:
        pass
    return None


def lp_refine(h0, target, maxit, topk, tlimit=None):
    h = h0.copy()
    best_h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    stag = 0
    for it in range(maxit):
        if tlimit is not None and (time.time() - t0) > tlimit:
            break
        h_lp = lp_step(h, target, topk)
        if h_lp is None:
            break
        best_t = best
        best_h_candidate = h.copy()
        for t in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            h_try = project_feasible(t * h_lp + (1.0 - t) * h, target)
            sc = compute_c5(h_try)
            if sc < best_t:
                best_t = sc
                best_h_candidate = h_try
        if best_t < best - 1e-14:
            best = best_t
            best_h = best_h_candidate.copy()
            h = best_h_candidate
            stag = 0
        else:
            stag += 1
            if stag > 15:
                break
    return best_h, best


def proj_grad_step(h0, target, maxit, topk, alpha, lr_init, momentum,
                   noise_p, noise_s, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best = compute_c5(h)
    lr = lr_init
    stag = 0
    for it in range(maxit):
        corr = overlap_corrs(h)
        nc = len(corr)
        if topk < nc:
            idx = np.argpartition(corr, -topk)[-topk:]
            vals = corr[idx]
        else:
            idx = np.arange(nc)
            vals = corr
        cmax = np.max(vals)
        w = np.exp(alpha * (vals - cmax))
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.ones(len(idx)) / len(idx)
        else:
            w = w / ws
        g = grad_multi_lag(h, idx, w)
        v = momentum * v + (1.0 - momentum) * g
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)
        if noise_p > 0.0 and rng.random() < noise_p:
            h_new = h_new + rng.randn(n) * noise_s
            h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best:
            best = sc
            best_h = h_new.copy()
            stag = 0
            lr = min(lr * 1.01, 5.0)
        else:
            stag += 1
            lr *= 0.98
        h = h_new
        if stag > patience:
            break
    return best_h, best


def interp_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _C_N, _C_TARGET
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remain():
        return budget_s - 30.0 - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    tgt0 = n0 / 2.0
    best_h = project_feasible(h_init, tgt0)
    best_sc = compute_c5(best_h)

    if remain() < 30:
        return best_h, best_sc, n0

    phase1_budget = min(remain() * 0.25, 300)
    h_new, sc = proj_grad_step(best_h, tgt0, 2000, 10, 50, 0.10, 0.90,
                                0.05, 0.02, 200, seed)
    if sc < best_sc:
        best_sc = sc
        best_h = h_new

    if remain() > 50 and phase1_budget > 20:
        h_new, sc = lp_refine(best_h, tgt0, 300, 50, tlimit=phase1_budget)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    if remain() > 100:
        h_new, sc = lbfgs_opt(best_h, tgt0, [(100, 60), (500, 30), (2000, 15)],
                               maxiter=3000)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    if remain() > 80:
        h_new, sc = lp_refine(best_h, tgt0, 200, 30, tlimit=remain() - 60)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    if remain() > 150:
        h_new, sc = slsqp_opt(best_h, tgt0, [(500, 40), (2000, 20), (5000, 10)],
                               maxiter=2000)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    if remain() > 100:
        h_new, sc = lp_refine(best_h, tgt0, 200, 20, tlimit=remain() - 80)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    for n_res in [600, 800]:
        if remain() < 150:
            break
        tgt_r = n_res / 2.0
        h_res = interp_h(best_h, n_res, tgt_r)
        sc_res = compute_c5(h_res)

        if remain() > 100:
            h_new, sc = proj_grad_step(h_res, tgt_r, 1500, 15, 30, 0.08, 0.90,
                                        0.03, 0.01, 150, seed + n_res)
            if sc < sc_res:
                h_res = h_new
                sc_res = sc

        if remain() > 100:
            h_new, sc = lbfgs_opt(h_res, tgt_r, [(200, 80), (1000, 40), (5000, 20)],
                                   maxiter=2000)
            if sc < sc_res:
                h_res = h_new
                sc_res = sc

        if remain() > 80:
            h_new, sc = lp_refine(h_res, tgt_r, 200, 60, tlimit=remain() - 60)
            if sc < sc_res:
                h_res = h_new
                sc_res = sc

        if remain() > 60:
            h_new, sc = slsqp_opt(h_res, tgt_r, [(500, 50), (2000, 25), (8000, 10)],
                                   maxiter=1500)
            if sc < sc_res:
                h_res = h_new
                sc_res = sc

        if remain() > 50:
            h_new, sc = lp_refine(h_res, tgt_r, 200, 30, tlimit=remain() - 40)
            if sc < sc_res:
                h_res = h_new
                sc_res = sc

        if sc_res < best_sc:
            best_sc = sc_res
            best_h = h_res

    if remain() > 200:
        n_res2 = 1000
        tgt_r2 = n_res2 / 2.0
        h_res2 = interp_h(best_h, n_res2, tgt_r2)
        sc_res2 = compute_c5(h_res2)

        if remain() > 150:
            h_new, sc = lbfgs_opt(h_res2, tgt_r2, [(200, 100), (1000, 50)],
                                   maxiter=1500)
            if sc < sc_res2:
                h_res2 = h_new
                sc_res2 = sc

        if remain() > 100:
            h_new, sc = lp_refine(h_res2, tgt_r2, 150, 80, tlimit=remain() - 80)
            if sc < sc_res2:
                h_res2 = h_new
                sc_res2 = sc

        if sc_res2 < best_sc:
            best_sc = sc_res2
            best_h = h_res2

    if remain() > 60:
        h_try = best_h + rng.randn(len(best_h)) * 0.002
        h_try = project_feasible(h_try, len(best_h) / 2.0)
        h_new, sc = proj_grad_step(h_try, len(best_h) / 2.0, 1000, 5, 200,
                                    0.02, 0.93, 0.0, 0.0, 100, seed + 9999)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    if remain() > 30:
        h_new, sc = lp_refine(best_h, len(best_h) / 2.0, 100, 15,
                               tlimit=remain() - 10)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_sc = compute_c5(h_final)
    return h_final, final_sc, len(h_final)
