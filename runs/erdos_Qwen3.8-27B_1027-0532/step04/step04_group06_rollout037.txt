import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == int(target_sum):
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > int(target_sum):
        need = current_sum - int(target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum) - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != int(target_sum):
        if final_sum > int(target_sum):
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - int(target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum) - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_lag_gradient(h, k, n):
    """Compute gradient of c(k) w.r.t. h for a specific lag k."""
    g = np.zeros(n)
    if k > 0:
        g[:n-k] -= h[k:]
        g[k:] += (1.0 - h[:n-k])
    elif k < 0:
        m = -k
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def smooth_objective_and_grad(h, n, alpha, top_m=20):
    """Compute smooth approximation of max correlation and its gradient."""
    corr = get_corr(h)
    center = n - 1
    c_max = np.max(corr)
    # Compute weights for top_m lags
    # corr is indexed from 0 to 2n-2, center is at n-1
    # Lag k corresponds to index center+k in corr
    shifted = alpha * (corr - c_max)
    # For numerical stability, clip
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    obj = c_max + log_sum / alpha
    
    # Weights
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Only use top_m lags for gradient
    top_indices = np.argsort(weights)[::-1][:top_m]
    top_weights = weights[top_indices]
    total_weight = np.sum(top_weights)
    top_weights = top_weights / total_weights  # renormalize
    
    grad = np.zeros(n)
    for idx, w in zip(top_indices, top_weights):
        k = int(idx) - center
        g_k = compute_lag_gradient(h, k, n)
        grad += w * g_k
    
    return float(obj), grad

    if len(candidates) == 0:
        candidates = np.array([center + 1])
    order = c[candidates][::-1].argsort()
    selected = candidates[order[:n_top]]
    lags = selected - center
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1])
    return lags

def binary_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 15)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = get_corr(h)
    center = n - 1
    worst_lags = get_top_lags(corr, center, 5)
    corr_refresh = 0
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.08:
                no_improve = 0
                for _ in range(rng.randint(2, 5)):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    current_c5 = compute_c5(h)
                corr = get_corr(h)
                worst_lags = get_top_lags(corr, center, 5)
            else:
                break
        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.60 and len(ones) > 0 and len(zeros) > 0:
            corr_refresh += 1
            if corr_refresh >= 4:
                corr = get_corr(h)
                worst_lags = get_top_lags(corr, center, 5)
                corr_refresh = 0
            k_star = int(worst_lags[rng.randint(len(worst_lags))])
            if k_star == 0:
                k_star = 1
            a_k = ones + k_star
            a_km = ones - k_star
            valid_a = (a_k >= 0) & (a_k < n) & (a_km >= 0) & (a_km < n)
            b_k = zeros + k_star
            b_km = zeros - k_star
            valid_b = (b_k >= 0) & (b_k < n) & (b_km >= 0) & (b_km < n)
            if np.sum(valid_a) > 0 and np.sum(valid_b) > 0:
                a_valid = ones[valid_a]
                a_score = h[np.clip(a_k, 0, n - 1)[valid_a]] + h[np.clip(a_km, 0, n - 1)[valid_a]]
                b_valid = zeros[valid_b]
                b_score = h[np.clip(b_k, 0, n - 1)[valid_b]] + h[np.clip(b_km, 0, n - 1)[valid_b]]
                a_score += rng.random(len(a_score)) * 0.3
                b_score += rng.random(len(b_score)) * 0.3
                a_idx = a_valid[np.argmin(a_score)]
                b_idx = b_valid[np.argmax(b_score)]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
            else:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
        elif r < 0.85 and len(ones) > 0 and len(zeros) > 0:
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
            h_new[a_idx] = 0.0
            h_new[b_idx] = 1.0
            moved = True
        elif r < 0.95 and len(ones) >= 2 and len(zeros) >= 2:
            ia = rng.choice(len(ones), 2, replace=False)
            ib = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[ia[0]]] = 0.0
            h_new[ones[ia[1]]] = 0.0
            h_new[zeros[ib[0]]] = 1.0
            h_new[zeros[ib[1]]] = 1.0
            moved = True
        if not moved:
            continue
        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def fast_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 200
    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5

def slsqp_refine(h_init, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    cons_f = lambda h_in: np.sum(h_in) - target
    cons_j = lambda h_in: np.ones(n)
    constraints = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_init.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        center = n - 1
        def obj_fn(h_in, a_val=alpha):
            corr = get_corr(h_in)
            c_max = np.max(corr)
            shifted = np.clip(a_val * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a_val)
        def grad_fn(h_in, a_val=alpha):
            corr = get_corr(h_in)
            c_max = np.max(corr)
            shifted = np.clip(a_val * (corr - c_max), -500.0, 0.0)
            exp_s = np.exp(shifted)
            total = np.sum(exp_s)
            if total < 1e-300:
                return np.zeros(n)
            weights = exp_s / total
            top_m = min(40, 2 * n - 1)
            top_idx = np.argsort(weights)[::-1][:top_m]
            top_w = weights[top_idx]
            top_w = top_w / np.sum(top_w)
            g = np.zeros(n)
            for idx, w in zip(top_idx, top_w):
                k = int(idx) - center
                g += w * compute_lag_gradient(h_in, k, n)
            return g
        try:
            res = minimize(obj_fn, h_cur, jac=grad_fn, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Binary SA at initial resolution
    h_bin = project_binary(h_init, target_init)
    sa_phases = [
        (80000, 0.025, 0.0001),
        (100000, 0.015, 0.00005),
        (80000, 0.008, 0.00002),
    ]
    h_sa = h_bin.copy()
    for (max_it, t_i, t_f) in sa_phases:
        if tleft() < 80:
            break
        allowed = min(180.0, tleft() * 0.35)
        h_sa, c5_sa = binary_sa(h_sa, target_init, max_it, t_i, t_f,
                                rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    if tleft() > 40:
        h_hc, c5_hc = fast_hill_climb(h_sa, target_init, 6000, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 2: SLSQP at initial resolution
    if tleft() > 60:
        alphas2 = [40, 100, 250, 600, 1500, 3000, 6000]
        h_ref, sc = slsqp_refine(best_h, n_init, target_init, alphas2, 400, 1e-14,
                                 tleft() * 0.4, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Multi-resolution SLSQP
    n_list = [800, 900, 1000, 1100, 1200]
    n_list = [x for x in n_list if x != n_init]
    for n_try in n_list:
        if tleft() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [60, 150, 400, 1000, 2500, 5000]
        maxit_try = max(150, min(350, int(tleft() * 0.15)))
        allowed = min(tleft() * 0.3, 250.0)
        h_ref, sc = slsqp_refine(h_interp, n_try, target_try, alphas_try,
                                 maxit_try, 1e-13, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Higher resolution attempts
    n_high_list = [1400, 1600, 1800]
    for n_try in n_high_list:
        if tleft() < 50:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [80, 200, 500, 1500, 3000]
        maxit_try = max(100, min(250, int(tleft() * 0.12)))
        allowed = min(tleft() * 0.35, 250.0)
        h_ref, sc = slsqp_refine(h_interp, n_try, target_try, alphas_try,
                                 maxit_try, 1e-13, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Final polish at best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 30:
        alphas_fine = [2000, 5000, 10000, 20000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas_fine,
                                 max(100, int(tleft() * 0.3)), 1e-15,
                                 min(tleft() * 0.5, 120.0), start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Perturbation + refine
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        for noise_scale in [0.01, 0.02, 0.03]:
            if tleft() < 15:
                break
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            alphas_p = [1000, 3000, 8000]
            h_ref, sc = slsqp_refine(h_pert, n_use, target_use, alphas_p,
                                     max(80, int(tleft() * 0.2)), 1e-14,
                                     min(30.0, tleft() * 0.4), start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 7: Final binary + continuous polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_f = project_binary(best_h, target_use)
        max_it = max(3000, int(tleft() * 2000))
        h_sa_f, c5_f = binary_sa(h_bin_f, target_use, max_it, 0.003, 0.00001,
                                 rng, time_limit=max(3.0, tleft() * 0.2), start_time=start_time)
        if tleft() > 12:
            alphas_last = [5000, 10000, 20000]
            h_ref, sc = slsqp_refine(h_sa_f, n_use, target_use, alphas_last,
                                     max(80, int(tleft() * 0.3)), 1e-15,
                                     max(3.0, tleft() * 0.5), start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            if c5_f < best_score:
                best_score = c5_f
                best_h = h_sa_f

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-9:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n_final
