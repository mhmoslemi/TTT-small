import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_optimize(h, target, alpha_list, maxiter, ftol, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    for alpha in alpha_list:
        if time.time() > deadline:
            break

        def fun(x):
            return lse_obj_and_grad(x, alpha)[0]

        def jac(x):
            return lse_obj_and_grad(x, alpha)[1]

        try:
            res = minimize(fun, h_cur, jac=jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        range_val = c_max - c_min
        if range_val < 1e-10:
            break
        threshold = c_max - active_frac * range_val
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        step = step_size / (1.0 + 0.015 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score


def subgrad_backtrack(h, target, n_steps, active_frac, deadline):

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    rng = np.random.RandomState(123)

            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))

            break

            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])

        improved = False

                best_h = h_new.copy()

                break

            h_cur = project_feasible(h_cur + 0.003 * rng.randn(n), target)

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_init(n, seed_val, target):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2, n)
    method = seed_val % 6
    if method == 0:
        h = 1.0 - np.abs(x - 1.0)
    elif method == 1:
        c1 = 0.4 + 0.3 * rng.random()
        c2 = 1.6 - 0.3 * rng.random()
        w = 0.1 + 0.2 * rng.random()
        h = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    elif method == 2:
        a = 0.1 + 0.5 * rng.random()
        b = a + 0.3 + 0.3 * rng.random()
        h = ((x >= a) & (x <= b)).astype(float)
        h = h + 0.1 * rng.random(n)
    elif method == 3:
        freq = 1 + rng.randint(3)
        phase = rng.random() * np.pi
        h = 0.5 + 0.4 * np.sin(np.pi * freq * x + phase)
    elif method == 4:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x)
    else:
        h = rng.uniform(0, 1, n)
    h = np.clip(h, 0, 1)
    return project_feasible(h, target)


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            h_fixed = h_cur.copy()
            x0 = h_cur[s:e].copy()

            def bf(x, idx=(s, e), fx=h_fixed, tg=target, al=alpha):
                ht = fx.copy()
                ht[idx[0]:idx[1]] = np.clip(x, 0.0, 1.0)
                ht = project_feasible(ht, tg)
                return lse_obj_and_grad(ht, al)[0]

            try:
                res = minimize(bf, x0, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur, cur_score
