def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def subgrad_active(h, active_indices, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in active_indices:
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
        else:
            mm = -m
            grad[:n - mm] += (1.0 - h[mm:])
            grad[mm:] -= h[:n - mm]
    return grad * dx / len(active_indices)

def nesterov_subgrad(h_start, target, time_limit, start_time, max_iters=500000, gamma0=None):
    n = len(h_start)
    dx = 2.0 / n
    if gamma0 is None:
        gamma0 = 0.05
    h = h_start.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    gamma = gamma0
    no_improve = 0
    patience = 8000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > patience:
            break
        
        corr = compute_corr(y)
        c_max = float(np.max(corr))
        threshold = c_max * 0.96
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 15:
            idx = np.argsort(corr[active])[::-1][:15]
            active = active[idx]
        
        grad = subgrad_active(y, active, dx)
        beta = 2.0 / (it + 2.0)
        h_new = project_box_sum(y - gamma * grad, target)
        y = (1.0 + beta) * h_new - beta * h
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 300 == 0:
                gamma *= 0.97
                if gamma < 1e-4:
                    gamma = gamma0 * 0.5
                    y = h.copy()
        
        if it > 0 and it % 15000 == 0:
            y = h.copy()
            gamma = gamma0 * 0.7
    
    return best_h, best_c5

                no_improve += 1
        else:
            h = project_binary(h, target)
            current_c5 = compute_c5(h)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve += 1

    return best_h, best_c5


def targeted_binary_search(h_start, target, time_limit, start_time, rng):
    n = len(h_start)
    h = project_binary(h_start, target) if np.any((h != np.round(h))) else h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 400

    while time.time() - start_time < time_limit:
        if stall > max_stall:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        corr = compute_corr(h)
        c_max = float(np.max(corr))
        top_lags = np.argsort(corr)[::-1][:10]

        lag = top_lags[rng.randint(len(top_lags))]
        m = lag - (n - 1)
        accepted = False

        if m > 0:
            valid = np.arange(n - m)
            mask = (h[valid] == 0) & (h[valid + m] == 1)
            if np.any(mask):
                candidates = valid[mask]
                i = candidates[rng.randint(len(candidates))]
                h[i] = 1.0
                h[i + m] = 0.0
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    stall = 0
                    accepted = True
            if not accepted:
                mask2 = (h[valid] == 1) & (h[valid + m] == 0)
                if np.any(mask2):
                    candidates = valid[mask2]
                    i = candidates[rng.randint(len(candidates))]
                    h[i] = 0.0
                    h[i + m] = 1.0
                    c5_new = compute_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        stall = 0
                        accepted = True
        elif m < 0:
            mm = -m
            valid = np.arange(mm, n)
            mask = (h[valid - mm] == 0) & (h[valid] == 1)
            if np.any(mask):
                candidates = valid[mask]
                i = candidates[rng.randint(len(candidates))]
                h[i - mm] = 1.0
                h[i] = 0.0
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    stall = 0
                    accepted = True
            if not accepted:
                mask2 = (h[valid - mm] == 1) & (h[valid] == 0)
                if np.any(mask2):
                    candidates = valid[mask2]
                    i = candidates[rng.randint(len(candidates))]
                    h[i - mm] = 0.0
                    h[i] = 1.0
                    c5_new = compute_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        stall = 0
                        accepted = True

        if not accepted:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0.0
            h[i0] = 1.0
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stall = 0
            else:
                h[i1] = 1.0
                h[i0] = 0.0
                stall += 1

    return best_h, best_c5


def cons_fun(x):
    return np.sum(x) - _cons_target


def cons_jac(x):
    return np.ones(_cons_n)


def smooth_max_obj(x, alpha, n, target):
    h = project_box_sum(x, target)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(x, alpha, n, target):
    h = project_box_sum(x, target)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-10:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad * dx


def slsqp_polish(h_start, target, time_limit, start_time):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    alphas = [200, 500, 1000, 2000, 4000, 8000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for _ in range(2):
            if time.time() - start_time > time_limit:
                break
            try:
                res = minimize(
                    smooth_max_obj, h, args=(alpha, n, target),
                    jac=lambda x: smooth_max_grad(x, alpha, n, target),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-18}
                )
                h_new = project_box_sum(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass

    return best_h, best_c5


def multi_start_subgrad(h_start, target, time_limit, start_time, rng):
    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)

    configs = [
        (0.03, 0.95, 6000),
        (0.05, 0.96, 8000),
        (0.08, 0.97, 5000),
        (0.12, 0.94, 4000),
        (0.04, 0.98, 7000),
    ]

    time_per = min(80.0, (time_limit - (time.time() - start_time)) / len(configs))
    for gamma0, thresh, pat in configs:
        if time.time() - start_time > time_limit - 30:
            break
        h_try = best_h + rng.randn(n) * 0.05
        h_try = project_box_sum(h_try, target)
        tl = time_limit - (time.time() - start_time) + time_per
        h_new, c5_new = nesterov_subgrad(h_try, target, tl, start_time,
                                         gamma0=gamma0, thresh=thresh, patience=pat)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Nesterov subgradient with multiple configs
    if tleft() > 120:
        h_sg, c5_sg = nesterov_subgrad(best_h, target, min(150.0, tleft()), start_time,
                                       gamma0=0.06, thresh=0.96, patience=8000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    # Phase 2: Binary SA
    if tleft() > 100:
        h_bin = project_binary(best_h, target)
        h_sa, c5_sa = binary_sa(h_bin, target, min(80.0, tleft()), start_time, rng)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Targeted binary local search
    if tleft() > 80:
        h_tb, c5_tb = targeted_binary_search(best_h, target, min(60.0, tleft()), start_time, rng)
        if c5_tb < best_score:
            best_score = c5_tb
            best_h = h_tb.copy()

    # Phase 4: More Nesterov subgradient (refined)
    if tleft() > 100:
        h_sg2, c5_sg2 = nesterov_subgrad(best_h, target, min(100.0, tleft()), start_time,
                                         gamma0=0.03, thresh=0.97, patience=6000)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2.copy()

    # Phase 5: Multi-start subgradient
    if tleft() > 150:
        h_ms, c5_ms = multi_start_subgrad(best_h, target, min(150.0, tleft()), start_time, rng)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 6: Second binary SA round
    if tleft() > 80:
        h_bin2 = project_binary(best_h, target)
        h_sa2, c5_sa2 = binary_sa(h_bin2, target, min(60.0, tleft()), start_time, rng,
                                  t_init=0.006, t_final=0.00002)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        if tleft() > 30:
            h_tb2, c5_tb2 = targeted_binary_search(best_h, target, min(30.0, tleft()), start_time, rng)
            if c5_tb2 < best_score:
                best_score = c5_tb2
                best_h = h_tb2.copy()

    # Phase 7: SLSQP polish
    if tleft() > 120:
        h_slq, c5_slq = slsqp_polish(best_h, target, min(180.0, tleft()), start_time)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    # Phase 8: Another round of subgradient + targeted search
    if tleft() > 80:
        h_sg3, c5_sg3 = nesterov_subgrad(best_h, target, min(60.0, tleft()), start_time,
                                         gamma0=0.02, thresh=0.98, patience=5000)
        if c5_sg3 < best_score:
            best_score = c5_sg3
            best_h = h_sg3.copy()
        if tleft() > 30:
            h_tb3, c5_tb3 = targeted_binary_search(best_h, target, min(30.0, tleft()), start_time, rng)
            if c5_tb3 < best_score:
                best_score = c5_tb3
                best_h = h_tb3.copy()

    # Phase 9: Multi-resolution
    n_candidates = []
    for factor in [0.8, 1.1, 1.25, 1.5]:
        nc = int(n * factor)
        if nc >= 200 and nc != n and nc % 2 == 0:
            n_candidates.append(nc)

    for n_try in n_candidates:
        if tleft() < 60:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)

        if tleft() > 40:
            h_sa_r, c5_sa_r = binary_sa(h_bin_try, target_try, min(25.0, tleft()), start_time, rng,
                                        t_init=0.01, t_final=0.0001)
        else:
            h_sa_r = h_bin_try
            c5_sa_r = compute_c5(h_sa_r)

        if tleft() > 30:
            h_sg_r, c5_sg_r = nesterov_subgrad(h_sa_r, float(target_try), min(25.0, tleft()), start_time,
                                                gamma0=0.05, thresh=0.96, patience=4000)
            if c5_sg_r < c5_sa_r:
                h_sa_r = h_sg_r
                c5_sa_r = c5_sg_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 10: Final SLSQP with high alpha
    if tleft() > 60:
        h_final_slq, c5_final = slsqp_polish(best_h, float(len(best_h) / 2.0),
                                              min(60.0, tleft()), start_time)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_slq.copy()

    # Phase 11: Final targeted binary search
    if tleft() > 20:
        h_ft, c5_ft = targeted_binary_search(best_h, float(len(best_h) / 2.0),
                                             min(20.0, tleft()), start_time, rng)
        if c5_ft < best_score:
            best_score = c5_ft
            best_h = h_ft.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
