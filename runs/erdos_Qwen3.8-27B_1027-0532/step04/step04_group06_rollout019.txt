```python
def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_fun_target = target
    cons_fun_n = n
    
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / cons_fun_n)
    
    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target
    
    def cons_j(h_in):
        return np.ones(cons_fun_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score
```

```python
def lse_gradient(h, n, top_k, alpha):
    corr = get_corr(h)
    center = n - 1
    # Don't exclude any lag, including center (lag 0)
    
    # Get top k lags by correlation value
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    lags = top_idx - center
    
    # Compute softmax weights (numerically stable)
    max_c = corr[top_idx[0]]
    exp_vals = np.exp(alpha * (corr[top_idx] - max_c))
    weights = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for w, k in zip(weights, lags):
        if k == 0:
            # dc_0/dh[j] = 1 - 2*h[j]
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            k_abs = -k
            grad[k_abs:] += w * (1.0 - h[:n-k_abs])
            grad[:n-k_abs] -= w * h[k_abs:]
    
    return grad * (2.0 / n)
```

This is cleaner. The lag 0 term naturally encourages binary-ness, while the other lags encourage correlation matching.
