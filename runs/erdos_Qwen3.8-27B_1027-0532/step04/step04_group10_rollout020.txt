        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Get top n_active lags

        grad = np.zeros(n)
        for idx in top_idx:
            grad += gradient_from_lag(best_h, idx, n)
        grad /= n_active
        
        # Adaptive learning rate
        # Use gradient descent with backtracking

                best_score = sc
                best_h = h_try
                improved = True
                break

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

        return project_feasible(h_start, target)

        _, g = lse_obj_grad(x, alpha, topk)
        return g

                       options={'maxiter': maxiter, 'ftol': 1e-14})

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad


def lbfgs_optimize(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol, m=20):
    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        return f + penalty * s * s, g + 2.0 * penalty * s
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12, 'maxls': 50, 'm': m})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def slsqp_optimize(h_start, target, alpha, topk, maxiter):
    n = len(h_start)

    def obj_func(x):
        f, _ = lse_obj_grad(x, alpha, topk)
        return f

    def obj_jac(x):
        _, g = lse_obj_grad(x, alpha, topk)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_start, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def active_set_gd(h, target, n_active, n_iters, base_lr):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argpartition(corr, -n_active)[-n_active:]
        grad = np.zeros(n)
        omh = 1.0 - best_h
        for ti in top_idx:
            lag = ti - (n - 1)
            if lag >= 0 and lag < n:
                grad[lag:] += omh[:n - lag]
                grad[:n - lag] -= best_h[lag:]
            elif lag < 0:
                m = -lag
                if m < n:
                    grad[:n - m] += omh[m:]
                    grad[m:] -= best_h[:n - m]
        grad /= n_active
        lr = base_lr * (1.0 + 0.5 * it)
        h_try = best_h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_try
        else:
            lr2 = lr * 0.1
            h_try2 = best_h - lr2 * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < best_score - 1e-14:
                best_score = sc2
                best_h = h_try2
    return best_h, best_score
