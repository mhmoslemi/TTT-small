import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-13)[0]
        else:
            idx = np.where(h > 1e-13)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def find_t_star(a_k, b_k):
    min_a = float(np.min(a_k))
    max_a = float(np.max(a_k))
    if min_a >= -1e-15:
        return 0.0
    if max_a <= 1e-15:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a_k * mid + b_k
        k_max = int(np.argmax(vals))
        s = a_k[k_max]
        if s > 1e-14:
            hi = mid
        elif s < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h_init, n, target, max_sweeps, rng, time_limit, start_time):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    corr = get_corr_array(h)
    best_h = h.copy()
    best_c5 = float(np.max(corr))
    k_range = np.arange(-(n - 1), n)
    n_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            valid_ik = (i - k_range >= 0) & (i - k_range < n)
            valid_ipk = (i + k_range >= 0) & (i + k_range < n)
            a_k = np.zeros(2 * n - 1)
            if np.any(valid_ik):
                a_k[valid_ik] = dx * (1.0 - h[i - k_range[valid_ik]])
            if np.any(valid_ipk):
                a_k[valid_ipk] -= dx * h[i + k_range[valid_ipk]]
            b_k = corr - h[i] * a_k
            old_val = h[i]
            t_star = find_t_star(a_k, b_k)
            if abs(t_star - old_val) > 1e-15:
                delta = t_star - old_val
                h[i] = t_star
                corr += delta * a_k
        h_old_sum = np.sum(h)
        h = project_feasible(h, target)
        if abs(np.sum(h) - h_old_sum) > 1e-10:
            corr = get_corr_array(h)
        c5_val = float(np.max(corr))
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            n_no_improve = 0
        else:
            n_no_improve += 1
        if n_no_improve > 200 and sweep > 50:
            break
    return best_h, best_c5
