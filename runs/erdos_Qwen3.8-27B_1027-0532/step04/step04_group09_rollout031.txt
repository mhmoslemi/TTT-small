import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_init(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n / 2.0)

def structured_init(n, rng):
    # Try different structured inits
    idx = rng.randint(0, 5)
    if idx == 0:
        # Step function: first half 1, second half 0 (or vice versa)
        h = np.zeros(n)
        h[:n//2] = 1.0
    elif idx == 1:
        # Sine
        x = np.linspace(0, 2*np.pi, n)
        h = 0.5 + 0.5*np.sin(x)
    elif idx == 2:
        # Two bumps
        x = np.linspace(0, 1, n)
        h = 0.5 + 0.5*np.sin(2*x)
    elif idx == 3:
        # Random blocks
        block = rng.randint(4, 10)
        h = np.zeros(n)
        for start in range(0, n, block):
            if rng.random() > 0.5:
                h[start:start+block] = 1.0
    else:
        # Triangle
        x = np.linspace(0, 1, n)
        h = 1.0 - 2.0*abs(x - 0.5)
    return project_feasible(h, n/2.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            grad[:m] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:m]
        else:
            kk = -k
            m = n - kk
            grad[kk:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[kk:]
    return obj, grad

def slsqp_optimize(h, target, alpha, maxiter=1000, ftol=1e-12):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                    'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except:
        return h

def multi_alpha_refine(h, target, alphas, maxiter, ftol, deadline):
    h_cur = h.copy()
    for a in alphas:
        if time.time() > deadline:
            break
        h_new = slsqp_optimize(h_cur, target, a, maxiter, ftol)
        if compute_c5(h_new) < compute_c5(h_cur):
            h_cur = h_new
    return h_cur

def subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = np.max(corr)
        c_min = np.min(corr)
        range_c = c_max - c_min
        if range_c < 1e-10:
            break
        threshold = c_max - active_frac * range_c
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                m = n - k
                grad[:m] -= h_cur[k:]
                grad[k:] += (1.0 - h_cur[:m])
            else:
                kk = -k
                m = n - kk
                grad[kk:] -= h_cur[:m]
                grad[:m] += (1.0 - h_cur[kk:])
        grad /= len(active)
        
        step = step_size / (1.0 + 0.01 * it)
        h_new = project_feasible(h_cur - step * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    x_old = np.linspace(0, 1, len(h_src))
    x_new = np.linspace(0, 1, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_descent(h, target, block_size, n_rounds, alpha, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    bounds = [(0.0, 1.0)] * block_size
    
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            
            def block_obj(x):
                ht = h_cur.copy()
                ht[s:e] = np.clip(x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                return lse_obj_and_grad(ht, alpha)[0]
            
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(block_obj, x0, bounds=bounds, method='L-BFGS-B',
                             options={'maxiter': 200, 'ftol': 1e-12})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur_score:
                    h_cur = ht
                    cur_score = sc
            except:
                pass
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def random_init(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return project_feasible(h, n / 2.0)


def mixed_init(n, rng):

def lse_obj_raw(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)

    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    return obj, w

    obj, _ = lse_obj_raw(x, alpha)
    cv = float(np.sum(x)) - target
    return obj + penalty * cv * cv

    n = len(x)
    _, w = lse_obj_raw(x, alpha)
    cv = float(np.sum(x)) - target
    grad = 2.0 * penalty * cv * np.ones(n)
    for idx in range(2 * n - 1):
        wi = w[idx]

        if k >= 0:
            m = n - k

            kk = -k
            m = n - kk
            grad[kk:] += wi * (1.0 - x[:m])
            grad[:m] -= wi * x[kk:]
    return grad


def slsqp_solve(h, target, alpha, penalty, maxiter, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    f = partial(obj_fn, alpha=alpha, target=target, penalty=penalty)
    g = partial(grad_fn, alpha=alpha, target=target, penalty=penalty)
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def multi_alpha_refine(h, target, alphas, penalty, maxiter, ftol, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        h_new = slsqp_solve(h_cur, target, a, penalty, maxiter, ftol)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
    return h_cur, cur_score


def subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))

        if range_c < 1e-12:
            break
        threshold = c_max - active_frac * range_c
        active = np.where(corr >= threshold)[0]

                m = n - k
                grad[:m] -= h_cur[k:]

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, n_dst / 2.0)
