def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[:final_sum - target_sum]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[:target_sum - final_sum]] = 1
    return h_r.astype(float)


def swap_delta_full(h, i, j, n):
    p_lo_i = max(0, i - (n - 1))
    p_hi_i = min(n - 1, i)
    m_i = i - np.arange(p_lo_i, p_hi_i + 1) + (n - 1)
    p_lo_j = max(0, j - (n - 1))
    p_hi_j = min(n - 1, j)
    m_j = j - np.arange(p_lo_j, p_hi_j + 1) + (n - 1)

    delta[m_i] -= (1.0 - h[p_lo_i:p_hi_i + 1])
    delta[m_j] += (1.0 - h[p_lo_j:p_hi_j + 1])

                break

        max_val = float(corr.max())
        top_idx = np.argsort(corr)[-top_k:][::-1]
        top_vals = corr[top_idx].copy()

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _cons_target

    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol,
                 time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def diverse_init(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for i in range(ones_count):
        frac = ((i + 1) / golden) % 1.0
        positions.add(int(frac * n))
    positions = sorted(positions)
    while len(positions) < ones_count:
        cands = [p for p in range(n) if p not in positions]
        if not cands:
            break
        positions.append(cands[rng.randint(len(cands))])
        positions = sorted(set(positions))
    positions = positions[:ones_count]
    h[positions] = 1.0
    return h

        h_sa_f, c5_f = sa_bottleneck(h_bin_final, target_use, max_it,
                                     0.005, 0.00002, rng,
                                     time_limit=max(5.0, tleft() - 12),
                                     start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 9: Final SLSQP
    if tleft() > 12:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, [3000, 8000, 15000],
                                 1000, 1e-15,
                                 time_limit=max(4.0, tleft() - 6),
                                 start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
