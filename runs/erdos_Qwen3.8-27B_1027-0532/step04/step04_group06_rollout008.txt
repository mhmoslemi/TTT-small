    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_R(h):
    return np.correlate(h, h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h).clip(0, 1).astype(int)
    s = int(h_r.sum())
    if s == target_sum:
        return h_r.astype(float)
    if s > target_sum:
        need = s - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - s
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smooth_objective_and_grad(h, alpha):
    """Compute F_alpha(h) and its gradient."""
    n = len(h)
    R = get_R(h)
    R_min = R.min()
    shifted = -alpha * (R - R_min)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    S = exp_s.sum()
    p = exp_s / S
    obj_val = -R_min + np.log(S) / alpha
    
    # Gradient: -sum_k p[k] * (h[j-k] + h[j+k])
    grad = np.zeros(n)
    center = n - 1
    # Only iterate over significant p values
    thresh = 1e-10
    for idx in range(len(p)):
        if p[idx] < thresh:
            continue
        lag = idx - center
        w = p[idx]
        if lag == 0:
            grad += 2.0 * w * h
        elif lag > 0:
            grad[:n - lag] += w * h[lag:]
            grad[lag:] += w * h[:n - lag]
        else:
            lag = -lag
            grad[:n - lag] += w * h[lag:]
            grad[lag:] += w * h[:n - lag]
    grad = -grad
    return obj_val, grad

def slsqp_step(h_start, n, target, alpha, maxiter=500, ftol=1e-12):
    def obj(h_in):
        R = get_R(h_in)
        R_min = R.min()
        shifted = -alpha * (R - R_min)
        shifted = np.clip(shifted, -700, 0)
        return float(-R_min + np.log(np.exp(shifted).sum()) / alpha)
    
    def obj_grad(h_in):
        n_l = len(h_in)
        R = get_R(h_in)
        R_min = R.min()
        shifted = -alpha * (R - R_min)
        shifted = np.clip(shifted, -700, 0)
        exp_s = np.exp(shifted)
        S = exp_s.sum()
        p = exp_s / S
        grad = np.zeros(n_l)
        center = n_l - 1
        for idx in range(len(p)):
            if p[idx] < 1e-10:
                continue
            lag = idx - center
            w = p[idx]
            if lag == 0:
                grad += 2.0 * w * h_in
            elif lag > 0:
                grad[:n_l - lag] += w * h_in[lag:]
                grad[lag:] += w * h_in[:n_l - lag]
            else:
                lag = -lag
                grad[:n_l - lag] += w * h_in[lag:]
                grad[lag:] += w * h_in[:n_l - lag]
        return -grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def targeted_binary_sa(h_start, target_sum, max_iters, rng, time_limit, start_time):

        if k_worst == 0:
            k_worst = 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        # Try a few random pairs
        improvements = 0
        for _ in range(20):
            a = ones[rng.randint(len(ones))]

            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                improvements += 1
            else:
                h[a] = 1.0
                h[b] = 0.0
        
        if improvements == 0:
            # Try targeted moves
            # For the worst lag, find a that "damages" R(k_worst) most
            a_dmg = np.zeros(len(ones))
            for i, a in enumerate(ones):
                d = 0
                if 0 <= a + k_worst < n and h[a + k_worst] == 1:
                    d += 1
                if 0 <= a - k_worst < n and h[a - k_worst] == 1:
                    d += 1
                a_dmg[i] = d
            
            # Find b that "helps" R(k_worst) most
            b_gain = np.zeros(len(zeros))
            for i, b in enumerate(zeros):
                g = 0
                if 0 <= b + k_worst < n and h[b + k_worst] == 1:
                    g += 1
                if 0 <= b - k_worst < n and h[b - k_worst] == 1:
                    g += 1
                b_gain[i] = g
            
            # Try the best (a with max damage, b with max gain)
            a_best = ones[np.argmax(a_dmg)]
            b_best = zeros[np.argmax(b_gain)]

                best_h = h.copy()

                h[a_best] = 1.0
                h[b_best] = 0.0

Hmm, I realize this binary SA is not very different from the previous one. Let me think of something more novel.

**New idea: Use the structure of the problem more directly.**

The key insight is that C5 = 1 - (2/n) * min_k R(k). So we want to maximize min_k R(k).

For a given h, the "bottleneck lag" k* = argmin R(k). The constraint R(k*) ≥ t can be written as:
Σ_i h[i]h[i+k*] ≥ t

This is a quadratic constraint. If we fix all variables except a small set, it becomes linear in the remaining variables.

**Idea: Coordinate descent on the "bottleneck" constraint.**

1. Fix all of h except a few coordinates.
2. For the bottleneck lag k*, the constraint Σ_i h[i]h[i+k*] ≥ t becomes linear in the free coordinates.
3. Optimize the free coordinates to increase t.
4. Update k* and repeat.

This is essentially a cutting-plane / column-generation approach.

But implementing this efficiently is complex. Let me think of something simpler.

**Simpler idea: Alternating minimization.**

Define the problem as: maximize t subject to R(k) ≥ t for all k, Σh = n/2, 0 ≤ h ≤ 1.

This is a non-convex QCQP (quadratically constrained quadratic program). We can solve it approximately by:

1. Fix k* (the active constraint). Solve: maximize t subject to R(k*) ≥ t, Σh = n/2, 0 ≤ h ≤ 1.
   This is equivalent to maximizing R(k*) = Σ_i h[i]h[i+k*].
   For fixed k*, this is maximize Σ_i h[i]h[i+k*] s.t. Σh = n/2, 0 ≤ h ≤ 1.
   
   This is still non-convex (product of variables). But we can use:
   - Let A(i) = h[i], B(i) = h[i+k*]. Maximize Σ A(i)*B(i) s.t. constraints.
   - This is a "complementarity" problem: we want A and B to be simultaneously large.

2. After optimizing for k*, update k* = argmin R(k) and repeat.

For step 1, we can use a simple heuristic: for the pairs (i, i+k*), set h[i] and h[i+k*] to be "opposite" of what they need to be... actually, we want them both large. So we should increase h[i] and h[i+k*] simultaneously.

A practical approach:
- For the bottleneck lag k*, identify pairs (i, i+k*) where at least one is small.
- Increase the smaller one (while respecting the sum constraint by decreasing something else).

This is essentially a coordinate ascent on R(k*).

Let me formalize:
- Current h with sum = n/2.
- k* = argmin R(k).
- For each pair (i, i+k*) with i+k* < n:
  - If h[i] < h[i+k*]: increase h[i] by ε, decrease some other h[j] by ε
  - If h[i+k*] < h[i]: increase h[i+k*] by ε, decrease some other h[j] by ε
  - Choose j to minimize the damage to other R(k) values.

The "damage" of decreasing h[j] by ε to R(k) is: ε * (h[j-k] + h[j+k]).

So the best j to decrease is the one with minimum (h[j-k] + h[j+k]) for the current k* (and other active lags).
