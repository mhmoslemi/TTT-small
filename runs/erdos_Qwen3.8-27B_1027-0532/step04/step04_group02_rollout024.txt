```python
def project_sum_clip(h, target_sum):
    """Project h onto {sum(h) = target_sum, 0 <= h <= 1}"""
    n = len(h)
    # Bisection on lambda
    lo = h.min() - target_sum
    hi = h.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any tiny numerical error
    diff = target_sum - h_proj.sum()
    if abs(diff) < 1e-10:
        return h_proj
    # Distribute the error
    if diff > 0:
        room = 1.0 - h_proj
        order = np.argsort(-room)[:int(np.ceil(abs(diff)))+1]
        for i in order:
            add = min(room[i], diff)
            h_proj[i] += add
            diff -= add
    else:
        room = h_proj.copy()
        order = np.argsort(room)[:int(np.ceil(abs(diff)))+1]
        for i in order:
            sub = min(room[i], -diff)
            h_proj[i] -= sub
            diff += sub
    return h_proj
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_binary(h, target):
    n = len(h)
    target = int(target)
    h_r = np.round(h).astype(int)
    s = int(h_r.sum())
    if s == target:
        return h_r.astype(float)
    if s > target:
        need = s - target
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target - s
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    s2 = int(h_r.sum())
    if s2 > target:
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:s2 - target]] = 0
    elif s2 < target:
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:target - s2]] = 1
    return h_r.astype(float)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    lo = -h.sum() - 1
    hi = n * (1.0 - h.mean()) + 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - h_p.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            room = 1.0 - h_p
            idx = np.argsort(-room)
            for i in idx:
                if residual <= 1e-13:
                    break
                a = min(room[i], residual)
                h_p[i] += a
                residual -= a
        else:
            idx = np.argsort(h_p)
            for i in idx:
                if -residual <= 1e-13:
                    break
                a = min(h_p[i], -residual)
                h_p[i] -= a
                residual += a
    return h_p

def smooth_grad(h, alpha, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ln = len(corr)
    if top_k is not None and top_k < ln:
        top_idx = np.argsort(-corr)[-top_k:]
        w = np.zeros(ln)
        cv = corr[top_idx]
        cv = cv - cv.max()
        w[top_idx] = np.exp(alpha * cv)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[top_idx] = 1.0 / len(top_idx)
        else:
            w /= w_sum
    else:
        cv_shifted = corr - corr.max()
        w = np.exp(alpha * cv_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[:] = 1.0 / ln
        else:
            w /= w_sum
    grad = np.zeros(n)
    for idx in range(ln):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nk = n - k
            grad[:nk] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:nk]
        else:
            jk = -k
            njk = n - jk
            grad[jk:] += wk * (1.0 - h[:njk])
            grad[:njk] -= wk * h[jk:]
    return grad

def smooth_optimize(h0, target, time_budget, start_time, alpha_init=50.0, alpha_final=3000.0):
    n = len(h0)
    h = project_sum(h0.copy(), target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    n_corr = 2 * n - 1
    top_k = max(20, n // 5)
    gamma = 1.0
    max_iter = 5000
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > time_budget:
            break
        frac = it / max(max_iter - 1, 1)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        g = smooth_grad(h, alpha, top_k=top_k)
        step = gamma * (2.0 / n) * 0.5
        h_new = project_sum(h - step * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
            gamma = min(gamma * 1.1, 5.0)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h.copy()
        else:
            gamma = max(gamma * 0.7, 1e-3)
    return h_best, c5_best

    target_use = n_use / 2.0
    if tleft() > 15:
        h_sm_end, c5_end = smooth_optimize(best_h, target_use, max(5.0, tleft() - 8), start_time,
                                           alpha_init=200.0, alpha_final=8000.0)
        if c5_end < best_c5:
            best_c5 = c5_end
            best_h = h_sm_end.copy()

    h_final = project_sum(best_h, target_use)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5 + 1e-12:
        h_final = best_h
        final_c5 = best_c5
    return h_final, final_c5, len(h_final)
```
