```python
    else:
        idx = np.where(h > 0.0)[0]

    h = h.copy()
    h[idx] += diff / len(idx)

    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]

        need = int(current_sum - target_sum)
```

This means j should be a position where:
- h[j+k] = 0 (or small)
- h[j-k] = 1 (or large)

We can score each candidate j by:
score(j) = ∑_{k in active_lags} w_k * (h[j+k] - (1-h[j-k]))

And pick the j with the most negative score (most damaging to active lags when turned off, meaning it helps most).

```python
def score_partner(j, h, active_lags, weights, n):
    """Score how good j is as a partner to turn off (1->0).
    More negative = better (turning j off helps more)."""
    score = 0.0
    for idx, k in enumerate(active_lags):
        k = int(k)
        w = weights[idx]
        # Δcorr_j(k) for turning j from 1 to 0:
        # = h[j+k] - (1-h[j-k]) (if we define it as the change)
        # Wait, for turning j from 1 to 0:
        # (1-h[j]) goes from 0 to 1: GAINS h[j+k]
        # h[j] in h[j+k] position: j+k = i means i=j+k, term is h[j+k]*(1-h[j+k])... no
        # Let me re-derive.
        # Turning j from 1 to 0:
        # (1-h[j]): was 0, now 1. Contributes +h[j+k] to corr(k) if 0≤j+k<n.
        # h[j] appears in h[i+k] when i+k=j, i=j-k. h[j] was 1, now 0.
        #   Term h[j]*(1-h[j-k]) goes from (1-h[j-k]) to 0. Change: -(1-h[j-k]).
        # Total Δ for turning j from 1 to 0:
        # Δ(k) = h[j+k]*[0≤j+k<n] - (1-h[j-k])*[0≤j-k<n]
        
        val = 0.0
        if 0 <= j + k < n:
            val += h[j + k]
        if 0 <= j - k < n:
            val -= (1.0 - h[j - k])
        score += w * val
    return score
```

Similarly, for choosing which position i to turn on (0→1):
Δ_i(k) = (1-h[i-k])*[0≤i-k<n] - h[i+k]*[0≤i+k<n]

We want Δ_i(k) to be very negative for the active lags (turning i on helps a lot).

score_i = ∑ w_k * Δ_i(k)

We want the most negative score_i (turning i on helps the active lags the most).

But we also need to consider the boundary: turning i on might HURT other lags. For a good pair (i,j):
Total change at lag k: Δ_i(k) + Δ_j(k)
We want max_k (corr[k] + Δ_i(k) + Δ_j(k)) to be minimized.

For a quick heuristic:
1. Find the i with the most negative score_i (best position to turn on)
2. Find the j with the most negative score_j (best position to turn off)
3. Check if the pair (i,j) actually improves the max
4. If not, try the next-best candidates

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def score_positions_on(h, active_lags, weights, n):
    score = np.zeros(n)
    for idx in range(len(active_lags)):
        k = int(active_lags[idx])
        w = weights[idx]
        if w < 1e-12:
            continue
        if k >= 0:
            if n - k > 0:
                score[k:] += w * (1.0 - h[:n-k])
                score[:n-k] -= w * h[k:]
        else:
            ka = -k
            if n - ka > 0:
                score[:n-ka] += w * (1.0 - h[ka:])
                score[ka:] -= w * h[:n-ka]
    return score

def directed_binary_search(h_start, target, max_iters, time_limit, start_time, rng, n_active=8, shortlist=40):
    n = len(h_start)
    dx = 2.0 / n
    h = project_binary(h_start, target)
    corr = get_corr(h)
    center = n - 1
    current_c5 = float(np.max(corr)) * dx
    best_c5 = current_c5
    best_h = h.copy()
    T = 0.3
    T_min = 0.01
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 8000:
            T = 0.4
            no_improve = 0
            for _ in range(30):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                h[i], h[j] = 1.0, 0.0
            corr = get_corr(h)
            current_c5 = float(np.max(corr)) * dx
            continue

        corr_work = corr.copy()
        corr_work[center] = -1e10
        top_idx = np.argsort(corr_work)[::-1][:n_active]
        active_lags = top_idx - center
        active_vals = corr_work[top_idx]
        cmax = active_vals[0]

        sw = np.exp(100.0 * (active_vals - cmax))
        sw /= sw.sum()

        score = score_positions_on(h, active_lags, sw, n)

        ones_mask = (h == 1)
        zeros_mask = (h == 0)

        zeros_scored = np.where(zeros_mask)[0]
        ones_scored = np.where(ones_mask)[0]

        if len(zeros_scored) == 0 or len(ones_scored) == 0:
            break

        i_cands = zeros_scored[np.argsort(score[zeros_scored])[:shortlist]]
        j_cands = ones_scored[np.argsort(-score[ones_scored])[:shortlist]]

        best_pair_score = current_c5
        best_i = -1
        best_j = -1
        best_corr = None

        n_i_try = min(len(i_cands), 12)
        n_j_try = min(len(j_cands), 12)

        for ii in range(n_i_try):
            i = i_cands[ii]
            for jj in range(n_j_try):
                j = j_cands[jj]
                if j == i:
                    continue
                h[i] = 1.0
                h[j] = 0.0
                nc = get_corr(h)
                nv = float(np.max(nc)) * dx
                h[i] = 0.0
                h[j] = 1.0
                if nv < best_pair_score - 1e-15:
                    best_pair_score = nv
                    best_i = i
                    best_j = j
                    best_corr = nc.copy()

        if best_i >= 0 and best_corr is not None:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = best_corr
            current_c5 = best_pair_score
            if current_c5 < best_c5 - 1e-12:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve = 0
                T = max(T * 0.995, T_min)
            else:
                delta = (current_c5 - best_c5) / dx
                if rng.random() < np.exp(-min(delta / T, 50.0)):
                    no_improve += 1
                    T = max(T * 0.998, T_min)
                else:
                    h[best_i] = 0.0
                    h[best_j] = 1.0
                    corr = get_corr(h)
                    current_c5 = float(np.max(corr)) * dx
                    no_improve += 1
        else:
            i = zeros_scored[rng.randint(len(zeros_scored))]
            j = ones_scored[rng.randint(len(ones_scored))]
            h[i] = 1.0
            h[j] = 0.0
            corr = get_corr(h)
            current_c5 = float(np.max(corr)) * dx
            delta = (current_c5 - best_c5) / dx
            if delta < 0 or rng.random() < np.exp(-min(delta / T, 50.0)):
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                T = max(T * 0.998, T_min)
            else:
                h[i] = 0.0
                h[j] = 1.0
                corr = get_corr(h)
                no_improve += 1

    return best_h, best_c5

def smooth_grad_descent(h_start, target, alphas, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = project_feasible(h_start.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    center = n - 1

    for alpha in alphas:
        step = 2.0 / alpha
        for it in range(5000):
            if time.time() - start_time > time_limit:
                break
            corr = get_corr(h) * dx
            corr[center] = -1e10
            c_max = np.max(corr)
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500, 0)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(wsum)] = 1.0
            else:
                weights /= wsum

            lags = np.arange(len(corr)) - center
            grad = np.zeros(n)
            for il in range(len(lags)):
                w = weights[il]
                if w < 1e-10:
                    continue
                k = int(lags[il])
                if k >= 0:
                    if n - k > 0:
                        grad[k:] += w * (1.0 - h[:n-k])
                        grad[:n-k] -= w * h[k:]
                else:
                    ka = -k
                    if n - ka > 0:
                        grad[:n-ka] += w * (1.0 - h[ka:])
                        grad[ka:] -= w * h[:n-ka]

            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            h = h_new
            step *= 0.9995

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    def _obj(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def _cons_f(h_in):
        return np.sum(h_in) - target

    def _cons_j(h_in):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_obj, h_cur, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
```
