Now, for the continuous refinement, I'll use a subgradient-based approach:
- The objective is f(h) = max_k corr_k * dx
- Subgradient: ∂f/∂h[i] = dx * (1/n_active) * Σ_{k active} ∂corr_k/∂h[i]

where "active" means corr_k is within ε of the max.

For the continuous case (h not necessarily binary):
∂corr_s/∂h[i] = (1-h[i+s]) * I[i+s valid] - h[i-s] * I[i-s valid, i-s≠i]

Wait, for continuous h:
corr_s = Σ_i h[i]*(1-h[i+s]) (for valid i, i+s)

∂corr_s/∂h[i] = (1-h[i+s]) * I[i+s valid] - h[i-s] * I[i-s valid]

The second term: when h[i] appears in (1-h[i-s')s term, i.e., when some h[p]*(1-h[i]) where p+s'=i, so p = i-s'.
∂/∂h[i] of h[i-s']*(1-h[i]) = -h[i-s']

So ∂corr_s/∂h[i] = (1-h[i+s])*I[i+s valid] - h[i-s]*I[i-s valid]

And the subgradient of max_k corr_k at h is:
(1/|A|) * Σ_{k∈A} ∂corr_k/∂h where A = {k : corr_k ≥ max - ε}

This gives us a direction to move h in the negative gradient direction (projected onto the feasible set).

For the continuous case with constraint Σh = n/2 and 0≤h≤1:
- Compute subgradient direction
- Take a step in that direction
- Project back onto feasible set

This is essentially the Frank-Wolfe (conditional gradient) method or projected subgradient.

Let me implement a more practical version:

```python
def subgradient_refine(h, max_iters, step_sizes, time_limit=None, start_time=None):
    n = len(h)
    h = project_feasible(h.copy(), n/2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        corr = compute_corr(h)
        corr_max = float(np.max(corr))
        current_c5 = corr_max * (2.0/n)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Active set: top correlations
        threshold = corr_max - 0.01 / n  # small epsilon
        active = np.where(corr >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            s = k - (n-1)  # shift
            for i in range(n):
                # First term: (1-h[i+s]) if i+s valid
                if 0 <= i+s < n:
                    grad[i] += (1.0 - h[i+s])
                # Second term: -h[i-s] if i-s valid
                if 0 <= i-s < n:
                    grad[i] -= h[i-s]
        
        grad /= len(active)
        
        # Step
        alpha = step_sizes[min(it, len(step_sizes)-1)]
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n/2.0)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
        else:
            # Halve step size or revert
            h = h
    
    return best_h, best_c5
```

This is O(n² * |active|) per iteration which is too slow for n=600. Let me vectorize the gradient computation.

The gradient for a single shift s:
grad_s[i] = (1-h[i+s])*I[0≤i+s<n] - h[i-s]*I[0≤i-s<n]

This can be computed with numpy slicing:
```python
def shift_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        # i+s valid: i in [0, n-1-s]
        grad[:n-s] += (1.0 - h[s:])
        # i-s valid: i-s in [0,n-1] → i in [s, n-1]
        grad[s:] -= h[:n-s]
    else:
        s_abs = -s
        # i+s = i-s_abs valid: i in [s_abs, n-1]
        grad[s_abs:] += (1.0 - h[:n-s_abs])
        # i-s = i+s_abs valid: i in [0, n-1-s_abs]
        grad[:n-s_abs] -= h[s_abs:]
    return grad
```

Then the total gradient is:
grad = (1/|A|) * Σ_{k∈A} shift_gradient(h, k-(n-1))

For |A| active shifts, this is O(n * |A|) per iteration. If |A| is small (say 10-50), this is fast.

But if |A| is large (say 100+), it gets expensive. We can limit |A| to the top K correlations.

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)


def compute_D_fft(h, w, n):
    g = 1.0 - 2.0 * h
    N = 1
    while N < 3 * n:
        N *= 2
    G = np.fft.rfft(g, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(G * np.conj(W), N)
    D = R[(np.arange(n) - (n - 1)) % N]
    return D


def pressure_search_binary(h, betas, max_iters, tabu_len=30, rng=None,
                           time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tabu_i = {}
    tabu_j = {}
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        beta = betas[min(it // 200, len(betas) - 1)] if len(betas) > 1 else betas[0]

        corr = compute_corr(h)
        corr_max = float(np.max(corr))
        current_c5 = corr_max * (2.0 / n)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        w = np.exp(beta * (corr - corr_max))
        D = compute_D_fft(h, w, n)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        D_ones = D[ones].copy()
        D_zeros = D[zeros].copy()

        for ti, t_iter in list(tabu_i.items()):
            if it - t_iter < tabu_len:
                mask = ones == ti
                if np.any(mask):
                    D_ones[mask] = -np.inf
            else:
                del tabu_i[ti]
        for tj, t_iter in list(tabu_j.items()):
            if it - t_iter < tabu_len:
                mask = zeros == tj
                if np.any(mask):
                    D_zeros[mask] = np.inf
            else:
                del tabu_j[tj]

        if np.all(np.isinf(D_ones)) or np.all(np.isinf(D_zeros)):
            i_best = ones[rng.randint(len(ones))]
            j_best = zeros[rng.randint(len(zeros))]
        else:
            i_best = ones[np.argmax(D_ones)]
            j_best = zeros[np.argmin(D_zeros)]

        h_old_max = corr_max
        h[i_best] = 0.0
        h[j_best] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))

        if new_max > h_old_max * 1.005:
            h[i_best] = 1.0
            h[j_best] = 0.0
            n_try = min(10, len(ones) * len(zeros))
            for _ in range(n_try):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h[ii] = 0.0
                h[jj] = 1.0
                nc = compute_corr(h)
                nm = float(np.max(nc))
                if nm <= h_old_max:
                    break
                h[ii] = 1.0
                h[jj] = 0.0

        tabu_i[i_best] = it
        tabu_j[j_best] = it

    return best_h, best_c5


def annealed_pressure(h, max_iters, time_limit=None, start_time=None, rng=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tabu_i = {}
    tabu_j = {}
    tabu_len = 20
    stall = 0
    max_stall = 400

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        ratio = it / float(max(max_iters - 1, 1))
        beta = 80.0 * (1.0 - ratio) + 5.0 * ratio

        corr = compute_corr(h)
        corr_max = float(np.max(corr))
        current_c5 = corr_max * (2.0 / n)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        w = np.exp(beta * (corr - corr_max))
        D = compute_D_fft(h, w, n)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        D_ones = D[ones].copy()
        D_zeros = D[zeros].copy()

        for ti, t_iter in list(tabu_i.items()):
            if it - t_iter < tabu_len:
                mask = ones == ti
                if np.any(mask):
                    D_ones[mask] = -np.inf
            else:
                del tabu_i[ti]
        for tj, t_iter in list(tabu_j.items()):
            if it - t_iter < tabu_len:
                mask = zeros == tj
                if np.any(mask):
                    D_zeros[mask] = np.inf
            else:
                del tabu_j[tj]

        if np.all(~np.isfinite(D_ones)) or np.all(~np.isfinite(D_zeros)):
            i_best = ones[rng.randint(len(ones))]
            j_best = zeros[rng.randint(len(zeros))]
        else:
            top_k = min(5, int(np.sum(np.isfinite(D_ones))))
            top_i_idx = np.argsort(D_ones)[::-1][:top_k]
            top_j_idx = np.argsort(D_zeros)[:top_k]

            best_delta = np.inf
            i_best = ones[top_i_idx[0]]
            j_best = zeros[top_j_idx[0]]
            for ii in range(min(top_k, len(top_i_idx))):
                for jj in range(min(top_k, len(top_j_idx))):
                    delta = D_zeros[top_j_idx[jj]] - D_ones[top_i_idx[ii]]
                    if delta < best_delta:
                        best_delta = delta
                        i_best = ones[top_i_idx[ii]]
                        j_best = zeros[top_j_idx[jj]]

        h[i_best] = 0.0
        h[j_best] = 1.0
        tabu_i[i_best] = it
        tabu_j[j_best] = it

    return best_h, best_c5


def sa_binary(h, target_sum, max_iters, t_init, t_final, rng,
              time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 6000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        ratio = it / float(max(max_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        move_type = rng.randint(5)
        h_new = h.copy()
        moved = False

        if move_type <= 2:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 3:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                h_new[i1] = 0
                h_new[i2] = 0
                h_new[j1] = 1
                h_new[j2] = 1
                moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def smooth_obj(h, alpha, target_sum):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol,
                 time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def subgradient_refine(h, max_iters, time_limit=None, start_time=None):
    n = len(h)
    h = project_feasible(h.copy(), n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0

    base_step = 0.5 / n

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > 500:
            break

        corr = compute_corr(h)
        corr_max = float(np.max(corr))
        current_c5 = corr_max * (2.0 / n)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        eps = 0.05 / n
        active = np.where(corr >= corr_max - eps)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] -= h[:n - s]
            else:
                sa = -s
                grad[sa:] += (1.0 - h[:n - sa])
                grad[:n - sa] -= h[sa:]

        grad /= float(len(active))
        grad -= np.mean(grad)

        step = base_step / (1.0 + 0.01 * it)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n / 2.0)

        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
        else:
            h = project_feasible(h - 0.1 * step * grad, n / 2.0)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        allowed = min(120, tleft() * 0.2)
        betas = [50, 30, 15, 8, 5]
        h_p4, c5_p4 = pressure_search_binary(h_bin, betas, max_iters=int(allowed * 1000),
                                             tabu_len=25, rng=rng,
                                             time_limit=allowed, start_time=start_time)
        if c5_p4 < best_score:
            best_score = c5_p4
            best_h = h_p4.copy()

    # Phase 5: Multi-restart at coarse level with random perturbations
    if tleft() > 80:
        for restart in range(3):
            if tleft() < 30:
                break
            h_c = interpolate_to(best_h, n_coarse, n_coarse // 2)
            h_c_bin = project_binary(h_c, n_coarse // 2)
            # Perturb
            n_pert = 15
            ones = np.where(h_c_bin == 1)[0]
            zeros = np.where(h_c_bin == 0)[0]
            for _ in range(n_pert):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_c_bin[i] = 0
                h_c_bin[j] = 1
                ones = np.where(h_c_bin == 1)[0]
                zeros = np.where(h_c_bin == 0)[0]

            allowed = min(40, tleft() * 0.15)
            h_r, c5_r = annealed_pressure(h_c_bin, max_iters=int(allowed * 2000),
                                          time_limit=allowed, start_time=start_time, rng=rng)
            c5_r_full = c5_r * (2.0 / n_coarse)
            if c5_r_full < best_score * 1.01:
                h_interp = interpolate_to(h_r, n_init, target_init)
                h_interp_bin = project_binary(h_interp, target_init)
                allowed2 = min(30, tleft() * 0.1)
                h_ref, c5_ref = annealed_pressure(h_interp_bin, max_iters=int(allowed2 * 800),
                                                  time_limit=allowed2, start_time=start_time, rng=rng)
                if c5_ref < best_score:
                    best_score = c5_ref
                    best_h = h_ref.copy()

    # Phase 6: Subgradient continuous refinement
    if tleft() > 50:
        h_cont = project_feasible(best_h, float(n_init) / 2.0)
        allowed = min(80, tleft() * 0.2)
        h_sg, c5_sg = subgradient_refine(h_cont, max_iters=int(allowed * 50),
                                         time_limit=allowed, start_time=start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    # Phase 7: Final pressure search
    if tleft() > 30:
        h_bin = project_binary(best_h, target_init)
        allowed = min(40, tleft() * 0.3)
        if allowed > 5:
            h_p7, c5_p7 = annealed_pressure(h_bin, max_iters=int(allowed * 1000),
                                            time_limit=allowed, start_time=start_time, rng=rng)
            if c5_p7 < best_score:
                best_score = c5_p7
                best_h = h_p7.copy()

    # Phase 8: SLSQP refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [50, 150, 500, 1500, 4000]
        allowed = min(60, tleft() - 10)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final binary polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(min(30000, max(500, tleft() * 2000)))
        h_sa_f, c5_f = sa_binary(h_bin_final, target_use, max_it, 0.005, 0.00001,
                                 rng, time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Final feasibility and evaluation
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-6:
        h_final = best_h
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
