import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_box_sum(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target} using bisection."""
    h = np.asarray(h, dtype=float)
    # Bisection on lambda: find lambda s.t. sum(clip(h + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune for any residual
    resid = target - np.sum(h_proj)
    if abs(resid) > 1e-10:
        if resid > 0:
            free = np.where(h_proj < 1.0 - 1e-12)[0]
        else:
            free = np.where(h_proj > 1e-12)[0]
        if len(free) > 0:
            h_proj[free] += resid / len(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def c5_smooth_grad(h, alpha, active_lags):
    """Compute smooth approximation of C5 and its gradient using active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Use only active lags
    active_corr = corr[active_lags]
    c_max = np.max(active_corr)
    shifted = np.clip(alpha * (active_corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i, lag_idx in enumerate(active_lags):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

A Barker sequence is a binary sequence with minimum aperiodic autocorrelation. But our problem is slightly different - we're looking at the cross-correlation between h and 1-h.

Actually, let me reconsider. For binary h:
C_k = sum_i h[i](1-h[i+k]) = sum_i h[i] - sum_i h[i]h[i+k] (over valid range)

The first term depends on k (it's the sum of h over the overlap region).
The second term is the autocorrelation.

So C_k ≈ (n-k)/2 - R(k) where R(k) is the autocorrelation.

To minimize max C_k, we want R(k) to be as large as possible relative to (n-k)/2.

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_box_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        if resid > 0:
            free = np.where(h < 1.0 - 1e-10)[0]
        else:
            free = np.where(h > 1e-10)[0]
        if len(free) > 0:
            h[free] += resid / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(len(corr))
    c_a = corr[idx] * dx
    c_max = np.max(c_a)
    shifted = np.clip(alpha * (c_a - c_max), -500.0, 0.0)
    ev = np.exp(shifted)
    tot = np.sum(ev)
    obj = c_max + np.log(tot) / alpha
    w = (ev / tot) * dx
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * omh[:n - lag]
                grad[:n - lag] -= wi * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += wi * omh[m:]
                grad[m:] -= wi * h[:n - m]
    return obj, grad


def heavyball_phase(h_start, target, alpha, topk, n_iters, lr_init, momentum, deadline):
    n = len(h_start)
    x = h_start.copy()
    v = np.zeros(n)
    lr = lr_init
    best = compute_c5(x)
    best_x = x.copy()
    no_improve = 0

    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, g = smooth_obj_grad(x, alpha, topk)
        v = momentum * v - lr * g
        x_new = project_box_sum(x + v, target)
        x = x_new

        if it % 20 == 0 or it == n_iters - 1:
            sc = compute_c5(x)
            if sc < best:
                best = sc
                best_x = x.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.7
                    no_improve = 0
                    if lr < 1e-5:
                        break
    return best_x, best


def fista_phase(h_start, target, alpha, topk, n_iters, lr_init, deadline):
    n = len(h_start)
    x = h_start.copy()
    y = x.copy()
    t = 1.0
    lr = lr_init
    best = compute_c5(x)
