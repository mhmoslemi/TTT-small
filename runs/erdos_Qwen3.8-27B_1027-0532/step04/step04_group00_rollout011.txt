```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc_1d(h, j, lam):
    """Find optimal h[j] minimizing max_k O_k + lam*h[j], given other h fixed."""
    n = len(h)
    # Compute a_k (linear coefficient of h[j] in O_k)
    a = np.zeros(n)
    n1 = n - j
    a[:n1] += 1.0 - h[j:j+n1]
    if j > 0:
        a[:j+1] -= h[j:-1:-1] if j+1 > 1 else np.array([h[0]])
    # Hmm, h[j:-1:-1] goes from j down to index 0. Let me verify.
    # h[5:-1:-1] = [h[5], h[4], h[3], h[2], h[1], h[0]] -- yes, 6 elements for j=5.
    # h[:j+1] has j+1 elements. ✓
    
    # b_k = O_k - a_k * h[j]
    O = compute_corr(h)[n-1:]  # O[k] for k=0,...,n-1
    b = O - a * h[j]
    c = a + lam
    
    # Find v in [0,1] minimizing max_k (c_k * v + b_k)
    # Binary search on the subgradient
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = c * mid + b
        active = np.argmax(vals)
        if c[active] < -1e-12:
            lo = mid
        elif c[active] > 1e-12:
            hi = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, target, n_passes=50, lam_iters=30):
    """Coordinate descent with Lagrange multiplier for sum constraint."""
    best_h = h.copy()
    best_score = compute_c5(h)
    
    lam = 0.0
    h_cur = h.copy()
    
    for p in range(n_passes):
        # Adjust lambda to satisfy sum constraint
        for _ in range(lam_iters):
            h_test = h_cur.copy()
            for j in range(n):
                h_test[j] = coord_desc_1d(h_test, j, lam)
            h_test = project_feasible(h_test, target)
            sum_err = h_test.sum() - target
            if abs(sum_err) < 1e-6:
                break
            # Adjust lambda
            lam -= 0.01 * np.sign(sum_err) * max(1, abs(sum_err))
            lam = np.clip(lam, -10, 10)
        
        sc = compute_c5(h_test)
        if sc < best_score:
            best_score = sc
            best_h = h_test.copy()
            h_cur = h_test.copy()
        else:
            # Try a few more lambda values
            improved = False
            for dl in [0.005, -0.005, 0.01, -0.01]:
                h_test2 = h_cur.copy()
                for j in range(n):
                    h_test2[j] = coord_desc_1d(h_test2, j, lam + dl)
                h_test2 = project_feasible(h_test2, target)
                sc2 = compute_c5(h_test2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_test2.copy()
                    h_cur = h_test2.copy()
                    lam += dl
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_score
```

Actually, the issue is that O depends on h, and as we change h[j], the O values change. But for the 1D optimization of h[j], we only need a_k and b_k, which are:
- a_k: depends on all h values (not just h[j])
- b_k = O_k - a_k * h[j]: depends on O_k which depends on all h

So we need to recompute O after each coordinate update. That's expensive.

Actually, let me think about this more carefully. When we update h[j] from old value to new value v:
- O_k changes only for k where h[j] appears (i.e., k ≤ n-1-j and k ≤ j, so k ≤ min(j, n-1-j))
- Specifically, O_k changes by: v*(1-h[j+k]) - old_h[j]*(1-h[j+k]) [if j+k < n]
  + h[j-k]*(1-v) - h[j-k]*(1-old_h[j]) [if j >= k]
  = (v - old_h[j]) * [(1-h[j+k]) - h[j-k]] (combining both terms when both apply)

So the change in O_k is (v - old_h[j]) * a_k, which is what we expect.

After updating h[j], the new O_k = old_O_k + (v - old_h[j]) * a_k.

This means I can update O incrementally! After setting h[j] = v:
O_new[k] = O_old[k] + (v - h[j]) * a_k

This is O(n) per coordinate update (to update O), plus O(n) to find v*. So O(n) per coordinate, O(n²) per pass. Much better!
