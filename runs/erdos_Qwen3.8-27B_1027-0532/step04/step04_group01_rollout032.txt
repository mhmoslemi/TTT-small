def project_binary(h, target_sum):
    n = len(h)
    target = int(target_sum)
    h_c = np.clip(h, 0.0, 1.0)
    threshold = 0.5

        return (h_c >= threshold).astype(np.float64)
    if n_ones < target:

        below_idx = np.where(h_c < threshold)[0]

            result = np.ones(n, dtype=np.float64)
            result[h_c < threshold] = 0.0
            # Need to flip some from 0 to 1... this shouldn't happen if we project properly
            return result
        top = below_idx[np.argsort(-h_c[below_idx])[:need]]
        result = np.zeros(n, dtype=np.float64)
        result[h_c >= threshold] = 1.0
        result[top] = 1.0
        return result

        result = np.ones(n, dtype=np.float64)
        result[h_c < threshold] = 0.0
        result[to_remove] = 0.0
