import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def con_eq(x, target):
    return np.sum(x) - target

def con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-16, deadline=None):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if deadline and time.time() > deadline:
            break
        def obj(x):
            return lse_obj(x, a, n, target)
        def ceq(x):
            return np.sum(x) - target
        cons = [{'type': 'eq', 'fun': ceq, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

            # - If i=j is valid (0≤j<n, 0≤j+k<n): h[j]*(1-h[j+k])
            # - If i+k=j is valid (i=j-k, 0≤j-k<n, 0≤j<n): h[j-k]*(1-h[j])

            # (1-h[j+k]) * [0≤j+k<n] - h[j-k] * [0≤j-k<n]
            # (The minus sign comes from the second term: h[j-k]*(1-h[j]) = h[j-k] - h[j-k]*h[j])
            
            # For m = 0 to 2n-2, k = m - (n-1):

            mask2 = (j - k_vals >= 0) & (j - k_vals < n)

            val0 = float(np.max(b))
            val1 = float(np.max(b + a))

            k0 = int(np.argmax(b))
            k1 = int(np.argmax(b + a))

                # Same active k at both ends, minimum is at boundary
                best_t = 0.0 if val0 < val1 else 1.0

                    best_t = 0.0 if val0 < val1 else 1.0

import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_g_n = 400
_g_target = 200
_g_alpha = 100.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

# SLSQP objective and constraint (using globals)
def _obj_full(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _con_eq(x):
    return np.sum(x) - _g_target

def _con_jac(x):
    return np.ones(_g_n)

def _obj_block(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _con_block_eq(x):
    return np.sum(x) - _g_block_target

def _con_block_jac(x):
    return np.ones(_g_e - _g_s)

def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-16, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_full, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def block_slsqp(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_target, _g_alpha, _g_block_target
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': _con_block_eq, 'jac': _con_block_jac}]
            try:
                res = minimize(_obj_block, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 300, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:

            except Exception:

    return h_cur, cur

    h_new = np.interp(t_new, t_old, h)
    return h_new

def periodic_h(n, period, duty):

    return h

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 400
_g_target = 200
_g_alpha = 100.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_full(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _con_eq(x):
    return np.sum(x) - _g_target


def _con_jac(x):
    return np.ones(_g_n)


def _obj_block(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _con_block_eq(x):
    return np.sum(x) - _g_block_target


def _con_block_jac(x):
    return np.ones(_g_e - _g_s)


def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-16, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_full, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur


def block_slsqp(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_target, _g_alpha, _g_block_target
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': _con_block_eq, 'jac': _con_block_jac}]
            try:
                res = minimize(_obj_block, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 300, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur


def coord_descent(h, n, target, n_sweeps, deadline):
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full').astype(float)
    cur = float(np.max(corr)) * (2.0 / n)
    n_corr = 2 * n - 1
    k_vals = np.arange(n_corr) - (n - 1)
    a_buf = np.zeros(n_corr)

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            a_buf[:] = 0.0
            mask1 = (j + k_vals >= 0) & (j + k_vals < n)
            mask2 = (j - k_vals >= 0) & (j - k_vals < n)
            a_buf[mask1] = 1.0 - h_cur[j + k_vals[mask1]]
            a_buf[mask2] -= h_cur[j - k_vals[mask2]]

            b = corr - h_cur[j] * a_buf
            val0 = float(np.max(b))
            val1 = float(np.max(b + a_buf))

            if val0 <= val1:
                best_t = 0.0
                best_val = val0
            else:
                best_t = 1.0
                best_val = val1

            k0 = int(np.argmax(b))
            k1 = int(np.argmax(b + a_buf))
            if k0 != k1:
                da = a_buf[k0] - a_buf[k1]
                if abs(da) > 1e-15:
                    t_cross = (b[k1] - b[k0]) / da
                    if 0.0 < t_cross < 1.0:
                        v = float(np.max(b + t_cross * a_buf))
                        if v < best_val:
                            best_val = v
                            best_t = t_cross

            for t_try in (0.25, 0.5, 0.75):
                v = float(np.max(b + t_try * a_buf))
                if v < best_val:
                    best_val = v
                    best_t = t_try

            old_t = h_cur[j]
            if best_t != old_t:
                h_cur[j] = best_t
                corr += (best_t - old_t) * a_buf

        if sweep % 20 == 0:
            h_cur = project_feasible(h_cur, target)
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full').astype(float)
        sc = float(np.max(corr)) * (2.0 / n)
        if sc < cur:
            cur = sc

    return h_cur, cur


def upscale(h, new_n):
    old_n = len(h)
    t_old = np.arange(old_n) + 0.5
    t_new = np.arange(new_n) * (old_n / new_n) + 0.5
    h_new = np.interp(t_new, t_old, h)
    return h_new


def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h


def periodic_h(n, period, duty):
    h = np.zeros(n)
    idx = np.arange(n)
    h[(idx % period) < duty] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        h[:total - n // 2] = 0.0
    elif total < n // 2:
        deficit = n // 2 - total
        zeros = np.where(h == 0)[0]
        h[zeros[:deficit]] = 1.0
    return h


def smooth_two_interval(n, L1, edge_frac):
    L2 = n // 2 - L1
    h = np.zeros(n)
    edge = max(int(L1 * edge_frac), 2)
    h[:L1] = 1.0
    h[L1 - edge:L1] = 0.5 * (1.0 + np.cos(np.pi * np.arange(edge) / edge))
    edge2 = max(int(L2 * edge_frac), 2)
    h[n - L2:] = 1.0
    h[n - L2:n - L2 + edge2] = 1.0 - 0.5 * (1.0 + np.cos(np.pi * np.arange(edge2) / edge2))
    return project_feasible(h, n / 2.0)
