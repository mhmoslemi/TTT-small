def project_feasible(h, target_sum):

from scipy.optimize import minimize
import time

def compute_corr_overlap(h):
    """Compute overlap correlations (lags 0 to n-1)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2*n-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_overlap(h)
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coord_descent_fast(h, n, target, n_sweeps=100, deadline=None, tolerance=1e-12):
    """Fast coordinate descent with incremental correlation tracking."""
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr_overlap(h_cur).copy()
    cur_score = float(np.max(corr)) * dx
    best_score = cur_score
    best_h = h_cur.copy()
    no_improve_count = 0
    
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        sweep_improved = False
        sum_error = 0.0
        
        for j in range(n):
            old_val = h_cur[j]
            
            # Compute a_k: coefficient of h[j] in each corr_k
            a = np.zeros(n)
            # From i=j term: valid for k < n-j
            nk1 = n - j
            if nk1 > 0:
                a[:nk1] += (1.0 - h_cur[j:j+nk1])
            # From i=j-k term: valid for k <= j
            nk2 = j + 1
            if nk2 > 0:
                a[:nk2] -= h_cur[j-nk2+1:j+1][::-1]
            
            # b_k = corr_k - a_k * old_val
            b = corr - a * old_val
            
            # Find optimal x for h[j] minimizing max_k(a_k*x + b_k) over [0,1]
            # The function f(x) = max_k(a_k*x + b_k) is convex piecewise linear
            # Minimum is at boundary or at a breakpoint
            
            # Evaluate at boundaries and find the "deepest" breakpoint
            f0 = float(np.max(b))
            f1 = float(np.max(a + b))
            
            # Find the k that's optimal at x=0 and x=1
            k0 = int(np.argmax(b))
            k1 = int(np.argmax(a + b))
            
            # If k0 != k1 and they cross in [0,1], that's a candidate
            candidates_x = [0.0, 1.0]
            if a[k0] != a[k1]:
                x_cross = (b[k0] - b[k1]) / (a[k1] - a[k0])
                if 0.0 <= x_cross <= 1.0:
                    candidates_x.append(x_cross)
            
            # Also try the current value and nearby
            candidates_x.append(old_val)
            
            best_x = old_val
            best_f = f0 if old_val == 0 else (f1 if old_val == 1 else float(np.max(a*old_val + b)))
            
            for x_c in candidates_x:
                x_c = float(np.clip(x_c, 0.0, 1.0))
                f_c = float(np.max(a * x_c + b))
                if f_c < best_f - tolerance:
                    best_f = f_c
                    best_x = x_c
            
            if abs(best_x - old_val) > 1e-14:
                delta = best_x - old_val
                h_cur[j] = best_x
                corr += delta * a
                sum_error += delta
                sweep_improved = True
        
        # Fix sum constraint
        if abs(sum_error) > 1e-14:
            # Redistribute to non-bound coordinates
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr_overlap(h_cur).copy()
        
        cur_score = float(np.max(corr)) * dx
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_h = h_cur.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 5:
                break
    
    return best_h, best_score

def momentum_subgradient(h, n, target, n_steps=2000, n_active=5, lr_init=0.01, deadline=None):
    """Momentum-based subgradient with adaptive learning rate."""
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    
    for step in range(n_steps):
        if deadline and time.time() > deadline:
            break
        
        corr = compute_corr_overlap(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        c_max = float(np.max(corr))
        
        # Softmax weights
        diffs = corr[top_idx] - c_max
        weights = np.exp(diffs / 0.01)
        weights /= weights.sum()
        
        # Gradient
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        # Momentum update
        v = mu * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
        else:
            lr *= 0.9
            v *= 0.5
            if lr < 1e-12:
                break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_active = None


def compute_corr_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2*n-1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_overlap(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _obj_lse_active(x):
    c = compute_corr_overlap(x)
    active_vals = c[_g_active]
    c_max = float(np.max(active_vals))
    shifted = _g_alpha * (active_vals - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    dx = 2.0 / _g_n
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha) * dx


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _obj_lse_all(x):
    c = compute_corr_overlap(x)
    c_max = float(np.max(c))
    shifted = _g_alpha * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    dx = 2.0 / _g_n
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha) * dx


def active_set_slsqp(h, n, target, n_active=10, maxiter=5000, ftol=1e-18, deadline=None):
    global _g_n, _g_target, _g_alpha, _g_active
    _g_n = n
    _g_target = target

    corr = compute_corr_overlap(h)
    top_k = np.argsort(corr)[::-1][:n_active]
    _g_active = np.sort(top_k)

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]

    for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
        if deadline and time.time() > deadline:
            break
        _g_alpha = alpha
        try:
            res = minimize(_obj_lse_active, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                corr = compute_corr_overlap(h_cur)
                _g_active = np.sort(np.argsort(corr)[::-1][:n_active])
        except Exception:
            pass

    return h_cur, cur_score


def full_lse_slsqp(h, n, target, maxiter=3000, ftol=1e-17, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n

def coord_descent_fast(h, n, target, n_sweeps=200, deadline=None, project_interval=5):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr_overlap(h_cur).copy()
    cur_score = float(np.max(corr)) * dx
    best_score = cur_score
    best_h = h_cur.copy()
    no_improve = 0

    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        sweep_improved = False

        for j in range(n):
            old_val = h_cur[j]

            a = np.zeros(n)
            nk1 = n - j
            if nk1 > 0:
                a[:nk1] += (1.0 - h_cur[j:j + nk1])
            nk2 = j + 1
            if nk2 > 0:
                a[:nk2] -= h_cur[j - nk2 + 1:j + 1][::-1]

            b = corr - a * old_val

            f0 = float(np.max(b))
            f1 = float(np.max(a + b))

            k0 = int(np.argmax(b))
            k1 = int(np.argmax(a + b))

            best_x = old_val
            best_f = float(np.max(a * old_val + b))

            if f0 < best_f - 1e-14:
                best_f = f0
                best_x = 0.0
            if f1 < best_f - 1e-14:
                best_f = f1
                best_x = 1.0

            if k0 != k1 and abs(a[k0] - a[k1]) > 1e-15:
                x_cross = (b[k0] - b[k1]) / (a[k1] - a[k0])
                if 0.0 < x_cross < 1.0:
                    f_cross = float(np.max(a * x_cross + b))
                    if f_cross < best_f - 1e-14:
                        best_f = f_cross
                        best_x = x_cross

            # Also try crossing of top-2 at current point
            vals_now = a * old_val + b
            top2 = np.argsort(vals_now)[::-1][:2]
            if len(top2) == 2 and abs(a[top2[0]] - a[top2[1]]) > 1e-15:
                x_cross2 = (b[top2[1]] - b[top2[0]]) / (a[top2[0]] - a[top2[1]])
                if 0.0 < x_cross2 < 1.0:
                    f_cross2 = float(np.max(a * x_cross2 + b))
                    if f_cross2 < best_f - 1e-14:
                        best_f = f_cross2
                        best_x = x_cross2

            if abs(best_x - old_val) > 1e-14:
                delta = best_x - old_val
                h_cur[j] = best_x
                corr += delta * a
                sweep_improved = True

        if (sweep + 1) % project_interval == 0 or sweep == n_sweeps - 1:
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr_overlap(h_cur).copy()
            cur_score = float(np.max(corr)) * dx
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_h = h_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15 and not sweep_improved:
                    break

    h_cur = project_feasible(h_cur, target)
    corr = compute_corr_overlap(h_cur).copy()
    cur_score = float(np.max(corr)) * dx
    if cur_score < best_score:
        best_score = cur_score
        best_h = h_cur.copy()

    return best_h, best_score


def momentum_subgradient(h, n, target, n_steps=3000, n_active=5, lr_init=0.008, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    mu = 0.85
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    sc_cur = best_score

    for step in range(n_steps):
        if deadline and time.time() > deadline:
            break

        corr = compute_corr_overlap(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        c_max = float(np.max(corr))

        diffs = corr[top_idx] - c_max
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k > 0:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx

        v = mu * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)

        sc_new = compute_c5(h_new)
        if sc_new < sc_cur - 1e-15:
            h_cur = h_new
            sc_cur = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
        else:
            lr *= 0.93
            v *= 0.3
            if lr < 1e-13:
                break

    return best_h, best_score


def paired_coord_descent(h, n, target, n_sweeps=100, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr_overlap(h_cur).copy()
    cur_score = float(np.max(corr)) * dx
    best_score = cur_score
    best_h = h_cur.copy()
    no_improve = 0

    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        sweep_improved = False

        for j in range(n - 1):
            j2 = n - 1 - j
            old_j = h_cur[j]
            old_j2 = h_cur[j2]

            a_j = np.zeros(n)
            nk1 = n - j
            if nk1 > 0:
                a_j[:nk1] += (1.0 - h_cur[j:j + nk1])
            nk2 = j + 1
            if nk2 > 0:
                a_j[:nk2] -= h_cur[j - nk2 + 1:j + 1][::-1]

            a_j2 = np.zeros(n)
            nk1 = n - j2
            if nk1 > 0:
                a_j2[:nk1] += (1.0 - h_cur[j2:j2 + nk1])
            nk2 = j2 + 1
            if nk2 > 0:
                a_j2[:nk2] -= h_cur[j2 - nk2 + 1:j2 + 1][::-1]

            b = corr - a_j * old_j - a_j2 * old_j2

            # Try paired moves: increase j, decrease j2 (and vice versa)
            deltas = [0.01, 0.05, 0.1, 0.03, 0.08]
            best_delta = 0.0
            best_val = float(np.max(corr))

            for d in deltas:
                # Forward: j += d, j2 -= d
                if old_j + d <= 1.0 and old_j2 - d >= 0.0:
                    new_corr = b + a_j * (old_j + d) + a_j2 * (old_j2 - d)
                    val = float(np.max(new_corr))
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d
                # Backward: j -= d, j2 += d
                if old_j - d >= 0.0 and old_j2 + d <= 1.0:
                    new_corr = b + a_j * (old_j - d) + a_j2 * (old_j2 + d)
                    val = float(np.max(new_corr))
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = -d

            if abs(best_delta) > 1e-6:
                h_cur[j] = old_j + best_delta
                h_cur[j2] = old_j2 - best_delta
                corr += best_delta * a_j - best_delta * a_j2
                sweep_improved = True

        h_cur = project_feasible(h_cur, target)
        corr = compute_corr_overlap(h_cur).copy()
        cur_score = float(np.max(corr)) * dx
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10 and not sweep_improved:
                break

    return best_h, best_score

            h_c, sc_c = coord_descent_fast(h_coarse, n_coarse, target_coarse, n_sweeps=300, deadline=start_time + 40)
            h_c, sc_c2 = active_set_slsqp(h_c, n_coarse, target_coarse, n_active=15, maxiter=5000, deadline=start_time + 40)
            if sc_c2 < sc_c:
                h_c, sc_c = h_c, sc_c2

            h_fine = interpolate_to(h_c, n, target)
            sc_fine = compute_c5(h_fine)

            if sc_fine < best_score:
                best_score = sc_fine
                best_h = h_fine

            if budget_s - 20.0 - (time.time() - start_time) > 100:
                h_cd2, sc_cd2 = coord_descent_fast(best_h, n, target, n_sweeps=300, deadline=start_time + 40)
                if sc_cd2 < best_score:
                    best_score = sc_cd2
                    best_h = h_cd2

    # Phase 7: Random restarts with CD
    deadline_p7 = start_time + (budget_s - 20.0) * 0.82
    for attempt in range(5):
        if budget_s - 20.0 - (time.time() - start_time) < 60:
            break
        scales = [0.03, 0.06, 0.02, 0.05, 0.04]
        h_pert = best_h.copy()
        h_pert += rng.randn(n) * scales[attempt]
        h_pert = project_feasible(h_pert, target)

        h_cd, sc_cd = coord_descent_fast(h_pert, n, target, n_sweeps=200, deadline=deadline_p7)
        h_as, sc_as = active_set_slsqp(h_cd, n, target, n_active=10, maxiter=3000, deadline=deadline_p7)
        if sc_as < sc_cd:
            sc_best = sc_as
            h_best = h_as
        else:
            sc_best = sc_cd
            h_best = h_cd
        if sc_best < best_score:
            best_score = sc_best
            best_h = h_best

    # Phase 8: Momentum subgradient
    if budget_s - 20.0 - (time.time() - start_time) > 100:
        deadline_p8 = start_time + (budget_s - 20.0) * 0.90
        h_mom, sc_mom = momentum_subgradient(best_h, n, target, n_steps=3000, n_active=5, lr_init=0.005, deadline=deadline_p8)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom

    # Phase 9: Final active-set SLSQP (very tight)
    if budget_s - 20.0 - (time.time() - start_time) > 80:
        deadline_p9 = start_time + (budget_s - 20.0) * 0.96
        h_as, sc_as = active_set_slsqp(best_h, n, target, n_active=30, maxiter=15000, ftol=1e-20, deadline=deadline_p9)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    # Phase 10: Final coordinate descent polish
    if budget_s - 20.0 - (time.time() - start_time) > 30:
        h_cd, sc_cd = coord_descent_fast(best_h, n, target, n_sweeps=300, deadline=start_time + budget_s - 22.0)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 11: Final momentum
    if budget_s - 20.0 - (time.time() - start_time) > 20:
        h_mom, sc_mom = momentum_subgradient(best_h, n, target, n_steps=1500, n_active=3, lr_init=0.002, deadline=start_time + budget_s - 22.0)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
