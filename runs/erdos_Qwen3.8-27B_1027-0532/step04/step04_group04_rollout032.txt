def compute_delta_c(h, a, b):
    """Compute ΔC vector when swapping h[a] (1→0) and h[b] (0→1)."""
    n = len(h)
    # ΔC[k] = h[a+k] - h[b+k] + h[a-k] - h[b-k] for all k
    # where out-of-range indices contribute 0
    # k ranges from -(n-1) to (n-1), stored as index k+(n-1) in a 2n-1 vector
    
    # h[a+k] is nonzero (== h[a+k]) only when 0 <= a+k < n, i.e., -a <= k < n-a
    # h[b+k] is nonzero only when -b <= k < n-b
    # h[a-k] is nonzero only when 0 <= a-k < n, i.e., a-n+1 < k <= a
    # h[b-k] is nonzero only when b-n+1 < k <= b
    
    # For the full vector of length 2n-1 (index 0 to 2n-2, representing k = -(n-1) to n-1):
    delta = np.zeros(2*n - 1)
    
    # Term 1: h[a+k] for valid k
    # a+k in [0, n-1] → k in [-a, n-1-a]
    # In the delta array (index = k + n - 1):
    # index in [n-1-a, 2n-2-a]
    lo1 = max(0, n-1-a)
    hi1 = min(2*n-2, 2*n-2-a)
    if lo1 <= hi1:
        k_range = np.arange(lo1 - (n-1), hi1 - (n-1) + 1)  # actual k values
        delta[lo1:hi1+1] += h[a + k_range]
    
    # Term 2: -h[b+k] for valid k
    lo2 = max(0, n-1-b)
    hi2 = min(2*n-2, 2*n-2-b)
    if lo2 <= hi2:
        k_range = np.arange(lo2 - (n-1), hi2 - (n-1) + 1)
        delta[lo2:hi2+1] -= h[b + k_range]
    
    # Term 3: h[a-k] for valid k
    # a-k in [0, n-1] → k in [a-(n-1), a]
    lo3 = max(0, a-(n-1))
    hi3 = min(2*n-2, a)
    if lo3 <= hi3:
        k_range = np.arange(lo3 - (n-1), hi3 - (n-1) + 1)
        delta[lo3:hi3+1] += h[a - k_range]
    
    # Term 4: -h[b-k] for valid k
    lo4 = max(0, b-(n-1))
    hi4 = min(2*n-2, b)
    if lo4 <= hi4:
        k_range = np.arange(lo4 - (n-1), hi4 - (n-1) + 1)
        delta[lo4:hi4+1] -= h[b - k_range]
    
    return delta
