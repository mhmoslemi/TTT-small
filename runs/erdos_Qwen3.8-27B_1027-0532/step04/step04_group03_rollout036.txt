def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_dx(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    h[idx] += per
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h).clip(0, 1).astype(int)
    cur = int(h_r.sum())
    if cur == target_sum:
        return h_r.astype(float)
    if cur > target_sum:
        need = cur - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - cur
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def subgrad_lags(h, lags):
    n = len(h)

            g[:n-m] -= h[m:]
        else:
            mm = -m
            g[:n-mm] += (1.0 - h[mm:])

def interp_to(h_src, n_dst, target_dst):

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h).clip(0, 1).astype(int)
    cur = int(h_r.sum())
    if cur == target_sum:
        return h_r.astype(float)
    if cur > target_sum:
        need = cur - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - cur
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)


def subgrad_lags(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        m = k - (n - 1)
        if m >= 0:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
        else:
            mm = -m
            g[:n - mm] += (1.0 - h[mm:])
            g[mm:] -= h[:n - mm]
    return g / max(len(lags), 1)


def grad_sa(h_start, target, max_it, t0, t1, rng, tl, st):
    n = len(h_start)
    h = project_feasible(h_start.copy(), target)
    bh = h.copy()
    bc5 = compute_c5(h)
    cc5 = bc5
    nip = 0
    max_nip = max(4000, max_it // 8)

    corr = compute_corr_dx(h)
    cm = np.max(corr)
    thr = cm * 0.96
    al = np.where(corr >= thr)[0]
    if len(al) == 0:
        al = np.array([int(np.argmax(corr))])
    g = subgrad_lags(h, al)

    for i in range(max_it):
        if time.time() - st > tl:
            break
        if nip > max_nip:
            if rng.random() < 0.02:
                nip = 0
            else:
                break

        r = min(i / max(max_it - 1, 1), 1.0)
        t = t0 * (t1 / t0) ** r

        if i % 100 == 0:
            corr = compute_corr_dx(h)
            cm = np.max(corr)
            thr = cm * (0.94 + 0.04 * (1.0 - r))
            al = np.where(corr >= thr)[0]
            if len(al) == 0:
                al = np.array([int(np.argmax(corr))])
            g = subgrad_lags(h, al)

        mt = rng.randint(7)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        if mt == 0:

            wi = np.maximum(-g[zeros], 0) + 0.01
            swd = wd.sum()
            swi = wi.sum()

                wd /= swd
                wi /= swi
                i1 = ones[rng.choice(len(ones), p=wd)]
                i0 = zeros[rng.choice(len(zeros), p=wi)]
            else:
                i1 = ones[rng.randint(len(ones))]
                i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0.0
            h[i0] = 1.0
        elif mt == 1:
            kh = al[rng.randint(len(al))]
            m = kh - (n - 1)
            if 0 < m < n:
                pr = np.where((h[:n - m] > 0.5) & (h[m:] < 0.5))[0]
                if len(pr) > 0:
                    p = pr[rng.randint(len(pr))]
                    h[p] = 0.0
                    h[p + m] = 1.0
                else:
                    h[ones[rng.randint(len(ones))]] = 0.0
                    h[zeros[rng.randint(len(zeros))]] = 1.0
            elif m < 0:

                pr = np.where((h[mm:] > 0.5) & (h[:n - mm] < 0.5))[0]

            else:

                h[ones[s1[1]]] = 0.0

        elif mt == 5:

                    else:

                        i0 = zeros[rng.randint(len(zeros))]

                else:

                    i0 = zeros[rng.randint(len(zeros))]

                    h[i0] = 1.0
        else:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0.0
            h[i0] = 1.0

        d = c5n - cc5

            if c5n < bc5:
                bc5 = c5n
                bh = h.copy()
                nip = 0

            nip += 1

def cons_fun(x):
    return np.sum(x) - _cons_target


def cons_jac(x):
    return np.ones(_cons_n)


def lse_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(corr))
    s = np.clip(alpha * (corr - cm), -500.0, 0.0)
    return cm + np.log(np.sum(np.exp(s))) / alpha


def prox_grad(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    bh = h.copy()
    bc5 = compute_c5(h)
    lr = 0.03
    nip = 0
    max_nip = 500

    for it in range(200000):
        if time.time() - start_time > time_limit:
            break
        if nip > max_nip:
            break

        corr = compute_corr_dx(h)
        cm = float(np.max(corr))
        thr = cm * 0.99
        al = np.where(corr >= thr)[0]
        if len(al) == 0:
            al = np.array([int(np.argmax(corr))])

        g = subgrad_lags(h, al)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        h_try = project_feasible(h - lr * g, target)
        c5_try = compute_c5(h_try)

        if c5_try < bc5 - 1e-13:
            bc5 = c5_try
            bh = h_try.copy()
            h = h_try
            nip = 0
            lr = min(lr * 1.08, 0.5)
        else:
            h_try2 = project_feasible(h - lr * 0.1 * g, target)
            c5_2 = compute_c5(h_try2)
            if c5_2 < bc5 - 1e-13:
                bc5 = c5_2
                bh = h_try2.copy()
                h = h_try2
                nip = 0
            else:
                nip += 1
                lr *= 0.92
                if lr < 1e-9:
                    break

    return bh, bc5


def slsqp_phase(h_start, target, time_limit, start_time):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    h = h_start.copy().astype(float)
    bh = h.copy()
    bc5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    alphas = [25, 50, 100, 200, 400, 800, 1600, 3200, 6400, 12000]
    for a in alphas:
        if time.time() - start_time > time_limit:
            break
        maxiter = 300 if a <= 800 else 200
        try:
            res = minimize(lse_obj, h, args=(a, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < bc5:
                bc5 = c5_new
                bh = h_new.copy()
                h = h_new
        except Exception:
            pass

    return bh, bc5


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smart_local_search(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    bh = h.copy()
    bc5 = compute_c5(h)
    fracs = [0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 0.95]

    for sweep in range(10):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr_dx(h)
        cm = np.max(corr)
        g = subgrad_lags(h, np.where(corr >= cm * 0.98)[0])

        top_positions = np.argsort(-np.abs(g))[:80]
        improved = False

        for pos in top_positions:
            if time.time() - start_time > time_limit:
                return bh, bc5
            old = h[pos]
            best_local = None
            best_local_c5 = None
            for f in fracs:
                if abs(f - old) < 0.02:
                    continue
                h[pos] = f
                h = project_feasible(h, target)
                c5 = compute_c5(h)
                if best_local_c5 is None or c5 < best_local_c5:
                    best_local_c5 = c5
                    best_local = f
            if best_local is not None and best_local_c5 < bc5 - 1e-11:
                h[pos] = best_local
                h = project_feasible(h, target)
                bc5 = best_local_c5
                bh = h.copy()
                improved = True
            else:
                h[pos] = old
                h = project_feasible(h, target)

        if not improved:
            break

    return bh, bc5
