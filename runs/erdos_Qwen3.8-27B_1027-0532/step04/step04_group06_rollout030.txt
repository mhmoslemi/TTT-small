The gradient w.r.t. h[j]:

- j appears as the "i" term when i = j is valid and j+k is valid: contributes -h[j+k]

This is O(n) per lag, and we only need top_k lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_grad_smoothed(h, corr, center, top_lags, alpha):
    """Compute gradient of log-sum-exp smoothed max for given top lags."""
    n = len(h)
    # Get correlations at top lags
    lag_corrs = np.array([corr[center + k] for k in top_lags])
    # Softmax weights
    max_lc = lag_corrs.max()
    weights = np.exp(alpha * (lag_corrs - max_lc))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(top_lags):
        k = int(k)
        if k > 0:
            # dC(k)/dh[j] = (1-h[j-k]) for j in [k, n) - h[j+k] for j in [0, n-k)
            grad[k:n] += weights[idx] * (1.0 - h[0:n-k])
            grad[0:n-k] -= weights[idx] * h[k:n]
        elif k < 0:
            kp = -k
            # For lag -kp: C(-kp) = sum_i h[i-kp]*(1-h[i]) for valid i
            # dC(-kp)/dh[j] = (1-h[j+kp]) for j+kp valid - h[j-kp] for j-kp valid
            # But wait, let me re-derive.
            # C(k) = sum over valid (i, i+k) of h[i+k]*(1-h[i])
            # For k = -kp: C(-kp) = sum over valid (i, i-kp) of h[i-kp]*(1-h[i])
            # j appears as i-kp when i = j+kp: contributes (1-h[j+kp]) if j+kp < n
            # j appears as i when i = j: contributes -h[j-kp] * ... wait
            # Actually C(-kp) = sum_i h[i-kp]*(1-h[i]) where i-kp >= 0 and i < n
            # = sum_m h[m]*(1-h[m+kp]) where m >= 0 and m+kp < n, m = i-kp
            # So C(-kp) = sum_m h[m]*(1-h[m+kp]) for m in [0, n-kp)
            # dC(-kp)/dh[j] = (1-h[j+kp]) if j < n-kp - h[j-kp] if j >= kp
            # Wait: d/dh[j] of sum_m h[m]*(1-h[m+kp])
            # = (1-h[j+kp]) if j+kp < n (from the h[m] term with m=j)
            # + h[j-kp] * (-1) if j-kp >= 0 (from the (1-h[m+kp]) term with m+kp = j, i.e., m = j-kp)
            # = (1-h[j+kp]) * [j+kp < n] - h[j-kp] * [j >= kp]
            # Which is the same formula as for positive k! Just with k = -kp:
            # dC(k)/dh[j] = (1-h[j-k]) [j-k valid] - h[j+k] [j+k valid]
            # For k = -kp: (1-h[j+kp]) [j+kp < n] - h[j-kp] [j-kp >= 0]
            # = (1-h[j+kp]) [j < n-kp] - h[j-kp] [j >= kp]
            grad[0:n-kp] += weights[idx] * (1.0 - h[kp:n])
            grad[kp:n] -= weights[idx] * h[0:n-kp]
        else:
            # k = 0: C(0) = sum h[i]*(1-h[i]), d/dh[j] = 1 - 2*h[j]
            grad += weights[idx] * (1.0 - 2.0 * h)
    
    return grad

def find_top_lags(corr, center, top_k):
    c = corr.copy()
    c[center] = -1e10  # exclude center (always 0 for binary, small for continuous)
    idx = np.argpartition(-c, top_k)[:top_k]
    lags = idx - center
    return lags

def adam_optimize(h0, target_sum, max_iters, alpha_start, alpha_end, 
                  lr, beta1, beta2, eps, time_limit, start_time, top_k=5):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ratio = it / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        
        corr = get_corr(h)
        center = n - 1
        top_lags = find_top_lags(corr, center, top_k)
        
        grad = compute_grad_smoothed(h, corr, center, top_lags, alpha)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - target
    
    def cons_j(h_in):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def find_top_lags(corr, center, top_k):
    c = corr.copy()
    c[center] = -1e10
    if len(c) <= top_k:
        # Handle edge case
        c[center] = 0.0
        return np.array([1, -1])[:top_k]
    idx = np.argpartition(-c, top_k)[:top_k]
    lags = idx - center
    return lags

def compute_grad(h, corr, center, top_lags, alpha):
    n = len(h)
    lag_corrs = np.array([corr[center + int(k)] for k in top_lags])
    max_lc = lag_corrs.max()
    exp_vals = np.exp(alpha * (lag_corrs - max_lc))
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_lags)):
        k = int(top_lags[idx])
        w = weights[idx]
        if k > 0:
            grad[k:n] += w * (1.0 - h[0:n-k])
            grad[0:n-k] -= w * h[k:n]
        elif k < 0:
            kp = -k
            grad[0:n-kp] += w * (1.0 - h[kp:n])
            grad[kp:n] -= w * h[0:n-kp]
        else:
            grad += w * (1.0 - 2.0 * h)
    
    return grad

def adam_optimize(h0, target_sum, max_iters, alpha_start, alpha_end,
                  lr, time_limit, start_time, top_k=5):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ratio = it / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        
        corr = get_corr(h)
        center = n - 1
        top_lags = find_top_lags(corr, center, top_k)
        
        grad = compute_grad(h, corr, center, top_lags, alpha)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 100
    
    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        h[a_idx] = 0.0
        h[b_idx] = 1.0

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - target
    
    def cons_j(h_in):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def find_top_lags(corr, center, top_k):
    c = corr.copy()
    c[center] = -1e10
    if len(c) <= top_k + 1:
        c[center] = 0.0
        idx = np.argsort(-c)[:top_k]
        return idx - center
    idx = np.argpartition(-c, top_k)[:top_k]
    return idx - center

def compute_grad(h, corr, center, top_lags, alpha):
    n = len(h)
    lag_corrs = np.array([corr[center + int(k)] for k in top_lags])
    max_lc = lag_corrs.max()
    exp_vals = np.exp(alpha * (lag_corrs - max_lc))
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for idx in range(len(top_lags)):
        k = int(top_lags[idx])
        w = weights[idx]
        if k > 0:
            grad[k:n] += w * (1.0 - h[0:n-k])
            grad[0:n-k] -= w * h[k:n]
        elif k < 0:
            kp = -k
            grad[0:n-kp] += w * (1.0 - h[kp:n])
            grad[kp:n] -= w * h[0:n-kp]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad

def adam_optimize(h0, target_sum, max_iters, alpha_start, alpha_end, lr,
                  time_limit, start_time, top_k=5):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    stall_limit = max(500, max_iters // 30)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > stall_limit:
            break
        ratio = it / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        corr = get_corr(h)
        center = n - 1
        top_lags = find_top_lags(corr, center, top_k)
        grad = compute_grad(h, corr, center, top_lags, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    def cons_f(h_in):
        return np.sum(h_in) - target
    def cons_j(h_in):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        else:
            h_curr = h_interp

        # Binary SA at this resolution
        if tleft() > 80:
            h_bin_level = project_binary(h_curr, target_level)
            sa_time = min(80.0, tleft() * 0.3)
            max_sa = int(max(5000, sa_time * 400))
            h_sa2, c5_sa2 = targeted_sa(h_bin_level, target_level, max_sa,
                                         0.008, 0.00003, rng,
                                         time_limit=sa_time, start_time=start_time)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()
                h_curr = h_sa2.copy()

    # Phase 3: Second pass Adam with different parameters
    n_use = len(best_h)
    if tleft() > 80:
        target_use = n_use / 2.0
        h_adam2, c5_adam2 = adam_optimize_multi(best_h, target_use,
                                                  int(max(2000, tleft() * 100)),
                                                  [50.0, 500.0, 2000.0],
                                                  [500.0, 5000.0, 20000.0],
                                                  [0.03, 0.015, 0.008],
                                                  time_limit=min(120.0, tleft() * 0.5),
                                                  start_time=start_time, top_k=7)
        if c5_adam2 < best_score:
            best_score = c5_adam2
            best_h = h_adam2.copy()

    # Phase 4: SLSQP at moderate resolution
    n_slq = min(len(best_h), 1200)
    target_slq = n_slq / 2.0
    h_slq_start = interpolate_to(best_h, n_slq, target_slq)
    if tleft() > 60:
        alphas_slq = [30, 100, 400, 1500, 5000]
        h_ref, sc = slsqp_refine(h_slq_start, n_slq, target_slq, alphas_slq,
                                  400, 1e-13,
                                  time_limit=min(80.0, tleft() * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Up-resolve and Adam again
    n_use = len(best_h)
    if tleft() > 60 and n_use < 4800:
        n_high = min(4800, max(2400, int(n_use * 1.5)))
        target_high = n_high / 2.0
        h_high = interpolate_to(best_h, n_high, target_high)
        allowed = max(20.0, min(100.0, tleft() * 0.5))
        h_adam3, c5_adam3 = adam_optimize_multi(h_high, target_high,
                                                  int(max(1000, allowed * 100)),
                                                  [200.0, 2000.0],
                                                  [2000.0, 20000.0],
                                                  [0.02, 0.01],
                                                  time_limit=allowed,
                                                  start_time=start_time, top_k=5)
        if c5_adam3 < best_score:
            best_score = c5_adam3
            best_h = h_adam3

    # Phase 6: Final binary SA polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 20:
        h_bin_final = project_binary(best_h, target_use)
        sa_time = max(5.0, min(40.0, tleft() - 15))
        max_it = int(max(3000, sa_time * 3000))
        h_sa_f, c5_f = targeted_sa(h_bin_final, target_use, max_it, 0.004, 0.00001,
                                    rng, time_limit=sa_time, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 10:
            h_hc, c5_hc = fast_hill_climb(best_h, target_use, 5000, rng)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc

    # Phase 7: Final SLSQP
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 15 and n_use <= 1200:
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, [2000, 8000],
                                  200, 1e-14,
                                  time_limit=max(3.0, tleft() - 5), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n_final
