def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        idx = np.argsort(-room)
        remaining = diff
        for i in idx:
            if remaining <= 1e-15:

            add = min(1.0 - h[i], remaining)
            h[i] += add
            remaining -= add
    else:

        idx = np.argsort(h)
        for i in idx:
            if remaining <= 1e-15:
                break
            sub = min(h[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

        ones = np.where(h_r == 1)[0]
