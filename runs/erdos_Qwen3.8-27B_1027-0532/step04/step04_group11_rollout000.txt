        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= w * h[i_start2-m:i_end2-m]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s > 0:
        w /= s
    return w

def grad_from_weights(h, weights):
    n = len(h)
    m_vals = np.arange(-(n - 1), n)
    threshold = 1e-12
    sig = weights > threshold
    sig_idx = np.where(sig)[0]
    sig_w = weights[sig_idx]
    grad = np.zeros(n)
    for k in range(len(sig_idx)):
        m = int(sig_idx[k] - (n - 1))
        w = float(sig_w[k])
        i_s = max(0, -m)
        i_e = min(n, n - m)
        if i_s < i_e:
            grad[i_s:i_e] += w * (1.0 - h[i_s + m:i_e + m])
        i_s2 = max(0, m)
        i_e2 = min(n, n + m)
        if i_s2 < i_e2:
            grad[i_s2:i_e2] -= w * h[i_s2 - m:i_e2 - m]
    return grad

def lse_objective_and_grad(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    obj = c_max + np.log(sum_exp) / alpha
    weights = exp_s / sum_exp
    grad = grad_from_weights(x, weights)
    return obj, grad

def proj_grad_desc(h, target, n_iter, alpha, lr_init, deadline, top_k=50):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        weights = softmax_weights(corr, alpha)
        if top_k > 0:
            idx = np.argsort(weights)[::-1][:top_k]
            sparse_w = np.zeros(len(weights))
            sparse_w[idx] = weights[idx]
            sparse_w /= sparse_w.sum()
            weights = sparse_w
        grad = grad_from_weights(h_cur, weights)
        step = lr
        improved = False
        for _ in range(15):
            h_new = project_feasible(h_cur - step * grad, target)
            new_score = compute_c5(h_new)
            if new_score < cur_score - 1e-15:
                h_cur = h_new
                cur_score = new_score
                improved = True
                stall = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h_cur.copy()
                break
            step *= 0.5
        if not improved:
            stall += 1
            lr = max(lr * 0.9, 1e-6)
            if stall > 50:
                break
        else:
            lr = min(lr * 1.05, 5.0)
    return best_h, best_score

def slsqp_optimize(h, target, alpha, maxiter=2000):
    n = len(h)
    def obj_and_jac(x):
        return lse_objective_and_grad(x, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = (lambda x: np.sum(x) - target, lambda x: np.ones(n))
    constraints = [{'type': 'eq', 'fun': cons[0], 'jac': cons[1]}]
    try:
        res = minimize(obj_and_jac, h, jac=None, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

def periodic_h(n, period, duty, phase=0):
    h = np.zeros(n)
    for i in range(n):
        if ((i + phase) % period) < duty:
            h[i] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        excess = total - n // 2
        ones_idx = np.where(h == 1.0)[0]
        h[ones_idx[:excess]] = 0.0
    elif total < n // 2:
        deficit = n // 2 - total
        zeros_idx = np.where(h == 0.0)[0]
        h[zeros_idx[:deficit]] = 1.0
    return h

def cosine_mod_h(n, L1, width):
    h = np.zeros(n)
    L2 = n // 2 - L1
    for i in range(L1):

    for i in range(L2):

        if t > 0.8:

        else:

    return project_feasible(h, n // 2.0)
