```python
    n = len(h)
    # Identify free variables
    free_mask = (h > free_threshold) & (h < 1.0 - free_threshold)
    free_idx = np.where(free_mask)[0]
    n_free = len(free_idx)
    
    if n_free == 0:
        return h, compute_c5(h)
    
    #Extract free variables
    h_free = h[free_idx].copy()
    
    # Fixed sum: sum(h[free_idx]) = target - sum(h[fixed])
    sum_fixed = np.sum(h[~free_mask])
    target_free = target - sum_fixed
    
    # Define objective and constraint on reduced problem
    def obj_grad(x_free):
        h_temp = h.copy()
        h_temp[free_idx] = np.clip(x_free, 0, 1)
        return lse_obj_and_grad(h_temp, alpha)

    bounds_free = [(0.0, 1.0)] * n_free
    res = minimize(lambda x: obj_grad(x)[0], h_free,
                   jac=lambda x: obj_grad(x)[1],
                   method='SLSQP', bounds=bounds_free,
                   constraints=[{'type': 'eq', 
                                 'fun': lambda x: np.sum(x) - target_free,
                                 'jac': lambda x: np.ones(n_free)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})

    h_new[free_idx] = np.clip(res.x, 0, 1)
    h_new = project_feasible(h_new, target)
    return h_new, compute_c5(h_new)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(300):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            proj = proj.copy()
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += wi * (1.0 - h[:n - k])
                grad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            if kk < n:
                grad[:n - kk] += wi * (1.0 - h[kk:])
                grad[kk:] -= wi * h[:n - kk]
    return obj, grad


def fista_optimize(h0, target, alpha, n_iter, deadline):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    L = alpha * 2.0 * n
    lr = 1.0 / L
    for it in range(n_iter):
        if time.time() > deadline:
            break
        f, g = lse_obj_and_grad(y, alpha)
        x_new = project_box_sum(y - lr * g, target)
        diff = x_new - y
        quad = 0.5 * L * np.dot(diff, diff)
        lin = np.dot(g, diff)
        if f + lin + quad < f - 1e-16:
            pass
        else:
            lr *= 0.5
            x_new = project_box_sum(y - lr * g, target)
            diff = x_new - y
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        y = x_new + (t / t_new - 1.0) * (x_new - x)
        y = project_box_sum(y, target)
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    return best_x, best_score


def slsqp_single(h, target, alpha, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_box_sum(res.x, target)
        sc = compute_c5(h_new)
        if sc < compute_c5(h):
            return h_new, sc
    except Exception:
        pass
    return h, compute_c5(h)


def reduced_slsqp(h, target, alpha, threshold, maxiter, deadline):
    n = len(h)
    free_mask = (h > threshold) & (h < 1.0 - threshold)
    n_free = int(free_mask.sum())
    if n_free < 5:
        return h, compute_c5(h)
    free_idx = np.where(free_mask)[0]
    h_free = h[free_idx].copy()
    sum_fixed = np.sum(h[~free_mask])
    target_free = target - sum_fixed
    if target_free < -1e-10 or target_free > n_free + 1e-10:
        return h, compute_c5(h)
    target_free = float(np.clip(target_free, 0.0, float(n_free)))

    def fun_reduced(x):
        h_temp = h.copy()
        h_temp[free_idx] = np.clip(x, 0.0, 1.0)
        return lse_obj_and_grad(h_temp, alpha)[0]

    def jac_reduced(x):
        h_temp = h.copy()
        h_temp[free_idx] = np.clip(x, 0.0, 1.0)
        _, g = lse_obj_and_grad(h_temp, alpha)
        return g[free_idx]

    def cfun_reduced(x):
        return np.sum(x) - target_free

    def cjac_reduced(x):
        return np.ones(n_free)

    bounds_free = [(0.0, 1.0)] * n_free
    try:
        res = minimize(fun_reduced, h_free, jac=jac_reduced, method='SLSQP',
                       bounds=bounds_free,
                       constraints=[{'type': 'eq', 'fun': cfun_reduced, 'jac': cjac_reduced}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = h.copy()
        h_new[free_idx] = np.clip(res.x, 0.0, 1.0)
        h_new = project_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < compute_c5(h):
            return h_new, sc
    except Exception:
        pass
    return h, compute_c5(h)


def binary_greedy(h_init, target, n_steps, alpha, deadline):
    n = len(h_init)
    h = np.round(h_init)
    h = project_box_sum(h, target)
    best_score = compute_c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        _, grad = lse_obj_and_grad(h, alpha)
        best_gain = 0.0
        best_j = -1
        best_dir = 0
        for j in range(n):
            if h[j] < 1.0:
                gain = -grad[j]
                if gain > best_gain:
                    best_gain = gain
                    best_j = j
                    best_dir = 1
            if h[j] > 0.0:
                gain = grad[j]
                if gain > best_gain:
                    best_gain = gain
                    best_j = j
                    best_dir = -1
        if best_j < 0 or best_gain <= 1e-10:
            break
        h[best_j] = float(best_dir)
        h = project_box_sum(h, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        else:
            h = best_h.copy()
            break
    return best_h, best_score


def active_subgrad(h, target, n_steps, step_base, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        rng = c_max - c_min
        if rng < 1e-12:
            break
        threshold = c_max - active_frac * rng
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[k:] += (1.0 - h_cur[:n - k])
                    grad[:n - k] -= h_cur[k:]
            else:
                kk = -k
                if kk < n:
                    grad[:n - kk] += (1.0 - h_cur[kk:])
                    grad[kk:] -= h_cur[:n - kk]
        grad /= float(len(active))
        step = step_base / np.sqrt(float(it + 1))
        h_new = project_box_sum(h_cur - step * grad, target)
        h_avg = project_box_sum(0.5 * (h_cur + h_new), target)
        sc1 = compute_c5(h_new)
        sc2 = compute_c5(h_avg)
        if sc1 <= sc2:
            h_cur = h_new
        else:
            h_cur = h_avg
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)
```
