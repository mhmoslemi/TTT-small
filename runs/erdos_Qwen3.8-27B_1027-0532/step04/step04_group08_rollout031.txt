def compute_corr(h):

def project_feasible(h, target_sum):

        else:

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def sm_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    wt = np.exp(shifted)

    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)

            grad[:n - s] -= w * h[s:]

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def sm_obj(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def sm_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = wt[s + n - 1]
        if w < 1e-10:
            continue
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            grad[:n + s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:n + s]
    return grad


def lbfgs_smooth(h, target, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h, compute_c5(h)
    try:
        res = minimize(sm_obj, sm_grad, h, args=(alpha,),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * len(h),
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

        ones = np.where(h_cur == 1.0)[0]

        if len(ones) == 0 or len(zeros) == 0:

        found = False

            if sc < best_score:
                best_score = sc
                best_h = h_ls

    if _rem() > 40:
        h_bin = project_feasible((best_h > 0.5).astype(float), target)
        h_sa, sc = sa_binary(h_bin, target, time.time() + min(30, _rem() * 0.5), rng,
                             T_init=0.03, T_final=0.0005, max_iters=30000)
        if sc < best_score:
            best_score = sc
            best_h = h_sa

    if _rem() > 30:
        for a in [100000, 300000]:
            if _rem() < 10:
                break
            h_new, sc = lbfgs_smooth(best_h, target, a, 200, time.time() + min(15, _rem() * 0.4))
            if sc < best_score:
                best_score = sc
                best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
