    corr = compute_corr(h)
    over = corr[n-1:]  # shape (n,)
    O_max = float(np.max(over))
    shifted = alpha * (over - O_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = exp_shifted / exp_shifted.sum()  # softmax weights
    
    grad = np.zeros(n)
    for i in range(n):
        # First sum: k from 0 to n-1-i
        # sum of s[k] * (1-h[i+k])
        if n-1-i >= 0:
            first = np.dot(s[:n-i], 1.0 - h[i:i+n-i])
        else:
            first = 0.0
        # Second sum: k from i to n-1
        # sum of s[k] * h[i-k]
        if i <= n-1:
            second = np.dot(s[i:], h[0:n-i])
        else:
            second = 0.0
        grad[i] = dx * (first - second)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_mu = 0.0
_g_h_full = None
_g_block_s = 0
_g_block_e = 0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)[n-1:])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_overlaps(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:]

def lse_value_and_grad(h, alpha, mu, target):
    n = _g_n
    dx = 2.0 / n
    over = get_overlaps(h)
    O_max = float(np.max(over))
    shifted = alpha * (over - O_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = exp_s / exp_s.sum()
    lse_val = O_max + float(np.log(exp_s.sum())) / alpha
    
    # Penalty
    sum_diff = np.sum(h) - target
    pen_val = mu * sum_diff * sum_diff
    total_val = lse_val + pen_val
    
    # Gradient of LSE
    grad_lse = np.zeros(n)
    for i in range(n):
        first = np.dot(s[:n-i], 1.0 - h[i:i+n-i]) if n-i > 0 else 0.0
        second = np.dot(s[i:], h[:n-i]) if n-i > 0 else 0.0
        grad_lse[i] = dx * (first - second)
    
    # Gradient of penalty
    grad_pen = 2.0 * mu * sum_diff * np.ones(n)
    
    return total_val, grad_lse + grad_pen

def lse_obj(x):
    val, _ = lse_value_and_grad(x, _g_alpha, _g_mu, _g_target)
    return val

def lse_obj_grad(x):
    val, grad = lse_value_and_grad(x, _g_alpha, _g_mu, _g_target)
    return val, grad

def block_lse_obj_grad(x):
    h_full = _g_h_full.copy()
    h_full[_g_block_s:_g_block_e] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    over = get_overlaps(h_full)
    n = _g_n
    O_max = float(np.max(over))
    shifted = _g_alpha * (over - O_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = exp_s / exp_s.sum()
    lse_val = O_max + float(np.log(exp_s.sum())) / _g_alpha
    return lse_val

def lbfgsb_round(h, n, target, alphas, mu, maxiter, ftol):
    global _g_n, _g_target, _g_alpha, _g_mu
    _g_n = n
    _g_target = target
    _g_mu = mu
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj_grad, h_cur, jac=True, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                cur_score = sc
        except Exception:

    return h_cur, cur_score

def momentum_subgrad(h, n, target, n_steps, n_active, lr_init, lr_decay, mom, temp0):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    temp = temp0
    for step in range(n_steps):
        over = get_overlaps(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if n - k > 0:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
            if k > 0:
                grad[k:] -= wk * h_cur[:n-k]
        grad *= dx
        v = mom * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_curr + 1e-12:
            h_cur = h_new
        else:
            lr *= lr_decay
            v *= mom * 0.5
            if lr < 1e-11:
                break
        temp *= 0.999
    return best_h, best_score

- h[i] appears in O_k when j+k = i, i.e., j = i-k (if i-k ≥ 0, i.e., i ≥ k): contributes dx × (-h[i-k])

def lse_gradient_vec(h, s, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        # s[k] contributes to grad[i] for i in [0, n-1-k] with +(1-h[i+k])
        # and for i in [k, n-1] with -h[i-k]
        if n - 1 - k >= 0:

        if k <= n-1:
            grad[k:] -= s[k] * h[:n-k]
    return grad * dx

_g_mu = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)[n-1:])) * dx


def get_overlaps(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:]


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value_and_grad(h, alpha, mu, target):
    n = _g_n
    dx = 2.0 / n
    over = get_overlaps(h)
    O_max = float(np.max(over))
    shifted = alpha * (over - O_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    exp_sum = float(exp_s.sum())
    s = exp_s / exp_sum
    lse_val = O_max + np.log(exp_sum) / alpha
    sum_diff = np.sum(h) - target
    pen_val = mu * sum_diff * sum_diff
    total_val = lse_val + pen_val
    grad = np.zeros(n)
    for k in range(n):
        if n - k > 0:
            grad[:n-k] += s[k] * (1.0 - h[k:n])
            grad[k:] -= s[k] * h[:n-k]
    grad = grad * dx + 2.0 * mu * sum_diff
    return total_val, grad

    val, _ = lse_value_and_grad(x, _g_alpha, _g_mu, _g_target)

    return val, grad

    global _g_n, _g_target, _g_alpha, _g_mu
    _g_n = n
    _g_target = target
    _g_mu = mu
    bounds = [(0.0, 1.0)] * n

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def momentum_subgrad(h, n, target, n_steps, n_active, lr_init, lr_decay, mom, temp0):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    temp = temp0
    for step in range(n_steps):
        over = get_overlaps(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if n - k > 0:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
