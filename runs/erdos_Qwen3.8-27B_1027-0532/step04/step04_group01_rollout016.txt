def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""

    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def make_thue_morse(n, rng):
    target = n // 2
    h = np.zeros(n)
    for i in range(n):
        # Number of 1s in binary representation of i
        t = bin(i).count('1') % 2
        h[i] = t
    # Adjust to have exactly target ones
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0
    return h
