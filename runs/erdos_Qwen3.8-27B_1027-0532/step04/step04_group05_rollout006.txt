```python
    n = len(h)
    a = 1.0 - h
    # Compute all v_k
    corr = np.correlate(h, a, mode='full') * dx  # length 2n-1
    # corr[k] corresponds to lag k-(n-1)
    # v_k = corr[k] for k = 0..2n-2, lag = k-(n-1)
    
    # Compute softmax weights
    max_corr = np.max(corr)
    log_weights = alpha * (corr - max_corr)
    log_weights = np.clip(log_weights, -500, 0)
    weights = np.exp(log_weights)
    weights /= np.sum(weights)
    
    # Now compute gradient
    # grad[j] = dx * sum_k weights[k] * (a[j+k-(n-1)]*1_{valid} - h[j-k+(n-1)]*1_{valid})

    # = sum_l' w[l'] * a[j+l']  (substituting l' = -lag)

    # w is defined on lags from -(n-1) to (n-1)

    # Create w as a sequence centered at 0
    w_seq = np.zeros(2*n)  # indexed 0 to 2n-1, center at n-1
    w_seq[n-1-(n-1):n-1+(n-1)+1] = weights  # w_seq[lag + n-1] = weights[lag]
```

```python
    n = len(h)
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full') * dx
    max_c = np.max(corr)
    logits = alpha * (corr - max_c)
    logits = np.clip(logits, -500, 0)
    w = np.exp(logits)
    w_sum = np.sum(w)
    w = w / w_sum
    
    # Gradient = dx * sum_k w[k] * (a[j + lag_k] - h[j - lag_k])
    # where lag_k = k - (n-1)
    
    grad = np.zeros(n)
```

Let me think about a **bilinear program** approach. The problem is:
min max_k h^T A_k (1-h)
where A_k is the shift matrix for lag k.

Expanding: h^T A_k (1-h) = h^T A_k - h^T A_k h

The first term is linear in h, the second is bilinear (quadratic but with zero diagonal in a sense).

This is a quadratic optimization problem with a max constraint. Specifically:
min_t t
s.t. h^T A_k - h^T A_k h <= t/dx for all k
     sum(h) = n/2
     0 <= h <= 1

The constraint h^T A_k - h^T A_k h <= t/dx is equivalent to:
h^T A_k h - h^T A_k + t/dx >= 0

This is a quadratic constraint. The matrix A_k is not necessarily positive or negative definite, so this is a non-convex QCQP.

However, we can use the fact that h^T A_k h = sum_{i,j} h_i (A_k)_{ij} h_j = sum_{i} h_i h_{i+k} (for the appropriate definition).

For the k=0 case: h^T A_0 h = sum_i h_i² (if A_0 is identity) or something similar. Let me be precise.

v_0 = dx * sum_i h_i (1-h_i) = dx * (sum h_i - sum h_i²) = dx * (n/2 - sum h_i²)

So minimizing v_0 is equivalent to maximizing sum h_i², which means making h as "spread out" as possible (close to 0 or 1).

For k≠0: v_k = dx * sum_i h_i (1-h_{i+k}) = dx * (sum_i h_i - sum_i h_i h_{i+k}) = dx * (n/2 - sum_i h_i h_{i+k})

So minimizing v_k is equivalent to maximizing sum_i h_i h_{i+k}, which means making h "correlated" with its shifted version.

There's a tension: we want h to be close to 0/1 (for k=0) but also correlated with its shifts (for k≠0).

This suggests the optimal solution might be a "staircase" or "gradual" function that is close to 0 or 1 most of the time but has a transition region.

Let me think about the L-BFGS-B implementation more carefully.

The objective function for L-BFGS-B:
f(h) = F_alpha(h) + mu * (sum(h) - target)²

where F_alpha(h) = (1/alpha) * log(sum_k exp(alpha * v_k))

The gradient:
∇f(h) = ∇F_alpha(h) + 2*mu*(sum(h)-target) * 1

I'll use scipy.optimize.minimize with method='L-BFGS-B', bounds=[(0,1)]*n.

For the alpha schedule: start with alpha=10, increase to alpha=10000 or more.

For mu: start with mu=100, increase over time.

Let me also think about a key insight: **the number of "active" k values**.

For a good solution, only a few k values will be near the maximum. The softmax weights w_k will be concentrated on these few k values. So in practice, the gradient computation can be approximated by considering only the top few k values.

But for accuracy, I'll compute the full sum.

Let me also think about **symmetry**. The problem has the symmetry h → 1-h (complement). If h is optimal, then 1-h is also optimal (with the same C_5 value). Wait, let me check:

By substitution y = x+k: = max_k ∫ h(y)(1-h)(y-k) dy = max_k ∫ h(y)(1-h)(y-k) dy

This is the same as C_5(h) with lag -k instead of k. Since we take max over all k, C_5(1-h) = C_5(h). ✓

So the problem is symmetric under h → 1-h. This means we can without loss of generality look for solutions with sum(h) = n/2 (which we already have as a constraint).

Another symmetry: h(x) → h(2-x) (reflection). This preserves the integral and the max (since the lags are symmetric).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, mu):
    """Compute smooth objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full') * dx
    max_c = np.max(corr)
    logits = alpha * (corr - max_c)
    logits = np.clip(logits, -500.0, 0.0)
    exp_logits = np.exp(logits)
    log_sum = np.log(np.sum(exp_logits))
    val = max_c + log_sum / alpha
    
    # Penalty for sum constraint
    sum_h = np.sum(h)
    val += mu * (sum_h - target) ** 2
    
    # Compute gradient of F_alpha
    w = exp_logits / np.sum(exp_logits)
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        # a[j + lag] term
        j0 = max(0, -lag)
        j1 = min(n, n - lag)
        if j1 > j0:
            grad[j0:j1] += wk * a[j0+lag:j1+lag]
        # -h[j - lag] term
        j0 = max(0, lag)
        j1 = min(n, n + lag)
        if j1 > j0:
            grad[j0:j1] -= wk * h[j0-lag:j1-lag]
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * mu * (sum_h - target)
    
    return val, grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha, mu, target):
    n = len(h)
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full') * dx
    max_c = np.max(corr)
    logits = alpha * (corr - max_c)
    logits = np.clip(logits, -500.0, 0.0)
    exp_l = np.exp(logits)
    w_sum = np.sum(exp_l)
    w = exp_l / w_sum
    val = max_c + np.log(w_sum) / alpha
    sum_diff = np.sum(h) - target
    val += mu * sum_diff * sum_diff
    
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        # a[j + lag]
        j0 = max(0, -lag)
        j1 = min(n, n - lag)
        if j1 > j0:
            grad[j0:j1] += wk * a[j0+lag:j1+lag]
        # -h[j - lag]
        j0 = max(0, lag)
        j1 = min(n, n + lag)
        if j1 > j0:
            grad[j0:j1] -= wk * h[j0-lag:j1-lag]
    grad *= dx
    grad += 2.0 * mu * sum_diff
    
    return val, grad

def lbfgsb_solve(h0, alpha, mu, target, maxiter=2000, gtol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def fun_and_grad(x):
        return smooth_obj_grad(x, alpha, mu, target)
    
    res = minimize(fun_and_grad, h0, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol,
                                           'ftol': 1e-15})
    return res.x, res.fun, res.nit, res.success

def lbfgsb_homotopy(h_start, alphas, mus, target, maxiter_per_step=2000):
    h = h_start.copy()
    for i, alpha in enumerate(alphas):
        mu = mus[min(i, len(mus)-1)]
        h, _, _, _ = lbfgsb_solve(h, alpha, mu, target, maxiter=maxiter_per_step)
        h = project_feasible(h, target)
    return h
