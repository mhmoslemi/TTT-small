**Fictitious Play approach:**
- Maintain a weight vector w over the n lags (w[k] ≥ 0, sum w = 1)
- At each iteration:
  1. Compute g(h) = sum_k w[k] * corr_k(h) (this is the "expected loss")
  2. Optimize h to minimize g(h) (this is smooth, so SLSQP works well)
  3. Update w: w[k] ∝ exp(T * corr_k(h)) for some temperature T
  4. As T → ∞, w concentrates on the worst lag

This is essentially the "exceedance" or "importance sampling" approach. The key advantage is that step 2 is a SMOOTH optimization problem (weighted sum of correlations), which SLSQP handles very well.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_w = None


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(x):
    corr = compute_corr(x)
    n = _g_n
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def sum_constraint(x):
    return np.sum(x) - _g_target


def sum_constraint_jac(x):
    return np.ones(_g_n)


def fp_obj(x):
    corr = compute_corr(x)
    n = _g_n
    over = corr[n-1:n-1+n]
    return float(_g_w @ over)


def fp_grad(x):
    n = _g_n
    w = _g_w
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        grad[:n-k] += wk * (1.0 - x[k:])
        grad[k:] -= wk * x[:n-k]
    return grad


def slsqp_lse(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def slsqp_fp(h, n, target, weights, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_w
    _g_n = n
    _g_target = target
    _g_w = weights
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    try:
        res = minimize(fp_obj, h, jac=fp_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        if compute_c5(h_new) <= compute_c5(h) + 0.001:
            return h_new
        return h
    except Exception:
        return h


def fictitious_play(h, n, target, n_iter, T_start, T_end, deadline, maxiter=200):
    global _g_n, _g_target, _g_w
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_vals = over * (2.0 / n)
        shifted = c_vals / T
        shifted -= np.max(shifted)
        w = np.exp(shifted)
        w /= w.sum()
        _g_w = w
        h_new = slsqp_fp(h_cur, n, target, w, maxiter, 1e-15, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_score


def adaptive_block(h, n, target, block_size, n_rounds, deadline, seed=42):
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(seed)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        made_progress = False
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            best_block = h_cur[s:e].copy()
            best_sc = cur_score
            sigma = 0.08
            for attempt in range(25):
                if time.time() > deadline:
                    break
                perturb = rng.randn(block_size) * sigma
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(best_block + perturb, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_block = h_test[s:e].copy()
                    cur_score = best_sc
                    h_cur = h_test
                    made_progress = True
                    sigma = min(sigma * 1.2, 0.3)
                else:
                    sigma *= 0.75
            if not made_progress and rd > 0:
                pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coordinate_sweep(h, n, target, n_sweeps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            corr = compute_corr(h_cur)
            over = corr[n-1:n-1+n]
            c_max = float(np.max(over)) * (2.0 / n)
            x0 = h_cur[i]
            best_x = x0
            best_f = c_max
            for trial in range(15):
                dx_trial = (1.0 - x0) * 0.1 * (2 ** trial) * (1 if trial % 2 == 0 else -1)
                if trial == 0:
                    dx_trial = x0 * 0.5
                x_trial = x0 + dx_trial
                if x_trial < -0.01 or x_trial > 1.01:
                    continue
                x_trial = np.clip(x_trial, 0.0, 1.0)
                h_test = h_cur.copy()
                h_test[i] = x_trial
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < best_f:
                    best_f = sc
                    best_x = x_trial
            if best_x != x0 and best_f < cur_score:
                h_cur[i] = best_x
                h_cur = project_feasible(h_cur, target)
                cur_score = best_f
    return h_cur, cur_score
