def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def make_qr_sequence(n, rng):

        else:

                if p % f == 0:
                    is_p = False
                    break
        if is_p:
            break
        p += 1
    h = np.zeros(n)

def compute_c5(h):

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    if x < 2:
        return False

def smooth_logsumexp_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def continuous_refine(h_init, n, target, t_start, time_limit):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        return smooth_logsumexp_objective(x, 200, n)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    alphas = [30, 80, 200, 500, 1000]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break

        def obj_a(x):
            return smooth_logsumexp_objective(x, alpha, n)

        try:
            res = minimize(obj_a, h_cur, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

        h_qr = make_qr_sequence(nt, rng)
        h_ls, c5_ls = local_search_v3(h_qr, nt, 200000, rng, t_alloc, start,
                                      t_init=0.02, t_final=0.00003)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

        # Also try resampled version of current best
        if time.time() < deadline - 30:
            h_res = resample_binary(best_h, nt, target_nt, rng)
            t_alloc2 = min(60, deadline - time.time())
            if t_alloc2 > 20:
                h_ls2, c5_ls2 = local_search_v3(h_res, nt, 150000, rng, t_alloc2,
                                                 start, t_init=0.015,
                                                 t_final=0.00002)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()

    # Phase 3: Frequency-optimal initialization + local search
    if time.time() < deadline - 60:
        n_use = len(best_h)
        t_alloc = min(100, deadline - time.time())
        if t_alloc > 30:
            h_freq = make_freq_optimal(n_use, rng, n_iters=2000)
            h_ls, c5_ls = local_search_v3(h_freq, n_use, 200000, rng, t_alloc,
                                          start, t_init=0.02, t_final=0.00003)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 4: More perturbation-based refinement on best
    for it in range(5):
        if time.time() > deadline:
            break
        n_use = len(best_h)
        target_use = n_use // 2.0
        t_alloc = min(80, deadline - time.time())
        if t_alloc < 30:
            break
        noise_scale = 0.3 + 0.1 * it
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert_bin = project_binary(h_pert, target_use)
        h_ls, c5_ls = local_search_v3(h_pert_bin, n_use, 150000, rng, t_alloc,
                                      start, t_init=0.012, t_final=0.00002)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 5: M-sequence based start
    if time.time() < deadline - 50:
        n_use = len(best_h)
        target_use = n_use // 2.0
        h_ms = make_msequence(n_use, rng)
        t_alloc = min(70, deadline - time.time())
        if t_alloc > 20:
            h_ls, c5_ls = local_search_v3(h_ms, n_use, 150000, rng, t_alloc,
                                          start, t_init=0.015, t_final=0.00002)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 6: Continuous refinement
    if time.time() < deadline - 40:
        n_use = len(best_h)
        target_use = n_use // 2.0
        t_alloc = min(80, deadline - time.time())
        if t_alloc > 25:
            h_cont, c5_cont = continuous_refine(best_h, n_use, target_use,
                                                start, time.time() + t_alloc)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()

    # Phase 7: Final binary local search polish
    if time.time() < deadline - 10:
        n_use = len(best_h)
        target_use = n_use // 2.0
        h_bin = project_binary(best_h, target_use)
        t_alloc = min(40, deadline - time.time())
        if t_alloc > 5:
            h_ls, c5_ls = local_search_v3(h_bin, n_use, 100000, rng, t_alloc,
                                          start, t_init=0.005, t_final=0.00001)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Final evaluation
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5:
        h_final = best_h
        final_c5 = best_c5

    return h_final, final_c5, n_final
