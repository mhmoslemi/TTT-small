import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def smooth_obj_func(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_grad_func(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)


def _eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def _eq_jac(x):
    return np.ones(_CONS_N)


def slsqp_smooth(h_start, target_sum, alphas, topks, maxiter, ftol, time_limit):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    n = _CONS_N
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    t0 = time.time()

    for i in range(len(alphas)):
        if time.time() - t0 > time_limit:
            break
        alpha = alphas[i]
        topk = topks[i]
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha, topk),
                           jac=smooth_grad_func, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def subgradient_opt(h_init, target_sum, max_iter, topk, alpha,
                    momentum, patience, seed, noise_scale, time_limit):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = 0.1
    stagnation = 0
    t0 = time.time()

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]

        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum

        grad = compute_grad_multi_lag(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)

        if noise_scale > 0.0:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)

            best_score = score
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 30:
            lr *= 0.93
            if lr < 0.0005:
                break
        elif stagnation > 15:
            lr *= 0.97
        else:
            lr = min(lr * 1.003, 0.5)

        if stagnation > patience:
            break

    return best_h, best_score


def lp_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i in range(m):
        idx = top_idx[i]
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c_vec, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target_sum)
    return None


def lp_refine(h_start, target_sum, max_iter, topk, time_limit, shrink=True):

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
