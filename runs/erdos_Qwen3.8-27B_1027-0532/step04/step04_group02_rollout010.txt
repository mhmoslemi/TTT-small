```python
def eval_swap_delta(h, corr, i1, i0, d_star, n):
    """Compute Δcorr[d_star] for swapping i1(1->0) and i0(0->1)"""
    delta = 0.0
    # Contribution from i1 (1->0):
    if 0 <= i1 - d_star < n:
        delta += -(1 - h[i1 - d_star])
    if 0 <= i1 + d_star < n:
        delta += h[i1 + d_star]
    # Contribution from i0 (0->1):
    if 0 <= i0 - d_star < n:
        delta += (1 - h[i0 - d_star])
    if 0 <= i0 + d_star < n:
        delta += -h[i0 + d_star]
    return delta
```

```python
def greedy_local_search(h, target_sum, max_passes=50, time_limit=None, start_time=None):
    """Greedy: try all swaps, pick the best one that improves C5"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr))
    
    for _ in range(max_passes):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        d_star = int(np.argmax(corr))
        d_star_lag = d_star - (n - 1)  # actual lag
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_i1 = -1
        best_i0 = -1
        
        for i1 in ones:
            for i0 in zeros:
                # Compute Δcorr[d_star] for this swap
                delta = 0.0
                # i1 goes 1->0
                p = i1 - d_star_lag
                if 0 <= p < n:
                    delta += -(1.0 - h[p])
                p = i1 + d_star_lag
                if 0 <= p < n:
                    delta += h[p]
                # i0 goes 0->1
                p = i0 - d_star_lag
                if 0 <= p < n:
                    delta += (1.0 - h[p])
                p = i0 + d_star_lag
                if 0 <= p < n:
                    delta += -h[p]
                
                if delta < best_delta:
                    best_delta = delta
                    best_i1 = i1
                    best_i0 = i0
        
        if best_i1 < 0:
            break  # No improving move found
        
        # Apply the swap
        h[best_i1] = 0
        h[best_i0] = 1
        
        # Update correlation (full recompute for safety)
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(corr))
        
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Shouldn't happen if delta was computed correctly
            # but due to the max being at a different position now,
            # the actual C5 might not decrease
            if new_c5 > best_c5:
                h[best_i1] = 1
                h[best_i0] = 0
                corr = np.correlate(h, 1.0 - h, mode='full')
                break
    
    return h, best_c5
```

```python
def fft_full_correlate(a, b):
    n_a = len(a)
    n_b = len(b)
    n = n_a + n_b - 1
    N = 1 << (n - 1).bit_length()
    C = np.fft.irfft(np.fft.rfft(a, N) * np.conj(np.fft.rfft(b, N)), N)
    return np.concatenate([C[N - n_b + 1:], C[:n_a]])
```

        h = project_binary(h, target_sum)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr))
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 12)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]
    recompute_every = max(50, n // 15)
    recompute_counter = 0

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.03:
                no_improve = 0
            else:
                break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        d_star = int(np.argmax(corr))
        d_lag = d_star - (n - 1)

        h_new = h.copy()
        moved = False

        if len(ones_cache) > 0 and len(zeros_cache) > 0:
            if rng.random() < 0.65:
                one_deltas = np.zeros(len(ones_cache))
                for idx, i in enumerate(ones_cache):
                    d = 0.0
                    pp = i - d_lag
                    if 0 <= pp < n:
                        d += -(1.0 - h[pp])
                    pp = i + d_lag
                    if 0 <= pp < n:
                        d += h[pp]
                    one_deltas[idx] = d
                zero_deltas = np.zeros(len(zeros_cache))
                for idx, i in enumerate(zeros_cache):
                    d = 0.0
                    pp = i - d_lag
                    if 0 <= pp < n:
                        d += (1.0 - h[pp])
                    pp = i + d_lag
                    if 0 <= pp < n:
                        d += -h[pp]
                    zero_deltas[idx] = d

                if rng.random() < 0.3:
                    top_k = min(5, len(ones_cache))
                    k_ones = np.argsort(one_deltas)[:top_k]
                    top_kz = min(5, len(zeros_cache))
                    k_zeros = np.argsort(zero_deltas)[:top_kz]
                    best_combo = 1e9
                    bi1, bi0 = k_ones[0], k_zeros[0]
                    for oi in k_ones:
                        for zi in k_zeros:
                            combo = one_deltas[oi] + zero_deltas[zi]
                            if combo < best_combo:
                                best_combo = combo
                                bi1, bi0 = oi, zi
                    i1a = ones_cache[bi1]
                    i0a = zeros_cache[bi0]
                else:
                    i1a = ones_cache[np.argmin(one_deltas)]
                    i0a = zeros_cache[np.argmin(zero_deltas)]
                h_new[i1a] = 0
                h_new[i0a] = 1
                moved = True
            else:
                i1a = ones_cache[rng.randint(len(ones_cache))]
                i0a = zeros_cache[rng.randint(len(zeros_cache))]
                h_new[i1a] = 0
                h_new[i0a] = 1
                moved = True

        if not moved:
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5

        if delta < -1e-14 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
