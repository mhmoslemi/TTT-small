```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    h_cur = h.copy()
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()

        diff = target_sum - h.sum()

def gradient_correlation(h, k_idx, n):
    """Compute gradient of correlation value at index k_idx w.r.t. h.
    The full correlation corr[m] where m in [0, 2n-2] corresponds to lag = m - (n-1).
    """
    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    if lag >= 0:
        # corr[m] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        # d/dh[j]: 
        #   if j >= lag and j <= n-1: h[j] appears as h[i+lag] with i=j-lag
        #     contribution: (1-h[j-lag])
        #   if j <= n-1-lag: h[j] appears in (1-h[i]) with i=j
        #     contribution: -h[j+lag]
        idx1 = np.arange(lag, n)
        grad[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, n - lag)
        grad[idx2] -= h[idx2 + lag]
    else:
        l = -lag
        # corr[m] = sum_{i=l}^{n-1} h[i-l] * (1-h[i])
        # = sum_{j=0}^{n-1-l} h[j] * (1-h[j+l])
        # d/dh[j]:
        #   if j <= n-1-l: h[j] appears as h[j], contribution: (1-h[j+l])
        #   if j >= l: h[j] appears in (1-h[j+l]) with j+l = i, contribution: -h[j-l]
        idx1 = np.arange(0, n - l)
        grad[idx1] += (1.0 - h[idx1 + l])
        idx2 = np.arange(l, n)
        grad[idx2] -= h[idx2 - l]
    return grad


def subgradient_polish(h, n, target, n_steps=500, n_active=5, temperature=0.005, lr_init=0.003, lr_decay=0.997):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):

        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Softmax weighting over top correlations
        top_idx = np.argsort(corr)[::-1][:n_active]
        shifted = corr[top_idx] / temperature
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            grad += w * gradient_correlation(h_cur, int(k), n)
        grad *= dx
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_curr:
            h_cur = h_new
        else:
            lr = max(lr * lr_decay, 1e-10)
            if lr < 1e-10:
                break
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def gradient_correlation(h, k_idx, n):
    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    if lag >= 0:
        idx1 = np.arange(lag, n)
        grad[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, n - lag)
        grad[idx2] -= h[idx2 + lag]
    else:
        l = -lag
        idx1 = np.arange(0, n - l)
        grad[idx1] += (1.0 - h[idx1 + l])
        idx2 = np.arange(l, n)
        grad[idx2] -= h[idx2 - l]
    return grad


def subgradient_polish(h, n, target, n_steps=500, n_active=5, temperature=0.004, lr_init=0.004, lr_decay=0.997):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        shifted = corr[top_idx] / temperature
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            grad += w * gradient_correlation(h_cur, int(k), n)
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_curr:
            h_cur = h_new
        else:
            lr = max(lr * lr_decay, 1e-10)
            if lr < 1e-10:
                break
    return best_h, best_score


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h, n, target, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def de_optimize(h_best, h_alt, n, target, pop_size, n_gens, F, CR, seed, deadline):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def memetic_refine(h, n, target, n_steps=50, lr=0.005):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for _ in range(n_steps):
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:3]
        shifted = corr[top_idx] / 0.005
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            grad += w * gradient_correlation(h_cur, int(k), n)
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
        else:
            lr *= 0.8
            if lr < 1e-9:
                break
    return best_h

            best_h = h_sg
        if tleft() > 80:
            h_ref = slsqp_round(best_h, n_final, target_final,
                                [500, 2000, 8000], 3000, 1e-16)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 6: Second DE with tighter neighborhood
    if tleft() > 200:
        h_p1 = best_h + rng.randn(n_final) * 0.03
        h_p1 = project_feasible(h_p1, target_final)
        h_p2 = best_h + rng.randn(n_final) * 0.06
        h_p2 = project_feasible(h_p2, target_final)
        h_de4 = de_optimize(best_h, h_p1, n_final, target_final,
                            20, 100, 0.35, 0.95, seed + 20, start_time + budget_s * 0.7)
        sc = compute_c5(h_de4)
        if sc < best_score:
            best_score = sc
            best_h = h_de4
        elif tleft() > 100:
            h_de5 = de_optimize(best_h, h_p2, n_final, target_final,
                                15, 80, 0.4, 0.95, seed + 30, start_time + budget_s * 0.78)
            sc = compute_c5(h_de5)
            if sc < best_score:
                best_score = sc
                best_h = h_de5

    # Phase 7: SLSQP + subgradient cycle
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n_final, target_final,
                            [300, 1000, 4000, 10000], 3000, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        if tleft() > 60:
            h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                              n_steps=500, n_active=4,
                                              temperature=0.002, lr_init=0.003, lr_decay=0.998)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 50:
        h_ref = slsqp_round(best_h, n_final, target_final,
                            [5000, 15000, 30000, 50000], 2000, 1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final subgradient
    if tleft() > 25:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                          n_steps=300, n_active=3,
                                          temperature=0.001, lr_init=0.002, lr_decay=0.999)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
