import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h = h.copy()
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_step(h, target, alpha, n, maxiter=2000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    
    def cfun(x):
        return np.sum(x) - target
    
    def cjac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def binary_local_search(h_bin, n, target, max_iter=50000):
    """Local search in binary space: flip pairs of bits."""
    h = h_bin.copy().astype(float)
    target = n // 2
    best_score = compute_c5(h)
    
    for it in range(max_iter):
        improved = False
        # Try flipping a 1 to 0 and a 0 to 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            sc = compute_c5(h)
            if sc < best_score - 1e-15:
                best_score = sc
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:

    return h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        def fun(x):
            return lse_obj_and_grad(x, alpha)[0]
        def jac(x):
            return lse_obj_and_grad(x, alpha)[1]
        def cfun(x):
            return np.sum(x) - target
        def cjac(x):
            return np.ones(n)
        try:
            res = minimize(fun, h_cur, jac=jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def binary_round(h, target):
    """Round h to binary, maintaining sum constraint."""
    n = len(h)
    # Sort by value, assign 1 to top n/2, 0 to bottom n/2
    # But we want to be close to the original
    # Use a threshold with tie-breaking
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    # Determine how many ones we need
    n_ones = int(round(target))
    # Take the top n_ones by value
    for i in range(n_ones):
        h_bin[sorted_idx[i]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_swaps=200000, rng=None):
    """Local search in binary space."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.copy().astype(float)
    corr = compute_corr(h)
    best_max = float(np.max(corr))
    dx = 2.0 / n
    best_c5 = best_max * dx
    stuck_count = 0
    
    for swap_idx in range(max_swaps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random i from ones, j from zeros
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        if new_max < best_max - 1e-10:
            corr = new_corr
            best_max = new_max
            best_c5 = new_max * dx
            stuck_count = 0
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
            stuck_count += 1
        
        if stuck_count > 5000:
            break
    
    return h, best_c5

def continuous_polish_from_binary(h_bin, target, deadline):
    """Take a binary solution and polish it in continuous space."""
    h = h_bin.copy().astype(float)
    h = project_feasible(h, target)
    # Do SLSQP from here
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    h, sc = slsqp_refine(h, target, alphas, 2000, 1e-16, deadline)
    return h, sc

def subgradient_polish(h, target, n_steps, step_size, deadline):
    """Active subgradient method."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - 0.01 * c_max)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        
        step = step_size / (1.0 + 0.01 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_score

        best_h = h5

    # Phase 6: Fine subgradient
    dl6 = start_time + 950.0
    h6, sc6 = subgrad_polish(best_h, target, 5000, 0.02, 0.01, dl6)
    if sc6 < best_score:
        best_score = sc6
        best_h = h6

    # Phase 7: Try higher resolution
    if time.time() < deadline_final - 150.0:
        n_new = 600
        target_new = n_new / 2.0
        h_interp = interpolate_h(best_h, n_new, target_new)
        sc_interp = compute_c5(h_interp)
        if sc_interp < best_score + 0.02:
            dl7 = min(deadline_final - 80.0, time.time() + 130.0)
            h7, sc7 = slsqp_refine(h_interp, target_new,
                                   [50, 500, 5000, 50000, 200000], 1500, 1e-15, dl7)
            if sc7 < best_score:
                best_score = sc7
                best_h = h7

    # Phase 8: Final high-alpha SLSQP on current best
    n_now = len(best_h)
    target_now = n_now / 2.0
    dl8 = deadline_final
    h8, sc8 = slsqp_refine(best_h, target_now, [200000, 500000, 1000000], 1500, 1e-16, dl8)
    if sc8 < best_score:
        best_score = sc8
        best_h = h8

    # Final projection and evaluation
    n_fin = len(best_h)
    best_h = project_feasible(best_h, n_fin / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n_fin
