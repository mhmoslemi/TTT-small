def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def cd_step(h, i0, C):
    """Find optimal h[i0] given C (current correlations) and other h values fixed."""
    n = len(h)
    h_i0 = h[i0]
    a_list = []
    b_list = []
    for k in range(1, n):
        # C[n-1+k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # h[i0] coefficient:
        a1 = 0.0
        if i0 + k < n:
            a1 += (1.0 - h[i0 + k])
        if i0 - k >= 0:
            a1 -= h[i0 - k]
        if abs(a1) > 1e-15:
            b1 = C[n-1+k] - a1 * h_i0
            a_list.append(a1)
            b_list.append(b1)
        # C[n-1-k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        a2 = 0.0
        if i0 - k >= 0:
            a2 += (1.0 - h[i0 - k])
        if i0 + k < n:
            a2 -= h[i0 + k]
        if abs(a2) > 1e-15:
            b2 = C[n-1-k] - a2 * h_i0
            a_list.append(a2)
            b_list.append(b2)
    
    if not a_list:
        return h_i0
    
    a_arr = np.array(a_list)
    b_arr = np.array(b_list)
    
    # Binary search on t
    t_low = 0.0
    t_high = float(np.max(C))  # safe upper bound
    
    for _ in range(40):
        t_mid = (t_low + t_high) / 2.0
        x_low = 0.0
        x_high = 1.0
        feasible = True
        for idx in range(len(a_arr)):
            if a_arr[idx] > 1e-15:
                x_high = min(x_high, (t_mid - b_arr[idx]) / a_arr[idx])
            elif a_arr[idx] < -1e-15:
                x_low = max(x_low, (t_mid - b_arr[idx]) / a_arr[idx])
            else:
                if b_arr[idx] > t_mid + 1e-15:
                    feasible = False
                    break
        if feasible and x_low <= x_high:
            t_high = t_mid
        else:
            t_low = t_mid
    
    # Find optimal x at t_high
    x_low = 0.0
    x_high = 1.0
    for idx in range(len(a_arr)):
        if a_arr[idx] > 1e-15:
            x_high = min(x_high, (t_high - b_arr[idx]) / a_arr[idx])
        elif a_arr[idx] < -1e-15:
            x_low = max(x_low, (t_high - b_arr[idx]) / a_arr[idx])
    
    x_low = max(x_low, 0.0)
    x_high = min(x_high, 1.0)
    x_opt = (x_low + x_high) / 2.0
    return x_opt

def update_C(C, h, i0, old_val, new_val):
    """Update C incrementally when h[i0] changes from old_val to new_val."""
    n = len(h)
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return C
    C = C.copy()
    for k in range(1, n):
        # C[n-1+k]: a1 * delta
        a1 = 0.0
        if i0 + k < n:
            a1 += (1.0 - h[i0 + k])
        if i0 - k >= 0:
            a1 -= h[i0 - k]
        C[n-1+k] += a1 * delta
        
        # C[n-1-k]: a2 * delta
        a2 = 0.0
        if i0 - k >= 0:
            a2 += (1.0 - h[i0 - k])
        if i0 + k < n:
            a2 -= h[i0 + k]
        C[n-1-k] += a2 * delta
    
    # C[n-1]: h[i](1-h[i]) changes
    # d/dh[i0] of h[i0](1-h[i0]) = 1 - 2*h[i0]
    # But this is a change in C[n-1] = sum h[j](1-h[j])
    # C[n-1] changes by: new*(1-new) - old*(1-old)
    C[n-1] += (new_val * (1 - new_val) - old_val * (1 - old_val))
    
    return C

def coordinate_descent(h_start, target, max_sweeps, tol=1e-10, time_limit=None):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    C = np.correlate(h, 1.0 - h, mode='full')
    
    for sweep in range(max_sweeps):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        max_before = np.max(C)
        for i in range(n):
            old = h[i]
            new = cd_step(h, i, C)
            if abs(new - old) > 1e-12:
                h[i] = new
                C = update_C(C, h, i, old, new)
        h = project_feasible(h, target)
        C = np.correlate(h, 1.0 - h, mode='full')
        max_after = np.max(C)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if max_after < max_before - tol:
            continue
        else:
            break
    
    return best_h, best_c5

def binary_local_search(h_start, target, max_iter, seed, time_limit=None):

def smooth_obj(x, alpha, topk):
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(x, alpha, topk):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        w = weights[idx_i]
        if w < 1e-14:
            continue
        lag = top_idx[idx_i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - x[:n - lag])
                grad[:n - lag] -= w * x[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - x[m:])
                grad[m:] -= w * x[:n - m]
    return grad

def slsqp_optimize(h_start, target, alpha_topk_pairs, maxiter, ftol):

    _CONS_N = len(h_start)
    _CONS_TARGET = target
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                smooth_obj, h_cur, args=(alpha, topk),
                jac=smooth_grad, method='SLSQP',

                options={'maxiter': maxiter, 'ftol': ftol}

            h_new = project_feasible(res.x, target)
            score = compute_c5(h_new)
            if score < best_score:

        except Exception:

def interpolate_h(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
