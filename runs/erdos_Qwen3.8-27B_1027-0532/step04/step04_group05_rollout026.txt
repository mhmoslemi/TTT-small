import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_k(h, k, n, dx):
    """Gradient of g_k with respect to h."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    grad *= dx
    return grad

def softmax_gradient_descent(h, n, target, max_iters, time_limit, top_count=3, temperature=50):
    """Gradient descent using softmax-weighted top-k gradients."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    stuck = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if stuck > 500:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        g = corr * dx
        sorted_idx = np.argsort(g)[::-1]
        top_idx = sorted_idx[:top_count]
        top_vals = g[top_idx]
        
        # Softmax weights
        max_val = top_vals.max()
        weights = np.exp(temperature * (top_vals - max_val + 1e-10))
        weights = weights / weights.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            k = ki - n_half
            grad += wi * compute_grad_k(h_cur, k, n, dx)
        
        # Line search
        improved = False
        for lr in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                stuck = 0
                improved = True
                break
            elif sc < compute_c5(h_cur) + 1e-10:
                # Accept even small improvements to current
                h_cur = h_try
        
        if not improved:
            stuck += 1
    
    return best_h, best_c5

def binary_local_search(h_bin, n, target, max_iters, time_limit, rng):
    """Systematic local search on binary solutions."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h_bin.copy()
    best_c5 = compute_c5(h_cur)
    
    ones = set(np.where(h_cur == 1.0)[0].tolist())
    zeros = set(np.where(h_cur == 0.0)[0].tolist())

            h_try[i] = 0.0
            h_try[j] = 1.0
            
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h_cur = h_try
                best_c5 = c5_try
                ones.remove(i)
                ones.add(j)
                zeros.add(i)
                zeros.remove(j)
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
    
    return h_cur, best_c5

def smooth_obj(h, alpha, target, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, target, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def hybrid_binary_fractional(h_frac, h_bin, n, target, time_limit, rng):

    start = time.time()
    best_h = h_bin.copy() if compute_c5(h_bin) < compute_c5(h_frac) else h_frac.copy()
    best_c5 = compute_c5(best_h)

            h_pert = h_work.copy()
            h_pert[idxs] = rng.uniform(0.2, 0.8, size=n_frac)

        if not improved:
            stuck += 1
            if stuck % 200 == 0:
                temp = max(5.0, temp * 0.7)

    return best_h, best_c5


def binary_local_search(h_bin, n, target, max_iters, time_limit, rng, use_sa=False, sa_temp=0.01):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_bin.copy()
    best_c5 = compute_c5(h_cur)
    cur_c5 = best_c5

    ones_arr = np.where(h_cur == 1.0)[0]
    zeros_arr = np.where(h_cur == 0.0)[0]
    no_improve = 0
    sa_schedule = sa_temp

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 800 and not use_sa:
            break
        if no_improve > 3000:
            break

        if use_sa:
            sa_schedule = max(0.001, sa_schedule * 0.9999)

        improved = False
        n_tries = 200
        for _ in range(n_tries):
            if len(ones_arr) == 0 or len(zeros_arr) == 0:
                break
            i = ones_arr[rng.randint(len(ones_arr))]
            j = zeros_arr[rng.randint(len(zeros_arr))]

            h_try = h_cur.copy()
            h_try[i] = 0.0
            h_try[j] = 1.0

            c5_try = compute_c5(h_try)
            delta = c5_try - cur_c5

            accept = False
            if c5_try < cur_c5 - 1e-15:
                accept = True
            elif use_sa and delta < 0.5 and rng.random() < np.exp(-delta / sa_schedule):
                accept = True

            if accept:
                h_cur = h_try
                cur_c5 = c5_try
                ones_arr = np.where(h_cur == 1.0)[0]
                zeros_arr = np.where(h_cur == 0.0)[0]
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

    return h_cur, best_c5


def penalty_obj(h, alpha, target, n, t_val):
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * (2.0 / n)
    excess = np.maximum(0.0, g - t_val)
    return float(alpha * np.sum(excess ** 2))


def penalty_slsqp(h_start, n, target, t_init, time_limit, rng):
    start = time.time()
    bounds = [(0.0, 1.0)] * n

    res = project_feasible(best_h_initial, n / 2.0)
    n_local = len(best_h_initial)
    target_local = n_local / 2.0
    bounds = [(0.0, 1.0)] * n_local

    def pf_fun(x):
        return np.sum(x) - target_local

    def pf_jac(x):
        return np.ones(n_local)

    cons = [{'type': 'eq', 'fun': pf_fun, 'jac': pf_jac}]
    h_cur = best_h_initial copy() if False else h_start.copy()
    t_val = t_init
    alpha = 10.0

    for phase in range(30):
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(penalty_obj, h_cur, args=(alpha, target, n, t_val),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
        t_val = max(best_c5_global * 0.99, t_val * 0.98)
        alpha *= 2.0

    return h_cur


best_h_initial = None
best_c5_global = 1.0


def smooth_obj_func(h, alpha, target, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n

    def sfun(x):
        return np.sum(x) - target

    def sjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha, target, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def run(seed=42, budget_s=1300, **kwargs):
    global best_h_initial, best_c5_global
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h_initial = h_init.copy()
    best_c5_global = compute_c5(h_init)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 250:
        h_sg, c5_sg = softmax_grad_descent(best_h, n, target,
                                            max_iters=80000, time_limit=min(100, tleft() * 0.2),
                                            top_count=3, temperature=50.0)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 200:
        h_slq = slsqp_optimize(best_h, n, target,
                                [200, 1000, 5000, 20000, 100000],
                                600, 1e-14, min(150, tleft() * 0.25))
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq
        best_c5_global = best_score

    if tleft() > 150:
        n_ones = int(round(target))
        idx_top = np.argsort(best_h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx_top[:n_ones]] = 1.0

        h_bs, c5_bs = binary_local_search(h_bin, n, target,
                                           max_iters=300000, time_limit=min(80, tleft() * 0.25),
                                           rng=rng, use_sa=False)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs

    if tleft() > 150:
        n_ones = int(round(target))
        idx_top = np.argsort(best_h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx_top[:n_ones]] = 1.0

        h_sa, c5_sa = binary_local_search(h_bin, n, target,
                                           max_iters=400000, time_limit=min(80, tleft() * 0.25),
                                           rng=rng, use_sa=True, sa_temp=0.05)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
        best_c5_global = best_score

    if tleft() > 120:
        h_pen = penalty_slsqp(best_h, n, target, best_score * 0.99,
                               min(80, tleft() * 0.3), rng)
        sc = compute_c5(h_pen)
        if sc < best_score:
            best_score = sc
            best_h = h_pen

    if tleft() > 120:
        h_sg2, c5_sg2 = softmax_grad_descent(best_h, n, target,
                                              max_iters=60000, time_limit=min(80, tleft() * 0.25),
                                              top_count=5, temperature=20.0)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    if tleft() > 100:
        for noise_level in [0.005, 0.01, 0.02, 0.05]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_level
            h_pert = project_feasible(h_pert, target)
            h_pol = slsqp_optimize(h_pert, n, target,
                                    [5000, 20000, 100000],
                                    500, 1e-15, min(50, tleft() * 0.2))
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    if tleft() > 60:
        n_ones = int(round(target))
        idx_top = np.argsort(best_h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx_top[:n_ones]] = 1.0
        h_bs2, c5_bs2 = binary_local_search(h_bin, n, target,
                                             max_iters=200000, time_limit=min(50, tleft() * 0.4),
                                             rng=rng, use_sa=True, sa_temp=0.02)
        if c5_bs2 < best_score:
            best_score = c5_bs2
            best_h = h_bs2

    if tleft() > 40:
        h_final_try = slsqp_optimize(best_h, n, target,
                                      [20000, 100000, 500000],
                                      400, 1e-15, min(35, tleft() * 0.5))
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    if tleft() > 25:
        h_sg3, c5_sg3 = softmax_grad_descent(best_h, n, target,
                                              max_iters=30000, time_limit=min(25, tleft() * 0.6),
                                              top_count=3, temperature=100.0)
        if c5_sg3 < best_score:
            best_score = c5_sg3
            best_h = h_sg3

    h_final = project_feasible(best_h, n / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
