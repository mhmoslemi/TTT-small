def slsqp_refine(h_start, n, target, time_limit=None, start_time=None):
    """SLSQP continuous relaxation."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        # Use soft-max of the top values
        top_p = max(5, n // 30)
        sorted_v = np.sort(corr)
        top = sorted_v[-top_p:]
        alpha = 500.0
        max_top = np.max(top)
        shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
        return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        # Numerical gradient (expensive but works)
        eps = 1e-7
        g = np.zeros(n)
        f0 = obj(x)
        for i in range(0, n, max(1, n // 100)):  # Subsample gradient
            x[i] += eps
            g[i] = (obj(x) - f0) / eps
            x[i] -= eps
        # Scale to approximate full gradient
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for attempt in range(3):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break

            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

            for oi, i1 in enumerate(ones):
                s = 0.0
                j = i1 - k_star
                if 0 <= j < n:
                    s -= (1.0 - h[j])
                j = i1 + k_star
                if 0 <= j < n:
                    s += h[j]
                i1_scores[oi] = s
            i1_scores += rng.random(len(ones)) * t * 2.0
            i1 = ones[np.argmin(i1_scores)]

            i0_scores = np.zeros(len(zeros))
            for zi, i0 in enumerate(zeros):
                s = 0.0
                j = i0 - k_star
                if 0 <= j < n:
                    s += (1.0 - h[j])
                j = i0 + k_star
                if 0 <= j < n:
                    s -= h[j]
                i0_scores[zi] = s
            i0_scores += rng.random(len(zeros)) * t * 2.0
            i0 = zeros[np.argmin(i0_scores)]

            h_new[i1] = 0.0
            h_new[i0] = 1.0
            moved = True

        elif move_type < 0.65 and len(ones) > 0 and len(zeros) > 0:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0.0
            h_new[i0] = 1.0
            moved = True

        elif move_type < 0.85 and len(ones) >= 2 and len(zeros) >= 2:
            worst_lags = np.argsort(-corr)[:3]
            i1a = ones[rng.randint(len(ones))]
            i1b = ones[rng.randint(len(ones))]
            attempts = 0
            while i1b == i1a and attempts < 10:
                i1b = ones[rng.randint(len(ones))]
                attempts += 1

            i0_scores = np.zeros(len(zeros))
            for zi, i0 in enumerate(zeros):
                s = 0.0
                for wi in worst_lags:
                    k = wi - (n - 1)
                    j = i0 - k
                    if 0 <= j < n:
                        s += (1.0 - h[j])
                    j = i0 + k
                    if 0 <= j < n:
                        s -= h[j]
                i0_scores[zi] = s
            i0_scores += rng.random(len(zeros)) * t * 1.5
            i0a = zeros[np.argmin(i0_scores)]

            i0b_scores = np.zeros(len(zeros))
            for zi, i0 in enumerate(zeros):
                if i0 == i0a:
                    i0b_scores[zi] = 1e9
                    continue
                s = 0.0
                for wi in worst_lags:
                    k = wi - (n - 1)
                    j = i0 - k
                    if 0 <= j < n:
                        s += (1.0 - h[j])
                    j = i0 + k
                    if 0 <= j < n:
                        s -= h[j]
                i0b_scores[zi] = s
            i0b_scores += rng.random(len(zeros)) * t * 1.5
            valid_mask = i0b_scores < 1e8
            if np.any(valid_mask):
                candidates = np.where(valid_mask)[0]
                i0b = zeros[candidates[np.argmin(i0b_scores[candidates])]]
            else:
                i0b = zeros[rng.randint(len(zeros))]
                if i0b == i0a:
                    i0b = zeros[(np.argmin(i0_scores) + 1) % len(zeros)]

            h_new[i1a] = 0.0
            h_new[i1b] = 0.0
            h_new[i0a] = 1.0
            h_new[i0b] = 1.0
            moved = True

        else:
            if len(ones) > 0 and len(zeros) > 0:
                worst_m = np.argsort(-corr)[:5]
                i1 = ones[rng.randint(len(ones))]
                i0_scores = np.zeros(len(zeros))
                for zi, i0 in enumerate(zeros):
                    s = 0.0
                    for wi in worst_m:
                        k = wi - (n - 1)
                        j = i0 - k
                        if 0 <= j < n:
                            s += (1.0 - h[j])
                        j = i0 + k
                        if 0 <= j < n:
                            s -= h[j]
                        j = i1 - k
                        if 0 <= j < n:
                            s -= (1.0 - h[j])
                        j = i1 + k
                        if 0 <= j < n:
                            s += h[j]
                    i0_scores[zi] = s
                i0_scores += rng.random(len(zeros)) * t * 2.0
                i0 = zeros[np.argmin(i0_scores)]
                h_new[i1] = 0.0
                h_new[i0] = 1.0
                moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5

        if delta < -1e-14 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def greedy_hill_climb(h_start, target_sum, max_iters, rng):
    n = len(h_start)
    h = project_binary(h_start.astype(float), target_sum).copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    stall = 0
    best_corr_new = None

    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        worst_idx = np.argmax(corr)
        k_star = worst_idx - (n - 1)

        best_delta = -1e-15
        best_i1 = -1
        best_i0 = -1
        n_tries = min(30, min(len(ones), len(zeros)))

        for _ in range(n_tries):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            if i1 == i0:
                continue
            h[i1] = 0.0
            h[i0] = 1.0
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            h[i1] = 1.0
            h[i0] = 0.0
            delta = c5_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_i1 = i1
                best_i0 = i0
                best_corr_new = corr_new.copy()

        if best_i1 >= 0 and best_delta < -1e-14:
            h[best_i1] = 0.0
            h[best_i0] = 1.0
            corr = best_corr_new
            best_c5 += best_delta
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break

    return best_h, best_c5

def _slsqp_obj(x, n, target, top_p, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    sorted_v = np.sort(corr)
    top = sorted_v[-top_p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, time_limit=None, start_time=None):
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    top_p_vals = [max(5, n // 40), max(10, n // 25), max(15, n // 15)]
    alphas = [200.0, 500.0, 1000.0, 2000.0]

    for tp in top_p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(
                    lambda x: _slsqp_obj(x, n, target, tp, alpha),
                    h_cur, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq',
                                  'fun': lambda x: np.sum(x) - target,
                                  'jac': lambda x: np.ones(n)}],
                    options={'maxiter': 200, 'ftol': 1e-18})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_binary(h_init, target_init).copy()
    best_score = compute_c5(best_h)

    starts = [best_h.copy()]
    starts.append(project_binary(thue_morse(n_init), target_init))
    starts.append(generate_start(n_init, rng, 'random'))
    starts.append(generate_start(n_init, rng, 'random'))
    starts.append(generate_start(n_init, rng, 'alt_pattern'))

    for si, h_s in enumerate(starts):
        if tleft() < 70:
            break
        allowed = min(180.0, tleft() * 0.28)
        max_it = int(min(200000, allowed * 3500))
        h_sa, c5_sa = fast_sa(h_s, target_init, max_it, 0.008, 0.00005,
                              rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

        if tleft() > 40:
            h_g, c5_g = greedy_hill_climb(h_sa, target_init, 4000, rng)
            if c5_g < best_score:
                best_score = c5_g
                best_h = h_g.copy()

    if tleft() > 30:
        h_g, c5_g = greedy_hill_climb(best_h, target_init, 5000, rng)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    n_candidates = []
    for nf in [0.7, 0.8, 0.9, 1.1, 1.2, 1.35, 1.5, 1.75, 2.0]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try != n_init and 200 <= n_try <= 3000:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 55:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(90.0, tleft() * 0.25)
        max_it = int(min(100000, allowed * 2500))
        h_sa, c5_sa = fast_sa(h_bin_try, target_try, max_it, 0.006, 0.00005,
                              rng, time_limit=allowed, start_time=start_time)

        if tleft() > 35:
            h_g, c5_g = greedy_hill_climb(h_sa, target_try, 3000, rng)
            if c5_g < c5_sa:
                h_sa = h_g
                c5_sa = c5_g

        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 50:
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 time_limit=tleft() - 35, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_bin = project_binary(best_h, target_use)

    if tleft() > 25:
        max_it = int(max(10000, tleft() * 2500))
        h_final_sa, c5_final = fast_sa(h_bin, target_use, max_it, 0.003, 0.00002,
                                       rng, time_limit=max(5.0, tleft() - 12), start_time=start_time)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_sa

    if tleft() > 12:
        h_g, c5_g = greedy_hill_climb(best_h, len(best_h) / 2.0, 5000, rng)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g

    n_final = len(best_h)
    h_final = project_binary(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
