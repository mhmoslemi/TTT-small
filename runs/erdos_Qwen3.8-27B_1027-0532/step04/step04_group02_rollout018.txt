Let me implement the FFT-based correlation:

```python
def corr_fft(h):
    n = len(h)
    g = 1.0 - h
    # Zero-pad to length at least 2n-1 for linear correlation
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    H = np.fft.rfft(h, fsize)
    G = np.fft.rfft(g, fsize)
    # Cross-correlation: IFFT(conj(H) * G)
    corr = np.fft.irfft(np.conj(H) * G, fsize)[:2*n-1]
    return corr
```

def corr_fft(h):
    n = len(h)
    g = 1.0 - h
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    H = np.fft.rfft(h, fsize)
    G = np.fft.rfft(g, fsize)
    r = np.fft.irfft(H * np.conj(G), fsize)
    full_corr = np.concatenate([r[fsize - (n - 1):fsize], r[0:n]])
    return full_corr * (2.0 / n)


def compute_c5(h):
    corr = corr_fft(h)
    return float(np.max(corr))


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, alpha, p):
    n = len(h)
    corr = corr_fft(h)
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_optimize(h_start, n, target, p_vals, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for p in p_vals:
        for alpha in alphas:
            if (time.time() - start_time) > time_limit:
                return h_cur, curr_score
            try:
                res = minimize(lambda x, a=alpha, pp=p: smooth_max_obj(x, a, pp), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,

                sc = compute_c5(h_new)
                if sc < curr_score:

                    h_cur = h_new

                                     [max(3, n_use // 25)], [1000, 3000, 5000],
                                     400, 1e-15, time_limit=min(12.0, tleft() * 0.7), start_time=start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-14:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
