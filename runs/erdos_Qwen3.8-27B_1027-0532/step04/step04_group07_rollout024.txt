def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def compute_ab_for_j(h, j, n):
    """Compute a[k] and b[k] such that for k != n-1: r(k) = a[k]*x + b[k] where x = h[j]"""
    x = h[j]
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = np.zeros(n_corr)

        if L == 0:
            continue
        elif L > 0:
            da = 0.0
            if j <= n - 1 - L:
                da += (1.0 - h[j + L])
            if L <= j:
                da -= h[j - L]
            a[k] = da
        else:
            m = -L
            da = 0.0
            if m <= j:
                da += (1.0 - h[j - m])
            if j + m < n:
                da -= h[j + m]
            a[k] = da
    
    b = corr - a * x
    # c0 for zero lag
    c0 = corr[n-1] - x * (1.0 - x)
    return a, b, c0

def coord_optimize(h, j, n, n_grid=200):
    a, b, c0 = compute_ab_for_j(h, j, n)
    grid = np.linspace(0.0, 1.0, n_grid)
    
    # Linear part (all lags except zero)
    idx_nonzero = np.concatenate([np.arange(0, n-1), np.arange(n, 2*n-1)])
    lin = a[idx_nonzero][None, :] * grid[:, None] + b[idx_nonzero][None, :]
    lin_max = np.max(lin, axis=1)
    
    # Quadratic part (zero lag)
    quad = grid - grid**2 + c0
    
    overall = np.maximum(lin_max, quad)
    idx = np.argmin(overall)
    
    # Refine
    lo = grid[max(0, idx - 1)]
    hi = grid[min(n_grid - 1, idx + 1)]
    fine = np.linspace(lo, hi, 500)
    lin = a[idx_nonzero][None, :] * fine[:, None] + b[idx_nonzero][None, :]
    lin_max = np.max(lin, axis=1)
    quad = fine - fine**2 + c0
    overall = np.maximum(lin_max, quad)
    idx2 = np.argmin(overall)
    
    return fine[idx2]

One more optimization: in the CD sweep, instead of recomputing np.correlate at each step (which is O(n²)), I can compute it once and then update it incrementally. When h[j] changes from old to new (delta = new - old):

```python
def update_corr(corr, h, j, old_val, new_val, n):
    """Update correlation after changing h[j] from old_val to new_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    
    for k in range(2*n-1):
        L = k - (n-1)
        if L == 0:
            corr[k] += delta * (1.0 - new_val - old_val)
        elif L > 0:
            dk = 0.0
            if j <= n - 1 - L:
                dk += (1.0 - h[j+L])
            if L <= j:
                dk -= h[j-L]
            corr[k] += delta * dk
        else:
            m = -L
            dk = 0.0
            if m <= j:
                dk += (1.0 - h[j-m])
            if j <= n - 1 - m:
                dk -= h[j+m]
            corr[k] += delta * dk
    
    return corr
```

            if L == 0:
                continue
            elif L > 0:
                dk = 0.0
                if j <= n - 1 - L:
                    dk += (1.0 - h[j+L])
                if L <= j:
                    dk -= h[j-L]
                a[k] = dk
            else:
                m = -L
                dk = 0.0
                if m <= j:
                    dk += (1.0 - h[j-m])
                if j <= n - 1 - m:
                    dk -= h[j+m]
                a[k] = dk
        
        b = corr - a * x_old
        c0 = corr[n-1] - x_old * (1.0 - x_old)
        
        # Grid search
        grid = np.linspace(0.0, 1.0, n_grid)
        idx_nz = np.concatenate([np.arange(0, n-1), np.arange(n, 2*n-1)])
        lin = a[idx_nz][None,:] * grid[:,None] + b[idx_nz][None,:]
        lin_max = np.max(lin, axis=1)
        quad = grid - grid**2 + c0
        overall = np.maximum(lin_max, quad)
        idx = np.argmin(overall)
        
        # Refine
        lo = grid[max(0, idx-1)]
        hi = grid[min(n_grid-1, idx+1)]
        fine = np.linspace(lo, hi, 500)
        lin = a[idx_nz][None,:] * fine[:,None] + b[idx_nz][None,:]
        lin_max = np.max(lin, axis=1)
        quad = fine - fine**2 + c0
        overall = np.maximum(lin_max, quad)
        idx2 = np.argmin(overall)
        x_new = fine[idx2]
        
        # Update
        if x_new != x_old:
            h[j] = x_new
            delta = x_new - x_old
            for k in range(n_corr):
                L = k - (n-1)
                if L == 0:
                    corr[k] += delta * (1.0 - x_new - x_old)
                elif L > 0:
                    dk = 0.0
                    if j <= n - 1 - L:
                        dk += (1.0 - h[j+L])
                    if L <= j:
                        dk -= h[j-L]
                    corr[k] += delta * dk
                else:
                    m = -L
                    dk = 0.0
                    if m <= j:
                        dk += (1.0 - h[j-m])
                    if j <= n - 1 - m:
                        dk -= h[j+m]
                    corr[k] += delta * dk
    
    return h, corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)

    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
