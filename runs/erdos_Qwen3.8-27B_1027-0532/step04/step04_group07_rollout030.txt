def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-15
            nf = free.sum()
            if nf == 0: break
            h[free] = np.minimum(h[free] + diff / nf, 1.0)
        else:
            free = h > 1e-15
            nf = free.sum()
            if nf == 0: break
            h[free] = np.maximum(h[free] + diff / nf, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return np.clip(h, 0.0, 1.0)

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_single_lag(h, lag):
    """Gradient of c_lag(h) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
    else:
        m = -lag
        if m == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]

def gradient_weighted(h, indices, weights):

    n = len(h)

    for i in range(len(indices)):

    shifted = np.clip(alpha * (vals - c_ref), -600.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha, topk):

    nc = len(corr)

    exp_v = np.exp(np.clip(alpha * (vals - c_ref), -600.0, 0.0))
    w = exp_v / np.sum(exp_v)
    return gradient_weighted(h, top, w)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nf = free.sum()
            if nf == 0: break
            h[free] = np.minimum(h[free] + diff / nf, 1.0)
        else:
            free = h > 1e-14
            nf = free.sum()
            if nf == 0: break
            h[free] = np.maximum(h[free] + diff / nf, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    return g

    n = len(h0)
    nc = 2 * n - 1

        topk = nc
    h = h0.copy()
    best_h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    stall = 0

            break
        
        corr = get_corr(h)
        # Select lags
        if topk < nc:
            top = np.argpartition(corr, -topk)[-topk:]

        m = len(top)

        b = np.zeros(m)
        for i in range(m):

            lag = idx - n_off

            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[idx]
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n + [(-1e10, 1e10)]

                h = h_trial

                found = True
                break

    return np.correlate(h, 1.0 - h, mode='full')


def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    return g


def grad_weighted(h, idxs, ws):
    n = len(h)
    g = np.zeros(n)
    n_off = n - 1
    for i in range(len(idxs)):
        w = ws[i]
        if w < 1e-15:
            continue
        lag = int(idxs[i]) - n_off
        if lag >= 0:
            if lag < n:
                g[lag:] += w * (1.0 - h[:n - lag])
                g[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                g[:n - m] += w * (1.0 - h[m:])
                g[m:] -= w * h[:n - m]
    return g


def slp_refine(h0, target, max_iter, time_limit=None, use_all_lags=True):
    n = len(h0)
    nc = 2 * n - 1
    h = h0.copy()
    best_h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    stall = 0
    n_off = n - 1

    for it in range(max_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        corr = get_corr(h)
        if use_all_lags:
            top = np.arange(nc)
        else:
            top = np.argpartition(corr, -nc)[-nc:]
        m = len(top)

        A = np.empty((m, n + 1))
        b = np.empty(m)
        for i in range(m):
            idx = top[i]
            lag = idx - n_off
            g = grad_lag(h, lag)
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[idx]

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n + [(-1e10, 1e10)]

        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_lp = project_feasible(res.x[:n], target)

        found = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            h_trial = project_feasible((1.0 - alpha) * h + alpha * h_lp, target)
            sc = compute_c5(h_trial)
            if sc < best - 1e-15:
                best = sc
                best_h = h_trial.copy()
                h = h_trial
                stall = 0
                found = True
                break
        if not found:
            h_trial = project_feasible(h_lp, target)
            sc = compute_c5(h_trial)
            if sc < best - 1e-15:
                best = sc
                best_h = h_trial.copy()
                h = h_trial
                stall = 0
            else:
                stall += 1
                if stall > 25:
                    break
    return best_h, best


def subgradient_opt(h0, target, max_iter, lr, topk, alpha, patience, momentum, seed, time_limit=None):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    stall = 0
    n_off = n - 1

    for it in range(max_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        corr = get_corr(h)
        nc = len(corr)
        if topk < nc:
            top = np.argpartition(corr, -topk)[-topk:]
        else:
            top = np.arange(nc)
        vals = corr[top]
        w = np.exp(alpha * (vals - np.max(vals)))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(top)) / len(top)
        else:
            w /= ws
        g = grad_weighted(h, top, w)
        v = momentum * v + (1.0 - momentum) * g
        h_new = project_feasible(h - lr * v, target)
        sc = compute_c5(h_new)
        if sc < best:
            best = sc
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.005, 5.0)
        else:
            stall += 1
            lr *= 0.98
        h = h_new
        if stall > patience:
            break
    return best_h, best


def smooth_obj(h, alpha, topk):
    corr = get_corr(h)
    nc = len(corr)
    if topk < nc:
        top = np.argpartition(corr, -topk)[-topk:]
        vals = corr[top]
    else:
        top = np.arange(nc)
        vals = corr
    c_ref = np.max(vals)
    shifted = np.clip(alpha * (vals - c_ref), -600.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_grad(h, alpha, topk):
    corr = get_corr(h)
    nc = len(corr)
    if topk < nc:
        top = np.argpartition(corr, -topk)[-topk:]
        vals = corr[top]
    else:
        top = np.arange(nc)
        vals = corr
    c_ref = np.max(vals)
    exp_v = np.exp(np.clip(alpha * (vals - c_ref), -600.0, 0.0))
    w = exp_v / np.sum(exp_v)
    return grad_weighted(h, top, w)


def _cons_eq(x):
    return np.sum(x) - _T_G


def _cons_eq_jac(x):
    return np.ones(_N_G)


def lbgfb_smooth(h0, target, alpha_list, maxiter):
    global _N_G, _T_G
    _N_G = len(h0)
    _T_G = target
    h = h0.copy()
    best = compute_c5(h)
    bounds = [(0.0, 1.0)] * len(h0)
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    for alpha, topk in alpha_list:
        try:
            res = minimize(smooth_obj, h, args=(alpha, topk),
                           jac=smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best:
                best = sc
                h = h_new
        except Exception:
            pass
    return h, best


def interp_to(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    x_new = np.linspace(0.0, 1.0, n_dst)
    return project_feasible(np.interp(x_new, x_old, h_src), target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _N_G, _T_G
    t_start = time.time()

    def remaining():
        return budget_s - 20.0 - (time.time() - t_start)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    best_h = h.copy()
    best = compute_c5(h)

    # Phase 1: Quick subgradient warm-up at original resolution
    if remaining() > 60:
        h_try = best_h + np.random.RandomState(seed).randn(n) * 0.003
        h_try = project_feasible(h_try, target)
        hl, sc = subgradient_opt(h_try, target, 1500, 0.06, 8, 30, 150, 0.90, seed)
        if sc < best:
            best = sc
            best_h = hl

    # Phase 2: SLP with all lags at original resolution
    if remaining() > 150:
        hl, sc = slp_refine(best_h, target, max_iter=5000, time_limit=remaining() - 100, use_all_lags=True)
        if sc < best:
            best = sc
            best_h = hl

    # Phase 3: SLSQP smooth refinement
    if remaining() > 150:
        alphas = [(200, 80), (1000, 40), (5000, 20)]
        hr, sc = lbgfb_smooth(best_h, target, alphas, 2000)
        if sc < best:
            best = sc
            best_h = hr

    # Phase 4: More SLP after SLSQP
    if remaining() > 100:
        hl, sc = slp_refine(best_h, target, max_iter=3000, time_limit=remaining() - 60, use_all_lags=True)
        if sc < best:
            best = sc
            best_h = hl

    # Phase 5: Try higher resolutions
    for n_new in [600, 800, 1000, 1200]:
        if remaining() < 120:
            break
        target_new = n_new / 2.0
        h_interp = interp_to(best_h, n_new, target_new)

        # SLP at new resolution
        if remaining() > 80:
            hl, sc = slp_refine(h_interp, target_new, max_iter=5000, time_limit=remaining() - 40, use_all_lags=True)
            if sc < best:
                best = sc
                best_h = hl

        # Quick SLSQP
        if remaining() > 60 and len(best_h) == n_new:
            hr, sc = lbgfb_smooth(best_h, target_new, [(500, 50), (2000, 25)], 1500)
            if sc < best:
                best = sc
                best_h = hr

        # SLP again
        if remaining() > 50:
            hl, sc = slp_refine(best_h, len(best_h) / 2.0, max_iter=3000, time_limit=remaining() - 20, use_all_lags=True)
            if sc < best:
                best = sc
                best_h = hl

    # Phase 6: Try a different starting point at medium resolution
    if remaining() > 150:
        n_alt = 800
        t_alt = n_alt / 2.0
        # Create a structured alternative: periodic blocks
        n_blocks = 7
        block_len = n_alt // n_blocks
        h_alt = np.zeros(n_alt)
        for b in range(n_blocks):
            start = b * block_len
            end = min(start + block_len, n_alt)
            fill = int((end - start) * 0.5)
            h_alt[start:start + fill] = 1.0
        h_alt = project_feasible(h_alt, t_alt)
        hl, sc = slp_refine(h_alt, t_alt, max_iter=5000, time_limit=remaining() - 50, use_all_lags=True)
        if sc < best:
            best = sc
            best_h = hl

    # Phase 7: Final SLP polish
    n_final = len(best_h)
    t_final = n_final / 2.0
    if remaining() > 30:
        hl, sc = slp_refine(best_h, t_final, max_iter=3000, time_limit=remaining() - 10, use_all_lags=True)
        if sc < best:
            best = sc
            best_h = hl

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
