def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)

    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    
    grad = np.zeros(n)

        w = weights[s]
        if w < 1e-10:
            continue
        lag = n - 1 - s

                grad[:jmax] += w * (1.0 - h[lag:lag+jmax])
            if lag > 0:
                grad[lag:] -= w * h[:n-lag]
        else:
            jmin = -lag
            grad[jmin:] += w * (1.0 - h[:n-jmin])
            grad[:n+lag] -= w * h[-lag:]
    return grad

        ratio = it / max(n_iters - 1, 1)

Let me implement incremental correlation updates:

When h changes at positions p1, p2, ..., pm (from old_val to new_val):
The change in corr[s] for each s:
corr[s] = Σ_i h[i] * (1-h[i+n-1-s])

For a single position p changing from 1 to 0:
- corr[n-1-k] changes by -(1-h[p+k]) for k where 0≤p+k<n [as the h[i] term]
- corr[n-1-k] changes by +h[p-k] for k where 0≤p-k<n [as the (1-h[i+k]) term, since h[p] was the h[i+k] part]

Wait: corr[n-1-k] = Σ_i h[i]*(1-h[i+k])
If h[p] changes from 1 to 0:
- As h[i] with i=p: term was 1*(1-h[p+k]), now 0. Change: -(1-h[p+k]) = h[p+k]-1
- As h[i+k] with i+k=p, i=p-k: term was h[p-k]*(1-1) = 0, now h[p-k]*(1-0) = h[p-k]. Change: +h[p-k]

So Δcorr[n-1-k] = (h[p+k]-1)·[p+k<n] + h[p-k]·[p-k≥0]
= -（1-h[p+k])·[p+k<n] + h[p-k]·[p-k≥0]

For h[p] changing from 0 to 1:
- As h[i] with i=p: term was 0, now 1*(1-h[p+k]). Change: +(1-h[p+k])
- As h[i+k] with i=p-k: term was h[p-k]*1, now h[p-k]*0. Change: -h[p-k]

So Δcorr[n-1-k] = (1-h[p+k])·[p+k<n] - h[p-k]·[p-k≥0]

These are O(n) updates per position changed. For a swap (2 positions), it's O(n) total.

    shifted = np.clip(shifted, -100.0, 0.0)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = exp_s / total
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-10:
            continue
        lag = n - 1 - s
        if lag >= 0:
            jmax = n - lag
            if jmax > 0:
                grad[:jmax] += w * (1.0 - h[lag:lag + jmax])
            if lag > 0:
                grad[lag:] -= w * h[:n - lag]
        else:
            jmin = -lag
            grad[jmin:] += w * (1.0 - h[:n - jmin])
            jm2 = n + lag
            if jm2 > 0:
                grad[:jm2] -= w * h[-lag:]
    return grad


def continuous_optimize(h_init, target, t_start, time_limit, alpha_start=5.0, alpha_end=300.0):
    n = len(h_init)
    h = project_box_sum(h_init.copy(), target)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_total = 400
    for it in range(n_total):
        if time.time() - t_start > time_limit:
            break
        ratio = it / max(n_total - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        momentum = 0.85 * (1.0 - 0.5 * ratio)
        grad = get_smooth_grad(h, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        eta = n / (2.0 * (1.0 + alpha * grad_norm))
        eta = min(eta, 2.0)
        h_old = h.copy()
        h = y - eta * grad
        y = h + momentum * (h - h_old)
        h = project_box_sum(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5


def targeted_local_search(h_init, target, max_iters, rng, t_start, time_limit, t_init=0.008, t_final=0.00003):
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_imp = 0
    max_no_imp = max(5000, max_iters // 8)
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if no_imp > max_no_imp:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_perturb = rng.randint(5, max(6, n // 25))
            for _ in range(n_perturb):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = int(ones[rng.randint(len(ones))])
                    pj = int(zeros[rng.randint(len(zeros))])
                    h[pi] = 0
                    h[pj] = 1
            h = project_binary(h, target_int).astype(np.float64)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            continue

        ratio = it / max(max_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        worst_s = int(np.argmax(corr))
        worst_lag = worst_s - (n - 1)
        h_saved = h.copy()
        move_type = rng.randint(10)

        if move_type < 5 and len(ones) > 0 and len(zeros) > 0:
            if worst_lag >= 0:
                valid_mask = ones < (n - worst_lag)
                valid_ones = ones[valid_mask]
                if len(valid_ones) > 0:
                    cand_ones = valid_ones[h[valid_ones + worst_lag] == 0]
                else:
                    cand_ones = ones
            else:
                valid_mask = ones >= (-worst_lag)
                valid_ones = ones[valid_mask]
                if len(valid_ones) > 0:
                    cand_ones = valid_ones[h[valid_ones + worst_lag] == 0]
                else:
                    cand_ones = ones
            if len(cand_ones) == 0:
                cand_ones = ones
            ii = int(cand_ones[rng.randint(len(cand_ones))])
            jj_arr = zeros.astype(np.int64)
            jl = jj_arr + worst_lag
            jr = jj_arr - worst_lag
            scores = np.zeros(len(jj_arr))
            valid_l = (jl >= 0) & (jl < n)
            if np.any(valid_l):
                scores[valid_l] += 1.0 - h[jl[valid_l]]
            valid_r = (jr >= 0) & (jr < n)
            if np.any(valid_r):
                scores[valid_r] -= h[jr[valid_r]]
            scores[jj_arr == ii] = 999.0
            n_best = max(1, len(scores) // 5)
            sorted_idx = np.argsort(scores)[:n_best]
            jj = int(jj_arr[sorted_idx[rng.randint(n_best)]])
            h[ii] = 0
            h[jj] = 1
        elif move_type < 7 and len(ones) > 0 and len(zeros) > 0:
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0
            h[jj] = 1
        elif move_type < 9 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])
            while i2 == i1:
                i2 = int(ones[rng.randint(len(ones))])
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            while j2 == j1:
                j2 = int(zeros[rng.randint(len(zeros))])
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
        else:
            if len(ones) >= 3 and len(zeros) >= 3:
                idx1 = rng.choice(len(ones), size=3, replace=False)
                idx0 = rng.choice(len(zeros), size=3, replace=False)
                for idx in idx1:
                    h[int(ones[idx])] = 0
                for idx in idx0:
                    h[int(zeros[idx])] = 1
            elif len(ones) > 0 and len(zeros) > 0:
                ii = int(ones[rng.randint(len(ones))])
                jj = int(zeros[rng.randint(len(zeros))])
                h[ii] = 0
                h[jj] = 1

        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        delta = new_c5 - cur_c5
        if delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-delta / temp)):
            corr = new_corr
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h = h_saved
            no_imp += 1
    return best_h, best_c5


def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while True:
        is_prime = True
        if p < 2:
            is_prime = False
        else:
            for f in range(2, int(np.sqrt(p)) + 1):
                if p % f == 0:
                    is_prime = False
                    break
        if is_prime:
            break
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones_idx = np.where(h == 1)[0]
        order = rng.permutation(len(ones_idx))[:current - target]
        h[ones_idx[order]] = 0.0
    return h


def multiply_by_h_prime(h, n):
    h_flip = 1.0 - h.copy()
    h_flip = project_feasible(h_flip, n / 2.0)
    return h_flip


def recombine(h1, h2, rng):
    n = len(h1)
    h_mix = np.zeros(n)
    for i in range(n):
        if rng.random() < 0.5:
            h_mix[i] = h1[i]
        else:
            h_mix[i] = h2[i]
    return h_mix


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    deadline = start + budget_s - 25.0
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h_init = np.array(initial_h_values, dtype=float)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def remaining():
        return deadline - time.time()

    # Phase 1: Continuous optimization (up to 250s)
    if remaining() > 150:
        t_lim = min(250.0, remaining() * 0.7)
        h_c, c5_c = continuous_optimize(h_init, target, start, t_lim, 5.0, 200.0)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
        h_c2, c5_c2 = continuous_optimize(h_c, target, start, t_lim, 50.0, 500.0)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2

    # Phase 2: Main targeted binary local search (up to 500s)
    if remaining() > 200:
        t_lim = min(500.0, remaining() * 0.7)
        h_b, c5_b = targeted_local_search(best_h, target, 1000000, rng, start, t_lim, 0.007, 0.00002)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 3: Continue from best with different params
    if remaining() > 150:
        t_lim = min(300.0, remaining() * 0.7)
        h_b, c5_b = targeted_local_search(best_h, target, 800000, rng, start, t_lim, 0.004, 0.00001)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 4: Try QR starting point
    if remaining() > 120:
        h_qr = make_qr_sequence(n, rng)
        t_lim = min(200.0, remaining() * 0.6)
        h_b, c5_b = targeted_local_search(h_qr, target, 500000, rng, start, t_lim, 0.01, 0.00005)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 5: Recombination and further search
    if remaining() > 100:
        h_qr2 = make_qr_sequence(n, rng)
        h_mix = recombine(best_h, h_qr2, rng)
        h_mix_bin = project_binary(h_mix, target)
        t_lim = min(150.0, remaining() * 0.6)
        h_b, c5_b = targeted_local_search(h_mix_bin, target, 400000, rng, start, t_lim, 0.005, 0.00002)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 6: Fine-tuning with very low temperature
    if remaining() > 40:
        t_lim = min(remaining() * 0.9, 100.0)
        h_b, c5_b = targeted_local_search(best_h, target, 500000, rng, start, t_lim, 0.002, 0.000005)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 7: Try complement (1-h) which should give same C5 but different landscape
    if remaining() > 60:
        h_comp = 1.0 - h_init
        h_comp_bin = project_binary(h_comp, target)
        t_lim = min(80.0, remaining() * 0.7)
        h_b, c5_b = targeted_local_search(h_comp_bin, target, 300000, rng, start, t_lim, 0.008, 0.00003)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Final: ensure feasibility and evaluate
    h_final = project_feasible(best_h, target)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5 + 1e-12:
        h_final = best_h
        final_c5 = best_c5
    return h_final, final_c5, n
