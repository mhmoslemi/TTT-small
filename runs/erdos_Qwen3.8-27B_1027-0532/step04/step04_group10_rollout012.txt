```python
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    current_max = np.max(corr) * dx
    
    for pass_num in range(max_passes):
        best_swap = None
        best_new_max = current_max
        improved = False
        
        for a_idx in range(len(ones)):
            a = ones[a_idx]
            for b_idx in range(len(zeros)):
                b = zeros[b_idx]
                if a == b:
                    continue
                # Compute change in corr for swap (a: 1->0, b: 0->1)
                # For each lag k:
                # Δc[k] from a: -(1-h[a-k])*[valid] + h[a+k]*[valid]  but h[a]=1 so 1-h[a-k] and h[a+k]
                # Actually since h is binary:
                # a changes 1->0:
                #   As h[n+lag]: n+lag=a => n=a-lag. Contribution was h[a]*(1-h[a-lag]) = (1-h[a-lag]). Now 0. Change: -(1-h[a-lag])
                #   As (1-h[n]): n=a. Contribution was h[a+lag]*(1-1) = 0. Now h[a+lag]*1. Change: +h[a+lag]
                # b changes 0->1:
                #   As h[n+lag]: n+lag=b => n=b-lag. Contribution was h[b]*(1-h[b-lag]) = 0. Now (1-h[b-lag]). Change: +(1-h[b-lag])
                #   As (1-h[n]): n=b. Contribution was h[b+lag]*(1-0) = h[b+lag]. Now h[b+lag]*0. Change: -h[b+lag]
                
                # For binary h:
                # Δc[lag] = -(1-h[a-lag])*[0<=a-lag<n] + h[a+lag]*[0<=a+lag<n]
                #            + (1-h[b-lag])*[0<=b-lag<n] - h[b+lag]*[0<=b+lag<n]
                
                # Since h is binary, (1-h[x]) = 1 - h[x]:
                # = -[0<=a-lag<n] + h[a-lag]*[0<=a-lag<n] + h[a+lag]*[0<=a+lag<n]
                #   + [0<=b-lag<n] - h[b-lag]*[0<=b-lag<n] - h[b+lag]*[0<=b+lag<n]
                
                # This is O(n) to compute for all lags. Let me vectorize.
                
                # Actually, let me just compute the new max directly for the affected lags.
                # The lags affected are those where a or b is in the valid range.
                # For position a: lags from a-(n-1) to a (as n+lag=a => lag=a-n, and n=a => lag can be anything valid)
                # Actually all lags from -(n-1) to (n-1) could be affected.
                
                # Let me just compute Δc for all lags:
                # Δc[lag] = -(1-h[a-lag])*[0<=a-lag<n] + h[a+lag]*[0<=a+lag<n]
                #            + (1-h[b-lag])*[0<=b-lag<n] - h[b+lag]*[0<=b+lag<n]
                
                # Vectorized:
                lags = np.arange(-(n-1), n)  # all possible lags
                # For position a:
                valid_a_as_shifted = (a - lags >= 0) & (a - lags < n)  # a-lag is valid index
                valid_a_as_base = (a + lags >= 0) & (a + lags < n)  # a+lag is valid index
                # For position b:
                valid_b_as_shifted = (b - lags >= 0) & (b - lags < n)
                valid_b_as_base = (b + lags >= 0) & (b + lags < n)
                
                # h[a-lag] where valid:
                a_shifted_idx = np.where(valid_a_as_shifted, a - lags, 0)
                a_base_idx = np.where(valid_a_as_base, a + lags, 0)
                b_shifted_idx = np.where(valid_b_as_shifted, b - lags, 0)
                b_base_idx = np.where(valid_b_as_base, b + lags, 0)
                
                dc = np.zeros(2*n-1)
                dc[valid_a_as_shifted] -= (1 - h[a_shifted_idx[valid_a_as_shifted]])
                dc[valid_a_as_base] += h[a_base_idx[valid_a_as_base]]
                dc[valid_b_as_shifted] += (1 - h[b_shifted_idx[valid_b_as_shifted]])
                dc[valid_b_as_base] -= h[b_base_idx[valid_b_as_base]]
                
                new_corr = corr + dc
                new_max_val = np.max(new_corr) * dx
                
                if new_max_val < best_new_max - 1e-12:
                    best_new_max = new_max_val
                    best_swap = (a, b, dc)
                    improved = True
        
        if not improved:
            break
        
        # Apply the best swap
        a, b, dc = best_swap
        h[a] = 0
        h[b] = 1
        corr = corr + dc
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        current_max = best_new_max
    
    return h, current_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_params = None


def _objective(x):
    alpha, topk, target, penalty = _g_params
    return _lse_objective(x, alpha, topk, target, penalty)


def _gradient(x):
    alpha, topk, target, penalty = _g_params
    return _lse_gradient(x, alpha, topk, target, penalty)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        if diff > 0:
            step = min(step, 1.0 - h[idx].min() - 1e-15)
        else:
            step = max(step, -h[idx].min() - 1e-15)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def _lse_objective(h, alpha, topk, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    f = c_max + np.log(total) / alpha
    s = np.sum(h) - target
    f += penalty * s * s
    return f


def _lse_gradient(h, alpha, topk, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    s = np.sum(h) - target
    grad += 2.0 * penalty * s
    return grad


def run_lbfgs(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol):
    global _g_params
    _g_params = (alpha, topk, target, penalty)
    try:
        res = minimize(_objective, h_start, jac=_gradient, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def round_to_binary(h, target):
    n = len(h)
    k = int(target)
    top_indices = np.argsort(h)[-k:]
    h_bin = np.zeros(n, dtype=float)
    h_bin[top_indices] = 1.0
    return h_bin


def binary_local_search(h_bin, max_time):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy().astype(float)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    current_score = float(np.max(corr)) * dx
    lags = np.arange(-(n - 1), n)
    start_t = time.time()

    for _ in range(2000):
        if time.time() - start_t > max_time:
            break
        best_improvement = 0.0
        best_a = -1
        best_b = -1
        best_dc = None

        for a in ones:
            for b in zeros:
                valid_a1 = (a - lags >= 0) & (a - lags < n)
                valid_a2 = (a + lags >= 0) & (a + lags < n)
                valid_b1 = (b - lags >= 0) & (b - lags < n)
                valid_b2 = (b + lags >= 0) & (b + lags < n)
                dc = np.zeros(2 * n - 1)
                if np.any(valid_a1):
                    dc[valid_a1] -= (1.0 - h[a - lags[valid_a1]])
                if np.any(valid_a2):
                    dc[valid_a2] += h[a + lags[valid_a2]]
                if np.any(valid_b1):
                    dc[valid_b1] += (1.0 - h[b - lags[valid_b1]])
                if np.any(valid_b2):
                    dc[valid_b2] -= h[b + lags[valid_b2]]
                new_corr = corr + dc
                new_max = float(np.max(new_corr)) * dx
                improvement = current_score - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_a = a
                    best_b = b
                    best_dc = dc

        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        corr = corr + best_dc
        ones.remove(best_a)
        ones.append(best_b)
        zeros.remove(best_b)
        zeros.append(best_a)
        current_score = current_score - best_improvement

    return h, current_score


def binary_local_search_fast(h_bin, max_time):

    lags = np.arange(-(n - 1), n)

        for a in ones:

            dc_a = np.zeros(2 * n - 1)
```
