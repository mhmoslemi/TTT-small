def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_corr_fft(h):
    n = len(h)
    dx = 2.0 / n
    ones = np.ones(n)
    n_fft = 2 * n - 1
    h_pad = np.zeros(n_fft)
    ones_pad = np.zeros(n_fft)
    h_pad[:n] = h
    ones_pad[:n] = ones - h
    h_f = np.fft.fft(h_pad)
    o_f = np.fft.fft(ones_pad)
    corr = np.fft.ifft(np.conj(h_f) * o_f).real
    # np.correlate gives c[k] = sum_i a[i+k-n_b+1]*b[i]
    # FFT gives circular correlation which for zero-padded signals matches linear correlation
    # But we need to match the ordering of np.correlate mode='full'
    # np.correlate(h, ones-h, mode='full')[k] = sum_i h[i] * (1-h[i+k-n+1])
    # The FFT result should be the same after proper indexing
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)

def delta_corr_single(h, i1, i0):
    """Compute the change in correlation when flipping h[i1]: 1->0 and h[i0]: 0->1.
    Returns the delta vector (change in corr for each lag)."""
    n = len(h)
    dx = 2.0 / n
    # corr[k] = sum_i h[i+k] * (1-h[i]) for k = -(n-1) to (n-1)
    # index in full array: k + (n-1)
    
    # Effect of h[i1] going 1->0:
    # As the "a" part: h[i1+k] where i1+k is the index into h
    #   For lag k (0-indexed in full array, corresponding to shift k-(n-1)):
    #   h[i + shift] * (1-h[i]) where shift = k - (n-1)
    #   If i + shift = i1, then i = i1 - shift = i1 - k + (n-1)
    #   The term was 1 * (1-h[i]) = (1-h[i]), now becomes 0
    #   So delta at lag k = -(1-h[i1-k+(n-1)]) if 0 <= i1-k+(n-1) < n
    #
    # As the "b" part: (1-h[i1]) where i = i1
    #   h[i+k] * (1-h[i]) where i = i1
    #   (1-h[i1]) was 0, now becomes 1
    #   delta at lag k = h[i1+k] if 0 <= i1+k < n
    
    # Effect of h[i0] going 0->1:
    # As the "a" part:
    #   delta at lag k = +(1-h[i0-k+(n-1)]) if 0 <= i0-k+(n-1) < n
    # As the "b" part:
    #   (1-h[i0]) was 1, now becomes 0
    #   delta at lag k = -h[i0+k] if 0 <= i0+k < n
    
    # Let's compute this vectorized.
    # Full array has indices 0 to 2n-2, corresponding to shifts -(n-1) to (n-1)
    delta = np.zeros(2*n-1)
    
    # h[i1] 1->0
    # As "a": shift s = k-(n-1), i = i1-s, valid if 0<=i<n, i.e., s <= i1 and s >= i1-n+1
    #   s ranges from max(-(n-1), i1-n+1) to min(n-1, i1)
    #   delta[k] -= (1-h[i]) = (1-h[i1-s])
    s_min_a = max(-(n-1), i1 - (n-1))  # = max(-(n-1), i1-n+1)
    s_max_a = min(n-1, i1)
    if s_min_a <= s_max_a:
        s_vals = np.arange(s_min_a, s_max_a + 1)
        i_vals = i1 - s_vals  # index into h for the (1-h[i]) part
        delta[s_vals + (n-1)] -= (1.0 - h[i_vals])
    
    # As "b": shift s, h[i1+s] valid if 0<=i1+s<n
    s_min_b = max(-(n-1), -i1)
    s_max_b = min(n-1, n-1-i1)
    if s_min_b <= s_max_b:
        s_vals = np.arange(s_min_b, s_max_b + 1)
        i_vals = i1 + s_vals  # index into h for h[i1+s]
        delta[s_vals + (n-1)] += h[i_vals]
    
    # h[i0] 0->1
    # As "a":
    s_min_a0 = max(-(n-1), i0 - (n-1))
    s_max_a0 = min(n-1, i0)
    if s_min_a0 <= s_max_a0:
        s_vals = np.arange(s_min_a0, s_max_a0 + 1)
        i_vals = i0 - s_vals
        delta[s_vals + (n-1)] += (1.0 - h[i_vals])
    
    # As "b":
    s_min_b0 = max(-(n-1), -i0)
    s_max_b0 = min(n-1, n-1-i0)
    if s_min_b0 <= s_max_b0:
        s_vals = np.arange(s_min_b0, s_max_b0 + 1)
        i_vals = i0 + s_vals
        delta[s_vals + (n-1)] -= h[i_vals]
    
    return delta * dx

def delta_corr_double(h, i1a, i1b, i0a, i0b):
    """Compute change in correlation for 2-swap: h[i1a],h[i1b] 1->0, h[i0a],h[i0b] 0->1."""
    d1 = delta_corr_single(h, i1a, i0a)
    d2 = delta_corr_single(h, i1b, i0b)
    # But we need to be careful: if i0a == i1b or similar overlaps
    # For simplicity, compute full delta
    n = len(h)
    dx = 2.0 / n
    delta = np.zeros(2*n-1)
    
    for i1, i0 in [(i1a, i0a), (i1b, i0b)]:
        s_min_a = max(-(n-1), i1 - (n-1))
        s_max_a = min(n-1, i1)
        if s_min_a <= s_max_a:
            s_vals = np.arange(s_min_a, s_max_a + 1)
            i_vals = i1 - s_vals
            delta[s_vals + (n-1)] -= (1.0 - h[i_vals])
        
        s_min_b = max(-(n-1), -i1)
        s_max_b = min(n-1, n-1-i1)
        if s_min_b <= s_max_b:
            s_vals = np.arange(s_min_b, s_max_b + 1)
            i_vals = i1 + s_vals
            delta[s_vals + (n-1)] += h[i_vals]
        
        s_min_a0 = max(-(n-1), i0 - (n-1))
        s_max_a0 = min(n-1, i0)
        if s_min_a0 <= s_max_a0:
            s_vals = np.arange(s_min_a0, s_max_a0 + 1)
            i_vals = i0 - s_vals
            delta[s_vals + (n-1)] += (1.0 - h[i_vals])
        
        s_min_b0 = max(-(n-1), -i0)

        if s_min_b0 <= s_max_b0:
            s_vals = np.arange(s_min_b0, s_max_b0 + 1)

        # Choose move type

def softmax_obj(x, alpha, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

def grad_softmax_obj(x, alpha, p):
    """Gradient of the softmax top-p objective."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    sorted_corr = np.sort(corr)
    top_mask = np.zeros(len(corr), dtype=bool)
    top_idx = np.argsort(corr)[-p:]
    top_mask[top_idx] = True
    max_top = np.max(corr[top_mask])
    shifted = np.clip(alpha * (corr[top_mask] - max_top), -500.0, 0.0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    # grad_corr = sum of weights * grad_corr_k
    # corr[k] = sum_i x[i] * (1-x[i+k]) * dx (approximately)
    # d(corr[k])/dx[j] = (1-x[j+k])*dx if i=j (x part) + x[j-k]*(-1)*dx if i+k=j (1-x part)
    # This is complex to compute efficiently. Let's just use finite differences in SLSQP.
    return None  # Let SLSQP handle gradients numerically

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, p_vals, alphas, maxiter, ftol,
                 time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(softmax_obj, h_cur, args=(alpha, p),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:

            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)

def softmax_obj(x, alpha, p):
    n = len(x)
    dx = 2.0 / n

    top = sorted_corr[-p:]

    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

            try:
                res = minimize(softmax_obj, h_cur, args=(alpha, p),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 10.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_score:
        best_score = c5_bin
        best_h = h_bin.copy()

    starts = [h_bin]
    for _ in range(2):
        h_r = np.zeros(n_init)
        perm = rng.permutation(n_init)
        h_r[perm[:n_init // 2]] = 1.0
        starts.append(h_r)
    tm = np.zeros(n_init)
    for i in range(n_init):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    tm = project_binary(tm, target_init)
    starts.append(tm)

    if n_init >= 8:
        h_block = np.zeros(n_init)
        bs = max(2, n_init // 100)
        for i in range(0, n_init, bs):
            if (i // bs) % 2 == 0:
                h_block[i:min(i + bs, n_init)] = 1.0
        h_block = project_binary(h_block, target_init)
        starts.append(h_block)

    total_sa_time = min(budget_s * 0.45, 580.0)
    sa_budget_each = total_sa_time / len(starts)

    for si, h_s in enumerate(starts):
        if tleft() < 100:
            break
        allowed = min(sa_budget_each, tleft() * 0.3)
        max_it = int(allowed * 12000)
        h_sa, c5_sa = fast_sa(h_s, target_init, max_it, 0.018, 0.00008,
                              rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_candidates = []
    for nf in [0.75, 0.9, 1.1, 1.3, 1.5, 2.0]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try >= 200:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 70:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(70.0, tleft() * 0.2)
        max_it = int(allowed * 8000)
        h_sa, c5_sa = fast_sa(h_bin_try, target_try, max_it, 0.012, 0.00005,
                              rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        p_vals = [max(3, n_use // 40), max(5, n_use // 25), max(10, n_use // 15)]
        alphas = [100, 300, 800, 2000]
        allowed_slq = min(150.0, tleft() - 50)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, p_vals, alphas,
                                  400, 1e-14, time_limit=allowed_slq,
                                  start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(10000, tleft() * 8000))
        allowed = max(5.0, tleft() - 15)
        h_sa_f, c5_f = fast_sa(h_bin_final, target_use, max_it, 0.006, 0.00003,
                               rng, time_limit=allowed, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        p_fine = [max(3, n_use // 30)]
        alphas_fine = [500, 1500, 4000]
        allowed_final = min(25.0, tleft() - 12)
        h_ref_f, sc_f = slsqp_refine(best_h, n_use, target_use, p_fine, alphas_fine,
                                      200, 1e-15, time_limit=allowed_final,
                                      start_time=start_time)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_ref_f

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
