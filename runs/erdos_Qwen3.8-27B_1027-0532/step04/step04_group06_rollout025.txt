                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stall = 0
            else:
                h[best_a] = 1.0
                h[best_b] = 0.0
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - target
    
    def cons_j(h_in):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    alphas = [30, 80, 200, 500, 1200, 3000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    
    h_bin = project_binary(h_init, target_init)
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: Long targeted SA from initial
    if tleft() > 300:
        allowed = min(350.0, tleft() * 0.5)
        h_sa, c5_sa = fast_sa_binary(h_bin, target_init, 300000, 0.015, 0.00003,
                                     rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 2: Steepest descent local search
    if tleft() > 100:
        allowed = min(150.0, tleft() * 0.3)
        h_ls, c5_ls = local_search_steepest(best_h, target_init, 500, rng,
                                            time_limit=allowed, start_time=start_time)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls.copy()
    
    # Phase 3: ILS - iterate perturbation + SA
    ils_rounds = min(6, int(max(2, tleft() / 120)))
    for il in range(ils_rounds):
        if tleft() < 80:
            break
        n_pert = rng.randint(5, 15)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_pert):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_pert[a] = 0.0
            h_pert[b] = 1.0
        allowed = min(100.0, tleft() * 0.3)
        h_sa, c5_sa = fast_sa_binary(h_pert, target_init, 100000, 0.01, 0.00005,
                                     rng, time_limit=allowed, start_time=start_time)
        if tleft() > 30:
            h_ls, c5_ls = local_search_steepest(h_sa, target_init, 200, rng,
                                                time_limit=min(40.0, tleft() * 0.2),
                                                start_time=start_time)
            if c5_ls < c5_sa:
                h_sa = h_ls
                c5_sa = c5_ls
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 4: Random restarts
    n_random = min(4, int(max(1, tleft() / 100)))
    for rs in range(n_random):
        if tleft() < 60:
            break
        h_rand = np.zeros(n_init)
        perm = rng.permutation(n_init)
        h_rand[perm[:n_init // 2]] = 1.0
        allowed = min(80.0, tleft() * 0.3)
        h_sa, c5_sa = fast_sa_binary(h_rand, target_init, 80000, 0.02, 0.0001,
                                     rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 5: Higher resolution attempts
    n_res_list = [int(n_init * 1.5), int(n_init * 2)]
    n_res_list = [x for x in n_res_list if x >= 200 and x <= 2000]
    
    for n_try in n_res_list:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(100.0, tleft() * 0.3)
        h_sa, c5_sa = fast_sa_binary(h_bin_try, target_try, 60000, 0.01, 0.00005,
                                     rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 6: SLSQP continuous refinement
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        allowed = min(100.0, tleft() * 0.4)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 7: Binary polish of continuous solution
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        allowed = min(50.0, tleft() * 0.5)
        h_sa_f, c5_f = fast_sa_binary(h_bin_final, target_use, 50000, 0.005, 0.00002,
                                      rng, time_limit=allowed, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
    
    # Final check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score
    
    return h_final, final_score, n_final
