def delta_O(s, a, b, h, n):
    """Compute delta in O(s) from swap (a:0->1, b:1->0) for shift s >= 0."""
    if s == 0:
        # O(0) = sum h[i](1-h[i]) = 0 for binary h, always 0
        return 0
    # Affected indices i where the term h[i](1-h[i+s]) changes
    affected = set()
    affected.add(a)  # h[a] changes
    affected.add(b)  # h[b] changes
    if a - s >= 0:
        affected.add(a - s)  # h[i+s] = h[a] changes
    if b - s >= 0:
        affected.add(b - s)  # h[i+s] = h[b] changes
    
    delta = 0
    for i in affected:
        if i < 0 or i > n - 1 - s:
            continue
        # Old value of term
        old = h[i] * (1 - h[i + s])
        # New value of term
        new_i = 1 - h[i] if i in (a, b) else h[i]
        # Wait, h[i] changes if i==a (0->1) or i==b (1->0)
        if i == a:
            new_h_i = 1
        elif i == b:
            new_h_i = 0
        else:
            new_h_i = h[i]
        js = i + s
        if js == a:
            new_h_js = 1
        elif js == b:
            new_h_js = 0
        else:
            new_h_js = h[js]
        new = new_h_i * (1 - new_h_js)
        delta += new - old
    return delta

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    """Project to binary with exactly target ones."""
    h_bin = np.zeros(len(h))
    # Take the top target values
    idx = np.argsort(-h)[:int(target)]
    h_bin[idx] = 1.0
    return h_bin

def binary_ls_first_improve(h, n, target, deadline, max_rounds=200):
    """First-improvement binary local search via 1-0 swaps."""
    h = h.copy()
    corr = compute_corr(h)
    best_raw = float(np.max(corr))
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]

            for o in ones:
                if time.time() > deadline:
                    return h
                h[z] = 1.0
                h[o] = 0.0
                c = float(np.max(compute_corr(h)))
                h[z] = 0.0
                h[o] = 1.0

                    h[z] = 1.0
                    h[o] = 0.0
                    best_raw = c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h

def slsqp_softmax(h, n, target, alpha, maxiter=300, deadline=None):
    """SLSQP optimization with soft-max objective."""
    dx = 2.0 / n
    
    def objective(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        soft = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        return soft

        return np.sum(x) - target
    
    bounds = [(0.0, 1.0)] * n

    """Momentum-based descent with soft-max gradient."""

    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    idx = np.argsort(-h)[:int(target)]
    h_bin = np.zeros(len(h))
    h_bin[idx] = 1.0
    return h_bin

def to_binary_greedy(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(-h)
    for i in range(int(target)):
        h_bin[order[i]] = 1.0
    return h_bin

def binary_ls_first_improve(h, n, target, deadline, max_rounds=500):
    h = h.copy()
    best_raw = float(np.max(compute_corr(h)))
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for z in zeros:
            for o in ones:
                if time.time() > deadline:
                    return h, best_raw
                h[z] = 1.0
                h[o] = 0.0
                c = float(np.max(compute_corr(h)))
                h[z] = 0.0
                h[o] = 1.0
                if c < best_raw - 1e-12:
                    h[z] = 1.0
                    h[o] = 0.0
                    best_raw = c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_raw

def binary_ls_best_improve(h, n, target, deadline, max_rounds=30):
    h = h.copy()
    best_raw = float(np.max(compute_corr(h)))
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_delta = 0.0
        best_pair = None
        for z in zeros:
            for o in ones:
                if time.time() > deadline:
                    return h, best_raw
                h[z] = 1.0
                h[o] = 0.0
                c = float(np.max(compute_corr(h)))
                h[z] = 0.0
                h[o] = 1.0
                if c < best_raw + best_delta - 1e-12:
                    best_delta = c - best_raw
                    best_pair = (z, o)
        if best_pair is not None:
            z, o = best_pair
            h[z] = 1.0
            h[o] = 0.0
            best_raw += best_delta
        else:
            break
    return h, best_raw

def binary_ls_anneal(h, n, target, deadline, rng, t0=0.008, cooling=0.995):
    h = h.copy()
    best_raw = float(np.max(compute_corr(h)))
    best_h = h.copy()
    cur_raw = best_raw
    t = t0
    while time.time() < deadline:
        if t < 1e-6:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        z = zeros[rng.randint(len(zeros))]
        o = ones[rng.randint(len(ones))]
        h[z] = 1.0
        h[o] = 0.0
        new_raw = float(np.max(compute_corr(h)))
        delta = new_raw - cur_raw
        if delta < 0 or rng.random() < np.exp(-delta / t):
            cur_raw = new_raw
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_h = h.copy()
        else:
            h[z] = 0.0
            h[o] = 1.0
        t *= cooling
    return best_h, best_raw

def slsqp_optimize(h0, n, target, alpha, maxiter=400, ftol=1e-14):
    def objective(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def constraint_eq(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constraint_eq}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h0

def slsqp_optimize_direct(h0, n, target, maxiter=400, ftol=1e-14):
    def objective(x):
        return compute_c5(x)

    def constraint_eq(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constraint_eq}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h0

def momentum_descent(h0, n, target, alpha_list, lr_init, momentum, deadline):
    h_cur = h0.copy()
    v = np.zeros(n)
    best_raw = float(np.max(compute_corr(h_cur)))
    best_h = h_cur.copy()
    lr = lr_init
    for alpha, count in alpha_list:
        for _ in range(count):
            if time.time() > deadline:
                return best_h
            h_look = project_feasible(h_cur + momentum * v, target)
            corr = compute_corr(h_look)
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
            wt = np.exp(shifted)
            wsum = wt.sum()
            if wsum < 1e-300:
                g = np.zeros(n)
            else:
                wt = wt / wsum
                g = np.zeros(n)
                n_full = len(corr)
                for si in range(n_full):
                    w = wt[si]
                    if w < 1e-11:
                        continue
                    s = si - (n - 1)
                    if s >= 0:
                        g[s:] += w * (1.0 - h_look[:n - s])
                        g[:n - s] -= w * h_look[s:]
                    else:
                        g[:n + s] += w * (1.0 - h_look[-s:])
                        g[-s:] -= w * h_look[:n + s]
            v = momentum * v - lr * g
            h_new = project_feasible(h_cur + v, target)
            sc = float(np.max(compute_corr(h_new)))
            if sc < best_raw:
                best_raw = sc
                best_h = h_new.copy()
                h_cur = h_new
                lr = lr_init
            else:
                lr *= 0.5
                if lr < 1e-14:
                    lr = lr_init
                    v *= 0.1
    return best_h

def block_descent(h0, n, target, block_size, deadline, rng, rounds=3):
    h_cur = h0.copy()
    cur_raw = float(np.max(compute_corr(h_cur)))
    n_blocks = n // block_size
    for rnd in range(rounds):
        if time.time() > deadline:
            break
        order = list(range(n_blocks))
        rng.shuffle(order)
        for bi in order:
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            prev_block = h_cur[s:e].copy()
            best_block = prev_block.copy()
            best_block_raw = cur_raw
            for trial in range(20):
                if time.time() > deadline:
                    break
                perturbed = prev_block + rng.uniform(-0.3, 0.3, block_size)
                perturbed = np.clip(perturbed, 0.0, 1.0)
                h_test = h_cur.copy()
                h_test[s:e] = perturbed
                h_test = project_feasible(h_test, target)
                sc = float(np.max(compute_corr(h_test)))
                if sc < best_block_raw - 1e-12:
                    best_block_raw = sc
                    best_block = perturbed.copy()
            if best_block_raw < cur_raw - 1e-12:
                h_cur[s:e] = best_block
                h_cur = project_feasible(h_cur, target)
                cur_raw = best_block_raw
    return h_cur, cur_raw
