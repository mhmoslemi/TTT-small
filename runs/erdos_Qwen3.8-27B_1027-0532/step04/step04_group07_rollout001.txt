def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h onto {sum(h) = target_sum, 0 <= h <= 1}
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-10
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, alpha, topk):
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[top_idx]
    else:
        vals = corr
    c_ref = np.max(vals)
    shifted = alpha * (vals - c_ref)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(x, alpha, topk):

    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]

def _grad_multi_lag(h, lag_indices, weights):

    grad = np.zeros(n)

        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def _grad_single_lag(h, lag_idx):

    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:

            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def projected_gradient_smooth(h, target_sum, alpha, topk, max_iter, lr, momentum, patience):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    stag = 0

def lp_step(h, target_sum, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = _grad_single_lag(h, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h - corr[idx]
    c = np.zeros(n + 1)

    b_eq = np.array([target_sum])

def _corr(h):

def _grad_single(h, corr_idx):

    lag = corr_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:

            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            g[:n - m] += (1.0 - h[m:])

    for i in range(len(corr_indices)):
        if weights[i] < 1e-15:

def lp_step(h, target, topk):
    n = len(h)
    c = _corr(h)
    nc = len(c)
    if topk < nc:
        ti = np.argpartition(c, -topk)[-topk:]

        ti = np.arange(nc)

    b = np.zeros(m)

        g = _grad_single(h, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h - c[idx]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]

    return None

                                        4000, 0.08, 0.92, 300, 0.02, 0.01, seed + 20000 + n_hi)
        if _rem() > 60:
            h_new2, sc2 = lp_cycle(h_new, target_hi, [50, 30, 20, 10], _rem() - 30)
            if sc2 < sc:
                h_new, sc = h_new2, sc2
        if _rem() > 40:
            h_new3, sc3 = slsqp_refine(h_new, target_hi, [(500, 15), (5000, 5)], 2000, 1e-15)
            if sc3 < sc:
                h_new, sc = h_new3, sc3
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Final polish at current best resolution
    n_final = len(best_h)
    target_final = n_final / 2.0
    if _rem() > 30:
        h_new, sc = lp_cycle(best_h, target_final, [10, 5, 3, 2], _rem() - 10)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final validation
    best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
