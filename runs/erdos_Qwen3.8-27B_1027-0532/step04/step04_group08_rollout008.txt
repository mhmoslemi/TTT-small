def simulated_annealing_binary(h, n_iter=10000, T_init=1.0, T_min=0.01, deadline=None):
    n = len(h)
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    overlap = compute_overlap(h)
    c5_raw = float(overlap.max())
    best_h = h.copy()
    best_c5 = c5_raw
    
    rng = np.random.RandomState(42)
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        T = T_init * (T_min / T_init) ** (it / n_iter)
        if T < T_min:
            break
        
        # Random swap
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        i = ones_pos[rng.randint(len(ones_pos))]
        j = zeros_pos[rng.randint(len(zeros_pos))]
        
        old_vi, old_vj = h[i], h[j]
        h[i], h[j] = old_vj, old_vi
        overlap = compute_overlap(h)
        new_c5 = float(overlap.max())
        delta = new_c5 - c5_raw
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5_raw = new_c5
            if c5_raw < best_c5:
                best_c5 = c5_raw
                best_h = h.copy()
        else:
            h[i], h[j] = old_vi, old_vj
    
    return best_h, best_c5 * dx

        h_new = project_feasible(h_cur - lr * sg, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
        else:
            lr *= 0.8
            if lr < 1e-12:
                break

    return best_h, best_c5


def _slsqp_soft_obj(x):
    corr = compute_overlap(x)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _slsqp_constraint(x):
    return np.sum(x) - _g_target


def slsqp_polish(h, target, alpha, maxiter, deadline):
    global _g_alpha, _g_target
    n = len(h)
    _g_alpha = alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    res = minimize(_slsqp_soft_obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _slsqp_constraint}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if time.time() > deadline:
        project_feasible(h, target)
        return project_feasible(h, target)
    h_new = project_feasible(res.x, target)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 30.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: ILS at original resolution
    dl1 = min(start_time + 250.0, deadline - 500.0)
    h_p1, c5_p1 = ils_optimize(best_h, max_rounds=50, deadline=dl1, seed=seed)
    if c5_p1 < best_c5:
        best_c5 = c5_p1
        best_h = h_p1

    # Phase 2: SA at original resolution
    if time.time() < deadline - 350.0:
        dl2 = min(start_time + 500.0, deadline - 350.0)
        h_p2, c5_p2 = sa_optimize(best_h, n_iter=300000, T_init=0.3, T_min=0.02,
                                  deadline=dl2, seed=seed + 100)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2

    # Phase 3: Higher resolution (2x)
    if time.time() < deadline - 200.0:
        n_hr = n * 2
        h_hr = np.repeat(best_h, 2)
        h_hr = to_binary(h_hr, n_hr)
        dl3 = min(start_time + 1050.0, deadline - 200.0)
        h_p3, c5_p3 = ils_optimize(h_hr, max_rounds=40, deadline=dl3, seed=seed + 200)
        if c5_p3 < best_c5:
            best_c5 = c5_p3
            best_h = h_p3

    # Phase 4: Even higher resolution (4x) if time allows
    if time.time() < deadline - 150.0:
        n_hr4 = n * 4
        h_hr4 = np.repeat(best_h, n_hr4 // len(best_h))
        if len(h_hr4) != n_hr4:
            h_hr4 = h_hr4[:n_hr4]
        h_hr4 = to_binary(h_hr4, n_hr4)
        dl4 = min(start_time + 1200.0, deadline - 150.0)
        h_p4, c5_p4 = ils_optimize(h_hr4, max_rounds=20, deadline=dl4, seed=seed + 300)
        if c5_p4 < best_c5:
            best_c5 = c5_p4
            best_h = h_p4

    # Phase 5: Subgradient polish
    if time.time() < deadline - 80.0:
        n_best = len(best_h)
        target_best = n_best / 2.0
        dl5 = min(start_time + 1250.0, deadline - 80.0)
        h_p5, c5_p5 = subgradient_polish(best_h, target_best, lr_init=0.02,
                                          n_iter=3000, deadline=dl5)
        if c5_p5 < best_c5:
            best_c5 = c5_p5
            best_h = h_p5

    # Phase 6: SLSQP polish
    if time.time() < deadline - 30.0:
        n_final = len(best_h)
        target_final = n_final / 2.0
        for a in [5000.0, 20000.0, 80000.0]:
            if time.time() > deadline - 10.0:
                break
            dl6 = min(time.time() + 60.0, deadline - 10.0)
            h_p6 = slsqp_polish(best_h, target_final, a, maxiter=200, deadline=dl6)
            c5_p6 = compute_c5(h_p6)
            if c5_p6 < best_c5:
                best_c5 = c5_p6
                best_h = h_p6

    # Final: ensure feasibility
    n_out = len(best_h)
    target_out = n_out / 2.0
    best_h = project_feasible(best_h, target_out)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_out
