E[# of 1s at positions that have a 0 at distance k] ≈ (n-k)/4 * (n/2 / n) ≈ (n-k)/4

So E[C5] ≈ n/4 * 2/n = 1/2 for k=0 (but for binary, k=0 gives 0!). Wait, for binary, O[0] = 0 always.

For k=1: E[O[1]] ≈ (n-1)/4 * 2/n ≈ 1/2
This is too high.

But we can do much better with structured sequences. For example, if we use a "block" structure:
h = [1,1,...,1, 0,0,...,0] with n/2 ones and n/2 zeros.
Then for k < n/2: O[k] = (n/2 - k) * 2/n = 1 - 2k/n
Maximum at k=0: O[0]=0 (binary), k=1: 1-2/n.
For k ≥ n/2: O[k] = 0 (no overlap of 1s and 0s).

So C5 ≈ (n-2)/(2n) ≈ 1/2 for large n. Still not good.

What about h = [1,0,1,0,...,1,0] (alternating)?
For k=1: O[1] = n/2 * 2/n = 1 (all 1s are followed by 0s). Bad!
For k=2: O[2] = 0 (1s are followed by 1s).
So C5 = 1. Terrible.

What about h = [1,1,0,0,1,1,0,0,...] (period 4)?
For k=1: O[1] = (n/4 - 2/n * ... ) hmm this gets complicated.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline=None):
    """SLSQP with log-sum-exp smoothing."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda: np.ones(n)}]
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if deadline and time.time() > deadline:
            break
        
        def obj(x):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = a * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / a
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps=500, n_active=3, lr_init=0.002, lr_decay=0.995, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            # For the full correlation, index k corresponds to lag = k - (n-1)
            lag = k - (n - 1)
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            elif lag > 0:
                # result[n-1+lag] = sum h[j+lag]*(1-h[j])
                # d/dh[p] = (1-h[p-lag]) for lag<=p<n, -h[p+lag] for p<n-lag
                for p in range(lag, n):
                    grad[p] += wk * (1.0 - h_cur[p - lag])
                for p in range(0, n - lag):
                    grad[p] -= wk * h_cur[p + lag]
            else:
                # lag < 0, let kl = -lag > 0
                kl = -lag
                # result[n-1-kl] = sum h[j]*(1-h[j+kl])
                # d/dh[p] = (1-h[p+kl]) for p<n-kl, -h[p-kl] for kl<=p
                for p in range(0, n - kl):
                    grad[p] += wk * (1.0 - h_cur[p + kl])
                for p in range(kl, n):
                    grad[p] -= wk * h_cur[p - kl]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-9:
                break
    return best_h, best_score

def coordinate_descent(h, n, target, n_sweeps=5, n_candidates=21, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h_save = h_cur.copy()
            old_val = h_cur[p]
            # Try candidate values
            candidates = np.linspace(0, 1, n_candidates)
            best_t = old_val
            best_sc = cur_score
            for t in candidates:
                h_cur[p] = t
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
                    h_best = h_cur.copy()
                h_cur = h_save.copy()
            if best_sc < cur_score - 1e-14:
                h_cur = h_best
                cur_score = best_sc
                improved = True
        if not improved:
            break
    return h_cur, cur_score

```python
def differential_evolution(h_best, n, target, pop_size=15, n_generations=50, 
                           F=0.7, CR=0.7, deadline=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_score = compute_c5(h_best)
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = h_best
    for i in range(1, pop_size):
        pop[i] = h_best + rng.randn(n) * 0.1
        pop[i] = project_feasible(pop[i], target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = 0
    
    for gen in range(n_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            # Mutation
            a, b = rng.choice([j for j in range(pop_size) if j != i], size=2, replace=False)
            v = pop[best_idx] + F * (pop[a] - pop[b])
            # Crossover
            j_rand = rng.randint(n)
            u = np.zeros(n)
            mask = (rng.rand(n) < CR) | (np.arange(n) == j_rand)
            u[mask] = v[mask]
            u[~mask] = pop[i][~mask]
            # Clip and project
            u = np.clip(u, 0, 1)
            u = project_feasible(u, target)
            # Selection
            sc_u = compute_c5(u)
            if sc_u < scores[i]:
                pop[i] = u
                scores[i] = sc_u
            # Update best
            if sc_u < scores[best_idx]:
                best_idx = i
        
        # Update global best
        if scores[best_idx] < best_score:
            best_score = scores[best_idx]
            h_best = pop[best_idx].copy()
    
    return h_best, best_score
```

Another thought: maybe I should try a different formulation. Instead of minimizing C5 directly, I can use a "barrier" approach:
- Start with a target t
- Try to find h such that all O[k] ≤ t
- If feasible, decrease t; if not, increase t
- This is a bisection on t

For a given t, the feasibility problem is:
minimize 0
subject to: h ∈ [0,1], ∑h = n/2, max_k O[k] ≤ t

This is equivalent to:
minimize 0
subject to: h ∈ [0,1], ∑h = n/2, O[k] ≤ t for all k

Each O[k] is a quadratic function of h (due to the h[j]*h[j+k] terms). So we have quadratic constraints, which makes this a QP (if we relax to second-order cone or use a QP solver).

Actually, O[k] = ∑ h[j](1-h[j+k]) = ∑h[j] - ∑h[j]h[j+k]
The constraint O[k] ≤ t is: ∑h[j] - ∑h[j]h[j+k] ≤ t, which is: ∑h[j]h[j+k] ≥ ∑h[j] - t

The LHS is a quadratic form (specifically, a bilinear form). This makes the constraint non-convex (since we're constraining a quadratic form to be ≥ a constant, which is a non-convex constraint).

However, if we fix all variables except one, each constraint becomes a single-variable quadratic (hence convex or concave in that variable), and we can solve the 1D problem exactly. This is exactly what coordinate descent does!

So the coordinate descent approach is actually solving the feasibility problem (can we achieve C5 ≤ t?) one coordinate at a time. If for a given t, coordinate descent converges to a feasible point, then C5 ≤ t is achievable.

This suggests a two-level optimization:
Outer: bisection on t
Inner: coordinate descent to find feasible h for target t

One more idea: **proximal gradient with exact line search for the non-smooth part**.

The objective C5(h) = max_k O[k](h) is the max of differentiable functions, so it's differentiable except at points where multiple functions are simultaneously maximal. At such points, it has a subgradient.

I can use:
1. Compute active set S = {k : O[k] = C5}
2. Compute gradient of the average of active functions: g = (1/|S|) ∑_{k∈S} ∇O[k]
3. Take a step: h_new = h - α * g
4. Project to feasible set
5. Use backtracking line search on α

This is essentially what the subgradient method does, but with exact gradient of active functions (not just subgradient of max).

Let me implement this with a twist: instead of using all active lags equally, weight them by how close they are to the maximum (soft max):

w_k = exp(β * (O[k] - C5)) / ∑ exp(β * (O[j] - C5))

with β large (e.g., 100-1000). This gives a smooth approximation to the active set.

def weighted_gradient(h, n, target, n_steps=1000, beta=200.0, 
                      lr_init=0.001, lr_decay=0.998, deadline=None):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stagnant = 0
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_curr = c_max * dx
        if c_curr < best_score - 1e-14:
            best_score = c_curr
            best_h = h_cur.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                break
        
        # Soft weights
        shifted = beta * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum
        
        # Gradient
        grad = np.zeros(n)
        nz_idx = np.where(weights > 1e-10)[0]
        for idx in nz_idx:
            wk = weights[idx]
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            elif lag > 0:
                grad[lag:] += wk * (1.0 - h_cur[:-lag])
                grad[:-lag] -= wk * h_cur[lag:]
            else:
                kl = -lag
                grad[:n-kl] += wk * (1.0 - h_cur[kl:])
                grad[kl:] -= wk * h_cur[:n-kl]
        grad *= dx
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score
