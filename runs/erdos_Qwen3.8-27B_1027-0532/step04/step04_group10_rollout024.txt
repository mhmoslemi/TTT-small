def proj(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    else:
        mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def corr_and_grad_single_lag(h, lag):

    n = len(h)
    # lag is the "true" lag: 0 means no shift, positive means h shifted right
    # In the full correlation, the index is n-1+lag
    k = n - 1 + lag
    if k < 0 or k >= 2*n - 1:
        return 0.0, np.zeros(n)

    return obj, grad

def lbfgs_optimize(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol=1e-16):

        return f, g
    try:

    except:

    n = len(h)

    score = compute_c5(h_best)
    
    for _ in range(2):  # 2 sweeps

            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            x0 = h_best[start:end].copy()
            
            # Fix other blocks, optimize this block
            def block_obj(x):
                h_tmp = h_best.copy()
                h_tmp[start:end] = x
                f, g_full = lse_obj_grad(h_tmp, alpha, topk, penalty)
                return f, g_full[start:end]
            
            block_bounds = [(0.0, 1.0)] * (end - start)
            try:
                res = minimize(block_obj, x0, jac=True, method='L-BFGS-B',

                sc = compute_c5(h_candidate)
                if sc < score:

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    half = n // 2
    penalty = 5000.0

    best_h = proj(h_init, target)
    best_sc = compute_c5(best_h)

    def tl():
        return budget_s - 15.0 - (time.time() - t0)

    bounds_full = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing alpha
    p1_configs = [(30, 50, 500), (80, 35, 450), (150, 30, 400), (300, 25, 350), (600, 20, 300), (1200, 15, 250)]
    for alpha, topk, iters in p1_configs:
        if tl() < 70:
            break
        _g['alpha'], _g['topk'], _g['pen'] = alpha, topk, penalty
        for attempt in range(3):
            if tl() < 35:
                break
            eps = 0.005 / (attempt + 1)
            hs = proj(best_h + rng.randn(n) * eps, target)
            xn = _do_lbfgs(hs, bounds_full, _obj, iters)
            hn = proj(xn, target)
            sc = compute_c5(hn)
            if sc < best_sc:
                best_sc, best_h = sc, hn

    # Phase 2: Anti-periodic constrained optimization
    if tl() > 70:
        h_a = proj_anti(best_h)
        sc_a = compute_c5(h_a)
        a0 = h_a[:half]
        if sc_a < best_sc:
            best_sc, best_h = sc_a, h_a
            a0 = best_h[:half]
        bounds_anti = [(0.0, 1.0)] * half
        for alpha, topk, iters in [(100, 30, 350), (300, 20, 300), (800, 15, 250), (2000, 10, 200)]:
            if tl() < 35:
                break
            _g['alpha'], _g['topk'], _g['pen'] = alpha, topk, penalty
            a_try = np.clip(a0 + rng.randn(half) * 0.004, 0.0, 1.0)
            an = _do_lbfgs(a_try, bounds_anti, _obj_anti, iters)
            an = np.clip(an, 0.0, 1.0)
            h_try = np.concatenate([an, 1.0 - an])
            sc = compute_c5(h_try)
            if sc < best_sc:
                best_sc, best_h = sc, h_try
                a0 = an

    # Phase 3: Momentum subgradient descent
    if tl() > 50:
        v = np.zeros(n)
        step = 0.015
        no_imp = 0
        for it in range(800):
            if tl() < 15:
                break
            h_new, v, _ = subgrad_step(best_h, target, step, v, 0.85)
            sc = compute_c5(h_new)
            if sc < best_sc - 1e-9:
                best_sc, best_h = sc, h_new
                no_imp = 0
                step = min(step * 1.02, 0.04)
            else:
                no_imp += 1
                step *= 0.96
                if no_imp > 80:
                    v = np.zeros(n)
                    step = 0.008
                    no_imp = 0
            if step < 0.0003:
                break

    # Phase 4: Block coordinate descent
    if tl() > 70:
        for nb in [4, 8, 16]:
            if tl() < 35:
                break
            bs = n // nb
            for sweep in range(2):
                if tl() < 15:
                    break
                for b in range(nb):
                    if tl() < 8:
                        break
                    s = b * bs
                    e = s + bs if b < nb - 1 else n
                    _g['blk_h'] = best_h
                    _g['blk_s'] = s
                    _g['blk_e'] = e
                    _g['alpha'], _g['topk'], _g['pen'] = 400, 25, penalty
                    x0 = best_h[s:e].copy()
                    bb = [(0.0, 1.0)] * (e - s)
                    xn = _do_lbfgs(x0, bb, _obj_block, 60)
                    h_try = best_h.copy()
                    h_try[s:e] = np.clip(xn, 0.0, 1.0)
                    h_try = proj(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_sc:
                        best_sc, best_h = sc, h_try

    # Phase 5: Multi-resolution exploration
    if tl() > 100:
        for n_try in [500, 650, 800]:
            if tl() < 70:
                break
            tgt_try = n_try / 2.0
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n_try)
            hi = np.interp(x_new, x_old, best_h)
            hi = proj(hi, tgt_try)
            bu = [(0.0, 1.0)] * n_try
            improved = False
            for alpha, topk in [(150, 35), (500, 20), (1500, 12)]:
                if tl() < 35:
                    break
                _g['alpha'], _g['topk'], _g['pen'] = alpha, topk, penalty
                hs = proj(hi + rng.randn(n_try) * 0.005, tgt_try)
                xn = _do_lbfgs(hs, bu, _obj, 300)
                hn = proj(xn, tgt_try)
                sc = compute_c5(hn)
                if sc < best_sc:
                    best_sc = sc
                    xb = np.linspace(0, 1, n)
                    best_h = np.interp(xb, x_new, hn)
                    best_h = proj(best_h, target)
                    hi = hn
                    improved = True
                elif not improved:
                    hi = hn

    # Phase 6: Fine momentum subgradient polish
    if tl() > 25:
        v = np.zeros(n)
        step = 0.004
        for it in range(400):
            if tl() < 8:
                break
            h_new, v, _ = subgrad_step(best_h, target, step, v, 0.88)
            sc = compute_c5(h_new)
            if sc < best_sc - 1e-10:
                best_sc, best_h = sc, h_new
                step = min(step * 1.01, 0.008)
            else:
                step *= 0.94
                if step < 0.0002:
                    break

    # Phase 7: Coordinate search on active coordinates
    if tl() > 15:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        if lag >= 0:
            rel = np.unique(np.concatenate([np.arange(0, n - lag), np.arange(lag, n)]))
        else:
            rel = np.unique(np.concatenate([np.arange(0, n + lag), np.arange(-lag, n)]))
        rel = rel[:180]
        for c in rel:
            if tl() < 6:
                break
            orig = best_h[c]
            for d in [0.002, -0.002, 0.0008, -0.0008, 0.0003, -0.0003]:
                ht = best_h.copy()
                ht[c] = np.clip(orig + d, 0.0, 1.0)
                ht = proj(ht, target)
                sc = compute_c5(ht)
                if sc < best_sc - 1e-11:
                    best_sc, best_h = sc, ht
                    break

    # Phase 8: Final high-alpha L-BFGS polish
    if tl() > 12:
        for alpha in [3000, 8000, 15000]:
            if tl() < 8:
                break
            _g['alpha'], _g['topk'], _g['pen'] = alpha, 8, penalty
            xn = _do_lbfgs(best_h, bounds_full, _obj, 200)
            hn = proj(xn, target)
            sc = compute_c5(hn)
            if sc < best_sc:
                best_sc, best_h = sc, hn

    # Phase 9: Anti-periodic final check
    if tl() > 10:
        h_a = proj_anti(best_h)
        sc_a = compute_c5(h_a)
        if sc_a < best_sc:
            best_sc, best_h = sc_a, h_a
        _g['alpha'], _g['topk'], _g['pen'] = 5000, 8, penalty
        a0 = best_h[:half]
        bounds_anti = [(0.0, 1.0)] * half
        an = _do_lbfgs(a0, bounds_anti, _obj_anti, 300)
        an = np.clip(an, 0.0, 1.0)
        h_try = np.concatenate([an, 1.0 - an])
        sc = compute_c5(h_try)
        if sc < best_sc:
            best_sc, best_h = sc, h_try

    best_h = proj(best_h, target)
    final_sc = compute_c5(best_h)
    return best_h, final_sc, n
