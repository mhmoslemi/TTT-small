def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:

    c_max = np.max(corr)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        if n - lag > 0:
            g[:n - lag] += (1.0 - h[lag:])
        if lag < n:
            g[lag:] -= h[:n - lag]
    else:
        L = -lag
        if n - L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
    return g

def smooth_gradient(h, alpha, top_k=80):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s < 1e-300:
        return np.zeros(n), c_max
    p = exp_s / s
    idx = np.argsort(p)[::-1][:top_k]
    g = np.zeros(n)
    for k in idx:

        if pk < 1e-10:

        g += pk * subgrad_lag(h, lag)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)

def con_eq_full(x, target):

def slsqp_optimize(h, n, target, alphas, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        try:
            res = minimize(lambda x: lse_obj_full(x, a, n), h_cur, method='SLSQP',
                          bounds=bounds,

                          options={'maxiter': maxiter, 'ftol': 1e-15})

    if elapsed() > budget_s - 60:
        return best_h, best_sc, n0

    # Phase 2: Quick SLSQP at original resolution
    dl2 = t0 + min(80, (budget_s - 60) * 0.1)
    if time.time() < dl2:
        h_sl, sc = slsqp_optimize(best_h, n0, target0, [100.0, 500.0], 1000)
        if sc < best_sc:
            best_sc = sc
            best_h = h_sl

    if elapsed() > budget_s - 60:
        return best_h, best_sc, n0

    # Phase 3: Upscale to 2x and optimize
    n1 = n0 * 2
    target1 = n1 // 2
    h1 = upscale(best_h, 2)
    h1 = project_feasible(h1, target1)
    best_h1 = h1.copy()
    best_sc1 = compute_c5(best_h1)

    dl3 = t0 + min(budget_s - 200, (budget_s) * 0.45)
    # Multiple rounds of gradient descent with decreasing alpha
    for round_idx in range(4):
        if time.time() > min(dl3, t0 + budget_s - 80):
            break
        alpha_round = 200.0 * (0.5 ** round_idx)
        lr_round = 0.03 / (1 + 0.3 * round_idx)
        h_gd, sc = gradient_descent(best_h1, target1, 8000, lr_round, alpha_round,
                                    min(time.time() + 60, dl3))
        if sc < best_sc1:
            best_sc1 = sc
            best_h1 = h_gd

    # Block SLSQP at higher resolution
    if time.time() < min(dl3, t0 + budget_s - 60):
        for bs in [6, 4]:
            if time.time() > min(dl3, t0 + budget_s - 40):
                break
            h_bl, sc = block_slsqp(best_h1, n1, target1, bs, 3, 100.0,
                                   min(time.time() + 40, t0 + budget_s - 40))
            if sc < best_sc1:
                best_sc1 = sc
                best_h1 = h_bl

    if best_sc1 < best_sc:
        best_sc = best_sc1
        best_h = best_h1

    if elapsed() > budget_s - 80:
        return best_h, best_sc, len(best_h)

    # Phase 4: Try 4x upsampling
    n2 = n0 * 4
    target2 = n2 // 2
    h2 = upscale(best_h, 4)
    h2 = project_feasible(h2, target2)
    best_h2 = h2.copy()
    best_sc2 = compute_c5(best_h2)

    dl4 = t0 + budget_s - 100
    # Gradient descent at 4x
    for round_idx in range(3):
        if time.time() > min(dl4, t0 + budget_s - 60):
            break
        alpha_r = 100.0 * (0.6 ** round_idx)
        lr_r = 0.02 / (1 + 0.3 * round_idx)
        h_gd, sc = gradient_descent(best_h2, target2, 5000, lr_r, alpha_r,
                                    min(time.time() + 50, dl4))
        if sc < best_sc2:
            best_sc2 = sc
            best_h2 = h_gd

    # Block SLSQP at 4x
    if time.time() < min(dl4, t0 + budget_s - 40):
        for bs in [5, 4]:
            if time.time() > min(dl4, t0 + budget_s - 30):
                break
            h_bl, sc = block_slsqp(best_h2, n2, target2, bs, 2, 200.0,
                                   min(time.time() + 30, t0 + budget_s - 30))
            if sc < best_sc2:
                best_sc2 = sc
                best_h2 = h_bl

    if best_sc2 < best_sc:
        best_sc = best_sc2
        best_h = best_h2

    if elapsed() > budget_s - 60:
        return best_h, best_sc, len(best_h)

    # Phase 5: Restart exploration at best resolution
    n_best = len(best_h)
    target_best = n_best // 2
    dl5 = t0 + budget_s - 80

    for attempt in range(20):
        if time.time() > dl5:
            break
        h_pert = perturbed_h(best_h, n_best, 0.05, rng)
        h_gd, sc = gradient_descent(h_pert, target_best, 3000, 0.04, 200.0,
                                    min(time.time() + 20, dl5))
        if sc < best_sc:
            best_sc = sc
            best_h = h_gd

    if elapsed() > budget_s - 60:
        return best_h, best_sc, len(best_h)

    # Phase 6: Final polish with block SLSQP
    n_fin = len(best_h)
    target_fin = n_fin // 2
    dl6 = t0 + budget_s - 30
    for bs in [4, 5, 3]:
        if time.time() > dl6:
            break
        h_bl, sc = block_slsqp(best_h, n_fin, target_fin, bs, 4, 500.0,
                               min(time.time() + 25, dl6))
        if sc < best_sc:
            best_sc = sc
            best_h = h_bl

    # Final projection and validation
    best_h = project_feasible(best_h, target_fin)
    final_sc = compute_c5(best_h)
    if final_sc < best_sc:
        best_sc = final_sc

    return best_h, best_sc, len(best_h)
