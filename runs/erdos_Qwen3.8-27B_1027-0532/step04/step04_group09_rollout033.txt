    return obj, grad

def subgrad_polish(h0, target, n_steps, step_size0, active_frac, deadline):

    h_out = np.interp(x_new, x_old, h_src)

        x0 = rng.uniform(0.3, 1.7)
        w = rng.uniform(0.2, 0.7)
        h = 0.5 + 0.4 * np.exp(-((x - x0) / w) ** 2)
    else:
        n_terms = 3
        h = np.zeros(n)
        for j in range(n_terms):
            f = rng.randint(1, 6)
            a = rng.uniform(0.1, 0.3)
            p = rng.uniform(0, 2 * np.pi)
            h += a * np.sin(np.pi * f * x + p)
        h = h - h.min() + 0.1
    return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline_final = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    deadline_p1 = start_time + (budget_s - 15.0) * 0.18
    coarse_levels = [40, 80, 160, 320]
    current_best_coarse = None
    for n_c in coarse_levels:
        target_c = n_c / 2.0
        if time.time() > deadline_p1 - 30:
            break
        level_dl = min(deadline_p1, time.time() + (budget_s - 15.0) * 0.18 / len(coarse_levels))
        best_at_level = None
        best_sc_level = 1.0
        ntries = max(3, int((deadline_p1 - time.time()) / max(10, n_c * 2)))
        ntries = min(ntries, 15)
        for trial in range(ntries):
            if time.time() > deadline_p1 - 10:
                break
            if trial == 0 and current_best_coarse is not None:
                h_c = interpolate_to(current_best_coarse, n_c, target_c)
            else:
                h_c = random_smooth_init(n_c, rng, trial % 5)
                h_c = project_fast(h_c, target_c)
            h_c, sc = fista_optimize(h_c, target_c,
                                     [20, 100, 500, 2000],
                                     min(2000, n_c * 50), level_dl)
            if sc < best_sc_level:
                best_sc_level = sc
                best_at_level = h_c.copy()
            if time.time() > level_dl:
                break
        if best_at_level is not None:
            sl_dl = min(deadline_p1, time.time() + 30)
            best_at_level, sc2 = slsqp_refine(best_at_level, target_c,
                                              [500, 3000], 500, 1e-14, sl_dl)
            if sc2 < best_sc_level:
                best_sc_level = sc2
            current_best_coarse = best_at_level

    if current_best_coarse is not None:
        h_xfer = interpolate_to(current_best_coarse, n_init, target_init)
        sc_xfer = compute_c5(h_xfer)
    else:
        h_xfer = best_h.copy()
        sc_xfer = best_score

    deadline_p2 = start_time + (budget_s - 15.0) * 0.45
    candidates = [best_h, h_xfer]
    h_pert = best_h + 0.05 * rng.randn(n_init)
    h_pert = project_fast(h_pert, target_init)
    candidates.append(h_pert)

    best_p2 = best_h.copy()
    best_sc_p2 = best_score
    per_cand_dl = (deadline_p2 - time.time()) / len(candidates)
    for cand in candidates:
        if time.time() > deadline_p2 - 10:
            break
        cd = min(deadline_p2 - 5, time.time() + per_cand_dl)
        h_r, sc = fista_optimize(cand, target_init,
                                 [50, 300, 1500, 8000], 3000, cd)
        if sc < best_sc_p2:
            best_sc_p2 = sc
            best_p2 = h_r

    h_sl, sc = slsqp_refine(best_p2, target_init,
                            [5000, 15000, 50000], 2000, 1e-16,
                            min(deadline_p2, time.time() + 60))
    if sc < best_sc_p2:
        best_sc_p2 = sc
        best_p2 = h_sl

    if best_sc_p2 < best_score:
        best_score = best_sc_p2
        best_h = best_p2

    deadline_p3 = start_time + (budget_s - 15.0) * 0.62
    h_sg, sc = subgrad_polish(best_h, target_init, 3000, 0.05, 0.02, deadline_p3)
    if sc < best_score:
        best_score = sc
        best_h = h_sg

    h_sl2, sc = slsqp_refine(best_h, target_init,
                             [30000, 80000, 200000], 2000, 1e-16,
                             min(deadline_final - 100, time.time() + 120))
    if sc < best_score:
        best_score = sc
        best_h = h_sl2

    deadline_p4 = start_time + (budget_s - 15.0) * 0.80
    h_sg2, sc = subgrad_polish(best_h, target_init, 3000, 0.02, 0.01, deadline_p4)
    if sc < best_score:
        best_score = sc
        best_h = h_sg2

    h_pert2 = best_h + 0.03 * rng.randn(n_init)
    h_pert2 = project_fast(h_pert2, target_init)
    h_r2, sc2 = fista_optimize(h_pert2, target_init,
                               [200, 1000, 5000], 2000,
                               min(deadline_final - 80, time.time() + 60))
    if sc2 < best_score:
        best_score = sc2
        best_h = h_r2

    if time.time() < deadline_final - 80:
        for n_new in [600, 800]:
            if time.time() > deadline_final - 50:
                break
            target_new = n_new / 2.0
            h_new = interpolate_to(best_h, n_new, target_new)
            sc_new = compute_c5(h_new)
            remaining = deadline_final - 30 - time.time()
            if remaining < 30:
                break
            h_r3, sc3 = fista_optimize(h_new, target_new,
                                       [100, 500, 3000, 15000],
                                       int(remaining * 0.4), time.time() + remaining * 0.4)
            sc3 = compute_c5(h_r3)
            h_r4, sc4 = slsqp_refine(h_r3, target_new,
                                     [10000, 50000], 1000, 1e-15,
                                     min(deadline_final - 20, time.time() + remaining * 0.4))
            if sc4 < best_score:
                best_score = sc4
                best_h = h_r4

    deadline_p6 = deadline_final - 5
    if time.time() < deadline_p6:
        h_final, sc_final = subgrad_polish(best_h, len(best_h) / 2.0,
                                           2000, 0.01, 0.005, deadline_p6)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    best_h = project_fast(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
