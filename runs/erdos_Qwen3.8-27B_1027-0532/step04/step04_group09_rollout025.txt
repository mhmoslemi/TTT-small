            kk = -k
            if kk < n:
                G[idx, kk:] += (-h_current[:n-kk])
                G[idx, :n-kk] += (1.0 - h_current[kk:])
    
    # LP: min t s.t. corr + G*(h-h_cur) <= t, 0<=h<=1, sum(h)=target
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Linearized constraints
    corr_vals = corr  # these are the current values
    lin_terms = G @ (h_var - h_current)
    constraints = [corr_vals[i] + lin_terms[i] <= t_var for i in range(2*n-1)]
    constraints += [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    prob = cp.Problem(cp.Minimize(t_var * dx), constraints)
    try:
        prob.solve(solver=cp.GLOP)  # or CPXPY_DEFAULT
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return h_current, compute_c5(h_current)
    
    if t_var.value is not None and h_var.value is not None:
        h_new = project_feasible(np.array(h_var.value), target)
        sc = compute_c5(h_new)
        return h_new, sc
    
    return h_current, compute_c5(h_current)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    return c_max + np.log(sum_exp) / alpha


def lse_grad(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * x[k:]
                grad[k:] += wi * (1.0 - x[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * x[:n - kk]
                grad[:n - kk] += wi * (1.0 - x[kk:])
    return grad


def lse_penalized_obj(x, alpha, target, mu):
    base = lse_obj(x, alpha)
    s = np.sum(x) - target
    return base + mu * s * s


def lse_penalized_grad(x, alpha, target, mu):
    base_grad = lse_grad(x, alpha)
    s = np.sum(x) - target
    return base_grad + 2.0 * mu * s * np.ones_like(x)


def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline, mu=1e4):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(lse_penalized_obj, h_cur, jac=lse_penalized_grad,
                          args=(alpha, target, mu), method='SLSQP',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def build_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                G[idx, :n - k] += (1.0 - h[k:])
                G[idx, k:] += (-h[:n - k])
        else:
            kk = -k
            if kk < n:
                G[idx, kk:] += (-h[:n - kk])
                G[idx, :n - kk] += (1.0 - h[kk:])
    return G


def lp_direction(h_cur, target, deadline):
    n = len(h_cur)
    dx = 2.0 / n
    corr = compute_corr(h_cur)
    G = build_grad_matrix(h_cur, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    lin_terms = G @ (h_var - h_cur)
    constraints = [corr[i] + lin_terms[i] <= t_var for i in range(2 * n - 1)]
    constraints += [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.array(h_var.value)


def lp_line_search(h_cur, target, h_dir, deadline):
    n = len(h_cur)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur_score
    for frac in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15]:
        if time.time() > deadline:
            break
        h_try = h_cur + frac * (h_dir - h_cur)
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_sc:
            best_sc = sc
            best_h = h_try.copy()
    return best_h, best_sc


def subgrad_momentum(h, target, n_steps, lr, beta, shake_interval, shake_size, deadline, active_frac=0.05):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    v = np.zeros(n)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        threshold = c_max - active_frac * (c_max - float(np.min(corr)))
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        w_top = corr[active]
        w_top = w_top - w_top.min() + 1e-8
        w_top = w_top / w_top.sum()
        G = build_grad_matrix(h_cur, n)
        grad = G[active].T @ w_top
        if it % shake_interval == 0 and it > 0:
            rng_shake = np.random.RandomState(it * 7919 + 13)
            grad += shake_size * rng_shake.randn(n)
        lr_eff = lr / (1.0 + 0.01 * it)
        v = beta * v - lr_eff * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_res_explore(h_base, n_base, deadline, seed):
    rng = np.random.RandomState(seed)
    best_h = h_base.copy()
    best_score = compute_c5(best_h)
    target_base = n_base / 2.0
    coarse_ns = [80, 40, 20]
    for n_coarse in coarse_ns:
        if time.time() > deadline:
            break
        h_down = interpolate_to(best_h, n_coarse, n_coarse / 2.0)
        n_tries = 12 if n_coarse <= 40 else 8
        for trial in range(n_tries):
            if time.time() > deadline:
                break
            h_pert = h_down.copy()
            frac = rng.uniform(0.02, 0.15)
            perm = rng.permutation(n_coarse)
            n_swap = max(2, int(frac * n_coarse))
            for _ in range(n_swap):
                i = perm[_ % n_coarse]
                j = perm[(_ + n_coarse // 2) % n_coarse]
                delta = (h_down[j] - h_down[i]) * rng.uniform(0.3, 1.0)
                h_pert[i] -= delta
                h_pert[j] += delta
            h_pert = project_feasible(h_pert, n_coarse / 2.0)
            sc_coarse = compute_c5(h_pert)
            if sc_coarse < compute_c5(h_down) - 1e-5:
                alpha_trial = [200, 1000, 5000]
                h_opt, sc_opt = slsqp_refine(h_pert, n_coarse / 2.0, alpha_trial, 300, 1e-10, min(deadline, time.time() + 8.0))
                if sc_opt < sc_coarse:
                    h_up = interpolate_to(h_opt, n_base, target_base)
                    sc_up = compute_c5(h_up)
                    if sc_up < best_score - 1e-5:
                        dl_inner = min(deadline, time.time() + 15.0)
                        h_ref, sc_ref = slsqp_refine(h_up, target_base, [1000, 5000, 20000], 500, 1e-12, dl_inner)
                        if sc_ref < best_score:
                            best_score = sc_ref
                            best_h = h_ref
                            h_down = interpolate_to(best_h, n_coarse, n_coarse / 2.0)
    return best_h, best_score


def random_structured_init(n, rng):
    candidates = []
    x = np.linspace(0, 2, n)
    h1 = np.ones(n)
    h1[:n // 2] = 1.0
    h1[n // 2:] = 0.0
    candidates.append(project_feasible(h1, n / 2.0))
    h2 = np.sin(np.pi * x) ** 2
    h2 = project_feasible(h2 / h2.max(), n / 2.0)
    candidates.append(h2)
    h3 = np.zeros(n)
    period = n // 4
    for i in range(n):
        if (i % (2 * period)) < period:
            h3[i] = 1.0
    candidates.append(project_feasible(h3, n / 2.0))
    h4 = np.abs(np.sin(np.pi * x * 2))
    candidates.append(project_feasible(h4 / h4.max(), n / 2.0))
    h5 = np.zeros(n)
    block = n // 8
    for i in range(0, n, 2 * block):
        h5[i:i + block] = 1.0
    candidates.append(project_feasible(h5, n / 2.0))
    h6 = rng.uniform(0, 1, n)
    candidates.append(project_feasible(h6, n / 2.0))
    return candidates
