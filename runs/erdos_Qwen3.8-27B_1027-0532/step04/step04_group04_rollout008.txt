```python
def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Fix if still not exact
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)
```

```python
def tabu_search_v3(h_init, n, target, max_iters, rng, tabu_horizon=50,
                   top_k=40, threshold=None, time_limit=None, start_time=None):
    h = h_init.copy()
    corr = compute_corr(h).astype(float)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_max = float(corr.max())
    stall = 0
    
    if threshold is None:
        threshold = int(current_max) - 1  # Try to get below current max
    
    tabu_expiry = np.full((n, n), -10**9, dtype=np.int64)
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > 3000:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Objective: lexicographic (count_above_threshold, max_corr)
        n_above = int(np.sum(corr > threshold))
        
        # Get top shifts for quick evaluation
        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = top_indices - (n - 1)
        top_vals = corr[top_indices].copy()
        
        # Precompute contrib
        contrib = np.zeros((n, top_k), dtype=np.float64)
        for idx in range(top_k):
            k = int(top_shifts[idx])
            if k >= 0:
                contrib[:n-k, idx] += h[k:]
                contrib[k:, idx] += h[:n-k]
            else:
                idx_neg = -k
                contrib[idx_neg:, idx] += h[:n-idx_neg]
                contrib[:n-idx_neg, idx] += h[idx_neg:]
        
        # Full sweep: evaluate all (i, j) pairs
        bi = len(ones)
        bj = len(zeros)
        
        # Process in batches
        batch = 150
        best_score = (n_above, current_max)
        best_i = -1
        best_j = -1
        
        for i_s in range(0, bi, batch):
            i_batch = ones[i_s:i_s+batch]
            ci = contrib[i_batch, :]  # (b, top_k)
            for j_s in range(0, bj, batch):
                j_batch = zeros[j_s:j_s+batch]
                cj = contrib[j_batch, :]  # (b, top_k)
                
                # new_vals shape: (len_i, len_j, top_k)
                new_vals = top_vals[None, None, :] + ci[:, None, :] - cj[None, :, :]
                new_max = new_vals.max(axis=2)  # (len_i, len_j)
                
                # Estimate n_above: count how many top values exceed threshold
                # (This is an underestimate since we only check top_k shifts)
                new_above = np.sum(new_vals > threshold, axis=2)  # (len_i, len_j)
                
                # Lexicographic score: (new_above, new_max)
                # We want to minimize this
                for a in range(len(i_batch)):
                    for b in range(len(j_batch)):
                        i_idx = int(i_batch[a])
                        j_idx = int(j_batch[b])
                        if tabu_expiry[i_idx, j_idx] > it:
                            continue
                        score = (new_above[a, b], new_max[a, b])
                        if score < best_score:
                            best_score = score
                            best_i = i_idx
                            best_j = j_idx
        
        if best_i < 0:
            stall += 1
            # Random move
            i = int(ones[rng.randint(bi)])
            j = int(zeros[rng.randint(bj)])
        else:
            i, j = best_i, best_j
        
        h[i] = 0
        h[j] = 1
        corr = compute_corr(h).astype(float)
        current_max = float(corr.max())
        new_c5 = current_max * dx
        
        tabu_expiry[i, j] = it + tabu_horizon
        tabu_expiry[j, i] = it + tabu_horizon
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
            # If we've reached the target, we can stop or continue to polish
            if new_c5 <= 0.38 and threshold == 114:
                # Keep going to try to improve further
                pass
        else:
            stall += 1
    
    return best_h, best_c5
```

```python
def update_corr_incremental(corr, h_old, i, j, n):
    """Update corr after swap h[i]: 1->0, h[j]: 0->1. Uses old h values."""
    for k in range(-(n-1), n):

        ik = i + k
        if 0 <= ik < n:
            d += h_old[ik]
        ik2 = i - k
        if 0 <= ik2 < n:
            d += h_old[ik2]
        jk = j + k
        if 0 <= jk < n:
            d -= h_old[jk]
        jk2 = j - k
        if 0 <= jk2 < n:
            d -= h_old[jk2]
        corr[n-1+k] += d
```
