def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    # corr[k] = sum_{valid p} h[p] * (1-h[p+s])

def beam_local_search(h_init, n, max_iters, rng, time_limit, t_start, beam_width=10, t0=0.01, tf=0.00005):
    target = n // 2
    h = project_binary(h_init, target)
    dx = 2.0 / n
    corr = corr_full(h)
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    no_imp = 0
    max_no_imp = max(2000, max_iters // 15)
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if no_imp > max_no_imp:
            # Perturbation
            h = best_h.copy()
            n_p = rng.randint(5, max(6, n // 30))
            for _ in range(n_p):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = int(ones[rng.randint(len(ones))])
                    pj = int(zeros[rng.randint(len(zeros))])
                    h[pi] = 0
                    h[pj] = 1
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            corr = corr_full(h)
            cur_c5 = float(np.max(corr)) * dx
            no_imp = 0
            continue
        
        # Generate beam_width candidate swaps
        best_new_c5 = cur_c5
        best_i = -1
        best_j = -1
        best_new_corr = None
        
        for _ in range(beam_width):
            # Choose swap type
            r = rng.random()
            if r < 0.4:
                # Random swap
                i_f = int(ones[rng.randint(len(ones))])
                j_f = int(zeros[rng.randint(len(zeros))])
            elif r < 0.7:
                # Bottleneck-directed: swap a contributing one with a random zero
                worst_idx = int(np.argmax(corr))
                s_w = worst_idx - (n - 1)
                if s_w >= 0:
                    valid = np.arange(0, n - s_w)
                else:
                    valid = np.arange(-s_w, n)
                contrib = valid[(h[valid] == 1)]
                if len(contrib) > 0:
                    i_f = int(contrib[rng.randint(len(contrib))])
                else:
                    i_f = int(ones[rng.randint(len(ones))])
                j_f = int(zeros[rng.randint(len(zeros))])
            elif r < 0.9:
                # 2-exchange
                i1 = int(ones[rng.randint(len(ones))])
                i2 = int(ones[rng.randint(len(ones))])
                j1 = int(zeros[rng.randint(len(zeros))])
                j2 = int(zeros[rng.randint(len(zeros))])
                if i1 == i2 or j1 == j2:
                    continue
                # Apply 2-exchange
                h[i1] = 0; h[i2] = 0
                h[j1] = 1; h[j2] = 1
                new_corr = corr_full(h)
                new_c5 = float(np.max(new_corr)) * dx
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_new_corr = new_corr
                    best_i = (i1, i2, j1, j2)  # special marker
                h[i1] = 1; h[i2] = 1
                h[j1] = 0; h[j2] = 0
                continue
            else:
                # Greedy: target worst lag, try best zero
                worst_idx = int(np.argmax(corr))
                s_w = worst_idx - (n - 1)
                if s_w >= 0:
                    valid = np.arange(0, n - s_w)
                else:
                    valid = np.arange(-s_w, n)
                contrib = valid[(h[valid] == 1)]
                if len(contrib) == 0:
                    continue
                i_f = int(contrib[rng.randint(len(contrib))])
                # Try a few zeros and pick the best
                for _t in range(3):

                    h[i_f] = 0
                    h[j_try] = 1

                    h[i_f] = 1
                    h[j_try] = 0
                    if new_c5_t < best_new_c5:
                        best_new_c5 = new_c5_t
                        best_new_corr = new_corr_t
                        best_i = i_f
                        best_j = j_try
                continue
            
            # Single swap
            h[i_f] = 0
            h[j_f] = 1
            new_corr = corr_full(h)
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_new_corr = new_corr
                best_i = i_f
                best_j = j_f
            h[i_f] = 1
            h[j_f] = 0
        
        if best_i >= 0:
            if isinstance(best_i, tuple):
                # 2-exchange
                h[best_i[0]] = 0; h[best_i[1]] = 0
                h[best_i[2]] = 1; h[best_i[3]] = 1
            else:
                h[best_i] = 0
                h[best_j] = 1
            corr = best_new_corr
            cur_c5 = best_new_c5
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
    
    return best_h, best_c5
