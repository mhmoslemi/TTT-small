        return lse_obj_and_grad(x, alpha)[0]

        return lse_obj_and_grad(x, alpha)[1]

        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:

                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgrad_nesterov(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        range_cd = c_max - c_min
        if range_cd < 1e-12:
            break
        threshold = c_max - active_frac * range_cd
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= float(len(active))
        mom = (it - 1) / (it + 2.0) if it > 0 else 0.0
        h_mom = h_cur + mom * (h_cur - h_prev)
        step = step_size / (1.0 + 0.005 * it)
        h_new = h_mom - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_prev = h_cur.copy()
        h_cur = h_new
    return best_h, best_score
