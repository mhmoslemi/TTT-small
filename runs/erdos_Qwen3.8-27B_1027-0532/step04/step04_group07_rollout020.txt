def lp_trust_region(h, target_sum, delta):
    n = len(h)
    n_corr = 2*n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Build the LP
    # Variables: h' (n) + t (1)
    # Objective: minimize t
    # Constraints:
    #   g_k · h' - t ≤ -b_k for all k (i.e., g_k · h' ≤ t - b_k)
    #   sum h' = target_sum
    #   h - delta ≤ h' ≤ min(1, h + delta)
    
    # Compute gradients
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    for idx in range(n_corr):
        g = compute_grad_single_lag(h, idx)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = g @ h - corr[idx]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    
    lower = np.maximum(0.0, h - delta)
    upper = np.minimum(1.0, h + delta)
    bounds_lp = list(zip(lower, upper)) + [(-1e6, 1e6)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        t_pred = res.x[n]
        return h_lp, t_pred
    return None, None

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def lp_step_all_lags(h, target_sum, delta):
    """LP step with all lags and trust region."""
    n = len(h)
    n_corr = 2 * n - 1
    corr = overlap_corrs(h)
    
    # Build constraint matrix
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    for idx in range(n_corr):
        g = compute_grad_single_lag(h, idx)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = g @ h - corr[idx]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    
    lower = np.maximum(0.0, h - delta)
    upper = np.minimum(1.0, h + delta)
    bounds_lp = list(zip(lower, upper)) + [(-1e6, 1e6)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum), res.x[n]
    return None, None

def lp_step_top_k(h, target_sum, topk, delta):

    n = len(h)

    corr = overlap_corrs(h)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_weighted(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def get_lp_indices(corr, topk=100, frac=0.005):
    n_corr = len(corr)
    c_max = np.max(corr)
    threshold = c_max * (1.0 - frac)
    near_max = np.where(corr >= threshold)[0]
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    return np.unique(np.concatenate([near_max, top_idx]))


def trust_region_lp(h_start, target_sum, max_iter, time_limit, topk=150, frac=0.005,
                    delta_init=0.05, delta_min=0.002):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    delta = delta_init
    t0 = time.time()
    good_streak = 0
    n = len(h)

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break

        corr = overlap_corrs(h)
        idx = get_lp_indices(corr, topk=topk, frac=frac)
        m = len(idx)

        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i in range(m):
            g = compute_grad_single_lag(h, idx[i])
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = g @ h - corr[idx[i]]

        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])

        lower = np.maximum(0.0, h - delta)
        upper = np.minimum(1.0, h + delta)
        bounds_lp = [(lo, hi) for lo, hi in zip(lower, upper)] + [(-1e6, 1e6)]

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            delta *= 0.5
            if delta < delta_min:
                break
            continue

        if not res.success:
            delta *= 0.5
            if delta < delta_min:
                break
            continue

        h_lp = res.x[:n]
        h_lp = project_feasible(h_lp, target_sum)
        score = compute_c5(h_lp)

        if score < best_score - 1e-13:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            good_streak += 1
            if good_streak % 10 == 0:
                delta = min(delta * 1.15, 0.25)
        else:
            h_frac = project_feasible(0.6 * h + 0.4 * h_lp, target_sum)

                best_score = sc_frac
                best_h = h_frac.copy()
                h = h_frac
            delta *= 0.6
            good_streak = 0
            if delta < delta_min:
                break

    return best_h, best_score

            h_new = project_feasible(h_new, target_sum)

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0

    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum

        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)

        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 2.0)
        else:
            stagnation += 1
            lr *= 0.99
        h = h_new
        if stagnation > patience:
            break

    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
