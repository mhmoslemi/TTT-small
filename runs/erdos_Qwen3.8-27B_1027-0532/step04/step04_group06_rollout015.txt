  - As second factor: if p-j in [0,n-1], delta at lag j is +h[p-j]

For p changing from 0 to 1:
- As first factor: if p+j in [0,n-1], delta at lag j is +(1-h[p+j])
- As second factor: if p-j in [0,n-1], delta at lag j is -h[p-j]

So for a swap (a: 1→0, b: 0→1):
total_delta[j] = [-(1-h[a+j]) if valid] + [h[a-j] if valid] + [(1-h[b+j]) if valid] + [-h[b-j] if valid]

With the caveat that if a+j == b, we should use h[b]=1 (the new value) instead of h[b]=0 (the old value).
Similarly for other interactions.

For simplicity, I'll handle the interaction cases separately (they're rare since a and b are specific positions).

        n_rand = max(2, n_cand - len(candidates))
        for _ in range(n_rand):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        for _ in range(2):
            if len(ones) >= 2 and len(zeros) >= 2:
                ia = rng.randint(len(ones))
                ib = rng.randint(len(zeros))
                candidates.append((ones[ia], zeros[ib]))
        
        improved = False
        best_candidate_swap = None
        best_candidate_max = current_max
        
        seen = set()
        for a_pos, b_pos in candidates:
            if a_pos == b_pos or (a_pos, b_pos) in seen:
                continue
            seen.add((a_pos, b_pos))
            h[a_pos] = 0.0
            h[b_pos] = 1.0
            corr_new = get_corr(h)
            cm = float(np.max(corr_new))
            if cm < best_candidate_max:
                best_candidate_max = cm
                best_candidate_swap = (a_pos, b_pos, corr_new)
            h[a_pos] = 1.0
            h[b_pos] = 0.0
        
        if best_candidate_swap is not None:
            a_pos, b_pos, corr_new = best_candidate_swap
            h[a_pos] = 0.0
            h[b_pos] = 1.0
            corr = corr_new
            current_max = best_candidate_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if current_max < best_max - 1e-12:
                best_max = current_max
                best_h = h.copy()
                no_improve = 0
                stall_count = 0
            else:
                no_improve += 1
                stall_count += 1
        else:
            no_improve += 1
            stall_count += 1
            if stall_count > 3:
                stall_count = 0
                max_no_improve = max(200, max_no_improve - 50)
    
    return best_h, best_max * dx

def binary_2swap_search(h_init, target_sum, time_limit, start_time, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    corr = get_corr(h)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    max_no_improve = 500
    
    while time.time() - start_time < time_limit:
        if no_improve > max_no_improve:
            for _ in range(rng.randint(2, 5)):
                if len(ones) > 0 and len(zeros) > 0:
                    h[ones[rng.randint(len(ones))]] = 0.0
                    h[zeros[rng.randint(len(zeros))]] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = get_corr(h)
            current_max = float(np.max(corr))
            no_improve = 0
            continue
        
        center = n - 1
        corr_work = corr.copy()
        corr_work[center] = -np.inf
        worst_idx = np.argsort(corr_work)[-3:]
        worst_lags = worst_idx - center
        worst_lags = worst_lags[worst_lags != 0]
        if len(worst_lags) == 0:
            worst_lags = np.array([1])
        k_star = int(worst_lags[0])
        
        a_k = ones + k_star
        a_k_valid = (a_k >= 0) & (a_k < n)
        a_km = ones - k_star
        a_km_valid = (a_km >= 0) & (a_km < n)
        
        a_delta = np.zeros(len(ones), dtype=np.float64)
        a_delta[a_k_valid] += -(1.0 - h[np.clip(a_k, 0, n-1)[a_k_valid]])
        a_delta[a_km_valid] += h[np.clip(a_km, 0, n-1)[a_km_valid]]
        
        b_k = zeros + k_star
        b_k_valid = (b_k >= 0) & (b_k < n)
        b_km = zeros - k_star
        b_km_valid = (b_km >= 0) & (b_km < n)
        
        b_delta = np.zeros(len(zeros), dtype=np.float64)
        b_delta[b_k_valid] += (1.0 - h[np.clip(b_k, 0, n-1)[b_k_valid]])
        b_delta[b_km_valid] += -h[np.clip(b_km, 0, n-1)[b_km_valid]]
        
        top_a = ones[np.argsort(a_delta)[:6]]
        top_b = zeros[np.argsort(b_delta)[:6]]
        
        improved = False
        for a1 in top_a[:4]:
            for b1 in top_b[:4]:
                if a1 == b1:
                    continue
                h[a1] = 0.0
                h[b1] = 1.0
                if len(ones) >= 2 and len(zeros) >= 2:
                    ones_tmp = np.where(h == 1)[0]
                    zeros_tmp = np.where(h == 0)[0]
                    
                    ak2 = ones_tmp + k_star
                    ak2v = (ak2 >= 0) & (ak2 < n)
                    akm2 = ones_tmp - k_star
                    akm2v = (akm2 >= 0) & (akm2 < n)
                    ad2 = np.zeros(len(ones_tmp), dtype=np.float64)
                    ad2[ak2v] += -(1.0 - h[np.clip(ak2, 0, n-1)[ak2v]])
                    ad2[akm2v] += h[np.clip(akm2, 0, n-1)[akm2v]]
                    
                    bk2 = zeros_tmp + k_star
                    bk2v = (bk2 >= 0) & (bk2 < n)
                    bkm2 = zeros_tmp - k_star
                    bkm2v = (bkm2 >= 0) & (bkm2 < n)
                    bd2 = np.zeros(len(zeros_tmp), dtype=np.float64)
                    bd2[bk2v] += (1.0 - h[np.clip(bk2, 0, n-1)[bk2v]])
                    bd2[bkm2v] += -h[np.clip(bkm2, 0, n-1)[bkm2v]]
                    
                    a2 = ones_tmp[np.argmin(ad2)]
                    b2 = zeros_tmp[np.argmin(bd2)]
                    if a2 != b2:
                        h[a2] = 0.0
                        h[b2] = 1.0
                        corr_new = get_corr(h)
                        cm = float(np.max(corr_new))
                        if cm < current_max:
                            corr = corr_new
                            current_max = cm
                            ones = np.where(h == 1)[0]
                            zeros = np.where(h == 0)[0]
                            if current_max < best_max - 1e-12:
                                best_max = current_max
                                best_h = h.copy()
                            no_improve = 0
                            improved = True
                            break
                        h[a2] = 1.0
                        h[b2] = 0.0
                else:
                    corr_new = get_corr(h)
                    cm = float(np.max(corr_new))
                    if cm < current_max:
                        corr = corr_new
                        current_max = cm
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        if current_max < best_max - 1e-12:
                            best_max = current_max
                            best_h = h.copy()
                        no_improve = 0
                        improved = True
                        break
                    h[a1] = 1.0
                    h[b1] = 0.0
                    break
                h[a1] = 1.0
                h[b1] = 0.0
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve % 100 == 0:
                max_no_improve = max(100, max_no_improve - 50)
    
    return best_h, best_max * dx

def soft_max_obj(h, alpha, dx, n, target):
    corr = get_corr(h)
    shifted = alpha * (corr - np.max(corr))
    shifted = np.clip(shifted, -500.0, 0.0)
    return np.max(corr) + np.log(np.sum(np.exp(shifted))) / alpha

def continuous_refine(h_init, target_sum, time_limit, start_time, alphas=None, maxiter=300):
    n = len(h_init)
    h = h_init.copy()
    curr_score = compute_c5(h)
    
    if alphas is None:
        alphas = [20, 50, 100, 250, 500, 1000, 2000, 5000]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(
                soft_max_obj, h, args=(alpha, 2.0/n, n, target_sum),
                method='SLSQP', bounds=[(0,1)]*n,
                constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum,
                             'jac':lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h = h_new
        except Exception:
            pass
    
    return h, curr_score

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst + 1)
    x_old = np.linspace(0.0, 1.0, n_src + 1)
    h_out = np.interp(x_new[:-1], x_old[:-1], h_src)
    return h_out

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    
    h_bin = project_binary(h_init, target_init)
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: Aggressive binary local search from initial (40% of time)
    t1_limit = budget_s * 0.35
    t1_deadline = start_time + t1_limit
    h_ls, sc_ls = binary_local_search(h_bin, target_init, t1_limit, start_time, rng, n_cand=12)
    if sc_ls < best_score:
        best_score = sc_ls
        best_h = h_ls.copy()
    
    # Phase 2: 2-swap search from best (15% of time)
    if tleft() > budget_s * 0.12:
        t2_limit = min(budget_s * 0.15, tleft() * 0.6)
        h_2s, sc_2s = binary_2swap_search(best_h, target_init, t2_limit, start_time, rng)
        if sc_2s < best_score:
            best_score = sc_2s
            best_h = h_2s.copy()
    
    # Phase 3: More local search passes with different parameters (20% of time)
    if tleft() > budget_s * 0.15:
        t3_limit = min(budget_s * 0.20, tleft() * 0.6)
        h_ls2, sc_ls2 = binary_local_search(best_h, target_init, t3_limit, start_time, rng, n_cand=16)
        if sc_ls2 < best_score:
            best_score = sc_ls2
            best_h = h_ls2.copy()
    
    # Phase 4: Random restarts (10% of time)
    if tleft() > budget_s * 0.10:
        n_restarts = 3
        t_per_restart = tleft() / (n_restarts + 1)
        for rs in range(n_restarts):
            if tleft() < 30:
                break
            h_rand = np.zeros(n_init)
            perm = rng.permutation(n_init)
            h_rand[perm[:n_init // 2]] = 1.0
            t_r_limit = min(t_per_restart, tleft() * 0.8)
            h_rs, sc_rs = binary_local_search(h_rand, target_init, t_r_limit, start_time, rng, n_cand=10)
            if sc_rs < best_score:
                best_score = sc_rs
                best_h = h_rs.copy()
    
    # Phase 5: Continuous SLSQP refinement (12% of time)
    if tleft() > budget_s * 0.10:
        t5_limit = min(budget_s * 0.12, tleft() * 0.5)
        alphas = [30, 80, 200, 500, 1200, 3000]
        h_cont, sc_cont = continuous_refine(best_h, target_init, t5_limit, start_time,
                                            alphas=alphas, maxiter=200)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()
    
    # Phase 6: Multi-resolution (13% of time)
    if tleft() > budget_s * 0.10:
        n_candidates = [400, 500, 700, 800, 900, 1000, 1200]
        n_candidates = [x for x in n_candidates if x != n_init and x >= 200]
        rng.shuffle(n_candidates)
        
        for n_try in n_candidates[:4]:
            if tleft() < 40:
                break
            target_try = n_try // 2
            if n_try % 2 != 0:
                target_try = n_try // 2
                n_try = target_try * 2
            target_try = n_try / 2.0
            
            h_interp = interpolate_to(best_h, int(n_try))
            h_bin_try = project_binary(h_interp, target_try)
            
            t_r = min(60.0, tleft() * 0.4)
            h_mr, sc_mr = binary_local_search(h_bin_try, target_try, t_r, start_time, rng, n_cand=10)
            
            if tleft() > 20:
                h_mr2, sc_mr2 = continuous_refine(h_mr, target_try, min(20.0, tleft()*0.3), start_time,
                                                   alphas=[100, 300, 800], maxiter=150)
                if sc_mr2 < sc_mr:
                    sc_mr = sc_mr2
                    h_mr = h_mr2
            
            if sc_mr < best_score:
                best_score = sc_mr
                best_h = h_mr.copy()
    
    # Phase 7: Final polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        t_final = min(30.0, tleft() * 0.5)
        h_pol, sc_pol = binary_local_search(h_bin_final, target_use, t_final, start_time, rng, n_cand=20)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol.copy()
        
        if tleft() > 10:
            h_ref, sc_ref = continuous_refine(best_h, target_use, min(10.0, tleft()*0.5), start_time,
                                               alphas=[500, 1500, 5000], maxiter=100)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
    
    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    if final_score > best_score + 1e-10:
        h_final = project_feasible(best_h, target_final)
        final_score = compute_c5(h_final)
    
    return h_final, final_score, n_final
