2. Compute the gradient g_k = ∇f_k(h0) for active lags
3. Linearize: f_k(h) ≈ f_k(h0) + g_k · (h - h0)
4. Solve: minimize t subject to f_k(h0) + g_k · (h-h0) ≤ t for active k, 0≤h≤1, sum(h)=target
5. This is a linear program!
6. Update h0 to the solution
7. Repeat

This is the cutting plane method for DC (difference of convex) or bilinear optimization. It should converge to a local optimum.

def lse_obj_grad(h, alpha, target, penalty, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    n = len(h)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(n_corr)
    corr_top = corr[idx]
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    s = np.sum(h) - target
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

        return lse_obj_grad(x, alpha, target, penalty, topk)
    try:

                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-19, 'gtol': 1e-13})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

    best_h = h.copy()
    best_score = compute_c5(h)
