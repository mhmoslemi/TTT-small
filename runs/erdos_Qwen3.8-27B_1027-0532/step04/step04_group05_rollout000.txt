def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:

        if len(idx2) > 0:

def smooth_max_value_and_grad(h, alpha, dx):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Soft max
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max = c_max + np.log(sum_exp) / alpha
    
    # Gradient: d/dh_i [soft_max]
    # For each k, f_k = sum_i h_i (1-h_{i+k}) dx
    # d f_k / d h_i = (1 - h_{i+k}) dx - h_{i-k} dx (when valid)
    # soft_max = (1/alpha) log sum_k exp(alpha * f_k)
    # d soft_max / d h_i = sum_k [exp(alpha*f_k) / sum_exp] * d f_k / d h_i
    #                    = sum_k p_k * [(1-h_{i+k}) - h_{i-k}] dx
    p = exp_shifted / sum_exp  # softmax weights over k
    
    n_half = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
