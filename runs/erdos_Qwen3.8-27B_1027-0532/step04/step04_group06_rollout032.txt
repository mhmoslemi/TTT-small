```python
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smoothed_obj_and_grad(h, alpha, top_k=30):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(h)
    corr = get_corr(h)
    center = n - 1
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    log_sum_exp = max_c + np.log(np.sum(np.exp(shifted))) / alpha
    probs = np.exp(shifted) / np.sum(np.exp(shifted))
    
    top_idx = np.argsort(probs)[::-1][:top_k]
    top_probs = probs[top_idx]
    top_shifts = top_idx - center
    
    grad = np.zeros(n)
    for p, s in zip(top_probs, top_shifts):
        if p < 1e-12:
            continue
        s = int(s)
        if s >= 0:
            m = n - s
            if m > 0:
                grad[s:] += p * (1.0 - h[:m])
                grad[:m] += p * (-h[s:])
        else:
            t = -s
            m = n - t
            if m > 0:
                grad[:m] += p * (1.0 - h[t:])
                grad[t:] += p * (-h[:m])
    
    return log_sum_exp, grad

def slsqp_continuous(h_start, target, alphas, top_k=30, maxiter=500, ftol=1e-16):
    """Run SLSQP with log-sum-exp smoothed objective at increasing alphas."""
    n = len(h_start)
    h_cur = h_start.astype(float).copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj(h_in):
            val, _ = smoothed_obj_and_grad(h_in, alpha, top_k)
            return val
        
        def grad(h_in):
            _, g = smoothed_obj_and_grad(h_in, alpha, top_k)
            return g
        
        def con_fun(h_in):
            return np.sum(h_in) - target
        
        def con_jac(h_in):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    """Binary simulated annealing with targeted moves."""
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(500, max_iters // 10)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    cached_corr = get_corr(h)
    center = n - 1
    
    # Find worst lags
    def get_worst_lags(corr, top_k=5):
        c = corr.copy()
        c[center] = 0  # exclude center? Actually no, center is valid
        # Actually, the center lag (shift 0) is always large, let's keep it
        top_idx = np.argsort(corr)[::-1][:top_k]
        return top_idx - center
    
    cached_worst = get_worst_lags(cached_corr)
    corr_refresh = 0
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
                for _ in range(rng.randint(2, 4)):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    current_c5 = compute_c5(h)
                cached_corr = get_corr(h)
                cached_worst = get_worst_lags(cached_corr)
            else:
                break
        
        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        h_new = h.copy()
        moved = False
        
        r = rng.random()
        if r < 0.55 and len(ones) > 0 and len(zeros) > 0:
            # Targeted move: move a 1 that's at a peak of worst lag to a 0 that reduces it
            corr_refresh += 1
            if corr_refresh >= 2:
                cached_corr = get_corr(h)
                cached_worst = get_worst_lags(cached_corr)
                corr_refresh = 0
            
            k_star = int(cached_worst[rng.randint(len(cached_worst))])
            a_k = ones + k_star
            a_km = ones - k_star
            valid_a = (a_k >= 0) & (a_k < n) & (a_km >= 0) & (a_km < n)
            b_k = zeros + k_star
            b_km = zeros - k_star
            valid_b = (b_k >= 0) & (b_k < n) & (b_km >= 0) & (b_km < n)
            
            if np.any(valid_a) and np.any(valid_b):
                a_valid = ones[valid_a]
                a_score = h[np.clip(a_k, 0, n-1)[valid_a]] + h[np.clip(a_km, 0, n-1)[valid_a]]
                b_valid = zeros[valid_b]
                b_score = -(h[np.clip(b_k, 0, n-1)[valid_b]] + h[np.clip(b_km, 0, n-1)[valid_b]])
                a_idx = a_valid[np.argmin(a_score)]
                b_idx = b_valid[np.argmin(b_score)]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
            else:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
        elif r < 0.85 and len(ones) > 0 and len(zeros) > 0:
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
            h_new[a_idx] = 0.0
            h_new[b_idx] = 1.0
            moved = True
        elif r < 0.95 and len(ones) >= 2 and len(zeros) >= 2:
            ia = rng.choice(len(ones), 2, replace=False)
            ib = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[ia[0]]] = 0.0
            h_new[ones[ia[1]]] = 0.0
            h_new[zeros[ib[0]]] = 1.0
            h_new[zeros[ib[1]]] = 1.0
            moved = True
        else:
            if len(ones) >= 3 and len(zeros) >= 3:
                ia = rng.choice(len(ones), 3, replace=False)
                ib = rng.choice(len(zeros), 3, replace=False)
                for idx in ia:
                    h_new[ones[idx]] = 0.0
                for idx in ib:
                    h_new[zeros[idx]] = 1.0
                moved = True
        
        if not moved:
            continue
        
        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5

def projected_subgradient(h_start, target, max_iters, step_init, step_decay, 
                          num_restarts, top_k=10, time_limit=None, start_time=None):
    """Projected subgradient method for the smoothed objective."""
    n = len(h_start)
    h = h_start.astype(float).copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for restart in range(num_restarts):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        h = best_h + rng.randn(n) * 0.01 * (restart + 1)
        h = project_feasible(h, target)
        
        step = step_init
        for i in range(max_iters // num_restarts):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            
            _, grad = smoothed_obj_and_grad(h, 50.0, top_k)
            h = h - step * grad
            h = project_feasible(h, target)
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            step *= step_decay
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smoothed_obj(h, alpha):
    corr = get_corr(h)
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def smoothed_grad(h, alpha, top_k=30):
    n = len(h)
    corr = get_corr(h)
    center = n - 1
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    probs = exp_shifted / np.sum(exp_shifted)
    
    top_idx = np.argsort(probs)[::-1][:top_k]
    top_probs = probs[top_idx]
    top_shifts = top_idx - center
    
    grad = np.zeros(n)
    for p, s in zip(top_probs, top_shifts):
        if p < 1e-10:
            continue
        s = int(s)
        if s >= 0:
            m = n - s
            if m > 0:
                grad[s:] += p * (1.0 - h[:m])
                grad[:m] -= p * h[s:]
        else:
            t = -s
            m = n - t
            if m > 0:
                grad[:m] += p * (1.0 - h[t:])
                grad[t:] -= p * h[:m]
    
    return grad

def slsqp_with_grad(h_start, target, alpha, maxiter, ftol, top_k=30):
    n = len(h_start)
    
    def obj(h_in):
        return smoothed_obj(h_in, alpha)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def smoothed_obj(h, alpha):
    corr = get_corr(h)
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def smoothed_grad(h, alpha, top_k=30):
    n = len(h)
    corr = get_corr(h)
    center = n - 1
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        return np.zeros(n)
    probs = exp_shifted / total
    top_idx = np.argsort(probs)[::-1][:top_k]
    top_probs = probs[top_idx]
    top_shifts = top_idx - center
    grad = np.zeros(n)
    for p, s in zip(top_probs, top_shifts):
        if p < 1e-10:
            continue
        s = int(s)
        if s >= 0:
            m = n - s
            if m > 0:
                grad[s:] += p * (1.0 - h[:m])
                grad[:m] -= p * h[s:]
        else:
            t = -s
            m = n - t
            if m > 0:
                grad[:m] += p * (1.0 - h[t:])
                grad[t:] -= p * h[:m]
    return grad

def slsqp_single(h_start, target, alpha, maxiter, top_k=30):
    n = len(h_start)
    def obj(h_in):
        return smoothed_obj(h_in, alpha)
    def grad(h_in):
        return smoothed_grad(h_in, alpha, top_k)
    def con_f(h_in):
        return np.sum(h_in) - target
    def con_j(h_in):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    h_new = h_start.copy()
    try:
        res = minimize(obj, h_new, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
    except Exception:
        pass
    return h_new

def lbfgsb_penalty(h_start, target, alpha, lambda_pen, maxiter=2000, top_k=30):
    n = len(h_start)
    def obj_pen(h_in):
        base = smoothed_obj(h_in, alpha)
        viol = np.sum(h_in) - target
        return base + lambda_pen * viol * viol

        base_g = smoothed_grad(h_in, alpha, top_k)
        viol = np.sum(h_in) - target

                                            use_lbfgsb_from_idx=0, lambda_pen=2000.0)
        if c5_ref5 < best_score:
            best_score = c5_ref5
            best_h = h_ref5.copy()

    # Phase 6: Try scaling the best solution to slightly different resolutions
    for scale in [0.85, 1.15, 1.3]:
        if tleft() < 40:
            break
        n_use = len(best_h)
        n_try = int(n_use * scale)
        if n_try == n_use or n_try < 200 or n_try > 2000:
            continue
        target_try = n_try / 2.0
        h_interp6 = interpolate_to(best_h, n_try, target_try)
        alphas6 = [100, 500, 2000, 10000, 50000]
        h_ref6, c5_ref6 = continuous_refine(h_interp6, target_try, alphas6,
                                            top_k=40, maxiter_per_alpha=250,
                                            use_lbfgsb_from_idx=0, lambda_pen=2000.0)
        if c5_ref6 < best_score:
            best_score = c5_ref6
            best_h = h_ref6.copy()

    # Phase 7: Final binary SA polish at best resolution
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin7 = project_binary(best_h, target_use)
        allowed7 = max(5.0, tleft() - 15)
        max_it7 = int(min(50000, allowed7 * 800))
        h_sa7, c5_sa7 = sa_binary(h_bin7, target_use, max_it7, 0.005, 0.00005,
                                   rng, time_limit=allowed7 * 0.6, start_time=start_time)
        if tleft() > 15:
            alphas7 = [1000, 10000, 100000]
            h_ref7, c5_ref7 = continuous_refine(h_sa7, target_use, alphas7,
                                                top_k=40, maxiter_per_alpha=200,
                                                use_lbfgsb_from_idx=0, lambda_pen=2000.0)
            if c5_ref7 < c5_sa7:
                c5_sa7 = c5_ref7
                h_sa7 = h_ref7
        if c5_sa7 < best_score:
            best_score = c5_sa7
            best_h = h_sa7.copy()

    # Phase 8: Ultra-fine final SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas8 = [10000, 100000, 500000]
        h_ref8, c5_ref8 = continuous_refine(best_h, target_use, alphas8,
                                            top_k=60, maxiter_per_alpha=200,
                                            use_lbfgsb_from_idx=0, lambda_pen=5000.0)
        if c5_ref8 < best_score:
            best_score = c5_ref8
            best_h = h_ref8.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        return best_h, best_score, n_final
    return h_final, final_score, n_final
```
