import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    # Sort by value, set top target_sum to 1, rest to 0
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:int(target_sum)]] = 1.0
    return h_bin

def smooth_max(corr, alpha):
    """Log-sum-exp smoothing of the max."""
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def adam_optimize(h_start, target_sum, lr, beta1, beta2, eps, alpha_start, alpha_end, 
                  max_iters, time_limit, start_time, top_k=10):
    """Adam optimizer for the smoothed max correlation."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target_sum)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    # Focus on top lags for gradient computation
    corr = get_corr(h)
    center = n - 1
    top_lags = np.argsort(-corr)[:top_k]
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        t += 1
        ratio = i / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        
        # Recompute top lags periodically
        if i % 20 == 0:
            corr = get_corr(h)
            top_lags = np.argsort(-corr)[:top_k]
        
        # Compute gradient of smooth_max
        corr = get_corr(h)
        c_max = np.max(corr)
        weights = np.exp(alpha * (corr - c_max))
        weights /= np.sum(weights)
        
        # Gradient of corr[k] w.r.t. h[i]
        # corr[k] = sum_j h[j] * (1-h[j-(k-center)]) for appropriate indices
        # d_corr[k]/d_h[i] = (1-h[i-(k-center)]) - h[i+(k-center)] (with boundary checks)
        
        grad = np.zeros(n)
        for k_idx in top_lags:
            if weights[k_idx] < 1e-10:
                continue
            k = k_idx - center  # actual lag
            w = weights[k_idx]
            # d/dh[i] of sum_j h[j]*(1-h[j-k])
            # = (1-h[i-k]) - h[i+k] (with boundary handling)
            # For h[i]: it appears in h[j] when j=i, contributing (1-h[i-k])
            #           and in (1-h[j-k]) when j-k=i, i.e., j=i+k, contributing -h[i+k]
            if k >= 0:
                # i-k valid when i >= k
                mask1 = np.arange(n) >= k
                grad[mask1] += w * (1.0 - h[np.arange(n)[mask1] - k])
                # i+k valid when i+k < n, i.e., i < n-k
                mask2 = np.arange(n) < (n - k)
                grad[mask2] -= w * h[np.arange(n)[mask2] + k]
            else:
                k_neg = -k
                # i+k_neg valid when i < n-k_neg
                mask1 = np.arange(n) < (n - k_neg)
                grad[mask1] += w * (1.0 - h[np.arange(n)[mask1] + k_neg])
                # i-k_neg valid when i >= k_neg
                mask2 = np.arange(n) >= k_neg
                grad[mask2] -= w * h[np.arange(n)[mask2] - k_neg]
        
        grad *= dx  # account for the dx factor in C5
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h = h_new  # still accept for exploration
            current_c5 = c5_new
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:int(target_sum)]] = 1.0
    return h_bin

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                    max_iters, time_limit, start_time, top_k=15,
                    beta1=0.8, beta2=0.999, eps=1e-8):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target_sum)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        t += 1
        ratio = i / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        
        corr = get_corr(h)
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        # Compute gradient only for significant weights
        significant = np.where(weights > 1e-6)[0]
        if len(significant) == 0:
            continuing breaks
            break
        
        grad = np.zeros(n)
        center = n - 1
        for k_idx in significant:
            k = k_idx - center
            w = weights[k_idx]
            if k == 0:
                # overlap(0) = sum h[i]*(1-h[i]), grad = 1-2h[i]
                grad += w * (1.0 - 2.0 * h)
            elif k > 0:

                if k < n:

                    grad[idx1] += w * (1.0 - h[idx1 - k])

                if k_neg < n:

def project_feasible_v2(h, target_sum):
    """Project onto {h : 0 <= h[i] <= 1, sum(h) = target_sum} using bisection."""
    h = h.astype(float).copy()
    # We want: sum(clip(h - tau, 0, 1)) = target_sum
    # Find tau by bisection
    lo = h.min() - 1
    hi = h.max() + 1
    for _ in range(50):
        tau = (lo + hi) / 2
        h_proj = np.clip(h - tau, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix any residual (should be tiny)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h_proj < 1.0)[0]
        else:
            idx = np.where(h_proj > 0.0)[0]
        if len(idx) > 0:
            h_proj[idx] += diff / len(idx)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def interpolate_to(h_src, n_dst, target_dst):

def compute_grad_topk(h, top_k_indices, weights, dx, n):
    grad = np.zeros(n)
    center = n - 1
    for k_idx in top_k_indices:
        k = k_idx - center
        w = weights[k_idx]
        if w < 1e-8:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < k < n:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            kn = -k
            if kn < n:
                grad[:n - kn] += w * (1.0 - h[kn:])
                grad[kn:] -= w * h[:n - kn]
    return grad * dx


def adam_optimize(h_start, target_sum, lr, alpha_start, alpha_end,
                  max_iters, time_limit, start_time, top_k,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h_start)
    dx = 2.0 / n
    h = project_feasible(h_start.astype(float).copy(), target_sum)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    center = n - 1
    corr = get_corr(h)
    top_idx = np.argsort(-corr)[:top_k * 3]

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        t += 1
        ratio = i / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio

        if i % 50 == 0:
            corr = get_corr(h)
            top_idx = np.argsort(-corr)[:top_k * 3]

        c_max = np.max(corr[top_idx])
        shifted = alpha * (corr[top_idx] - c_max)
        shifted = np.clip(shifted, -400.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= w_sum

        k_sel = np.where(w > 1e-6)[0]
        if len(k_sel) == 0:
            k_sel = np.array([int(np.argmax(w))])
        k_sel = k_sel[:top_k]
        w_sel = w[k_sel]
        idx_sel = top_idx[k_sel]

        grad = compute_grad_topk(h, idx_sel, w_sel, dx, n)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, target_sum)

        if i % 100 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def _find_worst_lags(corr, n, top_k=5):
    center = n - 1
    c = corr.copy()
    c[center] = 0.0
    top = np.argsort(-c)[:top_k * 2]
    lags = top - center
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1])
    return lags[:top_k]


def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    cons_target = target
    cons_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - cons_target,
             'jac': lambda x: np.ones(cons_n)}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x, a=alpha):
                c = np.correlate(x, 1.0 - x, mode='full')
                cm = np.max(c)
                sh = np.clip(a * (c - cm), -400.0, 0.0)
                return float(cm + np.log(np.sum(np.exp(sh))) / a)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin = project_binary(best_h, target_use)
        time_limit = max(5.0, tleft() * 0.4)
        h_sa, sc_sa = targeted_sa(h_bin, target_use, 100000, 0.005, 0.00001,
                                  rng, time_limit, start_time)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n_final
