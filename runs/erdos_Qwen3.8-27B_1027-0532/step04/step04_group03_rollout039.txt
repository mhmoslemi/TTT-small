import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_full(h):
    """Returns the full correlation vector scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) / 2.0
    result = np.clip(h - s, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-9)[0]
        else:
            idx = np.where(result > 1e-9)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def project_binary(h, target_sum):
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1.0)[0]
    zeros = np.where(h_r == 0.0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0
    return h_r

def lse_objective(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lse_gradient(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(n)
    weights = weights / w_sum
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k]
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
        else:
            mm = -m
            grad[:n-mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n-mm]
    return grad

def slsqp_optimize(h_start, target, time_limit, start_time, alpha_list):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    
    def obj(x):
        return lse_objective(x, alpha_list[0], n, target)
    
    # We'll just use the first alpha for the constraint
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alpha_list:
        if time.time() - start_time > time_limit:
            break
        _cons_n = n
        _cons_target = target
        
        def o(x):
            return lse_objective(x, alpha, n, target)
        
        def g(x):
            return lse_gradient(x, alpha, n, target)
        
        def c_eq(x):
            return np.sum(x) - target
        
        def c_eq_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
        
        try:
            res = minimize(o, h, jac=g, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

    corr_change = np.zeros(2*n-1)

    k_min_f = -i
    k_max_f = n-1-i

    r_max_f = k_max_f + (n-1)  # = 2n-2-i

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) / 2.0
    result = np.clip(h - s, 0.0, 1.0)
    residual = target_sum - np.sum(result)

def project_binary(h, target_sum):
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    current_sum = int(round(h_r.sum()))
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1.0)[0]
    zeros = np.where(h_r == 0.0)[0]
    target_int = int(target_sum)
    if current_sum > target_int:
        need = current_sum - target_int
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        need = target_int - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0
    return h_r

def _slsqp_obj(x):
    h = project_feasible(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _slsqp_grad(x):
    h = project_feasible(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(_g_n)
    weights = weights / w_sum
    grad = np.zeros(_g_n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k]
        m = k - (_g_n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:_g_n-m])
            grad[:_g_n-m] -= w * h[m:]
        else:
            mm = -m
            grad[:_g_n-mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:_g_n-mm]
    return grad

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_round(h_start, target, alphas, time_limit, start_time, maxiter=300):

                h = h_new

            new_seg[r - a_start] += v*(1-v) - h[i]*(1-h[i])
        else:
            h_idx = i + k
            new_seg[r - a_start] += delta * (1.0 - h[h_idx])

    for r in range(r2_start, r2_end+1):
        k = r - (n-1)
        if k == 0:
            pass  # Already handled in first factor
        else:
            j = i - k
            new_seg[r - a_start] -= h[j] * delta
    
    new_seg_max = float(np.max(new_seg))
    
    # Max of the rest
    rest_max = -np.inf
    if a_start > 0:
        rest_max = max(rest_max, float(np.max(corr[:a_start])))
    if a_end < 2*n-2:
        rest_max = max(rest_max, float(np.max(corr[a_end+1:])))
    
    return new_seg, new_seg_max, rest_max, (a_start, a_end)

h_i(1-h_{i+k}) = h_i - h_i*h_{i+k}

The product h_i*h_{i+k} is not convex. However, we can use the fact that for h_i, h_{i+k} ∈ [0,1]:
h_i*h_{i+k} ≥ 0 (trivial)
h_i*h_{i+k} ≤ h_i (since h_{i+k} ≤ 1)

So h_i(1-h_{i+k}) = h_i - h_i*h_{i+k} ≤ h_i (since h_i*h_{i+k} ≥ 0)

This gives an upper bound on each correlation, but not tight enough.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) / 2.0
    result = np.clip(h - s, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-9)[0]
        else:
            idx = np.where(result > 1e-9)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def project_binary(h, target_sum):
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    current_sum = int(round(h_r.sum()))
    target_int = int(target_sum)
    if current_sum == target_int:
        return h_r
    ones = np.where(h_r == 1.0)[0]
    zeros = np.where(h_r == 0.0)[0]
    if current_sum > target_int:
        need = current_sum - target_int
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:

        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0

    h = project_feasible(x, _g_target)
    n = _g_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(n)
    weights = weights / w_sum
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]

        m = k - (n - 1)

        else:
            mm = -m
            grad[:n-mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n-mm]
