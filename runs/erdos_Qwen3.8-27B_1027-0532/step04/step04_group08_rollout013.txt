        s = idx - (nn - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:nn-s])
            grad[:nn-s] -= w * h[s:]
        else:
            grad[:nn+s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:nn+s]
    return val, grad

def _pen_obj(x):
    h = np.clip(x, 0.0, 1.0)
    val, g = _sm_val_grad(h, _g_alpha)
    val += _g_penalty * (np.sum(h) - _g_target) ** 2
    return val

def _pen_grad(x):
    h = np.clip(x, 0.0, 1.0)
    val, g = _sm_val_grad(h, _g_alpha)
    pg = 2.0 * _g_penalty * (np.sum(h) - _g_target)
    return g + pg

def _pen_objgrad(x):
    return _pen_obj(x), _pen_grad(x)
