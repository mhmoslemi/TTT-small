    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:

    h = h.copy()

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    h = h.astype(float)
    h_r = np.round(h)

    if current == target_sum:
        return h_r
    if current > target_sum:
        need = int(current - target_sum)
        ones = np.where(h_r == 1.0)[0]

        order = np.argsort(h[ones])

            h_r[ones[order[k]]] = 0.0

For binary h, the full correlate at position m (0-indexed) is:

The number of terms is: min(m+1, 2n-1-m, n) for m = 0, ..., 2n-2.

So the max is likely at positions near the middle (m ≈ n-1), where all n terms contribute. At the boundaries (m=0 or m=2n-2), only 1 term contributes.

For the middle position m=n-1: corr[n-1] = sum_j h[j]*(1-h[j]) = sum_j h[j] - sum_j h[j]^2. For binary h, this is n/2 - n/2 = 0. So the zero-lag is actually 0 for binary!

The max will be at positions m = n-1 ± k for small k, where the number of terms is approximately n - k.

So effectively, for binary h, the relevant lags are k = ±1, ±2, ..., and the value at lag k is approximately:
corr[n-1+k] = sum_{j=0}^{n-1-k} h[j] * (1-h[j+k]) for k > 0
corr[n-1-k] = sum_{j=k}^{n-1} h[j] * (1-h[j-k]) for k > 0

These are the "one-sided" correlations, not the circular ones. They have n-k terms instead of n terms.

This is important! It means the problem is NOT the same as minimizing the circular autocorrelation. It's minimizing the maximum of the one-sided cross-correlations.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_array(h):
    n = len(h)

    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h = h.astype(float)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0.0, 1.0)
    current = h_r.sum()
    if current == target_sum:
        return h_r
    if current > target_sum:
        need = int(current - target_sum)
        ones = np.where(h_r == 1.0)[0]
        order = np.argsort(h[ones])
        for k in range(min(need, len(order))):
            h_r[ones[order[k]]] = 0.0
    else:
        need = int(target_sum - current)
        zeros = np.where(h_r == 0.0)[0]
        order = np.argsort(-h[zeros])
        for k in range(min(need, len(order))):
            h_r[zeros[order[k]]] = 1.0
    diff = target_sum - h_r.sum()
    if diff > 0:
        zeros = np.where(h_r == 0.0)[0]
        for k in range(int(diff)):
            h_r[zeros[k]] = 1.0
    elif diff < 0:
        ones = np.where(h_r == 1.0)[0]
        for k in range(int(-diff)):
            h_r[ones[k]] = 0.0
    return h_r
