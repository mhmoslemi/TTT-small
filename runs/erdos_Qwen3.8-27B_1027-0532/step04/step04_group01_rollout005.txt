            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if j1 == j2:
                continue
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            saved = [(i1, 1), (i2, 1), (j1, 0), (j2, 0)]
            valid = True
        else:
            k = min(3, len(ones), len(zeros))
            if k < 2:
                continue
            idx1 = rng.choice(len(ones), k, replace=False)
            idx0 = rng.choice(len(zeros), k, replace=False)
            for idx in idx1:
                pos = int(ones[idx])
                h[pos] = 0
                saved.append((pos, 1))
            for idx in idx0:
                pos = int(zeros[idx])
                h[pos] = 1
                saved.append((pos, 0))
            valid = True
        if not valid:
            continue
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        new_cost = new_c5 + flat_beta * float(np.std(new_corr)) * 2.0 / n
        delta = new_cost - cur_cost
        if delta < -1e-15 or (temp > 1e-14 and np.exp(-delta / temp) > rng.random()):
            corr = new_corr.astype(np.float64)
            cur_c5 = new_c5
            cur_cost = new_cost
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            for pos, old_val in saved:
                h[pos] = old_val
            no_imp += 1
    return best_h, best_c5


def soft_max_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if c_max <= 0:
        return 0.0
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def continuous_refine(h_init, time_limit, t_start):
    n = len(h_init)
    target = n / 2.0
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 500, 1000]
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return soft_max_objective(x, alpha, n)
            def cfun(x):
                return np.sum(x) - target
            def cjac(x):
                return np.ones(n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def targeted_refine(h_init, target, rng, time_limit, t_start):
    n = len(h_init)
    target_int = int(target)
    h = h_init.astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * 2.0 / n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(500000):
        if time.time() - t_start > time_limit:
            break
        worst_lag_idx = int(np.argmax(corr))
        worst_lag = worst_lag_idx - (n - 1)
        if abs(worst_lag) == 0:
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
        else:
            contrib = []
            for ci in range(len(ones)):
                oi = int(ones[ci])
                pos = oi + worst_lag
                if 0 <= pos < n and h[pos] == 0:
                    contrib.append(ci)
            if contrib:
                ii = int(ones[contrib[rng.randint(len(contrib))]])
            else:
                ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
        h[ii] = 0
        h[jj] = 1
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        if new_c5 < best_c5 - 1e-15:
            corr = new_corr.astype(np.float64)
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[ii] = 1
            h[jj] = 0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start + budget_s - 5.0

    h_init_arr = np.array(initial_h_values, dtype=float)
    n_init = len(h_init_arr)
    target_init = n_init / 2.0

    best_h = h_init_arr.copy()
    best_c5 = compute_c5(best_h)

    def try_update(h_new, c5_new):
        nonlocal best_h, best_c5
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 1: Local search from parent with different flat_beta
    remaining = budget_s - 30.0 - (time.time() - start)
    for fb in [0.3, 0.7, 1.2]:
        if time.time() > hard_deadline:
            break
        alloc = min(180.0, remaining / 3.0)
        h_try, c5_try = local_search_v2(h_init_arr, target_init, 400000, rng,
                                         alloc, start, t_init=0.03, t_final=0.00002, flat_beta=fb)
        try_update(h_try, c5_try)
        remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 2: QR sequence starting points
    remaining = budget_s - 30.0 - (time.time() - start)
    if remaining > 60:
        h_qr = make_qr(n_init, rng)
        alloc = min(150.0, remaining * 0.4)
        h_try, c5_try = local_search_v2(h_qr, target_init, 300000, rng,
                                         alloc, start, t_init=0.02, t_final=0.00002, flat_beta=0.5)
        try_update(h_try, c5_try)
        remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 3: Resolution sweep
    remaining = budget_s - 30.0 - (time.time() - start)
    n_candidates = [n_init - 2, n_init + 2, n_init + 4, n_init + 6,
                    n_init - 4, n_init + 8, n_init + 10, n_init - 6,
                    n_init + 12, n_init + 14, n_init + 16]
    n_candidates = [x for x in n_candidates if x >= 400 and x <= 800 and x % 2 == 0]
    for nt in n_candidates:
        if time.time() > hard_deadline:
            break
        if remaining < 50:
            break
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, nt)
        h_interp = np.interp(x_new, x_old, best_h)
        alloc = min(80.0, remaining * 0.25)
        h_try, c5_try = local_search_v2(h_interp, nt / 2.0, 200000, rng,
                                         alloc, start, t_init=0.02, t_final=0.00003, flat_beta=0.5)
        try_update(h_try, c5_try)
        remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 4: More diverse starts
    remaining = budget_s - 30.0 - (time.time() - start)
    if remaining > 80:
        n_use = len(best_h)
        h_rand = make_random_h(n_use, rng)
        alloc = min(100.0, remaining * 0.35)
        h_try, c5_try = local_search_v2(h_rand, n_use / 2.0, 250000, rng,
                                         alloc, start, t_init=0.04, t_final=0.00003, flat_beta=0.5)
        try_update(h_try, c5_try)
        remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 5: Targeted refinement on best
    remaining = budget_s - 30.0 - (time.time() - start)
    if remaining > 40:
        n_use = len(best_h)
        h_bin = project_binary(best_h, n_use / 2.0)
        alloc = min(120.0, remaining * 0.4)
        h_try, c5_try = targeted_refine(h_bin, n_use / 2.0, rng, alloc, start)
        try_update(h_try, c5_try)
        remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 6: Another round of local search on best with different params
    remaining = budget_s - 30.0 - (time.time() - start)
    if remaining > 50:
        n_use = len(best_h)
        h_bin = project_binary(best_h, n_use / 2.0)
        for fb in [0.2, 1.0]:
            if remaining < 40:
                break
            alloc = min(80.0, remaining * 0.4)
            h_try, c5_try = local_search_v2(h_bin, n_use / 2.0, 200000, rng,
                                             alloc, start, t_init=0.01, t_final=0.00001, flat_beta=fb)
            try_update(h_try, c5_try)
            remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 7: Continuous refinement
    remaining = budget_s - 30.0 - (time.time() - start)
    if remaining > 60:
        alloc = min(120.0, remaining * 0.5)
        h_cont, c5_cont = continuous_refine(best_h, alloc, start)
        if c5_cont < best_c5:
            best_h = h_cont
            best_c5 = c5_cont
        remaining = budget_s - 30.0 - (time.time() - start)

    # Phase 8: Final binary polish
    remaining = budget_s - 30.0 - (time.time() - start)
    if remaining > 20:
        n_use = len(best_h)
        h_bin_final = project_binary(best_h, n_use / 2.0)
        alloc = max(5.0, remaining * 0.6)
        h_try, c5_try = local_search_v2(h_bin_final, n_use / 2.0, 100000, rng,
                                         alloc, start, t_init=0.005, t_final=0.00001, flat_beta=0.5)
        try_update(h_try, c5_try)

    n_final = len(best_h)
    h_final = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5 + 1e-12:
        h_final = project_feasible(best_h, n_final / 2.0)
        final_c5 = compute_c5(h_final)

    return h_final, final_c5, n_final
