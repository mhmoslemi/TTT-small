    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_feasible(res.x, target_sum)
    return h_new, compute_c5(h_new)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def _remaining():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Persistent LP at initial resolution
    if _remaining() > 300:
        t1 = min(350.0, _remaining() - 30.0)
        h_new, score = lp_persistent(best_h, target_init, time_limit=t1,
                                     add_per_iter=5, max_constraints=2000, seed=seed)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: Subgradient refinement
    if _remaining() > 120:
        rng = np.random.RandomState(seed)
        h_try = project_feasible(best_h + rng.randn(n_init) * 0.005, target_init)
        h_new, score = subgradient_refine(h_try, target_init, max_iter=3000,
                                          lr_init=0.08, topk=5, alpha=30,
                                          noise_prob=0.01, noise_scale=0.005,
                                          patience=150, momentum=0.88, seed=seed + 1)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 60:
            h_try2 = project_feasible(best_h + rng.randn(n_init) * 0.003, target_init)
            h_new2, score2 = subgradient_refine(h_try2, target_init, max_iter=2000,
                                                lr_init=0.03, topk=3, alpha=100,
                                                noise_prob=0.0, noise_scale=0.0,
                                                patience=200, momentum=0.92, seed=seed + 2)
            if score2 < best_score:
                best_score = score2
                best_h = h_new2

    # Phase 3: Another LP round (should converge faster in better basin)
    if _remaining() > 200:
        t3 = min(250.0, _remaining() - 30.0)
        h_new, score = lp_persistent(best_h, target_init, time_limit=t3,
                                     add_per_iter=3, max_constraints=3000, seed=seed + 10)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 4: SLSQP smoothing
    if _remaining() > 100:
        h_ref, sc = slsqp_smooth(best_h, target_init, alpha=200, topk=0,
                                 maxiter=1500, ftol=1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Try higher resolution
    for n_try in [500, 600]:
        if _remaining() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        cur_score = compute_c5(h_interp)
        if _remaining() > 100:
            t5 = min(120.0, _remaining() - 30.0)
            h_new, score = lp_persistent(h_interp, target_try, time_limit=t5,
                                         add_per_iter=4, max_constraints=1500, seed=seed + 100 + n_try)
            if score < best_score:
                best_score = score
                best_h = h_new
        if _remaining() > 60:
            h_try_sg = project_feasible(h_interp + np.random.RandomState(seed + n_try).randn(n_try) * 0.005, target_try)
            h_new_sg, score_sg = subgradient_refine(h_try_sg, target_try, max_iter=1500,
                                                    lr_init=0.05, topk=4, alpha=50,
                                                    noise_prob=0.01, noise_scale=0.005,
                                                    patience=150, momentum=0.90, seed=seed + 200 + n_try)
            if score_sg < best_score:
                best_score = score_sg
                best_h = h_new_sg

    # Phase 6: Final polish
    if _remaining() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t_final = min(60.0, _remaining() - 10.0)
        h_new, score = lp_persistent(best_h, target_use, time_limit=t_final,
                                     add_per_iter=3, max_constraints=2000, seed=seed + 500)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc = slsqp_smooth(best_h, target_use, alpha=500, topk=0,
                                 maxiter=1000, ftol=1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
