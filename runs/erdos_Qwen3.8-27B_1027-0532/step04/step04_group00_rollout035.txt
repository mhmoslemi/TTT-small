import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_block_full = None
_g_block_start = 0
_g_block_end = 0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n - 1:n - 1 + n]
    return float(np.max(over)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def make_supertriangle(n, p):
    x = (np.arange(n, dtype=float) + 0.5) * 2.0 / n
    if p >= 1.0:
        a = (p + 1.0) / (2.0 * p)
        h = np.ones(n)
        left = x < a
        right = x > 2.0 - a
        h[left] = np.power(x[left] / a, p)
        h[right] = np.power((2.0 - x[right]) / a, p)
    else:
        base = np.minimum(x, 2.0 - x)
        h = np.power(base, p)
        h = h / h.mean() * 0.5
    return project_feasible(h, n / 2.0)


def make_sine_harmonic(n, freq, amp):
    x = (np.arange(n, dtype=float) + 0.5) * 2.0 / n
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    return project_feasible(h, n / 2.0)


def make_gaussian_bump(n, center, width):
    x = (np.arange(n, dtype=float) + 0.5) * 2.0 / n
    h = np.exp(-0.5 * ((x - center) / width) ** 2)
    h = h / h.sum() * (n / 2.0)
    return project_feasible(h, n / 2.0)


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def block_obj(x):
    h_full = _g_block_full.copy()
    h_full[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    corr = compute_corr(h_full)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_round(h, n, target, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_block_full, _g_block_start, _g_block_end, _g_target, _g_alpha
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = max(1, n // block_size)
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = min(s + block_size, n)
            _g_block_full = h_cur.copy()
            _g_block_start = s
            _g_block_end = e
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur


def worst_lag_surgery(h, n, target, max_steps=300, initial_epsilon=0.002):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_idx = np.argsort(over)[::-1][:5]
        grad = np.zeros(n)
        for k_i in k_idx:
            k = int(k_i)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= h_cur[j_neg - k]
        grad *= dx / len(k_idx)
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.01)
        else:
            eps *= 0.75
            if eps < 1e-8:
                break
    return h_cur, cur_score


def subgradient_polish(h, n, target, n_steps=600, n_active=4, lr_init=0.003, lr_decay=0.996):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        temp = 0.005
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
