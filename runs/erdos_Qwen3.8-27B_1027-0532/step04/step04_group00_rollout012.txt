def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def coordinate_descent(h, n, target, n_sweeps, lam_rate, top_lags, grid_size, deadline):
    """Coordinate descent with exact 1D subproblem for the max-of-lines objective."""
    h_cur = h.copy()
    dx = 2.0 / n
    lam = 0.0
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    t_grid = np.linspace(0.0, 1.0, grid_size)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        # Compute current overlaps to find top lags
        corr = compute_corr(h_cur)
        over = corr[n-1:]  # lags 0 to n-1
        top_k = np.argsort(over)[::-1][:top_lags]
        
        for i in range(n):
            # Compute a_k, b_k for all k in top_lags
            # a_k = coeff of h[i] in O_k, b_k = constant
            k_arr = top_k
            n_k = len(k_arr)
            
            # left_mask: k < n-i means h[i] appears as first factor in h[i](1-h[i+k])
            left_mask = (k_arr < (n - i)).astype(float)
            # right_mask: k <= i means h[i] appears as second factor (negated) in h[i-k](1-h[i])
            right_mask = (k_arr <= i).astype(float)
            
            # h[i+k] for each k in top_lags
            h_ipk = np.zeros(n_k)
            valid_left = k_arr < (n - i)
            if np.any(valid_left):
                h_ipk[valid_left] = h_cur[i + k_arr[valid_left]]
            
            # h[i-k] for each k in top_lags  
            h_ik = np.zeros(n_k)
            valid_right = k_arr <= i
            if np.any(valid_right):
                h_ik[valid_right] = h_cur[i - k_arr[valid_right]]
            
            a = left_mask * (1.0 - h_ipk) - right_mask * h_ik
            b = right_mask * h_ik
            
            # Solve min_t max_k ((a[k]+lam)*t + b[k])
            m = a + lam
            # Vectorized: m[:, None] * t_grid[None, :] + b[:, None] → shape (n_k, grid_size)
            vals = np.max(m[:, None] * t_grid[None, :] + b[:, None], axis=0)
            t_star = t_grid[np.argmin(vals)]
            h_cur[i] = t_star
        
        # Update lambda
        lam += lam_rate * (np.sum(h_cur) - target)
        # Clamp lambda to prevent divergence
        lam = np.clip(lam, -1.0, 1.0)
        # Clip h
        h_cur = np.clip(h_cur, 0.0, 1.0)
        
        # Periodic projection and score check
        if sweep % 10 == 9:
            h_proj = project_feasible(h_cur, target)
            sc = compute_c5(h_proj)
            if sc < best_score:
                best_score = sc
                best_h = h_proj.copy()
    
    # Final projection
    h_final = project_feasible(h_cur, target)
    sc = compute_c5(h_final)
    if sc < best_score:
        best_score = sc
        best_h = h_final.copy()
    
    return best_h, best_score


def pairwise_cd(h, n, target, n_sweeps, lam_rate, top_lags, deadline):
    """Pairwise coordinate descent: optimize pairs (i, i+1) or (i, i+k) at a time."""
    h_cur = h.copy()
    dx = 2.0 / n
    lam = 0.0
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    t_grid = np.linspace(0.0, 1.0, grid_size=50)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        top_k = np.argsort(over)[::-1][:top_lags]
        n_k = len(top_k)
        k_arr = top_k.astype(int)
        
        # Optimize consecutive pairs
        for i in range(0, n-1, 2):
            j = i + 1
            # Compute a_k, b_k for h[i] and h[j]
            # O_k = ... + a_i[k]*h[i] + a_j[k]*h[j] + b[k] ...
            
            # For h[i]:
            left_i = (k_arr < (n - i)).astype(float)
            right_i = (k_arr <= i).astype(float)
            h_ipk_i = np.zeros(n_k)
            vl = k_arr < (n - i)
            if np.any(vl):
                h_ipk_i[vl] = h_cur[i + k_arr[vl]]
            # When computing for h[i], h[j] is fixed (current value)
            # But h[i+k] might include h[j] if k = j-i = 1
            # We treat h[j] as fixed at its current value
            h_ik_i = np.zeros(n_k)
            vr = k_arr <= i
            if np.any(vr):
                h_ik_i[vr] = h_cur[i - k_arr[vr]]
            a_i = left_i * (1.0 - h_ipk_i) - right_i * h_ik_i
            b_i = right_i * h_ik_i
            
            # For h[j]:
            left_j = (k_arr < (n - j)).astype(float)
            right_j = (k_arr <= j).astype(float)
            h_ipk_j = np.zeros(n_k)
            vl = k_arr < (n - j)
            if np.any(vl):
                h_ipk_j[vl] = h_cur[j + k_arr[vl]]
            h_ik_j = np.zeros(n_k)
            vr = k_arr <= j
            if np.any(vr):
                h_ik_j[vr] = h_cur[j - k_arr[vr]]
            a_j = left_j * (1.0 - h_ipk_j) - right_j * h_ik_j
            b_j = right_j * h_ik_j
            
            # 2D subproblem: min_{t1,t2} max_k (a_i[k]*t1 + a_j[k]*t2 + b_i[k] + b_j[k] + lam*(t1+t2))
            # But wait, the constant b already includes terms from other coordinates.
            # Actually, I need to be more careful.
            # O_k = (terms not involving h[i] or h[j]) + a_i[k]*h[i] + a_j[k]*h[j]
            # The "terms not involving" include b_i[k] + b_j[k] only if they don't overlap.
            # Actually b_i[k] = h[i-k]*[i>=k] and b_j[k] = h[j-k]*[j>=k]
            # If i-k == j or j-k == i, there's overlap. Since j = i+1:
            # i-k = j → i-k = i+1 → k = -1 (impossible)
            # j-k = i → i+1-k = i → k = 1
            # So for k=1: b_j[1] = h[j-1] = h[i], which is a variable!
            # This means my decomposition is wrong for k=1.
            # 
            # Let me handle this differently: when k=j-i=1, the term h[i-k]*h[j-k]... 
            # Actually, let me just be very explicit.
            # O_k = sum_{x=0}^{n-1-k} h[x](1-h[x+k])
            # Terms involving h[i] or h[j=i+1]:
            #   x=i: h[i](1-h[i+k])
            #   x=j=i+1: h[i+1](1-h[i+1+k])
            #   x=i-k (if ≥ 0): h[i-k](1-h[i])
            #   x=j-k=i+1-k (if ≥ 0 and ≠ i): h[i+1-k](1-h[i+1])
            #
            # For k=1: x=i-k=i-1: h[i-1](1-h[i]), and x=j-k=i: h[i](1-h[i+1])
            #   The x=i term for h[i] is h[i](1-h[i+1]), which also involves h[j]!
            #   So O_1 has the term h[i](1-h[i+1]) which is bilinear in (h[i], h[j]).
            #
            # This means the 2D subproblem is NOT linear! It's quadratic.
            # 
            # So pairwise CD doesn't work as simply as I thought. Let me skip this
            # and stick with 1D coordinate descent.
            pass
        
        # Just do 1D CD
        for i in range(n):
            left_mask = (k_arr < (n - i)).astype(float)
            right_mask = (k_arr <= i).astype(float)
            h_ipk = np.zeros(n_k)
            vl = k_arr < (n - i)
            if np.any(vl):
                h_ipk[vl] = h_cur[i + k_arr[vl]]
            h_ik = np.zeros(n_k)
            vr = k_arr <= i
            if np.any(vr):
                h_ik[vr] = h_cur[i - k_arr[vr]]
            a = left_mask * (1.0 - h_ipk) - right_mask * h_ik
            b = right_mask * h_ik
            m = a + lam
            vals = np.max(m[:, None] * t_grid[None, :] + b[:, None], axis=0)
            t_star = t_grid[np.argmin(vals)]
            h_cur[i] = t_star
        
        lam += lam_rate * (np.sum(h_cur) - target)
        lam = np.clip(lam, -1.0, 1.0)
        h_cur = np.clip(h_cur, 0.0, 1.0)
        
        if sweep % 20 == 19:
            h_proj = project_feasible(h_cur, target)
            sc = compute_c5(h_proj)
            if sc < best_score:
                best_score = sc
                best_h = h_proj.copy()
    
    h_final = project_feasible(h_cur, target)
    sc = compute_c5(h_final)
    if sc < best_score:
        best_score = sc
        best_h = h_final.copy()
    
    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        c_curr = float(np.max(over)) * dx
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        if k_star == 0:
            grad = (1.0 - 2.0 * h_cur) * dx
        else:
            j_pos = np.arange(0, n - k_star)
            grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]
            grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.05, 0.01)
        else:
            eps *= 0.7
            if eps < 1e-8:
                break
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
