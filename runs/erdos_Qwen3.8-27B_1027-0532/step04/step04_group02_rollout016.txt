```python
def project_binary(h, target_sum):
    n = len(h)
    target = int(round(target_sum))
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target:
        need = current_sum - target
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Fix any remaining discrepancy
    final_sum = int(h_r.sum())
    if final_sum > target:
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:final_sum - target]] = 0
    elif final_sum < target:
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:target - final_sum]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    n = len(h)
    if abs(diff) <= 1.0:
        # Small adjustment: spread evenly
        step = diff / n
        h = np.clip(h + step, 0.0, 1.0)
        # Fix residual
        residual = target_sum - np.sum(h)
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                if residual <= 1e-14:
                    break
                add = min(room[i], residual)
                h[i] += add
                residual -= add
        else:
            room = h.copy()
            order = np.argsort(room)
            for i in order:
                if -residual <= 1e-14:
                    break
                sub = min(h[i], -residual)
                h[i] -= sub
                residual += sub
    else:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            remaining = diff
            for i in order:
                if remaining <= 1e-14:
                    break
                add = min(room[i], remaining)
                h[i] += add
                remaining -= add
        else:
            room = h.copy()
            order = np.argsort(room)
            remaining = -diff
            for i in order:
                if remaining <= 1e-14:
                    break
                sub = min(room[i], remaining)
                h[i] -= sub
                remaining -= sub
    return np.clip(h, 0.0, 1.0)

def greedy_local_search(h, target_sum, max_iter=5000, rng=None):
    """Greedy: repeatedly apply the best improving single swap."""
    n = len(h)
    h = h.copy()
    corr = compute_corr_fft(h)
    best_c5 = float(np.max(corr))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 200
    if rng is None:
        rng = np.random.RandomState(0)
    
    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try a few random swaps, keep the best
        best_delta = 0
        best_i1 = -1
        best_i0 = -1
        n_try = min(50, len(ones) * len(zeros))
        for _ in range(n_try):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0
            h[i0] = 1
            corr_new = compute_corr_fft(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_i1 = i1
                best_i0 = i0
            h[i1] = 1
            h[i0] = 0
        
        if best_i1 >= 0 and best_delta < -1e-10:
            h[best_i1] = 0
            h[best_i0] = 1
            corr = compute_corr_fft(h)
            best_c5 += best_delta
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
    
    return h, best_c5

def sa_search(h_start, target_sum, max_iters, t_init, t_final, rng,
              time_limit=None, start_time=None):
    """Simulated annealing with FFT correlation."""
    n = len(h_start)
    h = h_start.copy()
    if not np.allclose(h, np.round(h), atol=1e-6):
        h = project_binary(h, target_sum)
    
    corr = compute_corr_fft(h)
    current_c5 = float(np.max(corr))
    best_c5 = current_c5
    best_h = h.copy()
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]
    no_improve = 0
    max_no_improve = max(2000, max_iters // 20)
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.03:
                no_improve = 0
            else:
                break
        
        ratio = min(1.0, it / max(max_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio
        
        move_type = rng.randint(4)
        h_new = h.copy()
        moved = False
        
        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            # Single swap with correlation-aware targeting
            i1 = ones_cache[rng.randint(len(ones_cache))]
            worst_idx = int(np.argmax(corr))
            k_star = worst_idx - (n - 1)
            if rng.random() < 0.7:
                # Target zeros near the worst shift positions
                target_pos = (i1 + k_star) % n
                dist = np.abs(zeros_cache.astype(int) - target_pos).astype(float)
                dist = np.minimum(dist, n - dist)  # circular distance
                dist += rng.random(len(zeros_cache)) * 5.0
                i0 = zeros_cache[np.argmin(dist)]
            else:
                i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
            
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            # Double swap
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            i1a = ones_cache[idx1[0]]
            i1b = ones_cache[idx1[1]]
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            i0a = zeros_cache[idx0[0]]
            i0b = zeros_cache[idx0[1]]
            h_new[i1a] = 0
            h_new[i1b] = 0
            h_new[i0a] = 1
            h_new[i0b] = 1
            moved = True
            
        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            # Targeted swap: pick i1 that's involved in worst correlation
            # and i0 that would most reduce the worst correlation
            i1 = ones_cache[rng.randint(len(ones_cache))]
            # Score each zero candidate
            worst_indices = np.argsort(-corr)[:5]
            best_score = 1e9
            best_i0 = zeros_cache[0]
            num_cand = min(15, len(zeros_cache))
            cand_idx = rng.choice(len(zeros_cache), num_cand, replace=False)
            for ci in cand_idx:
                i0 = zeros_cache[ci]
                score = 0.0
                for wi in worst_indices:
                    k = wi - (n - 1)
                    # Estimate delta in corr[wi]
                    j1 = i1 + k
                    j2 = i0 + k
                    # Change: h[i1] 1->0, h[i0] 0->1
                    # corr[wi] change from i1: -(1-h[j1]) if j1 valid, +... 
                    # Simplified: just check if i0 is "good" relative to worst shifts
                    if 0 <= j1 < n and 0 <= j2 < n:
                        # i1 going to 0 removes h[i1]*(1-h[j1]) = 1*(1-h[j1]) from corr
                        # i0 going to 1 adds 1*(1-h[j2]) to corr
                        # Net: -(1-h[j1]) + (1-h[j2]) = h[j1] - h[j2]
                        score += h[j1] - h[j2]
                    elif 0 <= j1 < n:
                        score += -1.0  # removes (1-h[j1])
                    elif 0 <= j2 < n:
                        score += 1.0  # adds (1-h[j2])
                if score < best_score:
                    best_score = score
                    best_i0 = i0
            h_new[i1] = 0
            h_new[best_i0] = 1
            moved = True
            
        elif move_type == 3 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            # Block reversal: reverse a contiguous block
            # Pick two random positions and reverse between them
            pos_a = rng.randint(n)
            pos_b = rng.randint(n)
            lo = min(pos_a, pos_b)
            hi = max(pos_a, pos_b)
            block_len = hi - lo + 1
            if block_len < 2 or block_len > n // 3:
                # Too short or too long, just do a single swap
                i1 = ones_cache[rng.randint(len(ones_cache))]
                i0 = zeros_cache[rng.randint(len(zeros_cache))]
                h_new[i1] = 0
                h_new[i0] = 1
                moved = True
            else:
                h_new[lo:hi+1] = h_new[lo:hi+1][::-1]
                # Check if still valid (same number of ones in block)
                # Reversal preserves the count, so it's always valid
                moved = True
        
        if not moved:
            continue
        
        # Evaluate
        corr_new = compute_corr_fft(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5
        
        if delta < -1e-12 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5

def logsumexp_obj(h, alpha, p_top):
    """Smooth approximation of max correlation using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Use top p values for a closer approximation to max
    sorted_corr = np.sort(corr)
    # Take the top p
    top = sorted_corr[-p_top:]
    m = np.max(top)
    return float(m + np.log(np.sum(np.exp(alpha * (top - m)))) / alpha)
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if p_top is not None and p_top < len(corr):
        thresh = np.sort(corr)[-p_top]
        mask = corr >= thresh
        top = corr[mask]
        weights_full = np.zeros(len(corr))
        weights_full[mask] = np.exp(alpha * (top - np.max(top)))
    else:
        top = corr
        weights_full = np.exp(alpha * (corr - np.max(corr)))
    weights_full /= np.sum(weights_full)
    
    # Gradient computation
    # d corr[k] / dh[i] = (1-h[i+lag_k]) * [valid] - h[i-lag_k] * [valid]
    # grad[i] = dx * sum_k weights_full[k] * ((1-h[i+lag_k])*[valid] - h[i-lag_k]*[valid])
    
    lags = np.arange(2 * n - 1) - (n - 1)  # shape (2n-1,)
    
    # For each i, compute:
    # sum_k w[k] * (1-h[i+lags[k]]) * [0<=i+lags[k]<n]
    # = sum_k w[k] * (1-h)[i+lags[k]] (with (1-h)[j]=0 for j outside [0,n))
    
    # This is a weighted sum of shifted (1-h) vectors
    # = (w ⊛ (1-h))[i] where ⊛ is cross-correlation
    
    # Using np.correlate:
    # np.correlate(a, b, 'full')[k] = sum_j a[j+k-(n_b-1)] * b[j]
    # We want: sum_k w[k] * (1-h)[i + lags[k]]
    # Let's define it as: for each i, sum over k of w[k] * (1-h)[i + lags[k]]
    # Let j = i + lags[k], so lags[k] = j - i, k = j - i + (n-1)
    # = sum_j (1-h)[j] * w[j - i + (n-1)]
    # This is the cross-correlation of (1-h) and w... 
    
    # np.correlate(1-h, w, 'full')[i] = sum_j (1-h)[j+i-(len(w)-1)] * w[j]
    # = sum_j (1-h)[j+i-(2n-2)] * w[j]
    # Hmm, this doesn't match directly.
    
    # Let me just use a matrix approach for clarity, even if it's O(n²):
    # For position i and lag lags[k], (1-h)[i+lags[k]] is valid when 0<=i+lags[k]<n
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    # Vectorized: for each i, the contribution is
    # fwd[i] = sum_k w[k] * one_minus_h[i+lags[k]] * (0<=i+lags[k]<n)
    # bwd[i] = sum_k w[k] * h[i-lags[k]] * (0<=i-lags[k]<n)
    
    # Build the matrix M where M[i,k] = one_minus_h[i+lags[k]] if valid
    # This is n x (2n-1) matrix
    # M[i,k] = one_minus_h[i + lags[k]] if 0 <= i+lags[k] < n
    
    # Actually, let's use a smarter approach.
    # fwd = correlate(w, one_minus_h, 'full') evaluated at specific positions
    # Let me verify:
    # np.correlate(w, o) where w has length 2n-1 and o has length n
    # result[m] = sum_j w[j+m-(n-1)] * o[j] for m=0,...,3n-3
    # We want: sum_k w[k] * o[i+lags[k]] where lags[k] = k-(n-1)
    # = sum_k w[k] * o[i+k-(n-1)]
    # Let j = i+k-(n-1), k = j-i+(n-1)
    # = sum_j w[j-i+(n-1)] * o[j]
    # This matches np.correlate(w, o, 'full')[i] if we check the indexing...
    # np.correlate(w, o, 'full')[m] = sum_j w[j+m-(len(o)-1)] * o[j] = sum_j w[j+m-(n-1)] * o[j]
    # So we want m = i! But i ranges from 0 to n-1, and m ranges from 0 to (2n-2)+(n-1) = 3n-3
    # So fwd[i] = np.correlate(w, one_minus_h, 'full')[i]
    
    # Similarly: bwd[i] = sum_k w[k] * h[i-lags[k]] = sum_k w[k] * h[i-k+(n-1)]
    # Let j = i-k+(n-1), k = i+(n-1)-j
    # = sum_j h[j] * w[i+(n-1)-j]
    # np.correlate(h, w, 'full')[m] = sum_j h[j+m-(len(w)-1)] * w[j] = sum_j h[j+m-(2n-2)] * w[j]
    # We want: sum_j h[j] * w[i+(n-1)-j]
    # Let me try: np.correlate(w, h, 'full')[i+(n-1)]
    # = sum_j w[j+i+(n-1)-(n-1)] * h[j] = sum_j w[j+i] * h[j]
    # Hmm, that's sum_j w[j+i] * h[j], but we want sum_j h[j] * w[i+n-1-j]
    # These are different.
    
    # Let me try: np.correlate(h, w, 'full')[2n-2-i]
    # = sum_j h[j + (2n-2-i) - (2n-2)] * w[j] = sum_j h[j-i] * w[j]
    # Let k = j-i: sum_k h[k] * w[k+i]
    # That's sum_k w[k+i] * h[k], still not quite right.
    
    # OK let me just do it directly with a loop or efficient vectorization.
    
    # Efficient approach: use the fact that lags = -(n-1), ..., n-1
    # fwd[i] = sum_{j=max(0,i-(n-1))}^{min(n-1,i+(n-1))} w[j-i+(n-1)] * one_minus_h[j]
    
    # This is a sliding window convolution. Let's use FFT for this.
    # fwd = ifft(fft(w, 4n) * conj(fft(one_minus_h, 4n)))
    # The right part of this gives the cross-correlation
    
    size = 4 * n
    W = np.fft.rfft(w, size)
    O = np.fft.rfft(one_minus_h, size)
    corr1 = np.fft.irfft(W * np.conj(O), size)
    # corr1[m] = sum_j w[(m-j) mod size] * one_minus_h[j]
    # We want: sum_j w[j-i+(n-1)] * one_minus_h[j] for j in [0,n)
    # = sum_j one_minus_h[j] * w[j-i+(n-1)]
    # Let m - j = j' - i + (n-1) where j' = j... this is getting confusing.
    
    # Let me just use the direct O(n*2n) approach which is O(n²) but with good constants:
    # For each i in range(n):
    #   valid_fwd = (i + lags >= 0) & (i + lags < n)
    #   fwd[i] = sum(w[valid_fwd] * one_minus_h[i + lags[valid_fwd]])
    
    # This is O(n * 2n) = O(n²) = 720K for n=600. Fine.
    
    # Actually, let me use a cleaner vectorized approach:
    # Build a matrix of shape (n, 2n-1) where entry [i,k] = i + lags[k]
    # Then mask valid entries and do the sum.
    
    idx = np.arange(n)[:, None] + lags[None, :]  # shape (n, 2n-1)
    mask = (idx >= 0) & (idx < n)  # shape (n, 2n-1)
    
    # fwd[i] = sum_k w[k] * one_minus_h[idx[i,k]] * mask[i,k]
    # We can do this as: for each i, the valid idx values give positions into one_minus_h
    # This is essentially a gather operation.
    
    # Efficient: create a full matrix of one_minus_h values at the right positions
    # safe_idx = np.where(mask, idx, 0)  # avoid negative indices
    # vals = one_minus_h[safe_idx] * mask * w[None, :]
    # fwd = vals.sum(axis=1)
    
    safe_idx_fwd = np.where(mask, idx, 0)
    fwd = (one_minus_h[safe_idx_fwd] * mask.astype(float) * w[None, :]).sum(axis=1)
    
    idx_bwd = np.arange(n)[:, None] - lags[None, :]  # i - lags[k]
    mask_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    safe_idx_bwd = np.where(mask_bwd, idx_bwd, 0)
    bwd = (h[safe_idx_bwd] * mask_bwd.astype(float) * w[None, :]).sum(axis=1)
    
    grad = dx * (fwd - bwd)
    return grad
```

```python
def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def compute_corr_fft(h):
    n = len(h)
    size = 2 * n - 1
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    return corr * (2.0 / n)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    target = int(round(target_sum))
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target:
        need = current_sum - target
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target:
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:final_sum - target]] = 0
    elif final_sum < target:
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:target - final_sum]] = 1
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    n = len(h)
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add

def lse_obj_and_weights(h, alpha, p_top):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if p_top < len(corr):
        sorted_corr = np.sort(corr)
        thresh = sorted_corr[-p_top]
        weights_raw = np.where(corr >= thresh, np.exp(alpha * (corr - thresh)), 0.0)
    else:
        m = np.max(corr)
        weights_raw = np.exp(alpha * (corr - m))
    wsum = np.sum(weights_raw)
    if wsum < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
        val = float(np.max(corr))
    else:
        weights = weights_raw / wsum
        m = np.max(corr[weights_raw > 0]) if np.any(weights_raw > 0) else 0.0
        val = float(m + np.log(wsum) / alpha)
    return val, weights

    bwd = (h[safe_bwd] * mask_bwd.astype(np.float64) * weights[None, :]).sum(axis=1)

def continuous_opt(h_start, n, target, time_limit, start_time, rng):

    if (time.time() - start_time) < time_limit - 30:
        for alpha in [500, 1500]:
            if (time.time() - start_time) > time_limit:
                break
            try:
                res = minimize(lambda x: lse_objective(x, alpha, n - 1), h_cur,
                               jac=lambda x: lse_gradient(x, alpha, n - 1),
                               method='L-BFGS-B', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score - 1e-14:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass

    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = np.array(h_init, dtype=float)
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)
    best_bin_h = h_bin_start.copy()
    best_bin_score = compute_c5(h_bin_start)

    starts = [h_bin_start]
    h_tm = project_binary(thue_morse_pattern(n_init), target_init)
    starts.append(h_tm)
    for _ in range(4):
        h_rand = np.zeros(n_init)
        perm = rng.permutation(n_init)
        h_rand[perm[:n_init // 2]] = 1.0
        starts.append(h_rand)
    h_block = np.zeros(n_init)
    for i in range(0, n_init, 4):
        if ((i // 4) % 3) != 2:
            h_block[i:min(i + 3, n_init)] = 1.0
    starts.append(project_binary(h_block, target_init))

    sa_time_per = max(60.0, min(180.0, tleft() * 0.5 / len(starts)))

    for si in range(len(starts)):
        if tleft() < 80:
            break
        h_s = starts[si]
        max_it = int(min(200000, sa_time_per * 3000))
        t_init_sa = 0.02 if si == 0 else 0.025
        t_final_sa = 0.00005
        h_sa, c5_sa = sa_search(h_s, target_init, max_it, t_init_sa, t_final_sa,
                                rng, time_limit=sa_time_per, start_time=start_time)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            best_bin_h = h_sa.copy()
        if tleft() > 60:
            h_gl, c5_gl = greedy_ls(h_sa, target_init, 4000, rng)
            if c5_gl < best_bin_score:
                best_bin_score = c5_gl
                best_bin_h = h_gl.copy()
        if best_bin_score < best_score:
            best_score = best_bin_score
            best_h = best_bin_h.copy()

    if tleft() < 50:
        h_final = project_feasible(best_h, target_init)
        return h_final, compute_c5(h_final), n_init

    cont_time = min(400.0, tleft() * 0.5)
    h_cont, sc_cont = continuous_opt(best_bin_h, n_init, target_init,
                                     cont_time, start_time, rng)
    if sc_cont < best_score:
        best_score = sc_cont
        best_h = h_cont.copy()

    if tleft() > 80:
        h_bin_pert = best_bin_h + rng.randn(n_init) * 0.3
        h_bin_pert = project_binary(h_bin_pert, target_init)
        max_it2 = int(min(80000, tleft() * 0.3 * 2500))
        h_sa2, c5_sa2 = sa_search(h_bin_pert, target_init, max_it2, 0.01, 0.00003,
                                  rng, time_limit=tleft() * 0.25, start_time=start_time)
        if tleft() > 40:
            h_gl2, c5_gl2 = greedy_ls(h_sa2, target_init, 3000, rng)
            if c5_gl2 < c5_sa2:
                h_sa2 = h_gl2
                c5_sa2 = c5_gl2
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        if c5_sa2 < best_bin_score:
            best_bin_score = c5_sa2
            best_bin_h = h_sa2.copy()

    if tleft() > 60:
        cont_time2 = min(200.0, tleft() * 0.4)
        h_cont2, sc_cont2 = continuous_opt(best_h, n_init, target_init,
                                           cont_time2, start_time, rng)
        if sc_cont2 < best_score:
            best_score = sc_cont2
            best_h = h_cont2.copy()

        npert = min(4, int(max(1, tleft() // 50)))
        for _ in range(npert):
            if tleft() < 35:
                break
            noise = rng.choice([0.02, 0.05, 0.1])
            h_p = best_h + rng.randn(n_init) * noise
            h_p = project_feasible(h_p, target_init)
            ct = min(40.0, tleft() * 0.2)
            h_cp, sc_cp = continuous_opt(h_p, n_init, target_init, ct, start_time, rng)
            if sc_cp < best_score:
                best_score = sc_cp
                best_h = h_cp.copy()

    if tleft() > 30:
        h_sa_f, c5_f = sa_search(project_binary(best_h, target_init), target_init,
                                 int(max(5000, tleft() * 1500)), 0.005, 0.00002,
                                 rng, time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
```
