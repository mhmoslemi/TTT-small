def compute_c5(h):

    return float(np.max(corr[n - 1:])) * dx


def project_box_sum(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)


def upsample_step(h, factor):

        h_new = h_cur + lr * vel
        h_new = project_box_sum(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-13:
                break
    return best_h, best_score


def worst_lag_surgery(h_in, n, target, max_steps, eps_init, deadline):
    h_cur = h_in.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n - 1:]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = (1.0 - 2.0 * h_cur) * dx
        else:
            nk = n - k_star
            grad[:nk] = (1.0 - h_cur[k_star:]) * dx
            grad[k_star:] = -h_cur[:nk] * dx
        h_new = h_cur - eps * grad
        h_new = project_box_sum(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.06, 0.05)
        else:
            eps *= 0.7
            if eps < 1e-8:
                break
    return h_cur, cur_score


def slsqp_optimize(h_in, n, target, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    cur_n = n

    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        over = corr[cur_n - 1:]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(cur_n)

    bounds = [(0.0, 1.0)] * cur_n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fn, h_in, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() < deadline:
            return project_box_sum(res.x, target)
    except Exception:
        pass
    return h_in


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 25.0

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    n_init = len(initial_h_values)
    target_init = n_init / 2.0
    h_init = project_box_sum(np.array(initial_h_values, dtype=float), target_init)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    best_n = n_init

    # Phase 1: Quick polish at original resolution
    if tleft() > 120:
        h_sg, sc_sg = subgradient_momentum(h_init, n_init, target_init,
                                           n_steps=400, n_active=4, temp=0.005,
                                           lr_init=0.003, lr_decay=0.997,
                                           momentum=0.5, deadline=deadline)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 100:
        h_ref = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha=3000,
                               maxiter=2000, ftol=1e-16, deadline=deadline)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Upsample to 2x and optimize with mirror descent
    n2 = 2 * n_init
    target2 = n2 / 2.0
    h_up2 = upsample_step(best_h, 2)
    h_up2 = project_box_sum(h_up2, target2)

    if tleft() > 150:
        h_md2, sc_md2 = mirror_descent_minimax(h_up2, n2, target2,
                                               n_steps=600, deadline=deadline,
                                               lr_init=0.005, temp_init=0.03,
                                               temp_decay=0.998)
        if sc_md2 < best_score:
            best_score = sc_md2
            best_h = h_md2
            best_n = n2

    if tleft() > 100:
        h_sg2, sc_sg2 = subgradient_momentum(best_h, best_n, best_n / 2.0,
                                             n_steps=500, n_active=5, temp=0.004,
                                             lr_init=0.002, lr_decay=0.998,
                                             momentum=0.6, deadline=deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 3: Upsample to 4x original and optimize
    if tleft() > 250:
        n4 = 4 * n_init
        target4 = n4 / 2.0
        h_up4 = upsample_step(best_h, 2)
        h_up4 = project_box_sum(h_up4, target4)
        sc4 = compute_c5(h_up4)

        if tleft() > 150:
            h_md4, sc_md4 = mirror_descent_minimax(h_up4, n4, target4,
                                                   n_steps=800, deadline=deadline,
                                                   lr_init=0.004, temp_init=0.025,
                                                   temp_decay=0.9985)
            if sc_md4 < best_score:
                best_score = sc_md4
                best_h = h_md4
                best_n = n4
        elif sc4 < best_score:
            best_score = sc4
            best_h = h_up4
            best_n = n4

        if tleft() > 100:
            h_sg4, sc_sg4 = subgradient_momentum(best_h, best_n, best_n / 2.0,
                                                 n_steps=600, n_active=6, temp=0.003,
                                                 lr_init=0.001, lr_decay=0.999,
                                                 momentum=0.7, deadline=deadline)
            if sc_sg4 < best_score:
                best_score = sc_sg4
                best_h = h_sg4

    # Phase 4: Upsample to 8x original if time allows
    if tleft() > 400:
        n8 = 8 * n_init
        target8 = n8 / 2.0
        h_up8 = upsample_step(best_h, 2)
        h_up8 = project_box_sum(h_up8, target8)
        sc8 = compute_c5(h_up8)

        if tleft() > 200:
            h_md8, sc_md8 = mirror_descent_minimax(h_up8, n8, target8,
                                                   n_steps=1000, deadline=deadline,
                                                   lr_init=0.003, temp_init=0.02,
                                                   temp_decay=0.999)
            if sc_md8 < best_score:
                best_score = sc_md8
                best_h = h_md8
                best_n = n8
        elif sc8 < best_score:
            best_score = sc8
            best_h = h_up8
            best_n = n8

        if tleft() > 100:
            h_sg8, sc_sg8 = subgradient_momentum(best_h, best_n, best_n / 2.0,
                                                 n_steps=500, n_active=5, temp=0.003,
                                                 lr_init=0.001, lr_decay=0.999,
                                                 momentum=0.7, deadline=deadline)
            if sc_sg8 < best_score:
                best_score = sc_sg8
                best_h = h_sg8

    # Phase 5: Worst-lag surgery at current best
    if tleft() > 80:
        h_surg, sc_surg = worst_lag_surgery(best_h, best_n, best_n / 2.0,
                                            max_steps=200, eps_init=0.003, deadline=deadline)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 6: Final mirror descent with finer temperature
    if tleft() > 80:
        h_md_fin, sc_md_fin = mirror_descent_minimax(best_h, best_n, best_n / 2.0,
                                                     n_steps=400, deadline=deadline,
                                                     lr_init=0.002, temp_init=0.01,
                                                     temp_decay=0.999)
        if sc_md_fin < best_score:
            best_score = sc_md_fin
            best_h = h_md_fin

    # Phase 7: Random restarts
    if tleft() > 120:
        scales = [0.04, 0.08, 0.03, 0.06, 0.05]
        for attempt in range(len(scales)):
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(best_n) * scales[attempt]
            h_pert = project_box_sum(h_pert, best_n / 2.0)
            h_ref, sc_ref = subgradient_momentum(h_pert, best_n, best_n / 2.0,
                                                 n_steps=200, n_active=4, temp=0.005,
                                                 lr_init=0.002, lr_decay=0.997,
                                                 momentum=0.5, deadline=deadline)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 40:
                h_md_r, sc_md_r = mirror_descent_minimax(h_pert, best_n, best_n / 2.0,
                                                         n_steps=150, deadline=deadline,
                                                         lr_init=0.003, temp_init=0.015,
                                                         temp_decay=0.998)
                if sc_md_r < best_score:
                    best_score = sc_md_r
                    best_h = h_md_r

    # Phase 8: Final surgery + subgradient
    if tleft() > 50:
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, best_n, best_n / 2.0,
                                              max_steps=100, eps_init=0.002, deadline=deadline)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    if tleft() > 30:
        h_sg_fin, sc_sg_fin = subgradient_momentum(best_h, best_n, best_n / 2.0,
                                                   n_steps=300, n_active=4, temp=0.002,
                                                   lr_init=0.0005, lr_decay=0.999,
                                                   momentum=0.8, deadline=deadline)
        if sc_sg_fin < best_score:
            best_score = sc_sg_fin
            best_h = h_sg_fin

    # Final projection and return
    h_final = project_box_sum(best_h, best_n / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
