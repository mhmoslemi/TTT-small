from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, lag, n):
    k = abs(lag)
    if k == 0:
        return 1.0 - 2.0 * h
    if k >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        j = k
        grad[j:] += 1.0 - h[:n - j]
        grad[:n - j] -= h[j:]
    return grad

def full_lse_gradient(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    w_norm = w / w_sum
    f_val = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w_norm[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += w_norm[k] * lag_grad(x, lag, n)
    return f_val, grad

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _slsqp_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    w_norm = w / w_sum
    f_val = c_max + np.log(w_sum) / _g_alpha
    grad = np.zeros(_g_n)
    for k in range(len(corr)):
        if w_norm[k] < 1e-12:
            continue
        lag = k - (_g_n - 1)
        grad += w_norm[k] * lag_grad(x, lag, _g_n)
    return f_val, grad

def _slsqp_con_eq(x):
    return np.sum(x) - _g_target

def _slsqp_con_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h, n, target, alphas, maxiter, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    cur = compute_c5(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_eq, 'jac': _slsqp_con_jac}]
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_slsqp_obj_grad, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def pgd_topk(h, n, target, max_iter, deadline, rng, topk_start, topk_end, step_init):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur
    step = step_init
    stall_count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        t_frac = float(it) / max(max_iter, 1)
        K = int(topk_start + (topk_end - topk_start) * min(t_frac * 2.5, 1.0))
        K = max(1, min(K, len(corr)))
        top_idx = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            grad += lag_grad(h_cur, lag, n)
        grad /= float(K)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        if t_frac < 0.25:
            noise_scale = 0.008 * (1.0 - t_frac / 0.25)
            grad += noise_scale * rng.randn(n) / gnorm
        h_new = h_cur - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        new_sc = compute_c5(h_new)
        if new_sc < cur:
            h_cur = h_new
            cur = new_sc
            stall_count = 0
            step = min(step * 1.02, step_init * 2.5)
            if cur < best_sc:
                best_sc = cur
                best_h = h_cur.copy()
        else:
            stall_count += 1
            step *= 0.85
            if stall_count > 200 and t_frac < 0.6:
                n_pert = max(3, n // 60)
                idx_pert = rng.randint(0, n, size=n_pert)
                h_cur[idx_pert] += rng.normal(0, 0.04, n_pert)
                h_cur = np.clip(h_cur, 0.0, 1.0)
                h_cur = project_feasible(h_cur, target)
                cur = compute_c5(h_cur)
                stall_count = 0
    return best_h, best_sc

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0.0, float(old_n - 1), new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, new_n / 2.0)
    return h_new

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

def periodic_h(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        h[:total - n // 2] = 0.0
    elif total < n // 2:
        deficit = n // 2 - total
        mask = h == 0
        indices = np.where(mask)[0]
        for j in range(min(deficit, len(indices))):
            h[indices[j]] = 1.0
    return h

def multi_start_initial(n, rng):
    best_h = None
    best_sc = 1.0
    for L1 in range(20, n // 2, 7):
        ht = two_interval_h(n, L1)
