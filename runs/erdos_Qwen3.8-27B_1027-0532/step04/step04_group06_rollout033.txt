def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]

        return h
    h = h.copy()
    h[idx] += diff / len(idx)

def project_binary(h, target_sum):

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_fun_target = target
    cons_fun_n = n
    
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target
    
    def cons_j(h_in):
        return np.ones(cons_fun_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_continuous(h_start, target_sum, max_iters, step_init, step_final, rng, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ratio = i / max(max_iters - 1, 1)
        step = step_init * (step_final / step_init) ** ratio
        
        corr = get_corr(h)
        center = n - 1
        top_lags = find_worst_lags(corr, n, top_k=5)
        
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            g = np.zeros(n)
            for i_idx in range(n):
                val = 0.0
                if 0 <= i_idx - k < n:
                    val += (1.0 - h[i_idx - k])
                if 0 <= i_idx + k < n:
                    val -= h[i_idx + k]
                g[i_idx] = val
            grad += g
        grad /= len(top_lags)
        
        h_new = h - step * grad * dx
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def find_worst_lags(corr, n, top_k=5, exclude_center=True):
    center = n - 1
    c = corr.copy()
    if exclude_center:
        c[center] = 0.0
    cmax = c.max()
    if cmax == 0:
        return np.array([1])
    threshold = cmax * 0.85
    candidates = np.where(c >= threshold)[0]
    if len(candidates) == 0:
        candidates = np.array([center + 1])
    order = c[candidates][::-1].argsort()
    selected = candidates[order[:top_k]]
    lags = selected - center
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1])
    return lags

def multi_lag_score(h, ones, zeros, worst_lags, n):
    n_ones = len(ones)
    n_zeros = len(zeros)
    if n_ones == 0 or n_zeros == 0:
        return None
    n_lags = len(worst_lags)
    a_scores = np.zeros(n_ones)
    b_scores = np.zeros(n_zeros)
    for li in range(n_lags):
        k = abs(int(worst_lags[li]))
        if k == 0:
            k = 1
        a_k = ones + k
        a_km = ones - k
        valid_a = (a_k >= 0) & (a_k < n) & (a_km >= 0) & (a_km < n)
        if np.any(valid_a):
            a_scores[valid_a] += h[np.clip(a_k, 0, n-1)[valid_a]] - h[np.clip(a_km, 0, n-1)[valid_a]]
        b_k = zeros + k
        b_km = zeros - k
        valid_b = (b_k >= 0) & (b_k < n) & (b_km >= 0) & (b_km < n)
        if np.any(valid_b):
            b_scores[valid_b] -= (h[np.clip(b_k, 0, n-1)[valid_b]] + h[np.clip(b_km, 0, n-1)[valid_b]])
    return a_scores, b_scores

def block_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 12)

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    cached_corr = get_corr(h)
    cached_worst = find_worst_lags(cached_corr, n, top_k=4)
    corr_refresh = 0
    n_lag_refresh = 5

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.12:
                no_improve = 0
                for _ in range(rng.randint(3, 7)):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    current_c5 = compute_c5(h)
                cached_corr = get_corr(h)
                cached_worst = find_worst_lags(cached_corr, n, top_k=4)
            else:
                break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.50 and len(ones) > 0 and len(zeros) > 0:
            corr_refresh += 1
            if corr_refresh >= n_lag_refresh:
                cached_corr = get_corr(h)
                cached_worst = find_worst_lags(cached_corr, n, top_k=4)
                corr_refresh = 0

            scores = multi_lag_score(h, ones, zeros, cached_worst, n)
            if scores is not None:
                a_scores, b_scores = scores
                a_scores += rng.random(len(a_scores)) * 0.1
                b_scores += rng.random(len(b_scores)) * 0.1
                a_idx = ones[np.argmin(a_scores)]
                b_idx = zeros[np.argmin(b_scores)]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
            else:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
        elif r < 0.70 and len(ones) > 0 and len(zeros) > 0:
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
            h_new[a_idx] = 0.0
            h_new[b_idx] = 1.0
            moved = True
        elif r < 0.85 and len(ones) >= 2 and len(zeros) >= 2:
            block_size = rng.randint(2, max(3, min(12, n // 40)))
            found = False
            for attempt in range(3):
                a_start = ones[rng.randint(len(ones))]
                b_start = zeros[rng.randint(len(zeros))]
                if (a_start + block_size <= n and b_start + block_size <= n and
                    h[a_start:a_start+block_size].all() == 1 and
                    h[b_start:b_start+block_size].all() == 0):
                    h_new[a_start:a_start+block_size] = 0.0
                    h_new[b_start:b_start+block_size] = 1.0
                    moved = True
                    found = True
                    break
            if not found:
                ia = rng.choice(len(ones), 2, replace=False)
                ib = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[ia[0]]] = 0.0
                h_new[ones[ia[1]]] = 0.0
                h_new[zeros[ib[0]]] = 1.0
                h_new[zeros[ib[1]]] = 1.0
                moved = True
        else:
            if len(ones) >= 3 and len(zeros) >= 3:
                n_swap = rng.randint(2, 5)
                n_swap = min(n_swap, len(ones), len(zeros))
                ia = rng.choice(len(ones), n_swap, replace=False)
                ib = rng.choice(len(zeros), n_swap, replace=False)
                for idx in ia:
                    h_new[ones[idx]] = 0.0
                for idx in ib:
                    h_new[zeros[idx]] = 1.0
                moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def greedy_local_search(h_start, target_sum, max_iters, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    stall = 0
    max_stall = 40

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        corr = get_corr(h)
        worst_lags = find_worst_lags(corr, n, top_k=3)
        scores = multi_lag_score(h, ones, zeros, worst_lags, n)
        if scores is None:
            break
        a_scores, b_scores = scores
        a_idx = ones[np.argmin(a_scores)]
        b_idx = zeros[np.argmin(b_scores)]

        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)

        if c5_new < current_c5 - 1e-12:
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1

    return best_h, best_c5

def subgradient_continuous(h_start, target_sum, max_iters, step_init, step_final, rng, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        ratio = i / max(max_iters - 1, 1)
        step = step_init * (step_final / step_init) ** ratio

        corr = get_corr(h)
        center = n - 1
        top_lags = find_worst_lags(corr, n, top_k=5)

        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            g = np.zeros(n)
            for i_idx in range(n):
                val = 0.0
                if 0 <= i_idx - k < n:
                    val += (1.0 - h[i_idx - k])
                if 0 <= i_idx + k < n:
                    val -= h[i_idx + k]
                g[i_idx] = val
            grad += g
        grad /= len(top_lags)

        h_new = h - step * grad * dx
        h_new = project_feasible(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_fun_target = target
    cons_fun_n = n

    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target

    def cons_j(h_in):
        return np.ones(cons_fun_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def generate_sturmian(n, alpha, rng):
    rho = rng.random()
    h = np.zeros(n, dtype=int)
    for i in range(n):
        h[i] = int(np.floor((i + 1) * alpha + rho)) - int(np.floor(i * alpha + rho))
    target = n // 2
    current = h.sum()
    if current != target:
        if current > target:
            ones = np.where(h == 1)[0]
            h[ones[:current - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - current]] = 1
    return h.astype(float)

def generate_perturbation(h, n, noise_scale, rng):
    h_new = h + rng.randn(n) * noise_scale
    return project_feasible(h_new, n // 2.0)

def elapsed_budget(start_time, budget_s):
    return max(0.0, budget_s - 20.0 - (time.time() - start_time))

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)

    # Phase 1: SA with multi-lag targeted moves from initial
    h_sa_curr = h_bin_start.copy()
    sa_configs = [
        (130000, 0.022, 0.00008),
        (140000, 0.012, 0.00003),
        (110000, 0.016, 0.0001),
        (130000, 0.008, 0.00002),
        (100000, 0.005, 0.00001),
    ]

    for ci in range(len(sa_configs)):
        tl = elapsed_budget(start_time, budget_s)
        if tl < 100:
            break
        max_it, t_i, t_f = sa_configs[ci]
        allowed = min(230.0, tl * 0.27)
        h_sa, c5_sa = block_sa(h_sa_curr, target_init, max_it, t_i, t_f,
                               rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        h_sa_curr = h_sa

        tl = elapsed_budget(start_time, budget_s)
        if tl > 50:
            h_gl, c5_gl = greedy_local_search(h_sa, target_init, 6000, rng,
                                              min(35.0, tl * 0.2), start_time)
            if c5_gl < best_score:
                best_score = c5_gl
                best_h = h_gl.copy()
                h_sa_curr = h_gl

    # Phase 2: Structured and diverse initializations
    tl = elapsed_budget(start_time, budget_s)
    if tl > 150:
        starts = []
        sturmian_pairs = [(7, 13), (11, 17), (13, 23), (17, 29), (19, 31),
                          (11, 19), (5, 11), (17, 31), (13, 29), (7, 19)]
        for sn, sd in sturmian_pairs:
            alpha = sn / sd
            if abs(alpha - 0.5) > 0.08:
                h_st = generate_sturmian(n_init, alpha, rng)
                starts.append(h_st)

        for _ in range(4):
            h_rand = np.zeros(n_init)
            perm = rng.permutation(n_init)
            h_rand[perm[:n_init // 2]] = 1.0
            starts.append(h_rand)

        for noise in [0.15, 0.25, 0.35]:
            h_pert = generate_perturbation(h_init, n_init, noise, rng)
            starts.append(h_pert)

        rng.shuffle(starts)

        for h_start in starts:
            tl = elapsed_budget(start_time, budget_s)
            if tl < 60:
                break
            allowed = min(130.0, tl * 0.22)
            max_it = int(min(90000, allowed * 5000))
            h_sa, c5_sa = block_sa(h_start, target_init, max_it, 0.014, 0.00006,
                                   rng, time_limit=allowed, start_time=start_time)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()
            tl = elapsed_budget(start_time, budget_s)
            if tl > 25:
                h_gl, c5_gl = greedy_local_search(h_sa, target_init, 3000, rng,
                                                  min(18.0, tl * 0.15), start_time)
                if c5_gl < best_score:
                    best_score = c5_gl
                    best_h = h_gl.copy()

    # Phase 3: Multi-resolution
    tl = elapsed_budget(start_time, budget_s)
    if tl > 120:
        n_candidates = [400, 500, 700, 800, 900, 1000, 1100]
        n_candidates = [x for x in n_candidates if x != n_init and x >= 200]
        rng.shuffle(n_candidates)

        for n_try in n_candidates:
            tl = elapsed_budget(start_time, budget_s)
            if tl < 60:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin_try = project_binary(h_interp, target_try)

            allowed = min(100.0, tl * 0.25)
            max_it = int(min(60000, allowed * 4000))
            h_sa, c5_sa = block_sa(h_bin_try, target_try, max_it, 0.011, 0.00005,
                                   rng, time_limit=allowed, start_time=start_time)

            tl = elapsed_budget(start_time, budget_s)
            if tl > 20:
                h_gl, c5_gl = greedy_local_search(h_sa, target_try, 3000, rng,
                                                  min(15.0, tl * 0.15), start_time)
                if c5_gl < c5_sa:
                    h_sa, c5_sa = h_gl, c5_gl

            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()

    # Phase 4: Resolution cycling - upsample, SLSQP, downsample, SA
    tl = elapsed_budget(start_time, budget_s)
    if tl > 100:
        n_cycle = int(n_init * 1.5)
        target_cycle = n_cycle / 2.0
        h_up = interpolate_to(best_h, n_cycle, target_cycle)
        alphas_cycle = [30, 80, 200, 600, 1500]
        h_ref, sc = slsqp_refine(h_up, n_cycle, target_cycle, alphas_cycle,
                                  300, 1e-13, time_limit=min(60.0, tl * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

        tl = elapsed_budget(start_time, budget_s)
        if tl > 40:
            h_down = interpolate_to(best_h, n_init, target_init)
            h_bin_down = project_binary(h_down, target_init)
            h_sa, c5_sa = block_sa(h_bin_down, target_init, 50000, 0.008, 0.00003,
                                   rng, time_limit=min(50.0, tl * 0.5), start_time=start_time)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()

    # Phase 5: Subgradient continuous
    n_use = len(best_h)
    target_use = n_use / 2.0
    tl = elapsed_budget(start_time, budget_s)
    if tl > 80:
        allowed = min(100.0, tl * 0.25)
        h_sg, c5_sg = subgradient_continuous(best_h, target_use, 40000,
                                             0.4, 0.008, rng,
                                             time_limit=allowed, start_time=start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    # Phase 6: SLSQP refinement with aggressive smoothing
    n_use = len(best_h)
    target_use = n_use / 2.0
    tl = elapsed_budget(start_time, budget_s)
    if tl > 60:
        alphas = [15, 40, 100, 250, 600, 1500, 3000, 6000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 400, 1e-14,
                                 time_limit=tl - 30, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Try different resolutions with SLSQP
    n_use = len(best_h)
    n_refine_list = [int(n_use * 0.75), int(n_use * 1.3), int(n_use * 1.6)]
    n_refine_list = [x for x in n_refine_list if x >= 200 and abs(x - n_use) > 50]

    for n_try in n_refine_list:
        tl = elapsed_budget(start_time, budget_s)
        if tl < 40:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [50, 120, 300, 800, 2000, 5000]
        h_ref, sc = slsqp_refine(h_interp, n_try, target_try, alphas_try, 300, 1e-13,
                                 time_limit=tl - 25, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final polish - perturbation + refine
    n_use = len(best_h)
    target_use = n_use / 2.0
    fine_attempts = min(6, int(max(2, elapsed_budget(start_time, budget_s) / 30)))

    for fa in range(fine_attempts):
        tl = elapsed_budget(start_time, budget_s)
        if tl < 15:
            break
        noise = rng.choice([0.005, 0.01, 0.015, 0.025])
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        alphas_fine = [500, 1500, 4000, 8000]
        h_ref, sc = slsqp_refine(h_pert, n_use, target_use, alphas_fine, 200, 1e-15,
                                 time_limit=min(18.0, tl - 8), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final binary SA
    tl = elapsed_budget(start_time, budget_s)
    if tl > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(8000, tl * 2500))
        h_sa_f, c5_f = block_sa(h_bin_final, target_use, max_it, 0.004, 0.00001,
                                rng, time_limit=max(5.0, tl * 0.5), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

        tl = elapsed_budget(start_time, budget_s)
        if tl > 12:
            n_use2 = len(best_h)
            target_use2 = n_use2 / 2.0
            h_ref, sc = slsqp_refine(best_h, n_use2, target_use2,
                                     [3000, 8000, 15000, 30000], 150, 1e-15,
                                     time_limit=max(3.0, tl - 5), start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
