import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h).astype(np.float64)
    current_sum = int(np.round(h_r.sum()))
    if current_sum == target_sum:
        return np.clip(h_r, 0.0, 1.0)
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    diff2 = target_sum - int(h_r.sum())
    if diff2 > 0:
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:diff2]] = 1
    elif diff2 < 0:
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:(-diff2)]] = 0
    return h_r

def fast_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(2000, max_iters // 8)

    for i in range(max_iters):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve and rng.random() > 0.03:
            break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]

        old_i1, old_i0 = h[i1], h[i0]
        h[i1] = 0.0
        h[i0] = 1.0

        c5_new = compute_c5(h)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-max(delta, 0.0) / t)):
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i1] = old_i1
            h[i0] = old_i0
            no_improve += 1

    return best_h, best_c5

def lse_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return val * (2.0 / n)

def lse_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(n)
    weights /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w[:n - mm] * 0.0
            grad[mm:] -= w * h[:n - mm]
    return grad

def cons_fun(x):
    return np.sum(x) - _cons_target

def cons_jac(x):
    return np.ones(_cons_n)

def slsqp_optimize(h_start, target, alphas, time_limit, start_time, maxiter_per_call=200):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = float(target)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    for alpha in alphas:
        if (time.time() - start_time) > time_limit:
            break
        for attempt in range(2):
            if (time.time() - start_time) > time_limit:
                break
            try:
                h_init_call = h.copy()
                if attempt > 0:
                    h_init_call = h_init_call + 0.01 * np.random.RandomState(int(alpha) * 7 + attempt).randn(n)
                    h_init_call = project_feasible(h_init_call, target)
                res = minimize(
                    lse_obj, h_init_call, args=(alpha, n, target),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter_per_call, 'ftol': 1e-18, 'iprint': 0}
                )
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass
    return best_h, best_c5

def nesterov_with_snap(h_start, target, alpha, time_limit, start_time, momentum=0.85):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(int(alpha))

    grad0 = lse_grad(h, alpha, n, target)
    gnorm0 = np.linalg.norm(grad0)
    if gnorm0 < 1e-15:
        return best_h, best_c5
    lr = 0.3 / (alpha * dx * 5.0 + 1e-8)
    lr = min(lr, 2.0 / n)

    max_iters = 20000
    snap_every = 500

    for i in range(max_iters):
        if (time.time() - start_time) > time_limit:
            break

        y = project_feasible(h + momentum * v, target)
        grad_y = lse_grad(y, alpha, n, target)
        gn = np.linalg.norm(grad_y)
        if gn < 1e-14:
            break
        v = momentum * v - lr * grad_y / gn
        h_new = project_feasible(y + v, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new

        if (i + 1) % snap_every == 0:
            h_bin = project_binary(h, target)
            h_sa, c5_sa = fast_sa(h_bin, target, 2000, 0.005, 0.0001, rng,
                                   min(5.0, (time_limit - (time.time() - start_time)) * 0.1),
                                   start_time)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h = project_feasible(h_sa, target)
                y = h.copy()
                v = np.zeros(n)

        if (i + 1) % 200 == 0:
            obj_now = np.max(compute_corr_scaled(h))
            if obj_now > best_c5 + 0.05:
                h = best_h.copy()
                y = h.copy()
                v = np.zeros(n)
                lr *= 0.5

    return best_h, best_c5

def active_subgradient(h, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    threshold = c_max * 0.94
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += (1.0 - h[:n - m]) * dx
            grad[:n - m] -= h[m:] * dx
        else:
            mm = -m
            grad[:n - mm] += (1.0 - h[mm:]) * dx
            grad[mm:] -= h[:n - mm] * dx
    return grad / len(active)

def subgradient_optimize(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    no_improve = 0

    for i in range(50000):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > 400:
            break

        grad = active_subgradient(h, n, target)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn

        for step_frac in [1.0, 0.3, 0.1]:
            h_try = project_feasible(h - lr * step_frac * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                no_improve = 0
                lr = min(lr * 1.05, 0.5)
                break
        else:
            no_improve += 1
            lr *= 0.92
            if lr < 1e-7:
                break

    return best_h, best_c5

def fix_and_optimize(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    free_tol = 0.05

    free = np.where((h > free_tol) & (h < 1.0 - free_tol))[0]
    if len(free) == 0 or len(free) > n // 3:
        return best_h, best_c5

    fixed_mask = np.ones(n, dtype=bool)
    fixed_mask[free] = False
    fixed_vals = h[fixed_mask].copy()

    _cons_n = len(free)
    _cons_target = float(target - np.sum(fixed_vals))
    bounds = [(0.0, 1.0)] * len(free)

    def wrap_obj(x_free):
        x_full = np.zeros(n)
        x_full[fixed_mask] = fixed_vals
        x_full[free] = x_free
        return lse_obj(x_full, 2000, n, target)

    def wrap_cons_fun(x_free):
        return np.sum(x_free) - _cons_target

    def wrap_cons_jac(x_free):
        return np.ones(len(free))

    cons = [{'type': 'eq', 'fun': wrap_cons_fun, 'jac': wrap_cons_jac}]

    try:
        res = minimize(wrap_obj, h[free].copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-18, 'iprint': 0})
        h_full = h.copy()
        h_full[free] = res.x
        h_full = project_feasible(h_full, target)
        c5_new = compute_c5(h_full)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_full.copy()
    except Exception:
        pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
