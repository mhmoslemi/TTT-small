    best_c5 = compute_c5(h)

        if time.time() - start > time_limit:
            break

        bundle.append((h.copy(), c5_val, k_star, grad))

    dx = 2.0 / n

    best_h = h.copy()

        k_star = np.argmax(corr) - (n - 1)
        grad = compute_subgradient(h, k_star, n, dx)
        
        # Backtracking line search
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:

            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                lr = min(lr * 1.1, 0.1)
                break
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_h, best_c5

def active_set_lp(h, target, top_k_list, n):
    """Solve LP: min t s.t. sum h'_i * g_{i+k} * dx <= t for k in top_k_list,
       0<=h'<=1, sum(h')=target, where g = 1-h is fixed."""
    dx = 2.0 / n
    g = 1.0 - h
    
    # c = [0, 0, ..., 0, 1] (minimize t, which is variable n)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k in top_k_list:
    #   sum_i h'_i * g_{i+k} * dx <= t
    #   => sum_i h'_i * g_{i+k} * dx - t <= 0
    #   => [g_0*dx, g_1*dx, ..., g_{n-1}*dx, -1] @ [h'_0, ..., h'_{n-1}, t] <= 0
    
    A_rows = []
    b_rows = []
    for k in top_k_list:
        row = np.zeros(n + 1)
        # For lag k, the valid indices i are those with 0<=i<n and 0<=i+k<n
        # In the correlate convention: corr[k] = sum_i h[i] * (1-h[i+k])
        # where i ranges over valid indices
        # g[i+k] = 1-h[i+k]
        # The contribution to the constraint is h'_i * g[i+k] * dx
        lo = max(0, -k)  # i >= 0 and i >= -k
        hi = min(n, n - k)  # i < n and i+k < n => i < n-k
        for i in range(lo, hi):
            row[i] = g[i + k] * dx
        row[-1] = -1.0  # -t
        A_rows.append(row)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.array(b_rows) if b_rows else None
    
    # Equality: sum(h') = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= h'_i <= 1, t is free (or >= 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None

def compute_c5(h):

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

            valid2 = idxs >= k_star
        else:
            valid1 = idxs + k_star >= 0
            grad[valid1] += (1.0 - h[idxs[valid1] + k_star]) * dx
            valid2 = idxs + k_star < n
        grad[valid2] -= h[idxs[valid2] - k_star] * dx

        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = h - trial_lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                lr = min(lr * 1.05, 0.15)
                improved = True
                break
        if not improved:
            lr *= 0.7
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Fast active-set LP
    if tleft() > 150:
        t1 = min(200, tleft() * 0.28)
        h1, c5_1 = active_set_lp_phase(h, n, target, max_iter=50000, time_limit=t1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Subgradient refinement
    if tleft() > 80:
        h2, c5_2 = subgradient_phase(best_h, n, target, max_steps=50000, time_limit=min(60, tleft() * 0.15))
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: SLSQP moderate alpha
    if tleft() > 150:
        alphas = [50, 200, 800]
        h_cur = best_h.copy()
        for alpha in alphas:
            if tleft() < 50:
                break
            h_try = slsqp_optimize(h_cur, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try
            h_cur = h_try

    # Phase 4: More active-set LP
    if tleft() > 100:
        t4 = min(100, tleft() * 0.2)
        h4, c5_4 = active_set_lp_phase(best_h, n, target, max_iter=30000, time_limit=t4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Perturbation + SLSQP
    if tleft() > 120:
        for noise in [0.03, 0.06, 0.1, 0.15]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
            h_try = slsqp_optimize(h_pert, n, target, 300, maxiter=800)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try
            if tleft() > 40:
                h_lp, c5_lp = active_set_lp_phase(h_pert, n, target, max_iter=5000, time_limit=min(40, tleft() * 0.3))
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp

    # Phase 6: SLSQP high alpha
    if tleft() > 100:
        alphas2 = [2000, 10000]
        h_cur = best_h.copy()
        for alpha in alphas2:
            if tleft() < 40:
                break
            h_try = slsqp_optimize(h_cur, n, target, alpha, maxiter=2000)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try
            h_cur = h_try

    # Phase 7: Fine-tune with tiny perturbations + subgradient
    if tleft() > 60:
        for noise in [0.005, 0.01, 0.02]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
            h_sg, c5_sg = subgradient_phase(h_pert, n, target, max_steps=20000, time_limit=min(20, tleft() * 0.3))
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg

    # Phase 8: Final SLSQP polish
    if tleft() > 35:
        h_try = slsqp_optimize(best_h, n, target, 20000, maxiter=3000)
        sc = compute_c5(h_try)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_try

    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
