import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_one(h, target, alpha, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    def cfun(x):
        return np.sum(x) - target
    def cjac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def binary_push(h, target, n_iter, push_strength, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        # Compute subgradient of max
        active = np.where(corr >= c_max - 1e-8)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n-k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n-k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n-kk]
                    grad[:n-kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        
        # Gradient step with decreasing step size
        step = 0.1 / (1.0 + 0.01 * it)
        h_new = h_cur - step * grad
        
        # Binary push: amplify deviation from 0.5
        h_new = 0.5 + push_strength * (h_new - 0.5)
        
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h_cur = h_new
    
    return best_h, best_score

def local_search(h, target, n_iter, delta, deadline, rng):

    h_cur = h.copy()

def subgrad_bb(h, target, n_steps, deadline):
    """Subgradient with Barzilai-Borwein step size."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    h_prev = h_cur.copy()
    grad_prev = None
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        
        # Use top 5% active set
        sorted_idx = np.argsort(corr)[::-1]
        n_active = max(1, len(sorted_idx) // 20)
        active = sorted_idx[:n_active]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n-k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n-k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n-kk]
                    grad[:n-kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        
        # BB step size
        if grad_prev is not None:
            y = grad - grad_prev
            s = h_cur - h_prev
            yy = np.dot(y, y)
            sy = np.dot(s, y)
            if yy > 1e-16 and abs(sy) > 1e-16:
                step = sy / yy
                step = np.clip(step, 0.001, 0.5)
            else:
                step = 0.01
        else:
            step = 0.05
        
        h_new = project_feasible(h_cur - step * grad, target)
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h_prev = h_cur.copy()
        grad_prev = grad.copy()
        h_cur = h_new
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def exact_subgrad(h, frac=0.05):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    threshold = c_max - frac * (c_max - float(np.min(corr)))
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= h[k:]
                grad[k:] += (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= h[:n - kk]
                grad[:n - kk] += (1.0 - h[kk:])
    grad /= len(active)
    return grad


def slsqp_solve(h, target, alpha, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    def cfun(x):
        return np.sum(x) - target
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h
