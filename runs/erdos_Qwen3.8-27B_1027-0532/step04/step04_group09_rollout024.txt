```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
        # Binary search on lambda
        def f(lam):
            return np.sum(np.clip(h + lam, 0.0, 1.0)) - target_sum
        # Initial bounds
        lo, hi = -max(h, default=0) - 1, max(1 - h, default=0) + 1
        for _ in range(60):
            mid = (lo + hi) / 2
            if f(mid) > 0:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        mask = h > 0 if diff < 0 else h < 1
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
def lse_gradient_fft(h, alpha):
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    
    # W_pos[d] = w[n-1+d] for d = 0, ..., n-1
    W_pos = w[n-1:]  # length n
    # W_neg[d] = w[n-1-d] for d = 0, ..., n-1
    W_neg = w[n-1::-1]  # length n, W_neg[0]=w[n-1], W_neg[1]=w[n-2], ...
    
    # Compute convolutions via FFT
    # Z1[j] = sum_{d=0}^{j} W_pos[d] * h[j-d] = (h * W_pos)[j]
    # Z4[j] = sum_{d=1}^{j} W_neg[d] * h[j-d] = (h * W_neg)[j] - W_neg[0]*h[j]
    
    # For anti-convolutions:
    # Z2[j] = sum_{d=0}^{n-1-j} W_pos[d] * h[j+d]
    #        = (W_pos * h_rev)[n-1-j] where h_rev[i] = h[n-1-i]
    # Z3[j] = sum_{d=1}^{n-1-j} W_neg[d] * h[j+d]
    #        = (W_neg * h_rev)[n-1-j] - W_neg[0]*h[j]
    
    h_rev = h[::-1]
    
    # FFT size
    fft_size = 2 * n
    H = np.fft.rfft(h, fft_size)
    H_rev = np.fft.rfft(h_rev, fft_size)
    Wp = np.fft.rfft(W_pos, fft_size)
    Wn = np.fft.rfft(W_neg, fft_size)
    
    conv_h_Wp = np.fft.irfft(H * Wp, fft_size)[:n]  # (h * W_pos)[:n]
    conv_h_Wn = np.fft.irfft(H * Wn, fft_size)[:n]  # (h * W_neg)[:n]
    conv_Wp_Hrev = np.fft.irfft(Wp * H_rev, fft_size)  # (W_pos * h_rev)
    conv_Wn_Hrev = np.fft.irfft(Wn * H_rev, fft_size)  # (W_neg * h_rev)
    
    Z1 = conv_h_Wp
    Z4 = conv_h_Wn - W_neg[0] * h
    Z2 = conv_Wp_Hrev[n-1::-1]  # index n-1-j for j=0..n-1 means reverse
    Z3 = conv_Wn_Hrev[n-1::-1] - W_neg[0] * h
    
    # Prefix sums
    P_pos = np.cumsum(W_pos)  # P_pos[j] = sum_{d=0}^{j} W_pos[d]
    P_neg = np.zeros(n)
    P_neg[0] = 0
    P_neg[1:] = np.cumsum(W_neg[1:])  # P_neg[j] = sum_{d=1}^{j} W_neg[d]
    
    # grad[j] = P_pos[j] + P_neg[n-1-j] - Z1[j] - Z2[j] - Z3[j] - Z4[j]
    grad = P_pos + P_neg[::-1] - Z1 - Z2 - Z3 - Z4
    
    return grad
```

```python
import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        # Binary search for lambda
        lo = -1.0
        hi = 1.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s > target_sum:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Correct residual
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_grad_fft(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    
    W_pos = w[n-1:]
    W_neg = w[n-1::-1]
    
    h_rev = h[::-1]
    
    fft_size = 2 * n
    H = rfft(h, fft_size)
    H_rev = rfft(h_rev, fft_size)
    Wp = rfft(W_pos, fft_size)
    Wn = rfft(W_neg, fft_size)
    
    conv_h_Wp = irfft(H * Wp, fft_size)[:n]
    conv_h_Wn = irfft(H * Wn, fft_size)[:n]
    conv_Wp_Hrev = irfft(Wp * H_rev, fft_size)
    conv_Wn_Hrev = irfft(Wn * H_rev, fft_size)
    
    Z1 = conv_h_Wp
    Z4 = conv_h_Wn - W_neg[0] * h
    Z2 = conv_Wp_Hrev[n-1::-1]
    Z3 = conv_Wn_Hrev[n-1::-1] - W_neg[0] * h
    
    P_pos = np.cumsum(W_pos)
    P_neg = np.zeros(n)
    if n > 1:
        P_neg[1:] = np.cumsum(W_neg[1:])
    
    grad = P_pos + P_neg[::-1] - Z1 - Z2 - Z3 - Z4
    return grad, c_max + np.log(sum_exp) / alpha


def fista_lse(h_init, target_sum, alphas, iters_per_alpha, deadline):
    n = len(h_init)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_prev = h_cur.copy()
        t = 1
        step = 1.0 / (alpha * 2.0 + 1.0)
        
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            grad, _ = lse_grad_fft(h_cur, alpha)
            # Nesterov momentum
            if t > 1:
                momentum = (t - 1.0) / (t + 2.0)
            else:
                momentum = 0.0
            y = h_cur + momentum * (h_cur - h_prev)
            h_new = project_feasible(y - step * grad, target_sum)
            h_prev = h_cur
            h_cur = h_new
            t += 1
        
        score = compute_c5(h_cur)
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
    
    return best_h, best_score
```

Now let me think about the active-set polishing more carefully.

After FISTA with large alpha, the solution is good but might not be optimal. The active-set method works as follows:
1. Compute corr = correlate(h, 1-h, 'full')
2. Find the k* that achieves the maximum
3. Compute the subgradient of corr[k*] (or the average of top-m subgradients)
4. Take a step in the negative subgradient direction
5. Project

The gradient of corr[k] w.r.t. h[j]:
If lag = k - (n-1) >= 0:
  d(corr[k])/dh[j] = (1-h[j-lag])[j>=lag] - h[j+lag][j+lag<n]
If lag < 0, llag = -lag:
  d(corr[k])/dh[j] = (1-h[j+llag])[j+llag<n] - h[j-llag][j>=llag]

For a single k, this is O(n) to compute. For m active k's, it's O(m*n).

```python
def compute_single_corr_grad(h, k_full_index):
    """Compute gradient of corr[k] w.r.t. h."""
    n = len(h)
    lag = k_full_index - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n-lag])  # (1-h[j-lag]) for j >= lag
            grad[:n-lag] -= h[lag:]  # -h[j+lag] for j+lag < n
    else:
        llag = -lag
        if llag < n:
            grad[:n-llag] += (1.0 - h[llag:])  # (1-h[j+llag]) for j+llag < n
            grad[llag:] -= h[:n-llag]  # -h[j-llag] for j >= llag
    return grad
```

```python
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        lo, hi = -1.0, 1.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s > target_sum:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_grad_fft(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    
    W_pos = w[n-1:]
    W_neg = w[n-1::-1]
    h_rev = h[::-1]
    
    fft_size = 2 * n
    H = rfft(h, fft_size)
    H_rev = rfft(h_rev, fft_size)
    Wp = rfft(W_pos, fft_size)
    Wn = rfft(W_neg, fft_size)
    
    Z1 = irfft(H * Wp, fft_size)[:n]
    Z4 = irfft(H * Wn, fft_size)[:n] - W_neg[0] * h
    Z2 = irfft(Wp * H_rev, fft_size)[n-1::-1]
    Z3 = irfft(Wn * H_rev, fft_size)[n-1::-1] - W_neg[0] * h
    
    P_pos = np.cumsum(W_pos)
    P_neg = np.zeros(n)
    if n > 1:
        P_neg[1:] = np.cumsum(W_neg[1:])
    
    grad = P_pos + P_neg[::-1] - Z1 - Z2 - Z3 - Z4
    obj = c_max + np.log(sum_exp) / alpha
    return grad, obj
```

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        lo, hi = -1.0, 1.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s > target_sum:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_grad_fft(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    W_pos = w[n - 1:]
    W_neg = w[n - 1::-1]
    h_rev = h[::-1]
    fft_size = 2 * n
    H = rfft(h, fft_size)
    H_rev = rfft(h_rev, fft_size)
    Wp = rfft(W_pos, fft_size)
    Wn = rfft(W_neg, fft_size)
    Z1 = irfft(H * Wp, fft_size)[:n]
    Z4 = irfft(H * Wn, fft_size)[:n] - W_neg[0] * h
    Z2 = irfft(Wp * H_rev, fft_size)[n - 1::-1]
    Z3 = irfft(Wn * H_rev, fft_size)[n - 1::-1] - W_neg[0] * h
    P_pos = np.cumsum(W_pos)
    P_neg = np.zeros(n)
    if n > 1:
        P_neg[1:] = np.cumsum(W_neg[1:])
    grad = P_pos + P_neg[::-1] - Z1 - Z2 - Z3 - Z4
    obj = c_max + np.log(sum_exp) / alpha
    return grad, obj


def fista_optimize(h_init, target_sum, alpha, max_iters, deadline):
    n = len(h_init)
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    t = 1.0
    step = 0.5
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, _ = lse_grad_fft(h_cur, alpha)
        momentum = (t - 1.0) / (t + 2.0) if t > 1.0 else 0.0
        y = h_cur + momentum * (h_cur - h_prev)
        h_new = project_feasible(y - step * grad, target_sum)
        h_prev = h_cur
        h_cur = h_new
        t += 1.0
        if (it + 1) % 200 == 0:
            c5_val = compute_c5(h_cur)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_cur.copy()
    c5_val = compute_c5(h_cur)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_cur.copy()
    return best_h
