def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)

            h_new = project_feasible(h_new, target)
            
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                last_improve = it
            
            # Adaptive step
            if it % 100 == 0 and it > 100:
                step = min(step * 1.1, 2.0 / (2.0 * dx))
            
            h = h_new
            t = t_new
    
    return best_h, best_c5

def slsqp_with_grad(h_init, n, target, p, maxiter, ftol=1e-18):
    """SLSQP with analytic gradient for power-smoothed objective."""
    dx = 2.0 / n
    
    def obj_func(h):
        return smooth_obj_p(h, p)
    
    def obj_grad(h):
        return power_grad(h, p)
    
    def cons_eq(h):
        return np.sum(h) - target
    
    def cons_eq_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    
    try:
        res = minimize(obj_func, h_init, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_init
