from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            per = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            per = diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] + per, 0.0)
    return np.clip(h, 0.0, 1.0)

def compute_lag_grad(h, lag):
    """Gradient of corr_lag with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if 0 <= lag < n:
        # corr = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        a = h[lag:]
        b = 1.0 - h[:n-lag]
        grad[lag:] += b
        grad[:n-lag] -= a
    elif -n < lag < 0:
        m = -lag
        # corr = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        a = h[:n-m]
        b = 1.0 - h[m:]
        grad[:n-m] += b
        grad[m:] -= a
    return grad

    n = len(h)
    if 0 <= lag < n:
        return np.dot(h[lag:], 1.0 - h[:n-lag])
    elif -n < lag < 0:
        m = -lag
        return np.dot(h[:n-m], 1.0 - h[m:])

    shifted = alpha * (top_vals - c_ref)
    shifted = np.clip(shifted, -500.0, 0.0)
    obj = (c_ref + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / len(h))
    
    exp_shifted = np.exp(shifted)

def subgradient_descent(h_init, target_sum, max_iter, lr_init, topk, alpha,
                        noise_prob, noise_scale, patience, momentum, seed, time_limit=None):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    t0 = time.time()
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for i, idx in enumerate(top_indices):
            if w[i] < 1e-14:
                continue
            lag = idx - (n - 1)
            grad += w[i] * compute_lag_grad(h, lag)
        
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 5.0)
        else:
            stagnation += 1
            lr *= 0.995
        
        h = h_new
        if stagnation > patience:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            h[mask] = np.maximum(h[mask] + per, 0.0)
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if 0 <= lag < n:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    elif -n < lag < 0:
        m = -lag
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def smooth_max_obj(h, alpha, topk):
    corr = overlap_corrs(h)
    n_corr = len(corr)
    if topk <= n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    top_vals = corr[top_idx]
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha, topk):
    corr = overlap_corrs(h)
    n_corr = len(corr)

    weights = exp_s / total
    grad = np.zeros(len(h))
    for i in range(len(top_idx)):

        g = smooth_max_grad(x, alpha, topk) / len(x) / 2.0
        return val, g
    
    # Actually, let's not scale - just minimize the raw smooth max

                   args=(alpha, topk), method='L-BFGS-B',

                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})

        else:

            lag = idx - (n - 1)
            g = lag_grad(h, lag)
            A[i, :n] = -g  # we want g.h <= t - (g.h_cur - corr[idx])
            # Actually: corr_idx(h) ≈ corr[idx] + g . (h - h_cur)
            # = corr[idx] + g.h - g.h_cur
            # We want: corr[idx] + g.h - g.h_cur <= t
            # i.e.: g.h - t <= g.h_cur - corr[idx]
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[idx]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if not res.success:
            break
        h_lp = project_feasible(res.x[:n], target_sum)
        score = compute_c5(h_lp)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stag = 0
        else:
            h_half = project_feasible(0.7 * h + 0.3 * h_lp, target_sum)
            sc = compute_c5(h_half)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_half.copy()
                h = h_half
                stag = 0
            else:
                stag += 1
                if stag > 20:
                    break
    return best_h, best_score

def subgrad(h_init, target_sum, max_iter, lr, topk, alpha, noise_p, noise_s, patience, momentum, seed, time_limit=None):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stag = 0
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        corr = overlap_corrs(h)
        nc = len(corr)
        if topk < nc:
            ti = np.argpartition(corr, -topk)[-topk:]
        else:
            ti = np.arange(nc)
        tv = corr[ti]
        w = np.exp(alpha * (tv - np.max(tv)))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(ti)) / len(ti)
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(len(ti)):
            if w[i] < 1e-12:
                continue
            lag = ti[i] - (n - 1)
            grad += w[i] * lag_grad(h, lag)
        v = momentum * v + (1 - momentum) * grad
        h_new = project_feasible(h - lr * v, target_sum)
        if noise_p > 0 and rng.random() < noise_p:
            h_new = project_feasible(h_new + rng.randn(n) * noise_s, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stag = 0
            lr = min(lr * 1.005, 5.0)
        else:
            stag += 1
            lr *= 0.997
        h = h_new
        if stag > patience:
            break
    return best_h, best_score

def interp(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    xo = np.linspace(0, 1, len(h_src))
    xn = np.linspace(0, 1, n_dst)
    return project_feasible(np.interp(xn, xo, h_src), target_dst)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            h[mask] = np.maximum(h[mask] + per, 0.0)
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if 0 <= lag < n:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    elif -n < lag < 0:
        m = -lag
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad
