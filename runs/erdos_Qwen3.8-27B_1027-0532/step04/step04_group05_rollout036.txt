- $0 \leq h_i \leq 1$

Wait, the correct first-order Taylor expansion of $f(h) = \sum_i h_i h_{i+k}$ at $h_0$ is:

    if abs(diff) < 1e-14:

    # k_star is the shift (can be negative)
    dx = 2.0 / n
    grad = np.zeros(n)

        if 0 <= i + k_star < n:

Actually, I can do even better. Instead of looping over all shifts, I can accumulate the gradient more efficiently.

For shift k >= 0:
- grad[0:n-k] += w_k * dx * (1 - h[k:n])
- grad[k:n] -= w_k * dx * h[0:n-k]

For all positive shifts k=0,1,...,n-1 with weights w_k:
- grad[i] receives: sum over k where i < n-k (i.e., k < n-i) of w_k * dx * (1-h[i+k])
- minus: sum over k where i >= k (i.e., k <= i) of w_k * dx * h[i-k]

This is a convolution! Let me use FFT:
- The first term is a convolution of w with (1-h)
- The second term is a cross-correlation of w with h

Let me define:
w_pos[k] = weight for shift k (k >= 0)
w_neg[k] = weight for shift -k (k > 0)

Then:
grad[i] = dx * (conv(w_pos, 1-h)[i] - cross_corr(w_pos, h)[i]) + ... (similar for negative shifts)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]

            h[idx2] += residual / len(idx2)

def subgradient_descent(h_init, n, target, max_steps, deadline, multi_top=5):
    """Subgradient descent using weighted average of top multi_top shifts"""
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_scale = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        
        # Use top few shifts for the subgradient
        top_idx = np.argsort(corr)[-multi_top:]
        # Weights: exponential of (corr - c_max) * beta
        beta = 5.0  # temperature parameter
        shifted = beta * (corr[top_idx] - c_max)
        w = np.exp(shifted)
        w = w / np.sum(w)
        
        grad = np.zeros(n)
        for wt, k_idx in zip(w, top_idx):
            k = int(k_idx) - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[0:n-k] += wt * dx * (1.0 - h[k:n])
                    grad[k:n] -= wt * dx * h[0:n-k]
            else:
                kk = -k
                if n - kk > 0:
                    grad[kk:n] += wt * dx * (1.0 - h[0:n-kk])
                    grad[0:n-kk] -= wt * dx * h[kk:n]
        
        # Nesterov momentum step
        v = momentum * v - step_scale * grad
        h_new = h - v
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # still update h even if not improving best
        
        # Adaptive step size
        if c5_new > best_c5 * 1.001:
            step_scale *= 0.9
        step_scale = max(step_scale, 0.001)
    
    return best_h, best_c5

def active_set_qp(h_init, n, target, max_iter, deadline, threshold=0.003):
    """Active-set QP: solve proximal QP with linearized constraints"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr_vals)

        active_mask = corr_vals >= (c_max - threshold)
        active_indices = np.where(active_mask)[0]
        active_shifts = active_indices - (n - 1)
        
        m = len(active_shifts)
        if m == 0:
            k_star = int(np.argmax(corr_vals)) - (n - 1)
            active_shifts = [k_star]
            m = 1

        # For shift k: acf_k(h) >= mu where mu = 1/c_max_target... 
        # Actually, we want: corr_k(h) <= c_max for all k

        # Or: acf_k(h) >= (1 - c_max) / dx  ... hmm this isn't quite right

        # We want to minimize max_k corr_k(h)
        # At the current point, the active shifts have corr_k ≈ c_max
        # We want to find h' such that corr_k(h') <= c_max - delta for all k
        # Linearize: corr_k(h') ≈ corr_k(h) + grad_corr_k . (h' - h)
        # Constraint: corr_k(h) + grad_corr_k . h' - grad_corr_k . h <= c_max - delta
        # i.e., grad_corr_k . h' <= c_max - delta - corr_k(h) + grad_corr_k . h

        # dx * [(1-h_{j+k}) if valid - h_{j-k} if valid]

            if k >= 0:
                if n - k > 0:

                    g[k:n] -= h[0:n-k] * dx
            else:
                kk = -k
                if n - kk > 0:
                    g[kk:n] = (1.0 - h[0:n-kk]) * dx
                    g[0:n-kk] -= h[kk:n] * dx
            
            A[idx, :] = g
            b_vec[idx] = c_max - delta - float(corr_vals[idx + (n-1)]) + float(np.dot(g, h))
            # Wait, I need the index of this shift in corr_vals
            # corr_vals has indices 0 to 2n-2, shift k corresponds to index n-1+k
        
        # Fix the indexing: for shift k, the index in corr_vals is n-1+k
        # Let me redo:
        for idx, k in enumerate(active_shifts):
            k_val = int(k)
            corr_idx = (n - 1) + k_val
            if 0 <= corr_idx < len(corr_vals):
                curr_val = corr_vals[corr_idx]
            else:
                curr_val = c_max
            
            g = np.zeros(n)
            if k_val >= 0:
                if n - k_val > 0:

            else:

                break

            h = project_feasible(h, target)

def interpolate_to(h_src, n_dst, target_dst):
