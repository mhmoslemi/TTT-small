import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def kronecker_seq(n, alpha, density=0.5):
    h = np.zeros(n)
    for i in range(n):
        if (i * alpha) % 1.0 < density:
            h[i] = 1.0
    n_ones = int(h.sum())
    target = n // 2
    if n_ones < target:
        idx = np.where(h == 0.0)[0]
        for pos in idx[:target - n_ones]:
            h[pos] = 1.0
    elif n_ones > target:
        idx = np.where(h == 1.0)[0]
        for pos in idx[n_ones - target:]:
            h[pos] = 0.0
    return h

    if n_ones < target:
        idx = np.where(h == 0.0)[0]
        for pos in idx[:target - n_ones]:
            h[pos] = 1.0
    elif n_ones > target:
        idx = np.where(h == 1.0)[0]
        for pos in idx[n_ones - target:]:
            h[pos] = 0.0
    return h

def encode_binary(h):
    n = len(h)
    target = n // 2
    h = h.copy()
    n_ones = int((h > 0.5).sum())
    h_bin = (h > 0.5).astype(float)
    if n_ones > target:
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[::-1]
        for pos in order[target:]:
            h_bin[ones_idx[pos]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])
        for pos in order[:target - n_ones]:
            h_bin[zeros_idx[pos]] = 1.0
    return h_bin

def binary_sa(h, deadline, rng, n_iters, T_init, T_end, directed_frac=0.4):
    h = h.copy().astype(float)
    n = len(h)
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    corr = compute_corr(h)
    max_corr = float(np.max(corr))
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T_init * (T_end / T_init) ** (it / max(float(n_iters), 1.0))

        if directed_frac > 0 and it % 10 == 0 and max_corr > 0:
            threshold = 0.85 * max_corr
            active = np.where(corr > threshold)[0]
            hot_ones = set()
            hot_zeros = set()
            for k_idx in active:
                lag = int(k_idx) - (n - 1)
                if lag >= 0:
                    for i in range(n - lag):
                        if h[i] == 1.0 and h[i + lag] == 0.0:
                            hot_ones.add(i)
                            hot_zeros.add(i + lag)
                else:
                    t = -lag
                    for i in range(t, n):
                        if h[i] == 1.0 and h[i - t] == 0.0:
                            hot_ones.add(i)
                            hot_zeros.add(i - t)
            if len(hot_ones) > 0 and len(hot_zeros) > 0:
                i = list(hot_ones)[rng.randint(len(hot_ones))]
                j = list(hot_zeros)[rng.randint(len(hot_zeros))]
                if h[i] != 1.0 or h[j] != 0.0:
                    i = ones_list[rng.randint(len(ones_list))]
                    j = zeros_list[rng.randint(len(zeros_list))]
            else:
                i = ones_list[rng.randint(len(ones_list))]
                j = zeros_list[rng.randint(len(zeros_list))]
        else:
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]

        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)

        delta = new_score - cur_score
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                corr = compute_corr(h)
                max_corr = float(np.max(corr))
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_score

def grad_soft_max(h, alpha, topk=None):
    n = len(h)
    corr = compute_corr(h)
    if topk is not None and topk < len(corr):
        idx = np.argsort(-corr)[:topk]
        c_max = corr[idx[0]]
    else:
        c_max = float(np.max(corr))
        idx = np.arange(len(corr))
    shifted = np.full(len(corr), -30.0)
    shifted[idx] = np.clip(alpha * (corr[idx] - c_max), -30.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n), c_max
    wt = wt / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = wt[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            t = -lag
            grad[t:] += w * (1.0 - h[:n - t])
            grad[:n - t] -= w * h[t:]
    return grad, c_max

def continuous_refine(h, target, deadline, topk=None):
    n = len(h)
    best_h = project_feasible(h, target)
    best_score = compute_c5(best_h)
    h_cur = best_h.copy()

    for alpha in [500.0, 2000.0, 8000.0, 30000.0]:
        if time.time() > deadline:
            break
        lr = 0.05
        stall = 0
        for step in range(5000):
            if time.time() > deadline:
                break
            grad, _ = grad_soft_max(h_cur, alpha, topk=topk)
            h_new = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = h_new
                h_cur = h_new
                lr = min(lr * 1.05, 0.5)
                stall = 0
            else:
                lr *= 0.7
                stall += 1
                if stall > 40 or lr < 1e-14:
                    break
    return best_h, best_score

def nelder_block(h, target, deadline, rng, block_size=4, n_rounds=3):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    if n_blocks == 0:
        return h_cur, cur_score

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        order = list(range(n_blocks))
        rng.shuffle(order)
        for bi in order:
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size

            def _block_obj(x, s=s, e=e, h_ref=h_cur):
                h_full = h_ref.copy()
                h_full[s:e] = np.clip(x, 0.0, 1.0)
                h_full = project_feasible(h_full, target)
                return compute_c5(h_full)

            x0 = h_cur[s:e].copy()
            try:
                res = minimize(_block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_test
            except Exception:
                pass
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    n_ones = n // 2

    best_h = None
    best_score = float('inf')
    best_n = n

    # Phase 1: Binary SA at original resolution
    phase1_deadline = start_time + budget_s * 0.45

    # Generate candidates
    candidates = []
    h_bin_init = encode_binary(h_init)
    candidates.append(h_bin_init)
    alpha_gold = (np.sqrt(5.0) - 1.0) / 2.0
    candidates.append(kronecker_seq(n, alpha_gold, 0.5))
    candidates.append(kronecker_seq(n, np.sqrt(2.0) - 1.0, 0.5))
    candidates.append(thue_morse_seq(n))
    for _ in range(3):
        h_r = np.zeros(n)
        h_r[rng.choice(n, size=n_ones, replace=False)] = 1.0
        candidates.append(h_r)

    # Run SA on each candidate
    n_cands = len(candidates)
    per_cand_time = (phase1_deadline - start_time) / n_cands
    for ci, h_cand in enumerate(candidates):
        dl = start_time + (ci + 1) * per_cand_time
        if time.time() > dl - 5:
            break
        h_sa, sc = binary_sa(h_cand, dl, rng,
                             n_iters=500000, T_init=0.015, T_end=0.0005,
                             directed_frac=0.3)
        if sc < best_score:
            best_score = sc
            best_h = h_sa.copy()
            best_n = n

    # Phase 2: Continuous refinement at original resolution
    if remaining() > 100:
        cont_deadline = start_time + budget_s * 0.65
        h_ref, sc_ref = continuous_refine(best_h, target, cont_deadline)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref
            best_n = n

    # Phase 3: Nelder-Mead block polish
    if remaining() > 60:
        nm_deadline = start_time + budget_s * 0.78
        h_nm, sc_nm = nelder_block(best_h, target, nm_deadline, rng,
                                   block_size=4, n_rounds=4)
        if sc_nm < best_score:
            best_score = sc_nm
            best_h = h_nm
            best_n = n

    # Phase 4: Higher resolution
    if remaining() > 150:
        n2 = n * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0.0, 2.0, n)
        x_new = np.linspace(0.0, 2.0, n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, target2)
        h_bin2 = encode_binary(h_up)

        # Also try Kronecker at higher res
        h_kr2 = kronecker_seq(n2, alpha_gold, 0.5)

        # SA on upsampled
        dl2 = start_time + budget_s * 0.92
        h_sa2, sc2 = binary_sa(h_bin2, dl2, rng,
                               n_iters=400000, T_init=0.01, T_end=0.0003,
                               directed_frac=0.3)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_sa2
            best_n = n2

        # Quick SA on Kronecker
        if remaining() > 60:
            h_sa3, sc3 = binary_sa(h_kr2, start_time + budget_s - 30, rng,
                                   n_iters=200000, T_init=0.01, T_end=0.001,
                                   directed_frac=0.3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_sa3
                best_n = n2

        # Continuous refine at higher res
        if remaining() > 40:
            h_ref2, sc_ref2 = continuous_refine(best_h, target2,
                                                start_time + budget_s - 25)
            if sc_ref2 < best_score:
                best_score = sc_ref2
                best_h = h_ref2
                best_n = n2

    # Final polish with SLSQP if time allows
    if remaining() > 50:
        n_final = len(best_h)
        target_final = n_final / 2.0

        def _slsqp_obj(x):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = np.clip(50000.0 * (corr - c_max), -30.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / 50000.0

        def _slsqp_con(x):
            return np.sum(x) - target_final

        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_final,
                          constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                          options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_final)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                best_n = n_final
        except Exception:
            pass

    h_final = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
