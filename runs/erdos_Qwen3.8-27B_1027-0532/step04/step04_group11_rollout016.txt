def top_lag_gradient_descent(h, n, target, n_iters, top_k, deadline):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top-k lags
        top_indices = np.argsort(corr)[-top_k:]
        top_vals = corr[top_indices]
        # Weights (softmax of top values)
        weights = np.exp(100 * (top_vals - top_vals.max()))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for idx, wt in zip(top_indices, weights):
            offset = idx - (n - 1)
            # Term 1: (1 - h[m+offset]) for valid m+offset
            if offset >= 0:
                grad[:n-offset] += wt * (1.0 - h[offset:])
            else:
                grad[-offset:] += wt * (1.0 - h[:n+offset])
            # Term 2: -h[m-offset] for valid m-offset
            if offset >= 0:
                grad[offset:] -= wt * h[:n-offset]
            else:
                grad[:n+offset] -= wt * h[-offset:]
        
        # Step size (simple: scale to max step 0.5)
        max_grad = np.max(np.abs(grad))
        if max_grad > 1e-15:
            step = 0.3 / max_grad
            h_new = h - step * grad
        else:
            h_new = h.copy()
        
        h_new = project_feasible(h_new, float(n // 2))
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, offset, n):
    """Compute gradient of corr at given offset (lag) w.r.t. h."""
    grad = np.zeros(n)
    if offset >= 0:
        # Term 1: (1-h[m+offset]) for m = 0..n-1-offset
        grad[:n-offset] += (1.0 - h[offset:])
        # Term 2: -h[m-offset] for m = offset..n-1
        grad[offset:] -= h[:n-offset]
    else:
        # offset < 0, let a = -offset
        a = -offset
        # Term 1: (1-h[m+offset]) = (1-h[m-a]) for m = a..n-1
        grad[a:] += (1.0 - h[:n-a])
        # Term 2: -h[m-offset] = -h[m+a] for m = 0..n-1-a
        grad[:n-a] -= h[a:]
    return grad

def soft_grad_descent(h, n, target, n_iters, top_k, momentum, deadline):
    """Gradient descent with top-k lag weighting and momentum."""
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    velocity = np.zeros(n)
    
    prev_step = None
    prev_grad = None
    bb_alpha = 0.5  # Barzilai-Borwein parameter
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top-k lags
        if top_k <= len(corr):
            top_indices = np.argsort(corr)[-top_k:]
        else:
            top_indices = np.arange(len(corr))
        top_vals = corr[top_indices]
        
        # Softmax weights (temperature 1)
        w = np.exp(top_vals - top_vals.max())
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, wt in zip(top_indices, w):
            if wt > 1e-10:
                offset = idx - (n - 1)
                grad += wt * grad_for_lag(h, offset, n)
        
        # Barzilai-Borwein step size
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            alpha = 0.1
        else:
            if prev_step is not None and prev_grad is not None:
                s = h - prev_step  # step direction (approximate)
                y = grad - prev_grad
                sTs = np.dot(s, s)
                sTy = np.dot(s, y)
                if sTs > 1e-20 and abs(sTy) > 1e-20:
                    bb_alpha = sTs / sTy
            alpha = max(0.01, min(2.0, bb_alpha / max_grad))
        
        # Momentum
        velocity = momentum * velocity - alpha * grad
        h_new = h + velocity
        h_new = project_feasible(h_new, float(n // 2))
        
        prev_step = h.copy()
        prev_grad = grad.copy()
        h = h_new
        
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score

def block_slsqp_optimize(h, n, target, block_size, n_sweeps, alpha, deadline):
    """Optimize blocks of h using SLSQP with log-sum-exp objective."""
    global _g_n, _g_target, _g_alpha, _g_h_base, _g_s, _g_e, _g_block_target
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': _block_con_eq, 'jac': _block_con_jac}]
            try:

                sc = compute_c5(ht)

            except Exception:

    return h_cur, cur

def anneal_perturb(h, n, target, n_iter, t_start, t_end, deadline):
    """Simulated annealing with pair swaps."""
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_local = h_cur.copy()
    best_local_sc = cur
    rng = np.random.RandomState(42)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            j = (j + 1) % n
        delta = (rng.rand() - 0.5) * 0.4
        h_new = h_cur.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        dsc = sc_new - cur
        if dsc < 0 or rng.rand() < np.exp(-dsc / max(t, 1e-12)):
            h_cur = h_new
            cur = sc_new
            if cur < best_local_sc:
                best_local_sc = cur
                best_local = h_cur.copy()
    return best_local, best_local_sc

def upsample(h, new_n):
    """Upsample h from len(h) to new_n points."""
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n // 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, offset, n):
    grad = np.zeros(n)
    if offset >= 0:
        grad[:n - offset] += (1.0 - h[offset:])
        grad[offset:] -= h[:n - offset]
    else:
        a = -offset
        grad[a:] += (1.0 - h[:n - a])
        grad[:n - a] -= h[a:]
    return grad


def soft_grad_descent(h, n, target, n_iters, top_k, momentum, deadline):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    velocity = np.zeros(n)
    prev_h = h.copy()
    prev_grad = np.zeros(n)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_k <= len(corr):
            top_indices = np.argsort(corr)[-top_k:]
        else:
            top_indices = np.arange(len(corr))
        top_vals = corr[top_indices]
        w = np.exp(top_vals - top_vals.max())
        w /= w.sum()
        
        grad = np.zeros(n)
        for idx, wt in zip(top_indices, w):
            if wt > 1e-10:
                offset = idx - (n - 1)
                grad += wt * grad_for_lag(h, offset, n)
        
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            alpha = 0.1
        else:
            if it > 0:
                s = h - prev_h
                y = grad - prev_grad
                sTs = np.dot(s, s)

        prev_grad = grad.copy()
        h = h_new
