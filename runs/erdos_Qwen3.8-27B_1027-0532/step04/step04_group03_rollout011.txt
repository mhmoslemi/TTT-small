def continuous_sa(h_start, target, time_limit, start_time, rng):
    """Simulated annealing with continuous values."""
    n = len(h_start)
    h = project_feasible(h_start.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = 3000
    
    t_init = 0.02
    t_final = 0.00001
    max_iters = 100000
    
    for i in range(max_iters):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Random perturbation: add noise to a random subset
        n_perturb = rng.randint(3, 20)
        positions = rng.choice(n, n_perturb, replace=False)
        noise = rng.randn(n_perturb) * (0.05 + 0.3 * (1 - ratio))
        h[positions] += noise
        h_new = project_feasible(h, target)
        
        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            current_c5 = c5_new
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            h[positions] -= noise
            h = project_feasible(h, target)
            no_improve += 1
    
    return best_h, best_c5

def targeted_active_refine(h_start, target, time_limit, start_time):
    """Focus on the most active shifts and try to equalize them."""
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for round_num in range(50):
        if (time.time() - start_time) > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        
        # Get the top shifts
        top_k = np.argsort(corr)[-min(10, len(corr)):]
        
        # For each top shift, compute which positions contribute most
        # and try to reduce their contribution
        improved = False
        for k_idx in top_k:
            m = k_idx - (n - 1)
            if m >= 0:
                # c = sum h[i]*(1-h[i+m]) for i=0..n-1-m
                contributions = h[:n-m] * (1.0 - h[m:])
            else:
                mm = -m
                contributions = h[mm:] * (1.0 - h[:n-mm])
            
            # Find the position with the largest contribution
            if len(contributions) == 0:
                continue
            max_pos = np.argmax(contributions)
            
            # Try to reduce h at that position (if m>=0) or h at position max_pos+mm (if m<0)
            if m >= 0:
                pos = max_pos  # reduce h[pos] which is the left factor
                # Also try the right factor
                pos2 = max_pos + m
            else:
                mm = -m
                pos = max_pos + mm  # reduce h at right factor position
                pos2 = max_pos  # or the left factor
            
            for pos_try in [pos, pos2]:
                if pos_try < 0 or pos_try >= n:
                    continue
                h_save = h[pos_try]
                h[pos_try] = max(0, h_save - 0.1)
                h_proj = project_feasible(h, target)
                c5_try = compute_c5(h_proj)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_proj.copy()
                    h = h_proj
                    improved = True
                else:
                    h[pos_try] = h_save
                    h = project_feasible(h, target)
        
        if not improved:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

def coord_descent(h_start, target, time_limit, start_time):

    dx = 2.0 / n
    h = project_feasible(h_start.copy(), target)
    
    # Compute initial correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    
    for iteration in range(10000):
        if (time.time() - start_time) > time_limit:
            break
        
        improved = False
        for i in range(n):
            if (time.time() - start_time) > time_limit:
                break
            
            old_val = h[i]
            if old_val < 1e-10 or old_val > 1 - 1e-10:
                # Try the opposite extreme
                new_val = 0.0 if old_val < 0.5 else 1.0
            else:
                # Try a few values
                continue  # Skip for now, handle binary case first
            
            # Compute change in correlations
            delta = new_val - old_val
            if abs(delta) < 1e-15:
                continue
            
            # Update correlations
            # Left factor contributions: h[i] appears as left factor at shift k where i+k is valid
            # Right factor contributions: h[i] appears as right factor at shift k where i-k is valid
            
            # For the full correlation output index m (shift = m - (n-1)):
            # h[i] is a left factor when: i + shift >= 0 and i < n and i + shift < n
            #   shift = m - (n-1), so i + m - (n-1) >= 0 and i + m - (n-1) < n
            #   => m >= n-1-i and m <= n-1-i+n-1 = 2n-2-i
            # h[i] is a right factor when: i - shift >= 0 and i - shift < n
            #   => i - m + (n-1) >= 0 and i - m + (n-1) < n
            #   => m <= i + n - 1 and m >= i + n - 1 - n + 1 = i
            #   => m >= i and m <= i + n - 1
            
            # Left factor: m in [n-1-i, 2n-2-i]
            # Right factor: m in [i, i+n-1]
            
            # For left factor: Δc(m) += delta * (1 - h[i + shift]) where shift = m - (n-1)
            #   i + shift = i + m - (n-1)
            #   valid when 0 <= i + m - (n-1) < n
            #   => n-1-i <= m <= 2n-2-i (same as before)
            
            # For right factor: Δc(m) -= delta * h[i - shift] where shift = m - (n-1)
            #   i - shift = i - m + (n-1)
            #   valid when 0 <= i - m + (n-1) < n
            #   => i <= m <= i + n - 1
            
            left_start = n - 1 - i
            left_end = min(2*n - 2 - i, 2*n - 2)
            right_start = i
            right_end = min(i + n - 1, 2*n - 2)
            
            # Apply left factor changes
            for m in range(left_start, left_end + 1):
                shift = m - (n - 1)
                j = i + shift  # = i + m - (n-1)
                corr[m] += delta * (1.0 - h[j])

            if len(ones) < 3 or len(zeros) < 3:
                continue
            idx1 = rng.choice(len(ones), 3, replace=False)
            idx0 = rng.choice(len(zeros), 3, replace=False)
            for idx in idx1:
                h[ones[idx]] = 0
            for idx in idx0:
                h[zeros[idx]] = 1

        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or (temp > 1e-12 and rng.random() < np.exp(-delta / max(temp, 1e-12))):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                h[ones[rng.randint(len(ones))]] = 0
                h[zeros[rng.randint(len(zeros))]] = 1
                h = project_binary(h, target)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            no_improve += 1

    return best_h, best_c5


def correlated_perturb_optimize(h_base, target, time_limit, start_time, rng):
    n = len(h_base)
    h = h_base.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for round_num in range(30):
        if (time.time() - start_time) > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        top_frac = 0.997
        active = np.where(corr >= c_max * top_frac)[0]

        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            if m >= 0:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
            else:
                mm = -m
                grad[:n - mm] += (1.0 - h[mm:])
                grad[mm:] -= h[:, :n - mm] if False else h[:n - mm]
        grad /= len(active)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        direction = -grad / gn

        best_step_c5 = best_c5
        best_step_h = h.copy()

        for frac in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            h_try = project_feasible(h + frac * direction, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_step_c5:
                best_step_c5 = c5_try
                best_step_h = h_try.copy()

        if best_step_c5 < best_c5 - 1e-14:
            best_c5 = best_step_c5
            best_h = best_step_h.copy()
            h = best_step_h
        else:
            break

    return best_h, best_c5


def try_multiple_n(h_base, base_n, target, time_limit, start_time, rng, factors):
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)

    for f in factors:
        if (time.time() - start_time) > time_limit:
            break
        n_try = int(base_n * f)
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 100 or n_try == base_n:
            continue
        target_try = n_try / 2.0

        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, base_n)
        h_interp = np.interp(x_new, x_old, h_base)
        h_interp = project_feasible(h_interp, target_try)

        h_bin = project_binary(h_interp, target_try)

        t_left = time_limit - (time.time() - start_time)
        if t_left < 20:
            break

        h_sa, c5_sa = binary_local_search(h_bin, target_try,
                                           min(25.0, t_left * 0.3), start_time, rng)

        if t_left - (time.time() - start_time) > 20:
            h_slq, c5_slq = slsqp_multistage(h_sa, target_try,
                                              min(40.0, (time_limit - (time.time() - start_time)) * 0.5),
                                              start_time,
                                              alphas=[200, 500, 1000, 3000, 8000])
            if c5_slq < c5_sa:
                h_sa = h_slq
                c5_sa = c5_slq

        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    return best_h, best_c5
