def fast_swap_update(h, i, j, corr, n, dx):
    """Vectorized swap update"""
    k_vals = np.arange(2*n - 1) - (n - 1)  # shifts: -(n-1), ..., n-1

    ik = i + k_vals
    ik_valid = (ik >= 0) & (ik < n)
    d_i = np.where(ik_valid, -(1.0 - h[ik][ik_valid]), 0.0)
    
    ik2 = i - k_vals
    ik2_valid = (ik2 >= 0) & (ik2 < n)
    d_i += np.where(ik2_valid, h[ik2][ik2_valid], 0.0)
    
    # For h[j]: 0->1
    jk = j + k_vals
    jk_valid = (jk >= 0) & (jk < n)
    d_j = np.where(jk_valid, (1.0 - h[jk][jk_valid]), 0.0)

    jk2_valid = (jk2 >= 0) & (jk2 < n)
    d_j -= np.where(jk2_valid, h[jk2][jk2_valid], 0.0)
    
    corr += (d_i + d_j) * dx
    h[i] = 0.0
    h[j] = 1.0
    return float(corr.max())

def swap_delta(i, j, h, k_vals, n):
    d = np.zeros(2 * n - 1)
    ik = i + k_vals

    jk2 = j - k_vals
    v4 = (jk2 >= 0) & (jk2 < n)
    d[v4] -= h[jk2[v4]]
    return d

            corr += total
            new_max = float(corr.max())
            delta = new_max - current_max
            if delta < -1e-14 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
                h[i1a] = 0.0
                h[i0a] = 1.0
                h[i1b] = 0.0
                h[i0b] = 1.0
                current_max = new_max
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                corr -= total
                no_improve += 1

        elif r < 0.88 and len(ones) > 0 and len(zeros) > 0:
            L = int(rng.randint(2, min(7, n // 30)))
            start = int(rng.randint(0, n - L))
            seg = h[start:start + L].copy()
            n_ones_seg = int(seg.sum())
            if 0 < n_ones_seg < L:
                shift = int(rng.randint(1, min(8, n - L)))
                if rng.random() < 0.5:
                    shift = -shift
                new_start = start + shift
                if 0 <= new_start and new_start + L <= n:
                    dest = h[new_start:new_start + L].copy()
                    if int(dest.sum()) == n_ones_seg:
                        h[start:start + L] = dest
                        h[new_start:new_start + L] = seg
                        corr = compute_corr_full(h)
                        new_max = float(corr.max())
                        delta = new_max - current_max
                        if delta < -1e-14 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
                            current_max = new_max
                            ones = np.where(h == 1)[0].astype(np.int64)
                            zeros = np.where(h == 0)[0].astype(np.int64)
                            if new_max < best_max - 1e-14:
                                best_max = new_max
                                best_h = h.copy()
                                no_improve = 0
                            else:
                                no_improve += 1
                        else:
                            h[start:start + L] = seg
                            h[new_start:new_start + L] = dest
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        else:
            worst_idx = int(corr.argmax())
            k_star = worst_idx - (n - 1)
            if k_star >= 0:
                valid_o = ones[ones + k_star < n]
                if len(valid_o) > 0:
                    bad = valid_o[h[valid_o + k_star] == 0]
                else:
                    bad = np.array([], dtype=np.int64)
            else:
                valid_o = ones[ones - k_star < n]
                if len(valid_o) > 0:
                    bad = valid_o[h[valid_o - k_star] == 0]
                else:
                    bad = np.array([], dtype=np.int64)
            if len(bad) > 0:
                i = int(bad[rng.randint(len(bad))])
                j = i + k_star
                if 0 <= j < n and h[j] == 0:
                    d = swap_delta(i, j, h, k_vals, n) * dx
                    corr += d
                    new_max = float(corr.max())
                    delta = new_max - current_max
                    if delta < -1e-14 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
                        h[i] = 0.0
                        h[j] = 1.0
                        current_max = new_max
                        ones = np.where(h == 1)[0].astype(np.int64)
                        zeros = np.where(h == 0)[0].astype(np.int64)
                        if new_max < best_max - 1e-14:
                            best_max = new_max
                            best_h = h.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        corr -= d
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

    return best_h, best_max


def perturb_solution(h, n, rng, block_sizes):
    h_p = h.copy()
    n_blocks = int(rng.randint(2, 5))
    for _ in range(n_blocks):
        L = int(rng.choice(block_sizes))
        if L >= n:
            continue
        start = int(rng.randint(0, n - L))
        seg = h_p[start:start + L].copy()
        flip_seg = 1.0 - seg
        n_ones_old = int(seg.sum())
        n_ones_new = int(flip_seg.sum())
        delta = n_ones_new - n_ones_old
        h_p[start:start + L] = flip_seg
        if delta > 0:
            available = np.where(h_p == 1)[0]
            mask = (available >= start) & (available < start + L)
            available = available[~mask]
            if len(available) >= delta:
                to_zero = available[rng.choice(len(available), delta, replace=False)]
                h_p[to_zero] = 0.0
            else:
                h_p = project_binary(h_p, n / 2.0)
                return h_p
        elif delta < 0:
            available = np.where(h_p == 0)[0]
            mask = (available >= start) & (available < start + L)
            available = available[~mask]
            if len(available) >= -delta:
                to_one = available[rng.choice(len(available), -delta, replace=False)]
                h_p[to_one] = 1.0
            else:
                h_p = project_binary(h_p, n / 2.0)
                return h_p
    h_p = project_binary(h_p, n / 2.0)
    return h_p


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_smooth_max(h_start, n, target, alphas, p_frac, maxiter, ftol,
                     time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    corr_len = 2 * n - 1
    p = max(3, int(corr_len * p_frac))

    for alpha in alphas:
        if (time.time() - start_time) > time_limit:
            break
        try:
            def obj(x):
                c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
                idx = np.argsort(c)[-p:]
                top = c[idx]
                m = top.max()
                shifted = np.clip(alpha * (top - m), -500.0, 0.0)
                return m + np.log(np.sum(np.exp(shifted))) / alpha
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_binary(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_binary(h_out, target_dst)
