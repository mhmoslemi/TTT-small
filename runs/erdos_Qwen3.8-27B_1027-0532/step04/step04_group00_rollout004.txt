def project_feasible(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    h = h.astype(float).copy()
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    # Sort h
    h_sorted = np.sort(h)[::-1]
    # Try different thresholds
    # When h_i - tau > 1 => h_i - 1 > tau => tau < h_i - 1 (clip to 1)
    # When h_i - tau < 0 => h_i < tau (clip to 0)
    # When 0 <= h_i - tau <= 1 => h_i - 1 <= tau <= h_i
    
    # Cumulative sum approach
    cumsum = np.cumsum(h_sorted)
    # For threshold between h_sorted[k-1] and h_sorted[k]:
    # sum = sum_{i<k}(h_i - tau) + sum_{i>=k}(0) = cumsum[k-1] - k*tau + ... 
    # Actually: for i < k (the top k values), clip(h_i - tau, 0, 1) = h_i - tau
    # for i >= k, clip = 0
    # So sum = cumsum[k-1] - k*tau = target
    # tau = (cumsum[k-1] - target) / k
    
    # We need tau such that h_sorted[k-1] >= tau >= h_sorted[k] (for interior)
    # Or tau >= h_sorted[0] - 1 (all clipped to 1)
    # Or tau <= h_sorted[n-1] (all clipped to 0, but then sum=0)
    
    best_tau = None
    for k in range(1, n+1):
        tau = (cumsum[k-1] - target) / k
        if k == 1:
            lo = -np.inf
        else:
            lo = h_sorted[k]
        if k == n:
            hi = np.inf
        else:
            hi = h_sorted[k-1]
        # Also need tau such that h_i - tau <= 1 for all i < k
        # i.e., tau >= h_i - 1 for all i < k, i.e., tau >= h_sorted[k-1] - 1
        lo = max(lo, h_sorted[k-1] - 1.0)
        # And tau <= h_i for all i >= k (if k < n)
        # i.e., tau <= h_sorted[k] (already have this)
        if lo <= tau <= hi + 1e-12:
            best_tau = tau
            break
    
    if best_tau is None:

            else:

            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 2.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def arc_polish(h, target, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        sorted_idx = np.argsort(corr)[::-1]
        num_active = 3
        active = sorted_idx[:num_active]
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / 0.005)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        for eps_try in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = h_cur - eps_try * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

def coord_descent(h, target, deadline, max_sweeps=200):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            if old_val < 1e-10 or old_val > 1.0 - 1e-10:
                for trial in [0.0, 1.0]:
                    if abs(trial - old_val) < 1e-12:
                        continue
                    h_cur[idx] = trial
                    h_cur = project_feasible(h_cur, target)
                    sc = compute_c5(h_cur)
                    if sc < cur_score:
                        cur_score = sc
                    else:
                        h_cur[idx] = old_val
                        h_cur = project_feasible(h_cur, target)
            else:
                for fr in [0.3, 0.7, 0.5, 0.1, 0.9]:
                    trial = old_val * fr
                    h_cur[idx] = trial
                    h_cur = project_feasible(h_cur, target)
                    sc = compute_c5(h_cur)
                    if sc < cur_score - 1e-14:
                        cur_score = sc
                    else:
                        break
                h_cur = project_feasible(h_cur, target)
    return h_cur, cur_score

def coarse_to_fine(h_fine, n_fine, target_fine, deadline):
    n_coarse = n_fine // 4
    target_coarse = n_coarse / 2.0
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    h_coarse = np.interp(x_coarse, x_fine, h_fine)
    h_coarse = project_feasible(h_coarse, target_coarse)
    sc_coarse = compute_c5(h_coarse)
    h_coarse = fista_optimize(h_coarse, target_coarse, 500, 20000, deadline)[0]
    sc_coarse2 = compute_c5(h_coarse)
    h_up = np.interp(x_fine, x_coarse, h_coarse)
    h_up = project_feasible(h_up, target_fine)
    sc_up = compute_c5(h_up)
    if sc_up < sc_fine_global:
        return h_up
    return None

def run(seed=42, budget_s=1300, **kwargs):
    global sc_fine_global
    start_time = time.time()
    deadline_final = start_time + budget_s - 15.0

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    sc_fine_global = best_score

    def tleft():
        return deadline_final - time.time()

    # Phase 1: FISTA with increasing alpha (continuation)
    alphas_cont = [50, 200, 800, 3000, 10000, 30000]
    allocated = [80, 60, 60, 80, 80, 80]
    for i, (a, td) in enumerate(zip(alphas_cont, allocated)):
        if tleft() < td + 10:
            break
        h_new, sc_new = fista_optimize(best_h, target, a, 50000, start_time + (80 if i == 0 else 60) - 0.01)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        sc_fine_global = best_score

    # Phase 2: Arc polish (subgradient on top lags)
    if tleft() > 60:
        h_new, sc_new = arc_polish(best_h, target, 200, start_time + 60)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        sc_fine_global = best_score

    # Phase 3: More FISTA at high alpha
    if tleft() > 100:
        h_new, sc_new = fista_optimize(best_h, target, 50000, 50000, start_time + 100)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        sc_fine_global = best_score
        if tleft() > 60:
            h_new, sc_new = arc_polish(best_h, target, 200, start_time + 60)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
                sc_fine_global = best_score

    # Phase 4: Coarse-to-fine restart
    if tleft() > 150:
        n_coarse = n // 4
        target_coarse = n_coarse / 2.0
        x_coarse = np.linspace(0, 1, n_coarse)
        x_fine = np.linspace(0, 1, n)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        h_coarse = project_feasible(h_coarse, target_coarse)
        h_coarse, _ = fista_optimize(h_coarse, target_coarse, 500, 30000, start_time + 40)
        h_coarse, _ = fista_optimize(h_coarse, target_coarse, 5000, 30000, start_time + 80)
        h_up = np.interp(x_fine, x_coarse, h_coarse)
        h_up = project_feasible(h_up, n / 2.0)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up
        sc_fine_global = best_score
        if tleft() > 60:
            h_new, sc_new = fista_optimize(best_h, target, 10000, 30000, start_time + 60)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
                sc_fine_global = best_score

    # Phase 5: Perturbation-based restarts
    rng = np.random.RandomState(seed)
    if tleft() > 120:
        scales = [0.03, 0.06, 0.1, 0.05, 0.08]
        for s in scales:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n) * s
            h_pert = project_feasible(h_pert, target)
            h_new, sc_new = fista_optimize(h_pert, target, 2000, 20000, start_time + 25)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            sc_fine_global = best_score

    # Phase 6: Final high-alpha FISTA + arc polish
    if tleft() > 80:
        h_new, sc_new = fista_optimize(best_h, target, 50000, 50000, start_time + 50)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = arc_polish(best_h, target, 300, start_time + 30)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        sc_fine_global = best_score

    # Phase 7: Coordinate descent final polish
    if tleft() > 30:
        h_new, sc_new = coord_descent(best_h, target, start_time + max(tleft() - 5, 10), max_sweeps=200)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
