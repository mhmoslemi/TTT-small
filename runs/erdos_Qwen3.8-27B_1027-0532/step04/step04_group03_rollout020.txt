def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)


def smooth_max_obj(x, alpha, target, penalty):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    sum_h = np.sum(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    corr = corr * dx

    smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sum_penalty = penalty * (sum_h - target) ** 2
    return smooth_max + sum_penalty

    corr = compute_corr_scaled(h)

    corr_top = corr[top_idx]
    c_shifted = alpha * (corr_top - np.max(corr_top))
    c_shifted = np.clip(c_shifted, -500.0, 0.0)
    p = np.exp(c_shifted)
    p_sum = np.sum(p)
    if p_sum < 1e-30:
        return np.zeros(n)
    p = p / p_sum
    grad = np.zeros(n)
    one_minus_h = 1.0 - h

def lbfgsb_smooth_max(h_start, target, alpha, time_limit, start_time, penalty=500.0):

    bounds = [(0.0, 1.0)] * n
    x0 = h_start.astype(float).copy()
    eval_count = [0]

    def obj(x):
        eval_count[0] += 1
        return smooth_max_obj(x, alpha, target, penalty)

    def grad(x):
        eval_count[0] += 1
        return smooth_max_grad(x, alpha, target, penalty)

    best_h = project_feasible(x0, target)
    best_c5 = compute_c5(best_h)
    try:
        res = minimize(
            obj, x0, jac=grad,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 5000, 'maxfun': 20000, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxls': 50, 'iprint': -1}

        h_new = project_feasible(res.x, target)

            h_new[pos] = np.clip(h_new[pos] + rng.uniform(-0.3, 0.3), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / t):
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    gradient_coord_descent._stall = 0

    # Phase 1: Binary SA
    h_bin_start = project_binary(best_h, target)
    time_p1 = min(60.0, tleft() * 0.12)
    if time_p1 > 5:
        h_sa, c5_sa = sa_optimize(h_bin_start, target, 200000, 0.02, 0.0001,
                                  rng, time_limit=time_p1, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 1b: Hill climb
    if tleft() > 20:
        h_hc, c5_hc = hill_climb_binary(best_h, target, 10000, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 2: Second SA round with different temperature
    if tleft() > 40:
        time_p2 = min(40.0, tleft() * 0.08)
        h_sa2, c5_sa2 = sa_optimize(best_h, target, 150000, 0.008, 0.00005,
                                    rng, time_limit=time_p2, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        h_hc2, c5_hc2 = hill_climb_binary(best_h, target, 8000, rng)
        if c5_hc2 < best_score:
            best_score = c5_hc2
            best_h = h_hc2.copy()

    # Phase 3: L-BFGS-B with smooth max (main continuous optimization)
    if tleft() > 60:
        alpha_schedule = [20, 50, 100, 200, 400, 800, 1500, 3000]
        for alpha in alpha_schedule:
            if tleft() < 20:
                break
            time_allow = min(40.0, tleft() * 0.12)
            if time_allow < 5:
                time_allow = 5
            h_new, c5_new = lbfgsb_smooth_max(best_h, target, alpha, time_allow,
                                              start_time, penalty=500.0)
            if c5_new < best_score:
                best_score = c5_new
                best_h = h_new.copy()

    # Phase 4: Gradient-guided coordinate descent
    if tleft() > 30:
        time_p4 = min(60.0, tleft() * 0.12)
        h_cd, c5_cd = gradient_coord_descent(best_h, target, 500.0, time_p4, start_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Multi-resolution
    n_candidates = []
    for factor in [0.8, 1.1, 1.25, 1.5]:
        nc = int(n * factor)
        if nc >= 200 and nc != n and nc % 2 == 0:
            n_candidates.append(nc)

    for n_try in n_candidates:
        if tleft() < 40:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        time_sa = min(15.0, tleft() * 0.15)
        h_sa_r, c5_sa_r = sa_optimize(h_bin_try, target_try, 50000, 0.01, 0.0002,
                                      rng, time_limit=time_sa, start_time=start_time)

        if tleft() > 30:
            time_lb = min(30.0, tleft() * 0.15)
            h_lb_r, c5_lb_r = lbfgsb_smooth_max(h_sa_r, target_try, 300.0, time_lb,
                                                start_time, penalty=500.0)
            if c5_lb_r < c5_sa_r:
                h_sa_r = h_lb_r
                c5_sa_r = c5_lb_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 6: Another round of L-BFGS-B at higher alpha on current best
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alpha_schedule2 = [2000, 4000, 8000, 15000]
        for alpha in alpha_schedule2:
            if tleft() < 15:
                break
            time_allow = min(30.0, tleft() * 0.15)
            h_new, c5_new = lbfgsb_smooth_max(best_h, target_use, alpha, time_allow,
                                              start_time, penalty=1000.0)
            if c5_new < best_score:
                best_score = c5_new
                best_h = h_new.copy()

    # Phase 7: Continuous SA for exploration
    if tleft() > 30:
        time_p7 = min(40.0, tleft() * 0.12)
        h_csa, c5_csa = continuous_sa(best_h, len(best_h) / 2.0, time_p7, start_time, rng)
        if c5_csa < best_score:
            best_score = c5_csa
            best_h = h_csa.copy()

    # Phase 8: Final L-BFGS-B polish
    if tleft() > 20:
        n_final = len(best_h)
        target_final = n_final / 2.0
        for alpha in [5000, 10000, 20000]:
            if tleft() < 10:
                break
            time_allow = min(20.0, tleft() * 0.2)
            h_new, c5_new = lbfgsb_smooth_max(best_h, target_final, alpha, time_allow,
                                              start_time, penalty=2000.0)
            if c5_new < best_score:
                best_score = c5_new
                best_h = h_new.copy()

    # Phase 9: Final coordinate descent
    if tleft() > 15:
        n_fd = len(best_h)
        target_fd = n_fd / 2.0
        time_p9 = min(20.0, tleft() * 0.2)
        h_fd, c5_fd = gradient_coord_descent(best_h, target_fd, 1000.0, time_p9, start_time)
        if c5_fd < best_score:
            best_score = c5_fd
            best_h = h_fd.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
