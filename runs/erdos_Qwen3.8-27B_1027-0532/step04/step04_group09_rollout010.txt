    h_bin: binary array of length n with exactly n/2 ones
    """
    h = h_bin.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_score = np.max(corr) * dx
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Compute delta for each possible flip
        # delta_1[a] = change in corr if h[a] goes from 1 to 0
        # delta_0[b] = change in corr if h[b] goes from 0 to 1
        
        deltas_1 = np.zeros((len(ones), 2*n-1))
        deltas_0 = np.zeros((len(zeros), 2*n-1))
        
        for i, a in enumerate(ones):
            # h[a] = 1 -> 0
            # corr[k] = sum_i h[i]*(1-h[i+k])
            # Terms involving h[a]:
            # As first index: h[a]*(1-h[a+k]) = 1*(1-h[a+k]) for k where 0<=a+k<n => -a<=k<n-a
            # As second index: h[a-k]*(1-h[a]) = h[a-k]*0 = 0 for k where 0<=a-k<n => a-n+1<k<=a
            # After flip (h[a]=0):
            # As first index: 0*(1-h[a+k]) = 0
            # As second index: h[a-k]*1 = h[a-k]
            # So change = - (1-h[a+k]) for -a<=k<n-a (as first index term removed)
            #            + h[a-k] for a-n+1<k<=a (as second index term added)

                idx = k + n - 1
                deltas_1[i, idx] -= (1 - h[a+k])

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    h = (h > 0.5).astype(float)

    ones_count = int(h.sum())
    if ones_count > int(target):
        # Remove excess ones
        one_indices = np.where(h == 1)[0]
        excess = ones_count - int(target)
        for i in range(excess):
            h[one_indices[i]] = 0
    elif ones_count < int(target):
        zero_indices = np.where(h == 0)[0]
        deficit = int(target) - ones_count
        for i in range(deficit):
            h[zero_indices[i]] = 1

    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    h_bin = (h > 0.5).astype(float)
    ones_count = int(h_bin.sum())

            break

            for k in range(-b, n - b):

            for k in range(-(n-1), b + 1):

                if 0 <= b - k < n:

        best_a = -1
        best_b_idx = -1

                    delta_1[k + n - 1] += h[a - k]

            # Vectorized: for each b, compute max(base + deltas_0[i])
            new_corrs = base[np.newaxis, :] + deltas_0
            scores = np.max(new_corrs, axis=1)

            sc = scores[min_idx]
            if sc < best_total_sc:

            zeros = np.where(h == 0)[0]
            no_improve = 0
        else:

    h = h.copy()
    dx = 2.0 / n

            break

                c = 0.0
                if 0 <= i + k < n:

                if 0 <= i - k < n:
                    c -= h[i - k]
                coeffs[idx] = c

            # Subject to 0 <= h_new <= 1
            # Use bisection on the subgradient
            lo = -h[i]
            hi = 1.0 - h[i]

            a_b, b_b = lo, hi

                new_corr = corr + coeffs * mid
                max_val = np.max(new_corr)
                active = np.where(new_corr >= max_val - 1e-12)[0]
                min_coeff = np.min(coeffs[active])
                max_coeff = np.max(coeffs[active])
                if min_coeff > 1e-10:
                    b_b = mid
                elif max_coeff < -1e-10:
                    a_b = mid
                else:
                    a_b = b_b = mid
                    break
            
            h[i] = np.clip(h[i] + a_b, 0.0, 1.0)
        
        h = project_feasible(h, target)
    
    return h
