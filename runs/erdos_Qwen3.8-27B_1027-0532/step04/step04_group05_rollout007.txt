    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    scaled = alpha * (corr - c_max)

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

        h = np.clip(x, 0, 1)
        val, _ = smooth_obj_and_grad(h, alpha)

        _, g = smooth_obj_and_grad(h, alpha)
        return g

        return max(0.0, budget_s - 30.0 - (time.time() - start_time))
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: SLSQP with analytical gradient at original resolution
    if tleft() > 150:
        alphas_phase1 = [50, 200, 1000, 5000, 20000]
        h_p1 = slsqp_analytic(best_h, target, alphas_phase1, 1500, min(150, tleft() * 0.8))
        sc = compute_c5(h_p1)
        if sc < best_score:
            best_score = sc
            best_h = h_p1
    
    # Phase 2: CMA-ES in Fourier basis for global exploration
    if tleft() > 200:
        M_fourier = 20
        cma_budget = min(250, tleft() * 0.3)
        h_cma, c5_cma = cma_es_simple(best_h, n, M_fourier, cma_budget, target, seed + 1)
        if c5_cma < best_score:
            best_score = c5_cma
            best_h = h_cma
        # Refine CMA-ES result with SLSQP
        if tleft() > 80:
            h_ref = slsqp_analytic(h_cma, target, [500, 2000, 10000], 1000, min(80, tleft() * 0.5))
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    
    # Phase 3: Subgradient refinement
    if tleft() > 60:
        h_sg, c5_sg = subgradient_improve(best_h, n, target, max_steps=15000, time_limit=min(40, tleft() * 0.3))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
    
    # Phase 4: Try higher resolution (n=800)
    if tleft() > 300:
        n_high = 800
        target_high = n_high / 2.0
        h_high = interpolate_to(best_h, n_high, target_high)
        alphas_high = [100, 500, 2000, 8000, 30000]
        h_high_opt = slsqp_analytic(h_high, target_high, alphas_high, 1200, min(250, tleft() * 0.5))
        sc_high = compute_c5(h_high_opt)
        if sc_high < best_score:
            best_score = sc_high
            best_h = h_high_opt
            n = n_high
            target = target_high
    
    # Phase 5: CMA-ES at current best resolution
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        M_f2 = min(25, max(15, n_cur // 20))
        h_cma2, c5_cma2 = cma_es_simple(best_h, n_cur, M_f2, min(120, tleft() * 0.4), target_cur, seed + 2)
        if c5_cma2 < best_score:
            best_score = c5_cma2
            best_h = h_cma2
        if tleft() > 60:
            h_ref2 = slsqp_analytic(h_cma2, target_cur, [500, 2000, 8000, 30000], 1000, min(60, tleft() * 0.5))
            sc = compute_c5(h_ref2)
            if sc < best_score:
                best_score = sc
                best_h = h_ref2
    
    # Phase 6: Perturbation + SLSQP restarts
    for noise_level in [0.01, 0.03, 0.05]:
        if tleft() < 80:
            break
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pert = best_h + rng.randn(n_cur) * noise_level
        h_pert = project_feasible(h_pert, target_cur)
        h_r = slsqp_analytic(h_pert, target_cur, [200, 1000, 5000], 800, min(70, tleft() * 0.4))
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best_h = h_r
    
    # Phase 7: Try n=1000 if budget allows
    if tleft() > 250:
        n_vhigh = 1000
        target_vhigh = n_vhigh / 2.0
        h_vhigh = interpolate_to(best_h, n_vhigh, target_vhigh)
        alphas_vh = [200, 1000, 5000]
        h_vh_opt = slsqp_analytic(h_vhigh, target_vhigh, alphas_vh, 1000, min(200, tleft() * 0.5))
        sc_vh = compute_c5(h_vh_opt)
        if sc_vh < best_score:
            best_score = sc_vh
            best_h = h_vh_opt
            n = n_vhigh
            target = target_vhigh
    
    # Phase 8: Final SLSQP polish with very high alpha
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final = slsqp_analytic(best_h, target_cur, [5000, 20000, 80000], 1500, min(35, tleft() * 0.8))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final
    
    # Phase 9: Final subgradient
    if tleft() > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg2, c5_sg2 = subgradient_improve(best_h, n_cur, target_cur, max_steps=8000, time_limit=min(18, tleft() * 0.7))
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2
    
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, len(h_final)
