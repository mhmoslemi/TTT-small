def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]

    else:
        need = int(target_sum - current_sum)
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:

        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:

        order = np.argsort(-room)

        for i in order:

            sub = min(room[i], remaining)

    """Try swap i1->0, i0->1. Returns (new_corr, new_c5) if accepted, else None."""
    # Compute delta
    # delta[k] for k from -(n-1) to n-1
    # In our array, shift k is at index n-1+k
    delta = np.zeros(2*n-1)
    
    for k_range in range(-(n-1), n):
        k = k_range
        d = 0
        if 0 <= i0 + k < n:
            d += h[i0 + k]
        if 0 <= i1 + k < n:
            d -= h[i1 + k]
        if 0 <= i1 - k < n:
            d += h[i1 - k]
        if 0 <= i0 - k < n:
            d -= h[i0 - k]
        delta[n-1+k] = d
    
    return corr + delta, float(np.max(corr + delta))
