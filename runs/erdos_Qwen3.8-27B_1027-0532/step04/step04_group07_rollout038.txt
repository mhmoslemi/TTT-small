def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(80):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-10
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def coord_update_1d(h, i, n, corr):
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)

    ff_valid = (i + shifts >= 0) & (i + shifts < n)
    sf_valid = (i - shifts >= 0) & (i - shifts < n)

    base = corr.astype(float).copy()
    coeff = np.zeros(2 * n - 1)

    only_ff = ff_valid & ~sf_valid
    if np.any(only_ff):
        s = shifts[only_ff]
        base[only_ff] = corr[only_ff] - h[i] * (1.0 - h[i + s])
        coeff[only_ff] = (1.0 - h[i + s])

    only_sf = sf_valid & ~ff_valid
    if np.any(only_sf):
        s = shifts[only_sf]
        base[only_sf] = corr[only_sf] - h[i - s] * (1.0 - h[i])
        coeff[only_sf] = -h[i - s]

    both_nz = ff_valid & sf_valid & (shifts != 0)
    if np.any(both_nz):
        s = shifts[both_nz]
        base[both_nz] = corr[both_nz] - h[i] * ((1.0 - h[i + s]) - h[i - s]) - h[i - s]
        coeff[both_nz] = (1.0 - h[i + s]) - h[i - s]

    base_0 = corr[n - 1] - h[i] + h[i] * h[i]

    grid = np.concatenate([
        np.linspace(0.0, 1.0, 17),
        [h[i], max(0.0, h[i] - 0.05), min(1.0, h[i] + 0.05),
         max(0.0, h[i] - 0.02), min(1.0, h[i] + 0.02)]
    ])
    grid = np.unique(grid)

    aff = base[None, :] + coeff[None, :] * grid[:, None]
    aff_max = np.max(aff, axis=1)
    quad = base_0 + grid - grid * grid
    total = np.maximum(aff_max, quad)

    best_idx = np.argmin(total)
    best_t = grid[best_idx]

    if 0 < best_idx < len(grid) - 1:
        lo = grid[best_idx - 1]
        hi = grid[best_idx + 1]
        fine = np.linspace(lo, hi, 40)

        aff_max_f = np.max(aff_f, axis=1)
        quad_f = base_0 + fine - fine * fine
        total_f = np.maximum(aff_max_f, quad_f)
        bi = np.argmin(total_f)
        best_t = fine[bi]

            hi2 = fine[bi + 1]

            total_ff = np.maximum(aff_max_ff, quad_ff)

def smooth_max_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx

def smooth_max_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = weights / w_sum

    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-14:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            m = min(shift, n - 1)
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            m = -shift
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad * dx
