overlap(k) ≈ 1 - autocorrelation(k)

The self-overlap sum_i h[i]*h[i+k]*dx is a quadratic form: h^T A_k h where A_k is a shift matrix.

= maximize min_k (h^T A_k h)

This is a max-min quadratic problem! The inner minimization over k is the bottleneck.

For a fixed k, h^T A_k h is a quadratic form. The problem of maximizing the minimum of several quadratic forms is non-convex in general.

def momentum_descent(h_init, n, target, alphas, max_iters, lr_init, momentum, t_start, deadline):

    best_sc = compute_c5(h_cur)
    best_h = h_cur.copy()

    lr = lr_init
    
    for alpha in alphas:
        for it in range(max_iters):
            if time.time() > deadline:
                return best_h, best_sc
            h_look = project_feasible(h_cur + momentum * v, target)
            # Gradient of soft-max
            corr = compute_corr(h_look)
            c_max = float(np.max(corr))

                g = np.zeros(n)

                    if s >= 0:

                        g[:n-s] -= w * h_look[s:]
                    else:
                        g[:n+s] += w * (1.0 - h_look[-s:])

                best_h = h_new.copy()
                h_cur = h_new
            else:

                if lr < 1e-12:
                    lr = lr_init
                    v = np.zeros(n)

- If m+k is in [0, n-1-k], i.e., m is in [0, n-1-k]: the term h[m+k]*(1-h[m]) contributes -h[m+k]

So:
dO(-k)/dh[m] = (1-h[m-k]) * [k ≤ m ≤ n-1] - h[m+k] * [0 ≤ m ≤ n-1-k]

grad[m] = sum_{k≠0} w_k * dO(k)/dh[m]

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999
            if mask.sum() == 0:
                mask = h < 1.0
        else:
            mask = h > 0.0001
            if mask.sum() == 0:
                mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    n = len(h)
    dx = 2.0 / n
    return val * dx

def compute_soft_gradient(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        w = wt[s + n - 1]
        if w < 1e-10:
            continue
        if s >= 0:
            g[s:] += w * (1.0 - h[:n - s])
            g[:n - s] -= w * h[s:]
        else:
            g[:n + s] += w * (1.0 - h[-s:])
            g[-s:] -= w * h[:n + s]
    return g

def slp_optimize(h_init, target, max_slp_iters, deadline):

    dx = 2.0 / n
    h_cur = h_init.copy()
    best_sc = compute_c5(h_cur)
    best_h = h_cur.copy()

        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        n_lags = 2 * n - 2
        A_ub = np.zeros((n_lags, n + 1))
        b_ub = np.zeros(n_lags)
        lag_idx = 0
        for s in range(-(n - 1), n):
            if s == 0:
                continue
            row = np.zeros(n)
            if s > 0:
                for i in range(n - s):
                    row[i] += (1.0 - h_cur[i + s])
                    row[i + s] += -h_cur[i]
            else:
                for i in range(-s, n):
                    row[i] += (1.0 - h_cur[i + s])
                    row[i + s] += -h_cur[i]
            cur_overlap = corr[n - 1 + s] * dx
            A_ub[lag_idx, :n] = row * dx
            A_ub[lag_idx, n] = -1.0
            b_ub[lag_idx] = -cur_overlap
            lag_idx += 1
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))

def rmsprop_optimize(h_init, n, target, alphas, iters_per_alpha, lr, decay, deadline):
