import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:

    if diff > 0:

        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()

Actually, let me think about this more carefully. The influence of changing h[p] from 0 to 1 on corr[m] (where m is the index in the full correlation, corresponding to lag = m - (n-1)):

corr[m] = ∑_i h[i] * (1-h[i - (m-(n-1))])

Let lag = m - (n-1).

The terms involving h[p]:
1. i = p: contributes h[p] * (1-h[p-lag]) if p-lag is in [0, n-1]
2. i such that i - lag = p, i.e., i = p+lag: contributes h[p+lag] * (1-h[p]) if p+lag is in [0, n-1]

When h[p] changes from 0 to 1:
- Term 1 changes from 0 to (1-h[p-lag]): Δ₁ = (1-h[p-lag]) * I(p-lag ∈ [0,n-1])
- Term 2 changes from h[p+lag]*(1-0) to h[p+lag]*(1-1) = 0: Δ₂ = -h[p+lag] * I(p+lag ∈ [0,n-1])

So influence[p][m] = (1-h[p-lag]) * I(valid) - h[p+lag] * I(valid)

For a swap (a: 0→1, b: 1→0), the change in corr is:
influence[a] - influence[b] (element-wise)

We want to find swaps that reduce the maximum of corr.

Strategy:
1. Find the current max value M = max(corr)
2. Find the set S of lags where corr is close to M (within some epsilon)
3. For those lags, find swaps (a,b) where influence[a][m] - influence[b][m] < 0 for all m in S
4. Apply the best such swap

But finding the "best" swap is expensive. Instead, we can:
1. For each lag m in S, compute for each position p the value influence[p][m]
2. For a swap to help at lag m, we need influence[a][m] < influence[b][m]
3. So we want a with small influence at all worst lags, and b with large influence at all worst lags

This is like a multi-objective optimization. A simple heuristic:
- Score_a = sum of influence[a][m] over m in S
- Score_b = sum of influence[b][m] over m in S
- Choose a (among zeros) with minimum score, b (among ones) with maximum score
- Check if the swap actually helps

Actually, let me think about a simpler but effective approach:

For each iteration:
1. Find the worst lag m* (or top K worst lags)
2. For m*, compute influence[p][m*] for all p
3. Among zeros: find a that minimizes influence[a][m*] (most negative = best for reducing corr[m*])
4. Among ones: find b that maximizes influence[b][m*] (most positive = best for reducing corr[m*] when removed)
5. Try this swap, check if it reduces the overall max
6. If not, try next best candidates

This is O(n) per candidate, and we might try a few candidates per iteration.

For n=600 and ~1000 lags, this is very fast per iteration. We can do millions of iterations.

                    if rng.random() < np.exp(-exp_arg) if exp_arg < 0 else (rng.random() * np.exp(exp_arg) > 1.0):
                        accept = True

                if accept:
                    cur_max = new_max
                    corr = new_corr.astype(np.float64)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if new_max < best_max - 1e-12:
                        best_max = new_max
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[a] = 0
                    h[b] = 1
                    no_improve += 1
            else:
                no_improve += 1

    return best_h, best_max * dx


def block_swap_local_search(h_init, target, max_iters, rng, time_limit, t_start, K=5):
    n = len(h_init)
    dx = 2.0 / n
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)
    corr = compute_correlation(h).astype(np.float64)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    max_no_improve = max(2000, max_iters // 15)

    for it in range(max_iters):
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break

        if no_improve > max_no_improve:
            h = best_h.copy()
            npert = rng.randint(5, max(6, n // 20))
            ones_tmp = np.where(h == 1)[0]
            zeros_tmp = np.where(h == 0)[0]
            for _ in range(npert):
                if len(ones_tmp) > 0 and len(zeros_tmp) > 0:
                    pi = int(ones_tmp[rng.randint(len(ones_tmp))])
                    pj = int(zeros_tmp[rng.randint(len(zeros_tmp))])
                    h[pi] = 0
                    h[pj] = 1
            h = project_binary(h, target_int).astype(np.float64)
            corr = compute_correlation(h).astype(np.float64)
            cur_max = float(np.max(corr))
            no_improve = 0
            continue

        sorted_lags = np.argsort(-corr)
        worst_lags = sorted_lags[:K]

        block_len = rng.randint(2, max(3, n // 30) + 1)
        start = rng.randint(0, n - block_len)
        end = start + block_len

        block = h[start:end].copy()
        n_ones_in_block = int(np.sum(block))
        n_zeros_in_block = block_len - n_ones_in_block

        ones_outside = np.where((np.arange(n) < start) | (np.arange(n) >= end))[0]
        ones_outside = ones_outside[h[ones_outside] == 1]
        zeros_outside = ones_outside[h[ones_outside] == 0]

        if n_ones_in_block == 0 or n_zeros_in_block == 0:
            no_improve += 1
            continue
        if len(ones_outside) < n_zeros_in_block or len(zeros_outside) < n_ones_in_block:
            no_improve += 1
            continue

        swap_ones = rng.choice(ones_outside, n_zeros_in_block, replace=False)
        swap_zeros = rng.choice(zeros_outside, n_ones_in_block, replace=False)

        h[swap_ones] = 0
        h[swap_zeros] = 1
        h[start:end] = block

        h = project_binary(h, target_int).astype(np.float64)
        new_corr = compute_correlation(h)
        new_max = float(np.max(new_corr))

        if new_max < cur_max - 1e-12:
            cur_max = new_max
            corr = new_corr.astype(np.float64)
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_max * dx


def smooth_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [10, 30, 80, 200, 500, 1000, 2000]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                lambda x: smooth_objective(x, alpha, n),
                h_cur, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-14:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def make_initial_random(n, rng, target):
    h = rng.binomial(1, 0.5, n).astype(float)
    return project_binary(h, target)


def make_periodic_pattern(n, period_bits, rng):
    h = np.zeros(n)
    for i in range(n):
        h[i] = period_bits[i % len(period_bits)]
    return project_binary(h, n // 2)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    h_init_arr = np.array(initial_h_values, dtype=float)
    n_init = len(h_init_arr)
    target_init = n_init / 2.0

    best_h = h_init_arr.copy()
    best_c5 = compute_c5(best_h)

    remaining = budget_s - 60.0 - (time.time() - start)

    h_bin = project_binary(h_init_arr, target_init)
    h_ls, c5_ls = targeted_local_search(h_bin, target_init, 500000, rng,
                                         min(300.0, remaining), start, 0.01, 0.001, K=5, top_candidates=20)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    remaining = budget_s - 60.0 - (time.time() - start)

    h_block, c5_block = block_swap_local_search(best_h, target_init, 200000, rng,
                                                 min(120.0, remaining), start, K=5)
    if c5_block < best_c5:
        best_c5 = c5_block
        best_h = h_block.copy()

    remaining = budget_s - 60.0 - (time.time() - start)

    alt_inits = []
    h_qr = make_qr_sequence(n_init, rng)
    alt_inits.append(h_qr)
    h_rand = make_initial_random(n_init, rng, target_init)
    alt_inits.append(h_rand)

    for h_alt in alt_inits:
        if remaining < 60.0:
            break
        h_ls2, c5_ls2 = targeted_local_search(h_alt, target_init, 300000, rng,
                                               min(80.0, remaining), start, 0.01, 0.001, K=5, top_candidates=15)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
        remaining = budget_s - 60.0 - (time.time() - start)

    remaining = budget_s - 60.0 - (time.time() - start)

    for nt in [n_init, n_init + 4, n_init + 8]:
        if remaining < 70.0:
            break
        if nt == n_init:
            continue
        target_nt = nt / 2.0
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, nt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = project_binary(h_interp, target_nt)
        h_ls3, c5_ls3 = targeted_local_search(h_bin_try, target_nt, 200000, rng,
                                               min(80.0, remaining), start, 0.01, 0.001, K=5, top_candidates=15)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3.copy()
        remaining = budget_s - 60.0 - (time.time() - start)

    remaining = budget_s - 60.0 - (time.time() - start)

    if remaining > 100.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cur, sc = continuous_refine(best_h, n_use, target_use, min(120.0, remaining), start)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_cur
        remaining = budget_s - 60.0 - (time.time() - start)

    if remaining > 30.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        h_ls4, c5_ls4 = targeted_local_search(h_bin_final, target_use, 200000, rng,
                                               max(10.0, remaining - 10.0), start, 0.005, 0.0005, K=7, top_candidates=20)
        if c5_ls4 < best_c5:
            best_c5 = c5_ls4
            best_h = h_ls4.copy()

    remaining = budget_s - 30.0 - (time.time() - start)

    if remaining > 60.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cur2, sc2 = continuous_refine(best_h, n_use, target_use, max(10.0, remaining - 10.0), start)
        if sc2 < best_c5:
            best_c5 = sc2
            best_h = h_cur2

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5:
        h_final = best_h
        final_c5 = best_c5

    elapsed_total = time.time() - start
    if elapsed_total > budget_s - 5.0:
        return h_final, final_c5, n_final

    return h_final, final_c5, n_final
