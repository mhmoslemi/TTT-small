from scipy.optimize import minimize, differential_evolution
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_block_full = None
_g_block_start = 0
_g_block_end = 0
_g_de_target = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def block_obj_fn(x):
    h_full = _g_block_full.copy()
    h_full[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    corr = compute_corr(h_full)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline):

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

            h_de_coarse = project_feasible(res_de.x, target_coarse)
            h_de_fine = interpolate_to(h_de_coarse, n, target)
            deadline_de = start_time + (budget_s - 30.0) * 0.40
            h_de_ref, sc_de_ref = slsqp_round(h_de_fine, n, target,
                                               [500, 2000, 8000], 2500, 1e-15, deadline_de)
            if sc_de_ref < best_score:
                best_score = sc_de_ref
                best_h = h_de_ref
        except Exception:
            pass

    # Phase 3: Second DE run with different seed and coarser grid
    if budget_s - 30.0 - (time.time() - start_time) > 300.0:
        n_coarse2 = 60
        target_coarse2 = n_coarse2 / 2.0
        _g_de_target = target_coarse2
        bounds_de2 = [(0.0, 1.0)] * n_coarse2
        try:
            res_de2 = differential_evolution(de_obj_fn, bounds_de2, seed=seed + 1,
                                              popsize=15, maxiter=100,
                                              tol=1e-10, polish=True)
            h_de2_coarse = project_feasible(res_de2.x, target_coarse2)
            h_de2_fine = interpolate_to(h_de2_coarse, n, target)
            deadline_de2 = start_time + (budget_s - 30.0) * 0.50
            h_de2_ref, sc_de2_ref = slsqp_round(h_de2_fine, n, target,
                                                 [500, 2000, 8000], 2000, 1e-15, deadline_de2)
            if sc_de2_ref < best_score:
                best_score = sc_de2_ref
                best_h = h_de2_ref
        except Exception:
            pass

    # Phase 4: Subgradient polish
    if budget_s - 30.0 - (time.time() - start_time) > 200.0:
        dl = start_time + (budget_s - 30.0) * 0.55
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=800, n_active=5,
                                         lr_init=0.003, lr_decay=0.997, deadline=dl)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 5: Active set refinement
    if budget_s - 30.0 - (time.time() - start_time) > 150.0:
        h_as, sc_as = active_set_refine(best_h, n, target, n_active=5, n_iter=150)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    # Phase 6: Block SLSQP
    if budget_s - 30.0 - (time.time() - start_time) > 200.0:
        deadline_bd = start_time + (budget_s - 30.0) * 0.72
        for bs in [2, 3]:
            if budget_s - 30.0 - (time.time() - start_time) < 80.0:
                break
            h_bd, sc_bd = block_slsqp(best_h, n, target, bs, 6, 3000.0, deadline_bd)
            if sc_bd < best_score:
                best_score = sc_bd
                best_h = h_bd

    # Phase 7: Worst-lag surgery
    if budget_s - 30.0 - (time.time() - start_time) > 120.0:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, max_steps=400,
                                             initial_epsilon=0.003)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        deadline7 = start_time + (budget_s - 30.0) * 0.82
        h7, sc7 = slsqp_round(best_h, n, target, [5000, 20000], 2000, 1e-16, deadline7)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Final block descent with smaller blocks
    if budget_s - 30.0 - (time.time() - start_time) > 80.0:
        deadline8 = start_time + (budget_s - 30.0) * 0.90
        h_bd2, sc_bd2 = block_slsqp(best_h, n, target, 2, 4, 5000.0, deadline8)
        if sc_bd2 < best_score:
            best_score = sc_bd2
            best_h = h_bd2

    # Phase 9: Final subgradient
    if budget_s - 30.0 - (time.time() - start_time) > 50.0:
        deadline9 = start_time + (budget_s - 30.0) * 0.95
        h_sg2, sc_sg2 = subgradient_polish(best_h, n, target, n_steps=300, n_active=3,
                                           lr_init=0.001, lr_decay=0.999, deadline=deadline9)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 10: Final surgery
    if budget_s - 30.0 - (time.time() - start_time) > 20.0:
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, n, target, max_steps=200,
                                               initial_epsilon=0.001)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
