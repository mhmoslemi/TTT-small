def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-14)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def lse_subset_obj_grad(h, alpha, indices, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[indices]
    cmax = np.max(vals)
    shifted = np.clip(alpha * (vals - cmax), -500.0, 0.0)
    ev = np.exp(shifted)
    tot = np.sum(ev)
    obj = cmax + np.log(tot) / alpha
    w = ev / tot
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(indices)):
        wi = w[i]
        if wi < 1e-15:
            continue
        lag = indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:

                grad[:n - lag] -= wi * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += wi * omh[m:]
                grad[m:] -= wi * h[:n - m]
    return obj, grad, corr

    g = np.zeros(n)

    if lag >= 0:

            g[m:] -= h[:n - m]

        s = np.sum(x) - target
        return f + penalty * s * s, g + 2.0 * penalty * s

        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def subgradient_step(h, target, n, step_size):

    kmax = np.argmax(corr)
    lag = kmax - (n - 1)
    g = grad_lag(h, lag, n)
    h_try = h - step_size * g
    h_try = project_feasible(h_try, target)
    return h_try, kmax
