def compute_lag_effects(h, p):
    """For coordinate p, compute the coefficient of delta for each correlation lag."""
    n = len(h)
    effects = np.zeros(2*n - 1)
    # effects[k] = d(corr[k])/d(h[p])
    
    # For lag = 0 (k = n-1):
    effects[n-1] = 1 - 2*h[p]
    
    # For lag = m > 0 (k = n-1+m):
    # corr[n-1+m] = sum_{i=0}^{n-m-1} h[i] * (1-h[i+m])
    # d/dh[p]: 
    #   if p < n-m: (1-h[p+m])  [p appears as h[i] with i=p]
    #   if p >= m: -h[p-m]      [p appears as h[i+m] with i=p-m]
    for m in range(1, n):
        effect = 0.0
        if p + m < n:
            effect += (1 - h[p+m])
        if p >= m:
            effect -= h[p-m]
        effects[n-1+m] = effect
    
    # For lag = -m (k = n-1-m):
    # corr[n-1-m] = sum_{i=m}^{n-1} h[i] * (1-h[i-m])

    for m in range(1, n):
        effect = 0.0
        if p >= m:
            effect += (1 - h[p-m])
        if p + m < n:
            effect -= h[p+m]
        effects[n-1-m] = effect
    
    return effects

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):

        if w < 1e-14:

                v = h[p]

                if v > 0.01:
                    cands.append(0.0)
                if v < 0.99:
                    cands.append(1.0)
                if v > 0.25:
                    cands.append(max(0.0, v - 0.25))
                if v < 0.75:
                    cands.append(min(1.0, v + 0.25))
                if len(cands) > n_candidates:
                    cands = cands[:n_candidates]

                best_sc = best_score
                best_new = None
                for cv in cands:
                    if abs(cv - v) < 1e-6:
                        continue
                    h_try = h.copy()
                    h_try[p] = cv
                    h_try = project_feasible(h_try, target_sum)
                    corr_try = overlap_corrs(h_try)
                    sc = float(np.max(corr_try)) * dx
                    if sc < best_sc or (temp > 1e-9 and rng.random() < np.exp(-(sc - best_sc) / temp)):
                        best_sc = sc
                        best_new = (h_try, corr_try, sc)
                if best_new is not None:
                    h, corr_cur, sc = best_new
                    pass_improved = True
                    if sc < best_score:
                        best_score = sc
                        best_h = h.copy()
        else:
            for start_idx in range(0, n, block_size):
                p = order[start_idx % n]
                v = h[p]
                cands = []
                if v > 0.01:
                    cands.append(0.0)
                if v < 0.99:
                    cands.append(1.0)
                best_sc = best_score
                best_new = None
                for cv in cands:
                    if abs(cv - v) < 1e-6:
                        continue
                    h_try = h.copy()
                    h_try[p] = cv
                    h_try = project_feasible(h_try, target_sum)
                    corr_try = overlap_corrs(h_try)
                    sc = float(np.max(corr_try)) * dx
                    if sc < best_sc or (temp > 1e-9 and rng.random() < np.exp(-(sc - best_sc) / temp)):
                        best_sc = sc
                        best_new = (h_try, corr_try, sc)
                if best_new is not None:
                    h, corr_cur, sc = best_new
                    pass_improved = True
                    if sc < best_score:
                        best_score = sc
                        best_h = h.copy()

        if not pass_improved:
            no_improve_total += n
            temp *= sa_cooldown
        else:
            no_improve_total = 0
            temp *= 0.995

        if no_improve_total > max_no_improve:
            break

    return best_h, best_score

def block_local_search(h, target_sum, time_budget, rng, block_size=5, n_cand=3):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    no_improve = 0

    while time.time() - t0 < time_budget and no_improve < 3 * (n // block_size):
        positions = np.random.permutation(n // block_size) * block_size
        improved = False
        for pos in positions:
            if pos + block_size > n:
                pos = n - block_size
            v_block = h[pos:pos+block_size].copy()
            candidates = []
            candidates.append(np.zeros(block_size))
            candidates.append(np.ones(block_size))
            candidates.append(1.0 - v_block)
            
            best_sc = best_score
            best_new = None
            for cb in candidates[:n_cand]:
                if np.max(np.abs(cb - v_block)) < 1e-6:
                    continue
                h_try = h.copy()
                h_try[pos:pos+block_size] = cb
                h_try = project_feasible(h_try, target_sum)
                sc = compute_c5(h_try)
                if sc < best_sc:
                    best_sc = sc
                    best_new = h_try.copy()
            if best_new is not None:
                h = best_new
                improved = True
                if best_sc < best_score:
                    best_score = best_sc
                    best_h = h.copy()
        if not improved:
            no_improve += n // block_size
        else:
            no_improve = 0

    return best_h, best_score

def smooth_max_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = float(np.max(top_vals))
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = float(np.max(top_vals))
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)

def _cons_eq_fun(x):
    return np.sum(x) - _LP_TARGET

def _cons_eq_jac(x):
    return np.ones(_LP_N)

def slsqp_refine(h_start, target_sum, alpha_topk_pairs, maxiter, ftol, time_limit):
    global _LP_N, _LP_TARGET
    _LP_N = len(h_start)
    _LP_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _LP_N
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    t0 = time.time()
    for alpha, topk in alpha_topk_pairs:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(
                smooth_max_obj, h_cur, args=(alpha, topk),
                jac=smooth_max_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def subgradient_refine(h_start, target_sum, max_iter, lr_init, topk, alpha,
                       patience, momentum, seed, time_limit):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stag = 0
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
            top_vals = corr[top_idx]
        else:
            top_idx = np.arange(n_corr)
            top_vals = corr
        w = np.exp(alpha * (top_vals - np.max(top_vals)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_idx, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stag = 0
            lr = min(lr * 1.02, 2.0)
        else:
            stag += 1
            lr *= 0.98
        h = h_new
        if stag > patience:
            break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _LP_N, _LP_TARGET
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Discrete local search at n=400 (fast, targeted)
    if _remaining() > 120:
        h_new, sc = discrete_local_search(best_h, target_init,
                                           time_budget=min(100, _remaining() - 80),
                                           rng=rng, topk_lags=30, n_candidates=4,
                                           sa_temperature=0.001, sa_cooldown=0.97)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: LP refinement at n=400
    if _remaining() > 150:
        h_new, sc = lp_refine(best_h, target_init, max_iter=300, topk=50,
                               time_limit=min(120, _remaining() - 100))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: More discrete search with SA
    if _remaining() > 100:
        h_pert = best_h + rng.randn(n_init) * 0.005
        h_pert = project_feasible(h_pert, target_init)
        h_new, sc = discrete_local_search(h_pert, target_init,
                                           time_budget=min(80, _remaining() - 60),
                                           rng=rng, topk_lags=20, n_candidates=3,
                                           sa_temperature=0.003, sa_cooldown=0.95)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: SLSQP refinement at n=400
    if _remaining() > 150:
        h_ref, sc = slsqp_refine(best_h, target_init,
                                  [(100, 80), (500, 40), (2000, 20)],
                                  maxiter=2000, ftol=1e-15,
                                  time_limit=min(120, _remaining() - 100))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Block local search for larger moves
    if _remaining() > 80:
        h_new, sc = block_local_search(best_h, target_init,
                                        time_budget=min(60, _remaining() - 50),
                                        rng=rng, block_size=4, n_cand=3)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: LP with different topk
    if _remaining() > 80:
        h_new, sc = lp_refine(best_h, target_init, max_iter=200, topk=25,
                               time_limit=min(60, _remaining() - 50),
                               epsilon_seq=[0.003, 0.001, 0.0005, 0.0])
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Multi-resolution - go to n=600
    n600 = 600
    target600 = n600 / 2.0
    if _remaining() > 250:
        h600 = interpolate_to(best_h, n600, target600)
        h600, sc600 = discrete_local_search(h600, target600,
                                             time_budget=min(120, _remaining() - 100),
                                             rng=rng, topk_lags=40, n_candidates=4,
                                             sa_temperature=0.001, sa_cooldown=0.97)
        if _remaining() > 100:
            h600, sc600 = lp_refine(h600, target600, max_iter=200, topk=60,
                                     time_limit=min(80, _remaining() - 60))
        if sc600 < best_score:
            best_score = sc600
            best_h = h600

    # Phase 8: Multi-resolution - go to n=800
    n800 = 800
    target800 = n800 / 2.0
    if _remaining() > 200:
        src_h = best_h if len(best_h) == n600 else best_h
        h800 = interpolate_to(src_h, n800, target800)
        h800, sc800 = discrete_local_search(h800, target800,
                                             time_budget=min(100, _remaining() - 80),
                                             rng=rng, topk_lags=50, n_candidates=3,
                                             sa_temperature=0.002, sa_cooldown=0.96)
        if _remaining() > 80:
            h800, sc800 = lp_refine(h800, target800, max_iter=150, topk=50,
                                     time_limit=min(60, _remaining() - 50))
        if sc800 < best_score:
            best_score = sc800
            best_h = h800

    # Phase 9: Subgradient at current best resolution
    if _remaining() > 100:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_pert2 = best_h + rng.randn(n_cur) * 0.003
        h_pert2 = project_feasible(h_pert2, t_cur)
        h_new, sc = subgradient_refine(h_pert2, t_cur,
                                        max_iter=3000, lr_init=0.05, topk=5,
                                        alpha=30, patience=200, momentum=0.90,
                                        seed=seed + 7777,
                                        time_limit=min(80, _remaining() - 50))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 10: Final discrete polish
    if _remaining() > 50:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_new, sc = discrete_local_search(best_h, t_cur,
                                           time_budget=min(40, _remaining() - 25),
                                           rng=rng, topk_lags=20, n_candidates=3,
                                           sa_temperature=0.0, sa_cooldown=1.0)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 11: Final LP polish
    if _remaining() > 30:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_new, sc = lp_refine(best_h, t_cur, max_iter=100, topk=15,
                               time_limit=min(25, _remaining() - 10),
                               epsilon_seq=[0.001, 0.0005, 0.0])
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final projection and evaluation
    n_final = len(best_h)
    t_final = n_final / 2.0
    h_final = project_feasible(best_h, t_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
