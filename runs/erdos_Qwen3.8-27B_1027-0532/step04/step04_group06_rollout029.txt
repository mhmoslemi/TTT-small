    w /= w.sum()
    grad = np.zeros(n)
    for w_val, ti in zip(w, top_indices):
        k = int(ti - center)
        if k == 0:
            continue
        g = np.zeros(n)
        if k > 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] += -h[:n-k]
        else:
            m = -k
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] += -h[m:]
        grad += w_val * g
    return grad

def adam_continuous_opt(h_start, target_sum, n_iters, lr, beta1, beta2, eps, top_k, alpha_w_schedule, time_limit, start_time):
    """Adam-based continuous optimization focusing on top-k lags."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target_sum)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        t += 1
        
        # Adaptive alpha_w: increase over time (sharper focus on worst lag)
        progress = i / max(n_iters - 1, 1)
        alpha_w = alpha_w_schedule[0] + (alpha_w_schedule[1] - alpha_w_schedule[0]) * progress
        
        # Compute gradient
        grad = compute_grad_topk(h, top_k=top_k, alpha_w=alpha_w)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - target
    
    def cons_j(h_in):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def fast_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 150
    
    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
