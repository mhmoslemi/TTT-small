import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h = h.copy()
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_optimize(h, target, alpha, maxiter, deadline=None):

    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

        return lse_obj_and_grad(x, alpha)[1]
    
    def cfun(x):
        return np.sum(x) - target
    
    def cjac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        cur_sc = compute_c5(h)
        if sc < cur_sc:

    except Exception:
        pass
    return h, compute_c5(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
        else:
            mask = h > 1e-7
        cnt = int(mask.sum())
        if cnt == 0:
            mask = (h < 1.0) if diff > 0 else (h > 0.0)
            cnt = int(mask.sum())
            if cnt == 0:
                break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    obj = c_max + np.log(sum_exp) / alpha
    w = exp_v / sum_exp
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:

                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

        return lse_obj_and_grad(x, alpha)[1]

        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

                       bounds=bounds,

            best_score = sc

    h_cd, sc = coord_descent_maxcorr(best_h, target_init, 100, deadline_p5)
    if sc < best_score:
        best_score = sc
        best_h = h_cd

    # Phase 6: Try higher resolution (12% time)
    deadline_p6 = start_time + (budget_s - 20.0) * 0.90
    for n_new in [600, 800]:
        if time.time() > deadline_p6:
            break
        target_new = n_new / 2.0
        h_new = interpolate_to(best_h, n_new, target_new)
        sc_new = compute_c5(h_new)

        h_new, sc_new = coord_descent_maxcorr(h_new, target_new, 30, deadline_p6)
        for alpha in [5000, 50000]:
            if time.time() > deadline_p6:
                break
            h_new, sc_new = slsqp_optimize(h_new, target_new, alpha, 500, deadline_p6)
        h_new, sc_new = coord_descent_maxcorr(h_new, target_new, 30, deadline_p6)

        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Final polish (6% time)
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h, sc = active_subgrad_polish(best_h, target_final, 2000, 0.01, 0.01, deadline_final)
    if sc < best_score:
        best_score = sc

    h_cd, sc = coord_descent_maxcorr(best_h, target_final, 50, deadline_final)
    if sc < best_score:
        best_score = sc
        best_h = h_cd

    for alpha in [500000, 1000000]:
        if time.time() > deadline_final:
            break
        best_h, sc = slsqp_optimize(best_h, target_final, alpha, 500, deadline_final)
        if sc < best_score:
            best_score = sc

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
