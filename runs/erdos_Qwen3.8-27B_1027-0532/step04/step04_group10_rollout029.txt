        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

def lbfgs_step(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol):
    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f = f + penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

def topm_obj_grad(h, m):
    """Objective: sum of top-m correlation values. Gradient accordingly."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    # Get top m indices
    if m < n_corr:
        idx = np.argpartition(corr, -m)[-m:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    obj = np.sum(corr_top)
    
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):

            m_lag = -lag
            if m_lag < n:
                grad[:n - m_lag] += one_minus_h[m_lag:]
                grad[m_lag:] -= h[:n - m_lag]
    return obj, grad

def binary_local_search(h, target, n_rounds, rng):
    """Try to improve by flipping individual bits (in a soft sense)."""
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for r in range(n_rounds):
        improved = False
        # Random order
        order = rng.permutation(n)
        for coord in order:
            orig = best_h[coord]
            # Try moving toward 0 or 1
            if orig > 0.5:
                candidates = [0.0, 0.1, 0.3, orig - 0.2, orig - 0.1]
            else:
                candidates = [1.0, 0.9, 0.7, orig + 0.2, orig + 0.1]
            
            for val in candidates:
                h_try = best_h.copy()

                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try
                    improved = True
                    break
        if not improved:
            break
    
    return best_h, best_score
