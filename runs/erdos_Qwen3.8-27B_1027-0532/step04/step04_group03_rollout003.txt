import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smooth_obj(x, alpha, target, penalty):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sum_viol = np.sum(h) - target
    obj += penalty * sum_viol * sum_viol
    return obj

def smooth_grad(x, alpha, target, penalty):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
        else:
            mm = -m
            grad[:n-mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n-mm]
    sum_viol = np.sum(h) - target
    grad += 2.0 * penalty * sum_viol
    # Zero out gradient where h is at bounds (approximately)
    # Actually, since we clip, the gradient of clip is 0 at bounds
    grad[h <= 0.0] = 0.0
    grad[h >= 1.0] = 0.0
    return grad

def lbbf_optimize(h_start, target, alpha, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_obj, h_start, args=(alpha, target, penalty),
            jac=smooth_grad, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def continuous_sa(h_start, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = 5000

    for i in range(max_iters):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        move_type = rng.randint(4)
        
        if move_type == 0:
            # Random perturbation
            idx = rng.randint(n)
            delta = rng.uniform(-0.1, 0.1)
            h_new = h.copy()
            h_new[idx] += delta
            h_new = project_feasible(h_new, target)
        elif move_type == 1:
            # Swap-like (change two positions)
            i1, i0 = rng.choice(n, 2, replace=False)
            h_new = h.copy()
            delta = 0.2 * rng.uniform(0.5, 1.0)
            h_new[i1] -= delta
            h_new[i0] += delta
            h_new = project_feasible(h_new, target)
        elif move_type == 2:
            # Block perturbation
            block_size = rng.randint(2, 6)

            h_new = h.copy()
            h_new[start:start+block_size] += rng.uniform(-0.1, 0.1, block_size)
            h_new = project_feasible(h_new, target)
        else:

            h_new = h.copy()
            h_new[mask] += rng.uniform(-0.05, 0.05, mask.sum())
            h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        delta_c = c5_new - current_c5
        if delta_c < 0 or (t > 1e-12 and rng.random() < np.exp(-delta_c / t)):
            current_c5 = c5_new
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def slab_swap(h, target, n_swaps, rng, time_limit, start_time):
    """Try swapping slabs of h with other slabs"""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(n_swaps):
        if (time.time() - start_time) > time_limit:
            break
        # Pick two random positions and try to swap their values
        i, j = rng.choice(n, 2, replace=False)
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)


def smooth_objective(x, alpha, target, penalty):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sum_viol = np.sum(h) - target
    obj += penalty * sum_viol * sum_viol
    return obj


def smooth_gradient(x, alpha, target, penalty):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    sum_viol = np.sum(h) - target
    grad += 2.0 * penalty * sum_viol
    grad[h <= 0.0] = 0.0
    grad[h >= 1.0] = 0.0
    return grad


def lbbf_optimize(h_start, target, alpha, penalty, maxiter):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_objective, h_start, args=(alpha, target, penalty),
            jac=smooth_gradient, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start.copy()

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_score = c5_mr
            best_h = h_mr

    # Phase 8: Final intensive L-BFGS-B on best solution
    if remaining() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas_f = [2000, 5000, 10000, 20000]
        pens_f = [3000, 5000, 5000, 10000]
        iters_f = [300, 400, 400, 500]
        h_f, c5_f = multistart_lbfgs(best_h, target_use, alphas_f, pens_f, iters_f, remaining(), start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 9: Final local refinement
    if remaining() > 15:
        h_fl, c5_fl = local_refine_single(best_h, len(best_h) / 2.0, min(20.0, remaining() - 5), start_time)
        if c5_fl < best_score:
            best_score = c5_fl
            best_h = h_fl

    # Phase 10: Final pairwise swaps
    if remaining() > 10:
        h_fs, c5_fs = pairwise_swap(best_h, len(best_h) / 2.0, 5000, rng, min(10.0, remaining() - 5), start_time)
        if c5_fs < best_score:
            best_score = c5_fs
            best_h = h_fs

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
