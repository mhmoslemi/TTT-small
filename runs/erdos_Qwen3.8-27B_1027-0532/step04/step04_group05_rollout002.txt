    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_overlap_k(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.empty(n)
    idxs = np.arange(n)
    # Contribution from h[j] as the first factor
    v1 = (idxs - k >= 0) & (idxs - k < n)
    # Contribution from h[j] as the second factor (inside 1-h)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[:] = 0.0
    grad[v1] += (1.0 - h[idxs[v1] - k]) * dx
    grad[v2] -= h[idxs[v2] + k] * dx
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:

        return h
    h = h.copy()

    h = np.clip(h, 0.0, 1.0)
