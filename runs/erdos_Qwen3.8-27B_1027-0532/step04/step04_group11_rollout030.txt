                continue

            lower_from_neg = np.max((mid - intercepts[neg_mask]) / slopes[neg_mask])
        else:
            lower_from_neg = 0.0

            upper_from_pos = np.min((mid - intercepts[pos_mask]) / slopes[pos_mask])
        else:
            upper_from_pos = 1.0
        
        lower = max(0.0, lower_from_neg)
        upper = min(1.0, upper_from_pos)
        
        if lower <= upper:
            hi = mid  # y is feasible, try smaller
        else:
            lo = mid  # y is not feasible, try larger
    
    y_opt = (lo + hi) / 2
    
    # Find t that achieves y_opt
    if np.any(neg_mask):
        lower_from_neg = np.max((y_opt - intercepts[neg_mask]) / slopes[neg_mask])
    else:
        lower_from_neg = 0.0
    if np.any(pos_mask):
        upper_from_pos = np.min((y_opt - intercepts[pos_mask]) / slopes[pos_mask])
    else:
        upper_from_pos = 1.0
    
    lower = max(0.0, lower_from_neg)
    upper = min(1.0, upper_from_pos)
    t_opt = (lower + upper) / 2
    
    return t_opt, y_opt

    f1 = np.max(slopes + intercepts)
    lo = min(f0, f1)
    hi = max(f0, f1)
    if hi - lo < 1e-16:
        return 0.5
    for _ in range(45):
        mid = (lo + hi) / 2.0
        neg_mask = slopes < -1e-12
        pos_mask = slopes > 1e-12
        if np.any(neg_mask):
            lb = np.max((mid - intercepts[neg_mask]) / slopes[neg_mask])
        else:
            lb = 0.0
        if np.any(pos_mask):
            ub = np.min((mid - intercepts[pos_mask]) / slopes[pos_mask])
        else:
            ub = 1.0
        if max(lb, 0.0) <= min(ub, 1.0) + 1e-14:
            hi = mid
        else:
            lo = mid
    y_opt = (lo + hi) / 2.0
    neg_mask = slopes < -1e-12
    pos_mask = slopes > 1e-12
    if np.any(neg_mask):
        lb = np.max((y_opt - intercepts[neg_mask]) / slopes[neg_mask])
    else:
        lb = 0.0
    if np.any(pos_mask):
        ub = np.min((y_opt - intercepts[pos_mask]) / slopes[pos_mask])
    else:
        ub = 1.0
    t_opt = (max(lb, 0.0) + min(ub, 1.0)) / 2.0
    return float(np.clip(t_opt, 0.0, 1.0))


def coord_descent(h, target, lam, corr, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    corr_cur = corr.copy()
    n1 = n - 1
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            k_min = max(0, n1 - i)
            k_max = min(2 * n - 2, n1 + i)
            if k_min > k_max:
                continue
            valid_ks = np.arange(k_min, k_max + 1)
            js = i + valid_ks - n1
            m = 1.0 - h_cur[js]
            b = corr_cur[valid_ks] - h_cur[i] * m
            slopes = m + lam
            t_opt = find_min_envelope(slopes, b)
            delta = t_opt - h_cur[i]
            if abs(delta) > 1e-14:
                h_cur[i] = t_opt
                corr_cur[valid_ks] += delta * m
        diff = np.sum(h_cur) - target
        lam = lam - 0.05 * diff
        lam = float(np.clip(lam, -5.0, 5.0))
        h_cur = project_feasible(h_cur, target)
        corr_cur = corr_array(h_cur)
    sc = float(np.max(corr_cur)) * (2.0 / n)
    return h_cur, sc, lam, corr_cur


def smooth_max_obj(x, alpha):
    c = corr_array(x)
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lfbfgs_optimize(h, n, target, deadline, alphas):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        penalty = 1.0
        for p_iter in range(5):
            if time.time() > deadline:
                break

            def obj_pen(x):
                val = smooth_max_obj(x, alpha)
                sd = np.sum(x) - target
                return val + 0.5 * penalty * sd * sd

            try:
                res = minimize(obj_pen, h_cur, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 100, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur:
                    h_cur = h_new
                    cur = sc
                sd = abs(np.sum(h_cur) - target)
                if sd < 1e-8:
                    break
                penalty *= 5.0
            except Exception:
                break
    return h_cur, cur


def block_optimize(h, n, target, block_size, deadline, alpha=500.0):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for bi in range(n_blocks):
        if time.time() > deadline:
            break
        s = bi * block_size
        e = s + block_size
        rest_sum = np.sum(h_cur[:s]) + np.sum(h_cur[e:])
        block_target = target - rest_sum

        def block_obj(x):
            hf = h_cur.copy()
            hf[s:e] = np.clip(x, 0.0, 1.0)
            c = corr_array(hf)
            c_max = float(np.max(c))
            shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        def block_eq(x):
            return np.sum(x) - block_target

        bounds_b = [(0.0, 1.0)] * block_size
        cons = [{'type': 'eq', 'fun': block_eq}]
        x0 = h_cur[s:e].copy()
        try:
            res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
            ht = h_cur.copy()
            ht[s:e] = np.clip(res.x, 0.0, 1.0)
            ht = project_feasible(ht, target)
            sc = compute_c5(ht)
            if sc < cur:
                h_cur = ht
                cur = sc
        except Exception:
            pass
    return h_cur, cur


def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    if L1 > 0:
        h[:L1] = 1.0
    if L2 > 0:
        h[n - L2:] = 1.0
    return h


def periodic_h(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    total = np.sum(h)
    if total > n // 2:
        h[:int(total - n // 2)] = 0.0
    elif total < n // 2:
        deficit = int(n // 2 - total)
        idx = np.where(h == 0)[0]
        for j in range(min(deficit, len(idx))):
            h[idx[j]] = 1.0
    return h


def interpolate_h(h_in, n_out):
    n_in = len(h_in)
    x_in = np.linspace(0.0, 2.0, n_in)
    x_out = np.linspace(0.0, 2.0, n_out)
    h_out = np.interp(x_out, x_in, h_in)
    return project_feasible(h_out, n_out / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = two_interval_h(400, 100)

    n = len(h_init)
    target = float(n) / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    if elapsed() > budget_s - 100:
        return best_h, best_score, n

    # Phase 1: Coordinate descent at original resolution
    corr0 = corr_array(best_h)
    hl = 0.0
    dl1 = t0 + budget_s * 0.25
    h_cd, sc_cd, hl, corr_cd = coord_descent(best_h, target, hl, corr0, 300, dl1)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    if elapsed() > budget_s - 80:
        return best_h, best_score, n

    # Phase 2: L-BFGS polish at original resolution
    dl2 = t0 + budget_s * 0.35
    h_lf, sc_lf = lfbfgs_optimize(best_h, n, target, dl2, [200, 1000, 5000])
    if sc_lf < best_score:
        best_score = sc_lf
        best_h = h_lf

    if elapsed() > budget_s - 80:
        return best_h, best_score, n

    # Phase 3: Block SLSQP at original resolution
    dl3 = t0 + budget_s * 0.45
    h_blk, sc_blk = block_optimize(best_h, n, target, 8, dl3, alpha=1000.0)
    if sc_blk < best_score:
        best_score = sc_blk
        best_h = h_blk

    if elapsed() > budget_s - 60:
        return best_h, best_score, n

    # Phase 4: Multi-start exploration
    dl4 = t0 + budget_s * 0.65
    n_attempts = 6
    for att in range(n_attempts):
        if time.time() > dl4:
            break
        if att == 0:
            h_s = two_interval_h(n, rng.randint(n // 6, n // 3))
        elif att == 1:
            h_s = two_interval_h(n, rng.randint(n // 6, n // 3))
            h_s = h_s + rng.normal(0, 0.08, n)
            h_s = project_feasible(h_s, target)
        elif att == 2:
            p = rng.randint(4, 15)
            d = max(1, p // 3)
            h_s = periodic_h(n, p, d)
        else:
            h_s = best_h.copy()
            h_s += rng.normal(0, 0.05 + 0.03 * att, n)
            h_s = project_feasible(h_s, target)
        c_s = corr_array(h_s)
        h_try, sc_try, _, c_try = coord_descent(h_s, target, 0.0, c_s, 150, dl4)
        if sc_try < best_score and time.time() < t0 + budget_s * 0.75:
            h_pol, sc_pol = lfbfgs_optimize(h_try, n, target, t0 + budget_s * 0.75, [500, 2000])
            if sc_pol < best_score:
                best_score = sc_pol
                best_h = h_pol
        elif sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    if elapsed() > budget_s - 60:
        return best_h, best_score, n

    # Phase 5: Try higher resolution
    dl5 = t0 + budget_s * 0.82
    n_hi = 2 * n
    h_hi = interpolate_h(best_h, n_hi)
    target_hi = float(n_hi) / 2.0
    sc_hi0 = compute_c5(h_hi)
    if sc_hi0 < best_score + 0.005:
        c_hi = corr_array(h_hi)
        h_cd_hi, sc_cd_hi, _, _ = coord_descent(h_hi, target_hi, 0.0, c_hi, 200, dl5)
        if sc_cd_hi < best_score and time.time() < t0 + budget_s * 0.92:
            h_pol_hi, sc_pol_hi = lfbfgs_optimize(h_cd_hi, n_hi, target_hi, t0 + budget_s * 0.92, [500, 2000])
            if sc_pol_hi < best_score:
                best_score = sc_pol_hi
                best_h = h_pol_hi
        elif sc_cd_hi < best_score:
            best_score = sc_cd_hi
            best_h = h_cd_hi

    if elapsed() > budget_s - 40:
        return best_h, best_score, n

    # Phase 6: Final intensive coordinate descent
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    dl6 = t0 + budget_s * 0.92
    c_f = corr_array(best_h)
    h_cd_f, sc_cd_f, _, _ = coord_descent(best_h, target_final, 0.0, c_f, 400, dl6)
    if sc_cd_f < best_score:
        best_score = sc_cd_f
        best_h = h_cd_f

    if elapsed() > budget_s - 30:
        return best_h, best_score, n_final

    # Phase 7: Final SLSQP polish
    dl7 = t0 + budget_s - 10
    h_blk_f, sc_blk_f = block_optimize(best_h, n_final, target_final, 6, dl7, alpha=5000.0)
    if sc_blk_f < best_score:
        best_score = sc_blk_f
        best_h = h_blk_f

    best_h = project_feasible(best_h, target_final)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, len(best_h)
