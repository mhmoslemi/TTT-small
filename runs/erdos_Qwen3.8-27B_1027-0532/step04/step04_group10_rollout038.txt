def lse_obj_and_grad(h, alpha, topk):
    """Log-sum-exp smooth approximation to max(corr)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    n_corr = len(corr)
    
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    
    # Gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    
    return obj, grad

def optimize_lse(h_start, target, alpha, topk, maxiter, ftol):

    bounds = [(0.0, 1.0)] * n

        # Add constraint penalty
        s = np.sum(x) - target
        f += 1e6 * s * s
        g = g + 2e6 * s * np.ones(n)
        return f, g

def fw_step(h, grad, target, step_size):
    """Frank-Wolfe step on the hypersimplex."""
    n = len(h)
    # Linear minimizer: pick target indices with smallest grad
    v = np.zeros(n)
    v[np.argsort(grad)[:int(target)]] = 1.0
    return (1 - step_size) * h + step_size * v

def away_step(h, grad, target, step_size):
    """Away step: move away from the 'worst' component."""
    n = len(h)
    # Find the index with largest h_i * grad_i (most to move away from)
    # Simple version: find the index where h is most "committed" and grad is large
    positive = h > 0.1
    if not np.any(positive):
        return h
    # Among positive entries, find the one with largest grad
    idx = np.where(positive)[0]
    worst = idx[np.argmax(grad[idx])]
    # Move away: decrease h[worst] and distribute to others
    h_new = h.copy()
    amount = min(h[worst], step_size)
    h_new[worst] -= amount
    # Distribute to indices with smallest grad among those < 1
    other = np.where((np.arange(n) != worst) & (h < 0.9))[0]
    if len(other) > 0:
        other_sorted = np.argsort(grad[other])[:int(amount * 20)]  # distribute to best
        per = amount / len(other_sorted)
        h_new[other_sorted] += per
    return project_hypersimplex(h_new, target)

def shift_perturb(h, target, lag, epsilon):

    n = len(h)
    h_shifted = np.zeros(n)

    # Move h toward h_shifted to reduce disagreement at this lag

    hi = float(np.max(h)) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > k:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    result = np.clip(h - theta, 0.0, 1.0)
    diff = k - float(np.sum(result))
    if abs(diff) > 1e-12:
        free = np.where((result > 1e-12) & (result < 1.0 - 1e-12))[0]
        if len(free) > 0:
            result[free] += diff / len(free)
            result = np.clip(result, 0.0, 1.0)
    return result


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = float(np.max(corr_top))
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = float(np.sum(exp_vals))
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = int(idx[i]) - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad


def optimize_lbfgs(h_start, target, alpha, topk, maxiter, ftol, penalty):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = float(np.sum(x)) - target
        f += penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g

    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 50})
        return project_hypersimplex(res.x, target)
    except Exception:
        return project_hypersimplex(h_start, target)


def optimize_slsqp(h_start, target, alpha, topk, maxiter):
    n = len(h_start)

    def obj_func(x):
        f, _ = lse_obj_grad(x, alpha, topk)
        return f

    def obj_jac(x):
        _, g = lse_obj_grad(x, alpha, topk)
        return g

    def con_fun(x):
        return float(np.sum(x)) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_start, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_hypersimplex(res.x, target)
    except Exception:
        return project_hypersimplex(h_start, target)


def shift_perturb(h, target, lag, epsilon):
    n = len(h)
    h_shifted = np.zeros(n)
    if lag >= 0:
        h_shifted[:n - lag] = h[lag:]
    else:
        m = -lag
        h_shifted[m:] = h[:n - m]
    h_new = h + epsilon * (h_shifted - h)
    return project_hypersimplex(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_hypersimplex(h_init, target)
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 40.0 - (time.time() - start_time)

    # Phase 1: Multi-alpha L-BFGS-B with multiple restarts
    alpha_schedule = [15, 30, 60, 120, 250, 500, 1000, 2000, 5000]
    penalty = 2000.0
    for ai, alpha in enumerate(alpha_schedule):
        if time_left() < 70:
            break
        n_restart = 3
        if time_left() < 200:
            n_restart = 2
        if time_left() < 100:
            n_restart = 1
        noise_scale = 0.004 * (1.0 / (1.0 + 0.3 * ai))
        for r in range(n_restart):
            if time_left() < 40:
                break
            h_start = best_h + rng.randn(n) * noise_scale
            h_start = project_hypersimplex(h_start, target)
            h_new = optimize_lbfgs(h_start, target, alpha, 100, 400, 1e-16, penalty)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 2: Worst-lag targeted reduction
    if time_left() > 80:
        for rnd in range(8):
            if time_left() < 50:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top_lags = np.argsort(corr)[-5:]
            for tl in top_lags:
                if time_left() < 40:
                    break
                wl = int(tl) - (n - 1)
                for eps in [0.08, 0.04, 0.02, 0.01]:
                    h_try = shift_perturb(best_h, target, wl, eps)
                    sc = compute_c5(h_try)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_try
                for eps in [-0.04, -0.02, -0.01]:
                    h_try = shift_perturb(best_h, target, wl, eps)
                    sc = compute_c5(h_try)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_try
            if time_left() < 40:
                break
            h_new = optimize_lbfgs(best_h, target, 500, 80, 300, 1e-16, penalty)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 3: Frank-Wolfe style exploration
    if time_left() > 100:
        alpha_fw = 200
        for it in range(300):
            if time_left() < 30:
                break
            _, grad = lse_obj_grad(best_h, alpha_fw, 80)
            t = 2.0 / (it + 2.0)
            v = np.zeros(n)
            v[np.argsort(grad)[:int(target)]] = 1.0
            h_fw = (1.0 - t) * best_h + t * v
            h_fw = project_hypersimplex(h_fw, target)
            score = compute_c5(h_fw)
            if score < best_score:
                best_score = score
                best_h = h_fw
            if (it + 1) % 60 == 0 and time_left() > 50:
                h_new = optimize_lbfgs(best_h, target, 500, 80, 200, 1e-16, penalty)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new

    # Phase 4: SLSQP refinement
    if time_left() > 80:
        for alpha_s in [300, 1000]:
            if time_left() < 40:
                break
            h_new = optimize_slsqp(best_h, target, alpha_s, 60, 800)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 5: Coordinate-wise local search
    if time_left() > 40:
        for rnd in range(4):
            if time_left() < 20:
                break
            improved = False
            for coord in range(n):
                if time_left() < 15:
                    break
                orig = best_h[coord]
                for delta in [0.008, -0.008, 0.004, -0.004, 0.002, -0.002, 0.001, -0.001]:
                    h_try = best_h.copy()
                    h_try[coord] = np.clip(orig + delta, 0.0, 1.0)
                    h_try = project_hypersimplex(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_h = h_try
                        improved = True
                        break
            if not improved:
                break

    # Phase 6: Final high-alpha L-BFGS polish
    if time_left() > 30:
        for alpha_f in [3000, 10000]:
            if time_left() < 20:
                break
            h_new = optimize_lbfgs(best_h, target, alpha_f, 30, 300, 1e-17, penalty)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 7: Final SLSQP
    if time_left() > 25:
        h_new = optimize_slsqp(best_h, target, 2000, 30, 1000)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    best_h = project_hypersimplex(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
