import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[::(len(ones2)//(final_sum-target_sum)+1)][:int(final_sum-target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[::(len(zeros2)//(target_sum-final_sum)+1)][:int(target_sum-final_sum)]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

    if current_sum == int(target_sum):

    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > int(target_sum):
        need = current_sum - int(target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum) - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]

def project_feasible(h, target_sum):

def softmax_objective(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_c = np.sort(corr)
    top = sorted_c[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

def softmax_grad(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_idx = np.argsort(corr)
    top_indices = sorted_idx[-p:]
    top_corr = corr[top_indices]
    max_top = np.max(top_corr)
    exps = np.exp(alpha * (top_corr - max_top))
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(top_indices)):
        k = int(top_indices[idx]) - (n - 1)
        w = weights[idx] * dx
        if abs(w) < 1e-15:
            continue
        m1_s = max(0, -k)
        m1_e = min(n, n - k)
        if m1_s < m1_e:
            grad[m1_s:m1_e] += w * (1.0 - h[m1_s + k:m1_e + k])
        m2_s = max(0, k)
        m2_e = min(n, n + k)
        if m2_s < m2_e:
            grad[m2_s:m2_e] -= w * h[m2_s - k:m2_e - k]
    return grad

        max_it = int(min(30000, allowed * 2500))
        h_sa_r, c5_r = conflict_directed_sa(h_bin_r, target_r, max_it, 0.012, 0.00004,
                                            rng, time_limit=allowed, start_time=start_time,
                                            top_m=3, n_cand=5, block_prob=0.1)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_sa_r.copy()

    n_final_use = len(best_h)
    target_final_use = n_final_use / 2.0
    fine_attempts = min(5, int(max(1, tleft() / 40)))

    for fa in range(fine_attempts):
        if tleft() < 30:
            break
        noise = rng.choice([0.003, 0.008, 0.015])
        h_pert = best_h + rng.randn(n_final_use) * noise
        h_pert = project_feasible(h_pert, target_final_use)
        h_pert_bin = project_binary(h_pert, target_final_use)
        allowed = min(30.0, tleft() - 10)
        if allowed < 5:
            break
        max_it = int(min(20000, allowed * 3000))
        h_sa_f, c5_f = conflict_directed_sa(h_pert_bin, target_final_use, max_it,
                                            0.008, 0.00002, rng,
                                            time_limit=allowed, start_time=start_time,
                                            top_m=3, n_cand=6, block_prob=0.08)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    if tleft() > 15:
        n_fu = len(best_h)
        target_fu = n_fu / 2.0
        h_bin_final = project_binary(best_h, target_fu)
        max_it = int(max(3000, tleft() * 2000))
        h_sa_f2, c5_f2 = conflict_directed_sa(h_bin_final, target_fu, max_it,
                                              0.004, 0.00001, rng,
                                              time_limit=max(5.0, tleft() - 8),
                                              start_time=start_time,
                                              top_m=3, n_cand=8, block_prob=0.05)
        if c5_f2 < best_score:
            best_score = c5_f2
            best_h = h_sa_f2.copy()
        if tleft() > 8:
            h_hc_f, c5_hc_f = quick_hill_climb(best_h, target_fu, 2000, rng)
            if c5_hc_f < best_score:
                best_score = c5_hc_f
                best_h = h_hc_f.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
