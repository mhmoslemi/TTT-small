def _compute_overlap(h):
    """Compute the overlap integral C_5 for a step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with (1-h) using scipy.signal.correlate (more efficient than np.correlate for our needs)
    # Note: We use the same weighted mean as the original evaluator
    h_flat = h.astype(float)
    data = h_flat
    weights = 1.0 - h_flat
    return np.correlate(data, weights, mode='full')

def compute_c5(h):
    """Compute C_5 for step function h."""
    corr = _compute_overlap(h)
    return float(np.max(corr)) * (2.0 / len(h))

def _project_feasible(h, target_sum, max_iter=500, tol=1e-8):
    """Project step function h to have total integral sum(h)*dx = target_sum."""
    dx = 2.0 / len(h)
    h_flat = h.copy().astype(float)
    cur_sum = np.sum(h_flat)
    if abs(cur_sum - target_sum) < tol:
        return h_flat
    # Use iterative adjustment
    for _ in range(max_iter):
        if abs(cur_sum - target_sum) < tol:
            break
        if cur_sum < target_sum:
            # Add mass (if below target)
            mask = h_flat < 1.0
        else:
            # Remove mass (if above target)
            mask = h_flat > 0.0
        if not np.any(mask):
            break
        # Adjust by a small amount
        step = (target_sum - cur_sum) / (np.sum(mask) or 1.0)
        if step < 0:
            step = -step
        adjust_step = step / (len(h_flat) * (1.0 if step > 0 else 1.0))
        h_flat[mask] += adjust_step
        # Ensure bounds and clamp
        h_flat = np.clip(h_flat, 0.0, 1.0)
        cur_sum = np.sum(h_flat)
        if abs(cur_sum - target_sum) < tol or np.sum(mask) == 0:
            return h_flat
    return h_flat
