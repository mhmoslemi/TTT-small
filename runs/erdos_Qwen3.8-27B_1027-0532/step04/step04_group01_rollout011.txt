import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def continuous_refine_v2(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [30, 80, 200, 500, 1000, 2000]
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return smooth_objective(x, alpha, n)
            def cfun(x):
                return np.sum(x) - target
            def cjac(x):
                return np.ones(n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score


def gradient_refine(h_init, n, target, time_limit, t_start, lr_init=0.01, n_steps=2000):
    h = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h)
    
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        alpha = 200
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max) / dx
        shifted = np.clip(shifted, -500.0, 0.0)
        obj_val = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
        
        # Numerical gradient
        eps = 0.01
        grad = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            if h_plus.sum() > target + 1e-10:
                h_plus[i] = h[i]
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            if h_minus.sum() < target - 1e-10:
                h_minus[i] = h[i]
            
            def calc_obj(x):
                xx = project_feasible(x, target)
                return smooth_objective(xx, alpha, n)
            
            grad[i] = (calc_obj(h_plus) - calc_obj(h_minus)) / (h_plus[i] - h_minus[i] + 1e-15)
        
        # Projected gradient step
        lr = lr_init * 0.99 ** step
        h_new = h - lr * grad
        # Project onto constraint
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        
        new_score = compute_c5(h_new)
        if new_score < curr_score:
            h = h_new
            curr_score = new_score
    
    return h, curr_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def smooth_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def smooth_gradient(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        return np.zeros(n)
    softmax = weights / total
    grad = np.zeros(n)
    for idx in range(len(softmax)):
        w = softmax[idx]
        if w < 1e-8:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += w * (1.0 - h[lag:n])
                grad[lag:n] -= w * h[0:n - lag]
        else:
            l = -lag
            if n - l > 0:
                grad[l:n] += w * (1.0 - h[0:n - l])
                grad[0:n - l] -= w * h[l:n]
    return grad

    h = project_binary(h_init, target_int).astype(np.float64)
    one_minus_h = 1.0 - h

    ones = np.where(h == 1)[0]

        ratio = it / max(max_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        saved_change = []

            elif move_type == 3 and len(ones) > 0 and len(zeros) > 0:

def pg_refine(h_init, n, target, time_limit, t_start):
    h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(h)
    alphas = [40, 100, 300, 800, 2000, 5000]
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        dx = 2.0 / n
        base_lr = 4.0 * dx / alpha
        prev_f = smooth_objective(h, alpha, n)
        for step in range(800):
            if time.time() - t_start > time_limit:
                break
            grad = smooth_gradient(h, alpha, n)
            norm_grad = np.linalg.norm(grad)
            if norm_grad < 1e-12:
                break
            found = False
            for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
                lr = base_lr * lr_mult
                h_trial = h - lr * grad
                h_trial = np.clip(h_trial, 0.0, 1.0)
                h_trial = project_feasible(h_trial, target)
                f_trial = smooth_objective(h_trial, alpha, n)
                if f_trial < prev_f - 1e-16:
                    h = h_trial
                    prev_f = f_trial
                    found = True
                    break
            if not found:
                break
            score = compute_c5(h)
            if score < best_score:
                best_score = score
    return h, best_score


def slsqp_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 600, 1500]
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return smooth_objective(x, alpha, n)
            def cfun(x):
                return np.sum(x) - target
            def cjac(x):
                return np.ones(n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
