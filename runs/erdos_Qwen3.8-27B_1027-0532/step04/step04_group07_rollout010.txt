def grad_of_corr(h, lag_idx):
    # Gradient of the correlation at a specific lag
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])

    n_corr = len(corr)

        top_idx = np.argpartition(corr, -topk)[-topk:]

    else:

    c_ref = np.max(top_vals)

def project_feasible(h, target_sum):

    else:

The Hessian of -corr_k is: for the term sum_i h[i+k]*h[i], the Hessian has a 1 at positions (i, i+k) and (i+k, i) for each i. This is a symmetric matrix. The objective -corr_k = -sum h[i+k] + sum h[i+k]*h[i], which has a Hessian that's the matrix from the quadratic term. So minimizing a concave function over a convex polytope is a non-convex problem in general. The minimum is attained at a vertex of the polytope (for a single constraint) or on the boundary.

4. Repeat until no improvement

To maintain the sum constraint when flipping, I need to flip two bits: one from 0→1 and one from 1→0. So the move is a "swap": h[i] += 1, h[j] -= 1 (or vice versa).

This is actually a combinatorial optimization problem. For n=400, there are O(n^2) possible swaps, so a full neighborhood search takes O(n^2 * n) = O(n^3) per iteration, which is 64M operations for n=400. That's feasible for a few iterations.

But actually, I think the continuous relaxation is more powerful. The binary solutions have C₅ = max_corr * dx where max_corr is an integer (for binary h), so C₅ is a multiple of dx = 0.005. The continuous relaxation can achieve values between these multiples.

import numpy as np
from scipy.optimize import minimize, linprog
import time

_NG = 0
_TG = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(200):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def _cf(x):
    return np.sum(x) - _TG

def _cj(x):
    return np.ones(_NG)

def smooth_obj(x, alpha, topk):
    corr = get_corr(x)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(nc)
        vals = corr
    cr = np.max(vals)
    sh = np.clip(alpha * (vals - cr), -500.0, 0.0)
    return cr + np.log(np.sum(np.exp(sh))) / alpha

def smooth_grad(x, alpha, topk):
    corr = get_corr(x)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(nc)
        vals = corr
    cr = np.max(vals)
    es = np.exp(np.clip(alpha * (vals - cr), -500.0, 0.0))
    w = es / es.sum()
    g = np.zeros(len(x))
    for i in range(len(idx)):
        if w[i] > 1e-14:
            g += w[i] * grad_lag(x, idx[i])
    return g

def do_slsqp(h0, target, alpha, topk, maxiter):
    global _NG, _TG
    _NG = len(h0)
    _TG = target
    bounds = [(0.0, 1.0)] * _NG
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(smooth_obj, h0, args=(alpha, topk),
                       jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h0

def fixed_topk_lp(h_cur, target, topk):
    n = len(h_cur)
    corr = get_corr(h_cur)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(nc)
    m = len(idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i in range(m):
        g = grad_lag(h_cur, idx[i])
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx[i]]
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target])
    bnds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs')
        if res.success:
            return project_feasible(res.x[:n], target)
    except Exception:
        pass
    return None

def adaptive_lp(h_cur, target, gap=0.3, max_cons=250):
    n = len(h_cur)
    corr = get_corr(h_cur)
    c_max = np.max(corr)
    near = np.where(corr >= c_max - gap)[0]
    top5 = np.argpartition(corr, -5)[-5:]
    idx = np.unique(np.concatenate([near, top5]))
    if len(idx) > max_cons:
        idx = np.argpartition(corr, -max_cons)[-max_cons:]
    m = len(idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i in range(m):
        g = grad_lag(h_cur, idx[i])
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx[i]]
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target])
    bnds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs')
        if res.success:
            return project_feasible(res.x[:n], target)
    except Exception:
        pass
    return None

def lp_loop(h_start, target, max_iter, time_limit, mode='adaptive'):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stag = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if mode == 'adaptive':
            gap = max(0.1, 0.5 - 0.4 * it / max_iter)
            h_lp = adaptive_lp(h, target, gap=gap)
        else:
            topk = int(5 + 60 * (it / max_iter))
            h_lp = fixed_topk_lp(h, target, topk)
        if h_lp is None:
            stag += 1
            if stag > 15:
                break
            continue
        score = compute_c5(h_lp)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stag = 0
        else:
            h_avg = project_feasible(0.5 * h + 0.5 * h_lp, target)
            sc = compute_c5(h_avg)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_avg.copy()
                h = h_avg
                stag = 0
            else:
                stag += 1
                if stag > 25:
                    break
    return best_h, best_score

def do_subgrad(h0, target, max_it, lr, topk, alpha, mom, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    stag = 0
    for it in range(max_it):
        corr = get_corr(h)
        nc = len(corr)
        if topk < nc:
            idx = np.argpartition(corr, -topk)[-topk:]
            vals = corr[idx]
        else:
            idx = np.arange(nc)
            vals = corr
        w = np.exp(alpha * (vals - vals.max()))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(idx)) / len(idx)
        else:
            w = w / ws
        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] > 1e-14:
                g += w[i] * grad_lag(h, idx[i])
        v = mom * v + (1.0 - mom) * g
        h = project_feasible(h - lr * v, target)
        score = compute_c5(h)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h.copy()
            stag = 0
            lr = min(lr * 1.01, 1.5)
        else:
            stag += 1
            lr *= 0.995
        if stag > patience:
            break
    return best_h, best_score

def interp_to(h_src, n_dst, tgt):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), tgt)
    xo = np.linspace(0, 1, len(h_src))
    xn = np.linspace(0, 1, n_dst)
    return project_feasible(np.interp(xn, xo, h_src), tgt)

def bit_flip_search(h_bin, target, max_rounds=50):
    n = len(h_bin)
    h = h_bin.copy()
    best_score = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        rng = np.random.RandomState(rnd)
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for _ in range(min(200, len(ones) * len(zeros) // 10)):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] -= 1.0
            h[j] += 1.0
            sc = compute_c5(h)
            if sc < best_score - 1e-15:
                best_score = sc
                improved = True
            else:
                h[i] += 1.0
                h[j] -= 1.0
        if not improved:
            break
    return h, best_score
