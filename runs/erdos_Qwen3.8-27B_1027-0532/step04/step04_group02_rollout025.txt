    if not np.allclose(h, np.round(h)):
        h = project_binary(h, target_sum)
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(5000, max_iters // 10)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    dx = 2.0 / n
    center = n - 1

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.03:
                no_improve = 0
            else:
                break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        move_type = rng.randint(4)
        h_new = h.copy()
        moved = False

        if move_type == 0 and len(ones) > 0 and len(zeros) > 0:
            i1 = ones[rng.randint(len(ones))]
            if rng.random() < 0.7:
                worst_idx = np.argmax(corr)
                k_star = worst_idx - center
                if k_star > 0:
                    target = i1 + k_star
                    if 0 <= target < n and h[target] == 0:
                        i0 = target
                    else:
                        dist = np.abs(zeros.astype(int) - target).astype(float)
                        dist += rng.random(len(zeros)) * 10.0
                        i0 = zeros[np.argmin(dist)]
                elif k_star < 0:
                    target = i1 - k_star
                    if 0 <= target < n and h[target] == 0:
                        i0 = target
                    else:
                        dist = np.abs(zeros.astype(int) - target).astype(float)
                        dist += rng.random(len(zeros)) * 10.0
                        i0 = zeros[np.argmin(dist)]
                else:
                    i0 = zeros[rng.randint(len(zeros))]
            else:
                i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True

        elif move_type == 1 and len(ones) > 0 and len(zeros) > 0:
            worst_indices = np.argsort(-corr)[:top_m]
            i1 = ones[rng.randint(len(ones))]
            best_score_val = 1e18
            best_i0 = zeros[0]
            num_candidates = min(30, len(zeros))
            cand_idx = rng.choice(len(zeros), num_candidates, replace=False)
            for ci in cand_idx:
                i0 = zeros[ci]
                score = 0.0
                for wi in worst_indices[:top_m]:
                    k = wi - center
                    j1 = i1 + k
                    j2 = i0 + k
                    val1 = 0.0
                    val2 = 0.0
                    if 0 <= j1 < n:
                        val1 = h[j1]
                    if 0 <= j2 < n:
                        val2 = h[j2]
                    delta = -val1 + (1.0 - val2) + (1.0 - val1) - (1.0 - val2)
                    j1b = i1 - k
                    j2b = i0 - k
                    val1b = 0.0
                    val2b = 0.0
                    if 0 <= j1b < n:
                        val1b = h[j1b]
                    if 0 <= j2b < n:
                        val2b = h[j2b]
                    delta += -val1b + (1.0 - val2b) + (1.0 - val1b) - (1.0 - val2b)
                    score += max(0.0, corr[wi] + delta * dx - best_c5)
                if score < best_score_val:
                    best_score_val = score
                    best_i0 = i0
            h_new[i1] = 0
            h_new[best_i0] = 1
            moved = True

        elif move_type == 2 and len(ones) >= 2 and len(zeros) >= 2:
            idx1 = rng.choice(len(ones), 2, replace=False)
            i1a = ones[idx1[0]]
            i1b = ones[idx1[1]]
            if rng.random() < 0.6:
                worst_idx = np.argmax(corr)
                k_star = worst_idx - center
                t1 = (i1a + k_star) % n
                t2 = (i1b - k_star) % n
                d1 = np.abs(zeros.astype(int) - t1).astype(float) + rng.random(len(zeros)) * 8.0
                d2 = np.abs(zeros.astype(int) - t2).astype(float) + rng.random(len(zeros)) * 8.0
                i0a = zeros[np.argmin(d1)]
                i0b = zeros[np.argmin(d2)]
                if i0a == i0b:
                    i0b = zeros[rng.randint(len(zeros))]
            else:
                idx0 = rng.choice(len(zeros), 2, replace=False)
                i0a = zeros[idx0[0]]
                i0b = zeros[idx0[1]]
            h_new[i1a] = 0
            h_new[i1b] = 0
            h_new[i0a] = 1
            h_new[i0b] = 1
            moved = True

        elif move_type == 3 and len(ones) > 0 and len(zeros) > 0:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5

        if delta < -1e-14 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5


def top_p_smooth(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_optimize(h_start, n, target, p_vals, alphas, maxiter, ftol,
                   time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(lambda x, a=alpha, p=p: top_p_smooth(x, a, p), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, curr_score


def local_search_binary(h, target_sum, rng, max_iters, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    dx = 2.0 / n
    center = n - 1
    stall = 0
    max_stall = 200

    for _ in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if stall > max_stall:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        worst_idx = np.argmax(corr)
        k_star = worst_idx - center

        i1 = ones[rng.randint(len(ones))]
        if k_star > 0:
            target_pos = i1 + k_star
        elif k_star < 0:
            target_pos = i1 - k_star
        else:
            target_pos = i1

        dist = np.abs(zeros.astype(int) - target_pos).astype(float)
        dist += rng.random(len(zeros)) * 5.0
        i0 = zeros[np.argmin(dist)]

        h[i1] = 0
        h[i0] = 1
        corr_new = compute_corr(h)
        c5_new = float(np.max(corr_new))
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            corr = corr_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[i1] = 1
            h[i0] = 0
            stall += 1

    return h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()

    for si in range(n_starts):

        if remaining < 20:
            break
        h_s = starts[si]
        allowed = min(sa_time_per_start, remaining - 10)
        if allowed < 5:
            break
        max_it = int(min(200000, allowed * 3000))
        t_init_val = 0.008
        t_final_val = 0.00003
        h_sa, c5_sa = fast_sa(h_s, target_init, max_it, t_init_val, t_final_val,
                              rng, time_limit=allowed, start_time=start_time + elapsed)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

        remaining = budget_s * 0.65 - (time.time() - start_time)
        if remaining > 20:
            h_ls, c5_ls = local_search_binary(h_sa, target_init, rng, 10000,
                                              time_limit=min(15.0, remaining - 5),
                                              start_time=start_time)
            if c5_ls < best_score:
                best_score = c5_ls
                best_h = h_ls.copy()

    elapsed = time.time() - start_time
    remaining_for_multi = budget_s * 0.85 - elapsed
    if remaining_for_multi > 60:
        n_candidates = []
        for nf in [0.75, 0.85, 1.15, 1.3, 1.5]:
            n_try = int(n_init * nf)
            if n_try % 2 == 0 and n_try != n_init and n_try >= 200 and n_try <= 2000:
                n_candidates.append(n_try)
        n_candidates = sorted(set(n_candidates))
        time_per_n = remaining_for_multi / max(len(n_candidates), 1)
        for n_try in n_candidates:
            elapsed = time.time() - start_time
            if time.time() - start_time > budget_s * 0.85:
                break
            allowed = min(80.0, (budget_s * 0.85 - elapsed) - 10)
            if allowed < 10:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin_try = project_binary(h_interp, target_try)
            max_it = int(min(100000, allowed * 2500))
            h_sa, c5_sa = fast_sa(h_bin_try, target_try, max_it, 0.006, 0.00003,
                                  rng, time_limit=allowed, start_time=start_time)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0
    elapsed = time.time() - start_time

    if elapsed < budget_s * 0.9:
        allowed = min(budget_s * 0.9 - elapsed, 150.0)
        if allowed > 30:
            p_vals = [max(3, n_use // 40), max(5, n_use // 25), max(10, n_use // 15)]
            alphas = [80, 200, 500, 1200]
            h_ref, sc = slsqp_optimize(best_h, n_use, target_use, p_vals, alphas,
                                       500, 1e-14, time_limit=allowed, start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            n_use = len(best_h)
            target_use = n_use / 2.0

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.95:
        allowed = min(budget_s * 0.95 - elapsed, 60.0)
        if allowed > 10:
            h_bin_final = project_binary(best_h, target_use)
            max_it = int(max(5000, allowed * 2000))
            h_sa_f, c5_f = fast_sa(h_bin_final, target_use, max_it, 0.003, 0.00001,
                                   rng, time_limit=allowed, start_time=start_time)
            if c5_f < best_score:
                best_score = c5_f
                best_h = h_sa_f

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.98:
        n_use = len(best_h)
        target_use = n_use / 2.0
        allowed = min(budget_s * 0.98 - elapsed, 30.0)
        if allowed > 5:
            h_ls2, c5_ls2 = local_search_binary(best_h, target_use, rng,
                                                int(allowed * 3000),
                                                time_limit=allowed, start_time=start_time)
            if c5_ls2 < best_score:
                best_score = c5_ls2
                best_h = h_ls2

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-14:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
