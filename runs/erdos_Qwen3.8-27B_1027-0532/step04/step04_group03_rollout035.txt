def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def smoothed_obj_and_grad(h, alpha, mu, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / alpha
    penalty = mu * (np.sum(h) - target) ** 2
    obj += penalty
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights = weights / w_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        m = k - (n - 1)
        if m >= 0:
            wm = w * dx
            grad[m:] += wm * (1.0 - h[:n - m])
            grad[:n - m] -= wm * h[m:]
        else:
            mm = -m
            wm = w * dx
            grad[:n - mm] += wm * (1.0 - h[mm:])
            grad[mm:] -= wm * h[:n - mm]
    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - target)
    return obj, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:

            h[zeros_cache[idx0[0]]] = 1

        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:

            if h[i0_pos] == 1:

            h[i1] = 0
            h[i0_pos] = 1
            moved = True
        elif move_type == 3 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:

def lbfgsb_refine(h_start, target, time_limit, start_time, alpha_schedule=None, mu_base=500.0):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    if alpha_schedule is None:
        alpha_schedule = [20, 50, 100, 200, 400, 800, 1500, 3000, 5000]

    for alpha in alpha_schedule:
        if (time.time() - start_time) > time_limit - 10:
            break

        def obj_fn(x):
            o, _ = smoothed_obj_and_grad(x, alpha, mu_base, target)
            return o

        def grad_fn(x):
            _, g = smoothed_obj_and_grad(x, alpha, mu_base, target)
            return g

        try:
            result, fflag, info = fmin_l_bfgs_b(
                obj_fn, h, fprime=grad_fn,
                pgtol=1e-12, gtol=1e-10, maxiter=500, maxfun=2000,
                iprint=0, pgtol=1e-12
            )
            h_new = project_feasible(result, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    return best_h, best_c5


def hill_climb_binary(h, target_sum, max_iter, rng):

def weighted_perturb(h_start, target, time_limit, start_time, rng):

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
