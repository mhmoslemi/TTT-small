        for ii in range(len(active_idx)):
            if weights[ii] < 1e-10:
                continue
            s = int(active_idx[ii]) - (n - 1)
            w = weights[ii]
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
            else:
                m = -s
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def simulated_annealing(h, target, n_rounds, rng, cooling_rate=0.95):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    temp = 0.01
    for rnd in range(n_rounds):
        temp *= cooling_rate
        for _ in range(3):
            idx = rng.randint(n)
            delta = rng.uniform(-0.05, 0.05)
            h_try = h.copy()
            h_try[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            score_try = compute_c5(h_try)
            if score_try < best_score or rng.random() < np.exp(-(score_try - best_score) / max(temp, 1e-10)):
                h = h_try
                if score_try < best_score:
                    best_score = score_try
                    best_h = h_try.copy()
        if temp < 1e-6:
            break
    return best_h, best_score


def coordinate_descent(h, target, n_rounds, step_sizes):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for coord in range(n):
            for ds in step_sizes:
                h_try = h.copy()
                h_try[coord] = np.clip(h[coord] + ds, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Periodic initialization
    periods = [20, 25, 40, 50]
    for p in periods:
        if time_left() < 150:
            break
        if n % p != 0:
            continue
        h_peri, sc_peri = optimize_periodic(p, n, n_restarts=10, max_iter=600)
        if h_peri is not None and sc_peri < best_score:
            best_score = sc_peri
            best_h = h_peri

    # Phase 2: L-BFGS-B with increasing alpha
    alpha_schedule = [50, 100, 200, 400, 800, 1500, 3000]
    for alpha in alpha_schedule:
        if time_left() < 60:
            break
        for attempt in range(2):
            if time_left() < 30:
                break
            noise = 0.004 if attempt == 0 else 0.001
            h_start = best_h + rng.randn(len(best_h)) * noise
            h_start = project_feasible(h_start, target)
            n_cur = len(h_start)
            t_cur = n_cur / 2.0

            def obj_func(x):
                f, g = lse_obj_grad(x, alpha, topk=60)
                s = np.sum(x) - t_cur
                f += 5000.0 * s * s
                g = g + 10000.0 * s
                return f, g

            try:
                maxiter = 600 if alpha < 500 else 400
                res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                              bounds=[(0, 1)] * n_cur,
                              options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
                h_candidate = project_feasible(res.x, t_cur)
                score = compute_c5(h_candidate)
                if score < best_score:
                    best_score = score
                    best_h = h_candidate
            except Exception:
                pass

    # Phase 3: Subgradient polish on true max
    if time_left() > 80:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        best_h, sc = subgradient_polish(best_h, t_cur, n_steps=150)
        if sc < best_score:
            best_score = sc

    # Phase 4: Try higher resolution
    for n_try in [n + 100, n + 200]:
        if time_left() < 200:
            break
        t_try = n_try / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, t_try)
        h_try_best = h_interp
        sc_try = compute_c5(h_interp)

        for alpha in [200, 1000]:
            if time_left() < 60:
                break
            def obj_try(x):
                f, g = lse_obj_grad(x, alpha, topk=60)
                s = np.sum(x) - t_try
                f += 5000.0 * s * s
                g = g + 10000.0 * s
                return f, g
            try:
                res = minimize(obj_try, h_try_best + rng.randn(n_try) * 0.002,
                              jac=True, method='L-BFGS-B',
                              bounds=[(0, 1)] * n_try,
                              options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12})
                h_cand = project_feasible(res.x, t_try)
                sc = compute_c5(h_cand)
                if sc < sc_try:
                    sc_try = sc
                    h_try_best = h_cand
            except Exception:
                pass

        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try_best

    # Phase 5: Another round of L-BFGS at current best resolution
    if time_left() > 80:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        for alpha in [500, 2000, 5000]:
            if time_left() < 30:
                break
            h_start = best_h + rng.randn(n_cur) * 0.001
            h_start = project_feasible(h_start, t_cur)

            def obj_final(x):
                f, g = lse_obj_grad(x, alpha, topk=30)
                s = np.sum(x) - t_cur
                f += 10000.0 * s * s
                g = g + 20000.0 * s
                return f, g

            try:
                res = minimize(obj_final, h_start, jac=True, method='L-BFGS-B',
                              bounds=[(0, 1)] * n_cur,
                              options={'maxiter': 400, 'ftol': 1e-17, 'gtol': 1e-13})
                h_cand = project_feasible(res.x, t_cur)
                sc = compute_c5(h_cand)
                if sc < best_score:
                    best_score = sc
                    best_h = h_cand
            except Exception:
                pass

    # Phase 6: Subgradient polish again
    if time_left() > 60:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        best_h, sc = subgradient_polish(best_h, t_cur, n_steps=100)
        if sc < best_score:
            best_score = sc

    # Phase 7: Coordinate descent
    if time_left() > 40:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        best_h, sc = coordinate_descent(best_h, t_cur, n_rounds=3,
                                        step_sizes=[0.005, -0.005, 0.002, -0.002, 0.001, -0.001])
        if sc < best_score:
            best_score = sc

    # Phase 8: Simulated annealing
    if time_left() > 50:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        best_h, sc = simulated_annealing(best_h, t_cur, n_rounds=200, rng=rng)
        if sc < best_score:
            best_score = sc

    # Phase 9: Final L-BFGS
    if time_left() > 25:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0

        def obj_last(x):
            f, g = lse_obj_grad(x, 5000, topk=20)
            s = np.sum(x) - t_cur
            f += 20000.0 * s * s
            g = g + 40000.0 * s
            return f, g
        try:
            res = minimize(obj_last, best_h, jac=True, method='L-BFGS-B',
                          bounds=[(0, 1)] * n_cur,
                          options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14})
            h_cand = project_feasible(res.x, t_cur)
            sc = compute_c5(h_cand)
            if sc < best_score:
                best_score = sc
                best_h = h_cand
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
