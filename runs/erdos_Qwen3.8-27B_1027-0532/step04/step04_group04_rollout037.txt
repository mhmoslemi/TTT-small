Key insight for efficiency: I precompute g, and after each swap I update g in O(n) time. Then I can evaluate any candidate swap in O(n) time (just g[j,:]-g[i,:] and check max).

For the local search, the main loop is:
1. Find the current max of c and the lag(s) achieving it
2. Generate candidate swaps (i,j)
3. For each candidate, compute delta_c = g[j,:]-g[i,:] and check if max(c+delta_c) < current_max
4. Apply the best improving swap
5. Update c and g
6. Repeat until no improving swap found or time limit

For step 3, the check "max(c+delta_c) < current_max" can be made faster by only checking the active lags (where c is close to its max). If we define active set A = {k : c[k] > current_max - ε}, then we need c[k] + delta_c[k] < current_max for all k in A, i.e., delta_c[k] < current_max - c[k] for all k in A.

If the active set is small (say < 10 lags), this check is very fast.

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.clip(np.round(h), 0, 1).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r

def compute_g_full(h):
    n = len(h)
    g = np.zeros((n, 2*n-1), dtype=float)
    g[:, n-1] = 1.0 - 2.0 * h
    for d in range(1, n):
        g[d:, d+n-1] += (1.0 - h[:n-d])
        g[:n-d+1, d+n-1] += (-h[d:])
        g[:n-d+1, n-1-d] += (1.0 - h[d:])
        g[d:, n-1-d] += (-h[:n-d])
    return g

def update_g_swap(g, n, i, j):
    """Update g in-place after swap h[i]:1->0, h[j]:0->1."""
    # h[i] changes by -1, effect is -(-1)=+1 on g
    # g[i+lag, lag_col] += 1 for lag in [-i, n-1-i]
    lags1 = np.arange(-i, n-i)
    g[i + lags1, lags1 + n - 1] += 1.0
    # g[i-lag, lag_col] += 1 for lag in [-(n-1-i), i]
    lags2 = np.arange(-(n-1-i), i+1)
    g[i - lags2, lags2 + n - 1] += 1.0
    # h[j] changes by +1, effect is -(+1)=-1 on g
    lags3 = np.arange(-j, n-j)
    g[j + lags3, lags3 + n - 1] -= 1.0
    lags4 = np.arange(-(n-1-j), j+1)
    g[j - lags4, lags4 + n - 1] -= 1.0

def update_c_swap(c, n, i, j, g):
    """Compute new c after swap, using g (before update)."""
    # delta_c = g[j,:] - g[i,:]
    delta = g[j] - g[i]
    new_c = c + delta
    return new_c

def binary_local_search(h, n, target, rng, time_limit, start_time):

    h = h.copy()

    g = compute_g_full(h)
    current_max = float(c.max()) * (2.0 / n)
    best_max = current_max
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    max_stall = 20000

            break

            break

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.clip(np.round(h), 0, 1).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r

def compute_g_full(h):
    n = len(h)
    g = np.zeros((n, 2 * n - 1), dtype=float)
    g[:, n - 1] = 1.0 - 2.0 * h
    for d in range(1, n):
        g[d:, d + n - 1] += (1.0 - h[:n - d])
        g[:n - d + 1, d + n - 1] += (-h[d:])
        g[:n - d + 1, n - 1 - d] += (1.0 - h[d:])
        g[d:, n - 1 - d] += (-h[:n - d])
    return g

def update_g_swap(g, n, i, j):
    lags1 = np.arange(-i, n - i)
    g[i + lags1, lags1 + n - 1] += 1.0
    lags2 = np.arange(-(n - 1 - i), i + 1)
    g[i - lags2, lags2 + n - 1] += 1.0
    lags3 = np.arange(-j, n - j)
    g[j + lags3, lags3 + n - 1] -= 1.0
    lags4 = np.arange(-(n - 1 - j), j + 1)
    g[j - lags4, lags4 + n - 1] -= 1.0

def binary_ls_focused(h, n, target, rng, time_limit, start_time):
    h = h.copy()
    c = compute_corr(h)
    g = compute_g_full(h)
    current_max_raw = float(c.max())
    best_max = current_max_raw * (2.0 / n)
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 8000

    for it in range(10000000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        c_max_raw = float(c.max())
        active = np.where(c >= c_max_raw - 1.0)[0]

        found = False
        if len(ones) > 0 and len(zeros) > 0 and len(active) > 0:
            ones_scores = np.sum(g[ones][:, active], axis=1)
            zeros_scores = np.sum(g[zeros][:, active], axis=1)

            top_j = np.argsort(zeros_scores)[:30]

                    j = zeros[jj]

                        if new_c.max() < c_max_raw:

                            h[j] = 1
                            c = new_c
                            update_g_swap(g, n, i, j)

                            zeros = np.where(h == 0)[0]

                            stall = 0

def sa_binary_fast(h, n, target, rng, t_init, t_final, time_limit, start_time):

    g = compute_g_full(h)

            break

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
