def fast_correlate(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Pad to next power of 2
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.fft(a, fsize)
    B = np.fft.fft(b, fsize)
    # Correlation: c[k] = sum_n a[n+k] * b[n]

    # conv(a, b_flipped)[k] = sum_n a[n] * b_flipped[k-n] = sum_n a[n] * b[k-n]
    # For correlation: c[k] = sum_n a[n+k] * b[n] = sum_m a[m] * b[m-k]
    # = conv(a, flip(b))[k + m - 1]
    B_flip = np.fft.fft(b[::-1], fsize)
    C = np.fft.ifft(A * B_flip)
    return C[:size].real

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient_lse(h, alpha):
    """Compute exact gradient of the LSE-smoothed max of correlation."""
    n = len(h)
    corr = compute_corr(h)
    # Compute softmax weights
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # grad[j] += w * [(1-h[j]) * I(lag <= j <= n-1) - h[j+lag] * I(0 <= j <= n-1-lag)]
            lo = lag
            hi = n - 1
            # First part: (1-h[j]) for j in [lag, n-1]
            if lo <= hi:
                grad[lo:hi+1] += w * (1.0 - h[lo:hi+1])
            # Second part: -h[j+lag] for j in [0, n-1-lag]
            lo2 = 0
            hi2 = n - 1 - lag
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= w * h[lo2+lag:hi2+1+lag]
        else:
            l = -lag
            # grad[j] += w * [(1-h[j+l]) * I(0 <= j <= n-1-l) - h[j-l] * I(l <= j <= n-1)]
            lo2 = 0
            hi2 = n - 1 - l
            if lo2 <= hi2:
                grad[lo2:hi2+1] += w * (1.0 - h[lo2+l:hi2+1+l])
            lo = l
            hi = n - 1
            if lo <= hi:
                grad[lo:hi+1] -= w * h[lo-l:hi+1-l]
    
    return grad

def lse_smoothed_max(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def projected_gradient_descent(h_init, target, alpha, n_iters, lr_init=0.01, lr_decay=0.999, threshold_t=0.0):
    """Projected gradient descent with LSE smoothing and optional soft thresholding."""
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    
    for it in range(n_iters):
        # Compute gradient
        grad = compute_gradient_lse(h, alpha)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Soft thresholding (push toward 0 or 1)
        if threshold_t > 0:
            h_new = np.where(h_new > 0.5, h_new + threshold_t, h_new - threshold_t)
        
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h = h_new
        lr *= lr_decay
        if lr < 1e-10:
            break
    
    return best_h, best_score

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)

    n = len(h)
    corr = compute_corr(h)

    """Projected gradient descent with LSE smoothing and alpha annealing."""
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        lr = lr_init
        for it in range(iters_per_alpha):
            if deadline is not None and time.time() > deadline:
                return best_h, best_score
            grad = compute_gradient_lse(h, alpha)
            # Project gradient to preserve sum
            grad = grad - np.mean(grad)
            
            h_new = h - lr * grad
            if threshold_t > 0:

            h_new = project_feasible(h_new, target)

                best_h = h_new.copy()
            h = h_new
            lr *= 0.9995
            if lr < 1e-8:

    """Discrete pair-flip local search."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n = len(h)
