    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:

            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

```python
def compute_swap_delta(h, i0, j0, n):
    """Compute the change in correlation when h[i0] goes 1->0 and h[j0] goes 0->1.
    Returns an array of length 2n-1 (indexed by m=0..2n-2, corresponding to k=m-(n-1))."""
    delta = np.zeros(2*n-1)
    # Affected i values for a given k: i0, j0, i0-k, j0-k
    # For each such i to be valid: 0<=i<n and 0<=i+k<n
    # Equivalently: 0<=i<n and -i<=k<n-i
    
    # i0 as first element: valid for k in [-i0, n-1-i0]
    # i0 as second element (i+k=i0, so i=i0-k): valid for k in [i0-(n-1), i0] = [i0-n+1, i0]
    # j0 as first element: valid for k in [-j0, n-1-j0]
    # j0 as second element (i+k=j0, so i=j0-k): valid for k in [j0-n+1, j0]
    
    # For each k, the affected terms are subset of {i0, j0, i0-k, j0-k}
    # We need to compute the change for each affected (i, i+k) pair
    
    # Let's iterate over the k ranges where delta is non-zero
    # k range for i0 (first): [-i0, n-1-i0]
    # k range for i0 (second): [i0-n+1, i0]
    # k range for j0 (first): [-j0, n-1-j0]
    # k range for j0 (second): [j0-n+1, j0]
    
    # Union of these ranges
    k_min = max(-(n-1), max(-i0, i0-n+1, -j0, j0-n+1))
    k_max = min(n-1, max(n-1-i0, i0, n-1-j0, j0))
    
    for k in range(k_min, k_max + 1):
        m = k + (n-1)  # index in the correlation array
        d = 0.0
        # Check each affected i
        for i in set([i0, j0, i0-k, j0-k]):
            if 0 <= i < n and 0 <= i+k < n:
                ik = i + k
                old_val = h[i] * (1.0 - h[ik])
                # Compute new h[i] and h[ik]
                new_i = h[i]
                new_ik = h[ik]
                if i == i0:
                    new_i = 0.0
                elif i == j0:
                    new_i = 1.0
                if ik == i0:
                    new_ik = 0.0
                elif ik == j0:
                    new_ik = 1.0
                new_val = new_i * (1.0 - new_ik)
                d += new_val - old_val
        delta[m] = d
    return delta
```

This is O(n) per swap but with a small constant (at most 4 set elements per k). For n=600, this is about 600*4 = 2400 operations per swap evaluation.

OK so the correct algorithm is:
1. For each k, identify the set of affected i values: {i ∈ {i0, j0, i0-k, j0-k} : 0<=i<n and 0<=i+k<n}

4. Sum the changes

The key optimization: for most k, the set {i0, j0, i0-k, j0-k} has at most 2 valid elements (and they're distinct). Only for k = j0-i0 or k = i0-j0 do we get the "interaction" case where the same i appears from two sources.

```python
def compute_swap_delta(h, i0, j0, n):
    delta = np.zeros(2*n-1)
    # k ranges where delta might be non-zero
    # i0 valid as first: k in [-i0, n-1-i0]
    # i0-k = i valid: k in [i0-(n-1), i0]  
    # j0 valid as first: k in [-j0, n-1-j0]
    # j0-k = i valid: k in [j0-(n-1), j0]
    
    k_lo = max(-(n-1), -i0, i0-n+1, -j0, j0-n+1)
    k_hi = min(n-1, n-1-i0, i0, n-1-j0, j0)
    
    for k in range(k_lo, k_hi+1):
        m = k + n - 1
        d = 0.0
        # affected i values
        candidates = []
        if 0 <= i0 < n and 0 <= i0+k < n:
            candidates.append(i0)
        if 0 <= j0 < n and 0 <= j0+k < n:
            candidates.append(j0)
        if 0 <= i0-k < n and 0 <= i0 < n:
            candidates.append(i0-k)
        if 0 <= j0-k < n and 0 <= j0 < n:
            candidates.append(j0-k)
        
        for i in set(candidates):
            ik = i + k
            old_h_i = h[i]
            old_h_ik = h[ik]
            new_h_i = old_h_i
            new_h_ik = old_h_ik
            if i == i0: new_h_i = 0.0
            elif i == j0: new_h_i = 1.0
            if ik == i0: new_h_ik = 0.0
            elif ik == j0: new_h_ik = 1.0
            d += new_h_i * (1.0 - new_h_ik) - old_h_i * (1.0 - old_h_ik)
        delta[m] = d
    return delta
```

This is O(n) per swap with a constant factor of about 4-8. For n=600, that's about 2400-4800 operations. In Python, this might be about 1-5 microseconds per operation, so maybe 5-20 microseconds per swap evaluation. That means we can evaluate about 50,000-200,000 swaps per second.

    length = 2*n - 1
    delta = np.zeros(length)
    
    # k range for non-zero delta
    k_lo = max(-(n-1), -i0, i0-n+1, -j0, j0-n+1)
    k_hi = min(n-1, n-1-i0, i0, n-1-j0, j0)
    if k_lo > k_hi:
        return delta
    
    ks = np.arange(k_lo, k_hi+1)

    # Valid when 0 <= i0+k < n, i.e., k in [-i0, n-1-i0]
    valid_i0_first = (ks >= -i0) & (ks <= n-1-i0)

        # So actually the formula gives the right answer even at the interaction point!

    # Contribution from i0 as second (i=i0-k): valid for k in [max(k_lo, i0-n+1), min(k_hi, i0)]

        tmp[mask_j0s] = -h[idx]
    delta[ms] += tmp

    # At k = j0-i0: i0 is counted as both "first" (i=i0) and "second from j0" (i=j0-k=i0)
    # The double-counted term is: -(1-h[i0+k]) + (-h[j0-k]) = -(1-h[j0]) + (-h[i0]) = -1 + (-1) = -2

    # The double-counted term is: +(1-h[j0+k]) + h[i0-k] = +(1-h[i0]) + h[j0] = 0 + 0 = 0
    # Wait: 1-h[i0] = 1-1 = 0, and h[j0] = 0. So the formula gives 0.
    # Actual change: h[j0]*(1-h[i0]) old = 0*0 = 0, new = 1*1 = 1. Change = +1.

            if (i0-j0) >= i0-n+1 and (i0-j0) <= i0:  # i0 second is valid (i=i0-k=j0)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

- For the k* that achieves the max:

  - dc[k*]/dh[i] = -h[i-k*] if i-k* is valid (i appears as second element)
  - (sum of both if both are valid)

So the subgradient step is:
h[i] ← h[i] - η * sum_{k in argmax} (subgradient at k) * dx

from scipy.optimize import minimize
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def eval_swap(h, corr, i0, j0, n):
    """Evaluate the effect of swapping h[i0]:1->0, h[j0]:0->1.
    Returns the new max correlation value."""
    # Compute delta efficiently
    # delta[k] for k in relevant range
    # The affected k range:
    # i0 first: k in [-i0, n-1-i0]
    # j0 first: k in [-j0, n-1-j0]  
    # i0 second: k in [i0-n+1, i0]
    # j0 second: k in [j0-n+1, j0]
    
    # For efficiency, compute new_max = max(corr + delta)
    # delta is non-zero only in the union of the above ranges
    
    # Quick check: if the current max is at a position unaffected by the swap,
    # then new_max >= current_max (since other values might increase)
    # But we need to check if the affected positions exceed current_max
    
    # Let's compute the max of corr outside the affected range (should be <= current_max)
    # and the max of corr+delta inside the affected range
    
    # Affected range in terms of m (index in corr array, m = k + n - 1):
    # i0 first: m in [n-1-i0, n-1+n-1-i0] = [n-1-i0, 2n-2-i0]
    # j0 first: m in [n-1-j0, 2n-2-j0]
    # i0 second: m in [n-1+i0-n+1, n-1+i0] = [i0, n-1+i0]
    # j0 second: m in [j0, n-1+j0]
    
    m_lo = max(0, n-1-max(i0,j0), min(i0,j0))  # this is getting complicated
    # Let me just compute it differently
    
    # The affected m range:
    # From i0 first: m in [n-1-i0, 2n-2-i0]
    # From j0 first: m in [n-1-j0, 2n-2-j0]
    # From i0 second: m in [i0, n-1+i0]
    # From j0 second: m in [j0, n-1+j0]
    
    m_min = max(0, min(n-1-i0, n-1-j0, i0, j0))
    m_max = min(2*n-2, max(2*n-2-i0, 2*n-2-j0, n-1+i0, n-1+j0))
    
    # Compute delta for m in [m_min, m_max]
    ks = np.arange(m_min, m_max+1) - (n-1)
    delta = np.zeros(len(ks))
    
    # i0 first: valid when 0 <= i0+k < n
    mask = (i0 + ks >= 0) & (i0 + ks < n)
    delta[mask] -= (1.0 - h[i0 + ks[mask]])
    
    # j0 first: valid when 0 <= j0+k < n
    mask = (j0 + ks >= 0) & (j0 + ks < n)
    delta[mask] += (1.0 - h[j0 + ks[mask]])
    
    # i0 second: valid when 0 <= i0-k < n
    mask = (i0 - ks >= 0) & (i0 - ks < n)
    delta[mask] += h[i0 - ks[mask]]
    
    # j0 second: valid when 0 <= j0-k < n
    mask = (j0 - ks >= 0) & (j0 - ks < n)
    delta[mask] -= h[j0 - ks[mask]]
    
    # Correct for double-counting at interaction points
    # k = j0-i0: i0 first and j0 second refer to same term
    k_int1 = j0 - i0
    m_int1 = k_int1 + n - 1
    if m_min <= m_int1 <= m_max:
        # Check if both conditions are met
        if (i0 + k_int1 >= 0) and (i0 + k_int1 < n):  # i0 first valid
            if (j0 - k_int1 >= 0) and (j0 - k_int1 < n):  # j0 second valid (j0-k=i0)
                idx_in_range = m_int1 - m_min
                delta[idx_in_range] += 1.0
    
    # k = i0-j0: j0 first and i0 second refer to same term
    k_int2 = i0 - j0
    m_int2 = k_int2 + n - 1
    if m_min <= m_int2 <= m_max:
        if (j0 + k_int2 >= 0) and (j0 + k_int2 < n):  # j0 first valid
            if (i0 - k_int2 >= 0) and (i0 - k_int2 < n):  # i0 second valid (i0-k=j0)
                idx_in_range = m_int2 - m_min
                delta[idx_in_range] += 1.0
    
    # Now compute new max
    # corr[m_min:m_max+1] + delta
    affected_max = float(np.max(corr[m_min:m_max+1] + delta))
    # Unaffected max
    if m_min > 0:
        unaffected_left = float(np.max(corr[:m_min]))
    else:
        unaffected_left = 0.0
    if m_max < 2*n-2:
        unaffected_right = float(np.max(corr[m_max+1:]))
    else:
        unaffected_right = 0.0
    unaffected_max = max(unaffected_left, unaffected_right)
