f(x) = log(Σ exp(α * c_i) / p) / α for the top p values

Or even simpler: f(x) = mean of top p + σ * std of top p

This is a soft-max of the top p values, which is a smooth approximation of max(top p).

For the targeted approach, I want to find i0 (a zero position) that, when we move h[i1]=1 to h[i0]=0→1, reduces the maximum correlation.

The key insight: Δc[k] = h[i1+k] + h[i1-k] - h[i0+k] - h[i0-k]

For the worst lag k*, we want Δc[k*] < 0, i.e.:
h[i1+k*] + h[i1-k*] < h[i0+k*] + h[i0-k*]

So we want i0 to be a position where h[i0+k*] + h[i0-k*] is large (many ones at distance k* from i0).

A good heuristic:
1. For a given i1, compute s1 = h[i1+k*] + h[i1-k*] (bounded, at most 2)
2. For each candidate i0, compute s0 = h[i0+k*] + h[i0-k*]
3. Prefer i0 with s0 > s1

But we also need to check that the move doesn't increase other lags above the current max. So we should:
1. Generate a few top candidates based on the heuristic above
2. For each candidate, compute the full delta and check if max(c + delta) < current_max
3. Pick the best one

This gives a good balance between speed and quality.

def top_p_smooth(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

        allowed_time = min(200.0, tleft() * 0.3)
        max_it = int(min(2000000, allowed_time * 15000))
        t_init_val = sa_params_list[min(si, len(sa_params_list) - 1)][0]
        t_final_val = sa_params_list[min(si, len(sa_params_list) - 1)][1]
        h_sa, c5_sa = informed_sa_fast(h_s, target_init, max_it, t_init_val, t_final_val,
                                       rng, time_limit=allowed_time, start_time=start_time)
        if tleft() > 40:
            h_hc, c5_hc = quick_hill_climb_fast(h_sa, target_init, 10000, rng)
            if c5_hc < c5_sa:
                h_sa = h_hc
                c5_sa = c5_hc
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 100:
        p_vals = [max(3, n_init // 40), max(5, n_init // 25), max(8, n_init // 15)]
        alphas = [100, 300, 800, 2000]
        h_ref, sc = slsqp_refine(best_h, n_init, target_init, p_vals, alphas,
                                 800, 1e-14, time_limit=max(10.0, tleft() - 80), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()
            if tleft() > 60:
                h_sa2, c5_sa2 = informed_sa_fast(project_binary(best_h, target_init),
                                                 target_init,
                                                 int(min(1000000, max(10.0, tleft() - 50) * 15000)),
                                                 0.005, 0.0001, rng,
                                                 time_limit=max(5.0, tleft() - 40),
                                                 start_time=start_time)
                if c5_sa2 < best_score:
                    best_score = c5_sa2
                    best_h = h_sa2.copy()

    n_candidates = []
    for nf in [0.6, 0.75, 0.9, 1.1, 1.25, 1.5, 1.75]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try != n_init and n_try >= 200:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed_time = min(150.0, tleft() * 0.3)
        max_it = int(min(2000000, allowed_time * 12000))
        h_sa, c5_sa = informed_sa_fast(h_bin_try, target_try, max_it, 0.012, 0.0003,
                                       rng, time_limit=allowed_time, start_time=start_time)
        if tleft() > 40:
            h_hc, c5_hc = quick_hill_climb_fast(h_sa, target_try, 8000, rng)
            if c5_hc < c5_sa:
                h_sa = h_hc
                c5_sa = c5_hc
        if tleft() > 50:
            p_vals_r = [max(3, n_try // 35), max(5, n_try // 20)]
            alphas_r = [200, 600, 1500]
            h_ref_r, sc_r = slsqp_refine(h_sa, n_try, target_try, p_vals_r, alphas_r,
                                         500, 1e-14, time_limit=max(10.0, tleft() - 40),
                                         start_time=start_time)
            if sc_r < c5_sa:
                h_sa = h_ref_r
                c5_sa = sc_r
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        p_vals = [max(3, n_use // 30), max(5, n_use // 20), max(10, n_use // 12)]
        alphas = [200, 500, 1200, 3000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, p_vals, alphas,
                                 1000, 1e-15, time_limit=max(10.0, tleft() - 60), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    if tleft() > 50:
        alphas_sm = [500, 1500, 4000, 8000]
        h_ref2, sc2 = slsqp_softmax_refine(best_h, n_use, target_use, alphas_sm,
                                           600, 1e-15, time_limit=max(5.0, tleft() - 30),
                                           start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    n_refine_list = []
    for nf in [0.85, 1.1, 1.3, 1.55]:
        n_r = int(n_use * nf)
        if n_r % 2 == 0 and n_r != n_use and n_r >= 200:
            n_refine_list.append(n_r)

    for n_r in n_refine_list:
        if tleft() < 80:
            break
        target_r = n_r / 2.0
        h_interp = interpolate_to(best_h, n_r, target_r)
        h_bin_r = project_binary(h_interp, target_r)
        allowed = min(80.0, tleft() * 0.25)
        max_it = int(min(1500000, allowed * 12000))
        h_sa_r, c5_r = informed_sa_fast(h_bin_r, target_r, max_it, 0.008, 0.0002,
                                        rng, time_limit=allowed, start_time=start_time)
        if tleft() > 40:
            h_hc_r, c5_hc_r = quick_hill_climb_fast(h_sa_r, target_r, 8000, rng)
            if c5_hc_r < c5_r:
                h_sa_r = h_hc_r
                c5_r = c5_hc_r
        if tleft() > 30:
            p_vals_r = [max(3, n_r // 30), max(8, n_r // 15)]
            alphas_r = [300, 1000, 3000]
            h_ref_r, sc_r = slsqp_refine(h_sa_r, n_r, target_r, p_vals_r, alphas_r,
                                         400, 1e-15, time_limit=max(5.0, tleft() - 20),
                                         start_time=start_time)
            if sc_r < c5_r:
                h_sa_r = h_ref_r
                c5_r = sc_r
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_sa_r.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 60:
        num_pert = min(6, int(max(2, tleft() / 40)))
        for fa in range(num_pert):
            if tleft() < 25:
                break
            noise = rng.choice([0.005, 0.01, 0.015, 0.02])
            h_pert = best_h + rng.randn(n_use) * noise
            h_pert = project_feasible(h_pert, target_use)
            h_bin_pert = project_binary(h_pert, target_use)
            allowed_p = min(25.0, tleft() - 10)
            max_it_p = int(min(500000, allowed_p * 10000))
            h_sa_p, c5_p = informed_sa_fast(h_bin_pert, target_use, max_it_p, 0.003, 0.00005,
                                            rng, time_limit=allowed_p, start_time=start_time)
            if c5_p < best_score:
                best_score = c5_p
                best_h = h_sa_p.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 20:
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(50000, (tleft() - 10) * 12000))
        h_sa_f, c5_f = informed_sa_fast(h_bin_final, target_use, max_it, 0.003, 0.00003,
                                        rng, time_limit=max(5.0, tleft() - 15),
                                        start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    if tleft() > 10:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [max(3, n_use // 25)], [1000, 3000, 8000],
                                 300, 1e-15, time_limit=max(3.0, tleft() - 5),
                                 start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
