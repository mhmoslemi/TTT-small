def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)

        return h
    # Distribute the difference to elements that aren't at bounds
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    per = diff / len(idx)
    h[idx] += per
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)

    grad = np.zeros(n)
    n_half = n - 1
    
    for idx_k in range(2*n - 1):
        w = weights[idx_k]
        if w < 1e-15:
            continue
        k = idx_k - n_half  # shift, range [-(n-1), n-1]
        
        if k == 0:
            # d c_0 / d h[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            # d c_k / d h[j] = (1-h[j-k]) [j>=k] - h[j+k] [j<n-k]
            # First part: for j = k to n-1
            grad[k:] += w * (1.0 - h[:n-k]) * dx
            # Second part: for j = 0 to n-1-k
            grad[:n-k] -= w * h[k:] * dx
        else:
            m = -k  # m > 0
            # d c_{-m} / d h[j] = (1-h[j+m]) [j<n-m] - h[j-m] [j>=m]
            # First part: for j = 0 to n-1-m
            grad[:n-m] += w * (1.0 - h[m:]) * dx
            # Second part: for j = m to n-1
            grad[m:] -= w * h[:n-m] * dx
    
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    c_max = np.max(corr)
    # Get top M indices
    top_indices = np.argsort(corr)[-top_m:]
    # Renormalize weights over just these
    shifted = alpha * (corr[top_indices] - c_max)
    shifted = np.clip(shifted, -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n-k]) * dx
                grad[:n-k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n-m] * dx
    
    return grad

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def proj_grad_method(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    steps_per_alpha = 3000
    top_m_list = [10, 15, 20, 30, 50, 80, 120, 200, 2 * n - 1, 2 * n - 1, 2 * n - 1]

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]

        momentum_v = np.zeros(n)
        beta = 0.85
        stalled = 0

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break

            grad = smooth_grad(h, alpha, dx, top_m=top_m)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad

            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-12:
                break

            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 2.0)

            improved = False
            for _ in range(40):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5

            if not improved:
                stalled += 1
                if stalled > 50:
                    break

    return best_h, best_score


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30,
                   damping=0.6, reg=1e-4):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
