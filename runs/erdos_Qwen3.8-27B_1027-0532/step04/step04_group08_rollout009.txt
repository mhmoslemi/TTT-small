def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value descending, take top n/2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def gen_golden_ratio(n):
    """Generate a binary sequence based on golden ratio."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    i = np.arange(n)
    x = np.mod(i * phi, 1.0)
    # Threshold at 0.5, adjust to have exactly n/2 ones
    h = (x < 0.5).astype(float)
    # Adjust
    n_ones = int(h.sum())
    if n_ones > n // 2:
        # Remove some ones (those closest to threshold)
        one_positions = np.where(h == 1)[0]
        # Sort by x value (farthest from 0 are best to remove)
        order = np.argsort(-x[one_positions])
        h[one_positions[order[:n_ones - n//2]]] = 0.0
    elif n_ones < n // 2:
        zero_positions = np.where(h == 0)[0]
        order = np.argsort(x[zero_positions])
        h[zero_positions[order[:n//2 - n_ones]]] = 1.0
    return h

def gen_bitreversal(n):

    i = np.arange(n)

def gen_quadratic_residue(n):
    """Generate binary sequence from quadratic residues."""
    # Use a prime p >= n
    p = 251  # prime
    # If p < n, use a different approach
    i = np.arange(n)

    h = np.zeros(n)

        if val == 0:
            h[idx] = 0.0

            h[idx] = 1.0 if pow(val, (p-1)//2, p) == 1 else 0.0
    # Adjust to have exactly n/2 ones

        h[zero_pos[order[:n//2 - n_ones]]] = 1.0

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def make_binary(h, n):
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    return h_bin


def gen_golden_ratio(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    i = np.arange(n)
    x = np.mod(i * phi, 1.0)
    h = (x < 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > n // 2:
        one_pos = np.where(h == 1)[0]
        order = np.argsort(-x[one_pos])
        h[one_pos[order[:n_ones - n // 2]]] = 0.0
    elif n_ones < n // 2:
        zero_pos = np.where(h == 0)[0]
        order = np.argsort(x[zero_pos])
        h[zero_pos[order[:n // 2 - n_ones]]] = 1.0
    return h


def gen_quadratic_residue(n, p_base=251):
    p = p_base
    while p < n:
        p += 2
        found = True
        for q in range(2, int(np.sqrt(p)) + 1):
            if p % q == 0:
                found = False
                break
        if not found:
            p += 2
    i = np.arange(n)
    h = np.zeros(n)
    for idx in range(n):
        val = idx % p
        if val == 0:
            h[idx] = 0.0
        elif pow(val, (p - 1) // 2, p) == 1:
            h[idx] = 1.0
    n_ones = int(h.sum())
    if n_ones > n // 2:
        one_pos = np.where(h == 1)[0]
        rng_order = np.random.RandomState(0).permutation(len(one_pos))

    elif n_ones < n // 2:

        rng_order = np.random.RandomState(0).permutation(len(zero_pos))
        h[zero_pos[rng_order[:n // 2 - n_ones]]] = 1.0
    return h

def continuous_polish(h_start, n, max_time, target):
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    deadline = time.time() + max_time
    alphas = [1000, 5000, 20000, 100000]
    for a in alphas:
        if time.time() > deadline:
            break
        sub_deadline = min(60.0, deadline - time.time())
        if sub_deadline < 10:
            break
        h_cur = best_h.copy()
        h_cur = h_cur + 0.005 * np.random.randn(n)
        h_cur = project_feasible(h_cur, target)
        for _ in range(3):
            if time.time() > deadline - 5:
                break
            def obj(x):
                return soft_max_obj(x, a)

                return np.sum(x) - target
            try:

                               bounds=[(0.0, 1.0)] * n,

                               options={'maxiter': 300, 'ftol': 1e-14,

                h_new = project_feasible(res.x, target)

def active_set_gradient(h, n, target, max_time, lr_init=0.01):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    deadline = time.time() + max_time
    v = np.zeros(n)
    momentum = 0.9
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        thresh = 0.85 * c_max
        active_idx = np.where(corr > thresh)[0]
        if len(active_idx) == 0:
            break
        grad = np.zeros(n)
        n_pts = n
        for k_idx in active_idx:
            s = k_idx - (n_pts - 1)
            if s >= 0:
                grad[s:] += (1.0 - h_cur[:n_pts - s])
                grad[:n_pts - s] -= h_cur[s:]
            else:
                grad[:n_pts + s] += (1.0 - h_cur[-s:])
                grad[-s:] -= h_cur[:n_pts + s]
        grad /= float(len(active_idx))
        v = momentum * v - lr * grad
        h_new = project_feasible(h_cur + v, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if lr < 1e-10:
                lr = lr_init
                v = np.zeros(n)
                if stall > 50:
                    break
    return best_h, best_score
