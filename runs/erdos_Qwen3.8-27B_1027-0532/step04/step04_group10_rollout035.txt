def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        # Proportional increase
        room = 1.0 - h[idx]
        if room.sum() < diff:
            h[idx] = 1.0
            # Still need more, but can't (all at 1)
            return h
        h[idx] += diff * room / room.sum()
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff * h[idx] / h[idx].sum()
    return np.clip(h, 0.0, 1.0)

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:

    if diff > 0:

def project_feasible(h, target_sum):
