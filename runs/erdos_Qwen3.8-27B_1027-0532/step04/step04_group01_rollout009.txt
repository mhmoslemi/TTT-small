def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    """Project h to have sum = target, staying in [0,1]."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    current = np.sum(h)
    if abs(current - target) < 1e-12:
        return h
    diff = target - current
    if diff > 0:
        # Need to increase sum
        # Add to positions with most room
        room = 1.0 - h
        mask = room > 1e-10
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        # Need to decrease sum
        room = h
        mask = room > 1e-10
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h_init, n_passes=200, penalty=50.0, time_limit=300.0):

    best_h = h.copy()
    
    for p_pass in range(n_passes):
        if time.time() - t_start > time_limit:
            break
        
        # Shuffle order for better convergence
        order = np.random.permutation(n)
        
        for p in order:

            d_lo = max(-p, -(n-1))
            d_hi = min(n-p, n-1)

                d_vals2 = np.arange(d_lo2, d_hi2 + 1)
                c[d_vals2 + offset] += (-h[p - d_vals2])
            
            x_old = h[p]
            b = corr - c * x_old
            
            S = target_sum - (np.sum(h) - x_old)

            def slope_at(x):
                vals = c * x + b
                idx_max = np.argmax(vals)
                return c[idx_max] + 2.0 * penalty * (x + S)
            
            s0 = slope_at(0.0)
            s1 = slope_at(1.0)
            
            if s0 >= 0:
                x_opt = 0.0

                x_opt = 1.0

                    if slope_at(mid) < 0:
                        lo = mid
                    else:
                        hi2 = mid

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    current = np.sum(h)
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-10
    else:
        room = h
        mask = room > 1e-10
    if mask.sum() == 0:
        return h
    h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def coordinate_descent(h_init, n_passes=200, penalty=100.0, time_limit=300.0, rng=None):
    t_start = time.time()
    n = len(h_init)
    h = h_init.copy().astype(np.float64)
    target_sum = n / 2.0
    n_full = 2 * n - 1
    offset = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for p_pass in range(n_passes):
        if time.time() - t_start > time_limit:
            break

        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for p in order:
            c = np.zeros(n_full)

            d_lo = max(-p, -(n - 1))
            d_hi = min(n - p, n - 1)
            if d_hi > d_lo:
                d_vals = np.arange(d_lo, d_hi)
                c[d_vals + offset] += (1.0 - h[p + d_vals])

            d_lo2 = max(p - (n - 1), -(n - 1))
            d_hi2 = min(p, n - 1)
            if d_hi2 >= d_lo2:
                d_vals2 = np.arange(d_lo2, d_hi2 + 1)
                c[d_vals2 + offset] += (-h[p - d_vals2])

            x_old = h[p]
            b = corr - c * x_old
            S = target_sum - (np.sum(h) - x_old)

            idx0 = np.argmax(b)
            s0 = c[idx0] + 2.0 * penalty * (0.0 + S)
            idx1 = np.argmax(c + b)
            s1 = c[idx1] + 2.0 * penalty * (1.0 + S)

            if s0 >= 0:
                x_opt = 0.0
            elif s1 <= 0:
                x_opt = 1.0
            else:
                lo = 0.0
                hi2 = 1.0
                for _ in range(25):
                    mid = (lo + hi2) / 2.0
                    idx_mid = np.argmax(c * mid + b)
                    s_mid = c[idx_mid] + 2.0 * penalty * (mid + S)
                    if s_mid < 0:
                        lo = mid
                    else:
                        hi2 = mid
                x_opt = (lo + hi2) / 2.0

            dx_ = x_opt - x_old
            if abs(dx_) > 1e-12:
                h[p] = x_opt
                corr += c * dx_

        curr_c5 = compute_c5(h)
        if curr_c5 < best_c5 - 1e-14:
            best_c5 = curr_c5
            best_h = h.copy()

    return best_h, best_c5

def make_fourier_seed(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.full(n, 0.5)
    for freq in range(1, n // 2 + 1):
        amp = rng.uniform(0.0, 2.0 / max(freq, 1))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)
    h = project_sum(h, n / 2.0)
    return h
