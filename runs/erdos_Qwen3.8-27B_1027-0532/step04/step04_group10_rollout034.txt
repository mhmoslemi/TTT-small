def compute_correlations(h):

        return h

    h = h0.copy()
    m = np.zeros(n)

But wait, there's a subtlety: c_max is itself a function of h (it's the max of the top-k correlations). So the full gradient would include the derivative of c_max. However, at the point where c_max is achieved, the derivative of the max with respect to h is the same as the derivative of that particular correlation. And since we're using the log-sum-exp which is a smooth approximation, the c_max term's gradient is:

d(c_max)/dh[j] = sum_i I(c_i = c_max) * dc_i/dh[j]

But in the log-sum-exp, the weight w_i for the maximizing element is w_max = exp(0)/sum(exp(...)) = 1/sum(exp(...)), which is small when k > 1. So the gradient of the log-sum-exp is dominated by the "averaged" gradient over all top-k elements, which is correct.

However, there's a technical issue: the derivative of the log-sum-exp with respect to h should be:
df/dh = (1/alpha) * d/dh [log(sum exp(alpha*(c_i - c_max)))] + dc_max/dh

The first term: (1/alpha) * sum_i [alpha*exp(alpha*(c_i-c_max)) / sum_j exp(alpha*(c_j-c_max))] * (dc_i/dh - dc_max/dh)
= sum_i w_i * (dc_i/dh - dc_max/dh)

The second term: dc_max/dh = sum_i I(c_i = c_max) * dc_i/dh (subgradient)

So total: sum_i w_i * dc_i/dh - sum_i w_i * dc_max/dh + dc_max/dh
= sum_i w_i * dc_i/dh - (sum_i w_i) * dc_max/dh + dc_max/dh
= sum_i w_i * dc_i/dh - dc_max/dh + dc_max/dh  [since sum_i w_i = 1]
= sum_i w_i * dc_i/dh

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0 - 1e-9
        count = np.sum(mask)
        if count == 0:
            return h
        h[mask] += diff / count
    else:
        mask = h > 1e-9
        count = np.sum(mask)
        if count == 0:
            return h
        h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def compute_lse_and_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return obj, grad

def adam_optimize(h0, target, alpha, topk, n_iter, lr0, beta1, beta2, eps, lr_decay, noise_scale, seed):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    rng_local = np.random.RandomState(seed)
    for t in range(1, n_iter + 1):
        lr = lr0 / (1.0 + t * lr_decay)
        obj, g = compute_lse_and_grad(h, alpha, topk)
        if noise_scale > 0.0:
            g = g + noise_scale * rng_local.randn(n)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * g * g
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        step = h_new - h
        max_step = np.max(np.abs(step))
        if max_step > 0.1:
            h_new = h + step * (0.1 / max_step)
        h = project_feasible(h_new, target)
    return h, compute_c5(h)
