Let me try a different approach: **paired coordinate descent**. Pick two coordinates i and j, and optimize (h[i], h[j]) jointly subject to h[i] + h[j] = h[i]_old + h[j]_old (to maintain the sum). The 2D problem is:

This is again a 1D convex PL minimization! Same as before, just with different B and A.

The constraint is: max(0, c-1) <= t <= min(1, c).

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, i, n):
    """Compute B[k] for coordinate i: the coefficient of h[i] in corr[k]."""
    B = np.zeros(2*n - 1)

        if i + m < n:

        B[n-1-m] = b

So corr[k] = (number of ONES in [0, n-1-k]) - (number of ONES pairs at distance k)

To minimize max_k corr[k], we want to:
1. Spread the ones evenly (so that the count in any window is roughly proportional)
2. Minimize the number of close pairs at all distances

This is related to the problem of finding "well-distributed" subsets, or "low-discrepancy" sets.

A simple heuristic: use a "greedy" placement where we place ones as far apart as possible. For n=400, we'd want 200 ones spread over 400 positions, so roughly every other position. This gives a starting point of h = [1, 0, 1, 0, ...] or similar.

    # c[k] for k >= 0: index = n-1+k
    # B[n-1+k] = (1-h[i-k]) * [k<=i<=n-1] - h[i+k] * [i<=n-1-k]

    ks = np.arange(n)  # k = 0, 1, ..., n-1
    # For positive lags
    mask1 = ks <= i  # k <= i
    mask2 = (i + ks) < n  # i+k < n, i.e., i <= n-1-k
    b_pos = np.zeros(n)
    b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    b_pos[mask2] -= h[i + ks[mask2]]
    B[n-1:n-1+n] = b_pos
    
    ms = np.arange(1, n)  # m = 1, 2, ..., n-1
    # For negative lags
    mask3 = (i + ms) < n  # i+m < n, i.e., i <= n-1-m
    mask4 = ms <= i  # m <= i
    b_neg = np.zeros(n-1)
    b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n-1] = b_neg  # indices 0 to n-2 correspond to k = -(n-1) to -1

def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=3000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
