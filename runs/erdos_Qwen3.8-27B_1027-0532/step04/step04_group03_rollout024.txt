def waterfill_refine(h_start, target, time_limit, start_time, n_lags=5):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    no_improve = 0
    
    for it in range(100000):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > 1000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        c_max = float(np.max(corr))
        
        # Find top n_lags peaks
        threshold = c_max * 0.95
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Weighted gradient: weight by (c_k - target) where target = 2nd highest
        weights = np.zeros(len(corr))
        for k in active:
            excess = corr[k] - c_max * 0.9
            weights[k] = max(excess, 0.01)
        weights[k] / sum(weights)  # normalize
        
        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)  # actual lag
            w = weights[k] / weights[active].sum()
            if m >= 0:
                grad[m:] += w * (1.0 - h[:n - m])
                grad[:n - m] -= w * h[m:]
            else:
                mm = -m
                grad[:n - mm] += w * (1.0 - h[mm:])
                grad[mm:] -= w * h[:n - mm]
        
        # Project step
        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            no_improve = 0
            lr = min(lr * 1.03, 0.3)
        else:
            no_improve += 1
            lr *= 0.97
            if lr < 1e-7:
                lr = 0.02
    
    return best_h, best_c5

import time

_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            idx_sorted = ones[np.argsort(h[ones])][:need]
            h_r[idx_sorted] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            idx_sorted = zeros[np.argsort(-h[zeros])][:need]
            h_r[idx_sorted] = 1
    final_sum = h_r.sum()

    return np.sum(x) - _cons_target

    return np.ones(_cons_n)

def smoothened_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    logsumexp = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return logsumexp

def smoothened_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad

def slsqp_refine(h_start, target, time_limit, start_time, alphas=None):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    if alphas is None:
        alphas = [50, 100, 200, 400, 800, 1500, 3000, 6000, 10000, 20000]
    
    for alpha in alphas:
        if (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(
                smoothened_obj, h, args=(alpha, n, target),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def waterfill_refine(h_start, target, time_limit, start_time):

    h = h_start.copy().astype(float)
    best_h = h.copy()

    no_improve = 0

            break

            no_improve = 0
    
        corr_unscaled = np.correlate(h, 1.0 - h, mode='full')
        corr = corr_unscaled * (2.0 / n)
        c_max = float(np.max(corr))
        
        # Find lags exceeding 90% of max
        threshold = c_max * 0.90
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Weight by excess
        excess = np.maximum(corr[active] - c_max * 0.85, 0.001)
        w_tot = excess.sum()
        w_norm = excess / w_tot
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for j, k in enumerate(active):
            w = w_norm[j]
            m = k - (n - 1)
            if m >= 0:
                grad[m:] += w * (1.0 - h[:n - m])
                grad[:n - m] -= w * h[m:]
            else:
                mm = -m
                grad[:n - mm] += w * (1.0 - h[mm:])
                grad[mm:] -= w * h[:n - mm]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            no_improve = 0
            lr = min(lr * 1.02, 0.2)
        else:
            no_improve += 1
            lr *= 0.96
            if lr < 1e-8:
                lr = 0.01
                no_improve = 0
    
    return best_h, best_c5

    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)

    h_out = np.interp(x_new, x_old, h_src)

from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1

def cons_fun(x):
    return np.sum(x) - _cons_target

    return np.ones(_cons_n)

def smoothened_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_refine(h_start, target, time_limit, start_time, alphas=None):

    n = len(h_start)
    _cons_n = n
    _cons_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    if alphas is None:
        alphas = [100, 300, 600, 1200, 2400, 5000, 10000, 20000]
    
    for alpha in alphas:
        if (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(
                smoothened_obj, h, args=(alpha, n, target),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def waterfill_refine(h_start, target, time_limit, start_time):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[:int(final_sum - target_sum)]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)


def sa_optimize(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 10)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]

    for i in range(max_iters):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.03:
                no_improve = 0
            else:
                break

        h_prev = h.copy()
        ratio = (i + 1) / max(max_iters, 1)
        t = t_init * (t_final / t_init) ** ratio
        move_type = rng.randint(6)
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h[i1] = 0
            h[i0] = 1
            moved = True
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            h[ones_cache[idx1[0]]] = 0
            h[ones_cache[idx1[1]]] = 0
            h[zeros_cache[idx0[0]]] = 1
            h[zeros_cache[idx0[1]]] = 1
            moved = True
        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            radius = rng.randint(1, max(2, n // 20))
            direction = rng.choice([-1, 1])
            i0_pos = (i1 + direction * radius) % n
            if h[i0_pos] == 1:
                i0_pos = zeros_cache[rng.randint(len(zeros_cache))]
            h[i1] = 0
            h[i0_pos] = 1
            moved = True
        elif move_type == 3 and len(ones_cache) >= 3 and len(zeros_cache) >= 3:
            idx1 = rng.choice(len(ones_cache), 3, replace=False)
            idx0 = rng.choice(len(zeros_cache), 3, replace=False)
            for idx in idx1:
                h[ones_cache[idx]] = 0
            for idx in idx0:
                h[zeros_cache[idx]] = 1
            moved = True
        elif move_type == 4:
            block_len = rng.randint(1, min(6, max(1, n // 150)))
            pos1 = rng.randint(n)
            pos2 = rng.randint(n)
            valid = True
            b1_list = []
            b2_list = []
            for ext in range(block_len):
                p1 = (pos1 + ext) % n
                p2 = (pos2 + ext) % n
                if h[p1] != 1 or h[p2] != 0:
                    valid = False
                    break
                b1_list.append(p1)
                b2_list.append(p2)
            if valid and len(b1_list) > 0:
                for p in b1_list:
                    h[p] = 0
                for p in b2_list:
                    h[p] = 1
                moved = True
        else:
            if len(ones_cache) > 0 and len(zeros_cache) > 0:
                i1 = ones_cache[rng.randint(len(ones_cache))]
                radius = rng.randint(1, max(2, n // 15))
                offsets = rng.choice(np.arange(1, radius + 1), 2, replace=False)
                dir1 = rng.choice([-1, 1])
                dir2 = rng.choice([-1, 1])
                i0a = (i1 + dir1 * offsets[0]) % n
                i0b = (i1 + dir2 * offsets[1]) % n
                if h[i0a] == 0 and h[i0b] == 0:
                    h[i1] = 0
                    h[i0a] = 1
                    h[i0b] = 1
                    h[zeros_cache[rng.randint(len(zeros_cache))]] = 0
                    moved = True

        if not moved:
            continue

        c5_new = compute_c5(h)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h = h_prev
            no_improve += 1

    return best_h, best_c5


def cons_fun(x):
    return np.sum(x) - _cons_target


def cons_jac(x):
    return np.ones(_cons_n)


def smoothened_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def slsqp_refine(h_start, target, time_limit, start_time, alphas=None):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    if alphas is None:
        alphas = [100, 300, 800, 2000, 5000, 12000, 25000]

    for alpha in alphas:
        if (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(
                smoothened_obj, h, args=(alpha, n, target),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 400, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    return best_h, best_c5


def waterfill_refine(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.01
    no_improve = 0

    for it in range(80000):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > 400:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        c_max = float(np.max(corr))
        top_k = min(8, len(corr))
        active = np.argsort(corr)[-top_k:]
        active = np.sort(active)

        cutoff = c_max * 0.94
        excess = np.maximum(corr[active] - cutoff, 0.001)
        w_total = excess.sum()
        w_norm = excess / w_total

        grad = np.zeros(n)
        for j in range(len(active)):
            k = active[j]
            w = w_norm[j]
            m = k - (n - 1)
            if m >= 0:
                grad[m:] += w * (1.0 - h[:n - m])
                grad[:n - m] -= w * h[m:]
            else:
                mm = -m
                grad[:n - mm] += w * (1.0 - h[mm:])
                grad[mm:] -= w * h[:n - mm]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            no_improve = 0
            lr = min(lr * 1.03, 0.12)
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-8:
                break

    return best_h, best_c5


def targeted_swap(h_start, target_sum, max_iter, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if (time.time() - start_time) > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        m = k_max - (n - 1)

        improved = False
        for _ in range(30):
            if m >= 0:
                i = rng.randint(0, max(1, n - m))
                if h[i] == 1 and h[i + m] == 0:
                    h[i] = 0
                    h[i + m] = 1
                    c5_new = compute_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        improved = True
                    else:
                        h[i] = 1
                        h[i + m] = 0
                    break
            else:
                mm = -m
                i = rng.randint(0, max(1, n - mm))
                if h[i] == 0 and h[i + mm] == 1:
                    h[i] = 1
                    h[i + mm] = 0
                    c5_new = compute_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        improved = True
                    else:
                        h[i] = 0
                        h[i + mm] = 1
                    break
        if not improved:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0
            h[i0] = 1
            c5_new = compute_c5(h)
            if c5_new >= best_c5:
                h[i1] = 1
                h[i0] = 0

    return h, best_c5


def interpolation_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def hill_climb_binary(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 500

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0
        h[i0] = 1
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[i1] = 1
            h[i0] = 0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5


def make_sinusoidal_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    if tleft() > 150:
        h_sa, c5_sa = sa_optimize(h_bin, target_init, 100000, 0.02, 0.00005,
                                  rng, time_limit=min(160.0, tleft() * 0.15), start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 80:
        h_hc, c5_hc = hill_climb_binary(best_h, target_init, 30000, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    if tleft() > 100:
        h_ts, c5_ts = targeted_swap(best_h, target_init, 50000, rng,
                                     min(80.0, tleft() * 0.12), start_time)
        if c5_ts < best_score:
            best_score = c5_ts
            best_h = h_ts.copy()

    if tleft() > 120:
        h_wf, c5_wf = waterfill_refine(best_h, target_init, min(200.0, tleft() * 0.16), start_time)
        if c5_wf < best_score:
            best_score = c5_wf
            best_h = h_wf.copy()

    if tleft() > 150:
        h_slq, c5_slq = slsqp_refine(best_h, target_init, min(250.0, tleft() * 0.2), start_time,
                                      alphas=[100, 300, 800, 2000, 5000, 12000, 25000, 50000])
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    if tleft() > 100:
        h_sa2, c5_sa2 = sa_optimize(best_h, target_init, 80000, 0.01, 0.00003,
                                     rng, time_limit=min(130.0, tleft() * 0.13), start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        h_hc2, c5_hc2 = hill_climb_binary(best_h, target_init, 20000, rng)
        if c5_hc2 < best_score:
            best_score = c5_hc2
            best_h = h_hc2.copy()

    if tleft() > 80:
        h_wf2, c5_wf2 = waterfill_refine(best_h, target_init, min(100.0, tleft() * 0.12), start_time)
        if c5_wf2 < best_score:
            best_score = c5_wf2
            best_h = h_wf2.copy()

    if tleft() > 100:
        h_slq2, c5_slq2 = slsqp_refine(best_h, target_init, min(180.0, tleft() * 0.18), start_time,
                                        alphas=[5000, 15000, 40000, 100000, 200000])
        if c5_slq2 < best_score:
            best_score = c5_slq2
            best_h = h_slq2.copy()

    n_candidates = []
    for nv in [640, 720, 800, 900, 960, 1000, 1080, 1200]:
        if nv != n_init and nv % 2 == 0 and nv >= 200:
            n_candidates.append(nv)

    for n_try in n_candidates:
        if tleft() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolation_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        h_sa_r, c5_sa_r = sa_optimize(h_bin_try, target_try, 30000, 0.02, 0.0001,
                                       rng, time_limit=min(35.0, tleft() * 0.25), start_time=start_time)
        if tleft() > 30:
            h_wfr, c5_wfr = waterfill_refine(h_sa_r, target_try, min(35.0, tleft() * 0.25), start_time)
            if c5_wfr < c5_sa_r:
                h_sa_r = h_wfr
                c5_sa_r = c5_wfr
        if tleft() > 25:
            h_slq_r, c5_slq_r = slsqp_refine(h_sa_r, target_try, min(40.0, tleft() * 0.25), start_time,
                                               alphas=[500, 2000, 6000, 15000])
            if c5_slq_r < c5_sa_r:
                h_sa_r = h_slq_r
                c5_sa_r = c5_slq_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    if tleft() > 50:
        freqs = [3, 5, 7, 9, 13]
        for freq in freqs:
            if tleft() < 25:
                break
            phase = rng.uniform(0.0, 2.0 * np.pi)
            h_sine = make_sinusoidal_init(n_init, freq, phase)
            h_bin_s = project_binary(h_sine, target_init)
            h_sa_s, c5_sa_s = sa_optimize(h_bin_s, target_init, 30000, 0.03, 0.0002,
                                           rng, time_limit=min(20.0, tleft() * 0.35), start_time=start_time)
            if c5_sa_s < best_score:
                best_score = c5_sa_s
                best_h = h_sa_s.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin, c5_fin = slsqp_refine(best_h, target_use, min(50.0, tleft() - 5), start_time,
                                      alphas=[20000, 50000, 150000, 400000])
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_final
