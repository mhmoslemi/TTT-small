                best_h = h_try
                lr = min(lr * 1.03, 0.3)
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 20:
                    lr *= 0.7
                    stagnant = 0
                if lr < 0.001:
                    break

    # Phase 4: Higher resolution attempts
    for n2 in [600, 800]:
        if not time_ok(150):
            break
        t2 = n2 / 2.0
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n2)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, t2)
        bounds2 = [(0.0, 1.0)] * n2

        for alpha, topk, maxiter in [(100, 70, 250), (400, 50, 350), (1000, 40, 400)]:
            if not time_ok(50):
                break
            h_start = h_interp + rng.randn(n2) * 0.005
            h_start = project_feasible(h_start, t2)
            _obj_alpha = float(alpha)
            _obj_topk = int(topk)
            _obj_target = float(t2)
            _obj_penalty = 5000.0
            try:
                res = minimize(_objective, h_start, jac=True, method='L-BFGS-B',
                               bounds=bounds2, options={'maxiter': maxiter, 'ftol': 1e-17})
                h_new = project_feasible(res.x, t2)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = np.interp(x_old, x_new, h_new)
                    best_h = project_feasible(best_h, target)
            except Exception:
                pass

    # Phase 5: Diverse restarts with LSE at moderate alpha
    if time_ok(80):
        for restart in range(5):
            if not time_ok(25):
                break
            # Larger perturbation for diversity
            scale = 0.01 + 0.005 * restart
            h_start = best_h + rng.randn(n) * scale
            h_start = project_feasible(h_start, target)
            for alpha in [50, 200]:
                if not time_ok(20):
                    break
                _obj_alpha = float(alpha)
                _obj_topk = 80
                _obj_target = float(target)
                _obj_penalty = 3000.0
                try:
                    res = minimize(_objective, h_start, jac=True, method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 300, 'ftol': 1e-17})
                    h_new = project_feasible(res.x, target)
                    score = compute_c5(h_new)
                    if score < best_score:
                        best_score = score
                        best_h = h_new
                except Exception:
                    pass

    # Phase 6: SLSQP with equality constraint (separate from L-BFGS-B)
    if time_ok(60):
        for alpha in [200, 800]:
            if not time_ok(30):
                break
            h_start = best_h + rng.randn(n) * 0.002
            h_start = project_feasible(h_start, target)
            _obj_alpha = float(alpha)
            _obj_topk = 30
            _obj_target = float(target)
            _obj_penalty = 0.0

            def slsqp_obj(x):
                f, _ = lse_obj_and_grad(x, _obj_alpha, _obj_topk)
                return f

            def slsqp_jac(x):
                _, g = lse_obj_and_grad(x, _obj_alpha, _obj_topk)
                return g

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            try:
                res = minimize(slsqp_obj, h_start, jac=slsqp_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Final high-alpha polish
    if time_ok(40):
        for alpha in [5000, 8000]:
            if not time_ok(20):
                break
            h_start = best_h + rng.randn(n) * 0.001
            h_start = project_feasible(h_start, target)
            _obj_alpha = float(alpha)
            _obj_topk = 15
            _obj_target = float(target)
            _obj_penalty = 20000.0
            try:
                res = minimize(_objective, h_start, jac=True, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-13})
                h_new = project_feasible(res.x, target)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new
            except Exception:
                pass

    # Phase 8: Coordinated block perturbations
    if time_ok(40):
        block_size = max(n // 40, 4)
        for round_id in range(3):
            if not time_ok(15):
                break
            for b in range(0, n - block_size, block_size):
                if not time_ok(10):
                    break
                block = best_h[b:b+block_size].copy()
                for delta in [0.005, -0.005, 0.003, -0.003]:
                    h_try = best_h.copy()
                    h_try[b:b+block_size] = np.clip(block + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-9:
                        best_score = sc
                        best_h = h_try
                        break

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
