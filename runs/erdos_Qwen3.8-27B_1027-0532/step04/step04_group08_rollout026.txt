from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def binarize(h, n_ones):
    # Convert to binary, keeping the n_ones largest values as 1s
    idx = np.argsort(h)[::-1][:n_ones]
    h_bin = np.zeros(len(h))
    h_bin[idx] = 1.0
    return h_bin

def random_block_sequence(n, n_ones, rng, n_blocks):
    # Create a binary sequence with n_ones ones arranged in n_blocks blocks
    # Block sizes are random
    sizes = rng.multinomial(n_ones, np.ones(n_blocks)/n_blocks)
    gaps = rng.multinomial(n - n_ones, np.ones(n_blocks+1)/(n_blocks+1))
    h = np.zeros(n)
    pos = 0
    pos += gaps[0]
    for i in range(n_blocks):
        h[pos:pos+sizes[i]] = 1.0
        pos += sizes[i]
        pos += gaps[i+1]
    # Fix if we overflowed
    h = h[:n]
    # Fix count
    actual = int(h.sum())
    if actual > n_ones:
        # Remove excess ones
        one_positions = np.where(h == 1)[0]
        remove = rng.choice(one_positions, actual - n_ones, replace=False)
        h[remove] = 0
    elif actual < n_ones:
        # Add missing ones
        zero_positions = np.where(h == 0)[0]
        add = rng.choice(zero_positions, n_ones - actual, replace=False)
        h[add] = 1
    return h

def hill_climb(h, n, target_ones, max_iters, deadline):
    h = h.copy()
    best_score = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        improved = False
        # Try random swaps
        for attempt in range(200):
            if time.time() > deadline:
                break
            a = rng.choice(zeros)
            b = rng.choice(ones)
            h[a] = 1
            h[b] = 0
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                improved = True
                # Update sets
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                break
            else:
                h[a] = 0
                h[b] = 1
        if not improved:
            # Try more or break
            if attempt >= 199:
                break
    return h, best_score

2. As the second factor: when i+j=a, i.e., i=a-j, the term h[a-j]*(1-h[a]) changes from h[a-j]*1 to h[a-j]*0, i.e., decreases by h[a-j]

Similarly for h[b] changing from 1 to 0:
1. As first factor: when i=b, term changes from (1-h[b+j]) to 0, i.e., decreases by (1-h[b+j])
2. As second factor: when i=b-j, term changes from h[b-j]*(1-h[b]) = h[b-j]*0 to h[b-j]*1, i.e., increases by h[b-j]

So total change at shift j:
Δcorr[j] = (1-h[a+j]) - h[a-j] - (1-h[b+j]) + h[b-j]

Where terms are only included if the indices are valid (in [0, n-1]).

Note: h[a-j] is the value BEFORE the swap. If a-j = b, then h[a-j] = h[b] = 1 (before swap). If a-j = a, that means j=0 which we skip.

Similarly, h[b-j]: if b-j = a, then h[b-j] = h[a] = 0 (before swap).

This gives us an O(n) way to compute the delta for each swap, and then we just need to check if the new max is smaller.

import numpy as np
import time

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target_ones = n // 2

    def remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    def eval_mc(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(c))

    def binarize(h_in):
        hb = np.zeros(n)
        hb[np.argsort(h_in)[::-1][:target_ones]] = 1.0
        return hb

    def make_blocks(nb):
        hb = np.zeros(n)
        sizes = rng.multinomial(target_ones, np.ones(nb) / nb)
        for i in range(nb):
            if sizes[i] == 0:
                j = int(np.argmax(sizes))
                if sizes[j] > 1:
                    sizes[j] -= 1
                    sizes[i] = 1
        gaps = rng.multinomial(n - target_ones, np.ones(nb + 1) / (nb + 1))
        pos = 0
        for i in range(nb):
            pos += gaps[i]
            end = min(pos + int(sizes[i]), n)
            if end > pos:
                hb[pos:end] = 1
            pos = end
        actual = int(hb.sum())
        if actual > target_ones:
            op = np.where(hb == 1)[0]
            hb[rng.choice(op, actual - target_ones, replace=False)] = 0
        elif actual < target_ones:
            zp = np.where(hb == 0)[0]
            hb[rng.choice(zp, target_ones - actual, replace=False)] = 1
        return hb

    def hill_climb(h_start, mc_start, deadline, kick_interval=15, max_stall=400):
        hl = h_start.copy()
        mcl = mc_start
        ones = np.where(hl == 1)[0]
        zeros = np.where(hl == 0)[0]
        stall = 0
        while time.time() < deadline:
            found = False
            for _ in range(300):
                if time.time() > deadline:
                    return hl, mcl
                a = int(zeros[rng.randint(len(zeros))])
                b = int(ones[rng.randint(len(ones))])
                hl[a] = 1.0
                hl[b] = 0.0
                sc = eval_mc(hl)
                if sc < mcl:
                    mcl = sc
                    found = True
                    ones = np.where(hl == 1)[0]
                    zeros = np.where(hl == 0)[0]
                    stall = 0
                    break
                hl[a] = 0.0
                hl[b] = 1.0
            if not found:
                stall += 1
                if stall % kick_interval == 0:
                    a = int(zeros[rng.randint(len(zeros))])
                    b = int(ones[rng.randint(len(ones))])
                    hl[a] = 1.0
                    hl[b] = 0.0
                    mcl = eval_mc(hl)
                    ones = np.where(hl == 1)[0]
                    zeros = np.where(hl == 0)[0]
                    stall = 0
                if stall > max_stall:
                    break
        return hl, mcl

    def targeted_climb(h_start, mc_start, deadline):
        hl = h_start.copy()
        mcl = mc_start
        while time.time() < deadline:
            corr = np.correlate(hl, 1.0 - hl, mode='full')
            max_idx = int(np.argmax(corr))
            shift = max_idx - (n - 1)
            ones = np.where(hl == 1)[0]
            zeros = np.where(hl == 0)[0]
            found = False
            if shift > 0:
                contrib = [i for i in range(n - shift) if hl[i] == 1.0 and hl[i + shift] == 0.0]
            elif shift < 0:
                s = -shift
                contrib = [i for i in range(s, n) if hl[i] == 1.0 and hl[i - s] == 0.0]
            else:
                contrib = []
            if len(contrib) > 0:
                sample_size = min(len(contrib), 15)
                for a in rng.choice(np.array(contrib), sample_size, replace=False):
                    for _ in range(8):
                        b = int(zeros[rng.randint(len(zeros))])
                        if b == a:
                            continue
                        hl[int(a)] = 0.0
                        hl[b] = 1.0
                        sc = eval_mc(hl)
                        if sc < mcl:
                            mcl = sc
                            found = True
                            ones = np.where(hl == 1)[0]
                            zeros = np.where(hl == 0)[0]
                            break
                        hl[int(a)] = 1.0
                        hl[b] = 0.0
                    if found:
                        break
            if not found:
                for _ in range(100):
                    if time.time() > deadline:
                        return hl, mcl
                    a = int(zeros[rng.randint(len(zeros))])
                    b = int(ones[rng.randint(len(ones))])
                    hl[a] = 1.0
                    hl[b] = 0.0
                    sc = eval_mc(hl)
                    if sc < mcl:
                        mcl = sc
                        found = True
                        break
                    hl[a] = 0.0
                    hl[b] = 1.0
                if not found:
                    break
        return hl, mcl

    # Binarize initial
    h0 = binarize(h_init)
    mc0 = eval_mc(h0)
    best_h = h0.copy()
    best_mc = mc0

    # Phase 1: Intensive hill climb from binarized initial
    dl1 = time.time() + min(500.0, remaining())
    h1, mc1 = hill_climb(h0, mc0, dl1)
    if mc1 < best_mc:
        best_mc = mc1
        best_h = h1

    # Phase 1b: Targeted climb
    if best_mc > 76 and remaining() > 30:
        dl_bt = time.time() + min(200.0, remaining() * 0.5)
        ht, mct = targeted_climb(best_h, best_mc, dl_bt)
        if mct < best_mc:
            best_mc = mct
            best_h = ht
        # Follow up with random hill climb
        if remaining() > 30:
            dl_f = time.time() + min(100.0, remaining() * 0.4)
            hf, mcf = hill_climb(best_h, best_mc, dl_f)

                best_mc = mcf
                best_h = hf

    for trial in range(10):

            if t_alloc < 15:
                break
            dl2 = time.time() + t_alloc
            h2, mc2 = hill_climb(hb, mc_init, dl2)
            if mc2 < best_mc:
                best_mc = mc2
                best_h = h2
                if best_mc > 76 and remaining() > 30:
                    dl2b = time.time() + min(60.0, remaining() * 0.3)
                    h2b, mc2b = targeted_climb(best_h, best_mc, dl2b)
                    if mc2b < best_mc:
                        best_mc = mc2b
                        best_h = h2b

    # Phase 3: More random restarts with different seeds
    for trial in range(5):
        if remaining() < 60:
            break
        rng2 = np.random.RandomState(seed + 1000 + trial * 7)
        h_rand = np.zeros(n)
        h_rand[rng2.choice(n, target_ones, replace=False)] = 1.0
        mc_r = eval_mc(h_rand)
        if mc_r < best_mc + 10:
            t_alloc = min(60.0, remaining() * 0.2)
            if t_alloc < 15:
                break
            dl3 = time.time() + t_alloc
            h3, mc3 = hill_climb(h_rand, mc_r, dl3, kick_interval=5, max_stall=200)
            if mc3 < best_mc:
                best_mc = mc3
                best_h = h3

    # Phase 4: Final intensive polish
    if remaining() > 20:
        dl4 = time.time() + remaining() - 20
        h4, mc4 = hill_climb(best_h, best_mc, dl4, kick_interval=20, max_stall=800)
        if mc4 < best_mc:
            best_mc = mc4
            best_h = h4

    c5 = best_mc * 2.0 / n
    return best_h, c5, n
