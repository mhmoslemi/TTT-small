from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def is_prime_check(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True


def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while not is_prime_check(p):
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h


def make_structured_random(n, rng, density_bias=0.0):
    target = n // 2
    h = np.zeros(n)
    positions = np.arange(n)
    if density_bias > 0:
        weights = 1.0 + density_bias * np.sin(np.pi * positions / n)
    elif density_bias < 0:
        weights = 1.0 + abs(density_bias) * np.cos(2 * np.pi * positions / n)
    else:
        weights = np.ones(n)
    weights = weights / weights.sum()
    perm = rng.permutation(n)
    h[perm[:target]] = 1.0
    return h


def compute_corr_array(h):
    one_minus_h = 1.0 - h
    return np.correlate(h, one_minus_h, mode='full').astype(np.float64)


def get_top_lags(corr, n, k=10):
    idx = np.argsort(-corr)[:k]
    lags = idx - (n - 1)
    return lags, idx


def targeted_local_search(h_init, target, max_iters, rng, time_limit, t_start,
                          t_init=0.03, t_final=0.00005, top_k=8):
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)
    corr = compute_corr_array(h)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    best_h = h.copy()
    best_c5 = cur_c5
    best_corr = corr.copy()
    no_imp = 0
    max_no_imp = max(3000, max_iters // 8)

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for it in range(max_iters):
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        if no_imp > max_no_imp:
            perturb_strength = min(20, 3 + no_imp // 500)
            h = best_h.copy()
            for _ in range(perturb_strength):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = int(ones[rng.randint(len(ones))])
                    pj = int(zeros[rng.randint(len(zeros))])
                    if pi != pj:
                        h[pi] = 0.0
                        h[pj] = 1.0
            h = project_binary(h, target_int).astype(np.float64)
            corr = compute_corr_array(h)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            continue

        ratio = it / max(max_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio

        move_type = rng.randint(6)
        valid = False
        saved_state = None

        if move_type == 0 and len(ones) > 0 and len(zeros) > 0:
            top_lags, top_idx = get_top_lags(corr, n, top_k)
            contrib_score = np.zeros(len(ones))
            for li, li_idx in zip(top_lags, top_idx):
                for ci in range(len(ones)):
                    oi = int(ones[ci])
                    pos = oi + li
                    if 0 <= pos < n and h[pos] == 0:
                        contrib_score[ci] += 1.0 / (1 + abs(li))
            if contrib_score.max() > 0:
                weights = contrib_score + 0.01
                weights /= weights.sum()
                ci = rng.choice(len(ones), p=weights)
            else:
                ci = rng.randint(len(ones))
            ii = int(ones[ci])
            jj = int(zeros[rng.randint(len(zeros))])
            if ii == jj:
                continue
            saved_state = [(ii, 1.0), (jj, 0.0)]
            h[ii] = 0.0
            h[jj] = 1.0
            valid = True
        elif move_type == 1 and len(ones) > 0 and len(zeros) > 0:
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            if ii == jj:
                continue
            saved_state = [(ii, 1.0), (jj, 0.0)]
            h[ii] = 0.0
            h[jj] = 1.0
            valid = True
        elif move_type == 2 and len(ones) >= 2 and len(zeros) >= 2:
            top_lags, top_idx = get_top_lags(corr, n, min(top_k, 5))
            main_lag = int(top_lags[0])
            bad_ones = []
            for ci in range(len(ones)):
                oi = int(ones[ci])
                pos = oi + main_lag
                if 0 <= pos < n and h[pos] == 0:
                    bad_ones.append(ci)
            if len(bad_ones) > 0:
                ci = bad_ones[rng.randint(len(bad_ones))]
                ii = int(ones[ci])
            else:
                ii = int(ones[rng.randint(len(ones))])
            good_zeros = []
            for zi in range(len(zeros)):
                oz = int(zeros[zi])
                pos = oz + main_lag
                if 0 <= pos < n and h[pos] == 1:
                    good_zeros.append(zi)
            if len(good_zeros) > 0:
                zi = good_zeros[rng.randint(len(good_zeros))]
                jj = int(zeros[zi])
            else:
                jj = int(zeros[rng.randint(len(zeros))])
            if ii == jj:
                continue
            saved_state = [(ii, 1.0), (jj, 0.0)]
            h[ii] = 0.0
            h[jj] = 1.0
            valid = True
        elif move_type == 3 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])
            if i1 == i2:
                continue
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if j1 == j2:
                continue
            saved_state = [(i1, 1.0), (i2, 1.0), (j1, 0.0), (j2, 0.0)]
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            valid = True
        elif move_type == 4 and n > 20:
            block_len = rng.randint(2, max(3, n // 30) + 1)
            start = rng.randint(0, n - block_len)
            saved_state = []
            for pos in range(start, start + block_len):
                saved_state.append((pos, h[pos]))
                h[pos] = 1.0 - h[pos]
            ones_tmp = np.where(h == 1)[0]
            zeros_tmp = np.where(h == 0)[0]
            cur_sum = int(h.sum())
            if cur_sum != target_int:
                for pos, old_val in saved_state:
                    h[pos] = old_val
                continue
            valid = True
        else:
            top_lags, top_idx = get_top_lags(corr, n, top_k)
            if len(ones) > 0 and len(zeros) > 0:
                scores_i = np.zeros(len(ones))
                scores_j = np.zeros(len(zeros))
                for li, li_idx in zip(top_lags[:min(5, len(top_lags))], top_idx[:min(5, len(top_idx))]):
                    weight = 1.0 / (1 + abs(int(li)))
                    for ci in range(len(ones)):
                        oi = int(ones[ci])
                        pos = oi + int(li)
                        if 0 <= pos < n and h[pos] == 0:
                            scores_i[ci] += weight
                    for zi in range(len(zeros)):
                        oz = int(zeros[zi])
                        pos = oz + int(li)
                        if 0 <= pos < n and h[pos] == 1:
                            scores_j[zi] += weight
                wi = scores_i + 0.01
                wi /= wi.sum()
                wj = scores_j + 0.01
                wj /= wj.sum()
                ci = rng.choice(len(ones), p=wi)
                zi = rng.choice(len(zeros), p=wj)
                ii = int(ones[ci])
                jj = int(zeros[zi])
                if ii == jj:
                    continue
                saved_state = [(ii, 1.0), (jj, 0.0)]
                h[ii] = 0.0
                h[jj] = 1.0
                valid = True

        if not valid:
            continue

        new_corr = compute_corr_array(h)
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        delta = new_c5 - cur_c5

        accepted = False
        if delta < -1e-15:
            accepted = True
        elif temp > 1e-14 and delta < 0.005:
            if rng.random() < np.exp(-delta / temp):
                accepted = True

        if accepted:
            corr = new_corr
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                best_corr = corr.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            for pos, old_val in saved_state:
                h[pos] = old_val
            no_imp += 1

    return best_h, best_c5, best_corr


def smooth_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def _cons_fun(h, target):
    return np.sum(h) - target


def _cons_jac(h, target):
    return np.ones(len(h))


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 500, 1000, 2000]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return smooth_objective(x, alpha, n)

            def cfun(x):
                return _cons_fun(x, target)

            def cjac(x):
                return _cons_jac(x, target)

            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return h_interp
