∫_A (1 - 1_{A-k}(x)) dx = |A| - |A ∩ (A+k)| = 1 - |A ∩ (A+k)|

By the Cauchy-Schwarz / Brunn-Minkowski type arguments, there are fundamental limits. But for the discrete problem, we can do better.

**Key idea - use the autocorrelation structure**: 
If we think of h as a feature vector, the autocorrelation R[d] = sum_i h[i]h[i+d] should be as uniform as possible (for d ≠ 0) to minimize the maximum overlap.

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    """SLSQP optimization with log-sum-exp smoothing."""
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x):
                corr = get_corr(x)
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500, 0)
                return c_max + np.log(np.sum(np.exp(shifted))) / alpha
            
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def lbfgsb_optimize(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, pen_factor=1e4):
    """L-BFGS-B with penalty for the equality constraint."""
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x):
                corr = get_corr(x)
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500, 0)
                smooth_val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
                penalty = pen_factor * (np.sum(x) - target) ** 2
                return smooth_val + penalty
            
            h_new = minimize(obj, h_cur, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14, 'gtol': 1e-8}).x
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def hill_climb_binary(h, target_sum, max_iter, rng):

            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
