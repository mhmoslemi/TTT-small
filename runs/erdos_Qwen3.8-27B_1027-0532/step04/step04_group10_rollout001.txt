    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:

            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad


class LSEObjective:
    def __init__(self, alpha, topk, target, penalty):
        self.alpha = alpha
        self.topk = topk
        self.target = target
        self.penalty = penalty
    
    def __call__(self, x):
        f, g = lse_obj_grad(x, self.alpha, self.topk)
        s = np.sum(x) - self.target
        return f + self.penalty * s * s, g + 2.0 * self.penalty * s


class LSEObjectiveNoPenalty:
    def __init__(self, alpha, topk):
        self.alpha = alpha
        self.topk = topk
    
    def __call__(self, x):
        return lse_obj_grad(x, self.alpha, self.topk)


def run_lbfgs(h_start, target, alpha, topk, maxiter, ftol, penalty):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    obj = LSEObjective(alpha, topk, target, penalty)
    try:
        res = minimize(obj, h_start, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def run_slsqp(h_start, target, alpha, topk, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    obj = LSEObjectiveNoPenalty(alpha, topk)
    
    class SumConstraint:
        def __init__(self, target):
            self.target = target
        def fun(self, x):
            return np.sum(x) - self.target
        def jac(self, x):
            return np.ones(len(x))
    
    cons = [{'type': 'eq', 'fun': SumConstraint(target).fun, 'jac': SumConstraint(target).jac}]
    
    try:
        res = minimize(obj, h_start, jac=True, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def resample_h(h, n_old, n_new):
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h)


def multi_perturb_optimize(h, n, target, rng, n_restarts, alpha, topk, maxiter, penalty):
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(n_restarts):
        perturb_scale = 0.003
        h_try = project_feasible(h + rng.randn(n) * perturb_scale, target)
        h_new = run_lbfgs(h_try, target, alpha, topk, maxiter, 1e-15, penalty)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
    return best_h, best_score


def coordinate_descent(h, target, n_steps, step_sizes):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for step in step_sizes:
        for coord in range(n):
            orig = best_h[coord]
            for delta in [step, -step]:
                h_try = best_h.copy()
                new_val = np.clip(orig + delta, 0.0, 1.0)
                h_try[coord] = new_val
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try
                    break
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad


class LSEPenObj:
    def __init__(self, alpha, topk, target, penalty):
        self.alpha = alpha
        self.topk = topk
        self.target = target
        self.penalty = penalty

    def __call__(self, x):
        f, g = lse_obj_grad(x, self.alpha, self.topk)
        s = np.sum(x) - self.target
        return f + self.penalty * s * s, g + 2.0 * self.penalty * s


class LSEObj:
    def __init__(self, alpha, topk):
        self.alpha = alpha
        self.topk = topk

    def __call__(self, x):
        return lse_obj_grad(x, self.alpha, self.topk)


class SumEq:
    def __init__(self, target):
        self.target_val = target

    def fun(self, x):
        return np.sum(x) - self.target_val

    def jac(self, x):
        return np.ones(len(x))


def run_lbfgs(h_start, target, alpha, topk, maxiter, ftol, penalty):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    obj = LSEPenObj(alpha, topk, target, penalty)
    try:
        res = minimize(obj, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def run_slsqp(h_start, target, alpha, topk, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    obj = LSEObj(alpha, topk)
    cons_obj = SumEq(target)
    cons = [{'type': 'eq', 'fun': cons_obj.fun, 'jac': cons_obj.jac}]
    try:
        res = minimize(obj, h_start, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def resample_h(h, n_old, n_new):
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h)


def gradient_perturb(h, alpha, topk, scale, rng):
    _, grad = lse_obj_grad(h, alpha, topk)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return h
    direction = -grad / grad_norm
    h_try = project_feasible(h + scale * direction + rng.randn(len(h)) * scale * 0.1, np.sum(h))
    return h_try


def coord_search(h, target, deltas):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for d in deltas:
        for c in range(n):
            orig = best_h[c]
            for delta in [d, -d]:
                h_try = best_h.copy()
                h_try[c] = np.clip(orig + delta, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try
                    break
    return best_h, best_score
