def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]

    h = 1.0 - np.abs(x - 1.0)
    cands.append(project_feasible(h, n / 2.0))
    h = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(2 * np.pi * x)
    cands.append(project_feasible(h, n / 2.0))
    for _ in range(2):
        h = 0.3 + 0.4 * rng.rand(n)
        cands.append(project_feasible(h, n / 2.0))
    return cands


def targeted_lag_reduction(h, target, bounds, n_rounds, time_fn):
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(n_rounds):
        if time_fn() < 15:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top5 = np.argsort(corr)[-5:]
        for tidx in top5:
            if time_fn() < 10:
                break
            lag = tidx - (len(best_h) - 1)
            n = len(best_h)
            grad_one = np.zeros(n)
            ome = 1.0 - best_h
            if 0 <= lag < n:
                grad_one[lag:n] += ome[:n - lag]
                grad_one[:n - lag] -= best_h[lag:n]
            elif 0 < -lag < n:
                m = -lag
                grad_one[:n - m] += ome[m:n]
                grad_one[m:n] -= best_h[:n - m]
            for step in [0.03, 0.015, 0.008, 0.004]:
                h_try = best_h - step * grad_one
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try
                    break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    penalty = 5000.0

    # Phase 1: Polish with increasing alpha
    for alpha in [80, 200, 500, 1200, 3000]:
        if tl() < 50:
            break
        for attempt in range(3):
            if tl() < 20:
                break
            scale = 0.005 / (attempt + 1)
            h_start = best_h + rng.randn(n) * scale
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_step(h_start, target, alpha, 50, penalty, 600, bounds, 1e-16)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 2: Targeted lag reduction
    if tl() > 80:
        best_h, best_score = targeted_lag_reduction(best_h, target, bounds, 15, tl)

    # Phase 3: Multi-resolution exploration
    x_fine = np.linspace(0, 2, n)
    for n_coarse in [100, 150, 200, 300]:
        if tl() < 90:
            break
        target_c = n_coarse / 2.0
        bounds_c = [(0.0, 1.0)] * n_coarse
        x_coarse = np.linspace(0, 2, n_coarse)

        cands = make_diverse_candidates(n_coarse, rng)
        h_interp = np.interp(x_coarse, x_fine, best_h)
        cands.append(project_feasible(h_interp, target_c))

        for h_cand in cands:
            if tl() < 25:
                break
            for alpha in [200, 700]:
                if tl() < 12:
                    break
                h_try = h_cand + rng.randn(n_coarse) * 0.008
                h_try = project_feasible(h_try, target_c)
                h_new = lbfgs_step(h_try, target_c, alpha, 30, penalty, 500, bounds_c, 1e-15)
                h_up = np.interp(x_fine, x_coarse, h_new)
                h_up = project_feasible(h_up, target)
                score = compute_c5(h_up)
                if score < best_score:
                    best_score = score
                    best_h = h_up

    # Phase 4: High-alpha refinement at full resolution
    if tl() > 70:
        for alpha in [4000, 10000]:
            if tl() < 25:
                break
            for attempt in range(3):
                if tl() < 12:
                    break
                h_start = best_h + rng.randn(n) * 0.001
                h_start = project_feasible(h_start, target)
                h_new = lbfgs_step(h_start, target, alpha, 35, penalty, 500, bounds, 1e-17)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new

    # Phase 5: More targeted reduction
    if tl() > 60:
        best_h, best_score = targeted_lag_reduction(best_h, target, bounds, 20, tl)

    # Phase 6: SLSQP polish
    if tl() > 45:
        slq_alpha = 800
        slq_topk = 25

        def slq_obj(x):
            corr, idx = get_topk_corr(x, slq_topk)
            ct = corr[idx]
            cm = np.max(ct)
            sh = np.clip(slq_alpha * (ct - cm), -500.0, 0.0)
            ev = np.exp(sh)
            return cm + np.log(np.sum(ev)) / slq_alpha

        def slq_jac(x):
            corr, idx = get_topk_corr(x, slq_topk)
            ct = corr[idx]
            cm = np.max(ct)
            sh = np.clip(slq_alpha * (ct - cm), -500.0, 0.0)
            ev = np.exp(sh)
            tot = np.sum(ev)
            wts = ev / tot
            nn = len(x)
            g = np.zeros(nn)
            ome = 1.0 - x
            for i in range(len(idx)):
                w = wts[i]
                if w < 1e-15:
                    continue
                lag = idx[i] - (nn - 1)
                if 0 <= lag < nn:
                    g[lag:nn] += w * ome[:nn - lag]
                    g[:nn - lag] -= w * x[lag:nn]
                elif 0 < -lag < nn:
                    m = -lag
                    g[:nn - m] += w * ome[m:nn]
                    g[m:nn] -= w * x[:nn - m]
            return g

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(slq_obj, best_h.copy(), jac=slq_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new
        except Exception:
            pass

    # Phase 7: Final L-BFGS
    if tl() > 20:
        h_new = lbfgs_step(best_h, target, 8000, 20, penalty, 500, bounds, 1e-18)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 8: Micro coordinate search
    if tl() > 20:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top10 = np.argsort(corr)[-10:]
        for tidx in top10:
            if tl() < 10:
                break
            lag = tidx - (n - 1)
            if 0 <= lag < n:
                coords = np.arange(lag, n)
            elif 0 < -lag < n:
                m = -lag
                coords = np.arange(0, n - m)
            else:
                continue
            step_size = max(1, len(coords) // 15)
            for ci in coords[::step_size]:
                if tl() < 8:
                    break
                orig = best_h[ci]
                for delta in [0.008, -0.008, 0.004, -0.004, 0.002, -0.002]:
                    h_try = best_h.copy()
                    h_try[ci] = np.clip(orig + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_h = h_try
                        break

    # Final
    if tl() > 10:
        h_new = lbfgs_step(best_h, target, 10000, 15, penalty, 400, bounds, 1e-18)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
