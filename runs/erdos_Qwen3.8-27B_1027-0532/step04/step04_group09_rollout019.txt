    if S <= 0 or S >= len(v):

        x = np.clip(v, 0, 1)
        # Adjust sum if needed
        diff = S - np.sum(x)
        if diff > 0:

            x[mask] += diff / mask.sum()
        else:
            mask = x > 0

    """Compute gradient of log-sum-exp objective."""
    n = len(h)
    corr = compute_corr(h)
    m = len(corr)  # 2n-1
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()

    grad = np.zeros(n)
    
    # For each lag k (from -(n-1) to n-1), the contribution to grad[p] is:
    # If k >= 0:
    #   grad[p] += w[n-1+k] * (1-h[p+k]) for p <= n-1-k
    #   grad[p] += w[n-1+k] * (-h[p-k]) for k <= p <= n-1
    # If k < 0 (kk = -k):
    #   grad[p] += w[n-1-kk] * (1-h[p-kk]) for kk <= p <= n-1
    #   grad[p] += w[n-1-kk] * (-h[p+kk]) for p <= n-1-kk
    
    # Let's compute this using vectorized operations
    # For k >= 0: contribution to grad from first factor
    #   For p = 0 to n-1-k: grad[p] += w[n-1+k] * (1-h[p+k])
    #   This is: grad[0:n-k] += w[n-1+k] * (1-h[k:n])
    
    # For k >= 0: contribution to grad from second factor
    #   For p = k to n-1: grad[p] += w[n-1+k] * (-h[p-k])
    #   This is: grad[k:n] += w[n-1+k] * (-h[0:n-k])
    
    for k in range(n):
        m_idx = n - 1 + k  # correlation index for lag k
        wk = w[m_idx]
        if wk < 1e-15:
            continue
        # First factor: h[p], p = 0 to n-1-k
        grad[:n-k] += wk * (1.0 - h[k:])
        # Second factor: (1-h[p]), p = k to n-1, i.e., h[p-k] with p-k = 0 to n-1-k
        grad[k:] += wk * (-h[:n-k])
    
    # For k < 0, let kk = -k, so kk goes from 1 to n-1
    for kk in range(1, n):
        m_idx = n - 1 - kk  # correlation index for lag -kk
        wk = w[m_idx]
        if wk < 1e-15:
            continue
        # First factor: h[p], p = kk to n-1, h[p-kk] with p-kk = 0 to n-1-kk
        # c_{-kk} = sum_{i=0}^{n-1-kk} h[i+kk](1-h[i])
        # ∂/∂h[p]: h[p] appears as h[i+kk] when i+kk=p, i=p-kk
        #   contribution: (1-h[p-kk]) for p = kk to n-1
        # h[p] appears as (1-h[i]) when i=p
        #   contribution: -h[p+kk] for p = 0 to n-1-kk
        grad[kk:] += wk * (1.0 - h[:n-kk])
        grad[:n-kk] += wk * (-h[kk:])

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_box_sum(v, S):
    """Project v onto {0 <= x <= 1, sum(x) = S}."""
    v = np.asarray(v, dtype=float).copy()
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    sum_exp = float(np.sum(exp_v))

Now let me think about whether the momentum method will actually work well here. The key issue is:

1. The LSE objective is smooth but non-convex
2. The feasible set is a convex polytope (box ∩ hyperplane)
3. Gradient descent with projection should converge to a stationary point
4. The question is whether the stationary point is good enough

For large alpha, the LSE concentrates the weight on the max correlation, so the method effectively minimizes the max correlation. This is what we want.

One concern: with very high alpha, the LSE becomes almost non-smooth (it looks like a max), and gradient methods struggle. So there's a sweet spot for alpha.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_box_sum(v, S):
    v = np.asarray(v, dtype=float).copy()
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        grad[:n - k] += wk * (1.0 - h[k:])
        grad[k:] += wk * (-h[:n - k])
    for kk in range(1, n):
        wk = w[n - 1 - kk]
        if wk < 1e-15:
            continue
        grad[kk:] += wk * (1.0 - h[:n - kk])
        grad[:n - kk] += wk * (-h[kk:])
    return obj, grad

        x_new = project_box_sum(y - step * grad_y, target)

    bounds = [(0.0, 1.0)] * n

        return lse_obj_and_grad(x, alpha)[1]
