**Better approach: Incremental correlation update**

If we can update the correlation incrementally when we swap two positions, we can make each iteration O(n) instead of O(n²).

When we swap h[a]=1→0 and h[b]=0→1:
For each lag k, c[k] changes by Δc[k] as computed above.
The non-zero Δc[k] values are only for k in a specific range:
- k ≤ a (a as target): contributes h[a-k]-1
- k ≤ n-1-a (a as source): contributes h[a+k]
- k ≤ b (b as target): contributes 1-h[b-k]
- k ≤ n-1-b (b as source): contributes -h[b+k]
- Special cases when |a-b|=k

So for a given swap (a,b), the lags k that are affected are those where at least one of:
- k ≤ a (and a-k is valid)
- k ≤ n-1-a (and a+k is valid)
- k ≤ b
- k ≤ n-1-b

This is essentially all k from 0 to max(a, b, n-1-a, n-1-b), which could be up to n-1. So it's still O(n) per swap to update all affected lags.

But the key is: we only need to track the MAX of c[k], not all of them. So if we maintain a data structure that tracks the current max, we can update it in O(n) per swap (just update the affected lags and check if the max changed).

This gives us O(n) per iteration instead of O(n²), allowing many more iterations.

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]

    for k in range(1, n):

        full_delta[center - k] += delta[k]
    
    return full_delta

                d_partial[1:n-a] += h[a+1:n]
            if a >= 1:
                d_partial[1:a+1] += h[a-1::-1] - 1.0

            b_neg = zeros - worst_k
            b_pos = zeros + worst_k
            candidate_b = zeros
            scores = np.zeros(len(candidate_b))
            vn = (b_neg >= 0) & (b_neg < n)
            scores[vn] += h[b_neg[vn]]
            vn2 = (b_pos >= 0) & (b_pos < n)
            scores[vn2] += h[b_pos[vn2]]
            scores += rng.random(len(scores)) * 0.05
            top_b = candidate_b[np.argsort(scores)[-min(20, len(candidate_b)):]]

            for b in top_b:
                d_b = np.zeros(n)
                if n - 1 - b >= 1:
                    d_b[1:n-b] -= h[b+1:n]
                    if b > a and (b - a) < (n - a):
                        d_b[b-a] += 1.0
                if b >= 1:
                    d_b[1:b+1] += 1.0 - h[b-1::-1]
                if a > b and (a - b) <= a:
                    pass
                d_full = d_partial + d_b
                if a > b and (a - b) <= a:
                    d_full[a-b] += 1.0
                new_c5_val = (half_corr + d_full)
                max_at_worst = new_c5_val[worst_k]
                new_max = np.max(new_c5_val)
                dc = new_max * dx - c5
                if dc < best_delta_c5:
                    best_delta_c5 = dc
                    best_a = a
                    best_b = b

        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            half_corr = get_half_corr(h)
            c5 = np.max(half_corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_fun_target = target
    cons_fun_n = n
    dx_val = 2.0 / n

    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full') * dx_val
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def obj_grad(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full') * dx_val
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / np.sum(exp_shifted)
        grad = np.zeros(n)
        for k in range(n):
            p_k = p[n - 1 + k]
            if k > 0:
                p_k += p[n - 1 - k]
            if p_k < 1e-15:
                continue
            g = np.zeros(n)
            if n - k > 0:
                g[k:] = 1.0 - h_in[:n-k]
            if k < n:
                g[:n-k] -= h_in[k:]
            grad += p_k * g * dx_val
        return np.sum(exp_shifted) / np.sum(exp_shifted) * 0 + grad

    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target

        return np.ones(cons_fun_n)
