    h = h0.copy()
    h = project_feasible(h, target)
    y = h.copy()
    t = 1.0
    L = 1.0  # Initial step size (will be adjusted)
    
    for it in range(max_iter):
        f = pnorm_value(y, p, dx)
        g = pnorm_grad(y, p, dx)
        
        # Gradient step
        z = y - g / L
        # Project
        h_new = project_feasible(z, target)
        
        # Check convergence
        if np.max(np.abs(h_new - h)) < tol:
            break
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = h_new + ((t-1)/t_new) * (h_new - h)
        h = h_new
        t = t_new
    
    h = project_feasible(h, target)
    score = compute_c5(h)
    return h, score
