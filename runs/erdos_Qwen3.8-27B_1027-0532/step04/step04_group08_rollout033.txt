def evaluate_swap(corr, h, a, b, n):
    """Compute the change in correlation when swapping h[a]:1->0, h[b]:0->1.
    Returns the new correlation vector (or just the max)."""
    # c_m = sum_n h[n+m] * (1-h[n])
    # Changing h[a] from 1 to 0 and h[b] from 0 to 1:
    # Δc_m = -(1-h[a-m]) + (1-h[b-m]) + h[a+m] - h[b+m]
    # where each term is 0 if index out of [0, n-1]
    
    dm = np.zeros(2*n - 1)
    # -(1-h[a-m]) for valid a-m
    # a-m in [0, n-1] => m in [a-(n-1), a]
    lo1 = max(0, a - (n-1))

        idx = np.arange(lo1, hi1+1)
        a_minus_m = a - idx
        dm[lo1:hi1+1] -= (1.0 - h[a_minus_m])
    
    # +(1-h[b-m]) for valid b-m
    lo2 = max(0, b - (n-1))
    hi2 = min(b, 2*n-2)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2+1)
        b_minus_m = b - idx
        dm[lo2:hi2+1] += (1.0 - h[b_minus_m])
    
    # +h[a+m] for valid a+m
    lo3 = max(0, -a)
    hi3 = min(n-1-a, 2*n-2)
    if lo3 <= hi3:
        idx = np.arange(lo3, hi3+1)

        idx = np.arange(lo4, hi4+1)
        b_plus_m = b + idx
        dm[lo4:hi4+1] -= h[b_plus_m]
    
    return dm

def greedy_pass(h, corr, n, dx):
    """One pass of greedy swap. Returns best h found."""
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    # Precompute L and R matrices
    # L[i, m] = 1-h[i-k] where k=m-(n-1), valid if 0<=i-k<n
    # R[i, m] = h[i+k] where k=m-(n-1), valid if 0<=i+k<n
    L = np.zeros((n, 2*n-1))
    R = np.zeros((n, 2*n-1))
    for i in range(n):
        # L[i]: m from max(0, i-n+1) to min(2n-2, i+n-1)
        lo = max(0, i - n + 1)
        hi = min(2*n-2, i + n - 1)
        if lo <= hi:
            idx = i - (np.arange(lo, hi+1) - (n-1))  # = i - m + n - 1
            L[i, lo:hi+1] = 1.0 - h[idx]
        
        # R[i]: m from max(0, n-1-i) to min(2n-2, n-1+i)
        lo = max(0, n - 1 - i)
        hi = min(2*n-2, n - 1 + i)
        if lo <= hi:
            idx = i + (np.arange(lo, hi+1) - (n-1))  # = i + m - n + 1
            R[i, lo:hi+1] = h[idx]
    
    # For each swap (a, b), compute delta
    # delta[a,b] = -L[a] + L[b] + R[a] - R[b] + corrections
    best_val = float(np.max(corr))
    best_a = -1
    best_b = -1
    
    # Process in batches
    batch_size = 2000
    all_pairs = [(a, b) for a in ones for b in zeros]
    # Actually this is 200*200 = 40000 pairs. Let me process them.
    
    for start in range(0, len(all_pairs), batch_size):
        batch = all_pairs[start:start+batch_size]
        if not batch:
            break
        batch_a = np.array([p[0] for p in batch])
        batch_b = np.array([p[1] for p in batch])
        
        # delta shape: (batch_size, 2n-1)
        delta = -L[batch_a] + L[batch_b] + R[batch_a] - R[batch_b]
        
        # Add corrections
        # -2 at m = n-1 (k=0)
        delta[:, n-1] -= 2.0
        # +1 at m = n-1+a-b
        for i, (a, b) in enumerate(batch):

            if 0 <= m_ab < 2*n-1:
                delta[i, m_ab] += 1.0
            m_ba = n - 1 + b - a
            if 0 <= m_ba < 2*n-1:
                delta[i, m_ba] += 1.0
        
        new_corr = corr[None, :] + delta

                a, b = batch[i]
                best_a = a
                best_b = b

    # Apply the best swap
    h = h.copy()
    h[best_a] = 0.0
    h[best_b] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    return h, corr, best_val

    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:

        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

        valid = (idx >= 0) & (idx < n)
        delta[valid] -= h[idx[valid]]
        delta[n - 1] -= 2.0
        m_ab = n - 1 + a - b
        m_ba = n - 1 + b - a
        if 0 <= m_ab < 2 * n - 1:
            delta[m_ab] += 1.0
        if 0 <= m_ba < 2 * n - 1:
            delta[m_ba] += 1.0
        new_corr = corr + delta
        new_val = float(np.max(new_corr))
        accept = new_val < current_val
        if not accept and temp > 1e-8:
            log_ratio = (current_val - new_val) / temp
            if log_ratio > -700 and rng.random() < np.exp(log_ratio):
                accept = True
        if accept:
            h[a] = 0.0
            h[b] = 1.0
            ones[ones.index(a)] = b
            zeros[zeros.index(b)] = a
            corr = new_corr
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
    return best_h, best_val * dx


def soft_max_obj(x, target, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def cons_eq(x, target):
    return np.sum(x) - target


def continuous_optimize(h, n, target, deadline, alpha=5000):
    def obj(x):
        return soft_max_obj(x, target, alpha)

    def cons(x):
        return cons_eq(x, target)

    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': 400, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if time.time() < deadline:
            return h_new, sc
    except Exception:
        pass
    return h, compute_c5(h)


def make_hump_structure(n, n_humps, rng):
    h = np.zeros(n)
    total_ones = n // 2
    lengths = np.ones(n_humps, dtype=int)
    remaining = total_ones - n_humps
    for _ in range(remaining):
        lengths[rng.randint(n_humps)] += 1
    positions = []
    used = 0
    sorted_len = lengths.copy()
    min_gap = max(2, n // (n_humps * 4))
    for i in range(n_humps):
        gaps = n - used - (n_humps - i) * min_gap
        if gaps <= 0:
            min_gap = 1
        g = rng.randint(0, max(1, gaps - (n_humps - i - 1) * min_gap)) + (1 if i > 0 else 0)
        if i == 0:
            g = rng.randint(0, max(1, n - sum(lengths)))
        pos = used + g
        if pos + sorted_len[i] > n:
            pos = n - sorted_len[i]
        if pos < 0:
            pos = 0
        positions.append(pos)
        used = pos + sorted_len[i]
    for pos, ln in zip(positions, lengths):
        h[pos:pos + ln] = 1.0
    if h.sum() != n // 2:
        h = to_binary(h, n)
    return h

    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binary local search from initial thresholding
    h_bin = to_binary(best_h, n)
    dl = start_time + _tl() * 0.25
    h_bin, sc = greedy_binary_search(h_bin, n, max_iters=60, deadline=dl)
    if sc < best_score:
        best_score = sc
        best_h = h_bin.copy()

    # Phase 2: Continuous polish from binary
    if _tl() > 80:
        dl = start_time + _tl() * 0.25
        h_c, sc = continuous_optimize(h_bin, n, target, dl, alpha=5000)
        if sc < best_score:
            best_score = sc
            best_h = h_c

    # Phase 3: Simulated annealing on binary
    if _tl() > 150:
        h_sa_start = to_binary(best_h, n)
        for trial in range(4):
            if _tl() < 40:
                break
            dl = start_time + _tl() * 0.2
            h_sa, sc = simulated_annealing_binary(h_sa_start, n, 0.08, 0.0005, 80000, rng, deadline=dl)
            if sc < best_score:
                best_score = sc
                best_h = h_sa.copy()
            h_sa_start = h_sa
            # Quick greedy polish after SA
            if _tl() > 30:
                h_g, sc2 = greedy_binary_search(h_sa_start, n, max_iters=20, deadline=start_time + _tl() * 0.15)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_g.copy()
                    h_sa_start = h_g

    # Phase 4: More binary local search
    if _tl() > 100:
        h_bin2 = to_binary(best_h, n)
        dl = start_time + _tl() * 0.25
        h_bin2, sc = greedy_binary_search(h_bin2, n, max_iters=60, deadline=dl)
        if sc < best_score:
            best_score = sc
            best_h = h_bin2.copy()

    # Phase 5: Structured multi-start
    if _tl() > 120:
        for nh in [5, 7, 10, 8]:
            if _tl() < 30:
                break
            h_struct = make_hump_structure(n, nh, rng)
            dl = start_time + _tl() * 0.2
            h_struct, sc = greedy_binary_search(h_struct, n, max_iters=30, deadline=dl)
            if sc < best_score:
                best_score = sc
                best_h = h_struct.copy()

    if _tl() > 100:
        for trial in range(6):
            if _tl() < 25:
                break

            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            dl = start_time + _tl() * 0.15

                best_h = h_rand.copy()
