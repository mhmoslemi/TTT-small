g_{-m}(h) = sum_{i'=0}^{n-1-m} h[i'+m] * (1-h[i']) * dx

∂/∂h[j]:
- j = i'+m: valid if 0 ≤ j-m ≤ n-1-m, i.e., m ≤ j ≤ n-1: contributes (1-h[j-m]) * dx
- j = i': valid if 0 ≤ j ≤ n-1-m: contributes -h[j+m] * dx

So ∂g_{-m}/∂h[j] = dx * [(1-h[j-m]) * 1_{j ≥ m} - h[j+m] * 1_{j ≤ n-1-m}]

In general, for lag k (positive or negative):
∂g_k/∂h[j] = dx * [(1-h[j+k]) * 1_{0≤j+k<n, j in range} - h[j-k] * 1_{0≤j-k<n, j in range}]

Where "j in range" means j is a valid index in the sum. Let me just use the simpler condition:
- First term: j+k is valid (0 ≤ j+k < n) AND j is valid (0 ≤ j < n) AND j is in the "first" part of the sum
- Second term: j-k is valid AND j is valid AND j is in the "second" part

Actually, the simplest correct formulation:
For any lag k, g_k(h) = sum_i h[i] * (1-h[i+k]) * dx where the sum is over all i such that 0≤i<n and 0≤i+k<n.

∂g_k/∂h[j] = dx * [(1-h[j+k]) if 0≤j<n and 0≤j+k<n] + [-h[j-k] if 0≤j<n and 0≤j-k<n]
= dx * [(1-h[j+k]) * 1_{j+k ≥ 0, j+k < n} - h[j-k] * 1_{j-k ≥ 0, j-k < n}]

This is symmetric and correct for both positive and negative k. Let me verify:
- For k > 0: j+k < n means j < n-k (first part), j-k ≥ 0 means j ≥ k (second part). ✓
- For k < 0 (k = -m): j+k = j-m ≥ 0 means j ≥ m (first part), j-k = j+m < n means j < n-m (second part). ✓

Great. So the subgradient is:
grad[j] = dx * [(1 - h[j+k*]) if (0 ≤ j+k* < n) else 0] - dx * [h[j-k*] if (0 ≤ j-k* < n) else 0]

2. Compute subgradient at k*
3. Update: h_new = h - lr * grad

For the projection, I'll use the water-filling approach:
- Clip to [0,1]
- If sum > n/2, reduce values (starting from highest) until sum = n/2
- If sum < n/2, increase values (starting from lowest) until sum = n/2

Actually, I think a better approach is to use the "smoothed" objective and do gradient descent:

f_smooth(h) = (1/alpha) * log(sum_k exp(alpha * g_k(h)))

∇f_smooth(h) = (1/sum_k exp(alpha*g_k)) * sum_k exp(alpha*g_k) * ∇g_k(h)
= sum_k p_k * ∇g_k(h) where p_k = exp(alpha*g_k) / sum_j exp(alpha*g_j)

This is a weighted sum of subgradients, with weights proportional to exp(alpha * g_k). For large alpha, it concentrates on the max, recovering the subgradient of the max.

This is actually a smooth (differentiable) objective, so we can use standard gradient descent or even L-BFGS!

```python
def projected_smooth_descent(h_init, n, target, alpha, max_iter, lr, rng, time_limit, start_time):
    """Projected gradient descent on smooth max correlation."""
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Compute weights (top-M for efficiency)
        corr = compute_corr(h)
        max_corr = np.max(corr)
        shifted = alpha * (corr - max_corr)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Only use top M weights for efficiency
        M = min(20, len(corr))
        top_idx = np.argsort(-weights)[:M]
        top_weights = weights[top_idx] / np.sum(weights[top_idx])
        
        # Compute gradient
        grad = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            w = top_weights[i]
            k = k_idx - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[0:n-k] += w * (2.0/n) * (1 - h[k:n])
                if k < n:
                    grad[k:n] -= w * (2.0/n) * h[0:n-k]
            else:
                m = -k
                if m < n:
                    grad[m:n] += w * (2.0/n) * (1 - h[0:n-m])
                if n - m > 0:
                    grad[0:n-m] -= w * (2.0/n) * h[m:n]
        
        # Update
        h = h - lr * grad
        h = project_feasible(h, target)
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score
```

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    # Iterative water-filling
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        offset = diff / n
        h = np.clip(h + offset, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    # Final adjustment
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to increase some values
            room = 1.0 - h
            order = np.argsort(-room)
            remaining = diff
            for i in order:
                if remaining <= 1e-12:
                    break
                add = min(room[i], remaining)
                h[i] += add
                remaining -= add
        else:
            order = np.argsort(h)
            remaining = -diff
            for i in order:
                if remaining <= 1e-12:
                    break
                sub = min(h[i], remaining)
                h[i] -= sub
                remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1

    max_corr = np.max(corr)

def slsqp_smooth(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    """SLSQP optimization with smooth max objective."""
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(_smooth_obj_alpha, h_cur, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score

def _smooth_obj_alpha(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500, 0)
    return max_corr + np.log(np.sum(np.exp(shifted))) / alpha

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(30):
        if abs(diff) < 1e-12:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            remaining = diff
            for i in order:
                if remaining <= 1e-12:
                    break
                add = min(room[i], remaining)
                h[i] += add
                remaining -= add
        else:
            order = np.argsort(h)
            remaining = -diff
            for i in order:
                if remaining <= 1e-12:
                    break
                sub = min(h[i], remaining)
                h[i] -= sub
                remaining -= sub
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == int(target_sum):
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    target_int = int(target_sum)
    if current_sum > target_int:
        need = int(current_sum - target_int)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_int - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_int:
        if final_sum > target_int:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_int]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_int - final_sum]] = 1
    return h_r.astype(float)


def thue_morse_pattern(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = np.binary_repr(i).count('1') % 2
    return t.astype(float)


def smooth_grad(h, alpha, top_m=40):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-300:
        return np.zeros(n)
    weights = weights / total_w
    if len(corr) > top_m:
        top_idx = np.argsort(-weights)[:top_m]
        mask = np.zeros(len(corr), dtype=bool)
        mask[top_idx] = True
        weights = np.where(mask, weights, 0.0)
        sw = np.sum(weights)
        if sw > 0:
            weights = weights / sw
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-12:
            continue
        k = ki - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[0:nk] += w * dx * (1.0 - h[k:n])
            if k < n:
                grad[k:n] -= w * dx * h[0:nk]
        else:
            m = -k
            nm = n - m
            if m < n:
                grad[m:n] += w * dx * (1.0 - h[0:nm])
            if nm > 0:
                grad[0:nm] -= w * dx * h[m:n]
    return grad


def projected_gradient_method(h_init, n, target, alpha_seq, lr, rng, time_limit, start_time):
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    max_no_improve = 5000
    it = 0
    while time.time() - start_time < time_limit:
        if it >= len(alpha_seq):
            it = len(alpha_seq) - 1
        alpha = alpha_seq[it]
        if no_improve > max_no_improve:
            h = h + rng.randn(n) * 0.05
            h = project_feasible(h, target)
            no_improve = 0
            it = 0
        grad = smooth_grad(h, alpha, top_m=40)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if it < len(alpha_seq) - 1 and it % 200 == 0:
                it += 1
        else:
            h = h_new
            no_improve += 1
        it += 1
    return best_h, best_score


def informed_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None, top_m=5):
    n = len(h_start)
    h = project_binary(h_start.copy(), target_sum)
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(5000, max_iters // 8)
    ones_cache = np.where(h == 1.0)[0]
    zeros_cache = np.where(h == 0.0)[0]

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.03:
                no_improve = 0
            else:
                break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        move_type = rng.randint(4)
        h_new = h.copy()
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            if rng.random() < 0.7:
                worst_idx = np.argmax(corr)
                k_star = worst_idx - (n - 1)
                target_pos = (i1 + k_star) % n
                dist = np.abs(zeros_cache.astype(int) - target_pos).astype(float)
                dist += rng.random(len(zeros_cache)) * 10.0
                i0 = zeros_cache[np.argmin(dist)]
            else:
                i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0.0
            h_new[i0] = 1.0
            moved = True
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            i1a = ones_cache[idx1[0]]
            i1b = ones_cache[idx1[1]]
            if rng.random() < 0.6:
                worst_idx = np.argmax(corr)
                k_star = worst_idx - (n - 1)
                t1 = (i1a + k_star) % n
                t2 = (i1b - k_star) % n
                d1 = np.abs(zeros_cache.astype(int) - t1).astype(float) + rng.random(len(zeros_cache)) * 8.0
                d2 = np.abs(zeros_cache.astype(int) - t2).astype(float) + rng.random(len(zeros_cache)) * 8.0
                i0a = zeros_cache[np.argmin(d1)]
                i0b = zeros_cache[np.argmin(d2)]
                if i0a == i0b:
                    i0b = zeros_cache[rng.randint(len(zeros_cache))]
            else:
                idx0 = rng.choice(len(zeros_cache), 2, replace=False)
                i0a = zeros_cache[idx0[0]]
                i0b = zeros_cache[idx0[1]]
            h_new[i1a] = 0.0
            h_new[i1b] = 0.0
            h_new[i0a] = 1.0
            h_new[i0b] = 1.0
            moved = True
        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            worst_indices = np.argsort(-corr)[:top_m]
            i1 = ones_cache[rng.randint(len(ones_cache))]
            best_sv = 1e9
            best_i0 = zeros_cache[0]
            num_cand = min(30, len(zeros_cache))
            cand_idx = rng.choice(len(zeros_cache), num_cand, replace=False)
            for ci in cand_idx:
                i0 = zeros_cache[ci]
                score_val = 0.0
                for wi in worst_indices[:top_m]:
                    k = wi - (n - 1)
                    delta = 0.0
                    if 0 <= i1 + k < n:
                        delta -= (1.0 - h[i1 + k])
                    if 0 <= i1 - k < n:
                        delta += h[i1 - k]
                    if 0 <= i0 + k < n:
                        delta += (1.0 - h[i0 + k])
                    if 0 <= i0 - k < n:
                        delta -= h[i0 - k]
                    score_val += max(0.0, corr[wi] + delta * (2.0 / n) - best_c5)
                if score_val < best_sv:
                    best_sv = score_val
                    best_i0 = i0
            h_new[i1] = 0.0
            h_new[best_i0] = 1.0
            moved = True
        elif move_type == 3 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0.0
            h_new[i0] = 1.0
            moved = True

        if not moved:
            continue
        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5

            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1.0)[0]
            zeros_cache = np.where(h == 0.0)[0]
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5


def _slsqp_obj(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    shifted = np.clip(alpha * (corr - mc), -500.0, 0.0)
    return mc + np.log(np.sum(np.exp(shifted))) / alpha


def _con_fun(h):
    return np.sum(h) - _cons_target


def _con_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    x = np.linspace(0.0, 2.0 * np.pi * 5, n)

    h_sin = project_binary(h_sin, n / 2.0)
    starts.append(h_sin)

        h_rand = np.zeros(n)
        perm = rng.permutation(n)

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()

    n_points = n_init
    starts = generate_starts(n_points, rng, 6)

    for si in range(len(starts)):
        if tleft() < 80:
            break
        h_s = starts[si]
        allowed_cont = min(50.0, tleft() * 0.25)
        if allowed_cont > 8:
            alpha_seq = [50.0, 100.0, 300.0, 800.0, 2000.0, 5000.0]
            h_cont, sc_cont = projected_gradient_method(
                h_s, n_points, target_init, alpha_seq,
                lr=100.0, rng=rng,
                time_limit=allowed_cont, start_time=start_time)
            if sc_cont < best_score:
                best_score = sc_cont
                best_h = h_cont.copy()
        h_for_sa = project_binary(best_h if compute_c5(best_h) < compute_c5(h_s) else h_s, target_init)
        allowed_sa = min(120.0, tleft() * 0.35)
        if allowed_sa > 8:
            max_it = int(min(100000, allowed_sa * 2500))
            h_sa, c5_sa = informed_sa(h_for_sa, target_init, max_it,
                                      0.015, 0.00005, rng,
                                      time_limit=allowed_sa, start_time=start_time)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()

    if tleft() > 60:
        alphas = [200.0, 500.0, 1500.0, 4000.0]
        h_ref, sc_ref = slsqp_refine(best_h, n_points, target_init, alphas,
                                     400, 1e-14,
                                     time_limit=max(10.0, tleft() - 40), start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    n_candidates = []
    for nf in [0.75, 0.9, 1.1, 1.3, 1.5]:
        n_try = int(n_points * nf)
        if n_try % 2 == 0 and n_try != n_points and n_try >= 200:

    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:

            break

            sc_candidate = min(sc_cont, compute_c5(h_bin_try))
        else:
            candidate = h_bin_try

        allowed_sa = min(60.0, tleft() * 0.3)
        if allowed_sa > 5:
            max_it = int(min(60000, allowed_sa * 2000))
            h_sa, c5_sa = informed_sa(project_binary(candidate, target_try), target_try, max_it,
                                      0.012, 0.00008, rng,
                                      time_limit=allowed_sa, start_time=start_time)

    if final_score > best_score + 1e-14:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n_final
