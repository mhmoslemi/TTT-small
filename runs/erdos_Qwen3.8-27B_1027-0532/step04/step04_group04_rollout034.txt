    """Project v onto {x : lb <= x <= ub, sum(x) = target_sum}."""
    n = len(v)
    # Bisection on the multiplier
    lo = lb.min() - target_sum / n
    hi = ub.max() - target_sum / n
    
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, lb, ub)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(v - (lo + hi) / 2, lb, ub)
    # Final adjustment
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # Distribute among free variables
        free = (x > lb + 1e-10) & (x < ub - 1e-10)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, lb, ub)
    return x

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return float(np.sum(h)) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Vectorized best-improvement LS from binary initial
    h_bin = project_binary(h_init, target)
    allowed1 = min(120, max(20, tleft() * 0.12))
    h_ls, c5_ls = vectorized_best_ls(h_bin, max_rounds=10000, rng=rng,
                                      time_limit=allowed1, start_time=start_time,
                                      top_k=30, max_cand=50)
    if c5_ls < best_score:
        best_score = c5_ls
        best_h = h_ls.copy()

    # Phase 2: SA from LS result
    if tleft() > 100:
        allowed2 = min(200, max(30, tleft() * 0.2))
        max_it2 = int(min(300000, allowed2 * 1500))
        h_sa, c5_sa = sa_binary(h_ls, target, max_it2, 0.012, 0.00003, rng,
                                 time_limit=allowed2, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        # Follow-up LS on SA result
        if tleft() > 80:
            h_ls2, c5_ls2 = vectorized_best_ls(h_sa, max_rounds=5000, rng=rng,
                                                time_limit=min(60, tleft() * 0.08),
                                                start_time=start_time, top_k=40, max_cand=50)
            if c5_ls2 < best_score:
                best_score = c5_ls2
                best_h = h_ls2.copy()

    # Phase 3: Iterated Local Search with varied perturbation sizes
    for ils_iter in range(15):
        if tleft() < 60:
            break
        n_pert = max(3, int(3 + ils_iter * 1.5))
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        allowed3 = min(40, max(5, tleft() * 0.06))
        h_ils, c5_ils = vectorized_best_ls(h_pert, max_rounds=8000, rng=rng,
                                            time_limit=allowed3, start_time=start_time,
                                            top_k=35, max_cand=50)
        if c5_ils < best_score:
            best_score = c5_ils
            best_h = h_ils.copy()

        if tleft() < 50:
            break
        # Quick SA polish on ILS result
        allowed3b = min(25, max(5, tleft() * 0.04))
        max_it3 = int(min(30000, allowed3b * 1500))
        h_sa3, c5_sa3 = sa_binary(h_ils, target, max_it3, 0.006, 0.00005, rng,
                                   time_limit=allowed3b, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 4: Diverse starting points
    if tleft() > 120:
        diverse_starts = []
        tm = build_thue_morse(n)
        diverse_starts.append(('tm', tm))
        gm = build_golden(n, rng)
        diverse_starts.append(('golden', gm))
        # Periodic patterns
        pat7 = build_periodic(n, [1, 1, 0, 1, 0, 0, 0])
        diverse_starts.append(('per7', pat7))
        pat11 = build_periodic(n, [1, 0, 1, 1, 0, 0, 1, 0, 0, 1, 0])
        diverse_starts.append(('per11', pat11))
        # Random
        h_rand = np.zeros(n)
        rand_pos = rng.choice(n, n // 2, replace=False)
        h_rand[rand_pos] = 1.0
        diverse_starts.append(('rand', h_rand))

        for name, h_start in diverse_starts:
            if tleft() < 60:
                break
            h_b = project_binary(h_start, target)
            allowed4 = min(50, max(10, tleft() * 0.08))
            h_d, c5_d = vectorized_best_ls(h_b, max_rounds=6000, rng=rng,
                                            time_limit=allowed4, start_time=start_time,
                                            top_k=30, max_cand=50)
            if c5_d < best_score:
                best_score = c5_d
                best_h = h_d.copy()

    # Phase 5: SLSQP refinement on best solution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 50, 150, 500, 1500, 5000, 10000]
        allowed5 = min(250, max(30, tleft() * 0.35))
        h_ref, sc_ref = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-14,
                                      time_limit=allowed5, start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 6: Final SA + LS + SLSQP
    if tleft() > 40:
        h_bin_f = project_binary(best_h, int(len(best_h) / 2))
        allowed6 = min(max(10, tleft() * 0.4), 100)
        max_it6 = int(min(50000, allowed6 * 2000))
        h_sa_f, c5_f = sa_binary(h_bin_f, int(len(best_h) / 2), max_it6,
                                  0.005, 0.00002, rng,
                                  time_limit=allowed6, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final_ls, c5_final_ls = vectorized_best_ls(project_binary(best_h, int(n_use // 2)),
                                                      max_rounds=5000, rng=rng,
                                                      time_limit=min(30, tleft() * 0.5),
                                                      start_time=start_time, top_k=40, max_cand=50)
        if c5_final_ls < best_score:
            best_score = c5_final_ls
            best_h = h_final_ls.copy()

        if tleft() > 15:
            alphas_final = [3000, 8000, 15000]
            h_ref_f, sc_f = slsqp_refine(best_h, len(best_h), float(len(best_h)) / 2.0,
                                          alphas_final, 2000, 1e-15,
                                          time_limit=max(5.0, tleft() - 5), start_time=start_time)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_ref_f.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
