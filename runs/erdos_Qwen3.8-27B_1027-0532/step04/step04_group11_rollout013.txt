```python
def compute_lse_grad_fast(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    ew = np.exp(shifted)
    w_sum = float(ew.sum())
    if w_sum < 1e-300:
        w_sum = 1e-300
    weights = ew / w_sum
    
    offset = n - 1
    # slope[i, m] = (1-h[i+m-offset])*[valid] - h[i-m+offset]*[valid]
    # Using broadcasting:
    i_idx = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    idx1 = i_idx + m_idx - offset  # (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i_idx + offset - m_idx  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # s[i,m] = (1-h[idx1[i,m]])*[valid1] - h[idx2[i,m]]*[valid2]
    s = np.zeros((n, 2*n-1))
    # For valid1: s[i,m] += 1 - h[idx1[i,m]]
    # We need h[idx1] but only where valid1
    # Use np.where or masked assignment
    
    # h as a padded array for safe indexing
    h_pad = np.zeros(2*n)  # extra padding
    h_pad[n:2*n] = h
    # Actually, let's just use np.take with clipping
    idx1_clipped = np.clip(idx1, 0, n-1)
    idx2_clipped = np.clip(idx2, 0, n-1)
    
    s = np.where(valid1, 1.0 - h[idx1_clipped], 0.0) - np.where(valid2, h[idx2_clipped], 0.0)
    
    grad = s @ weights  # (n,) 
    obj = c_max + np.log(w_sum) / alpha
    return grad, obj
```

This is fully vectorized! The matrix s has shape (n, 2n-1). For n=800, that's 800 × 1599 ≈ 1.28M elements. Creating this matrix takes O(n²) time and memory, which is fine.

The matrix-vector product s @ weights is O(n²) as well.

Total: O(n²) per gradient computation, fully vectorized in numpy. For n=800, this should take about 5-10ms. We can do 1000+ iterations in a few seconds.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    # Use periodic extension for smooth interpolation
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(x_new, x_ext, h_ext)
    return h_new

def compute_lse_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    ew = np.exp(shifted)
    w_sum = float(ew.sum())
    if w_sum < 1e-300:
        w_sum = 1e-300
    weights = ew / w_sum
    
    offset = n - 1
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    idx1 = i_idx + m_idx - offset
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i_idx + offset - m_idx
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    
    s = np.where(valid1, 1.0 - h[idx1c], 0.0) - np.where(valid2, h[idx2c], 0.0)
    grad = s.dot(weights)
    obj = c_max + np.log(w_sum) / alpha
    del s
    return grad, obj

def peaking_step(h, target, gamma=2.0):
    h_new = h ** gamma / (h ** gamma + (1.0 - h) ** gamma)
    # Handle h=0 and h=1
    h_new = np.where(h == 0.0, 0.0, h_new)
    h_new = np.where(h == 1.0, 1.0, h_new)
    return project_feasible(h_new, target)

def gradient_descent(h, n, target, alpha_schedule, lr_init, deadline, n_max_iter=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    target_sum = target
    lr = lr_init
    stagnant = 0
    
    it = 0
    while time.time() < deadline:
        if n_max_iter and it >= n_max_iter:
            break
        
        # Determine current alpha
        frac = it / max(n_max_iter, 1) if n_max_iter else 0.0
        # Interpolate through alpha schedule
        n_alpha = len(alpha_schedule)
        if frac < 1.0 and n_alpha > 1:
            pos = frac * (n_alpha - 1)
            ai = int(pos)
            if ai >= n_alpha - 1:
                ai = n_alpha - 2
            frac_a = pos - ai
            alpha = alpha_schedule[ai] * (1 - frac_a) + alpha_schedule[ai+1] * frac_a
        else:
            alpha = alpha_schedule[-1]
        
        grad, obj = compute_lse_grad(h_cur, alpha)
        
        # Gradient step with adaptive lr
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Check every 50 iterations
        if it % 50 == 0:
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                stagnant = 0
            elif sc < cur_score:
                stagnant = 0
            else:
                stagnant += 50
            
            cur_score = sc
            
            # Adapt lr
            if stagnant > 200:
                lr *= 0.5
                stagnant = 0
            
            # Peaking step every 500 iterations
            if it % 500 == 499 and it > 0:
                h_peaked = peaking_step(h_new, target_sum, gamma=1.5)
                sc_p = compute_c5(h_peaked)
                if sc_p < sc:
                    h_new = h_peaked
                    h_cur = h_new
                    cur_score = sc_p
                    if sc_p < best_score:
                        best_score = sc_p
                        best_h = h_new.copy()
                    it += 1
                    continue
        
        h_cur = h_new
        it += 1
    
    return best_h, best_score

    return np.ones(_g_n)

def slsqp_round(h, n, target, alphas, maxiter=2000, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def block_slsqp_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_target, _g_alpha, _g_block_target
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 300, 'ftol': 1e-14})
                ht = h_cur.copy()

                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(x_new, x_ext, h_ext)
    return h_new


def compute_lse_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    ew = np.exp(shifted)
    w_sum = float(ew.sum())
    if w_sum < 1e-300:
        w_sum = 1e-300
    weights = ew / w_sum
    offset = n - 1
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    idx1 = i_idx + m_idx - offset
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i_idx + offset - m_idx
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    s = np.where(valid1, 1.0 - h[idx1c], 0.0) - np.where(valid2, h[idx2c], 0.0)
    grad = s.dot(weights)
    obj = c_max + np.log(w_sum) / alpha
    return grad, obj


def peaking_step(h, target, gamma):
    eps = 1e-10
    h_safe = np.clip(h, eps, 1.0 - eps)
    h_new = h_safe ** gamma / (h_safe ** gamma + (1.0 - h_safe) ** gamma)
    h_new = np.where(h <= 0.0, 0.0, h_new)
    h_new = np.where(h >= 1.0, 1.0, h_new)
    return project_feasible(h_new, target)

def slsqp_round(h, n, target, alphas, maxiter=2000, ftol=1e-15, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur
