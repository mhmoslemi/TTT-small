def project_binary(h, target_sum):

def delta_c_swap(h, p, q, n):
    """O(n) vectorized delta computation for swap p:1->0, q:0->1"""
    ks = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    pk = p + ks
    m = (pk >= 0) & (pk < n)
    delta[m] += h[pk[m]]
    pmk = p - ks
    m = (pmk >= 0) & (pmk < n)
    delta[m] += h[pmk[m]]
    qmk = q - ks
    m = (qmk >= 0) & (qmk < n)
    delta[m] -= h[qmk[m]]
    qk = q + ks
    m = (qk >= 0) & (qk < n)
    delta[m] -= h[qk[m]]

def delta_c_swap(h, p, q, n):
    ls = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # p-ℓ in range: contribution h[p-ℓ] - 1
    idx = p - ls
    m = (idx >= 0) & (idx < n)
    delta[m] += h[idx[m]] - 1.0
    
    # p+ℓ in range: contribution h[p+ℓ]
    idx = p + ls
    m = (idx >= 0) & (idx < n)
    delta[m] += h[idx[m]]
    
    # q-ℓ in range: contribution 1 - h[q-ℓ]
    idx = q - ls
    m = (idx >= 0) & (idx < n)
    delta[m] += 1.0 - h[idx[m]]
    
    # q+ℓ in range: contribution -h[q+ℓ]
    idx = q + ls
    m = (idx >= 0) & (idx < n)
    delta[m] -= h[idx[m]]
    
    return delta
