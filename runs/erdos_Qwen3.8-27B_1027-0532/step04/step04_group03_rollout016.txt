def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]

        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if m >= 0:
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
        else:
            mm = -m
            grad[mm:] += w * (1.0 - h[:n - mm])
            grad[:n - mm] -= w * h[mm:]
    return grad

def lsbfgs_optimize(h_start, target, betas, max_iter_per_beta, time_limit, start_time):
    """L-BFGS-B optimization with log-sum-exp smoothing"""
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in betas:
        if time.time() - start_time > time_limit:
            break
        
        def obj(x):
            h_p = np.clip(x, 0.0, 1.0)
            diff = target - np.sum(h_p)
            penalty = 100.0 * diff * diff
            corr = np.correlate(h_p, 1.0 - h_p, mode='full') * (2.0 / n)
            c_max = np.max(corr)
            shifted = np.clip(beta * (corr - c_max), -500, 0)
            return c_max + np.log(np.sum(np.exp(shifted))) / beta + penalty
        
        def grad(x):
            h_p = np.clip(x, 0.0, 1.0)
            diff = target - np.sum(h_p)
            penalty_grad = 200.0 * diff * np.ones(n)
            corr = np.correlate(h_p, 1.0 - h_p, mode='full') * (2.0 / n)
            c_max = np.max(corr)
            shifted = np.clip(beta * (corr - c_max), -500, 0)
            weights = np.exp(shifted) / np.sum(np.exp(shifted))
            g = corr_gradient(h_p, weights) * (2.0 / n)
            # Only compute grad where 0 < h < 1
            mask = (h_p > 0.0) & (h_p < 1.0)
            g = g * mask
            return g + penalty_grad * mask
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iter_per_beta, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = best_h.copy()
        except Exception:
            pass
    
    return best_h, best_c5
```

Here's the idea:

- Move: pick a random position i, change h[i] by a small random amount δ, then project to feasible set
- Or: pick two positions i, j, transfer a small amount from one to the other (preserving the sum)
- Accept/reject based on SA criterion

- Then do continuous SA starting from that binary solution (initialized with small random perturbations)
- The continuous SA should be able to fine-tune the values

For the continuous SA, the key is the move size. If it's too small, we'll be slow to explore. If it's too large, we'll rarely accept moves.

            break
        if no_improve > 50000:
            break

    return best_h, best_c5

But copying h each iteration is expensive (O(n) per iteration). For n=600 and 500K iterations, that's 300M bytes copied, which is fine.

Actually, I realize the continuous SA might not be that effective because the landscape is very flat in most directions (changing a single h[i] by a tiny amount barely changes C₅). The effective "bumps" in the landscape are at the boundaries (h[i] = 0 or h[i] = 1).

A better approach for continuous optimization might be:
1. Use the log-sum-exp formulation with L-BFGS-B (which handles the flat landscape well via the Hessian approximation)
2. Use a "threshold" approach: try setting individual h[i] to 0 or 1 and see if it improves C₅

Let me combine these ideas. My final algorithm:

1. **Continuous L-BFGS-B** (200s): Optimize the log-sum-exp with increasing beta
2. **Binary SA** (300s): Round to binary, then SA with good moves
3. **Continuous L-BFGS-B again** (100s): Polish the binary solution
4. **Individual flip search** (100s): Try setting each h[i] to 0 or 1 (one at a time)
5. **Multi-n exploration** (200s): Try different resolutions
6. **Final SA + polish** (200s): Last round of improvements

For step 4, the "individual flip search" is:
- For each position i:
  - Try h[i] = 0 (if currently > 0), project, compute C₅
  - Try h[i] = 1 (if currently < 1), project, compute C₅
  - Keep the best
- Repeat a few times

This is O(n²) per pass (n positions × O(n) for correlation), so O(n³) per pass. For n=600, that's 216M per pass. Too slow for many passes.

But we can make it faster: only try positions where h[i] is "close to" 0 or 1 (e.g., 0 < h[i] < 0.3 or 0.7 < h[i] < 1). This reduces the number of positions to try.

Actually, for a near-binary solution, most positions are already at 0 or 1. The fractional positions are the ones worth trying. So the individual flip search is most useful right after the L-BFGS-B step, when there are a few fractional positions.

OK, I think I have a good plan. Let me write the final code now.

Let me also add one more idea: **cooperative moves in SA**. Instead of single swaps, try moving a block:
- Pick a contiguous block of k consecutive 1s
- Pick a contiguous block of k consecutive 0s
- Swap them (shift the 1s to where the 0s were)

This is a more powerful move that can make larger structural changes.

For n=600, a block of size 3-5 would be reasonable. The number of possible block positions is about n, and the number of target positions is about n, so there are O(n²) possible block moves. We can sample from this set.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fast(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if m >= 0:
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:])
            if m > 0 and n - m > 0:
                grad[m:] -= w * h[:n - m]
        else:
            mm = -m
            if n - mm > 0:
                grad[mm:] += w * (1.0 - h[:n - mm])
            if n - mm > 0:
                grad[:n - mm] -= w * h[mm:]
    return grad

            mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
            g *= mask
            return obj, g

        try:

                best_h = h_new.copy()
            h = best_h.copy()

def to_binary(h, target):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == int(target):
        return h_r.astype(float)
    if current_sum > target:
        # Need to flip (current_sum - target) ones to zeros
        excess = current_sum - int(target)
        ones = np.where(h_r == 1)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[ones])[:excess]
        h_r[ones[order]] = 0
    else:
        # Need to flip (target - current_sum) zeros to ones
        deficit = int(target) - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:deficit]
        h_r[zeros[order]] = 1
    return h_r.astype(float)
