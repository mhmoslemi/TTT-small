def nesterov_refine(h_start, target, time_limit, start_time, alpha=200):

    v = np.zeros(n)
    best_h = h.copy()

    lr = 0.02
    gamma = 0.9
    no_improve = 0
    max_no_improve = 500
    mu = 1.0 / (1.0 + np.sqrt(10.0))

    for it in range(50000):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            break

        y = h + mu * v
        corr = compute_corr_scaled(y)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)

        active_indices = np.where(weights > 0.01 * w_sum / len(weights))[0]
        active_weights = weights[active_indices] / w_sum

        grad = compute_correlation_gradient(y, active_indices, active_weights, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break

        lr_adapt = lr / (1.0 + 0.1 * it / max_no_improve)
        h_try = project_feasible(h - lr_adapt * grad / grad_norm, target)
        new_mu = (1.0 + mu) / (1.0 + np.sqrt(1.0 + 10.0 * lr_adapt / grad_norm))

            best_h = h.copy()

            lr *= 0.97
            if lr < 1e-7:
                break
            h = project_feasible(best_h + 0.1 * v, target)

    no_improve = 0
    max_no_improve = 3000

    for i in range(n_iter):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            break

        n_perturb = rng.randint(1, 4)
        positions = rng.choice(n, n_perturb, replace=False)
        h_save = h[positions].copy()
        h[positions] += rng.randn(n_perturb) * lr
        h_proj = project_feasible(h, target)
        c5_try = compute_c5(h_proj)

        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_proj.copy()
            h = h_proj
            no_improve = 0
            lr = min(lr * 1.01, 0.3)
        else:
            h[positions] = h_save
            no_improve += 1
            if no_improve % 500 == 0:
                lr = max(lr * 0.9, 0.005)

    return best_h, best_c5


def topk_equalization(h_start, target, time_limit, start_time, K=10):
    n = len(h_start)
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    for outer in range(20):
        if (time.time() - start_time) > time_limit - 5:
            break
        corr = compute_corr_scaled(h)
        topk = np.argsort(corr)[::-1][:K]
        top_max = float(np.max(corr[topk]))
        top_mean = float(np.mean(corr[topk]))
        spread = top_max - top_mean
        if spread < 1e-10:
            break

        alpha = 500
        try:
            res = minimize(
                smoothmax_obj, h,
                args=(alpha, n, target, topk),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            elif c5_new < best_c5 + 1e-6:
                h = h_new
        except Exception:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def periodic_construction(n, period, rng):
    h = np.zeros(n)
    for p in range(period):
        frac = rng.rand()
        if frac < 0.4:
            pos = (p + 1) // 2
            if pos < period:
                h[p * (n // period) + min(pos, n // period - 1)] = 1.0
        elif frac < 0.7:
            for j in range(n // period):
                if (j + p) % 2 == 0:
                    h[p * (n // period) + j] = 1.0
        else:
            for j in range(n // period):
                if (j % 3 != p % 3):
                    h[p * (n // period) + j] = 1.0
    return project_binary(h, n // 2)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Binary SA from initial
    h_bin_start = project_binary(h_init, target)
    time_p1 = min(50.0, tleft() * 0.1)
    h_sa, c5_sa = sa_optimize(h_bin_start, target, 80000, 0.012, 0.00003,
                              rng, time_limit=time_p1, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Hill climb
    if tleft() > 30:
        h_hc, c5_hc = hill_climb_binary(best_h, target, 6000, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 2: Second SA with lower temperature
    if tleft() > 60:
        time_p2 = min(40.0, tleft() * 0.12)
        h_sa2, c5_sa2 = sa_optimize(best_h, target, 60000, 0.006, 0.00002,
                                    rng, time_limit=time_p2, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        h_hc2, c5_hc2 = hill_climb_binary(best_h, target, 5000, rng)
        if c5_hc2 < best_score:
            best_score = c5_hc2
            best_h = h_hc2.copy()

    # Phase 3: SLSQP with smooth max - primary optimization
    if tleft() > 100:
        time_p3 = min(350.0, tleft() * 0.3)
        alpha_sched = [20, 40, 80, 160, 320, 640, 1280, 2560, 5120]
        h_slq, c5_slq = slsqp_refine(best_h, target, time_p3, start_time,
                                     topk=None, alpha_schedule=alpha_sched, maxiter_per_alpha=500)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    # Phase 4: Top-K equalization
    if tleft() > 80:
        time_p4 = min(200.0, tleft() * 0.2)
        for K in [8, 15, 30]:
            if tleft() < 30:
                break
            h_eq, c5_eq = topk_equalization(best_h, target, min(60.0, tleft() * 0.3), start_time, K=K)
            if c5_eq < best_score:
                best_score = c5_eq
                best_h = h_eq.copy()

    # Phase 5: Nesterov accelerated gradient
    if tleft() > 60:
        time_p5 = min(150.0, tleft() * 0.2)
        for alpha_n in [100, 300, 800]:
            if tleft() < 20:
                break
            h_ne, c5_ne = nesterov_refine(best_h, target, min(50.0, tleft() * 0.4), start_time, alpha=alpha_n)
            if c5_ne < best_score:
                best_score = c5_ne
                best_h = h_ne.copy()

    # Phase 6: Stochastic continuous perturbation
    if tleft() > 50:
        time_p6 = min(100.0, tleft() * 0.2)
        h_sp, c5_sp = stochastic_continuous_perturb(best_h, target, time_p6, start_time, rng)
        if c5_sp < best_score:
            best_score = c5_sp
            best_h = h_sp.copy()

    # Phase 7: Multi-resolution
    n_candidates = []
    for factor in [0.9, 1.1, 1.25, 1.5, 0.75]:
        nc = int(n * factor)
        if nc >= 100 and nc != n and nc % 2 == 0:
            n_candidates.append(nc)

    for n_try in n_candidates:
        if tleft() < 60:
            break
        target_try = int(n_try) // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)

        time_sa = min(25.0, tleft() * 0.15)
        h_sa_r, c5_sa_r = sa_optimize(h_bin_try, target_try, 40000, 0.008, 0.0001,
                                      rng, time_limit=time_sa, start_time=start_time)
        if tleft() > 40:
            time_slq = min(80.0, tleft() * 0.25)
            alpha_sched_r = [50, 200, 800, 3000]
            h_slq_r, c5_slq_r = slsqp_refine(h_sa_r, float(target_try), time_slq, start_time,
                                             topk=None, alpha_schedule=alpha_sched_r, maxiter_per_alpha=300)
            if c5_slq_r < c5_sa_r:
                c5_sa_r = c5_slq_r
                h_sa_r = h_slq_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 8: Final intensive SLSQP with high alpha and top-K
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        _g_n = n_use
        _g_target = target_use
        corr = compute_corr_scaled(best_h)
        K_final = 12
        topk_final = np.argsort(corr)[::-1][:K_final]

        time_p8 = min(120.0, tleft() * 0.4)
        h_final_slq, c5_final = slsqp_refine(best_h, target_use, time_p8, start_time,
                                             topk=topk_final,
                                             alpha_schedule=[500, 1000, 2000, 5000, 10000],
                                             maxiter_per_alpha=600)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_slq.copy()

    # Phase 9: Final Nesterov polish
    if tleft() > 25:
        h_pol, c5_pol = nesterov_refine(best_h, len(best_h) / 2.0,
                                        min(40.0, tleft() - 5), start_time, alpha=1500)
        if c5_pol < best_score:
            best_score = c5_pol
            best_h = h_pol.copy()

    # Phase 10: Final stochastic polish
    if tleft() > 15:
        h_st, c5_st = stochastic_continuous_perturb(best_h, len(best_h) / 2.0,
                                                    min(20.0, tleft() - 3), start_time, rng, n_iter=20000)
        if c5_st < best_score:
            best_score = c5_st
            best_h = h_st.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
