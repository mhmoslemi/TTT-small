        s_star = max_idx - (n - 1)  # the shift
        max_val = corr[max_idx]
        
        ones = np.where(h == 1)[0]

        # A_i = h[i+s*] - (1-h[i-s*]) for valid i (h[i]=1)
        # B_j = (1-h[j-s*]) - h[j+s*] for valid j (h[j]=0)

                    best_h = h.copy()
            else:

            stall += 1

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)

    zeros = np.where(h_r == 0)[0]

def flip_delta(h, i, new_val):
    """O(n) computation of change in correlation when h[i] is set to new_val."""
    n = len(h)
    old_val = 1.0 - new_val
    diff = new_val - old_val
    delta = np.zeros(2 * n - 1)
    
    # h[i] as h[t+s] factor: t = i-s, valid when 0<=i-s<n
    s_lo = max(-(n-1), i - (n-1))
    s_hi = min(n-1, i)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi + 1)
        t = i - s
        idx = s + (n - 1)
        delta[idx] += diff * (1.0 - h[t])

        j = i + s

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)


def flip_delta(h, i, new_val):
    n = len(h)
    old_val = 1.0 - new_val
    diff = new_val - old_val
    delta = np.zeros(2 * n - 1)
    s_lo = max(-(n - 1), i - (n - 1))
    s_hi = min(n - 1, i)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi + 1)
        t = i - s
        idx = s + (n - 1)
        delta[idx] += diff * (1.0 - h[t])
    s_lo2 = max(-(n - 1), -i)
    s_hi2 = min(n - 1, n - 1 - i)
    if s_lo2 <= s_hi2:
        s = np.arange(s_lo2, s_hi2 + 1)
        j = i + s
        idx = s + (n - 1)
        delta[idx] += -diff * h[j]
    return delta


def delta_pair(h, i, j):
    return flip_delta(h, i, 0.0) + flip_delta(h, j, 1.0)

            j_pool = zeros[j_mask]

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
