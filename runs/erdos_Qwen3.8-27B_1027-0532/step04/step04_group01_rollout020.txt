def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def update_corr_single_flip(corr, h, pos, n):
    """Update correlation after flipping h[pos]. Returns new corr."""
    # h[pos] changes from val to 1-val
    val = h[pos]
    # corr[k] = sum_j h[j] * (1-h)[j + k - (n-1)]
    # When h[pos] flips:
    # - As first element (j = pos): remove val * (1-h)[pos + k - (n-1)]
    #   i.e., for k = 0..2n-2, corr[k] -= val * (1-h)[pos + k - (n-1)] if valid
    # - As second element (j + k - (n-1) = pos, so k = pos - j + (n-1)):
    #   corr[k] += (1-val) * h[j] where j = pos - k + (n-1)
    #   i.e., for k = 0..2n-2, corr[k] += (1-val) * h[pos - k + (n-1)] if valid
    
    new_corr = corr.copy()
    # First element contribution: h[pos] * (1-h)[pos + k - (n-1)]
    # For k from 0 to 2n-2: index in (1-h) is pos + k - (n-1)
    # Valid when 0 <= pos + k - (n-1) < n
    # i.e., (n-1) - pos <= k <= n - 1 + pos
    k_min = max(0, (n - 1) - pos)
    k_max = min(2 * n - 2, n - 1 + pos)
    indices = np.arange(k_min, k_max + 1)
    h_indices = pos + indices - (n - 1)  # indices into (1-h)
    new_corr[indices] -= val * (1.0 - h[h_indices])
    
    # Second element contribution: h[j] * (1-h)[pos] where j = pos - k + (n-1)
    # For k from 0 to 2n-2: index in h is pos - k + (n-1)
    # Valid when 0 <= pos - k + (n-1) < n
    # i.e., pos - (n-1) <= k <= pos + (n-1)
    k_min2 = max(0, pos - (n - 1))
    k_max2 = min(2 * n - 2, pos + (n - 1))
    indices2 = np.arange(k_min2, k_max2 + 1)
    h_indices2 = pos - indices2 + (n - 1)  # indices into h
    new_corr[indices2] += (1.0 - val) * h[h_indices2]
    
    return new_corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def is_prime_check(x):
    if x < 2:

    if x < 4:

    return True

        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

                tau_fj = ks_fj - (n - 1)
                fwd_j = jj + tau_fj
                valid_fj = (fwd_j >= 0) & (fwd_j < n)
                cfj = np.zeros(len(ks_fj))
                cfj[valid_fj] = h[fwd_j[valid_fj]] * delta_vj

                corr_trial = corr.copy()
                corr_trial[ks_i] += ci
                corr_trial[ks_j] += cj
                corr_trial[ks_fi] += cfi
                corr_trial[ks_fj] += cfj

                nm = float(np.max(corr_trial))
                if nm < best_new_max - 1e-12:
                    best_new_max = nm
                    best_jj = jj

            if best_jj >= 0 and best_new_max < cur_max - 1e-12:
                # Apply the move
                jj = best_jj
                val_i = 1.0
                val_j = 0.0
                delta_vi = -1.0
                delta_vj = 1.0

                k_lo_i = max(0, ii)
                k_hi_i = min(2 * n - 2, ii + n - 1)
                ks_i = np.arange(k_lo_i, k_hi_i + 1)
                tau_i = ks_i - (n - 1)
                comp_i = ii - tau_i
                valid_ci = (comp_i >= 0) & (comp_i < n)
                ci = np.zeros(len(ks_i))
                ci[valid_ci] = delta_vi * (1.0 - h[comp_i[valid_ci]])

                k_lo_j = max(0, jj)
                k_hi_j = min(2 * n - 2, jj + n - 1)
                ks_j = np.arange(k_lo_j, k_hi_j + 1)
                tau_j = ks_j - (n - 1)
                comp_j = jj - tau_j
                valid_cj = (comp_j >= 0) & (comp_j < n)
                cj = np.zeros(len(ks_j))
                cj[valid_cj] = delta_vj * (1.0 - h[comp_j[valid_cj]])

                k_lo_fi = max(0, (n - 1) - ii)
                k_hi_fi = min(2 * n - 2, 2 * (n - 1) - ii)
                ks_fi = np.arange(k_lo_fi, k_hi_fi + 1)
                tau_fi = ks_fi - (n - 1)
                fwd_i = ii + tau_fi
                valid_fi = (fwd_i >= 0) & (fwd_i < n)
                cfi = np.zeros(len(ks_fi))
                cfi[valid_fi] = h[fwd_i[valid_fi]] * delta_vi

                k_lo_fj = max(0, (n - 1) - jj)
                k_hi_fj = min(2 * n - 2, 2 * (n - 1) - jj)
                ks_fj = np.arange(k_lo_fj, k_hi_fj + 1)
                tau_fj = ks_fj - (n - 1)
                fwd_j = jj + tau_fj
                valid_fj = (fwd_j >= 0) & (fwd_j < n)
                cfj = np.zeros(len(ks_fj))
                cfj[valid_fj] = h[fwd_j[valid_fj]] * delta_vj

                corr[ks_i] += ci
                corr[ks_j] += cj
                corr[ks_fi] += cfi
                corr[ks_fj] += cfj
                h[ii] = 0.0
                h[jj] = 1.0
                cur_max = best_new_max
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()

        if not improved:
            break

    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def continuous_refine(h_init, time_limit, t_start):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

    def cons_fun(x, t):
        return np.sum(x) - t

    def cons_jac(x, t):
        return np.ones(len(x))

    alphas = [15, 40, 100, 250, 500, 1000]
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return smooth_obj(x, alpha)

            def cfun(x):
                return np.sum(x) - target

            def cjac(x):
                return np.ones(n)

            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 300, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def gradient_descent_continuous(h_init, time_limit, t_start, lr_init, n_steps):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target).astype(np.float64)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        c_max = np.max(corr)

        alpha = 100.0
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            break
        p = p / p_sum

        # Compute gradient: for each position j, grad[j] = sum_k p[k] * d(corr[k])/d(h[j])
        # d(corr[k])/d(h[j]) = (1-h)[j - tau_k] - h[j + tau_k] (with validity)
        # where tau_k = k - (n-1)
        tau = np.arange(2 * n - 1) - (n - 1)

        # Use FFT for speed
        # grad_part1[j] = sum_k p[k] * (1-h)[j - tau[k]] (when valid)
        # This is a cross-correlation of p and (1-h)
        # grad_part2[j] = sum_k p[k] * h[j + tau[k]] (when valid)
        # This is a cross-correlation of p and h (with sign flip)

        # corr_p_1h = correlate(p, 1-h) gives sum_k p[k] * (1-h)[k - tau]... 
        # Actually let's just do it directly for n=600 (it's O(n^2) = 360000, fast enough for a few steps)

        grad = np.zeros(n)
        one_minus_h = 1.0 - h
        for j in range(n):
            comp_idx = j - tau
            valid1 = (comp_idx >= 0) & (comp_idx < n)
            grad_val = np.sum(p[valid1] * one_minus_h[comp_idx[valid1]])
            fwd_idx = j + tau
            valid2 = (fwd_idx >= 0) & (fwd_idx < n)
            grad_val -= np.sum(p[valid2] * h[fwd_idx[valid2]])
            grad[j] = grad_val

        lr = lr_init * (0.95 ** step)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    safe_limit = budget_s - 20.0

    h_init_arr = np.array(initial_h_values, dtype=float)
    n_init = len(h_init_arr)
    target_init = n_init / 2.0

    best_h = h_init_arr.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast local search from initial (most important)
    remaining = safe_limit - (time.time() - start)
    t_allocate = min(350.0, remaining * 0.35)
    h_bin = project_binary(h_init_arr, target_init)
    h_ls, c5_ls = fast_local_search_v2(h_bin, target_init, t_allocate, start, rng, 0.015, 0.00003, 10000000)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    # Phase 2: Greedy sweep from best
    remaining = safe_limit - (time.time() - start)
    if remaining > 60:
        t_allocate = min(150.0, remaining * 0.2)
        h_gs, c5_gs = greedy_sweep(best_h, len(best_h) / 2.0, t_allocate, start, rng)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()

    # Phase 3: Fast local search from QR sequence
    remaining = safe_limit - (time.time() - start)
    if remaining > 80:
        h_qr = make_qr_sequence(n_init, rng)
        t_allocate = min(120.0, remaining * 0.2)
        h_ls, c5_ls = fast_local_search_v2(h_qr, target_init, t_allocate, start, rng, 0.012, 0.00002, 10000000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: Try different n values
    remaining = safe_limit - (time.time() - start)
    n_trials = [n_init + 2, n_init + 4, n_init + 6, n_init + 8]
    for nt in n_trials:
        if remaining < 70:
            break
        h_qr = make_qr_sequence(nt, rng)
        t_allocate = min(80.0, remaining * 0.15)
        h_ls, c5_ls = fast_local_search_v2(h_qr, nt / 2.0, t_allocate, start, rng, 0.012, 0.00002, 10000000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = safe_limit - (time.time() - start)

    # Phase 5: Local search from random initialization
    remaining = safe_limit - (time.time() - start)
    if remaining > 80:
        h_rand = make_random_binary(n_init, rng)
        t_allocate = min(100.0, remaining * 0.15)
        h_ls, c5_ls = fast_local_search_v2(h_rand, target_init, t_allocate, start, rng, 0.02, 0.00003, 10000000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 6: More local search from best (perturbed restarts)
    remaining = safe_limit - (time.time() - start)
    n_use = len(best_h)
    target_use = n_use / 2.0
    for restart in range(4):
        if remaining < 60:
            break
        noise = rng.randn(n_use) * 0.3
        h_pert = project_binary(best_h + noise, target_use)
        t_allocate = min(80.0, remaining * 0.15)
        h_ls, c5_ls = fast_local_search_v2(h_pert, target_use, t_allocate, start, rng, 0.01, 0.00002, 10000000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = safe_limit - (time.time() - start)

    # Phase 7: Higher resolution trial
    remaining = safe_limit - (time.time() - start)
    if remaining > 100:
        n_hi = n_init + 40
        x_old = np.linspace(0.0, 1.0, n_use)
        x_new = np.linspace(0.0, 1.0, n_hi)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_hi = project_binary(h_interp, n_hi / 2.0)
        t_allocate = min(100.0, remaining * 0.25)
        h_ls, c5_ls = fast_local_search_v2(h_bin_hi, n_hi / 2.0, t_allocate, start, rng, 0.012, 0.00003, 10000000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = safe_limit - (time.time() - start)

    # Phase 8: Continuous refinement
    remaining = safe_limit - (time.time() - start)
    if remaining > 100:
        n_use = len(best_h)
        t_allocate = min(120.0, remaining * 0.3)
        h_cont, c5_cont = continuous_refine(best_h, t_allocate, start)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 9: Gradient descent on continuous
    remaining = safe_limit - (time.time() - start)
    if remaining > 60:
        n_use = len(best_h)
        t_allocate = min(60.0, remaining * 0.3)
        h_grad, c5_grad = gradient_descent_continuous(best_h, t_allocate, start, 0.001, 200)
        if c5_grad < best_c5:
            best_c5 = c5_grad
            best_h = h_grad.copy()

    # Phase 10: Final binary local search polish
    remaining = safe_limit - (time.time() - start)
    if remaining > 15:
        n_use = len(best_h)
        h_bin_final = project_binary(best_h, n_use / 2.0)
        t_allocate = max(5.0, remaining - 10.0)
        h_ls, c5_ls = fast_local_search_v2(h_bin_final, n_use / 2.0, t_allocate, start, rng, 0.005, 0.00001, 10000000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final: ensure feasibility and compute exact C5
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5:
        h_final = best_h
        final_c5 = best_c5

    # If final_c5 is still worse than best (due to projection), try binary
    h_bin_check = project_binary(h_final, n_final / 2.0)
    c5_bin = compute_c5(h_bin_check)
    if c5_bin < final_c5:
        h_final = h_bin_check
        final_c5 = c5_bin

    return h_final, final_c5, n_final
