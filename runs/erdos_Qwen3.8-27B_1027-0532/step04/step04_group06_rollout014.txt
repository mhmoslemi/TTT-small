    try:
        res = minimize(obj_pen, h_cur, args=(alpha, dx, target_sum, M),
                       jac=grad_pen, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target_sum)
        if time.time() - start_time > time_limit:
            return h_new
        return h_new
    except Exception:
        return h_cur

def find_worst_lags(corr, n, top_k=3):
    center = n - 1
    c = corr.copy()
    c[center] = 0.0
    threshold = c.max() * 0.85
    candidates = np.where(c >= threshold)[0]
    if len(candidates) == 0:
        candidates = np.array([center + 1])
    order = c[candidates][::-1].argsort()
    selected = candidates[order[:top_k]]
    lags = selected - center
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1])
    return lags

def targeted_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(1500, max_iters // 25)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    cached_corr = np.correlate(h, 1.0 - h, mode='full')
    cached_worst = find_worst_lags(cached_corr, n, top_k=3)
    corr_refresh = 0

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.1:
                no_improve = 0
                for _ in range(rng.randint(3, 7)):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    current_c5 = compute_c5(h)
                cached_corr = np.correlate(h, 1.0 - h, mode='full')
                cached_worst = find_worst_lags(cached_corr, n, top_k=3)
            else:
                break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** (ratio + 1e-8)
        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.55 and len(ones) > 0 and len(zeros) > 0:
            corr_refresh += 1
            if corr_refresh >= 2:
                cached_corr = np.correlate(h, 1.0 - h, mode='full')
                cached_worst = find_worst_lags(cached_corr, n, top_k=3)
                corr_refresh = 0
            k_star = int(cached_worst[rng.randint(len(cached_worst))])
            if k_star == 0:
                k_star = 1
            a_k = ones + k_star
            a_km = ones - k_star
            valid_a = (a_k >= 0) & (a_k < n) & (a_km >= 0) & (a_km < n)
            b_k = zeros + k_star
            b_km = zeros - k_star
            valid_b = (b_k >= 0) & (b_k < n) & (b_km >= 0) & (b_km < n)
            if np.sum(valid_a) > 0 and np.sum(valid_b) > 0:
                a_valid = ones[valid_a]
                ak_clipped = np.clip(a_k, 0, n - 1)[valid_a]
                akm_clipped = np.clip(a_km, 0, n - 1)[valid_a]
                a_score = h[ak_clipped] + h[akm_clipped]
                b_valid = zeros[valid_b]
                bk_clipped = np.clip(b_k, 0, n - 1)[valid_b]
                bkm_clipped = np.clip(b_km, 0, n - 1)[valid_b]
                b_score = h[bk_clipped] + h[bkm_clipped]
                a_score += rng.random(len(a_score)) * 0.15
                b_score += rng.random(len(b_score)) * 0.15
                a_idx = a_valid[np.argmin(a_score)]
                b_idx = b_valid[np.argmin(b_score)]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
            else:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
        elif r < 0.80 and len(ones) > 0 and len(zeros) > 0:
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
            h_new[a_idx] = 0.0
            h_new[b_idx] = 1.0
            moved = True
        elif r < 0.93 and len(ones) >= 2 and len(zeros) >= 2:
            ia = rng.choice(len(ones), 2, replace=False)
            ib = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[ia[0]]] = 0.0
            h_new[ones[ia[1]]] = 0.0
            h_new[zeros[ib[0]]] = 1.0
            h_new[zeros[ib[1]]] = 1.0
            moved = True
        else:
            if len(ones) >= 3 and len(zeros) >= 3:
                ia = rng.choice(len(ones), 3, replace=False)
                ib = rng.choice(len(zeros), 3, replace=False)
                h_new[ones[ia]] = 0.0
                h_new[zeros[ib]] = 1.0
                moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / (t + 1e-15))):
            h = h_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    dx_init = 2.0 / n_init

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    if compute_c5(h_bin) < best_score:
        best_score = compute_c5(h_bin)
        best_h = h_bin.copy()

    sa_configs = [
        (80000, 0.018, 0.00008),
        (100000, 0.01, 0.00004),
        (60000, 0.025, 0.0005),
        (120000, 0.006, 0.00002),
    ]

    h_sa_curr = h_bin.copy()
    for ci, (max_it, t_i, t_f) in enumerate(sa_configs):
        if tleft() < 80:
            break
        allowed = min(150.0, tleft() * 0.25)
        h_sa, c5_sa = targeted_sa(h_sa_curr, target_init, max_it, t_i, t_f,
                                  rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        h_sa_curr = h_sa

    h_cont = best_h.copy()
    h_cont = project_feasible(h_cont, target_init)

    alphas = [15, 40, 100, 250, 600, 1500, 4000, 10000, 25000, 60000]
    maxiters = [3000, 2500, 2500, 2000, 2000, 1500, 1500, 1000, 1000, 800]

    for ai, (alpha, maxit) in enumerate(zip(alphas, maxiters)):
        if tleft() < 40:
            break
        allowed = max(10.0, min(60.0, tleft() * 0.12))
        h_new = run_lbfgsb(h_cont, target_init, n_init, dx_init, alpha, maxit, allowed, start_time)
        h_new = project_feasible(h_new, target_init)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cont = h_new.copy()

        h_bin_check = project_binary(h_cont, target_init)
        sc_bin = compute_c5(h_bin_check)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin_check.copy()

    n_restarts = 4
    for rs in range(n_restarts):
        if tleft() < 60:
            break
        noise = rng.choice([0.02, 0.05, 0.1, 0.15])
        h_pert = best_h + rng.randn(n_init) * noise
        h_pert = project_feasible(h_pert, target_init)
        alphas_r = [100, 500, 2000, 8000, 20000]
        maxiters_r = [1500, 1200, 1000, 800, 600]
        h_r = h_pert.copy()
        for alpha_r, maxit_r in zip(alphas_r, maxiters_r):
            if tleft() < 30:
                break
            allowed = max(5.0, min(30.0, tleft() * 0.15))
            h_r = run_lbfgsb(h_r, target_init, n_init, dx_init, alpha_r, maxit_r, allowed, start_time)
            h_r = project_feasible(h_r, target_init)
            sc_r = compute_c5(h_r)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r.copy()

    if tleft() > 120:
        n_alt = [int(n_init * 1.5), int(n_init * 2)]
        for n_try in n_alt:
            if tleft() < 80:
                break
            target_try = n_try / 2.0
            dx_try = 2.0 / n_try
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin_try = project_binary(h_interp, target_try)

            allowed_sa = max(5.0, min(60.0, tleft() * 0.15))
            h_sa_try, c5_sa_try = targeted_sa(h_bin_try, target_try, 30000, 0.01, 0.0001,
                                              rng, time_limit=allowed_sa, start_time=start_time)
            h_cont_try = h_sa_try.copy()
            sc_try = c5_sa_try
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_sa_try.copy()

            alphas_try = [50, 200, 800, 3000, 10000]
            maxiters_try = [1500, 1200, 1000, 800, 500]
            for alpha_t, maxit_t in zip(alphas_try, maxiters_try):
                if tleft() < 30:
                    break
                allowed_t = max(5.0, min(30.0, tleft() * 0.12))
                h_cont_try = run_lbfgsb(h_cont_try, target_try, n_try, dx_try, alpha_t, maxit_t,
                                        allowed_t, start_time)
                h_cont_try = project_feasible(h_cont_try, target_try)
                sc_t = compute_c5(h_cont_try)
                if sc_t < best_score:
                    best_score = sc_t
                    best_h = h_cont_try.copy()
                h_bin_t = project_binary(h_cont_try, target_try)
                sc_bt = compute_c5(h_bin_t)
                if sc_bt < best_score:
                    best_score = sc_bt
                    best_h = h_bin_t.copy()

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        dx_use = 2.0 / n_use
        alphas_final = [50000, 100000, 200000, 500000]
        maxiters_final = [800, 600, 500, 400]
        h_final_cont = best_h.copy()
        for alpha_f, maxit_f in zip(alphas_final, maxiters_final):
            if tleft() < 15:
                break
            allowed_f = max(3.0, min(15.0, tleft() * 0.3))
            h_final_cont = run_lbfgsb(h_final_cont, target_use, n_use, dx_use, alpha_f, maxit_f,
                                      allowed_f, start_time)
            h_final_cont = project_feasible(h_final_cont, target_use)
            sc_f = compute_c5(h_final_cont)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_final_cont.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
