def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Now need to adjust sum to target_sum
    # Use bisection on a threshold
    def project_with_sum(threshold):
        h_proj = np.clip(h - threshold, 0.0, 1.0)
        return h_proj
    
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(project_with_sum(mid))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = project_with_sum((lo+hi)/2)
    # Fix any residual
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h_proj
            order = np.argsort(-room)
            remaining = diff
            for i in order:
                if remaining <= 1e-12:
                    break
                add = min(room[i], remaining)
                h_proj[i] += add
                remaining -= add
        else:
            order = np.argsort(h_proj)
            remaining = -diff
            for i in order:
                if remaining <= 1e-12:
                    break
                sub = min(h_proj[i], remaining)
                h_proj[i] -= sub
                remaining -= sub
    return np.clip(h_proj, 0.0, 1.0)

The algorithm:

3. Solve the LP with all 2n-1 linearized constraints
4. Update h
5. Recompute C(ℓ)
6. If max C(ℓ) decreased, continue; else, add only the worst constraint and re-solve
7. Repeat until convergence

def linearized_subproblem(h_curr, n, target):
    """Solve the linearized LP subproblem"""
    from scipy.optimize import linprog
    
    corr = compute_corr(h_curr)  # C(ℓ) for all ℓ, shape (2n-1,)
    n_corr = 2 * n - 1
    
    # Compute gradients of C(ℓ) wrt h for all ℓ
    # ∂C(ℓ)/∂h[m] = (1-h[m+lag])*I[0≤m+lag<n] - h[m-lag]*I[0≤m-lag<n]
    # where lag = k - (n-1) for the k-th correlation value
    
    # Build A_ub matrix: (2n-1) x (n+1)
    # Row for lag ℓ: [∂C(ℓ)/∂h[0], ..., ∂C(ℓ)/∂h[n-1], -1]
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for k in range(n_corr):
        lag = k - (n - 1)
        # ∂C(k)/∂h[m] = (1-h[m+lag])*I[0≤m+lag<n] - h[m-lag]*I[0≤m-lag<n]
        grad_lag = np.zeros(n)
        if lag >= 0:
            # m+lag valid: m in [0, n-1-lag]
            grad_lag[:n-lag] += (1 - h_curr[lag:])
            # m-lag valid: m in [lag, n-1]
            grad_lag[lag:] -= h_curr[:n-lag]
        else:
            # m+lag valid: m in [-lag, n-1]
            grad_lag[-lag:] += (1 - h_curr[:n+lag])
            # m-lag valid: m in [0, n-1+lag]
            grad_lag[:n+lag] -= h_curr[-lag:]
        
        A_ub[k, :n] = grad_lag
        A_ub[k, n] = -1
        b_ub[k] = -corr[k]  # constraint: grad_lag · h - t ≤ -C_curr[k]... 
        # Wait, let me redo:
        # C_lin(k) = C_curr[k] + grad_lag · (h - h_curr)
        # We want C_lin(k) ≤ t
        # C_curr[k] + grad_lag · h - grad_lag · h_curr ≤ t
        # grad_lag · h - t ≤ grad_lag · h_curr - C_curr[k]
        
        b_ub[k] = grad_lag @ h_curr - corr[k]
    
    # Objective: minimize t → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 ≤ h[i] ≤ 1, t free
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h_curr

Let me think about the gradient computation more carefully.

For the top-p softmax of C(k):
f(h) = (1/α) * log(Σ_{k∈S} exp(α*C(k))) where S is the set of top-p lags

∂f/∂h[m] = Σ_{k∈S} p_k * ∂C(k)/∂h[m]
where p_k = exp(α*C(k)) / Σ_{j∈S} exp(α*C(j))

∂C(k)/∂h[m] = dx * [(1-h[m+ℓ_k])*I[0≤m+ℓ_k<n] - h[m-ℓ_k]*I[0≤m-ℓ_k<n]]

So:
∂f/∂h[m] = dx * Σ_{k∈S} p_k * [(1-h[m+ℓ_k])*I[valid] - h[m-ℓ_k]*I[valid]]
= dx * [Σ_{k∈S} p_k*I[0≤m+ℓ_k<n] - Σ_{k∈S} p_k*h[m+ℓ_k]*I[0≤m+ℓ_k<n] - Σ_{k∈S} p_k*h[m-ℓ_k]*I[0≤m-ℓ_k<n]]

I can compute this efficiently by iterating over the p lags:
For each k in S:
  ℓ = lag_k
  w = dx * p_k
  For m where m+ℓ is valid: add w*(1-h[m+ℓ]) to grad[m]
  For m where m-ℓ is valid: subtract w*h[m-ℓ] from grad[m]

This is O(n*p) total.

Alternatively, I can use FFT-based cross-correlation:
- Define w[ℓ] = dx * p_{ℓ+(n-1)} (for the p active lags, 0 elsewhere)
- First term: (w * 1)[m] = Σ_ℓ w[ℓ] * I[0≤m+ℓ<n] = moving sum of w
- Second term: (w * h)[m] = Σ_ℓ w[ℓ] * h[m+ℓ] * I[0≤m+ℓ<n] = cross-correlation
- Third term: (w_flipped * h)[m] = Σ_ℓ w[ℓ] * h[m-ℓ] * I[0≤m-ℓ<n]

For p << n, the direct O(n*p) approach is faster than FFT.
For p ≈ n, FFT is faster.

I'll use the direct approach for p ≤ n/10 and FFT for larger p.
