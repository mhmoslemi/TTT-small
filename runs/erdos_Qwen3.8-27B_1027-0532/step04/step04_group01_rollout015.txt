def swap_delta_fast(h, i, j, n):
    """Fast vectorized delta for swap."""
    # Use the fact that delta[k] = h[i+k] - h[j+k] + h[i-k] - h[j-k]
    # where out-of-range = 0
    
    # Create a zero-padded version of h with enough room
    # We need h[x] for x in [-(n-1), 2n-2] (to cover all i+k, i-k, j+k, j-k)
    h_pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    # h_pad[x + (n-1)] = h[x] for 0 <= x < n, else 0
    
    # Now h[i+k] for k in [-(n-1), n-1]:
    # i+k ranges from i-(n-1) to i+n-1
    # In h_pad: h_pad[(i+k)+(n-1)] = h_pad[i+k+n-1]
    # So the array of h[i+k] for k=-(n-1),...,n-1 is h_pad[i : i+2n-1]
    # Wait: k=-(n-1) → i+k = i-n+1, h_pad[i-n+1+n-1] = h_pad[i]
    # k=n-1 → i+k = i+n-1, h_pad[i+n-1+n-1] = h_pad[i+2n-2]
    # So it's h_pad[i : i+2n-1], which has length 2n-1. ✓
    
    h_ik = h_pad[i:i+2*n-1]
    h_jk = h_pad[j:j+2*n-1]
    
    # h[i-k] for k in [-(n-1), n-1]:
    # i-k ranges from i+n-1 down to i-n+1
    # In h_pad: h_pad[(i-k)+(n-1)] = h_pad[i-k+n-1]
    # k=-(n-1) → i-k = i+n-1, h_pad[i+n-1+n-1] = h_pad[i+2n-2]
    # k=n-1 → i-k = i-n+1, h_pad[i-n+1+n-1] = h_pad[i]
    # So it's h_pad[i+2n-2 : i-1 : -1]... reversed!
    # = h_pad[i : i+2n-1][::-1]
    
    h_ik_neg = h_pad[i:i+2*n-1][::-1]
    h_jk_neg = h_pad[j:j+2*n-1][::-1]
    
    return h_ik - h_jk + h_ik_neg - h_jk_neg

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = target - current
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def make_h_pad(h, n):
    """Create zero-padded h: length 3n-2, h_pad[x] = h[x-(n-1)] for x in [n-1, 2n-2]."""
    return np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])


def swap_delta_single(h_pad, i, j, n):
    """Delta for single swap: h[i]:1->0, h[j]:0->1."""
    seg_i = h_pad[i:i+2*n-1]
    seg_j = h_pad[j:j+2*n-1]
    seg_i_rev = seg_i[::-1]
    seg_j_rev = seg_j[::-1]
    return seg_i - seg_j + seg_i_rev - seg_j_rev


def swap_delta_double(h_pad, i1, i2, j1, j2, n):
    """Delta for double swap: h[i1],h[i2]:1->0, h[j1],h[j2]:0->1."""
    delta = np.zeros(2*n-1)
    for i, j in [(i1, j1), (i2, j2)]:
        seg_i = h_pad[i:i+2*n-1]
        seg_j = h_pad[j:j+2*n-1]
        delta += seg_i - seg_j + seg_i[::-1] - seg_j[::-1]
    return delta

    h = project_binary(h_init, target_int).astype(np.float64)
    
    h_pad = make_h_pad(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(np.max(corr))
    cur_c5 = cur_max * 2.0 / n
    
    best_h = h.copy()
    best_corr = corr.copy()
    best_c5 = cur_c5
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    max_no_imp = max(3000, len(ones) * max_no_imp_factor)
    no_imp = 0
    it = 0
    
    while True:
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        
        if no_imp > max_no_imp:
            # Try to escape: restart from best with larger perturbation
            h = best_h.copy()
            corr = best_corr.copy()
            cur_max = float(np.max(corr))
            cur_c5 = cur_max * 2.0 / n
            h_pad = make_h_pad(h, n)
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            
            # Perturb
            n_perturb = rng.randint(5, max(6, n // 20))
            for _ in range(n_perturb):
                if ones and zeros:
                    pi = ones[rng.randint(len(ones))]
                    pj = zeros[rng.randint(len(zeros))]
                    h[pi], h[pj] = 0.0, 1.0
                    # Swap in lists
                    ones[ones.index(pi)] = pj  # This is O(n) though...
                    # Let me just rebuild
            h_pad = make_h_pad(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = float(np.max(corr))
            cur_c5 = cur_max * 2.0 / n
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            no_imp = 0
            continue
        
        # Temperature
        # Use a cooling schedule based on elapsed time
        frac = min(1.0, elapsed / time_limit)
        temp = t_init * (t_final / t_init) ** frac
        
        # Choose move
        move_type = rng.random()
        
        if move_type < 0.5 and len(ones) > 0 and len(zeros) > 0:

Let me also think about the continuous refinement step. The previous code uses SLSQP which is slow for large n. A better approach might be:
1. Start from the best binary solution
2. Allow small continuous perturbations (h[i] ∈ [0, 1], not just {0, 1})
3. Use projected gradient descent on the smooth max objective

The smooth max is: f(h) = (1/alpha) * log(sum_k exp(alpha * corr[k] * dx)) which approximates max_k corr[k] * dx when alpha is large.

The gradient of corr[k] with respect to h[i] is: (1-h[i+k]) - h[i-k] (as derived earlier, with boundary handling).

So the gradient of f with respect to h[i] is:
∂f/∂h[i] = (dx/alpha) * sum_k p_k * ((1-h[i+k]) - h[i-k])
where p_k = exp(alpha * corr[k] * dx) / sum_j exp(alpha * corr[j] * dx) (softmax weights).

This can be computed efficiently! For each i, we need to compute a weighted sum of (1-h[i+k]) - h[i-k] over k, with weights p_{i-related}.

Actually, this is a convolution-like operation. Let me think...

∂(corr[k])/∂h[i] = (1-h[i+k]) - h[i-k] (for valid indices)

So ∂f/∂h[i] = (dx/alpha) * sum_k p_k * ((1-h[i+k]) - h[i-k])

The first sum: sum_k p_k * (1-h[i+k]) = sum_k p_k - sum_k p_k * h[i+k]
The second sum: sum_k p_k * h[i-k]

These are convolutions of h with p (and 1-p). I can compute them using FFT!
