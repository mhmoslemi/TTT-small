```python
def project_box_sum(v, target):
    """Project v onto {h: 0 <= h <= 1, sum(h) = target}"""
    n = len(v)
    lo = -v.max() - 1
    hi = -v.min() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_box_sum(v, target):
    """Project v onto {h: 0 <= h <= 1, sum(h) = target} using bisection."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad(h, alpha, top_w=80):
    """Compute LSE objective value and gradient."""
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    val = c_max + log_sum / alpha
    weights = np.exp(shifted)
    w_sum = weights.sum()
    weights /= w_sum
    
    grad = np.zeros(n)
    top_idx = np.argsort(weights)[::-1][:top_w]
    
    for m in top_idx:
        w = weights[m]
        if w < 1e-14:
            break
        k = abs(int(m) - (n - 1))
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            j_pos = np.arange(0, n - k)
            grad[j_pos] += w * (1.0 - h[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= w * h[j_neg - k]
    
    return val, grad

def fista_run(h0, target, alphas, iters_per_alpha, top_w=80, max_time=None, start_time=None):
    """Run FISTA with multiple alpha values (annealing)."""
    n = len(h0)
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        h_cur = h.copy()
        y = h_cur.copy()
        t = 1.0
        step = 1.0
        
        for it in range(iters_per_alpha):
            if max_time is not None and start_time is not None:
                if time.time() - start_time > max_time:
                    break
            
            val, grad = lse_grad(y, alpha, top_w)
            y_new = y - step * grad
            h_new = project_box_sum(y_new, target)
            
            # Backtracking
            val_new, _ = lse_grad(h_new, alpha, top_w)
            if val_new > val + 1e-12:
                step *= 0.5
                continue
            
            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h_new + (t / t_new - 1.0) * (h_new - h_cur)
            y = project_box_sum(y, target)  # Keep y in feasible set
            h_cur = h_new
            t = t_new
            
            score = compute_c5(h_cur)
            if score < best_score:
                best_score = score
                best_h = h_cur.copy()
        
        h = h_cur
    
    return best_h, best_score

def slsqp_round(h, n, target, alphas, maxiter=3000, ftol=1e-16, deadline=None):
    """SLSQP optimization with LSE objective."""
    bounds = [(0.0, 1.0)] * n
    h_cur = project_box_sum(h.copy(), target)
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        def cons_eq(x):
            return np.sum(x) - target
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps=500, n_active=5, lr_init=0.003, lr_decay=0.997, deadline=None):
    """Subgradient method using top active lags."""
    h_cur = project_box_sum(h.copy(), target)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            k = abs(int(m) - (n - 1))
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        h_new = h_cur - lr * grad
        h_new = project_box_sum(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    
    return best_h, best_score

    return project_box_sum(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_n = 0
_g_target = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _slsqp_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _slsqp_con(x):
    return np.sum(x) - _g_target


def _slsqp_jac(x):
    return np.ones(_g_n)


def lse_grad(h, alpha, top_w=80):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    val = c_max + log_sum / alpha
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(weights)[::-1][:top_w]
    for m in top_idx:
        w = weights[m]
        if w < 1e-14:
            break
        k = abs(int(m) - (n - 1))
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            j_pos = np.arange(0, n - k)
            grad[j_pos] += w * (1.0 - h[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= w * h[j_neg - k]
    return val, grad


def fista_run(h0, target, alphas, iters_per_alpha, top_w=80, max_time=100.0, start_time=None):
    n = len(h0)
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    if start_time is None:
        start_time = time.time()
    for alpha in alphas:
        if time.time() - start_time > max_time:
            break
        h_cur = h.copy()
        y = h_cur.copy()
        t = 1.0
        step = 2.0 / alpha * 10.0
        step = min(step, 5.0)
        for it in range(iters_per_alpha):
            if time.time() - start_time > max_time:
                break
            val, grad = lse_grad(y, alpha, top_w)
            y_new = y - step * grad
            h_new = project_box_sum(y_new, target)
            val_new, _ = lse_grad(h_new, alpha, top_w)
            if val_new > val + 1e-10:
                step *= 0.5
                if step < 1e-8:
                    step = 1e-8
                continue
            if val_new < val - step * 1e-3:
                step = min(step * 1.1, 10.0)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h_new + (t / t_new - 1.0) * (h_new - h_cur)
            y = project_box_sum(y, target)

            t = t_new

        h = h_cur

def slsqp_round(h, n, target, alphas, maxiter=3000, ftol=1e-16, deadline=None):
    global _g_alpha, _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
    h_cur = project_box_sum(h.copy(), target)
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:

    return h_cur, cur_score


def subgradient_polish(h, n, target, n_steps=800, n_active=5, lr_init=0.004, lr_decay=0.998, deadline=None):
    h_cur = project_box_sum(h.copy(), target)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.004)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            k = abs(int(m) - (n - 1))
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)

                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]

        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)
```
