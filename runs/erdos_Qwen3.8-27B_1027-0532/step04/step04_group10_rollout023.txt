    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    # Distribute the difference more carefully
    if diff > 0:
        # Add to elements with most room
        room = 1.0 - h
        total_room = room.sum()
        if total_room < diff:
            h = np.clip(h + diff / len(h), 0, 1)
        else:
            h += diff * room / total_room
    else:
        # Remove from elements with most value
        val = h
        total_val = val.sum()
        if total_val < -diff:
            h = np.clip(h + diff / len(h), 0, 1)
        else:
            h += diff * val / total_val

    n = len(h)

    s = np.sum(h) - target
    obj += penalty * s * s
    return obj

                j_range = np.arange(lag, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room > 1e-10:
            h += diff * room / total_room
        else:
            h += diff / len(h)
    else:
        total_val = h.sum()
        if total_val > 1e-10:
            h += diff * h / total_val
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)


def lse_obj_and_grad(h, alpha, topk, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(nc)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total_exp = np.sum(exp_vals)
    obj = c_max + np.log(total_exp) / alpha
    weights = exp_vals / total_exp
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * omh[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * omh[m:]
                grad[m:] -= w * h[:n - m]
    s = np.sum(h) - target
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad

        return f

        _, g = lse_obj_and_grad(x, alpha, topk, target, penalty)
        return g
    try:

                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 8000})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

            orig = h[i]
