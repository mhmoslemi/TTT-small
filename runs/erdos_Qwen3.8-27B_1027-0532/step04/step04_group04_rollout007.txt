```python
    # Total change to c[k]: -(1-h[p+k-(n-1)]) + h[p+k-(n-1)] = -(1-h[p+k-(n-1)]) + h[p+k-(n-1)]
    # = -1 + 2*h[p+k-(n-1)]
    
    # Similarly for h[q] going from 0 to 1:
    #   Term i=q: was 0, now (1-h[q+k-(n-1)]). Change: +(1-h[q+k-(n-1)])
    #   Terms where i+k-(n-1)=q, i.e., i=q+k-(n-1): was h[i]*(1-0)=h[i], now h[i]*(1-1)=0. Change: -h[q+k-(n-1)]
    # Total change to c[k]: (1-h[q+k-(n-1)]) - h[q+k-(n-1)] = 1 - 2*h[q+k-(n-1)]
    
    # So for the swap (p: 1→0, q: 0→1):
    # Δc[k] = (-1 + 2*h[p+k-(n-1)]) + (1 - 2*h[q+k-(n-1)])
    #       = 2*h[p+k-(n-1)] - 2*h[q+k-(n-1)]
    #       = 2*(h[p+k-(n-1)] - h[q+k-(n-1)])
    
    # Confirmed! Same as before.
    
    # To DECREASE c[k], we want h[p+k-(n-1)] < h[q+k-(n-1)]
    # i.e., h[p+k-(n-1)]=0 and h[q+k-(n-1)]=1
    
    # Score for p (want to minimize): sum_i w_i * h[p+k_i-(n-1)] (lower is better)
    # Score for q (want to minimize): sum_i w_i * (-h[q+k_i-(n-1)]) = -sum_i w_i * h[q+k_i-(n-1)] (lower is better = higher h values)
    
    # So: best p minimizes sum_i w_i * h[p+k_i-(n-1)]
    #     best q maximizes sum_i w_i * h[q+k_i-(n-1)]
    
    # Build the shift offsets
    offsets = top_indices - (n - 1)  # actual lags
    
    # For all p in ones, compute their "cost vector" across top shifts
    # cost_p[i] = h[p + offsets[i]] if valid, else 0
    # score_p = dot(cost_p, weights)
    
    # Vectorized: for each p, the positions to check are p + offsets
    p_positions = ones[:, None] + offsets[None, :]  # shape (n_ones, K)
    valid_p = (p_positions >= 0) & (p_positions < n)
    h_at_p = np.where(valid_p, h[np.clip(p_positions, 0, n-1)], 0)
    score_p = h_at_p @ weights  # lower is better
    
    q_positions = zeros[:, None] + offsets[None, :]
    valid_q = (q_positions >= 0) & (q_positions < n)
    h_at_q = np.where(valid_q, h[np.clip(q_positions, 0, n-1)], 0)
    score_q = h_at_q @ weights  # higher is better
    
    best_p_idx = np.argmin(score_p)
    best_q_idx = np.argmax(score_q)
    
    return ones[best_p_idx], zeros[best_q_idx]
```

```python
def compute_gradient(h, p, alpha, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-10:
            continue
        # d c_k / d h[i] for all i
        m1 = k - (n-1)  # shift for "direct" term
        m2 = -(k - (n-1))  # shift for "indirect" term
        # Direct: (1-h[i+m1]) for valid i
        i_range_direct = np.arange(max(0, -m1), min(n, n-m1))

        # Indirect: -h[i+m2] for valid i
        i_range_indirect = np.arange(max(0, -m2), min(n, n-m2))
        grad[i_range_indirect] += -p[k] * h[i_range_indirect + m2]
    return grad * dx / alpha
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 500

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_vals = corr[top_indices]
        offsets = top_indices - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        p_pos = ones[:, None] + offsets[None, :]
        valid_p = (p_pos >= 0) & (p_pos < n)
        h_at_p = np.where(valid_p, h[np.clip(p_pos, 0, n - 1)], 0.0)
        score_p = h_at_p @ top_vals

        q_pos = zeros[:, None] + offsets[None, :]
        valid_q = (q_pos >= 0) & (q_pos < n)
        h_at_q = np.where(valid_q, h[np.clip(q_pos, 0, n - 1)], 0.0)
        score_q = h_at_q @ top_vals

        top10_p = np.argsort(score_p)[:10]
        top10_q = np.argsort(-score_q)[:10]

        best_score = np.inf
        best_pi, best_qi = None, None
        for pi in top10_p:
            for qi in top10_q:
                p = ones[pi]
                q = zeros[qi]
                delta = 2.0 * (h_at_p[pi] - h_at_q[qi])
                new_top = top_vals + delta
                ms = float(new_top.max())
                if ms < best_score:
                    best_score = ms
                    best_pi, best_qi = pi, qi

        if best_pi is None:
            stall += 1
            continue

        if best_score >= max_val:
            stall += 1
            if stall > 100:
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p] = 0
                h[q] = 1
                corr = compute_corr(h)
                if corr.max() < max_val:
                    stall = 0
                    c5_new = float(corr.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                else:
                    h[p] = 1
                    h[q] = 0
            continue

        p = ones[best_pi]
        q = zeros[best_qi]
        h[p] = 0
        h[q] = 1
        corr = compute_corr(h)
        new_max = float(corr.max())
        new_c5 = new_max * dx
        stall = 0
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return best_h, best_c5


def sa_optimize(h, target, max_iters, t_init, t_final, rng, time_limit=None, start_time=None, focus_k=20):

    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 10000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        ratio = (it + 1) / max(max_iters, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        move_type = rng.randint(8)
        h_new = h.copy()
        moved = False

        if move_type <= 4:

            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type <= 6:
            top_idx = np.argsort(corr)[-focus_k:][::-1]

            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / len(h))


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

            c_max = float(corr.max())
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

                if p[k] < 1e-12:
                    continue
                m = k - (n - 1)
                i_direct = np.arange(max(0, -m), min(n, n - m))
                grad[i_direct] += p[k] * (1.0 - h[i_direct + m])
                i_indirect = np.arange(max(0, m), min(n, n + m))
                grad[i_indirect] -= p[k] * h[i_indirect - m]
            grad *= dx / alpha

            lr = 1.0 / (alpha * dx + 1.0)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fib_init(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = []
    for i in range(ones_count):
        frac = ((i + 1.0) * golden) % 1.0
        pos = int(frac * n)
        pos = pos % n
        positions.append(pos)
    positions = list(set(positions))
    while len(positions) < ones_count:
        candidates = [p for p in range(n) if p not in set(positions)]
        if not candidates:
            break
        positions.append(candidates[rng.randint(len(candidates))])
    positions = positions[:ones_count]
    h[positions] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Greedy from binary initial
    h_bin = project_binary(h_init, target_init)
    allowed = min(120, tleft() * 0.20)
    h_g, c5_g = vec_greedy(h_bin, max_iters=50000, rng=rng,
                           time_limit=allowed, start_time=start_time, top_k=50)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA refinement
    allowed = min(250, tleft() * 0.30)
    max_it = int(min(200000, allowed * 3000))
    h_sa, c5_sa = sa_optimize(h_g, target_init, max_it, 0.012, 0.00002, rng,
                              time_limit=allowed, start_time=start_time, focus_k=20)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Multiple SA restarts with perturbation
    for restart in range(5):
        if tleft() < 60:
            break
        n_pert = max(3, 3 + restart * 2)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        allowed = min(40, tleft() * 0.10)
        h_rg, c5_rg = vec_greedy(h_pert, max_iters=20000, rng=rng,
                                 time_limit=allowed, start_time=start_time, top_k=40)
        if c5_rg < best_score:
            best_score = c5_rg
            best_h = h_rg.copy()

        if tleft() < 40:
            break
        allowed2 = min(30, tleft() * 0.08)
        h_rs, c5_rs = sa_optimize(h_rg, target_init,
                                  int(min(80000, allowed2 * 3000)),
                                  0.006, 0.00001, rng,
                                  time_limit=allowed2, start_time=start_time, focus_k=15)
        if c5_rs < best_score:
            best_score = c5_rs
            best_h = h_rs.copy()

    # Phase 4: Resolution exploration
    res_candidates = [n_init + 2, n_init - 2, n_init + 4, n_init - 4,
                      n_init + 6, n_init + 8, n_init - 6, n_init + 10]
    res_candidates = [r for r in res_candidates if r >= 100 and r % 2 == 0 and r != n_init]
    res_candidates = sorted(set(res_candidates))[:6]

    for n_try in res_candidates:
        if tleft() < 50:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(30, tleft() * 0.08)
        h_g2, c5_g2 = vec_greedy(h_bin_try, max_iters=15000, rng=rng,
                                 time_limit=allowed, start_time=start_time, top_k=40)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 40:
            break
        allowed2 = min(20, tleft() * 0.05)
        h_s2, c5_s2 = sa_optimize(h_g2, target_try,
                                  int(min(40000, allowed2 * 3000)),
                                  0.008, 0.00002, rng,
                                  time_limit=allowed2, start_time=start_time, focus_k=15)
        if c5_s2 < best_score:
            best_score = c5_s2
            best_h = h_s2.copy()

    # Phase 5: Fibonacci initialization + optimization
    if tleft() > 80:
        h_fib = fib_init(n_init, rng)
        h_fib_bin = project_binary(h_fib, target_init)
        allowed = min(50, tleft() * 0.10)
        h_fg, c5_fg = vec_greedy(h_fib_bin, max_iters=20000, rng=rng,
                                 time_limit=allowed, start_time=start_time, top_k=50)
        if c5_fg < best_score:
            best_score = c5_fg
            best_h = h_fg.copy()

        if tleft() > 40:
            allowed2 = min(30, tleft() * 0.06)
            h_fs, c5_fs = sa_optimize(h_fg, target_init,
                                      int(min(50000, allowed2 * 3000)),
                                      0.008, 0.00002, rng,
                                      time_limit=allowed2, start_time=start_time, focus_k=15)
            if c5_fs < best_score:
                best_score = c5_fs
                best_h = h_fs.copy()

    # Phase 6: Projected gradient smooth refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 50, 100, 300]
        allowed = min(tleft() - 20, 100)
        h_pg, sc_pg = projected_gradient(best_h, n_use, target_use, alphas,
                                         50, time_limit=allowed, start_time=start_time)
        if sc_pg < best_score:
            best_score = sc_pg
            best_h = h_pg.copy()

    # Phase 7: SLSQP fine-tune
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [500, 1500, 5000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-14,
                                 time_limit=min(tleft() - 15, 60), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final binary SA
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_final = project_binary(best_h, target_use)
        allowed = max(5.0, tleft() - 10)
        h_sf, c5_f = sa_optimize(h_bin_final, target_use,
                                 int(min(60000, allowed * 3000)),
                                 0.005, 0.00001, rng,
                                 time_limit=allowed, start_time=start_time, focus_k=10)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sf

    # Final projected gradient polish
    if tleft() > 12:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_pg2, sc2 = projected_gradient(best_h, n_use, target_use,
                                        [1000, 3000], 30,
                                        time_limit=max(3.0, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pg2

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
```
