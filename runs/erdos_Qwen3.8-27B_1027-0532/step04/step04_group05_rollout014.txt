def smooth_max(corr, alpha):
    """Smooth approximation of max using log-sum-exp."""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def binary_round(h, target):
    """Round h to binary, then adjust to meet target sum."""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    # Adjust to meet target
    diff = target - np.sum(h_bin)
    if diff > 0:
        # Need more 1s
        sorted_idx = np.argsort(h)[::-1]  # highest h values first
        for i in range(int(np.ceil(diff))):
            if i < len(sorted_idx):
                h_bin[sorted_idx[i]] = 1.0
    elif diff < 0:
        # Need fewer 1s
        sorted_idx = np.argsort(h)[::-1]  # highest h values first
        for i in range(int(np.floor(-diff))):
            if i < len(sorted_idx):
                h_bin[sorted_idx[i]] = 0.0
    return h_bin


def random_binary(n, target, rng):
    """Generate random binary with exact target sum."""
    n_ones = int(target)
    h = np.zeros(n)
    indices = rng.choice(n, size=n_ones, replace=False)
    h[indices] = 1.0
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def chebyshev_expand(h, n_cheb):
    """Expand h in Chebyshev polynomials on [0, 2]."""
    x = np.linspace(-1, 1, len(h))
    coeffs = np.zeros(n_cheb)
    for k in range(n_cheb):
        t = np.cos(k * np.arccos(x))
        coeffs[k] = np.mean(t * h) * (2 / len(h) if k > 0 else 1 / len(h))
    return coeffs


def chebyshev_reconstruct(coeffs, n):
    x = np.linspace(-1, 1, n)
    h = np.zeros(n)
    for k, c in enumerate(coeffs):
        h += c * np.cos(k * np.arccos(x))
    return h

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def build_overlap_constraints(h, n, dx, top_m=None):
    g = 1.0 - h
    constraints_data = []
    corr = np.correlate(h, g, mode='full') * dx
    if top_m is not None and top_m < 2 * n - 1:
        idx = np.argsort(corr)[::-1][:top_m]
    else:
        idx = np.arange(2 * n - 1)
    for m in idx:
        k = m - (n - 1)
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints_data.append(coeff)
    return constraints_data, idx

    for it in range(max_iter):
        if time.time() - start > time_limit:

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

def smooth_max_obj(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    start = time.time()
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break

        def obj_func(h, a=alpha):
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            return smooth_max_obj(corr, a)

        try:
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol,
                                   'disp': False})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc <= best_c5 + 1e-13:
                h_cur = h_new
                best_c5 = sc
        except Exception:
            pass

    return h_cur

        idxs = np.arange(n)

        idx_v2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    # Phase 6: Higher resolution
    if tleft() > 200:
        for n_try in [500, 700, 1000]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)

            h_lp, c5_lp = active_set_lp(h_interp, n_try, target_try,
                                         max_iter=8000, time_limit=min(80, tleft() * 0.3),
                                         top_m=200, damping_init=1.0, reg=1e-5, eps=1e-9)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try

            if tleft() > 80:
                h_ref = slsqp_refine(best_h, n, target,
                                      alphas=[500, 2000, 10000, 50000],
                                      maxiter_per_alpha=1500, ftol=1e-16,
                                      time_limit=min(50, tleft() * 0.3))
                sc = compute_c5(h_ref)
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref

    # Phase 7: Binary-based exploration
    if tleft() > 80:
        h_bin = binary_round(best_h, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
            n = len(best_h)
            target = n / 2.0

        h_ref = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                              alphas=[100, 500, 2000, 10000, 50000],
                              maxiter_per_alpha=2000, ftol=1e-16,
                              time_limit=min(50, tleft() * 0.5))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final LP with extremely tight settings
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp, c5_lp = active_set_lp(best_h, n_use, target_use,
                                     max_iter=15000, time_limit=min(40, tleft() * 0.6),
                                     top_m=100, damping_init=1.0, reg=1e-10, eps=1e-10)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 9: Final SLSQP with extreme alpha
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_refine(best_h, n_use, target_use,
                                alphas=[50000, 200000, 500000, 2000000],
                                maxiter_per_alpha=2000, ftol=1e-17,
                                time_limit=min(25, tleft() * 0.7))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final subgradient micro-adjustment
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_step(best_h, n_use, target_use, time_budget=min(10, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
