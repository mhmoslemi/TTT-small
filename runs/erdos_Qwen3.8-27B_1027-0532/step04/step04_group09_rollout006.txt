import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def grad_corr_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    if 0 <= k < n:
        grad[:n - k] -= h[k:]
        grad[k:] += (1.0 - h[:n - k])
    elif -n < k < 0:
        kk = -k
        grad[kk:] -= h[:n - kk]
        grad[:n - kk] += (1.0 - h[kk:])
    return grad

def slsqp_step(h, target, alpha, maxiter, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur_score = compute_c5(h)

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new, sc
    except Exception:
        pass
    return h, cur_score

def active_set_opt(h, target, m_active, n_iters, deadline, use_nesterov=True):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    h_prev = h_cur.copy()
    momentum = 0.0

    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        top_idx = np.argsort(corr)[::-1][:m_active]

        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            grad += grad_corr_k(h_cur, k)
        grad /= len(top_idx)

        if use_nesterov:
            grad = grad + momentum * (h_cur - h_prev)

        step = 0.03 / (1.0 + 0.01 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

        if sc < c_max * (2.0 / n):
            h_prev = h_cur.copy()
            h_cur = h_new
            target_mom = 0.9 if it > 50 else 0.5
            momentum = 0.8 * momentum + 0.2 * target_mom
        else:
            h_cur = project_feasible(h_cur - 0.005 * grad, target)
            momentum *= 0.5

    return best_h, best_score

def perturb_smooth(h, target, scale, rng, n_harm=8):
    n = len(h)
    noise = np.zeros(n)
    x = np.linspace(0, 2, n)
    for j in range(1, n_harm + 1):
        amp = rng.randn() * scale / j
        phase = rng.uniform(0, 2 * np.pi)
        noise += amp * np.cos(j * np.pi * x + phase)
    h_new = h + noise
    return project_feasible(h_new, target)

def cosine_start(n, a1=0.5, a2=0.0, a3=0.0):
    x = np.linspace(0, 2, n)
    h = 0.5 + a1 * np.cos(np.pi * x) + a2 * np.cos(2 * np.pi * x) + a3 * np.cos(3 * np.pi * x)
    return project_feasible(h, n / 2.0)

def random_smooth_start(n, rng, n_harm=6):
    x = np.linspace(0, 2, n)
    h = 0.5 * np.ones(n)
    for j in range(1, n_harm + 1):
        amp = rng.randn() * 0.2 / j
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(j * np.pi * x + phase)
    return project_feasible(h, n / 2.0)

def topk_active_refine(h, target, n_refine, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    for it in range(n_refine):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        active = np.argsort(corr)[::-1][:3]

        sub_grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            sub_grad += grad_corr_k(h_cur, k)
        sub_grad /= len(active)

        for step_mult in [1.0, 0.5, 0.25, 0.1]:
            step = 0.02 * step_mult / (1.0 + 0.005 * it)
            h_try = project_feasible(h_cur - step * sub_grad, target)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try.copy()
                h_cur = h_try
                break
        else:
            h_cur = project_feasible(h_cur - 0.003 * sub_grad, target)

    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
