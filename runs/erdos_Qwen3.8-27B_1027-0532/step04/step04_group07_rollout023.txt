def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            nfree = mask.sum()
            if nfree == 0:
                break
            inc = min(diff / nfree, 1.0 - h[mask]).max() if nfree > 0 else 0
            # Simple: add equal amount to all free
            amt = diff / nfree
            h[mask] = np.minimum(h[mask] + amt, 1.0)
        else:
            mask = h > 1e-15
            nfree = mask.sum()
            if nfree == 0:
                break
            amt = diff / nfree
            h[mask] = np.maximum(h[mask] + amt, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(x, alpha, topk):
    corr = get_corrs(x)
    nc = len(corr)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    m = vals.max()
    s = np.exp(np.clip(alpha * (vals - m), -500, 0))
    return m + np.log(s.sum()) / alpha

def smooth_grad(x, alpha, topk):
    corr = get_corrs(x)
    nc = len(corr)
    n = len(x)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(nc)
        vals = corr
    m = vals.max()
    s = np.exp(np.clip(alpha * (vals - m), -500, 0))
    w = s / s.sum()
    grad = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-15:
            continue
        lag = idx[i] - (n - 1)

                grad[lag:] += wi * (1.0 - x[:n-lag])

                grad[mm:] -= wi * x[:n-mm]

    n = len(h0)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    best_h = h0.copy()

            res = minimize(

                jac=smooth_grad, method='L-BFGS-B',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 2000}
            )
            h_new = project_feasible(res.x, target)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new
                h0 = h_new
        except Exception:
            pass
    return best_h, best_score

Nesterov's method:
- y_k = x_k + β_k (x_k - x_{k-1})  [extrapolation]
- g_k = ∇f(y_k)  [gradient at extrapolated point]
- x_{k+1} = P(y_k - α_k g_k)  [projected step]
- α_k = 1/L  [step size, L is Lipschitz constant]
- β_k = (1 - √(1 + 4t_k²)) / (1 + √(1 + 4t_k²))  [momentum]
  where t_{k+1} = (1 + √(1 + 4t_k²)) / 2, t_0 = 1

Or the simpler FISTA variant:
- y_k = x_k + (t_{k-1} - 1)/(t_k) * (x_k - x_{k-1})
- x_{k+1} = P(y_k - (1/L) * ∇f(y_k))
- t_{k+1} = (1 + √(1 + 4t_k²)) / 2

The key advantage of Nesterov/FISTA is the O(1/k²) rate vs O(1/k) for vanilla gradient descent.

For the step size 1/L, we need the Lipschitz constant of the gradient. For the smooth objective (log-sum-exp), the Hessian is bounded, so the Lipschitz constant is bounded. We can estimate it empirically.

        phases_b = [
            (5, 400),
            (10, 300),
            (20, 200),
            (50, 120),
        ]
        h_new, score = nesterov_opt(best_h, target, phases_b,
                                     max_iter_per_phase=300,
                                     t0=start_time, time_limit=remaining() - 230)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: LP refinement broad
    if remaining() > 200:
        h_new, score = lp_refine(best_h, target, max_iter=500, topk=150,
                                  t0=start_time, time_limit=remaining() - 180)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 3: Momentum with medium settings
    if remaining() > 250:
        h_new, score = momentum_opt(best_h, target, alpha=100, topk=50,
                                     max_iter=5000, lr_init=0.05, momentum=0.92,
                                     t0=start_time, time_limit=remaining() - 230,
                                     noise_prob=0.01, noise_scale=0.005, seed=seed)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 4: LP refinement medium
    if remaining() > 200:
        h_new, score = lp_refine(best_h, target, max_iter=500, topk=60,
                                  t0=start_time, time_limit=remaining() - 180)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 5: Nesterov narrow
    if remaining() > 250:
        phases_n = [
            (100, 40),
            (500, 20),
            (2000, 10),
            (10000, 5),
        ]
        h_new, score = nesterov_opt(best_h, target, phases_n,
                                     max_iter_per_phase=500,
                                     t0=start_time, time_limit=remaining() - 230)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 6: LP refinement narrow
    if remaining() > 200:
        h_new, score = lp_refine(best_h, target, max_iter=800, topk=30,
                                  t0=start_time, time_limit=remaining() - 180)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 7: SLSQP fine-tuning
    if remaining() > 250:
        pairs = [(500, 30), (2000, 15), (10000, 8), (50000, 4)]
        h_new, score = slsqp_refine(best_h, target, pairs, maxiter=1000,
                                     t0=start_time, time_limit=remaining() - 230)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 8: Final LP polish
    if remaining() > 150:
        h_new, score = lp_refine(best_h, target, max_iter=500, topk=15,
                                  t0=start_time, time_limit=remaining() - 130)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: Very fine momentum
    if remaining() > 150:
        h_new, score = momentum_opt(best_h, target, alpha=5000, topk=5,
                                     max_iter=10000, lr_init=0.01, momentum=0.95,
                                     t0=start_time, time_limit=remaining() - 130,
                                     seed=seed + 100)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 10: Final LP
    if remaining() > 80:
        h_new, score = lp_refine(best_h, target, max_iter=300, topk=10,
                                  t0=start_time, time_limit=remaining() - 60)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 11: Ultra-fine LP
    if remaining() > 50:
        h_new, score = lp_refine(best_h, target, max_iter=200, topk=5,
                                  t0=start_time, time_limit=remaining() - 35)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
