    return compute_grad_multi_lag(x, top_idx, weights)

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def slsqp_opt(h_start, target_sum, alpha, topk, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    try:
        res = minimize(
            smooth_obj, h_start, args=(alpha, topk),
            jac=smooth_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def lbfgs_opt(h_start, target_sum, alpha, topk, maxiter, penalty):
    """L-BFGS-B with penalty for sum constraint."""
    n = len(h_start)
    
    def obj(x):
        base = smooth_obj(x, alpha, topk)
        pen = penalty * (np.sum(x) - target_sum) ** 2
        return base + pen
    
    def grad(x):
        base_g = smooth_grad(x, alpha, topk)
        pen_g = 2.0 * penalty * (np.sum(x) - target_sum)
        return base_g + pen_g * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj, h_start, jac=grad, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def lp_refinement(h_start, target_sum, max_iterations, topk, time_limit=None):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    for it in range(max_iterations):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        n = len(h)
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        m = len(top_idx)
        A = np.zeros((m, n + 1))
        b = np.zeros(m)
        for i, idx in enumerate(top_idx):
            g = compute_grad_single_lag(h, idx)
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[idx]
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success:
            h_lp = res.x[:n]
            h_lp = project_feasible(h_lp, target_sum)
            score = compute_c5(h_lp)

    best_h = h.copy()

    stagnation = 0

        n_corr = len(corr)
        if topk < n_corr:

        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        
        if nesterov:
            v_new = momentum * v + (1.0 - momentum) * grad
            h_new = h - lr * (v_new + momentum * (v_new - v))
        else:
            v = momentum * v + (1.0 - momentum) * grad
            h_new = h - lr * v
            v_new = v
        
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)

            best_score = score
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        if score < best_score + 1e-10:
            lr = min(lr * 1.005, 5.0)
        else:
            lr *= 0.98
        h = h_new
        v = v_new
        if stagnation > patience:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):

        w = weights[i]
        if w < 1e-14:

        lag = lag_indices[i] - (n - 1)

    return np.ones(_CONS_N)


def slsqp_opt(h_start, target_sum, alpha, topk, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    try:
        res = minimize(
            smooth_obj, h_start, args=(alpha, topk),
            jac=smooth_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)


def lbfgs_opt(h_start, target_sum, alpha, topk, maxiter, penalty):
    n = len(h_start)

    def obj(x):
        base = smooth_obj(x, alpha, topk)
        pen = penalty * (np.sum(x) - target_sum) ** 2
        return base + pen

    def grad(x):
        base_g = smooth_grad(x, alpha, topk)
        pen_g = 2.0 * penalty * (np.sum(x) - target_sum)
        return base_g + pen_g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj, h_start, jac=grad, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10}
        )
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)


def lp_refinement(h_start, target_sum, max_iterations, topk, time_limit=None):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    for it in range(max_iterations):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        n = len(h)
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        m = len(top_idx)
        A = np.zeros((m, n + 1))
        b = np.zeros(m)
        for i, idx in enumerate(top_idx):
            g = compute_grad_single_lag(h, idx)
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[idx]
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success:
            h_lp = res.x[:n]
            h_lp = project_feasible(h_lp, target_sum)
            score = compute_c5(h_lp)
            improved = False
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_lp.copy()
                h = h_lp
                improved = True
            else:
                for t in [0.5, 0.25, 0.1, 0.05]:
                    h_t = project_feasible((1 - t) * h + t * h_lp, target_sum)
                    sc_t = compute_c5(h_t)
                    if sc_t < best_score - 1e-15:
                        best_score = sc_t
                        best_h = h_t.copy()
                        h = h_t
                        improved = True
                        break
            if improved:
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 20:
                    break
        else:
            break
    return best_h, best_score


def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed, nesterov=True):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    v_new = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)

        if nesterov:
            v_new = momentum * v + (1.0 - momentum) * grad
            h_new = h - lr * (v_new + momentum * (v_new - v))
        else:
            v_new = momentum * v + (1.0 - momentum) * grad
            h_new = h - lr * v_new

        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        if score < best_score + 1e-10:
            lr = min(lr * 1.003, 5.0)
        else:
            lr *= 0.985
        h = h_new
        v = v_new
        if stagnation > patience:
            break
    return best_h, best_score


def sa_refinement(h_init, target_sum, max_iter, temp_init, temp_min, cooldown,
                  step_scale, seed, time_limit=None):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    temp = temp_init
    t0 = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        if temp < temp_min:
            break
        r = rng.random()
        if r < 0.5:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-step_scale, step_scale)
            hi_new = np.clip(h[i] + delta, 0.0, 1.0)
            hj_new = np.clip(h[j] - delta, 0.0, 1.0)
            actual_delta = hi_new - h[i]
            h[i] = hi_new
            h[j] = np.clip(h[j] - actual_delta, 0.0, 1.0)
        else:
            i = rng.randint(n)
            delta = rng.uniform(-step_scale, step_scale)
            h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            others = np.delete(np.arange(n), i)
            free = (h[others] > 1e-14) & (h[others] < 1.0 - 1e-14)
            if free.any():
                idx = others[free][:min(10, free.sum())]
                per = diff / len(idx)
                h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
        score = compute_c5(h)
        delta_e = score - current_score
        if delta_e < 0 or rng.random() < np.exp(-delta_e / (temp + 1e-12)):
            current_score = score
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h = best_h if score > best_score + 0.01 else h
        temp *= cooldown
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_start(n, target, n_blocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    per_block = n // n_blocks
    for b in range(n_blocks):
        start = b * per_block
        end = min(start + per_block, n)
        if b % 2 == 0:
            h[start:end] = 1.0
    h = h + rng.randn(n) * 0.1
    return project_feasible(h, target)


def sinusoidal_start(n, target, freq, phase, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2 * np.pi * freq, n) + phase
    h = 0.5 + 0.4 * np.sin(x)
    h = h + rng.randn(n) * 0.05
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _CONS_N, _CONS_TARGET
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    subgrad_configs = [
        (2000, 0.10, 5, 30, 0.02, 0.01, 200, 0.93),
        (2000, 0.06, 8, 50, 0.01, 0.005, 250, 0.90),
        (1500, 0.15, 3, 20, 0.04, 0.02, 150, 0.94),
        (1500, 0.04, 10, 80, 0.0, 0.0, 300, 0.85),
        (1500, 0.08, 5, 40, 0.01, 0.005, 200, 0.95),
        (1000, 0.03, 3, 100, 0.0, 0.0, 200, 0.97),
    ]

    for i, cfg in enumerate(subgrad_configs):
        if _remaining() < 60:
            break
        h_try = best_h + rng.randn(n_init) * 0.004
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                        max_iter=cfg[0], lr_init=cfg[1], topk=cfg[2],
                                        alpha=cfg[3], noise_prob=cfg[4], noise_scale=cfg[5],
                                        patience=cfg[6], momentum=cfg[7], seed=seed + i * 100)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 150:
        lbfgs_configs = [
            (30, 8, 200), (50, 5, 100), (100, 3, 50), (200, 2, 20), (500, 1, 10)
        ]
        for i, (alpha, topk, pen) in enumerate(lbfgs_configs):
            if _remaining() < 70:
                break
            h_new, score = lbfgs_opt(best_h, target_init, alpha, topk, 500, pen)
            if score < best_score:
                best_score = score
                best_h = h_new
            if _remaining() > 60:
                h_new2, score2 = run_subgradient(best_h, target_init,
                                                  max_iter=800, lr_init=0.03, topk=topk,
                                                  alpha=alpha, noise_prob=0.0, noise_scale=0.0,
                                                  patience=100, momentum=0.96, seed=seed + 3000 + i)
                if score2 < best_score:
                    best_score = score2
                    best_h = h_new2

    if _remaining() > 180:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=400, topk=40,
                                       time_limit=min(120, _remaining() - 80))
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 150:
        slsqp_configs = [(100, 5), (300, 3), (600, 2), (1000, 1), (2000, 1), (5000, 1)]
        for alpha, topk in slsqp_configs:
            if _remaining() < 70:
                break
            h_new, score = slsqp_opt(best_h, target_init, alpha, topk, 1500, 1e-16)
            if score < best_score:
                best_score = score
                best_h = h_new

    if _remaining() > 100:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=300, topk=15,
                                       time_limit=min(70, _remaining() - 40))
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 100:
        h_new, score = sa_refinement(best_h, target_init,
                                      max_iter=3000, temp_init=0.02, temp_min=0.001,
                                      cooldown=0.998, step_scale=0.1, seed=seed + 7000,
                                      time_limit=min(80, _remaining() - 30))
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 150:
        shake_configs = [
            (0.02, [(50, 5), (200, 2)]),
            (0.04, [(30, 8), (100, 3)]),
            (0.06, [(20, 10), (50, 5)]),
        ]
        for i, (perturb, slqps) in enumerate(shake_configs):
            if _remaining() < 80:
                break
            h_shake = best_h + rng.randn(n_init) * perturb
            h_shake = project_feasible(h_shake, target_init)
            sc_shake = compute_c5(h_shake)
            for alpha, topk in slqps:
                if _remaining() < 40:
                    break
                h_new, score = slsqp_opt(h_shake, target_init, alpha, topk, 1000, 1e-16)
                if score < best_score:
                    best_score = score
                    best_h = h_new
            h_new, score = run_subgradient(h_shake, target_init,
                                            max_iter=500, lr_init=0.05, topk=3,
                                            alpha=50, noise_prob=0.0, noise_scale=0.0,
                                            patience=80, momentum=0.95, seed=seed + 8000 + i)
            if score < best_score:
                best_score = score
                best_h = h_new

    if _remaining() > 100:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=200, topk=10,
                                       time_limit=min(60, _remaining() - 30))
        if score < best_score:
            best_score = score
            best_h = h_new

    for n_try in [600, 800]:
        if _remaining() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        sc_interp = compute_c5(h_interp)
        if sc_interp < best_score:
            best_score = sc_interp
            best_h = h_interp

        if _remaining() > 80:
            h_new, score = run_subgradient(best_h, target_try,
                                            max_iter=1200, lr_init=0.06, topk=5,
                                            alpha=40, noise_prob=0.01, noise_scale=0.005,
                                            patience=150, momentum=0.94, seed=seed + 5000 + n_try)
            if score < best_score:
                best_score = score
                best_h = h_new

        if _remaining() > 60:
            h_new2, score2 = lbfgs_opt(best_h, target_try, 100, 3, 300, 50)
            if score2 < best_score:
                best_score = score2
                best_h = h_new2

        if _remaining() > 50:
            h_new3, score3 = lp_refinement(best_h, target_try,
                                            max_iterations=150, topk=25,
                                            time_limit=min(40, _remaining() - 30))
            if score3 < best_score:
                best_score = score3
                best_h = h_new3

        if _remaining() > 40:
            h_new4, score4 = slsqp_opt(best_h, target_try, 200, 2, 800, 1e-16)
            if score4 < best_score:
                best_score = score4
                best_h = h_new4

    n_use = len(best_h)
    target_use = n_use / 2.0

    if _remaining() > 50:
        h_try = best_h + rng.randn(n_use) * 0.002
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient(h_try, target_use,
                                        max_iter=2000, lr_init=0.015, topk=2,
                                        alpha=200, noise_prob=0.0, noise_scale=0.0,
                                        patience=200, momentum=0.97, seed=seed + 10000)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 30:
        h_new, score = lp_refinement(best_h, target_use,
                                       max_iterations=100, topk=8,
                                       time_limit=min(20, _remaining() - 10))
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 20:
        h_new, score = slsqp_opt(best_h, target_use, 5000, 1, 1000, 1e-17)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
