```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                lambda x: lse_obj_and_grad(x, alpha)[0],
                h_cur,
                jac=lambda x: lse_obj_and_grad(x, alpha)[1],
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                             'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def binarize(h):
    """Convert to binary solution using threshold + sum adjustment"""
    n = len(h)
    target = n // 2
    # Sort by value, take top n/2 as 1s
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa_binary(h_init, n_iter, T0, T1, deadline, rng):
    """Simulated annealing on binary solutions"""
    n = len(h_init)
    target = n // 2
    h = binarize(h_init)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    cur_score = best_score
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        
        # Pick random 1 and random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Make the swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            cur_score = new_score
            # Update ones and zeros
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_score

def local_search_binary(h, n_rounds, deadline, rng):
    """Local search: try all single-swap moves, keep best improvement"""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        
        # Try a random sample of swaps (all would be O(n^2) = 40000 per round)
        # Actually for n=400, that's 200*200 = 40000 swaps, each O(n) = 400
        # Total: 16M operations per round. With 10 rounds: 160M. Should be ok in a few seconds.
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        for idx_i in range(n_ones):
            if time.time() > deadline:
                return h_cur, cur_score
            i = ones[idx_i]
            for idx_j in range(n_zeros):
                j = zeros[idx_j]
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                sc = compute_c5(h_cur)
                h_cur[i] = 1.0
                h_cur[j] = 0.0
                improvement = cur_score - sc
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h_cur[best_i] = 0.0
            h_cur[best_j] = 1.0
            cur_score -= best_improvement
    
    return h_cur, cur_score
```

```python
        # Now for swap (i, j): delta = prof[i] - prof[j]
        # new_max = max(corr + prof[i] - prof[j])
        
        zero_prof = prof[zeros]  # (n_zeros, 2n-1)
        
        for i in ones:
            # new_max for all j: max(corr + prof[i] - zero_prof, axis=1)
            tmp = corr[None, :] + prof[i][None, :] - zero_prof  # (n_zeros, 2n-1)
            new_maxes = tmp.max(axis=1)
            min_new = new_maxes.min()
            if min_new < M - best_improvement:
                j_idx = int(new_maxes.argmin())
                best_improvement = M - min_new
                best_i = i
                best_j = zeros[j_idx]
        
        if best_i < 0:
            break  # No improving swap found
        
        # Apply the swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        ones.remove(best_i)
        zeros.remove(best_j)
        ones.append(best_j)
        zeros.append(best_i)
        
        # Recompute corr
        corr = compute_corr(h)
        M = float(corr.max())
    
    return h, M * dx
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def lse_obj(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            grad[kk:] -= wi * h[:n - kk]
            grad[:n - kk] += wi * (1.0 - h[kk:])
    return grad

    try:
        res = minimize(
            lambda x: lse_obj(x, alpha),
            h,
            jac=lambda x: lse_grad(x, alpha),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return project_feasible(res.x, target)
    except Exception:
        return h

def compute_profiles(h):
    """Compute profile matrix: prof[p][k+(n-1)] = h[p+k] + h[p-k] (0 if out of bounds)"""
    n = len(h)
    k_range = np.arange(-(n-1), n)
    prof = np.zeros((n, 2*n-1))
    for p in range(n):
        vk = (k_range + p >= 0) & (k_range + p < n)
        prof[p, vk] += h[k_range[vk] + p]
        vk2 = (p - k_range >= 0) & (p - k_range < n)
        prof[p, vk2] += h[p - k_range[vk2]]
    return prof
