def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1

    return c_max / n + np.log(np.sum(np.exp(shifted))) / alpha

def _smooth_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr / n - c_max / n), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad / n

def fast_local_search(h_start, target, time_limit, start_time):

    #              or i <= n-1-m (as (1-h[i]) with h[i+m])

    #   m >= 0: i in [m, n-1] (as h[i]) or i in [0, n-1-m] (in (1-h[i]))
    #   m < 0: i in [0, n-1+mm] (as h[i]) or i in [mm, n-1] (in (1-h[i]))

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def _cons_fun(x):
    return np.sum(x) - _cons_target

def _cons_jac(x):
    return np.ones(_cons_n)

def _smooth_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max) / n, -500.0, 0.0)
    return c_max / n + np.log(np.sum(np.exp(shifted))) / alpha

def _smooth_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr - c_max) / n, -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad / n

def slsqp_optimize(h_start, target, time_limit, start_time, alpha_schedule=None, maxiter=400):
    global _cons_n, _cons_target
    n = len(h_start)

            res = minimize(
                _smooth_obj, h, args=(alpha, n, target),

            h_new = project_feasible(res.x, target)

            if c5_new < best_c5:

                best_h = h_new.copy()
                h = h_new

        h_s2, c5_s2 = slsqp_optimize(
            best_h, target, t5, start_time,
            alpha_schedule=[10000, 30000, 80000, 200000],
            maxiter=600
        )
        if c5_s2 < best_score:
            best_score = c5_s2
            best_h = h_s2.copy()

    # Phase 6: More targeted local search
    if tleft() > 20:
        t6 = min(40.0, tleft() * 0.1)
        h_ls2, c5_ls2 = targeted_local_search(best_h, target, t6, start_time)
        if c5_ls2 < best_score:
            best_score = c5_ls2
            best_h = h_ls2.copy()

    # Phase 7: Multi-resolution
    if tleft() > 80:
        for n_try in [800, 1000, 1200]:
            if tleft() < 60:
                break
            if n_try % 2 != 0:
                n_try += 1
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin_try = project_binary(h_interp, target_try)
            t_sa_r = min(15.0, tleft() * 0.1)
            h_sa_r, c5_sa_r = sa_optimize(h_bin_try, target_try, 15000, 0.008, 0.0001,
                                          rng, time_limit=t_sa_r, start_time=start_time)
            if tleft() > 40:
                t_slq_r = min(50.0, tleft() * 0.25)
                h_slq_r, c5_slq_r = slsqp_optimize(
                    h_sa_r, target_try, t_slq_r, start_time,
                    alpha_schedule=[2000, 10000, 50000, 100000],
                    maxiter=300
                )
                if c5_slq_r < c5_sa_r:
                    h_sa_r = h_slq_r
                    c5_sa_r = c5_slq_r
            if tleft() > 15:
                h_ls_r, c5_ls_r = targeted_local_search(h_sa_r, target_try, min(20.0, tleft()*0.1), start_time)
                if c5_ls_r < c5_sa_r:
                    h_sa_r = h_ls_r
                    c5_sa_r = c5_ls_r
            if c5_sa_r < best_score:
                best_score = c5_sa_r
                best_h = h_sa_r.copy()

    # Phase 8: Final SLSQP + local search
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t8 = min(tleft() - 10, 80.0)
        h_f, c5_f = slsqp_optimize(
            best_h, target_use, t8, start_time,
            alpha_schedule=[50000, 100000, 200000, 500000],
            maxiter=500
        )
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()
    if tleft() > 10:
        h_fl, c5_fl = targeted_local_search(best_h, len(best_h)/2.0, min(10.0, tleft()-3), start_time)
        if c5_fl < best_score:
            best_score = c5_fl
            best_h = h_fl.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score
    return best_h, best_score, n_final
