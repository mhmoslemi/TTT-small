def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    # Bisection on lambda
    lam_lo = h.min() - target_sum
    lam_hi = h.max() + target_sum
    for _ in range(100):
        lam = (lam_lo + lam_hi) / 2
        projected = np.clip(h - lam, 0.0, 1.0)
        s = projected.sum()
        if abs(s - target_sum) < 1e-15:
            break
        if s > target_sum:
            lam_lo = lam
        else:
            lam_hi = lam
    return np.clip(h - lam, 0.0, 1.0)

    h = np.clip(x, 0.0, 1.0)
    # Smooth max
    corr_dx = compute_corr_dx(h)
    max_c = np.max(corr_dx)
    shifted = np.clip(alpha * (corr_dx - max_c), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    f_val = max_c + np.log(w_sum) / alpha
    f_grad = np.zeros(n)
    if w_sum > 1e-300:
        weights = weights / w_sum
        for k in range(len(corr_dx)):
            w = weights[k]

                    f_grad[:n-m] += w * (1.0 - h[m:])

                if n - mm > 0:

                    f_grad[:n-mm] -= w * h[mm:]

    s = np.sum(h) - target

    grad = f_grad + (lam + 2.0 * mu * s) * np.ones(n)
