def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0

    corr = np.correlate(x, 1.0 - x, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0
_g_h_full = None
_g_block_start = 0
_g_block_end = 0
_g_block_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    a = _g_alpha
    shifted = np.clip(a * corr, -700.0, 700.0)
    mv = np.max(shifted)
    return float((mv + np.log(np.sum(np.exp(shifted - mv)))) / a)

def smooth_grad(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    a = _g_alpha
    shifted = np.clip(a * corr, -700.0, 700.0)
    mv = np.max(shifted)
    ex = np.exp(shifted - mv)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(len(corr)):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        lo1 = max(0, s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 + s:hi1 + s])
        lo2 = max(0, -s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * x[lo2 - s:hi2 - s]
    return grad

def con_eq(x):
    return np.sum(x) - _g_target

def con_jac(x):
    return np.ones(len(x))

def block_obj(x):
    h = _g_h_full.copy()
    h[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = _g_alpha
    shifted = np.clip(a * corr, -700.0, 700.0)
    mv = np.max(shifted)
    return float((mv + np.log(np.sum(np.exp(shifted - mv)))) / a)

def block_grad(x):
    h = _g_h_full.copy()
    h[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = _g_alpha
    shifted = np.clip(a * corr, -700.0, 700.0)
    mv = np.max(shifted)
    ex = np.exp(shifted - mv)
    w = ex / np.sum(ex)
    b = _g_block_end - _g_block_start
    grad = np.zeros(b)
    for k in range(len(corr)):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        for j in range(b):
            ji = _g_block_start + j
            term = 0.0
            if 0 <= ji + s < n:
                term += (1.0 - h[ji + s])
            if 0 <= ji - s < n:
                term -= h[ji - s]
            if term != 0.0:
                grad[j] += wk * term
    return grad

def block_con_eq(x):
    return np.sum(x) - _g_block_target

def block_con_jac(x):
    return np.ones(len(x))

def full_slsqp(h, n, target, alpha, maxiter=2000):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = float(target)
    _g_alpha = float(alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP',
                      jac=smooth_grad, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def block_slsqp(h, n, target, block_size, alpha, n_rounds, deadline):
    global _g_h_full, _g_block_start, _g_block_end, _g_alpha, _g_block_target
    _g_alpha = float(alpha)
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    target_f = float(target)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_full = h_cur
            _g_block_start = s
            _g_block_end = e
            bt = target_f - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            bt = float(np.clip(bt, 0.0, float(block_size)))
            _g_block_target = bt
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
            try:
                res = minimize(block_obj, x0, method='SLSQP',
                              jac=block_grad, bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 400, 'ftol': 1e-14})
                h_new = h_cur.copy()
                h_new[s:e] = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, target_f)
                sc = compute_c5(h_new)
                if sc < cur:
                    h_cur = h_new
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, new_n // 2)

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

def swap_descent(h, n, target, n_trials, deadline, seed_val=123):
    rng = np.random.RandomState(seed_val)
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    deltas = np.array([0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3])
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            j = (j + 1) % n
        for d in deltas:
            for sign in [1, -1]:
                h_try = h_cur.copy()
                hv_i = h_try[i] + sign * d
                hv_j = h_try[j] - sign * d
                if hv_i < -0.01 or hv_i > 1.01 or hv_j < -0.01 or hv_j > 1.01:
                    continue
                h_try[i] = np.clip(hv_i, 0.0, 1.0)
                h_try[j] = np.clip(hv_j, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < cur:
                    h_cur = h_try
                    cur = sc
    return h_cur, cur

            best_h = ht

    if remaining() < 80:
        return best_h, best_score, n

    # Phase 2: Full SLSQP with analytical gradient at n
    for alpha in [50, 200, 500]:
        if remaining() < 30:
            break
        h_opt = full_slsqp(best_h, n, target, alpha, maxiter=2000)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    if remaining() < 80:
        return best_h, best_score, n

    # Phase 3: Block SLSQP at n
    deadline3 = t0 + budget_s * 0.35
    for bs in [8, 6]:
        if time.time() > deadline3:
            break
        h_bd, sc = block_slsqp(best_h, n, target, bs, 1000.0, 3, deadline3)
        if sc < best_score:
            best_score = sc
            best_h = h_bd

    if remaining() < 80:
        return best_h, best_score, n

    # Phase 4: Upscale to 2n and optimize
    n2 = n * 2
    h_up = upsample(best_h, n2)
    target2 = n2 // 2
    best_h2 = h_up.copy()
    best_score2 = compute_c5(best_h2)

    for alpha in [100, 400, 1000]:
        if remaining() < 30:
            break
        h_opt = full_slsqp(best_h2, n2, target2, alpha, maxiter=1500)
        sc = compute_c5(h_opt)
        if sc < best_score2:
            best_score2 = sc
            best_h2 = h_opt

    if remaining() < 80:
        if best_score2 < best_score:
            return best_h2, best_score2, n2
        return best_h, best_score, n

    # Phase 5: Block SLSQP at 2n
    deadline5 = t0 + budget_s * 0.55
    for bs in [6, 4]:
        if time.time() > deadline5:
            break
        h_bd, sc = block_slsqp(best_h2, n2, target2, bs, 2000.0, 2, deadline5)
        if sc < best_score2:
            best_score2 = sc
            best_h2 = h_bd

    # Decide which resolution is better
    if best_score2 < best_score:
        best_h = best_h2
        best_score = best_score2
        n = n2
        target = target2

    if remaining() < 60:
        return best_h, best_score, n

    # Phase 6: Higher alpha SLSQP on best resolution
    for alpha in [2000, 5000]:
        if remaining() < 30:
            break
        h_opt = full_slsqp(best_h, n, target, alpha, maxiter=2000)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    if remaining() < 60:
        return best_h, best_score, n

    # Phase 7: Fine block SLSQP
    deadline7 = t0 + budget_s * 0.82
    for bs in [4, 3]:
        if time.time() > deadline7:
            break
        h_bd, sc = block_slsqp(best_h, n, target, bs, 5000.0, 3, deadline7)
        if sc < best_score:
            best_score = sc
            best_h = h_bd

    if remaining() < 40:
        h_final = project_feasible(best_h, target)
        return h_final, compute_c5(h_final), len(h_final)

    # Phase 8: Swap-based local search
    deadline8 = t0 + budget_s - 30
    h_sw, sc_sw = swap_descent(best_h, n, target, 50000, deadline8, seed_val=seed + 1)
    if sc_sw < best_score:
        best_score = sc_sw
        best_h = h_sw

    # Phase 9: Final SLSQP polish
    if remaining() > 30:
        h_opt = full_slsqp(best_h, n, target, 10000, maxiter=2000)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, len(best_h)
