def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def find_top_lags(corr, center, k):
    c = corr.copy()
    c[center] = -np.inf
    top_idx = np.argsort(c)[-k:]
    lags = top_idx - center
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1])
    return lags

def compute_lag_gradient(h, lag):
    n = len(h)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        grad = np.zeros(n)
        grad[lag:] += (1.0 - h[:-lag])
        grad[:-lag] -= h[lag:]
        return grad
    else:
        k = -lag
        grad = np.zeros(n)
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
        return grad

def compute_subgradient(h, n_top_lags=5):
    corr = get_corr(h)
    center = len(h) - 1
    lags = find_top_lags(corr, center, n_top_lags)
    grad = np.zeros(len(h))
    for lag in lags:
        grad += compute_lag_gradient(h, int(lag))
    return grad / len(lags), lags

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - target

    def cons_j(h_in):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fast_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 150
    grad, _ = compute_subgradient(h, n_top_lags=3)

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        if stall % 10 == 0:
            grad, _ = compute_subgradient(h, n_top_lags=3)
        one_scores = grad[ones] + rng.random(len(ones)) * 0.1
        zero_scores = grad[zeros] + rng.random(len(zeros)) * 0.1
        a_idx = ones[np.argmax(one_scores)]
        b_idx = zeros[np.argmin(zero_scores)]
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5

def make_golden_init(n, rng):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    offsets = np.arange(n) * phi
    frac = offsets - np.floor(offsets)
    h[frac < 0.5] = 1.0
    return project_binary(h, n // 2)

def make_periodic_init(n, rng):
    periods = [3, 4, 5, 7, 8]
    p = periods[rng.randint(len(periods))]
    h = np.zeros(n)
    for i in range(n):
        if (i % p) < p // 2:
            h[i] = 1.0
    return project_binary(h, n // 2)
