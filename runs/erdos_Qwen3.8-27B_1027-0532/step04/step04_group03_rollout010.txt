def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones_idx = np.where(h_r == 1)[0]
    zeros_idx = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones_idx):
            order = np.argsort(h[ones_idx])[:need]
            h_r[ones_idx[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros_idx):
            order = np.argsort(-h[zeros_idx])[:need]
            h_r[zeros_idx[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

    n = len(h)
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_vals = corr * dx
    c_max = np.max(c_vals)
    shifted = np.clip(alpha * (c_vals - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lse_grad(h, alpha, target_sum, top_k=30):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    c_max = np.max(c_vals)
    shifted = np.clip(alpha * (c_vals - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)

                        dists = np.abs(zeros.astype(int) - (i_c - mm))
                        thresh = max(mm + 5, 15)
                        candidates = zeros[dists > thresh]
                        if len(candidates) == 0:
                            candidates = zeros
                        pos_to_set = candidates[rng.randint(len(candidates))]
                        h[i_c] = 0
                        h[pos_to_set] = 1
                    else:
                        ones = np.where(h == 1)[0]
                        dists = np.abs(ones.astype(int) - (i_c - mm))
                        thresh = max(mm + 5, 15)
                        candidates = ones[dists > thresh]
                        if len(candidates) == 0:
                            candidates = ones
                        pos_to_unset = candidates[rng.randint(len(candidates))]
                        h[i_c - mm] = 1
                        h[pos_to_unset] = 0
                else:
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        h[ones[rng.randint(len(ones))]] = 0
                        h[zeros[rng.randint(len(zeros))]] = 1
            else:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    h[ones[rng.randint(len(ones))]] = 0
                    h[zeros[rng.randint(len(zeros))]] = 1

        elif move_type < 0.72:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            start = ones[rng.randint(len(ones))]
            left = start
            while left > 0 and h[left - 1] == 1:
                left -= 1
            right = start
            while right < n - 1 and h[right + 1] == 1:
                right += 1
            blen = right - left + 1
            if blen < 2:
                h[start] = 0
                h[zeros[rng.randint(len(zeros))]] = 1
            else:
                max_shift = max(3, n // 30)
                shift = rng.randint(-max_shift, max_shift + 1)
                if shift == 0:
                    shift = rng.choice([-1, 1])
                new_left = left + shift
                new_right = right + shift
                if new_left >= 0 and new_right < n:
                    feasible = True
                    src_set = set(range(left, right + 1))
                    for p in range(new_left, new_right + 1):
                        if h[p] == 1 and p not in src_set:
                            feasible = False
                            break
                    if feasible:
                        for p in range(new_left, new_right + 1):
                            if h[p] == 0:
                                h[p] = 1
                        for p in range(left, right + 1):
                            if p < new_left or p > new_right:
                                if h[p] == 1:
                                    h[p] = 0

        elif move_type < 0.92:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                h[ones[rng.randint(len(ones))]] = 0
                h[zeros[rng.randint(len(zeros))]] = 1
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_pts = rng.randint(2, 4)
            if len(ones) >= n_pts and len(zeros) >= n_pts:
                sw1 = rng.choice(ones, n_pts, replace=False)
                sw0 = rng.choice(zeros, n_pts, replace=False)
                h[sw1] = 0
                h[sw0] = 1

        c5_new = compute_c5(h)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-14 and rng.random() < np.exp(-max(0.0, delta) / t)):
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h = h_save
            no_improve += 1

    return best_h, best_c5

def continuous_refine(h_start, target_sum, deadline, top_k=30):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    no_improve = 0
    max_no_improve = 200

    for it in range(100000):
        if time.time() > deadline:
            break
        if no_improve > max_no_improve:
            break

        grad = lse_grad(h, 300.0, target_sum, top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        grad_normed = grad / grad_norm

        h_try = project_feasible(h - lr * grad_normed, target_sum)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            no_improve = 0
            lr = min(lr * 1.03, 0.3)
        else:
            h_try2 = project_feasible(h - lr * 0.2 * grad_normed, target_sum)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
                h = h_try2
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.92
                if lr < 1e-7:
                    break

    return best_h, best_c5

def slsqp_refine(h_start, target_sum, deadline):
    n = len(h_start)
    best_h = h_start.copy().astype(float)
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        hp = project_feasible(x, target_sum)
        return lse_obj(hp, 2000.0, target_sum)

    def obj_grad(x):
        hp = project_feasible(x, target_sum)
        return lse_grad(hp, 2000.0, target_sum, top_k=40)

    def cons_fun(x):
        return np.sum(x) - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = best_h.copy()

    for alpha_try in [800, 1500, 3000, 5000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Binary SA with conflict-directed moves
    h_bin = project_binary(h_init, target)
    deadline_p1 = start_time + min(200.0, budget_s * 0.2)
    h_sa1, c5_sa1 = cd_sa(h_bin, target, 500000, 0.02, 0.0001, rng, deadline_p1, top_k=5)
    if c5_sa1 < best_score:
        best_score = c5_sa1
        best_h = h_sa1.copy()

    # Phase 2: Second SA round with different params
    if tleft() > 100:
        deadline_p2 = start_time + min(200.0, budget_s * 0.4)
        h_sa2, c5_sa2 = cd_sa(best_h, target, 400000, 0.008, 0.00005, rng, deadline_p2, top_k=8)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 3: Continuous refinement
    if tleft() > 80:
        deadline_p3 = start_time + min(180.0, budget_s * 0.55)
        h_cont, c5_cont = continuous_refine(best_h, target, deadline_p3, top_k=30)
        if c5_cont < best_score:
            best_score = c5_cont
            best_h = h_cont.copy()

    # Phase 4: SLSQP refinement
    if tleft() > 80:
        deadline_p4 = start_time + min(250.0, budget_s * 0.7)
        h_slq, c5_slq = slsqp_refine(best_h, target, deadline_p4)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    # Phase 5: Multi-resolution attempts
    n_candidates = []
    for factor in [0.9, 1.1, 1.25, 1.5]:
        nc = int(n * factor)
        if nc >= 100 and nc % 2 == 0 and nc != n:
            n_candidates.append(nc)

    for n_try in n_candidates:
        if tleft() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        deadline_sa = start_time + min(60.0, (tleft() - 40) * 0.4)
        h_sa_r, c5_sa_r = cd_sa(h_bin_try, target_try, 200000, 0.01, 0.0005, rng, deadline_sa, top_k=5)

        if tleft() > 50:
            deadline_slq = start_time + min(80.0, (tleft() - 30) * 0.5)
            h_slq_r, c5_slq_r = slsqp_refine(h_sa_r, target_try, deadline_slq)
            if c5_slq_r < c5_sa_r:
                h_sa_r = h_slq_r
                c5_sa_r = c5_slq_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 6: Final SA + continuous refinement on best
    if tleft() > 40:
        h_bin_final = project_binary(best_h, len(best_h) / 2.0)
        deadline_final_sa = start_time + min(40.0, max(10.0, tleft() * 0.4))
        h_fin, c5_fin = cd_sa(h_bin_final, len(best_h) / 2.0, 200000, 0.005, 0.0001, rng, deadline_final_sa, top_k=6)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    if tleft() > 30:
        deadline_final_cont = start_time + min(30.0, max(10.0, tleft() * 0.5))
        h_fin2, c5_fin2 = continuous_refine(best_h, len(best_h) / 2.0, deadline_final_cont, top_k=40)
        if c5_fin2 < best_score:
            best_score = c5_fin2
            best_h = h_fin2.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
