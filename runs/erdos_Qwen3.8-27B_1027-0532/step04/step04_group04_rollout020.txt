import time

    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:int(target_sum)]
    h_bin[idx] = 1.0
    return h_bin

    n = len(h)
    corr = compute_corr(h)
    c_max = float(corr.max())
    shifted = alpha * (corr - c_max)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = p[s]
        if w < 1e-8:
            continue
        k = s - (n - 1)
        if k == 0:
            grad += w * (1 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:-k]
        else:
            k_abs = -k
            grad[k_abs:] += w * (1.0 - h[:n-k_abs])
            grad[:n-k_abs] -= w * h[k_abs:]
    
    return grad

def projected_gradient(h_init, alphas, max_iters_per_alpha, lr_init, lr_decay, time_limit, start_time, momentum=0.9):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        lr = lr_init
        for it in range(max_iters_per_alpha):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            
            grad = grad_smooth_max(h, alpha)
            v = momentum * v - lr * grad
            h_new = project_feasible(h + v, target)

                best_h = h_new.copy()
            
            h = h_new
            lr *= lr_decay

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    target = int(target_sum)
    # Sort by value descending, take top target
    idx = np.argsort(h)[::-1][:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.arange(n_src, dtype=float) / n_src
    x_dst = np.arange(n_dst, dtype=float) / n_dst
    return np.interp(x_dst, x_src, h_src)

    n = len(h)

            grad[:n-ka] -= w * h[ka:]
