After swapping h[a] and h[b]:

For h[a]: 1→0:

- Where a appears as h[i+s]: i=a-s, so term h[i]*(1-h[a]) goes from h[a-s]*0 to h[a-s]*1. Change: +h[a-s] if a-s is valid.

import numpy as np

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = wt[s + n - 1]
        if w < 1e-14:
            continue
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            t = -s
            grad[:n-t] += w * (1.0 - h[t:])
