def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def solve_1d_convex(a_vals, b_vals, d_lo, d_hi):
    m = len(a_vals)
    if m == 0:
        return 0.0
    best_d = 0.0
    best_f = float(np.max(b_vals))
    f_lo = np.max(a_vals * d_lo + b_vals)
    f_hi = np.max(a_vals * d_hi + b_vals)
    if f_lo < best_f:
        best_f = f_lo
        best_d = d_lo
    if f_hi < best_f:
        best_f = f_hi
        best_d = d_hi
    for j in range(m):
        for l in range(j + 1, m):
            da = a_vals[j] - a_vals[l]
            if abs(da) < 1e-15:
                continue
            d_cross = (b_vals[l] - b_vals[j]) / da
            if d_lo < d_cross < d_hi:
                f_cross = np.max(a_vals * d_cross + b_vals)
                if f_cross < best_f:
                    best_f = f_cross
                    best_d = d_cross
    return best_d


def coord_descent_step(h, corr, i, topk, n):
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    a_vals = np.zeros(m)
    for j, idx in enumerate(top_idx):
        lag = idx - (n - 1)
        a = 0.0
        if lag >= 0:
            if lag < n:
                if i <= n - 1 - lag:
                    a += (1.0 - h[i + lag])
                if i >= lag:
                    a -= h[i - lag]
        else:
            mm = -lag
            if mm < n:
                if i <= n - 1 - mm:
                    a += (1.0 - h[i + mm])
                if i >= mm:
                    a -= h[i - mm]
        a_vals[j] = a
    b_vals = corr[top_idx].copy()
    d_lo = -h[i]
    d_hi = 1.0 - h[i]
    best_d = solve_1d_convex(a_vals, b_vals, d_lo, d_hi)
    return np.clip(h[i] + best_d, 0.0, 1.0)


def run_coord_descent(h_init, target_sum, max_sweeps, topk, time_limit, start_time, seed):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if (time.time() - start_time) > time_limit:
            break
        corr = overlap_corrs(h)
        order = rng.permutation(n)
        for i in order:
            old_max = float(np.max(corr))
            old_val = h[i]
            new_val = coord_descent_step(h, corr, i, topk, n)
            if abs(new_val - old_val) > 1e-14:
                h[i] = new_val
                corr = overlap_corrs(h)
                new_max = float(np.max(corr))
                if new_max > old_max + 1e-10:
                    h[i] = old_val
                    corr = overlap_corrs(h)
        h = project_feasible(h, target_sum)
        score = compute_c5(h)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def run_sa_coord_descent(h_init, target_sum, max_sweeps, topk, time_limit, start_time, seed, T_init, T_final):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(seed)
    n_total = max_sweeps
    for sweep in range(max_sweeps):
        if (time.time() - start_time) > time_limit:
            break
        T = T_init * (T_final / T_init) ** (sweep / max(1, n_total - 1))
        corr = overlap_corrs(h)
        order = rng.permutation(n)
        for i in order:
            old_max = float(np.max(corr))
            old_val = h[i]
            new_val = coord_descent_step(h, corr, i, topk, n)
            if abs(new_val - old_val) > 1e-14:
                h[i] = new_val
                corr = overlap_corrs(h)
                new_max = float(np.max(corr))
                if new_max <= old_max + 1e-10:
                    pass
                else:
                    delta = (new_max - old_max) * (2.0 / n)
                    if rng.random() > np.exp(-delta / max(T, 1e-10)):
                        h[i] = old_val
                        corr = overlap_corrs(h)
        h = project_feasible(h, target_sum)
        score = compute_c5(h)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def lp_linearized_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None


def lp_refinement(h_start, target_sum, max_iterations, topk, time_limit, start_time):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    stagnation = 0
    for it in range(max_iterations):
        if (time.time() - start_time) > time_limit:
            break
        h_lp = lp_linearized_step(h, target_sum, topk)
        if h_lp is None:
            break
        score = compute_c5(h_lp)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stagnation = 0
        else:
            h_half = project_feasible(0.5 * h + 0.5 * h_lp, target_sum)
            sc_half = compute_c5(h_half)
            if sc_half < best_score - 1e-15:
                best_score = sc_half
                best_h = h_half.copy()
                h = h_half
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 15:
                    break
    return best_h, best_score


def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def active_set_smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)


def slsqp_active_set(h_start, target_sum, alpha_topk_pairs, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons_fun = np.sum(h_cur) - target_sum
    cons = [{'type': 'eq',
             'fun': cons_eq_wrapper,
             'jac': cons_jac_wrapper}]
    global _SLQ_N, _SLQ_TARGET
    _SLQ_N = n
    _SLQ_TARGET = target_sum
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        if (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(
                active_set_smooth_obj, h_cur, args=(alpha, topk),
                jac=active_set_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def cons_eq_wrapper(x):
    return np.sum(x) - _SLQ_TARGET


def cons_jac_wrapper(x):
    return np.ones(_SLQ_N)


_SLQ_N = 0
_SLQ_TARGET = 0.0


def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    seed, time_limit, start_time):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    momentum = 0.90
    for it in range(max_iter):
        if (time.time() - start_time) > time_limit:
            break
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.005, 2.0)
        else:
            lr *= 0.98
            h = h_new
        if lr < 1e-6:
            break
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _SLQ_N, _SLQ_TARGET
    start_time = time.time()

    def _elapsed():
        return time.time() - start_time

    def _remaining():
        return budget_s - _elapsed()

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent with moderate topk
    if _remaining() > 400:
        h_new, score = run_coord_descent(best_h, target_init,
                                         max_sweeps=50000, topk=60,
                                         time_limit=_remaining() - 350,
                                         start_time=start_time, seed=seed)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: Subgradient for exploration + coord descent
    if _remaining() > 350:
        rng = np.random.RandomState(seed + 1)
        h_try = best_h + rng.randn(n_init) * 0.015
        h_try = project_feasible(h_try, target_init)
        h_sg, sc_sg = run_subgradient(h_try, target_init,
                                      max_iter=5000, lr_init=0.2, topk=5,
                                      alpha=30, seed=seed + 10,
                                      time_limit=80, start_time=start_time)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if _remaining() > 200:
            h_new, score = run_coord_descent(best_h, target_init,
                                             max_sweeps=30000, topk=40,
                                             time_limit=_remaining() - 150,
                                             start_time=start_time, seed=seed + 20)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 3: LP refinement
    if _remaining() > 250:
        h_new, score = lp_refinement(best_h, target_init,
                                     max_iterations=300, topk=50,
                                     time_limit=120, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 100:
            h_new, score = run_coord_descent(best_h, target_init,
                                             max_sweeps=20000, topk=30,
                                             time_limit=_remaining() - 80,
                                             start_time=start_time, seed=seed + 30)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 4: SLSQP with smooth max
    if _remaining() > 180:
        alphas = [(100, 50), (500, 30), (2000, 15), (5000, 8)]
        h_ref, sc = slsqp_active_set(best_h, target_init, alphas,
                                     maxiter=3000, ftol=1e-16,
                                     time_limit=_remaining() - 120,
                                     start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: SA coordinate descent for escape
    if _remaining() > 200:
        h_new, score = run_sa_coord_descent(best_h, target_init,
                                            max_sweeps=20000, topk=40,
                                            time_limit=100,
                                            start_time=start_time, seed=seed + 50,
                                            T_init=0.01, T_final=0.0001)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 80:
            h_new, score = run_coord_descent(best_h, target_init,
                                             max_sweeps=30000, topk=25,
                                             time_limit=_remaining() - 60,
                                             start_time=start_time, seed=seed + 60)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 6: LP with tighter topk
    if _remaining() > 150:
        h_new, score = lp_refinement(best_h, target_init,
                                     max_iterations=200, topk=25,
                                     time_limit=80, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 7: Try higher resolution
    for n_try in [500, 600, 800]:
        if _remaining() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_cd, sc_cd = run_coord_descent(h_interp, target_try,
                                        max_sweeps=10000, topk=40,
                                        time_limit=40, start_time=start_time,
                                        seed=seed + 100 + n_try)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
        if _remaining() > 80:
            h_lp, sc_lp = lp_refinement(best_h, len(best_h) / 2.0,
                                        max_iterations=100, topk=30,
                                        time_limit=30, start_time=start_time)
            if sc_lp < best_score:
                best_score = sc_lp
                best_h = h_lp

    # Phase 8: Final coordinate descent polishing
    n_final = len(best_h)
    target_final = n_final / 2.0
    if _remaining() > 60:
        h_new, score = run_coord_descent(best_h, target_final,
                                         max_sweeps=50000, topk=15,
                                         time_limit=_remaining() - 40,
                                         start_time=start_time, seed=seed + 200)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: Final LP polish
    if _remaining() > 30:
        h_new, score = lp_refinement(best_h, target_final,
                                     max_iterations=50, topk=10,
                                     time_limit=_remaining() - 10,
                                     start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
