import numpy as np
from scipy.optimize import minimize
import time

# Top-level helper functions

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)[n-1:])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def active_coord_descent(h, n_sweeps, top_m, target, deadline):
    """Coordinate descent focusing on top-m active lags."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_score = compute_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        corr_over = corr[n-1:]
        
        # Find top-m active lags
        top_idx = np.argsort(corr_over)[::-1][:top_m]
        top_vals = corr_over[top_idx]
        
        for j in range(n):
            # For each active lag k, compute O_k as function of t=h[j]
            # O_k(t) = a_k * t + b_k (linear in t)
            a = np.zeros(top_m)
            b = np.zeros(top_m)
            for idx, k in enumerate(top_idx):
                k = int(k)
                # coefficient of t: (1-h[j+k])*dx * 1_{j+k<n} + (-h[j-k])*dx * 1_{j>=k}
                ak = 0.0
                if j + k < n:
                    ak += (1.0 - h[j+k]) * dx
                if j >= k:
                    ak += -h[j-k] * dx
                a[idx] = ak
                b[idx] = corr_over[k] - ak * h[j]
            
            # Minimize max_i(a[i]*t + b[i]) over t in [0,1]
            t_new = _min_1d_convex(b, a, h[j])
            h[j] = np.clip(t_new, 0.0, 1.0)
        
        h = project_feasible(h, target)
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
    
    return h, cur_score

def _min_1d_convex(b, a, t_init):
    """Minimize max_i(a[i]*t + b[i]) over t in [0,1]. 
    a and b are 1D arrays of length m (small, like 3-5)."""
    m = len(a)
    # Candidates: 0, 1, t_init, and all crossing points
    candidates = [0.0, 1.0, t_init]
    for i in range(m):
        for j in range(i+1, m):
            if abs(a[i] - a[j]) > 1e-15:
                tc = (b[j] - b[i]) / (a[i] - a[j])
                if 0.0 <= tc <= 1.0:
                    candidates.append(tc)
    
    best_t = t_init
    best_val = np.inf
    for tc in candidates:
        val = np.max(a * tc + b)
        if val < best_val:
            best_val = val
            best_t = tc
    return best_t

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                _lse_objective, best_h, args=(alpha, dx),
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (target,)}],
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

def _lse_objective(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    over = corr[len(x)-1:]
    c_max = np.max(over)
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def _sum_constraint(x, target):
    return np.sum(x) - target

def _sum_jac(x, target):
    return np.ones(len(x))

def nesterov_subgradient(h, n, target, n_steps, n_active, lr_init, deadline):
    """Nesterov-accelerated subgradient method."""
    dx = 2.0 / n
    h = h.copy()
    h_mom = h.copy()  # momentum position
    v = np.zeros(n)   # velocity
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Nesterov step: evaluate gradient at y = h + v/(lr_next)
        # Simplified: just use h_mom for gradient evaluation
        corr = compute_corr(h_mom)
        corr_over = corr[n-1:]
        
        # Softmax weights on top active lags
        top_idx = np.argsort(corr_over)[::-1][:n_active]
        diffs = corr_over[top_idx] - corr_over[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= np.sum(weights)
        
        # Subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += w * (1.0 - h_mom[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= w * h_mom[j_neg - k]
        grad *= dx
        
        # Update
        v = 0.9 * v - lr * grad
        h_new = h_mom + v
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h_mom = h_new
        h = h_new
        lr *= 0.999
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_optimize(h, n, target, block_size, n_rounds, deadline):
    """Optimize blocks of coordinates using Nelder-Mead."""
    h = h.copy()
    cur_score = compute_c5(h)
    n_blocks = n // block_size
    
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            x0 = h[s:e].copy()
            
            def _block_obj(x, s=s, e=e):
                hf = h.copy()
                hf[s:e] = np.clip(x, 0.0, 1.0)
                hf = project_feasible(hf, target)
                return compute_c5(hf)
            
            try:
                res = minimize(_block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 1000, 'xatol': 1e-10, 'fatol': 1e-12})
                h_test = h.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    cur_score = sc
                    h = h_test
            except Exception:
                pass
    return h, cur_score

    h = np.clip(h.astype(float), 0.0, 1.0)

def min_1d_convex(b, a, t_init):
    m = len(a)
    candidates = [0.0, 1.0, t_init]
    for i in range(m):
        for j in range(i + 1, m):
            ds = a[i] - a[j]
            if abs(ds) > 1e-15:
                tc = (b[j] - b[i]) / ds
                if -0.01 <= tc <= 1.01:
                    candidates.append(np.clip(tc, 0.0, 1.0))
    best_t = t_init
    best_val = np.inf
    for tc in candidates:
        val = np.max(a * tc + b)
        if val < best_val:
            best_val = val
            best_t = tc
    return best_t

def slsqp_round(h, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha, _g_dx
    n = len(h)
    _g_n = n
    _g_target = target
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for alpha in alphas:
        if time.time() > deadline:
            break
        _g_alpha = alpha
        try:
            res = minimize(lse_objective, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score


def block_obj_func(x):
    hf = _g_block_full.copy()
    hf[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    hf = project_feasible(hf, _g_target)
    n = len(hf)
    dx = 2.0 / n
    corr = compute_corr(hf)
    over = corr[n-1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha) * dx


def block_descent(h, target, block_size, n_rounds, alpha, deadline):
    global _g_block_full, _g_block_start, _g_block_end, _g_target, _g_alpha
    n = len(h)
    _g_target = target
    _g_alpha = alpha
    _g_n = n
    _g_dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_block_full = h_cur.copy()
            _g_block_start = s
            _g_block_end = e
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(block_obj_func, x0, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_test
            except Exception:
                pass
    return h_cur, cur_score


def nesterov_subgrad(h, target, n_steps, n_active, lr_init, decay, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    mu = 0.9
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        corr_over = corr[n-1:]
        top_idx = np.argsort(corr_over)[::-1][:n_active]
        diffs = corr_over[top_idx] - corr_over[top_idx[0]]
        w = np.exp(diffs / 0.004)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            jp = np.arange(0, n - k)
            grad[jp] += wi * (1.0 - h_cur[jp + k])
            jn = np.arange(k, n)
            grad[jn] -= wi * h_cur[jn - k]
        grad *= dx
        v = mu * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
        lr *= decay
        if lr < 1e-10:
            break
    return best_h, best_score


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def worst_lag_grad(h, k_star):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    jp = np.arange(0, n - k_star)
    grad[jp] += (1.0 - h[jp + k_star])
    jn = np.arange(k_star, n)
    grad[jn] -= h[jn - k_star]
    return grad * dx


def gradient_surgery(h, target, max_steps, eps_init, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        corr_over = corr[n-1:]
        k_star = int(np.argmax(corr_over))
        if k_star == 0:
            break
        grad = worst_lag_grad(h_cur, k_star)
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc
            eps = min(eps * 1.03, 0.02)
        else:
            eps *= 0.6
            if eps < 1e-8:
                break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_dx
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    deadline_1 = start_time + budget_s * 0.12
    h_ref, sc = slsqp_round(best_h, target, [50, 200, 800], 3000, 1e-15, deadline_1)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    deadline_2 = start_time + budget_s * 0.35
    h_cd, sc = active_coord_descent(best_h, 5000, 5, target, deadline_2)
    if sc < best_score:
        best_score = sc
        best_h = h_cd

    if tleft() > 150:
        deadline_3 = start_time + budget_s * 0.45
        h_ref, sc = slsqp_round(best_h, target, [500, 2000, 8000], 3000, 1e-16, deadline_3)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    deadline_4 = start_time + budget_s * 0.55
    h_cd, sc = active_coord_descent(best_h, 8000, 7, target, deadline_4)
    if sc < best_score:
        best_score = sc
        best_h = h_cd

    if tleft() > 120:
        deadline_5 = start_time + budget_s * 0.62
        h_sg, sc = nesterov_subgrad(best_h, target, 600, 4, 0.004, 0.998, deadline_5)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    if tleft() > 100:
        deadline_6 = start_time + budget_s * 0.68
        h_ref, sc = slsqp_round(best_h, target, [2000, 8000, 20000], 2500, 1e-16, deadline_6)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    deadline_7 = start_time + budget_s * 0.78
    h_cd, sc = active_coord_descent(best_h, 6000, 5, target, deadline_7)
    if sc < best_score:
        best_score = sc
        best_h = h_cd

    if tleft() > 100:
        deadline_8 = start_time + budget_s * 0.83
        for bs in [3, 4]:
            if tleft() < 50:
                break
            h_bd, sc = block_descent(best_h, target, bs, 5, 1000.0, deadline_8)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    if tleft() > 80:
        deadline_9 = start_time + budget_s * 0.88
        h_sg, sc = nesterov_subgrad(best_h, target, 400, 5, 0.002, 0.999, deadline_9)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    if tleft() > 60:
        deadline_10 = start_time + budget_s * 0.92
        h_cg, sc = gradient_surgery(best_h, target, 200, 0.002, deadline_10)
        if sc < best_score:
            best_score = sc
            best_h = h_cg

    deadline_11 = start_time + budget_s * 0.95
    h_cd, sc = active_coord_descent(best_h, 5000, 5, target, deadline_11)
    if sc < best_score:
        best_score = sc
        best_h = h_cd

    if tleft() > 30:
        deadline_12 = start_time + budget_s * 0.97
        h_ref, sc = slsqp_round(best_h, target, [5000, 20000], 2000, 1e-17, deadline_12)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 15:
        h_cd, sc = active_coord_descent(best_h, 2000, 7, target, start_time + budget_s - 10.0)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    final_h = project_feasible(best_h, target)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
