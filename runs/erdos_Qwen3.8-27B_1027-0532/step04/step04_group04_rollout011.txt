def compute_grad_smoothed(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    exp_corr = np.exp(alpha * (corr - c_max))
    denom = np.sum(exp_corr)
    weights = exp_corr / denom  # length 2n-1
    
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:
            continue
        s = m - (n - 1)  # actual shift
        # dO(s)/dh[j] = (1-h[j+s])*1[0<=j+s<n] - h[j-s]*1[0<=j-s<n]
        # For term 1: (1-h[j+s]) for valid j
        #   j+s valid means 0 <= j+s < n, i.e., -s <= j < n-s
        #   j range: max(0, -s) to min(n-1, n-1-s)
        lo1 = max(0, -s)
        hi1 = min(n-1, n-1-s)
        if lo1 <= hi1:

        hi2 = min(n-1, n+s-1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2-s:hi2-s+1]
    
    grad *= dx
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

def smooth_obj_lr(h, alpha, target_sum, n):
    """Smoothed max objective with log-sum-exp, using line search."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Use log-sum-exp for smoothness
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    return val

def smooth_grad(h, alpha, target_sum, n):
    """Gradient of the smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    exp_corr = np.exp(alpha * (corr - c_max))
    denom = float(np.sum(exp_corr))
    W = exp_corr / denom
    
    grad = np.zeros(n)
    for m in range(len(W)):
        w = W[m]
        if w < 1e-14:
            continue
        s = m - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n - 1, n - 1 - s)

        hi2 = min(n - 1, n + s - 1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2-s:hi2-s+1]
    
    grad *= dx
    return grad

def lbfgs_continuous(h_start, n, target_sum, alphas, maxiter_per_alpha, time_limit=None, start_time=None):
    """Run L-BFGS-B with progressively increasing alpha."""
    h_cur = h_start.copy()
    curr_score = smooth_obj_lr(h_cur, alphas[0], target_sum, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 
                    'jac': lambda h: np.ones(n)}]
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(
                smooth_obj_lr, h_cur,
                args=(alpha, target_sum, n),
                jac=lambda h: smooth_grad(h, alpha, target_sum, n),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
            )
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

            for x in grid:

                if val < best_val:
                    best_val = val
                    best_x = x
            
            h[i] = best_x
            if best_x != h[i]:
                improved = True
        
        # Project to feasible set
        h = project_feasible(h, target_sum)
        
        if not improved:
            break
    
    return h

def coord_descent_1d(h, alpha, target_sum, n, max_sweeps, time_limit=None, start_time=None):
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        for i in range(n):
            # Compute the gradient components for coordinate i
            # g[s] = (1-h[i+s])*1[valid] - h[i-s]*1[valid]
            g = np.zeros(2 * n - 1)
            h_old = h[i]
            
            # Term 1: (1-h[i+s]) for valid i+s
            s_range = range(-(i), n - 1 - i + 1)  # s values where i+s is valid
            for s in s_range:
                m = s + (n - 1)
                g[m] += (1.0 - h[i + s])
            
            # Term 2: -h[i-s] for valid i-s
            s_range2 = range(i - (n - 1), i + 1)  # s values where i-s is valid
            for s in s_range2:
                m = s + (n - 1)
                g[m] -= h[i - s]
            
            # Now the 1D function is: f(x) = smoothed_max(corr + (x-h_old) * g)
            # Use scipy.optimize.minimize_scalar to find the best x in [0, 1]
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            def obj_1d(x):
                new_corr = corr + (x - h_old) * g
                c_max = float(np.max(new_corr))
                shifted = np.clip(alpha * (new_corr - c_max), -500, 0)
                return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
            
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(obj_1d, bounds=(0, 1), method='bounded')
            if res.success:
                h[i] = res.x
        
        h = project_feasible(h, target_sum)
    
    return h
