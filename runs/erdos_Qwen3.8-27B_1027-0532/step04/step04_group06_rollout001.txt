def fft_corr(h):
    """FFT-based correlation, equivalent to np.correlate(h, 1-h, mode='full')"""
    n = len(h)
    size = 1
    while size < 2*n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    C = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(C), size)[:2*n-1]
    return corr

from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Fix if still not exact
    final_sum = h_r.sum()
    if final_sum > target_sum:
        extra = final_sum - target_sum
        h_r[np.where(h_r == 1)[0][:extra]] = 0
    elif final_sum < target_sum:
        extra = target_sum - final_sum
        h_r[np.where(h_r == 0)[0][:extra]] = 1
    return h_r.astype(float)

def efficient_local_search(h_start, target_sum, max_iters, rng, time_limit, start_time):
    """Local search that tries best-improvement swaps focused on worst lags."""
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = get_corr(h)
    center = n - 1
    
    stall = 0
    max_stall = 50
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        # Find worst lags
        c = corr.copy()
        c[center] = 0  # exclude center (lag 0)
        top_lags = np.argsort(c)[-3:][::-1] - center
        top_lags = top_lags[top_lags != 0]
        if len(top_lags) == 0:
            top_lags = np.array([1, -1])
        
        # For each top lag, find best swap
        best_delta = 0
        best_a, best_b = -1, -1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for k in top_lags:
            k = int(k)
            # Δc[k_lag] = h[a-k] + h[a+k] - h[b-k] - h[b+k]
            # We want to MINIMIZE the correlation, so we want Δc to be as negative as possible
            # for the worst lags. But we also need to not increase other lags too much.
            
            # For a swap (a: 1→0, b: 0→1), the change at lag k is:
            # Δc[k] = h[a-k] + h[a+k] - h[b-k] - h[b+k]
            # (with boundary handling: out-of-range terms are 0)
            
            # To minimize c at lag k, we want:
            # h[a-k] + h[a+k] small (a has few 1-nighbors at distance k)
            # h[b-k] + h[b+k] large (b has many 1-nighbors at distance k)
            
            a_k = ones - k
            a_pk = ones + k
            valid_a = (a_k >= 0) & (a_k < n) & (a_pk >= 0) & (a_pk < n)
            
            b_k = zeros - k
            b_pk = zeros + k
            valid_b = (b_k >= 0) & (b_k < n) & (b_pk >= 0) & (b_pk < n)
            
            if np.sum(valid_a) == 0 or np.sum(valid_b) == 0:
                continue
            
            a_valid = ones[valid_a]
            a_score = h[np.clip(a_k, 0, n-1)[valid_a]] + h[np.clip(a_pk, 0, n-1)[valid_a]]
            
            b_valid = zeros[valid_b]
            b_score = h[np.clip(b_k, 0, n-1)[valid_b]] + h[np.clip(b_pk, 0, n-1)[valid_b]]
            
            # Best a: minimize a_score (few 1s at distance k)
            # Best b: maximize b_score (many 1s at distance k)
            best_a_idx = a_valid[np.argmin(a_score)]
            best_b_idx = b_valid[np.argmax(b_score)]
            
            # Estimate the change: Δc[k] = a_score_min - b_score_max
            est_delta = a_score.min() - b_score.max()
            
            if est_delta < best_delta:
                best_delta = est_delta
                best_a = best_a_idx
                best_b = best_b_idx
        
        if best_a == -1:
            # Random swap
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a] = 0
            h[b] = 1
            new_corr = get_corr(h)
            new_c5 = float(np.max(new_corr)) * (2.0/n)
            if new_c5 > best_c5:
                h[a] = 1
                h[b] = 0
            else:
                h[a] = 0
                h[b] = 1
                corr = new_corr
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            continue
        
        # Try the best swap
        h[best_a] = 0
        h[best_b] = 1
        new_corr = get_corr(h)
        new_c5 = float(np.max(new_corr)) * (2.0/n)
        
        if new_c5 <= best_c5 + 1e-12:  # accept if not worse
            corr = new_corr
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[best_a] = 1
            h[best_b] = 0
            stall += 1
    
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def find_worst_lags(corr, n, top_k=5):
    c = corr.copy()
    c[n-1] = 0
    idx = np.argsort(c)[-top_k:][::-1]
    lags = idx - (n-1)
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1, -1])
    return lags

    zeros = np.where(h == 0)[0]

    max_stall = max(100, max_iters // 30)

        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            # Restart from best
            h = best_h.copy()
            current_c5 = best_c5
            corr = get_corr(h)

            zeros = np.where(h == 0)[0]
            stall = 0

            for _ in range(rng.randint(2, 8)):
                if len(ones) > 0 and len(zeros) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    h[a] = 0
                    h[b] = 1
            corr = get_corr(h)
            current_c5 = float(np.max(corr)) * dx
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            continue

        t = t_init * (t_final / t_init) ** ratio

        moved = False
        
        if r < 0.55 and len(ones) > 0 and len(zeros) > 0:

            if k == 0:

            valid_a = (a_k >= 0) & (a_k < n) & (a_pk >= 0) & (a_pk < n)
            b_k = zeros - k
            b_pk = zeros + k
            valid_b = (b_k >= 0) & (b_k < n) & (b_pk >= 0) & (b_pk < n)
            
            if np.sum(valid_a) > 0 and np.sum(valid_b) > 0:

                b_valid = zeros[valid_b]

                b_score = b_score + rng.random(len(b_score)) * 0.3

                b_idx = b_valid[np.argmax(b_score)]

Actually wait - the objective is max_k ∑ h[i](1-h[i+k]) which is NOT linear in h. Let me reconsider.

∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx

The first term is constant (= 1). The second term ∫ h(x)h(x+k) dx is quadratic in h (it's a bilinear form). So the objective is:
max_k (1 - B_k(h,h)) where B_k(h,h) = ∫ h(x)h(x+k) dx.

Since B_k is a quadratic form (specifically, h^T Q_k h where Q_k is the shift matrix), the objective is:
max_k (1 - h^T Q_k h)

This is the maximum of concave quadratic functions, which is NOT convex in general. So the problem is non-convex, and binary solutions may not be optimal.

However, in practice, for this type of problem, the optimal solution is often close to binary. The SLSQP approach can find non-binary solutions that are slightly better.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, k=200):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(k * (corr - c_max), -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / k) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]

def slsqp_obj(h_in, alpha, target):
    corr = np.correlate(h_in, 1.0 - h_in, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_cons_f(h_in, target):
    return np.sum(h_in) - target

def slsqp_cons_j(h_in, target):
    return np.ones(len(h_in))

def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_f, 'jac': slsqp_cons_j, 'args': (target,)}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
