Hmm, actually I realize the "omega" trick from the LSE literature might help. Let me look at this more carefully.

The standard approach for minimizing max of functions is to use the "smooth maximum" with the omega parameter:

f_alpha(x) = (1/alpha) * log(sum exp(alpha * f_i(x)))

But a better variant is:
f_alpha(x) = max_i f_i(x) + (1/alpha) * log(sum exp(alpha * (f_i(x) - max_j f_j(x))))

This has the property that as alpha → ∞, f_alpha → max f_i, and the gradient is well-behaved.

The "omega" trick modifies this to:
f_omega = (1 - omega) * max_i f_i + omega * (1/alpha) * log(sum exp(alpha * f_i))

where omega = 1 / (1 + log(sum exp(alpha * (f_i - max))))

This makes the function closer to the max when the max is well-separated, and closer to the average when they're close.

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    n = len(h)
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15:
            return np.clip(h + diff / n, 0.0, 1.0)
        h += diff * room / total
    else:
        mass = h.copy()
        total = mass.sum()
        if total < 1e-15:
            return np.clip(h + diff / n, 0.0, 1.0)
        h += diff * mass / total
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk is not None and topk < n_corr:
        part = np.argpartition(corr, -topk)[-topk:]
        cu = corr[part]
    else:
        cu = corr
        part = np.arange(n_corr)
    c_max = cu.max()
    sh = np.clip(alpha * (cu - c_max), -500.0, 0.0)
    ev = np.exp(sh)
    te = ev.sum()
    obj = (c_max + np.log(te) / alpha) * dx
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(part)):
        w = ev[i] / te
        if w < 1e-15:
            continue
        lag = part[i] - (n - 1)
        if lag >= 0 and lag < n:
            grad[lag:] += w * omh[:n - lag]
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0:
            m = -lag
            if m < n:
                grad[m:] += w * omh[:n - m]
                grad[:n - m] -= w * h[m:]
