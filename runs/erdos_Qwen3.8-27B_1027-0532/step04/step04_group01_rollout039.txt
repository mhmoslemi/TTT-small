    """Compute delta in correlation array when flipping h[j] (0→1)."""
    delta = np.zeros(2*n - 1)
    old = h[j]
    sign = 1.0 if old == 0 else -1.0  # 0→1 is +1, 1→0 is -1
    
    # Term 1: for L in [max(j-n+1, -(n-1)), min(j, n-1)], L != 0
    # delta[L+n-1] += sign * (1-h[j-L])
    L_lo = max(j - n + 1, -(n-1))
    L_hi = min(j, n-1)
    if L_lo <= L_hi:
        L_range = np.arange(L_lo, L_hi + 1)
        idx_range = L_range + n - 1
        j_minus_L = j - L_range
        mask = (j_minus_L >= 0) & (j_minus_L < n) & (L_range != 0)
        delta[idx_range[mask]] += sign * (1.0 - h[j_minus_L[mask]])
    
    # Term 2: for L in [max(-j, -(n-1)), min(n-j-1, n-1)], L != 0
    # delta[L+n-1] += sign * (-h[j+L])
    L_lo2 = max(-j, -(n-1))
    L_hi2 = min(n - j - 1, n-1)
    if L_lo2 <= L_hi2:
        L_range2 = np.arange(L_lo2, L_hi2 + 1)
        idx_range2 = L_range2 + n - 1
        j_plus_L = j + L_range2
        mask2 = (j_plus_L >= 0) & (j_plus_L < n) & (L_range2 != 0)

Actually, let me think about this differently. The correlation array is:
corr[i] = sum_n h[n + L_i] * (1-h[n]) where L_i = i - (n-1)

If I flip h[j], the change in corr is:
- corr[i] changes by: (new_h[j] - old_h[j]) * [contribution of h[j] to corr[i]]

The contribution of h[j] to corr[i] (i.e., the coefficient of h[j] in the expression for corr[i]):
- h[j] appears as h[n+L] when n+L = j, i.e., n = j-L. The coefficient is (1-h[n]) = (1-h[j-L]).
- h[j] appears as (1-h[n]) when n = j. The coefficient is -h[n+L] = -h[j+L].

So the coefficient of h[j] in corr[i] is:
coef[i, j] = (1-h[j-L_i]) * [0 ≤ j-L_i < n] - h[j+L_i] * [0 ≤ j+L_i < n]

(with the L=0 correction: when L_i=0, both terms refer to the same position, and the net coefficient is 0 for binary)

Wait, for the change:
Δcorr[i] = (new_h[j] - old_h[j]) * coef[i, j]

For binary 0→1: Δcorr[i] = coef[i, j]
For binary 1→0: Δcorr[i] = -coef[i, j]

So I need to compute coef[i, j] for all i, j. This is a (2n-1) × n matrix. For n=600, that's 1200 × 600 = 720,000 entries. Once computed, I can do steepest descent very efficiently:

For each j:
  new_corr = corr ± coef[:, j]  (± depending on flip direction)
  new_max = max(new_corr)

This is O(n) per j (just a max of a length-2n-1 array), so O(n²) total per step.

```python
def compute_coef_matrix(h, n):
    """Compute coef[i, j] = coefficient of h[j] in corr[i]."""
    # corr[i] = sum_n h[n + L_i] * (1-h[n]), L_i = i - (n-1)
    # coef[i, j] = derivative of corr[i] w.r.t. h[j]
    # = (1-h[j-L_i]) * [0 <= j-L_i < n] - h[j+L_i] * [0 <= j+L_i < n]
    #   with L_i = 0 correction (net zero for binary)
    
    coefs = np.zeros((2*n-1, n))
    lags = np.arange(2*n-1) - (n-1)  # L_i for each i
    
    for j in range(n):
        # Term 1: (1-h[j-L]) for valid L, where 0 <= j-L < n
        # j-L >= 0 → L <= j
        # j-L < n → L > j-n
        # L in [j-n+1, j] ∩ [-(n-1), n-1]
        valid1 = (lags >= j-n+1) & (lags <= j) & (j-lags >= 0) & (j-lags < n)
        # Term 2: -h[j+L] for valid L, where 0 <= j+L < n
        # j+L >= 0 → L >= -j
        # j+L < n → L < n-j
        valid2 = (lags >= -j) & (lags < n-j) & (j+lags >= 0) & (j+lags < n)
        
        coef_j = np.zeros(2*n-1)
        idx1 = j - lags[valid1]
        coef_j[valid1] += (1.0 - h[idx1])
        idx2 = j + lags[valid2]
        coef_j[valid2] -= h[idx2]
        
        # L=0 correction: at i = n-1, L=0
        # Both conditions are satisfied, and the net effect is 0 for binary
        # coef = (1-h[j]) - h[j] = 1-2h[j], but actual change is 0
        # So subtract (1-2*h[j]) at i=n-1
        coef_j[n-1] -= (1.0 - 2.0 * h[j])
        
        coefs[:, j] = coef_j
    
    return coefs
```

Wait, this is O(n²) to compute, which for n=600 is 360,000 operations. That's fine.

Alternative: at each step, compute the delta for all j simultaneously.

For a given state h, the delta for flipping position j (0→1) is:
delta_j[i] = (1-h[j-L_i]) * [valid1] - h[j+L_i] * [valid2] (with L=0 correction)

I can compute this in a vectorized way:

```python
def all_deltas_0to1(h, n):
    """Compute delta arrays for all positions j being flipped 0→1."""
    lags = np.arange(2*n-1) - (n-1)
    deltas = np.zeros((n, 2*n-1))
    
    for j in range(n):
        valid1 = (j - lags >= 0) & (j - lags < n) & (lags != 0)
        valid2 = (j + lags >= 0) & (j + lags < n) & (lags != 0)
        
        if np.any(valid1):
            deltas[j, valid1] += (1.0 - h[j - lags[valid1]])
        if np.any(valid2):
            deltas[j, valid2] -= h[j + lags[valid2]]
    
    return deltas
```

This is O(n²) per call. For n=600, that's 360,000 + overhead. In numpy, the inner operations are vectorized over lags (length 1199), so each j takes ~μs. Total: ~0.6 ms per full computation.

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros_idx = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros_idx])[:need]
        h_r[zeros_idx[order]] = 1
    else:
        need = current - target
        ones_idx = np.where(h_r == 1)[0]
        order = np.argsort(h[ones_idx])[:need]
        h_r[ones_idx[order]] = 0
    return h_r.astype(float)

def steep_descent(h_init, max_steps, time_limit, t_start):
    """Greedy steepest descent for binary h."""
    n = len(h_init)
    h = h_init.copy().astype(float)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    for step in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        
        # Compute deltas for all positions
        lags = np.arange(2*n-1) - (n-1)
        
        # For efficiency, compute new_max for each candidate flip
        zero_idx = np.where(h == 0)[0]
        one_idx = np.where(h == 1)[0]
        
        best_new_max = cur_max + 1e-15
        best_j = -1
        best_dir = 0  # 1 for 0→1, -1 for 1→0
        
        for j in zero_idx:
            # delta for 0→1 at position j
            v1 = (j - lags >= 0) & (j - lags < n) & (lags != 0)
            v2 = (j + lags >= 0) & (j + lags < n) & (lags != 0)
            delta = np.zeros(2*n-1)
            if np.any(v1):
                delta[v1] += (1.0 - h[j - lags[v1]])
            if np.any(v2):
                delta[v2] -= h[j + lags[v2]]
            
            new_max = np.max(corr + delta)
            if new_max < best_new_max:
                best_new_max = new_max
                best_j = j
                best_dir = 1
        
        for j in one_idx:
            # delta for 1→0 at position j (negative of 0→1 delta)
            v1 = (j - lags >= 0) & (j - lags < n) & (lags != 0)
            v2 = (j + lags >= 0) & (j + lags < n) & (lags != 0)
            delta = np.zeros(2*n-1)
            if np.any(v1):
                delta[v1] += (1.0 - h[j - lags[v1]])
            if np.any(v2):
                delta[v2] -= h[j + lags[v2]]
            
            new_max = np.max(corr - delta)
            if new_max < best_new_max:
                best_new_max = new_max
                best_j = j
                best_dir = -1
        
        if best_j == -1:
            break  # stuck
        
        # Apply flip
        h[best_j] = 1.0 - h[best_j]
        corr = compute_corr(h)  # recompute (O(n log n) with FFT or O(n²) direct)
        cur_max = float(np.max(corr))
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max

One more consideration: pair moves. When single moves get stuck, I want to try pairs. The delta for flipping both j and k is:
delta_pair = delta_j + delta_k + cross_term

The cross term is non-zero only when j and k are at a lag distance from each other (specifically, when |j-k| = |L| for some L where both contribute).

For the cross term: when we flip both j and k, the correlation at lag L changes by:
- delta_j at lag L (change due to flipping j)
- delta_k at lag L (change due to flipping k)
- cross term: this occurs when the "contribution" of j to corr[L] involves k, or vice versa.

Specifically, the contribution of h[j] to corr[L] includes (1-h[j-L]) and -h[j+L]. If j-L = k or j+L = k, then flipping k also affects this term.

    return np.correlate(h, 1.0 - h, mode='full')

    h_r = np.clip(h_r, 0, 1)

    if current < target:

        zeros_idx = np.where(h_r == 0)[0]

        order = np.argsort(h[ones_idx])[:need]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def steep_descent(h_init, max_steps, time_limit, t_start):
    n = len(h_init)
    h = h_init.copy().astype(np.float64)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    lags = np.arange(2*n-1) - (n-1)
    
    for step in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        
        zero_idx = np.where(h == 0)[0]
        one_idx = np.where(h == 1)[0]
        
        best_new_max = cur_max + 1e-15
        best_j = -1
        best_dir = 0
        
        # Try all 0→1 flips (vectorized)
        if len(zero_idx) > 0:
            J = zero_idx[:, None].astype(np.int64)
            L = lags[None, :].astype(np.int64)
            j_minus_L = J - L
            valid1 = (j_minus_L >= 0) & (j_minus_L < n) & (L != 0)
            j_plus_L = J + L
            valid2 = (j_plus_L >= 0) & (j_plus_L < n) & (L != 0)
            
            safe_idx1 = np.where(valid1, j_minus_L, 0)
            term1 = np.where(valid1, 1.0 - h[safe_idx1], 0.0)
            safe_idx2 = np.where(valid2, j_plus_L, 0)
            term2 = np.where(valid2, -h[safe_idx2], 0.0)
            deltas = term1 + term2
            
            new_maxes = np.max(corr[None, :] + deltas, axis=1)
            best_local = np.argmin(new_maxes)
            if new_maxes[best_local] < best_new_max:
                best_new_max = new_maxes[best_local]
                best_j = int(zero_idx[best_local])
                best_dir = 1
        
        # Try all 1→0 flips (vectorized)
        if len(one_idx) > 0:

            L = lags[None, :].astype(np.int64)
            j_minus_L = J - L
            valid1 = (j_minus_L >= 0) & (j_minus_L < n) & (L != 0)
            j_plus_L = J + L
            valid2 = (j_plus_L >= 0) & (j_plus_L < n) & (L != 0)

            term1 = np.where(valid1, 1.0 - h[safe_idx1], 0.0)

            term2 = np.where(valid2, -h[safe_idx2], 0.0)
            deltas = term1 + term2

            best_local = np.argmin(new_maxes)
            if new_maxes[best_local] < best_new_max:

                best_j = int(one_idx[best_local])

    h = h_init.copy().astype(np.float64)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    lags = np.arange(2*n-1) - (n-1)
    no_imp = 0
    total_iters = 0
    
    while time.time() - t_start < time_limit:
        total_iters += 1
        T = t_init * (t_final / t_init) ** (total_iters / max(total_iters, 1))
        # Actually, let's use a fixed schedule based on time or iterations
        # Use: T = t_final + (t_init - t_final) * exp(-total_iters / (time_limit * 50))
        T = t_final + (t_init - t_final) * np.exp(-total_iters / max(1, float(int((time_limit) * 100))))

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros_idx = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros_idx])[:need]
        h_r[zeros_idx[order]] = 1
    else:
        need = current - target
        ones_idx = np.where(h_r == 1)[0]
        order = np.argsort(h[ones_idx])[:need]
        h_r[ones_idx[order]] = 0
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def steep_descent(h_init, max_steps, time_limit, t_start):
    n = len(h_init)
    h = h_init.copy().astype(np.float64)
    corr = compute_corr(h).astype(np.float64)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    lags = np.arange(2 * n - 1) - (n - 1)

    for step in range(max_steps):
        if time.time() - t_start > time_limit:
            break

        zero_idx = np.where(h == 0)[0]
        one_idx = np.where(h == 1)[0]

        best_new_max = cur_max + 1e-15
        best_j = -1
        best_dir = 0

        if len(zero_idx) > 0:
            J = zero_idx[:, None].astype(np.int64)
            L = lags[None, :].astype(np.int64)
            j_minus_L = J - L
            valid1 = (j_minus_L >= 0) & (j_minus_L < n) & (L != 0)
            j_plus_L = J + L
            valid2 = (j_plus_L >= 0) & (j_plus_L < n) & (L != 0)
            safe_idx1 = np.where(valid1, j_minus_L, 0)
            term1 = np.where(valid1, 1.0 - h[safe_idx1], 0.0)
            safe_idx2 = np.where(valid2, j_plus_L, 0)
            term2 = np.where(valid2, -h[safe_idx2], 0.0)
            deltas = term1 + term2
            new_maxes = np.max(corr[None, :] + deltas, axis=1)
            best_local = int(np.argmin(new_maxes))
            if new_maxes[best_local] < best_new_max:
                best_new_max = new_maxes[best_local]
                best_j = int(zero_idx[best_local])
                best_dir = 1

        if len(one_idx) > 0:
            J = one_idx[:, None].astype(np.int64)
            L = lags[None, :].astype(np.int64)
            j_minus_L = J - L
            valid1 = (j_minus_L >= 0) & (j_minus_L < n) & (L != 0)
            j_plus_L = J + L
            valid2 = (j_plus_L >= 0) & (j_plus_L < n) & (L != 0)
            safe_idx1 = np.where(valid1, j_minus_L, 0)
            term1 = np.where(valid1, 1.0 - h[safe_idx1], 0.0)
            safe_idx2 = np.where(valid2, j_plus_L, 0)
            term2 = np.where(valid2, -h[safe_idx2], 0.0)

            new_maxes = np.max(corr[None, :] - deltas, axis=1)

def is_prime(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True


def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while not is_prime(p):
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    return project_binary(h, target)


def make_msequence(n, rng):
    m = max(2, int(np.log2(n)))
    while True:
        poly = rng.randint(1 << m, 1 << (m + 1))
        if is_prime(1 << m) or True:
            state = 1
            seq = []
            for _ in range(n * 4):
                seq.append(state & 1)
                feedback = 0
                for i in range(m):
                    if (poly >> i) & 1:
                        feedback ^= (state >> i) & 1
                feedback = feedback & 1
                state = (state >> 1) | (feedback << (m - 1))
            if len(set(seq[:n])) == 2:
                break
        m += 1
        if m > 16:
            return project_binary(rng.random(n), n // 2)
    h = np.array(seq[:n], dtype=float)
    return project_binary(h, n // 2)


def make_random_binary(n, rng):
    h = rng.random(n)
    return project_binary(h, n // 2)


def smooth_objective(x, alpha, n, target):
    dx = 2.0 / n
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [15, 40, 100, 250, 600, 1200]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                lambda x: smooth_objective(x, alpha, n, target),
                h_cur, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score
