    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_gradient(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    total_w = weights.sum()
    if total_w < 1e-300:
        return np.zeros(len(h))
    weights /= total_w
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        w = weights[n-1+k]
        if w < 1e-12:
            continue
        if n - k > 0:
            grad[:n-k] += w * (1.0 - h[k:k+n-k])
        if k < n:
            grad[k:] -= w * h[:n-k]
    return grad

def nesterov_gradient(h, n, target, alpha, n_steps, lr_init, deadline):
    """Nesterov accelerated projected gradient descent on LSE objective."""
    h_cur = h.copy()
    m = np.zeros(n)
    t = 1.0
    lr = lr_init
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Extrapolate
        beta = (t - 1) / (t + 2)
        y = h_cur + beta * m
        
        # Gradient at y
        grad = lse_gradient(y, alpha)
        
        # Step
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Update momentum
        m = h_new - h_cur
        h_cur = h_new
        t += 1
        
        # Adaptive learning rate
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
        
        # Decay alpha schedule is handled externally
    return best_h, best_score

def slsqp_optimize(h, n, target, alpha, maxiter, ftol):
    """SLSQP on LSE objective."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return lse_objective(x, alpha)
    
    def obj_grad(x):
        return lse_gradient(x, alpha)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def active_set_gradient(h, n, target, n_active, n_steps, lr_init, deadline):
    """Gradient descent focused on the top n_active lags."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        # Get top n_active lags
        top_idx = np.argsort(over)[::-1][:n_active]
        # Compute weighted gradient
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if n - k > 0:
                grad[:n-k] += wk * (1.0 - h_cur[k:k+n-k])
                grad[k:] -= wk * h_cur[:n-k]
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.01, 0.3)
        else:
            lr *= 0.95
            if lr < 1e-9:
                break
    
    return best_h, best_score

def fourier_init(n, seed, n_terms=15):
    """Generate a Fourier-based initial solution."""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_terms+1):
        a = rng.randn() * 0.15 / np.sqrt(k)
        b = rng.randn() * 0.15 / np.sqrt(k)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_perturb(h, n, target, rng, scale=0.1):
    """Apply a structured perturbation (block shifts, frequency changes)."""
    h_new = h.copy()
    method = rng.randint(4)
    if method == 0:
        # Random block shift
        bs = rng.randint(5, n//10)
        start = rng.randint(0, n - bs)
        shift = rng.randint(-n//5, n//5 + 1)
        if shift > 0:
            h_new[start:start+bs] = h_new[start+shift:start+bs+shift]
            h_new[start+bs+shift-start:start+bs+shift] = h_new[start:start+shift]
        else:
            h_new[start+shift:start+bs+shift] = h_new[start:start+bs]
    elif method == 1:
        # Frequency perturbation
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.uniform(1, 5)
        amp = scale * 0.3
        h_new += amp * np.cos(np.pi * freq * x + rng.uniform(0, 2*np.pi))
    elif method == 2:
        # Local bump
        center = rng.randint(0, n)
        width = rng.randint(10, n//5)
        amp = rng.uniform(-0.2, 0.2)
        for i in range(n):
            d = abs(i - center)
            if d < width:
                h_new[i] += amp * np.cos(np.pi * d / width)
    else:
        # Gaussian noise
        h_new += rng.randn(n) * scale
    return project_feasible(h_new, target)

def fourier_init(n, seed, n_terms=20):

        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)

def structured_perturb(h, n, target, rng, scale=0.1):
    h_new = h.copy()
    method = rng.randint(5)
    if method == 0:
        bs = rng.randint(5, max(6, n // 10))
        start = rng.randint(0, n - bs - 1)
        shift = rng.randint(-n // 5, n // 5)
        if shift > 0:
            h_new[start+shift:start+bs+shift] = h_new[start:start+bs]
        elif shift < 0:
            h_new[start:start+bs+shift] = h_new[start-shift:start+bs]
    elif method == 1:
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.uniform(1, 6)
        amp = scale * 0.3
        phase = rng.uniform(0, 2 * np.pi)
        h_new += amp * np.cos(np.pi * freq * x + phase)
    elif method == 2:
        center = rng.randint(0, n)
        width = rng.randint(10, max(11, n // 5))
        amp = rng.uniform(-0.15, 0.15)
        for i in range(n):
            d = abs(i - center)
            if d < width:
                h_new[i] += amp * np.cos(np.pi * d / (2.0 * width))
    elif method == 3:
        # Two-point swap perturbation
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i != j:
            swap = rng.uniform(0.05, scale)
            h_new[i] += swap
            h_new[j] -= swap
    else:
        h_new += rng.randn(n) * scale * 0.5
    return project_feasible(h_new, target)

```python
def cutting_plane_step(h, n, target, n_active, step_scale):
    """One cutting plane step: linearize top constraints and solve LP."""
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    top_idx = np.argsort(over)[::-1][:n_active]
    c_max = over[top_idx[0]]
    
    # Build LP: min C s.t. a_i' h + C >= b_i for i in active, sum(h)=n/2, 0<=h<=1
    # Variables: [h(0), ..., h(n-1), C]  -> total n+1 variables
    # Minimize: C  -> c = [0, 0, ..., 0, 1]
    # Constraints:
    #   For each active k: -sum_j (df_k/df_h[j]) * h[j] + C >= -f_k(h*)
    #   Wait, I want f_k_approx(h) <= C
    #   f_k_approx(h) = f_k(h*) + grad_k'(h-h*) <= C
    #   grad_k'h + (f_k(h*) - grad_k'h*) <= C
    #   -grad_k'h + C >= f_k(h*) - grad_k'h*
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize C
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: grad_k' h - C <= f_k(h*) - grad_k' h*
    A_ub = []
    b_ub = []
    for k in top_idx:
        k = int(k)
        nk = n - k
        gk = np.zeros(n)
        gk[:nk] = (1.0 - h[k:k+nk])
        gk[k:] -= h[:nk]
        rhs = over[k] - gk @ h  # = f_k(h*) - gk' h*  (should be ~0)
        row = np.zeros(n + 1)
        row[:n] = gk
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(rhs)
    
    # Sum constraint: sum(h) = n/2 -> handled separately or as equality
    # For linprog, use A_eq, b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds_lp = [(0, 1)] * n + [(0, None)]  # h in [0,1], C >= 0
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
    
    if res.status == 0:
        h_lp = res.x[:n]
        # Take a step from h toward h_lp
        h_new = h + step_scale * (h_lp - h)
        h_new = project_feasible(h_new, target)
        return h_new
    return None
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_target = 0.0
_g_alpha = 0.0
_g_n = 0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _lse_grad(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(_g_n)
    weights /= total
    n = _g_n
    grad = np.zeros(n)
    for k in range(n):
        w = weights[n - 1 + k]
        if w < 1e-12:
            continue
        nk = n - k
        grad[:nk] += w * (1.0 - x[k:k + nk])
        grad[k:] -= w * x[:nk]
    return grad


def _cons_eq_fun(x):
    return np.sum(x) - _g_target


def _cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h, n, target, alpha, maxiter, ftol):
    global _g_target, _g_alpha, _g_n
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def nesterov_gradient(h, n, target, alpha, n_steps, lr_init, deadline):
    global _g_target, _g_alpha, _g_n
    _g_n = n
    _g_target = target
    _g_alpha = alpha

    h_cur = h.copy()
    m = np.zeros(n)
    t = 1.0
    lr = lr_init
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

    for step in range(n_steps):
        if time.time() > deadline:
            break
        beta = (t - 1) / (t + 2)
        y = h_cur + beta * m
        grad = _lse_grad(y)
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        m = h_new - h_cur
        h_cur = h_new
        t += 1

        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.93
            if lr < 1e-8:
                break

def active_set_gradient(h, n, target, n_active, n_steps, lr_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            nk = n - k
            grad[:nk] += wk * (1.0 - h_cur[k:k + nk])
            grad[k:] -= wk * h_cur[:nk]

        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.005, 0.3)
        else:
            lr *= 0.97
            if lr < 1e-10:
                break

def cutting_plane_step(h, n, target, n_active, step_scale, deadline):
    corr = compute_corr(h)
    over = corr[n - 1:n - 1 + n]
    top_idx = np.argsort(over)[::-1][:n_active]
    c_max_val = over[top_idx[0]]

    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0

    A_ub = []
    b_ub = []
    for k in top_idx:
        k = int(k)
        nk = n - k
        gk = np.zeros(n)
        gk[:nk] = (1.0 - h[k:k + nk])
        gk[k:] -= h[:nk]
        rhs = over[k] - gk @ h
        row = np.zeros(n + 1)
        row[:n] = gk
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(rhs)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    bounds_lp = [(0, 1)] * n + [(0, None)]

    try:
        res = linprog(c_vec, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
        if res.status == 0:
            h_lp = res.x[:n]
            h_new = h + step_scale * (h_lp - h)
            h_new = project_feasible(h_new, target)
            return h_new
    except Exception:
        pass
    return None


def cutting_plane_loop(h, n, target, n_active, n_steps, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

    for step in range(n_steps):
        if time.time() > deadline:
            break
        found = False
        for scale in [0.5, 0.3, 0.15, 0.07, 0.03]:
            h_new = cutting_plane_step(h_cur, n, target, n_active, scale, deadline)

                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_new.copy()
                    h_cur = h_new
                    found = True
                    break

def fourier_init(n, seed, n_terms=20):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_terms + 1):
        a = rng.randn() * 0.12 / np.sqrt(k)
        b = rng.randn() * 0.12 / np.sqrt(k)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, rng, scale=0.1):
    h_new = h.copy()

        if shift > 0:
            h_new[start + shift:start + bs + shift] = h_new[start:start + bs]
        elif shift < 0:
            h_new[start:start + bs + shift] = h_new[start - shift:start + bs]
    elif method == 1:

            h_new[j] -= swap_amt
