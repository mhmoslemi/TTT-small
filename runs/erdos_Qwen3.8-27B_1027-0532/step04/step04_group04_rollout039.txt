def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    f = corr * dx
    f_max = f.max()
    shifted = alpha * (f - f_max)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    p = exp_shifted / S
    F_val = f_max + np.log(S) / alpha
    one_minus_h = 1.0 - h

            a_max = max(a_max, (u[i] - 1.0) / max(diff_u[i], 1e-15))
            a_min = min(a_min, (v[i] - 1.0) / max(diff_v[i], 1e-15))
    a = max(a_min, min(a_max, 1.0))
    a = max(0.0, min(1.0, a))
    h_new = a * u + (1.0 - a) * h_curr
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def pen_fista(h_init, target, alphas, iters_per_alpha, time_limit, start_time):
    h = project_box_sum(h_init.copy(), target)
    n = len(h)
    best_score = compute_c5(h)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        L = 2.0 + alpha
        h_curr = h.copy()
        h_prev = h.copy()
        t_acc = 1.0
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc)) * 0.5
            y = h_curr + ((t_acc - 1.0) / t_new) * (h_curr - h_prev)
            h_prev = h_curr.copy()
            F_y, grad_y = smooth_max_and_grad(y, alpha)
            h_new = lyapunov_step(h_curr, h_prev, grad_y, alpha, L)
            h_curr = h_new
            t_acc = t_new
            if it % 100 == 0:
                sc = compute_c5(h_curr)
                if sc < best_score:
                    best_score = sc
                    best_h = h_curr.copy()
        sc = compute_c5(h_curr)
        if sc < best_score:
            best_score = sc
            best_h = h_curr.copy()
        h = h_curr.copy()
    return best_h, best_score

def fista_with_sum_penalty(h_init, target, alphas_f, pen_coef, iters_per_alpha, time_limit, start_time):
    h = h_init.copy()
    n = len(h)
    best_score = compute_c5(h)
    best_h = h.copy()
    for alpha in alphas_f:
        if time.time() - start_time > time_limit:
            break
        L = 4.0 + alpha * 0.3
        step = 1.0 / L
        h_curr = h.copy()
        h_prev = h.copy()
        t_acc = 1.0
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc)) * 0.5
            y = h_curr + ((t_acc - 1.0) / t_new) * (h_curr - h_prev)
            h_prev = h_curr.copy()
            F_y, grad_f = smooth_max_and_grad(y, alpha)
            sum_dev = np.sum(y) - target
            grad = grad_f + pen_coef * sum_dev * np.ones(n)
            h_prop = y - step * grad
            h_curr = project_box_sum(h_prop, target)
            t_acc = t_new
            if it % 200 == 0:
                sc = compute_c5(h_curr)
                if sc < best_score:
                    best_score = sc
                    best_h = h_curr.copy()
        sc = compute_c5(h_curr)
        if sc < best_score:
            best_score = sc
            best_h = h_curr.copy()
        h = h_curr.copy()
    return best_h, best_score

def binary_local_search(h, target, rng, time_limit, start_time, max_stall=600):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    tabu = {}
    tabu_len = 80
    iteration = 0
    while time.time() - start_time < time_limit:
        iteration += 1
        if stall > max_stall:
            n_pert = min(12, 2 + stall // 80)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = compute_corr(h)
            stall = 0
            continue
        top_k = 3
        top_idx = np.argsort(corr)[-top_k:][::-1]
        max_val = float(corr[top_idx[0]])
        k_star = int(top_idx[0]) - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        found = False
        if len(i_cand) > 0 and len(j_cand) > 0:
            n_try = min(300, len(i_cand) * len(j_cand))
            for _ in range(n_try):
                i = i_cand[rng.randint(len(i_cand))]
                j = j_cand[rng.randint(len(j_cand))]
                if tabu.get((i, j), -9999) >= iteration - tabu_len:
                    continue
                if tabu.get((j, i), -9999) >= iteration - tabu_len:
                    continue
                h[i] = 0
                h[j] = 1
                new_corr = compute_corr(h)
                new_max = float(new_corr.max())
                if new_max < max_val or new_max * (2.0 / n) < best_c5:
                    tabu[(i, j)] = iteration
                    corr = new_corr
                    if new_max * (2.0 / n) < best_c5:
                        best_c5 = new_max * (2.0 / n)
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                    found = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        if not found:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            if corr.max() * (2.0 / n) < best_c5:
                best_c5 = float(corr.max()) * (2.0 / n)
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
                h[i] = 1
                h[j] = 0
    return best_h, best_c5

def sa_binary(h, target, rng, max_iters, t_init, t_final, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    while time.time() - start_time < time_limit:
        ratio = (time.time() - start_time) / max(time_limit, 1e-6)
        t = t_init * (t_final / t_init) ** min(ratio, 1.0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        move_type = rng.randint(5)
        h_new = h.copy()
        moved = False
        if move_type <= 2:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 3:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        if not moved:
            continue
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-14 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    if current > target_sum:
        need = current - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final = int(h_r.sum())
    if final != target_sum:
        if final > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final]] = 1
    return h_r.astype(float)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0
    target_int = int(n) / 2

    h_feas = project_box_sum(h_init, target)
    best_h = h_feas.copy()
    best_score = compute_c5(h_feas)

    h_bin = project_binary(h_feas, target_int)
    h_dbg, sc_dbg = binary_local_search(h_bin, target_int, rng, min(30.0, tleft() * 0.15), start_time, max_stall=400)
    if sc_dbg < best_score:
        best_score = sc_dbg
        best_h = h_dbg.copy()

    if tleft() > 60:
        alphas = [5.0, 15.0, 40.0, 100.0, 300.0, 800.0, 2000.0]
        h_cont, sc_cont = fista_optimize(h_dbg, target, alphas, 2000, 2.0, min(tleft() * 0.35, 300.0), start_time)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()

    if tleft() > 40:
        h_bin2 = project_binary(best_h, target_int)
        h_bl2, sc_bl2 = binary_local_search(h_bin2, target_int, rng, min(40.0, tleft() * 0.2), start_time, max_stall=500)
        if sc_bl2 < best_score:
            best_score = sc_bl2
            best_h = h_bl2.copy()

    if tleft() > 50:
        alphas2 = [50.0, 200.0, 500.0, 1500.0, 5000.0]
        h_c2, sc_c2 = fista_optimize(h_best_h if 'h_best_h' in dir() else best_h, target, alphas2, 1500, 4.0, min(tleft() * 0.25, 250.0), start_time)
        if sc_c2 < best_score:
            best_score = sc_c2
            best_h = h_c2.copy()

    if tleft() > 30:
        h_bin3 = project_binary(best_h, target_int)
        h_sa, sc_sa = sa_binary(h_bin3, target_int, rng, 100000, 0.01, 0.0001, min(30.0, tleft() * 0.3), start_time)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()

    if tleft() > 40:
        for pert_round in range(3):
            if tleft() < 20:
                break
            h_pert = best_h.copy()
            n_pert = 5 + pert_round * 3
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            h_block = project_box_sum(h_pert, target)
            h_bl3, sc_bl3 = binary_local_search(project_binary(h_block, target_int), target_int, rng,
                                                min(15.0, tleft() * 0.15), start_time, max_stall=300)
            if sc_bl3 < best_score:
                best_score = sc_bl3
                best_h = h_bl3.copy()
            alphas3 = [100.0, 500.0, 2000.0]
            h_c3, sc_c3 = fista_optimize(h_block, target, alphas3, 1000, 4.0, min(15.0, tleft() * 0.15), start_time)
            if sc_c3 < best_score:
                best_score = sc_c3
                best_h = h_c3.copy()

    if tleft() > 30:
        alphas4 = [200.0, 1000.0, 3000.0, 8000.0]
        h_c4, sc_c4 = fista_optimize(best_h, target, alphas4, 2000, 6.0, min(tleft() * 0.4, 300.0), start_time)
        if sc_c4 < best_score:
            best_score = sc_c4
            best_h = h_c4.copy()

    if tleft() > 15:
        h_bin_final = project_binary(best_h, target_int)
        h_sa_f, sc_sa_f = sa_binary(h_bin_final, target_int, rng, 50000, 0.005, 0.00003,
                                    max(5.0, tleft() * 0.5), start_time)
        if sc_sa_f < best_score:
            best_score = sc_sa_f
            best_h = h_sa_f.copy()

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
