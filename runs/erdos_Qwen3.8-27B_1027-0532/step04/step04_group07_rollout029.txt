For the 1D minimization: we want to find t ∈ [0,1] that minimizes max_m (c[m] + Δc[m]).

import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    p = lag_idx - (n - 1)
    grad = np.zeros(n)
    if p >= 0:
        if p < n:
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] -= h[p:]
    else:
        m = -p
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        p = lag_indices[i] - (n - 1)
        if p >= 0:
            if p < n:
                grad[p:] += w * (1.0 - h[:n - p])
                grad[:n - p] -= w * h[p:]
        else:
            m = -p
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smooth_max_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)


def slsqp_solve(h_start, target_sum, alpha_topk_pairs, maxiter, ftol, bounds_cache=None):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                smooth_max_obj, h_cur, args=(alpha, topk),
                jac=smooth_max_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,

    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.02, 5.0)
        else:
            stagnation += 1
            lr *= 0.985
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score

            else:
                h_new2 = project_feasible(h + rng.randn(n) * lr, target_sum)
                score2 = compute_c5(h_new2)
                if score2 < best_score - 1e-15:
                    best_score = score2
                    best_h = h_new2.copy()
                    h = h_new2
                    stagnation = 0
                    lr *= 1.1
                else:
                    stagnation += 1
                    lr *= 0.9
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def cyclic_coord_descent(h_init, target_sum, n_sweeps, n_grid, time_limit=None):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    grid = np.linspace(0.0, 1.0, n_grid)

    for sweep in range(n_sweeps):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        for i in range(n):
            old_val = h[i]
            p_offsets = np.arange(n_corr) - (n - 1)
            idx1 = i - p_offsets
            idx2 = i + p_offsets
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            h1 = np.where(valid1, h[idx1[np.maximum(idx1, 0) & (idx1 < n)] if np.any(valid1) else np.zeros(0)], 0.0)
            h2 = np.where(valid2, h[idx2[np.maximum(idx2, 0) & (idx2 < n)] if np.any(valid2) else np.zeros(0)], 0.0)
            if np.any(valid1):
                h_vals_1 = h[np.clip(idx1, 0, n - 1)]
                h_vals_1 = np.where(valid1, h_vals_1, 0.0)
            else:
                h_vals_1 = np.zeros(n_corr)
            if np.any(valid2):
                h_vals_2 = h[np.clip(idx2, 0, n - 1)]
                h_vals_2 = np.where(valid2, h_vals_2, 0.0)
            else:
                h_vals_2 = np.zeros(n_corr)
            coef = (1.0 - h_vals_1) * valid1 - h_vals_2 * valid2
            corr = overlap_corrs(h)
            d_vals = corr - old_val * coef
            best_t = old_val
            best_cmax = np.max(corr)
            for t in grid:
                if t == old_val:
                    continue
                new_cmax = np.max(d_vals + t * coef)
                if new_cmax < best_cmax - 1e-14:
                    best_cmax = new_cmax
                    best_t = t
            h[i] = best_t
        h = project_feasible(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
