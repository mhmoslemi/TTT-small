    #   - pos is the second index: need pos-k in [0,n) → k in (pos-n+1, pos]

        const = 0.0

        if 0 <= pos + k < n:

                pass

    best_idx = np.argmin(g)

_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h2, 0.0, 1.0)
    return h


def coord_descent(h_init, n, target, max_sweeps, time_limit, grid_size=100, start_time=None):
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    ks = np.arange(-(n - 1), n)
    k0 = n - 1
    t_grid = np.linspace(0.0, 1.0, grid_size)
    mask_k0 = np.ones(2 * n - 1, dtype=bool)
    mask_k0[k0] = False
    pos_order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        np.random.shuffle(pos_order)
        for pos in pos_order:
            pk = pos + ks
            mk = pos - ks
            valid_p = (pk >= 0) & (pk < n)
            valid_m = (mk >= 0) & (mk < n)
            c1 = np.zeros(2 * n - 1)
            c2 = np.zeros(2 * n - 1)
            c1[valid_p] = 1.0 - h[pk[valid_p]]
            c2[valid_m] = h[mk[valid_m]]
            d = (c1 - c2) * dx
            d[k0] = 0.0
            C = corr - h[pos] * d
            C[k0] = corr[k0] - h[pos] * (1.0 - h[pos]) * dx
            f0 = C[k0] + t_grid * dx - t_grid**2 * dx
            f_lin = C[mask_k0][:, None] + d[mask_k0][:, None] * t_grid[None, :]
            g_vals = np.maximum(f0[None, :], f_lin).max(axis=0)
            new_val = t_grid[np.argmin(g_vals)]
            old_val = h[pos]
            if abs(new_val - old_val) > 1e-10:
                delta = new_val - old_val
                h[pos] = new_val
                corr[mask_k0] += delta * d[mask_k0]
                corr[k0] += delta * (1.0 - new_val - old_val) * dx

        h = project_feasible(h, target)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')
        elif sweep > 30 and sweep % 20 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_check = float(np.max(corr)) * dx
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h.copy()

    return best_h, best_c5


def binary_sa(h_init, n, target, time_limit, start_time=None, seed=42):
    if start_time is None:
        start_time = time.time()
    rng = np.random.RandomState(seed)
    ones = np.where(h_init >= 0.5)[0]
    zeros = np.where(h_init < 0.5)[0]
    if len(ones) != len(zeros) and len(ones) + len(zeros) == n:
        all_idx = np.arange(n)
        shuffled = rng.permutation(all_idx)
        ones = shuffled[:n // 2]
        zeros = shuffled[n // 2:]
    h = np.zeros(n)
    h[ones] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    T = 5.0
    T_min = 0.01
    cooling = 0.995
    steps_no_improve = 0

    while time.time() - start_time < time_limit:
        if steps_no_improve > 10000:
            T = max(T_min, T * 0.5)
            steps_no_improve = 0
        p1 = rng.choice(ones)
        p2 = rng.choice(zeros)
        if p1 == p2:
            continue
        ones_list = list(ones)
        zeros_list = list(zeros)
        oi = list(ones).index(p1)
        zi = list(zeros).index(p2)
        ones[oi] = p2
        zeros[zi] = p1
        h[p1] = 0.0
        h[p2] = 1.0
        ks = np.arange(-(n - 1), n)
        delta = 2.0 / n
        for pos_val, sign in [(p1, -1.0), (p2, 1.0)]:
            pk = pos_val + ks
            valid_p = (pk >= 0) & (pk < n)
            c1 = np.zeros(2 * n - 1)
            c1[valid_p] = 1.0 - h[pk[valid_p]]
            if sign > 0:
                c1[valid_p] += 1.0
            else:
                c1[valid_p] -= 1.0
            mk = pos_val - ks
            valid_m = (mk >= 0) & (mk < n)
            c2 = np.zeros(2 * n - 1)
            c2[valid_m] = h[mk[valid_m]]
            if sign > 0:
                c2[valid_m] -= 1.0
            else:
                c2[valid_m] += 1.0
            d = (c1 - c2) * delta
            k0 = n - 1
            d[k0] = 0.0
            mask_k0 = np.ones(2 * n - 1, dtype=bool)
            mask_k0[k0] = False
            old = 1.0 if sign > 0 else 0.0
            new = 0.0 if sign > 0 else 1.0
            corr[mask_k0] += (new - old) * d[mask_k0]
            corr[k0] += (new - old) * (1.0 - new - old) * delta
        cur_c5 = float(np.max(corr)) * delta
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            steps_no_improve = 0
        else:
            delta_c = cur_c5 - best_c5
            if delta_c / max(T, 1e-10) < 50 and rng.rand() < np.exp(-delta_c / max(T, 1e-10)):
                steps_no_improve += 1
            else:
                h[p1] = 1.0
                h[p2] = 0.0
                ones[oi] = p1
                zeros[zi] = p2
                for pos_val, sign in [(p1, 1.0), (p2, -1.0)]:
                    pk = pos_val + ks
                    valid_p = (pk >= 0) & (pk < n)
                    c1 = np.zeros(2 * n - 1)
                    c1[valid_p] = 1.0 - h[pk[valid_p]]
                    if sign > 0:
                        c1[valid_p] += 1.0
                    else:
                        c1[valid_p] -= 1.0
                    mk = pos_val - ks
                    valid_m = (mk >= 0) & (mk < n)
                    c2 = np.zeros(2 * n - 1)
                    c2[valid_m] = h[mk[valid_m]]
                    if sign > 0:
                        c2[valid_m] -= 1.0
                    else:
                        c2[valid_m] += 1.0
                    d = (c1 - c2) * delta
                    k0 = n - 1
                    d[k0] = 0.0
                    mask_k0 = np.ones(2 * n - 1, dtype=bool)
                    mask_k0[k0] = False
                    old2 = 0.0 if sign > 0 else 1.0
                    new2 = 1.0 if sign > 0 else 0.0
                    corr[mask_k0] += (new2 - old2) * d[mask_k0]
                    corr[k0] += (new2 - old2) * (1.0 - new2 - old2) * delta
                steps_no_improve += 1
        T = max(T_min, T * cooling)

    return best_h, best_c5


def smooth_logsumexp_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_polish(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_logsumexp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def subgradient_refine(h_init, n, target, max_steps, time_limit, start_time=None):
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h[idxs[idx_valid2] + k]
        if k == 0:
            grad -= h
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5]:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        if not improved and step > 200:
            break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.7, reg=1e-6, start_time=None):
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg
    g = 1.0 - h

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if it % 10 == 0:
            g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 30:
            damping = max(0.1, damping * 0.85)
            stuck_count = 0
            if damping < 0.12:
                damping = min(0.9, damping * 1.5)
        if it % 500 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    deadline = start_time + budget_s - 30.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: Coordinate descent (main new approach)
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                     max_sweeps=2000, time_limit=cd_time,
                                     grid_size=100, start_time=start_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Finer coordinate descent
    if tleft() > 120:
        cd_time2 = min(150, tleft() * 0.25)
        h_cd2, c5_cd2 = coord_descent(best_h, n, target,
                                       max_sweeps=2000, time_limit=cd_time2,
                                       grid_size=200, start_time=start_time)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 3: Binary SA for diverse exploration
    if tleft() > 150:
        sa_time = min(120, tleft() * 0.2)
        h_bin, c5_bin = binary_sa(best_h, n, target,
                                   time_limit=sa_time, start_time=start_time, seed=seed)
        # From binary solution, run coordinate descent to relax
        if c5_bin < 0.39 or tleft() > 100:
            cd_time3 = min(80, tleft() * 0.15)
            h_bin_relaxed, c5_bin_rel = coord_descent(h_bin, n, target,
                                                       max_sweeps=500, time_limit=cd_time3,
                                                       grid_size=150, start_time=start_time)
            if c5_bin_rel < best_score:
                best_score = c5_bin_rel
                best_h = h_bin_relaxed
            if c5_bin < best_score:
                best_score = c5_bin
                best_h = h_bin

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_slq, c5_slq = slsqp_polish(best_h, n, target,
                                      [50, 200, 1000, 5000, 20000],
                                      maxiter=1000)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq

    # Phase 5: Subgradient refinement
    if tleft() > 80:
        sg_time = min(60, tleft() * 0.2)
        h_sg, c5_sg = subgradient_refine(best_h, n, target,
                                          max_steps=30000, time_limit=sg_time,
                                          start_time=start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Another round of coordinate descent
    if tleft() > 80:
        cd_time4 = min(100, tleft() * 0.2)
        h_cd3, c5_cd3 = coord_descent(best_h, n, target,
                                       max_sweeps=1500, time_limit=cd_time4,
                                       grid_size=200, start_time=start_time)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 7: Multi-start with perturbations
    if tleft() > 120:
        rng = np.random.RandomState(seed + 1)
        for noise_level in [0.05, 0.1, 0.2]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_level
            h_pert = project_feasible(h_pert, target)
            cd_time5 = min(40, tleft() * 0.2)
            h_p, c5_p = coord_descent(h_pert, n, target,
                                       max_sweeps=500, time_limit=cd_time5,
                                       grid_size=100, start_time=start_time)
            if c5_p < best_score:
                best_score = c5_p
                best_h = h_p

    # Phase 8: LP for structured improvement
    if tleft() > 150:
        lp_time = min(120, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=3000, time_limit=lp_time,
                                      damping_init=0.8, reg=1e-5, start_time=start_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
        # Follow up with coordinate descent
        if tleft() > 60:
            cd_time6 = min(60, tleft() * 0.2)
            h_lp_cd, c5_lp_cd = coord_descent(h_lp, n, target,
                                               max_sweeps=500, time_limit=cd_time6,
                                               grid_size=200, start_time=start_time)
            if c5_lp_cd < best_score:
                best_score = c5_lp_cd
                best_h = h_lp_cd

    # Phase 9: Try higher resolution
    if tleft() > 200:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            cd_time7 = min(80, tleft() * 0.25)
            h_hr, c5_hr = coord_descent(h_interp, n_try, target_try,
                                         max_sweeps=800, time_limit=cd_time7,
                                         grid_size=100, start_time=start_time)
            if c5_hr < best_score:
                best_score = c5_hr
                best_h = h_hr
            if tleft() < 60:
                break
            cd_time8 = min(60, tleft() * 0.2)
            h_hr2, c5_hr2 = coord_descent(best_h, len(best_h), len(best_h) / 2.0,
                                           max_sweeps=500, time_limit=cd_time8,
                                           grid_size=200, start_time=start_time)
            if c5_hr2 < best_score:
                best_score = c5_hr2
                best_h = h_hr2

    # Phase 10: Final SLSQP + coordinate descent polish
    if tleft() > 40:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_slq_fin, c5_fin = slsqp_polish(best_h, n_final, target_final,
                                          [500, 2000, 10000], maxiter=500)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_slq_fin
    if tleft() > 20:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_cd_fin, c5_fin = coord_descent(best_h, n_final, target_final,
                                          max_sweeps=300, time_limit=min(30, tleft() * 0.7),
                                          grid_size=300, start_time=start_time)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
