def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

```python
def project_box_hyperplane(x, target_sum):
    """Project x onto {h : 0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(x)
    h = x.copy()
    # Bisection on the shift parameter
    lo = x.min() - target_sum / n
    hi = x.max() - target_sum / n
    for _ in range(50):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0, 1)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h
```

    h = project_box_hyperplane(h_init.copy(), target)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_c5 = float(np.max(corr)) * dx
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        
        max_iter = 500
        step_size = 1.0 / alpha
        
        for it in range(max_iter):
            if time.time() - t_start > time_limit:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            shifted = alpha * (corr - np.max(corr)) * dx
            shifted = np.clip(shifted, -500, 0)
            weights = np.exp(shifted)
            weights = weights / weights.sum()
            
            # Compute gradient of smoothed objective
            grad = np.zeros(n)
            for k in range(2*n-1):
                if weights[k] < 1e-10:
                    continue

                    for i in range(lag, n):
                        grad[i] += w * (-h[i - lag])

                        grad[i] += w * (1 - h[i - lag_abs])
                    for i in range(0, n - lag_abs):
                        grad[i] += w * (-h[i + lag_abs])

            h = project_box_hyperplane(h_new, target)
            
            new_c5 = compute_c5(h)
            if new_c5 < curr_c5:

def project_binary(h, target_sum):
    n = len(h)

            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def local_search(h_init, target, max_iters, rng, time_limit, t_start,
                 t_init=0.02, t_final=0.00005):
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init.copy(), target_int).astype(np.float64)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_imp = 0
    max_no_imp = max(3000, max_iters // 8)
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if no_imp > max_no_imp:
            if rng.random() < 0.5:
                h = best_h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                n_pert = rng.randint(2, max(3, n // 50) + 1)
                for _ in range(n_pert):
                    if len(ones) > 0 and len(zeros) > 0:
                        pi = int(ones[rng.randint(len(ones))])
                        pj = int(zeros[rng.randint(len(zeros))])
                        h[pi] = 0
                        h[pj] = 1
                h = project_binary(h, target_int).astype(np.float64)
                corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                cur_c5 = float(np.max(corr)) * 2.0 / n
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            no_imp = 0
            continue
        
        ratio = min(it / max(max_iters - 1, 1), 1.0)
        temp = t_init * (t_final / t_init) ** ratio
        
        move_type = rng.randint(6)
        saved = None
        valid = False
        
        if move_type == 0 and len(ones) > 0 and len(zeros) > 0:
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0; h[jj] = 1
            saved = [(ii, 1), (jj, 0)]
            valid = True
        elif move_type == 1 and len(ones) > 0 and len(zeros) > 0:
            # Lag-focused
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - (n - 1)
            found = False
            if worst_lag >= 0:
                bad = [ci for ci in range(len(ones)) 
                       if int(ones[ci]) + worst_lag < n and h[int(ones[ci]) + worst_lag] == 0]
                if bad:
                    ci = bad[rng.randint(len(bad))]
                    ii = int(ones[ci])
                    good = [cj for cj in range(len(zeros))
                            if int(zeros[cj]) + worst_lag < n and h[int(zeros[cj]) + worst_lag] == 1]
                    if good:
                        cj = good[rng.randint(len(good))]
                        jj = int(zeros[cj])
                        h[ii] = 0; h[jj] = 1
                        saved = [(ii, 1), (jj, 0)]
                        valid = True
                        found = True
            if not found:
                ii = int(ones[rng.randint(len(ones))])
                jj = int(zeros[rng.randint(len(zeros))])
                h[ii] = 0; h[jj] = 1
                saved = [(ii, 1), (jj, 0)]
                valid = True
        elif move_type == 2 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])
            if i1 == i2:
                continue
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if j1 == j2 or j1 == i1 or j2 == i2:
                continue
            h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            saved = [(i1, 1), (i2, 1), (j1, 0), (j2, 0)]
            valid = True
        elif move_type == 3 and len(ones) >= 2:
            # Adjacent swap (swap h[i] and h[i+1])
            i = rng.randint(n - 1)
            if h[i] != h[i + 1]:
                h[i], h[i+1] = h[i+1], h[i]
                saved = [(i, h[i+1]), (i+1, h[i])]
                valid = True
        elif move_type == 4 and len(ones) > 0 and len(zeros) > 0:
            # Multi-lag focused
            top3 = np.argsort(corr)[-3:]
            for tl in top3:
                lag = int(tl) - (n - 1)
                if abs(lag) >= n:
                    continue
                if lag >= 0:
                    bad = [ci for ci in range(len(ones))
                           if int(ones[ci]) + lag < n and h[int(ones[ci]) + lag] == 0]
                else:
                    l = -lag
                    bad = [ci for ci in range(len(ones))
                           if int(ones[ci]) >= l and h[int(ones[ci]) - l] == 0]
                if bad:
                    ci = bad[rng.randint(len(bad))]
                    ii = int(ones[ci])
                    jj = int(zeros[rng.randint(len(zeros))])
                    h[ii] = 0; h[jj] = 1
                    saved = [(ii, 1), (jj, 0)]
                    valid = True
                    break
        elif move_type == 5 and len(ones) >= 3 and len(zeros) >= 3:
            idx1 = rng.choice(len(ones), 3, replace=False)
            idx0 = rng.choice(len(zeros), 3, replace=False)
            saved = []
            for idx in idx1:
                pos = int(ones[idx])
                h[pos] = 0
                saved.append((pos, 1))
            for idx in idx0:
                pos = int(zeros[idx])
                h[pos] = 1
                saved.append((pos, 0))
            valid = True
        
        if not valid:
            continue
        
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        delta = new_c5 - cur_c5
        
        accept = delta < -1e-15
        if not accept and temp > 1e-14:
            accept = rng.random() < min(1.0, np.exp(-delta / temp))
        
        if accept:
            corr = new_corr.astype(np.float64)
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            for pos, old_val in saved:
                h[pos] = old_val
            no_imp += 1
    
    return best_h, best_c5
