def bin_proj(h, tgt):

def feas_proj(h, tgt):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    d = tgt - h.sum()
    if d > 1e-14:
        rm = 1.0 - h
        for i in np.argsort(-rm):
            if d <= 1e-14:
                break
            a = min(rm[i], d)
            h[i] += a
            d -= a
    elif d < -1e-14:
        rm = h.copy()
        for i in np.argsort(-rm):
            if -d <= 1e-14:
                break
            a = min(rm[i], -d)
            h[i] -= a
            d += a
    return np.clip(h, 0.0, 1.0)

def slsqp_refine(h_start, n, tgt, tlim, t0):
    global _n_g, _tg
    _n_g = n
    _tg = tgt
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    h_cur = h_start.copy()
    cur_sc = c5_val(h_cur)

    alphas = [80, 200, 500, 1200, 3000, 8000]
    p_list = [max(3, n // 40), max(5, n // 20), max(10, n // 10), n - 1]

    for p in p_list:
        for alpha in alphas:
            if time.time() - t0 > tlim:
                break

            def obj(x):
                c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
                return top_p_smooth(c, p, alpha)

            try:
                res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
                h_new = feas_proj(res.x, tgt)
                sc = c5_val(h_new)
                if sc < cur_sc:
                    cur_sc = sc
                    h_cur = h_new
            except Exception:
                pass

    return h_cur, cur_sc
