In numpy, for term1: indices are i+(k-(n-1)) for k=0,...,2n-2, which is i-(n-1), i-(n-2), ..., i+(n-1). So it's h[i-(n-1) : i+n] (clipped to [0, n-1]).

This is a contiguous slice! So:
- start1 = max(0, i-(n-1)), end1 = min(n, i+n)
- term1 has 1s at positions k where i+k-(n-1) is in [start1, end1-1] and h[i+k-(n-1)] = 1

Actually, the simplest O(n) implementation is:

```python
def compute_delta(h, i, j, n):
    # i goes 1->0, j goes 0->1
    k = np.arange(2*n-1) - (n-1)  # k holds m values: -(n-1), ..., n-1
    # h[i+m] where m = k, 0 if out of range
    idx1 = i + k
    term1 = np.where((idx1 >= 0) & (idx1 < n), h[idx1.clip(0, n-1)], 0.0)
    idx2 = i - k
    term2 = np.where((idx2 >= 0) & (idx2 < n), h[idx2.clip(0, n-1)], 0.0)
    idx3 = j - k
    term3 = np.where((idx3 >= 0) & (idx3 < n), h[idx3.clip(0, n-1)], 0.0)
    idx4 = j + k
    term4 = np.where((idx4 >= 0) & (idx4 < n), h[idx4.clip(0, n-1)], 0.0)
    return term1 + term2 - term3 - term4
```

This is O(n) and vectorized. For n=600, this is very fast.

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0

            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def fast_greedy_v2(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if k_star == 0:
            stall += 1
            continue

        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        top_i = ones[np.argsort(A[ones])[:min(20, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(20, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def sa_v2(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def sa_focused(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if it % 5 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def embed_search(h_coarse, n_fine, target_fine):
    n_coarse = len(h_coarse)
    ratio = n_fine // n_coarse
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * ratio
        end = start + ratio
        if end > n_fine:
            end = n_fine
        if ratio == 1:
            h_fine[i] = h_coarse[i]
        elif h_coarse[i] > 0.5:
            h_fine[start:end] = 1.0
        else:
            h_fine[start:end] = 0.0
    return project_binary(h_fine, target_fine)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Fast greedy
    allowed = min(80, tleft() * 0.12)
    h_g, c5_g = fast_greedy_v2(best_h, max_iters=50000, rng=rng,
                               time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Focused SA (long run)
    allowed = min(250, tleft() * 0.30)
    n_sa = int(min(5000000, allowed * 50000))
    h_sa, c5_sa = sa_focused(best_h, n_sa, 0.02, 0.0001, rng,
                             time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Greedy + SA alternating (iterated)
    for it in range(12):
        if tleft() < 60:
            break
        # Greedy polish
        allowed = min(30, tleft() * 0.08)
        h_g2, c5_g2 = fast_greedy_v2(best_h, max_iters=20000, rng=rng,
                                     time_limit=allowed, start_time=start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 40:
            break

        # Perturb
        n_pert = rng.randint(3, 8)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        # SA from perturbed
        allowed = min(80, tleft() * 0.12)
        n_sa2 = int(min(2000000, allowed * 40000))
        h_sa2, c5_sa2 = sa_v2(h_pert, n_sa2, 0.015, 0.00005, rng,
                              time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Coarse-to-fine
    if tleft() > 120:
        n_coarse = 100
        target_coarse = n_coarse // 2
        x_new = np.linspace(0, 1, n_coarse)
        x_old = np.linspace(0, 1, n_init)
        h_coarse = np.interp(x_new, x_old, best_h)
        h_coarse_bin = project_binary(h_coarse, target_coarse)

        allowed = min(40, tleft() * 0.15)
        h_cg, c5_cg = fast_greedy_v2(h_coarse_bin, max_iters=10000, rng=rng,
                                     time_limit=allowed * 0.4, start_time=start_time)
        allowed2 = min(30, tleft() * 0.10)
        n_sac = int(min(500000, allowed2 * 30000))
        h_cs, c5_cs = sa_v2(h_cg, n_sac, 0.01, 0.0001, rng,
                            time_limit=allowed2, start_time=start_time)
        if c5_cs < c5_cg:
            h_coarse_best = h_cs
        else:
            h_coarse_best = h_cg

        h_fine = embed_search(h_coarse_best, n_init, target_init)
        allowed3 = min(40, tleft() * 0.12)
        h_fg, c5_fg = fast_greedy_v2(h_fine, max_iters=15000, rng=rng,
                                     time_limit=allowed3 * 0.5, start_time=start_time)
        allowed4 = min(30, tleft() * 0.08)
        n_saf = int(min(500000, allowed4 * 30000))
        h_fs, c5_fs = sa_v2(h_fg, n_saf, 0.01, 0.00005, rng,
                            time_limit=allowed4, start_time=start_time)
        if c5_fs < c5_fg:
            h_embed = h_fs
            c5_embed = c5_fs
        else:
            h_embed = h_fg
            c5_embed = c5_fg
        if c5_embed < best_score:
            best_score = c5_embed
            best_h = h_embed.copy()

    # Phase 5: Try different resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init + 12]:
        if tleft() < 50:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(25, tleft() * 0.08)
        h_g3, c5_g3 = fast_greedy_v2(h_bin_try, max_iters=10000, rng=rng,
                                     time_limit=allowed * 0.5, start_time=start_time)
        allowed2 = min(20, tleft() * 0.06)
        n_sa3 = int(min(300000, allowed2 * 30000))
        h_s3, c5_s3 = sa_v2(h_g3, n_sa3, 0.01, 0.0001, rng,
                            time_limit=allowed2, start_time=start_time)
        if c5_s3 < c5_g3:
            h_try = h_s3
            c5_try = c5_s3
        else:
            h_try = h_g3
            c5_try = c5_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 6: SLSQP refinement on continuous
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 50, 150, 500, 1500, 5000]
        allowed = min(tleft() - 25, 180)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final SA burst on binary projection
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(1000000, max(10000, (tleft() - 15) * 40000)))
        h_sa_f, c5_f = sa_focused(h_bin_final, n_sa_f, 0.008, 0.00002,
                                  rng, time_limit=max(5.0, tleft() - 12), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 8: Final SLSQP if time
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [2000, 5000, 10000], 1500, 1e-15,
                                 time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
