def slsqp_optimize(h_start, n, target, alpha, maxiter=500):
    """SLSQP optimization with analytical gradient."""
    def obj_and_grad(x):
        val = smooth_obj(x, alpha)
        g = smooth_grad(x, alpha)
        return val, g
    
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj_and_grad, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = np.clip(res.x, 0.0, 1.0)
        # Project to feasible
        diff = target - np.sum(h_opt)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h_opt < 1.0)[0]
            else:
                idx = np.where(h_opt > 0.0)[0]
            if len(idx) > 0:
                h_opt[idx] += diff / len(idx)
                h_opt = np.clip(h_opt, 0.0, 1.0)
        return h_opt
    except Exception:
        return h_start
