The terms exp(alpha * (corr - c_max)) will be exp(-alpha * gap) for non-max terms. If the gap is 0.001 (in overlap units), then alpha * gap = 200, and exp(-200) ≈ 1e-87. This is fine numerically. But if the gap is 0, we have exp(0) = 1. The issue is when there are many terms close to the max - the log of their sum might not be accurately computed.

A better approach: use the "log-sum-exp trick" properly:
log_sum_exp(x) = max(x) + log(sum(exp(x - max(x))))

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

_sl_n = 0
_sl_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _eq_fun(x):
    return np.sum(x) - _sl_target


def _eq_jac(x):
    return np.ones(_sl_n)


def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    overlaps = corr * dx
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    global _sl_n, _sl_target
    _sl_n = n
    _sl_target = target
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def alt_lp_active(h_init, n, target, max_iter, time_limit, M_init, M_max, damping_init):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    M = M_init
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        overlaps = compute_all_overlaps(h)
        top_k = np.argsort(overlaps)[::-1][:M]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k_idx in top_k:
            k = k_idx - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(cp.sum(coeff * h_var) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=3000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=3000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 40:
            damping = max(0.3, damping * 0.88)
            if damping < 0.35:
                damping = 0.5
            stuck = 0

        if it % 50 == 0:
            M = min(M + 10, M_max)

    return best_h, best_c5


def coord_line_search(h, j, target):
    n = len(h)
    lo_delta = -h[j] + 1e-8
    hi_delta = 1.0 - h[j] - 1e-8
    if hi_delta - lo_delta < 1e-6:
        return h, compute_c5(h)

    def obj_1d(delta):
        h_try = h.copy()
        h_try[j] = h[j] + delta
        h_try = project_feasible(h_try, target)
        return compute_c5(h_try)

    try:
        res = minimize_scalar(obj_1d, bounds=(lo_delta, hi_delta),
                              method='bounded', options={'xatol': 1e-7})
        if res.success and res.fun < 0.5:
            h_try = h.copy()
            h_try[j] = h[j] + res.x
            h_try = project_feasible(h_try, target)
            return h_try, float(res.fun)
    except Exception:
        pass
    return h, compute_c5(h)


def coordinate_descent(h_start, n, target, max_passes, time_limit):
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break

        overlaps = compute_all_overlaps(h)
        top_k = np.argsort(overlaps)[::-1][:15]

        active_coords = set()
        for k_idx in top_k:
            k = k_idx - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            active_coords.update(range(lo, hi))
            active_coords.update(range(lo + k, hi + k))

        coords = sorted(active_coords)
        improved = False

        for j in coords:
            if time.time() - start > time_limit:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue

            h_new, sc = coord_line_search(h, j, target)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new
                improved = True

        if not improved:
            break

    return best_h, best_c5


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half

        grad = np.zeros(n)
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        lrs = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8]
        for lr in lrs:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100 and step % 200 == 0:
            h_cur = best_h.copy()

    return best_h, best_c5


def structured_perturb(h, n, rng, mode):
    h_new = h.copy()
    if mode == 0:
        shift = rng.randint(-n // 10, n // 10 + 1)
        h_new = np.roll(h, shift)
    elif mode == 1:
        mask = rng.rand(n) < 0.3
        h_new[mask] = 1.0 - h_new[mask]
    elif mode == 2:
        center = rng.randint(n // 4, 3 * n // 4)
        width = rng.randint(n // 20, n // 5)
        idx = np.arange(n)
        dist = np.abs(idx - center)
        bell = np.exp(-dist ** 2 / (2.0 * width ** 2))
        h_new = h + rng.randn(n) * 0.1 * bell
    elif mode == 3:
        h_new = h ** rng.uniform(0.5, 2.0)
    h_new = project_feasible(h_new, n / 2.0)
    return h_new
