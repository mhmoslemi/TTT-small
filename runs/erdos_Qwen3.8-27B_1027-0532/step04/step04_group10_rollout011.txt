**Alternative: Direct minimax with line search**

The problem min_h max_k C(h, k) can be attacked with a line search on t:
- Is there an h with max_k C(h,k) <= t?
- This is: min_h max_k C(h,k) subject to constraints

**Alternative: Frank-Wolfe / Conditional Gradient**

Since the feasible set {h : 0<=h<=1, sum(h)=n/2} is a polytope, we can use Frank-Wolfe:
1. Start with any feasible h
2. Compute gradient of the (smoothed) objective
3. Solve the linear minimization over the polytope (which is easy: pick the coordinates with largest/smallest gradient)
4. Move towards the linear minimizer
5. Repeat

The linear minimization over {0<=h<=1, sum(h)=n/2} is: pick h[i]=1 for the n/2 coordinates with smallest gradient, and h[i]=0 for the rest.

This is actually quite efficient and might work well! Let me think about this more.

Frank-Wolfe for our problem:
- Objective: f(h) = LSE approximation of max_k C(h,k)
- Gradient: ∇f(h) (computed as before)
- Linear minimization: h_v = 1 where ∇f is smallest (top n/2 smallest), 0 elsewhere
- Update: h_new = (1-t)*h + t*h_v where t is the step size

This is attractive because:
1. Each iteration is O(n log n) for the gradient + O(n log n) for the linear minimization
2. No need for bound handling (the polytope projection is exact)
3. Naturally maintains feasibility
4. Can be combined with shrinking/strong Wolfe line search

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        headroom = 1.0 - h
        mask = headroom > 1e-10
        if not np.any(mask):
            return h
        total_head = headroom[mask].sum()
        h[mask] += (diff / total_head) * headroom[mask]
    else:
        mask = h > 1e-10
        if not np.any(mask):
            return h
        total_mass = h[mask].sum()
        h[mask] += (diff / total_mass) * h[mask]
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    n = len(h)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

        if extra > 0:
            rem = sorted_idx[target_int:target_int + int(extra)]
            h_v[rem] = 1.0
    return h_v

    n = len(h)
    target_int = int(target)

        if time.time() - start_time > time_limit:
            break
        _, grad = lse_obj_grad(h_fw, alpha, topk)
        h_v = fw_linear_min(grad, target_int)
        if len(active) > 0:
            grad_active = grad[active]
            away_idx = active[np.argmin(grad_active)]
        else:
            away_idx = -1
        t_fw = 2.0 / (it + 2)
        if away_idx >= 0 and h_fw[away_idx] > 1e-6:

            t_away = min(t_away, t_fw)
            h_away = h_fw - t_away * h_fw
            h_away[away_idx] = max(0.0, h_fw[away_idx] - t_away * h_fw[away_idx])
            h_away = (1.0 - t_fw) * h_fw + t_fw * h_v - (t_away - t_fw * h_fw[away_idx]) * np.zeros(n)
            h_away = h_fw - t_away * h_fw + t_fw * (h_v + h_fw)
            h_away = (1.0 - t_away) * h_fw + t_fw * h_v
            h_away[away_idx] = h_fw[away_idx] * (1.0 - t_away)
            h_try_away = project_feasible(h_away, target)
            f_away, _ = lse_obj_grad(h_try_away, alpha, topk)
            h_try_fw = project_feasible((1.0 - t_fw) * h_fw + t_fw * h_v, target)
            f_fw, _ = lse_obj_grad(h_try_fw, alpha, topk)
            if f_away < f_fw:

            else:
                h_fw = h_try_fw
        else:
            h_try_fw = project_feasible((1.0 - t_fw) * h_fw + t_fw * h_v, target)
            h_fw = h_try_fw
        active = np.where(h_fw > 1e-8)[0]
    return h_fw


def lbfgs_optimize(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol, start_time, time_limit):
    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f = f + 0.5 * penalty * s * s
        g = g + penalty * s * np.ones_like(x)
        return f, g

    callback_count = [0]

    def cb(xk):
        callback_count[0] += 1
        if time.time() - start_time > time_limit:
            return True
        return False

    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, callback=cb,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

                break
            target_try = n_try / 2.0
            bounds_try = [(0.0, 1.0)] * n_try
            x_old = np.linspace(0, 1, best_n)
            x_new = np.linspace(0, 1, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, target_try)

            for alpha in [100, 400, 1500]:
                if remaining() < 40:
                    break
                h_start = h_interp + rng.randn(n_try) * 0.003
                h_start = project_feasible(h_start, target_try)
                h_new = lbfgs_optimize(h_start, target_try, alpha, 100, penalty, 200,
                                       bounds_try, 1e-15, start_time, remaining() - 15)
                score_hi = compute_c5(h_new)
                if score_hi < best_score - 1e-8:
                    best_score = score_hi
                    best_h = h_new
                    best_n = n_try
                elif score_hi < best_score:
                    # Check at original resolution
                    x_down = np.linspace(0, 1, best_n)
                    h_down = np.interp(x_down, x_new, h_new)
                    h_down = project_feasible(h_down, best_n / 2.0)
                    score_down = compute_c5(h_down)
                    if score_down < best_score:
                        best_score = score_down
                        best_h = h_down
                        best_n = best_n

    # Phase 4: Frank-Wolfe polish at best resolution
    if remaining() > 100:
        n_curr = best_n
        target_curr = n_curr / 2.0
        bounds_curr = [(0.0, 1.0)] * n_curr
        for alpha in [200, 500, 1000]:
            if remaining() < 40:
                break
            h_fw = fw_away_step(best_h, target_curr, alpha, 80, 400, start_time, remaining() - 10)
            score_fw = compute_c5(h_fw)
            if score_fw < best_score:
                best_score = score_fw
                best_h = h_fw
                best_n = n_curr

    # Phase 5: Pairwise adjustment on top correlations
    if remaining() > 80:
        n_curr = best_n
        target_curr = n_curr / 2.0
        for _round in range(5):
            if remaining() < 20:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top_lags = np.argsort(corr)[-25:]
            for ki in top_lags:
                if remaining() < 10:
                    break
                lag = ki - (n_curr - 1)
                abs_lag = abs(lag)
                if abs_lag == 0 or abs_lag >= n_curr:
                    continue
                h_try = best_h.copy()
                if lag > 0:
                    for i in range(n_curr - abs_lag):
                        j = i + abs_lag
                        if abs(h_try[i] - h_try[j]) > 0.02:
                            delta = 0.01 * np.sign(h_try[i] - h_try[j])
                            h_try[i] -= delta
                            h_try[j] += delta
                else:
                    m = abs_lag
                    for i in range(m, n_curr):
                        j = i - m
                        if abs(h_try[i] - h_try[j]) > 0.02:
                            delta = 0.01 * np.sign(h_try[i] - h_try[j])
                            h_try[i] -= delta
                            h_try[j] += delta
                h_try = project_feasible(h_try, target_curr)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try
                    best_n = n_curr
                    break

    # Phase 6: High-alpha L-BFGS polish
    if remaining() > 40:
        n_curr = best_n
        target_curr = n_curr / 2.0
        bounds_curr = [(0.0, 1.0)] * n_curr
        for alpha in [3000, 8000, 15000]:
            if remaining() < 20:
                break
            h_new = lbfgs_optimize(best_h, target_curr, alpha, 120, 5000.0, 250,
                                   bounds_curr, 1e-17, start_time, remaining() - 10)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new
                best_n = n_curr

    # Phase 7: Micro coordinate descent
    if remaining() > 25:
        n_curr = best_n
        target_curr = n_curr / 2.0
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_10 = np.argsort(corr)[-10:]
        for ki in top_10:
            if remaining() < 10:
                break
            lag = ki - (n_curr - 1)
            abs_lag = abs(lag)
            if abs_lag == 0 or abs_lag >= n_curr:
                continue
            for i in range(0, n_curr - abs_lag, max(1, (n_curr - abs_lag) // 25)):
                if remaining() < 8:
                    break
                j = i + abs_lag
                for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                    h_try = best_h.copy()
                    old_i = h_try[i]
                    old_j = h_try[j]
                    h_try[i] = min(1.0, max(0.0, old_i + delta))
                    h_try[j] = min(1.0, max(0.0, old_j - delta))
                    h_try = project_feasible(h_try, target_curr)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-11:
                        best_score = sc
                        best_h = h_try
                        best_n = n_curr
                        break

    # Phase 8: Final L-BFGS at very high alpha
    if remaining() > 15:
        n_curr = best_n
        target_curr = n_curr / 2.0
        bounds_curr = [(0.0, 1.0)] * n_curr
        h_new = lbfgs_optimize(best_h, target_curr, 20000, 80, 5000.0, 150,
                               bounds_curr, 1e-18, start_time, remaining() - 5)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
            best_n = n_curr

    best_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(best_h)
    final_n = len(best_h)
    return best_h, final_score, final_n
