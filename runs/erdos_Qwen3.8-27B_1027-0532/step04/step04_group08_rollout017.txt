import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def lse_corr_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lse_corr_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)

    wsum = wt.sum()

        if w < 1e-15:

        s = idx - (n - 1)
        if s >= 0:

            # d/dh[j] = (1-h[j+s]) for j in [0, n-1-s) minus h[j-s] for j in [s, n-1]

    n = len(h0)
    x = project_box_sum(h0, target)
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_f = lse_corr_obj(x, alpha)
    # Estimate step size
    step = 1.0 / (alpha * n)  # rough Lipschitz estimate
    
    for k in range(max_iter):
        if time.time() > deadline:
            break

        # Backtracking line search

            # Check sufficient decrease (or just use fixed step for simplicity)
            break  # Skip backtracking for speed

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)

    if wsum < 1e-300:
        return np.zeros(n)

            continue
        s = idx - (n - 1)

        else:
            sp = -s
            grad[sp:] += w * (1.0 - h[:n-sp])
            grad[:n-sp] -= w * h[sp:]
    return grad

    # Start from binary pattern
    h_bin = (h_init > 0.5).astype(float)

    if h_bin.sum() != target:

            h_bin[j] = 0.0
