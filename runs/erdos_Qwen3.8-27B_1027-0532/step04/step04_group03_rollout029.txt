def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        return h
    h = h.copy()
    h[idx] += diff / len(idx)

def project_binary(h, target_sum):

def subgradient_refine(h_start, target, time_limit, start_time):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    alpha_schedule = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]
    for alpha in alpha_schedule:
        if (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(
                soft_max_obj, h, args=(alpha, n, float(target)),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-15}
            )
            if res.success or res.fun < best_c5 * (1 + 0.01):
                h_new = project_feasible(res.x, float(target))
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
        except Exception:
            pass

    return best_h, best_c5


def targeted_fractional_refine(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    cors = np.linspace(0.01, 0.49, 20)

    for _round in range(50):
        if (time.time() - start_time) > time_limit:
            break
        corr = compute_corr_scaled(h)
        c_max = float(np.max(corr))
        active_shifts = np.where(corr >= c_max * 0.95)[0]

        for s_idx in active_shifts[:5]:
            s = int(s_idx) - (n - 1)
            if s == 0:
                continue
            if abs(s) >= n:
                continue

            improving = False
            for c_val in cors:
                if (time.time() - start_time) > time_limit:
                    break
                if s > 0:
                    valid_range = range(0, n - s)
                else:
                    valid_range = range(0, n + s)

                for i in valid_range:
                    j = i + s
                    if h[i] > 0.5 and h[j] < 0.5:
                        h_save = h[i]
                        h[i] = c_val
                        h[j] = 1.0 - c_val
                        h_proj = project_feasible(h, target)
                        c5_try = compute_c5(h_proj)
                        if c5_try < best_c5 - 1e-10:
                            best_c5 = c5_try
                            best_h = h_proj.copy()
                            h = h_proj
                            improving = True
                            break
                        else:
                            h[i] = h_save
                            h[j] = 1.0 - h_save
                    elif h[i] < 0.5 and h[j] > 0.5:
                        h_save_i = h[i]
                        h_save_j = h[j]
                        h[i] = 1.0 - c_val
                        h[j] = c_val
                        h_proj = project_feasible(h, target)
                        c5_try = compute_c5(h_proj)
                        if c5_try < best_c5 - 1e-10:
                            best_c5 = c5_try
                            best_h = h_proj.copy()
                            h = h_proj
                            improving = True
                            break
                        else:
                            h[i] = h_save_i
                            h[j] = h_save_j

                if improving:
                    break

        if not improving and _round > 5:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Binary SA from initial (focused, longer)
    h_bin_start = project_binary(h_init, int(target))
    time_p1 = min(120.0, max(30.0, tleft() * 0.12))
    h_sa1, c5_sa1 = sa_optimize(h_bin_start, int(target), 200000, 0.02, 0.00002,
                                rng, time_limit=time_p1, start_time=start_time)
    if c5_sa1 < best_score:
        best_score = c5_sa1
        best_h = h_sa1.copy()

    # Hill climb from SA result
    if tleft() > 60:
        h_hc1, c5_hc1 = hill_climb_binary(best_h, int(target), 10000, rng)
        if c5_hc1 < best_score:
            best_score = c5_hc1
            best_h = h_hc1.copy()

    # Phase 2: Second SA round (lower temperature, longer)
    if tleft() > 100:
        time_p2 = min(100.0, max(30.0, tleft() * 0.12))
        h_sa2, c5_sa2 = sa_optimize(best_h, int(target), 150000, 0.005, 0.00001,
                                    rng, time_limit=time_p2, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        h_hc2, c5_hc2 = hill_climb_binary(best_h, int(target), 8000, rng)
        if c5_hc2 < best_score:
            best_score = c5_hc2
            best_h = h_hc2.copy()

    # Phase 3: Subgradient continuous refinement
    if tleft() > 80:
        time_p3 = min(150.0, max(30.0, tleft() * 0.15))
        h_sg, c5_sg = subgradient_refine(best_h, target, time_p3, start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    # Phase 4: SLSQP soft-max refinement
    if tleft() > 80:
        time_p4 = min(200.0, max(40.0, tleft() * 0.2))
        h_slq, c5_slq = soft_max_refine(best_h, target, time_p4, start_time)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    # Phase 5: Targeted fractional refinement
    if tleft() > 60:
        time_p5 = min(120.0, max(20.0, tleft() * 0.15))
        h_tf, c5_tf = targeted_fractional_refine(best_h, target, time_p5, start_time)
        if c5_tf < best_score:
            best_score = c5_tf
            best_h = h_tf.copy()

    # Phase 6: Another round of subgradient + SLSQP
    if tleft() > 80:
        time_p6a = min(80.0, max(20.0, tleft() * 0.15))
        h_sg2, c5_sg2 = subgradient_refine(best_h, target, time_p6a, start_time)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2.copy()
        if tleft() > 50:
            time_p6b = min(100.0, max(30.0, tleft() * 0.2))
            h_slq2, c5_slq2 = soft_max_refine(best_h, target, time_p6b, start_time)
            if c5_slq2 < best_score:
                best_score = c5_slq2
                best_h = h_slq2.copy()

    # Phase 7: Multi-n exploration
    n_base = len(best_h)
    n_candidates = []
    for factor in [0.9, 0.95, 1.05, 1.1, 1.15, 1.2, 1.3, 1.5]:
        nc = int(n_base * factor)
        if nc >= 200 and nc != n_base and nc % 2 == 0:
            if nc not in n_candidates:
                n_candidates.append(nc)

    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, int(target_try))

        time_sa = min(30.0, max(10.0, tleft() * 0.15))
        h_sa_r, c5_sa_r = sa_optimize(h_bin_try, int(target_try), 50000, 0.01, 0.00005,
                                      rng, time_limit=time_sa, start_time=start_time)

        if tleft() > 50:
            time_slq = min(40.0, max(15.0, tleft() * 0.2))
            h_slq_r, c5_slq_r = soft_max_refine(h_sa_r, target_try, time_slq, start_time)
            if c5_slq_r < c5_sa_r:
                h_sa_r = h_slq_r
                c5_sa_r = c5_slq_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 8: Final intensive refinement on best solution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        global _cons_n, _cons_target
        _cons_n = n_use
        _cons_target = float(target_use)
        bounds = [(0.0, 1.0)] * n_use
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        h_cur = best_h.copy()

        for alpha in [5000, 10000, 20000]:
            if tleft() < 20:
                break
            try:
                res = minimize(
                    soft_max_obj, h_cur, args=(alpha, n_use, float(target_use)),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 150, 'ftol': 1e-16}
                )
                h_new = project_feasible(res.x, float(target_use))
                c5_new = compute_c5(h_new)
                if c5_new < best_score:
                    best_score = c5_new
                    best_h = h_new
                    h_cur = h_new
            except Exception:
                pass

    # Phase 9: Final targeted fractional
    if tleft() > 20:
        h_tf2, c5_tf2 = targeted_fractional_refine(best_h, len(best_h) / 2.0,
                                                    min(25.0, tleft() - 5), start_time)
        if c5_tf2 < best_score:
            best_score = c5_tf2
            best_h = h_tf2.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
