import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0
_g_alpha = 1.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def full_con_eq(x):
    return np.sum(x) - _g_target

def full_con_jac(x):
    return np.ones(_g_n)

def block_con_eq(x):
    return np.sum(x) - _g_block_target

def block_con_jac(x):
    return np.ones(_g_e - _g_s)

def block_obj(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': full_con_eq, 'jac': full_con_jac}]
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_target, _g_alpha, _g_block_target
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 300, 'ftol': 1e-13})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0
_g_alpha = 1.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def full_con_eq(x):
    return np.sum(x) - _g_target

def full_con_jac(x):
    return np.ones(_g_n)

def block_con_eq(x):
    return np.sum(x) - _g_block_target

def block_con_jac(x):
    return np.ones(_g_e - _g_s)

def block_obj(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': full_con_eq, 'jac': full_con_jac}]
    h_cur = h.copy()
    cur = compute_c5(h_cur)

        try:

            sc = compute_c5(h_new)
            if sc < cur:

        except Exception:

        # Pick random a from ones, b from zeros
        ones = np.where(h == 1)[0]

        if len(ones) == 0 or len(zeros) == 0:

        a = ones[rng.randint(len(ones))]

Actually, let me be smarter. I don't need to recompute the full correlation. I can compute the delta:

When I swap h[a]: 1→0 and h[b]: 0→1:
The change in corr[k] for each k is:
δ[k] = (change from h[a]) + (change from h[b])

For h[a]: 1→0:
- corr[k] had term h[a]*(1-h[a+j]) where j = n-1-k (the "second index" relative to a)
  Wait, I'm getting confused with the indexing. Let me just use the formula I derived:

corr[k] = Σ_i h[i] * (1-h[i+n-1-k])

When h[a] changes from 1 to 0:
- The term where i=a: was 1*(1-h[a+n-1-k]), now 0. Change: -(1-h[a+n-1-k]) = h[a+n-1-k]
  Valid when 0 ≤ a+n-1-k ≤ n-1, i.e., n-1-a ≤ k ≤ n-1+a → k ∈ [n-1-a, n-1+a]
- The term where i+n-1-k=a, i.e., i=k-n+1+a: was h[k-n+1+a]*(1-1)=0, now h[k-n+1+a]*(1-0)=h[k-n+1+a]. Change: +h[k-n+1+a]
  Valid when 0 ≤ k-n+1+a ≤ n-1, i.e., n-1-a ≤ k ≤ n-1+a... wait that's the same range? No:
  0 ≤ k-n+1+a ≤ n-1 → -a ≤ k-n+1 ≤ n-1-a → n-1+a ≤ k ≤ 2n-2-a... hmm that doesn't seem right.
  
  Let me redo: 0 ≤ k-(n-1)+a ≤ n-1 → (n-1)-a ≤ k ≤ (n-1)+a

So both terms are valid for k ∈ [n-1-a, n-1+a]. 

Similarly for h[b]: 0→1:
- The term where i=b: was 0, now 1*(1-h[b+n-1-k]). Change: +(1-h[b+n-1-k])
  Valid when k ∈ [n-1-b, n-1+b]
- The term where i=k-(n-1)+b: was h[k-n+1+b]*(1-0)=h[k-n+1+b], now h[k-n+1+b]*(1-1)=0. Change: -h[k-n+1+b]
  Valid when k ∈ [n-1-b, n-1+b]

So:
δ[k] = h[a+n-1-k] + h[k-n+1+a] + (1-h[b+n-1-k]) - h[k-n+1+b]

for k in the appropriate ranges (all clipped to [0, 2n-2]).

This is O(1) per k, so O(n) total for computing δ. Then new_max = max(corr + δ) is also O(n).

So each swap evaluation is O(n) instead of O(n^2)! That's a huge speedup.

For n=800: O(800) per swap, so 100,000 swaps take 80M operations. In Python with numpy... maybe 1-2 seconds if vectorized properly.
