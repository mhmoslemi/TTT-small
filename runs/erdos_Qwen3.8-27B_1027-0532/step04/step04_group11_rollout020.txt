def compute_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    w = np.exp(np.clip(alpha * (corr - c_max), -700.0, 0.0))
    ws = float(np.sum(w))
    if ws < 1e-300:
        return np.zeros(n)
    p = w / ws  # length 2n-1
    
    # A[i] = sum_{j=i}^{min(2n-2, i+n-1)} p[j]
    # Using prefix sum
    prefix = np.zeros(2*n)  # prefix[k] = sum_{j=0}^{k-1} p[j]
    prefix[1:] = np.cumsum(p)
    A = np.zeros(n)
    for i in range(n):
        j_lo = i  # since i >= 0
        j_hi = min(2*n-2, i+n-1)
        A[i] = prefix[j_hi+1] - prefix[j_lo]
    
    # B[i] = conv(p, h)[i+n-1]
    B_full = np.convolve(p, h, mode='full')  # length (2n-1)+(n-1) = 3n-2
    B = B_full[n-1 : 2*n-1]  # B[i] = B_full[i+n-1] for i=0,...,n-1
    
    # C[i] = conv(p_rev, h)[n-1+i]
    p_rev = p[::-1]
    C_full = np.convolve(p_rev, h, mode='full')  # length 3n-2
    C = C_full[n-1 : 2*n-1]  # C[i] = C_full[n-1+i] for i=0,...,n-1
    
    grad = dx * (A - B - C)
    return grad
