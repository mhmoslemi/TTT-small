        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def fast_sa_binary(h_start, n, time_limit, start_time, rng, t_init=0.05, t_final=0.0005):
    target = n // 2
    h = project_binary(h_start, target).copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    corr = get_corr(h).astype(np.float64)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    
    no_improve = 0
    max_no_improve = 30000
    accepted_since_refresh = 0
    
    total_iters = 0
    n_corr = 2 * n - 1
    center = n - 1
    
    while time.time() - start_time < time_limit:
        total_iters += 1
        
        if no_improve > max_no_improve:
            # Perturbation
            if len(ones) > 0 and len(zeros) > 0:
                n_pert = rng.randint(3, 8)
                for _ in range(n_pert):
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    h[a] = 0.0
                    h[b] = 1.0
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                corr = get_corr(h)
                current_max = float(np.max(corr))
            no_improve = 0
            continue
        
        ratio = min(1.0, total_iters / 300000.0)
        t = t_init * (t_final / t_init) ** ratio
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Move generation
        r = rng.random()
        
        if r < 0.65:
            # Targeted: focus on worst lag
            # Find worst lag(s) excluding center
            c_temp = corr.copy()
            c_temp[center] = -np.inf
            worst_k = int(np.argmax(c_temp))
            k_val = worst_k - center
            if k_val == 0:
                k_val = 1
            
            # Score positions for this lag
            # For a (flipping 1->0): effect on corr[worst_k] is
            #   -(1-h[a+worst_k-n+1]) + h[a-worst_k+n-1]
            # We want this to be negative (reduce corr)
            # So prefer a where h at "neighbor" positions is low
            
            ak_off = ones + k_val
            ak_off2 = ones - k_val
            valid_ak = (ak_off >= 0) & (ak_off < n)
            valid_ak2 = (ak_off2 >= 0) & (ak_off2 < n)
            
            a_score = np.zeros(len(ones))
            if np.any(valid_ak):
                a_score[valid_ak] += h[ak_off[valid_ak]]
            if np.any(valid_ak2):
                a_score[valid_ak2] += h[ak_off2[valid_ak2]]
            # Lower a_score = better (neighbors are 0)
            
            bk_off = zeros + k_val
            bk_off2 = zeros - k_val
            valid_bk = (bk_off >= 0) & (bk_off < n)
            valid_bk2 = (bk_off2 >= 0) & (bk_off2 < n)
            
            b_score = np.zeros(len(zeros))
            if np.any(valid_bk):
                b_score[valid_bk] += h[bk_off[valid_bk]]
            if np.any(valid_bk2):
                b_score[valid_bk2] += h[bk_off2[valid_bk2]]
            # Higher b_score = better (neighbors are 1)
            
            # Add noise
            a_score += rng.random(len(a_score)) * 0.3
            b_score -= rng.random(len(b_score)) * 0.3
            
            a_idx = ones[np.argmin(a_score)]
            b_idx = zeros[np.argmax(b_score)]
            
        elif r < 0.85:
            # Random move
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
        elif r < 0.95 and len(ones) >= 2 and len(zeros) >= 2:
            # 2-swap: flip two 1s to 0 and two 0s to 1
            ia = rng.choice(len(ones), 2, replace=False)
            ib = rng.choice(len(zeros), 2, replace=False)
            a1, a2 = ones[ia[0]], ones[ia[1]]
            b1, b2 = zeros[ib[0]], zeros[ib[1]]
            # Try the move
            h[a1] = 0; h[b1] = 1
            h[a2] = 0; h[b2] = 1
            new_max = float(np.max(get_corr(h)))
            if not (new_max < current_max or (t > 1e-10 and rng.random() < np.exp(-(new_max - current_max) / t))):
                h[a1] = 1; h[b1] = 0
                h[a2] = 1; h[b2] = 0
                no_improve += 1
                continue
            h = h.copy()  # keep it
            corr = get_corr(h)
            current_max = new_max
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            accepted_since_refresh += 1
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            continue
        else:
            # Small perturbation: flip 3
            if len(ones) >= 1 and len(zeros) >= 1:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]
            else:
                continue
        
        if a_idx == b_idx:
            continue
        
        # Compute incremental delta
        h[a_idx] = 0.0  # will restore if rejected
        h[b_idx] = 1.0
        
        # Compute delta_corr
        # For a flip at position p with old value v_old and new value v_new:
        # delta = v_new - v_old
        # Effect on corr[k]:
        #   From i=p: delta * (1 - h_old[p+k-n+1]) if valid
        #   From i+k-n+1=p (i = p-k+n-1): -delta * h_old[i] if valid
        # Note: h_old is the state BEFORE the flip
        
        # Since we already changed h, we need to compute using old values.
        # For a: old value was 1, new is 0, delta = -1
        # For b: old value was 0, new is 1, delta = +1
        # The neighbors' values in h are still the old values (since we only changed a and b)
        # UNLESS a's neighbor is b or b's neighbor is a. We'll ignore this edge case (rare).
        
        delta_corr = np.zeros(n_corr)
        
        # Flip a (1->0, delta=-1)
        # Contribution 1: i=a, affects k where a+k-n+1 in [0,n)
        ks = max(0, n - 1 - a_idx)
        ke = min(2*n-2, 2*n-2-a_idx)
        if ks <= ke:
            idx = np.arange(ks + n - 1 - a_idx, ke + n - 1 - a_idx + 1)
            delta_corr[ks:ke+1] += -1.0 * (1.0 - h[idx])
        
        # Contribution 2: i+k-n+1=a (i=a-k+n-1), affects k where a-k+n-1 in [0,n)
        # a-k+n-1 >= 0 => k <= a+n-1
        # a-k+n-1 <= n-1 => k >= a
        ks = max(0, a_idx)
        ke = min(2*n-2, a_idx + n - 1)
        if ks <= ke:
            # i = a-n+1+k... no, i = a-k+n-1
            # For k=ks: i = a-ks+n-1
            # For k=ke: i = a-ke+n-1
            idx = a + n - 1 - np.arange(ks, ke+1)
            # idx is decreasing from a+n-1-ks to a+n-1-ke
            delta_corr[ks:ke+1] += 1.0 * h[idx]
        
        # Flip b (0->1, delta=+1)
        ks = max(0, n - 1 - b_idx)
        ke = min(2*n-2, 2*n-2-b_idx)
        if ks <= ke:
            idx = np.arange(ks + n - 1 - b_idx, ke + n - 1 - b_idx + 1)
            delta_corr[ks:ke+1] += 1.0 * (1.0 - h[idx])
        
        ks = max(0, b_idx)
        ke = min(2*n-2, b_idx + n - 1)
        if ks <= ke:
            idx = b + n - 1 - np.arange(ks, ke+1)
            delta_corr[ks:ke+1] += -1.0 * h[idx]
        
        new_max = float(np.max(corr + delta_corr))
        delta_obj = new_max - current_max
        
        accept = False
        if delta_obj < -1e-12:
            accept = True
        elif t > 1e-12:
            if rng.random() < min(1.0, np.exp(-delta_obj / t)):
                accept = True
        
        if accept:
            corr = corr + delta_corr
            current_max = new_max
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            accepted_since_refresh += 1
            if current_max < best_max - 1e-12:
                best_max = current_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if accepted_since_refresh >= 100:
                corr = get_corr(h)
                current_max = float(np.max(corr))
                accepted_since_refresh = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            no_improve += 1
    
    return best_h, best_max * (2.0 / n)
