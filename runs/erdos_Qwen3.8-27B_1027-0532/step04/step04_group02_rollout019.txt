def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

    """Compute delta in correlation at lag k for swap a:1->0, b:0->1.
    k is the lag index in the full correlation array (0 to 2n-2).
    Actual lag value is k - (n-1)."""
    lag = k - (n - 1)
    delta = 0.0
    # h[a] going 1->0: removes (1-h[a+lag]) at lag, adds h[a-lag] at lag... 
    # Wait, I need to be more careful with the indexing.
    # Let me use the lag directly.
    # C(lag) = sum_{i} h[i]*(1-h[i+lag]) for valid i
    # Effect of h[a]: 1->0 on C(lag):
    #   - term i=a: -(1-h[a+lag]) if 0<=a+lag<n
    #   - term i=a-lag: +h[a-lag] if 0<=a-lag<n  (because (1-h[a]) changes from 0 to 1)
    # Effect of h[b]: 0->1 on C(lag):
    #   - term i=b: +(1-h[b+lag]) if 0<=b+lag<n
    #   - term i=b-lag: -h[b-lag] if 0<=b-lag<n
    
    if 0 <= a + lag < n:
        delta += -(1.0 - h[a + lag])
    if 0 <= a - lag < n:
        delta += h[a - lag]
    if 0 <= b + lag < n:
        delta += (1.0 - h[b + lag])
    if 0 <= b - lag < n:
        delta += -h[b - lag]
    return delta

def swap_delta(h, a, b, k, n):
    """Compute change in C(k) for simultaneous swap h[a]:1->0, h[b]:0->1.
    k is the lag (can be negative)."""
    # Find all unique i values that are affected
    candidates = set()
    if 0 <= a < n and 0 <= a+k < n:
        candidates.add(a)
    if 0 <= a-k < n and 0 <= a < n:
        candidates.add(a-k)
    if 0 <= b < n and 0 <= b+k < n:
        candidates.add(b)
    if 0 <= b-k < n and 0 <= b < n:
        candidates.add(b-k)
    
    delta = 0.0
    for i in candidates:
        # Before: h[i] * (1 - h[i+k])
        # After: h'[i] * (1 - h'[i+k])
        h_i = 1.0 if i == a else (0.0 if i == b else h[i])
        h_ik = 1.0 if (i+k) == a else (0.0 if (i+k) == b else h[i+k])
        before = h[i] * (1.0 - h[i+k])
        after = h_i * (1.0 - h_ik)
        delta += after - before
    return delta

            best_c5 = c5
            best_h = h.copy()
            pert_size = max(3, pert_size // 2)
        else:
            h = best_h.copy()
            pert_size = min(60, int(pert_size * 1.4) + 2)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_flip = min(pert_size, len(ones) // 2, len(zeros) // 2)
            if n_flip < 1:
                break
            flip_ones = rng.choice(len(ones), n_flip, replace=False)
            flip_zeros = rng.choice(len(zeros), n_flip, replace=False)
            h[ones[flip_ones]] = 0
            h[zeros[flip_zeros]] = 1

    return best_h, best_c5


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def smooth_max_obj(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_c = np.sort(corr)
    top = sorted_c[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_refine(h_start, n, target, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    p_vals = [max(3, n // 50), max(5, n // 30), max(10, n // 15)]
    alphas = [50, 100, 300, 800, 2000]

    for p in p_vals:
        for alpha in alphas:
            if time.time() - start_time > time_limit - 20:
                return h_cur, curr_score
            try:
                res = minimize(lambda x: smooth_max_obj(x, alpha, p), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score - 1e-15:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass

    if time.time() - start_time < time_limit - 20:
        for alpha in [500, 1500, 5000]:
            if time.time() - start_time > time_limit - 15:
                break
            try:
                res = minimize(lambda x: smooth_max_obj(x, alpha, len(x) - 1), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score - 1e-15:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass

    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_binary(h_init, target_init).copy()
    best_score = compute_c5(best_h)

    h_greedy, score_greedy = fast_greedy(best_h, target_init, max_iter=50000,
                                         top_m=8, time_limit=120.0, start_time=start_time, rng=rng)
    if score_greedy < best_score:
        best_score = score_greedy
        best_h = h_greedy.copy()

    if tleft() > 200:
        h_ils, score_ils = ils_search(best_h, target_init, rng,
                                      time_limit=min(300.0, tleft() * 0.5),
                                      start_time=start_time, top_m=8)
        if score_ils < best_score:
            best_score = score_ils
            best_h = h_ils.copy()

    if tleft() > 100:
        h_sa, score_sa = sa_search(best_h, target_init, max_iter=80000,
                                   t_init=0.01, t_final=0.00005, rng=rng,
                                   time_limit=min(120.0, tleft() * 0.4),
                                   start_time=start_time, top_m=5)
        if score_sa < best_score:
            best_score = score_sa
            best_h = h_sa.copy()
        h_g2, score_g2 = fast_greedy(h_sa, target_init, max_iter=30000,
                                     top_m=10, time_limit=min(60.0, tleft() * 0.3),
                                     start_time=start_time, rng=rng)
        if score_g2 < best_score:
            best_score = score_g2
            best_h = h_g2.copy()

    if tleft() > 120:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, score_ref = slsqp_refine(best_h, n_use, target_use,
                                        time_limit=min(150.0, tleft() * 0.5),
                                        start_time=start_time)
        if score_ref < best_score:
            best_score = score_ref
            best_h = h_ref

    n_candidates = []
    for nf in [0.75, 0.85, 1.1, 1.25, 1.5, 2.0]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try != n_init and n_try >= 200:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)
        h_g, score_g = fast_greedy(h_bin, target_try, max_iter=30000,
                                   top_m=8, time_limit=min(60.0, tleft() * 0.3),
                                   start_time=start_time, rng=rng)
        if score_g < score_g:
            pass
        if tleft() > 40 and score_g < compute_c5(h_bin):
            h_ils2, score_ils2 = ils_search(h_g, target_try, rng,
                                            time_limit=min(40.0, tleft() * 0.3),
                                            start_time=start_time, top_m=8)
            if score_ils2 < score_g:
                h_g = h_ils2
                score_g = score_ils2
        if score_g < best_score:
            best_score = score_g
            best_h = h_g.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        h_ref2, score_ref2 = slsqp_refine(best_h, n_use, target_use,
                                          time_limit=min(100.0, tleft() * 0.5),
                                          start_time=start_time)
        if score_ref2 < best_score:
            best_score = score_ref2
            best_h = h_ref2

    n_final_refines = []
    for nf in [0.9, 1.1, 1.3]:
        n_r = int(len(best_h) * nf)
        if n_r % 2 == 0 and n_r != len(best_h) and n_r >= 200:
            n_final_refines.append(n_r)

    for n_r in n_final_refines:
        if tleft() < 50:
            break
        target_r = n_r / 2.0
        h_interp = interpolate_to(best_h, n_r, target_r)
        h_bin_r = project_binary(h_interp, target_r)
        h_gr, score_gr = fast_greedy(h_bin_r, target_r, max_iter=20000,
                                     top_m=8, time_limit=min(40.0, tleft() * 0.3),
                                     start_time=start_time, rng=rng)
        if tleft() > 30:
            h_ref_r, score_ref_r = slsqp_refine(h_gr, n_r, target_r,
                                                time_limit=min(30.0, tleft() * 0.3),
                                                start_time=start_time)
            if score_ref_r < score_gr:
                h_gr = h_ref_r
                score_gr = score_ref_r
        if score_gr < best_score:
            best_score = score_gr
            best_h = h_gr.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0
    fine_attempts = min(6, int(max(1, tleft() / 40)))

    for fa in range(fine_attempts):
        if tleft() < 30:
            break
        noise = rng.choice([0.005, 0.01, 0.02])
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        h_rp, score_rp = slsqp_refine(h_pert, n_use, target_use,
                                      time_limit=min(35.0, tleft() - 15),
                                      start_time=start_time)
        if score_rp < best_score:
            best_score = score_rp
            best_h = h_rp

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(5000, (tleft() - 10) * 5000))
        h_final_g, score_final_g = fast_greedy(h_bin_final, target_use, max_iter=max_it,
                                               top_m=10, time_limit=max(5.0, tleft() - 10),
                                               start_time=start_time, rng=rng)
        if score_final_g < best_score:
            best_score = score_final_g
            best_h = h_final_g

    if tleft() > 10:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref3, score_ref3 = slsqp_refine(best_h, n_use, target_use,
                                          time_limit=max(5.0, tleft() - 5),
                                          start_time=start_time)
        if score_ref3 < best_score:
            best_score = score_ref3
            best_h = h_ref3

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-14:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
