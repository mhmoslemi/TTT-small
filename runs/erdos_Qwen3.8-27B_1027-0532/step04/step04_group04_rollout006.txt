def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - float(np.sum(h))
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.int32)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            extra = final_sum - target_sum
            ones = np.where(h_r == 1)[0]
            h_r[ones[:extra]] = 0
        else:
            extra = target_sum - final_sum
            zeros = np.where(h_r == 0)[0]
            h_r[zeros[:extra]] = 1
    return h_r.astype(float)

def smooth_max_corr(h, alpha):
    """Compute smooth max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    return lse, corr

    val, _ = smooth_max_corr(h, alpha)
    penalty = penalty_w * (float(np.sum(h)) - target_sum) ** 2
    return val + penalty

                    return penalty_objective(h, alpha, target, penalty_w)

def quasi_random_init(n, rng):
    """Quasi-random initialization using golden ratio."""
    h = np.zeros(n)
    ones_count = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for i in range(ones_count * 2):
        frac = ((i + 0.5) / golden) % 1.0
        pos = int(frac * n)
        if pos not in positions:
            positions.add(pos)
            if len(positions) == ones_count:
                break
    # Fill remaining
    if len(positions) < ones_count:
        available = [p for p in range(n) if p not in positions]
        while len(positions) < ones_count and available:
            pos = available[rng.randint(len(available))]
            positions.add(pos)
            available.remove(pos)
    for pos in positions:
        h[pos] = 1.0
    return h

        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
    
    return best_h

def two_swap_local_search(h, max_iters, rng, time_limit=None, start_time=None):
    """Local search using 2-swap moves (swap two 1s with two 0s)."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > 1000:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        max_val = float(corr.max())
        found = False
        
        for _ in range(200):
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            o1, o2 = ones[i1], ones[i2]
            z1, z2 = zeros[j1], zeros[j2]
            
            h[o1] = 0
            h[o2] = 0
            h[z1] = 1
            h[z2] = 1
            corr_new = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(corr_new.max())
            
            if new_max < max_val:
                corr = corr_new
                if new_max * dx < best_c5:
                    best_c5 = new_max * dx
                    best_h = h.copy()
                stall = 0
                found = True
                break
            else:
                h[o1] = 1
                h[o2] = 1
                h[z1] = 0
                h[z2] = 0
                stall += 1
        
        if not found:
            stall += 1
    
    return best_h, best_c5
