import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def compute_grad_active(h, active_indices):
    """Compute subgradient as average over active lags."""
    n = len(h)
    grad = np.zeros(n)
    for m in active_indices:
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        js = np.arange(lo, hi)
        grad[js] += (1 - 2 * h[m - js])
    grad /= len(active_indices)
    return grad

def subgradient_bb(h_start, target, max_iter, k_active, eps_rel, initial_step, 
                   time_check=None, time_deadline=None):
    """Subgradient method with Barzilai-Borwein step size."""
    n = len(h_start)
    h = h_start.copy()
    h_prev = h.copy()
    s_prev = None
    best_h = h.copy()
    best_score = compute_c5(h)
    
    step = initial_step
    momentum = 0.0
    
    for t in range(max_iter):
        if time_check and time.time() > time_deadline:
            break
        
        # Compute correlations
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(c)
        current_score = c_max * (2.0 / n)
        
        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
        
        # Find active set
        threshold = c_max - eps_rel * c_max
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]
        
        # Limit to top-k if too many
        if len(active_indices) > k_active:
            top_k = np.argpartition(c, -k_active)[-k_active:]
            active_indices = top_k
        
        # Compute subgradient
        s = compute_grad_active(h, active_indices)
        s_norm = np.linalg.norm(s)
        
        if s_norm < 1e-15:
            break
        
        # BB step size
        if t > 0 and s_prev is not None:
            dh = h - h_prev
            ds = s - s_prev
            num = np.dot(dh, dh)
            den = np.dot(dh, ds)
            if den > 1e-10 and num > 1e-10:
                step = num / den
                step = np.clip(step, 1e-8, 50.0)
        
        # Update with momentum
        if momentum > 0 and t > 0:
            v = momentum * (h - h_prev) - step * s
        else:
            v = -step * s
        
        h_new = project_feasible(h + v, target)
        
        h_prev = h.copy()
        h = h_new
        s_prev = s.copy()
        
        # Occasionally reset momentum or step
        if t % 50 == 0:
            step = max(1e-4, step * 0.8)
            momentum = min(0.9, momentum + 0.05) if t > 100 else 0.0
    
    return best_h, best_score

def lse_obj_grad_fast(h, alpha, topk):
    """Log-sum-exp objective and gradient, optimized."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    n = len(h)
    
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        m = idx[i]
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        js = np.arange(lo, hi)
        grad[js] += w * (1 - 2 * h[m - js])
    
    return obj, grad

def lbfgs_optimize(h_start, target, alpha, topk, maxiter, bounds, ftol):
    def obj_func(x):
        f, g = lse_obj_grad_fast(x, alpha, topk)
        return f, g
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

def coordinate_ascent(h, target, max_rounds, time_check=None, time_deadline=None):
    """Pairwise coordinate swaps."""
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time_check and np.any(time.time() > time_deadline):
            break
        improved = False
        # Try swapping pairs
        for i in range(0, n, 2):  # stride 2 for speed
            if time_check and time.time() > time_deadline:
                break
            j = (i + 1) % n
            h_try = best_h.copy()
            h_try[i], h_try[j] = best_h[j], best_h[i]
            sc = compute_c5(h_try)
            if sc < best_score - 1e-9:
                best_score = sc
                best_h = h_try
                improved = True
        
        if not improved:
            break
    
    return best_h, best_score

    rng = np.random.RandomState(seed)
    
    def time_left():

```python
def project_box_sum(h, target_sum):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}."""
    n = len(h)
    h = h.copy()
    
    def sum_at_tau(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0))
    
    # Binary search for tau
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_tau(mid)
        if abs(s - target_sum) < 1e-13:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix any remaining numerical error
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
        idx = np.where((result > 1e-10) & (result < 1 - 1e-10))[0]
        if len(idx) > 0:
            result[idx] += diff / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result
```

This is a much better projection than the simple "spread evenly" approach. It's the Euclidean projection onto the intersection of the box and the hyperplane.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_box_sum(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    result = np.clip(h - tau, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        idx = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(idx) > 0:
            result[idx] += diff / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result


def grad_at_lag(h, m):
    n = len(h)
    lo = max(0, m - n + 1)
    hi = min(n, m + 1)
    g = np.zeros(n)
    if lo < hi:
        js = np.arange(lo, hi)
        g[js] = 1.0 - 2.0 * h[m - js]
    return g


def subgradient_bb_optimize(h_start, target, max_iter, k_active, eps_frac, step0, deadline):
    n = len(h_start)
    h = h_start.copy()
    h_prev = h.copy()
    s_prev = None
    best_h = h.copy()
    best_score = compute_c5(h)
    step = step0
    momentum = 0.0
    stall_count = 0

    for t in range(max_iter):
        if time.time() > deadline:
            break

        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(c)
        cur = c_max * (2.0 / n)

        if cur < best_score - 1e-12:
            best_score = cur
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 200:
            step *= 0.7
            momentum = max(0.0, momentum - 0.1)
            stall_count = 0

        threshold = c_max - eps_frac * c_max
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > k_active:
            active_idx = np.argpartition(c, -k_active)[-k_active:]

        s = np.zeros(n)
        for m in active_idx:
            s += grad_at_lag(h, m)
        s /= len(active_idx)
        s_norm = np.linalg.norm(s)

        if s_norm < 1e-14:
            break

        if t > 0 and s_prev is not None:
            dh = h - h_prev
            ds = s - s_prev
            num = float(np.dot(dh, dh))
            den = float(np.dot(dh, ds))
            if den > 1e-12 and num > 1e-12:
                step = num / den
                step = min(max(step, 1e-6), 30.0)

        if momentum > 0.01 and t > 20:
            v = momentum * (h - h_prev) - step * s
        else:
            v = -step * s

        h_new = project_box_sum(h + v, target)
        h_prev = h.copy()
        h = h_new
        s_prev = s.copy()

        if t % 100 == 0 and t > 0:
            step = max(1e-5, step * 0.9)
            if t > 200:
                momentum = min(0.85, momentum + 0.03)

    return best_h, best_score


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    n = len(h)
    if topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(nc)
    ct = corr[idx]
    c_max = np.max(ct)
    shifted = np.clip(alpha * (ct - c_max), -500.0, 0.0)
    ev = np.exp(shifted)
    total = np.sum(ev)
    obj = c_max + np.log(total) / alpha
    w = ev / total

    grad = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-14:
            continue
        m = idx[i]
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        if lo < hi:
            js = np.arange(lo, hi)
            grad[js] += wi * (1.0 - 2.0 * h[m - js])
    return obj, grad


def run_lbfgs(h_start, target, alpha, topk, maxiter, bounds, ftol):
    try:
        res = minimize(lse_obj_grad, h_start, args=(alpha, topk), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_box_sum(res.x, target)
    except Exception:
        return project_box_sum(h_start, target)


def pairwise_swap(h, target, max_rounds, deadline):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            j = (i + 1 + i % 7) % n
            if j == i:
                j = (i + 2) % n
            h_try = best_h.copy()
            h_try[i], h_try[j] = best_h[j], best_h[i]
            sc = compute_c5(h_try)
            if sc < best_score - 1e-10:
                best_score = sc
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_score
