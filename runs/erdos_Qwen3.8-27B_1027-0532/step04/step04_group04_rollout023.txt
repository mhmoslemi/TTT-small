    # I'll just use a simple loop over k for clarity (n=600 means 1199 iterations, each O(n) = O(n^2) total)
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[k + (n-1)]
        if pk < 1e-30:
            continue
        # First term: (1-h[j-k]) for j in [max(0,k), min(n-1, k+n-1)]
        js = max(0, k)
        je = min(n-1, k + n - 1)
        if js <= je:
            grad[js:je+1] += pk * (1.0 - h[js-k:je-k+1])
        # Second term: -h[j+k] for j in [max(0,-k), min(n-1, n-1-k)]
        js2 = max(0, -k)
        je2 = min(n-1, n-1-k)
        if js2 <= je2:
            grad[js2:je2+1] -= pk * h[js2+k:je2+k+1]
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0
_alpha_cur = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)


def _lse_obj_and_grad(h_flat, alpha):
    h = h_flat
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    exp_sum = float(np.sum(exp_shifted))
    p = exp_shifted / exp_sum
    obj = max_corr + float(np.log(exp_sum)) / alpha
    penalty = 500.0 * (np.sum(h) - _cons_target) ** 2
    obj += penalty
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-30:
            continue
        k = m - (n - 1)
        js = max(0, k)
        je = min(n - 1, k + n - 1)
        if js <= je:
            grad[js:je + 1] += pm * (1.0 - h[js - k:je - k + 1])
        js2 = max(0, -k)
        je2 = min(n - 1, n - 1 - k)
        if js2 <= je2:
            grad[js2:je2 + 1] -= pm * h[js2 + k:je2 + k + 1]
    grad += 1000.0 * (np.sum(h) - _cons_target)
    return obj, grad


def _slsqp_obj(h_flat):
    h = h_flat
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    shifted = _alpha_cur * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    return max_corr + float(np.log(np.sum(exp_shifted))) / _alpha_cur


def _slsqp_grad(h_flat):
    h = h_flat
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    shifted = np.clip(_alpha_cur * (corr - max_corr), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    exp_sum = float(np.sum(exp_shifted))
    p = exp_shifted / exp_sum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-30:
            continue
        k = m - (n - 1)
        js = max(0, k)
        je = min(n - 1, k + n - 1)
        if js <= je:
            grad[js:je + 1] += pm * (1.0 - h[js - k:je - k + 1])
        js2 = max(0, -k)
        je2 = min(n - 1, n - 1 - k)
        if js2 <= je2:
            grad[js2:je2 + 1] -= pm * h[js2 + k:je2 + k + 1]
    return grad


def _slsqp_cons_fun(h_flat):
    return np.sum(h_flat) - _cons_target


def _slsqp_cons_jac(h_flat):
    return np.ones(_cons_n)


def lbfgsb_optimize(h_start, target, alpha, maxiter, time_limit, start_time):
    global _cons_target
    _cons_target = float(target)
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    try:
        res = minimize(_lse_obj_and_grad, h_cur, method='L-BFGS-B',
                       bounds=bounds, args=(alpha,),
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
        if time.time() - start_time < time_limit:
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new
    except Exception:
        pass
    return best_h, best_sc


def slsqp_optimize(h_start, target, alpha, maxiter, time_limit, start_time):
    global _cons_n, _cons_target, _alpha_cur
    _cons_n = len(h_start)
    _cons_target = float(target)
    _alpha_cur = alpha
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * _cons_n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    try:
        res = minimize(_slsqp_obj, h_cur, method='SLSQP', jac=_slsqp_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new
    except Exception:
        pass
    return best_h, best_sc


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fast_greedy(h, max_iters, rng, time_limit=None, start_time=None, top_k=60):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 600

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            top_idx = int(np.argmax(corr))
            k2 = top_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k2 < n]
            j_pool = [j for j in zeros if 0 <= j + k2 < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            if corr.max() < max_val:
                stall = 0
                new_c5 = float(corr.max()) * (2.0 / n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1
                h[j] = 0
            continue

        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = top_indices - (n - 1)
        top_vals = corr[top_indices]

        found_improve = False
        n_try = min(300, len(i_cand) * len(j_cand))
        for _ in range(n_try):
            i = i_cand[rng.randint(len(i_cand))]
            j = j_cand[rng.randint(len(j_cand))]
            pos_i = i + top_shifts
            pos_j = j + top_shifts
            valid_i = (pos_i >= 0) & (pos_i < n)
            valid_j = (pos_j >= 0) & (pos_j < n)
            c1 = np.where(valid_i, 1.0 - h[np.clip(pos_i, 0, n - 1)], 0.0)
            c0 = np.where(valid_j, 1.0 - h[np.clip(pos_j, 0, n - 1)], 0.0)
            new_top_vals = top_vals + c0 - c1
            if new_top_vals.max() < max_val - 1e-10:
                h[i] = 0
                h[j] = 1
                corr = compute_corr(h)
                new_max = float(corr.max())
                new_c5 = new_max * (2.0 / n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                stall = 0
                found_improve = True
                break
        if not found_improve:
            stall += 1

    return best_h, best_c5


def sa_focal(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 12000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        move_type = rng.randint(8)
        h_new = h.copy()
        moved = False

        if move_type <= 4:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 5:
            top_indices = np.argsort(corr)[-10:][::-1]
            top_shifts = top_indices - (n - 1)
            k_s = top_shifts[rng.randint(min(10, len(top_shifts)))]
            i_pool = [i for i in ones if 0 <= i + k_s < n]
            j_pool = [j for j in zeros if 0 <= j + k_s < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 6:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def freq_perturb(h, rng, n_modes=5):
    n = len(h)
    h_new = h.copy()
    fft_h = np.fft.rfft(h)
    freqs = np.arange(len(fft_h))
    non_dc = np.where(freqs > 0)[0]
    selected = rng.choice(non_dc, min(n_modes, len(non_dc)), replace=False)
    for f in selected:
        phase = rng.uniform(-np.pi, np.pi)
        mag_scale = 1.0 + rng.uniform(0.2, 0.8)
        fft_h[f] = fft_h[f] * mag_scale * np.exp(1j * phase)
    h_new = np.real(np.fft.irfft(fft_h, n=n))
    return project_feasible(h_new, float(n) / 2.0)


def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()

                best_score = sc_full
                best_h = h_check.copy()

                best_h = h_new.copy()

            alloc = min(30, tleft() * 0.25)

            best_h = h_g.copy()

    # Phase 5: SA on binary solution

        h_sa, c5_sa = sa_focal(best_h, target_init, max_it, 0.012, 0.00003,

            best_h = h_ig.copy()

    # Phase 7: Multi-resolution try (n=300, n=400)
    for n_try in [300, 400]:
        if tleft() < 40:

        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        for alpha in [20, 200, 2000]:

            alloc = min(15, tleft() * 0.15)
            h_tri, sc_tri = lbfgsb_optimize(h_interp, target_try, alpha, 400, alloc, start_time)
            h_back = interpolate_to(h_tri, n_init, target_init)
            sc_back = compute_c5(h_back)
            if sc_back < best_score:
                best_score = sc_back
                best_h = h_back.copy()

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        alloc = min(20, tleft() * 0.5)
        h_new, sc = slsqp_optimize(best_h, target_init, 5000, 3000, alloc, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 9: Final binary polish
    if tleft() > 20:
        h_bin_final = project_binary(best_h, target_init)
        alloc = min(10, tleft() * 0.5)
        h_f, c5_f = fast_greedy(h_bin_final, max_iters=20000, rng=rng,
                                time_limit=alloc, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > 0.38090:
        h_final = project_feasible(h_init.copy(), target_final)
        final_score = compute_c5(h_final)
