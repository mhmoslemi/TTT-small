import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr(h, k_idx, n):
    lag = k_idx - (n - 1)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g = np.zeros(n)
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
        return g
    else:
        m = -lag
        g = np.zeros(n)
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
        return g

def subgrad_topk(h, top_idx):
