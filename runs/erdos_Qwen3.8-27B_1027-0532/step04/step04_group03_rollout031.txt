import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)


def lsq_obj(x, alpha, target):
    h = project_feasible(x, target)
    corr = compute_corr_scaled(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return c_max + log_sum / alpha


def lsq_grad(x, alpha, target):
    h = project_feasible(x, target)
    corr = compute_corr_scaled(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-10:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * dx * (1.0 - h[:n - m])
            grad[:n - m] -= w * dx * (1.0 - h[m:])
        else:
            mm = -m
            grad[:n - mm] += w * dx * (1.0 - h[mm:])
            grad[mm:] -= w * dx * (1.0 - h[:n - mm])
    return grad


def cons_fun(x):
    return np.sum(x) - _cons_target


def cons_jac(x):
    return np.ones(_cons_n)

            best_score = c5_s5
            best_h = h_s5.copy()
    elif remaining > 40:
        time_p5 = min(80.0, remaining * 0.4)
        alphas_p5 = [8000, 15000, 30000, 50000]
        h_s5, c5_s5 = slsqp_optimize(best_h, target0, alphas_p5, time_p5, start_time, maxiter_per_alpha=400)
        if c5_s5 < best_score:
            best_score = c5_s5
            best_h = h_s5.copy()

    # Phase 6: Local perturbation
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        time_p6 = min(50.0, remaining * 0.5)
        h_lp, c5_lp = local_perturb(best_h, target_cur, time_p6, start_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp.copy()

    # Phase 7: Final SLSQP polish
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 15:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        time_p7 = min(40.0, remaining * 0.7)
        alphas_p7 = [20000, 50000]
        h_f, c5_f = slsqp_optimize(best_h, target_cur, alphas_p7, time_p7, start_time, maxiter_per_alpha=300)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
