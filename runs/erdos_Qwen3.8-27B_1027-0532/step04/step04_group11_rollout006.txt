```python
            break
```

```python
def active_set_descent(h, n, target, n_iter, tol=1e-6, deadline=None):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_val = np.max(corr)
        # Active set: shifts where corr is close to max
        active = np.where(corr >= max_val - tol * max_val)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            tau = k - (n - 1)
            # ∂c(tau)/∂h[i] = (1-h[i-tau]) - h[i+tau] (with bounds)
            if tau >= 0:
                # h[i+tau] for i in [0, n-tau), h[i-tau] for i in [tau, n)
                # (1-h[i-tau]) contributes to i in [tau, n)
                # -h[i+tau] contributes to i in [0, n-tau)
                grad[tau:] += (1.0 - h_cur[:n-tau])  # (1-h[i-tau]) for i=tau..n-1, h[i-tau] = h[0..n-tau-1]
                grad[:n-tau] -= h_cur[tau:]  # -h[i+tau] for i=0..n-tau-1, h[i+tau] = h[tau..n-1]
            else:
                s = -tau
                # h[i-s] for i in [s, n), h[i+s] for i in [0, n-s)
                grad[s:] += (1.0 - h_cur[:n-s])  # (1-h[i-tau]) = (1-h[i+s])... wait
                # Let me redo this
                # tau < 0, let s = -tau > 0
                # c(tau) = Σ_j h[j+tau] (1-h[j]) = Σ_j h[j-s] (1-h[j])
                # valid j: j-s >= 0 and j >= 0, so j in [s, n)
                # ∂/∂h[i]: 
                #   from h[j-s]: j-s = i → j = i+s, contribution (1-h[j]) = (1-h[i+s]), valid when i+s < n
                #   from (1-h[j]): j = i, contribution -h[j-s] = -h[i-s], valid when i-s >= 0
                grad[:n-s] += (1.0 - h_cur[s:])  # (1-h[i+s]) for i=0..n-s-1
                grad[s:] -= h_cur[:n-s]  # -h[i-s] for i=s..n-1
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Projected gradient step
        # We want to minimize, so move in -grad direction
        # But we need to maintain sum(h) = target
        # Remove the mean component (project onto the constraint)
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        # Try a few step sizes
        best_step = 0
        best_sc = cur
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = h_cur - lr * grad_centered
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_step = lr
        
        if best_step > 0:
            h_cur = project_feasible(h_cur - best_step * grad_centered, target)
            cur = compute_c5(h_cur)
    
    return h_cur, cur
```

```python
import numpy as np
from scipy.optimize import minimize
import time
```

```python
_g_h_base = None
_g_s = 0
_g_e = 0
_g_alpha = 1.0
_g_target = 0.0
_g_n = 0
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999999
        else:
            mask = h > 1e-8
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_full(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def lse_obj_block(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def con_full_eq(x):
    return np.sum(x) - _g_target


def con_full_jac(x):
    return np.ones(_g_n)


def con_block_eq(x):
    return np.sum(x) - _g_block_target


def con_block_jac(x):
    return np.ones(_g_e - _g_s)


def multi_shift_descent(h, n, target, n_iter, deadline, top_m=8, max_step=0.03):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_val = float(np.max(corr))
        tol = 0.001 * max_val
        active = np.where(corr >= max_val - tol)[0]
        top_k = np.argsort(corr)[-top_m:]
        improved = False
        for k in top_k:
            tau = int(k) - (n - 1)
            if tau == 0:
                continue
            if tau > 0:
                a = h_cur[:n - tau].copy()
                b = h_cur[tau:].copy()
                misalign = a - b
                h_try = h_cur.copy()
                h_try[:n - tau] -= max_step * misalign
                h_try[tau:] += max_step * misalign
            else:
                s = -tau
                a = h_cur[s:].copy()
                b = h_cur[:n - s].copy()
                misalign = a - b
                h_try = h_cur.copy()
                h_try[s:] -= max_step * misalign
                h_try[:n - s] += max_step * misalign
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < cur - 1e-15:
                h_cur = h_try
                cur = sc
                improved = True
                break
        if not improved:
            h_try = h_cur.copy()
            idx = np.random.choice(n, size=min(10, n), replace=False)
            for i in idx:
                h_try[i] += np.random.randn() * 0.03
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < cur - 1e-15:
                h_cur = h_try
                cur = sc
    return h_cur, cur


def slsqp_full(h, n, target, alpha, maxiter=2000, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_full_eq, 'jac': con_full_jac}]
    try:
        res = minimize(lse_obj_full, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def slsqp_block_single(h, n, target, s, e, alpha, maxiter=400):

    _g_h_base = h
    _g_s = s
    _g_e = e
    _g_alpha = alpha
    _g_block_target = target - np.sum(h[:s]) - np.sum(h[e:])
    bs = e - s
    bounds = [(0.0, 1.0)] * bs
    cons = [{'type': 'eq', 'fun': con_block_eq, 'jac': con_block_jac}]

        res = minimize(lse_obj_block, h[s:e].copy(), method='SLSQP',

        ht[s:e] = np.clip(res.x, 0.0, 1.0)

def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        order = list(range(n_blocks))
        np.random.shuffle(order)
        for bi in order:
            if time.time() > deadline:

            s = bi * block_size
            e = s + block_size
            ht = slsqp_block_single(h_cur, n, target, s, e, alpha, 400)
            sc = compute_c5(ht)
            if sc < cur - 1e-14:
                h_cur = ht
                cur = sc
    return h_cur, cur

    L2 = n // 2 - L1

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = project_feasible(h_new, new_n / 2.0)
    return h_new
```
