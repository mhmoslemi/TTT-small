def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]

    # Fix any remaining discrepancy

def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)

        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0

        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]

        move_type = rng.randint(5)

        if move_type <= 2:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
        elif move_type == 3:
            block_len = rng.randint(1, min(6, len(ones) // 3 + 1))
            i_start = ones[rng.randint(len(ones))]
            j_start = zeros[rng.randint(len(zeros))]
            valid = True
            ib = []
            jb = []
            for d in range(block_len):
                ip = i_start + d
                jp = j_start + d
                if 0 <= ip < n and 0 <= jp < n and h[ip] == 1 and h[jp] == 0:
                    ib.append(ip)
                    jb.append(jp)
                else:
                    valid = False
                    break
            if valid and len(ib) == block_len:
                h_new[ib] = 0.0
                h_new[jb] = 1.0
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0.0
                h_new[j] = 1.0
        else:
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[i1]] = 0.0
            h_new[ones[i2]] = 0.0
            h_new[zeros[j1]] = 1.0
            h_new[zeros[j2]] = 1.0

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n

    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + float(np.sum(np.exp(shifted))) / alpha

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    h_cur = h_start.copy().astype(float)
    h_cur = project_feasible(h_cur, target)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)

    h_rand = random_binary_init(n, rng)

    best_h = h_bin.copy()
    best_score = compute_c5(best_h)

    for h_start in [h_bin, h_tm, h_rand]:
        if tleft() < 60:
            break
        allowed = min(90, tleft() * 0.15)
        h_g, c5_g = top_k_greedy(h_start, target, max_iters=50000, rng=rng,
                                  time_limit=allowed, start_time=start_time, top_k=8)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    if tleft() > 100:
        allowed = min(200, tleft() * 0.3)
        h_sa, c5_sa = sa_block(best_h, target, int(min(80000, allowed * 400)),
                               0.012, 0.00003, rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    for ils_iter in range(12):
        if tleft() < 60:
            break
        n_pert = max(3, int(3 + ils_iter * 2))
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        allowed = min(35, tleft() * 0.08)
        h_ig, c5_ig = top_k_greedy(h_pert, target, max_iters=20000, rng=rng,
                                    time_limit=allowed, start_time=start_time, top_k=8)
        if c5_ig < best_score:
            best_score = c5_ig
            best_h = h_ig.copy()

        if tleft() < 40:
            break
        allowed2 = min(25, tleft() * 0.05)
        h_sa2, c5_sa2 = sa_block(h_ig, target, int(min(30000, allowed2 * 600)),
                                  0.005, 0.00002, rng, time_limit=allowed2, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    if tleft() > 120:
        n2 = 1000
        target2 = n2 // 2
        h_interp2 = interpolate_to(best_h, n2, target2)
        h_bin2 = project_binary(h_interp2, target2)
        allowed = min(80, tleft() * 0.15)
        h_g2, c5_g2 = top_k_greedy(h_bin2, target2, max_iters=30000, rng=rng,
                                    time_limit=allowed, start_time=start_time, top_k=8)
        if tleft() > 50:
            allowed2 = min(60, tleft() * 0.12)
            h_sa3, c5_sa3 = sa_block(h_g2, target2, int(min(40000, allowed2 * 400)),
                                      0.01, 0.00002, rng, time_limit=allowed2, start_time=start_time)
            if c5_sa3 < c5_g2:
                g2_best, g2_score = h_sa3, c5_sa3
            else:
                g2_best, g2_score = h_g2, c5_g2
            if g2_score < best_score:
                best_score = g2_score
                best_h = g2_best.copy()

    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 60, 200, 600, 1500, 4000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-13,
                                  time_limit=min(tleft() - 20, 150), start_time=start_time)
        if sc < best_score:

            best_h = h_ref

        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        max_it = int(min(40000, max(500, tleft() * 800)))
        h_sa_f, c5_f = sa_block(h_bin_f, target_use, max_it, 0.005, 0.00001,
                                 rng, time_limit=max(5.0, tleft() - 12), start_time=start_time)
