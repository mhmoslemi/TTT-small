def _slsqp_obj(x, alpha, topk):

def _slsqp_grad(x, alpha, topk):

    n_corr = len(corr)

    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return multi_lag_grad(x, top_idx, weights)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(200):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / float(nf)
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / float(nf)
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def multi_lag_grad(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def lp_step(h, target_sum, topk, top_idx_override=None):
    n = len(h)
    corr = get_corrs(h)
    n_corr = len(corr)
    if top_idx_override is not None:
        top_idx = top_idx_override
    elif topk >= n_corr:
        top_idx = np.arange(n_corr)
    else:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    m_constr = len(top_idx)
    A = np.zeros((m_constr, n + 1))
    b = np.zeros(m_constr)
    for i, idx in enumerate(top_idx):
        g = grad_lag(h, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = float(g @ h) - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target_sum)
    return None

                                     topk_list=[100, 200, 350], rng=rng)
            if score < best_score:
                best_score = score
                best_h = h_new

    if _remaining() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [(5000, 30), (10000, 15)]
        h_ref, sc = slsqp_polish(best_h, target_use, alphas2, maxiter=1500)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _remaining() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_new, score = lp_phase(best_h, target_use, _remaining() - 10,
                                 topk_list=[300, 500], rng=rng)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, float(len(best_h)) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
