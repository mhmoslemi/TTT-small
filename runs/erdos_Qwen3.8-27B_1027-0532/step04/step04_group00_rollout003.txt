import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_block_h = None
_g_block_s = 0
_g_block_e = 0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _block_obj(x):
    h_full = _g_block_h.copy()
    h_full[_g_block_s:_g_block_e] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    corr = compute_corr(h_full)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.015, lr_init * 3.0)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        j_pos = np.arange(0, n - k_star)
        grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
        j_neg = np.arange(k_star, n)
        grad[j_neg] -= h_cur[j_neg - k_star]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.04, 0.025)
        else:
            eps *= 0.55
            if eps < 1e-8:
                break
    return h_cur, cur_score


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_block_h, _g_block_s, _g_block_e, _g_target, _g_alpha
    _g_target = target
    _g_alpha = float(alpha)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_block_h = h_cur.copy()
            _g_block_s = s
            _g_block_e = e
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(_block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 5
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 5)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.25, target)
            else:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.4, 0.05, 0.95), target)

    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def penalty_slsqp(h, n, target, alphas, pen_coeff, maxiter, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)

        def _pen_obj(x):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = _g_alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha
            pen = pen_coeff * (np.sum(x) - target) ** 2
            return lse + pen

        try:
            res = minimize(_pen_obj, h_cur, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur
