        c_max = np.max(corr)
        # Get top K indices
        sorted_idx = np.argsort(corr)[::-1][:K]
        weights = np.exp(alpha * (corr[sorted_idx] - c_max))
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for idx, w in zip(sorted_idx, weights):
            if w < 1e-12:
                continue
            m = idx - (n - 1)
            if m == 0:
                grad += w * (1.0 - 2.0 * h)

    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_grad_topK(h, corr_raw, K, alpha):
    """Compute gradient of log-sum-exp of top K correlations w.r.t. h."""
    n = len(h)
    c_max = np.max(corr_raw)
    # Get top K indices by correlation value
    top_idx = np.argsort(corr_raw)[::-1][:K]
    vals = corr_raw[top_idx] - c_max
    weights = np.exp(alpha * vals)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        return np.zeros(n)
    weights = weights / w_sum
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        m = idx - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
        else:
            p = -m
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad

                break

1. Use SLSQP with the equality constraint (slower but handles constraints)
2. Use L-BFGS-B and incorporate the sum constraint into the objective via a penalty
3. Use L-BFGS-B on a reparameterized problem where the sum is automatically satisfied

Option 3 is elegant: I can parameterize h' = h - mean(h) + target/n, so that sum(h') = target automatically. But this might violate bounds.

Actually, let me use a penalty approach with L-BFGS-B:
- Objective: log-sum-exp of top-K correlations + penalty * (sum(h) - target)^2
- This avoids the constraint entirely and L-BFGS-B is faster than SLSQP
