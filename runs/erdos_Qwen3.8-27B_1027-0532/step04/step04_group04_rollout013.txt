The smooth approximation to max:
f_alpha(x) = (1/alpha) * log(sum(exp(alpha * x_i)))

This converges to max(x_i) as alpha → ∞.

For the correlation vector corr(h), we want to minimize f_alpha(corr(h)).

The gradient of f_alpha with respect to h:
df/dh_i = (1/alpha) * d/dh_i [log(sum_p exp(alpha * corr[p]))]
= (1/alpha) * [sum_p alpha * exp(alpha*corr[p]) * d(corr[p])/dh_i] / [sum_p exp(alpha*corr[p])]
= [sum_p exp(alpha*corr[p]) * d(corr[p])/dh_i] / [sum_p exp(alpha*corr[p])]

This is a weighted average of d(corr[p])/dh_i, weighted by softmax(alpha * corr).

d(corr[p])/dh_i: from corr[p] = sum_n h[n+s]*(1-h[n]) with s = p-(n-1):
- If i appears as n+s (i.e., s = i-n for some n): d/dh_i contributes (1-h[n])
- If i appears as n (i.e., n = i): d/dh_i contributes -h[i+s]

So d(corr[p])/dh_i = (1-h[i-s])*[0<=i-s<n] - h[i+s]*[0<=i+s<n]

Note this is exactly -A(i) (from before, without the swap). Interesting.

So the gradient is:
g_i = sum_p softmax_p * [(1-h[i-s_p])*[valid] - h[i+s_p]*[valid]]

where s_p = p - (n-1).

This can be computed in O(n²) time (for each of n positions i, sum over n+1 values of p).

But actually, we can vectorize this:
g = sum_p softmax_p * [(1-h[i-s_p]) - h[i+s_p]] (with appropriate masking)

This is a convolution-like operation! Specifically:
g_i = sum_p softmax_p * (1-h[i-s_p]) - sum_p softmax_p * h[i+s_p]

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _global_target

def _cons_jac(h):
    return np.ones(_global_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Fix if still off
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[:final_sum - target_sum]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def focused_greedy(h, rng, time_limit, start_time, max_iters=50000):
    """Greedy local search focused on reducing the maximum correlation."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        
        p_star = int(np.argmax(corr))
        s_star = p_star - (n - 1)
        max_val = float(corr[p_star])
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute A(i) for all ones, B(j) for all zeros
        i_arr = ones
        j_arr = zeros
        
        idx_im = i_arr - s_star
        idx_ip = i_arr + s_star
        valid_im = (idx_im >= 0) & (idx_im < n)
        valid_ip = (idx_ip >= 0) & (idx_ip < n)
        
        A = np.zeros(len(i_arr))
        if np.any(valid_im):
            A[valid_im] -= (1.0 - h[idx_im[valid_im]])
        if np.any(valid_ip):
            A[valid_ip] += h[idx_ip[valid_ip]]
        
        idx_jm = j_arr - s_star
        idx_jp = j_arr + s_star
        valid_jm = (idx_jm >= 0) & (idx_jm < n)
        valid_jp = (idx_jp >= 0) & (idx_jp < n)
        
        B = np.zeros(len(j_arr))
        if np.any(valid_jm):
            B[valid_jm] += (1.0 - h[idx_jm[valid_jm]])
        if np.any(valid_jp):
            B[valid_jp] -= h[idx_jp[valid_jp]]
        
        # Try top candidates
        # Sort by A (ascending) and B (ascending)
        top_i_count = min(20, len(i_arr))
        top_j_count = min(20, len(j_arr))
        
        top_i = np.argsort(A)[:top_i_count]
        top_j = np.argsort(B)[:top_j_count]
        
        found_improve = False
        best_local_c5 = max_val
        
        for ii in top_i:
            if found_improve:
                break
            for jj in top_j:
                i = i_arr[ii]
                j = j_arr[jj]

                    continue  # won't help at p_star

                h[i] = 0
                h[j] = 1

                    best_local_c5 = new_max

                    stall = 0
                    break
                else:
                    # Revert
                    h[i] = 1
                    h[j] = 0
        
        if not found_improve:

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1

                best_c5 = new_max
                best_h = h.copy()
                stall = 0
            else:
                h[i] = 1
                h[j] = 0
                stall += 1

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def quasi_random_init(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for i in range(ones_count):
        frac = ((i + 0.5) / golden) % 1.0
        pos = int(frac * n) % n
        positions.add(pos)
    while len(positions) < ones_count:
        positions.add(rng.randint(n))
    h[list(positions)] = 1.0
    return h
