import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_project(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        if free.any():
            result[free] += diff / float(free.sum())
            result = np.clip(result, 0.0, 1.0)
    return result


def grad_single(h, lag_idx):
    n = len(h)
    L = lag_idx - (n - 1)
    g = np.zeros(n)
    if 0 <= L < n:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    elif -n < L < 0:
        m = -L
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g


def grad_multi(h, indices, weights):
    n = len(h)
    g = np.zeros(n)
    for i in range(len(indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        L = indices[i] - (n - 1)
        if 0 <= L < n:
            g[L:] += w * (1.0 - h[:n - L])
            g[:n - L] -= w * h[L:]
        elif -n < L < 0:
            m = -L
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] -= w[:n - m]
    return g


def slp_refine(h, target_sum, topk, max_iters, deadline):
    best_h = h.copy()
    best_score = compute_c5(h)
    n = len(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = overlap_corrs(best_h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        m = len(top_idx)
        A = np.zeros((m, n + 1))
        b = np.zeros(m)
        for i, idx in enumerate(top_idx):
            g = grad_single(best_h, idx)
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ best_h - corr[idx]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_lp = res.x[:n]
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            h_try = alpha * h_lp + (1.0 - alpha) * best_h
            s = compute_c5(h_try)
            if s < best_score - 1e-13:
                best_h = h_try
                best_score = s
                improved = True
                break
        if not improved:
            break
    return best_h, best_score


def slp_adaptive(h, target_sum, deadline):
    best_h = h.copy()
    best_score = compute_c5(h)
    n = len(h)
    topk_schedule = [80, 60, 40, 25, 15, 10, 5, 3]
    iters_per = [25, 40, 50, 60, 60, 50, 40, 30]
    for li, topk in enumerate(topk_schedule):
        n_iters = iters_per[li] if li < len(iters_per) else 30
        best_h, best_score = slp_refine(best_h, target_sum, topk, n_iters, deadline)
    return best_h, best_score


def subgrad_nesterov(h, target_sum, max_iter, lr_init, topk, alpha_sm, momentum, patience, deadline, rng):
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = overlap_corrs(y)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        top_vals = corr[top_idx]
        w = np.exp(alpha_sm * (top_vals - np.max(top_vals)))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum
        g = grad_multi(y, top_idx, w)
        v = momentum * v + (1.0 - momentum) * g
        y_new = y - cur_lr * v
        if rng is not None and rng.random() < 0.02:
            y_new = y_new + rng.randn(n) * 0.005
        y_new = exact_project(y_new, target_sum)
        s = compute_c5(y_new)
        if s < best_score - 1e-14:
            best_score = s
            best_h = y_new.copy()
            stagnation = 0
            cur_lr = min(cur_lr * 1.015, 1.5)
        else:
            stagnation += 1
            cur_lr *= 0.985
        y = y_new
        if stagnation > patience:
            break
    return best_h, best_score


def upsample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)


def make_sine_start(n, target, freq, rng):
    t = np.arange(n) / n
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    h = h + rng.randn(n) * 0.05
    return exact_project(h, target)


def make_bimodal_start(n, target, rng):
    h = np.zeros(n)
    width = n // 4
    pos1 = rng.randint(0, n // 2)
    pos2 = rng.randint(n // 2, n - width)
    h[pos1:pos1 + width] = 1.0
    h[pos2:pos2 + width] = 1.0
    h = h + rng.randn(n) * 0.01
    return exact_project(h, target)
