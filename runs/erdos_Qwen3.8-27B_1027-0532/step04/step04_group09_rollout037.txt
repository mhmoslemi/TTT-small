import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def fista_optimize(h0, target, alpha, n_iter, step, deadline):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_obj = float('inf')
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = lse_obj_and_grad(y, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_new
        y_new = project_feasible(h + mom * (h - h_prev) - step * grad, target)
        h_prev = h
        h = y_new
        
        # Check actual objective (not LSE)
        actual = compute_c5(h)
        if actual < best_obj:
            best_obj = actual
            best_h = h.copy()
        
        t = t_new
        y = h + mom * (h - h_prev)
    
    return best_h, best_obj

def slsqp_optimize(h, target, alpha, maxiter, ftol, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    
    def cfun(x):
        return np.sum(x) - target
    
    def cjac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        cur = compute_c5(h)
        if sc < cur:
            return h_new, sc
    except Exception:
        pass
    return h, compute_c5(h)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

Hmm, actually I want to double-check one thing: the L-BFGS-B with penalty approach. The issue is that as mu (penalty parameter) increases, the problem becomes ill-conditioned. L-BFGS-B might struggle with very large mu.

A better approach: use an **augmented Lagrangian** method.

minimize f(h) + mu * (sum(h) - target) + (rho/2) * (sum(h) - target)^2

where mu is the Lagrange multiplier (updated each outer iteration) and rho is the penalty parameter.

The augmented Lagrangian subproblem is still box-constrained (solvable by L-BFGS-B), and the outer loop updates mu = mu + rho * (sum(h) - target).

This converges much better than pure penalty.

```python
def aug_lagrangian_optimize(h0, target, alpha, rho, n_outer, deadline):
    n = len(h0)
    h = h0.copy()
    mu = 0.0
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        
        def fun(x):
            base, grad = lse_obj_and_grad(x, alpha)
            s = np.sum(x) - target
            return base + mu * s + 0.5 * rho * s * s
        
        def jac(x):
            base, grad = lse_obj_and_grad(x, alpha)
            s = np.sum(x) - target
            grad = grad + (mu + rho * s) * np.ones(n)
            return grad
        
        try:
            res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = np.clip(res.x, 0.0, 1.0)
            s = np.sum(h_new) - target
            mu = mu + rho * s
            h = h_new
            if abs(s) < 1e-10:
                break
        except Exception:
            pass
    
    h = project_feasible(h, target)
    return h, compute_c5(h)
```

This is better than pure penalty because the Lagrange multiplier mu tracks the constraint, so rho doesn't need to be as large.

Typical rho values: 1e2 to 1e4. With n_outer = 10-20 and 200 L-BFGS-B iterations each, this should converge well.

import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + lam, 0.0, 1.0))
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    h_proj = np.clip(h + lam, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_proj < 0.9999999
        else:
            mask = h_proj > 1e-07
        cnt = int(mask.sum())
        if cnt > 0:
            h_proj = h_proj.copy()
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    best_h = h.copy()
    best_score = compute_c5(h)
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})

        sc = compute_c5(h_new)
        if sc < best_score:
            best_h = h_new
            best_score = sc
    except Exception:
        pass
    return best_h, best_score
