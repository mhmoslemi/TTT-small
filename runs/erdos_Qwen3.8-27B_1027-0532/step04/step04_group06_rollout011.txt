def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

def upsample(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)

def downsamples(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.clip(np.round(h), 0, 1).astype(float)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Verify
    while h_r.sum() > target_sum:
        ones = np.where(h_r == 1)[0]
        h_r[ones[np.argmax(h[ones])]] = 0  # flip the one with highest h
    while h_r.sum() < target_sum:
        zeros = np.where(h_r == 0)[0]
        h_r[zeros[np.argmin(h[zeros])]] = 1  # flip the zero with lowest h

            # For each k, d corr[k]/dh[i] = (1-h[i+k-n+1])*1_{valid} - h[i+k-n+1-i+... 
            # This is complex. Let's use numerical gradient or skip.
            return None  # Let SLSQP use numerical gradient

def upsample(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)

def subgradient_refine(h_start, target, max_iters, step_init, step_final, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ratio = i / max(max_iters - 1, 1)
        step = step_init * (step_final / step_init) ** ratio
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        center = n - 1
        worst_lags = find_worst_lags(corr, n, top_k=3)
        
        grad = np.zeros(n)
        for k in worst_lags:
            k = int(k)
            g = np.zeros(n)
            for i_idx in range(n):
                val = 0.0
                if 0 <= i_idx - k < n:
                    val += (1.0 - h[i_idx - k])
                if 0 <= i_idx + k < n:
                    val -= h[i_idx + k]
                g[i_idx] = val
            grad += g
        grad /= len(worst_lags)
        
        h_new = h - step * grad * dx
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
