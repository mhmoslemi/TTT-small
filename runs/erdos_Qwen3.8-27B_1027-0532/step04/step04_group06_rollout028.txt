```python
def project_box_sum(v, S):
    """Project v onto {x: 0 <= x[i] <= 1, sum(x) = S}."""
    n = len(v)
    # Bisection on tau: find tau such that sum(clamp(v - tau, 0, 1)) = S
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)
```
