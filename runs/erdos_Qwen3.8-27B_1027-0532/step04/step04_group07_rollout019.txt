import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(50):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def active_set_smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)

def slsqp_active_set(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                active_set_smooth_obj, h_cur, args=(alpha, topk),
                jac=active_set_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def lp_linearized_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None

def lp_refinement(h_start, target_sum, max_iterations, topk, time_limit=None):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    for it in range(max_iterations):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        h_lp = lp_linearized_step(h, target_sum, topk)
        if h_lp is None:
            break
        score = compute_c5(h_lp)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stagnation = 0
        else:
            h_half = project_feasible(0.5 * h + 0.5 * h_lp, target_sum)
            sc_half = compute_c5(h_half)
            if sc_half < best_score - 1e-15:
                best_score = sc_half
                best_h = h_half.copy()
                h = h_half
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 15:
                    break
    return best_h, best_score

def run_subgradient_rmsprop(h_init, target_sum, max_iter, lr_init, topk, alpha,
                            noise_prob, noise_scale, patience, momentum, seed,
                            time_limit=None):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    sq_grad = np.zeros(n)  # RMSProp second moment
    beta2 = 0.9
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    t0 = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        sq_grad = beta2 * sq_grad + (1.0 - beta2) * (v ** 2)
        h_new = h - lr * v / (np.sqrt(sq_grad) + eps)
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(50):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:

                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag

                grad[m:] -= w * h[:n - m]
    return grad

    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:

    else:
        top_idx = np.arange(n_corr)

**Idea: Use a quadratic trust region approach.**

Instead of linearizing, use a second-order approximation:
C_k(h + Δ) ≈ C_k(h) + ∇C_k·Δ + Δ^T H_k Δ

where H_k is the Hessian of C_k with respect to h.

For C_k(h) = ∑ h[i](1-h[i+lag]), the Hessian is tridiagonal (only involves pairs (i, i+lag)). This is a sparse Hessian!

But implementing a full trust region method with this sparse Hessian is complex. Let me think of something simpler.

**Idea: Use the fact that the LP gives a lower bound on the objective increase.**

When we solve the LP, we get:

        return h_lp

        h_bt = project_feasible(h_cur + alpha * (h_lp - h_cur), target_sum)
        s_bt = compute_c5(h_bt)
        if s_bt < s_cur:
            return h_bt
    return h_cur

import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(50):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def _cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def _cons_eq_jac(x):
    return np.ones(_CONS_N)


def _smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def _smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)


def slsqp_phase(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                _smooth_obj, h_cur, args=(alpha, topk),
                jac=_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def lp_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target_sum)
    return None

    best_h = h.copy()

def rmsprop_subgrad(h_init, target_sum, max_iter, lr, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed,
                    time_limit=None):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    sq = np.zeros(n)
    beta2 = 0.9
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    stag = 0
    t0 = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            ti = np.argpartition(corr, -topk)[-topk:]
        else:
            ti = np.arange(n_corr)
        tv = corr[ti]
        w = np.exp(alpha * (tv - np.max(tv)))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(ti)) / len(ti)
        else:
            w = w / ws
        g = compute_grad_multi_lag(h, ti, w)
        v = momentum * v + (1.0 - momentum) * g
        sq = beta2 * sq + (1.0 - beta2) * (v * v)
        h_new = h - lr * v / (np.sqrt(sq) + eps)
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()
            stag = 0
            lr = min(lr * 1.03, 5.0)
        else:
            stag += 1
            lr *= 0.97
        h = h_new
        if stag > patience:
            break
    return best_h, best_score

def interp_res(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, n_dst / 2.0)

                             time_limit=min(100, _rem() - 250))

            best_score = sc
            best_h = h_new

    # Another full LP pass
    if _rem() > 250:
        h_new, sc = full_lp_loop(best_h, target, 5000,
                                  time_limit=min(80, _rem() - 200))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: SLSQP active set refinement
    if _rem() > 200:
        alphas = [(100, 50), (500, 30), (2000, 15)]
        h_ref, sc = slsqp_phase(best_h, target, alphas, 3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Perturb and refine (escape local minima)
    if _rem() > 180:
        h_new, sc = perturb_and_refine(best_h, target, 20, 0.005, 40,
                                        time_limit=min(40, _rem() - 140),
                                        seed=seed + 500)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_new, sc = perturb_and_refine(best_h, target, 20, 0.002, 20,
                                        time_limit=min(40, _rem() - 100),
                                        seed=seed + 600)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Try higher resolution
    if _rem() > 180:
        n_high = 500
        target_high = n_high / 2.0
        h_hi = interp_res(best_h, n_high)
        h_hi, sc_hi = rmsprop_subgrad(h_hi, target_high, 3000, 0.2, 5,
                                       30, 0.02, 0.01, 200, 0.9, seed + 700,
                                       time_limit=min(40, _rem() - 130))
        h_hi, sc_hi = lp_loop(h_hi, target_high, 3000, 60,
                               time_limit=min(40, _rem() - 90))
        h_hi, sc_hi = slsqp_phase(h_hi, target_high, [(200, 40), (1000, 20)], 2000, 1e-15)
        h_hi, sc_hi = full_lp_loop(h_hi, target_high, 3000,
                                    time_limit=min(30, _rem() - 60))
        if sc_hi < best_score:
            best_score = sc_hi
            best_h = h_hi

    if _rem() > 120:
        n_high2 = 600
        target_high2 = n_high2 / 2.0
        h_hi2 = interp_res(best_h, n_high2)
        h_hi2, sc_hi2 = rmsprop_subgrad(h_hi2, target_high2, 2000, 0.15, 5,
                                         30, 0.02, 0.01, 150, 0.9, seed + 800,
                                         time_limit=min(30, _rem() - 80))
        h_hi2, sc_hi2 = lp_loop(h_hi2, target_high2, 2000, 50,
                                 time_limit=min(30, _rem() - 40))
        h_hi2, sc_hi2 = full_lp_loop(h_hi2, target_high2, 2000,
                                      time_limit=max(0, _rem() - 20))
        if sc_hi2 < best_score:
            best_score = sc_hi2
            best_h = h_hi2

    # Phase 7: Final polish at best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0

    if _rem() > 60:
        h_new, sc = full_lp_loop(best_h, target_use, 5000,
                                  time_limit=min(30, _rem() - 40))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if _rem() > 40:
        alphas_final = [(5000, 10), (20000, 5)]
        h_ref, sc = slsqp_phase(best_h, target_use, alphas_final, 5000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _rem() > 20:
        h_new, sc = lp_loop(best_h, target_use, 2000, 10,
                             time_limit=max(0, _rem() - 10))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
