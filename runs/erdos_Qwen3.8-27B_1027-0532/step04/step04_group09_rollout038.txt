Wait, I realize the gradient computation in the previous code might have an issue. Let me verify.

The objective is C₅(h) = max_k ∑_i h[i] * (1-h[i+k]) (times dx, but that's a constant factor).

For a specific lag k (k ≥ 0):
f_k(h) = ∑_{i=0}^{n-1-k} h[i] * (1 - h[i+k])

∂f_k/∂h[j] for a specific j:
- If j appears as h[i] (first factor): i = j, so we need j+k < n, contribution: (1-h[j+k])
- If j appears as h[i+k] (second factor): i+k = j, so i = j-k, we need j-k ≥ 0, contribution: -h[j-k]

So ∂f_k/∂h[j] = (1-h[j+k]) * 1[j+k < n] - h[j-k] * 1[j-k ≥ 0]

For k < 0, let k' = -k:
f_k(h) = ∑_{i=-k}^{n-1} h[i] * (1 - h[i+k]) = ∑_{i=k'}^{n-1} h[i] * (1 - h[i-k'])

∂f_k/∂h[j]:
- As first factor (h[i]): i = j, need j ≥ k', contribution: (1-h[j-k'])
- As second factor (h[i-k']): i-k' = j, so i = j+k', need j+k' < n, contribution: -h[j+k']

So ∂f_k/∂h[j] = (1-h[j-k']) * 1[j ≥ k'] - h[j+k'] * 1[j+k' < n]

  So grad[j] += wi * (1-h[j-k]) for j ≥ k.

Which matches my derivation for k<0 (with k'=kk). Good.

def slsqp_optimize(h, target, alpha, maxiter=2000, ftol=1e-16):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    
    def cfun(x):
        return np.sum(x) - target
    
    def cjac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:

            break
        h_new, sc = slsqp_optimize(h_cur, target, alpha, maxiter, ftol)
        if sc < cur_score:
            cur_score = sc
            h_cur = h_new
    return h_cur, cur_score

    n = len(h)
    corr = compute_corr(h)
    # Get top n_top worst lags

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):

        c_max = float(np.max(corr))
        c_min = float(np.min(corr))

        grad = np.zeros(n)
        for idx in active:

            if k >= 0:

                    grad[k:] += (1.0 - h_cur[:n - k])

def generate_init_cosine(n, rng):
    """Generate a smooth initialization using random cosine series"""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for j in range(1, 8):
        amp = rng.uniform(0.02, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * j * x / 2 + phase)
    h = h - np.mean(h)  # zero mean perturbation
    h = 0.5 + h
    return project_feasible(h, n / 2.0)

def generate_init_bump(n, rng):
    """Generate a bump-like initialization"""
    x = np.linspace(0, 2, n)
    center = 1.0
    width = rng.uniform(0.3, 0.8)
    h = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    h = h - np.min(h)
    h = 0.5 + 0.4 * (h - np.mean(h))
    return project_feasible(h, n / 2.0)

def generate_init_two_peak(n, rng):
    """Generate a two-peak initialization"""
    x = np.linspace(0, 2, n)
    c1 = rng.uniform(0.3, 0.7)
    c2 = 2.0 - c1
    w = rng.uniform(0.15, 0.4)
    h = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    h = 0.5 + 0.3 * (h - np.mean(h))
    return project_feasible(h, n / 2.0)

def generate_init_symmetric(n, rng):
    """Generate a symmetric initialization h(x) = h(2-x)"""
    half = n // 2
    x = np.linspace(0, 1, half + 1)
    h_half = 0.5
    for j in range(1, 6):
        amp = rng.uniform(0.02, 0.12)
        h_half += amp * np.cos(np.pi * j * x)
    h_half = project_feasible(h_half, (half + 1) / 2.0)
    if n % 2 == 0:
        h = np.concatenate([h_half, h_half[-2::-1]])
    else:
        h = np.concatenate([h_half, h_half[-1::-1]])
    return project_feasible(h, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_one(h, target, alpha, maxiter, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur_score = compute_c5(h)

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new, sc
    except Exception:
        pass
    return h, cur_score


def slsqp_sequence(h, target, alpha_list, maxiter, ftol, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_cur, cur_score = slsqp_one(h_cur, target, alpha, maxiter, ftol)
    return h_cur, cur_score


def active_subgrad(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        n_act = max(len(active), 1)
        grad /= n_act
        step = step_size / (1.0 + 0.003 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score

                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
