            best_score = c5_r
            best_h = h_r

    # Phase 8: Final L-BFGS-B with very high alpha
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas_fin2 = [5000, 10000, 20000]
        time_fin2 = min(80.0, tleft() * 0.7)
        h_r, c5_r = lsbfgs_refine(best_h, n_use, target_use, time_fin2, start_time, alphas_fin2)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r

    # Phase 9: Final Nesterov polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_r, c5_r = nesterov_refine(best_h, n_use, target_use, min(40.0, tleft() * 0.8), start_time)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_exact(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
