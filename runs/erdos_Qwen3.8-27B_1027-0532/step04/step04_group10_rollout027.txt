Let me also think about the gradient computation. The gradient of the correlation at lag k with respect to h is:
- For lag ≥ 0: grad[i] += (1-h[i+k]) for i < n-k, grad[i+k] -= h[i] for i < n-k
- For lag < 0 (i.e., -k > 0): grad[i] += (1-h[i-k]) for i ≥ k, grad[i-k] -= h[i] for i ≥ k

This can be computed in O(n) per lag. If we average over K lags, it's O(Kn).

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return h
    if abs(np.sum(h) - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(clipped)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h_new = np.clip(h - tau, 0.0, 1.0)
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-13:
        non_bound = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        cnt = np.sum(non_bound)
        if cnt > 0:
            h_new[non_bound] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new


def grad_corr_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            m = n - lag
            grad[:m] += (1.0 - h[lag:])
            grad[lag:] -= h[:m]
    else:
        k = -lag
        if k < n:
            m = n - k
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    return grad


def lse_objective_and_grad(h, alpha, topk, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    s = np.sum(h) - target
    obj += penalty * s * s
    weights = exp_vals / total
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = int(idx[i]) - (n - 1)
        grad += w * grad_corr_at_lag(h, lag)
    grad += 2.0 * penalty * s
    return obj, grad


def lbfgs_optimize(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol):
    def obj_func(x):
        return lse_objective_and_grad(x, alpha, topk, target, penalty)
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 20})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def subgradient_step(h, target, topk, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_actual = min(topk, len(corr))
    idx = np.argpartition(corr, -topk_actual)[-topk_actual:]
    grad = np.zeros(n)
    for ki in idx:
        lag = int(ki) - (n - 1)
        grad += grad_corr_at_lag(h, lag)
    grad /= topk_actual
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, target)
    return h_new


def momentum_descent(h_start, target, alpha0, n_iters, momentum=0.85):
    n = len(h_start)
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha = alpha0
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk = min(15, len(corr))
        idx = np.argpartition(corr, -topk)[-topk:]
        grad = np.zeros(n)
        for ki in idx:
            lag = int(ki) - (n - 1)
            grad += grad_corr_at_lag(h, lag)
        grad /= topk
        v = momentum * v - alpha * grad
        h_new = project_feasible(h + v, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        alpha *= 0.998
    return best_h, best_score


def make_block_start(n, target, pattern):
    h = np.zeros(n)
    k = int(target)
    if pattern == 0:
        h[:k] = 1.0
    elif pattern == 1:
        h[:k//2] = 1.0
        h[n - k//2:] = 1.0
        if k % 2 == 1:
            h[k//2] = 1.0
    elif pattern == 2:
        h[::2][:k] = 1.0
    elif pattern == 3:
        h[n//3:2*n//3] = 1.0
        remaining = k - int(np.sum(h))
        if remaining > 0:
            h[:remaining] = 1.0
        elif remaining < 0:
            h[n//3:n//3+(-remaining)] = 0.0
    else:
        idx = np.random.choice(n, k, replace=False)
        h[idx] = 1.0
    return project_feasible(h, target)
