def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

    k = 1.0
    y = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        t_now = it / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t_now

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    for _ in range(200):
        if diff > 0:
            mask = h < 1.0
            slack = np.sum(1.0 - h[mask])
        else:
            mask = h > 0.0
            slack = np.sum(h[mask])
        if slack < 1e-15:
            break
        if abs(diff) <= slack:
            h[mask] += diff / np.sum(mask.astype(float))
            h = np.clip(h, 0.0, 1.0)
            break
        else:
            h = h.copy()
            h[mask] = 1.0 if diff > 0 else 0.0
            diff = target_sum - np.sum(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        if (diff > 0 and adj >= 1.0 - h[mask].min()) or (diff < 0 and adj <= h[mask].max()):
            h[mask] = 1.0 if diff > 0 else 0.0
        else:
            h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_smoothed(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def fista_optimize(h_init, target, alpha_start, alpha_end, deadline,
                   max_iters=200000, step_size=0.49, restart_period=2000):
    n = len(h_init)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    h_cur = h_init.copy()
    y = h_init.copy()
    k_mom = 1.0
    momentum = 0.0
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_frac = min(it / max(max_iters - 1, 1), 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** t_frac

        if it > 0 and it % restart_period == 0:
            k_mom = 1.0
            y = h_cur.copy()

        obj, grad = grad_smoothed(y, alpha)

        h_new = y - step_size * grad
        h_new = project_feasible(h_new, target)

        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_mom * k_mom))
        y = h_new + ((k_mom - 1.0) / k_new) * (h_new - h_cur)
        y = project_feasible(y, target)

        h_cur = h_new
        k_mom = k_new

        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            k_mom = 1.0
            y = h_new.copy()
            no_improve = 0

    return best_h, best_score


def slsqp_single(h, target, alpha, maxiter=500, ftol=1e-15):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return grad_smoothed(x, alpha)[0]

    def jac(x):
        return grad_smoothed(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def interpolate_to(h_src, n_dst, target_dst):
    x_old = np.linspace(0.0, 1.0, len(h_src))
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def active_set_polish(h_init, target, n_steps, step_size, deadline):
    n = len(h_init)
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        span = c_max - c_min
        if span < 1e-12:
            break
        threshold = c_max - 0.01 * span
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= max(len(active), 1)

        lr = step_size / (1.0 + 0.01 * it)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    return best_h, best_score


def multi_start_try(h_base, target, n_attempts, alpha, deadline, rng):
    n = len(h_base)
    best_h = h_base.copy()
    best_score = compute_c5(best_h)

        noise = rng.normal(0, 0.01, n)

            best_h = h_try
