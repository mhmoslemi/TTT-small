def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_coord(h, i, n, grid_size=5000):
    """Solve the 1D subproblem for coordinate i."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    old_val = h[i]
    
    # Compute A and B for each lag
    # Lag l ranges from -(n-1) to (n-1), correlation index is n-1+l
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    
    for l in range(-(n-1), n):
        corr_idx = n - 1 + l
        if l == 0:
            # corr_0(x) = base_0 + x - x^2
            base_0 = corr[corr_idx] - old_val * (1 - old_val)
            A[corr_idx] = base_0
            B[corr_idx] = 1.0  # coefficient of x
            # C (coefficient of x^2) is -1, stored separately
        else:
            # corr_l(x) = A_l + B_l * x
            r = 1.0 if (0 <= i + l < n) else 0.0  # h[i] as first factor
            s = 1.0 if (0 <= i - l < n) else 0.0  # h[i] in 1-h second factor
            
            if r > 0:
                h_right = h[min(i + l, n-1)] if (0 <= i+l < n) else 0
            if s > 0:
                h_left = h[max(i - l, 0)] if (0 <= i-l < n) else 0
            
            # A_l = base_l + h[i-l]*s (for l≠0)
            # base_l = corr[corr_idx] - old_val*(1-h[i+l])*r - h[i-l]*(1-old_val)*s
            contrib = 0.0
            if r > 0:
                contrib += old_val * (1 - h[i+l])
            if s > 0:
                contrib += h[i-l] * (1 - old_val)
            base_l = corr[corr_idx] - contrib
            
            A[corr_idx] = base_l + (h[i-l] if s > 0 else 0.0)
            B[corr_idx] = ((1 - h[i+l]) if r > 0 else 0.0) - (h[i-l] if s > 0 else 0.0)
    
    # Now solve: minimize max(A[center] + x - x², max_{l≠0}(A[l] + B[l]*x)) over x∈[0,1]
    center = n - 1
    A_0 = A[center]
    # g_0(x) = A_0 + x - x²
    
    # Linear functions for l≠0
    mask = np.ones(2*n-1, dtype=bool)
    mask[center] = False
    A_lin = A[mask]
    B_lin = B[mask]
    
    x_grid = np.linspace(0.0, 1.0, grid_size)
    val_quad = A_0 + x_grid - x_grid**2
    # val_lin[j] = max_k (A_lin[k] + B_lin[k] * x_grid[j])
    val_lin = np.max(A_lin[None, :] + x_grid[:, None] * B_lin[None, :], axis=1)
    f_grid = np.maximum(val_quad, val_lin)
    
    best_idx = np.argmin(f_grid)
    x_star = x_grid[best_idx]
    
    return x_star

def coord_descent_pass(h, n, grid_size=5000, order=None, time_limit=None, start_time=None):
    """One pass of coordinate descent."""
    if order is None:
        order = np.arange(n)

        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        x_star = solve_1d_coord(h, i, n, grid_size)
        h[i] = x_star
    
    return h

    A[:] = base + np.where(s, h_left, 0.0)

    # For l=0: A_0 = base_0, B_0 = 1 (for x - x^2 term)
    A[center] = base[center]
    B[center] = 1.0
    
    # Solve 1D problem
    A_0 = A[center]
    mask = np.ones(2*n-1, dtype=bool)
    mask[center] = False
    A_lin = A[mask]
    B_lin = B[mask]
    
    x_grid = np.linspace(0.0, 1.0, grid_size)
    val_quad = A_0 + x_grid - x_grid**2
    val_lin = np.max(A_lin[None, :] + x_grid[:, None] * B_lin[None, :], axis=1)
    f_grid = np.maximum(val_quad, val_lin)
    
    best_idx = np.argmin(f_grid)
    return x_grid[best_idx]

import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_coord(h, i, n, x_grid):
    """Solve the 1D subproblem for coordinate i given the grid."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    old_val = h[i]
    
    lags = np.arange(-(n-1), n)
    i_plus_l = i + lags
    i_minus_l = i - lags
    
    r = (i_plus_l >= 0) & (i_plus_l < n)
    s = (i_minus_l >= 0) & (i_minus_l < n)
    
    # Avoid out-of-bounds access
    i_plus_l_clipped = np.clip(i_plus_l, 0, n-1)
    i_minus_l_clipped = np.clip(i_minus_l, 0, n-1)
    h_right = h[i_plus_l_clipped]
    h_left = h[i_minus_l_clipped]
    
    # Current contribution of h[i] to each correlation
    contrib = np.where(r, old_val * (1.0 - h_right), 0.0) + np.where(s, h_left * (1.0 - old_val), 0.0)
    center = n - 1
    contrib[center] = old_val * (1.0 - old_val)  # fix double counting at l=0

    A = base + np.where(s, h_left, 0.0)

    mask[center] = False
    A_lin = A[mask]
    B_lin = B[mask]
    
    # Evaluate f(x) on grid
    A_0 = base[center]
    val_quad = A_0 + x_grid - x_grid**2
    val_lin = np.max(A_lin[None, :] + x_grid[:, None] * B_lin[None, :], axis=1)
    f_grid = np.maximum(val_quad, val_lin)
    
    return x_grid[np.argmin(f_grid)], f_grid[np.argmin(f_grid)]

def coord_descent(h, n, max_passes, grid_size, time_limit, start_time, tol=1e-10):
    """Run coordinate descent for the C5 minimization."""
    x_grid = np.linspace(0.0, 1.0, grid_size)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(0)
    
    for pass_num in range(max_passes):
        if time.time() - start_time > time_limit:
            break
        
        # Random order for stochastic coordinate descent
        order = rng.permutation(n)
        
        max_improvement = 0.0
        
        for i in order:
            x_star, f_val = solve_1d_coord(h, i, n, x_grid)
            h[i] = x_star
            max_improvement = max(max_improvement, abs(x_star - h[i]))
        
        # Project to maintain sum constraint
        h = project_feasible(h, n / 2.0)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
        
        if max_improvement < tol:
            break
    
    return h, best_c5

def smooth_obj(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def slsqp_refine(h, n, target, alphas, maxiter, time_limit, start_time):
    """Refine using SLSQP with log-sum-exp smooth approximation."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, target), h,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
