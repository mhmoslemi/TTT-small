from scipy.optimize import minimize, linprog
import time

_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

# Cutting-plane LP
def lp_cutting_plane(h_start, target_sum, max_iterations, time_limit):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    
    # Initial cuts: top lags
    corr = overlap_corrs(h)
    n_corr = len(corr)
    initial_k = min(100, n_corr)
    top_idx = np.argpartition(corr, -initial_k)[-initial_k:]
    
    cuts = list(top_idx)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    t0 = time.time()
    
    for it in range(max_iterations):
        if time.time() - t0 > time_limit:
            break
        
        # Build LP
        m = len(cuts)
        A = np.zeros((m, n + 1))
        b = np.zeros(m)
        
        for i, idx in enumerate(cuts):
            g = compute_grad_single_lag(h, idx)
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[idx]
        
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        
        if not res.success:
            break
        
        h_new = project_feasible(res.x[:n], target_sum)
        score = compute_c5(h_new)
        
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        # Check for new cuts
        corr_new = overlap_corrs(h_new)
        max_val = np.max(corr_new)

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)

            free = h < 1.0 - 1e-14

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:

            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        return h

            free = h < 1.0 - 1e-14

        else:
            free = h > 1e-14

    n = len(h)

    else:

        b_eq = np.array([target_sum])
        bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if not res.success:
            if m > 15:
                keep = np.argsort(np.abs(b))[:max(15, m * 2 // 3)]
                cut_set = set([cut_list[i] for i in keep])
            stagnant += 2
            if stagnant > 15:
                return best_h, best_score
            continue
        h_new = project_feasible(res.x[:n], target_sum)
        score = compute_c5(h_new)
        improved = False
        if score < best_score - 1e-11:
            best_score = score
            best_h = h_new.copy()
            improved = True
        corr_new = overlap_corrs(h_new)
        t_lp = res.x[n]
        max_idx = int(np.argmax(corr_new))
        if corr_new[max_idx] > t_lp + 0.3:
            if max_idx not in cut_set:
                cut_set.add(max_idx)
                improved = True
        if len(cut_set) < max_cuts:
            excess = corr_new - t_lp
            bad = np.where(excess > 1.5)[0]
            for idx in bad[:8]:
                if int(idx) not in cut_set:
                    cut_set.add(int(idx))
                    improved = True
        h = h_new
        corr = corr_new
        if not improved:
            stagnant += 1
            if stagnant > 50:
                break
        else:
            stagnant = 0
    return best_h, best_score


def local_search(h_init, target_sum, max_passes, time_limit, delta_list):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    for p in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit:
                return best_h, best_score
            orig = h[j]
            for delta in delta_list:
                new_val = orig + delta
                if new_val < -0.01 or new_val > 1.01:
                    continue
                h[j] = max(0.0, min(1.0, new_val))
                h_try = project_feasible(h.copy(), target_sum)
                score = compute_c5(h_try)
                if score < best_score - 1e-13:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    orig = h[j]
        if not improved:
            break
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    subgrad_configs = [
        (0.08, 5, 30, 0.90, 2000, 300),
        (0.04, 10, 80, 0.85, 2000, 300),
        (0.15, 3, 15, 0.92, 1500, 200),
        (0.02, 15, 150, 0.80, 1500, 250),
    ]

    for i, cfg in enumerate(subgrad_configs):
        if _remaining() < 480:
            break
        h_try = best_h + rng.randn(n_init) * 0.004
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                        max_iter=cfg[5], lr_init=cfg[0], topk=cfg[1],
                                        alpha=cfg[2], patience=cfg[4],
                                        momentum=cfg[3], seed=seed + i * 7,
                                        noise_prob=0.02, noise_scale=0.003)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 250:
        h_new, score = lp_cutting_plane(best_h, target_init,
                                         max_iterations=10000,
                                         time_limit=_remaining() - 220,
                                         max_cuts=250, init_topk=80)
        if score < best_score:
            best_score = score
            best_h = h_new

    for round_num in range(5):
        if _remaining() < 120:
            break
        if _remaining() > 70:
            h_new, score = lp_cutting_plane(best_h, target_init,
                                             max_iterations=5000,
                                             time_limit=max(20, (_remaining() - 50) // 2),
                                             max_cuts=350, init_topk=150)
            if score < best_score:
                best_score = score
                best_h = h_new
        if _remaining() > 60:
            h_try = best_h + rng.randn(n_init) * 0.002
            h_try = project_feasible(h_try, target_init)
            h_new, score = run_subgradient(h_try, target_init,
                                            max_iter=1500, lr_init=0.02, topk=3,
                                            alpha=100, patience=150, momentum=0.93,
                                            seed=seed + 100 + round_num * 13)
            if score < best_score:
                best_score = score
                best_h = h_new

    if _remaining() > 60:
        h_new, score = local_search(best_h, target_init, max_passes=30,
                                     time_limit=_remaining() // 3,
                                     delta_list=[-1.0, 1.0])
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 50:
        h_new, score = local_search(best_h, target_init, max_passes=15,
                                     time_limit=min(_remaining() - 30, 40),
                                     delta_list=[-0.5, 0.5, -0.25, 0.25])
        if score < best_score:
            best_score = score
            best_h = h_new

    for n_try in [500, 600, 700]:
        if _remaining() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_new, score = run_subgradient(h_interp, target_try,
                                        max_iter=1500, lr_init=0.05, topk=5,
                                        alpha=30, patience=200, momentum=0.90,
                                        seed=seed + 2000 + n_try)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 100:
            h_new, score = lp_cutting_plane(best_h, target_try,
                                             max_iterations=5000,
                                             time_limit=_remaining() - 80,
                                             max_cuts=200, init_topk=100)
            if score < best_score:
                best_score = score
                best_h = h_new

    if _remaining() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_new, score = lp_cutting_plane(best_h, target_use,
                                         max_iterations=5000,
                                         time_limit=_remaining() - 30,
                                         max_cuts=400, init_topk=200)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_try = best_h + rng.randn(n_use) * 0.002
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient(h_try, target_use,
                                        max_iter=3000, lr_init=0.01, topk=2,
                                        alpha=200, patience=300, momentum=0.94,
                                        seed=seed + 9999)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
