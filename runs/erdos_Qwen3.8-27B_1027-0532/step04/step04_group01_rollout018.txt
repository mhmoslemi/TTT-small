```python
import numpy as np
from scipy.optimize import minimize
import time

def corr_full_fft(a, b):
    """Compute full linear cross-correlation using FFT."""
    n = len(a)
    N = 2 * n
    A = np.fft.fft(a, N)
    B = np.fft.fft(b, N)
    circ = np.fft.ifft(A * np.conj(B)).real
    neg = circ[n+1:2*n]
    pos = circ[0:n]
    return np.concatenate([neg, pos])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full_fft(h, 1.0 - h)
    return float(np.max(corr)) * dx

def delta_flip_to_one(h, a, n):
    """Compute change in correlation when h[a] goes from 0 to 1."""
    k = np.arange(-(n-1), n)  # all lags
    delta = np.zeros(2*n-1)
    # First term: (1-h[a+k]) where 0<=a+k<n
    ak = a + k
    valid1 = (ak >= 0) & (ak < n)
    delta[valid1] += (1.0 - h[ak[valid1]])
    # Second term: -h[a-k] where 0<=a-k<n and k!=0
    ak_minus = a - k
    valid2 = (ak_minus >= 0) & (ak_minus < n) & (k != 0)
    delta[valid2] -= h[ak_minus[valid2]]
    return delta

def delta_swap(h, a, b, n):
    """
    Compute change in correlation when h[a] goes 1->0 and h[b] goes 0->1.
    a and b must be different.
    h[a] should be 1, h[b] should be 0.
    """
    k = np.arange(-(n-1), n)
    
    # Delta for a going 1->0: -(1-h[a+k])*I(valid) + h[a-k]*I(valid, k!=0)
    # But we need to be careful: when computing delta for b, h[a] has already changed.
    # Actually, for the total delta, we can compute independently:
    # delta_total = delta_a(1->0) + delta_b(0->1)
    # But these interact: when computing delta for b, if b+k == a or b-k == a, h[a] has changed.
    
    # Let me just compute the "before" and "after" for the affected lags.
    # Actually, the cleanest way: compute delta assuming h is unchanged,
```

```python
    # (1-h[b+k]) where 0<=b+k<n
    bk = b + k
    mask_bk = (bk >= 0) & (bk < n)
    delta[mask_bk] += (1.0 - h[bk[mask_bk]])
    
    # h[a-k] where 0<=a-k<n and k!=0
    ak_m = a - k
    mask_ak = (ak_m >= 0) & (ak_m < n) & (k != 0)
    delta[mask_ak] += h[ak_m[mask_ak]]
    
    # -(1-h[a+k]) where 0<=a+k<n
    ak = a + k
    mask_aks = (ak >= 0) & (ak < n)
    delta[mask_aks] -= (1.0 - h[ak[mask_aks]])
    
    # -h[b-k] where 0<=b-k<n and k!=0
    bk_m = b - k
    mask_bks = (bk_m >= 0) & (bk_m < n) & (k != 0)
    delta[mask_bks] -= h[bk_m[mask_bks]]
    
    # Correction: +2 at k=a-b
    idx = (a - b) + (n - 1)
    if 0 <= idx < 2*n-1:
        delta[idx] += 2.0
    
    # Correction: -1 at k=0
    idx0 = n - 1  # index of k=0
    delta[idx0] -= 1.0
    
    return delta
```

This is O(n) per swap and fully vectorized! Great.
