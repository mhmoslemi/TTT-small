    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

```python
def proximal_subgradient_step(h, n, target, epsilon=1e-6, max_outer=60, max_inner=60):
    """One step of proximal subgradient method with exact projection."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr)) - (n - 1)
    
    # Compute subgradient
    g = np.zeros(n)
    idxs = np.arange(n)
    k = k_star
    v1 = (idxs - k)
    v2 = (idxs + k)
    m1 = (v1 >= 0) & (v1 < n)
    m2 = (v2 >= 0) & (v2 < n)
    g[m1] += (1.0 - h[v1[m1]]) * dx
    g[m2] -= h[v2[m2]] * dx
    
    # Project h onto {h : g^T h <= g^T h - eps, 0<=h<=1, sum(h)=target}
    c = g @ h - epsilon
    
    # Bisection on lambda
    lo, hi = -100.0, 100.0
    for _ in range(max_outer):
        lam = (lo + hi) / 2
        # Inner: find mu such that sum(clip(h - lam*g - mu, 0, 1)) = target
        # Bisection on mu
        mu_lo, mu_hi = h.min() - lam*g.min() - 1, h.max() - lam*g.max() + 1
        # Actually, mu should be in [min(h_i - lam*g_i) - 1, max(h_i - lam*g_i) + 1]
        shifted = h - lam * g
        mu_lo = shifted.min() - 1
        mu_hi = shifted.max() + 1
        for _ in range(max_inner):
            mu = (mu_lo + mu_hi) / 2
            h_proj = np.clip(shifted - mu, 0, 1)
            s = np.sum(h_proj)
            if s > target:
                mu_lo = mu
            else:
                mu_hi = mu
        h_proj = np.clip(shifted - mu, 0, 1)
        val = g @ h_proj
        if val > c:
            lo = lam
        else:
            hi = lam
    
    h_new = np.clip(h - lo * g - mu, 0, 1)
    h_new = project_feasible(h_new, target)
    return h_new
```

def alternating_lp_fast(h_init, n, target, max_iter, time_limit, damping=0.8):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        
        # Build LP: minimize [0, ..., 0, 1] @ [h, t]
        # subject to: A @ [h, t] <= b
        # where A rows are: [coeff_k, -1] for each k, and [0,...,0,1] for t >= 0
        # plus box constraints
        
        # Using linprog: minimize c @ x subject to A_ub @ x <= b_ub, A_eq @ x == b_eq, bounds
        x_size = n + 1  # [h_0, ..., h_{n-1}, t]
        c = np.zeros(x_size)
        c[-1] = 1.0  # minimize t
        
        # Inequality constraints: for each k, sum_i coeff_i * h_i - t <= 0
        # i.e., [coeff, -1] @ [h, t] <= 0
        A_ub_rows = []
        b_ub_vals = []
        for k in range(-(n-1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                row = np.zeros(x_size)
                row[:n] = coeff
                row[n] = -1.0
                A_ub_rows.append(row)
                b_ub_vals.append(0.0)
        
        # t >= 0 => -t <= 0
        row_t = np.zeros(x_size)
        row_t[n] = -1.0
        A_ub_rows.append(row_t)
        b_ub_vals.append(0.0)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, x_size))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                h_new = np.clip(h_new, 0, 1)
                h = (1 - damping) * h + damping * h_new
                h = project_feasible(h, target)
                c5_val = compute_c5(h)
                if c5_val < best_c5 - 1e-14:
                    best_c5 = c5_val
                    best_h = h.copy()
        except Exception:
            break
    
    return best_h, best_c5

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:

    if len(idx) == 0:
        return h
    h = h.copy()
