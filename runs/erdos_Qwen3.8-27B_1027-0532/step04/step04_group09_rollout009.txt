        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_step(h, target, alpha, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def multi_alpha_slsqp(h, target, alpha_list, maxiter, ftol, deadline):
    n = len(h)
    h_cur = h.copy()
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_cur = slsqp_step(h_cur, target, alpha, n, maxiter, ftol)
    return h_cur


def topk_gradient(h, k_frac):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    c_min = float(np.min(corr))
    spread = c_max - c_min
    if spread < 1e-14:
        return zeros_grad(n)
    threshold = c_max - k_frac * spread
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for idx in active:
        ki = idx - (n - 1)
        if ki >= 0:
            if ki < n:
                grad[:n - ki] -= h[ki:]
                grad[ki:] += (1.0 - h[:n - ki])
        else:
            kk = -ki
            if kk < n:
                grad[kk:] -= h[:n - kk]
                grad[:n - kk] += (1.0 - h[kk:])
    grad /= len(active)
    return grad


def zeros_grad(n):
    return np.zeros(n)


def fine_active_polish(h, target, n_steps, deadline, k_frac_start, k_frac_end):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step_base = 0.05 / n

    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = k_frac_start + (k_frac_end - k_frac_start) * (it / max(n_steps, 1))
        grad = topk_gradient(h_cur, frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = step_base / (1.0 + 0.01 * it)
        h_new = project_feasible(h_cur - step * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score


def backtrack_active_polish(h, target, n_steps, deadline, k_frac):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    for it in range(n_steps):
        if time.time() > deadline:
            break
        grad = topk_gradient(h_cur, k_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        cur_score = compute_c5(h_cur)
        step = 0.1
        improved = False
        for _ls in range(15):
            h_try = project_feasible(h_cur - step * grad, target)
            sc = compute_c5(h_try)
            if sc < cur_score * (1.0 - 1e-10):
                h_cur = h_try
                improved = True
                break
            step *= 0.6
        if not improved:
            step = 0.01
            h_cur = project_feasible(h_cur - step * grad, target)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score


def perturbation_restart(h, target, n_perturb, deadline, alpha_list):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h_cur) if False else compute_c5(h)
    rng_local = np.random.RandomState(12345)

    for p in range(n_perturb):
        if time.time() > deadline:
            break
        noise = rng_local.normal(0, 0.02, size=n)
        h_pert = project_feasible(h + noise, target)
        h_new = multi_alpha_slsqp(h_pert, target, alpha_list, 500, 1e-13,
                                  min(deadline, time.time() + 60))
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score


def coarse_to_fine(n_final, deadline, rng):
    levels = [24, 48, 96, 192]
    levels = [lv for lv in levels if lv < n_final]
    if not levels:
        levels = [n_final // 4]
    levels.append(n_final)

    h_cur = rng.uniform(0, 1, size=levels[0])
    h_cur = project_feasible(h_cur, levels[0] / 2.0)

    for i, lvl in enumerate(levels):
        target = lvl / 2.0
        if len(h_cur) < lvl:
            x_old = np.linspace(0, 1, len(h_cur))
            x_new = np.linspace(0, 1, lvl)
            h_cur = np.interp(x_new, x_old, h_cur)
            h_cur = project_feasible(h_cur, target)
        elif len(h_cur) > lvl:
            x_old = np.linspace(0, 1, len(h_cur))
            x_new = np.linspace(0, 1, lvl)
            h_cur = np.interp(x_new, x_old, h_cur)
            h_cur = project_feasible(h_cur, target)

        remaining = deadline - time.time()
        if remaining < 5:
            break
        alphas = [20, 100, 500, 2000]
        dl = time.time() + min(remaining * 0.8, 120)
        h_cur = multi_alpha_slsqp(h_cur, target, alphas, 400, 1e-12, dl)

    if len(h_cur) != n_final:
        x_old = np.linspace(0, 1, len(h_cur))
        x_new = np.linspace(0, 1, n_final)
        h_cur = np.interp(x_new, x_old, h_cur)
        h_cur = project_feasible(h_cur, n_final / 2.0)
    return h_cur


def block_coordinate_descent(h, target, block_size, n_rounds, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    alpha = 5000.0

    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            x0 = h_cur[s:e].copy()

            def block_obj(x):
                h_temp = h_cur.copy()
                h_temp[s:e] = np.clip(x, 0.0, 1.0)
                h_temp = project_feasible(h_temp, target)
                return lse_obj_and_grad(h_temp, alpha)[0]

            try:
                res = minimize(block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
        if time.time() > deadline:
            break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 10.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP warmup on initial (15%)
    dl1 = start_time + (budget_s - 10.0) * 0.15
    alphas_p1 = [100, 500, 2000, 10000]
    h_ref = multi_alpha_slsqp(best_h, target, alphas_p1, 1500, 1e-14, dl1)
    sc = compute_c5(h_ref)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 2: Multi-resolution exploration (20%)
    dl2 = start_time + (budget_s - 10.0) * 0.35
    if time.time() < dl2:
        h_c2 = coarse_to_fine(n, dl2, rng)
        sc2 = compute_c5(h_c2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_c2
            h_ref2 = multi_alpha_slsqp(best_h, target, [5000, 20000], 1000, 1e-14, dl2)
            sc2b = compute_c5(h_ref2)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h_ref2

    # Phase 3: SLSQP with high alpha (20%)
    dl3 = start_time + (budget_s - 10.0) * 0.55
    alphas_p3 = [20000, 50000, 150000, 400000]
    h_ref3 = multi_alpha_slsqp(best_h, target, alphas_p3, 2000, 1e-15, dl3)
    sc3 = compute_c5(h_ref3)
    if sc3 < best_score:
        best_score = sc3
        best_h = h_ref3

    # Phase 4: Fine active-set polish with backtracking (15%)
    dl4 = start_time + (budget_s - 10.0) * 0.70
    h_pol, sc4 = backtrack_active_polish(best_h, target, 3000, dl4, 0.03)
    if sc4 < best_score:
        best_score = sc4
        best_h = h_pol

    # Phase 5: Perturbation restart (10%)
    dl5 = start_time + (budget_s - 10.0) * 0.80
    h_pert, sc5 = perturbation_restart(best_h, target, 3, dl5, [5000, 20000, 80000])
    if sc5 < best_score:
        best_score = sc5
        best_h = h_pert

    # Phase 6: Block coordinate descent (10%)
    dl6 = start_time + (budget_s - 10.0) * 0.90
    h_bd, sc6 = block_coordinate_descent(best_h, target, 6, 8, dl6)
    if sc6 < best_score:
        best_score = sc6
        best_h = h_bd

    # Phase 7: Final fine polish (remaining)
    dl7 = deadline
    h_fin, sc7 = fine_active_polish(best_h, target, 5000, dl7, 0.01, 0.005)
    if sc7 < best_score:
        best_score = sc7
        best_h = h_fin

    # Final SLSQP if time remains
    if time.time() < deadline - 30:
        h_final_sl = multi_alpha_slsqp(best_h, target, [100000, 300000], 1500, 1e-15, deadline)
        sc_fin = compute_c5(h_final_sl)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_final_sl

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
