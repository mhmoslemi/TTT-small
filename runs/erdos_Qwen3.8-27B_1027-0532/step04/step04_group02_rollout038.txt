            h[i] += add
            remaining -= add

**Key SA improvement: "Correlation-aware" block moves**

For a block move (swap two blocks), we can pre-compute which block swaps would reduce the worst correlations. Specifically:

For the worst shift k*, corr[k*] = sum_j h[j]*(1-h[j+k*]).
If we swap block A = h[i:i+L] with block B = h[j:j+L], the change in corr[k*] is:
Δcorr[k*] = sum over affected positions of (new value * (1 - neighbor) - old value * (1 - neighbor))

This is computable but might be expensive for large blocks. For small blocks (L=2,3), it's feasible.

Actually, for the SA, I think the most impactful improvement is to use **inversion moves** (reversing a block). These are:
- Very cheap to compute (just reverse an array slice)
- Preserve the sum constraint
- Can make significant changes to the correlation structure
- Have been underutilized in the previous approach

Let me also add a **"flush" move**: identify the positions contributing most to the worst correlation and try to "flush" them (change 1→0 or 0→1) in a coordinated way.

def sa_v2(h_start, target_sum, max_iters, t_init, t_final, rng,
          time_limit=None, start_time=None):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    current_c5 = best_c5
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max(5000, max_iters // 8):
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Adaptive move selection
        r = rng.random()
        h_new = h.copy()
        moved = False
        
        if r < 0.25:
            # Single guided swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                # Target the worst 3 shifts
                worst_ks = np.argsort(-corr)[:3]
                for wk in worst_ks:
                    k = int(wk) - (n - 1)
                    # Find j where h[j]=1 and h[j+k]=0 (contributing to corr at shift k)
                    # Try to swap h[j] with h[j+k]
                    found = False
                    for _ in range(5):
                        jj = ones[rng.randint(len(ones))]
                        target = (jj + k) % n
                        if 0 <= target < n and h[target] == 0:
                            h_new[jj] = 0
                            h_new[target] = 1
                            found = True
                            moved = True
                            break
                    if found:
                        break
                if not moved:
                    i1 = ones[rng.randint(len(ones))]
                    i0 = zeros[rng.randint(len(zeros))]
                    h_new[i1] = 0
                    h_new[i0] = 1
                    moved = True
        elif r < 0.50:
            # Inversion
            L_list = [3, 4, 5, 7, 8, 10, 12, 15, 20]
            L = L_list[rng.randint(len(L_list))]
            if L < n:
                i = rng.randint(0, n - L)
                h_new[i:i+L] = h[i:i+L][::-1]
                moved = True
        elif r < 0.70:
            # Block swap
            L_list = [2, 3, 4, 5, 6, 8, 10]
            L = L_list[rng.randint(len(L_list))]
            if 2 * L < n:
                i = rng.randint(0, n - L)
                j = rng.randint(0, n - L)
                if abs(i - j) >= L:
                    h_new[i:i+L], h_new[j:j+L] = h[j:j+L].copy(), h[i:i+L].copy()
                    moved = True
        elif r < 0.85:
            # Cyclic shift
            L_list = [3, 5, 7, 10, 15]
            L = L_list[rng.randint(len(L_list))]
            if L < n:
                i = rng.randint(0, n - L)
                h_new[i:i+L] = np.roll(h[i:i+L], 1)
                moved = True
        else:
            # Double swap (2-for-2)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= 2 and len(zeros) >= 2:
                i1a, i1b = rng.choice(len(ones), 2, replace=False)
                i0a, i0b = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1a]] = 0
                h_new[ones[i1b]] = 0
                h_new[zeros[i0a]] = 1
                h_new[zeros[i0b]] = 1
                moved = True
        
        if not moved:
            continue
        
        # Verify and fix sum
        if abs(h_new.sum() - target_sum) > 1e-10:
            h_new = project_binary(h_new, target_sum)
        
        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5
        
        if delta < -1e-12 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5

Now for the continuous optimization, let me use a more efficient approach:

**Gradient-based approach with projected gradient:**

Since the constraint is simple (box + equality), we can use:
1. Compute gradient of the smooth max objective
2. Take a gradient step
3. Project back to feasible set

The gradient of soft_max(corr*dx, α) with respect to h is:
∂/∂h[m] soft_max = sum_k p_k * (∂corr[k]/∂h[m]) * dx

where p_k = exp(α*corr[k]*dx) / sum_j exp(α*corr[j]*dx)

And ∂corr[k]/∂h[m]:
For corr = np.correlate(h, 1-h, mode='full'):
corr[i] = sum_j h[j] * (1-h[j+i-n+1]) for valid j

So ∂corr[i]/∂h[m] = (1-h[m+i-n+1]) [if valid] + h[m] * (-1) * (1 if m = j and i-n+1 = 0... )

This is getting complicated. Let me just use SLSQP with numerical gradients, which is what the previous code does, but with a better parameter sweep.

Actually, let me try a different approach for continuous optimization: **ADMM (Alternating Direction Method of Multipliers)**.

The problem:
minimize t
subject to: corr[k]*dx <= t for all k
             sum(h) = target
             0 <= h <= 1

We can relax the constraints with penalties and alternate:
1. Update h: minimize t + (ρ/2)*sum_k max(0, corr[k]*dx - t)^2 + (σ/2)*(sum(h)-target)^2 + barrier + (ρ/2)*||h - h_dual||^2
2. Update t: t = max_k corr[k]*dx (or a smoothed version)
3. Update dual variables

This is appealing but complex to implement correctly. Let me stick with SLSQP for now but improve the parameters and add more restarts.

import numpy as np
from scipy.optimize import minimize
import time

# Global variables for optimizer callbacks
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_p = 0

def _obj_lse(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = _g_alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(max_c + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _obj_power(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.sum(corr**_g_p) ** (1.0/_g_p))

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def generate_starts(n, rng, h_init):
    starts = []
    # Start 1: project initial
    h_bin = project_binary(h_init, n / 2.0)
    starts.append(h_bin)
    
    # Start 2: random
    h_rand = np.zeros(n)
    perm = rng.permutation(n)
    h_rand[perm[:n//2]] = 1.0
    starts.append(h_rand)
    
    # Start 3: Thue-Morse
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = bin(i).count('1') % 2
    h_tm = t.astype(float)
    h_tm = project_binary(h_tm, n / 2.0)
    starts.append(h_tm)
    
    # Start 4: low-discrepancy (Weyl sequence based)
    h_ward = np.zeros(n)
    golden = (1 + np.sqrt(5)) / 2
    for i in range(n):
        h_ward[i] = 1.0 if (i / n * golden) % 1.0 < 0.5 else 0.0
    h_ward = project_binary(h_ward, n / 2.0)
    starts.append(h_ward)
    
    # Start 5: perturbed initial
    h_pert = h_bin + rng.randn(n) * 0.3
    h_pert = project_binary(h_pert, n / 2.0)
    starts.append(h_pert)
    
    return starts

def slsqp_refine(h_start, n, target, time_limit, start_time):
    global _g_n, _g_target, _g_alpha, _g_p
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    alphas = [20, 50, 100, 200, 500, 1000, 2000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

import numpy as np

_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _obj_lse(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = _g_alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(max_c + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_refine(h_start, n, target, time_limit, start_time):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons_list = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    alphas = [15, 30, 60, 120, 250, 500, 1000, 2000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons_list,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-15:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score
