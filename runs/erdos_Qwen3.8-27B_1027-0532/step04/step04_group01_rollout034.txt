```python
def compute_gradient(h, alpha, n):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    # corr has length 2n-1
    
    # Softmax weights
    shifted = alpha * corr * dx
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    
    # For each lag L from 1 to n-1:
    for L in range(1, n):
        p_pos = p[n-1+L]  # positive lag
        p_neg = p[n-1-L]  # negative lag
        
        if p_pos < 1e-15 and p_neg < 1e-15:
            continue
        
        # f1(j,L) for shift = +L
        # j ∈ [0, min(L-1, n-1-L)]: -h[j+L]
        # j ∈ [max(L,0), n-1-L]: (1-h[j-L]) - h[j+L]  (when L ≤ n-1-L)
        # j ∈ [max(L, n-L), n-1]: (1-h[j-L])
        
        # f2(j,L) for shift = -L
        # j ∈ [0, min(L-1, n-1-L)]: (1-h[j+L])
        # j ∈ [max(L,0), n-1-L]: (1-h[j+L]) - h[j-L]
        # j ∈ [max(L, n-L), n-1]: -h[j-L]
        
        # Case analysis based on L vs n/2
        if 2*L <= n-1:  # L ≤ (n-1)/2
            # Region 1: j ∈ [0, L-1]
            if L > 0:
                r1 = slice(0, L)
                grad[r1] += (-h[r1+L] * p_pos + (1-h[r1+L]) * p_neg)
            # Region 2: j ∈ [L, n-1-L]
            r2 = slice(L, n-L)
            grad[r2] += ((1-h[r2-L]) - h[r2+L]) * p_pos + ((1-h[r2+L]) - h[r2-L]) * p_neg
            # Region 3: j ∈ [n-L, n-1]
            r3 = slice(n-L, n)
            grad[r3] += (1-h[r3-L]) * p_pos + (-h[r3-L]) * p_neg
        else:  # L > (n-1)/2
            # Region 1: j ∈ [0, n-1-L]
            r1 = slice(0, n-L)
            grad[r1] += (-h[r1+L] * p_pos + (1-h[r1+L]) * p_neg)
            # Region "neither": j ∈ [n-L, L-1]
            r_neither = slice(n-L, L)
            # No contribution (both conditions fail)
            # Region 2: j ∈ [L, n-1]
            r2 = slice(L, n)
            grad[r2] += (1-h[r2-L]) * p_pos + (-h[r2-L]) * p_neg
    
    grad *= dx
    return grad
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def compute_logsumexp_grad(h, alpha, n):
    """Compute gradient of (1/alpha)*log(sum(exp(alpha*corr*dx)))"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    shifted = alpha * corr * dx
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    
    for L in range(1, n):
        p_pos = p[n-1+L]
        p_neg = p[n-1-L]
        
        if p_pos < 1e-10 and p_neg < 1e-10:
            continue
        
        if 2*L <= n-1:
            # Region 1: j ∈ [0, L)
            if L > 0:
                r1 = slice(0, L)
                grad[r1] += (-h[r1+L]) * p_pos + (1-h[r1+L]) * p_neg
            # Region 2: j ∈ [L, n-L)
            r2 = slice(L, n-L)
            grad[r2] += ((1-h[r2-L]) - h[r2+L]) * p_pos + ((1-h[r2+L]) - h[r2-L]) * p_neg
            # Region 3: j ∈ [n-L, n)
            r3 = slice(n-L, n)
            grad[r3] += (1-h[r3-L]) * p_pos + (-h[r3-L]) * p_neg
        else:
            # Region 1: j ∈ [0, n-L)
            r1 = slice(0, n-L)
            grad[r1] += (-h[r1+L]) * p_pos + (1-h[r1+L]) * p_neg
            # Region "neither": j ∈ [n-L, L) - no contribution
            # Region 2: j ∈ [L, n)
            r2 = slice(L, n)
            grad[r2] += (1-h[r2-L]) * p_pos + (-h[r2-L]) * p_neg
    
    grad *= dx
    return grad
```

The projection step is the tricky part. I need to project onto:
- 0 ≤ h[i] ≤ 1 for all i
- sum(h) = n/2

This is a constrained projection problem. The projection onto the intersection of the box [0,1]^n and the hyperplane sum(h) = n/2 can be done as follows:

1. Start with the unconstrained projection: h_proj = h - lr * grad_step (the Adam update)
2. Project onto the box: clip to [0,1]
3. Project onto the hyperplane: adjust to make sum = n/2

But steps 2 and 3 don't commute (projecting onto the box might violate the sum constraint, and vice versa). The correct approach is to find the projection onto the intersection, which is:
min ||x - y||² subject to 0 ≤ x[i] ≤ 1 and sum(x) = n/2

This can be solved by: find λ such that sum(clip(y[i] - λ, 0, 1)) = n/2. Then x[i] = clip(y[i] - λ, 0, 1).

This is a 1D root-finding problem (bisection on λ), which is fast.

```python
def project_onto_box_and_sum(y, target_sum):
    """Project y onto {x: 0 ≤ x[i] ≤ 1, sum(x) = target_sum}"""
    n = len(y)
    # Find lambda such that sum(clip(y[i] - lambda, 0, 1)) = target_sum
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(y - mid, 0.0, 1.0)
```

```python
def adam_continuous_optimize(h_init, n, target, time_limit, t_start, n_iterations=200000):
    dx = 2.0 / n
    h = h_init.copy().astype(np.float64)
    
    # Adam parameters
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    alpha = 50.0  # smoothing parameter
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    no_imp = 0
    max_no_imp = 5000
    
    while t_count < n_iterations:
        if time.time() - t_start > time_limit:
            break
        
        t_count += 1
        
        # Adaptive alpha: increase over time
        alpha = 50.0 + 20.0 * t_count / n_iterations
        
        # Compute gradient
        grad = compute_logsumexp_grad(h, alpha, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_count)
        v_hat = v / (1 - beta2**t_count)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h_new = project_onto_box_and_sum(h_new, target)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        h = h_new
        
        # Periodic restart from best
        if no_imp > max_no_imp:
            h = best_h.copy()
            no_imp = 0
            # Increase learning rate slightly
            lr = max(0.001, lr * 0.5)
    
    return best_h, best_c5
```
