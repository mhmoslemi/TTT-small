    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    return corr

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.8, reg=1e-5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg
    n_half = n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Active set: constraints within margin of max
        corr = get_corr(h)
        cmax = corr.max()
        margin = max(0.0005, 0.03 * cmax)
        active_k = np.where(corr >= cmax - margin)[0] - n_half
        
        # Limit to top 50 active constraints for speed
        if len(active_k) > 50:
            top_idx = np.argsort(corr)[::-1][:50]
            active_k = top_idx - n_half
        
        for k in active_k:
            k = int(k)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        # Add a few random constraints
        rng_local = np.random.RandomState(it)
        n_random = min(5, n - len(active_k))
        all_k = np.arange(-(n-1), n)
        random_k = rng_local.choice(all_k, size=n_random, replace=False)
        for k in random_k:
            k = int(k)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-8, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Adapt damping
        if stuck_count > 50:
            damping = max(0.3, damping * 0.9)
            stuck_count = 0
        if it % 1000 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)
        if current_reg < 1e-7 and it % 500 == 0:
            current_reg = min(1e-5, current_reg * 1.5)
    
    return best_h, best_c5


def newton_equalize(h_init, n, target, n_active, max_iter, time_limit):
    """Use Newton's method to equalize the top n_active constraints."""
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = get_corr(h)
        # Find top n_active shifts
        top_idx = np.argsort(corr)[::-1][:n_active]
        active_k = top_idx - n_half
        target_val = np.mean(corr[top_idx])  # Target: equalize to mean
        
        # Build Jacobian: for each active constraint, gradient w.r.t. h
        # f_k(h) = sum_i h[i] * (1-h[i-k]) * dx (linearized around current h)
        # df_k/dh[j] = (1-h[j-k]) * dx - h[j] * (-1) * delta(j in overlap for h[i-k]) ... 
        # Actually for the bilinear form: f_k = sum_i h[i]*(1-h[i-k])*dx
        # df_k/dh[j] = (1-h[j-k])*dx (if j in range) + h[j+k]*dx (if j+k in range) * (-1) * ... 
        # Let me be more careful:
        # f_k = sum_i h[i] * (1 - h[i-k]) * dx
        # = sum_i h[i] * dx - sum_i h[i]*h[i-k]*dx
        # 
        # df_k/dh[j] = dx - (h[j-k]*dx + h[j+k]*dx) [if valid indices]
        # Wait, that's for the full bilinear form. Let me redo:
        # f_k = sum_i h_i * (1 - h_{i-k}) * dx (summing over valid i)
        # df_k/dh_j = sum_i [d(h_i)/d(h_j) * (1-h_{i-k}) + h_i * d(1-h_{i-k})/d(h_j)] * dx
        # = (1 - h_{j-k}) * dx - sum_i h_i * delta(i-k = j) * dx
        # = (1 - h_{j-k}) * dx - h_{j+k} * dx (if j+k is valid)
        
        J = np.zeros((n_active, n))
        for a, k in enumerate(active_k):
            k = int(k)
            for j in range(n):
                val = 0.0
                # First term: (1 - h[j-k]) * dx if j-k is valid
                if 0 <= j - k < n:
                    val += (1.0 - h[j - k]) * dx
                # Second term: -h[j+k] * dx if j+k is valid
                if 0 <= j + k < n:
                    val -= h[j + k] * dx
                J[a, j] = val
        
        # We want to solve: J @ dh = -r where r = corr[active] - target_val
        # Subject to: sum(dh) = 0 (to maintain sum constraint)
        # And: 0 <= h + dh <= 1
        
        r = corr[top_idx] - target_val
        
        # Use least squares with the sum constraint
        # Minimize ||J @ dh + r||^2 s.t. sum(dh) = 0, h+dh in [0,1]
        
        # Simple approach: use scipy minimize with bounds
        # Or: use the KKT system
        
        # For simplicity, use a damped Newton step:
        # dh = -alpha * J^+ @ r (pseudo-inverse)
        # Then project to satisfy sum(dh) = 0
        
        # Better: solve the constrained LSQ
        # [J; 1^T] @ dh = [-r; 0]
        A = np.vstack([J, np.ones((1, n))])
        b = np.concatenate([-r, [0.0]])
        
        try:
            dh, residuals, rank, sv = np.linalg.lstsq(A, b, rcond=None)
        except Exception:
            break
        
        # Damped step
        for alpha in [0.5, 0.3, 0.1, 0.05, 0.01]:
            h_try = h + alpha * dh
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                break
    
    return best_h, best_c5


def power_mean_obj(h, p):
    corr = get_corr(h)
    # Clip to avoid negative values (shouldn't happen but just in case)
    corr_pos = np.maximum(corr, 0.0)
    return float(np.sum(corr_pos ** p) ** (1.0 / p))

def power_mean_grad(h, p):
    corr = get_corr(h)
    corr_pos = np.maximum(corr, 0.0)
    cpm = power_mean_obj(h, p)
    
    # d/dh_j of (sum corr_k^p)^(1/p)
    # = (1/p) * (sum corr_k^p)^(1/p - 1) * p * sum_k corr_k^(p-1) * d(corr_k)/dh_j
    # = (sum corr_k^p)^(1/p - 1) * sum_k corr_k^(p-1) * d(corr_k)/dh_j
    
    n = len(h)
    dx = 2.0 / n
    n_half = n - 1
    
    grad = np.zeros(n)
    weights = corr_pos ** (p - 1) / (cpm ** (p - 1)) if cpm > 0 else np.zeros_like(corr)
    
    for k_idx in range(len(corr)):
        if weights[k_idx] == 0:
            continue
        k = k_idx - n_half
        w = weights[k_idx]
        for j in range(n):
            if 0 <= j - k < n:
                grad[j] += w * (1.0 - h[j - k]) * dx
            if 0 <= j + k < n:
                grad[j] -= w * h[j + k] * dx
    
    return grad

def bisection_lp(h_init, n, target, B_init, B_min, time_limit):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    
    B = B_init
    while B > B_min and time.time() - start < time_limit:
        g = 1.0 - h
        h_var = cp.Variable(n)
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Include all constraints with bound B
        n_half = n - 1
        dx = 2.0 / n
        for k in range(-(n-1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= B)
        
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-8)
        except:
            break
        
        if prob.status == 'infeasible':
            B = B + (B - B_min) * 0.5  # B was too tight
            continue
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = project_feasible(h_new, target)
        c5 = compute_c5(h)
        
        if c5 <= B:
            B = max(B_min, c5 * 0.999)  # Try to tighten
        else:
            B = min(c5, B + 0.001)
    
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.85, reg=1e-5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg
    n_half = n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        corr = get_corr(h)
        cmax = corr.max()
        margin = max(0.0003, 0.025 * cmax)
        active_mask = corr >= cmax - margin
        active_k = np.where(active_mask)[0] - n_half

        if len(active_k) > 40:
            top_idx = np.argsort(corr)[::-1][:40]
            active_k = top_idx - n_half

        for k in active_k:
            k = int(k)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        rng_local = np.random.RandomState(it * 7 + 3)
        all_k = np.arange(-(n-1), n)
        n_rand = 8
        random_k = rng_local.choice(all_k, size=n_rand, replace=False)
        for k in random_k:
            k = int(k)
            if k in active_k:
                continue
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iterations=10000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0

            stuck_count += 1

        if it % 800 == 0 and current_reg > 1e-10:

        if current_reg < 5e-8 and it % 400 == 0 and it > 800:
            current_reg = min(1e-4, current_reg * 2.0)

    return best_h, best_c5

def equalize_active(h_init, n, target, n_active, max_iter, time_limit):
    start = time.time()

    n_half = n - 1
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

        if time.time() - start > time_limit:
            break

        corr = get_corr(h)
        top_idx = np.argsort(corr)[::-1][:n_active]
        active_k = (top_idx - n_half).astype(int)
        target_val = np.mean(corr[top_idx])

        J = np.zeros((n_active, n))
        for a, k in enumerate(active_k):
            for j in range(n):
                val = 0.0
                if 0 <= j - k < n:
                    val += (1.0 - h[j - k]) * dx
                if 0 <= j + k < n:
                    val -= h[j + k] * dx
                J[a, j] = val

        r = corr[top_idx] - target_val
        A_mat = np.vstack([J, np.ones((1, n))])
        b_vec = np.concatenate([-r, [0.0]])

        try:
            dh, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except Exception:
            break

        improved = False
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            h_try = h + alpha * dh
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)

                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                break

            break

def smooth_obj(h, alpha):
    corr = get_corr(h)
    c_max = corr.max()
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
