    """Find optimal h[j] that minimizes max_k c_k."""
    old = h[j]
    n = len(h)
    dx = 2.0 / n
    
    # Precompute alpha_k and beta_k for all k
    # k ranges from -(n-1) to (n-1), indexed as m = k + (n-1), m in [0, 2n-2]
    M = 2 * n - 1  # number of correlation values
    
    def c_at_t(t):
        """Compute all correlation values when h[j] = t."""
        h_new = h.copy()
        h_new[j] = t
        return np.correlate(h_new, 1.0 - h_new, mode='full')
    
    # This is O(n) per evaluation due to correlate being O(n)
    # For golden section search, we need ~30 evaluations
    # Total: ~30 * O(n) = O(30n) per coordinate
    # For all n coordinates: O(30n²) per pass
    # For n=400: O(30*160000) = 4.8M operations per pass
    # That's fast enough!
    
    # Golden section search
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(c_at_t(c1)))
    f2 = float(np.max(c_at_t(c2)))
    for _ in range(40):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(c_at_t(c1)))
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(c_at_t(c2)))
    
    t_opt = (a + b) / 2.0
    h[j] = t_opt

    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return lse_obj_full(x, n, target, alpha, pen)
    
    def obj_grad(x):
        return lse_grad(x, n, target, alpha, pen)
    
    def con_eq(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)

    h_cur = h0.copy()
    cur = compute_c5(h_cur)
    for alpha in alphas:
        _alpha = alpha
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:

                cur = sc
        except Exception:
            pass
