    wsum = wt.sum()

        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = wt[s + n - 1]
        if w < 1e-12:
            continue
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            grad[:n + s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:n + s]
    return grad

def gradient_descent(h, deadline, alpha_list, lr_init, momentum, target):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init
    for alpha, count in alpha_list:
        for _ in range(count):
            if time.time() > deadline:
                return best_h, best_score
            h_look = project_feasible(h_cur + momentum * v, target)
            g = grad_soft_max(h_look, alpha)
            v = momentum * v - lr * g
            h_new = project_feasible(h_cur + v, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
            else:
                lr *= 0.5
                if lr < 1e-12:
                    lr = lr_init
                    break
    return best_h, best_score

def _slsqp_obj(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _slsqp_cons(x, target):
    return np.sum(x) - target

def perturbation_binary(h, n_flips, rng):
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(n_flips):
        if len(ones) > 1 and len(zeros) > 1:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    def _rem():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_parent = len(h_init)
    target_parent = n_parent / 2.0

    best_h = project_feasible(h_init, target_parent)
    best_score = compute_c5(best_h)

    # Phase 1: Intensive binary local search at parent resolution
    if _rem() > 100:
        h_bin = make_binary(best_h, target_parent)
        n_rounds = 40
        for rnd in range(n_rounds):
            if _rem() < 30:
                break
            if rnd < 5:
                dl = time.time() + min(12.0, _rem() * 0.2)
                h_ls, sc = local_search_exhaustive(h_bin, dl, max_passes=30)
            else:
                dl = time.time() + min(10.0, _rem() * 0.2)
                h_ls, sc = local_search(h_bin, dl, n_zero_sample=50, max_passes=150, top_k=7)
            if sc < best_score:
                best_score = sc
                best_h = h_ls
            if _rem() > 50:
                h_bin = perturbation_binary(best_h, max(3, n_parent // 30), rng)

    # Phase 2: Simulated annealing at parent resolution
    if _rem() > 80:
        h_bin = make_binary(best_h, target_parent)
        dl = time.time() + min(50.0, _rem() * 0.3)
        h_sa, sc = simulated_annealing(h_bin, dl, T_init=0.05, T_decay=0.997, T_min=0.003)
        if sc < best_score:
            best_score = sc
            best_h = h_sa
        # Follow-up local search on SA result
        if _rem() > 40:
            dl2 = time.time() + min(20.0, _rem() * 0.3)
            h_ls2, sc2 = local_search_exhaustive(h_sa, dl2, max_passes=50)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ls2

    # Phase 3: Higher resolution n=800
    if _rem() > 120:
        n_new = 800
        target_new = n_new / 2.0
        x_old = np.linspace(0, 2, n_parent, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_feasible(h_up, target_new)
        h_bin_new = make_binary(h_up, target_new)
        for rnd in range(20):
            if _rem() < 40:
                break
            dl = time.time() + min(10.0, _rem() * 0.2)
            if rnd < 3:
                h_ls, sc = local_search_exhaustive(h_bin_new, dl, max_passes=20)
            else:
                h_ls, sc = local_search(h_bin_new, dl, n_zero_sample=50, max_passes=100, top_k=7)
            if sc < best_score:
                best_score = sc
                best_h = h_ls
            if _rem() > 50:
                h_bin_new = perturbation_binary(best_h, max(4, n_new // 30), rng)

    # Phase 4: Even higher resolution n=1200
    if _rem() > 100:
        n_new2 = 1200
        target_new2 = n_new2 / 2.0
        x_old2 = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new2 = np.linspace(0, 2, n_new2, endpoint=False)
        h_up2 = np.interp(x_new2, x_old2, best_h, period=2.0)
        h_up2 = project_feasible(h_up2, target_new2)
        h_bin2 = make_binary(h_up2, target_new2)
        for rnd in range(10):
            if _rem() < 40:
                break
            dl = time.time() + min(8.0, _rem() * 0.2)
            h_ls, sc = local_search(h_bin2, dl, n_zero_sample=50, max_passes=80, top_k=7)
            if sc < best_score:
                best_score = sc
                best_h = h_ls
            if _rem() > 40:
                h_bin2 = perturbation_binary(best_h, max(4, n_new2 // 30), rng)

    # Phase 5: Gradient descent (non-binary refinement)
    if _rem() > 50:
        n = len(best_h)
        target = n / 2.0
        dl = time.time() + min(30.0, _rem() * 0.3)
        h_gd, sc = gradient_descent(best_h, dl, [(500, 40), (2000, 40), (8000, 40)],
                                     lr_init=0.003, momentum=0.85, target=target)
        if sc < best_score:
            best_score = sc
            best_h = h_gd

    # Phase 6: SLSQP polish with soft-max
    if _rem() > 30:
        n = len(best_h)
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
        for alpha in [3000, 15000, 60000]:
            if _rem() < 15:
                break
            try:
                res = minimize(_slsqp_obj, best_h, args=(alpha,), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Final binary local search
    if _rem() > 20:
        n = len(best_h)
        target = n / 2.0
        h_bin_final = make_binary(best_h, target)
        dl = time.time() + min(15.0, _rem() * 0.4)
        h_ls, sc = local_search_exhaustive(h_bin_final, dl, max_passes=50)
        if sc < best_score:
            best_score = sc
            best_h = h_ls

    # Phase 8: One more SA + LS cycle if time allows
    if _rem() > 40:
        n = len(best_h)
        target = n / 2.0
        h_bin = make_binary(best_h, target)
        h_bin = perturbation_binary(h_bin, max(2, n // 40), rng)
        dl_sa = time.time() + min(15.0, _rem() * 0.3)
        h_sa, sc_sa = simulated_annealing(h_bin, dl_sa, T_init=0.03, T_decay=0.998)
        dl_ls = time.time() + min(10.0, _rem() * 0.3)
        h_ls, sc_ls = local_search_exhaustive(h_sa, dl_ls, max_passes=40)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, len(best_h)
