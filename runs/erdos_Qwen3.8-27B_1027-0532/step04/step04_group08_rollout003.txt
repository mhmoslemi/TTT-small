f(h) = max_k q_k(h) where q_k(h) = sum_i h[i](1-h[i+k]) * dx

This is a nonconvex problem (max of quadratics). But locally, we can use the fact that at the optimum, several q_k's are active (equal to the max), and the subgradient is the average of their gradients.

The gradient of q_k with respect to h[i] is:
- (1-h[i+k]) * dx if 0 ≤ i+k < n (from the term h[i](1-h[i+k]))
- -h[i-k] * dx if 0 ≤ i-k < n (from the term h[i-k](1-h[i]))

So the subgradient at the optimum is:
SG[h[i]] = dx * sum_{k active} [(1-h[i+k]) * 1[i+k valid] - h[i-k] * 1[i-k valid]] / |active|

For the KKT conditions, we need:
SG[h[i]] + lambda * 1[i at upper bound] - mu * 1[i at lower bound] + nu = 0
where lambda ≥ 0 (for upper bound), mu ≥ 0 (for lower bound), and nu is the Lagrange multiplier for the sum constraint.

This means that at the optimum, for all i not at bounds:
sum_{k active} [(1-h[i+k]) - h[i-k]] / |active| = -nu/dx (a constant)

This is a useful necessary condition! I can use it to check if I'm at a local optimum, and to guide the optimization.

    except Exception:
        return h_start


def nesterov_opt(h, n, target, alpha, lr_init, n_iter, deadline):
    h_cur = project_feasible(h.copy(), target)
    y = h_cur.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t_k = 2.0 / (float(i) + 2.0)
        y = project_feasible(h_cur + t_k * (h_cur - y), target)
        _, g = soft_max_obj_and_grad(y, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        lr = lr_init
        h_new = project_feasible(y - lr * g, target)
        sc_new = compute_c5(h_new)
        for _ in range(25):
            if sc_new < compute_c5(y) - 1e-10:
                break
            lr *= 0.5
            h_new = project_feasible(y - lr * g, target)
            sc_new = compute_c5(h_new)
            if lr < 1e-12:
                break
        h_cur = h_new
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
    return best_h


def directed_binary_local_search(h, target, deadline, max_rounds=500, n_trials=40):
    n = len(h)
    h_cur = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_cur[sorted_idx[:int(target)]] = 1.0
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        worst_idx = int(np.argmax(corr))
        s = worst_idx - (n - 1)
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        if s >= 0:
            overlap_src = np.where((h_cur[:n - s] == 1.0) & (h_cur[s:] == 0.0))[0]
            overlap_dst = overlap_src + s
        else:
            sa = -s
            overlap_src = np.where((h_cur[sa:] == 1.0) & (h_cur[:n - sa] == 0.0))[0] + sa
            overlap_dst = np.where((h_cur[sa:] == 1.0) & (h_cur[:n - sa] == 0.0))[0]
        best_trial = best_score
        best_i, best_j = -1, -1
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            if len(overlap_dst) > 0 and rng_global.random() < 0.7:
                j = int(overlap_dst[rng_global.randint(len(overlap_dst))])
            else:
                j = int(zeros[rng_global.randint(len(zeros))])
            if len(overlap_src) > 0 and rng_global.random() < 0.7:
                i = int(overlap_src[rng_global.randint(len(overlap_src))])
            else:
                i = int(ones[rng_global.randint(len(ones))])
            if i == j or h_cur[i] != 1.0 or h_cur[j] != 0.0:
                continue
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            sc = compute_c5(h_cur)
            h_cur[i] = 1.0
            h_cur[j] = 0.0
            if sc < best_trial:
                best_trial = sc
                best_i, best_j = i, j
        if best_i >= 0:
            h_cur[best_i] = 0.0
            h_cur[best_j] = 1.0
            best_score = best_trial
            best_h = h_cur.copy()
        else:
            found = False
            for _ in range(20):
                if time.time() > deadline:
                    break
                i = int(ones[rng_global.randint(len(ones))])
                j = int(zeros[rng_global.randint(len(zeros))])
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                sc = compute_c5(h_cur)
                h_cur[i] = 1.0
                h_cur[j] = 0.0
                if sc < best_score:
                    best_score = sc
                    h_cur[i] = 0.0
                    h_cur[j] = 1.0
                    best_h = h_cur.copy()
                    found = True
                    break
            if not found:
                break
    return best_h, best_score


def simulated_annealing(h_init, target, deadline, total_iter=60000):
    n = len(h_init)
    h = np.zeros(n)
    sorted_idx = np.argsort(-h_init)
    h[sorted_idx[:int(target)]] = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    T_start = 0.008
    T_end = 0.000005
    n_phases = 40
    iters_per = max(100, total_iter // n_phases)
    for phase in range(n_phases):
        T = T_start * (T_end / T_start) ** (float(phase) / float(n_phases))
        for _ in range(iters_per):
            if time.time() > deadline:
                return best_h
            i = int(ones[rng_global.randint(len(ones))])
            j = int(zeros[rng_global.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            new_sc = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = new_sc - cur_score
            if delta < 0 or rng_global.random() < np.exp(-delta / T):
                h[i] = 0.0
                h[j] = 1.0
                cur_score = new_sc
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if new_sc < best_score:
                    best_score = new_sc
                    best_h = h.copy()
    return best_h


rng_global = np.random.RandomState(0)


def run(seed=42, budget_s=1300, **kwargs):
    global rng_global
    rng_global = np.random.RandomState(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    def remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: SLSQP with increasing alpha
    alphas_p1 = [50, 200, 1000, 5000, 20000, 80000]
    for a in alphas_p1:
        if remaining() < 60:
            break
        h_new = slsqp_solve(best_h, n, target, a, 400, 1e-14)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Directed binary local search
    if remaining() > 120:
        dl = time.time() + min(150.0, remaining() * 0.3)
        h_bin, sc_bin = directed_binary_local_search(best_h, target, dl, max_rounds=300, n_trials=50)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 3: Nesterov accelerated gradient
    if remaining() > 120:
        dl = time.time() + min(120.0, remaining() * 0.3)
        for a_nest in [2000, 10000, 50000]:
            if remaining() < 40:
                break
            dl2 = time.time() + min(60.0, remaining() * 0.25)
            h_nest = nesterov_opt(best_h, n, target, a_nest, 0.05, 5000, dl2)
            sc = compute_c5(h_nest)
            if sc < best_score:
                best_score = sc
                best_h = h_nest

    # Phase 4: Simulated annealing
    if remaining() > 150:
        dl = time.time() + min(200.0, remaining() * 0.35)
        h_sa = simulated_annealing(best_h, target, dl, total_iter=80000)
        sc = compute_c5(h_sa)
        if sc < best_score:
            best_score = sc
            best_h = h_sa

    # Phase 5: SLSQP polish with very high alpha
    if remaining() > 60:
        for a in [100000, 300000, 800000]:
            if remaining() < 30:
                break
            h_new = slsqp_solve(best_h, n, target, a, 500, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: Binary local search from continuous best
    if remaining() > 80:
        dl = time.time() + min(100.0, remaining() * 0.3)
        h_bin, sc_bin = directed_binary_local_search(best_h, target, dl, max_rounds=400, n_trials=60)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 7: Multi-restart SLSQP with perturbations
    if remaining() > 90:
        for pert_mag in [0.02, 0.01, 0.005]:
            if remaining() < 40:
                break
            h_pert = project_feasible(best_h + rng_global.randn(n) * pert_mag, target)
            h_new = slsqp_solve(h_pert, n, target, 50000, 300, 1e-14)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Final binary polish
    if remaining() > 40:
        dl = time.time() + min(40.0, remaining() * 0.5)
        h_bin, sc_bin = directed_binary_local_search(best_h, target, dl, max_rounds=200, n_trials=100)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 9: Final SLSQP
    if remaining() > 20:
        h_new = slsqp_solve(best_h, n, target, 500000, 300, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
