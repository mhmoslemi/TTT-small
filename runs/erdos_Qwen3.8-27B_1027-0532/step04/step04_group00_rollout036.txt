def gradient_of_max_overlap(h, n):
    """Compute gradient of max_k O(k) approximately using top-k subgradient."""
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:2*n-1]
    c_max = float(np.max(over))
    # Soft max gradient
    alpha = 500.0
    weights = np.exp(alpha * (over - c_max))
    weights = np.clip(weights, 0, 1e300)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(n)
        weights[np.argmax(over)] = 1.0
    else:
        weights /= wsum
    
    grad = np.zeros(n)
    for k in range(n):
        if weights[k] < 1e-10:
            continue
        w = weights[k]
        if k == 0:
            # O(0) = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # dO(0)/dh[i] = 1 - 2*h[i]
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            # O(k) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # dO(k)/dh[i] for i < n-k: (1-h[i+k])
            # dO(k)/dh[i] for i >= k: -h[i-k]
            # These can overlap for i in [k, n-k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += w * (1.0 - h[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= w * h[j_neg - k]
    grad *= dx
    return grad, c_max * dx

def subgradient_polish(h, n, target, n_steps=500, n_active=5, lr_init=0.003, lr_decay=0.998, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stall_count = 0
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:2*n-1]
        c_curr = float(np.max(over)) * dx
        
        if c_curr < best_score - 1e-12:
            best_score = c_curr
            best_h = h_cur.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                # Adaptive: reduce n_active and increase temp
                n_active = max(3, n_active - 1)
                lr = max(lr * 1.5, lr_init * 0.5)
                stall_count = 0
        
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_curr + 1e-14:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps=300, initial_epsilon=0.002, deadline=None):

                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def active_balance(h, n, target, n_steps=200, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:2*n-1]
        c_max = float(np.max(over))
        if c_max * dx < best_score:
            best_score = c_max * dx
            best_h = h_cur.copy()
        thresh = c_max - 0.02
        active = np.where(over >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(over))])
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            if k == 0:
                grad[:] += (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= h_cur[j_neg - k]
        grad /= len(active)
        grad *= dx
        improved = False
        for lr in [0.004, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n_par = len(initial_h_values)
    h_init = project_feasible(np.array(initial_h_values, dtype=float), n_par / 2.0)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    phase1_end = start_time + 80
    n1 = 80
    t1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, t1)
    pop = [h1.copy()]
    sc_pop = [compute_c5(h1)]
    for i in range(20):
        if i < 10:
            hp = h1 + rng.randn(n1) * 0.12
        else:
            x = np.linspace(0, 2, n1)
            f = 1.5 + (i - 10) * 0.8
            hp = 0.5 + 0.35 * np.sin(f * np.pi * x + rng.randn() * 0.3) + rng.randn(n1) * 0.04
        hp = project_feasible(hp, t1)
        pop.append(hp)
        sc_pop.append(compute_c5(hp))
    for gen in range(40):
        if time.time() > phase1_end:
            break
        bi = int(np.argmin(sc_pop))
        for i in range(1, len(pop)):
            a, b, c = rng.choice(len(pop), 3, replace=False)
            a, b, c = int(a), int(b), int(c)
            if i in (a, b, c):
                continue
            v = pop[a] + 0.75 * (pop[b] - pop[c])
            v = project_feasible(v, t1)
            trial = pop[i].copy()
            jr = rng.randint(n1)
            for j in range(n1):
                if rng.rand() < 0.85 or j == jr:
                    trial[j] = v[j]
            trial = project_feasible(trial, t1)
            sc = compute_c5(trial)
            if sc <= sc_pop[i]:
                pop[i] = trial
                sc_pop[i] = sc
    bi = int(np.argmin(sc_pop))
    h1_up = interpolate_to(pop[bi], n_par, n_par / 2.0)
    sc1_up = compute_c5(h1_up)
    if sc1_up < best_score:
        best_h = h1_up
        best_score = sc1_up

    if tleft() > 120:
        n2 = 200
        t2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, t2)
        h2 = slsqp_opt(h2, n2, t2, [50, 200, 800, 3000], 1500, 1e-15,
                       deadline=start_time + 200)
        h2_sg, sc2_sg = subgradient_polish(h2, n2, t2, n_steps=250, n_active=5,
                                            lr_init=0.003, lr_decay=0.998,
                                            deadline=start_time + 260)
        if sc2_sg < compute_c5(h2):
            h2 = h2_sg
        h2_up = interpolate_to(h2, n_par, n_par / 2.0)
        if compute_c5(h2_up) < best_score:
            best_h = h2_up
            best_score = compute_c5(h2_up)

    n3 = 600
    t3 = n3 / 2.0
    h3 = interpolate_to(best_h, n3, t3)
    sc3_init = compute_c5(h3)

    h3 = slsqp_opt(h3, n3, t3, [50, 200, 800], 2000, 1e-15,
                   deadline=start_time + budget_s - 500)
    sc3 = compute_c5(h3)

    if tleft() > 200:
        h3_sg, sc3_sg = subgradient_polish(h3, n3, t3, n_steps=400, n_active=5,
                                            lr_init=0.002, lr_decay=0.999,
                                            deadline=start_time + budget_s - 400)
        if sc3_sg < compute_c5(h3):
            h3 = h3_sg
            sc3 = sc3_sg

    if tleft() > 150:
        h3_surg, sc3_surg = worst_lag_surgery(h3, n3, t3, max_steps=200,
                                               initial_epsilon=0.002,
                                               deadline=start_time + budget_s - 350)
        if sc3_surg < compute_c5(h3):
            h3 = h3_surg
            sc3 = sc3_surg

    if tleft() > 120:
        h3_bal, sc3_bal = active_balance(h3, n3, t3, n_steps=150,
                                          deadline=start_time + budget_s - 300)
        if sc3_bal < compute_c5(h3):
            h3 = h3_bal
            sc3 = sc3_bal

    if tleft() > 100:
        h3 = slsqp_opt(h3, n3, t3, [2000, 8000, 20000], 1500, 1e-16,
                       deadline=start_time + budget_s - 200)
        sc3 = compute_c5(h3)

    if sc3 < best_score:
        best_score = sc3
        best_h = h3.copy()

    if tleft() > 100:
        h3_sg2, sc3_sg2 = subgradient_polish(h3, n3, t3, n_steps=350, n_active=3,
                                              lr_init=0.001, lr_decay=0.999,
                                              deadline=start_time + budget_s - 130)
        if sc3_sg2 < compute_c5(h3):
            h3 = h3_sg2
            sc3 = sc3_sg2

    if tleft() > 80:
        h3_bd = block_descent(h3, n3, t3, 3, 3, 3000.0, start_time + budget_s - 80)
        sc3_bd = compute_c5(h3_bd)
        if sc3_bd < compute_c5(h3):
            h3 = h3_bd
            sc3 = sc3_bd

    if tleft() > 60:
        h3 = slsqp_opt(h3, n3, t3, [5000, 15000], 1500, 1e-17,
                       deadline=start_time + budget_s - 50)
        sc3 = compute_c5(h3)

    if tleft() > 40:
        h3_surg2, sc3_surg2 = worst_lag_surgery(h3, n3, t3, max_steps=150,
                                                 initial_epsilon=0.001,
                                                 deadline=start_time + budget_s - 35)
        if sc3_surg2 < compute_c5(h3):
            h3 = h3_surg2
            sc3 = sc3_surg2

    if tleft() > 25:
        h3_sg3, sc3_sg3 = subgradient_polish(h3, n3, t3, n_steps=200, n_active=4,
                                              lr_init=0.0008, lr_decay=0.999,
                                              deadline=start_time + budget_s - 20)
        if sc3_sg3 < compute_c5(h3):
            h3 = h3_sg3
            sc3 = sc3_sg3

    sc3_final = compute_c5(h3)
    if sc3_final < best_score:
        best_score = sc3_final
        best_h = h3.copy()

    n_par_check = n_par
    h_par = interpolate_to(best_h, n_par_check, n_par_check / 2.0)
    h_par = slsqp_opt(h_par, n_par_check, n_par_check / 2.0, [5000, 15000, 30000], 1500, 1e-17,
                      deadline=start_time + budget_s - 15)
    sc_par = compute_c5(h_par)

    if sc_par < best_score:
        best_h = h_par
        best_score = sc_par
        final_n = n_par_check
    else:
        final_n = len(best_h)

    best_h = project_feasible(best_h, final_n / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, final_n
