def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp(h_current, n):
    """Solve the LP with c = 1 - h_current fixed."""
    c = 1.0 - h_current
    target = n / 2
    
    # Build the constraint matrix
    # Variables: h[0], ..., h[n-1], t (index n)
    # Minimize: t (so c_vec = [0,...,0, 1])
    # Constraints: A_ub @ vars <= b_ub
    #   For each shift j: sum of h[...]*c[...] <= t
    #   i.e., sum of h[...]*c[...] - t <= 0
    # Equality: sum(h) = target
    # Bounds: 0 <= h[i] <= 1, t >= 0
    
    n_corr = 2 * n - 1
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for j in range(n_corr):
        s = j - (n - 1)  # shift
        if s >= 0:
            # h[n+s] * c[n] for n in 0..n-1-s
            for ni in range(n - s):
                A_ub[j, ni + s] += c[ni]
        else:
            # h[n+s] * c[n] for n in -s..n-1
            for ni in range(-s, n):
                A_ub[j, ni + s] += c[ni]
        A_ub[j, n] = -1.0  # -t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h_current

def alternating_sequence(h_init, n, max_iters, deadline):
    """Alternating LP minimization."""
    h_cur = h_init.copy()
    target = n / 2
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = solve_lp(h_cur, n)
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        # Check convergence
        if np.max(np.abs(h_new - h_cur)) < 1e-10:
            break
        h_cur = h_new
    
    return best_h, best_score

    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0

    def con_eq(x):
        return np.sum(x) - target

        return np.ones(n)

        c_max = float(np.max(corr))

                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h_init, n, n_passes=500, deadline=None, tol=1e-12):
    """Coordinate descent with exact 1D subproblem solving."""
    h_cur = h_init.copy()
    target = n / 2
    corr_base = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_max = np.max(corr_base)
    best_h = h_cur.copy()
    best_max = cur_max
    n_corr = 2 * n - 1
    
    for p in range(n_passes):
        if deadline is not None and time.time() > deadline:
            break
        
        for i in range(n):
            # Compute d[k] for this coordinate i
            s = np.arange(n_corr) - (n - 1)
            idx1 = i - s
            idx2 = i + s
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            d = np.zeros(n_corr)
            if np.any(valid1):
                d[valid1] += 1.0 - h_cur[idx1[valid1]]
            if np.any(valid2):
                d[valid2] -= h_cur[idx2[valid2]]
            
            old_val = h_cur[i]
            
            # Minimize max_k (d[k] * (x - old_val) + corr_base[k]) over x in [0,1]
            # Let o = x - old_val, range is [-old_val, 1-old_val]
            lo, hi = -old_val, 1.0 - old_val
            
            # Quick check: is the current point already optimal?
            # Subgradient at o=0 contains 0 if min(d[active]) <= 0 <= max(d[active])
            # where active = argmax of corr_base
            active = corr_base == np.max(corr_base)
            if np.sum(active) > 1 or np.max(corr_base) == corr_base[np.argmax(corr_base)]:
                if np.min(d[active]) <= 0 and np.max(d[active]) >= 0:
                    # Current point is optimal (0 is in subgradient)
                    # But we might still want to move to maintain feasibility
                    continue
            
            # Ternary search
            for _ in range(35):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                if np.max(d * m1 + corr_base) < np.max(d * m2 + corr_base):
                    hi = m2
                else:
                    lo = m1
            
            x_opt = (lo + hi) / 2 + old_val
            x_opt = float(np.clip(x_opt, 0.0, 1.0))
            
            if abs(x_opt - old_val) > 1e-14:
                h_cur[i] = x_opt
                corr_base += d * (x_opt - old_val)
        
        # Project to feasibility after each pass
        h_cur = project_feasible(h_cur, target)
        corr_base = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cur_max = np.max(corr_base)
        if cur_max < best_max:
            best_max = cur_max
            best_h = h_cur.copy()
        
        if cur_max - best_max < tol:
            break
    
    return best_h, float(best_max) * (2.0 / n)

        return best_h, best_score, best_n

    # Phase 5: Try higher resolution n=600
    n2 = 600
    t_old = np.linspace(0, 2, n0)
    t_new = np.linspace(0, 2, n2)
    h_up = np.interp(t_new, t_old, best_h)
    h_up = project_feasible(h_up, n2 / 2.0)

    d5 = t0 + 950
    h_cd3, sc_cd3 = coord_descent(h_up, n2, 300, d5)
    if sc_cd3 < best_score:
        best_score = sc_cd3
        best_h = h_cd3
        best_n = n2

    # Phase 6: SLSQP at higher resolution
    d6 = t0 + 1150
    if elapsed() < d6 - 30:
        h_sls2, sc_sls2 = slsqp_round(best_h, best_n, best_n / 2.0, [500, 2000], 1000)
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2

    # Phase 7: Final coordinate descent polish
    d7 = t0 + budget_s - 20
    h_cd4, sc_cd4 = coord_descent(best_h, best_n, 300, d7)
    if sc_cd4 < best_score:
        best_score = sc_cd4
        best_h = h_cd4
        best_n = best_n

    best_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, best_n
