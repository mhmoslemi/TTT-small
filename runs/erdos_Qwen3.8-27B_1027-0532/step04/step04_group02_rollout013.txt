def find_best_swap(h, corr, s_range):

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_val = np.inf
    best_i, best_j = -1, -1
    
    for i in ones:
        # Compute P_i[s] = h[i+s] + h[i-s]
        P_i = np.zeros(2*n-1)
        idx1 = i + s_range
        mask1 = (idx1 >= 0) & (idx1 < n)
        P_i[mask1] += h[idx1[mask1]]
        idx2 = i - s_range
        mask2 = (idx2 >= 0) & (idx2 < n)
        P_i[mask2] += h[idx2[mask2]]
        
        target = corr + P_i  # shape (2n-1,)
        
        # For all j in zeros: compute Q_j[s] = h[j+s] + h[j-s]
        # Vectorized:
        j_arr = zeros[:, None]  # (n_zeros, 1)
        js = j_arr + s_range[None, :]  # (n_zeros, 2n-1)
        jsk = j_arr - s_range[None, :]
        
        Q = np.zeros((len(zeros), 2*n-1))

- Original: h[i]=1, h[j]=0
- After swap: h[i]=0, h[j]=1
- corr[s] = sum_x h[x] * (1-h[x+s]) for valid x

- Term where x+s=j, i.e., x=j-s: was h[j-s]*(1-h[j]) = h[j-s]*1 = h[j-s], now h[j-s]*(1-h[j]) = h[j-s]*0 = 0. Change: -h[j-s]
- Term where x=j: was h[j]*(1-h[j+s]) = 0*(1-h[j+s]) = 0, now h[j]*(1-h[j+s]) = 1*(1-h[j+s]) = 1-h[j+s]. Change: +(1-h[j+s])
- Term where x+s=i, i.e., x=i-s: was h[i-s]*(1-h[i]) = h[i-s]*0 = 0, now h[i-s]*(1-h[i]) = h[i-s]*1 = h[i-s]. Change: +h[i-s]

Total delta[s] = (h[i+s]-1) + (-h[j-s]) + (1-h[j+s]) + h[i-s]
= h[i+s] - 1 - h[j-s] + 1 - h[j+s] + h[i-s]
= h[i+s] + h[i-s] - h[j+s] - h[j-s]
= (h[i+s] + h[i-s]) - (h[j+s] + h[j-s])

Great, this confirms the formula: delta[s] = P_i[s] - Q_j[s] where P_i[s] = h[i+s] + h[i-s] and Q_j[s] = h[j+s] + h[j-s].

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)


def project_feasible(h, target_sum):

    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-14:

            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:

        order = np.argsort(-room)

        for i in order:

            sub = min(room[i], remaining)
            h[i] -= sub
