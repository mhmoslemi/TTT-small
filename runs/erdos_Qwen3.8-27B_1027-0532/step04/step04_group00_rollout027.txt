    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def level_lags(h, n, target, n_active=5, max_iters=200, lr=0.01):
    """Try to level the worst n_active lags by finding a direction that reduces all of them."""
    best_h = h.copy()
    best_score = compute_c5(h)
    h_cur = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        c_max = corr[top_idx[0]]
        # Check if lags are already well-leveled
        spread = corr[top_idx[0]] - corr[top_idx[-1]]
        if spread < 1e-6:
            break
        
        # Compute gradient for each top lag
        grads = []
        for k in top_idx:
            k = int(k)
            lag = k - (n - 1)
            if lag == 0:
                g = (1.0 - 2.0 * h_cur) * dx
            elif lag > 0:
                g = np.zeros(n)
                j_pos = np.arange(0, n - lag)
                g[j_pos] += (1.0 - h_cur[j_pos + lag]) * dx
                j_neg = np.arange(lag, n)
                g[j_neg] -= h_cur[j_neg - lag] * dx
            else:
                l = -lag
                g = np.zeros(n)
                j_a = np.arange(0, n - l)
                g[j_a] -= h_cur[j_a + l] * dx
                j_b = np.arange(l, n)
                g[j_b] += (1.0 - h_cur[j_b - l]) * dx
            grads.append(g)
        
        # Use the average gradient (equal weights) as descent direction
        avg_grad = np.mean(grads, axis=0)
        
        # Project to be orthogonal to the constraint (sum = target)
        # The constraint is sum(h) = target, so the normal is (1,1,...,1)
        # Project: avg_grad -= (avg_grad . ones / n) * ones
        mean_comp = np.mean(avg_grad)
        avg_grad_proj = avg_grad - mean_comp
        
        # Step
        h_new = h_cur - lr * avg_grad_proj
        h_new = np.clip(h_new, 0.0, 1.0)
        # Re-project to satisfy sum constraint
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    return best_h, best_score

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

            grad = np.zeros(n)
            j_a = np.arange(0, n - l)
            grad[j_a] -= h_cur[j_a + l] * dx
            j_b = np.arange(l, n)
            grad[j_b] += (1.0 - h_cur[j_b - l]) * dx
        
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-13:
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
            eps = min(eps * 1.04, 0.05)
        else:
            eps *= 0.85
            if eps < 1e-8:
                break
    return best_h, best_score

def level_lags(h, n, target, n_active=5, max_iters=300, lr=0.005):
    best_h = h.copy()
    best_score = compute_c5(h)
    h_cur = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        grads = []
        for k in top_idx:
            k = int(k)
            lag = k - (n - 1)
            if lag == 0:
                g = (1.0 - 2.0 * h_cur) * dx
            elif lag > 0:
                g = np.zeros(n)
                j_pos = np.arange(0, n - lag)
                g[j_pos] += (1.0 - h_cur[j_pos + lag]) * dx
                j_neg = np.arange(lag, n)
                g[j_neg] -= h_cur[j_neg - lag] * dx
            else:
                l = -lag
                g = np.zeros(n)
                j_a = np.arange(0, n - l)
                g[j_a] -= h_cur[j_a + l] * dx
                j_b = np.arange(l, n)
                g[j_b] += (1.0 - h_cur[j_b - l]) * dx
            grads.append(g)
        
        avg_grad = np.mean(grads, axis=0)
        mean_comp = np.mean(avg_grad)
        avg_grad_proj = avg_grad - mean_comp
        
        h_new = h_cur - lr * avg_grad_proj
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
        else:
            lr *= 0.92
            if lr < 1e-8:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)
    
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    print(f"Initial: {best_score:.6f}")
    
    # Phase 1: L-BFGS-B with increasing alpha (continuation method)
    # This is the main workhorse - fast with analytical gradients
    alpha_schedule_1 = [5, 20, 50, 100, 200, 500, 1000]
    for a in alpha_schedule_1:
        if tleft() < 60:
            break
        h_ref = lbfgs_smooth(best_h, n, target, a, penalty=1e6, maxiter=2000)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    print(f"After L-BFGS-B phase 1: {best_score:.6f}")
    
    # Phase 2: SLSQP with moderate alpha (handles equality exactly)
    if tleft() > 200:
        for a in [500, 1000, 2000, 5000]:
            if tleft() < 60:
                break
            h_ref = slsqp_smooth(best_h, n, target, a, maxiter=2000)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    print(f"After SLSQP phase 2: {best_score:.6f}")
    
    # Phase 3: Subgradient polish
    if tleft() > 100:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=500, n_active=4,
                                         lr_init=0.003, lr_decay=0.996)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    print(f"After subgradient phase 3: {best_score:.6f}")
    
    # Phase 4: Level lags
    if tleft() > 80:
        h_lvl, sc_lvl = level_lags(best_h, n, target, n_active=5, max_iters=300, lr=0.005)
        if sc_lvl < best_score:
            best_score = sc_lvl
            best_h = h_lvl
    print(f"After level lags phase 4: {best_score:.6f}")
    
    # Phase 5: Worst lag surgery
    if tleft() > 80:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, max_steps=400, initial_epsilon=0.003)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
    print(f"After surgery phase 5: {best_score:.6f}")
    
    # Phase 6: High-alpha L-BFGS-B
    if tleft() > 150:
        for a in [5000, 10000, 20000, 50000]:

                best_h = h_ref
    print(f"After high-alpha phase 6: {best_score:.6f}")
    
    # Phase 7: Multi-resolution
    if tleft() > 250:
        n_coarse = max(n // 4, 20)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        for a in [10, 50, 200, 500, 1000, 3000, 5000, 10000]:

                                                        n_steps=200, n_active=3,
                                                        lr_init=0.005, lr_decay=0.995)
        if sc_coarse_sg < compute_c5(h_coarse):
            h_coarse = h_coarse_sg

        sc_up = compute_c5(h_up)
