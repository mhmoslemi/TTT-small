        frac = min(1.0, (time.time() - t_start) / total)
        T = max(t0 * (t1 / t0) ** frac, 1e-12)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        corr_new = _corr(h)
        mc_new = float(np.max(corr_new))
        delta = mc_new - curr_mc
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            corr = corr_new
            curr_mc = mc_new
            if mc_new < best_mc:
                best_mc = mc_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_mc


def _il_binary(h, n, deadline, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_mc = float(np.max(_corr(h_cur)))
    while time.time() < deadline:
        rem = deadline - time.time()
        if rem < 3:
            break
        phase_time = min(15.0, 0.35 * rem)
        phase_end = time.time() + phase_time
        h_new, mc_new = _sa_pair_swap(h_cur, n, phase_end, rng, 2.0, 0.003)
        if mc_new < best_mc:
            best_mc = mc_new
            best_h = h_new.copy()
        h_cur = h_new.copy()
        n_pert = rng.randint(3, 18)
        for _ in range(n_pert):
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_cur[a], h_cur[b] = h_cur[b], h_cur[a]
    return best_h, best_mc


def _il_binary_gauss(h, n, deadline, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_mc = float(np.max(_corr(h_cur)))
    while time.time() < deadline:
        rem = deadline - time.time()
        if rem < 3:
            break
        phase_time = min(12.0, 0.3 * rem)
        phase_end = time.time() + phase_time
        h_new, mc_new = _sa_pair_swap(h_cur, n, phase_end, rng, 1.5, 0.005)
        if mc_new < best_mc:
            best_mc = mc_new
            best_h = h_new.copy()
        noise = rng.randn(n) * 0.3
        h_pert = (h_new + noise >= 0.5).astype(np.float64)
        h_pert = _fix_binary(h_pert, n, rng)
        h_cur = h_pert
    return best_h, best_mc


def _soft_obj(x):
    corr = _corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _block_direct_obj(x):
    h_full = _g_h_block.copy()
    h_full[_g_bstart:_g_bend] = np.clip(x, 0.0, 1.0)
    h_full = _project_feasible(h_full, _g_target)
    return _c5(h_full)


def _block_soft_obj(x):
    h_full = _g_h_block.copy()
    h_full[_g_bstart:_g_bend] = np.clip(x, 0.0, 1.0)
    h_full = _project_feasible(h_full, _g_target)
    corr = _corr(h_full)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _momentum_step(h, n, target, alpha, lr, momentum, n_iters, deadline):
    h_cur = h.copy()
    v = np.zeros(n)
    best_score = _c5(h_cur)
    best_h = h_cur.copy()
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        corr = _corr(h_cur)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        wt = np.exp(shifted)
        wsum = wt.sum()
        if wsum < 1e-300:
            break
        wt = wt / wsum
        grad = np.zeros(n)
        for s in range(-(n - 1), n):
            w = wt[s + n - 1]
            if w < 1e-12:
                continue
            if s >= 0:
                grad[s:] += w * (1.0 - h_cur[:n - s])
                grad[:n - s] -= w * h_cur[s:]
            else:
                grad[:n + s] += w * (1.0 - h_cur[-s:])
                grad[-s:] -= w * h_cur[:n + s]
        v = momentum * v - lr * grad
        h_new = _project_feasible(h_cur + v, target)
        sc = _c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_score


def _slsqp_polish(h, n, target, alpha, maxiter, deadline):
    _g_alpha = alpha
    _g_target = float(target)
    time_left = deadline - time.time()
    if time_left < 5:
        return h, _c5(h)
    try:
        res = minimize(_soft_obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _cons_eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = _project_feasible(res.x, target)
        sc = _c5(h_new)
        return h_new, sc
    except Exception:
        return h, _c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_h_block, _g_bstart, _g_bend, _g_target

    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    target = n // 2
    _g_target = float(target)

    best_h = h_init.copy()
    best_c5 = _c5(best_h)

    def _rem():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: Binary ILS with multiple starting points
    start_points = []

    h_b = (h_init >= 0.5).astype(np.float64)
    h_b = _fix_binary(h_b.copy(), n, rng)
    start_points.append(h_b)

    for _ in range(2):
        h_r = np.zeros(n)
        sel = rng.choice(n, target, replace=False)
        h_r[sel] = 1.0
        start_points.append(h_r)

    h_blk = np.zeros(n)
    pos = 0
    rem_ones = target
    block_len = rng.randint(4, 12)
    while pos < n and rem_ones > 0:
        b = min(block_len, n - pos, rem_ones)
        h_blk[pos:pos + b] = 1.0
        rem_ones -= b
        pos += b
        gap = rng.randint(4, 12)
        pos += min(gap, n - pos)
        block_len = rng.randint(3, 15)
    h_blk = _fix_binary(h_blk.copy(), n, rng)
    start_points.append(h_blk)

    n_starts = len(start_points)
    alloc = _rem() * 0.65
    time_per = max(15.0, alloc / n_starts)

    for si in range(n_starts):
        if _rem() < 50:
            break
        dl = time.time() + min(time_per, _rem() * 0.35)
        if dl <= time.time():
            break
        h_new, mc_new = _il_binary(start_points[si], n, dl, rng)
        c5_new = mc_new * 2.0 / n
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        if _rem() > 80 and si == 0:
            dl2 = time.time() + min(time_per * 0.5, _rem() * 0.2)
            if dl2 > time.time():
                h_new2, mc_new2 = _il_binary_gauss(start_points[si], n, dl2, rng)
                c5_new2 = mc_new2 * 2.0 / n
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2

    # Phase 2: More ILS on the best binary
    if _rem() > 60:
        h_bin_best = (best_h >= 0.5).astype(np.float64)
        h_bin_best = _fix_binary(h_bin_best.copy(), n, rng)
        mc_bin = float(np.max(_corr(h_bin_best)))
        if mc_bin * 2.0 / n < best_c5:
            best_c5 = mc_bin * 2.0 / n
            best_h = h_bin_best
        dl2 = time.time() + min(60.0, _rem() * 0.4)
        if dl2 > time.time():
            h_new, mc_new = _il_binary(h_bin_best, n, dl2, rng)
            c5_new = mc_new * 2.0 / n
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 3: Continuous SLSQP refinement
    if _rem() > 40:
        for a in [50000, 200000]:
            if _rem() < 20:
                break
            dl = time.time() + min(25.0, _rem() * 0.4)
            h_new, sc = _slsqp_polish(best_h, n, target, a, 400, dl)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new

    # Phase 4: Momentum gradient descent
    if _rem() > 50:
        dl = time.time() + min(50.0, _rem() * 0.5)
        for a in [5000, 20000]:
            if _rem() < 20:
                break
            h_new, sc = _momentum_step(best_h, n, target, a, 0.002, 0.8,
                                       200, min(dl, time.time() + 25.0))
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new

    # Phase 5: Block coordinate descent
    if _rem() > 40:
        _g_alpha = 100000.0
        for bs in [4, 3, 5]:
            if _rem() < 20:
                break
            n_blocks = n // bs
            h_bd = best_h.copy()
            cur_sc = _c5(h_bd)
            order = list(range(n_blocks))
            rng.shuffle(order)
            for bi in order:
                if _rem() < 10:
                    break
                s = bi * bs
                e = s + bs
                _g_h_block = h_bd
                _g_bstart = s
                _g_bend = e
                x0 = h_bd[s:e].copy()
                try:
                    res = minimize(_block_soft_obj, x0, method='Nelder-Mead',
                                  options={'maxiter': 1500, 'xatol': 1e-11, 'fatol': 1e-14})
                    h_test = h_bd.copy()
                    h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                    h_test = _project_feasible(h_test, target)
                    sc = _c5(h_test)
                    if sc < cur_sc:
                        cur_sc = sc
                        h_bd = h_test
                except Exception:
                    pass
            if cur_sc < best_c5:
                best_c5 = cur_sc
                best_h = h_bd

    # Phase 6: Final SLSQP polish
    if _rem() > 20:
        for a in [200000, 800000]:
            if _rem() < 10:
                break
            dl = time.time() + min(15.0, _rem() * 0.6)
            h_new, sc = _slsqp_polish(best_h, n, target, a, 300, dl)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new

    # Phase 7: Final active-set style
    if _rem() > 15:
        _g_alpha = 500000.0
        dl = time.time() + min(12.0, _rem() * 0.5)
        h_new, sc = _momentum_step(best_h, n, target, 50000, 0.0005, 0.7,
                                   100, dl)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new

    h_final = _project_feasible(best_h, target)
    final_c5 = _c5(h_final)
    return h_final, final_c5, n
