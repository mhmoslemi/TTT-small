                    pi = int(ons[rng.randint(len(ons))])
                    pj = int(zrs[rng.randint(len(zrs))])
                    h[pi] = 0
                    h[pj] = 1
            h = project_binary(h, target_int).astype(np.float64)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            no_imp = 0
            temp = t_init * 0.6
            continue

        r = rng.random()
        saved = None
        valid = False

        if r < 0.45:
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - (n - 1)
            if worst_lag >= 0:
                lo, hi = worst_lag, n
            else:
                lo, hi = 0, n + worst_lag
            if hi > lo:
                candidates = []
                for p in range(lo, hi):
                    q = p - worst_lag
                    if h[p] == 1 and h[q] == 0:
                        candidates.append(p)
            else:
                candidates = []
            if len(candidates) > 0:
                ii = int(candidates[rng.randint(len(candidates))])
            elif len(ones) > 0:
                ii = int(ones[rng.randint(len(ones))])
            else:
                continue
            if len(zeros) == 0:
                continue
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0
            h[jj] = 1
            saved = [(ii, 1.0), (jj, 0.0)]
            valid = True

        elif r < 0.65:
            if len(ones) < 2 or len(zeros) < 2:
                continue
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - (n - 1)
            if worst_lag >= 0:
                lo, hi = worst_lag, n
            else:
                lo, hi = 0, n + worst_lag
            candidates = []
            for p in range(lo, hi):
                q = p - worst_lag
                if h[p] == 1 and h[q] == 0:
                    candidates.append(p)
            if len(candidates) >= 2:
                sel = rng.choice(len(candidates), 2, replace=False)
                i1 = int(candidates[sel[0]])
                i2 = int(candidates[sel[1]])
            else:
                sel = rng.choice(len(ones), 2, replace=False)
                i1 = int(ones[sel[0]])
                i2 = int(ones[sel[1]])
            sel0 = rng.choice(len(zeros), 2, replace=False)
            j1 = int(zeros[sel0[0]])
            j2 = int(zeros[sel0[1]])
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            saved = [(i1, 1.0), (i2, 1.0), (j1, 0.0), (j2, 0.0)]
            valid = True

        elif r < 0.80:
            if len(ones) == 0 or len(zeros) == 0:
                continue
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0
            h[jj] = 1
            saved = [(ii, 1.0), (jj, 0.0)]
            valid = True

        elif r < 0.90:
            if len(ones) < 3 or len(zeros) < 3:
                continue
            sel1 = rng.choice(len(ones), 3, replace=False)
            sel0 = rng.choice(len(zeros), 3, replace=False)
            saved = []
            for idx in sel1:
                pos = int(ones[idx])
                h[pos] = 0
                saved.append((pos, 1.0))
            for idx in sel0:
                pos = int(zeros[idx])
                h[pos] = 1
                saved.append((pos, 0.0))
            valid = True

        else:
            blen = rng.randint(2, max(3, n // 60))
            start = rng.randint(0, max(1, n - blen))
            block = h[start:start + blen].copy()
            bs = block.sum()
            if bs == 0.0 or bs == float(blen):
                continue
            h[start:start + blen] = 1.0 - block
            saved = list(zip(range(start, start + blen), block))
            diff = int(float(blen) - 2.0 * bs)
            if diff > 0:
                avail = np.where(h == 1)[0]
                mask = (avail < start) | (avail >= start + blen)
                avail = avail[mask]
                if len(avail) < diff:
                    for pos, old in saved:
                        h[pos] = old
                    valid = False
                else:
                    idxs = rng.choice(len(avail), diff, replace=False)
                    for idx in idxs:
                        pos = int(avail[idx])
                        h[pos] = 0
                        saved.append((pos, 1.0))
                    valid = True
            elif diff < 0:
                avail = np.where(h == 0)[0]
                mask = (avail < start) | (avail >= start + blen)
                avail = avail[mask]
                if len(avail) < -diff:
                    for pos, old in saved:
                        h[pos] = old
                    valid = False
                else:
                    idxs = rng.choice(len(avail), -diff, replace=False)
                    for idx in idxs:
                        pos = int(avail[idx])
                        h[pos] = 1
                        saved.append((pos, 0.0))
                    valid = True
            else:
                valid = True

        if not valid:
            no_imp += 1
            temp = max(temp * cooldown, t_final)
            continue

        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        delta = new_c5 - cur_c5

        accept = False
        if delta < -1e-15:
            accept = True
        elif temp > 1e-14 and delta < 2.0:
            if rng.random() < np.exp(-delta / temp):
                accept = True

        if accept:
            corr = new_corr.astype(np.float64)
            cur_c5 = new_c5
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
            temp = max(temp * cooldown, t_final)
        else:
            for pos, old_val in saved:
                h[pos] = old_val
            no_imp += 1
            temp = max(temp * cooldown, t_final)

    return best_h, best_c5


def smooth_obj(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max) / dx, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def smooth_obj_jac(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max) / dx, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    denom = float(np.sum(exp_shifted))
    result = np.zeros(len(x))
    for k in range(len(corr)):
        w = exp_shifted[k] / denom
        lo = max(0, k - (n - 1))
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = i + k - (n - 1)
            if 0 <= j < n:
                result[i] += w * (2 * x[i] - 1) * dx
                result[j] -= w * (2 * x[j] - 1) * dx
    return result


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 500, 1000, 2000]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-12:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    h_init_arr = np.array(initial_h_values, dtype=float)
    n_init = len(h_init_arr)
    target_init = n_init / 2.0

    best_h = h_init_arr.copy()
    best_c5 = compute_c5(best_h)

    h_bin = project_binary(h_init_arr, target_init)
    h_ls, c5_ls = fast_local_search(h_bin, int(target_init),
                                     min(180.0, budget_s - 80.0), start, rng,
                                     t_init=0.04, t_final=0.000005, cooldown=0.99997)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    elapsed = time.time() - start
    remaining = budget_s - 20.0 - elapsed

    if remaining > 50:
        h_qr = make_qr_sequence(n_init, rng)
        h_ls, c5_ls = fast_local_search(h_qr, int(n_init / 2.0),
                                         min(100.0, remaining), start, rng,
                                         t_init=0.03, t_final=0.00001, cooldown=0.99997)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    elapsed = time.time() - start
    remaining = budget_s - 20.0 - elapsed

    n_use = len(best_h)
    target_use = int(n_use / 2.0)

    n_trials = int(min(10, max(2, remaining / 55.0)))
    for trial in range(n_trials):
        elapsed = time.time() - start
        remaining = budget_s - 20.0 - elapsed
        if remaining < 45:
            break
        noise_scale = 0.12 + 0.04 * trial
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert_bin = project_binary(h_pert, target_use)
        time_alloc = min(70.0, remaining - 5.0)
        h_ls, c5_ls = fast_local_search(h_pert_bin, target_use, time_alloc, start, rng,
                                         t_init=0.025, t_final=0.00001, cooldown=0.99997)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    elapsed = time.time() - start
    remaining = budget_s - 20.0 - elapsed

    for dt in [2, -2, 4, -4, 6, -6, 10, -10, 8, -8, 14, -14, 20, -20, 30, -30]:
        nt = n_init + dt
        if nt < 200 or nt > 1200:
            continue
        if remaining < 45:
            break
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, nt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = project_binary(h_interp, nt / 2.0)
        time_alloc = min(55.0, remaining - 5.0)
        h_ls, c5_ls = fast_local_search(h_bin_try, int(nt / 2.0), time_alloc, start, rng,
                                         t_init=0.03, t_final=0.00001, cooldown=0.99997)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining -= time_alloc + 1.0

    elapsed = time.time() - start
    remaining = budget_s - 10.0 - elapsed

    if remaining > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cont, sc_cont = continuous_refine(best_h, n_use, target_use,
                                             min(remaining - 5.0, 80.0), start)
        if sc_cont < best_c5:
            best_c5 = sc_cont
            best_h = h_cont.copy()

    elapsed = time.time() - start
    remaining = budget_s - 5.0 - elapsed
    if remaining > 8:
        n_use = len(best_h)
        h_bin_final = project_binary(best_h, n_use / 2.0)
        h_ls, c5_ls = fast_local_search(h_bin_final, int(n_use / 2.0), remaining, start, rng,
                                         t_init=0.008, t_final=0.000001, cooldown=0.99999)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    n_final = len(best_h)
    if abs(best_h.sum() - n_final / 2.0) > 0.01:
        best_h = project_binary(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
