import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def slsqp_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)

def slsqp_solve(h_start, target_sum, alpha_list, maxiter):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for alpha, topk in alpha_list:
        try:
            res = minimize(
                slsqp_obj, h_cur, args=(alpha, topk),
                jac=slsqp_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-14}

            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def lp_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:

    else:

    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)

        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target_sum)
    return None

                    if sc < best_score - 1e-16:

                        best_h = h_frac.copy()
                        h = h_frac
                        break
                else:

    return best_h, best_score

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

from scipy.optimize import minimize, linprog
import time

_G_N = 0
_G_TARGET = 0.0
_G_START = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:

            if lag < n:

        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def _slsqp_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)

        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def _slsqp_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)
