**Another idea: Frank-Wolfe (Conditional Gradient) method**

The problem is:
min_h max_k f_k(h)
subject to h ∈ P = {h : 0 ≤ h ≤ 1, 1^T h = n/2}

where f_k(h) = ∑_i h_i (1-h_{i+k}) dx.

Frank-Wolfe iterates:
1. Compute g = subgradient of max_k f_k at current h (i.e., gradient of the active f_k*)
2. Solve the linear minimization: δ = argmin_{d ∈ P} g^T d
3. Update: h ← (1-γ)h + γδ

The linear minimization over P is easy: d_i = 1 if g_i < min(g), d_i = 0 if g_i > max(g), and partially filled for the median value (to satisfy the sum constraint).

This is a nice approach because:
- Each iteration is O(n) (compute gradient + solve LP)
- It naturally handles the constraints
- It's guaranteed to converge (though slowly)

def block_lp_cd(h_init, n, target, block_size, max_passes, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    n_half = n - 1
    idx_arr = np.arange(n)
    
    for sweep in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        n_blocks = n // block_size
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            
            start_idx = b * block_size
            end_idx = start_idx + block_size
            block = h[start_idx:end_idx].copy()
            
            # Compute coeff matrix: (n_full) x block_size
            coeff = np.zeros((n_full, block_size))
            for j in range(block_size):
                p = start_idx + j
                s = np.arange(-(n-1), n)
                # (1-h[p+s]) for valid
                idx1 = p + s
                valid1 = (idx1 >= 0) & (idx1 < n)
                term1 = np.zeros(n_full)
                term1[valid1] = 1.0 - h[idx1[valid1]]
                # -h[p-s] for valid
                idx2 = p - s
                valid2 = (idx2 >= 0) & (idx2 < n)
                term2 = np.zeros(n_full)
                term2[valid2] = -h[idx2[valid2]]
                coeff[:, j] = term1 + term2
            
            # Current correlations
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            
            # LP: min t s.t. corr + dx * coeff @ delta <= t, 0 <= block+delta <= 1, sum(delta)=0
            # Variables: [delta(0), ..., delta(m-1), t]
            m = block_size
            # Objective: [0, ..., 0, 1]
            c_vec = np.zeros(m + 1)
            c_vec[-1] = 1.0
            
            # Inequality constraints: A_ub @ x <= b_ub
            # corr + dx * coeff @ delta - t <= 0  =>  dx*coeff @ delta - t <= -corr
            A_ub = np.zeros((n_full, m + 1))

            A_ub[:, m] = -1.0
            b_ub = -corr

            A_eq = np.zeros((1, m + 1))
            A_eq[0, :m] = 1.0

Let me also add an important technique: **adaptive grid refinement for the pairwise CD**. Instead of a fixed grid, use the fact that the objective is convex PL to find the exact minimum.

For a convex PL function f(δ) = max_s (a_s + b_s * δ), the minimum is at a breakpoint or boundary. The breakpoints are where two linear pieces cross. If I sort the pieces by slope (b_s), the "upper envelope" changes at most n-1 times. So there are at most n-1 breakpoints on the upper envelope.

I can find these by:
1. Sort the linear functions by slope
2. Walk through them, maintaining the current "active" function
3. When a new function becomes dominant, record the breakpoint

This is O(n log n) per pair, which is better than the grid approach.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_mask = int(np.sum(mask))
        if n_mask == 0:
            break
        step = diff / n_mask
        h = h.copy()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def frank_wolfe(h_init, n, target, max_iter, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    idx_arr = np.arange(n)
    n_target = int(target)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        
        g = np.zeros(n)
        k = k_star
        valid1 = (idx_arr + k >= 0) & (idx_arr + k < n)
        g[valid1] += (1.0 - h[idx_arr[valid1] + k])
        valid2 = (idx_arr - k >= 0) & (idx_arr - k < n)
        g[valid2] -= h[idx_arr[valid2] - k]
        g *= dx
        
        idx_sel = np.argpartition(g, n_target)[:n_target]
        d = np.zeros(n)
        d[idx_sel] = 1.0
        
        gamma_t = 2.0 / (it + 2.0)
        h_new = (1.0 - gamma_t) * h + gamma_t * d
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def block_lp_cd(h_init, n, target, block_size, max_passes, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    s = np.arange(-(n-1), n)
    
    # Precompute coefficient vectors for each position
    # coeff[p, m] = (1-h[p+s_m])_{valid} - h[p-s_m]_{valid}
    # We'll recompute this as h changes
    
    for sweep in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        n_blocks = n // block_size
        # Randomize block order
        block_order = np.random.permutation(n_blocks)
        
        any_improvement = False
        for b in block_order:
            if time.time() - start > time_limit:
                break
            
            start_idx = b * block_size
            end_idx = start_idx + block_size
            block = h[start_idx:end_idx].copy()
            m = block_size
            
            # Compute coeff matrix (n_full x m)
            coeff = np.zeros((n_full, m))
            for j in range(m):
                p = start_idx + j
                idx1 = p + s
                v1 = (idx1 >= 0) & (idx1 < n)
                term1 = np.zeros(n_full)
                term1[v1] = 1.0 - h[idx1[v1]]
                idx2 = p - s
                v2 = (idx2 >= 0) & (idx2 < n)
                term2 = np.zeros(n_full)
                term2[v2] = -h[idx2[v2]]
                coeff[:, j] = term1 + term2
            
            # LP: min t s.t. dx*coeff@delta - t <= -corr, bounds on delta
            # Variables: [delta_0, ..., delta_{m-1}, t]
            c_vec = np.zeros(m + 1)
            c_vec[-1] = 1.0
            
            A_ub = np.zeros((n_full, m + 1))
            A_ub[:, :m] = dx * coeff
            A_ub[:, m] = -1.0
            b_ub = -corr
            
            bounds = [(-block[j], 1.0 - block[j]) for j in range(m)]
            bounds.append((0.0, None))
            
            A_eq = np.zeros((1, m + 1))
            A_eq[0, :m] = 1.0
            b_eq = np.array([0.0])
            
            try:
                res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             A_eq=A_eq, b_eq=b_eq, method='highs')
                if res.status == 0:
                    delta = res.x[:m]
                    new_block = np.clip(block + delta, 0.0, 1.0)
                    # Fix sum
                    adj = 0.0
                    for j in range(m):
                        adj += new_block[j] - block[j]
                    # Actually, the LP constraint sum(delta)=0 should handle this
                    h[start_idx:end_idx] = new_block
                    any_improvement = True
            except:
                pass
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if not any_improvement and sweep > 5:
            break
    
    return best_h, best_c5

def pairwise_cd(h_init, n, target, max_sweeps, time_limit, grid_points=100, pairs_per_sweep=10000):

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    s = np.arange(-(n-1), n)
    rng = np.random.RandomState(123)
    
    # Precompute B: B[p, m] = h[p+s_m]_{valid} + h[p-s_m]_{valid}
    def compute_B(hh):
        Bmat = np.zeros((n, n_full))
        for p in range(n):
            idx1 = p + s
            v1 = (idx1 >= 0) & (idx1 < n)
            Bmat[p, v1] += hh[idx1[v1]]
            idx2 = p - s
            v2 = (idx2 >= 0) & (idx2 < n)
            Bmat[p, v2] += hh[idx2[v2]]
        return Bmat
    
    B = compute_B(h)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Select pairs (random)
        all_i = np.arange(n)
        all_j = np.arange(n)

        interior_mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        interior_idxs = np.where(interior_mask)[0]

            break

        n_pairs = max(n_pairs, 100)

        for _ in range(n_pairs):
            i = interior_idxs[rng.randint(len(interior_idxs))]
            j = interior_idxs[rng.randint(len(interior_idxs))]
            if i == j:
                continue
            
            A = B[j] - B[i]
            
            delta_lo = max(-h[i], h[j] - 1.0)
            delta_hi = min(1.0 - h[i], h[j])
            if delta_hi - delta_lo < 1e-12:

            delta_grid = np.linspace(delta_lo, delta_hi, grid_points)
            obj = np.max(corr[None, :] + delta_grid[:, None] * A[None, :], axis=1)
            best_idx = np.argmin(obj)
            best_delta = delta_grid[best_idx]
            
            if abs(best_delta) > 1e-10:
                h[i] += best_delta
                h[j] -= best_delta
                h[i] = np.clip(h[i], 0.0, 1.0)
                h[j] = np.clip(h[j], 0.0, 1.0)
                # Update B rows for i and j
                for p in [i, j]:
                    B[p, :] = 0.0
                    idx1 = p + s
                    v1 = (idx1 >= 0) & (idx1 < n)
                    B[p, v1] += h[idx1[v1]]
                    idx2 = p - s
                    v2 = (idx2 >= 0) & (idx2 < n)
                    B[p, v2] += h[idx2[v2]]
                any_improvement = True
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if not any_improvement:
            break
    
    return best_h, best_c5

def smooth_max(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h_init, n, target, alphas, maxiter_per_alpha, time_limit):

    h_cur = h_init.copy()
    best_h = h_cur.copy()

    bounds = [(0.0, 1.0)] * n
    
    # Use the constraint as a penalty instead (faster)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj(hh):
            val = smooth_max(hh, alpha)
            penalty = 1000.0 * (np.sum(hh) - target) ** 2
            return val + penalty
        
        def obj_grad(hh):
            # Approximate gradient
            eps = 1e-7
            n_pts = len(hh)
            grad = np.zeros(n_pts)
            for i in range(n_pts):
                hh[i] += eps
                f1 = obj(hh)
                hh[i] -= 2*eps
                f2 = obj(hh)
                hh[i] += eps
                grad[i] = (f1 - f2) / (2*eps)
            return grad

            h_new = project_feasible(res.x, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_mask = int(np.sum(mask))
        if n_mask == 0:
            break
        step = diff / n_mask
        h = h.copy()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def frank_wolfe(h_init, n, target, max_iter, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    idx_arr = np.arange(n)
    n_target = int(target)
    momentum_buf = np.zeros(n)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - n_half

        g = np.zeros(n)
        k = k_star
        valid1 = (idx_arr + k >= 0) & (idx_arr + k < n)
        g[valid1] += (1.0 - h[idx_arr[valid1] + k])
        valid2 = (idx_arr - k >= 0) & (idx_arr - k < n)
        g[valid2] -= h[idx_arr[valid2] - k]
        g *= dx

        idx_sel = np.argpartition(g, n_target)[:n_target]
        d = np.zeros(n)
        d[idx_sel] = 1.0

        gamma_t = 2.0 / (it + 2.0)
        h_old = h.copy()
        h = (1.0 - gamma_t) * h + gamma_t * d
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5:

            best_h = h.copy()

    return best_h, best_c5


def compute_B_matrix(h, n):
    n_full = 2 * n - 1
    s = np.arange(-(n - 1), n)
    Bmat = np.zeros((n, n_full))
    for p in range(n):
        idx1 = p + s
        v1 = (idx1 >= 0) & (idx1 < n)
        Bmat[p, v1] += h[idx1[v1]]
        idx2 = p - s
        v2 = (idx2 >= 0) & (idx2 < n)
        Bmat[p, v2] += h[idx2[v2]]
    return Bmat


def block_lp_cd(h_init, n, target, block_size, max_passes, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    s = np.arange(-(n - 1), n)
    rng = np.random.RandomState(77)

    for sweep in range(max_passes):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        n_blocks = n // block_size
        block_order = rng.permutation(n_blocks)
        any_improvement = False

        for b in block_order:
            if time.time() - start > time_limit:
                break
            start_idx = b * block_size
            end_idx = start_idx + block_size
            block = h[start_idx:end_idx].copy()
            m = block_size

            coeff = np.zeros((n_full, m))
            for j in range(m):
                p = start_idx + j
                idx1 = p + s
                v1 = (idx1 >= 0) & (idx1 < n)
                term1 = np.zeros(n_full)
                term1[v1] = 1.0 - h[idx1[v1]]
                idx2 = p - s
                v2 = (idx2 >= 0) & (idx2 < n)
                term2 = np.zeros(n_full)
                term2[v2] = -h[idx2[v2]]
                coeff[:, j] = term1 + term2

            c_vec = np.zeros(m + 1)
            c_vec[-1] = 1.0
            A_ub = np.zeros((n_full, m + 1))
            A_ub[:, :m] = dx * coeff
            A_ub[:, m] = -1.0
            b_ub = -corr
            bounds = [(-block[j], 1.0 - block[j]) for j in range(m)]
            bounds.append((0.0, None))
            A_eq = np.zeros((1, m + 1))
            A_eq[0, :m] = 1.0
            b_eq = np.array([0.0])

            try:
                res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                              A_eq=A_eq, b_eq=b_eq, method='highs')
                if res.status == 0:
                    delta = res.x[:m]
                    new_block = np.clip(block + delta, 0.0, 1.0)
                    h[start_idx:end_idx] = new_block
                    any_improvement = True
            except Exception:

        c5_val = compute_c5(h)

            best_h = h.copy()

def pairwise_cd(h_init, n, target, max_sweeps, time_limit, grid_points=80):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    s = np.arange(-(n - 1), n)

    B = compute_B_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        interior_mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        interior_idxs = np.where(interior_mask)[0]

        if len(interior_idxs) < 2:
            break

        any_improvement = False
        n_interior = len(interior_idxs)

        for ii in range(n_interior):
            i = interior_idxs[ii]
            if time.time() - start > time_limit:
                break
            for jj in range(ii + 1, n_interior):
                j = interior_idxs[jj]

                A = B[j] - B[i]
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                delta_grid = np.linspace(delta_lo, delta_hi, grid_points)
                obj = np.max(corr[None, :] + delta_grid[:, None] * A[None, :], axis=1)
                best_idx = np.argmin(obj)
                best_delta = delta_grid[best_idx]

                if abs(best_delta) > 1e-10:
                    h[i] += best_delta
                    h[j] -= best_delta
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    for p in [i, j]:
                        B[p, :] = 0.0
                        idx1 = p + s
                        v1 = (idx1 >= 0) & (idx1 < n)
                        B[p, v1] += h[idx1[v1]]
                        idx2 = p - s
                        v2 = (idx2 >= 0) & (idx2 < n)
                        B[p, v2] += h[idx2[v2]]
                    any_improvement = True

        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        if not any_improvement:
            break

    return best_h, best_c5


def smooth_max_obj(hh, alpha):
    corr = np.correlate(hh, 1.0 - hh, mode='full')
    dx = 2.0 / len(hh)
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    bounds = [(0.0, 1.0)] * n

        def obj_fn(hh):
            val = smooth_max_obj(hh, alpha)

            return val + penalty
