    h_best = h_cur.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    smooth_val = c_max + log_sum / alpha
    p_k = exp_vals / np.sum(exp_vals)
    return smooth_val, p_k

def compute_gradient(h, p_k):
    n = len(h)
    grad = np.ones(n)
    for l in range(n):
        if l == 0:
            w_l = p_k[n-1]
        else:
            w_l = p_k[n-1+l] + p_k[n-1-l]
        if abs(w_l) < 1e-15:
            continue
        grad[:n-l] -= w_l * h[l:]
        grad[l:] -= w_l * h[:n-l]
    return grad

def pgd_phase(h, n, target, alphas, iters_per_alpha, deadline, lr_base=0.01):
    h_cur = h.copy()
    h_best = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        h_mom = h_cur.copy()
        v = np.zeros(n)  # velocity for momentum
        
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            smooth_val, p_k = smooth_max_and_weights(corr, alpha)
            grad = compute_gradient(h_cur, p_k)
            
            # Momentum
            v = 0.9 * v + grad
            h_new = project_feasible(h_cur - lr_base * v, target)
            
            # Accept if it improves
            sc_new = compute_c5(h_new)
            if sc_new < best_score + 0.001:  # Allow small increases for exploration
                h_cur = h_new
                if sc_new < best_score:
                    best_score = sc_new
                    h_best = h_cur.copy()
        
        # Reset momentum between alpha changes
        v = np.zeros(n)
    
    return h_best, best_score

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

def compute_gradient(h, p_k):
    n = len(h)
    grad = np.ones(n)
    for l in range(n):
        if l == 0:
            w_l = p_k[n-1]
        else:
            w_l = p_k[n-1+l] + p_k[n-1-l]
        if abs(w_l) < 1e-15:
            continue
        grad[:n-l] -= w_l * h[l:]
        grad[l:] -= w_l * h[:n-l]
    return grad

def pgd_optimize(h, n, target, alpha, n_iter, deadline, lr=0.05):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    v = np.zeros(n)
    momentum = 0.85
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        exp_vals = np.exp(shifted)
        p_k = exp_vals / np.sum(exp_vals)
        grad = compute_gradient(h_cur, p_k)
        
        v = momentum * v + grad
        h_new = project_feasible(h_cur - lr * v, target)
        
        sc_new = compute_c5(h_new)
        if sc_new <= best_score + 0.01:
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
        
        # Decay momentum slightly
        if it % 1000 == 0:
            v *= 0.5
    
    return best_h, best_score

def coord_descent_optimize(h, n, target, n_sweeps, grid_size, deadline):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    lags = np.arange(-(n-1), n)

        if time.time() > deadline:
            break
        
        improved = False
        for j in range(n):
            h_j = h_cur[j]
            best_hj = h_j
            best_val = np.max(corr)

            # Use a local search around current value
            if h_j < 0.1:
                t_grid = np.linspace(0, 0.5, grid_size)

                delta = t - h_j
                # Compute change in corr
                m1 = (j + lags >= 0) & (j + lags < n)
                m2 = (j - lags >= 0) & (j - lags < n)
                
                change = np.zeros(2*n-1)
                change[m1] += delta * (1.0 - h_cur[j + lags[m1]])
                m2_nz = m2 & (lags != 0)
                change[m2_nz] -= delta * h_cur[j - lags[m2_nz]]
                # lag=0 correction (the quadratic term)
                change[n-1] = delta * (1.0 - 2.0*h_j) - delta*delta
                
                new_corr = corr + change
                val = np.max(new_corr)
                if val < best_val:
                    best_val = val
                    best_hj = t

                m1 = (j + lags >= 0) & (j + lags < n)
                m2 = (j - lags >= 0) & (j - lags < n)
                
                change = np.zeros(2*n-1)

                m2_nz = m2 & (lags != 0)
                change[m2_nz] -= delta * h_cur[j - lags[m2_nz]]
                change[n-1] = delta * (1.0 - 2.0*h_j) - delta*delta
                
                corr += change
                h_cur[j] = best_hj
                improved = True
        
        # Project to maintain sum constraint
        h_cur = project_feasible(h_cur, target)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        
        if not improved:
            break
    
    return best_h, best_score
