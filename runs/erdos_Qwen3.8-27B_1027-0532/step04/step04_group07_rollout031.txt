import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(80):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def compute_all_grads(h):
    n = len(h)
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for idx in range(n_corr):
        lag = idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grads[idx, lag:] += (1.0 - h[:n - lag])
                grads[idx, :n - lag] -= h[lag:]
        else:
            m = -lag
            if m < n:
                grads[idx, :n - m] += (1.0 - h[m:])
                grads[idx, m:] -= h[:n - m]
    return grads

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def active_set_smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)

def slsqp_active_set(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                active_set_smooth_obj, h_cur, args=(alpha, topk),
                jac=active_set_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def lp_linearized_step(h_cur, target_sum, topk=None):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk is None or topk >= n_corr:
        top_idx = np.arange(n_corr)
    else:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None

def run_subgradient_adagrad(h_init, target_sum, max_iter, lr_init, lr_decay, topk, alpha,
                            noise_prob, noise_scale, patience, seed):

    n = len(h_init)
    h = h_init.copy()
    G = np.zeros(n)  # AdaGrad accumulator
    rho = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    t_iter = 0
    stagnation = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        G += grad * grad
        h_new = h - lr / np.sqrt(G + rho) * grad
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        h = h_new
        t_iter += 1
        lr = lr_init / (1.0 + lr_decay * t_iter)
        if stagnation > patience:
            break
    return best_h, best_score

from scipy.optimize import minimize, linprog
import time

_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(80):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def compute_all_grads(h):
    n = len(h)
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for idx in range(n_corr):
        lag = idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grads[idx, lag:] += (1.0 - h[:n - lag])
                grads[idx, :n - lag] -= h[lag:]
        else:
            m = -lag
            if m < n:
                grads[idx, :n - m] += (1.0 - h[m:])
                grads[idx, m:] -= h[:n - m]
    return grads


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def active_set_smooth_grad(x, alpha, topk):

    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]

    m = len(top_idx)

    A = np.zeros((m, n + 1))
    A[:, :n] = grads[top_idx]
    A[:, n] = -1.0
    b = np.array([float(grads[idx] @ h_cur) - corr[idx] for idx in top_idx])
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])

                                                    seed=seed + 3000 + n_try)
            if score < best_score:
                best_score = score
                best_h = h_new

        rem = remaining_time(start_time, budget_s)
        if rem > 50:
            n_cur = len(best_h)
            t_cur = n_cur / 2.0
            lp_b = min(30, rem - 40)
            h_new, score = lp_refinement(best_h, t_cur,
                                          max_iterations=150, topk=None,
                                          time_limit=lp_b)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 9: Final polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    rem = remaining_time(start_time, budget_s)
    if rem > 40:
        h_new, score = lp_refinement(best_h, target_use,
                                      max_iterations=100, topk=30,
                                      time_limit=min(25, rem - 30))
        if score < best_score:
            best_score = score
            best_h = h_new

    rem = remaining_time(start_time, budget_s)
    if rem > 25:
        alphas3 = [(5000, 10), (10000, 5)]
        h_ref, sc = slsqp_active_set(best_h, target_use, alphas3, 3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    rem = remaining_time(start_time, budget_s)
    if rem > 10:
        h_new, score = lp_refinement(best_h, target_use,
                                      max_iterations=50, topk=20,
                                      time_limit=max(3, rem - 8))
        if score < best_score:
            best_score = score
            best_h = h_new

    # Final fine subgradient
    rem = remaining_time(start_time, budget_s)
    if rem > 15:
        h_try = best_h + rng.randn(n_use) * 0.002
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient_adagrad(h_try, target_use,
                                                max_iter=1500, lr_init=0.1,
                                                lr_decay=0.02, topk=3,
                                                alpha=200, noise_prob=0.0,
                                                noise_scale=0.0, patience=200,
                                                seed=seed + 9999)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
