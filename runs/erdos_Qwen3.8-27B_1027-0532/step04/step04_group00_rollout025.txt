    dx = 2.0 / n
    corr = compute_corr_overlap(h)
    over = corr[n-1:2*n-1]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        if adj > 0 and 1.0 - h[mask].max() < adj:
            adj = (1.0 - h[mask]) / cnt * target_sum / h[mask].sum() if h[mask].sum() > 0 else 0
            h[mask] = np.clip(h[mask] + adj, 0, 1)

            h[mask] = np.clip(h[mask] - adj, 0, 1)
        else:
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)

        e = s + block_size
        offset = rng.randn() * scale
        h_new[s:e] += offset
    h_new = project_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    global _glob_n, _glob_target, _glob_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 15

    n = len(initial_h_values)
    target = n / 2.0
    _glob_n = n
    _glob_target = target

    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Warm start SLSQP (adaptive alpha sweep)
    p1_deadline = start_time + 0.15 * (budget_s - 15)
    h_ref, sc = warm_start_slsqp(best_h, n, target, p1_deadline)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 2: Multi-resolution refinement
    p2_deadline = start_time + 0.30 * (budget_s - 15)
    if time.time() < p2_deadline:
        h_ref, sc = multi_resolve_refine(best_h, n, target, p2_deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Active-set gradient descent with annealing
    p3_deadline = start_time + 0.50 * (budget_s - 15)
    if time.time() < p3_deadline - 30:
        h_ref, sc = active_set_gradient_descent(best_h, n, target, n_steps=2000, top_m=4,
                                                temp_init=0.01, temp_decay=0.999,
                                                deadline=p3_deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        # Follow up with SLSQP
        h_ref, sc = slsqp_optimize(best_h, n, target, [3000, 8000], 3000, 1e-16, p3_deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Second multi-resolution pass
    p4_deadline = start_time + 0.65 * (budget_s - 15)
    if time.time() < p4_deadline - 30:
        n_coarse = n // 2
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_resample(best_h, n_coarse, target_coarse)
        h_coarse, sc_c = slsqp_optimize(h_coarse, n_coarse, target_coarse,
                                         [100, 500, 2000, 6000], 3000, 1e-15, p4_deadline)
        h_fine = interpolate_resample(h_coarse, n, target)
        h_fine, sc_f = slsqp_optimize(h_fine, n, target, [2000, 6000, 15000], 3000, 1e-16,
                                       p4_deadline)
        sc = compute_c5(h_fine)
        if sc < best_score:
            best_score = sc
            best_h = h_fine

    # Phase 5: Structured perturbations + local optimization
    p5_deadline = start_time + 0.80 * (budget_s - 15)
    for attempt in range(5):
        if time.time() > p5_deadline:
            break
        scales = [0.03, 0.05, 0.02, 0.04, 0.06]
        h_pert = structured_perturb(best_h, n, target, rng, scales[attempt])
        h_ref, sc = slsqp_optimize(h_pert, n, target, [200, 1000, 4000, 10000], 2000, 1e-15,
                                    deadline - 60)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Final active-set gradient + SLSQP
    p6_deadline = deadline - 30
    if time.time() < p6_deadline:
        h_ref, sc = active_set_gradient_descent(best_h, n, target, n_steps=1000, top_m=5,
                                                temp_init=0.005, temp_decay=0.9995,
                                                deadline=p6_deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Coordinate refinement
    if time.time() < deadline - 20:
        h_ref, sc = coordinate_refine(best_h, n, target, max_rounds=3, deadline=deadline - 20)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final SLSQP polish with very high alpha
    if time.time() < deadline - 5:
        h_ref, sc = slsqp_optimize(best_h, n, target, [20000, 50000], 5000, 1e-17, deadline - 2)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
