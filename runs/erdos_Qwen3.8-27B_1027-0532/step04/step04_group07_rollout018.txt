    h_bin = h_bin.astype(np.int8).copy()
    target_sum = n // 2
    
    corr = overlap_corrs(h_bin.astype(float))
    current_max = np.max(corr)
    
    lr = compute_left_right_fast(h_bin, n)
    
    t0 = time.time()
    improved = True
    passes = 0
    while improved and passes < max_passes:
        if time.time() - t0 > time_limit:
            break
        improved = False
        passes += 1
        
        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        
        # For each zero position a, compute base = corr - lr[a]
        # For each one position b, new_max = max(base + lr[b])
        # Find the pair (a, b) that gives the smallest new_max
        
        best_new_max = current_max  # We want to find a swap that reduces this
        best_a = -1
        best_b = -1
        
        # Process in batches
        for a in zeros:
            base = corr - lr[a]  # shape: (2n-1,)
            # For each b in ones: new_max_b = max(base + lr[b])
            # Vectorized: new_maxs = np.max(base[None, :] + lr[ones, :], axis=1)
            new_maxs = np.max(base[None, :] + lr[ones, :], axis=1)
            min_new_max = np.min(new_maxs)
            if min_new_max < best_new_max:
                best_new_max = min_new_max
                best_a = a
                best_b = ones[np.argmin(new_maxs)]
        
        if best_new_max < current_max - 1e-10:
            # Apply the swap
            h_bin[best_a] = 1
            h_bin[best_b] = 0
            # Update corr
            corr = corr + lr[best_b] - lr[best_a]
            current_max = best_new_max
            # Update lr
            # When h[best_a] goes 0->1:
            #   lr[i, L] changes by: (h[i+L] + h[i-L]) changes
            #   lr[best_a, L] was computed with h[best_a]=0, now h[best_a]=1
            #   For position i: lr[i,L] = h[i+L] + h[i-L]
            #   If i+L == best_a: lr[i,L] increases by 1
            #   If i-L == best_a: lr[i,L] increases by 1
            # When h[best_b] goes 1->0:
            #   Similarly, lr[i,L] decreases by 1 if i+L == best_b or i-L == best_b
            
            # For the a position (0->1):
            # lr[best_a, L] changes: it was h[best_a+L]+h[best_a-L] (with h[best_a]=0)
            # Now it should be h[best_a+L]+h[best_a-L]+2 (the +2 is because h[best_a] appears in both terms when L=0)
            # Wait, lr[i,L] = h[i+L] + h[i-L]. For i=best_a:
            #   lr[best_a, L] = h[best_a+L] + h[best_a-L]
            #   This doesn't involve h[best_a] itself unless L=0 (then it's 2*h[best_a])
            #   So when h[best_a] changes from 0 to 1: lr[best_a, 0] increases by 2.
            #   For other L: lr[best_a, L] doesn't change (h[best_a+L] and h[best_a-L] are other positions).
