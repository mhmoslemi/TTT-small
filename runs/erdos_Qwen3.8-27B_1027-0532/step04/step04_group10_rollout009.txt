import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    for _ in range(50):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def downsample(h, n_fine, n_coarse):
    """Downsample by block averaging."""
    k = n_fine // n_coarse
    padded = n_coarse * k
    h_trunc = h[:padded].reshape(n_coarse, k)
    return np.mean(h_trunc, axis=1)

def upsample(h, n_coarse, n_fine):
    """Upsample by replication."""
    k = n_fine // n_coarse
    return np.repeat(h, k)

def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        ct = corr[idx]
    else:
        idx = np.arange(n_corr)
        ct = corr

    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n-lag]
                grad[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n-m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n-m]
    return obj, grad

def optimize_lse(h_start, target, alpha, topk, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    penalty = 5000.0
    
    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f += penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g
    
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

def optimize_power(h_start, target, p, topk, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    penalty = 5000.0
    
    def obj_func(x):

def slsqp_optimize(h_start, target, alpha, topk, maxiter, ftol):
    n = len(h_start)
    def obj_func(x):
        f, _ = lse_obj_grad(x, alpha, topk)
        return f
    def obj_jac(x):
        _, g = lse_obj_grad(x, alpha, topk)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_start, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def downsample(h, n_fine, n_coarse):
    k = n_fine // n_coarse
    padded = n_coarse * k
    h_trunc = h[:padded].reshape(n_coarse, k)
    return np.mean(h_trunc, axis=1)


def upsample(h, n_coarse, n_fine):
    k = n_fine // n_coarse
    return np.repeat(h, k)


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        ct = corr[idx]
    else:
        idx = np.arange(n_corr)
        ct = corr
    c_max = np.max(ct)
    shifted = np.clip(alpha * (ct - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:

        lag = idx[i] - (n - 1)

        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

        return f, g

def optimize_power(h_start, target, p, topk, maxiter, ftol):

    bounds = [(0.0, 1.0)] * n
    penalty = 5000.0

        _, g = lse_obj_grad(x, alpha, topk)
        return g

        return np.sum(x) - target

        return np.ones(n)

            best_h_full, n_full / 2.0, 1000, 10, 300, 0.005)

    if time_left() > 20:
        h_new = optimize_lse(best_h_full, n_full / 2.0, 3000, 10, 200, 1e-17)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h_full = h_new

    best_h_full = project_feasible(best_h_full, n_full / 2.0)
    final_score = compute_c5(best_h_full)
    return best_h_full, final_score, n_full
