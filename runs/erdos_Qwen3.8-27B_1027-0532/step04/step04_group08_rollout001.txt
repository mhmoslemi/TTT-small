    n = len(h)

    a_lo = max(0.0, c - 1.0)

        h_new[i] = a
        h_new[i+1] = c - a
        return compute_c5(h_new)

    a, b = a_lo, a_hi
    c_val = b - gr * (b - a)
    d_val = a + gr * (b - a)
    fc = obj(c_val)

        if fc < fd:
            b = d_val
            d_val = c_val
            fd = fc
            c_val = b - gr * (b - a)

        else:
            a = c_val
            c_val = d_val
            fc = fd
            d_val = a + gr * (b - a)

        # corr_full[s] = sum_i a[i] * v[i - (s - (n-1))]  ...

    # The full correlation is: C[s] = sum_n a[n] * conj(v[n - s + (nv-1)])
    # For real signals without conjugation: C[s] = sum_n a[n] * v[n - s + (nv-1)]
    # where s ranges from 0 to na+nv-2

    # C[s] = sum_n h[n] * (1-h[n - s + n - 1])  for valid indices
    # Let k = s - (n-1) (the lag), then s = k + n - 1
    # C[k+n-1] = sum_n h[n] * (1-h[n-k]) for valid n (where n-k is in [0, n-1])

    # C[k+n-1] = sum_n h[n] * (1-h[n-k]) for n-k in [0, n-1], i.e., n in [k, n-1+k] ∩ [0, n-1]

    # dC/dh[m] = sum over terms where h[m] appears
    # h[m] appears as h[n] when n=m: contributes (1-h[m-k]) if m-k in [0,n-1]
    # h[m] appears as h[n-k] when n-k=m, i.e., n=m+k: contributes -h[m+k] if m+k in [0,n-1]

            # d/dh[m] for m = n_idx-lag: coefficient is -h[n_idx]

            grad[n_idx - lag] -= h[n_idx]

            grad[n_idx + (-lag)] -= h[n_idx]

from scipy.optimize import minimize

# Global variables for objective functions
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _soft_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def pairwise_cd(h, n, target, max_passes, deadline):
    """Pairwise coordinate descent with golden section search."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n - 1):
            if time.time() > deadline:
                break
            c = h_cur[i] + h_cur[i + 1]
            a_lo = max(0.0, c - 1.0)
            a_hi = min(1.0, c)
            if a_hi - a_lo < 1e-12:
                continue
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = a_lo
            b_g = a_hi
            m1 = b_g - gr * (b_g - a_g)
            m2 = a_g + gr * (b_g - a_g)
            
            h_cur[i] = m1
            h_cur[i+1] = c - m1
            f1 = compute_c5(h_cur)
            
            h_cur[i] = m2
            h_cur[i+1] = c - m2
            f2 = compute_c5(h_cur)

                    break
                if f1 < f2:
                    b_g = m2
                    m2 = m1
                    f2 = f1

                    h_cur[i+1] = c - m1
                    f1 = compute_c5(h_cur)
                else:

                    m1 = m2
                    f1 = f2
                    m2 = a_g + gr * (b_g - a_g)
                    h_cur[i] = m2
                    h_cur[i+1] = c - m2
                    f2 = compute_c5(h_cur)
            
            # Set to optimal
            a_opt = (a_g + b_g) / 2.0
            h_cur[i] = a_opt
            h_cur[i+1] = c - a_opt
            sc = compute_c5(h_cur)
            if sc < best_score - 1e-14:
                best_score = sc
                improved_any = True

    return h_cur, compute_c5(h_cur)

def active_set_descent(h, n, target, deadline, eps=1e-6, max_iter=500):
    """Gradient descent on the active set of the maximax problem."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - eps)[0]
        
        # Compute gradient of average active correlation
        grad = np.zeros(n)
        for s in active:
            lag = int(s) - (n - 1)
            if lag >= 0:
                for idx in range(lag, n):
                    grad[idx] += (1.0 - h_cur[idx - lag])
                    grad[idx - lag] -= h_cur[idx]

                    grad[idx] += (1.0 - h_cur[idx + abs_lag])
                    grad[idx + abs_lag] -= h_cur[idx]
