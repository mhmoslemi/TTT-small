import numpy as np

_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    """Project to binary with exact sum using argsort."""
    n = len(h)
    target_sum = int(target_sum)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    idx = np.argsort(-h)
    h_r = np.zeros(n)
    h_r[idx[:target_sum]] = 1.0
    return h_r

def project_simplex_box(h, target_sum):
    """Project onto {x : 0 <= x_i <= 1, sum(x) = target_sum} via bisection."""
    lo = -max(h) - 1.0
    hi = -min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_optimize(h_start, target, max_iters, time_limit, start_time, rng, top_k=5, alpha_init=2.0):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    alpha = alpha_init
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Find top K correlation values
        K = min(top_k, 2 * n - 1)
        top_idx = np.argsort(corr)[-K:][::-1]
        top_vals = corr[top_idx]
        
        # Softmax weights over top K
        scale = np.exp(alpha * (top_vals - top_vals[0]))
        weights = scale / scale.sum()
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            s = int(idx) - (n - 1)
            if s >= 0:
                if s < n:
                    g[:n - s] += w * (1.0 - h[s:])
                    g[s:] -= w * h[:n - s]
            else:
                sa = -s
                if sa < n:
                    g[sa:] += w * (1.0 - h[:n - sa])
                    g[:n - sa] -= w * h[sa:]
        
        # Step size: polynomial decay
        lr = alpha_init / (1.0 + 0.01 * it)
        
        h_new = h - lr * g
        h_new = project_simplex_box(h_new, float(target))
        
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Occasionally snap to binary and check
        if it % 50 == 0:
            h_bin = project_binary(h_new, target)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
        
        h = h_new
        corr = corr_new
    
    return best_h, best_c5

Now let me think about the SLSQP phase. The key improvement here is to use a **threshold-based smooth objective**:

f(h, alpha, threshold) = (1/alpha) * log(sum_k exp(alpha * max(0, corr_k - threshold)))

This focuses the optimization on the correlations that are above the threshold, which are the ones actually causing the max to be large.

Actually, the standard log-sum-exp already does this to some extent (exponentially weighting the larger values). Let me just use a large alpha.

But there's another issue: SLSQP with n=600 variables and a log-sum-exp objective with alpha=5000 might have numerical issues. Let me use a safer formulation.

For the smooth objective, I'll use:
f(h) = c_max + (1/alpha) * log(sum_k exp(alpha * (corr_k - c_max)))

where c_max = max(corr_k). This is the standard log-sum-exp formulation that avoids overflow.

Actually, let me use a different approach for the SLSQP: instead of the full log-sum-exp, use a **piecewise smooth** objective:
f(h) = sum of top-K correlations / K

This is simpler and might work just as well. The subgradient is the average of the K correlation vectors that are in the top-K.

But for SLSQP to work, we need a differentiable objective. The "sum of top-K" isn't differentiable at the points where the ranking changes.

Let me stick with the log-sum-exp approach but be careful about numerical stability.

```python
def _smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Numerically stable log-sum-exp
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    return lse

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def update_corr_swap(corr, h, i, j, n):
    """Update corr in-place after h[i]: 1→0, h[j]: 0→1."""
    m = np.arange(2 * n - 1)
    
    # h[i]: 1→0, i as first index
    lo1, hi1 = (n-1) - i, 2*(n-1) - i
    mask1 = (m >= lo1) & (m <= hi1)
    if mask1.any():
        corr[mask1] += h[i + m[mask1] - (n-1)] - 1.0
    
    # h[i]: 1→0, i as second index
    lo2, hi2 = i, i + n - 1
    mask2 = (m >= lo2) & (m <= hi2)
    if mask2.any():
        corr[mask2] += h[i - m[mask2] + (n-1)]
    
    # h[j]: 0→1, j as first index
    lo3, hi3 = (n-1) - j, 2*(n-1) - j
    mask3 = (m >= lo3) & (m <= hi3)

    mask4 = (m >= lo4) & (m <= hi4)
    if mask4.any():
        corr[mask4] -= h[j - m[mask4] + (n-1)]
    
    h[i] = 0.0
    h[j] = 1.0

One more thought: for the SLSQP, I should use the **analytical gradient** of the smooth objective to speed things up.

The smooth objective is:
f(h) = (1/alpha) * log(sum_k exp(alpha * c_k))

where c_k = corr[m_k] for each shift k.

The gradient of f with respect to h is:
∂f/∂h[i] = (1/alpha) * (1/sum_k exp(alpha*c_k)) * sum_k alpha * exp(alpha*c_k) * (∂c_k/∂h[i])
= (sum_k w_k * ∂c_k/∂h[i]) where w_k = exp(alpha*c_k) / sum_j exp(alpha*c_j)

And ∂c_k/∂h[i] is the subgradient we computed earlier for a single shift k.

So the gradient is a weighted average of the shift-specific subgradients, with softmax weights.

For SLSQP to be fast, providing the analytical gradient is crucial. Let me implement this:

```python
def _smooth_obj_and_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    
    # Numerically stable log-sum-exp
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -1000.0, 0.0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    f_val = c_max + log_sum / alpha
    
    # Weights
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: weighted average of shift subgradients
    g = np.zeros(n)
    for k_idx in range(2 * n -
