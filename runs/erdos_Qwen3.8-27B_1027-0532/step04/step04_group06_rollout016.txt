def find_worst_lags(corr, n, top_k=5):
    center = n - 1
    c = corr.copy()
    c[center] = 0.0
    idx = np.argsort(c)[::-1][:top_k]

class SmoothedObj:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha
        self.dx = 2.0 / n
        self.fft_size = 1
        while self.fft_size < 3 * n:
            self.fft_size *= 2

    def __call__(self, h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(self.alpha * (corr - c_max), -500.0, 0.0)
        w_sum = np.sum(np.exp(shifted))
        return c_max + np.log(w_sum) / self.alpha

    def gradient(self, h):
        n = self.n
        alpha = self.alpha
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        probs = weights / w_sum

        P = np.zeros(self.fft_size)
        P[:2 * n - 1] = probs
        H = np.zeros(self.fft_size)
        H[:n] = h

        rH = np.fft.rfft(H)
        conv_A = np.fft.irfft(np.fft.rfft(P) * rH, self.fft_size)
        A = conv_A[:n]

        p_flip = probs[::-1]
        P2 = np.zeros(self.fft_size)
        P2[:2 * n - 1] = p_flip
        conv_B = np.fft.irfft(np.fft.rfft(P2) * rH, self.fft_size)
        B = conv_B[:n]

        prefix = np.zeros(2 * n - 1)
        cs = np.cumsum(probs)
        prefix[0] = probs[0]
        prefix[1:] = cs[:-1]
        total_w = cs[-1]

        for j in range(n):
            d_min = max(j - (n - 1), 0)
            d_max = min(j, 2 * n - 2)
            if d_min == 0:
                prefix_sum = prefix[d_max]
            else:
                prefix_sum = prefix[d_max] - prefix[d_min - 1]
            A[j] = total_w - prefix_sum + A[j]

        grad = (1.0 - A - B) * self.dx
        return grad

class ConsWrap:
    def __init__(self, target):
        self.target = target

    def f(self, h):
        return np.sum(h) - self.target

    def j(self, h):
        return np.ones(len(h))

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    dx = 2.0 / n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        obj_fn = SmoothedObj(n, alpha)
        cons_w = ConsWrap(target)

        try:
            cons_dict = {
                'type': 'eq',
                'fun': lambda x, w=cons_w: w.f(x),
                'jac': lambda x, w=cons_w: w.j(x)
            }
        except Exception:
            pass

        attempt = 0
        while attempt < 3 and time.time() - start_time < time_limit:
            attempt += 1
            try:
                res = minimize(
                    obj_fn, h_cur,
                    jac=lambda x, o=obj_fn: o.gradient(x),

def structured_start(n, rng, mode):
    h = np.zeros(n)
    half = n // 2
    if mode == 0:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    elif mode == 1:
        for i in range(half):
            h[(i * 2) % n] = 1.0
            if (i * 2 + 1) % n != (i * 2) % n:
                h[(i * 2 + 1) % n] = 1.0
        if h.sum() > half:
            h[h > 0][:int(h.sum() - half)] = 0
        elif h.sum() < half:
            zeros = np.where(h == 0)[0]
            h[zeros[:half - h.sum()]] = 1.0
    elif mode == 2:
        golden = 0.6180339887
        positions = np.zeros(half, dtype=int)
        for i in range(half):
            positions[i] = int((i * golden) % 1.0 * n)
        h[positions % n] = 1.0
        if h.sum() < half:
            zeros = np.where(h == 0)[0]
            h[zeros[:half - h.sum()]] = 1.0
    elif mode == 3:
        period = max(3, n // 100)
        for i in range(n):
            if (i % period) < period // 2:
                h[i] = 1.0
        if h.sum() > half:
            ones = np.where(h > 0)[0]
            rng_choice = rng.choice(len(ones), int(h.sum() - half), replace=False)
            h[ones[rng_choice]] = 0
        elif h.sum() < half:
            zeros = np.where(h == 0)[0]
            h[zeros[:half - h.sum()]] = 1.0
    else:
        h[:] = 0.5
    return project_feasible(h, half)

        rH = np.fft.rfft(H)

        conv_A = np.fft.irfft(np.fft.rfft(P) * rH, self.fft_size)
        A = conv_A[:n].copy()

        p_flip = probs[::-1]
        P2 = np.zeros(self.fft_size)
        P2[:2 * n - 1] = p_flip
        conv_B = np.fft.irfft(np.fft.rfft(P2) * rH, self.fft_size)
        B = conv_B[:n]

        cs = np.cumsum(probs)
        total_w = cs[-1]

        for j in range(n):
            d_min = max(j - (n - 1), 0)
            d_max = min(j, 2 * n - 2)
            if d_min == 0:
                prefix_sum = cs[d_max]
            else:
                prefix_sum = cs[d_max] - cs[d_min - 1]
            A[j] = total_w - prefix_sum + A[j]

        grad = (1.0 - A - B) * self.dx
        return grad

class ConsFun:
    def __init__(self, target):
        self.target = target
    def __call__(self, h):
        return np.sum(h) - self.target

class ConsJac:
    def __init__(self, n):
        self.n = n
    def __call__(self, h):
        return np.ones(self.n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    dx = 2.0 / n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons_f = ConsFun(target)
    cons_j = ConsJac(n)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        obj_fn = SmoothedObj(n, alpha)
        attempt = 0
        while attempt < 2 and time.time() - start_time < time_limit:
            attempt += 1
            try:
                res = minimize(
                    obj_fn, h_cur,
                    jac=obj_fn.gradient,
                    method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': cons_f, 'jac': cons_j}],
                    options={'maxiter': maxiter, 'ftol': ftol}
                )
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new.copy()
                break
            except Exception:
                break

    return h_cur, curr_score

def targeted_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 15)

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    cached_corr = get_corr(h)
    cached_worst = find_worst_lags(cached_corr, n, top_k=5)
    corr_refresh = 0

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
                for _ in range(rng.randint(3, 8)):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    current_c5 = compute_c5(h)
                cached_corr = get_corr(h)
                cached_worst = find_worst_lags(cached_corr, n, top_k=5)
            else:
                break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.55 and len(ones) > 0 and len(zeros) > 0:
            corr_refresh += 1
            if corr_refresh >= 5:
                cached_corr = get_corr(h)
                cached_worst = find_worst_lags(cached_corr, n, top_k=5)
                corr_refresh = 0

            k_star = int(cached_worst[rng.randint(len(cached_worst))])
            if k_star == 0:
                k_star = 1

            a_k = ones + k_star
            a_km = ones - k_star
            valid_a = (a_k >= 0) & (a_k < n) & (a_km >= 0) & (a_km < n)
            b_k = zeros + k_star
            b_km = zeros - k_star
            valid_b = (b_k >= 0) & (b_k < n) & (b_km >= 0) & (b_km < n)

            if np.sum(valid_a) > 0 and np.sum(valid_b) > 0:
                a_valid = ones[valid_a]
                a_score = h[np.clip(a_k, 0, n - 1)[valid_a]] + h[np.clip(a_km, 0, n - 1)[valid_a]]
                b_valid = zeros[valid_b]
                b_score = -(h[np.clip(b_k, 0, n - 1)[valid_b]] + h[np.clip(b_km, 0, n - 1)[valid_b]])
                a_score += rng.random(len(a_score)) * 0.15
                b_score += rng.random(len(b_score)) * 0.15
                a_idx = a_valid[np.argmin(a_score)]
                b_idx = b_valid[np.argmin(b_score)]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True
            else:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]
                h_new[a_idx] = 0.0
                h_new[b_idx] = 1.0
                moved = True

        elif r < 0.85 and len(ones) > 0 and len(zeros) > 0:
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
            h_new[a_idx] = 0.0
            h_new[b_idx] = 1.0
            moved = True

        elif r < 0.95 and len(ones) >= 2 and len(zeros) >= 2:
            ia = rng.choice(len(ones), 2, replace=False)
            ib = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[ia[0]]] = 0.0
            h_new[ones[ia[1]]] = 0.0
            h_new[zeros[ib[0]]] = 1.0
            h_new[zeros[ib[1]]] = 1.0
            moved = True

        else:
            if len(ones) >= 3 and len(zeros) >= 3:
                ia = rng.choice(len(ones), 3, replace=False)
                ib = rng.choice(len(zeros), 3, replace=False)
                for idx in ia:
                    h_new[ones[idx]] = 0.0
                for idx in ib:
                    h_new[zeros[idx]] = 1.0
                moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def fast_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 150

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_start(n, rng, mode):
    h = np.zeros(n)
    half = n // 2
    if mode == 0:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    elif mode == 1:
        for i in range(half):
            pos = (i * 2) % n
            h[pos] = 1.0
            pos2 = (i * 2 + 1) % n
            if pos2 != pos:
                h[pos2] = 1.0
        if h.sum() > half:
            ones = np.where(h > 0)[0]
            pick = rng.choice(len(ones), int(h.sum() - half), replace=False)
            h[ones[pick]] = 0
        elif h.sum() < half:
            zeros = np.where(h == 0)[0]
            h[zeros[:half - h.sum()]] = 1.0
    elif mode == 2:
        golden = 0.6180339887
        positions = np.zeros(half, dtype=int)
        for i in range(half):
            positions[i] = int((i * golden) % 1.0 * n)
        h[positions % n] = 1.0
        if h.sum() < half:
            zeros = np.where(h == 0)[0]
            h[zeros[:half - h.sum()]] = 1.0
        elif h.sum() > half:
            ones = np.where(h > 0)[0]
            pick = rng.choice(len(ones), int(h.sum() - half), replace=False)
            h[ones[pick]] = 0
    else:
        h[:] = 0.5
    return project_feasible(h, float(half))
