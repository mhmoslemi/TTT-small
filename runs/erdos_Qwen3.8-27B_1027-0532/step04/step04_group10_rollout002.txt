import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_with_index(h):

    idx = np.argmax(corr)
    return float(corr[idx]) * dx, idx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        per_elem = diff / len(idx)
        h[idx] = np.clip(h[idx] + per_elem, 0.0, 1.0)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:

        per_elem = diff / len(idx)
        h[idx] = np.clip(h[idx] + per_elem, 0.0, 1.0)
    return h

def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

- Terms where k=i: h[i]*(1-h[i+d]) changes from a*(1-h[i+d]) to b*(1-h[i+d]), change = (b-a)*(1-h[i+d])
- Terms where k+d=i, i.e., k=i-d: h[k]*(1-h[k+d]) = h[i-d]*(1-h[i]) changes from h[i-d]*(1-a) to h[i-d]*(1-b), change = h[i-d]*((1-b)-(1-a)) = h[i-d]*(a-b)

So the change in corr[d] from flipping h[i] from a to b is:
(b-a)*(1-h[i+d]) + h[i-d]*(a-b) = (b-a)*((1-h[i+d]) - h[i-d])

For a flip from 1 to 0 (a=1, b=0): change = -1*(1-h[i+d]) + h[i-d]*1 = h[i-d] - (1-h[i+d]) = h[i-d] + h[i+d] - 1

For a flip from 0 to 1 (a=0, b=1): change = 1*(1-h[i+d]) - h[i-d] = 1 - h[i+d] - h[i-d]

So if we flip position i from 1 to 0 and position j from 0 to 1:
For each lag d, the change in corr[d] is:
(h[i-d] + h[i+d] - 1) [from flipping i] + (1 - h[j-d] - h[j+d]) [from flipping j]
= h[i-d] + h[i+d] - h[j-d] - h[j+d]

(where we define h[k]=0 for k out of range)

This means we can precompute for each position i the "effect vector" e_i where e_i[d] = the change in corr[d] from flipping h[i]. Then a swap (i: 1→0, j: 0→1) changes corr by -e_i + e_j (approximately, since the two flips interact).

Actually the interaction is that when we flip both i and j, the effect on corr[d] isn't simply additive because flipping i changes h[i] which affects the computation for j and vice versa. But if i and j are far apart (|i-j| is large relative to the "support" of the correlation), the interaction is zero.

- Adding h[j]=1: similar but positive

Let me define: For a given lag d (with offset from center), the "sensitivity" of corr at that lag to position i is:

So to reduce corr[d*], we want to:
- Decrease h[i] where ∂corr[d*]/∂h[i] > 0 (i.e., 1-h[i+d*] > h[i-d*])
- Increase h[j] where ∂corr[d*]/∂h[j] < 0 (i.e., 1-h[j+d*] < h[j-d*])

    """Compute ∂corr[lag_offset]/∂h for all positions."""
    n = len(h)
    sens = np.zeros(n)
    if lag_offset >= 0:
        d = lag_offset
        if d >= n:
            return sens
        # First term: (1-h[p-d]) for p in [d, n-1]
