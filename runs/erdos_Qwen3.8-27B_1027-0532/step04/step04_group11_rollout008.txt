Actually, for a smooth objective with Lipschitz continuous gradient, a fixed step size of 1/L (where L is the Lipschitz constant) works. The Lipschitz constant of the gradient is related to alpha and the correlation structure.

For the log-sum-exp objective g(h) = (1/α) log(Σ exp(α c_k(h))):
∇g(h) = Σ p_k ∇c_k(h)
Each ∇c_k(h) is bounded (since h ∈ [0,1] and the correlation is a bilinear form).

The Lipschitz constant of ∇g is bounded by α * L_c where L_c is the largest eigenvalue of the Hessian of the correlation (which is at most 2n for the worst case, but typically much less).

In practice, a step size of 1/α should work, and we can use backtracking if needed.

```python
def gradient_descent_smoothed(h, target, alphas, iterations_per_alpha, t0, budget):
    n = len(h)
    for alpha in alphas:
        for it in range(iterations_per_alpha):
            if time.time() - t0 > budget - 30:
                return h
            # Compute correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            # Compute softmax weights
            shifted = alpha * (corr - corr.max())
            p = np.exp(shifted)
            p_sum = p.sum()
            if p_sum == 0:
                p = np.ones_like(corr) / len(corr)
            else:
                p /= p_sum
            # Compute gradient
            grad = compute_gradient_direct(h, p, n)
            # Projected gradient step
            step = 1.0 / (alpha + 1.0)
            h_new = project_onto_simplex_box(h - step * grad, target)
            # Check improvement
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                h = h_new
            else:
                step *= 0.5
                h_new = project_onto_simplex_box(h - step * grad, target)
                h = h_new
    return h
```

```python
def compute_gradient_direct(h, p, n):
    """Compute gradient of smoothed objective.
    p is the softmax weights over correlation values (length 2n-1).
    p[k] corresponds to lag k-(n-1).
    """
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for lag in range(-(n-1), n):
            pk = p[n-1+lag]
            if pk < 1e-15:
                continue
            # First term: (1-h[j+lag]) if 0 <= j+lag < n
            if 0 <= j + lag < n:
                val += pk * (1.0 - h[j+lag])
            # Second term: -h[j-lag] if 0 <= j-lag < n
            if 0 <= j - lag < n:
                val += pk * (-h[j-lag])
        grad[j] = val
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, new_n / 2.0)
    return h_new

def smooth_obj_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = alpha * (corr - c_max)
    p_raw = np.exp(shifted)
    p_sum = float(p_raw.sum())
    val = c_max + np.log(p_sum) / alpha
    p = p_raw / p_sum
    idx = np.abs(np.arange(n)[:, None] - np.arange(n)[None, :])
    T = p[n - 1 + idx]
    r = T @ h
    grad = 1.0 - 2.0 * r
    return val, grad

def grad_descent_smoothed(h, target, alphas, iters, t0, deadline):
    n = len(h)
    h_cur = h.copy()
    for alpha in alphas:
        step_base = 0.5 / max(alpha, 1.0)

                return h_cur

            h_trial = project_feasible(h_cur - step * grad, target)
            val_trial, _ = smooth_obj_grad(h_trial, alpha, n)
