    for i in range(n):
        period = (2 * period + 1) % (2 * n + 1)
        h[i] = 1.0 if (period & 1) == 1 else 0.0
    current = int(h.sum())
    target = n // 2
    if current < target:
        zeros = np.where(h == 0)[0]
        need = target - current
        order = rng.permutation(len(zeros))[:need]
        if len(order) > 0:
            h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        need = current - target
        order = rng.permutation(len(ones))[:need]
        if len(order) > 0:
            h[ones[order]] = 0.0
    return h

def targeted_local_search(h_init, target, rng, time_limit, t_start,
                          t_init=0.05, t_final=0.001, batch_size=20):
    """Local search with guided move generation focused on worst lags."""
    n = len(h_init)
    dx = 2.0 / n
    h = project_binary(h_init, target).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(np.max(corr))
    
    best_h = h.copy()
    best_max = cur_max
    best_c5 = cur_max * dx
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_imp = 0
    max_no_imp = 3000
    
    it = 0
    while True:
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        it += 1
        
        if no_imp > max_no_imp:
            # Restart from best with perturbation
            h = best_h.copy()
            n_pert = rng.randint(2, max(3, n // 30))
            for _ in range(n_pert):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = int(ones[rng.randint(len(ones))])
                    pj = int(zeros[rng.randint(len(zeros))])
                    h[pi] = 0.0
                    h[pj] = 1.0
            h = project_binary(h, target).astype(np.float64)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            continue
        
        # Temperature
        progress = min(it / 100000.0, 1.0)
        temp = t_init * (t_final / t_init) ** progress
        
        # Generate candidate swaps
        best_delta = 0
        best_a = -1
        best_b = -1
        accepted = False
        
        for _ in range(batch_size):
            # Choose swap strategy
            strategy = rng.randint(4)
            
            if strategy == 0:
                # Random swap
                if len(ones) == 0 or len(zeros) == 0:
                    continue
                a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
            elif strategy == 1:
                # Target worst lag: find a 1 that contributes to worst lag
                worst_idx = int(np.argmax(corr))
                worst_k = worst_idx - (n - 1)
                # Find positions i where h[i]=1 and h[i+worst_k]=0 (if valid)
                if worst_k >= 0:
                    valid_i = range(n - worst_k)
                else:
                    valid_i = range(-worst_k, n)
                # Find contributing ones
                contributors = []
                for ci in range(len(ones)):
                    oi = int(ones[ci])
                    pos = oi + worst_k
                    if 0 <= pos < n and h[pos] == 0:
                        contributors.append(ci)
                if len(contributors) > 0:
                    ci = contributors[rng.randint(len(contributors))]
                    a = int(ones[ci])
                else:
                    a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
            elif strategy == 2:
                # Target second-worst lag
                corr_copy = corr.copy()
                top_idx = int(np.argmax(corr_copy))
                corr_copy[top_idx] = -np.inf
                second_idx = int(np.argmax(corr_copy))
                worst_k = second_idx - (n - 1)
                contributors = []
                for ci in range(len(ones)):
                    oi = int(ones[ci])
                    pos = oi + worst_k
                    if 0 <= pos < n and h[pos] == 0:
                        contributors.append(ci)
                if len(contributors) > 0:
                    ci = contributors[rng.randint(len(contributors))]
                    a = int(ones[ci])
                else:
                    a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
            else:
                # Near-neighbor swap: swap a 1 with a nearby 0
                if len(ones) == 0:
                    continue
                a = int(ones[rng.randint(len(ones))])
                dist = rng.randint(1, max(2, n // 20) + 1)
                dir = rng.choice([-1, 1])
                b = (a + dir * dist) % n
                if h[b] != 0:
                    b = int(zeros[rng.randint(len(zeros))])
            
            if a == b:
                continue
            
            # Compute new correlation after swap
            h[a] = 0.0
            h[b] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr))
            
            # Revert
            h[a] = 1.0
            h[b] = 0.0
            
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_a = a
                best_b = b
                
            if best_delta < -1e-10:
                # Found improvement, break early
                break
        
        if best_a == -1:
            no_imp += 1
            continue
        
        # Accept or reject based on SA
        if best_delta < 0 or (temp > 1e-12 and rng.random() < np.exp(-best_delta / (temp * dx * cur_max + 1e-30))):
            h[best_a] = 0.0
            h[best_b] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            c5_val = cur_max * dx
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
    
    return best_h, best_c5

def smooth_max_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    soft_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(soft_max * dx)

def continuous_refine(h_init, time_limit, t_start):
    n = len(h_init)
    target = n / 2.0
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in [10, 30, 100, 300, 1000]:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return smooth_max_objective(x, alpha, n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    h_init_arr = np.array(initial_h_values, dtype=float)
    n_init = len(h_init_arr)
    target_init = n_init / 2.0
    
    best_h = h_init_arr.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick multi-start (150s)
    for it in range(5):
        elapsed = time.time() - start
        if elapsed > 150:
            break
        remaining_phase = 150 - elapsed
        
        if it == 0:
            h_try = h_init_arr.copy()
        elif it == 1:
            h_try = make_qr_sequence(n_init, rng)
        elif it == 2:
            h_try = make_msequence(n_init, rng)
        else:
            h_try = best_h + rng.randn(n_init) * 0.3
            h_try = project_binary(h_try, target_init)
        
        run_time = min(30.0, remaining_phase)
        h_ls, c5_ls = targeted_local_search(h_try, target_init, rng,
                                            run_time, start,
                                            t_init=0.03, t_final=0.001,
                                            batch_size=15)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 2: Intensive search on best (500s)
    elapsed = time.time() - start

        run_time = min(500.0, budget_s - 300 - elapsed)

                                            run_time, start,

            h_interp = np.interp(x_new, x_old, best_h)
            h_bin = project_binary(h_interp, target_nt)
            
            run_time = min(40.0, budget_s - 200 - elapsed)

                                                run_time, start,
