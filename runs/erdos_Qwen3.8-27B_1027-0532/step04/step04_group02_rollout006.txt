def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-12:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-12:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int).astype(float)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights (numerically stable)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    smooth_val = c_max + np.log(np.sum(exp_c)) / alpha
    
    # Gradient computation
    # dc[k_index]/dh[i] where k = k_index - (n-1)
    # = (1-h[i-k]) * [0 <= i-k < n] - h[i+k] * [0 <= i+k < n]
    # multiplied by dx
    grad = np.zeros(n)
    for i in range(n):
        # First term: k such that 0 <= i-k < n, i.e., i-n < k <= i
        # k ranges from max(-(n-1), i-n+1) to min(n-1, i)
        k_lo = max(-(n-1), i-(n-1))  # = max(-(n-1), i-n+1)
        k_hi = min(n-1, i)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi+1)
            w_idx = ks + (n-1)
            contrib1 = np.dot(w[w_idx], 1.0 - h[i - ks])
        else:
            contrib1 = 0.0
        # Second term: k such that 0 <= i+k < n, i.e., -i <= k < n-i
        k_lo2 = max(-(n-1), -i)
        k_hi2 = min(n-1, n-1-i)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2+1)
            w_idx2 = ks2 + (n-1)
            contrib2 = np.dot(w[w_idx2], h[i + ks2])
        else:
            contrib2 = 0.0
        grad[i] = dx * (contrib1 - contrib2) / alpha
    return smooth_val, grad

def pgd_optimize(h_start, target_sum, alpha_schedule, lr_schedule, steps, time_limit=None, start_time=None):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    total_steps = steps
    
    for s in range(total_steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        progress = s / max(total_steps - 1, 1)
        alpha = alpha_schedule[0] + (alpha_schedule[1] - alpha_schedule[0]) * progress
        lr = lr_schedule[0] + (lr_schedule[1] - lr_schedule[0]) * progress
        
        val, grad = smooth_max_and_grad(h, alpha)
        # Subtract mean gradient to handle sum constraint
        grad_centered = grad - np.mean(grad)
        h_new = h - lr * grad_centered
        h_new = project_feasible(h_new, target_sum)
        
        actual_val = compute_c5(h_new)
        if actual_val < best_val:
            best_val = actual_val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val

Actually, for binary h, a "block swap" means: select two intervals of the same length, one containing all ones and the other all zeros, and swap them. But finding such intervals is expensive.

A simpler "block move": select a contiguous block of h (of some length L), reverse it, or rotate it. This preserves the number of ones.

import numpy as np
from scipy.optimize import minimize
from numpy.lib.stride_tricks import sliding_window_view
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            if remaining <= 1e-12:
                break
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
    else:
        room = h.copy()
        order = np.argsort(-room)
        remaining = -diff
        for i in order:
            if remaining <= 1e-12:
                break
            sub = min(room[i], remaining)
            h[i] -= sub
            remaining -= sub
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    smooth_val = c_max + np.log(np.sum(exp_c)) / alpha
    
    # Efficient gradient using sliding windows
    # S[i] = w[i:i+n] for i in range(n), shape (n, n)
    S = sliding_window_view(w, n)  # shape (2n-1 - n + 1, n) = (n, n)
    flipped_g = (1.0 - h)[::-1]
    term1 = S.dot(flipped_g)  # shape (n,)
    term2 = S[::-1].dot(h)  # shape (n,)
    grad = dx / alpha * (term1 - term2)
    
    return smooth_val, grad

def pgd_optimize(h_start, target_sum, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    # Adaptive alpha schedule: start low, increase
    # Use cosine annealing on alpha
    total_time = time_limit
    step_time = 0.002  # estimate per step
    max_steps = int(total_time / step_time * 0.7)
    
    alpha_start = 20.0
    alpha_end = 500.0
    lr_start = 0.5
    lr_end = 0.01
    
    for s in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        progress = s / max(max_steps - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress ** 0.7
        lr = lr_start * (lr_end / lr_start) ** progress
        
        val, grad = smooth_max_and_grad(h, alpha)
        grad_centered = grad - np.mean(grad)
        h_new = h - lr * grad_centered
        h_new = project_feasible(h_new, target_sum)
        
        actual_val = compute_c5(h_new)
        if actual_val < best_val - 1e-12:
            best_val = actual_val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val
