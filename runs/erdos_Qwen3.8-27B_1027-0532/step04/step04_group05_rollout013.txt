    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for l in range(2*n - 1):
        k = l - (n - 1)  # lag
        w = weights[l] * dx
        if w < 1e-18:
            continue

        lo1 = max(0, k)
        hi1 = min(n, n + k)  # exclusive

        lo2 = max(0, -k)
        hi2 = min(n, n - k)  # exclusive
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + k : hi2 + k]
    
    return grad

    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    n_half = n - 1
    for l in range(2 * n - 1):
        s = l - n_half
        w = weights[l]
        if w < 1e-18:

            grad[s:n] += dx * w * (1.0 - h[0:n-s])
            # Part 2: grad[0:n-s] -= dx*w*h[s:n]
            grad[0:n-s] -= dx * w * h[s:n]

            grad[0:n-s_abs] += dx * w * (1.0 - h[s_abs:n])

            grad[s_abs:n] -= dx * w * h[0:n-s_abs]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    value = c_max + np.log(sum_exp) / alpha
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    n_half = n - 1
    for l in range(2 * n - 1):
        s = l - n_half
        w = weights[l]
        if w < 1e-18:
            continue
        if s >= 0:
            if n - s > 0:
                grad[s:n] += dx * w * (1.0 - h[0:n-s])
                grad[0:n-s] -= dx * w * h[s:n]

                grad[0:n-s_abs] += dx * w * (1.0 - h[s_abs:n])
                grad[s_abs:n] -= dx * w * h[0:n-s_abs]
    return value, grad

def slsqp_analytic(h0, n, target, alpha, maxiter, time_limit):
    start_t = time.time()
    def obj(h):
        if time.time() - start_t > time_limit:
            raise RuntimeError("timeout")
        v, _ = smooth_max_obj_and_grad(h, alpha, n)
        return v
    def grad(h):
        if time.time() - start_t > time_limit:
            raise RuntimeError("timeout")
        _, g = smooth_max_obj_and_grad(h, alpha, n)
        return g
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target)
        return h_out
    except Exception:
        return h0

def nesterov_projected_grad(h0, n, target, alpha, max_steps, time_limit, lr_base=0.01):
    start_t = time.time()
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        y = h + v
        _, grad = smooth_max_obj_and_grad(y, alpha, n)
        lr = lr_base / (1.0 + 0.01 * step)
        t = (step + 1) / (step + 1 + 1.0)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        v = (h_new - h) * t
        h = h_new
        sc = compute_c5(h)
        if sc < best_val - 1e-14:
            best_val = sc
            best_h = h.copy()
    return best_h, best_val

def binary_local_search(h0, n, target, max_steps, time_limit, window=3):
    start_t = time.time()
    h_bin = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))

        z_pick = rng.choice(len(zeros_idx), size=n_flip, replace=False)

        z_positions = zeros_idx[z_pick]

        h_try[o_positions] = 0.0

        sc = compute_c5(h_try)

            best_c5 = sc
            best_h = h_try
            ones_idx = np.where(best_h > 0.5)[0]
            zeros_idx = np.where(best_h < 0.5)[0]
    return best_h, best_c5
