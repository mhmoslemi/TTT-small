def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def obj_and_grad(h, alpha, topk, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        total = 1e-300
    obj = (c_max + np.log(total) / alpha) * dx
    weights = exp_vals / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    grad *= dx
    s = np.sum(h) - target
    obj += penalty * s * s
    grad += 2.0 * penalty * s
    return obj, grad


def lse_jac_wrapper(x, alpha, topk, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    f, g = obj_and_grad(h, alpha, topk, target, penalty)
    return f, g


def do_lbfgs(h_start, target, alpha, topk, penalty, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_jac_wrapper, h_start.copy(),
                       args=(alpha, topk, target, penalty),
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def mom_pgd(h, target, topk, lr0, n_steps, momentum, decay):

    dx = 2.0 / n
    best_h = h.copy()

    v = np.zeros(n)

                    count += 1

    # Phase 6: Simulated annealing with L-BFGS refinement
    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        temp = 0.035
        for sa_round in range(20):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 25:
                break
            h_pert = best_h + rng.randn(n) * temp
            h_pert = project_feasible(h_pert, target)
            h_new = do_lbfgs(h_pert, target, 300, 25, 5000, 250, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                temp = max(0.01, temp * 0.85)
            else:
                temp = max(0.01, temp * 0.92)

    # Phase 7: High-alpha L-BFGS polish
    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        for alpha in [3000, 5000, 8000]:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 20:
                break
            h_start = best_h + rng.randn(n) * 0.001
            h_start = project_feasible(h_start, target)
            h_new = do_lbfgs(h_start, target, alpha, 12, 10000, 250, 1e-17)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Coordinate descent
    elapsed = time.time() - start_time
    if budget_s - elapsed > 25:
        for rnd in range(3):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 10:
                break
            improved_any = False
            for coord in range(n):
                elapsed = time.time() - start_time
                if budget_s - elapsed < 5:
                    break
                orig = best_h[coord]
                for delta in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.003, -0.003]:
                    h_try = best_h.copy()
                    h_try[coord] = np.clip(orig + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_h = h_try
                        improved_any = True
                        break
            if not improved_any:
                break

    # Phase 9: Final L-BFGS
    elapsed = time.time() - start_time
    if budget_s - elapsed > 15:
        h_new = do_lbfgs(best_h, target, 5000, 10, 10000, 200, 1e-17)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
