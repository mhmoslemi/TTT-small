Actually, there's a subtlety: the delta formula uses h[i-m] etc. which are the OLD values. But h[i] is being changed from 1 to 0, and h[j] from 0 to 1. The formula I derived earlier was:
Δcorr[m] = h[i-m] + h[i+m] - h[j-m] - h[j+m]

where all h values are the old (pre-swap) values. This is correct because:
- The term h[i] in corr[m] is at position p=i, contributing h[i]*(1-h[i-m]) = 1*(1-h[i-m]). After swap, it's 0.
  But h[i] also appears in h[p-m] for p = i+m, where the term is h[i+m]*(1-h[i]). Since h[i] was 1, this was 0. After swap, h[i]=0, so it becomes h[i+m]*1 = h[i+m].
  
So the total change from removing h[i]=1 is:
- At p=i: -(1-h[i-m]) = h[i-m] - 1
- At p=i+m: +h[i+m] (if i+m is in range)
Total from i: h[i-m] - 1 + h[i+m]

And the total change from adding h[j]=1 is:
- At p=j: +(1-h[j-m]) = 1 - h[j-m]
- At p=j+m: -h[j+m] (if j+m is in range)
Total from j: 1 - h[j-m] - h[j+m]

Grand total: h[i-m] - 1 + h[i+m] + 1 - h[j-m] - h[j+m] = h[i-m] + h[i+m] - h[j-m] - h[j+m]

Yes, this confirms the formula. And it uses OLD h values.

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.int32)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(np.float64)
    if current < target:
        need = target - current
        zeros_idx = np.where(h_r == 0)[0]
        if len(zeros_idx) < need:
            return h_r.astype(np.float64)
        scores = h[zeros_idx]
        order = np.argsort(-scores)[:need]
        h_r[zeros_idx[order]] = 1
    else:
        need = current - target
        ones_idx = np.where(h_r == 1)[0]
        if len(ones_idx) < need:
            return h_r.astype(np.float64)
        scores = h[ones_idx]
        order = np.argsort(scores)[:need]
        h_r[ones_idx[order]] = 0
    return h_r.astype(np.float64)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def is_prime_check(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True


def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while not is_prime_check(p):
        p += 1
    h = np.zeros(n, dtype=np.float64)
    for i in range(2, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

def smooth_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 500, 1000]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                lambda x: smooth_objective(x, alpha, n),
                h_cur, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    reserve = 40.0

    h_init_arr = np.array(initial_h_values, dtype=np.float64)
    n_init = len(h_init_arr)
    target_init = n_init // 2

    best_h = project_binary(h_init_arr, target_init)
    best_c5 = compute_c5(best_h)

    remaining = budget_s - reserve - (time.time() - start)
    if remaining > 5:
        h_ls, c5_ls = fast_gradient_search(best_h, n_init, min(200.0, remaining), start, rng, top_k=3, max_no_imp=200)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    remaining = budget_s - reserve - (time.time() - start)
    if remaining > 60:
        h_qr = make_qr_sequence(n_init, rng)
        h_ls, c5_ls = fast_gradient_search(h_qr, n_init, min(120.0, remaining), start, rng, top_k=3, max_no_imp=200)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    remaining = budget_s - reserve - (time.time() - start)
    n_tries = [n_init + 2, n_init + 4, n_init + 6, n_init + 10, n_init + 20]
    for nt in n_tries:
        if remaining < 50:
            break
        h_qr = make_qr_sequence(nt, rng)
        h_ls, c5_ls = fast_gradient_search(h_qr, nt, min(80.0, remaining), start, rng, top_k=3, max_no_imp=150)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - reserve - (time.time() - start)

    remaining = budget_s - reserve - (time.time() - start)
    n_current = len(best_h)
    target_current = n_current // 2
    for restart in range(8):
        if remaining < 40:
            break
        noise_scale = 0.3 + 0.1 * restart
        h_pert = best_h + rng.randn(n_current) * noise_scale
        h_pert_bin = project_binary(h_pert, target_current)
        h_ls, c5_ls = fast_gradient_search(h_pert_bin, n_current, min(60.0, remaining), start, rng, top_k=4, max_no_imp=150)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - reserve - (time.time() - start)

    remaining = budget_s - reserve - (time.time() - start)
    if remaining > 50:
        n_final = len(best_h)
        target_final = n_final // 2
        h_cur, score = continuous_refine(best_h, n_final, target_final, min(60.0, remaining), start)
        if score < best_c5:
            best_c5 = score
            best_h = h_cur

    remaining = budget_s - reserve - (time.time() - start)
    if remaining > 10:
        n_final = len(best_h)
        target_final = n_final // 2
        h_bin = project_binary(best_h, target_final)
        h_ls, c5_ls = fast_gradient_search(h_bin, n_final, max(3.0, remaining - 5.0), start, rng, top_k=2, max_no_imp=100)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5:
        h_final = best_h
        final_c5 = best_c5

    return h_final, final_c5, n_final
