def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def binarize(h, target):
    # Round to binary, then repair
    h_bin = (h >= 0.5).astype(float)
    # Adjust to get exactly target ones
    diff = target - h_bin.sum()
    if diff > 0:
        # Need to add 'diff' ones: pick the h values closest to 0.5 from below
        zeros = np.where(h_bin == 0)[0]
        # Sort by h value descending (those closest to 0.5)
        order = np.argsort(-h[zeros])
        for i in range(int(diff)):
            h_bin[zeros[order[i]]] = 1.0
    elif diff < 0:
        # Need to remove '|diff|' ones
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for i in range(int(-diff)):
            h_bin[ones[order[i]]] = 0.0
    return h_bin

def sa_optimize(h0, target, n_steps, t_init, t_final, deadline, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_corr_max(h)
    cur_score = best_score
    
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Geometric cooling
        t = t_init * (t_final / t_init) ** (step / max(n_steps, 1))
        
        # Pick a random swap
        a = ones_pos[rng.randint(len(ones_pos))]
        b = zeros_pos[rng.randint(len(zeros_pos))]
        
        # Apply swap
        h[a] = 0.0
        h[b] = 1.0
        
        new_score = compute_corr_max(h)
        
        if new_score <= cur_score:
            # Accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # Update position lists
            ones_pos = np.append(ones_pos, b)
            ones_pos = np.delete(ones_pos, np.where(ones_pos == a)[0])
            zeros_pos = np.append(zeros_pos, a)
            zeros_pos = np.delete(zeros_pos, np.where(zeros_pos == b)[0])
        else:
            # Maybe accept
            delta = new_score - cur_score
            if delta > 0 and rng.random() < np.exp(-delta / max(t, 1e-10)):
                cur_score = new_score
                ones_pos = np.append(ones_pos, b)
                ones_pos = np.delete(ones_pos, np.where(ones_pos == a)[0])
                zeros_pos = np.append(zeros_pos, a)

            else:
                # Reject
                h[a] = 1.0
                h[b] = 0.0
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return compute_corr_max(h) * dx

def project_binary(h, target):
    """Project to binary with exactly target ones."""
    h = h.copy()
    ones = np.sum(h == 1.0)
    if ones == target:
        return h
    if ones < target:
        # Need to add (target - ones) ones
        zeros = np.where(h == 0.0)[0]
        for i in range(int(target - ones)):
            h[zeros[i]] = 1.0
    else:
        # Need to remove (ones - target) ones
        ones_idx = np.where(h == 1.0)[0]
        for i in range(int(ones - target)):
            h[ones_idx[i]] = 0.0
    return h

def binarize_from_fractional(h, target):
    """Convert fractional h to binary with exactly target ones."""
    h_bin = np.zeros_like(h)
    # Sort by value descending, assign top 'target' to 1
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    return h_bin

def sa_binary(h0, target, n_iter, t_start, t_end, deadline, rng):
    """Simulated annealing on binary vector."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_corr_max(h)
    cur_score = best_score
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    for i in range(n_iter):
        if time.time() > deadline:
            break
        
        # Temperature
        frac = i / max(n_iter - 1, 1)
        t = t_start * (t_end / t_start) ** frac
        
        # Random swap
        ia = rng.randint(n_ones)
        ib = rng.randint(n_zeros)
        a = ones_list[ia]
        b = zeros_list[ib]
        
        # Perform swap
        h[a] = 0.0
        h[b] = 1.0
        new_score = compute_corr_max(h)
        
        if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / t):
            # Accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # Update lists
            ones_list[ia] = b
            zeros_list[ib] = a
        else:
            # Reject
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_score

    zeros_list = list(np.where(h == 0)[0])

def compute_corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def compute_c5_from_corr_max(corr_max, n):
    return corr_max * 2.0 / n

def binarize_topk(h, target):
    n = len(h)
    h_bin = np.zeros(n, dtype=np.float64)
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    return h_bin

def sa_binary(h0, target, n_iter, t_start, t_end, deadline, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_corr_max(h)
    cur_score = best_score
    
    ones_idx = np.where(h == 1)[0].tolist()
    zeros_idx = np.where(h == 0)[0].tolist()
    n_ones = len(ones_idx)
    n_zeros = len(zeros_idx)
    
    for i in range(n_iter):
        if time.time() > deadline:
            break
        
        frac = i / max(n_iter - 1, 1)
        t = t_start * (t_end / t_start) ** frac
        
        ia = rng.randint(n_ones)
        ib = rng.randint(n_zeros)
        a = ones_idx[ia]
        b = zeros_idx[ib]
        
        h[a] = 0.0
        h[b] = 1.0
        new_score = compute_corr_max(h)
        
        if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / t):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones_idx[ia] = b
            zeros_idx[ib] = a
        else:
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_score

Actually, one more important consideration: the `np.correlate(h, 1-h, mode='full')` function. Let me make sure I understand what it computes correctly.

From numpy docs:
np.correlate(a, v, mode='full'):
"""
Cross-correlation of two 1-dimensional sequences.
This is the cross-correlation of a and v, defined as:
c_{av}[k] = \sum_n a[n+k] \overline{v[n]}
"""

So c[k] = Σ_n a[n+k] * conj(v[n]) for valid n.

With a=h and v=(1-h):
c[k] = Σ_n h[n+k] * (1-h[n])

For mode='full', k ranges from -(len(v)-1) to len(a)-1, i.e., from -(n-1) to n-1.

The output is indexed from 0 to 2n-2, where:
- output[n-1+k] corresponds to correlation at lag k

So output[n-1] = c[0] = Σ_n h[n] * (1-h[n]) = 0 for binary h. ✓

output[n-1+1] = c[1] = Σ_n h[n+1] * (1-h[n]) (for valid n, i.e., n=0 to n-2)
This counts: number of n where h[n+1]=1 and h[n]=0.

output[n-1-1] = c[-1] = Σ_n h[n-1] * (1-h[n]) (for valid n, i.e., n=1 to n-1)
This counts: number of n where h[n-1]=1 and h[n]=0.

So the max of the full correlation array gives us the max over all lags of the "1-then-0" count (for positive lags) or "0-then-1" count (for negative lags).

This is exactly what we want to minimize.

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

    n = len(h0)
    h_cur = h0.copy()
    cur_score = compute_corr_max(h_cur)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    for alpha in alpha_list:
        if time.time() > deadline:
            break

        def fun(x):
            return lse_obj_and_grad(x, alpha)[0]

        def jac(x):
            return lse_obj_and_grad(x, alpha)[1]

        try:
            res = minimize(fun, h_cur, jac=jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = np.clip(res.x, 0.0, 1.0)
            diff = target - np.sum(h_new)

                mask = h_new < 1.0 if diff > 0 else h_new > 0.0
                cnt = int(mask.sum())
                if cnt > 0:
                    h_new[mask] += diff / cnt
                    h_new = np.clip(h_new, 0.0, 1.0)
            sc = compute_corr_max(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

        h5, s5 = fractional_refine(h_frac_start, float(target), deadline, [2000, 8000, 30000])
        if s5 < best_score:
            best_score = s5
            best_h = h5

    final_c5 = best_score * 2.0 / n
    return best_h, final_c5, n
