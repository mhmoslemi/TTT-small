        else:
            # i from -m to n-1, i+m from 0 to n-1+m

                h[ones[idx1[0]]] = 0
                h[ones[idx1[1]]] = 0
                h[zeros[idx0[0]]] = 1
                h[zeros[idx0[1]]] = 1
                c5_new = compute_c5(h)
                delta = c5_new - current_c5
                if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        no_improve = 0
                    else:

                pos = i1 + ext
                if 0 <= pos < n and h[pos] == 1:

                    i0 = zeros[rng.randint(len(zeros))]

def continuous_optimize(h_start, target, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h = h_start.copy().astype(float)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # L-BFGS-B with increasing alpha and penalty
    alphas = [30, 60, 120, 250, 500, 1000]
    mu = 50.0
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def obj(x):
            return lse_obj_only(x, alpha, mu, target, n)
        
        def grad(x):
            return lse_grad_only(x, alpha, mu, target, n)
        
        try:
            res = minimize(
                obj, h, jac=grad, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 5000}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
        
        # Increase penalty
        mu = min(mu * 2.0, 5000.0)
    
    return best_h, best_c5

def directed_local_search(h_start, target, time_limit, start_time, rng):
    n = len(h_start)
    h = project_feasible(h_start.copy(), target)
    best_h = h.copy()

    stall = 0
    max_stall = 200
    
    for _ in range(100000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        # Try to find a position where a small change improves C5
        # Focus on positions where h is near 0 or 1
        near_extreme = np.where((h < 0.3) | (h > 0.7))[0]
        if len(near_extreme) == 0:
            near_extreme = np.arange(n)
        
        # Random subset
        n_try = min(50, len(near_extreme))
        idxs = near_extreme[rng.choice(len(near_extreme), n_try, replace=False)]
        
        improved = False
        for idx in idxs:
            # Try moving h[idx] toward 0 or 1
            if h[idx] > 0.5:
                delta = h[idx] - 0.5
                h[idx] = 0.5
            else:
                delta = 0.5 - h[idx]
                h[idx] = 0.5
            
            h_proj = project_feasible(h, target)
            c5_try = compute_c5(h_proj)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_proj.copy()
                h = h_proj
                improved = True
                stall = 0
            else:
                h[idx] += delta  # revert
                stall += 1
        
        if not improved:

            idx = rng.randint(n)

                best_h = h_proj.copy()
                h = h_proj
                stall = 0

                h = best_h.copy()
                stall += 1
