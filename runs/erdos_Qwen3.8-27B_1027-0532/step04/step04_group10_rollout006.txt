from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    n = len(h)
    
    # Get top-k indices
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    
    # Compute softmax weights (shifted for numerical stability)
    c_shift = np.max(corr_top)
    shifted = alpha * (corr_top - c_shift)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    
    obj = c_shift + np.log(total) / alpha
    weights = exp_vals / total
    
    # Compute gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        lag = idx[i] - (n - 1)
        m = abs(lag)
        if m < n:
            # For lag m: sum_{j=0}^{n-1-m} h[j+m] * (1-h[j])
            # grad[p] = (1-h[p]) * [m <= p <= n-1] - h[p] * [0 <= p <= n-1-m]
            if m == 0:
                grad += w * (one_minus_h - h)
            else:
                grad[m:] += w * one_minus_h[:n-m]
                grad[:n-m] -= w * h[m:]
    
    return obj, grad

def run_lse_opt(h_start, target, alpha, topk, maxiter, ftol):

    bounds = [(0.0, 1.0)] * n

        s = np.sum(x) - target
        f = f + 5000.0 * s * s
        g = g + 10000.0 * s
        return f, g
    
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

    """Project from coarse grid to fine grid using interpolation."""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return project_feasible(h_fine, n_fine / 2.0)

    """Project from fine grid to coarse grid by averaging."""

        h_coarse = h_fine.reshape(n_coarse, k).mean(axis=1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        lag = idx[i] - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        grad[m:] += w * one_minus_h[:n - m]
        grad[:n - m] -= w * h[m:]
    return obj, grad


def lse_optimize(h_start, target, alpha, topk, maxiter, ftol=1e-14):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    penalty = 5000.0

    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f += penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g

    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def down_project(h_fine, n_coarse):
    n_f = len(h_fine)
    x_f = np.linspace(0, 1, n_f)
    x_c = np.linspace(0, 1, n_coarse)
    h_c = np.interp(x_c, x_f, h_fine)
    return project_feasible(h_c, n_coarse / 2.0)


def up_project(h_coarse, n_fine):
    n_c = len(h_coarse)
    x_c = np.linspace(0, 1, n_c)
    x_f = np.linspace(0, 1, n_fine)
    h_f = np.interp(x_f, x_c, h_coarse)
    return project_feasible(h_f, n_fine / 2.0)
