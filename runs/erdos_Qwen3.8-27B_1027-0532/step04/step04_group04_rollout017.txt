def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Soft-max approximation
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def spectral_perturb(h, rng, n_freqs=10):

    return lse + lam * sum_err * sum_err

def logsumexp_grad(h, alpha, lam, target):

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def fast_greedy(h, max_iters, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            if stall > 80:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = compute_corr(h)
                if corr.max() < max_val:
                    stall = 0
                    if corr.max() * (2.0 / n) < best_c5:
                        best_c5 = float(corr.max()) * (2.0 / n)
                        best_h = h.copy()
                else:
                    h[i] = 1
                    h[j] = 0
                continue
            break
        found_improve = False
        n_try = min(150, len(i_cand) * len(j_cand))
        for _ in range(n_try):
            i = i_cand[rng.randint(len(i_cand))]
            j = j_cand[rng.randint(len(j_cand))]
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            new_max = float(corr.max())
            if new_max < max_val:
                new_c5 = new_max * (2.0 / n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                stall = 0
                found_improve = True
                break
            else:
                h[i] = 1
                h[j] = 0
        if not found_improve:
            stall += 1
    return best_h, best_c5

def sa_block(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 6000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        move_type = rng.randint(10)
        h_new = h.copy()
        moved = False
        if move_type <= 4:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            if rng.random() < 0.6:
                i_pool = [i for i in ones if 0 <= i + k_star < n]
                j_pool = [j for j in zeros if 0 <= j + k_star < n]
                if len(i_pool) > 0 and len(j_pool) > 0:
                    i = i_pool[rng.randint(len(i_pool))]
                    j = j_pool[rng.randint(len(j_pool))]
                    h_new[i] = 0
                    h_new[j] = 1
                    moved = True
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type <= 7:
            if len(ones) >= 2 and len(zeros) >= 2:
                idx1 = rng.randint(len(ones))
                idx2 = rng.randint(len(ones) - 1)
                if idx2 >= idx1:
                    idx2 += 1
                i1, i2 = ones[idx1], ones[idx2]
                idx3 = rng.randint(len(zeros))
                idx4 = rng.randint(len(zeros) - 1)
                if idx4 >= idx3:
                    idx4 += 1
                j1, j2 = zeros[idx3], zeros[idx4]
                h_new[i1] = 0
                h_new[i2] = 0
                h_new[j1] = 1
                h_new[j2] = 1
                moved = True
        else:
            block_len = rng.randint(2, min(10, n // 4))
            start1 = rng.randint(0, n - block_len)
            for _ in range(30):
                start2 = rng.randint(0, n - block_len)
                if abs(start2 - start1) >= block_len:
                    block1 = h[start1:start1 + block_len]
                    block2 = h[start2:start2 + block_len]
                    if abs(block1.sum() - block2.sum()) < 1e-10:
                        h_new[start1:start1 + block_len] = block2
                        h_new[start2:start2 + block_len] = block1
                        moved = True
                        break
        if not moved:
            continue
        if abs(h_new.sum() - h.sum()) > 1e-10:
            continue
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _smooth_obj(h, alpha, lam, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    sum_err = float(np.sum(h)) - target
    return lse + lam * sum_err * sum_err

def _smooth_grad(h, alpha, lam, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    p = exp_s / sum_exp
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    p_len = len(p)
    for j in range(n):
        l_lo = max(-(n-1), -j)
        l_hi = min(n-1, n-1-j)
        if l_hi >= l_lo:
            ls = np.arange(l_lo, l_hi + 1)
            idx = n - 1 + ls
            h_idx = j + ls
            grad[j] += np.sum(p[idx] * one_minus_h[h_idx])
        l_lo2 = max(-(n-1), j - (n-1))
        l_hi2 = min(n-1, j)
        if l_hi2 >= l_lo2:
            ls2 = np.arange(l_lo2, l_hi2 + 1)
            idx2 = n - 1 + ls2
            h_idx2 = j - ls2
            grad[j] -= np.sum(p[idx2] * h[h_idx2])
    sum_err = float(np.sum(h)) - target
    grad += 2.0 * lam * sum_err
    return grad

def lbfgs_refine(h_start, alpha, lam, target, maxiter, time_limit=None, start_time=None):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    try:
        res = minimize(_smooth_obj, h_cur, args=(alpha, lam, target),
                       jac=_smooth_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': maxiter * 2})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < curr_score:
            return h_new, sc
    except Exception:
        pass
    return h_cur, curr_score

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - _cons_target, 'jac': lambda h: np.ones(_cons_n)}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            def obj_packed(h, a):
                corr = np.correlate(h, 1.0 - h, mode='full')
                c_max = float(np.max(corr))
                shifted = np.clip(a * (corr - c_max), -700.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / a
            res = minimize(obj_packed, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def spectral_perturb(h, rng, n_freqs=10):
    n = len(h)
    H = np.fft.rfft(h)
    freqs = np.arange(1, len(H))
    selected = rng.choice(freqs, size=min(n_freqs, len(freqs)), replace=False)
    for f in selected:
        noise_mag = rng.uniform(0.1, 0.5) * max(abs(H[f]), 0.01)
        angle = rng.uniform(0, 2 * np.pi)
        H[f] += noise_mag * np.exp(1j * angle)
    h_new = np.fft.irfft(H, n=n)
    return project_feasible(h_new, np.sum(h))

    # Phase 5: Iterated local search with diverse perturbations
    for ils_iter in range(6):
        if tleft() < 60:
            break
        pert_type = rng.randint(3)
        h_pert = best_h.copy()
        if pert_type == 0:
            n_pert = max(3, int(4 + ils_iter * 2))
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
        elif pert_type == 1:
            h_pert = spectral_perturb(h_pert, rng, n_freqs=rng.randint(5, 15))
            h_pert = project_binary(h_pert, int(len(h_pert) / 2))
        else:
            n_blocks = rng.randint(2, 5)
            block_size = len(h_pert) // (n_blocks * 2)
            if block_size >= 2:
                for b in range(n_blocks):
                    s1 = b * 2 * block_size
                    s2 = s1 + block_size
                    h_pert[s1:s1+block_size], h_pert[s2:s2+block_size] = \
                        h_pert[s2:s2+block_size].copy(), h_pert[s1:s1+block_size].copy()
                h_pert = project_binary(h_pert, int(len(h_pert) / 2))
        allowed = min(40, tleft() * 0.06)
        h_ig, c5_ig = fast_greedy(h_pert, max_iters=15000, rng=rng,
                                  time_limit=allowed * 0.5, start_time=start_time)
        if c5_ig < best_score:
            best_score = c5_ig
            best_h = h_ig.copy()
        if tleft() < 40:
            break
        allowed2 = min(25, tleft() * 0.04)
        h_sa3, c5_sa3 = sa_block(h_ig, int(len(h_ig) / 2), int(min(40000, allowed2 * 2000)),
                                 0.008, 0.00005, rng,
                                 time_limit=allowed2, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()
        if tleft() < 30:
            break
        n_use = len(best_h)
        target_use_f = float(n_use) / 2.0
        h_one, sc_one = lbfgs_refine(best_h, 10000, 100.0, target_use_f, 300,
                                     time_limit=min(tleft() - 10, 30), start_time=start_time)
        if sc_one < best_score:
            best_score = sc_one
            best_h = h_one.copy()

    # Phase 6: Try different resolutions
    n_candidates = [n_init + 2, n_init - 2, n_init + 4, n_init - 4]
    n_candidates = [n for n in n_candidates if n >= 200 and n != n_init and n % 2 == 0]
    for n_try in n_candidates:
        if tleft() < 50:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(40, tleft() * 0.06)
        h_g2, c5_g2 = fast_greedy(h_bin_try, max_iters=15000, rng=rng,
                                  time_limit=allowed * 0.5, start_time=start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()
        if tleft() < 30:
            break
        target_try_f = float(n_try) / 2.0
        h_one2, sc_one2 = lbfgs_refine(h_g2, 5000, 80.0, target_try_f, 300,
                                       time_limit=min(tleft() - 10, 25), start_time=start_time)
        if sc_one2 < best_score:
            best_score = sc_one2
            best_h = h_one2.copy()

    # Phase 7: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, [8000, 20000, 50000],
                                 2000, 1e-16,
                                 time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
