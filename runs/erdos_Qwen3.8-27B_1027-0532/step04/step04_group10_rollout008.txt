from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def soft_max_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

    return np.clip(h, 0.0, 1.0)


def soft_max_obj_grad(h, alpha, topk):

        idx = np.argpartition(corr, -topk)[-topk:]

    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += (1.0 - h[lag:]) * w
                grad[lag:] += -h[:n - lag] * w
        else:
            m = -lag
            if m < n:
                grad[m:] += (1.0 - h[:n - m]) * w
                grad[:n - m] += -h[m:] * w
    return obj, grad


def nested_lbfgs(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol):
    def obj_func(x):
        f, g = soft_max_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f = f + penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def nesterov_loop(h_init, target, alpha, topk, lr, momentum, max_iter, time_fn):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    beta = momentum
    best_h = h.copy()
    best_score = compute_c5(h)
    stale = 0
    for it in range(max_iter):
        if time_fn() < 25:
            break
        obj, grad = soft_max_obj_grad(h, alpha, topk)
        s = np.sum(h) - target
        grad = grad + 500.0 * s
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        v = beta * v + (1.0 - beta) * (-lr * grad_normed)
        h_new = h + v
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score - 1e-10:
            best_score = score
            best_h = h_new.copy()
            stale = 0
            h = h_new
        else:
            stale += 1
            v *= 0.5
            if stale > 80:
                beta = max(0.0, beta - 0.1)
                stale = 0
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 55.0 - (time.time() - start_time)

    # Phase 1: Coarse-to-fine alpha with L-BFGS
    alphas_p1 = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]
    topks_p1 = [80, 60, 50, 40, 30, 25, 20, 15, 10]
    for i, (alpha, topk) in enumerate(zip(alphas_p1, topks_p1)):
        if time_left() < 50:
            break
        n_tries = 2 if time_left() > 300 else 1
        for t in range(n_tries):
            if time_left() < 30:
                break
            noise_scale = 0.004 - 0.0003 * i
            h_start = best_h + rng.randn(n) * noise_scale
            h_start = project_feasible(h_start, target)
            h_new = nested_lbfgs(h_start, target, alpha, topk, 2000.0, 400, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 2: Nesterov accelerated gradient
    if time_left() > 120:
        for alpha, topk, lr, mom, mi in [
            (500, 30, 0.4, 0.85, 400),
            (1000, 20, 0.3, 0.8, 350),
            (2000, 15, 0.2, 0.7, 300),
            (5000, 10, 0.1, 0.5, 250),
        ]:
            if time_left() < 40:
                break
            h_start = best_h + rng.randn(n) * 0.002
            h_start = project_feasible(h_start, target)
            h_new, score = nesterov_loop(h_start, target, alpha, topk, lr, mom, mi, time_left)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 3: Multi-resolution exploration
    res_list = [350, 450, 550, 700]
    for n_try in res_list:
        if time_left() < 100:
            break
        target_try = n_try / 2.0
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_try)
        bounds_try = [(0.0, 1.0)] * n_try

        for alpha, topk in [(100, 40), (500, 25), (2000, 15)]:
            if time_left() < 40:
                break
            h_start = h_interp + rng.randn(n_try) * 0.005
            h_start = project_feasible(h_start, target_try)
            h_new = nested_lbfgs(h_start, target_try, alpha, topk, 2000.0, 250, bounds_try, 1e-16)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                x_back = np.linspace(0, 1, n)
                best_h = np.interp(x_back, x_new, h_new)
                best_h = project_feasible(best_h, target)
            h_interp = h_new

    # Phase 4: Structured spectral perturbations
    if time_left() > 80:
        x = np.linspace(0, 2, n)
        freqs = [1, 2, 3, 4, 5, 6, 7, 8, 10, 12, 14, 16, 20, 25]
        amps = [0.03, 0.015, 0.008, 0.004]
        for _ in range(4):
            if time_left() < 30:
                break
            improved_round = False
            for f in freqs:
                for a in amps:
                    if time_left() < 15:
                        break
                    for phase in [0.0, np.pi / 2.0, np.pi]:
                        pert = a * np.sin(np.pi * f * x + phase)
                        h_try = project_feasible(best_h + pert, target)
                        sc = compute_c5(h_try)
                        if sc < best_score - 1e-9:
                            best_score = sc
                            best_h = h_try
                            improved_round = True
                            break
            if improved_round:
                if time_left() < 25:
                    break
                h_new = nested_lbfgs(best_h, target, 3000, 12, 5000.0, 300, bounds, 1e-17)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new

    # Phase 5: Very high alpha L-BFGS polish
    if time_left() > 50:
        for alpha in [5000, 10000, 20000]:
            if time_left() < 25:
                break
            for topk in [10, 5]:
                if time_left() < 15:
                    break
                h_start = best_h + rng.randn(n) * 0.0005
                h_start = project_feasible(h_start, target)
                h_new = nested_lbfgs(h_start, target, alpha, topk, 10000.0, 300, bounds, 1e-18)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new

    # Phase 6: SLSQP with tight constraint
    if time_left() > 60:
        for alpha in [500, 2000, 8000]:
            if time_left() < 30:
                break
            n_sl = n
            def obj_sls(x):
                f, _ = soft_max_obj_grad(x, alpha, 10)
                return f
            def jac_sls(x):
                _, g = soft_max_obj_grad(x, alpha, 10)
                return g
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n_sl)}]
            try:
                res = minimize(obj_sls, best_h, jac=jac_sls, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Fine coordinate descent
    if time_left() > 30:
        for round_id in range(4):
            if time_left() < 15:
                break
            improved_any = False
            for coord in range(0, n, 1):
                if time_left() < 10:
                    break
                orig = best_h[coord]
                for delta in [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005]:
                    h_try = best_h.copy()
                    h_try[coord] = np.clip(orig + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_h = h_try
                        improved_any = True
                        break
            if not improved_any:
                break

    # Phase 8: Final L-BFGS
    if time_left() > 20:
        h_new = nested_lbfgs(best_h, target, 15000, 8, 20000.0, 200, bounds, 1e-18)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
