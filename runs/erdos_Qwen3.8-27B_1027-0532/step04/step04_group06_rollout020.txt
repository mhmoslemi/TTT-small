```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def fast_binary_sa(h_init, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    """Fast SA for binary h using np.correlate for C5 evaluation."""
    n = len(h_init)
    dx = 2.0 / n
    h = project_binary(h_init, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    stall = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max(5000, max_iters // 5):
            break
        
        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Propose swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h[a] = 0.0
        h[b] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    
    return best_h, best_c5

def smooth_grad_descent(h_init, target_sum, n_iters, alpha, lr_init, lr_final, time_limit, start_time, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.astype(float), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        ratio = i / max(n_iters - 1, 1)
        lr = lr_init * (lr_final / lr_init) ** ratio
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Log-sum-exp for stability
        max_corr = np.max(corr)
        shifted = alpha * (corr - max_corr)
        log_sum_exp = max_corr + np.log(np.sum(np.exp(shifted))) / alpha
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Gradient
        grad = np.zeros(n)
        center = n - 1
        # For each lag p, add weights[p] * (gradient of corr[p])
        # corr[center + d] for d > 0: grad[j] += (1-h[j-d])*1_{j>=d} - h[j+d]*1_{j<=n-1-d}
        # corr[center - d] for d > 0: grad[j] += (1-h[j+d])*1_{j<=n-1-d} - h[j-d]*1_{j>=d}
        # corr[center]: grad[j] += 1 - 2*h[j]
        
        for d in range(1, n):
            w_pos = weights[center + d]  # weight for corr[center+d]
            w_neg = weights[center - d]  # weight for corr[center-d]
            if w_pos < 1e-10 and w_neg < 1e-10:
                continue
            
            # corr[center+d] = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # grad at j: (1-h[j-d])*1_{j>=d} - h[j+d]*1_{j<=n-1-d}
            if w_pos > 1e-10:
                # First term: (1-h[j-d]) for j >= d
                j_range = np.arange(d, n)
                grad[j_range] += w_pos * (1.0 - h[j_range - d])
                # Second term: -h[j+d] for j <= n-1-d
                j_range2 = np.arange(0, n - d)
                grad[j_range2] += w_pos * (-h[j_range2 + d])
            
            # corr[center-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # grad at j: (1-h[j+d])*1_{j<=n-1-d} - h[j-d]*1_{j>=d}
            if w_neg > 1e-10:
                j_range = np.arange(0, n - d)
                grad[j_range] += w_neg * (1.0 - h[j_range + d])
                j_range2 = np.arange(d, n)
                grad[j_range2] += w_neg * (-h[j_range2 - d])
        
        # Center lag
        grad += weights[center] * (1.0 - 2.0 * h)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
```

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

                    zeros_arr = np.where(h == 0)[0]
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    for pos, val in old_vals:
                        h[pos] = val
                    stall += 1
                continue
        h[a] = 0.0
        h[b] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - current_c5
        if delta < -1e-15 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            current_c5 = c5_new
            idx_a = np.where(ones_arr == a)[0]
            if len(idx_a) > 0:
                ones_arr = np.delete(ones_arr, idx_a[0])
            ones_arr = np.append(ones_arr, b)
            idx_b = np.where(zeros_arr == b)[0]
            if len(idx_b) > 0:
                zeros_arr = np.delete(zeros_arr, idx_b[0])
            zeros_arr = np.append(zeros_arr, a)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    return best_h, best_c5

def smooth_gradient_step(h, alpha, top_k=10):
    n = len(h)
    corr = get_corr(h)
    center = n - 1
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    sorted_ids = np.argsort(-weights)[:top_k]
    for p in sorted_ids:
        w = weights[p]
        if w < 1e-8:
            continue
        d = int(p) - center
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            j1 = np.arange(d, n)
            grad[j1] += w * (1.0 - h[j1 - d])
            j2 = np.arange(0, n - d)
            grad[j2] += w * (-h[j2 + d])
        else:
            dd = -d
            j1 = np.arange(0, n - dd)
            grad[j1] += w * (1.0 - h[j1 + dd])
            j2 = np.arange(dd, n)
            grad[j2] += w * (-h[j2 - dd])
    return grad

def gradient_descent(h_init, target_sum, n_iters, alpha, lr_init, lr_final, time_limit, start_time, rng):
    n = len(h_init)
    h = project_feasible(h_init.astype(float), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for i in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = i / max(n_iters - 1, 1)
        lr = lr_init * (lr_final / lr_init) ** ratio
        grad = smooth_gradient_step(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        if rng.random() < 0.1:
            h_new = h_new + rng.randn(n) * 0.01
            h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_target = target
    cons_n = n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj_smooth(h_in, a=alpha):
                corr = np.correlate(h_in, 1.0 - h_in, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
            def cons_f(h_in):
                return np.sum(h_in) - cons_target
            def cons_j(h_in):
                return np.ones(cons_n)
            cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
            res = minimize(obj_smooth, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_structured_start(n, rng, pattern='quadratic'):
    h = np.zeros(n)
    target = n // 2
    if pattern == 'quadratic':
        p = 4 * n + 3
        while p % 4 != 3 or p < n:
            p += 4
        for i in range(n):
            if pow(i * i, p - 1, p) == 1 or i == 0:
                h[i] = 1.0
        h = project_binary(h, target)
    elif pattern == 'alternating':
        h[::2] = 1.0
        h = project_binary(h, target)
    elif pattern == 'blocks':
        block_size = max(2, n // 20)
        for i in range(0, n, block_size * 2):
            h[i:i + block_size] = 1.0
        h = project_binary(h, target)
    else:
        perm = rng.permutation(n)
        h[perm[:target]] = 1.0
    return h

def hill_climb_binary(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    stall = 0
    for _ in range(max_iter):
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break
        if stall > 200:
            break
        a = ones_arr[rng.randint(len(ones_arr))]
        b = zeros_arr[rng.randint(len(zeros_arr))]
        h[a] = 0.0
        h[b] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            idx_a = np.where(ones_arr == a)[0]
            if len(idx_a) > 0:
                ones_arr = np.delete(ones_arr, idx_a[0])
            ones_arr = np.append(ones_arr, b)
            idx_b = np.where(zeros_arr == b)[0]
            if len(idx_b) > 0:
                zeros_arr = np.delete(zeros_arr, idx_b[0])
            zeros_arr = np.append(zeros_arr, a)
            stall = 0
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)

    # Phase 1: Targeted SA at native resolution
    if tleft() > 150:
        allowed = min(250.0, tleft() * 0.4)
        h_sa, c5_sa = targeted_binary_sa(h_bin_start, target_init, 150000, 0.015, 0.00005,
                                         rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 2: Fast SA with different temps at native resolution
    temp_configs = [
        (0.02, 0.0001, 100000),
        (0.008, 0.00003, 120000),
        (0.004, 0.00001, 80000),
    ]
    for tc in temp_configs:
        if tleft() < 80:
            break
        ti, tf, mi = tc
        allowed = min(180.0, tleft() * 0.3)
        h_fa, c5_fa = fast_binary_sa(best_h, target_init, mi, ti, tf,
                                     rng, time_limit=allowed, start_time=start_time)
        if c5_fa < best_score:
            best_score = c5_fa
            best_h = h_fa.copy()

    # Phase 3: Structured starts at native resolution
    patterns = ['quadratic', 'alternating', 'blocks']
    for pat in patterns:
        if tleft() < 60:
            break
        h_struct = make_structured_start(n_init, rng, pat)
        allowed = min(120.0, tleft() * 0.2)
        h_sa, c5_sa = targeted_binary_sa(h_struct, target_init, 80000, 0.012, 0.00005,
                                         rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Hill climb polish at native resolution
    if tleft() > 50:
        h_hc, c5_hc = hill_climb_binary(best_h, target_init, 20000, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 5: Multi-resolution SA
    n_candidates = [1024, 2048, 1500]
    for n_try in n_candidates:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(200.0, tleft() * 0.3)
        h_sa, c5_sa = targeted_binary_sa(h_bin_try, target_try, 100000, 0.01, 0.00004,
                                         rng, time_limit=allowed, start_time=start_time)
        if tleft() > 40:
            h_hc, c5_hc = hill_climb_binary(h_sa, target_try, 10000, rng)
            if c5_hc < c5_sa:
                h_sa = h_hc
                c5_sa = c5_hc
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Gradient descent on continuous relaxation
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * 0.05
        h_pert = project_feasible(h_pert, target_use)
        allowed = min(150.0, tleft() * 0.25)
        h_gd, c5_gd = gradient_descent(h_pert, target_use, 5000, 200.0, 0.3, 0.01,
                                       time_limit=allowed, start_time=start_time)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd.copy()

    # Phase 7: SLSQP refinement
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 80:
        alphas = [50, 200, 800, 3000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 400, 1e-13,
                                 time_limit=min(120.0, tleft() - 30), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Binary polish after SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        h_fa, c5_fa = fast_binary_sa(h_bin_final, target_use, 50000, 0.005, 0.00002,
                                     rng, time_limit=min(60.0, tleft() - 15), start_time=start_time)
        if c5_fa < best_score:
            best_score = c5_fa
            best_h = h_fa.copy()

    # Phase 9: Final gradient + SLSQP combo
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert2 = best_h + rng.randn(n_use) * 0.02
        h_pert2 = project_feasible(h_pert2, target_use)
        h_gd2, c5_gd2 = gradient_descent(h_pert2, target_use, 2000, 500.0, 0.1, 0.005,
                                         time_limit=min(40.0, tleft() * 0.4), start_time=start_time)
        if c5_gd2 < best_score:
            best_score = c5_gd2
            best_h = h_gd2.copy()
        if tleft() > 30:
            h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, [1000, 5000], 200, 1e-14,
                                       time_limit=min(30.0, tleft() - 10), start_time=start_time)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        final_score = best_score
        h_final = best_h
    return h_final, final_score, n_final
