def adam_subgrad(h, n, target, n_steps=2000, top_m=8, lr=0.005,
                 beta1=0.9, beta2=0.999, eps=1e-8, temp=0.003):
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stuck_count = 0
    
    for step in range(n_steps):
        over = compute_overlaps(h)
        current_score = float(np.max(over)) * dx
        
        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        if stuck_count > 200:
            break
        
        # Softmax weights on top M
        top_idx = np.argsort(over)[::-1][:top_m]
        top_vals = over[top_idx]
        w = np.exp((top_vals - top_vals[0]) / temp)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            if wi > 1e-10:
                grad += wi * dS_k(h, int(ki), n)
        grad *= dx
        
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < current_score:
            h = h_new
        else:
            # Try smaller step
            h_new2 = h - (lr * 0.5) * m_hat / (np.sqrt(v_hat) + eps)
            h_new2 = project_feasible(h_new2, target)
            sc_new2 = compute_c5(h_new2)
            if sc_new2 < current_score:
                h = h_new2
            # else: keep current
    
    return best_h, best_score
