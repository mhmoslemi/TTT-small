def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):

    corr = compute_corr(h)
    best_max = float(np.max(corr))
    
    dx = 2.0 / n
    best_c5 = best_max * dx

        if time.time() > deadline:
            break

        zeros = np.where(h == 0)[0]

                h[a] = 0
                h[b] = 1
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                gain = best_max - new_max
                if gain > best_gain:

                    best_swap = (a, b)
                    best_new_corr = new_corr
                    best_new_max = new_max
                # Restore
                h[a] = 1
                h[b] = 0

            h[a] = 0
            h[b] = 1
            corr = best_new_corr
            best_max = best_new_max
            best_c5 = best_max * dx

                print(f"Round {rnd}: max={best_max:.1f}, C5={best_c5:.6f}")
        else:
            break

def compute_delta_remove(h):
    n = len(h)
    # shifts[idx] = idx - (n-1) for idx = 0, ..., 2n-2
    shifts = np.arange(-(n-1), n)  # shifts[0]=-n+1, shifts[n-1]=0, shifts[2n-2]=n-1
    
    # For each position a:
    # delta[a, idx] = -(1-h[a-shifts[idx]]) * I(0≤a-shifts[idx]<n) + h[a+shifts[idx]] * I(0≤a+shifts[idx]<n)
    
    rows = np.arange(n)
    # a - shift = rows[:,None] - shifts[None,:]
    a_minus_shift = rows[:, None] - shifts[None, :]  # (n, 2n-1)
    a_plus_shift = rows[:, None] + shifts[None, :]  # (n, 2n-1)
    
    mask1 = (a_minus_shift >= 0) & (a_minus_shift < n)
    mask2 = (a_plus_shift >= 0) & (a_plus_shift < n)
    
    ams_safe = np.where(mask1, a_minus_shift, 0)
    aps_safe = np.where(mask2, a_plus_shift, 0)
    
    D = np.zeros((n, 2*n-1))
    D -= (1.0 - h[ams_safe]) * mask1
    D += h[aps_safe] * mask2
    return D
