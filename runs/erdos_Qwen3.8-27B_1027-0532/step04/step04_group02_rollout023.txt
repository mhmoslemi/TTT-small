def delta_corr(h, i1, i0, n):
    """Compute the change in the full correlation vector when swapping h[i1]=1->0 and h[i0]=0->1.
    Returns an array of length 2n-1 (the change in each correlation value).
    Uses the formula: Δcorr[i] for i=0..2n-2, where t = i-(n-1)
    """
    t = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    
    # b_new[i0-t] - b_new[i1-t]
    # b_new = 1 - h_new where h_new[i1]=0, h_new[i0]=1
    
    # For b_new[x]: 
    #   if x == i1: b_new = 1 (h_new[i1] = 0)
    #   if x == i0: b_new = 0 (h_new[i0] = 1)
    #   else: b_new = 1 - h[x]
    
    idx0 = i0 - t  # positions in b_new
    idx1 = i1 - t
    
    b_new_0 = np.empty(2*n-1)
    b_new_1 = np.empty(2*n-1)
    
    # Default: 1-h[x] for valid x
    mask0 = (idx0 >= 0) & (idx0 < n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    
    b_new_0[:] = 0.0
    b_new_1[:] = 0.0
    b_new_0[mask0] = 1.0 - h[idx0[mask0]]
    b_new_1[mask1] = 1.0 - h[idx1[mask1]]
    
    # Override for special cases
    # b_new[i1] = 1
    b_new_0[idx0 == i1] = 1.0
    b_new_1[idx1 == i1] = 1.0
    # b_new[i0] = 0
    b_new_0[idx0 == i0] = 0.0
    b_new_1[idx1 == i0] = 0.0
    
    # h[i1+t] - h[i0+t] (using OLD h)
    idx_h1 = i1 + t
    idx_h0 = i0 + t
    
    h_part = np.zeros(2*n-1)
    mask_h1 = (idx_h1 >= 0) & (idx_h1 < n)
    mask_h0 = (idx_h0 >= 0) & (idx_h0 < n)
    h_part[mask_h1] = h[idx_h1[mask_h1]]
    h_part[mask_h0] -= h[idx_h0[mask_h0]]
    
    return b_new_0 - b_new_1 + h_part

                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
        if not improved:
            break
    return h, best_c5


def top_p_smooth(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = fast_corr(h) * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    if alpha < 1e-6:
        return float(max_top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_top_p(h_start, n, target, p_vals, alphas, maxiter, ftol,
                time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(lambda x: top_p_smooth(x, alpha, p), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
                h_bin = project_binary(h_new, target)
                sc_bin = compute_c5(h_bin)
                if sc_bin < curr_score:
                    curr_score = sc_bin
                    h_cur = h_bin
            except Exception:
                pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def quick_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 150

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-14
        best_pair = None
        worst_k = int(np.argmax(corr))
        t_star = worst_k - (n - 1)
        candidates = 0
        for i1 in ones[:min(len(ones), 50)]:
            if candidates > 200:
                break
            for i0 in zeros[:min(len(zeros), 50)]:
                delta = 0.0
                j1m = i1 - t_star
                j0m = i0 - t_star
                j1p = i1 + t_star
                j0p = i0 + t_star
                if 0 <= j1m < n and j1m != i0:
                    delta += h[j1m]
                if 0 <= j0m < n and j0m != i1:
                    delta -= h[j0m]
                if 0 <= j1p < n and j1p != i0:
                    delta += h[j1p]
                if 0 <= j0p < n and j0p != i1:
                    delta -= h[j0p]
                new_val = corr[worst_k] + delta * (2.0 / n)
                gain = best_c5 - max(new_val, np.max(np.delete(corr, worst_k)))
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i1, i0)
                candidates += 1
        if best_pair is not None and best_gain > 0:
            h[best_pair[1]] = 1
            h[best_pair[0]] = 0
            corr = compute_corr(h)
            new_c5 = float(np.max(corr))
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stall = 0
                continue
        h[best_pair[1]] = 1
        h[best_pair[0]] = 0
        stall += 1
        if stall > max_stall:
            break
    return h, best_c5


def multi_time_sa(h_start, target_sum, total_time, rng, start_time):
    n = len(h_start)
    h = h_start.copy()
    if not np.allclose(h, np.round(h).astype(float)):
        h = project_binary(h, target_sum)
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    current_c5 = best_c5

    phases = [
        (0.4, 0.02, 0.002),
        (0.3, 0.002, 0.0003),
        (0.3, 0.0003, 0.00005),
    ]

    for frac, t_init, t_final in phases:
        phase_time = total_time * frac
        max_iters = int(phase_time * 3000)
        best_h, best_c5 = informed_sa(h, target_sum, max_iters, t_init, t_final,
                                       rng, time_limit=phase_time, start_time=start_time)
        corr = compute_corr(best_h)
        current_c5 = float(np.max(corr))

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = np.array(h_init, dtype=float)
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)
    bin_score = compute_c5(h_bin_start)
    if bin_score < best_score:
        best_score = bin_score
        best_h = h_bin_start.copy()

    starts = generate_starts(n_init, rng, 8)

    for si in range(len(starts)):
        if tleft() < 120:
            break
        h_s = starts[si].copy()
        ph_score = compute_c5(h_s)
        if ph_score < best_score:
            best_score = ph_score
            best_h = h_s.copy()
        allowed_time = min(150.0, tleft() * 0.35)
        h_sa, c5_sa = multi_time_sa(h_s, target_init, allowed_time, rng, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        if tleft() > 60:
            h_gr, c5_gr = greedy_refine(best_h, target_init, max_rounds=15,
                                         time_limit=min(30.0, tleft() - 50), start_time=start_time)
            if c5_gr < best_score:
                best_score = c5_gr
                best_h = h_gr.copy()
            h_hc, c5_hc = quick_hill_climb(best_h, target_init, 3000, rng)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 100:
        p_vals = [max(3, n_use // 60), max(5, n_use // 35), max(8, n_use // 20)]
        alphas = [100, 300, 800, 2000]
        h_ref, sc = slsqp_top_p(best_h, n_use, target_use, p_vals, alphas,
                                500, 1e-14, time_limit=min(80.0, tleft() - 60), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    n_candidates = []
    for nf in [0.75, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try != n_init and n_try >= 100:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed_time = min(100.0, tleft() * 0.3)
        h_sa, c5_sa = multi_time_sa(h_bin_try, target_try, allowed_time, rng, start_time)
        if tleft() > 40:
            h_gr, c5_gr = greedy_refine(h_sa, target_try, max_rounds=10,
                                         time_limit=min(20.0, tleft() - 30), start_time=start_time)
            if c5_gr < c5_sa:
                h_sa = h_gr
                c5_sa = c5_gr
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        p_vals2 = [max(3, n_use // 50), max(5, n_use // 25), max(10, n_use // 15)]
        alphas2 = [200, 600, 1500, 3000]
        h_ref2, sc2 = slsqp_top_p(best_h, n_use, target_use, p_vals2, alphas2,
                                   400, 1e-15, time_limit=min(60.0, tleft() - 50), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    fine_attempts = min(6, int(max(2, tleft() / 40)))
    for fa in range(fine_attempts):
        if tleft() < 30:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        noise = rng.choice([0.005, 0.01, 0.02])
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        h_pert_bin = project_binary(h_pert, target_use)
        p_fine = [max(3, n_use // 40)]
        alphas_fine = [500, 1500, 3000]
        h_ref, sc = slsqp_top_p(h_pert_bin, n_use, target_use, p_fine, alphas_fine,
                                 300, 1e-15, time_limit=min(25.0, tleft() - 15), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        allowed_final = max(5.0, tleft() - 15)
        h_sa_f, c5_f = informed_sa(h_bin_final, target_use, int(allowed_final * 3000),
                                    0.003, 0.00002, rng,
                                    time_limit=allowed_final, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()
        if tleft() > 12:
            h_gr_f, c5_gr_f = greedy_refine(best_h, target_use, max_rounds=10,
                                             time_limit=max(3.0, tleft() - 8), start_time=start_time)
            if c5_gr_f < best_score:
                best_score = c5_gr_f
                best_h = h_gr_f.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h.copy()
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
