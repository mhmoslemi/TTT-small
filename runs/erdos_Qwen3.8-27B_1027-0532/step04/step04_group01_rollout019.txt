The effect of swapping (a: 1→0, b: 0→1) on lag k:
- If a is a "contributing" position for lag k (h[a]=1, h[a+k]=0): removing it reduces corr_k by 1
- If b is a "resolving" position for lag k (h[b-k]=1, h[b]=0): adding it reduces corr_k by 1 (because h[b] becomes 1, so (1-h[b]) becomes 0)

```python
def eval_swap_effect(h, a, b, n, lags):
    """Compute change in corr at given lags for swap a(1->0), b(0->1)"""
    effects = np.zeros(len(lags))
    for idx, k in enumerate(lags):
        delta = 0.0
        # a going from 1 to 0
        # As first index: was contributing (1-h[a+k]) if valid
        if 0 <= a + k < n:
            delta -= (1.0 - h[a + k])
        # As second index: was contributing h[a-k] (since 1-h[a] was 0, now 1)
        if 0 <= a - k < n:
            delta += h[a - k]
        # b going from 0 to 1
        # As first index: now contributing (1-h[b+k]) if valid
        if 0 <= b + k < n:
            delta += (1.0 - h[b + k])
        # As second index: was contributing h[b-k] (since 1-h[b] was 1, now 0)
        if 0 <= b - k < n:
            delta -= h[b - k]
        effects[idx] = delta
    return effects
```

```python
def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.int8)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(np.float64)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        if need >= len(zeros):
            return np.ones(n, dtype=np.float64)  # shouldn't happen
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        if need >= len(ones):
            return np.zeros(n, dtype=np.float64)
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(np.float64)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)
```

```python
    """Find a swap that reduces the most critical lags."""
    # For each top lag k, compute the set of contributing ones (a) and zeros (b)
    # A swap (a, b) reduces lag k if:
    #   - a is a contributing one for k (h[a]=1, h[a+k]=0), OR
    #   - b is a "zero end" for k (h[b-k]=1, h[b]=0)
    
    n_lags = len(top_lags)
    
    # Compute pressure for each position
    one_pressure = np.zeros(n)  # How many top lags this one contributes to
    zero_relief = np.zeros(n)   # How many top lags this zero "blocks"
    
    for k in top_lags:
        if k > 0:
            i_range = range(0, n - k)
        elif k < 0:
            i_range = range(-k, n)
        else:
            continue  # Lag 0 always has corr=0 for binary
        
        for i in i_range:
            if 0 <= i + k < n:
                if h[i] == 1 and h[i + k] == 0:
                    one_pressure[i] += 1
                    zero_relief[i + k] += 1
    
    # Find the best one to flip (high pressure) and best zero to flip (high relief)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Pick from top-pressure ones
    one_order = np.argsort(-one_pressure[ones])
    zero_order = np.argsort(-zero_relief[zeros])

    best_a, best_b = -1, -1
    best_score = 0
```

```python
def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.int8)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(np.float64)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        if need >= len(zeros):
            return np.ones(n, dtype=np.float64)
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        if need >= len(ones):
            return np.zeros(n, dtype=np.float64)
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(np.float64)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)

    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]

    h = h.copy()

        if it % 300 == 1:
            top_lag_idx = np.argsort(corr)[-n_top_lags:]
            lags_cached = (top_lag_idx - (n - 1)).astype(int)
            corr_add = corr[top_lag_idx].copy()

        r = rng.random()
        a = -1
        b = -1

        if r < lag_aware_frac:
            k = int(lags_cached[rng.randint(n_top_lags)])
            if k > 0:
                i_lo, i_hi = 0, n - k - 1
            elif k < 0:
                i_lo, i_hi = -k, n - 1
            else:
                i_lo, i_hi = 0, n - 1
            found = False
            for _ in range(40):
                i = rng.randint(i_lo, i_hi + 1)
                if h[i] == 1 and 0 <= i + k < n and h[i + k] == 0:
                    a = i
                    found = True
                    break
            if not found:
                a = int(ones_arr[rng.randint(len(ones_arr))])
            b = int(zeros_arr[rng.randint(len(zeros_arr))])

        elif r < lag_aware_frac + (1.0 - lag_aware_frac - greedy_frac) * 0.5:
            k = int(lags_cached[rng.randint(n_top_lags)])
            if k > 0:
                i_lo, i_hi = 0, n - k - 1
            elif k < 0:
                i_lo, i_hi = -k, n - 1
            else:
                i_lo, i_hi = 0, n - 1
            found = False
            for _ in range(40):
                i = rng.randint(i_lo, i_hi + 1)
                if h[i] == 1 and 0 <= i + k < n and h[i + k] == 0:
                    a = i
                    found = True
                    break
            if not found:
                a = int(ones_arr[rng.randint(len(ones_arr))])
            if rng.random() < 0.5:
                if 0 <= a + k < n and h[a + k] == 0:
                    b = a + k
                    if b == a:
                        b = int(zeros_arr[rng.randint(len(zeros_arr))])
            else:
                if k != 0 and 0 <= a - k < n and h[a - k] == 1:
                    a2 = a - k
                    a = a2
                    b = int(zeros_arr[rng.randint(len(zeros_arr))])
                else:
                    b = int(zeros_arr[rng.randint(len(zeros_arr))])

        elif r < 1.0 - greedy_frac:
            a = int(ones_arr[rng.randint(len(ones_arr))])
            b = int(zeros_arr[rng.randint(len(zeros_arr))])

        else:
            best_s = cur_max + 1.0
            best_a, best_b = -1, -1
            for _ in range(5):
                ca = int(ones_arr[rng.randint(len(ones_arr))])
                cb = int(zeros_arr[rng.randint(len(zeros_arr))])
                if ca == cb:
                    continue
                h[ca] = 0
                h[cb] = 1
                nm = float(np.max(compute_corr_full(h)))
                h[ca] = 1
                h[cb] = 0
                if nm < best_s:
                    best_s = nm
                    best_a, best_b = ca, cb
            if best_a >= 0:
                a, b = best_a, best_b
            else:
                a = int(ones_arr[rng.randint(len(ones_arr))])
                b = int(zeros_arr[rng.randint(len(zeros_arr))])

        if a < 0 or a == b:
            continue

        h[a] = 0
        h[b] = 1
        new_corr = compute_corr_full(h)
        new_max = float(np.max(new_corr))
        delta = (new_max - cur_max) * 2.0 / n

        if delta < -1e-15 or (T > t_min and rng.random() < np.exp(-delta / T)):
            corr = new_corr
            cur_max = new_max
            cur_c5 = cur_max * 2.0 / n
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[a] = 1
            h[b] = 0
            no_imp += 1

        T = max(t_min, T * cooling)

        if no_imp > max_no_imp:
            h = best_h.copy()
            corr = compute_corr_full(h)
            cur_max = float(np.max(corr))
            cur_c5 = cur_max * 2.0 / n
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            no_imp = 0
            n_pert = rng.randint(3, max(4, n // 40) + 1)
            for _ in range(n_pert):
                pa = int(ones_arr[rng.randint(len(ones_arr))])
                pb = int(zeros_arr[rng.randint(len(zeros_arr))])
                h[pa] = 0
                h[pb] = 1
            corr = compute_corr_full(h)
            cur_max = float(np.max(corr))
            cur_c5 = cur_max * 2.0 / n
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]

    return best_h, best_c5


def pressure_local_search(h_init, target, time_limit, t_start, rng):
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int)

    corr = compute_corr_full(h)
    cur_max = float(np.max(corr))
    cur_c5 = cur_max * 2.0 / n
    best_h = h.copy()
    best_c5 = cur_c5

    T = 0.008
    T_min = 0.000002
    cooling = 0.9999
    no_imp = 0
    n_top = 12

    top_idx = np.argsort(corr)[-n_top:]
    top_lags = (top_idx - (n - 1)).astype(int)

    it = 0
    while time.time() - t_start < time_limit:
        it += 1

        if it % 500 == 1:
            top_idx = np.argsort(corr)[-n_top:]
            top_lags = (top_idx - (n - 1)).astype(int)

        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]

        one_pressure = np.zeros(n)
        zero_relief = np.zeros(n)

        for k in top_lags:
            k = int(k)
            if k > 0:
                for i in range(0, n - k):
                    if h[i] == 1 and h[i + k] == 0:
                        one_pressure[i] += 1.0
                        zero_relief[i + k] += 1.0
            elif k < 0:
                for i in range(-k, n):
                    if h[i] == 1 and h[i + k] == 0:
                        one_pressure[i] += 1.0
                        zero_relief[i + k] += 1.0

        op = one_pressure[ones_arr]
        zr = zero_relief[zeros_arr]

        a = -1
        b = -1
        best_score = -1.0

        for _ in range(15):
            ia = int(ones_arr[rng.randint(len(ones_arr))])
            ib = int(zeros_arr[rng.randint(len(zeros_arr))])
            if ia == ib:
                continue
            score = one_pressure[ia] + zero_relief[ib]
            if score > best_score:
                best_score = score
                a = ia
                b = ib

        if a < 0:
            a = int(ones_arr[rng.randint(len(ones_arr))])
            b = int(zeros_arr[rng.randint(len(zeros_arr))])

        if a == b:
            continue

        h[a] = 0
        h[b] = 1
        new_corr = compute_corr_full(h)
        new_max = float(np.max(new_corr))
        delta = (new_max - cur_max) * 2.0 / n

        if delta < -1e-15 or (T > T_min and rng.random() < np.exp(-delta / T)):
            corr = new_corr
            cur_max = new_max
            cur_c5 = new_max * 2.0 / n
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[a] = 1
            h[b] = 0
            no_imp += 1

        T = max(T_min, T * cooling)

        if no_imp > 5000:
            h = best_h.copy()
            corr = compute_corr_full(h)
            cur_max = float(np.max(corr))
            cur_c5 = cur_max * 2.0 / n
            no_imp = 0
            n_pert = rng.randint(4, max(5, n // 30) + 1)
            ones_t = np.where(h == 1)[0]
            zeros_t = np.where(h == 0)[0]
            for _ in range(n_pert):
                pa = int(ones_t[rng.randint(len(ones_t))])
                pb = int(zeros_t[rng.randint(len(zeros_t))])
                h[pa] = 0
                h[pb] = 1
            corr = compute_corr_full(h)
            cur_max = float(np.max(corr))
            cur_c5 = cur_max * 2.0 / n

    return best_h, best_c5


def continuous_polish(h_init, target, time_limit, t_start):
    n = len(h_init)
    h = project_feasible(h_init.astype(np.float64), target)
    dx = 2.0 / n

    def soft_max_obj(x, alpha):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr_x)
        shifted = alpha * (corr_x - cmax)
        shifted = np.clip(shifted, -500.0, 0.0)
        return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    alphas = [30, 80, 200, 500]
    h_cur = h.copy()
    sc_cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                lambda x: soft_max_obj(x, alpha),
                h_cur,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            h_new = project_feasible(res.x, target)
            sc_new = compute_c5(h_new)
            if sc_new < sc_cur:
                sc_cur = sc_new
                h_cur = h_new
        except Exception:
            pass

    return h_cur, sc_cur


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    target = n // 2

    best_h = project_binary(h_init, target)
    best_c5 = compute_c5(best_h)

    elapsed = time.time() - start
    phase1_budget = budget_s * 0.55

    h1, c1 = local_search_v2(best_h, target, phase1_budget, start,
                             rng, t_init=0.012, t_min=0.000003,
                             cooling=0.99985, max_no_imp=4000,
                             greedy_frac=0.15, lag_aware_frac=0.55)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    elapsed = time.time() - start
    remaining = budget_s - 30.0 - elapsed

    phase2_budget = min(remaining * 0.35, budget_s * 0.2)
    h2, c2 = pressure_local_search(best_h, target, phase2_budget,
                                   time.time(), rng)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2

    elapsed = time.time() - start
    remaining = budget_s - 30.0 - elapsed

    alt_starts = []
    alt_starts.append(make_thue_morse(n))
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts.append(make_beatty(n, golden))
    alt_starts.append(make_qr_sequence(n, rng))
    h_rand = np.zeros(n)
    rand_flip = rng.randint(0, 2, n)
    h_rand[rand_flip == 1] = 1.0
    h_rand = project_binary(h_rand, target)
    alt_starts.append(h_rand)

    for h_alt in alt_starts:
        elapsed = time.time() - start
        remaining = budget_s - 30.0 - elapsed
        if remaining < 60.0:
            break
        alt_budget = min(remaining * 0.3, 120.0)
        ha, ca = local_search_v2(h_alt, target, alt_budget, time.time(),
                                 rng, t_init=0.010, t_min=0.000005,
                                 cooling=0.9998, max_no_imp=3000,
                                 greedy_frac=0.2, lag_aware_frac=0.5)
        if ca < best_c5:
            best_c5 = ca
            best_h = ha

    elapsed = time.time() - start
    remaining = budget_s - 30.0 - elapsed

    for dn in [2, -2, 4, -4, 6]:
        elapsed = time.time() - start
        remaining = budget_s - 30.0 - elapsed
        if remaining < 50.0:
            break
        nt = n + dn
        if nt < 100:
            continue
        target_t = nt // 2
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, nt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin = project_binary(h_interp, target_t)
        hn, cn = local_search_v2(h_bin,
                                 float(target_t),
                                 min(remaining * 0.2, 80.0),
                                 time.time(), rng,
                                 t_init=0.010, t_min=0.00001,
                                 cooling=0.9998, max_no_imp=2500,
                                 greedy_frac=0.2, lag_aware_frac=0.5)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    elapsed = time.time() - start
    remaining = budget_s - 30.0 - elapsed

    if remaining > 40.0:
        polish_budget = min(remaining * 0.6, 100.0)
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, cp = continuous_polish(best_h, target_use, polish_budget,
                                   time.time())
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    elapsed = time.time() - start
    remaining = budget_s - 20.0 - elapsed

    if remaining > 15.0:
        n_final_try = len(best_h)
        t_final = n_final_try / 2.0
        h_bin_f = project_binary(best_h, t_final)
        hf, cf = local_search_v2(h_bin_f, t_final, remaining,
                                 time.time(), rng,
                                 t_init=0.005, t_min=0.000001,
                                 cooling=0.99995, max_no_imp=3000,
                                 greedy_frac=0.25, lag_aware_frac=0.6)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    n_out = len(best_h)
    target_out = n_out / 2.0
    h_out = project_feasible(best_h, target_out)
    c_out = compute_c5(h_out)
    if c_out > best_c5:
        h_out = best_h
        c_out = best_c5

    return h_out, c_out, n_out
```
