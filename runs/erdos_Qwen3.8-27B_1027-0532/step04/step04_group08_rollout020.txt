    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

                h_cur = h_new
                v = v_new
            else:
                lr *= 0.6
                if lr < 1e-14:
                    v = np.zeros(n)
                    lr = lr_init
                    break
    return best_h, best_score

    best_score = compute_c5(h_cur)

            continue

                return h_cur, best_score

                    return h_cur, best_score

                    h_full[s:e] = np.clip(x, 0.0, 1.0)

                    pass

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def soft_max_gradient(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)

    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)

def nesterov_descent(h_init, target, alpha_list, lr_init, beta, deadline):

    v = np.zeros(n)

def _block_objective(x):
    h_full = _g_h_block.copy()
    h_full[_g_bstart:_g_bend] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    return compute_c5(h_full)


def block_descent(h_init, target, block_sizes, n_rounds, deadline, rng):
    n = len(h_init)
    h_cur = h_init.copy()
    best_score = compute_c5(h_cur)
    for bs in block_sizes:
        n_blocks = n // bs
        if n_blocks == 0:
            continue
        for rnd in range(n_rounds):
            if time.time() > deadline:
                return h_cur, best_score
            order = list(range(n_blocks))
            rng.shuffle(order)
            for bi in order:
                if time.time() > deadline:
                    return h_cur, best_score
                s = bi * bs
                e = s + bs
                _g_h_block = h_cur
                _g_bstart = s
                _g_bend = e
                x0 = h_cur[s:e].copy()
                try:
                    res = minimize(_block_objective, x0, method='Nelder-Mead',
                                   options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-16})
                    h_test = h_cur.copy()
                    h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                    h_test = project_feasible(h_test, target)
                    sc = compute_c5(h_test)
                    if sc < best_score:
                        best_score = sc
                        h_cur = h_test
                except Exception:
                    pass
    return h_cur, best_score


def active_set_refine(h_init, target, deadline):
    n = len(h_init)
    h_cur = h_init.copy()
    best_score = compute_c5(h_cur)
    n_minus_1 = n - 1
    for frac in [0.92, 0.88, 0.82, 0.75, 0.65, 0.50]:
        if time.time() > deadline:
            return h_cur, best_score
        for it in range(150):
            if time.time() > deadline:
                return h_cur, best_score
            corr = compute_corr(h_cur)
            c_max_val = float(np.max(corr))
            if c_max_val < 1e-10:
                return h_cur, best_score
            thresh = frac * c_max_val
            active = np.where(corr > thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for j in active:
                s = j - n_minus_1
                if s >= 0:
                    m = n - s
                    grad[s:s + m] += (1.0 - h_cur[:m])
                    grad[:m] -= h_cur[s:s + m]
                else:
                    d = -s
                    m = n - d
                    grad[:m] += (1.0 - h_cur[d:d + m])
                    grad[d:d + m] -= h_cur[:m]
            grad /= float(len(active))
            grad -= np.mean(grad)
            lr = 0.08
            improved = False
            for _ in range(20):
                h_new = project_feasible(h_cur - lr * grad, target)
                sc = compute_c5(h_new)
                if sc < best_score - 1e-15:
                    best_score = sc
                    h_cur = h_new
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                break
    return h_cur, best_score


def _soft_obj(x):
    h = project_feasible(x, _g_target)
    return soft_max_value(h, _g_alpha)


def _cons_eq(x):
    return np.sum(x) - _g_target


def _make_binary(h, target, rng):
    n = len(h)
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h)[::-1]
    h_bin[sorted_idx[:int(target)]] = 1.0
    return h_bin


def binary_sa(h_bin, n, target, max_iters, deadline, seed):
    rng = np.random.RandomState(seed)
    h_cur = h_bin.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    temp = 0.008
    cool = 0.9998
    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_trial = h_cur.copy()
            h_trial[i] = 0.0
            h_trial[j] = 1.0
        else:
            num_ones = int(np.sum(h_cur == 1.0))
            if num_ones < 2:
                continue
            i = ones[rng.randint(len(ones))]
            j = ones[rng.randint(len(ones))]
            k = zeros[rng.randint(len(zeros))]
            l = zeros[rng.randint(len(zeros))]
            if i == j or k == l:
                continue
            h_trial = h_cur.copy()
            h_trial[i] = 0.0
            h_trial[j] = 0.0
            h_trial[k] = 1.0
            h_trial[l] = 1.0
        sc = compute_c5(h_trial)
        if sc < best_score:
            best_score = sc
            best_h = h_trial.copy()
        delta = sc - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            h_cur = h_trial
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
        temp *= cool
        if temp < 1e-9:
            temp = 1e-9
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_target, _g_alpha, _g_h_block, _g_bstart, _g_bend
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    _g_target = target

    def _rem():
        return budget_s - 25.0 - (time.time() - start_time)

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binary simulated annealing from parent
    if _rem() > 250:
        h_bin = _make_binary(best_h, target, rng)
        dl = time.time() + min(120.0, _rem() * 0.35)
        h_sa, sc_sa = binary_sa(h_bin, n, target, 800000, dl, seed)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 2: Random binary SA (different start)
    if _rem() > 200:
        h_rbin = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h_rbin[idx] = 1.0
        dl = time.time() + min(80.0, _rem() * 0.3)
        h_sa2, sc_sa2 = binary_sa(h_rbin, n, target, 400000, dl, seed + 1)
        if sc_sa2 < best_score:
            best_score = sc_sa2
            best_h = h_sa2

    # Phase 3: Nesterov gradient descent (main optimization)
    if _rem() > 250:
        dl = time.time() + min(130.0, _rem() * 0.4)
        h_new, sc_new = nesterov_descent(best_h, target,
                                         [(50, 80), (300, 100), (1500, 100), (8000, 80)],
                                         0.004, 0.85, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 4: SLSQP with soft-max objective
    if _rem() > 180:
        for a in [500, 3000, 15000, 60000]:
            if _rem() < 25:
                break
            _g_alpha = a
            try:
                res = minimize(_soft_obj, best_h, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq', 'fun': _cons_eq}],
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Phase 5: Active set refinement
    if _rem() > 120:
        dl = time.time() + min(70.0, _rem() * 0.4)
        h_as, sc_as = active_set_refine(best_h, target, dl)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    # Phase 6: Nesterov with high alpha
    if _rem() > 120:
        dl = time.time() + min(70.0, _rem() * 0.45)
        h_new, sc_new = nesterov_descent(best_h, target,
                                         [(3000, 120), (15000, 100), (80000, 80)],
                                         0.0015, 0.8, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Block coordinate descent
    if _rem() > 120:
        dl = time.time() + min(80.0, _rem() * 0.5)
        h_bd, sc_bd = block_descent(best_h, target, [4, 6, 3], 2, dl, rng)
        if sc_bd < best_score:
            best_score = sc_bd
            best_h = h_bd

    # Phase 8: Perturbation + re-optimization (escape local optima)
    if _rem() > 150:
        for trial in range(4):
            if _rem() < 40:
                break
            scale = 0.03 + 0.02 * trial
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = project_feasible(h_pert, target)
            dl = time.time() + min(35.0, _rem() * 0.25)
            h_new, sc_new = nesterov_descent(h_pert, target,
                                             [(2000, 120), (10000, 80), (50000, 60)],
                                             0.002, 0.82, dl)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 9: Second SLSQP pass
    if _rem() > 60:
        for a in [30000, 150000]:
            if _rem() < 15:
                break
            _g_alpha = a
            try:
                res = minimize(_soft_obj, best_h, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq', 'fun': _cons_eq}],
                               options={'maxiter': 400, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Phase 10: Final active set + block polish
    if _rem() > 30:
        dl = time.time() + min(20.0, _rem() * 0.5)
        h_as, sc_as = active_set_refine(best_h, target, dl)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    if _rem() > 20:
        dl = time.time() + min(15.0, _rem() * 0.6)
        h_bd, sc_bd = block_descent(best_h, target, [3, 4], 1, dl, rng)
        if sc_bd < best_score:
            best_score = sc_bd
            best_h = h_bd

    # Phase 11: Final Nesterov polish
    if _rem() > 15:
        dl = time.time() + min(10.0, _rem() * 0.7)
        h_new, sc_new = nesterov_descent(best_h, target,
                                         [(100000, 100)],
                                         0.0005, 0.7, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
