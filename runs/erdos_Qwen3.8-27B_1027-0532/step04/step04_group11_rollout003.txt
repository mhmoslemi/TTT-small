        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_g_target = 0
_g_alpha = 0

    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _con_eq(x):
    return np.sum(x) - _g_target

def _con_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, n, target, alpha, maxiter=3000):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:
        return h.copy()

def slsqp_schedule(h, target, alphas, deadline):

    # Phase 6: Full SLSQP with very high alpha for global polish
    deadline6 = t0 + 1020
    for alpha in [50000, 100000]:
        if time.time() > deadline6:
            break
        h_new = slsqp_full(best_h, n, target, alpha, maxiter=2000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if elapsed() > budget_s - 60:
        return best_h, best_score, n

    # Phase 7: Targeted subgradient on worst correlation
    deadline7 = t0 + 1080
    h_sg, sc_sg = targeted_subgradient(best_h, n, target, 500, 0.02, deadline7)
    if sc_sg < best_score:
        best_score = sc_sg
        best_h = h_sg

    # Snapping heuristic
    h_snap = snap_binary(best_h, target, 0.03)
    sc_snap = compute_c5(h_snap)
    if sc_snap < best_score:
        best_score = sc_snap
        best_h = h_snap

    if elapsed() > budget_s - 50:
        return best_h, best_score, n

    # Phase 8: Random perturbations + quick polish
    deadline8 = t0 + budget_s - 60
    for attempt in range(20):
        if time.time() > deadline8:
            break
        h_pert = best_h.copy()
        ns = np.random.randint(10, 50)
        si = np.random.randint(0, n - ns)
        h_pert[si:si + ns] += np.random.normal(0, 0.05, ns)
        h_pert = project_feasible(h_pert, target)
        h_ref = slsqp_full(h_pert, n, target, 20000, maxiter=1000)
        sc_ref = compute_c5(h_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 9: Final coordinate descent
    deadline9 = t0 + budget_s - 30
    h_c, sc_c = coord_descent_fast(best_h, n, target, 50000.0, 3, deadline9)
    if sc_c < best_score:
        best_score = sc_c
        best_h = h_c

    # Final block polish
    h_b, sc_b = block_descent(best_h, n, target, 2, 50000.0, 3, t0 + budget_s - 10, 'sequential')
    if sc_b < best_score:
        best_score = sc_b
        best_h = h_b

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score
    return best_h, best_score, n
