_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_obj(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def binary_sa(h_bin, deadline, rng, n_iter, T_start, cooling):
    h_cur = h_bin.copy()
    best_h = h_cur.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr))
    score = best_score
    T = T_start

    one_pos = np.where(h_cur == 1.0)[0].tolist()
    zero_pos = np.where(h_cur == 0.0)[0].tolist()
    n_one = len(one_pos)
    n_zero = len(zero_pos)

    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooling

        ii = rng.randint(n_one)
        jj = rng.randint(n_zero)
        i = one_pos[ii]
        j = zero_pos[jj]

        h_cur[i] = 0.0
        h_cur[j] = 1.0

        corr = compute_corr(h_cur)
        new_score = float(np.max(corr))

            score = new_score
            one_pos[ii] = j
            zero_pos[jj] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0

    return best_h, best_score

        order = np.argsort(-h[zero_idx])[:diff]
        h_bin[zero_idx[order]] = 1.0
    elif diff < 0:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])[:(-diff)]
        h_bin[one_idx[order]] = 0.0
    return h_bin


def slsqp_polish(h0, target, alpha, maxiter):
    global _g_target, _g_alpha
    _g_target = target
    _g_alpha = alpha
    n = len(h0)
    try:

                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h0
