import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_all_lags(h):
    """Return array of C_m for all lags (length 2n-1)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_gradients_vectorized(h):
    """Compute gradients of C_m w.r.t. h for all lags. Returns (2n-1, n) array."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    
    for j in range(2*n-1):
        m = j - (n-1)
        if m == 0:
            grads[j] = dx * (1.0 - 2.0 * h)
        elif m > 0:
            g = np.zeros(n)
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
            grads[j] = dx * g
        else:
            k = -m
            g = np.zeros(n)
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
            grads[j] = dx * g
    
    return grads

def lp_iteration(h, n, target):
    """One LP iteration: solve LP and return new h (before damping)."""
    dx = 2.0 / n
    C_all = compute_c5_all_lags(h)  # shape (2n-1,)
    grads = compute_gradients_vectorized(h)  # shape (2n-1, n)
    
    # Build LP:
    # min t
    # s.t. grad[j] @ h_new - t <= -C[j] + grad[j] @ h for all j
    #      sum(h_new) = target
    #      0 <= h_new[i] <= 1
    #      t >= 0 (or some lower bound)
    
    # Variables: [h_0, h_1, ..., h_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality: grad[j] @ h - t <= -C[j] + grad[j] @ h
    # A_ub row: [grad[j][0], ..., grad[j][n-1], -1]
    # b_ub[j]: -C[j] + grad[j] @ h
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -C_all + grads @ h
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    else:
        return h.copy()

def lp_optimize(h, n, target, max_iter=200, deadline=None, tol=1e-8):
    """Run LP iterations with adaptive damping."""
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur
    no_improve = 0
    alpha = 0.5

            break
        
        h_lp = lp_iteration(h_cur, n, target)
        h_lp = project_feasible(h_lp, target)

            if best_step_sc is None or sc < best_step_sc:

                best_step_sc = sc

            cur = best_step_sc
            no_improve = 0

            if no_improve > 30:

        if cur < best_sc:
            best_sc = cur
            best_h = h_cur.copy()
    
    return best_h, best_sc

def upscale(h, old_n, new_n):
    """Upscale h from old_n points to new_n points using linear interpolation."""
    x_old = np.linspace(0, 2, old_n)
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n // 2)

def slsqp_obj(x, n, target, alpha):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    # Add penalty for constraint violation
    penalty = 1000.0 * (np.sum(h) - target)**2
    return val * (2.0/n) + penalty

def slsqp_polish(h, n, target, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    best_h = h.copy()
    best_sc = compute_c5(best_h)
    for alpha in [50, 200, 1000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj, best_h, args=(n, target, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_sc

        if sc2 < best_score:

            best_h = h2

    # Final feasibility check

    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score
    else:
        best_h = best_h  # keep the one with best score
    
    return best_h, best_score, len(best_h)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_grads(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        m = j - (n - 1)
        if m == 0:
            grads[j] = dx * (1.0 - 2.0 * h)
        elif m > 0:
            g = np.zeros(n)
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
            grads[j] = dx * g
        else:
            k = -m
            g = np.zeros(n)
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
            grads[j] = dx * g
    return grads

def solve_lp(h, n, target):
    C_all = compute_c5_all(h)
    grads = compute_grads(h)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -C_all + grads @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return h.copy()

def lp_optimize(h, n, target, max_iter=300, deadline=None):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur
    no_improve = 0
    alpha = 0.5
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        h_lp = solve_lp(h_cur, n, target)
        h_lp = project_feasible(h_lp, target)
        best_step = h_cur.copy()
        best_step_sc = cur
        steps = [alpha, max(alpha * 0.5, 0.1), max(alpha * 0.25, 0.05), 1.0]
        for a in steps:
            h_try = (1.0 - a) * h_cur + a * h_lp
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_step_sc:
                best_step_sc = sc
                best_step = h_try
        if best_step_sc < cur - 1e-12:
            h_cur = best_step
            cur = best_step_sc
            no_improve = 0
            alpha = min(alpha * 1.05, 1.0)
        else:
            no_improve += 1
            alpha = max(alpha * 0.7, 0.05)
            if no_improve > 40:
                break
        if cur < best_sc:
            best_sc = cur
            best_h = h_cur.copy()
    return best_h, best_sc

def upscale(h, old_n, new_n):
    x_old = np.linspace(0, 2, old_n)
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n // 2)

def slsqp_obj_wrapped(x, n, target, alpha):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    penalty = 500.0 * (np.sum(h) - target) ** 2
    return val * (2.0 / n) + penalty

def slsqp_eq(x, n, target):
    return np.sum(x) - target

def slsqp_polish(h, n, target, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'args': (n, target)}]
    best_h = h.copy()
    best_sc = compute_c5(best_h)
    for alpha in [50, 200, 1000, 3000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj_wrapped, best_h, args=(n, target, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_sc

def coordinate_descent(h, n, target, n_sweeps=20, deadline=None):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            orig = h_cur[i]
            candidates = [0.0, 1.0, max(0.0, orig - 0.1), min(1.0, orig + 0.1)]
            best_val = None
            best_candidate = None
            for cv in candidates:
                if abs(cv - orig) < 1e-10:
                    continue
                h_try = h_cur.copy()
                h_try[i] = cv
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if best_candidate is None or sc < best_val:
                    best_val = sc
                    best_candidate = h_try
            if best_candidate is not None and best_val < cur - 1e-13:
                h_cur = best_candidate
                cur = best_val
                improved = True
        if cur < best_sc:
            best_sc = cur
            best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_sc

def random_perturb_refine(h, n, target, rng, n_attempts=20, deadline=None):
    best_h = h.copy()
    best_sc = compute_c5(best_h)
    for _ in range(n_attempts):
        if deadline is not None and time.time() > deadline:
            break
        h_p = best_h.copy()
        k = rng.randint(10, 60)
        s = rng.randint(0, n - k)
        h_p[s:s+k] += rng.normal(0, 0.1, k)
        h_p = project_feasible(h_p, target)
        h_lp, sc_lp = lp_optimize(h_p, n, target, max_iter=30, deadline=min(deadline, time.time() + 20) if deadline else None)
        if sc_lp < best_sc:
            best_sc = sc_lp
            best_h = h_lp
    return best_h, best_sc
