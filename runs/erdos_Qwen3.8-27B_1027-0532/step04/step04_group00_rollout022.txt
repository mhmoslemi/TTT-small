def project_feasible(h, target_sum):

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def obj_slsqp(x, alpha):
    return lse_value(x, alpha)


def grad_slsqp(x, alpha):
    return lse_grad(x, alpha)


def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_slsqp, h, method='SLSQP', jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h


def trust_constr_round(h, n, target, alpha, maxiter):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_slsqp, h, method='trust-constr', jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'xtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 8
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.02, 0.015)
        else:
            eps *= 0.85
            if eps < 1e-9:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_lag_minimize(h, n, target, m_active, alpha, maxiter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for outer in range(15):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:m_active]
        active_max = corr[top_m[0]]
        def obj_ml(x):
            c = compute_corr(x)
            c_active = c[top_m]
            am = float(np.max(c_active))
            shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
            return am + float(np.log(np.sum(np.exp(shifted)))) / alpha + 1e-6 * float(np.max(c))
        def objg_ml(x):
            c = compute_corr(x)
            c_active = c[top_m]
            am = float(np.max(c_active))
            shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
            w = np.exp(shifted)
            w /= w.sum()
            dx = 2.0 / n
            g = np.zeros(n)
            for wk, m in zip(w, top_m):
                m = int(m)
                k = m - (n - 1)
                if k >= 0:
                    g[:n - k] += wk * (1.0 - x[k:])
                    g[k:] -= wk * x[:n - k]
                else:
                    ell = -k
                    g[ell:] += wk * (1.0 - x[:n - ell])
                    g[:n - ell] -= wk * x[ell:]
            g *= dx
            return g
        bounds = [(0.0, 1.0)] * n
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_ml, h_cur, method='SLSQP', jac=objg_ml,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_candidate = project_feasible(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < best_score:
                best_score = sc
                h_cur = h_candidate
        except Exception:
            break
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 12.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuation (fast, handles bounds natively)
    alphas_p1 = [15, 40, 100, 300, 800, 2000, 5000, 12000]
    for alpha in alphas_p1:
        if time.time() > deadline - 300:
            break
        for mu in [5.0, 50.0, 200.0]:
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 200)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP with analytical gradient
    alphas_p2 = [100, 500, 2000, 8000, 20000]
    for alpha in alphas_p2:
        if time.time() > deadline - 200:
            break
        h_new = slsqp_round(best_h, n, target, alpha, 3000, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Multi-lag focused optimization
    if time.time() < deadline - 150:
        h_ml, sc_ml = multi_lag_minimize(best_h, n, target, 6, 1000, 1000, deadline - 150)
        if sc_ml < best_score:
            best_score = sc_ml
            best_h = h_ml

    # Phase 4: Subgradient polish with adaptive temperature
    if time.time() < deadline - 120:
        temps = [0.01, 0.005, 0.002, 0.001]
        for temp in temps:
            if time.time() > deadline - 100:
                break
            h_sg, sc_sg = subgradient_polish(best_h, n, target, 1500, temp, 0.004, 0.997, deadline - 100)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 5: Worst-lag surgery
    if time.time() < deadline - 80:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 400, 0.003, deadline - 80)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 6: SLSQP refinement after surgery
    if time.time() < deadline - 60:
        for alpha in [5000, 20000, 50000]:
            if time.time() > deadline - 50:
                break
            h_new = slsqp_round(best_h, n, target, alpha, 2000, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Try higher resolution n=600
    if time.time() < deadline - 250:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [30, 100, 500, 2000, 8000]:
            if time.time() > deadline - 100:
                break
            for mu in [10.0, 100.0]:
                h2_new = lbfgsb_round(best_h2, n2, target2, alpha, mu, 150)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 80:
            h2_new = slsqp_round(best_h2, n2, target2, 3000, 1500, 1e-15)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 60:
            h_sg2, sc_sg2 = subgradient_polish(best_h2, n2, target2, 1000, 0.003, 0.003, 0.997, deadline - 60)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2
        if time.time() < deadline - 40:
            h_surg2, sc_surg2 = worst_lag_surgery(best_h2, n2, target2, 200, 0.002, deadline - 40)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 8: Random perturbations + re-optimization
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(4):
            if time.time() > deadline - 40:
                break
            scales = [0.08, 0.12, 0.06, 0.15]
            h_pert = best_h + rng.randn(n_cur) * scales[attempt]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = lbfgsb_round(h_pert, n_cur, target_cur, 500, 50.0, 200)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_ref2 = slsqp_round(best_h, n_cur, target_cur, 5000, 1500, 1e-15)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    # Phase 9: Final subgradient + surgery
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_polish(best_h, n_cur, target_cur, 800, 0.002, 0.002, 0.998, deadline - 15)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if time.time() < deadline - 10:
            h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 200, 0.001, deadline - 10)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 10: Final SLSQP polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_round(best_h, n_cur, target_cur, 10000, 1000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
