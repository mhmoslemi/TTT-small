def binarize_with_sum(h, target_ones):
    """Convert h to binary, choosing the top target_ones values to be 1."""
    h_bin = np.zeros(len(h))
    # Pick the top target_ones positions
    idx = np.argsort(h)[::-1][:target_ones]
    h_bin[idx] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_iters, deadline, seed):
    """Local search on binary vector using swaps."""
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(np.max(corr))
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    iters = 0
    no_improve = 0
    while iters < max_iters and time.time() < deadline:
        # Pick a random one and zero to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))

            best_h = h.copy()
            corr = new_corr

            zeros = np.where(h == 0.0)[0]

            h[j] = 0.0

def continuous_refine(h, n, deadline, alpha=5000.0):
    """Refine continuous values using SLSQP with soft-max."""
    target = n / 2.0
    
    def _obj(x):
        hx = np.clip(x, 0.0, 1.0)
        hx = project_feasible(hx, target)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def _cons(x):
        return np.sum(x) - target
    
    h0 = project_feasible(h, target)
    try:
        res = minimize(_obj, h0, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _cons}],
                      options={'maxiter': 500, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h0

                #   dc/dh[i] += (1-h[i+l])
                # If i = p (valid if l<=i<=n-1):
                #   dc/dh[i] += -h[i-l]
                grad[:n-l] += (1.0 - h_cur[l:])
                grad[l:] -= h_cur[:n-l]
        
        grad /= top_k
        
        # Gradient step with projection
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
    
    return best_h

def block_swap_search(h_bin, n, deadline, seed):
    """Try swapping blocks of consecutive 1s and 0s."""
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(np.max(corr))
    best_h = h.copy()
    
    for block_len in [2, 3, 4, 5]:
        for trial in range(200):
            if time.time() > deadline:
                return best_h
            # Find a block of 1s and a block of 0s
            ones_mask = (h == 1.0)
            zeros_mask = (h == 0.0)
            
            # Find consecutive runs
            one_runs = []
            zero_runs = []
            
            i = 0
            while i < n:
                if ones_mask[i]:
                    j = i
                    while j < n and ones_mask[j]:
                        j += 1
                    if j - i >= block_len:
                        one_runs.append(i)
                    i = j
                elif zeros_mask[i]:
                    j = i
                    while j < n and zeros_mask[j]:
                        j += 1
                    if j - i >= block_len:
                        zero_runs.append(i)
                    i = j
                else:
                    i += 1
            
            if len(one_runs) == 0 or len(zero_runs) == 0:
                break
            
            i_start = one_runs[rng.randint(len(one_runs))]
            j_start = zero_runs[rng.randint(len(zero_runs))]
            
            # Swap blocks
            h[i_start:i_start+block_len] = 0.0
            h[j_start:j_start+block_len] = 1.0
            
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr))
            
            if new_max <= best_max:
                best_max = new_max
                best_h = h.copy()
                corr = new_corr
            else:
                # Revert
                h[i_start:i_start+block_len] = 1.0
                h[j_start:j_start+block_len] = 0.0
    
    return best_h

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize_by_rank(h, target_ones):
    h_bin = np.zeros(len(h))
    idx = np.argsort(h)[::-1][:target_ones]
    h_bin[idx] = 1.0
    return h_bin


def compute_max_corr(corr):
    return float(np.max(corr))


def sa_binary_search(h_bin, n, deadline, seed, t_init=5.0, t_final=0.01, max_steps=50000):
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_val = compute_max_corr(corr)
    best_val = cur_val
    best_h = h.copy()

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    n_steps = max_steps
    log_t_init = np.log(t_init)
    log_t_final = np.log(t_final)
    step_log = (log_t_final - log_t_init) / max(n_steps, 1)

    for step in range(n_steps):
        if time.time() > deadline:
            break
        t = np.exp(log_t_init + step * step_log)

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

def continuous_optimize(h, n, deadline, alphas=None):
    target = n / 2.0
    h_cur = project_feasible(h.copy(), target)
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

    if alphas is None:
        alphas = [500, 2000, 8000, 30000, 100000]

    for alpha in alphas:
        if time.time() > deadline - 10:
            break

        def _obj(x):
            hx = np.clip(x, 0.0, 1.0)
            hx = project_feasible(hx, target)
            corr = np.correlate(hx, 1.0 - hx, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        def _cons(x):
            return np.sum(x) - target

        try:
            res = minimize(_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _cons}],
                          options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    return best_h, best_score

    target = n / 2.0

    v = np.zeros(n)
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init

    for step in range(n_steps):

            break

        corr = np.correlate(h_look, 1.0 - h_look, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        wt = np.exp(shifted)
        wsum = wt.sum()
        if wsum < 1e-300:
            break
        wt = wt / wsum

        grad = np.zeros(n)
        for s in range(-(n - 1), n):
            w = wt[s + n - 1]
            if w < 1e-12:
                continue
            if s >= 0:
                grad[s:] += w * (1.0 - h_look[:n - s])
                grad[:n - s] -= w * h_look[s:]
            else:
                grad[:n + s] += w * (1.0 - h_look[-s:])
                grad[-s:] -= w * h_look[:n + s]

        v = momentum * v - lr * grad
        h_new = project_feasible(h_cur + v, target)

            best_score = sc
            best_h = h_new.copy()

        else:
            lr *= 0.6
            if lr < 1e-10:
                lr = lr_init
                v *= 0.5

    return best_h, best_score
