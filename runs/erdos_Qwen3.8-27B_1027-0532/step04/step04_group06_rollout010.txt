def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_delta_corr(h, a, b, n):
    """Compute the change in full correlation when flipping h[a]:1->0 and h[b]:0->1.
    Returns a vector of length 2n-1 with the changes at each position."""
    # The full correlation has 2n-1 entries indexed 0 to 2n-2
    # Lag ℓ = k - (n-1) where k is the index in the full array
    # ΔC(ℓ) = -(1-h[a-ℓ])*I(valid) + h[a+ℓ]*I(valid) + (1-h[b-ℓ])*I(valid) - h[b+ℓ]*I(valid)

    # a-ℓ = a - k + (n-1), valid if 0 <= a-k+(n-1) < n
    # a+ℓ = a + k - (n-1), valid if 0 <= a+k-(n-1) < n
    # b-ℓ = b - k + (n-1), valid if 0 <= b-k+(n-1) < n
    # b+ℓ = b + k - (n-1), valid if 0 <= b+k-(n-1) < n

    a_plus = a + ell   # a + k - (n-1)

    b_plus = b + ell

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - target

    def cons_j(h_in):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)

    phase1_time = budget_s * 0.55
    h_p1, c5_p1 = multi_restart_sa(h_bin_start, target_init, n_init,
                                   time_limit=phase1_time, start_time=start_time, rng=rng)
    if c5_p1 < best_score:
        best_score = c5_p1
        best_h = h_p1.copy()

    if tleft() < 50:
        pass
    else:
        n_high = n_init + 300
        if n_high % 2 != 0:
            n_high += 1
        target_high = n_high / 2.0
        h_interp = interpolate_to(best_h, n_high, target_high)
        h_bin_high = project_binary(h_interp, target_high)
        time_high = min(budget_s * 0.2, tleft() - 100)
        if time_high > 30:
            h_hi, c5_hi = multi_restart_sa(h_bin_high, target_high, n_high,
                                           time_limit=time_high, start_time=start_time, rng=rng)
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_hi.copy()

    if tleft() < 40:
        pass
    else:
        n_med = n_init - 100
        if n_med < 200:
            n_med = 200
        if n_med % 2 != 0:
            n_med += 1
        if abs(n_med - n_init) > 30:
            target_med = n_med / 2.0
            h_interp_med = interpolate_to(best_h, n_med, target_med)
            h_bin_med = project_binary(h_interp_med, target_med)
            time_med = min(budget_s * 0.12, tleft() - 50)
            if time_med > 20:
                h_md, c5_md = multi_restart_sa(h_bin_med, target_med, n_med,
                                               time_limit=time_med, start_time=start_time, rng=rng)
                if c5_md < best_score:
                    best_score = c5_md
                    best_h = h_md.copy()

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas = [50, 150, 500, 1500, 4000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 alphas, 300, 1e-14,
                                 time_limit=max(30.0, tleft() - 30), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        time_final = max(5.0, tleft() - 10)
        h_f, c5_f = fast_binary_sa(h_bin_final, target_use, n_use,
                                   time_limit=time_final, start_time=start_time, rng=rng,
                                   t_init=0.003, t_final=0.0001, max_no_improve=5000)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas_fine = [3000, 8000, 20000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use,
                                   alphas_fine, 200, 1e-15,
                                   time_limit=max(5.0, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
