```python
def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted) / np.exp(shifted).sum()
    
    # grad[i] = sum_k weights[k] * d(corr[k])/dh[i]
    # corr[k] = sum_j h[j+k-(n-1)] * (1-h[j]) for valid j
    # d(corr[k])/dh[i] = (1-h[i+s]) * [0<=i-s<n] - h[i+s] * [0<=i+s<n] where s=k-(n-1)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        s = k - (n-1)
        w = weights[k]
        if w < 1e-15:
            continue
        # Term 1: i appears as h[i] where i = j+s, so j = i-s
        # Need 0 <= i < n and 0 <= j = i-s < n
        # Contribution: w * (1 - h[j]) = w * (1 - h[i-s])
        # Term 2: i appears as h[i] where i = j, so j = i
        # Need 0 <= j = i < n and 0 <= j+s = i+s < n
        # Contribution: -w * h[i+s]
        
        # For term 1: valid i range is max(0, s) to min(n-1, n-1+s)
        lo1 = max(0, s)
        hi1 = min(n, n+s)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-s:hi1-s])
        
        # For term 2: valid i range is max(0, -s) to min(n, n-s)
        lo2 = max(0, -s)
        hi2 = min(n, n-s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2+s:hi2+s]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Iteratively adjust
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        # Distribute proportionally to room
        if diff > 0:
            room = 1.0 - h[mask]
        else:
            room = h[mask]
        total_room = room.sum()
        if total_room < 1e-14:
            break
        adj = diff * room / total_room
        h[mask] += adj
        diff = target_sum - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.zeros(n)
    # Sort by value descending, take top target_sum
    idx = np.argsort(-h)
    h_r[idx[:target_sum]] = 1.0
    return h_r

def binary_greedy(h, target_sum, time_limit, start_time, rng, max_iters=50000):
    """Fast binary greedy: always swap to reduce the max correlation."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 500:
            break
            
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        
        # Find beneficial swap: i with h[i]=1 and h[i+s]=0 (reduces corr at shift s)
        # and j with h[j]=0 and h[j+s]=1 (also reduces)
        # We want to set h[i]=0 and h[j]=1
        # This reduces corr at shift s by: (1-h[i+s]) + h[j+s] = 1 + 1 = 2
        # But it may increase corr at other shifts
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find i in ones where h[i+s] == 0 (contributes to positive part of corr at shift s)
        i_cand = []
        for i in ones:
            if 0 <= i + s < n and h[i + s] == 0:
                i_cand.append(i)
        
        # Find j in zeros where h[j+s] == 1 (contributes to the (1-h) part)
        j_cand = []
        for j in zeros:
            if 0 <= j + s < n and h[j + s] == 1:
                j_cand.append(j)
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            # Random move
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            if new_corr.max() < max_val:
                corr = new_corr
                stall = 0
                if new_corr.max() * (2.0/n) < best_c5:
                    best_c5 = float(new_corr.max()) * (2.0/n)
                    best_h = h.copy()
            else:
                h[j] = 0
                h[i] = 1
            continue
        
        # Try a swap
        improved = False
        n_try = min(50, len(i_cand) * len(j_cand))
        for _ in range(n_try):
            i = i_cand[rng.randint(len(i_cand))]
            j = j_cand[rng.randint(len(j_cand))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(new_corr.max())
            if new_max < max_val:
                corr = new_corr
                stall = 0
                if new_max * (2.0/n) < best_c5:
                    best_c5 = new_max * (2.0/n)
                    best_h = h.copy()
                improved = True
                break
            else:
                h[j] = 0
                h[i] = 1
        
        if not improved:
            stall += 1
    
    return best_h, best_c5
```

```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        if diff > 0:
            room = 1.0 - h[mask]
        else:
            room = h[mask]
        total_room = room.sum()
        if total_room < 1e-14:
            break
        h[mask] += diff * room / total_room
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def project_binary(h, target_sum):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target_sum]] = 1.0
    return h_bin

    n = len(h)
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(corr.max())
    best_h = h.copy()
    stall = 0
    temp = 0.1
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 2000:
            break
            
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Strategy 1: Focused swap on max shift
        if s > 0:
            i_cand = [i for i in ones if i + s < n and h[i + s] == 0]
            j_cand = [j for j in zeros if j + s < n and h[j + s] == 1]
        elif s < 0:
            i_cand = [i for i in ones if i + s >= 0 and h[i + s] == 0]
            j_cand = [j for j in zeros if j + s >= 0 and h[j + s] == 1]
        else:
            i_cand = list(ones)
            j_cand = list(zeros)
        
        improved = False
        
        if len(i_cand) > 0 and len(j_cand) > 0:
            n_try = min(100, len(i_cand) * len(j_cand))
            for _ in range(n_try):
                i = i_cand[rng.randint(len(i_cand))]
                j = j_cand[rng.randint(len(j_cand))]
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = float(new_corr.max())
                if new_max < max_val:
                    corr = new_corr
                    stall = 0
                    if new_max < best_max:
                        best_max = new_max
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    h[j] = 0
                    h[i] = 1
        
        if not improved:
            # Strategy 2: Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                stall += 1
                continue
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(new_corr.max())
            if new_max <= max_val + temp:
                corr = new_corr
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    stall = 0
            else:
                h[j] = 0
                h[i] = 1
            stall += 1
        
        # Cooling
        temp *= 0.999
    
    return best_h, best_max * (2.0 / n)
```
