def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(80):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def coord_descent(h_init, target_sum, n_sweeps, topk_cross, rng, time_limit=None, t0=None):
    N = len(h_init)
    h = h_init.astype(float).copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * (2.0 / N)
    flip_h = None

    for sweep in range(n_sweeps):
        if t0 is not None and time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        flip_h = h[::-1].copy()
        order = rng.permutation(N)
        proj_counter = 0
        for i in order:
            grad = np.zeros(2 * N - 1)
            grad[i:i + N] += (1.0 - flip_h)
            grad[N - 1 - i:2 * N - 1 - i] -= h

            d_lo = -h[i]
            d_hi = 1.0 - h[i]
            if d_hi - d_lo < 1e-14:

            candidates = [d_lo, d_hi, 0.0]
            top_idx = np.argpartition(corr, -topk_cross)[-topk_cross:]
            for a in range(min(len(top_idx), 12)):

                ca = corr[ka]

                if val < best_f - 1e-15:

                    best_d = d

        if lag >= 0:
            if lag < n:
                g[lag:] += (1.0 - h_cur[:n - lag])
                g[:n - lag] -= h_cur[lag:]
        else:
            mm = -lag
            if mm < n:
                g[:n - mm] += (1.0 - h_cur[mm:])
                g[mm:] -= h_cur[:n - mm]
        A[ii, :n] = g
        A[ii, n] = -1.0
        b[ii] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None


def lp_refinement(h_init, target_sum, max_iterations, topk, time_limit=None, t0=None):
    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    if t0 is None:
        t0 = time.time()
    stagnation = 0
    for it in range(max_iterations):
        if t0 is not None and time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        h_lp = lp_linearized_step(h, target_sum, topk)
        if h_lp is None:
            break
        score = compute_c5(h_lp)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stagnation = 0
        else:
            h_half = project_feasible(0.5 * h + 0.5 * h_lp, target_sum)
            sc_half = compute_c5(h_half)
            if sc_half < best_score - 1e-14:
                best_score = sc_half
                best_h = h_half.copy()
                h = h_half
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 15:
                    break
    return best_h, best_score


def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    target = n_dst / 2.0
    return project_feasible(h_out, target)


def subgradient_refine(h_init, target_sum, max_iter, lr_init, topk, alpha,
                       patience, momentum, rng):
    n = len(h_init)
    h = h_init.astype(float).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_vals = corr[top_indices]
        w = np.exp(alpha * (top_vals - np.max(top_vals)))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for ii in range(len(top_indices)):
            if w[ii] < 1e-14:
                continue
            idx = top_indices[ii]
            lag = idx - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[lag:] += w[ii] * (1.0 - h[:n - lag])
                    grad[:n - lag] -= w[ii] * h[lag:]
            else:
                mm = -lag
                if mm < n:
                    grad[:n - mm] += w[ii] * (1.0 - h[mm:])
                    grad[mm:] -= w[ii] * h[:n - mm]
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.005, 2.0)
        else:
            stagnation += 1
            lr *= 0.995
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    if _remaining() < 50:
        return best_h, best_score, n_init

    # Phase 1: Aggressive coordinate descent
    h_new, score = coord_descent(best_h, target_init, n_sweeps=3000,
                                  topk_cross=20, rng=rng,
                                  time_limit=_remaining() - 100, t0=start_time)
    if score < best_score:
        best_score = score
        best_h = h_new

    if _remaining() < 100:
        return best_h, best_score, len(best_h)

    # Phase 2: LP refinement
    h_new, score = lp_refinement(best_h, target_init, max_iterations=300,
                                  topk=50, time_limit=_remaining() - 200, t0=start_time)
    if score < best_score:
        best_score = score
        best_h = h_new

    # Phase 3: More coordinate descent
    if _remaining() > 150:
        h_new, score = coord_descent(best_h, target_init, n_sweeps=3000,
                                      topk_cross=15, rng=rng,
                                      time_limit=_remaining() - 120, t0=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 4: Subgradient with perturbation to escape local optima
    if _remaining() > 200:
        h_pert = best_h + rng.randn(n_init) * 0.02
        h_pert = project_feasible(h_pert, target_init)
        h_new, score = subgradient_refine(h_pert, target_init, max_iter=3000,
                                           lr_init=0.05, topk=10, alpha=50,
                                           patience=300, momentum=0.90, rng=rng)
        if score < best_score:
            best_score = score
            best_h = h_new
            if _remaining() > 100:
                h_new2, score2 = coord_descent(best_h, target_init, n_sweeps=2000,
                                                topk_cross=15, rng=rng,
                                                time_limit=_remaining() - 80, t0=start_time)
                if score2 < best_score:
                    best_score = score2
                    best_h = h_new2

    # Phase 5: Try higher resolution
    for n_try in [480, 560]:
        if _remaining() < 200:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to_n(best_h, n_try)
        h_new, score = coord_descent(h_interp, target_try, n_sweeps=2000,
                                      topk_cross=20, rng=rng,
                                      time_limit=_remaining() - 150, t0=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
            if _remaining() > 150:
                h_new2, score2 = lp_refinement(best_h, target_try, max_iterations=200,
                                                topk=40, time_limit=_remaining() - 100, t0=start_time)
                if score2 < best_score:
                    best_score = score2
                    best_h = h_new2
                if _remaining() > 100:
                    h_new3, score3 = coord_descent(best_h, target_try, n_sweeps=3000,
                                                    topk_cross=12, rng=rng,
                                                    time_limit=_remaining() - 80, t0=start_time)
                    if score3 < best_score:
                        best_score = score3
                        best_h = h_new3

    # Phase 6: Final polishing with multiple short coordinate descent passes
    if _remaining() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for s in range(3):
            if _remaining() < 30:
                break
            h_try = best_h + rng.randn(n_cur) * 0.002
            h_try = project_feasible(h_try, target_cur)
            h_new, score = coord_descent(h_try, target_cur, n_sweeps=2000,
                                          topk_cross=10, rng=rng,
                                          time_limit=_remaining() - 25, t0=start_time)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 7: Final LP polish
    if _remaining() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new, score = lp_refinement(best_h, target_cur, max_iterations=100,
                                      topk=30, time_limit=_remaining() - 5, t0=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Final: one last coordinate descent pass
    if _remaining() > 15:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new, score = coord_descent(best_h, target_cur, n_sweeps=500,
                                      topk_cross=8, rng=rng,
                                      time_limit=_remaining() - 3, t0=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
