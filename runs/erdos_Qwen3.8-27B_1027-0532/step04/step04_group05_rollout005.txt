import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    """Compute all correlation values (uncorrected for dx)"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    sum_exp = np.sum(exp_shifted)

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.7, reg=1e-6):

    """L-BFGS-B optimization with adaptive alpha."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n_iter = 0
    alpha = alpha_start
    n_alphas = 0
    max_alphas = int(np.log2(alpha_end / alpha_start)) + 1
    
    for a in range(max_alphas):
        if time.time() - start > time_limit:
            break
        alpha = alpha_start * (2 ** a)
        alpha = min(alpha, alpha_end)
        
        def obj_grad(x):
            x = np.clip(x, 0.0, 1.0)
            obj, grad = logsumexp_corr(x, alpha, mu, target)
            return obj, grad
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_grad, h, method='L-BFGS-B',
                          bounds=bounds, jac=True,
                          options={'maxiter': max_iter // max_alphas,
                                   'ftol': 1e-15, 'gtol': 1e-10,
                                   'maxfun': 50000})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        
        if time.time() - start > time_limit:
            break
    
    return best_h, best_c5

def slsqp_analytic(h_init, n, target, alpha, mu, maxiter):
    """SLSQP with analytical gradient."""
    h = h_init.copy()
    
    def obj(x):
        obj_val, _ = logsumexp_corr(x, alpha, 0.0, target)
        return obj_val
    
    def grad(x):
        _, g = logsumexp_corr(x, alpha, 0.0, target)
        return g
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
