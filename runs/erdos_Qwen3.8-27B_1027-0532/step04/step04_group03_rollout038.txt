def targeted_perturb(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

        # Full correlation has 2n-1 entries, index n-1 is shift 0
        # We want the worst non-zero shifts
        shifts = np.arange(len(corr)) - (n - 1)
        mask = shifts != 0
        corr_m = np.where(mask, corr, -1.0)
        # Get top 5 worst
        top_idx = np.argsort(corr_m)[-5:]
        
        improved = False
        for ti in top_idx:
            k = shifts[ti]
            if k == 0:
                continue
            # For shift k, the contribution is h[i]*(1-h[(i+k)%n])
            # Find positions where h[i] is high and h[(i+k)%n] is low
            for i in range(n):
                j = (i + k) % n
                contrib = h[i] * (1.0 - h[j])
                if contrib > 0.3:  # significant contribution
                    # Try to reduce: decrease h[i] or increase h[j]
                    delta = 0.02
                    h_save = h.copy()
                    if h[i] > 0.5:
                        h[i] = max(0.0, h[i] - delta)
                        h[j] = min(1.0, h[j] + delta)
                        c5_try = compute_c5(h)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best_h = h.copy()
                            improved = True
                        else:
                            h = h_save
                    elif h[j] < 0.5:
                        h[i] = min(1.0, h[i] + delta)
                        h[j] = max(0.0, h[j] - delta)
                        c5_try = compute_c5(h)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best_h = h.copy()
                            improved = True
                        else:
                            h = h_save
        if not improved:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
