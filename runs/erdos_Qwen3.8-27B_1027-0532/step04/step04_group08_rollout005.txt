def compute_deltas(h, n):
    """Compute delta for flipping each position."""
    nc = 2*n - 1
    shifts = np.arange(nc) - (n-1)  # -n+1 to n-1
    delta = np.zeros((n, nc))  # delta[i] = change in corr if h[i] flips
    
    for pos in range(n):
        current_val = h[pos]
        # Part 1: pos is the "h[i]" in the product h[i]*(1-h[i+shift])
        # Valid when 0 <= pos+shift <= n-1
        # shift is in [-pos, n-1-pos]
        # s = shift + n - 1 is in [n-1-pos, 2n-2-pos]
        s_start = n - 1 - pos
        s_end = 2*n - 2 - pos
        s_valid = np.arange(max(0, s_start), min(nc, s_end + 1))
        if len(s_valid) > 0:
            psa = pos + shifts[s_valid]  # pos + shift, should be in [0, n-1]
            if current_val == 1.0:
                delta[pos, s_valid] -= (1.0 - h[psa])
            else:
                delta[pos, s_valid] += (1.0 - h[psa])
        
        # Part 2: pos is the "h[i+shift]" in the product h[i]*(1-h[i+shift])
        # i = pos - shift, valid when 0 <= pos-shift <= n-1
        # shift is in [pos-(n-1), pos]
        # s = shift + n - 1 is in [pos, pos+n-1]
        s_start2 = pos
        s_end2 = pos + n - 1
        s_valid2 = np.arange(max(0, s_start2), min(nc, s_end2 + 1))
        if len(s_valid2) > 0:
            psm = pos - shifts[s_valid2]  # pos - shift, should be in [0, n-1]
            if current_val == 1.0:
                delta[pos, s_valid2] += h[psm]
            else:
                delta[pos, s_valid2] -= h[psm]
    
    return delta

            h_cur[zeros[best_bi]] = 1.0
            corr = compute_corr(h_cur)
            cur_val = float(np.max(corr))
            if cur_val * dx < best_c5:
                best_c5 = cur_val * dx
                best_h = h_cur.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        T *= decay
        if stuck > 200:
            T = max(T, T0 * 0.3)
            stuck = 0
    return best_h, best_c5


def subgradient_refine(h, deadline, max_iter=3000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    step = 0.05
    v = np.zeros(n)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        thresh = c_max * 0.98
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for s_idx in active:
            s = s_idx - (n - 1)
            if s >= 0:
                grad[s:] += (1.0 - h_cur[:n - s])
                grad[:n - s] -= h_cur[s:]
            else:
                grad[:n + s] += (1.0 - h_cur[-s:])
                grad[-s:] -= h_cur[:n + s]
        grad /= float(len(active))
        v = 0.8 * v - step * grad
        h_new = project_feasible(h_cur + v, target)
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
            h_cur = h_new
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.9
        if it % 500 == 499:
            h_rebin = make_binary(h_cur, n)
            c5r = compute_c5(h_rebin)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_rebin.copy()
                h_cur = h_rebin.copy()
    return best_h, best_c5


def _soft_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    _g_target = target

    deadline = start_time + budget_s - 25

    def _remaining():
        return deadline - time.time()

    h_bin0 = make_binary(h_init, n)
    best_score = compute_c5(h_bin0)
    best_h = h_bin0.copy()

    c5_init = compute_c5(project_feasible(h_init, target))
    if c5_init < best_score:
        best_score = c5_init
        best_h = project_feasible(h_init, target).copy()

    # Phase 1: Greedy binary local search
    if _remaining() > 300:
        dl = time.time() + min(200.0, _remaining() * 0.5)
        h_g, c5_g = greedy_local_search(h_bin0, dl)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g

    # Phase 2: Try different binary initializations
    if _remaining() > 200:
        h_poly = np.zeros(n)
        freqs = [1, 2, 3, 5, 7, 11, 13]
        for f in freqs:
            h_poly += np.sin(2.0 * np.pi * f * np.arange(n) * dx / 2.0)
        h_mix = 0.5 * h_bin0 + 0.5 * (h_poly - h_poly.min()) / (h_poly.max() - h_poly.min())
        h_alt = make_binary(h_mix, n)
        if _remaining() > 100:
            dl = time.time() + min(60.0, _remaining() * 0.3)
            h_g2, c5_g2 = greedy_local_search(h_alt, dl)
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2

    # Phase 3: Simulated annealing from best
    if _remaining() > 150:
        best_bin = make_binary(best_h, n)
        dl = time.time() + min(100.0, _remaining() * 0.4)
        h_sa, c5_sa = sa_local_search(best_bin, dl, T0=0.02, T_min=0.002, decay=0.98)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 4: More greedy from SA result
    if _remaining() > 100:
        best_bin = make_binary(best_h, n)
        dl = time.time() + min(60.0, _remaining() * 0.4)
        h_g3, c5_g3 = greedy_local_search(best_bin, dl)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3

    # Phase 5: Subgradient continuous refinement
    if _remaining() > 80:
        dl = time.time() + min(80.0, _remaining() * 0.5)
        h_sub, c5_sub = subgradient_refine(best_h, dl)
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 6: SLSQP soft-max polishing
    if _remaining() > 40:
        for alpha in [500, 2000, 10000, 50000]:
            if _remaining() < 10:
                break
            _g_alpha = alpha
            try:
                res = minimize(_soft_obj, best_h, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq', 'fun': _cons_eq}],
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Final binary local search
    if _remaining() > 20:
        best_bin = make_binary(best_h, n)
        c5_cb = compute_c5(best_bin)
        if c5_cb < best_score:
            best_score = c5_cb
            best_h = best_bin
        dl = time.time() + min(20.0, _remaining() * 0.6)
        h_gf, c5_gf = greedy_local_search(best_h, dl)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gf

    # Phase 8: Final SLSQP
    if _remaining() > 10:
        _g_alpha = 100000
        try:
            res = minimize(_soft_obj, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': _cons_eq}],
                           options={'maxiter': 200, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
