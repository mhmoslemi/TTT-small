def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def corr_and_grad_topk(h, topk):
    """Compute top-k correlation values and their gradients."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(len(corr))
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in idx:
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += one_minus_h[:n-lag]
                grad[:n-lag] -= h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n-m] += one_minus_h[m:]
                grad[m:] -= h[:n-m]
    return corr, idx, grad

Wait, one more idea: **the "shrinkage" approach**. Instead of optimizing h directly, I can:
1. Compute the correlation vector c
2. For the lags that are at or near the maximum, compute the gradient
3. Take a step in the direction that reduces those specific lags
4. Project back to feasible set
5. Repeat

This is essentially a subgradient method for the max function. It might converge slower than LSE but could be more accurate in the end.

The direction to reduce the max is: -∇(max_k c_k) = -∇c_{k*} where k* = argmax c_k.

But this only uses one lag at a time. A better approach: use a weighted sum of the top few lags' gradients.

        top_idx = np.argpartition(corr, -k)[-k:]

            grad += weights[i] * grad_corr_lag(h, lag)
        # Gradient step with projection
        h = h - lr * grad
        h = project_feasible(h, target)

import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def grad_corr_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def lse_obj_grad(h, target, alpha, topk, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(len(corr))
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    s = np.sum(h) - target
    obj += penalty * s * s
    weights = exp_vals / total
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        grad += w * grad_corr_lag(h, lag)
    grad += 2.0 * penalty * s
    return obj, grad

def hinge_obj_grad(h, target, threshold, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    above = corr > threshold
    excess = np.where(above, corr - threshold, 0.0)
    obj = np.sum(excess ** 2)
    s = np.sum(h) - target
    obj += penalty * s * s
    grad = np.zeros(n)
    for i in np.where(above)[0]:
        lam = corr[i] - threshold
        if lam > 0:
            grad += 2.0 * lam * grad_corr_lag(h, i - (n - 1))
    grad += 2.0 * penalty * s
    return obj, grad

def _lse_func(x):
    f, g = lse_obj_grad(x, _opt['target'], _opt['alpha'], _opt['topk'], _opt['penalty'])
    return f, g

def _hinge_func(x):
    f, g = hinge_obj_grad(x, _opt['target'], _opt['threshold'], _opt['penalty'])
    return f, g

def optimize_lse(h0, target, alpha, topk, penalty, maxiter, ftol):
    n = len(h0)
    _opt['target'] = target
    _opt['alpha'] = alpha
    _opt['topk'] = topk
    _opt['penalty'] = penalty
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_func, h0, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

def optimize_hinge(h0, target, threshold, penalty, maxiter, ftol):
    n = len(h0)
    _opt['target'] = target
    _opt['threshold'] = threshold
    _opt['penalty'] = penalty
    bounds = [(0.0, 1.0)] * n

                      bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

def topk_grad_descent(h, target, k, lr, n_iter, alpha_temp):
    n = len(h)
    h_cur = h.copy()
    for it in range(n_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        if k < len(corr):
            top_idx = np.argpartition(corr, -k)[-k:]
        else:
            top_idx = np.arange(len(corr))
        max_c = cpp = np.max(corr[top_idx])
        weights = np.exp(np.clip((corr[top_idx] - max_c) / alpha_temp, -100.0, 0.0))
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights = weights / wsum
        grad = np.zeros(n)
        for j in range(len(top_idx)):
            if weights[j] < 1e-15:
                continue
            lag = top_idx[j] - (n - 1)
            grad += weights[j] * grad_corr_lag(h_cur, lag)
        h_cur = h_cur - lr * grad
        h_cur = project_feasible(h_cur, target)
    return h_cur

def block_coord_descent(h, target, alpha, topk, penalty, block_size, n_sweeps):
    n = len(h)
    h_cur = h.copy()
    for sweep in range(n_sweeps):
        for start in range(0, n - block_size + 1, block_size):
            end = start + block_size
            block = h_cur[start:end].copy()
            for step in range(5):
                full_corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                if topk < len(full_corr):
                    tidx = np.argpartition(full_corr, -topk)[-topk:]
                    ctop = full_corr[tidx]
                else:
                    tidx = np.arange(len(full_corr))
                    ctop = full_corr
                cmax = np.max(ctop)
                shifted = np.clip(alpha * (ctop - cmax), -500.0, 0.0)
                ev = np.exp(shifted)
                wt = ev / np.sum(ev)
                bgrad = np.zeros(block_size)
                for j in range(len(tidx)):
                    if wt[j] < 1e-15:
                        continue
                    lag = tidx[j] - (n - 1)
                    full_g = grad_corr_lag(h_cur, lag)
                    bgrad += wt[j] * full_g[start:end]
                block = np.clip(block - 0.01 * bgrad, 0.0, 1.0)
            h_cur[start:end] = block
        h_cur = project_feasible(h_cur, target)
    return h_cur
