To minimize this over h[j] ∈ [0, 1] (with the sum constraint handled separately), we need to find the minimum of the upper envelope of lines {base_k + h[j] * s_k}.

The minimum of the upper envelope of lines occurs at:
- An intersection of two lines (where base_k + x*s_k = base_l + x*s_l, i.e., x = (base_l - base_k)/(s_k - s_l))
- Or at x = 0 or x = 1

There are O(n) lines, so O(n) intersection points (at most). We can find the minimum in O(n log n) time.

But we also need to handle the sum constraint. If we change h[j] by δ, we need to change h[i] by -δ for some i. This makes the problem two-variable optimization, which is more complex.

For simplicity, let me handle the sum constraint by:
1. Target a specific adjustment: we want sum(h) = n/2
2. When we change h[j], we also change h[i] by the exact amount needed
3. The combined change makes the objective a function of one variable (δ = change in h[j])

The slope of c[k] with respect to δ (change in h[j], with h[i] adjusted by -δ) is:

where s_k is the sensitivity at position j and s'_k is the sensitivity at position i.

So c[k] = base_k + δ * (s_k[j] - s_k[i])

And we want to minimize max_k (base_k + δ * d_k) over δ, where d_k = s_k[j] - s_k[i].

This is the same problem: minimize the upper envelope of lines {base_k + δ * d_k} over the feasible range of δ.

The feasible range of δ:
- h[j] + δ ∈ [0, 1] → δ ∈ [-h[j], 1-h[j]]
- h[i] - δ ∈ [0, 1] → δ ∈ [h[i]-1, h[i]]
- Combined: δ ∈ [max(-h[j], h[i]-1), min(1-h[j], h[i])]

The minimum of the upper envelope is at a breakpoint or boundary. We can find it in O(n log n).

```python
def exact_line_search(h, j, i, n):
    """Find the optimal delta for changing h[j] += delta, h[i] -= delta.
    Returns (best_delta, best_score)."""
    # Compute base_k and slope d_k for each lag k
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Sensitivity of corr[k] to h[j]:
    # sens_j[k] = (1-h[j+s]) * I(0<=j+s<n) - h[j-s] * I(0<=j-s<n)
    # where s = k - n + 1
    sens_j = np.zeros(2*n-1)
    sens_i = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        s = k - n + 1
        # sens_j[k]
        val = 0.0
        if 0 <= j + s < n:
            val += (1.0 - h[j + s])
        if 0 <= j - s < n:
            val -= h[j - s]
        sens_j[k] = val
        # sens_i[k]
        val = 0.0
        if 0 <= i + s < n:
            val += (1.0 - h[i + s])
        if 0 <= i - s < n:
            val -= h[i - s]
        sens_i[k] = val
    
    d = sens_j - sens_i  # slope of corr[k] w.r.t. delta
    
    # base_k = corr[k] - h[j]*sens_j[k] + h[i]*sens_i[k]
    # Wait, that's not right. Let me think again.
    # When h[j] changes by delta and h[i] changes by -delta:
    # new_corr[k] = corr[k] + delta * sens_j[k] - delta * sens_i[k]
    #             = corr[k] + delta * (sens_j[k] - sens_i[k])
    #             = corr[k] + delta * d[k]
    # So base_k = corr[k] and slope = d[k]
    
    base = corr  # current correlation values
    
    # Feasible range for delta
    lo = max(-h[j], h[i] - 1.0)
    hi = min(1.0 - h[j], h[i])
    
    if lo >= hi:
        return 0.0, float(np.max(corr)) * (2.0/n)
    
    # Find minimum of max_k (base[k] + delta * d[k]) for delta in [lo, hi]
    # This is the minimum of the upper envelope of lines.
    
    # Evaluate at boundaries
    best_delta = 0.0
    best_val = float('inf')
    
    for test_d in [lo, hi]:
        val = float(np.max(base + test_d * d))
        if val < best_val:
            best_val = val
            best_delta = test_d
    
    # Check breakpoints
    # Breakpoints are where base[k] + delta*d[k] = base[l] + delta*d[l]
    # delta = (base[l] - base[k]) / (d[k] - d[l])
    # Only consider breakpoints in [lo, hi]
    
    # For efficiency, only consider the top few lags (those near the max)
    top_idx = np.argsort(-base)[:min(50, len(base))]  # top 50 lags
    
    for a in top_idx:
        for b in top_idx:
            if a == b:
                continue
            dd = d[a] - d[b]
            if abs(dd) < 1e-12:
                continue
            delta = (base[b] - base[a]) / dd
            if lo <= delta <= hi:
                val = float(np.max(base + delta * d))
                if val < best_val:
                    best_val = val
                    best_delta = delta
    
    return best_delta, best_val * (2.0/n)
```

import numpy as np
from scipy.optimize import minimize
import time

_G_n = 0
_G_target = 0
_G_alpha = 100.0

def _con_eq(x):
    return np.sum(x) - _G_target

def _con_jac(x):
    return np.ones(_G_n)

def _lse_objective(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_G_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _G_alpha

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_round(h, n, target, alphas, maxiter=3000, ftol=1e-15):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    h_cur = project_feasible(h.copy(), target)
    cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    for a in alphas:
        _G_alpha = a
        try:
            res = minimize(_lse_objective, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

def periodic_h(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        h[:total - n // 2] = 0.0
    elif total < n // 2:
        deficit = n // 2 - total
        indices = np.where(h == 0)[0]
        for j in range(deficit):
            h[indices[j]] = 1.0
    return h
