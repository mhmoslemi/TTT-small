```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_corr(h):
    """Compute all correlation values (without dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def get_active_lags(corr, threshold=0.001):
    """Get lags within threshold of max."""
    max_val = np.max(corr)
    threshold_abs = max_val - threshold * max_val
    return np.where(corr >= threshold_abs)[0]

def slp_step(h0, n_lags_to_cons, tol=1e-8, max_slp_iters=50):
    """Successive Linear Programming to minimize max correlation."""
    n = len(h0)
    h = h0.copy()
    for it in range(max_slp_iters):
        corr = compute_corr(h)
        # Get top n_lags_to_cons lags
        top_idx = np.argsort(corr)[-n_lags_to_cons:]
        
        # Build linearization
        # For each active lag, compute gradient
        # corr[k] = sum_i h[i]*(1-h[i+k]) for appropriate i
        # In the correlation indexing: corr[lag_idx] where lag_idx = k + (n-1)
        
        A_rows = []
        b_vals = []
        for lag_idx in top_idx:
            lag = lag_idx - (n - 1)
            if lag >= 0:
                # corr = sum_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
                # = sum h[i] - sum h[i]*h[i+lag] for i=0..n-1-lag
                # grad w.r.t. h[j]: 
                # For j in 0..n-1-lag: 1 - h[j+lag]
                # For j in n-lag..n-1: -h[j-lag]
                grad = np.zeros(n)
                for i in range(n - lag):
                    grad[i] += 1.0 - h[i + lag]
                for j in range(n - lag, n):
                    grad[j] += -h[j - lag]
            else:
                m = -lag
                # corr = sum_{i=m}^{n-1} h[i] * (1 - h[i-m])
                # = sum h[i] - sum h[i]*h[i-m] for i=m..n-1
                grad = np.zeros(n)
                for i in range(m, n):
                    grad[i] += 1.0 - h[i - m]
                for j in range(0, m):
                    grad[j] += -h[j + m]
            
            A_rows.append(grad)
            b_vals.append(corr[lag_idx])
        
        A = np.array(A_rows)  # shape (n_lags_to_cons, n)
        b = np.array(b_vals)  # shape (n_lags_to_cons,)
        
        # LP: minimize T
        # subject to: b + A @ (h_new - h) <= T  for each row
        #             sum(h_new) = n/2
        #             0 <= h_new <= 1
        h_var = cp.Variable(n)
        T_var = cp.Variable()
        constraints = [
            h_var >= 0, h_var <= 1,
            cp.sum(h_var) == n / 2.0
        ]
        for i in range(len(top_idx)):
            constraints.append(b[i] + A[i] @ (h_var - h) <= T_var)
        
        prob = cp.Problem(cp.Minimize(T_var), constraints)
        try:
            prob.solve(solver=cp.GLOP)  # or CBC
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = h_var.value
        h_new = np.clip(h_new, 0, 1)
        # Project to feasible
        diff = n/2.0 - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        
        if np.max(np.abs(h_new - h)) < tol:
            h = h_new
            break
        h = h_new
    
    return h
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        idx = np.where(h > 1e-12)[0]

    return np.clip(h, 0.0, 1.0)

def corr_and_grad(h, lags):
    """Compute correlation values and gradients at specified lags."""
    n = len(h)

    corr_vals = corr_full[lags]

        else:
            L = -lag
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] += -w * h[L:]
    return obj, grad

def slp_optimize(h0, max_iters=30, n_active=10, solver='GLOP'):
    """Successive Linear Programming to minimize max correlation."""
    import cvxpy as cp
    n = len(h0)
    h = h0.copy()
    target = n / 2.0
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        
        # Build linearization matrix
        A = np.zeros((len(top_idx), n))
        b = np.zeros(len(top_idx))
        for i, lag_idx in enumerate(top_idx):
            lag = lag_idx - (n - 1)
            b[i] = corr[lag_idx]
            if lag >= 0:
                L = lag
                A[i, :n-L] += 1.0 - h[L:]
                A[i, L:] += -h[:n-L]
            else:
                L = -lag
                A[i, L:] += 1.0 - h[:n-L]
                A[i, :n-L] += -h[L:]
        
        h_var = cp.Variable(n)
        T_var = cp.Variable()
        constraints = [
            h_var >= 0, h_var <= 1,
            cp.sum(h_var) == target
        ]
        for i in range(len(top_idx)):
            constraints.append(b[i] + A[i] @ (h_var - h) <= T_var)
        
        prob = cp.Problem(cp.Minimize(T_var), constraints)
        try:
            prob.solve(solver=solver)
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        diff = target - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        
        step = np.max(np.abs(h_new - h))
        h = h_new
        if step < 1e-10:
            break
    
    return project_feasible(h, target)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) > 0:
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(n_corr)
    vals = corr[idx]
    c_max = np.max(vals)
    shifted = np.clip(alpha * (vals - c_max), -600.0, 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    obj = c_max + np.log(total) / alpha
    weights = exps / total
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            L = lag
            if L < n:
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] += -w * h[:n - L]
        else:
            L = -lag
            if L < n:
                grad[L:] += w * (1.0 - h[:n - L])
                grad[:n - L] += -w * h[L:]
    return obj, grad


def _obj_lse(x, alpha, topk, penalty, target):
    f, g = lse_obj_grad(x, alpha, topk)
    s = np.sum(x) - target
    f = f + penalty * s * s
    g = g + 2.0 * penalty * s
    return f, g


def lse_optimize(h_start, target, alpha, topk, penalty, bounds, maxiter=500):
    try:
        res = minimize(_obj_lse, h_start, jac=True,
                       args=(alpha, topk, penalty, target),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def slp_optimize(h0, max_iters=20, n_active=12, solver='GLOP'):
    n = len(h0)
    h = h0.copy()
    target = n / 2.0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        A = np.zeros((len(top_idx), n))
        b = np.zeros(len(top_idx))
        for i, lag_idx in enumerate(top_idx):
            li = int(lag_idx)
            lag = li - (n - 1)
            b[i] = corr[li]
            if lag >= 0:
                L = lag
                if L < n:
                    A[i, :n - L] = 1.0 - h[L:]
                    A[i, L:] = -h[:n - L]
            else:
                L = -lag
                if L < n:
                    A[i, L:] = 1.0 - h[:n - L]
                    A[i, :n - L] = -h[L:]
        h_var = cp.Variable(n)
        T_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for i in range(len(top_idx)):
            constraints.append(b[i] + A[i] @ (h_var - h) <= T_var)
        prob = cp.Problem(cp.Minimize(T_var), constraints)
        solved = False
        for s in [solver, 'CBC', 'ECOS', 'CLARABEL']:
            try:
                prob.solve(solver=s)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0, 1)
        diff = target - np.sum(h_new)
        h_new = h_new + diff / n
        h_new = np.clip(h_new, 0, 1)
        step_norm = np.max(np.abs(h_new - h))
        h = h_new
        if step_norm < 1e-11:
            break
    return project_feasible(h, target)


def optimize_at_resolution(h_init, n_res, target_res, bounds_res, time_limit):
    start_t = time.time()
    h_best = project_feasible(h_init, target_res)
    score_best = compute_c5(h_best)

    def elapsed():
        return time.time() - start_t

    if time_limit - elapsed() > 30:
        h_s = slp_optimize(h_best, max_iters=15, n_active=15)
        sc = compute_c5(h_s)
        if sc < score_best:
            score_best = sc
            h_best = h_s

    configs = [
        (40, 10, 5000, 400),
        (100, 8, 10000, 400),
        (200, 6, 20000, 500),
        (400, 5, 30000, 500),
        (800, 4, 50000, 500),
        (1500, 3, 80000, 400),
        (3000, 3, 100000, 300),
    ]
    for alpha, topk, penalty, maxit in configs:
        if time_limit - elapsed() < 30:
            break
        for trial in range(2):
            if time_limit - elapsed() < 20:
                break
            pert = 0.003 * (trial + 1)
            h_start = h_best + np.random.RandomState(int(elapsed() * 1000 + trial)) .randn(n_res) * pert
            h_start = project_feasible(h_start, target_res)
            h_new = lse_optimize(h_start, target_res, alpha, topk, penalty, bounds_res, maxit)
            sc = compute_c5(h_new)
            if sc < score_best:
                score_best = sc
                h_best = h_new

    if time_limit - elapsed() > 40:
        h_s2 = slp_optimize(h_best, max_iters=20, n_active=8)
        sc = compute_c5(h_s2)
        if sc < score_best:
            score_best = sc
            h_best = h_s2
        if time_limit - elapsed() > 20:
            h_new = lse_optimize(h_best, target_res, 800, 4, 50000, bounds_res, 400)
            sc = compute_c5(h_new)
            if sc < score_best:
                score_best = sc
                h_best = h_new

    return h_best, score_best

        if time_left() > 40:
```
