import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(v, target):
    n = len(v)
    v = v.astype(float).copy()
    sums = np.sum(v)
    if abs(sums - target) < 1e-12:
        return np.clip(v, 0.0, 1.0)
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def corr_grad_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k > 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    elif k < 0:
        m = -k
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        g = 1.0 - 2.0 * h
    return g

def subgradient_step(h, top_k=10):
    n = len(h)
    corr = get_corr(h)
    center = n - 1
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for idx in sorted_idx:
        k = int(idx - center)
        g += corr_grad_single(h, k)
    g /= top_k
    return g, corr

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    c_max = float(corr.max())
    shifted = alpha * (corr - c_max)
    exp_shifted = np.exp(shifted - shifted.max())
    norm = np.sum(exp_shifted)
    weights = exp_shifted / norm
    obj_val = c_max + np.log(norm) / alpha + (shifted.max() - alpha * (corr.max() - c_max)) / alpha
    obj_val = c_max + np.log(norm) / alpha
    center = n - 1
    g = np.zeros(n)
    top_k = min(30, n * 2 - 1)
    top_idx = np.argsort(weights)[::-1][:top_k]
    for idx in top_idx:
        w = weights[idx]
        if w < 1e-15:
            break
        k = int(idx - center)
        g += w * corr_grad_single(h, k)
    return obj_val, g, corr

def slsqp_optimize(h_start, target, alpha, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj(h_in):
        val, _, _ = lse_obj_and_grad(h_in, alpha)
        return val

    def obj_grad(h_in):
        _, g, _ = lse_obj_and_grad(h_in, alpha)
        return g

    def cons_f(h_in):
        return np.sum(h_in) - target

    def cons_j(h_in):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    try:
        res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_box_sum(res.x, target)
        sc = compute_c5(h_new)
        if sc < curr_score:
            curr_score = sc
            h_cur = h_new
    except Exception:
        pass
    return h_cur, curr_score

def subgradient_descent(h_start, target, max_iters, step_init, step_final, time_limit, start_time, cooldown=5):
    n = len(h_start)
    h = project_box_sum(h_start, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    max_no_improve = max(500, max_iters // 30)

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ratio = i / max(max_iters - 1, 1)
        step = step_init * (step_final / step_init) ** ratio

        g, corr = subgradient_step(h, top_k=10)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        h_new = h - step * g / g_norm * 0.5
        h_new = project_box_sum(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            no_improve += 1
            step *= 0.95

    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    target_dst = n_dst / 2.0
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def fast_hill_climb_binary(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 80

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)

            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5

def targeted_sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):

    for i in range(max_iters):

            b_km = zeros - k_star

def optimize_continuous_block(h_start, target, time_limit, start_time):
