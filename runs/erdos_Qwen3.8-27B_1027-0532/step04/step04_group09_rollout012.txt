    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

        sc = compute_c5(h_new)
        if sc < cur_score:

    except Exception:

    return h, cur_score

    best_sc = compute_c5(h_cur)
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n):
            j = rng.randint(n)
            old_val = h_cur[j]
            candidates = np.clip(old_val + 0.05 * rng.randn(8), 0.0, 1.0)
            candidates = np.unique(np.round(candidates, 6))
            best_local = old_val
            best_local_sc = compute_c5(h_cur)
            for cv in candidates:
                h_cur[j] = cv
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = cv
            h_cur[j] = best_local
            h_cur = project_feasible(h_cur, target)
            sc_now = compute_c5(h_cur)
            if sc_now < best_sc:
                best_sc = sc_now
                improved = True
        if not improved:
            break
    return h_cur, best_sc


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    t_end = start_time + budget_s - 30.0

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    t0 = n0 / 2.0
    best_h = h0.copy()
    best_sc = compute_c5(best_h)

    dl1 = start_time + 180.0
    best_h, best_sc = slsqp_seq(best_h, t0, [10, 50, 200, 1000, 5000, 20000, 60000], 2000, 1e-15, dl1)

    dl2 = start_time + 430.0
    for trial in range(15):
        if time.time() > dl2:
            break
        noise_scale = 0.012 + 0.003 * trial
        noise = noise_scale * rng.randn(n0)
        h_p = project_feasible(best_h + noise, t0)
        sc_p = compute_c5(h_p)
        if sc_p > best_sc * 1.06:
            continue
        h_r, sc_r = slsqp_seq(h_p, t0, [500, 3000, 12000, 50000], 1000, 1e-15, min(dl2, time.time() + 18.0))
        if sc_r < best_sc:
            best_sc = sc_r
            best_h = h_r

    dl3 = start_time + 500.0
    best_h, best_sc = subgrad_polish(best_h, t0, 2000, 0.02, 0.04, dl3)

    n_f = 800
    t_f = n_f / 2.0
    h_f = interp_to(best_h, n_f, t_f)
    sc_f = compute_c5(h_f)

    dl4 = start_time + 720.0
    h_f, sc_f = slsqp_seq(h_f, t_f, [200, 1000, 5000, 20000, 80000], 1200, 1e-15, dl4)
    if sc_f < best_sc:
        best_sc = sc_f
        best_h = h_f

    n_c = len(best_h)
    t_c = n_c / 2.0

    dl5 = start_time + 880.0
    best_h, best_sc = subgrad_polish(best_h, t_c, 3000, 0.012, 0.03, dl5)

    dl6 = start_time + 1020.0
    best_h, best_sc = slsqp_seq(best_h, t_c, [30000, 100000, 300000, 800000], 1500, 1e-16, dl6)

    dl7 = start_time + 1120.0
    best_h, best_sc = coordinate_polish(best_h, t_c, 3, dl7, rng)

    dl8 = t_end
    for trial in range(8):
        if time.time() > dl8:
            break
        noise = 0.004 * rng.randn(n_c)
        h_p = project_feasible(best_h + noise, t_c)
        sc_p = compute_c5(h_p)
        if sc_p > best_sc * 1.02:
            continue
        h_r, sc_r = slsqp_seq(h_p, t_c, [10000, 50000, 200000, 500000], 600, 1e-16, min(dl8, time.time() + 20.0))
        if sc_r < best_sc:
            best_sc = sc_r
            best_h = h_r

    if time.time() < t_end - 15:
        best_h, best_sc = subgrad_polish(best_h, t_c, 1500, 0.004, 0.01, t_end)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_sc = compute_c5(best_h)
    return best_h, final_sc, len(best_h)
