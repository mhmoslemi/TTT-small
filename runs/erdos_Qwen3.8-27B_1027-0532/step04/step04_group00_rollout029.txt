    active = np.where(w > threshold)[0]
    for k in active:
        grad += w[k] * _subgrad_of_lag(x, int(k))
    return grad

def _lbfgsb_optimize(h, n, target, alphas, maxiter_each, deadline):
    global _g_alpha
    h_cur = h.copy()
    cur_score = _compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_objective, h_cur, jac=_lse_gradient, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = _project_feasible(res.x, target)
            sc = _compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def _subgradient_polish(h, n, target, n_steps, deadline, lr_init=0.003,
                        n_active=4, temp=0.005):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = _compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    n_corr = 2 * n - 1
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = _compute_corr_full(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = max(n_active, 2)
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            if wi < 1e-6:
                continue
            grad += wi * _subgrad_of_lag(h_cur, int(ki))
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = _project_feasible(h_new, target)
        s_new = _compute_c5(h_new)
        if s_new < c_curr - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.75
            if lr < 1e-10:
                break
    return best_h, best_score

def _worst_lag_surgery(h, n, target, max_steps, deadline, eps_init=0.002):
    h_cur = h.copy()
    cur_score = _compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    n_corr = 2 * n - 1
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = _compute_corr_full(h_cur)
        k_star = int(np.argmax(corr))
        grad = _subgrad_of_lag(h_cur, k_star) * dx
        h_new = h_cur - eps * grad
        h_new = _project_feasible(h_new, target)
        sc_new = _compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.04, 0.02)
        else:
            eps *= 0.7
            if eps < 1e-8:
                break
    return h_cur, cur_score

def _block_descent(h, n, target, block_size, n_rounds, deadline, nr=500.0):
    global _g_alpha
    h_cur = h.copy()
    cur_score = _compute_c5(h_cur)
    n_blocks = n // block_size
    bounds_bl = [(0.0, 1.0)] * block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            x0 = h_cur[s:e].copy()
            try:
                _g_alpha = float(nr)
                res = minimize(_partial_block_obj, x0, args=(h_cur, s, e, n, target),
                               method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = _project_feasible(h_test, target)
                sc = _compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur, cur_score

def _partial_block_obj(x, h_full, s, e, n, target):
    h_test = h_full.copy()
    h_test[s:e] = np.clip(x, 0.0, 1.0)
    h_test = _project_feasible(h_test, target)
    return _compute_c5(h_test)

def _run_de(n, target, pop_size, max_gens, deadline, rng, h_init=None):
    pop = np.zeros((pop_size, n))
    scores = np.zeros(pop_size)
    for i in range(pop_size):
        if i == 0 and h_init is not None:
            pop[i] = h_init.copy()
        elif i % 4 == 1 and h_init is not None:
            pop[i] = h_init + rng.randn(n) * 0.08
        elif i % 4 == 2:
            pop[i] = (rng.rand(n) > 0.5).astype(float)
        elif i % 4 == 3:
            nblocks = 2 + rng.randint(4)
            edges = sorted(rng.randint(1, n, nblocks - 1))
            pop[i] = np.zeros(n)
            for j in range(nblocks):
                lo = edges[j-1] if j > 0 else 0
                hi = edges[j] if j < len(edges) else n
                if rng.rand() > 0.4:
                    pop[i][lo:hi] = 1.0
        else:
            pop[i] = rng.rand(n)
            pop[i] = (pop[i] > 0.4).astype(float)
        pop[i] = _project_feasible(pop[i], target)
        scores[i] = _compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best_score = float(scores[best_idx])
    F = 0.8
    CR = 0.9
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            cands = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(cands, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            j0 = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[j0] = True
            trial[mask] = v[mask]
            trial = _project_feasible(trial, target)
            s = _compute_c5(trial)
            if s <= scores[i]:
                pop[i] = trial
                scores[i] = s
                if s < best_score:
                    best_score = s
                    best_idx = int(i)
        F = max(0.35, F * 0.997)
    return pop[best_idx].copy(), best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n_final = 400
    target_final = n_final / 2.0

    if 'initial_h_values' in dir() or True:
        try:
            h_parent = np.array(initial_h_values, dtype=float)
        except NameError:
            h_parent = None
    else:
        h_parent = None

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    if h_parent is not None:
        h_400 = _project_feasible(h_parent.copy(), target_final)
        best_h = h_400.copy()
        best_score = _compute_c5(best_h)
    else:
        h_400 = np.zeros(n_final)
        h_400[:n_final//2] = 1.0
        best_h = h_400.copy()
        best_score = _compute_c5(best_h)

    # Phase 1: DE at n=30
    n1 = 30
    t1 = n1 / 2.0
    h1 = _interpolate_to(best_h, n1, t1) if h_parent is not None else None
    deadline1 = start_time + 55.0
    h1, s1 = _run_de(n1, t1, 30, 5000, deadline1, rng, h_init=h1)
    h1_up = _interpolate_to(h1, n_final, target_final)
    sc1 = _compute_c5(h1_up)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1_up

    # Phase 2: DE at n=60
    if tleft() > 100:
        n2 = 60
        t2 = n2 / 2.0
        h2 = _interpolate_to(best_h, n2, t2)
        deadline2 = start_time + 120.0
        h2, s2 = _run_de(n2, t2, 25, 4000, deadline2, rng, h_init=h2)
        h2_up = _interpolate_to(h2, n_final, target_final)
        sc2 = _compute_c5(h2_up)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2_up

    # Phase 3: DE at n=120
    if tleft() > 100:
        n3 = 120
        t3 = n3 / 2.0
        h3 = _interpolate_to(best_h, n3, t3)
        deadline3 = start_time + 200.0
        h3, s3 = _run_de(n3, t3, 20, 3000, deadline3, rng, h_init=h3)
        h3_up = _interpolate_to(h3, n_final, target_final)
        sc3 = _compute_c5(h3_up)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3_up

    # Phase 4: L-BFGS-B at n=400 with adaptive alpha
    if tleft() > 120:
        deadline4 = start_time + 340.0
        alphas4 = [10, 50, 200, 800]
        h4, sc4 = _lbfgsb_optimize(best_h, n_final, target_final, alphas4, 3000, deadline4)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Subgradient polish
    if tleft() > 100:
        deadline5 = start_time + 450.0
        h5, sc5 = _subgradient_polish(best_h, n_final, target_final, 2000, deadline5,
                                       lr_init=0.003, n_active=4, temp=0.004)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: L-BFGS-B again with higher alpha
    if tleft() > 100:
        deadline6 = start_time + 580.0
        alphas6 = [500, 2000, 8000]
        h6, sc6 = _lbfgsb_optimize(best_h, n_final, target_final, alphas6, 3000, deadline6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Surgery + subgradient
    if tleft() > 80:
        deadline7 = start_time + 680.0
        h7, sc7 = _worst_lag_surgery(best_h, n_final, target_final, 300, deadline7,
                                      eps_init=0.002)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
        if tleft() > 50:
            h7b, sc7b = _subgradient_polish(best_h, n_final, target_final, 1000,
                                             start_time + 760.0, lr_init=0.002,
                                             n_active=3, temp=0.003)
            if sc7b < best_score:
                best_score = sc7b
                best_h = h7b

    # Phase 8: Block descent
    if tleft() > 80:
        deadline8 = start_time + 900.0
        for bs in [3, 4]:
            if tleft() < 50:
                break
            h8, sc8 = _block_descent(best_h, n_final, target_final, bs, 6, deadline8, nr=500.0)
            if sc8 < best_score:
                best_score = sc8
                best_h = h8

    # Phase 9: L-BFGS-B with very high alpha (near-exact max)
    if tleft() > 60:
        deadline9 = start_time + 1050.0
        alphas9 = [3000, 10000, 30000]
        h9, sc9 = _lbfgsb_optimize(best_h, n_final, target_final, alphas9, 2000, deadline9)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Final subgradient + surgery
    if tleft() > 40:
        deadline10 = start_time + budget_s - 30.0
        h10, sc10 = _subgradient_polish(best_h, n_final, target_final, 800, deadline10,
                                         lr_init=0.001, n_active=3, temp=0.002)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10
        if tleft() > 15:
            h10b, sc10b = _worst_lag_surgery(best_h, n_final, target_final, 200,
                                              start_time + budget_s - 5.0, eps_init=0.001)
            if sc10b < best_score:
                best_score = sc10b
                best_h = h10b

    # Multi-start: try a few random perturbations if time permits
    if tleft() > 60:
        deadline_ms = start_time + budget_s - 5.0
        for attempt in range(4):
            if tleft() < 30:
                break
            noise = 0.04 + 0.03 * attempt
            h_pert = best_h + rng.randn(n_final) * noise
            h_pert = _project_feasible(h_pert, target_final)
            h_ref, sc_ref = _lbfgsb_optimize(h_pert, n_final, target_final,
                                              [100, 500, 2000], 1500, deadline_ms)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Final projection and verification
    h_final = _project_feasible(best_h, target_final)
    final_score = _compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    return h_final, best_score, len(h_final)
