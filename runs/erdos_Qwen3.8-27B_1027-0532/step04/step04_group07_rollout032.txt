import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = int(free.sum())
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)

def compute_grad_lag(h, lag_idx):

    if k < nc:

    nc = len(corr)
    k = min(topk, nc)
    if k < nc:

def lp_step(h, target_sum, topk, t_bound=None):
    n = len(h)
    corr = overlap_corrs(h)
    nc = len(corr)
    k = min(topk, nc)
    if k < nc:
        top_idx = np.argpartition(corr, -k)[-k:]
    else:
        top_idx = np.arange(nc)
    m = len(top_idx)
    # Build LP: minimize t s.t. 
    #   grad_k · h + (h_k^2 terms linearized) ≤ t  for each active k
    #   sum(h) = target_sum
    #   0 ≤ h ≤ 1, t free
    # Linearization: corr_k(h) ≈ corr_k(h_cur) + grad_k(h_cur) · (h - h_cur)
    # So: corr_k(h_cur) + grad_k · (h - h_cur) ≤ t
    #   => grad_k · h ≤ t + (grad_k · h_cur - corr_k(h_cur))
    #   => grad_k · h - t ≤ grad_k · h_cur - corr_k(h_cur)
    
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_lag(h, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = float(g @ h) - corr[idx]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))

    bounds_lp = [(0.0, 1.0)] * n + [(0.0, 1e6)]

                  bounds=bounds_lp, method='highs')

    return None, None

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = int(free.sum())
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = int(free.sum())
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def _smooth_obj(h, alpha, topk):
    corr = overlap_corrs(h)
    nc = len(corr)
    k = min(topk, nc)
    if k < nc:
        top_idx = np.argpartition(corr, -k)[-k:]
        vals = corr[top_idx]
    else:
        vals = corr
    c_ref = float(np.max(vals))
    shifted = np.clip(alpha * (vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def _smooth_grad(h, alpha, topk):
    corr = overlap_corrs(h)
    nc = len(corr)
    k = min(topk, nc)
    if k < nc:
        top_idx = np.argpartition(corr, -k)[-k:]
        vals = corr[top_idx]
        idx = top_idx
    else:
        vals = corr
        idx = np.arange(nc)
    c_ref = float(np.max(vals))
    shifted = np.clip(alpha * (vals - c_ref), -500.0, 0.0)
    ev = np.exp(shifted)
    total = float(np.sum(ev))
    weights = ev / total
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w > 1e-14:
            grad += w * compute_grad_lag(h, idx[i])
    return grad


def _cons_eq(x, target):
    return np.sum(x) - target


def _cons_eq_jac(x, n):
    return np.ones(n)


def slsqp_smooth(h_start, target_sum, alpha, topk, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(
            _smooth_obj, h_start, args=(alpha, topk),
            jac=_smooth_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start, compute_c5(h_start)


def lp_cutting_step(h, target_sum, topk):
    n = len(h)
    corr = overlap_corrs(h)
    nc = len(corr)
    k = min(topk, nc)
    if k < nc:
        top_idx = np.argpartition(corr, -k)[-k:]
    else:
        top_idx = np.arange(nc)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_lag(h, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = float(g @ h) - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target_sum)
    return None


def lp_refine(h_start, target_sum, topk_schedule, max_iter, time_limit, t0,

                h = h_lp
