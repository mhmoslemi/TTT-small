    """SLSQP with FFT-based gradient."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj(x):
            corr = fast_over(x)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        
        def obj_grad(x):

            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            exp_s = np.exp(shifted)
            sum_exp = float(np.sum(exp_s))
            w = exp_s / sum_exp
            N = _next_pow2(2 * n - 1)

            B = np.fft.irfft(np.conj(W) * H, N)[:n]

                best_score = sc

from scipy.optimize import minimize
import time

def _npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def fast_over(h):
    n = len(h)
    N = _npow2(2 * n)
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(np.conj(H) * G, N)[:n]

def compute_c5(h):
    n = len(h)
    return float(np.max(fast_over(h))) * (2.0 / n)

def project_sum_box(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target:
            hi = mid
        else:
            lo = mid
    p = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = target - np.sum(p)
    if abs(d) > 1e-12:
        m = (p > 1e-12) & (p < 1.0 - 1e-12)
        if m.any():
            p[m] += d / m.sum()
            np.clip(p, 0.0, 1.0, out=p)
    return p

    cmax = float(corr.max())

    s = np.clip(s, -700.0, 0.0)

            W = np.fft.rfft(w, N)
            H = np.fft.rfft(x, N)
            G1 = np.fft.rfft(1.0 - x, N)

            B = np.fft.irfft(np.conj(W) * H, N)[:n]
