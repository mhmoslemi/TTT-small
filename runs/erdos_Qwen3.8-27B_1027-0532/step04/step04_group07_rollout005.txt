import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        nfree = free.sum()
        if nfree == 0:
            break
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, lag_k):
    """Compute gradient of corr[lag_k] w.r.t. h"""
    n = len(h)
    lag = lag_k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def lp_step(h_cur, target_sum, topk, step_fraction=1.0):
    n = len(h_cur)
    corr = get_corrs(h_cur)
    n_corr = len(corr)
    if topk >= n_corr:
        idx = np.arange(n_corr)
    else:
        idx = np.argpartition(corr, -topk)[-topk:]
    
    m = len(idx)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, k in enumerate(idx):
        g = grad_for_lag(h_cur, k)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g.dot(h_cur) - corr[k]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    h_lp = res.x[:n]
    if step_fraction < 1.0:
        h_lp = step_fraction * h_lp + (1.0 - step_fraction) * h_cur
    return project_feasible(h_lp, target_sum)

def smooth_obj(h, alpha, topk):
    corr = get_corrs(h)
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    c_ref = np.max(vals)
    shifted = np.clip(alpha * (vals - c_ref), -500, 0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, alpha, topk):

    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(n_corr)
        vals = corr
    c_ref = np.max(vals)
    exp_sh = np.exp(np.clip(alpha * (vals - c_ref), -500, 0))
    w = exp_sh / exp_sh.sum()
    grad = np.zeros(len(h))
    for i, k in enumerate(idx):
        grad += w[i] * grad_for_lag(h, k)
    return grad

def slsqp_opt(h_start, target_sum, alpha_list, topk_list, maxiter, time_limit):

    _N = len(h_start)
    _T = target_sum
    h = h_start.copy()
    best_h = h.copy()

    t0 = time.time()

            h_new = project_feasible(res.x, _T)
            sc = compute_c5(h_new)

                best_h = h_new.copy()
                h = h_new
        except:
            pass
    return best_h, best_score

    x_new = np.linspace(0, 1, n_dst)

def binary_local_search(h_init, target, max_rounds, time_limit):
    """Convert to binary and do swap-based local search"""
    n = len(h_init)
    h = h_init.copy()
    # Make binary
    idx_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx_sorted[:int(target)]] = 1.0
    best_h = h_bin.copy()
    best_score = compute_c5(h_bin)
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        one_pos = np.where(h_bin == 1.0)[0]
        zero_pos = np.where(h_bin == 0.0)[0]
        # Random subset of swaps
        rng = np.random.RandomState(rnd + 12345)
        n_swaps = min(5000, len(one_pos) * len(zero_pos))
        for _ in range(n_swaps):
            i = one_pos[rng.randint(len(one_pos))]
            j = zero_pos[rng.randint(len(zero_pos))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            sc = compute_c5(h_bin)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_bin.copy()
                one_pos = np.where(h_bin == 1.0)[0]
                zero_pos = np.where(h_bin == 0.0)[0]
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:

import numpy as np
from scipy.optimize import minimize, linprog
import time

_N = 0
_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        nfree = free.sum()
        if nfree == 0:
            break
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, lag_k):
    n = len(h)
    lag = lag_k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def _eq_fun(x):
    return np.sum(x) - _T

def _eq_jac(x):
    return np.ones(_N)

def smooth_obj(h, alpha, topk):
    corr = get_corrs(h)
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    c_ref = np.max(vals)
    shifted = np.clip(alpha * (vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, alpha, topk):
    corr = get_corrs(h)
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(n_corr)
        vals = corr
    c_ref = np.max(vals)
    exp_sh = np.exp(np.clip(alpha * (vals - c_ref), -500.0, 0.0))
    w = exp_sh / exp_sh.sum()
    return grad_multi_lag(h, idx, w)

def lp_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = get_corrs(h_cur)
    n_corr = len(corr)
    if topk >= n_corr:
        idx = np.arange(n_corr)
    else:
        idx = np.argpartition(corr, -topk)[-topk:]
    m = len(idx)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, k in enumerate(idx):
        g = grad_for_lag(h_cur, k)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g.dot(h_cur) - corr[k]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return project_feasible(res.x[:n], target_sum)

def lp_refine(h_start, target_sum, max_iter, topk, time_limit):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stale = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        h_new = lp_step(h, target_sum, topk)
        if h_new is None:
            break
        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:

            best_h = h_new.copy()
            h = h_new
            stale = 0
        else:
            h_half = project_feasible(0.5 * h + 0.5 * h_new, target_sum)
            sc_half = compute_c5(h_half)
            if sc_half < best_score - 1e-14:
                best_score = sc_half
                best_h = h_half.copy()
                h = h_half
                stale = 0
            else:
                stale += 1
                if stale > 15:
                    break
    return best_h, best_score

def slsqp_opt(h_start, target_sum, alpha_list, topk_list, maxiter_per, time_limit):
    global _N, _T
    _N = len(h_start)
    _T = target_sum
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * _N
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for alpha, topk in zip(alpha_list, topk_list):
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(smooth_obj, h, args=(alpha, topk),
                          jac=smooth_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per, 'ftol': 1e-14})
            h_new = project_feasible(res.x, _T)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    x_old = np.linspace(0.0, 1.0, len(h_src))
    x_new = np.linspace(0.0, 1.0, n_dst)
    return project_feasible(np.interp(x_new, x_old, h_src), target_dst)

def binary_local_search(h_init, target, max_rounds, time_limit, seed):

        zero_pos = np.where(h_bin == 0.0)[0]
        if len(one_pos) == 0 or len(zero_pos) == 0:
            break
        n_try = min(1500, len(one_pos) * len(zero_pos))
        improved = False
        for _ in range(n_try):

            j = zero_pos[rng.randint(len(zero_pos))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            sc = compute_c5(h_bin)

                best_score = sc
                best_h = h_bin.copy()
                one_pos = np.where(h_bin == 1.0)[0]
                zero_pos = np.where(h_bin == 0.0)[0]
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return best_h, best_score

def subgradient_refine(h_init, target, max_iter, lr, topk, alpha, momentum, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    stagnation = 0
    for it in range(max_iter):
        corr = get_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        tv = corr[top_idx]
        w = np.exp(alpha * (tv - np.max(tv)))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= ws
        g = grad_multi_lag(h, top_idx, w)
        v = momentum * v + (1.0 - momentum) * g
        h_new = project_feasible(h - lr * v, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.02, 2.0)
        else:
            stagnation += 1
            lr *= 0.98
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    remaining = lambda: budget_s - 25.0 - elapsed()

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    # Phase 1: Aggressive LP at current resolution
    if remaining() > 180:
        h_new, sc = lp_refine(h, target, max_iter=400, topk=120, time_limit=remaining() - 160)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    # Phase 2: SLSQP continuation
    if remaining() > 200:
        h_new, sc = slsqp_opt(h, target,
                              alpha_list=[10, 50, 200, 1000, 5000],
                              topk_list=[100, 60, 30, 15, 8],
                              maxiter_per=3000, time_limit=remaining() - 180)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    # Phase 3: LP refinement with tighter topk
    if remaining() > 150:
        h_new, sc = lp_refine(h, target, max_iter=500, topk=40, time_limit=remaining() - 130)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    # Phase 4: Subgradient with momentum for diversification
    if remaining() > 120:
        rng = np.random.RandomState(seed)
        h_pert = project_feasible(h + rng.randn(n) * 0.005, target)
        h_new, sc = subgradient_refine(h_pert, target, max_iter=2000, lr=0.05,
                                        topk=10, alpha=50, momentum=0.92,
                                        patience=200, seed=seed)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    # Phase 5: Binary local search
    if remaining() > 80:
        h_bin, sc_bin = binary_local_search(h, target, max_rounds=100,
                                             time_limit=remaining() - 60, seed=seed + 77)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
            h = h_bin
            # Refine from binary with LP
            if remaining() > 80:
                h_new, sc = lp_refine(h, target, max_iter=200, topk=50, time_limit=remaining() - 60)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
                    h = h_new

    # Phase 6: Try higher resolution n=600
    if remaining() > 250:
        n600 = 600
        h600 = interpolate_to(h, n600, n600 / 2.0)
        h_new, sc = lp_refine(h600, n600 / 2.0, max_iter=200, topk=80,
                               time_limit=remaining() - 230)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new
        if remaining() > 120:
            h_new, sc = slsqp_opt(h, len(h) / 2.0,
                                   alpha_list=[50, 300, 2000],
                                   topk_list=[60, 30, 12],
                                   maxiter_per=2000, time_limit=remaining() - 100)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h = h_new
        if remaining() > 80:
            h_new, sc = lp_refine(h, len(h) / 2.0, max_iter=300, topk=30,
                                   time_limit=remaining() - 60)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h = h_new

    # Phase 7: Another binary search attempt at current best
    if remaining() > 80:
        n_cur = len(h)
        h_bin, sc_bin = binary_local_search(h, n_cur / 2.0, max_rounds=80,
                                             time_limit=remaining() - 60, seed=seed + 999)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
            h = h_bin
            if remaining() > 60:
                h_new, sc = lp_refine(h, n_cur / 2.0, max_iter=200, topk=40,
                                       time_limit=remaining() - 40)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
                    h = h_new

    # Phase 8: Final LP polish
    if remaining() > 50:
        n_use = len(h)
        h_new, sc = lp_refine(h, n_use / 2.0, max_iter=200, topk=15,
                               time_limit=remaining() - 15)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    # Phase 9: Tiny final polish
    if remaining() > 20:
        n_use = len(h)
        h_new, sc = lp_refine(h, n_use / 2.0, max_iter=80, topk=6,
                               time_limit=remaining() - 5)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    h_final = project_feasible(h, len(h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
