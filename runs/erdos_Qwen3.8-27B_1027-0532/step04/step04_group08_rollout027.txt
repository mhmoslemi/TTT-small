def softmax_obj(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def softmax_grad(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = wt[s]
        if w < 1e-14:
            continue
        k = s - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - x[k:])
            grad[k:] -= w * x[:n-k]
        else:
            j = -k
            grad[j:] += w * (1.0 - x[:n-j])
            grad[:n-j] -= w * x[j:]
    return grad

def active_set_grad(x, active_indices):
    """Gradient of max using only active shifts."""
    n = len(x)
    grad = np.zeros(n)
    count = len(active_indices)
    if count == 0:
        return grad
    for s in active_indices:
        k = s - (n - 1)
        if k >= 0:
            grad[:n-k] += (1.0 - x[k:])
            grad[k:] -= x[:n-k]
        else:
            j = -k
            grad[j:] += (1.0 - x[:n-j])
            grad[:n-j] -= x[j:]
    grad /= count
    return grad

def find_active_set(corr, c_max, frac=0.01):
    """Find indices of correlations within frac of c_max."""
    threshold = c_max * (1 - frac)
    return np.where(corr >= threshold)[0]

def active_set_optimize(h0, n, target, max_iter=500, tol=1e-10, deadline=None):
    """Optimize using active-set gradient method with backtracking line search."""
    h = project_feasible(h0, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        active = find_active_set(corr, c_max, frac=0.005)
        g = active_set_grad(h, active)
        
        # Backtracking line search
        best_local = best_score
        best_local_h = best_h
        lr = 0.01
        for ls in range(30):
            h_trial = project_feasible(h - lr * g, target)
            sc = compute_c5(h_trial)
            if sc < best_local - 1e-14:
                best_local = sc
                best_local_h = h_trial
                break
            lr *= 0.5
        
        if sc < best_score - 1e-12:
            h = best_local_h
            best_score = sc
            best_h = h.copy()
        else:
            # Try smaller active set
            active_tight = find_active_set(corr, c_max, frac=0.001)
            g2 = active_set_grad(h, active_tight)
            lr = 0.005
            improved = False
            for ls in range(20):
                h_trial = project_feasible(h - lr * g2, target)
                sc = compute_c5(h_trial)
                if sc < best_score - 1e-12:
                    h = h_trial
                    best_score = sc
                    best_h = h.copy()
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                break
    
    return best_h, best_score

from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def _vec_grad_shift(h, k):
    """Vectorized gradient of c(k)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] = 1.0 - h[k:]
        g[k:] -= h[:n-k]
    else:
        j = -k
        g[j:] = 1.0 - h[:n-j]
        g[:n-j] -= h[j:]
    return g

def fista_softmax(h0, n, target, alphas, max_iter_per_alpha=200, deadline=None):
    """FISTA (accelerated proximal gradient) for the softmax objective."""
    h = project_feasible(h0, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        # Setup
        h_prev = h.copy()
        t_k = 1.0
        L = alpha * n  # Lipschitz constant estimate (rough)
        step = 1.0 / L
        
        for it in range(max_iter_per_alpha):
            if deadline is not None and time.time() > deadline:
                break
            
            # Compute gradient at y = h + (t_{k-1}-1)/t_k * (h - h_prev)
            t_prev = t_k
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
            y = h + (t_prev - 1.0) / t_k * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            
            corr = compute_corr(y)
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
            wt = np.exp(shifted)
            wsum = wt.sum()
            if wsum < 1e-300:
                break
            wt = wt / wsum
            
            g = np.zeros(n)
            for s in range(2 * n - 1):
                w = wt[s]
                if w < 1e-12:
                    continue
                k = s - (n - 1)
                g += w * _vec_grad_shift(y, k)
            
            h_prev = h.copy()
            h_new = h - step * g
            h = project_feasible(h_new, target)
            
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
        
        h = best_h
    
    return best_h, best_score
