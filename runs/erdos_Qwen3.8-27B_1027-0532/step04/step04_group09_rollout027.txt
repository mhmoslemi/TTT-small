    cur_score = compute_c5(h_cur)
    ones = np.where(h_cur > 0.5)[0]
    zeros = np.where(h_cur < 0.5)[0]
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        sc = compute_c5(h_cur)
        if sc < cur_score - 1e-12:
            cur_score = sc
            ones = np.where(h_cur > 0.5)[0]
            zeros = np.where(h_cur < 0.5)[0]
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
    return h_cur, cur_score


def binary_double_swap(h_bin, n_trials, deadline, rng):
    n = len(h_bin)
    h_cur = h_bin.copy()
    cur_score = compute_c5(h_cur)
    ones = np.where(h_cur > 0.5)[0]
    zeros = np.where(h_cur < 0.5)[0]
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if len(set([i1, i2, j1, j2])) < 4:
            continue
        h_cur[i1] = 0.0
        h_cur[i2] = 0.0
        h_cur[j1] = 1.0
        h_cur[j2] = 1.0
        sc = compute_c5(h_cur)
        if sc < cur_score - 1e-12:
            cur_score = sc
            ones = np.where(h_cur > 0.5)[0]
            zeros = np.where(h_cur < 0.5)[0]
        else:
            h_cur[i1] = 1.0
            h_cur[i2] = 1.0
            h_cur[j1] = 0.0
            h_cur[j2] = 0.0
    return h_cur, cur_score


def interpolate_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def perturbation_restart(h, target, rng, n_restarts, deadline, alpha_list):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        noise = rng.randn(n) * 0.15
        h_pert = project_feasible(h + noise, target)
        for alpha in alpha_list:
            if time.time() > deadline:
                break
            h_pert, sc = slsqp_one(h_pert, target, alpha, n, 800, 1e-13, deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pert.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline_final = start_time + budget_s - 30.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha (25% time)
    dl1 = start_time + (budget_s - 30.0) * 0.25
    alpha_list_1 = [20, 60, 200, 600, 2000, 6000, 20000, 60000]
    for alpha in alpha_list_1:
        if time.time() > dl1:
            break
        h_new, sc = slsqp_one(best_h, target, alpha, n, 2000, 1e-15, dl1)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Momentum active-set gradient (15% time)
    dl2 = start_time + (budget_s - 30.0) * 0.40
    h_mom, sc = momentum_active_grad(best_h, target, 4000, 0.03, 0.6, 0.03, dl2)
    if sc < best_score:
        best_score = sc
        best_h = h_mom

    # Phase 3: Binary local search (15% time)
    dl3 = start_time + (budget_s - 30.0) * 0.55
    h_bin = binary_round(best_h, target)
    h_bin, sc_bin = binary_swap_search(h_bin, 30000, dl3, rng)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin
    h_bin2, sc_bin2 = binary_double_swap(h_bin, 20000, dl3, rng)
    if sc_bin2 < best_score:
        best_score = sc_bin2
        best_h = h_bin2

    # Phase 4: Interpolate binary back and refine
    if time.time() < dl3 + (budget_s - 30.0) * 0.05:
        h_unbin = best_h + 1e-6 * rng.randn(n)
        h_unbin = project_feasible(h_unbin, target)
        dl4 = start_time + (budget_s - 30.0) * 0.65
        for alpha in [500, 2000, 10000, 50000, 200000]:
            if time.time() > dl4:
                break
            h_new, sc = slsqp_one(h_unbin, target, alpha, n, 1500, 1e-14, dl4)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 5: Multi-resolution exploration (20% time)
    dl5 = start_time + (budget_s - 30.0) * 0.85
    n_candidates = [300, 400, 500, 640]
    for n_try in n_candidates:
        if time.time() > dl5 - 120:
            break
        target_try = n_try / 2.0
        h_try = interpolate_to_n(best_h, n_try, target_try)
        sc_try = compute_c5(h_try)
        if sc_try >= best_score + 0.005:
            continue
        for alpha in [200, 1000, 5000, 20000, 80000]:
            if time.time() > dl5 - 60:
                break
            h_try, sc_try = slsqp_one(h_try, target_try, alpha, n_try, 1200, 1e-14, dl5 - 40)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try
        h_try, sc_try = momentum_active_grad(h_try, target_try, 2000, 0.02, 0.5, 0.02, dl5 - 30)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    # Phase 6: Perturbation restarts (10% time)
    dl6 = deadline_final
    h_pert, sc_pert = perturbation_restart(best_h, target, rng, 3, dl6,
                                            [2000, 10000, 50000])
    if sc_pert < best_score:
        best_score = sc_pert
        best_h = h_pert

    # Final polish
    if time.time() < deadline_final - 10:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_fin, sc_fin = momentum_active_grad(best_h, t_fin, 3000, 0.01, 0.7, 0.01, deadline_final - 2)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
