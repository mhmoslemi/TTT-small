import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int).copy()
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Verify
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            extra_ones = np.where(h_r == 1)[0]
            h_r[extra_ones[:final_sum - target_sum]] = 0
        else:
            extra_zeros = np.where(h_r == 0)[0]
            h_r[extra_zeros[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        # Distribute proportionally to room
        total_room = room.sum()
        if total_room > 0:
            h += diff * room / total_room
        else:
            idx = np.argmax(h)
            h[idx] += diff
    else:
        total_h = h.sum()
        if total_h > 0:
            h += diff * h / total_h
        else:
            h[:] = target_sum / len(h)
    return np.clip(h, 0.0, 1.0)

            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
