    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def targeted_local_search(h_init, target, max_iters, rng, time_limit, t_start, t_init=0.02, t_final=0.00005):
    """Local search with targeted moves that focus on the worst lags."""
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_imp = 0
    max_no_imp = 3000
    
    for it in range(max_iters):
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        if no_imp > max_no_imp:
            # Restart with perturbation
            n_perturb = rng.randint(2, max(3, n // 100) + 1)
            h = best_h.copy()
            for _ in range(n_perturb):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = int(ones[rng.randint(len(ones))])
                    pj = int(zeros[rng.randint(len(zeros))])
                    h[pi] = 0
                    h[pj] = 1
            h = project_binary(h, target_int).astype(np.float64)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            continue
        
        ratio = it / max(max_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        
        # Generate candidate moves
        candidates = []
        
        # Find worst lag
        worst_idx = int(np.argmax(corr))
        worst_lag = worst_idx - (n - 1)
        
        # Targeted move: try to fix the worst lag
        # Find positions contributing to worst lag
        if len(ones) > 0 and len(zeros) > 0:
            # For the worst lag, find ones that have a zero at offset worst_lag
            contrib_ones = []
            for ci in range(len(ones)):
                oi = int(ones[ci])
                pos = oi + worst_lag
                if 0 <= pos < n and h[pos] == 0:
                    contrib_ones.append(ci)
            
            if contrib_ones:
                # Pick a few candidates from contributing ones
                n_cand = min(len(contrib_ones), 5)
                for _ in range(n_cand):
                    ci = contrib_ones[rng.randint(len(contrib_ones))]
                    ii = int(ones[ci])
                    jj = int(zeros[rng.randint(len(zeros))])
                    candidates.append((ii, jj))
            
            # Also try random moves
            for _ in range(3):
                ii = int(ones[rng.randint(len(ones))])
                jj = int(zeros[rng.randint(len(zeros))])
                candidates.append((ii, jj))
        
        # Evaluate candidates and pick the best
        best_delta = 0  # We accept if delta < 0 (improvement) or with SA probability
        best_move = None
        best_new_corr = None
        best_new_c5 = cur_c5
        
        for (ii, jj) in candidates:
            if ii == jj:
                continue
            if h[ii] != 1 or h[jj] != 0:
                continue
            h[ii] = 0
            h[jj] = 1
            test_corr = np.correlate(h, 1.0 - h, mode='full')
            test_c5 = float(np.max(test_corr)) * 2.0 / n
            h[ii] = 1
            h[jj] = 0
            
            delta = test_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (ii, jj)
                best_new_corr = test_corr.astype(np.float64)
                best_new_c5 = test_c5
        
        if best_move is not None and (best_delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-best_delta / temp))):
            ii, jj = best_move
            h[ii] = 0
            h[jj] = 1
            corr = best_new_corr
            cur_c5 = best_new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
    
    return best_h, best_c5


def smooth_max_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Soft-max approximation
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    soft_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(soft_max * dx)


def continuous_optimize(h_init, n, target, time_limit, t_start):
    """Use SLSQP to optimize in continuous space."""
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [30, 100, 300, 800, 2000, 5000]
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            # Use SLSQP with the soft-max objective
            def obj(x):
                return smooth_max_objective(x, alpha, n)
            
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score


def make_qr_sequence(n, rng):
    """Quadratic residue sequence."""
    def is_prime(x):
        if x < 2: return False
        if x < 4: return True
        if x % 2 == 0 or x % 3 == 0: return False
        i = 5
        while i * i <= x:
            if x % i == 0 or x % (i+2) == 0: return False
            i += 6
        return True
    
    target = n // 2
    p = n + 1
    while not is_prime(p):
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h


def make_msequence(n, rng):
    """Maximum length sequence."""
    h = np.zeros(n)
    period = 1
    for i in range(n):
        period = (2 * period + 1) % (2 * n + 1)
        h[i] = 1.0 if (period & 1) == 1 else 0.0
    current = int(h.sum())
    target = n // 2
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

Δcount_s = -h[a+s] + h[a-s] + (1-h[b+s]) - h[b-s] (with validity checks)

For a swap (a→0, b→1):
Δcorr_s = (h[a+s] - 1) + h[a-s] + (1 - h[b+s]) - h[b-s]
 = h[a+s] + h[a-s] - h[b+s] - h[b-s] - 1

(all terms with invalid indices treated as 0)

This is O(1) per lag s, so O(n) per swap. Much faster than O(n²) for the full correlation.

But we need the NEW max, not just the change at one lag. The new correlation at lag s is:
new_corr_s = old_corr_s + Δcorr_s (approximately, for the specific swap)

Wait, this is exact for binary h! Because the correlation is just a count, and the change from a single swap is exactly as computed.

So the algorithm becomes:
1. For each candidate swap (a, b):
   a. Compute Δcorr_s for all s (O(n) operations)
   b. new_corr = old_corr + Δcorr
   c. new_max = max(new_corr)
2. Pick the swap with the smallest new_max

This is O(n) per candidate instead of O(n²). Much faster!

def evaluate_swap(corr, h, a, b, n):
    """Evaluate the effect of swapping h[a] (1→0) and h[b] (0→1) on the correlation.
    Returns the new correlation array and new max."""
    # corr has length 2n-1, with index n-1+s corresponding to lag s
    # corr[n-1+s] = sum_i h[i]*(1-h[i+s]) for valid i

    # Δcorr_s = (h[a+s]-1) * [a+s valid] + h[a-s] * [a-s valid] 
    #         + (1-h[b+s]) * [b+s valid] - h[b-s] * [b-s valid]

    for s in range(-(n-1), n):
        if s == 0:
            continue
        # h[a] was 1, now 0
        # term h[a+s]: if valid, contributes h[a+s] - 1 (was (1-h[a+s]) * h[a] = 1-h[a+s], now 0)

        #   i=a-s: h[a-s]*(1-h[a]) = h[a-s]*0 = 0 if a-s valid → change: +h[a-s]*(1-0) = h[a-s]

        # Change: -(1-h[a+s]) = h[a+s]-1
        
        # The term i=a-s contributes h[a-s]*(1-h[a]) to corr_s (when a-s is in [0,n))
        # Before: h[a-s]*(1-1) = 0, After: h[a-s]*(1-0) = h[a-s]
        # Change: +h[a-s]
        
        # Similarly for b:
        # The term i=b: Before 0, After (1-h[b+s])
        # Change: +(1-h[b+s])
        # The term i=b-s: Before h[b-s], After 0
        # Change: -h[b-s]

    # Better approach: vectorized
    # For all lags s from -(n-1) to (n-1):
    # delta_s = 0
    # if 0 <= a+s < n: delta_s += h[a+s] - 1
    # if 0 <= a-s < n: delta_s += h[a-s]
    # if 0 <= b+s < n: delta_s += 1 - h[b+s]
    # if 0 <= b-s < n: delta_s += -h[b-s]
    
    # Vectorized version:
    s = np.arange(-(n-1), n)  # all possible lags
    
    # Terms from position a (was 1, now 0):
    # (a+s) valid: 0 <= a+s < n
    mask_a_plus = (a + s >= 0) & (a + s < n)
    mask_a_minus = (a - s >= 0) & (a - s < n)

    new_corr = corr + delta
    return new_corr, float(np.max(new_corr))
