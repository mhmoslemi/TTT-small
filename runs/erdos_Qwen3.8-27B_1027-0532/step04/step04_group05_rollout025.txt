        else:
            stuck += 1

    return best_h, best_c5


def momentum_subgradient(h_init, n, target, max_steps, time_limit,
                         n_top=15, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        top_vals = overlaps[top_idx]

        w = 0.85 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            cv = 0.0
            if k >= 0:
                idx1 = idxs[idxs + k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 + k])
            else:
                idx1 = idxs[idxs - k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 - k])

            if k >= 0:
                idx2 = idxs[idxs >= k]
                grad[idx2] += w[i] * (-y[idx2 - k])
            else:
                idx2 = idxs[idxs + k >= 0]
                grad[idx2] += w[i] * (-y[idx2 + k])

        grad *= dx

        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break

        c5_y = float(np.max(overlaps))
        polyak_lr = (c5_y - best_c5) / (norm_grad ** 2 + 1e-20)
        lr = min(max(0.01, polyak_lr), 0.5)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try

            if step % 500 == 0:
                for lr2 in [lr * 1.5, lr * 0.5, lr * 2.0, lr * 0.3]:
                    h_try2 = h - lr2 * grad
                    h_try2 = project_feasible(h_try2, target)
                    sc2 = compute_c5(h_try2)
                    if sc2 < best_c5 - 1e-14:
                        best_c5 = sc2
                        best_h = h_try2.copy()
                        h = h_try2
                        break
        else:
            h_prev = h.copy()

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def shake_and_restore(h, n, target, rng, noise_level, n_shake):
    h_shake = h.copy()
    for _ in range(n_shake):
        mask = rng.random(n) < 0.3
        h_shake[mask] += rng.randn(mask.sum()) * noise_level
        h_shake = project_feasible(h_shake, target)
    return h_shake


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Multi-resolution LP (coarse to fine)
    if tleft() > 200:
        for n_coarse in [100, 200]:
            if tleft() < 80:
                break
            target_c = n_coarse / 2.0
            h_c = interpolate_to(best_h, n_coarse, target_c)
            lp_time = min(60, tleft() * 0.2)
            h_c, c5_c = alternating_lp_refined(h_c, n_coarse, target_c,
                                                max_iter=2000, time_limit=lp_time,
                                                n_top=100, damping_start=0.9,
                                                damping_end=0.75, reg_start=1e-3,
                                                reg_end=1e-7)
            h_fine = interpolate_to(h_c, n, target)
            sc = compute_c5(h_fine)
            if sc < best_score:
                best_score = sc
                best_h = h_fine

    # Phase 2: Refined LP on full resolution
    if tleft() > 150:
        lp_time = min(200, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_refined(best_h, n, target,
                                              max_iter=5000, time_limit=lp_time,
                                              n_top=160, damping_start=0.95,
                                              damping_end=0.85, reg_start=1e-4,
                                              reg_end=1e-8)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: Momentum subgradient
    if tleft() > 100:
        sg_time = min(120, tleft() * 0.25)
        h_sg, c5_sg = momentum_subgradient(best_h, n, target,
                                            max_steps=50000, time_limit=sg_time,
                                            n_top=20, momentum=0.9)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: SLSQP with aggressive alpha
    if tleft() > 120:
        h_ref = slsqp_optimize(best_h, n, target,
                                [50, 200, 1000, 5000, 20000, 100000],
                                maxiter=2000, ftol=1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: LP again after SLSQP
    if tleft() > 120:
        lp_time = min(100, tleft() * 0.3)
        h_lp2, c5_lp2 = alternating_lp_refined(best_h, n, target,
                                                max_iter=3000, time_limit=lp_time,
                                                n_top=140, damping_start=0.9,
                                                damping_end=0.8, reg_start=1e-5,
                                                reg_end=1e-9)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 6: Shake + re-optimize (multiple attempts)
    for attempt in range(5):
        if tleft() < 80:
            break
        noise = 0.03 + attempt * 0.02
        h_shake = shake_and_restore(best_h, n, target, rng, noise, n_shake=3)
        # Quick LP
        h_sh, c5_sh = alternating_lp_refined(h_shake, n, target,
                                              max_iter=1000, time_limit=40,
                                              n_top=120, damping_start=0.9,
                                              damping_end=0.7, reg_start=1e-3,
                                              reg_end=1e-6)
        if c5_sh < best_score:
            best_score = c5_sh
            best_h = h_sh
        else:
            # Try SLSQP from shaken point
            if tleft() > 60:
                h_rs = slsqp_optimize(h_shake, n, target,
                                       [100, 500, 2000, 10000],
                                       maxiter=1500, ftol=1e-14)
                sc = compute_c5(h_rs)
                if sc < best_score:
                    best_score = sc
                    best_h = h_rs

    # Phase 7: Second momentum pass
    if tleft() > 80:
        sg_time = min(80, tleft() * 0.3)
        h_sg2, c5_sg2 = momentum_subgradient(best_h, n, target,
                                              max_steps=50000, time_limit=sg_time,
                                              n_top=25, momentum=0.85)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    # Phase 8: Final SLSQP polish
    if tleft() > 50:
        h_final_try = slsqp_optimize(best_h, n, target,
                                      [500, 2000, 10000, 50000, 200000],
                                      maxiter=1500, ftol=1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 9: Final LP polish
    if tleft() > 40:
        h_lp3, c5_lp3 = alternating_lp_refined(best_h, n, target,
                                                max_iter=2000, time_limit=min(35, tleft() * 0.8),
                                                n_top=100, damping_start=0.95,
                                                damping_end=0.9, reg_start=1e-6,
                                                reg_end=1e-10)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 10: Try higher resolution
    if tleft() > 150:
        for n_high in [500, 600]:
            if tleft() < 100:
                break
            target_h = n_high / 2.0
            h_hi = interpolate_to(best_h, n_high, target_h)
            h_hi, c5_hi = alternating_lp_refined(h_hi, n_high, target_h,
                                                  max_iter=2000, time_limit=min(60, tleft() * 0.3),
                                                  n_top=150, damping_start=0.9,
                                                  damping_end=0.8, reg_start=1e-4,
                                                  reg_end=1e-7)
            h_hi = slsqp_optimize(h_hi, n_high, target_h,
                                   [100, 500, 2000, 10000],
                                   maxiter=1000, ftol=1e-14)
            sc = compute_c5(h_hi)
            if sc < best_score:
                best_score = sc
                best_h = h_hi
                n = n_high
                target = n_high / 2.0
                break

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
