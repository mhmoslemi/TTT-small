        h = np.clip(h, 0.0, 1.0)
    return h

        h_cur = project_feasible(res.x, target)
        s = np.sum(h_cur) - target

        if abs(s) < 1e-10:
            mu = max(mu * 0.5, 1e-6)
        else:
            mu = min(mu * 2.0, 1e6)
    
    return h_cur, compute_c5(h_cur)

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    h_r, sc = lbfgsb_augmented(best_h, target2, 50000, 4, 200.0, 500, dl6)
    if sc < best_score:
        best_score = sc
        best_h = h_r
    h_r, sc = lbfgsb_augmented(best_h, target2, 150000, 4, 200.0, 600, dl6)
    if sc < best_score:
        best_score = sc
        best_h = h_r

    n3 = 1200
    target3 = n3 / 2.0
    h3 = interpolate_to(best_h, n3, target3)
    sc3 = compute_c5(h3)
    if sc3 < best_score:
        best_score = sc3
        best_h = h3

    dl7 = start_time + budget_s * 0.78
    h3r, sc3 = lbfgsb_augmented(h3, target3, 10000, 4, 200.0, 500, dl7)
    if sc3 < best_score:
        best_score = sc3
        best_h = h3r
    h3r, sc3 = lbfgsb_augmented(best_h, target3, 40000, 4, 300.0, 600, dl7)
    if sc3 < best_score:
        best_score = sc3
        best_h = h3r

    dl8 = start_time + budget_s * 0.88
    h_sg3, sc = subgrad_polish(best_h, target3, 6000, 0.02, 0.01, dl8)
    if sc < best_score:
        best_score = sc
        best_h = h_sg3

    n_now = len(best_h)
    target_now = n_now / 2.0
    h_r, sc = lbfgsb_augmented(best_h, target_now, 100000, 3, 500.0, 400, deadline_final)
    if sc < best_score:
        best_score = sc
        best_h = h_r

    if time.time() < deadline_final - 30:
        h_sg4, sc = subgrad_polish(best_h, target_now, 3000, 0.01, 0.005, deadline_final)
        if sc < best_score:
            best_score = sc
            best_h = h_sg4

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
