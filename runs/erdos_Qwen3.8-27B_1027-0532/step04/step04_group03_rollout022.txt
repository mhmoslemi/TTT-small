import numpy as np
from scipy.optimize import minimize
import time

_def_n = 0
_def_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def logsumexp_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def logsumexp_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    dx = 2.0 / n
    for k in range(len(corr)):
        w = (weights[k] / w_sum) * dx
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad

def slsqp_optimize(h_start, target, time_limit, start_time, alpha_schedule):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    for alpha in alpha_schedule:
        if (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(
                logsumexp_obj, h, args=(alpha, n, target),
                jac=lambda x: logsumexp_grad(x, alpha, n, target),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def active_set_optimize(h_start, target, time_limit, start_time):
    """Optimize focusing on active (near-max) correlations"""
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for alpha in [200, 500, 1000, 2000, 5000, 10000, 20000]:
        if (time.time() - start_time) > time_limit:
            break
        
        # Identify active set
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = float(np.max(corr))
        threshold = c_max - 0.001  # Lags within 0.001 of max
        active = np.where(corr >= threshold)[0]
        
        # Build objective with only active lags (weighted)
        # Use equal weights for all active lags
        weights = np.ones(len(active)) / len(active)
        
        def obj_active(x):
            h_t = project_feasible(x, target)
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full') * dx
            return float(np.sum(weights * corr_t[active]))
        
        def grad_active(x):
            h_t = project_feasible(x, target)
            grad = np.zeros(n)
            for idx, k in enumerate(active):
                w = weights[idx] * dx
                m = k - (n - 1)
                if m >= 0:
                    grad[m:] += w * (1.0 - h_t[:n - m])
                    grad[:n - m] -= w * h_t[m:]
                else:
                    mm = -m
                    grad[:n - mm] += w * (1.0 - h_t[mm:])
                    grad[mm:] -= w * h_t[:n - mm]
            return grad
        
        def con_fun(x):
            return np.sum(x) - target
        
        def con_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(
                obj_active, h, jac=grad_active,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def gradient_coord_descent(h_start, target, time_limit, start_time):
    """Coordinate descent using gradient of the max correlation"""
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    lr = 0.1
    
    for outer in range(200):
        if (time.time() - start_time) > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - 0.0005)[0]
        
        # Compute gradient of average of active correlations
        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            if m >= 0:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
            else:
                mm = -m
                grad[:n - mm] += (1.0 - h[mm:])
                grad[mm:] -= h[:n - mm]
        grad /= len(active)
        
        # Find the position with largest negative gradient (most beneficial to increase)
        # and largest positive gradient (most beneficial to decrease)
        
        # Try a few steps
        improved = False
        for step_mult in [1.0, 0.5, 0.25, 0.1]:
            h_try = project_feasible(h - lr * step_mult * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        
        if not improved:
            lr *= 0.8
            if lr < 1e-8:
                break
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)

            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0

        if need <= len(zeros):

            h_r[zeros[order]] = 1

def slsqp_optimize(h_start, target, time_limit, start_time, alpha_schedule):
    global _g_n, _g_target, _g_alpha
    n = len(h_start)
    _g_n = n
    _g_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    
    for alpha in alpha_schedule:
        if (time.time() - start_time) > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(
                logsumexp_obj, h, jac=logsumexp_grad,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0
_g_active = None
_g_weights = None


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)


def eq_con_fun(x):
    return np.sum(x) - _g_target


def eq_con_jac(x):
    return np.ones(_g_n)


def lse_obj(x):
    alpha = _g_alpha
    n = _g_n
    target = _g_target
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def lse_grad(x):
    alpha = _g_alpha
    n = _g_n
    target = _g_target
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    dx = 2.0 / n
    for k in range(len(corr)):
        w = (weights[k] / w_sum) * dx
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad


def active_obj(x):
    n = _g_n
    target = _g_target
    dx = 2.0 / n
    active = _g_active
    weights = _g_weights
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(weights * corr[active]))


def active_grad(x):
    n = _g_n
    target = _g_target
    dx = 2.0 / n
    active = _g_active
    weights = _g_weights
    h = project_feasible(x, target)
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        w = weights[idx] * dx
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad

                        i1 = ones_cache[rng.randint(len(ones_cache))]
                        i0 = zeros_cache[rng.randint(len(zeros_cache))]

                        h[i0] = 1

                zeros_cache = np.where(h == 0)[0]
                current_c5 = compute_c5(h)
            else:
                break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        move_type = rng.randint(6)
        h_save = h.copy()
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]

            h[i0] = 1

            h[zeros_cache[idx0[0]]] = 1

        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:

def slsqp_optimize(h_start, target, time_limit, start_time, alpha_schedule):
    global _g_n, _g_target, _g_alpha
    n = len(h_start)
    _g_n = n
    _g_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]

    for alpha in alpha_schedule:
        if (time.time() - start_time) > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(
                lse_obj, h, jac=lse_grad,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5


def active_set_optimize(h_start, target, time_limit, start_time):

def gradient_coord_descent(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = 0.05

    for outer in range(1000):
        if (time.time() - start_time) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - 0.0003)[0]

        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            if m >= 0:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
            else:
                mm = -m
                grad[:n - mm] += (1.0 - h[mm:])
                grad[mm:] -= h[:n - mm]
        grad /= len(active)

        improved = False
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * step_mult * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                break

        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_h, best_c5


def surgical_fractional(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for outer in range(50):
        if (time.time() - start_time) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - 0.0002)[0]

        best_improvement = 0.0
        best_pos = -1
        best_val = -1.0

        for pos in range(n):
            if h[pos] <= 0.001 or h[pos] >= 0.999:
                continue
            for frac in [h[pos] - 0.1, h[pos] - 0.05, h[pos] + 0.05, h[pos] + 0.1]:
                if frac < 0 or frac > 1:
                    continue
                h[pos] = frac
                h_proj = project_feasible(h, target)
                c5_try = compute_c5(h_proj)
                imp = best_c5 - c5_try
                if imp > best_improvement:
                    best_improvement = imp
                    best_pos = pos
                    best_val = frac
                h[pos] = h_start[min(pos, n-1)] if False else h[pos]

        if best_pos >= 0 and best_improvement > 1e-10:
            h[best_pos] = best_val
            h = project_feasible(h, target)
            best_c5 = compute_c5(h)
            best_h = h.copy()
        else:
            break

    return best_h, best_c5
