import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def find_optimal_t(c1, b, lam):
    """Find t in [0,1] minimizing max_L (c1_L*t + b_L) + lam*t"""
    # Check boundaries
    vals_0 = b
    f0 = np.max(vals_0)
    active_0 = vals_0 >= f0 - 1e-12
    if np.max(c1[active_0] + lam) >= 0:
        return 0.0
    
    vals_1 = c1 + b
    f1 = np.max(vals_1)
    active_1 = vals_1 >= f1 - 1e-12
    if np.min(c1[active_1] + lam) <= 0:
        return 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = c1 * mid + b
        f_mid = np.max(vals)
        active = vals >= f_mid - 1e-12
        left_d = np.min(c1[active] + lam)
        right_d = np.max(c1[active] + lam)
        
        if left_d > 1e-13:
            hi = mid
        elif right_d < -1e-13:
            lo = mid
        else:
            break
    
    return (lo + hi) * 0.5

def coord_descent_pass(h, target, lam=0.0, order=None):
    """One pass of coordinate descent."""
    n = len(h)
    n_corr = 2 * n - 1
    corr_full = np.correlate(h, 1.0 - h, mode='full').copy()
    s = np.arange(-(n-1), n)
    mask_pos = s >= 0
    mask_neg = s < 0
    s_pos = s[mask_pos]
    m_neg = -s[mask_neg]
    
    if order is None:
        order = list(range(n))
    
    for i in order:
        # Compute c1 for coordinate i
        c1 = np.zeros(n_corr)
        
        # s >= 0 part
        ip = i + s_pos
        valid1 = ip < n
        c1_pos = np.zeros(len(s_pos))
        if np.any(valid1):
            c1_pos[valid1] = 1.0 - h[ip[valid1]]
        im = i - s_pos
        valid2 = im >= 0
        if np.any(valid2):
            c1_pos[valid2] += -h[im[valid2]]
        c1[mask_pos] = c1_pos
        
        # s < 0 part
        im2 = i - m_neg
        valid3 = (im2 >= 0) & (im2 <= n - 1 - m_neg)
        c1_neg = np.zeros(len(m_neg))
        if np.any(valid3):
            c1_neg[valid3] = 1.0 - h[im2[valid3]]
        valid4 = i <= n - 1 - m_neg
        if np.any(valid4):
            c1_neg[valid4] += -h[i + m_neg[valid4]]
        c1[mask_neg] = c1_neg
        
        # Find optimal t
        b = corr_full - c1 * h[i]
        t_new = find_optimal_t(c1, b, lam)
        
        # Update
        delta = t_new - h[i]
        if abs(delta) > 1e-14:
            h[i] = t_new
            corr_full += c1 * delta
    
    return h

def lse_obj_grad(h, alpha, topk, penalty, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    n = len(h)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    s = np.sum(h) - target
    obj += penalty * s * s
    weights = exp_vals / total
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * omh[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * omh[m:]
                grad[m:] -= w * h[:n - m]
    grad += 2.0 * penalty * s
    return obj, grad
