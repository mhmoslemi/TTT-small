def compute_c5(h):
    n = len(h)

    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, k, dx):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif k > 0:
        grad[:n-k] += (1.0 - h[k:]) * dx
        grad[k:] -= h[:n-k] * dx
    else:
        m = -k
        grad[m:] += (1.0 - h[:n-m]) * dx
        grad[:n-m] -= h[m:] * dx
    return grad

def smoothed_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    n_half = n - 1
    grad = np.zeros(n)
    significant = np.where(weights > 1e-8)[0]
    for si in significant:
        k = int(si) - n_half
        grad += weights[si] * grad_for_lag(h, k, dx)
    return grad
