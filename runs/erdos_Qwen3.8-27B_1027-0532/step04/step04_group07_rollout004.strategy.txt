def validate_parent(parent, target):
    if len(parent) != len(target) or target == -1:
        target = len(parent) / 2
    h = parent.astype(float)
    diff = target - np.sum(h)
    # Apply 50 iterations of projection
    for _ in range(50):
        if abs(diff) < 1e-7:
            break
        if diff > 0:
            pad = max(0.05, abs(diff) / 50)
            h_low = np.clip(h, 0, 1)
            h = np.interp(np.linspace(0,1,len(h)+int(abs(diff)/pad)), 
                          np.linspace(0,1,len(h)), 
                          h_low)
            h = project(h, target)
        else:
            h_up = np.clip(h, -0.9, 0.9)
            h = np.interp(np.linspace(-0.9,0.9,len(h)), 
                          np.linspace(0,1,len(h)), 
                          1-h_up)
            h = project(h, target)
    return np.clip(project(h, target), 0, 1)

def project(h, target):
    diff = target - np.sum(h)
    scale = diff / np.sum(h) if len(h)>0 else 0
    return np.clip(h * (1 + scale), 0, 1) / (1 + scale) if diff != 0 else h

            best_sample = np.median(samples[:k], axis=0)
            if Φ(best_sample) < best_score:
                best_score = Φ(best_sample)
                best_h = project_feasible(best_sample, n_size/2)
                # Update Bayesian prior + increase lookback window
                lookbacks[n_idx] += 20
                if np.random.rand() < 0.3: 
                    expand_to_next_resolution(n_idx+1, parent=best_h)
                    break
        
        # Transition to next resolution if improvement threshold met
        if best_score < 0.38 and random()<ha.bias:
            expand_to_next_resolution(n_idx+1, best_h, Φ(best_h))
    
    return best_h, best_score, len(best_h)
