import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:2*n-1])) * dx

def compute_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2*n-1]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_obj(x, n, target, alpha):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    O = corr[n-1:2*n-1]
    c_max = float(np.max(O))
    shifted = alpha * (O - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_refine(h, n, target, alphas, maxiter=3000):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        def obj(x):
            return _lse_obj(x, n, target, a)
        def con_fun(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_polish(h, n, target, n_steps=500, n_active=3, lr_init=0.003, lr_decay=0.997):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        O = compute_overlap(h_cur)
        c_curr = float(np.max(O)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(O)[::-1][:n_active]
        diffs = O[top_idx] - O[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best_h = h_r

    if tleft() > 100:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=500, n_active=5,
                                         lr_init=0.004, lr_decay=0.997)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_cd = coord_descent(best_h, n, target, n_sweeps=150, deadline=start_time + budget_s * 0.88)
        sc = compute_c5(h_cd)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 60:
        deadline_final = start_time + budget_s * 0.95
        h_cd = coord_descent(best_h, n, target, n_sweeps=300, mortgage=0.8, deadline=deadline_final)
        sc = compute_c5(h_cd)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 30:
        h_sls = slsqp_refine(best_h, n, target, [5000, 15000, 40000], maxiter=2000)
        sc = compute_c5(h_sls)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
