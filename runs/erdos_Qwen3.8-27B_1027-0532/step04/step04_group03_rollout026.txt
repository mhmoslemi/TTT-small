import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= current_sum:
            ones = np.where(h_r == 1)[0]
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= (len(h_r) - current_sum):
            zeros = np.where(h_r == 0)[0]
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    diff = h_r.sum() - target_sum
    if diff != 0:
        if diff > 0:
            ones = np.where(h_r == 1)[0]
            h_r[ones[:int(diff)]] = 0
        else:
            zeros = np.where(h_r == 0)[0]
            h_r[zeros[:int(-diff)]] = 1
    return h_r.astype(float)

def smooth_objective(h, alpha, mu, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smooth_val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = mu * (np.sum(h) - target) ** 2
    return smooth_val + penalty

def smooth_gradient(h, alpha, mu, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    grad_out = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            if m == 0:
                grad_out += w * dx * (1.0 - 2.0 * h)
            else:
                grad_out[:n - m] += w * dx * (1.0 - h[m:])
                grad_out[m:] -= w * dx * h[:n - m]
        else:
            mm = -m
            if mm == 0:
                grad_out += w * dx * (1.0 - 2.0 * h)
            else:
                grad_out[mm:] += w * dx * (1.0 - h[:n - mm])
                grad_out[:n - mm] -= w * dx * h[mm:]
    grad_out += 2.0 * mu * (np.sum(h) - target)
    return grad_out

def l_bfgsb_refine(h_start, target, time_limit, start_time, max_alpha=15000):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    alpha_schedule = [30, 60, 120, 250, 500, 1000, 2000, 5000, max_alpha]
    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit:
            break
        mu = 500.0 * (ai + 1)
        def obj_fn(x):
            return smooth_objective(x, alpha, mu, n, target)
        def grad_fn(x):
            return smooth_gradient(x, alpha, mu, n, target)
        max_iter = max(100, 800 - ai * 50)
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iter, 'gtol': 1e-9, 'disp': False})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
        if time.time() - start_time > time_limit:
            break
    return best_h, best_c5

def fast_corr_update(h, j, new_val, corr):
    n = len(h)
    old_val = h[j]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    n_corr = len(corr)
    new_corr = corr.copy()
    for k in range(n_corr):
        m = k - (n - 1)
        if m >= 0:
            contrib = 0.0
            if j <= n - 1 - m:
                contrib += (1.0 - h[j + m])
            if j >= m:
                contrib -= h[j - m]
            new_corr[k] += delta * contrib
        else:
            mm = -m
            contrib = 0.0
            if j >= mm:
                contrib += (1.0 - h[j - mm])
            if j <= n - 1 - mm:
                contrib -= h[j + mm]
            new_corr[k] += delta * contrib
    return new_corr

def efficient_coord_descent(h_start, target, time_limit, start_time, n_passes_max=50):
    n = len(h_start)

    h = h_start.copy().astype(float)

                              0.6, 0.7, 0.8, 0.85, 0.9, 0.95, 0.98, 0.99, 1.0])
    for pass_num in range(n_passes_max):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:

            old_val = h[j]

                    corr = new_corr
                    best_c5 = new_max
                    improved = True

        if not improved:

    h = project_feasible(h, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_step(h, target, top_k, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    active = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        if m >= 0:
            if m == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] -= h[:n - m]
        else:
            mm = -m
            if mm == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[mm:] += (1.0 - h[:n - mm])
                grad[:n - mm] -= h[mm:]
    return grad / len(active), c_max

def nesterov_subgradient(h_start, target, time_limit, start_time, top_k=30):
    n = len(h_start)
    h = h_start.copy().astype(float)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = 0.5
    for t in range(1, 200000):
        if time.time() - start_time > time_limit:
            break
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) ** 2)) / 2.0
        momentum = (t - 1) / beta
        step = t0 / (1.0 + 0.1 * t)
        grad, c_max = subgradient_step(y, target, top_k, n)
        v_new = momentum * v - step * grad
        y_new = y + v_new
        y_new = project_feasible(y_new, target)
        h = y_new
        y = y_new
        v = v_new
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if t % 500 == 0 and best_c5 - c5_now > 0.01:
            t0 *= 0.5
    return best_h, best_c5

        h_sa, c5_sa = sa_optimize_binary(h_bin, target, 80000, 0.02, 0.00005,
                                         rng, min(80, tleft() * 0.15), start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 2: Main L-BFGS-B refinement (most important)
    if tleft() > 150:
        t_allow = min(350.0, tleft() * 0.45)
        h_ref, c5_ref = l_bfgsb_refine(best_h, target, t_allow, start_time, max_alpha=15000)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref.copy()

    # Phase 3: Nesterov subgradient for continuous refinement
    if tleft() > 100:
        t_allow = min(150.0, tleft() * 0.25)
        h_ng, c5_ng = nesterov_subgradient(best_h, target, t_allow, start_time, top_k=40)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng.copy()

    # Phase 4: Second L-BFGS-B round with higher alpha
    if tleft() > 100:
        t_allow = min(200.0, tleft() * 0.35)
        h_ref2, c5_ref2 = l_bfgsb_refine(best_h, target, t_allow, start_time, max_alpha=30000)
        if c5_ref2 < best_score:
            best_score = c5_ref2
            best_h = h_ref2.copy()

    # Phase 5: Fast coordinate descent for final polishing
    if tleft() > 40:
        t_allow = min(120.0, tleft() * 0.4)
        h_cd, c5_cd = efficient_coord_descent(best_h, target, t_allow, start_time, n_passes_max=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Try different resolutions
    n_candidates = [int(n * f) for f in [0.85, 0.9, 1.1, 1.2, 1.4]]
    n_candidates = sorted(set(nc for nc in n_candidates if nc >= 200 and nc % 2 == 0 and nc != n))
    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        if tleft() > 150:
            t_allow = min(120.0, tleft() * 0.4)
            h_r, c5_r = l_bfgsb_refine(h_interp, target_try, t_allow, start_time, max_alpha=10000)
            if c5_r < best_score:
                best_score = c5_r
                best_h = h_r.copy()
        if tleft() > 60:
            t_allow2 = min(60.0, tleft() * 0.3)
            h_r2, c5_r2 = efficient_coord_descent(best_h, len(best_h) / 2.0, t_allow2, start_time, n_passes_max=20)
            if c5_r2 < best_score:
                best_score = c5_r2
                best_h = h_r2.copy()

    # Phase 7: Final L-BFGS-B with very high alpha
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t_allow = min(80.0, tleft() * 0.6)
        h_final_ref, c5_final = l_bfgsb_refine(best_h, target_use, t_allow, start_time, max_alpha=50000)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_ref.copy()

    # Phase 8: Final coordinate descent
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t_allow = min(30.0, tleft() - 5)
        h_cd2, c5_cd2 = efficient_coord_descent(best_h, target_use, t_allow, start_time, n_passes_max=20)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    n_final = len(h_final)
    return h_final, final_score, n_final
