def project_feasible(h, target_sum):

        return h

def compute_grad_lag(h, lag):

    if lag >= 0:
        m = n - lag

            grad[:m] += (1.0 - h[lag:lag+m])
            grad[lag:lag+m] -= h[:m]
    else:
        m = -lag

    return grad

            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:lag + m])
            grad[lag:lag + m] -= h[:m]
    else:
        m = -lag
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smoothmax_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def smoothmax_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    n = len(x)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if weights[i] < 1e-15:
            continue
        lag = top_idx[i] - (n - 1)
        grad += weights[i] * compute_grad_lag(x, lag)
    return grad


def slsqp_smoothmax(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                smoothmax_obj, h_cur, args=(alpha, topk),
                jac=smoothmax_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def subgradient_run(h_init, target_sum, max_iter, lr, topk, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    stagnation = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
            top_vals = corr[top_idx]
        else:
            top_idx = np.arange(n_corr)
            top_vals = corr
        w = np.exp(top_vals - np.max(top_vals))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-15:
                continue
            grad += w[i] * compute_grad_lag(h, top_idx[i] - (n - 1))
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
            lr *= 0.995
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def lp_refinement(h_start, target_sum, max_iterations, topk, time_limit=None, tol=1e-14):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    for it in range(max_iterations):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        m = len(top_idx)
        A = np.zeros((m, n + 1))
        b = np.zeros(m)
        for i in range(m):
            g = compute_grad_lag(h, top_idx[i] - (n - 1))
            A[i, :n] = g
            A[i, n] = -1.0
            b[i] = g @ h - corr[top_idx[i]]
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        try:
            res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        if res.success:
            h_lp = np.clip(res.x[:n], 0.0, 1.0)
            h_lp = project_feasible(h_lp, target_sum)
            score = compute_c5(h_lp)
            if score < best_score - tol:
                best_score = score
                best_h = h_lp.copy()
                h = h_lp
                stagnation = 0
            else:
                h_half = project_feasible(0.5 * h + 0.5 * h_lp, target_sum)
                sc_half = compute_c5(h_half)
                if sc_half < best_score - tol:
                    best_score = sc_half
                    best_h = h_half.copy()
                    h = h_half
                    stagnation = 0
                else:
                    h_quarter = project_feasible(0.9 * h + 0.1 * h_lp, target_sum)
                    sc_q = compute_c5(h_quarter)
                    if sc_q < best_score - tol:
                        best_score = sc_q
                        best_h = h_quarter.copy()
                        h = h_quarter
                        stagnation = 0
                    else:
                        stagnation += 1
                        if stagnation > 20:
                            break
        else:
            break
    return best_h, best_score


def dual_mw(h_init, target_sum, n_outer, n_inner, gamma, topk_init, time_limit=None):
    n = len(h_init)
    n_corr = 2 * n - 1
    h = h_init.copy()
    t0 = time.time()
    corr = overlap_corrs(h)
    top_idx = np.argpartition(corr, -topk_init)[-topk_init:]
    p = np.zeros(n_corr)
    p[top_idx] = 1.0 / float(topk_init)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(n_outer):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        for s in range(n_inner):
            grad = np.zeros(n)
            for k in range(n_corr):
                if p[k] < 1e-12:
                    continue
                grad += p[k] * compute_grad_lag(h, k - (n - 1))
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            eta = min(0.5 / gnorm, 0.5)
            h_new = project_feasible(h - eta * grad, target_sum)
            obj_new = float(np.dot(p, overlap_corrs(h_new)))
            obj_old = float(np.dot(p, overlap_corrs(h)))
            if obj_new >= obj_old:
                h_new = project_feasible(h - 0.1 * eta * grad, target_sum)
            h = h_new
        corr = overlap_corrs(h)
        c_max = np.max(corr)
        p = p * np.exp(-gamma * (corr - c_max))
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(n_corr) / float(n_corr)
        else:
            p = p / p_sum
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def structured_perturb(h, target_sum, topk, scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    corr = overlap_corrs(h)
    top_idx = np.argpartition(corr, -topk)[-topk:]
    pert = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            pert[:m] -= scale * (1.0 - h[lag:lag + m])
            pert[lag:lag + m] += scale * h[:m]
        else:
            m = -lag
            pert[m:] -= scale * (1.0 - h[:n - m])
            pert[:n - m] += scale * h[m:]
    disp = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            disp[:m] += (1.0 - h[lag:lag + m])
            disp[lag:lag + m] -= h[:m]
        else:
            m = -lag
            disp[m:] += (1.0 - h[:n - m])
            disp[:n - m] -= h[m:]
    if np.max(np.abs(disp)) > 1e-8:
        pert = pert / np.max(np.abs(disp)) * min(scale, 0.01)
    h_new = h - pert + rng.randn(n) * scale * 0.05
    return project_feasible(h_new, target_sum)


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Subgradient exploration with multiple configs
    sg_configs = [
        (3000, 0.12, 8, 300, 0.88),
        (2500, 0.06, 5, 250, 0.90),
        (2000, 0.20, 12, 200, 0.85),
        (2000, 0.04, 3, 200, 0.92),
    ]
    for i, cfg in enumerate(sg_configs):
        if _remaining() < 40:
            break
        h_try = best_h + rng.randn(n_init) * 0.003
        h_try = project_feasible(h_try, target_init)
        h_new, score = subgradient_run(h_try, target_init,
                                        max_iter=cfg[0], lr=cfg[1], topk=cfg[2],
                                        patience=cfg[3], momentum=cfg[4], seed=seed + i * 137)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: LP refinement (broad)
    if _remaining() > 120:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=400, topk=100,
                                       time_limit=_remaining() - 100)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 3: Dual multiplicative weights
    if _remaining() > 100:
        budget_mw = min(_remaining() - 80, 200)
        h_new, score = dual_mw(best_h, target_init,
                                n_outer=80, n_inner=4, gamma=5.0,
                                topk_init=20, time_limit=budget_mw)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 60:
            h_new2, score2 = dual_mw(best_h, target_init,
                                      n_outer=60, n_inner=6, gamma=10.0,
                                      topk_init=10, time_limit=_remaining() - 50)
            if score2 < best_score:
                best_score = score2
                best_h = h_new2

    # Phase 4: LP refinement (medium)
    if _remaining() > 100:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=400, topk=60,
                                       time_limit=_remaining() - 80)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 5: SLSQP with high alpha, small topk
    if _remaining() > 120:
        alphas = [(500, 30), (2000, 15), (5000, 8), (10000, 5)]
        h_ref, sc = slsqp_smoothmax(best_h, target_init, alphas, 3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: LP refinement (narrow)
    if _remaining() > 80:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=500, topk=30,
                                       time_limit=_remaining() - 60)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 7: Structured perturbations + local refine
    if _remaining() > 100:
        for attempt in range(5):
            if _remaining() < 30:
                break
            h_pert = structured_perturb(best_h, target_init, topk=5,
                                         scale=0.005, seed=seed + 1000 + attempt * 71)
            if _remaining() > 40:
                h_s, sc_s = subgradient_run(h_pert, target_init,
                                             max_iter=1500, lr=0.05, topk=5,
                                             patience=150, momentum=0.90,
                                             seed=seed + 2000 + attempt * 101)
            else:
                h_s, sc_s = h_pert, compute_c5(h_pert)
            if sc_s < best_score:
                best_score = sc_s
                best_h = h_s
            if _remaining() > 40:
                h_lp, sc_lp = lp_refinement(h_s, target_init,
                                             max_iterations=200, topk=50,
                                             time_limit=_remaining() - 30)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_h = h_lp

    # Phase 8: SLSQP very high alpha
    if _remaining() > 80:
        alphas2 = [(3000, 10), (10000, 5), (20000, 3)]
        h_ref, sc = slsqp_smoothmax(best_h, target_init, alphas2, 5000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final LP polish
    if _remaining() > 40:
        h_new, score = lp_refinement(best_h, target_init,
                                       max_iterations=500, topk=15,
                                       time_limit=_remaining() - 20)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 10: Multi-resolution attempt
    for n_try in [500, 640]:
        if _remaining() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_new, score = subgradient_run(h_interp, target_try,
                                        max_iter=2000, lr=0.08, topk=5,
                                        patience=200, momentum=0.90,
                                        seed=seed + 5000 + n_try)
        if score < best_score:
            best_score = score
            best_h = h_new
            target_init = target_try
        if _remaining() > 50:
            h_new2, score2 = lp_refinement(best_h, target_init,
                                            max_iterations=300, topk=80,
                                            time_limit=_remaining() - 40)
            if score2 < best_score:
                best_score = score2
                best_h = h_new2

    # Phase 11: Final polish on best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if _remaining() > 30:
        h_new, score = lp_refinement(best_h, target_use,
                                       max_iterations=200, topk=10,
                                       time_limit=_remaining() - 10)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
