def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_polytope(y, target_sum):
    """Project y onto {h in [0,1]^n : sum(h) = target_sum}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - target_sum / n - 1.0
    hi = y.max() + target_sum / n + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

    A = np.correlate(h_pad, w, mode='valid')
    B = np.correlate(h_pad, w_rev, mode='valid')

The projection onto {h ∈ [0,1]^n : sum(h) = S} where S = n/2:

- f(λ) = 0 for λ ≥ max(y_i) (all clips are 0)
- f is continuous and non-increasing in λ
- f is piecewise linear

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def project_polytope(y, target_sum):
    n = len(y)
    if abs(np.sum(y) - target_sum) < 1e-12 and np.all(y >= -1e-12) and np.all(y <= 1.0 + 1e-12):
        return np.clip(y, 0.0, 1.0)
    lo = float(y.min()) - target_sum / n - 2.0
    hi = float(y.max()) + target_sum / n + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = alpha * (corr - M)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    G = np.sum(exp_s)
    val = dx * (M + np.log(G) / alpha)
    w = exp_s / G
    w_rev = w[::-1]
    h_pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    A = np.correlate(h_pad, w, mode='valid')
    B = np.correlate(h_pad, w_rev, mode='valid')
    grad = (dx / alpha) * (1.0 - A - B)
    return val, grad


def projected_grad_descent(h_init, alpha, n, target, max_iters, t_start, time_limit):
    dx = 2.0 / n
    h = project_polytope(h_init.copy(), target)
    val, grad = smooth_obj_and_grad(h, alpha, n)
    best_val = val
    best_h = h.copy()
    eta = 1.0 / (2.0 * alpha * dx)
    no_improve = 0
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        h_new = project_polytope(h - eta * grad, target)
        new_val, new_grad = smooth_obj_and_grad(h_new, alpha, n)
        if new_val > val + 1e-15:
            eta_half = eta * 0.5
            h_new = project_polytope(h - eta_half * grad, target)
            new_val, new_grad = smooth_obj_and_grad(h_new, alpha, n)
            eta = eta_half
        h = h_new
        val = new_val
        grad = new_grad
        if val < best_val - 1e-13:
            best_val = val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break
    return best_h, best_val

def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while True:
        is_p = True
        if p < 2:
            is_p = False
        elif p < 4:
            is_p = True
        elif p % 2 == 0 or p % 3 == 0:
            is_p = False
        else:
            i = 5
            while i * i <= p:
                if p % i == 0 or p % (i + 2) == 0:
                    is_p = False
                    break
                i += 6
        if is_p:
            break
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

        h_ls3, c5_ls3 = local_search_binary(h_qr_nt, nt / 2.0, 100000, rng, t5, start + elapsed, 0.015, 0.00005)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3.copy()
            n_init = nt
        elapsed = time.time() - start
        remaining = budget_s - 50.0 - elapsed

    elapsed = time.time() - start
    remaining = budget_s - 40.0 - elapsed

    n_use = len(best_h)
    target_use = n_use / 2.0
    for it in range(4):
        if remaining < 60.0:
            break
        noise_scale = 0.3 + 0.2 * it
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert_bin = project_binary(h_pert, target_use)
        t6 = min(60.0, remaining * 0.4)
        h_ls4, c5_ls4 = local_search_binary(h_pert_bin, target_use, 100000, rng, t6, start + elapsed, 0.012, 0.00005)
        if c5_ls4 < best_c5:
            best_c5 = c5_ls4
            best_h = h_ls4.copy()
        elapsed = time.time() - start
        remaining = budget_s - 40.0 - elapsed

    elapsed = time.time() - start
    remaining = budget_s - 30.0 - elapsed

    if remaining > 80.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cont_final, c5_final = continuous_refine(best_h, n_use, target_use, remaining, start + elapsed)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_cont_final.copy()

    elapsed = time.time() - start
    remaining = budget_s - 10.0 - elapsed

    if remaining > 15.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        h_ls5, c5_ls5 = local_search_binary(h_bin_final, target_use,
                                             max(1000, int(remaining * 100)),
                                             rng, remaining, start + elapsed, 0.005, 0.00002)
        if c5_ls5 < best_c5:
            best_c5 = c5_ls5
            best_h = h_ls5.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_polytope(best_h, target_final)
    final_c5, _ = compute_c5(h_final)
    if final_c5 > best_c5 + 1e-12:
        h_final = best_h
        final_c5 = best_c5

    return h_final, final_c5, n_final
