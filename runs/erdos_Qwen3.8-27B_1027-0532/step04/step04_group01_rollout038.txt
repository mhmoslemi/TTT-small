def project_binary(h, target_sum):

    current = int(h_r.sum())
    target = int(target_sum)

        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]

    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)

        idx = np.where(h > 0.0)[0]

        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def continuous_optimize(h_init, n, target, time_limit, t_start):
    """Try continuous optimization with smooth max."""
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [50, 100, 200, 500, 1000, 2000]
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                dx = 2.0 / n
                c_vals = corr * dx
                c_max = np.max(c_vals)
                # Smooth max: (c_max + (1/alpha)*log(sum(exp(alpha*(c - c_max)))))
                # But this is just c_max since the max term dominates
                # Better: use log-sum-exp
                shifted = alpha * (c_vals - c_max)
                shifted = np.clip(shifted, -500.0, 0.0)
                smooth = c_max + np.log(np.sum(np.exp(shifted))) / alpha
                return float(smooth)
            
            def cfun(x):
                return np.sum(x) - target
            
            def cjac(x):
                return np.ones(n)
            
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.int64)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(np.float64)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        if len(zeros) < need:
            return h_r.astype(np.float64)
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        if len(ones) < need:
            return h_r.astype(np.float64)
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(np.float64)


def project_feasible(h, target_sum):

    if diff > 0:

    h = h.copy()

def _is_prime(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True

                h = best_h.copy()
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                cur_c5 = float(np.max(corr)) * 2.0 / n
                n_perturb = rng.randint(perturb_level, perturb_level + 8)
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                for _ in range(n_perturb):
                    if len(ones) > 0 and len(zeros) > 0:
                        pi = int(ones[rng.randint(len(ones))])
                        pj = int(zeros[rng.randint(len(zeros))])
                        h[pi] = 0
                        h[pj] = 1
                h = project_binary(h, target_int).astype(np.float64)
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                cur_c5 = float(np.max(corr)) * 2.0 / n
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
            perturb_level = min(12, perturb_level + 1)
            no_imp = 0
            continue

        ratio = it / max(max_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        r = rng.random()

        if r < 0.45:
            # Target single worst lag aggressively
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - (n - 1)
            contrib = []
            for ci in range(len(ones)):
                oi = int(ones[ci])
                tp = oi - worst_lag
                if 0 <= tp < n and h[tp] == 0:
                    contrib.append(ci)
            if len(contrib) > 0:
                ii = int(ones[contrib[rng.randint(len(contrib))]])
            else:
                ii = int(ones[rng.randint(len(ones))])
            # Choose target zero: prefer zeros that are "far" from other ones
            jj = int(zeros[rng.randint(len(zeros))])
            accepted = _try_swap(h, g, corr, cur_c5, ii, jj, n, temp, rng)
            if accepted:
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                new_c5 = float(np.max(corr)) * 2.0 / n
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_imp = 0
                    perturb_level = max(3, perturb_level - 1)
                else:
                    no_imp += 1
            else:
                no_imp += 1

        elif r < 0.65:
            # Target top-K worst lags
            k = min(5, n_corr)
            top_k_idx = np.argsort(corr)[-k:]
            wi = int(top_k_idx[rng.randint(k)])
            wl = wi - (n - 1)
            contrib = []
            for ci in range(len(ones)):
                oi = int(ones[ci])
                tp = oi - wl
                if 0 <= tp < n and h[tp] == 0:
                    contrib.append(ci)
            if len(contrib) > 0:
                ii = int(ones[contrib[rng.randint(len(contrib))]])
            else:
                ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            accepted = _try_swap(h, g, corr, cur_c5, ii, jj, n, temp, rng)
            if accepted:
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                new_c5 = float(np.max(corr)) * 2.0 / n
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1

        elif r < 0.80:
            # Random swap
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            accepted = _try_swap(h, g, corr, cur_c5, ii, jj, n, temp, rng)
            if accepted:
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                new_c5 = float(np.max(corr)) * 2.0 / n
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1

        elif r < 0.92:
            # Double swap
            if len(ones) < 2 or len(zeros) < 2:
                no_imp += 1
                continue
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if i1 == i2 or j1 == j2 or i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                no_imp += 1
                continue
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            g = 1.0 - h
            new_corr = np.correlate(h, g, mode='full')
            new_c5 = float(np.max(new_corr)) * 2.0 / n
            delta = new_c5 - cur_c5
            if delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-delta / temp)):
                corr = new_corr.astype(np.float64)
                cur_c5 = new_c5
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                h[i1] = 1
                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                no_imp += 1

        else:
            # Block move: move a contiguous group of ones
            if len(ones) == 0 or len(zeros) == 0:
                no_imp += 1
                continue
            blen = rng.randint(2, 5)
            start_i = int(ones[rng.randint(len(ones))])
            block = []
            for d in range(blen):
                pos = start_i + d
                if 0 <= pos < n and h[pos] == 1:
                    block.append(pos)
            if len(block) < 2:
                for d in range(blen):
                    pos = start_i - d
                    if 0 <= pos < n and h[pos] == 1:
                        block.append(pos)
            if len(block) < 2:
                no_imp += 1
                continue
            # Find target location
            jj = int(zeros[rng.randint(len(zeros))])
            new_positions = []
            for d in range(len(block)):
                pos = jj - len(block) // 2 + d
                if 0 <= pos < n and h[pos] == 0 and pos not in block:
                    new_positions.append(pos)
            if len(new_positions) < len(block):
                no_imp += 1
                continue
            for pos in block:
                h[pos] = 0
            for pos in new_positions:
                h[pos] = 1
            g = 1.0 - h
            new_corr = np.correlate(h, g, mode='full')
            new_c5 = float(np.max(new_corr)) * 2.0 / n
            delta = new_c5 - cur_c5
            if delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-delta / temp)):
                corr = new_corr.astype(np.float64)
                cur_c5 = new_c5
                ones = np.where(h == 1)[0].astype(np.int64)
                zeros = np.where(h == 0)[0].astype(np.int64)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                for pos in block:
                    h[pos] = 1
                for pos in new_positions:
                    h[pos] = 0
                g = 1.0 - h
                corr = np.correlate(h, g, mode='full').astype(np.float64)
                no_imp += 1

    return best_h, best_c5


def _try_swap(h, g, corr, cur_c5, ii, jj, n, temp, rng):
    if ii == jj:
        return False
    h[ii] = 0
    h[jj] = 1
    g = 1.0 - h
    new_corr = np.correlate(h, g, mode='full')
    new_c5 = float(np.max(new_corr)) * 2.0 / n
    delta = new_c5 - cur_c5
    if delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-delta / temp)):
        return True
    else:
        h[ii] = 1
        h[jj] = 0
        g[ii] = 0
        g[jj] = 1
        return False


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 500, 1200, 3000]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def _smooth_obj(x, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    target0 = n0 / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive local search from initial (use ~25% of budget)
    t1 = min(300.0, budget_s * 0.25)
    h_ls, c5_ls = enhanced_local_search(h0, target0, 800000, rng, t1, start, 0.06, 0.000005)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    remaining = budget_s - 40.0 - (time.time() - start)

    # Phase 2: Try multiple initializations at n0
    inits = []
    inits.append(('qr', make_qr_sequence(n0, rng)))
    inits.append(('ms', make_mel_seq(n0, rng)))
    h_pert = best_h + rng.randn(n0) * 0.4
    inits.append(('pert', h_pert))
    h_rand = np.zeros(n0)
    h_rand[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    inits.append(('rand', h_rand))
    inits.append(('qr2', make_qr_sequence(n0 + 2, rng)))

    for name, h_try in inits:
        if time.time() - start > budget_s - 80.0:
            break
        tl = min(150.0, (budget_s - 80.0) - (time.time() - start))
        if tl < 40:
            break
        nt = len(h_try)
        h_ls, c5_ls = enhanced_local_search(h_try, nt / 2.0, 400000, rng,
                                            tl, start, 0.05, 0.00001)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    remaining = budget_s - 40.0 - (time.time() - start)

    # Phase 3: Try larger n values by interpolation
    n_candidates = []
    for delta in [20, 50, 100, 150]:
        nc = n0 + delta
        if nc % 2 == 0:
            n_candidates.append(nc)

    for nc in n_candidates:
        if remaining < 100.0:
            break
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, nc)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin = project_binary(h_interp, nc / 2.0)
        tl = min(150.0, remaining * 0.6)
        h_ls, c5_ls = enhanced_local_search(h_bin, nc / 2.0, 300000, rng,
                                            tl, start, 0.04, 0.00001)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - 40.0 - (time.time() - start)

    # Phase 4: Continuous refinement
    remaining = budget_s - 40.0 - (time.time() - start)
    if remaining > 80.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        tl = min(150.0, remaining)
        h_cont, sc_cont = continuous_refine(best_h, n_use, target_use, tl, start)
        if sc_cont < best_c5:
            best_c5 = sc_cont
            best_h = h_cont.copy()

    remaining = budget_s - 40.0 - (time.time() - start)

    # Phase 5: Final binary polish
    if remaining > 30.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        tl = max(15.0, remaining - 5.0)
        h_ls, c5_ls = enhanced_local_search(h_bin_final, target_use, 200000, rng,
                                            tl, start, 0.02, 0.000002)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5:
        h_final = best_h.copy()
        final_c5 = best_c5

    return h_final, final_c5, n_final
