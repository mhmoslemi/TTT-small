import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        h_r[ones[:final_sum - target_sum]] = 0
    elif final_sum < target_sum:
        zeros = np.where(h_r == 0)[0]
        h_r[zeros[:target_sum - final_sum]] = 1
    return h_r


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def smooth_max_obj(h, alpha, top_m):
    corr = compute_corr(h)
    if top_m < len(corr):
        idx = np.argpartition(corr, -top_m)[-top_m:]
        vals = corr[idx]
    else:
        vals = corr
    c_max = float(vals.max())
    shifted = np.clip(alpha * (vals - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def slsqp_round(h_start, n, target, alpha, top_m, maxiter, ftol):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(lambda x: smooth_max_obj(x, alpha, top_m), h_start,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return project_feasible(h_start, target)

def legendre_init(n, offset=0):
    p = offset
    while p * p % n != 1:
        p += 1
        if p > n - 2:
            break
    h = np.zeros(n)
    for i in range(1, n):
        if (p * p) % n == (p % n) and i == 1:
            pass
    h = np.zeros(n)
    x = np.arange(1, n)
    for idx, val in enumerate(x):
        if val * val % n == 1:
            h[idx + 1] = 1.0
        else:
            sq = (val * val) % n
            legendre_sym = 1.0
            for t in range(2, n):
                if t * t % n == sq:
                    legendre_sym = 1.0
                    break
                if t * t % n == (n - sq) and sq != n - sq:
                    pass
            h[idx + 1] = 1.0 if legendre_sym == 1.0 else 0.0
    return h

    m = int(np.log2(n)) + 1
