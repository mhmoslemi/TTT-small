        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def cdclp_topk(h_init, n, target, time_limit, K_start=20, K_max=150, damping=0.7):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    K = K_start

    for it in range(10000):
        if time.time() - start > time_limit:
            break

        corr_vals = compute_corr_scaled(h)
        K_cur = min(K, len(corr_vals))
        top_idx = np.argsort(corr_vals)[-K_cur:]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        g = 1.0 - h
        for idx in top_idx:
            k = idx - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                if np.any(coeff != 0):
                    constraints.append(coeff @ h_var <= t_var)

        objective = t_var + 1e-6 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False, eps=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=20000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_new.copy()

        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        if it % 40 == 0:
            K = min(K_max, K + 5)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def nesterov_subgradient(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    n_half = (n - 1)
    idxs = np.arange(n)

    h_cur = h.copy()
    h_mom = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    momentum = 0.9
    lr = 0.05
    lr_decay = 0.9995

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        k = k_star

        grad = np.zeros(n)
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        h_mom = momentum * h_mom + (1 - momentum) * h_cur
        h_try = h_mom - lr * grad
        h_try = project_feasible(h_try, target)
        h_cur = h_try
        lr *= lr_decay

        sc = compute_c5(h_cur)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_cur.copy()

    return best_h, best_c5


def pairwise_cd(h_init, n, target, max_sweeps, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.time() - start > time_limit:
                break
            j = (i + rng.randint(1, n - 1)) % n
            if j == i:
                continue

            a = h[i] + h[j]
            lo = max(0.0, a - 1.0)
            hi = min(1.0, a)
            if hi - lo < 1e-10:
                continue

            h_copy = h.copy()

            def obj_1d(xi):
                h_copy[i] = xi
                h_copy[j] = a - xi
                c = compute_c5(h_copy)
                h_copy[i] = h[i]
                h_copy[j] = h[j]
                return c

            x0 = h[i]
            result = minimize_scalar(obj_1d, bounds=(lo, hi),
                                      method='bounded',
                                      options={'maxiter': 40, 'xatol': 1e-6})
            if result.fun < best_c5 - 1e-13:
                h[i] = result.x
                h[j] = a - result.x
                h = project_feasible(h, target)
                best_c5 = result.fun
                best_h = h.copy()
                improved = True

        if not improved and sweep > 3:
            break

    return best_h, best_c5


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = (n - 1)
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        k = k_star

        grad = np.zeros(n)
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 50:
            break

    return best_h, best_c5


def local_perturbation_search(h_init, n, target, max_tries, time_limit, seed=12345):
    start = time.time()
    rng = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break

        n_pert = rng.choice([1, 2, 3, 5, 8, 13, 21, 34])
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3, 0.5])
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: CDCLP with top-K (main workhorse)
    if tleft() > 200:
        lp_time = min(300, tleft() * 0.3)
        h_lp, c5_lp = cdclp_topk(h_init, n, target,
                                   time_limit=lp_time, K_start=20, K_max=120, damping=0.7)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Nesterov subgradient
    if tleft() > 100:
        h_nest, c5_nest = nesterov_subgradient(best_h, n, target,
                                                max_steps=50000, time_limit=min(80, tleft() * 0.1))
        if c5_nest < best_score:
            best_score = c5_nest
            best_h = h_nest

    # Phase 3: SLSQP with moderate alphas
    if tleft() > 120:
        alphas = [50, 100, 300, 1000, 3000]
        h_ref = slsqp_round(best_h, n, target, alphas, 2000, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Pairwise coordinate descent (key improvement)
    if tleft() > 100:
        h_cd, c5_cd = pairwise_cd(best_h, n, target,
                                   max_sweeps=500, time_limit=min(200, tleft() * 0.25),
                                   seed=seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Second CDCLP round
    if tleft() > 150:
        lp_time = min(150, tleft() * 0.2)
        h_lp2, c5_lp2 = cdclp_topk(best_h, n, target,
                                     time_limit=lp_time, K_start=30, K_max=100, damping=0.5)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 6: More pairwise CD
    if tleft() > 80:
        h_cd2, c5_cd2 = pairwise_cd(best_h, n, target,
                                     max_sweeps=500, time_limit=min(120, tleft() * 0.2),
                                     seed=seed + 1)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 7: SLSQP polish
    if tleft() > 80:
        alphas2 = [100, 500, 2000, 5000, 10000]
        h_ref2 = slsqp_round(best_h, n, target, alphas2, 1500, 1e-15)
        sc = compute_c5(h_ref2)
        if sc < best_score:
            best_score = sc
            best_h = h_ref2

    # Phase 8: Subgradient fine-tuning
    if tleft() > 40:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                           max_steps=50000, time_limit=min(30, tleft() * 0.3))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 9: Final pairwise CD
    if tleft() > 30:
        h_cd3, c5_cd3 = pairwise_cd(best_h, n, target,
                                     max_sweeps=200, time_limit=min(20, tleft() * 0.4),
                                     seed=seed + 2)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 10: Perturbation + SLSQP restart
    if tleft() > 60:
        for noise in [0.01, 0.03, 0.05]:
            if tleft() < 40:
                break
            rng = np.random.RandomState(seed + noise * 1000)
            h_pert = best_h + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
            h_r = slsqp_round(h_pert, n, target, [100, 500, 2000], 1000, 1e-14)
            sc = compute_c5(h_r)
            if sc < best_score:
                best_score = sc
                best_h = h_r

    # Final: one more pass of pairwise CD on best
    if tleft() > 20:
        h_final_cd, c5_final = pairwise_cd(best_h, n, target,
                                            max_sweeps=100, time_limit=min(15, tleft() * 0.5),
                                            seed=seed + 3)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
