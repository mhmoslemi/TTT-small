import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def soft_max_obj(h, alpha, target_sum):
    """Smooth max objective with penalty for sum constraint."""
    corr = compute_corr(h)
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -100.0, 0.0)
    sm = m + float(np.log(np.sum(np.exp(shifted))) / alpha)
    # Add penalty for sum constraint
    penalty = 1000.0 * (np.sum(h) - target_sum) ** 2
    return sm + penalty

def soft_max_grad(h, alpha, target_sum):
    """Gradient of smooth max objective."""
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -100.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Compute gradient: sum_k weights[k] * d(corr[k])/dh[p]
    # d(corr(o))/dh[p] = (1-h[p+o])*[p+o valid] - h[p-o]*[p-o valid]
    # where o = k - (n-1)
    
    grad = np.zeros(n)

    # Loop over offsets
    for k in range(2 * n - 1):

                grad[0:n-o_neg] -= wo * h[o_neg:n]

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smooth_obj_and_grad(h, alpha, target_sum, penalty=500.0):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -100.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = exp_vals.sum()
    weights = exp_vals / w_sum
    sm_val = m + float(np.log(w_sum) / alpha)
    
    # Gradient of smooth max
    grad_sm = np.zeros(n)
    for k in range(2 * n - 1):
        wo = weights[k]
        if wo < 1e-12:
            continue
        o = k - (n - 1)
        if o >= 0:
            if o < n:
                grad_sm[0:n-o] += wo * (1.0 - h[o:n])
            if o > 0:
                grad_sm[o:n] -= wo * h[0:n-o]
        else:
            o_neg = -o
            if o_neg < n:
                grad_sm[o_neg:n] += wo * (1.0 - h[0:n-o_neg])
                grad_sm[0:n-o_neg] -= wo * h[o_neg:n]
    
    # Penalty for sum constraint
    sum_h = np.sum(h)
    pen_val = penalty * (sum_h - target_sum) ** 2
    pen_grad = 2.0 * penalty * (sum_h - target_sum) * np.ones(n)
    
    return sm_val + pen_val, grad_sm + pen_grad
