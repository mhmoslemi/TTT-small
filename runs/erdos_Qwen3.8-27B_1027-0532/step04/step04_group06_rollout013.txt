```python
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    center = n - 1
    # Only process lags with significant weight
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    for m in active:
        w = weights[m]
        k = m - center
        # d(corr[m])/dh[i] = (1-h[i-k])*1[0<=i-k<n] - h[i+k]*1[0<=i+k<n]
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-k:hi1-k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2+k:hi2+k]
    return grad

def _eq_con_fun(h, target):
    return np.sum(h) - target

def _eq_con_jac(h, n):
    return np.ones(n)
```

One more thought: for the binary local search, I want to make it fast. The key operation is: given a binary h, find a pair (i, j) with h[i]=1, h[j]=0 such that swapping them reduces C5.

The "targeted" approach: focus on the worst lag k*. A swap (i, j) reduces the correlation at lag k* if:
- h[i]=1 and (h[i+k*]=1 or h[i-k*]=1) [i contributes to overlap at k*]
- h[j]=0 and (h[j+k*]=0 and h[j-k*]=0) [j doesn't create overlap at k*]

So the ideal move is: remove a 1 that's part of an overlap at the worst lag, and place it at a position that doesn't create any overlap at that lag.

Let me implement this efficiently using numpy.

For a given worst lag k*:
- "Bad" ones: positions i where h[i]=1 and (h[i+k*]=1 or h[i-k*]=1)
- "Good" zeros: positions j where h[j]=0 and h[j+k*]=0 and h[j-k*]=0

If there's a bad one and a good zero, swap them. This is guaranteed to reduce the correlation at lag k* (or at least not increase it).

But we also need to check that it doesn't increase other lags too much. For a quick check, we can just compute C5 after the swap.

Let me code this up:

```python
def targeted_local_search(h, target_sum, max_rounds, rng, time_limit, start_time, aggressiveness=0.7):
    n = len(h)
    h = project_binary(h, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        corr = get_corr(h)
        center = n - 1
        # Find top lags
        top_k = 5
        sorted_lags = np.argsort(corr)[::-1]
        worst_lags = (sorted_lags[:top_k] - center)
        worst_lags = worst_lags[worst_lags != 0]
        if len(worst_lags) == 0:
            worst_lags = np.array([1])
        
        improved = False
        for k_star in worst_lags:
            k_star = int(k_star)
            if abs(k_star) >= n:
                continue
            
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # Find "bad" ones (contribute to overlap at k_star)
            bad_mask = np.zeros(len(ones), dtype=bool)
            for i_idx in range(len(ones)):
                i = ones[i_idx]
                ik = i + k_star
                ikm = i - k_star
                if 0 <= ik < n and h[ik] == 1:
                    bad_mask[i_idx] = True
                if 0 <= ikm < n and h[ikm] == 1:
                    bad_mask[i_idx] = True
            
            # Find "good" zeros (don't create overlap at k_star)
            good_mask = np.zeros(len(zeros), dtype=bool)
            for j_idx in range(len(zeros)):
                j = zeros[j_idx]
                jk = j + k_star
                jkm = j - k_star
                ok = True
                if 0 <= jk < n and h[jk] == 1:
                    ok = False
                if 0 <= jkm < n and h[jkm] == 1:
                    ok = False
                good_mask[j_idx] = ok
            
            if np.any(bad_mask) and np.any(good_mask):
                # Try random bad one and good zero
                bad_ones = ones[bad_mask]
                good_zeros = zeros[good_mask]
                
                # Try a few random pairs
                for attempt in range(20):
                    a = bad_ones[rng.randint(len(bad_ones))]
                    b = good_zeros[rng.randint(len(good_zeros))]
                    h[a] = 0.0
                    h[b] = 1.0
                    c5_new = compute_c5(h)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[a] = 1.0
                        h[b] = 0.0
            
            if not improved:
                # Try random swaps
                if len(ones) > 0 and len(zeros) > 0:
                    for attempt in range(10):
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                        c5_new = compute_c5(h)
                        if c5_new < best_c5 - 1e-12:
                            best_c5 = c5_new
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[a] = 1.0
                            h[b] = 0.0
        
        if not improved:
            # Try multi-bits swap
            if len(ones) >= 2 and len(zeros) >= 2:
                ia = rng.choice(len(ones), 2, replace=False)
                ib = rng.choice(len(zeros), 2, replace=False)
                a1, a2 = ones[ia[0]], ones[ia[1]]
                b1, b2 = zeros[ib[0]], zeros[ib[1]]
                h[a1] = 0; h[a2] = 0; h[b1] = 1; h[b2] = 1
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                else:
                    h[a1] = 1; h[a2] = 1; h[b1] = 0; h[b2] = 0
    
    return best_h, best_c5
```

This is O(n) per swap attempt (due to compute_c5), and we do maybe 20-30 attempts per round, with top_k * max_rounds rounds. For n=600, each compute_c5 takes ~50μs, so 30 attempts * 50μs = 1.5ms per round. With 10000 rounds, that's 15 seconds. Reasonable.
