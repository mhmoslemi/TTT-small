import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, target):
    """Convert continuous h to binary with exactly target ones."""
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[int(target) - 1] if int(target) < len(h) else 0
    h_bin = (h >= threshold).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Remove excess ones (those closest to threshold)
        ones_idx = np.where(h_bin == 1)[0]
        ones_vals = h[ones_idx]
        order = np.argsort(ones_vals)
        for i in range(n_ones - target):
            h_bin[ones_idx[order[i]]] = 0
    elif n_ones < target:
        # Add missing ones (those closest to threshold from below)
        zeros_idx = np.where(h_bin == 0)[0]
        zeros_vals = h[zeros_idx]
        order = np.argsort(zeros_vals)[::-1]
        for i in range(target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1
    return h_bin

def sa_local_search(h_bin, target, n_iter, t_start, t_end, rng, deadline):
    """Simulated annealing on binary configuration."""
    n = len(h_bin)
    h_cur = h_bin.copy()
    h_best = h_bin.copy()
    c5_cur = compute_c5(h_cur)
    c5_best = c5_cur
    
    ones_list = np.where(h_cur == 1)[0]
    zeros_list = np.where(h_cur == 0)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        
        # Pick a random one and a random zero to swap
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h_cur[i] = 0
        h_cur[j] = 1
        
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            c5_cur = c5_new
            ones_list = np.where(h_cur == 1)[0]
            zeros_list = np.where(h_cur == 0)[0]
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_cur.copy()
        else:
            h_cur[i] = 1
            h_cur[j] = 0
    
    return h_best, c5_best

    # Precompute correlation for faster evaluation
    # We'll recompute each time since n is small
    
    step = 0
    total_steps = int((deadline - time.time()) / 0.0001)  # rough estimate
    total_steps = max(total_steps, 1000)
    
    while time.time() < deadline:
        step += 1
        t = t_start * (t_end / t_start) ** (step / total_steps)
        
        # Choose move type
        if rng.random() < 0.3:
            # Targeted move: focus on worst lag
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            k_max = int(np.argmax(corr))
            k = k_max - (n - 1)
            # Find positions contributing to this lag
            if k >= 0:
                # corr at lag k: h[i]=1, h[i+k]=0 for i in [0, n-1-k)
                contrib = [i for i in range(n - k) if h_cur[i] == 1 and h_cur[i+k] == 0]
            else:
                m = -k
                contrib = [i for i in range(m, n) if h_cur[i] == 1 and h_cur[i-m] == 0]
            
            if not contrib:
                i = ones[rng.randint(len(ones))]
            else:
                i = contrib[rng.randint(len(contrib))]
            
            j = zeros[rng.randint(len(zeros))]
        else:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h_cur[i] = 0
        h_cur[j] = 1
        
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            c5_cur = c5_new
            # Update ones and zeros
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_cur.copy()
        else:
            h_cur[i] = 1
            h_cur[j] = 0
    
    return h_best, c5_best


def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, n_dst / 2.0)
