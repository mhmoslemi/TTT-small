import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_optimize(h, target, alpha, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]
    
    def cfun(x):
        return np.sum(x) - target
    
    def cjac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def nesterov_subgrad(h, target, n_steps, active_frac, learning_rate, deadline):
    n = len(h)
    h_cur = h.copy()
    h_mom = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    beta = 0.85
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        
        # Nesterov step
        h_new = h_cur - learning_rate / np.sqrt(1 + it) * grad
        h_new = project_feasible(h_new, target)
        
        h_mom = h_new + beta * (h_new - h_cur)
        h_mom = project_feasible(h_mom, target)
        
        sc = compute_c5(h_mom)
        if sc < best_score:
            best_score = sc
            best_h = h_mom.copy()
        
        h_cur = h_mom
    
    return best_h, best_score

def shake_and_optimize(h, target, n_shakes, shake_sigma, alpha, maxiter, deadline):

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_optimize(h, target, alpha, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def adaptive_slsqp(h, target, n, deadline, alpha_start=10, alpha_end=500000,
                   maxiter=3000, ftol=1e-16, stall_iters=2):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    alpha = alpha_start
    stall_count = 0
    prev_score = best_score

    while time.time() < deadline:
        h_new, sc = slsqp_optimize(h_cur, target, alpha, n, maxiter, ftol)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
            stall_count = 0
        else:
            stall_count += 1

        if sc < prev_score - 1e-8:
            stall_count = max(0, stall_count - 1)
        prev_score = min(prev_score, sc)

        if stall_count >= stall_iters:
            alpha = min(alpha * 3.0, alpha_end)
            stall_count = 0
        else:
            alpha = min(alpha * 1.5, alpha_end)

        if alpha >= alpha_end and stall_count > 3:
            break

    return h_cur, best_score


def nesterov_subgrad(h, target, n_steps, active_frac, lr0, deadline):
    n = len(h)
    h_cur = h.copy()
    h_mom = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    beta = 0.8

    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        c_range = c_max - c_min
        if c_range < 1e-12:
            break
        threshold = c_max - active_frac * c_range
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= len(active)

        step = lr0 / np.sqrt(1.0 + 0.5 * it)
        h_candidate = h_mom - step * grad
        h_candidate = project_feasible(h_candidate, target)
        new_mom = h_candidate + beta * (h_candidate - h_cur)
        new_mom = project_feasible(new_mom, target)

                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])

        g_norm_sq = float(np.dot(grad, grad))

        current_score = compute_c5(h_cur)

            best_h = h_new.copy()
            f_lower = max(f_lower, sc - 0.1)
