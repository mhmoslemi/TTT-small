                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h_start


def structured_restarts(h_base, n, rng, target, num_starts):
    starts = [h_base.copy()]
    x = np.linspace(0, 2, n)
    for i in range(num_starts):
        freq = rng.randint(1, 15)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        pert = amp * np.sin(np.pi * freq * x / 2.0 + phase)
        h_try = project_feasible(h_base + pert, target)
        starts.append(h_try)
    starts.append(np.clip(h_base + rng.randn(n) * 0.05, 0, 1))
    starts[-1] = project_feasible(starts[-1], target)
    return starts


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 40.0 - (time.time() - start_time)

    # Phase 1: Annealed LSE with dynamic active set (broad to narrow)
    phases = [
        (0.4, [10, 25, 60]),
        (0.25, [25, 60, 150]),
        (0.15, [60, 150, 400]),
        (0.08, [150, 400, 800]),
        (0.03, [400, 1000, 2000]),
    ]

    for tf, alphas in phases:
        if time_left() < 80:
            break
        corr = get_correlations(best_h)
        active = get_active_set(corr, tf, min_size=8)
        for alpha in alphas:
            if time_left() < 40:
                break
            h_start = best_h + rng.randn(n) * 0.004
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_active(h_start, target, alpha, active, 5000.0, 350, bounds, 1e-15)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new
                corr = get_correlations(best_h)
                active = get_active_set(corr, tf, min_size=8)

    # Phase 2: Subgradient with momentum (diversification)
    if time_left() > 120:
        h_sg, score_sg = subgradient_momentum(best_h, target, 800, 0.002, 0.005, 0.7, 15)
        if score_sg < best_score:
            best_score = score_sg
            best_h = h_sg

    # Phase 3: Try structured restarts at current best
    if time_left() > 150:
        starts = structured_restarts(best_h, n, rng, target, 4)
        for h_st in starts:
            if time_left() < 50:
                break
            corr_st = get_correlations(h_st)
            active_st = get_active_set(corr_st, 0.1, min_size=10)
            for alpha in [100, 300]:
                if time_left() < 30:
                    break
                h_new = lbfgs_active(h_st, target, alpha, active_st, 5000.0, 250, bounds, 1e-15)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new

    # Phase 4: Multi-resolution exploration
    res_list = [500, 600, 700, 800]
    for n_try in res_list:
        if time_left() < 100:
            break
        target_try = n_try / 2.0
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_try)
        bounds_try = [(0.0, 1.0)] * n_try

        for alpha in [40, 120, 350]:
            if time_left() < 40:
                break
            corr_try = get_correlations(h_interp)
            active_try = get_active_set(corr_try, 0.12, min_size=10)
            h_start = h_interp + rng.randn(n_try) * 0.006
            h_start = project_feasible(h_start, target_try)
            h_new = lbfgs_active(h_start, target_try, alpha, active_try, 5000.0, 200, bounds_try, 1e-14)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                x_new_back = np.linspace(0, 1, n_try)
                x_old_back = np.linspace(0, 1, n)
                best_h = np.interp(x_old_back, x_new_back, h_new)
                best_h = project_feasible(best_h, target)

    # Phase 5: SLSQP on tight active set
    if time_left() > 80:
        corr = get_correlations(best_h)
        c_max = np.max(corr)
        c_range = c_max - np.min(corr)
        active = np.where(corr >= c_max - 0.03 * c_range)[0]
        if len(active) < 8:
            active = np.argpartition(corr, -12)[-12:]
        h_new = slaqp_refine(best_h, target, 250, active, 600)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 6: High-alpha LSE polish (very tight active set)
    if time_left() > 60:
        for alpha in [500, 1000, 2500, 5000]:
            if time_left() < 25:
                break
            corr = get_correlations(best_h)
            top_k = 8
            active = np.argpartition(corr, -top_k)[-top_k:]
            h_start = best_h + rng.randn(n) * 0.001
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_active(h_start, target, alpha, active, 20000.0, 250, bounds, 1e-16)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 7: Local coordinate search
    if time_left() > 40:
        deltas = [0.008, -0.008, 0.004, -0.004, 0.002, -0.002]
        for round_id in range(3):
            if time_left() < 20:
                break
            improved = False
            step_size = 2 if round_id == 0 else 1
            for coord in range(0, n, step_size):
                if time_left() < 15:
                    break
                orig = best_h[coord]
                for delta in deltas:
                    h_try = best_h.copy()
                    h_try[coord] = np.clip(orig + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-9:
                        best_score = sc
                        best_h = h_try
                        improved = True
                        break
            if not improved:
                break

    # Phase 8: Final SLSQP
    if time_left() > 30:
        corr = get_correlations(best_h)
        active = np.argpartition(corr, -10)[-10:]
        h_new = slaqp_refine(best_h, target, 1500, active, 1000)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: Final ultra-high-alpha L-BFGS
    if time_left() > 15:
        corr = get_correlations(best_h)
        active = np.argpartition(corr, -6)[-6:]
        h_new = lbfgs_active(best_h, target, 8000, active, 50000.0, 200, bounds, 1e-17)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
