```python
def pnorm_gradient(h, p, dx):
    n = len(h)
    c_all = np.correlate(h, 1.0 - h, mode='full') * dx
    c_safe = np.maximum(c_all, 1e-15)
    
    sum_cp = np.sum(c_safe ** p)
    if sum_cp < 1e-15:
        return np.zeros(n)
    
    # alpha_k for k = -(n-1), ..., n-1
    # c_all[j] corresponds to k = j - (n-1)
    alpha = (c_safe ** (p-1)) * (sum_cp ** (1.0/p - 1.0)) * dx
    
    # T1[i] = sum_k alpha[k] * (1-h[i+k]) * [0<=i+k<n]
    # T2[i] = sum_k alpha[k] * h[i-k] * [0<=i-k<n]
    
    # For T1: for each k, it contributes to T1[i] for i in [max(0,-k), min(n, n-k))
    # T1[i] += alpha[k] * (1-h[i+k])
    
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    
    n_half = n - 1
    for j in range(2 * n - 1):
        k = j - n_half
        aj = alpha[j]
        if aj < 1e-18:
            continue
        # T1 contribution: i+k in [0,n) => i in [-k, n-k) intersect [0, n)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            T1[lo1:hi1] += aj * (1.0 - h[lo1+k:hi1+k])
        
        # T2 contribution: i-k in [0,n) => i in [k, k+n) intersect [0, n)
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if hi2 > lo2:
            T2[lo2:hi2] += aj * h[lo2-k:hi2-k]
    
    return T1 - T2
```

3. The continuation from small p to large p naturally "discovers" the active set

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()

        idx = np.where(h < 1.0 - 1e-10)[0]
    else:

        return h
