            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    for _ in range(abs(final_sum - target_sum)):
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def deterministic_greedy(h, max_iters, time_limit=None, start_time=None, n_candidates=50):
    """Greedy that tries the top n_candidates swaps at the worst lag."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 500
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates that would directly reduce R(k_star)
        # i in ones where h[i+k_star]=0 (removes a 1 from R(k_star))
        # j in zeros where h[j+k_star]=1 (removes a 1 from R(k_star))
        i_valid = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_valid = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        
        if len(i_valid) == 0 or len(j_valid) == 0:
            stall += 1
            break
        
        # Get top lags to check
        top_frac = corr / max_val
        top_mask = top_frac > 0.9
        top_indices = np.where(top_mask)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_shifts = top_indices - (n - 1)
        top_vals = corr[top_indices].astype(float).copy()
        
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        # Try limited candidates
        i_sample = i_valid[:n_candidates] if len(i_valid) > n_candidates else i_valid
        j_sample = j_valid[:n_candidates] if len(j_valid) > n_candidates else j_valid
        
        for i in i_sample:
            for j in j_sample:
                new_max = max_val
                for t, k in enumerate(top_shifts):
                    delta = 0.0
                    if 0 <= i + k < n:
                        delta -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        delta += h[i - k]
                    if 0 <= j + k < n:
                        delta += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        delta -= h[j - k]
                    val = top_vals[t] + delta
                    if val > new_max:
                        new_max = val
                        break
                if new_max < max_val - best_improvement:
                    best_improvement = max_val - new_max
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(corr.max()) * (2.0 / n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def targeted_sa(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    """SA with moves targeted at the worst correlation lags."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 10000
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose move type
        r = rng.random()
        h_new = h.copy()
        moved = False
        
        if r < 0.5:
            # Targeted move: swap within the worst lag
            worst_lags = np.argsort(corr)[-5:]
            k = worst_lags[rng.randint(len(worst_lags))] - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k < n]
            j_pool = [j for j in zeros if 0 <= j + k < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif r < 0.8:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            # Pair swap (4 positions)
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        
        if not moved:
            continue
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Smooth max using log-sum-exp
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def thue_morse_init(n, rng):
    """Generate Thue-Morse based initialization."""
    h = np.zeros(n)
    # Thue-Morse: a(n) = popcount(n) mod 2
    tm = np.array([np.binary_repr(i).count('1') % 2 for i in range(n)], dtype=float)
    # We need exactly n/2 ones
    ones_count = int(tm.sum())
    target = n // 2
    if ones_count > target:
        # Remove some ones
        ones_idx = np.where(tm == 1)[0]
        rng.shuffle(ones_idx)
        tm[ones_idx[:ones_count - target]] = 0
    elif ones_count < target:
        zeros_idx = np.where(tm == 0)[0]
        rng.shuffle(zeros_idx)
        tm[zeros_idx[:target - ones_count]] = 1
    return tm

            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def legendre_init(n, rng):
    p = 1
    while p < n:
        p += 2
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime and p >= n:
            break
    if p < n:
        p = n + 1 if (n + 1) % 2 == 1 else n + 2
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2

    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0

    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        remaining = n - p
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy from initial
    allowed = min(60, tleft() * 0.12)
    h_g, c5_g = greedy_full(best_h, max_iters=50000, time_limit=allowed,
                            start_time=start_time, scan_width=20)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA from greedy
    allowed = min(200, tleft() * 0.25)
    max_it = int(min(300000, allowed * 1500))
    h_sa, c5_sa = sa_targeted(best_h, target_init, max_it, 0.012, 0.00003, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Second SA round with different params
    if tleft() > 100:
        allowed = min(150, tleft() * 0.18)
        max_it = int(min(200000, allowed * 1500))
        h_sa2, c5_sa2 = sa_targeted(best_h, target_init, max_it, 0.006, 0.00001, rng,
                                    time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Try alternative n values
    n_options = [n_init + 2, n_init + 4, n_init + 8, n_init - 2, n_init + 6]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n_init]

    for n_try in n_options:
        if tleft() < 60:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(30, tleft() * 0.08)
        h_g2, c5_g2 = greedy_full(h_bin, max_iters=20000, time_limit=allowed,
                                  start_time=start_time, scan_width=15)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 40:
            continue
        allowed2 = min(40, tleft() * 0.08)
        max_it2 = int(min(80000, allowed2 * 2000))
        h_sa3, c5_sa3 = sa_targeted(h_g2, target_try, max_it2, 0.01, 0.0001, rng,
                                    time_limit=allowed2, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 5: Legendre init + optimization
    if tleft() > 120:
        h_leg = legendre_init(n_init, rng)
        allowed = min(40, tleft() * 0.08)
        h_g3, c5_g3 = greedy_full(h_leg, max_iters=30000, time_limit=allowed,
                                  start_time=start_time, scan_width=20)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()
        if tleft() > 60:
            allowed2 = min(60, tleft() * 0.08)
            max_it3 = int(min(100000, allowed2 * 1500))
            h_sa4, c5_sa4 = sa_targeted(h_g3, target_init, max_it3, 0.01, 0.00005, rng,
                                        time_limit=allowed2, start_time=start_time)
            if c5_sa4 < best_score:
                best_score = c5_sa4
                best_h = h_sa4.copy()

    # Phase 6: ILS perturbations
    if tleft() > 100:
        for pert_iter in range(5):
            if tleft() < 40:
                break
            n_pert = 3 + pert_iter
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(30, tleft() * 0.06)
            h_pg, c5_pg = greedy_full(h_pert, max_iters=15000, time_limit=allowed,
                                      start_time=start_time, scan_width=15)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

            if tleft() < 30:
                break
            allowed2 = min(25, tleft() * 0.04)
            max_it4 = int(min(50000, allowed2 * 2000))
            h_ps, c5_ps = sa_targeted(h_pg, int(len(h_pg) / 2), max_it4,
                                      0.005, 0.00002, rng,
                                      time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 50, 150, 400, 1000, 3000, 8000, 15000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 5000, 1e-15,
                                 time_limit=min(tleft() - 20, 300), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Binary SA on the refined solution's binary projection
    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(60, tleft() - 10)
        max_it5 = int(min(200000, allowed * 2000))
        h_sf, c5_sf = sa_targeted(h_bin_f, target_use, max_it5, 0.005, 0.00002,
                                  rng, time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 9: Final SLSQP polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [5000, 10000, 20000, 50000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 3000, 1e-16,
                                   time_limit=max(5.0, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
