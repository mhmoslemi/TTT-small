import numpy as np

_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

        max_val = float(corr.max())

        top_indices = np.argsort(corr)[-top_k:][::-1]

        top_shifts = top_indices - (n - 1)

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    se = float(np.log(np.sum(np.exp(shifted))) / alpha)
    return c_max + se


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def quasi_random_init(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = []
    for i in range(ones_count):
        frac = ((i + 1) / golden) % 1.0
        positions.append(int(frac * n))
    positions = sorted(set(positions))
    while len(positions) < ones_count:
        candidates = [p for p in range(n) if p not in set(positions)]
        if not candidates:
            break
        pos = candidates[rng.randint(len(candidates))]
        positions.append(pos)
    positions = positions[:ones_count]
    h[positions] = 1.0
    return h


def thue_morse_init(n, rng):
    """Generate Thue-Morse based initialization."""
    # Thue-Morse sequence: t(k) = popcount(k) mod 2
    # We want n/2 ones in total, so we map Thue-Morse values
    h = np.zeros(n)
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    # Count ones in Thue-Morse
    ones = np.where(tm == 1)[0]
    zeros = np.where(tm == 0)[0]
    target = n // 2
    if len(ones) >= target:
        # Shuffle and take first target
        rng.shuffle(ones)
        h[ones[:target]] = 1.0
    else:
        h[ones] = 1.0
        need = target - len(ones)
        rng.shuffle(zeros)
        h[zeros[:need]] = 1.0
    return h


def crossover(h1, h2, target, rng):
    """Combine two solutions: for each position, take from the one that gives better local correlation."""
    n = len(h1)
    h_new = np.zeros(n)
    # Split into random segments and take from h1 or h2
    splits = sorted(rng.choice(n - 1, size=min(10, n // 50), replace=False))
    splits = [0] + list(splits) + [n]
    for s in range(len(splits) - 1):
        a, b = splits[s], splits[s + 1]
        # Take from h1 if it has fewer 1s in this segment, else h2
        if h1[a:b].sum() <= h2[a:b].sum():
            h_new[a:b] = h1[a:b]
        else:
            h_new[a:b] = h2[a:b]
    return project_binary(h_new, target)


def worst_shift_analysis(corr, n, top_m=20):
    """Return the top m worst shifts and their values."""
    idx = np.argsort(corr)[-top_m:][::-1]
    shifts = idx - (n - 1)
    vals = corr[idx]
    return shifts, vals

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = top_indices - (n - 1)
        top_vals = corr[top_indices]

        found_improve = False
        n_try = min(400, max(80, len(ones) // 3))

        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            pos_i = i + top_shifts
            pos_j = j + top_shifts
            valid_i = (pos_i >= 0) & (pos_i < n)
            valid_j = (pos_j >= 0) & (pos_j < n)
            c1 = np.where(valid_i, 1.0 - h[np.clip(pos_i, 0, n - 1)], 0.0)
            c0 = np.where(valid_j, 1.0 - h[np.clip(pos_j, 0, n - 1)], 0.0)
            new_top_vals = top_vals + c0 - c1
            if new_top_vals.max() < max_val - 1e-10:
                h[i] = 0
                h[j] = 1
                corr = compute_corr(h)
                new_max = float(corr.max())
                if new_max < max_val:
                    new_c5 = new_max * (2.0 / n)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    stall = 0
                    found_improve = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        if not found_improve:
            stall += 1

    return best_h, best_c5


def sa_advanced(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 12000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        move_type = rng.randint(10)
        h_new = h.copy()
        moved = False

        if move_type <= 2:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 3:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif move_type <= 5:
            i = ones[rng.randint(len(ones))]
            d = rng.randint(1, 8) * (1 if rng.random() < 0.5 else -1)
            j = i + d
            if 0 <= j < n and h[j] == 0:
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 6:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        elif move_type == 7:
            bl = rng.randint(2, 5)
            for _ in range(10):
                s1 = rng.randint(0, n - bl + 1)
                s2 = rng.randint(0, n - bl + 1)
                if abs(s1 - s2) >= bl and np.all(h[s1:s1+bl] == 1) and np.all(h[s2:s2+bl] == 0):
                    h_new[s1:s1+bl] = 0
                    h_new[s2:s2+bl] = 1
                    moved = True
                    break
        else:
            i = ones[rng.randint(len(ones))]
            d = rng.randint(10, max(11, n // 8)) * (1 if rng.random() < 0.5 else -1)
            j = i + d
            if 0 <= j < n and h[j] == 0:
                h_new[i] = 0
                h_new[j] = 1
                moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def subgradient_polish(h_start, target, max_iters, time_limit=None, start_time=None):
    n = len(h)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    eta = 0.01
    dx = 2.0 / n

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        corr = compute_corr(h)
        c5_cur = float(corr.max()) * dx

        if c5_cur < best_score:
            best_score = c5_cur
            best_h = h.copy()

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + k_star < n:
                grad[i] += (1.0 - h[i + k_star])
            if 0 <= i - k_star < n:
                grad[i] -= h[i - k_star]
        grad *= dx

        eta = 0.01 / (1.0 + 0.001 * it)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, target)

        score_new = compute_c5(h_new)
        if score_new <= best_score + 0.001:
            h = h_new

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def quasi_random_init(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = []
    for i in range(ones_count):
        frac = ((i + 1) / golden) % 1.0
        positions.append(int(frac * n))
    positions = sorted(set(positions))
    while len(positions) < ones_count:
        candidates = [p for p in range(n) if p not in set(positions)]
        if not candidates:
            break
        pos = candidates[rng.randint(len(candidates))]
        positions.append(pos)
    positions = positions[:ones_count]
    h[positions] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    h_g, c5_g = fast_greedy(h_bin, max_iters=50000, rng=rng,
                            time_limit=min(80, tleft() * 0.1), start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    allowed = min(120, tleft() * 0.18)
    max_it = int(min(250000, allowed * 2500))
    h_sa, c5_sa = sa_advanced(h_g, target_init, max_it, 0.02, 0.00003, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    for ils_iter in range(10):
        if tleft() < 40:
            break
        n_pert = max(2, int(3 + ils_iter * 1.5))
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            pt = rng.randint(3)
            if pt == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
            elif pt == 1:
                i = ones[rng.randint(len(ones))]
                d = rng.randint(1, 10) * (1 if rng.random() < 0.5 else -1)
                j = i + d
                if 0 <= j < n_init and h_pert[j] == 0:
                    h_pert[i] = 0
                    h_pert[j] = 1
            else:
                bl = rng.randint(2, 5)
                s1 = rng.randint(0, n_init - bl)
                s2 = rng.randint(0, n_init - bl)
                if abs(s1 - s2) >= bl and np.all(h_pert[s1:s1+bl] == 1) and np.all(h_pert[s2:s2+bl] == 0):
                    h_pert[s1:s1+bl] = 0
                    h_pert[s2:s2+bl] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        allowed = min(40, tleft() * 0.08)
        h_ig, c5_ig = fast_greedy(h_pert, max_iters=20000, rng=rng,
                                  time_limit=allowed, start_time=start_time)
        if c5_ig < best_score:
            best_score = c5_ig
            best_h = h_ig.copy()

        if tleft() < 30:
            break
        allowed2 = min(20, tleft() * 0.06)
        h_sa2, c5_sa2 = sa_advanced(h_ig, target_init, int(min(50000, allowed2 * 3000)),
                                    0.005, 0.00005, rng,
                                    time_limit=allowed2, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    n_candidates = [n_init + 4, n_init + 8, n_init - 4, n_init + 12, n_init + 6, n_init - 8]
    n_candidates = [n for n in n_candidates if n >= 200 and n != n_init and n % 2 == 0]

    for n_try in n_candidates:
        if tleft() < 35:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(25, tleft() * 0.08)
        h_g2, c5_g2 = fast_greedy(h_bin_try, max_iters=15000, rng=rng,
                                  time_limit=allowed, start_time=start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    if tleft() > 50:
        h_qr = quasi_random_init(n_init, rng)
        allowed = min(25, tleft() * 0.08)
        h_qg, c5_qg = fast_greedy(h_qr, max_iters=10000, rng=rng,
                                  time_limit=allowed, start_time=start_time)
        if c5_qg < best_score:
            best_score = c5_qg
            best_h = h_qg.copy()

    n_use = len(best_h)
    target_use = float(n_use) / 2.0

    if tleft() > 40:
        alphas_coarse = [10, 25, 50, 100, 200, 500, 1000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas_coarse, 3000, 1e-14,
                                 time_limit=min(tleft() - 25, 120), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    if tleft() > 35:
        h_sg, sc_sg = subgradient_polish(best_h, target_use, 30000,
                                         time_limit=min(tleft() - 20, 60), start_time=start_time)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg.copy()

    if tleft() > 30:
        alphas_fine = [200, 500, 1000, 2000, 4000, 8000, 15000, 25000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas_fine, 5000, 1e-15,
                                   time_limit=min(tleft() - 20, 100), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    if tleft() > 20:
        h_sg2, sc_sg2 = subgradient_polish(best_h, target_use, 50000,
                                           time_limit=min(tleft() - 15, 50), start_time=start_time)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2.copy()

    if tleft() > 18:
        alphas_final = [5000, 10000, 20000, 40000, 80000]
        h_ref3, sc3 = slsqp_refine(best_h, n_use, target_use, alphas_final, 5000, 1e-15,
                                   time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    if tleft() > 10:
        h_sg3, sc_sg3 = subgradient_polish(best_h, target_use, 20000,
                                           time_limit=max(3.0, tleft() - 5), start_time=start_time)
        if sc_sg3 < best_score:
            best_score = sc_sg3
            best_h = h_sg3.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
