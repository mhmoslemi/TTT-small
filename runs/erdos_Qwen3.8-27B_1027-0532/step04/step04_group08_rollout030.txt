```python
    # d/dh[i]: (1-h[i+k]) - h[i-k] (with bounds)
    grad = np.zeros(n)

        w = wt[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n-1)
        # For each i, the contribution to grad[i] is:
        # w * [(1-h[i+k]) if i+k<n else 0] - w * [h[i-k] if i-k>=0 else 0]
        if k >= 0:
            # i ranges from 0 to n-1-k for first term
            # i ranges from k to n-1 for second term
            grad[:n-k] += w * (1.0 - h[k:n])
            grad[k:] -= w * h[:n-k]
        else:
            j = -k
            grad[j:] += w * (1.0 - h[:n-j])
            grad[:n-j] -= w * h[j:]
    return grad
```

For the pairwise CD, I'll implement it as follows:
- For a pair (i, j), keep s = h[i] + h[j] fixed
- Compute the 1D function f(t) = max_m corr[m](t) where h[i]=t, h[j]=s-t
- Minimize f(t) over t ∈ [max(0,s-1), min(1,s)] using golden section or similar

The key computation for f(t):
For each shift m (with k = m - (n-1)):
corr[m](t) = base[m] + t * a[m] + (s-t) * b[m] + (quadratic terms if applicable)

where:
- a[m] = coefficient of h[i] = (1-h[i+k]) if i+k valid, minus h[i-k] if i-k valid
- b[m] = coefficient of h[j] = (1-h[j+k]) if j+k valid, minus h[j-k] if j-k valid
- base[m] = corr[m] - h[i]*a[m] - h[j]*b[m] (removing the i and j contributions)

But wait, there are edge cases where i+k == j or j+k == i, in which case the term h[i+k] or h[j+k] is actually h[j] or h[i], creating a quadratic term.

Let me handle this properly. For shift k:
- If i+k == j: the term h[i]*(1-h[i+k]) = h[i]*(1-h[j]) = h[i] - h[i]*h[j] = t - t*(s-t) = t - st + t²
  This contributes t² to the correlation.
- If j+k == i: the term h[j]*(1-h[j+k]) = h[j]*(1-h[i]) = h[j] - h[i]*h[j] = (s-t) - t*(s-t) = s - t - st + t²
  
So for these special shifts, there's a +t² term, making the function convex (good for minimization).
For other shifts, the function is linear in t.

The max of convex functions is convex, so f(t) is convex! We can use standard 1D convex minimization.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_obj_smooth(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def grad_soft_max(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = wt[m_idx]
        if w < 1e-13:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:n])
            grad[k:] -= w * h[:n - k]
        else:
            j = -k
            grad[j:] += w * (1.0 - h[:n - j])
            grad[:n - j] -= w * h[j:]
    return grad


def shift_gradient(h, m_idx):
    n = len(h)
    k = m_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:n])
        grad[k:] -= h[:n - k]
    else:
        j = -k
        grad[j:] += (1.0 - h[:n - j])
        grad[:n - j] -= h[j:]
    return grad


def pairwise_cd_pass(h, target, n, n_sweeps, deadline, rng):
    dx = 2.0 / n
    best_score = compute_c5(h)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            i = int(idx)
            j = int(rng.randint(n))
            if i == j:
                continue
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-12:
                continue

            def f_of_t(t_val):
                h[i] = t_val
                h[j] = s - t_val
                return float(np.max(compute_corr(h)))

            try:
                res = minimize_scalar(f_of_t, bounds=(t_lo, t_hi),
                                      method='bounded', options={'xatol': 1e-10})
                h[i] = float(res.x)
                h[j] = s - float(res.x)
                sc = float(res.fun) * dx
                if sc < best_score:
                    best_score = sc
            except Exception:
                h[i] = t_hi * 0.5 + t_lo * 0.5
                h[j] = s - h[i]
    return h, best_score


def shift_focused_descent(h, target, n, deadline, lr_init=0.01):
    dx = 2.0 / n
    best_score = compute_c5(h)
    h_cur = h.copy()
    for it in range(2000):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        m_star = int(np.argmax(corr))
        g = shift_gradient(h_cur, m_star)
        lr = lr_init
        improved = False
        for _ in range(15):
            h_new = project_feasible(h_cur - lr * g, target)
            new_corr = compute_corr(h_new)
            new_max = float(np.max(new_corr))
            if new_max < c_max:
                h_cur = h_new
                sc = new_max * dx
                if sc < best_score:
                    best_score = sc
                c_max = new_max
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h_cur, best_score


def momentum_descent(h, target, n, deadline, alpha=5000, lr_init=0.005, momentum=0.85):
    dx = 2.0 / n
    best_score = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stall = 0
    for it in range(5000):
        if time.time() > deadline:
            break
        h_look = project_feasible(h + momentum * v, target)
        g = grad_soft_max(h_look, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 40:
                lr = lr_init
                v = np.zeros(n)
                stall = 0
            if lr < 1e-13:
                lr = lr_init
                v = np.zeros(n)
                stall = 0
    return best_h, best_score


def binary_swap_search(h, target, n, deadline, rng):
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    best_score = compute_c5(h_bin)
    corr = compute_corr(h_bin)
    while time.time() < deadline:
        c_max_dt = float(np.max(corr))
        if c_max_dt * dx >= best_score + 1e-10:
            h_bin = (h > 0.5).astype(float)
            h_bin = project_feasible(h_bin, target)
            corr = compute_corr(h_bin)
            continue
        worst_m = int(np.argmax(corr))
        k = worst_m - (n - 1)
        if k >= 0:
            conflicts = np.array([i for i in range(n - k)
                                  if h_bin[i] == 1.0 and h_bin[i + k] == 0.0])
        else:
            j = -k
            conflicts = np.array([i for i in range(j, n)
                                  if h_bin[i] == 1.0 and h_bin[i - j] == 0.0])
        if len(conflicts) == 0:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        improved = False
        for _ in range(30):
            i = int(conflicts[rng.randint(len(conflicts))])
            j = int(zeros[rng.randint(len(zeros))])
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_corr = compute_corr(h_bin)
            new_max = float(np.max(new_corr))
            if new_max * dx < best_score:
                best_score = new_max * dx
                corr = new_corr
                improved = True
                conflicts = np.array([i for i in range(n - k)
                                      if h_bin[i] == 1.0 and h_bin[i + k] == 0.0]) if k >= 0 else \
                            np.array([i for i in range(-k, n)
                                      if h_bin[i] == 1.0 and h_bin[i - k] == 0.0])
                ones = np.where(h_bin == 1.0)[0]
                zeros = np.where(h_bin == 0.0)[0]
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            h_bin = (h > 0.5).astype(float)
            h_bin = project_feasible(h_bin, target)
            break
    return h_bin, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    h = project_feasible(h0, target)
    best_score = compute_c5(h)
    best_h = h.copy()

    def tl():
        return deadline - time.time()

    # Phase 1: L-BFGS-B with soft-max, increasing alpha
    for alpha in [50, 200, 1000, 5000, 20000, 80000]:
        if tl() < 80:
            break

        def obj_lbfgs(x):
            return soft_max_obj_smooth(x, alpha)

        def grad_lbfgs(x):
            return grad_soft_max(x, alpha)

        x0 = project_feasible(best_h, target)
        try:
            res = minimize(obj_lbfgs, x0, jac=grad_lbfgs, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-12,
                                    'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Shift-focused descent (reduce worst shift)
    if tl() > 80:
        for lr_init in [0.01, 0.005, 0.002]:
            if tl() < 30:
                break
            dl = time.time() + min(40.0, tl() * 0.3)
            h_sf, sc = shift_focused_descent(best_h, target, n, dl, lr_init)
            if sc < best_score:
                best_score = sc
                best_h = h_sf

    # Phase 3: Pairwise coordinate descent (multiple passes)
    if tl() > 100:
        dl_pw = min(time.time() + tl() * 0.35, deadline - 60)
        h_pw, sc_pw = pairwise_cd_pass(best_h, target, n, 3, dl_pw, rng)
        if sc_pw < best_score:
            best_score = sc_pw
            best_h = h_pw

    # Phase 4: Momentum gradient descent with adaptive alpha
    if tl() > 80:
        for alpha, lr_i, mom in [(2000, 0.003, 0.85), (10000, 0.002, 0.80), (50000, 0.001, 0.75)]:
            if tl() < 40:
                break
            dl = time.time() + min(50.0, tl() * 0.3)
            h_md, sc = momentum_descent(best_h, target, n, dl, alpha, lr_i, mom)
            if sc < best_score:
                best_score = sc
                best_h = h_md

    # Phase 5: More pairwise CD (finer)
    if tl() > 80:
        dl_pw2 = min(time.time() + tl() * 0.3, deadline - 40)
        h_pw2, sc_pw2 = pairwise_cd_pass(best_h, target, n, 4, dl_pw2, rng)
        if sc_pw2 < best_score:
            best_score = sc_pw2
            best_h = h_pw2

    # Phase 6: Binary swap search
    if tl() > 60:
        dl_bin = min(time.time() + tl() * 0.25, deadline - 30)
        h_bin, sc_bin = binary_swap_search(best_h, target, n, dl_bin, rng)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 7: L-BFGS-B polish on best solution
    if tl() > 40:
        for alpha in [50000, 200000, 500000]:
            if tl() < 20:
                break

            def obj_p(x):
                return soft_max_obj_smooth(x, alpha)

            def grad_p(x):
                return grad_soft_max(x, alpha)

            x0 = project_feasible(best_h, target)
            try:
                res = minimize(obj_p, x0, jac=grad_p, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-13})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Phase 8: Final shift-focused + pairwise polish
    if tl() > 20:
        dl = time.time() + min(15.0, tl() * 0.4)
        h_sf, sc = shift_focused_descent(best_h, target, n, dl, 0.003)
        if sc < best_score:
            best_score = sc
            best_h = h_sf

    if tl() > 10:
        h_pw3, sc_pw3 = pairwise_cd_pass(best_h, target, n, 2, time.time() + min(10.0, tl() * 0.5), rng)
        if sc_pw3 < best_score:
            best_score = sc_pw3
            best_h = h_pw3

    # Phase 9: Final momentum with very high alpha
    if tl() > 15:
        dl = time.time() + min(12.0, tl() * 0.5)
        h_md, sc = momentum_descent(best_h, target, n, dl, 100000, 0.0005, 0.7)
        if sc < best_score:
            best_score = sc
            best_h = h_md

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
```
