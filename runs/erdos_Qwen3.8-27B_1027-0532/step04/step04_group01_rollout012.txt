Let me also think about the **spectral approach**. If we write h in terms of its DFT:
h[j] = (1/n) ∑_k H[k] e^{2πijk/n}

Then the autocorrelation is:
(h ⋆ h)[k] = ∑_j h[j] h[j+k] = (1/n) ∑_m |H[m]|² e^{2πimk/n}

And the overlap at lag k (for the full correlation, not circular):
overlap[k] = ∑_j h[j](1-h[j+k]) = ∑_j h[j] - (h ⋆ h)[k] (approximately, ignoring boundary effects)

For circular correlation (which is a good approximation for "interior" lags):
overlap_circ[k] = sum(h) - (1/n) ∑_m |H[m]|² e^{2πimk/n}

We want to minimize max_k overlap_circ[k] = sum(h) - min_k (1/n) ∑_m |H[m]|² e^{2πimk/n}

This is equivalent to maximizing min_k Re(∑_m |H[m]|² e^{2πimk/n})

The term (1/n) ∑_m |H[m]|² e^{2πimk/n} is the circular autocorrelation at lag k. So we want to maximize the minimum autocorrelation (excluding lag 0).

This is related to **low autocorrelation sequences** (CAS) in radar/communications. The problem of minimizing the maximum sidelobe of the autocorrelation is well-studied!

For binary sequences, optimal sequences include:
- Barker sequences (length 1-13)
- Zadoff-Chu sequences
- Zadoff sequences
- Legendre sequences
- m-sequences

For the specific problem of minimizing the maximum aperiodic autocorrelation, the best known sequences are related to **Sidon sequences** and **Möbius sequences**.

But our problem is slightly different because:
1. We have exactly n/2 ones (balanced)
2. We're minimizing the max of h(x)(1-h(x+k)), not the full autocorrelation
3. We have boundary effects (aperiodic, not circular)

Still, the spectral view suggests that we want |H[k]|² to be as flat as possible (for k ≠ 0), which is the property of "perfect sequences" or "low PAPR sequences."

This gives another initialization strategy: use known low-autocorrelation sequences as starting points.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.asarray(h, dtype=np.float64)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)


def project_binary(h, target):
    n = len(h)
    h_r = np.round(h).astype(np.int64)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    elif current > target:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(np.float64)


def smooth_max_value(h, alpha):
    n = len(h)
    dx = 2.0 / n

    z = alpha * dx * corr
    z_max = float(np.max(z))
    exp_z = np.exp(z - z_max)
    return float((z_max + np.log(np.sum(exp_z))) / alpha)

def make_qr_init(n, rng):
    def is_prime(x):
        if x < 2:
            return False
        if x < 4:
            return True
        if x % 2 == 0 or x % 3 == 0:
            return False
        i = 5
        while i * i <= x:
            if x % i == 0 or x % (i + 2) == 0:
                return False
            i += 6
        return True

    p = n + 1
    while not is_prime(p):
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    target = n // 2
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

        # Also try binary search from this
        if remaining > 100:
            h_b3 = project_binary(h_c3, target_init)
            tl3b = min(60.0, remaining * 0.08)
            h_ls3, c5_ls3 = binary_local_search(h_b3, target_init, tl3b, t_start, rng)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()
        remaining = budget_s - (time.time() - t_start)

    # Phase 4: Try different n values
    n_trials = [n_init + 4, n_init + 12, n_init + 32, n_init + 64, n_init + 128]
    for nt in n_trials:
        if remaining < 70:
            break
        target_nt = nt / 2.0
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, nt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_nt)
        tl4 = min(50.0, remaining * 0.06)
        h_c4, c5_c4 = continuous_optimize(h_interp, target_nt, tl4, t_start, seed=seed + nt)
        if c5_c4 < best_c5:
            best_c5 = c5_c4
            best_h = h_c4.copy()
        if remaining > 60:
            h_b4 = project_binary(h_c4, target_nt)
            tl4b = min(40.0, remaining * 0.04)
            h_ls4, c5_ls4 = binary_local_search(h_b4, target_nt, tl4b, t_start, rng)
            if c5_ls4 < best_c5:
                best_c5 = c5_ls4
                best_h = h_ls4.copy()
        remaining = budget_s - (time.time() - t_start)

    # Phase 5: Final binary polish
    if remaining > 30:
        n_f = len(best_h)
        t_f = n_f / 2.0
        h_bin_f = project_binary(best_h, t_f)
        tl5 = max(10.0, remaining - 10.0)
        h_ls_f, c5_ls_f = binary_local_search(h_bin_f, t_f, tl5, t_start, rng)
        if c5_ls_f < best_c5:
            best_c5 = c5_ls_f
            best_h = h_ls_f.copy()

    # Phase 6: Final continuous polish
    remaining = budget_s - (time.time() - t_start)
    if remaining > 20:
        n_f = len(best_h)
        t_f = n_f / 2.0
        tl6 = max(5.0, remaining - 5.0)
        h_c6, c5_c6 = continuous_optimize(best_h.copy(), t_f, tl6, t_start, seed=seed + 999)
        if c5_c6 < best_c5:
            best_c5 = c5_c6
            best_h = h_c6.copy()

    n_out = len(best_h)
    h_out = project_feasible(best_h, n_out / 2.0)
    c5_out = compute_c5(h_out)
    if c5_out > best_c5:
        h_out = best_h
        c5_out = best_c5

    if time.time() - t_start > budget_s - 2:
        return best_h, best_c5, n_out

    return h_out, c5_out, n_out
