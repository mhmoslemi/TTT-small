```python
def binary_local_search(h_binary, n, max_iters=10000, deadline=None):
    """Efficient binary local search using M-matrix."""
    h = h_binary.copy()
    F = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    current_max = F.max()
    
    # Build M matrix: M[idx, p] = change in F[idx] when flipping h[p] from 1 to 0
    # M[idx, p] = [p ≤ idx ≤ n-1+p] * (h[n-1+p-idx] - 1) + [n-1-p ≤ idx ≤ 2(n-1)-p] * h[idx+p-(n-1)]
    
    # Precompute M
    M = np.zeros((2*n-1, n))
    for p in range(n):
        # First term: idx from p to n-1+p
        idx_range1 = range(max(p, 0), min(n-1+p, 2*n-1))
        for idx in idx_range1:
            j = n-1+p-idx  # this is h[j]
            if 0 <= j < n:
                M[idx, p] += h[j] - 1
        # Second term: idx from n-1-p to 2(n-1)-p
        idx_range2 = range(max(n-1-p, 0), min(2*(n-1)-p, 2*n-1))
        for idx in idx_range2:
            j = idx+p-(n-1)  # this is h[j]
            if 0 <= j < n:
                M[idx, p] += h[j]
    
    # Local search
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_score = current_max
        best_p = -1
        best_q = -1
        
        # Try all pair flips (p in ones, q in zeros)
        # Score for flip (p,q): max(F + M[:,p] - M[:,q])
        
        # Vectorized: for each p, compute F + M[:,p], then for each q compute max of that minus M[:,q]
        # This is (n/2 × n/2) scores, each requiring max over (2n-1) elements
        
        # To keep memory reasonable, do it in batches
        batch_size = 50
        for p_start in range(0, len(ones), batch_size):
            p_batch = ones[p_start:p_start+batch_size]
            # F + M[:, p_batch] has shape (2n-1, batch_size)
            F_plus = F[:, None] + M[:, p_batch]  # (2n-1, batch)
            # Subtract M[:, q] for all q in zeros: shape (2n-1, batch, len(zeros))
            # This might be too large, so do sub-batches
            for q_start in range(0, len(zeros), batch_size):
                q_batch = zeros[q_start:q_start+batch_size]
                # scores[i,j] = max(F_plus[:,i] - M[:,q_batch[j]])
                # = max(F + M[:,p_i] - M[:,q_j])
                diffs = F_plus[:, :, None] - M[:, q_batch][None, :, :]  # (2n-1, batch_p, batch_q)
                scores = diffs.max(axis=0)  # (batch_p, batch_q)
                min_idx = np.unravel_index(scores.argmin(), scores.shape)
                if scores[min_idx] < best_score:
                    best_score = scores[min_idx]
                    best_p = p_batch[min_idx[0]]
                    best_q = q_batch[min_idx[1]]
        
        if best_p >= 0 and best_score < current_max:
            h[best_p] = 0
            h[best_q] = 1
            F = F + M[:, best_p] - M[:, best_q]
            current_max = F.max()
        else:
            # No improving move; try random (annealing)
            break
    
    return h, current_max
```
