def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

    else:
        idx = np.where(h > 0.0)[0]

        return h

    return np.clip(h, 0.0, 1.0)

import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def cons_fun(x):
    return np.sum(x) - _cons_target

def cons_jac(x):
    return np.ones(_cons_n)

def obj_lse(x, alpha, n, target):
    h = x  # assume already feasible
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    dx = 2.0 / n
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    lse = np.log(np.sum(np.exp(shifted)))
    return c_max * dx + lse * dx / alpha

def grad_lse(x, alpha, n, target):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    dx = 2.0 / n
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))

        if w < 1e-12:

        m = k - (n - 1)

            grad[m:] += w * (1.0 - h[:n - m])

        else:

    step_sizes = [0.01, 0.005, 0.001]
    
    for ss in step_sizes:
        for sweep in range(3):
            if (time.time() - start_time) > time_limit:
                return best_h, best_c5
            for i in range(n):
                if (time.time() - start_time) > time_limit:
                    return best_h, best_c5
                h_save = h[i]
                # Try + and -
                for direction in [1, -1]:
                    new_val = h[i] + direction * ss
                    if new_val < 0 or new_val > 1:
                        continue
                    h[i] = new_val
                    h_proj = project_feasible(h, target)
                    c5_try = compute_c5(h_proj)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_proj.copy()
                        h = h_proj
                    else:
                        h[i] = h_save
                h[i] = h_save
                # After each coordinate, use the projected best
                h = project_feasible(best_h, target) if compute_c5(best_h) < compute_c5(h) else h
    
    return best_h, best_c5

def multi_start_refine(h_start, target, n_start, time_limit, start_time, rng):
    global _cons_n, _cons_target
    n = len(h_start)
    best_h = h_start.copy().astype(float)
    best_c5 = compute_c5(best_h)
    
    for s in range(n_start):
        if (time.time() - start_time) > time_limit:
            break
        # Generate perturbation
        noise_scale = 0.1 * (1.0 - s / n_start)
        h_pert = best_h + noise_scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, target)
        
        # Optimize the perturbed solution
        time_left = time_limit - (time.time() - start_time)
        if time_left < 30:
            break
        h_opt, c5_opt = slsqp_optimize(h_pert, target, min(60.0, time_left * 0.7), start_time,
                                       alphas=[5000, 20000, 50000])
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5
