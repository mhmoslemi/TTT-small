def project_feasible_l2(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum} in L2 sense."""
    h = np.asarray(h, dtype=float)
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo, hi = -max(h.max(), 1.0) - 1, max(-h.min(), 1.0) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-15:
            break
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

Now, let me think about the subgradient method with Nesterov acceleration:

Standard subgradient: h_{t+1} = project(h_t - lr_t * g_t)
Nesterov: 
  y_t = h_t + (t-1)/(t+2) * (h_t - h_{t-1})
  h_{t+1} = project(y_t - lr_t * g(y_t))

For our non-smooth problem (max of smooth functions), the subgradient at h is:
g(h) = sum_k w_k * grad O_k(h) where w_k comes from the subdifferential of max

The subdifferential of max(x_1, ..., x_n) at x is the convex hull of {e_i : x_i = max}. So w_k is any probability distribution over the argmax set.

For a practical method, we can use:
- w_k = 1 (for the argmax) — Bregman subgradient
- w_k ∝ exp(O_k / t) — softmax approximation (smoother)

The softmax approach gives a "smoothed subgradient" which is actually the gradient of the log-sum-exp. So the subgradient method with softmax weights IS the gradient descent on the log-sum-exp! The difference is that we can adapt the temperature.

So my "subgradient with adaptive temperature" is actually just gradient descent on the log-sum-exp with changing alpha. The Nesterov acceleration applies to this smooth function.

FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) on the log-sum-exp:
  L = Lipschitz constant of grad(lse)
  y_t = h_t + (t_{t-1} - 1) / (t_t + 2) * (h_t - h_{t-1})
  h_{t+1} = project(y_t - (1/L) * grad_lse(y_t))
  t_{t+1} = (1 + sqrt(1 + 4*t_t²)) / 2

The Lipschitz constant L of grad_lse is bounded by alpha * (max eigenvalue of Hessian of O_k). Since O_k is bilinear with coefficient dx, the Hessian eigenvalues are O(dx). So L ≈ alpha * C * dx for some constant C.

For n=400, dx = 0.005. If alpha = 1000, L ≈ 1000 * 1 * 0.005 = 5. So lr = 1/L ≈ 0.2.

    """Nesterov-accelerated gradient descent on log-sum-exp with projection."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    t_moment = 1.0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Anneal alpha
        frac = step / max(n_steps - 1, 1)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_moment * t_moment)) / 2.0
        beta = (t_moment - 1.0) / t_new

        f_cur = lse_value(h_cur, alpha)

        h_cur = h_new
        t_moment = t_new

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_block_full = None
_g_block_start = 0
_g_block_end = 0
_g_block_alpha = 0.0


def compute_overlaps(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_l2(h, target):
    h = np.asarray(h, dtype=float)
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def lse_value(h, alpha):
    over = compute_overlaps(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_value_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    over = compute_overlaps(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    val = c_max + float(np.log(exp_s.sum())) / alpha
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    active = w > 1e-10
    idx = np.where(active)[0]
    for k in idx:
        k = int(k)
        wk = w[k]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h) * dx
        else:
            jp = np.arange(0, n - k)
            grad[jp] += wk * (1.0 - h[jp + k]) * dx
            jn = np.arange(k, n)
            grad[jn] -= wk * h[jn - k] * dx
    return val, grad


def _slsqp_obj(x):
    v, _ = lse_value_grad(x, _g_alpha)
    return v


def _slsqp_grad(x):
    _, g = lse_value_grad(x, _g_alpha)
    return g


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_at_alpha(h, alpha, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_alpha = alpha
    n = len(h)
    _g_n = n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        res = minimize(_slsqp_obj, h.copy(), jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_l2(res.x, _g_target)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            return h_new
    except Exception:
        pass
    return h


def slsqp_continuation(h, alphas, maxiter, ftol, deadline):
    h_cur = h.copy()
    for a in alphas:
        if time.time() > deadline:
            break
        h_new = slsqp_at_alpha(h_cur, a, maxiter, ftol)
        if compute_c5(h_new) < compute_c5(h_cur) - 1e-12:
            h_cur = h_new
    return h_cur


def nesterov_lse(h, n_steps, alpha_init, alpha_final, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    h_prev = h.copy()
    grad_prev = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    t_mom = 1.0

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(n_steps - 1, 1)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        y = h_cur + beta * (h_cur - h_prev)
        y_clipped = np.clip(y, 0.0, 1.0)

        val_y, grad_y = lse_value_grad(y_clipped, alpha)

        lr = 1.0 / (alpha * dx * 4.0 + 1e-6)
        lr = min(lr, 0.5)
        h_new = y - lr * grad_y
        h_new = project_l2(h_new, target)

        f_new = lse_value(h_new, alpha)
        back = 0
        while f_new > val_y * (1.0 + 1e-8) + 1e-12 and back < 15:
            lr *= 0.5
            h_new = y - lr * grad_y
            h_new = project_l2(h_new, target)
            f_new = lse_value(h_new, alpha)
            back += 1

        h_prev = h_cur.copy()
        h_cur = h_new
        t_mom = t_new

        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()

    return best_h


def _block_obj(x):
    hf = _g_block_full.copy()
    hf[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    return lse_value(hf, _g_block_alpha)


def _block_grad(x):
    hf = _g_block_full.copy()
    hf[_g_block_start:_g_block_end] = np.clip(x, 0.0, 1.0)
    _, g = lse_value_grad(hf, _g_block_alpha)
    return g[_g_block_start:_g_block_end]


def block_descent(h, block_sizes, n_rounds, alpha, deadline):
    global _g_block_full, _g_block_start, _g_block_end, _g_block_alpha
    n = len(h)
    target = n / 2.0
    _g_block_alpha = alpha
    h_cur = h.copy()
    best_score = compute_c5(h_cur)

    for bs in block_sizes:
        if time.time() > deadline:
            break
        n_blocks = n // bs
        for rd in range(n_rounds):
            if time.time() > deadline:
                break
            for bi in range(n_blocks):
                s = bi * bs
                e = s + bs
                _g_block_full = h_cur.copy()
                _g_block_start = s
                _g_block_end = e
                x0 = h_cur[s:e].copy()
                try:
                    bounds = [(0.0, 1.0)] * bs
                    res = minimize(_block_obj, x0, jac=_block_grad, method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 150, 'gtol': 1e-8})
                    h_test = h_cur.copy()
                    h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                    h_test = project_l2(h_test, target)
                    sc = compute_c5(h_test)
                    if sc < best_score - 1e-12:
                        h_cur = h_test
                        best_score = sc
                except Exception:
                    pass
    return h_cur


def worst_lag_surgery(h, max_steps, initial_eps, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = initial_eps

    for step in range(max_steps):
        if time.time() > deadline:
            break
        over = compute_overlaps(h_cur)
        k_star = int(np.argmax(over))

        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = (1.0 - 2.0 * h_cur) * dx
        else:
            jp = np.arange(0, n - k_star)
            grad[jp] += (1.0 - h_cur[jp + k_star]) * dx
            jn = np.arange(k_star, n)
            grad[jn] -= h_cur[jn - k_star] * dx

        h_new = h_cur - eps * grad
        h_new = project_l2(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.05)
        else:
            eps *= 0.6
            if eps < 1e-8:
                break
    return h_cur


def subgradient_adaptive(h, n_steps, temp_init, temp_final, lr_init, lr_decay, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(n_steps - 1, 1)
        temp = temp_init * (temp_final / temp_init) ** frac

        over = compute_overlaps(h_cur)
        shifted = over / temp
        shifted -= float(np.max(shifted))
        w = np.exp(shifted)
        w /= w.sum()

        grad = np.zeros(n)
        active = w > 1e-8
        for k in np.where(active)[0]:
            k = int(k)
            wk = w[k]
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur) * dx
            else:
                jp = np.arange(0, n - k)
                grad[jp] += wk * (1.0 - h_cur[jp + k]) * dx
                jn = np.arange(k, n)
                grad[jn] -= wk * h_cur[jn - k] * dx

        h_new = h_cur - lr * grad
        h_new = project_l2(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < compute_c5(h_cur) - 1e-13:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break

        if sc_new < best_score:
            best_score = sc_new
            best_h = h_cur.copy()

    return best_h, best_score


def simulated_annealing(h, n_iter, temp_init, temp_final, rng, deadline):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    over = compute_overlaps(h_cur)
    best_over = over.copy()

    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        temp = temp_init * (temp_final / temp_init) ** frac

        move_type = rng.randint(4)
        h_try = h_cur.copy()

        if move_type == 0:
            i = rng.randint(n)
            delta = rng.randn() * 0.05
            h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
        elif move_type == 1:
            i = rng.randint(n - 1)
            h_try[i], h_try[i+1] = h_try[i+1], h_try[i]
        elif move_type == 2:
            L = rng.randint(5, 20)
            s = rng.randint(0, n - L)
            h_try[s:s+L] = 1.0 - h_try[s:s+L]
        else:
            freq = rng.randint(2, 8)
            amp = 0.03 * rng.rand()
            phase = rng.rand() * 2.0 * np.pi
            idx = np.arange(n)
            h_try += amp * np.sin(2.0 * np.pi * freq * idx / n + phase)
            h_try = np.clip(h_try, 0.0, 1.0)

        h_try = project_l2(h_try, target)
        sc_try = compute_c5(h_try)
        sc_cur = compute_c5(h_cur)

        if sc_try < sc_cur or rng.rand() < np.exp(-(sc_try - sc_cur) / temp):
            h_cur = h_try
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_cur.copy()

    return best_h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_l2(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_l2(h_out, target_dst)
