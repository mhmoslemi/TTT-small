This is a convex piecewise-linear function of d (max of linear functions). The minimum is where the subgradient contains 0, which can be found by:

- Or ternary search (since it's unimodal)

4. If 0 is in the convex hull of {g_diff(s*) : s* in argmax(corr)}, we're at the optimum
5. Otherwise, move in the direction that reduces the max

        if lo2 <= hi2:
            s_idx = np.arange(lo2, hi2 + 1)
            G[p, s_idx] -= h[p - s_idx + n - 1]
    return G

def pair_swap_optimal(h, i, j, corr, G, n):
    c = h[i] + h[j]
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-12:
        return None
    g_diff = G[i] - G[j]
    if np.all(np.abs(g_diff) < 1e-14):
        return None
    lo, hi = d_lo, d_hi
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr + m1 * g_diff)
        f2 = np.max(corr + m2 * g_diff)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d_opt = (lo + hi) / 2.0
    if abs(d_opt) < 1e-14:
        return None
    new_max = np.max(corr + d_opt * g_diff)
    return d_opt, new_max

def _soft_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x, target):
    return np.sum(x) - target

def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    h = best_h.copy()

    # Phase 1: Pair swap local search
    while time_left() > 150:
        corr = compute_corr(h)
        G = precompute_gradients(h)
        current_max = float(np.max(corr))
        improved = False
        n_improvements = 0

        # Build pair list with potential filter
        # Focus on pairs where one coord is high and other is low
        hi_mask = h > 0.55
        lo_mask = h < 0.45
        hi_indices = np.where(hi_mask)[0]
        lo_indices = np.where(lo_mask)[0]

        pairs = []
        for i in hi_indices:
            for j in lo_indices:
                if i != j:
                    pairs.append((i, j))

        # Also add some random mixed pairs
        for _ in range(min(2000, n * 5)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j and abs(h[i] - h[j]) > 0.1:
                pairs.append((i, j))

        rng.shuffle(pairs)

        seen = set()
        for (i, j) in pairs:
            if time_left() < 150:
                break
            key = (min(i, j), max(i, j))
            if key in seen:
                continue
            seen.add(key)

            c = h[i] + h[j]
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-10:
                continue

            g_diff = G[i] - G[j]
            max_abs_gd = float(np.max(np.abs(g_diff)))
            if max_abs_gd < 1e-10:
                continue

            # Quick potential check: max possible reduction
            potential = (d_hi - d_lo) * max_abs_gd
            if potential < 1e-6:
                continue

            lo, hi = d_lo, d_hi
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(corr + m1 * g_diff)
                f2 = np.max(corr + m2 * g_diff)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            d_opt = (lo + hi) / 2.0
            if abs(d_opt) < 1e-12:
                continue

            new_max = float(np.max(corr + d_opt * g_diff))
            if new_max < current_max - 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr = compute_corr(h)
                G = precompute_gradients(h)
                current_max = float(np.max(corr))
                improved = True
                n_improvements += 1

        # Update best
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        if not improved:
            # Try a random perturbation to escape local optimum
            if time_left() > 200:
                n_perturb = rng.randint(5, 20)
                h_pert = best_h.copy()
                for _ in range(n_perturb):
                    idx = rng.randint(n)
                    h_pert[idx] += rng.uniform(-0.1, 0.1)
                h_pert = project_feasible(h_pert, target)
                sc = compute_c5(h_pert)
                if sc < best_score + 0.001:
                    h = h_pert.copy()
                else:
                    h = best_h.copy()
            else:
                break

    # Phase 2: Momentum gradient with soft-max
    if time_left() > 80:
        _g_alpha = 5000.0
        corr = compute_corr(h)
        current_max = float(np.max(corr))
        v = np.zeros(n)
        lr = 0.002
        momentum = 0.85
        deadline = time.time() + min(60.0, time_left() * 0.5)

        for alpha in [2000, 8000, 30000, 100000]:
            if time.time() > deadline:
                break
            _g_alpha = alpha
            for _ in range(200):
                if time.time() > deadline or time_left() < 50:
                    break
                corr = compute_corr(h)
                c_max = float(np.max(corr))
                shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
                wt = np.exp(shifted)
                wsum = wt.sum()
                if wsum < 1e-300:
                    break
                wt = wt / wsum
                grad = np.zeros(n)
                for s_idx in range(2 * n - 1):
                    w = wt[s_idx]
                    if w < 1e-10:
                        continue
                    s = s_idx - (n - 1)
                    if s >= 0:
                        grad[s:] += w * (1.0 - h[:n - s])
                        grad[:n - s] -= w * h[s:]
                    else:
                        grad[:n + s] += w * (1.0 - h[-s:])
                        grad[-s:] -= w * h[:n + s]
                v = momentum * v - lr * grad
                h_new = project_feasible(h + v, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                    h = h_new
                    lr *= 1.05
                else:
                    lr *= 0.5
                    if lr < 1e-14:
                        lr = 0.002
            if time.time() > deadline:
                break

    # Phase 3: SLSQP polish
    if time_left() > 40:
        for a in [20000, 80000, 300000]:
            if time_left() < 20:
                break
            _g_alpha = a
            try:
                res = minimize(_soft_obj, best_h, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq', 'fun': lambda x: _cons_eq(x, target)}],
                               options={'maxiter': 300, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Phase 4: Final pair swap polish
    if time_left() > 30:
        h = best_h.copy()
        for _ in range(3):
            if time_left() < 20:
                break
            corr = compute_corr(h)
            G = precompute_gradients(h)
            current_max = float(np.max(corr))
            improved = False
            # Try all pairs with |h[i]-h[j]| > 0.05
            for i in range(n):
                if time_left() < 20:
                    break
                for j in range(i + 1, n):
                    if time_left() < 20:
                        break
                    if abs(h[i] - h[j]) < 0.05:
                        continue
                    c = h[i] + h[j]
                    d_lo = max(-h[i], h[j] - 1.0)
                    d_hi = min(1.0 - h[i], h[j])
                    if d_lo >= d_hi - 1e-10:
                        continue
                    g_diff = G[i] - G[j]
                    lo, hi = d_lo, d_hi
                    for _ in range(35):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        f1 = np.max(corr + m1 * g_diff)
                        f2 = np.max(corr + m2 * g_diff)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    d_opt = (lo + hi) / 2.0
                    if abs(d_opt) < 1e-12:
                        continue
                    new_max = float(np.max(corr + d_opt * g_diff))
                    if new_max < current_max - 1e-13:
                        h[i] += d_opt
                        h[j] -= d_opt
                        h = np.clip(h, 0.0, 1.0)
                        h = project_feasible(h, target)
                        corr = compute_corr(h)
                        G = precompute_gradients(h)
                        current_max = float(np.max(corr))
                        improved = True
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
            if not improved:
                break

    # Final projection and evaluation
    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
