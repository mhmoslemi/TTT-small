def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)

def overlap_corrs(h):

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

**Idea: Coordinate descent**
Optimize one coordinate at a time. For a fixed j, corr[k] depends on h[j] through:
- h[j]*(1-h[j-k]) for appropriate k (when j is the first factor)
- h[j+k]*(1-h[j]) for appropriate k (when j is in the (1-h) factor)

So for fixed j, the objective max_k corr[k] is a piecewise linear function of h[j] (since each corr[k] is linear in h[j] when other variables are fixed). We can find the optimal h[j] by checking the breakpoints. This is exact coordinate descent!

Let me think about this more carefully. For a fixed coordinate j:
corr[k] = sum_i h[i](1-h[i+k])

The terms involving h[j]:
- If j appears as h[i] (i.e., i=j): contributes h[j]*(1-h[j+k]) = h[j] - h[j]*h[j+k]
- If j appears as h[i+k] (i.e., i+k=j, so i=j-k): contributes h[j-k]*(1-h[j]) = h[j-k] - h[j-k]*h[j]

So corr[k] has terms:
- h[j]*(1-h[j+k]) when the pair (j, j+k) is in the correlation sum
- h[j-k]*(1-h[j]) when the pair (j-k, j) is in the correlation sum

Both are linear in h[j]! (Since h[j+k] and h[j-k] are fixed when we're optimizing h[j].)

So for each k, corr[k] is a linear function of h[j]: corr[k] = a_k * h[j] + b_k where a_k and b_k depend on the other variables.

The objective is max_k (a_k * h[j] + b_k). This is a convex piecewise linear function of h[j] ∈ [0,1]. We can find its minimum by checking where the slopes change (i.e., where pairs of lines intersect and the active set changes).

Actually, for a convex piecewise linear function on [0,1], the minimum is either at 0, at 1, or at a point where two lines cross (a "kink"). The number of kinks is at most the number of pairs of lines that cross within [0,1], which is O(m^2) where m is the number of active lags.

But there's a simpler approach: the minimum of max_k (a_k * x + b_k) over x ∈ [0,1] can be found by bisection. For a given value t, check if there exists x ∈ [0,1] such that a_k * x + b_k ≤ t for all k. This is a system of linear inequalities in x, which can be checked in O(m) time.

Actually even simpler: since it's a 1D problem, we can just evaluate at all the kink points. The kink points are where a_i * x + b_i = a_j * x + b_j for some pair (i,j), i.e., x = (b_j - b_i)/(a_i - a_j). Filter to [0,1], sort, and evaluate the objective at each.

With m active lags, there are O(m^2) kink points. For m=10, that's 55 points. For m=50, that's 1225 points. Each evaluation is O(m). So total is O(m^3) per coordinate update. For n=400 coordinates and m=50 active lags, that's 400 * 125000 = 50 million operations per full sweep. That's feasible.

Alternative: use the "free" coordinate approach. Optimize h[j] freely, then project back to the feasible set. This is like a projected coordinate descent.

def equalize_step(h_cur, target_sum, topk, max_norm):
    """Find a small step that equalizes the top correlations."""
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    top_vals = corr[top_idx]
    c_max = np.max(top_vals)
    m = len(top_idx)
    
    # Build A and b
    A = np.zeros((m, n))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :] = g
        b[i] = c_max - corr[idx]  # We want to increase low ones, decrease high ones
    
    # Actually, we want to minimize the spread. Let's minimize sum_i (corr[i] - mean)^2
    mean_corr = np.mean(top_vals)
    b2 = corr[top_idx] - mean_corr  # We want A @ dh ≈ -b2 (to reduce spread)
    
    # Solve: minimize ||A @ dh + b2||^2 subject to sum(dh) = 0
    # Using Lagrange: A^T A dh + b2 = lambda * 1
    # augment: [2*A^T*A + lambda*I ... ] 
    # Simpler: use constrained LSQ
    # Minimize ||A @ dh||^2 - 2 * b2^T A @ dh  (drop constant)
    # = (dh)^T A^T A dh - 2 (A^T b2)^T dh
    # With constraint 1^T dh = 0
    # KKT: [2 A^T A, 1; 1^T, 0] [dh; lambda] = [2 A^T b2; 0]
    
    ATA = A.T @ A
    ATb2 = A.T @ b2
    ones = np.ones(n)
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = 2 * ATA
    KKT[:n, n] = ones
    KKT[n, :n] = ones
    rhs = np.zeros(n+1)
    rhs[:n] = -2 * ATb2  # We want to go in direction -gradient to reduce spread
    
    try:
        sol = np.linalg.solve(KKT, rhs)
        dh = sol[:n]
    except:
        dh = np.linalg.lstsq(np.vstack([A, np.zeros((1,n)) + 1]), np.concatenate([b2, [0]]))[0]
    
    # Clip norm
    dh_norm = np.linalg.norm(dh)
    if dh_norm > max_norm:
        dh = dh * max_norm / dh_norm
    
    h_new = project_feasible(h_cur + dh, target_sum)
    return h_new

import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smooth_max_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    if total < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)


def slsqp_optimize(h_start, target_sum, alpha_topk_pairs, maxiter, ftol, time_limit):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    t0 = time.time()
    for alpha, topk in alpha_topk_pairs:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(
                smooth_max_obj, h_cur, args=(alpha, topk),
                jac=smooth_max_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def lp_linearized_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):

    n = len(h_init)

    v = np.zeros(n)
    best_h = h.copy()

    lr = lr_init
    stagnation = 0
    for it in range(max_iter):

        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)

            lr *= 0.985

def coordinate_descent(h_init, target_sum, topk, max_sweeps=20, time_limit=None):

    h = h_init.copy()
    best_h = h.copy()

    t0 = time.time()
    for sweep in range(max_sweeps):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        improved = False
        for j in range(n):
            corr = overlap_corrs(h)
            n_corr = len(corr)
            if topk < n_corr:
                top_idx = np.argpartition(corr, -topk)[-topk:]
            else:
                top_idx = np.arange(n_corr)
            c_vals = corr[top_idx]
            a_coeffs = np.zeros(len(top_idx))
            b_coeffs = np.zeros(len(top_idx))
            for i, idx in enumerate(top_idx):
                g = compute_grad_single_lag(h, idx)
                a_coeffs[i] = g[j]

            kink_points = set()
            kink_points.add(0.0)
            kink_points.add(1.0)
            for i in range(len(a_coeffs)):
                for j2 in range(i + 1, len(a_coeffs)):
                    da = a_coeffs[i] - a_coeffs[j2]

                            kink_points.add(xk)

            best_x = h[j]

                    best_x = xk

                h = project_feasible(h, target_sum)
                improved = True
        score = compute_c5(h)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()

    return best_h, best_score

                                         [(1000, 30), (5000, 15), (10000, 8)], 2000, 1e-16,
                                         time_limit=_remaining() - 15)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    n_use = len(best_h)
    target_use = n_use / 2.0

    if _remaining() > 30:
        h_try = best_h + rng.randn(n_use) * 0.002
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient(h_try, target_use,
                                        max_iter=2000, lr_init=0.015, topk=5,
                                        alpha=200, noise_prob=0.0, noise_scale=0.0,
                                        patience=250, momentum=0.93, seed=seed + 9000)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 20:
        h_new, score = lp_refinement_bisection(best_h, target_use,
                                                topk=20, max_iterations=150,
                                                time_limit=_remaining() - 10)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 12:
        h_ref, sc = slsqp_optimize(best_h, target_use,
                                     [(20000, 5), (50000, 3)], 1000, 1e-16,
                                     time_limit=_remaining() - 5)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
