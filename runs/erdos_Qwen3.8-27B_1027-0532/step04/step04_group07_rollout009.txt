_CONS_TARGET = 0.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = np.sum(free)
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = np.sum(free)
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):

def smooth_max_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)

def slsqp_optimize(h_start, target_sum, alpha, topk, maxiter):

    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h

        if abs(diff) < 1e-12:
            break

                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = np.sum(free)
            if nfree == 0:

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]

    return grad

    n_corr = len(corr)

    m = len(top_idx)

    b = np.zeros(m)

        g = compute_grad_single_lag(h, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h - corr[idx]
    c = np.zeros(n + 1)

def smooth_max_obj(x, alpha, topk):
    corr = overlap_corrs(x)

                patience=cfg[6], momentum=cfg[7], seed=seed + 1000 + i * 200,
                time_limit=max(20, _remaining() - 50)
            )
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 4: More LP with different topk schedules
    if _remaining() > 120:
        topk_schedules = [
            [5, 10, 15, 8, 12],
            [30, 20, 10, 5, 15],
            [10, 5, 20, 8, 12, 6],
        ]
        for sched in topk_schedules:
            if _remaining() < 50:
                break
            h_new, score = lp_refinement_alt(best_h, target_init, sched, 35, time_limit=max(15, _remaining() - 40))
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 5: Binary snap + LP
    if _remaining() > 100:
        for threshold in [0.12, 0.20, 0.30]:
            if _remaining() < 50:
                break
            h_snapped = binary_snap(best_h, target_init, threshold)
            h_new, score = lp_refinement_alt(h_snapped, target_init, [20, 10, 15], 40, time_limit=max(15, _remaining() - 40))
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 6: Coarse-to-fine
    if _remaining() > 250:
        for n_coarse in [80, 120]:
            if _remaining() < 120:
                break
            target_coarse = n_coarse / 2.0
            h_coarse = interpolate_h(best_h, n_coarse, target_coarse)
            h_coarse, sc = slsqp_optimize(h_coarse, target_coarse, alpha=20, topk=8, maxiter=800)
            if _remaining() < 60:
                break
            h_fine = interpolate_h(h_coarse, n_init, target_init)
            h_fine, sc2 = lp_refinement_alt(h_fine, target_init, [20, 10, 15, 5], 30, time_limit=max(20, _remaining() - 40))
            if sc2 < best_score:
                best_score = sc2
                best_h = h_fine

    # Phase 7: SLSQP polish with various smoothness
    if _remaining() > 80:
        for alpha, topk, maxit in [(40, 15, 800), (200, 8, 600), (500, 5, 400), (1000, 3, 300)]:
            if _remaining() < 40:
                break
            h_new, score = slsqp_optimize(best_h, target_init, alpha=alpha, topk=topk, maxiter=maxit)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 8: Final LP polish - very small topk for precision
    if _remaining() > 60:
        h_new, score = lp_refinement_alt(best_h, target_init, [3, 5, 2, 4, 3], 50, time_limit=max(10, _remaining() - 20))
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: Try higher resolution
    if _remaining() > 200:
        for n_try in [500, 640]:
            if _remaining() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_h(best_h, n_try, target_try)
            h_new, score = lp_refinement_alt(h_interp, target_try, [25, 15, 10, 20], 30, time_limit=max(30, _remaining() - 80))
            if score < best_score:
                best_score = score
                best_h = h_new
            if _remaining() > 60:
                h_new, score = slsqp_optimize(best_h, len(best_h) / 2.0, alpha=80, topk=12, maxiter=800)
                if score < best_score:
                    best_score = score
                    best_h = h_new
                if _remaining() > 40:
                    h_new, score = lp_refinement_alt(best_h, len(best_h) / 2.0, [10, 5, 8], 30, time_limit=max(10, _remaining() - 25))
                    if score < best_score:
                        best_score = score
                        best_h = h_new

    # Phase 10: Final micro-refinement
    if _remaining() > 30:
        h_new, score = lp_refinement_alt(best_h, len(best_h) / 2.0, [2, 3, 5], 40, time_limit=max(5, _remaining() - 10))
        if score < best_score:
            best_score = score
            best_h = h_new

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
