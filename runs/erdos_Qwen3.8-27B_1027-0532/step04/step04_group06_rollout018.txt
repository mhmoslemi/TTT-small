Let me think about the subgradient method more carefully:

The objective is F(h) = max_k (corr_k * dx) where corr_k = sum_i h[i] * (1-h[i+k]).

The subdifferential of F at h is:
∂F(h) = {sum_{k ∈ K(h)} λ_k * ∂(corr_k * dx)/∂h : λ_k ≥ 0, sum λ_k = 1}
where K(h) = {k : corr_k * dx = F(h)} (the set of active lags).

A natural choice is λ_k = 1/|K(h)| for all active k (the average subgradient).

But we can do better: we can weight the lags by how "close" they are to the maximum. For example:

As β → ∞, this approaches the uniform weight on active lags. For finite β, it gives a smooth approximation.

The gradient of (corr_k * dx) w.r.t. h[i] is:

```python
def subgradient_step(h, target, step, beta, n_lags=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx  # This is the actual C5 values for each lag
    
    if n_lags is None:
        n_lags = n  # use all lags
    center = n - 1
    # Focus on lags around center (exclude very small and very large lags)
    # Actually, all lags from 0 to 2n-2 correspond to shifts from -(n-1) to (n-1)
    # But the meaningful ones are those where the overlap is significant
    
    # Compute softmax weights
    c_shifted = c - c.max()
    weights = np.exp(beta * c_shifted)
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(-n+1, n):
        if weights[center + k] < 1e-10:
            continue
        # grad_k[i] = (1-h[i+k]) - h[i-k]
        # Need to handle boundaries
        i_range = np.arange(n)
        i_plus = i_range + k
        i_minus = i_range - k
        mask_plus = (i_plus >= 0) & (i_plus < n)
        mask_minus = (i_minus >= 0) & (i_minus < n)
        g = np.zeros(n)
        g[mask_plus] += (1.0 - h[i_plus[mask_plus]])
        g[mask_minus] -= h[i_minus[mask_minus]]
        grad += weights[center + k] * g
    
    h_new = h - step * grad * dx
    h_new = project_feasible(h_new, target)
    return h_new, F(h), grad
```

    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for lag_idx in range(n * 2 - 1):
        w = weights[lag_idx]
        if w < 1e-12:
            continue
        k = lag_idx - center
        # grad contribution at position i: (1-h[i+k]) - h[i-k]
        # For the first term: i+k in [0,n) → i in [-k, n-k)

        i_first = np.arange(max(0, -k), min(n, n - k))

        grad[i_first] += w * (1.0 - h[i_first + k])
        grad[i_second] -= w * h[i_second - k]
    return grad

def softmax_subgradient(h, target, n, beta, step, momentum_dict):
    """One step of momentum subgradient descent."""
    dx = 2.0 / n
    corr = get_corr(h)
    c = corr * dx
    
    # Softmax weights
    c_shifted = c - c.max()
    weights = np.exp(beta * c_shifted)
    total_w = weights.sum()
    if total_w < 1e-30:
        weights = np.ones(len(c)) / len(c)
    else:
        weights /= total_w
    
    # Compute weighted gradient
    # grad[i] = sum_k weights[k] * [(1-h[i+k]) - h[i-k]]
    # where k ranges over all valid lags
    grad = np.zeros(n)
    center = n - 1
    for lag_idx in range(2 * n - 1):
        w = weights[lag_idx]
        if w < 1e-8:
            continue
        k = lag_idx - center
        # First term: (1-h[i+k]) for valid i
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end > i_start:
            grad[i_start:i_end] += w * (1.0 - h[i_start + k:i_end + k])
        # Second term: -h[i-k] for valid i
        i_start2 = max(0, k)
        i_end2 = min(n, n + k)
        if i_end2 > i_start2:
            grad[i_start2:i_end2] -= w * h[i_start2 - k:i_end2 - k]
    
    # Nesterov momentum
    if 'v' not in momentum_dict:
        momentum_dict['v'] = np.zeros(n)
    v = momentum_dict['v']
    
    # Update
    h_new = h - step * grad * dx
    v_new = 0.9 * v - step * grad * dx
    h_new = h + v_new  # This is the Nesterov update
    
    # Wait, Nesterov is:

    # h_new = Proj(y - step * grad(y))
    # v_new = h_new - h
    # Let me use a simpler momentum:
    # v = beta_m * v - step * grad
    # h_new = h + v
    
    v_new = 0.85 * v - step * grad * dx
    h_new = h + v_new
    h_new = project_feasible(h_new, target)
    momentum_dict['v'] = v_new
    
    return h_new, float(np.max(corr)) * dx

            bk = zeros + k_star
            bkv = (bk >= 0) & (bk < n)
            bkm_arr = zeros - k_star
            bkmv = (bkm_arr >= 0) & (bkm_arr < n)
            valid_b = bkv & bkmv
            if np.sum(valid_b) > 0:
                b_cands = zeros[valid_b]
                b_scores = -(h[np.clip(bk[valid_b], 0, n - 1)] + h[np.clip(bkm_arr[valid_b], 0, n - 1)])
                b_idx = b_cands[np.argmin(b_scores)]
            else:
                b_idx = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            a_idx = ones[rng.randint(len(ones))]
            b_idx = zeros[rng.randint(len(zeros))]
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                ia = rng.choice(len(ones), 2, replace=False)
                ib = rng.choice(len(zeros), 2, replace=False)
                a1, a2 = int(ones[ia[0]]), int(ones[ia[1]])
                b1, b2 = int(zeros[ib[0]]), int(zeros[ib[1]])
                d1 = fast_delta_corr(h, a1, b1)
                h[a1] = 0.0
                h[b1] = 1.0
                d2 = fast_delta_corr(h, a2, b2)
                new_corr = corr + d1 + d2
                new_c5 = float(np.max(new_corr)) * dx
                delta_c5 = new_c5 - current_c5
                if delta_c5 < 0 or (t > 1e-12 and rng.random() < np.exp(-delta_c5 / t)):
                    h[a2] = 0.0
                    h[b2] = 1.0
                    corr = new_corr
                    current_c5 = new_c5
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if current_c5 < best_c5 - 1e-12:
                        best_c5 = current_c5
                        best_h = h.copy()
                    no_improve = 0 if current_c5 < best_c5 else no_improve + 1
                else:
                    h[a1] = 1.0
                    h[b1] = 0.0
                    no_improve += 1
                continue
            else:
                a_idx = ones[rng.randint(len(ones))]
                b_idx = zeros[rng.randint(len(zeros))]

        a_idx = int(a_idx)
        b_idx = int(b_idx)
        delta = fast_delta_corr(h, a_idx, b_idx)
        new_corr = corr + delta
        new_c5 = float(np.max(new_corr)) * dx
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0 or (t > 1e-12 and rng.random() < np.exp(-delta_c5 / t)):
            h[a_idx] = 0.0
            h[b_idx] = 1.0
            corr = new_corr
            current_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if current_c5 < best_c5 - 1e-12:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def softmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    c = corr * dx
    c_shifted = c - c.max()
    weights = np.exp(beta * c_shifted)
    total_w = weights.sum()
    if total_w < 1e-30:
        weights = np.ones(len(c)) / len(c)
    else:
        weights /= total_w
    grad = np.zeros(n)
    center = n - 1
    for lag_idx in range(2 * n - 1):
        w = weights[lag_idx]
        if w < 1e-10:
            continue
        k = lag_idx - center
        i_s = max(0, -k)
        i_e = min(n, n - k)
        if i_e > i_s:
            grad[i_s:i_e] += w * (1.0 - h[i_s + k:i_e + k])
        i_s2 = max(0, k)
        i_e2 = min(n, n + k)
        if i_e2 > i_s2:
            grad[i_s2:i_e2] -= w * h[i_s2 - k:i_e2 - k]
    return grad * dx, float(np.max(c)) 

def subgradient_momentum(h_start, target, max_iters, beta_start, beta_end, step_start, step_end, time_limit, start_time):
    n = len(h_start)
    h_curr = project_feasible(h_start.copy(), target)
    best_h = h_curr.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    mu = 0.85

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = i / max(max_iters - 1, 1)
        step = step_start * (step_end / step_start) ** ratio
        beta = beta_start + (beta_end - beta_start) * ratio

        grad, _ = softmax_grad(h_curr, beta)
        v = mu * v - step * grad
        h_new = project_feasible(h_curr + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - target

    def cons_j(h_in):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_local_search(h, target, n_blocks, rng, time_limit, start_time):
    n = len(h)
    h = project_binary(h, target).astype(float)
    corr = get_corr(h).copy()
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    current_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    block_size = n // n_blocks
    improved = True
    while improved and time.time() - start_time < time_limit:
        improved = False
        for bi in range(n_blocks):
            start = bi * block_size
            end = start + block_size if bi < n_blocks - 1 else n
            block_ones = np.where((ones >= start) & (ones < end))[0]
            block_zeros = np.where((zeros >= start) & (zeros < end))[0]
            if len(block_ones) == 0 or len(block_zeros) == 0:
                continue
            for _ in range(200):
                a = ones[block_ones[0] + rng.randint(len(block_ones))]
                b = zeros[block_zeros[0] + rng.randint(len(block_zeros))]
                delta = fast_delta_corr(h, a, b)
                new_corr = corr + delta
                new_c5 = float(np.max(new_corr)) * dx
                if new_c5 < current_c5:
                    h[a] = 0.0
                    h[b] = 1.0
                    corr = new_corr
                    current_c5 = new_c5
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    block_ones = np.where((ones >= start) & (ones < end))[0]
                    block_zeros = np.where((zeros >= start) & (zeros < end))[0]
                    if len(block_ones) == 0 or len(block_zeros) == 0:
                        break
                    improved = True
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)

    sa_configs = [
        (300000, 0.015, 0.00005, 300),
        (400000, 0.008, 0.00002, 300),
        (500000, 0.004, 0.00001, 200),
    ]

    h_sa_curr = h_bin_start.copy()
    phase_time = tleft() * 0.35
    per_sa = phase_time / len(sa_configs)
    for ci, (max_it, t_i, t_f, _) in enumerate(sa_configs):
        if tleft() < 60:
            break
        allowed = min(per_sa, tleft() * 0.25)
        h_sa, c5_sa = targeted_sa_fast(h_sa_curr, target_init, max_it, t_i, t_f,
                                       rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        h_sa_curr = h_sa

    if tleft() < 30:
        h_final = project_feasible(best_h, len(best_h) / 2.0)
        return h_final, compute_c5(h_final), len(h_final)

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        allowed = min(tleft() * 0.25, 200.0)
        beta_sched = [5, 20, 80, 300, 1000]
        h_sg, c5_sg = subgradient_momentum(best_h, target_use, 50000,
                                           5.0, 2000.0, 0.3, 0.005,
                                           time_limit=allowed, start_time=start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    n_candidates = [400, 500, 700, 800, 900, 1000, 1200]
    n_candidates = [x for x in n_candidates if abs(x - n_init) > 30]
    rng.shuffle(n_candidates)
    n_try_max = 5

    for ni, n_try in enumerate(n_candidates[:n_try_max]):
        if tleft() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed_bin = min(60.0, tleft() * 0.2)
        h_sa, c5_sa = targeted_sa_fast(h_bin_try, target_try, 200000, 0.01, 0.00003,
                                       rng, time_limit=allowed_bin, start_time=start_time)
        if tleft() > 30:
            allowed_sg = min(40.0, tleft() * 0.15)
            h_sg, c5_sg = subgradient_momentum(h_sa, target_try, 20000,
                                               10.0, 1000.0, 0.2, 0.003,
                                               time_limit=allowed_sg, start_time=start_time)
            if c5_sg < c5_sa:
                h_sa, c5_sa = h_sg, c5_sg
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 50:
        alphas = [50, 150, 400, 1000, 3000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 300, 1e-14,
                                 time_limit=min(60.0, tleft() - 20), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 30:
        for fa in range(3):
            if tleft() < 12:
                break
            noise = 0.01
            h_pert = best_h + rng.randn(len(best_h)) * noise
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_ref, sc = slsqp_refine(h_pert, len(h_pert), len(h_pert) / 2.0,
                                     [500, 2000, 5000], 200, 1e-15,
                                     time_limit=min(15.0, tleft() - 8), start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(20000, tleft() * 2000))
        h_sa_f, c5_f = targeted_sa_fast(h_bin_final, target_use, max_it, 0.003, 0.00001,
                                        rng, time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 10:
            h_ref, sc = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                     [2000, 5000, 10000], 150, 1e-15,
                                     time_limit=max(3.0, tleft() - 5), start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = min(final_score, best_score)

    return h_final, final_score, n_final
