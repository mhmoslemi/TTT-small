                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    t0 = time.time()
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = two_interval_h(400, 100)

    n = len(h_init)
    target = n // 2
    _g_n = n
    _g_target = target

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: FISTA top-K at original resolution
    deadline1 = t0 + budget_s * 0.30
    for K in [10, 25, 50]:
        if time.time() > deadline1:
            break
        for step in [0.3, 0.4, 0.5]:
            if time.time() > deadline1:
                break
            h_new, sc = fista_topk(best_h, n, target, K, 200000, step, deadline1)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP polish
    deadline2 = t0 + budget_s * 0.45
    for alpha in [200, 1000]:
        if time.time() > deadline2:
            break
        h_pol = slsqp_polish(best_h, n, target, alpha, 3000)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: More FISTA with finer K
    deadline3 = t0 + budget_s * 0.55
    for K in [5, 10, 20]:
        if time.time() > deadline3:
            break
        for step in [0.3, 0.4]:
            if time.time() > deadline3:
                break
            h_new, sc = fista_topk(best_h, n, target, K, 100000, step, deadline3)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Block SLSQP for local refinement
    deadline4 = t0 + budget_s * 0.65
    h_bd, sc_bd = block_slsqp(best_h, n, target, 5, 10, 500.0, deadline4)
    if sc_bd < best_score:
        best_score = sc_bd
        best_h = h_bd

    # Phase 5: Upsample to 800 points
    n2 = 800
    target2 = n2 // 2
    if time.time() < t0 + budget_s * 0.90:
        h2 = upsample(best_h, n2)
        score2 = compute_c5(h2)

        deadline5 = t0 + budget_s * 0.85
        for K in [10, 25, 50]:
            if time.time() > deadline5:
                break
            for step in [0.3, 0.4, 0.5]:
                if time.time() > deadline5:
                    break
                h_new2, sc2 = fista_topk(h2, n2, target2, K, 100000, step, deadline5)
                if sc2 < score2:
                    score2 = sc2
                    h2 = h_new2

        if score2 < best_score:
            best_score = score2
            best_h = h2
            n = n2
            target = target2

    # Phase 6: Final SLSQP polish at best resolution
    if time.time() < t0 + budget_s - 30:
        for alpha in [500, 2000]:
            if time.time() > t0 + budget_s - 30:
                break
            h_pol = slsqp_polish(best_h, n, target, alpha, 2000)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 7: Final FISTA polish
    if time.time() < t0 + budget_s - 20:
        deadline7 = t0 + budget_s - 10
        for K in [10, 20]:
            if time.time() > deadline7:
                break
            h_new, sc = fista_topk(best_h, n, target, K, 50000, 0.3, deadline7)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    best_h = project_feasible(best_h, len(best_h) // 2)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score
    return best_h, best_score, len(best_h)
