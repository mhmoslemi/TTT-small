    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]

        return h

Let me also think about the bisection projection:

To project h onto {x: sum(x) = target, 0 ≤ x ≤ 1}:
- Find τ such that sum(clamp(h - τ, 0, 1)) = target
- clamp(v, 0, 1) = min(max(v, 0), 1)
- f(τ) = sum(clamp(h - τ, 0, 1)) is monotonically decreasing in τ
- f(0) = sum(min(h, 1)) ≥ sum(h) ≥ target (assuming sum(h) ≥ target)
- f(max(h)) = sum(max(h - max(h), 0)) = 0 ≤ target

So we bisection on τ ∈ [min(h) - 1, max(h) + 1] or something like that.

Actually, more precisely:
- When τ → -∞: clamp(h - τ, 0, 1) → 1 for all i, so sum → n
- When τ → +∞: clamp(h - τ, 0, 1) → 0 for all i, so sum → 0

So we need τ such that sum(clamp(h - τ, 0, 1)) = target.
Bisection on τ ∈ [min(h) - 1, max(h) + 1] should work.

def project_simplex_box(h, target_sum):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_simplex_box(h, target_sum):
    h = h.astype(float)
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(64):

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:

        else:
            hi = mid

    n = len(h_p)
    corr = get_corr(h_p)
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):

        if w < 1e-13:

    best_score = compute_c5(best_h)

    # Phase 1: FISTA with increasing alpha from continuous start
    alpha_schedule = [30, 60, 100, 200, 400, 800, 1500, 3000]
    for alpha in alpha_schedule:
        if tleft() < 80:
            break
        n_iter = min(8000, int(tleft() * 1.5))
        lr_val = 1.0 / (2.0 + 10.0 * n / max(alpha, 10))
        h_f, c5_f = fista_smooth(best_h, alpha, target, n_iter,
                                   time_limit=tleft() * 0.8, start_time=start_time, lr=lr_val)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()

    # Phase 2: Subgradient refinement
    if tleft() > 80:
        h_sg, c5_sg = subgradient_refine(best_h, target,
                                          time_limit=min(120, tleft() * 0.3), start_time=start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Binary SA
    if tleft() > 60:
        h_bin = project_binary(best_h, target)
        h_sa, c5_sa = sa_optimize(h_bin, target, 80000, 0.02, 0.0005,
                                   rng, time_limit=min(120, tleft() * 0.3), start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        h_hc, c5_hc = hill_climb(best_h, target, 5000, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 4: High-alpha FISTA from best binary
    for alpha in [5000, 10000, 20000, 40000]:
        if tleft() < 60:
            break
        n_iter = min(6000, int(tleft() * 2))
        lr_val = 1.0 / (2.0 + 10.0 * n / max(alpha, 10))
        h_f2, c5_f2 = fista_smooth(best_h, alpha, target, n_iter,
                                     time_limit=tleft() * 0.7, start_time=start_time, lr=lr_val)
        if c5_f2 < best_score:
            best_score = c5_f2
            best_h = h_f2.copy()

    # Phase 5: Alternating continuous-binary
    for rnd in range(6):
        if tleft() < 50:
            break
        # Continuous
        alpha = 5000 + 3000 * rnd
        n_iter = min(4000, int(tleft() * 1.5))
        lr_val = 1.0 / (2.0 + 10.0 * n / max(alpha, 10))
        h_f3, c5_f3 = fista_smooth(best_h, alpha, target, n_iter,
                                     time_limit=min(tleft() * 0.4, 80), start_time=start_time, lr=lr_val)
        if c5_f3 < best_score:
            best_score = c5_f3
            best_h = h_f3.copy()
        # Binary
        if tleft() > 40:
            h_b2 = project_binary(best_h, target)
            h_s2, c5_s2 = sa_optimize(h_b2, target, 30000, 0.01, 0.0003,
                                       rng, time_limit=min(40, tleft() * 0.3), start_time=start_time)
            if c5_s2 < best_score:
                best_score = c5_s2
                best_h = h_s2.copy()

    # Phase 6: Multi-n exploration
    for n_try in [n - 4, n - 2, n + 2, n + 4]:
        if tleft() < 60:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        h_sa_r, c5_sa_r = sa_optimize(h_bin_try, target_try, 20000, 0.015, 0.0005,
                                       rng, time_limit=min(30, tleft() * 0.25), start_time=start_time)
        if tleft() > 30:
            alpha_r = 10000
            lr_r = 1.0 / (2.0 + 10.0 * n_try / alpha_r)
            h_f_r, c5_f_r = fista_smooth(h_sa_r, alpha_r, target_try, 3000,
                                           time_limit=min(30, tleft() * 0.3), start_time=start_time, lr=lr_r)
            if c5_f_r < c5_sa_r:
                h_sa_r = h_f_r
                c5_sa_r = c5_f_r
        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 7: Final intensive FISTA
    if tleft() > 30:
        for alpha in [30000, 60000, 100000]:
            if tleft() < 20:
                break
            n_use = len(best_h)
            target_use = n_use / 2.0
            lr_val = 1.0 / (2.0 + 10.0 * n_use / alpha)
            h_fin, c5_fin = fista_smooth(best_h, alpha, target_use, 3000,
                                          time_limit=min(30, tleft() * 0.4), start_time=start_time, lr=lr_val)
            if c5_fin < best_score:
                best_score = c5_fin
                best_h = h_fin.copy()

    # Phase 8: Final local search
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h = best_h.copy().astype(float)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for _ in range(min(300, int(tleft() * 5))):
            if time.time() - start_time > budget_s - 10:
                break
            if len(ones) == 0:
                break
            i = ones[rng.randint(len(ones))]
            old_v = h[i]
            h[i] = 0.0
            h_p = project_simplex_box(h, target_use)
            c5_try = compute_c5(h_p)
            if c5_try < best_score - 1e-12:
                best_score = c5_try
                best_h = h_p.copy()
            else:
                h[i] = old_v

    # Final projection
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_simplex_box(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score
    else:
        h_final2 = project_feasible(best_h, target_final)
        final_score2 = compute_c5(h_final2)
        if final_score2 < best_score:
            best_h = h_final2
            best_score = final_score2

    return best_h, best_score, n_final
