from scipy.optimize import minimize

    dx = 2.0 / len(h)
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0
        else:

        if cnt == 0:
            break

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = alpha * exp_v[idx] / sum_exp
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

    return lse_obj_and_grad(x, _g_alpha)[0]

def _jac(x):
    return lse_obj_and_grad(x, _g_alpha)[1]

def _cfun(x):
    return np.sum(x) - _g_target

def _cjac(x):
    return np.ones(_g_n)

def slsqp_step(h, target, alpha, maxiter, ftol):
    global _g_alpha, _g_n, _g_target
    _g_alpha = alpha
    _g_n = len(h)
    _g_target = target
    bounds = [(0.0, 1.0)] * _g_n
    try:
        res = minimize(_fun, h, jac=_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_new = slsqp_step(h_cur, target, alpha, maxiter, ftol)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
    return h_cur, cur_score

def subgrad_polish(h, target, n_steps, step_size, active_frac, deadline, momentum=0.0):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    vel = np.zeros(n)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        if c_max - c_min < 1e-12:
            break
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= max(1, len(active))
        step = step_size / (1.0 + 0.03 * it)
        vel = momentum * vel + (1.0 - momentum) * grad
        h_new = h_cur - step * vel
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score

    x = np.linspace(0, 2, n)

    return project_feasible(np.clip(h, 0, 1), n / 2.0)

    h_out = np.interp(np.linspace(0, 1, n_dst), np.linspace(0, 1, n_src), h_src)

def perturb(h, target, noise_level, rng):

So h[i] appears in this sum:
- As h[j] when j = i, i.e., i ranges from 0 to n-1-k: contribution is h[i] * (1-h[i+k])
- As h[j+k] when j+k = i, i.e., j = i-k, so i ranges from k to n-1: contribution is h[i-k] * (1-h[i])

So the total dependency of the correlation at lag k on h[i] is:
- (1-h[i+k]) * h[i] (if i ≤ n-1-k) [linear in h[i]]
- (1-h[i]) * h[i-k] (if i ≥ k) [linear in h[i]]

Plus the correlation at lag -k:
sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])

h[i] appears:
- As h[j+k] when j+k = i, j = i-k, i ranges from k to n-1: contribution h[i] * (1-h[i-k])
- As h[j] when j = i, i ranges from 0 to n-1-k: contribution (1-h[i]) * h[i+k]

So the total for h[i] across all lags is a sum of terms, each linear in h[i]. Therefore, the full objective (max over all lags) is the max of a set of piecewise-linear functions of h[i] (when all other variables are fixed).

This means coordinate descent with exact 1D optimization (bisection or ternary search on the max of linear pieces) is well-defined and exact!

Let me implement this:

For coordinate i, the objective as a function of h[i] = t is:
max_k (a_k * t + b_k)

where a_k and b_k are constants (depending on all other h[j], j ≠ i).

This is a convex piecewise-linear function of t, and its minimum on [0,1] occurs at a "kink" point (where two pieces cross) or at a boundary.

To find the optimal t:
1. Compute all the a_k and b_k for each lag k
2. Find the minimum of max_k (a_k * t + b_k) over t ∈ [0,1]
3. This is a 1D convex minimization - can be done with bisection on the subgradient

Actually, for a max of linear functions, the minimum is where the subgradient contains 0, i.e., where the set of active functions (those achieving the max) has both positive and negative slopes (or one function has slope 0).

This is actually quite efficient to compute! For n=400, each coordinate update requires computing O(n) values (one per lag), and finding the optimal t is O(n log n) or O(n) with a simple scan.

Total cost of one full coordinate descent sweep: O(n²) (n coordinates × O(n) work per coordinate).
Same as one SLSQP function evaluation!

But coordinate descent can make consistent progress without the overhead of SLSQP's quadratic step computation.

def coord_descent_step(h, target, n_sweeps=10, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            # Compute the optimal value for h[i]
            t_opt = _optimal_coord(h_cur, i, target)
            if abs(t_opt - h_cur[i]) > 1e-10:
                # Check if moving h[i] to t_opt keeps us feasible
                # We need sum(h) = target
                # If we change h[i] by delta, we need to adjust another coordinate
                # For simplicity, just project after each update
                h_cur[i] = t_opt
                # Quick projection
                diff = target - np.sum(h_cur)
                if abs(diff) > 1e-10:
                    # Adjust one other coordinate
                    j = (i + 1) % n
                    if diff > 0 and h_cur[j] < 1.0:
                        adj = min(diff, 1.0 - h_cur[j])
                        h_cur[j] += adj
                        diff -= adj
                    elif diff < 0 and h_cur[j] > 0.0:
                        adj = max(diff, -h_cur[j])
                        h_cur[j] += adj
                        diff -= adj
                    if abs(diff) > 1e-10:
                        # Full projection
                        h_cur = project_feasible(h_cur, target)
                improved = True
        sc = compute_c5(h_cur)
        if sc < cur_score:
            cur_score = sc
        elif not improved:
            break
    
    return h_cur, cur_score

def _optimal_coord(h, i, target):
    n = len(h)
    # For each lag k, compute a_k and b_k such that corr[k] = a_k * h[i] + b_k
    # corr at lag k (k >= 0): sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
    # corr at lag k (k < 0, i.e., lag = -|k|): sum_{j=0}^{n-1-|k|} h[j+|k|]*(1-h[j])
    
    # For efficiency, compute for each lag the coefficient of h[i] and the constant term
    # We want to minimize max_k (a_k * t + b_k) over t in [0,1]
    
    a_vals = []
    b_vals = []
    
    for k in range(-(n-1), n):
        abs_k = abs(k)
        if abs_k >= n:
            continue
        # For this lag, h[i] contributes:
        # Case 1: i is the "first" index (h[j] where j=i)
        #   For lag k >= 0: h[i] appears as h[j] when j=i, valid if i <= n-1-k (k>=0)
        #       term: h[i] * (1-h[i+k]), so a = (1-h[i+k]), b = 0
        #   For lag k < 0: h[i] appears as h[j] when j=i, valid if i <= n-1-|k| (k<0, lag=-|k|)
        #       In corr at lag -|k|: sum h[j+|k|]*(1-h[j]), h[i] = h[j+|k|] when j=i-|k|
        #       term: h[i] * (1-h[i-|k|]), so a = (1-h[i-|k|]), b = 0
        # Case 2: i is the "second" index (h[j+k] where j+k=i, or 1-h[j] where j=i)
        #   For lag k >= 0: (1-h[i]) appears when i = j+k, j=i-k, valid if i >= k
        #       term: h[i-k] * (1-h[i]), so a = -h[i-k], b = h[i-k]
        #   For lag k < 0: (1-h[i]) appears when i = j, valid if i <= n-1-|k|
        #       In corr at lag -|k|: sum h[j+|k|]*(1-h[j]), (1-h[i]) when j=i
        #       term: h[i+|k|] * (1-h[i]), so a = -h[i+|k|], b = h[i+|k|]
        
        a = 0.0
        b = 0.0
        
        if k >= 0:
            # h[i] as first index: valid if i <= n-1-k
            if i <= n - 1 - k:
                a += (1.0 - h[i + k])
                b += 0.0
            # (1-h[i]) as second: valid if i >= k and i <= n-1
            if i >= k:
                a += -h[i - k]
                b += h[i - k]
        else:
            kk = -k  # |k|
            # corr at lag -kk: sum_{j=0}^{n-1-kk} h[j+kk]*(1-h[j])
            # h[i] as h[j+kk]: j = i-kk, valid if i-kk >= 0 and i-kk <= n-1-kk, i.e., kk <= i <= n-1
            if kk <= i:  # i >= kk
                a += (1.0 - h[i - kk])
                b += 0.0
            # (1-h[i]) as (1-h[j]): j = i, valid if i <= n-1-kk
            if i <= n - 1 - kk:
                a += -h[i + kk]
                b += h[i + kk]
        
        if a != 0.0 or b != 0.0:
            a_vals.append(a)
            b_vals.append(b)
    
    a_arr = np.array(a_vals)
    b_arr = np.array(b_vals)
    
    # Minimize max_k (a_k * t + b_k) over t in [0,1]
    # This is a 1D convex problem (max of linear functions)
    # Use bisection on the subgradient
    
    def f_of_t(t):
        return float(np.max(a_arr * t + b_arr))
    
    # Subgradient at t: the set {a_k : k in argmax_k (a_k*t + b_k)}
    # Minimum is where 0 is in the subgradient, i.e., min active a <= 0 <= max active a
    
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = a_arr * mid + b_arr
        c_max = np.max(vals)
        active = np.where(vals >= c_max - 1e-14)[0]
        if len(active) == 0:
            break
        min_a = np.min(a_arr[active])
        max_a = np.max(a_arr[active])
        if min_a <= 0.0 and max_a >= 0.0:
            # 0 is in subgradient, mid is optimal
            break
        elif min_a > 0.0:
            # All active slopes positive, need to decrease t
            hi = mid
        else:
            # All active slopes negative, need to increase t
            lo = mid
    
    t_opt = (lo + hi) / 2.0
    return t_opt

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 1.0
_g_n = 0
_g_target = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = alpha * exp_v[idx] / sum_exp
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def _fun(x):
    return lse_obj_and_grad(x, _g_alpha)[0]


def _jac(x):
    return lse_obj_and_grad(x, _g_alpha)[1]


def _cfun(x):
    return np.sum(x) - _g_target


def _cjac(x):
    return np.ones(_g_n)


def slsqp_step(h, target, alpha, maxiter, ftol):
    global _g_alpha, _g_n, _g_target
    _g_alpha = alpha
    _g_n = len(h)
    _g_target = target
    bounds = [(0.0, 1.0)] * _g_n
    try:
        res = minimize(_fun, h, jac=_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_new = slsqp_step(h_cur, target, alpha, maxiter, ftol)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
    return h_cur, cur_score


def subgrad_polish(h, target, n_steps, step_size, active_frac, deadline, momentum=0.0):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    vel = np.zeros(n)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        if c_max - c_min < 1e-12:
            break
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= max(1, len(active))
        step = step_size / (1.0 + 0.03 * it)
        vel = momentum * vel + (1.0 - momentum) * grad
        h_new = h_cur - step * vel
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score

    x = np.linspace(0, 2, n)

        h = rng.random(n)

    elif mode == 2:

    return project_feasible(np.clip(h, 0, 1), n / 2.0)


def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    h_out = np.interp(np.linspace(0, 1, n_dst), np.linspace(0, 1, n_src), h_src)
    return project_feasible(h_out, n_dst / 2.0)


def perturb(h, target, noise_level, rng):
    h_new = h + noise_level * rng.randn(len(h))
