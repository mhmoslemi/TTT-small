```python
def project_feasible(h, target_sum):

def project_binary(h, target_sum):

def find_worst_lags(corr, n, top_k=10, threshold_frac=0.95):
    center = n - 1
    c = corr.copy()
    c[center] = 0.0
    max_val = c.max()
    if max_val <= 0:
        return np.array([1])
    threshold = max_val * threshold_frac
    candidates = np.where(c >= threshold)[0]
    order = np.argsort(c[candidates])[::-1]
    selected = candidates[order[:top_k]]
    lags = selected - center
    lags = lags[lags != 0]
    if len(lags) == 0:

    n = len(h)
    grad = np.zeros(n)
    if k > 0:

        if k < n:
            grad[:n-k] += (1.0 - h[k:n])  # (1 - h[j+k]) for j=0..n-k-1
        # Contribution from -h[j-k]*h[j] for j=k..n-1:
        grad[k:] -= h[:n-k]  # -h[j-k] for j=k..n-1
    elif k < 0:
        a = -k
        if a < n:
            grad[a:] += (1.0 - h[:n-a])
            grad[:n-a] -= h[a:]
    return grad

def compute_subgradient(h, top_k=10, threshold_frac=0.95):

    corr = get_corr(h)
    n = len(h)
    worst_lags = find_worst_lags(corr, n, top_k=top_k, threshold_frac=threshold_frac)
    grad = np.zeros(n)
    for k in worst_lags:
        grad += subgradient_for_lag(h, int(k))
    grad /= len(worst_lags)
    return grad, worst_lags

    best_h = h.copy()

        if time.time() - start_time > time_limit:

        grad, _ = compute_subgradient(y, top_k=15, threshold_frac=0.9)

        y = y - step * grad
        y = project_feasible(y, target_sum)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        h = y
        t_mom = t_new
        step *= 0.999  # slow decay
        
    return best_h, best_c5
```

```python
def compute_flip_scores(h, ones, zeros, worst_lags, n):
    """Compute scores for flipping each one to zero and each zero to one"""
    a_scores = np.zeros(len(ones))
    b_scores = np.zeros(len(zeros))
    
    for k in worst_lags:
        k = int(k)
        
        # For ones (flipping 1->0), we want positions where the "cost" is high
        # Cost of flipping h[a] from 1 to 0 at lag k:
        #   removes (1-h[a+k]) from f_k and adds h[a-k] to f_k
        #   Net change in f_k: -(1-h[a+k]) + h[a-k]
        #   We want this to be very negative (reduces f_k a lot)
        #   So score = -(1-h[a+k]) + h[a-k] (lower = better)
        
        a_k = ones + k
        a_km = ones - k
        valid_a_k = (a_k >= 0) & (a_k < n)
        valid_a_km = (a_km >= 0) & (a_km < n)
        
        contrib = np.zeros(len(ones))
        contrib[valid_a_k] -= (1.0 - h[np.clip(a_k, 0, n-1)[valid_a_k]])
        contrib[valid_a_km] += h[np.clip(a_km, 0, n-1)[valid_a_km]]
        a_scores += contrib
        
        # For zeros (flipping 0->1), we want positions where the "gain" is high
        # Gain of flipping h[b] from 0 to 1 at lag k:
        #   adds (1-h[b+k]) to f_k and removes h[b-k] from f_k
        #   Net change in f_k: +(1-h[b+k]) - h[b-k]
        #   We want this to be very negative (reduces f_k)
        #   So score = (1-h[b+k]) - h[b-k] (lower = better)
        
        b_k = zeros + k
        b_km = zeros - k
        valid_b_k = (b_k >= 0) & (b_k < n)
        valid_b_km = (b_km >= 0) & (b_km < n)
        
        contrib_b = np.zeros(len(zeros))
        contrib_b[valid_b_k] += (1.0 - h[np.clip(b_k, 0, n-1)[valid_b_k]])
        contrib_b[valid_b_km] -= h[np.clip(b_km, 0, n-1)[valid_b_km]]
        b_scores += contrib_b
    
    return a_scores, b_scores
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]

def theue_morse_n(n):
    seq = np.zeros(n, dtype=np.float64)
    for i in range(n):
        x = i
        c = 0
        while x > 0:
            c += x & 1
            x >>= 1
        if c % 2 == 0:
            seq[i] = 1.0
    return seq

def golden_ratio_h(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    i = np.arange(n, dtype=np.float64)
    frac = (i * phi) % 1.0
    h = (frac < 0.5).astype(np.float64)
    return h

def subgradient_continuous_fast(h_init, target_sum, max_iters, time_limit, start_time,
                                 step_init=0.02, top_k=15):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_h = h.copy()
    step = step_init

    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        ratio = i / max(max_iters - 1, 1)
        step = step_init * (0.001) ** ratio

        corr = get_corr(h)
        worst_lags = find_worst_lags(corr, n, top_k=top_k, threshold_frac=0.88)

        grad = np.zeros(n)
        for k in worst_lags:
            k = int(k)
            if k > 0:
                grad[:n - k] += (1.0 - h[k:])
                grad[k:] -= (1.0 - h[:n - k])
            else:
                a = -k
                grad[a:] += (1.0 - h[:n - a])
                grad[:n - a] -= (1.0 - h[a:])
        grad /= len(worst_lags)

        h_new = h - step * grad * dx
        h_new = project_feasible(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        s = h_new - prev_h
        y = grad
        ss = np.dot(s, s)
        sy = np.dot(s, y)
        if sy > 1e-12:
            step_bb = ss / sy
            step = 0.5 * (step + step_bb)
            step = min(step, step_init)

        prev_h = h
        h = h_new

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    def logsumexp_objective(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        if alpha < 1e-3:
            return float(c_max)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - target

    def cons_j(h_in):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(logsumexp_objective, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Fast SA from initial binary projection
    h_bin_start = project_binary(h_init, target_init)
    max_iters1 = int(min(500000, (tleft() * 0.25) * 8000))
    h_sa, c5_sa = fast_sa_binary(h_bin_start, target_init, max_iters1,
                                  time_limit=tleft() * 0.25, start_time=start_time,
                                  rng=rng, t_init=0.05, t_final=0.0005)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 2: Fast SA from structured starts
    structured_starts = []

    h_tm = theue_morse_n(n_init)
    h_tm_bin = project_binary(h_tm, target_init)
    structured_starts.append(h_tm_bin)

    h_gr = golden_ratio_h(n_init)
    h_gr_bin = project_binary(h_gr, target_init)
    structured_starts.append(h_gr_bin)

    # Random start
    h_rand = np.zeros(n_init)
    perm = rng.permutation(n_init)
    h_rand[perm[:n_init // 2]] = 1.0
    structured_starts.append(h_rand)

    for si, h_s in enumerate(structured_starts):
        if tleft() < 80:
            break
        allowed = min(200.0, tleft() * 0.2)
        max_it = int(min(400000, allowed * 8000))
        h_sa2, c5_sa2 = fast_sa_binary(h_s, target_init, max_it,
                                        time_limit=allowed, start_time=start_time,
                                        rng=rng, t_init=0.04, t_final=0.0004)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 3: Continuous subgradient on best so far
    if tleft() > 100:
        allowed = min(120.0, tleft() * 0.2)
        h_sg, c5_sg = subgradient_continuous_fast(best_h, target_init, 20000,
                                                   time_limit=allowed, start_time=start_time,
                                                   step_init=0.015, top_k=15)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()
        # Also try from the binary solution
        h_bin3 = project_binary(best_h, target_init)
        h_sg2, c5_sg2 = subgradient_continuous_fast(h_bin3, target_init, 20000,
                                                     time_limit=min(80.0, tleft() * 0.15),
                                                     start_time=start_time,
                                                     step_init=0.01, top_k=20)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2.copy()

    # Phase 4: Fast SA on the continuous solution's binary projection
    if tleft() > 80:
        h_bin4 = project_binary(best_h, target_init)
        allowed = min(180.0, tleft() * 0.2)
        max_it4 = int(min(400000, allowed * 8000))
        h_sa4, c5_sa4 = fast_sa_binary(h_bin4, target_init, max_it4,
                                        time_limit=allowed, start_time=start_time,
                                        rng=rng, t_init=0.02, t_final=0.0002)
        if c5_sa4 < best_score:
            best_score = c5_sa4
            best_h = h_sa4.copy()

    # Phase 5: Multi-resolution attempts
    n_candidates = [400, 800, 1000, 1200]
    n_candidates = [x for x in n_candidates if x != n_init and x >= 200]

    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(150.0, tleft() * 0.25)
        max_it = int(min(300000, allowed * 6000))
        h_sa_r, c5_sa_r = fast_sa_binary(h_bin_try, target_try, max_it,
                                          time_limit=allowed, start_time=start_time,
                                          rng=rng, t_init=0.03, t_final=0.0003)
        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 6: SLSQP refinement on best
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas = [50, 150, 400, 1000, 3000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 300, 1e-14,
                                  time_limit=min(100.0, tleft() - 30), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 7: Final SA polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it_f = int(min(200000, max(10000, (tleft() - 15) * 6000)))
        h_sa_f, c5_f = fast_sa_binary(h_bin_final, target_use, max_it_f,
                                       time_limit=max(10.0, tleft() - 15), start_time=start_time,
                                       rng=rng, t_init=0.01, t_final=0.0001)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 8: Very final SLSQP if time
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas_f = [500, 2000, 5000]
        h_rf, sc_f = slsqp_refine(best_h, n_use, target_use, alphas_f, 200, 1e-15,
                                   time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_rf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
```
