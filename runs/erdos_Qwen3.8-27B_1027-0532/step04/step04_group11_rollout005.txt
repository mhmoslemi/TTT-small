```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def block2_obj(x0, h, s, c):
    # Block at positions s, s+1 with h[s]+h[s+1] = c
    # x0 = h[s], h[s+1] = c - x0
    # Returns max correlation
    h[s] = x0
    h[s+1] = c - x0
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def block2_optimize(h, s, c, n_grid=200):
    # Optimize block at s, s+1 with sum = c
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if lo >= hi:
        return h, 0.0
    x0 = h[s]  # current value (should satisfy constraint)
    # Clip to feasible range
    x0 = np.clip(x0, lo, hi)
    
    # Grid search
    xs = np.linspace(lo, hi, n_grid)
    best_x0 = x0
    best_val = block2_obj(x0, h.copy(), s, c)
    for x in xs:
        val = block2_obj(x, h, s, c)
        if val < best_val:
            best_val = val
            best_x0 = x
    h[s] = best_x0
    h[s+1] = c - best_x0
    return h, best_val
```

```python
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, new_n):
    """Upscale h to new_n points using linear interpolation + projection."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    h_new = np.interp(new_x, old_x, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = new_n // 2
    h_new = project_feasible(h_new, float(target))
    return h_new

def block2_func(x0, h_base, s, c, n):
    """Compute max correlation for block at s, s+1 with h[s]=x0, h[s+1]=c-x0."""
    h = h_base.copy()
    h[s] = x0
    h[s+1] = c - x0
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def block2_optimize(h, s, c, n):
    """Optimize the block at positions s, s+1 with sum constraint c."""
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if lo >= hi - 1e-15:
        return h
    h_base = h.copy()
    h_base[s] = 0.0
    h_base[s+1] = 0.0
    # Use minimize_scalar on [lo, hi]
    def obj(x0):
        return block2_func(x0, h_base, s, c, n)
    res = minimize_scalar(obj, bounds=(lo, hi), method='bounded', options={'maxiter': 50})
    x0_opt = np.clip(res.x, lo, hi)
    h[s] = x0_opt
    h[s+1] = c - x0_opt
    return h

def block_descent(h, n, target, block_size, n_sweeps, deadline):
    """Block coordinate descent with blocks of given size."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size

                    a = np.clip(a, 0, 1)

                    h_test = h_cur.copy()
                    h_test[s] = a
                    h_test[s+1] = b
                    h_test[s+2] = cc
                    corr = np.correlate(h_test, 1.0 - h_test, mode='full')
                    return float(np.max(corr))
                
                from scipy.optimize import minimize as _min
                best_block = None
                best_val = 1e10
                # Try a few starting points
                starts = [(x0, x1), (c/3, c/3), (0, c), (c, 0), (0.5, c-0.5)]
                for st in starts:
                    st = [np.clip(st[0], 0, 1), np.clip(st[1], 0, 1)]
                    try:
                        res = _min(obj_2d, np.array(st), method='Nelder-Mead', 
                                  options={'maxiter': 100, 'xatol': 1e-6, 'fatol': 1e-10})
                        if res.fun < best_val:
                            best_val = res.fun
                            best_block = res.x
                    except:
                        pass
                if best_block is not None:
                    a = np.clip(best_block[0], 0, 1)
                    b = np.clip(best_block[1], 0, 1)
                    cc = np.clip(c - a - b, 0, 1)
                    # Project to sum = c
                    s2 = a + b + cc
                    if abs(s2 - c) > 1e-12:

                    h_cur[s] = a

                    h_cur[s+2] = cc
            
            # Check improvement

            if new_score < cur_score:
                cur_score = new_score
    
    # Final feasibility projection
    h_cur = project_feasible(h_cur, float(target))
```

```python
def project_feasible(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

def upscale_linear(h, new_n):

    if old_n == new_n:
        return h.copy()

    new_x = np.linspace(0, 2, new_n)

    h_new = np.clip(h_new, 0.0, 1.0)
    target = new_n // 2
    h_new = project_feasible(h_new, float(target))

def lse_objective(x, target, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def con_eq(x):
    return np.sum(x) - x.shape[0] // 2

def con_jac(x):
    return np.ones(x.shape[0])

def slsqp_optimize(h, n, target, alpha, maxiter=2000):

    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    try:
        res = minimize(lambda x: lse_objective(x, target, alpha, n), h,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, float(target))
        return h_new
    except:
        return h.copy()

def block2_sweep(h, n, target, deadline, block_size=2):
    """One sweep of block coordinate descent over all blocks."""
    h_cur = h.copy()
    n_blocks = n // block_size
    for bi in range(n_blocks):
        if time.time() > deadline:
            break
        s = bi * block_size
        c = float(np.sum(h_cur[s:s+block_size]))
        c = np.clip(c, 0.0, float(block_size))
        
        if block_size == 2:
            lo = max(0.0, c - 1.0)
            hi = min(1.0, c)
            if lo < hi - 1e-12:
                h_base = h_cur.copy()
                h_base[s] = 0.0
                h_base[s+1] = 0.0
                
                def obj(x0):
                    h_test = h_base.copy()
                    h_test[s] = x0
                    h_test[s+1] = c - x0
                    corr = np.correlate(h_test, 1.0 - h_test, mode='full')
                    return float(np.max(corr))
                
                try:
                    res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                        options={'maxiter': 30, 'xatol': 1e-8})
                    x0_opt = np.clip(res.x, lo, hi)
                    h_cur[s] = x0_opt
                    h_cur[s+1] = c - x0_opt
                except:
                    pass
    return h_cur
```
