from scipy.optimize import minimize
import time

_cons_target = 0.0

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def project_polytope(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    h = h.copy()
    n = len(h)
    # Binary search on lambda
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:
        # Adjust the element with the most room
        if diff > 0:
            room = 1.0 - h_proj
        else:
            room = h_proj
        order = np.argsort(-room)
        remaining = abs(diff)
        for i in order:
            if remaining <= 1e-14:
                break
            adj = min(room[i], remaining)
            h_proj[i] += np.sign(diff) * adj
            remaining -= adj
    return h_proj

def project_binary(h, target_sum):
    """Round h to binary with exactly target_sum ones."""
    n = len(h)
    h_r = np.round(h).astype(int)
    current = h_r.sum()
    if current == target_sum:
        return h_r.astype(float)
    if current > target_sum:
        need = current - target_sum
        ones = np.where(h_r == 1)[0]
        # Convert the ones with lowest h values to 0
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

    """Compute log-sum-exp of correlations (smooth max)."""
    corr = compute_corr(h)
    # Shift for numerical stability
    c = corr - np.max(corr)
    return np.max(corr) + np.log(np.sum(np.exp(alpha * c))) / alpha

def projected_gradient(h, target_sum, alpha, step_size, n_steps):
    """Projected gradient descent on log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        # Soft weights
        c_shifted = corr - np.max(corr)
        weights = np.exp(alpha * c_shifted)
        weights /= weights.sum()
        
        # Compute gradient of log-sum-exp = weighted average of gradients
        # grad[i] = sum_m weights[m] * dcorr_m/dh[i] * dx
        # For corr at lag k (index n-1+k in full output):
        # dC(k)/dh[i] = [i>=k](1-h[i-k]) - [i<=n-1-k]h[i+k]  for k>=1
        # dC(-k)/dh[i] = [i<=n-1-k](1-h[i+k]) - [i>=k]h[i-k]  for k>=1
        # dC(0)/dh[i] = 1 - 2h[i]
        
        # This is O(n * 2n) = O(n^2) which for n=600 is 720K ops
        # Let's vectorize it.
        
        grad = np.zeros(n)
        
        # k=0 term: weight is weights[n-1]
        w0 = weights[n-1]
        grad += w0 * (1.0 - 2.0 * h_cur) * dx
        
        # For k = 1 to n-1:
        # C(k) at index n-1+k: dC(k)/dh[i] = [i>=k](1-h[i-k]) - [i<=n-1-k]h[i+k]
        # C(-k) at index n-1-k: dC(-k)/dh[i] = [i<=n-1-k](1-h[i+k]) - [i>=k]h[i-k]
        
        for k in range(1, n):
            wk_pos = weights[n-1+k]  # weight for C(k)
            wk_neg = weights[n-1-k]  # weight for C(-k)
            if wk_pos < 1e-15 and wk_neg < 1e-15:
                continue
            # dC(k)/dh[i] for i=0..n-1:
            # [i>=k](1-h[i-k]) - [i<=n-1-k]h[i+k]
            dk = np.zeros(n)
            dk[k:] = 1.0 - h_cur[:n-k]   # [i>=k](1-h[i-k]) for i=k..n-1
            dk[:n-k] -= h_cur[k:]         # [i<=n-1-k]h[i+k] for i=0..n-1-k
            grad += wk_pos * dk * dx
            
            dm = np.zeros(n)
            dm[:n-k] = 1.0 - h_cur[k:]    # [i<=n-1-k](1-h[i+k]) for i=0..n-1-k
            dm[k:] -= h_cur[:n-k]          # [i>=k]h[i-k] for i=k..n-1
            grad += wk_neg * dm * dx
        
        # Gradient step
        h_cur = h_cur - step_size * grad
        h_cur = project_polytope(h_cur, target_sum)
        
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    
    return best_h, best_score

def project_polytope(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -3.0, 3.0

        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h_p.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h_p
        else:
            room = h_p
        order = np.argsort(-room)
        remaining = abs(diff)
        for idx in order:
            if remaining <= 1e-14:
                break
            adj = min(room[idx], remaining)
            h_p[idx] += np.sign(diff) * adj
            remaining -= adj
    return h_p

    n = len(h)
    dx = 2.0 / n
    h_cur = h.astype(float).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step_size = 0.05
    momentum = np.zeros(n)
    beta = 0.9

    for step in range(n_steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        corr = compute_corr(h_cur)
        mx = np.max(corr)
        c_shifted = corr - mx
        weights = np.exp(alpha * c_shifted)
        weights /= weights.sum()
        active = np.where(weights > 1e-6)[0]

        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            w = weights[m]
            if k == 0:
                grad += w * (1.0 - 2.0 * h_cur) * dx
            elif k > 0:
                dk = np.zeros(n)
                dk[k:] = 1.0 - h_cur[:n - k]
                dk[:n - k] -= h_cur[k:]
                grad += w * dk * dx
            else:

                dm = np.zeros(n)
                dm[:n - kk] = 1.0 - h_cur[kk:]
                dm[kk:] -= h_cur[:n - kk]
                grad += w * dm * dx

        momentum = beta * momentum + (1 - beta) * grad
        h_trial = h_cur - step_size * momentum
        h_trial = project_polytope(h_trial, target_sum)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol,
                   time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(lse_objective, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_polytope(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_polytope(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_polytope(h_out, target_dst)
