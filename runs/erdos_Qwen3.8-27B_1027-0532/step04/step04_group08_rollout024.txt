    """Compute Δcorr when h[i]: 1->0 and h[j]: 0->1."""
    r_h = h[::-1]  # reversed h

    # = r_h[(n-1-i)+s], valid when 0 <= (n-1-i)+s < n
    shift1 = n - 1 - i
    # valid s: max(0, -shift1) <= s < n - shift1 = i+1
    # i.e., s from max(0, i+1-n) to i (inclusive), but s is already 0..2n-2
    # since i < n, max(0, i+1-n) = 0
    # so s from 0 to i
    
    # Term 2: h[j-s+(n-1)]
    shift2 = n - 1 - j
    # valid s: 0 to j
    
    # Term 3: -(1-h)[i+s-(n-1)]
    shift3 = i - (n - 1)
    # valid s: max(0, -shift3) to n-1-shift3 = n-1-i+(n-1) = 2n-2-i
    # -shift3 = n-1-i
    # so s from n-1-i to 2n-2-i
    
    # Term 4: +(1-h)[j+s-(n-1)]
    shift4 = j - (n - 1)
    # valid s: n-1-j to 2n-2-j
    
    # Let me just use the direct indexing approach with np.where
    s = np.arange(2 * n - 1)
    
    idx_a = i - s + (n - 1)  # for h[i-s+(n-1)]
    idx_b = j - s + (n - 1)  # for h[j-s+(n-1)]
    idx_c = i + s - (n - 1)  # for (1-h)[i+s-(n-1)]
    idx_d = j + s - (n - 1)  # for (1-h)[j+s-(n-1)]
    
    va = (idx_a >= 0) & (idx_a < n)
    vb = (idx_b >= 0) & (idx_b < n)
    vc = (idx_c >= 0) & (idx_c < n)
    vd = (idx_d >= 0) & (idx_d < n)
    
    delta = np.zeros(2 * n - 1)
    delta += np.where(va, h[np.clip(idx_a, 0, n-1)], 0.0)
    delta -= np.where(vb, h[np.clip(idx_b, 0, n-1)], 0.0)
    delta -= np.where(vc, one_h[np.clip(idx_c, 0, n-1)], 0.0)
    delta += np.where(vd, one_h[np.clip(idx_d, 0, n-1)], 0.0)
    
    # Cross terms
    s_ij = j - i + (n - 1)  # for [s == j-i+n-1]
    s_ji = i - j + (n - 1)  # for [s == i-j+n-1]
    s_mid = n - 1  # for [s == n-1]
    
    if 0 <= s_ij < 2*n-1:
        delta[s_ij] += 1.0
    if 0 <= s_ji < 2*n-1:
        delta[s_ji] += 1.0
    delta[s_mid] -= 2.0
    
    return delta

import time

    h_bin[sorted_idx[:int(target)]] = 1.0
    return h_bin

    """Steepest descent on binary space via single swaps."""
    h = h.copy()
    score = compute_c5(h)
    
    for _ in range(200):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        best_i = -1
        best_j = -1
        
        for i in ones:

            for j in zeros:

                h[j] = 1.0
                new_score = compute_c5(h)
                h[j] = 0.0
                if new_score < best_score:

            h[i] = 1.0
        
        if best_i < 0:
            break
        
        h[best_i] = 0.0
        h[best_j] = 1.0
        score = best_score

def binary_simulated_annealing(h, n, deadline, t0=0.002, t1=0.00001, steps=50000):
    """Simulated annealing on binary space."""
    h = h.copy()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    rng = np.random.RandomState(12345)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

            score = new_score

            zeros = np.where(h == 0.0)[0]
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_score

    for step in range(n_steps):
        if time.time() > deadline:
            break
        progress = step / float(n_steps)
        T = t_init * (t_final / t_init) ** progress
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - score
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            score = new_score
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

def pair_swap_search(h, deadline, rng, n_tries=50000):
    n = len(h)
    h = h.copy()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            continue
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        new_score = compute_c5(h)
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_h = h.copy()
            score = new_score
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
    return best_h, best_score

def continuous_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    score = compute_c5(h)
    alpha = 50000.0

    def soft_obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_eq(x):
        return np.sum(x) - target

    if time.time() < deadline - 5:
        try:
            res = minimize(soft_obj, h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cons_eq}],
                          options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            new_score = compute_c5(h_new)
            if new_score < score:
                h = h_new
                score = new_score
        except Exception:
            pass
    return h, score

def greedy_construction(n, target, deadline):
    h = np.zeros(n)
    score = 0.0
    for step in range(int(target)):
        if time.time() > deadline:
            break
        best_pos = 0
        best_score = score
        for pos in range(n):
            if h[pos] > 0:
                continue
            h[pos] = 1.0
            new_score = compute_c5(h)
            h[pos] = 0.0
            if new_score < best_score:
                best_score = new_score
                best_pos = pos
        h[best_pos] = 1.0
        score = best_score
    return h, score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binarize initial and steepest descent
    if tl() > 60:
        h_bin = binarize(h_init, target)
        dl = time.time() + min(50.0, tl() * 0.3)
        h_sd, sc_sd = binary_steepest_descent(h_bin, dl)
        if sc_sd < best_score:
            best_score = sc_sd
            best_h = h_sd.copy()

    # Phase 2: SA from best binary
    if tl() > 80:
        h_bin = binarize(best_h, target)
        dl = time.time() + min(25.0, tl() * 0.15)
        h_sd, sc_sd = binary_steepest_descent(h_bin, dl)
        if sc_sd < best_score:
            best_score = sc_sd
            best_h = h_sd.copy()

        for t_i, t_f, ns, s in [(0.003, 0.000005, 120000, 100),
                                  (0.0015, 0.00001, 100000, 200),
                                  (0.005, 0.00002, 150000, 300)]:
            if tl() < 25:
                break
            h_bin = binarize(best_h, target)
            dl = time.time() + min(45.0, tl() * 0.2)
            h_sa, sc_sa = binary_sa(h_bin, dl, t_i, t_f, ns, s)
            if sc_sa < best_score:
                best_score = sc_sd = sc_sa
                best_score = sc_sa
                best_h = h_sa.copy()

    # Phase 3: Pair swap search
    if tl() > 60:
        h_bin = binarize(best_h, target)
        dl = time.time() + min(35.0, tl() * 0.2)
        h_ps, sc_ps = pair_swap_search(h_bin, dl, rng, n_tries=80000)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps.copy()

    # Phase 4: Steepest descent after pair swaps
    if tl() > 40:
        h_bin = binarize(best_h, target)
        dl = time.time() + min(30.0, tl() * 0.25)
        h_sd, sc_sd = binary_steepest_descent(h_bin, dl)
        if sc_sd < best_score:
            best_score = sc_sd
            best_h = h_sd.copy()

    # Phase 5: Continuous polish
    if tl() > 25:
        h_pol, sc_pol = continuous_polish(best_h, time.time() + min(20.0, tl() * 0.35))
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol.copy()

    # Phase 6: Multi-start from random binaries
    if tl() > 80:
        for attempt in range(4):
            if tl() < 25:
                break
            h_rand = np.zeros(n)
            positions = rng.choice(n, size=int(target), replace=False)
            h_rand[positions] = 1.0
            dl = time.time() + min(25.0, tl() * 0.2)
            h_sd, sc_sd = binary_steepest_descent(h_rand, dl)
            if sc_sd < best_score:
                best_score = sc_sd
                best_h = h_sd.copy()
            if tl() > 25:
                h_bin = binarize(h_sd, target)
                sa_dl = time.time() + min(25.0, tl() * 0.2)
                h_sa, sc_sa = binary_sa(h_bin, sa_dl, 0.002, 0.00001, 60000, seed=1000 + attempt)
                if sc_sa < best_score:
                    best_score = sc_sa
                    best_h = h_sa.copy()

    # Phase 7: Greedy construction as another starting point
    if tl() > 50:
        dl = time.time() + min(20.0, tl() * 0.3)
        h_greedy, sc_greedy = greedy_construction(n, target, dl)
        if sc_greedy < best_score:
            best_score = sc_greedy
            best_h = h_greedy.copy()
        if tl() > 30:
            dl2 = time.time() + min(20.0, tl() * 0.3)
            h_sd, sc_sd = binary_steepest_descent(h_greedy, dl2)
            if sc_sd < best_score:
                best_score = sc_sd
                best_h = h_sd.copy()

    # Phase 8: More SA from the best found
    if tl() > 50:
        h_bin = binarize(best_h, target)
        dl = time.time() + min(35.0, tl() * 0.35)
        h_sa, sc_sa = binary_sa(h_bin, dl, 0.001, 0.000001, 120000, seed=9999)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()
        if tl() > 20:
            h_bin = binarize(best_h, target)
            dl = time.time() + min(15.0, tl() * 0.3)
            h_sd, sc_sd = binary_steepest_descent(h_bin, dl)
            if sc_sd < best_score:
                best_score = sc_sd
                best_h = h_sd.copy()

    # Phase 9: Final continuous polish
    if tl() > 15:
        h_pol, sc_pol = continuous_polish(best_h, time.time() + min(10.0, tl() * 0.5))
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol.copy()

    # Phase 10: One last binary steepest descent if time permits
    if tl() > 15:
        h_bin = binarize(best_h, target)
        dl = time.time() + min(10.0, tl() * 0.5)
        h_sd, sc_sd = binary_steepest_descent(h_bin, dl)
        if sc_sd < best_score:
            best_score = sc_sd
            best_h = h_sd.copy()
        if tl() > 8:
            h_pol, sc_pol = continuous_polish(best_h, time.time() + min(5.0, tl() * 0.6))
            if sc_pol < best_score:
                best_score = sc_pol
                best_h = h_pol.copy()

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
