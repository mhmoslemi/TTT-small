def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while True:
        is_p = p >= 2
        if is_p and p < 4:
            is_p = True
        elif is_p and (p % 2 == 0 or p % 3 == 0):
            is_p = False
        elif is_p:
            i = 5
            while i * i <= p:
                if p % i == 0 or p % (i + 2) == 0:
                    is_p = False
                    break
                i += 6
        if is_p:
            break
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h


def make_random_binary(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    current = int(h.sum())
    target = n // 2
    if current < target:
        zeros = np.where(h == 0)[0]

    elif current > target:

    return h

def continuous_optimize(h_init, n, target, time_limit, t_start):
    dx = 2.0 / n
    h_cur = np.clip(h_init.astype(float), 0.0, 1.0)
    h_cur = h_cur + (target - np.sum(h_cur)) / n
    h_cur = np.clip(h_cur, 0.0, 1.0)
    curr_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 150, 400, 1000, 2000]
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, n, target),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'gtol': 1e-8})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = h_new + (target - np.sum(h_new)) / n
            h_new = np.clip(h_new, 0.0, 1.0)
            sc = compute_c5(h_new)
            if sc < curr_c5:
                curr_c5 = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_c5


def _smooth_obj(x, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

        h_ls, c5_ls = fast_local_search(h_pert_bin, target_use, 250000, rng,
                                         min(90.0, remaining), start,
                                         t_init=4.0 + restart, t_final=0.03,
                                         swap2_prob=0.15, targeted_prob=0.35)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - 30.0 - (time.time() - start)

    remaining = budget_s - 30.0 - (time.time() - start)

    if remaining > 60.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cont, c5_cont = continuous_optimize(best_h, n_use, target_use,
                                              min(50.0, remaining - 10.0), start)
        h_cont_bin = project_binary(h_cont, target_use)
        h_ls, c5_ls = fast_local_search(h_cont_bin, target_use, 100000, rng,
                                         min(40.0, remaining - 10.0), start,
                                         t_init=2.0, t_final=0.02, swap2_prob=0.1,
                                         targeted_prob=0.4)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - 30.0 - (time.time() - start)

    if remaining > 10.0:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        h_ls, c5_ls = fast_local_search(h_bin_final, target_use, 80000, rng,
                                         max(5.0, remaining - 5.0), start,
                                         t_init=1.0, t_final=0.01, swap2_prob=0.1,
                                         targeted_prob=0.4)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    n_final = len(best_h)
    h_final = project_binary(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 <= best_c5:
        best_h = h_final
        best_c5 = final_c5

    return best_h, best_c5, n_final
