        h_pert = best_h + rng.randn(n_use) * 0.002
        h_pert = project_feasible(h_pert, tgt_use)
        h_new, score = lp_refinement(h_pert, tgt_use, max_iterations=200, topk=20,
                                     time_limit=_remaining() - 60)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 50:
            h_ref, sc = slsqp_smooth(best_h, len(best_h) / 2.0, alpha=2000, topk=10,
                                     maxiter=3000, ftol=1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 12: Final LP polish
    if _remaining() > 30:
        tgt_final = len(best_h) / 2.0
        h_new, score = lp_refinement(best_h, tgt_final, max_iterations=100, topk=5,
                                     time_limit=_remaining() - 10)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
