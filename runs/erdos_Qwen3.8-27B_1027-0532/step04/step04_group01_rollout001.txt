        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0
    return h

def make_random_binary(n, rng):
    h = np.zeros(n, dtype=np.int8)
    target = n // 2
    positions = rng.choice(n, target, replace=False)
    h[positions] = 1
    return h

def fast_lsa(h_int, target, rng, time_limit, t_start,
             t_init=0.01, t_final=0.00001, max_iters=1000000):
    n = len(h_int)
    h = h_int.copy().astype(np.int8)
    h_f = h.astype(np.float64)
    corr = np.correlate(h_f, 1.0 - h_f, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    no_imp = 0
    max_no_imp = 5000
    nm1 = n - 1
    corr_len = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if no_imp > max_no_imp:
            h = best_h.copy()
            h_f = h.astype(np.float64)
            corr = np.correlate(h_f, 1.0 - h_f, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones_list = np.where(h == 1)[0].tolist()
            zeros_list = np.where(h == 0)[0].tolist()
            npert = rng.randint(5, max(6, n//25 + 1))
            for _ in range(npert):
                if ones_list and zeros_list:
                    pi = ones_list[rng.randint(len(ones_list))]
                    pj = zeros_list[rng.randint(len(zeros_list))]
                    h[pi] = 0; h[pj] = 1
                    ones_list.remove(pi); zeros_list.remove(pj)
                    zeros_list.append(pi); ones_list.append(pj)
            h_f = h.astype(np.float64)
            corr = np.correlate(h_f, 1.0 - h_f, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            no_imp = 0
            continue
        
        if not ones_list or not zeros_list:
            break
        
        ratio = min(it / max(max_iters - 1, 1), 1.0)
        temp = t_init * (t_final / t_init) ** ratio
        
        mt = rng.randint(3)
        
        if mt == 0:
            # Directed single swap
            worst_m = int(np.argmax(corr))
            worst_shift = worst_m - nm1
            if worst_shift >= 0:
                lo, hi = 0, n - worst_shift
            else:
                lo, hi = -worst_shift, n
            
            contrib = []
            for oi in ones_list:
                if lo <= oi < hi:
                    p2 = oi + worst_shift
                    if 0 <= p2 < n and h[p2] == 0:
                        contrib.append(oi)
            
            if contrib:
                ii = contrib[rng.randint(len(contrib))]
            else:
                ii = ones_list[rng.randint(len(ones_list))]
            jj = zeros_list[rng.randint(len(zeros_list))]
            
        elif mt == 1:
            ii = ones_list[rng.randint(len(ones_list))]
            jj = zeros_list[rng.randint(len(zeros_list))]
            
        else:
            # Double swap
            if len(ones_list) < 2 or len(zeros_list) < 2:
                continue
            i1 = ones_list[rng.randint(len(ones_list))]
            i2 = ones_list[rng.randint(len(ones_list))]
            if i1 == i2: continue
            j1 = zeros_list[rng.randint(len(zeros_list))]
            j2 = zeros_list[rng.randint(len(zeros_list))]
            if j1 == j2: continue
            
            h[i1]=0; h[i2]=0; h[j1]=1; h[j2]=1
            h_f[i1]=0; h_f[i2]=0; h_f[j1]=1; h_f[j2]=1
            new_corr = np.correlate(h_f, 1.0-h_f, mode='full')
            new_c5 = float(np.max(new_corr)) * 2.0 / n
            d = new_c5 - cur_c5
            if d < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-d/temp)):
                corr = new_corr.astype(np.float64)
                cur_c5 = new_c5
                ones_list = np.where(h==1)[0].tolist()
                zeros_list = np.where(h==0)[0].tolist()
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5; best_h = h.copy(); no_imp = 0
                else:
                    no_imp += 1
            else:
                h[i1]=1; h[i2]=1; h[j1]=0; h[j2]=0
                h_f[i1]=1; h_f[i2]=1; h_f[j1]=0; h_f[j2]=0
                no_imp += 1
            continue
        
        # Single swap: compute delta incrementally
        # Flip ii: 1->0, Flip jj: 0->1
        delta = np.zeros(corr_len, dtype=np.float64)
        
        # ii as first arg (a[ii]): -(1-h[ii+m-nm1]) for m in [nm1-ii, nm1+ii]
        mlo = max(nm1 - ii, 0)
        mhi = min(nm1 + ii, corr_len - 1)
        if mlo <= mhi:
            idx = ii + np.arange(mlo, mhi+1) - nm1
            delta[mlo:mhi+1] -= (1.0 - h_f[idx])
        
        # ii as second arg (b[ii]): +h[ii-m+nm1] for m in [ii-nm1, ii+nm1]
        mlo = max(ii - nm1, 0)
        mhi = min(ii + nm1, corr_len - 1)
        if mlo <= mhi:
            idx = ii - np.arange(mlo, mhi+1) + nm1
            delta[mlo:mhi+1] += h_f[idx]
        
        # jj as first arg (a[jj]): +(1-h[jj+m-nm1]) for m in [nm1-jj, nm1+jj]
        mlo = max(nm1 - jj, 0)
        mhi = min(nm1 + jj, corr_len - 1)
        if mlo <= mhi:
            idx = jj + np.arange(mlo, mhi+1) - nm1
            delta[mlo:mhi+1] += (1.0 - h_f[idx])
        
        # jj as second arg (b[jj]): -h[jj-m+nm1] for m in [jj-nm1, jj+nm1]
        mlo = max(jj - nm1, 0)
        mhi = min(jj + nm1, corr_len - 1)
        if mlo <= mhi:
            idx = jj
