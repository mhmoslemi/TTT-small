    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

**Algorithm:**
1. Start from initial_h_values (n=600)
2. Project to binary
3. Phase 1: Gradient-based continuous optimization (log-sum-exp with increasing α)

   - This gives a good continuous solution
4. Phase 2: Round to binary and do smart local search

   - Multi-lag aware moves
5. Phase 3: Try different n values and periodic patterns

import numpy as np
from scipy.optimize import minimize, fmin_bfgs
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def smooth_c5(h, alpha):
    """Compute smooth approximation of C5"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    log_sum = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(log_sum * dx)

def smooth_c5_grad(h, alpha):
    """Compute smooth approximation of C5 and its gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        f_val = c_max * dx
        return f_val, np.zeros(n)
    p = p / p_sum
    log_sum = c_max + np.log(p_sum) / alpha
    f_val = float(log_sum * dx)
    
    # Compute gradient efficiently using top lags
    # grad[m] = dx * sum_k p[k] * [(1-h[m+k])*I[valid] - h[m-k]*I[valid]]
    
    # Use top lags for efficiency
    n_top = min(100, 2 * n - 1)
    top_idx = np.argsort(-p)[:n_top]
    
    A = np.zeros(n)  # A[m] = sum_k p[k] * h[m+k] * I[m+k valid]
    B = np.zeros(n)  # B[m] = sum_k p[k] * h[m-k] * I[m-k valid]
    S1 = np.zeros(n)  # S1[m] = sum_k p[k] * I[m+k valid]
    
    for idx in top_idx:
        k = idx - (n - 1)
        pk = p[idx]
        if pk < 1e-15:
            break
        if k >= 0:
            # m+k valid means m in [0, n-1-k]
            S1[:n-k] += pk
            A[:n-k] += pk * h[k:]
            # m-k valid means m in [k, n-1]
            B[k:] += pk * h[:n-k]
        else:
            kk = -k
            # m+k valid means m in [kk, n-1]
            S1[kk:] += pk
            A[kk:] += pk * h[:n-kk]
            # m-k valid means m in [0, n-1-kk]
            B[:n-kk] += pk * h[kk:]
    
    grad = dx * (S1 - A - B)
    return f_val, grad

def gradient_descent(h_init, target, alphas, steps_per_alpha, step_sizes, rng, time_limit, t_start):
    """Run projected gradient descent on smooth approximation"""
    n = len(h_init)
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if time.time() - t_start > time_limit:
                break
            f_val, grad = smooth_c5_grad(h, alpha)
            # Take step
            # Adaptive step size
            step = step_sizes[0] / (1.0 + 0.01 * alpha)
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            
            # Check improvement (use actual C5, not smooth)
            new_score = compute_c5(h_new)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
            h = h_new
        
        if time.time() - t_start > time_limit:
            break
    
    return best_h, best_score

def smart_local_search(h_init, target, rng, time_limit, t_start):

    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)

    zeros = np.where(h == 0)[0]
    
    no_imp = 0
    max_no_imp = 3000
    it = 0
    
    while time.time() - t_start < time_limit:
        it += 1
        if no_imp > max_no_imp:
            # Perturb and restart

            for _ in range(n_perturb):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = int(ones[rng.randint(len(ones))])
                    pj = int(zeros[rng.randint(len(zeros))])
                    h[pi] = 0.0
                    h[pj] = 1.0
            h = project_binary(h, target_int).astype(np.float64)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            max_no_imp = min(max_no_imp + 1000, 20000)
            continue
        
        # Find top lags
        top_k = min(5, len(corr))
        top_lags = np.argsort(-corr)[:top_k]
        
        # Strategy: for the worst lag, find a contributing 1 and a non-contributing 0
        move_made = False
        for lag_idx in top_lags:
            L = int(lag_idx) - (n - 1)
            if L == 0:
                continue
            
            # Find 1s that contribute to this lag: h[i]=1 AND h[i+L]=0
            # and 0s that don't add to this lag: h[j+L]=1 (so flipping j to 1 doesn't help this lag)
            
            # Sample a few candidates for speed
            n_try = 10
            for _ in range(n_try):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = int(ones[rng.randint(len(ones))])
                # Check if ii contributes to lag L
                idx_plus = ii + L
                if 0 <= idx_plus < n and h[idx_plus] == 0:
                    # ii contributes to lag L, good
                    # Now find j that doesn't add to lag L
                    j_candidates = []
                    for _j in range(min(20, len(zeros))):
                        jj = int(zeros[rng.randint(len(zeros))])
                        idx_j_plus = jj + L
                        if 0 <= idx_j_plus < n and h[idx_j_plus] == 1:
                            j_candidates.append(jj)
                        # Also check if jj would reduce another top lag
                        if len(j_candidates) >= 3:
                            break
                    if len(j_candidates) == 0:
                        jj = int(zeros[rng.randint(len(zeros))])
                    else:
                        jj = j_candidates[rng.randint(len(j_candidates))]
                    
                    if ii == jj:
                        continue
                    
                    # Try the swap
                    h[ii] = 0.0
                    h[jj] = 1.0
                    one_minus_h[ii] = 1.0
                    one_minus_h[jj] = 0.0
                    
                    new_corr = np.correlate(h, one_minus_h, mode='full')
                    new_c5 = float(np.max(new_corr)) * 2.0 / n

                            best_h = h.copy()
                        no_imp = 0
                        move_made = True
                        break
                    else:
                        # Revert
                        h[ii] = 1.0
                        h[jj] = 0.0
                        one_minus_h[ii] = 0.0
                        one_minus_h[jj] = 1.0

                break

**Key insight for better local search:**

The change in corr[k] when we swap (i: 1→0, j: 0→1) is:
Δcorr[k] = h[i+k] + h[i-k] - 1 - h[j+k] - h[j-k] (approximately, ignoring self-interactions)

For the worst lag L, we want Δcorr[L] < 0, i.e.:
h[i+L] + h[i-L] < 1 + h[j+L] + h[j-L]

Since h[i+L]=0 (because i contributes to lag L, meaning h[i]=1 and h[i+L]=0):
0 + h[i-L] < 1 + h[j+L] + h[j-L]
h[i-L] < 1 + h[j+L] + h[j-L]

This is almost always true (since LHS ≤ 1 and RHS ≥ 1). So the constraint at the worst lag is easy to satisfy.

The harder constraint is at OTHER lags. The swap increases corr at some lags and decreases at others. We need the net effect on the MAX to be negative.

A practical approach: after making the swap, check the new max. If it's lower, accept. If not, revert. This is what the previous code does.

To make it faster, we can:
1. Pre-compute which lags are "active" (near the max)
2. Only check those lags for the swap's effect
3. If all active lags decrease, accept without full recomputation

Let me implement this:

```python
def fast_swap_check(h, ones, zeros, i, j, active_lags, active_corr, n):
    """Check if swap (i:1->0, j:0->1) reduces all active lags"""
    # For each active lag L, compute the change
    # Δcorr[L] ≈ h[i+L] + h[i-L] - 1 - h[j+L] - h[j-L]
    # (using old values, with 0 for out-of-range)
    
    for L in active_lags:
        di_plus = h[i+L] if 0 <= i+L < n else 0
        di_minus = h[i-L] if 0 <= i-L < n else 0
        dj_plus = h[j+L] if 0 <= j+L < n else 0
        dj_minus = h[j-L] if 0 <= j-L < n else 0
        
        delta = di_plus + di_minus - dj_plus - dj_minus
        # If delta > 0, this lag increases, which is bad
        # But we also need to account for the fact that i and j are being swapped
        # Self-interaction: if i+L == j or j+L == i, we need to adjust
        if i + L == j:
            # h[i+L] in old h is h[j] = 0, but in new h it's h[j] = 1
            # So the actual delta is different
            delta -= 1  # correction
        if j + L == i:
            delta += 1
        
        if delta > 1e-10:
            return False  # This lag would increase
    return True
```

def project_binary(h, target_sum):
