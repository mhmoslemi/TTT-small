where d = N-1-k for the k-th element of the full correlation.

Now, for the smooth surrogate:
f(h) = max_k c_k + (1/α) log(sum_k exp(α(c_k - c_max)))

The gradient:
df/dh[i] = sum_k w_k * dc_k/dh[i]

where w_k = exp(α(c_k - c_max)) / sum_j exp(α(c_j - c_max)) (the softmax weights)

And dc_k/dh[i] = (1-h[i-d_k]) * I(0<=i-d_k<N) - h[i+d_k] * I(0<=i+d_k<N)
where d_k = N-1-k.

This is O(N * 2N) = O(N²) to compute the full gradient, which for N=400 is 160,000 operations per gradient evaluation. That's fine.

Actually, I can compute this more efficiently using the correlation structure. Note that:
sum_k w_k * (1-h[i-d_k]) * I(0<=i-d_k<N) = sum_{valid d} w_{N-1-d} * (1-h[i-d])

This is like a weighted sum of (1-h) with weights w. Similarly for the other term.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Softmax weights
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp  # softmax weights
    
    obj = c_max + np.log(sum_exp) / alpha
    
    # Gradient
    # dc_k/dh[i] = (1-h[i-d_k]) * I(0<=i-d_k<n) - h[i+d_k] * I(0<=i+d_k<n)
    # where d_k = n-1-k
    grad = np.zeros(n)
    # For efficiency, use the structure:
    # For each k, d_k = n-1-k
    # grad[i] += w[k] * [(1-h[i-d_k]) * I(0<=i-d_k<n) - h[i+d_k] * I(0<=i+d_k<n)]
    
    # Let's compute this as:
    # term1[i] = sum_k w[k] * (1-h[i-d_k]) * I(0<=i-d_k<n)
    #           = sum_{valid d} w[n-1-d] * (1-h[i-d])
    # term2[i] = sum_k w[k] * h[i+d_k] * I(0<=i+d_k<n)
    #           = sum_{valid d} w[n-1-d] * h[i+d]
    
    # For term1: for each d from -(n-1) to n-1 (where d = n-1-k, k from 0 to 2n-2):
    #   w[n-1-d] * (1-h[i-d]) for valid i-d
    # This is a weighted "cross-correlation" of w with (1-h)
    
    # Actually, let me just do it directly for clarity (n=400 or 800, O(n^2) is fine)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = n - 1 - k
        # term1: (1-h[i-d]) for i where 0 <= i-d < n, i.e., d <= i < d+n
        # term2: -h[i+d] for i where 0 <= i+d < n, i.e., -d <= i < n-d
        if d >= 0:
            # term1: i ranges from d to n-1
            grad[d:] += wk * (1.0 - h[:n-d])
            # term2: i ranges from 0 to n-1-d
            if n - 1 - d >= 0:
                grad[:n-d] += -wk * h[d:]
        else:
            dd = -d  # positive
            # term1: i ranges from 0 to n-1-dd
            grad[:n-dd] += wk * (1.0 - h[dd:])
            # term2: i ranges from dd to n-1
            grad[dd:] += -wk * h[:n-dd]
    
    return obj, grad

def gradient_descent(h, target, alpha_schedule, n_steps_per_alpha, lr, momentum, deadline, 
                     perturb_interval=200, perturb_size=0.05):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_score = compute_c5(best_h)
    v = np.zeros(n)  # momentum
    
    step_count = 0
    for alpha in alpha_schedule:
        for step in range(n_steps_per_alpha):
            if time.time() > deadline:
                return best_h, best_score
            
            obj, grad = smooth_obj_and_grad(h_cur, alpha)
            
            # Nesterov momentum
            v = momentum * v - lr * grad
            h_new = h_cur + v
            h_new = project_feasible(h_new, target)
            
            # Check improvement
            h_cur = h_new
            step_count += 1
            
            if step_count % 50 == 0:
                sc = compute_c5(h_cur)
                if sc < best_score:
                    best_score = sc
                    best_h = h_cur.copy()
            
            # Periodic perturbation
            if step_count % perturb_interval == 0:
                idx = np.random.randint(0, n, size=10)
                h_cur[idx] += np.random.randn(10) * perturb_size
                h_cur = project_feasible(h_cur, target)
                v = np.zeros(n)  # reset momentum
    
    sc = compute_c5(h_cur)
    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()
    
    return best_h, best_score

    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h


def fourier_h(n, K, amplitudes, phases):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(K):
        h += amplitudes[k] * np.cos(2.0 * np.pi * (k + 1) * x / 2.0 + phases[k])
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def block_gd(h, target, block_size, deadline, alpha, n_rounds, lr):
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            block = h_cur[s:e].copy()
            for sub in range(30):
                obj, w = corr_and_weights(h_cur, alpha)
                grad = grad_from_weights(h_cur, w)
                block -= lr * grad[s:e]
                block = np.clip(block, 0.0, 1.0)
                h_cur[s:e] = block
                h_cur = project_feasible(h_cur, target)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = two_interval_h(400, 100)

    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: GD at original resolution
    deadline1 = t0 + budget_s * 0.18
    h1, c5_1 = gd_nesterov(best_h, target, deadline1,
                           20, 2000, 2000, 0.006, 0.9, 300, 12, 0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    if time_left() < 80:
        best_h = project_feasible(best_h, n / 2.0)
        return best_h, compute_c5(best_h), n

    # Phase 2: SLSQP refinement at original resolution
    for alpha in [500, 2000, 5000]:
        if time_left() < 150:
            break
        h2, c5_2 = slsqp_refine(best_h, n / 2.0, alpha, 1500)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    if time_left() < 100:
        best_h = project_feasible(best_h, n / 2.0)
        return best_h, best_c5, n

    # Phase 3: Upsample and optimize at higher resolution
    h_up = upsample(best_h, 2)
    n_up = len(h_up)
    target_up = n_up / 2.0

    deadline3 = t0 + budget_s * 0.62
    h3, c5_3 = gd_nesterov(h_up, target_up, deadline3,
                           20, 8000, 5000, 0.004, 0.9, 250, 25, 0.08)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3
        n = n_up
        target = target_up

    if time_left() < 120:
        best_h = project_feasible(best_h, target)
        return best_h, best_c5, n

    # Phase 4: Block GD at higher resolution
    deadline4 = t0 + budget_s * 0.78
    for bs in [16, 8]:
        if time_left() < 100:
            break
        h4, c5_4 = block_gd(best_h, target, bs, deadline4,
                            alpha=2000, n_rounds=3, lr=0.003)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    if time_left() < 100:
        best_h = project_feasible(best_h, target)
        return best_h, best_c5, n

    # Phase 5: SLSQP at higher resolution
    for alpha in [2000, 10000]:
        if time_left() < 120:
            break
        h5, c5_5 = slsqp_refine(best_h, target, alpha, 1500)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    if time_left() < 80:
        best_h = project_feasible(best_h, target)
        return best_h, best_c5, n

    # Phase 6: Multi-restart at higher resolution with diverse inits
    deadline6 = t0 + budget_s * 0.90
    for attempt in range(5):
        if time_left() < 100:
            break
        h_pert = best_h.copy()
        pn = np.random.randint(30, 80)
        si = np.random.randint(0, n - pn)
        h_pert[si:si + pn] += np.random.randn(pn) * 0.1
        h_pert = project_feasible(h_pert, target)
        h6, c5_6 = gd_nesterov(h_pert, target, deadline6,
                               100, 10000, 1500, 0.003, 0.9, 500, 15, 0.05)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    if time_left() < 60:
        best_h = project_feasible(best_h, target)
        return best_h, best_c5, n

    # Phase 7: Final high-alpha GD polish
    deadline7 = t0 + budget_s * 0.96
    h7, c5_7 = gd_nesterov(best_h, target, deadline7,
                           10000, 200000, 2000, 0.0008, 0.97, 10000, 3, 0.01)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7

    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
