- As second factor at position i+s: affects corr for shift s where i is valid
  Change: h[i] * (v_old - v_new) = h[p-s] * (v_old - v_new)

For a: 1→0 (v_new - v_old = -1):
- As first factor: -1 * (1 - h[a+s]) for valid a+s
- As second factor: h[a-s] * 1 for valid a-s (since v_old - v_new = 1)

For b: 0→1 (v_new - v_old = +1):
- As first factor: +1 * (1 - h[b+s]) for valid b+s
- As second factor: h[b-s] * (-1) for valid b-s

So the total change vector can be computed as:
Δcorr = -[1 - h[a+shifts_valid]] + h[a-shifts_valid] + [1 - h[b+shifts_valid]] - h[b-shifts_valid]

This can be computed in O(n) using array slicing.

def flip_delta(h, a, b, n):
    """Compute the change in correlation when h[a]: 1->0, h[b]: 0->1"""
    delta = np.zeros(2*n - 1)
    # Shifts range from -(n-1) to (n-1), stored as indices 0 to 2n-2
    # Shift s is stored at index s + n - 1
    
    # a as first factor: shift s where a+s is in [0, n-1], i.e., s in [-a, n-1-a]
    s_start = max(-a, -(n-1))  # = -a (since a >= 0)
    s_end = min(n-1-a, n-1)  # = n-1-a (since a >= 0)
    for s in range(s_start, s_end + 1):
        idx = s + n - 1
        delta[idx] += -(1 - h[a + s])
    
    # a as second factor: shift s where a-s is in [0, n-1], i.e., s in [a-n+1, a]
    s_start = max(a - (n-1), -(n-1))
    s_end = min(a, n-1)
    for s in range(s_start, s_end + 1):
        idx = s + n - 1
        delta[idx] += h[a - s]
    
    # b as first factor: shift s where b+s is in [0, n-1]
    s_start = -b
    s_end = n - 1 - b
    for s in range(s_start, s_end + 1):
        idx = s + n - 1
        delta[idx] += (1 - h[b + s])
    
    # b as second factor: shift s where b-s is in [0, n-1]
    s_start = b - (n-1)
    s_end = b
    for s in range(s_start, s_end + 1):
        idx = s + n - 1
        delta[idx] += -h[b - s]
    
    return delta

def flip_delta(h, a, b, n):
    """Change in correlation when h[a]: 1->0, h[b]: 0->1.
    Correlation: c_k = sum_n h[n+k] * (1-h[n]) for valid n.
    Output index for shift k is k + n - 1 (since k ranges from -(n-1) to n-1)."""
    delta = np.zeros(2*n - 1)
    
    # a as first factor h[n+k]: n+k=a, n=a-k, k in [a-n+1, a]
    k_min = max(a - n + 1, -(n-1))
    k_max = min(a, n-1)
    if k_min <= k_max:
        k_vals = np.arange(k_min, k_max+1)
        idx = k_vals + n - 1
        delta[idx] -= (1.0 - h[a - k_vals])
    
    # a as second factor (1-h[n]): n=a, k in [-a, n-1-a]
    k_min = max(-a, -(n-1))
    k_max = min(n - 1 - a, n-1)
    if k_min <= k_max:
        k_vals = np.arange(k_min, k_max+1)
        idx = k_vals + n - 1
        delta[idx] += h[a + k_vals]
    
    # b as first factor h[n+k]: n+k=b, n=b-k, k in [b-n+1, b]
    k_min = max(b - n + 1, -(n-1))
    k_max = min(b, n-1)
    if k_min <= k_max:
        k_vals = np.arange(k_min, k_max+1)
        idx = k_vals + n - 1
        delta[idx] += (1.0 - h[b - k_vals])
    
    # b as second factor (1-h[n]): n=b, k in [-b, n-1-b]
    k_min = max(-b, -(n-1))
    k_max = min(n - 1 - b, n-1)
    if k_min <= k_max:
        k_vals = np.arange(k_min, k_max+1)
        idx = k_vals + n - 1
        delta[idx] -= h[b + k_vals]
    
    return delta

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
