        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)

def compute_A(h, w, lags, n):
    """Compute A[i] = sum_m w[m] * (-(1-h[i+k_m]) + h[i-k_m])"""
    i_idx = np.arange(n)[:, None].astype(int)
    k_vals = lags[None, :].astype(int)
    ipk = i_idx + k_vals
    ik = i_idx - k_vals
    valid_pk = (ipk >= 0) & (ipk < n)
    valid_k = (ik >= 0) & (ik < n)
    ipk_c = np.clip(ipk, 0, n - 1)
    ik_c = np.clip(ik, 0, n - 1)
    term1 = np.where(valid_pk, -(1.0 - h[ipk_c]), 0.0)
    term2 = np.where(valid_k, h[ik_c], 0.0)
    return (term1 + term2) @ w

def mw_local_search(h, w, lags, n, max_inner=10000):
    """Best-move local search with efficient A updates."""
    m = len(lags)
    A = compute_A(h, w, lags, n)
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    for it in range(max_inner):
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        A_ones = A[ones_idx]
        A_zeros = A[zeros_idx]
        min_p = np.argmin(A_ones)
        max_p = np.argmax(A_zeros)
        if A_ones[min_p] >= A_zeros[max_p] - 1e-12:
            break
        a = int(ones_idx[min_p])
        b = int(zeros_idx[max_p])
        h[a] = 0
        h[b] = 1
        # Update A for removal at a
        idx1 = a - lags
        v1 = (idx1 >= 0) & (idx1 < n)
        A[idx1[v1]] -= w[v1]
        idx2 = a + lags
        v2 = (idx2 >= 0) & (idx2 < n)
        A[idx2[v2]] -= w[v2]
        # Update A for addition at b
        idx3 = b - lags
        v3 = (idx3 >= 0) & (idx3 < n)
        A[idx3[v3]] += w[v3]
        idx4 = b + lags
        v4 = (idx4 >= 0) & (idx4 < n)
        A[idx4[v4]] += w[v4]
        # Update index arrays
        ones_idx = np.delete(ones_idx, min_p)
        zeros_idx = np.delete(zeros_idx, max_p)
        ones_idx = np.append(ones_idx, b)
        zeros_idx = np.append(zeros_idx, a)
    
    return h, A

def multisplit_search(h_init, target_sum, n, time_limit, start_time, rng):
    """Multiplicative weight + local search."""
    h = h_init.copy()
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    best_h = h.copy()
    
    # Initialize weights
    w = np.clip(corr, 1e-8, None)
    w /= np.sum(w)
    
    no_improve = 0
    max_no_improve = 50
    
    eta_start = 1.0
    eta_end = 0.1
    
    for outer in range(10000):
        if time.time() - start_time > time_limit:
            break
        
        # Anneal learning rate
        ratio = min(outer / 500.0, 1.0)
        eta = eta_start * (eta_end / eta_start) ** ratio
        
        h, A = mw_local_search(h, w, lags, n, max_inner=5000)
        
        corr = compute_corr(h)
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Reset weights and optionally perturb
            w = np.clip(corr, 1e-8, None)
            w /= np.sum(w)
            no_improve = 0
            # Small perturbation
            if rng.random() < 0.5:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    n_swap = min(3, len(ones), len(zeros))
                    swap_ones = rng.choice(ones, n_swap, replace=False)
                    swap_zeros = rng.choice(zeros, n_swap, replace=False)
                    h[swap_ones] = 0
                    h[swap_zeros] = 1
        
        # Update weights
        scale = max(float(np.max(corr)), 1e-10)
        w = w * np.exp(eta * corr / scale)
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            w = np.ones(m) / m
        else:
            w /= w_sum
    
    return best_h, best_c5

def informed_sa(h_start, target_sum, max_iters, t_init, t_final, rng,
                time_limit=None, start_time=None):

    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]
    
    for it in range(max_iters):

                break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
            else:
                break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        h_new = h.copy()
        moved = False
        i1 = ones_cache[rng.randint(len(ones_cache))]
        i0 = zeros_cache[rng.randint(len(zeros_cache))]
        h_new[i1] = 0
        h_new[i0] = 1
        moved = True
        if moved:
            corr_new = compute_corr(h_new)
            c5_new = float(np.max(corr_new))
            delta = c5_new - current_c5
            if delta < -1e-12 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
                h = h_new
                corr = corr_new
                current_c5 = c5_new
                ones_cache = np.where(h == 1)[0]
                zeros_cache = np.where(h == 0)[0]
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

def top_p_smooth(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_smooth(h_start, n, target, alphas, p_vals, maxiter, ftol,
                 time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(lambda x: top_p_smooth(x, alpha, p), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, curr_score

    h = h_start.copy()
    if not np.allclose(h, np.round(h)):
        h = project_binary(h, target_sum)
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(2000, max_iters // 12)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.04:
                no_improve = 0
            else:
                break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        move_type = rng.randint(4)
        h_new = h.copy()
        moved = False
        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            worst_idx = np.argmax(corr)
            k_star = worst_idx - (n - 1)
            target_pos1 = (i1 + k_star) % n
            target_pos2 = (i1 - k_star) % n
            dist1 = np.abs(zeros_cache.astype(float) - target_pos1)
            dist2 = np.abs(zeros_cache.astype(float) - target_pos2)
            dist = np.minimum(dist1, dist2) + rng.random(len(zeros_cache)) * 10.0
            i0 = zeros_cache[np.argmin(dist)]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 1 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 2 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            i1a = ones_cache[idx1[0]]
            i1b = ones_cache[idx1[1]]
            worst_idx = np.argmax(corr)
            k_star = worst_idx - (n - 1)
            t1 = (i1a + k_star) % n
            t2 = (i1b - k_star) % n
            d1 = np.abs(zeros_cache.astype(float) - t1) + rng.random(len(zeros_cache)) * 8.0
            d2 = np.abs(zeros_cache.astype(float) - t2) + rng.random(len(zeros_cache)) * 8.0
            i0a = zeros_cache[np.argmin(d1)]
            i0b = zeros_cache[np.argmin(d2)]
            if i0a == i0b:
                i0b = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1a] = 0
            h_new[i1b] = 0
            h_new[i0a] = 1
            h_new[i0b] = 1
            moved = True
        elif move_type == 3 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            worst_indices = np.argsort(-corr)[:top_m]
            i1 = ones_cache[rng.randint(len(ones_cache))]
            best_score_val = 1e9
            best_i0 = zeros_cache[0]
            num_cand = min(30, len(zeros_cache))
            cand_idx = rng.choice(len(zeros_cache), num_cand, replace=False)
            for ci in cand_idx:
                i0 = zeros_cache[ci]
                score = 0.0
                for wi in worst_indices[:top_m]:
                    k = wi - (n - 1)
                    j1 = i1 + k
                    j2 = i0 + k
                    val1 = 0.0
                    val2 = 0.0
                    if 0 <= j1 < n:
                        val1 = h[j1]
                    if 0 <= j2 < n:
                        val2 = h[j2]
                    delta = -val1 + (1.0 - val2) + (1.0 - val1) - (1.0 - val2)
                    j1b = i1 - k
                    j2b = i0 - k
                    val1b = 0.0
                    val2b = 0.0
                    if 0 <= j1b < n:
                        val1b = h[j1b]
                    if 0 <= j2b < n:
                        val2b = h[j2b]
                    delta += -val1b + (1.0 - val2b) + (1.0 - val1b) - (1.0 - val2b)
                    score += max(0.0, corr[wi] + delta * (2.0 / n) - best_c5)
                if score < best_score_val:
                    best_score_val = score
                    best_i0 = i0
            h_new[i1] = 0
            h_new[best_i0] = 1
            moved = True
        if not moved:
            continue
        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5
        if delta < -1e-12 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5


def top_p_smooth(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = float(np.max(top))
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return max_top + float(np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_smooth(h_start, n, target, alphas, p_vals, maxiter, ftol,
                 time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(lambda x, al=alpha, pv=p: top_p_smooth(x, al, pv), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score - 1e-15:
                    curr_score = sc
                    h_cur = h_new.copy()
            except Exception:
                pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Multiplicative weight search
    if tleft() > 80:
        mw_time = min(tleft() * 0.55, 650.0)
        h_mw, c5_mw = multisplit_search(best_h, target, n, mw_time, start_time, rng)
        if c5_mw < best_score:
            best_score = c5_mw
            best_h = h_mw.copy()

    # Phase 2: Second MW pass with different params
    if tleft() > 80:
        mw_time2 = min(tleft() * 0.25, 250.0)
        h_mw2, c5_mw2 = multisplit_search(best_h, target, n, mw_time2, start_time, rng)
        if c5_mw2 < best_score:
            best_score = c5_mw2
            best_h = h_mw2.copy()

    # Phase 3: SA refinement
    if tleft() > 60:
        sa_time = min(tleft() * 0.3, 200.0)
        max_it = int(sa_time * 2500)
        h_sa, c5_sa = informed_sa(best_h, target, max_it, 0.012, 0.00003,
                                  rng, time_limit=sa_time, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: SLSQP
    if tleft() > 50:
        p_vals = [max(3, n // 60), max(5, n // 35), max(8, n // 25)]
        alphas = [100, 300, 800, 2000, 5000]
        h_sl, sc_sl = slsqp_smooth(best_h, n, target, alphas, p_vals,
                                   300, 1e-14, time_limit=max(5.0, tleft() - 25),
                                   start_time=start_time)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl.copy()

    # Phase 5: Try alternate resolution
    if tleft() > 80:
        n_alt = int(n * 0.8)
        if n_alt % 2 == 0 and n_alt >= 200:
            h_interp = interpolate_to(best_h, n_alt, n_alt / 2.0)
            h_bin_alt = project_binary(h_interp, n_alt / 2.0)
            alt_time = min(tleft() * 0.3, 150.0)
            h_a, c5_a = multisplit_search(h_bin_alt, n_alt / 2.0, n_alt, alt_time, start_time, rng)
            if alt_time > 30:
                sa_t2 = min(40.0, tleft() * 0.2)
                if time.time() - start_time + sa_t2 < budget_s - 30:
                    max_it2 = int(sa_t2 * 2000)
                    h_sa2, c5_sa2 = informed_sa(h_a, n_alt / 2.0, max_it2, 0.01, 0.00005,
                                                rng, time_limit=sa_t2, start_time=start_time)
                    if c5_sa2 < c5_a:
                        h_a = h_sa2
                        c5_a = c5_sa2
            if c5_a < best_score:
                best_score = c5_a
                best_h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_alt), h_a)
                best_h = project_binary(best_h, target)
                best_score = compute_c5(best_h)

    # Phase 6: Final SA
    if tleft() > 25:
        h_bin_f = project_binary(best_h, target)
        max_it_f = int(max(3000, tleft() * 2000))
        h_sf, c5_sf = informed_sa(h_bin_f, target, max_it_f, 0.005, 0.00002,
                                  rng, time_limit=max(5.0, tleft() - 12), start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Phase 7: Final SLSQP polish
    if tleft() > 20:
        p_vals_f = [max(3, n // 50), max(8, n // 25)]
        alphas_f = [500, 1500, 4000]
        h_ref, sc_ref = slsqp_smooth(best_h, n, target, alphas_f, p_vals_f,
                                     200, 1e-15, time_limit=max(3.0, tleft() - 8),
                                     start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
