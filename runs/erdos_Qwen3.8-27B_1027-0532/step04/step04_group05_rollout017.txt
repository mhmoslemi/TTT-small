    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    shifted = alpha * (corr - np.max(corr))
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient of f_alpha w.r.t. h
    # For each j: sum_k w_k * [(1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n)] * dx
    grad = np.zeros(n)

        if wk < 1e-15:

def augmented_lagrangian_smooth_max(h_init, n, target, alpha, time_limit, max_outer=50, max_inner=300):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    mu = 1.0
    lam = 0.0
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for outer in range(max_outer):
        if time.time() - start > time_limit:
            break
        
        def obj_and_grad(x):
            val, _ = smooth_max_value_and_grad(x, alpha)
            # Add augmented Lagrangian term
            gap = np.sum(x) - target - lam / mu
            val += 0.5 * mu * gap * gap
            grad = smooth_max_grad(x, alpha)
            grad += mu * gap * np.ones(n)
            return val, grad
        
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=[(0,1)]*n, jac=True,

        if compute_c5(h) < best_score:
            best_score = compute_c5(h)
            best_h = h.copy()
        
        if abs(gap) < 1e-8:
            break
        
        mu = min(mu * 2, 1e6)
    
    return best_h, best_score

from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    es = np.exp(shifted)
    w = es / es.sum()
    val = c_max + np.log(es.sum()) / alpha
    grad = np.zeros(n)
    n_half = n - 1
    for k_idx in range(2 * n - 1):
        wi = w[k_idx]
        if wi < 1e-14:
            continue
        k = k_idx - n_half
        if k >= 0:
            if k < n:
                grad[:n-k] += wi * (1.0 - h[k:])

            if kn < n:

    return val, grad

def lbfgs_smooth(h_init, n, target, alpha, maxiter, time_limit, mu_start=10.0, mu_step=2.0, max_mu_iters=5):
    """Minimize smooth max + penalty using L-BFGS-B."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for mu_iter in range(max_mu_iters):
        if time.time() - start > time_limit:
            break
        mu = mu_start * (mu_step ** mu_iter)
        
        def obj(x):
            v, _ = smooth_max_and_grad(x, alpha)
            return v + 0.5 * mu * (np.sum(x) - target)**2
        
        def grad(x):
            _, g = smooth_max_and_grad(x, alpha)
            g = g + mu * (np.sum(x) - target)
            return g
        
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        h = project_feasible(res.x, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score

from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            h[:] = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    es = np.exp(shifted)
    w = es / es.sum()
    val = c_max + np.log(es.sum()) / alpha
    grad = np.zeros(n)
    n_half = n - 1
    for k_idx in range(2 * n - 1):
        wi = w[k_idx]
        if wi < 1e-14:
            continue
        k = k_idx - n_half
        if k >= 0:
            if k < n:
                grad[:n-k] += wi * (1.0 - h[k:])
                grad[k:] -= wi * h[:n-k]
        else:
            kn = -k
            if kn < n:
                grad[kn:] += wi * (1.0 - h[:n-kn])
                grad[:n-kn] -= wi * h[kn:]
    return val, grad

def lbfgs_optimize(h_init, n, target, alpha, time_limit, maxiter=500):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for mu in [10.0, 50.0, 200.0, 1000.0]:
        if time.time() - start > time_limit:
            break
        
        def obj(x):
            v, _ = smooth_max_and_grad(x, alpha)
            return v + 0.5 * mu * (np.sum(x) - target) ** 2

            g = g.copy()
            g += mu * (np.sum(x) - target)
            return g

    return best_h, best_score
