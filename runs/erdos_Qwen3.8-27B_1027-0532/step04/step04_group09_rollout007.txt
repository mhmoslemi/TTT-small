import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_optimize(h, target, alpha, n, maxiter, ftol, deadline):
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def active_subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        rng_span = c_max - c_min
        if rng_span < 1e-14:
            break
        threshold = c_max - active_frac * rng_span
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        step = step_size / (1.0 + 0.03 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def random_feasible(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n / 2.0)


def structured_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.choice([3, 4, 5, 6, 8])
    block_len = n // n_blocks
    remaining = n - block_len * n_blocks
    offsets = []
    pos = 0
    for i in range(n_blocks):
        bl = block_len + (1 if i < remaining else 0)
        if i % 2 == 0:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.uniform(0, 0.1, n)
    return project_feasible(h, n / 2.0)


def multi_resolution_search(n_levels, n_random, n_struct, alpha, maxiter, deadline, rng, init_h=None):
    best_h = None
    best_score = float('inf')

    if init_h is not None:
        n_first = n_levels[0]
        h_interp = interpolate_to(init_h, n_first, n_first / 2.0)
        h_opt, sc = slsqp_optimize(h_interp, n_first / 2.0, alpha, n_first, maxiter, 1e-10, deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    for level_n in n_levels:
        if time.time() > deadline:
            break
        target = level_n / 2.0

        candidates = []
        if best_h is not None:
            candidates.append(interpolate_to(best_h, level_n, target))

        for r in range(n_random):
            if time.time() > deadline:
                break
            candidates.append(random_feasible(level_n, rng))

        for r in range(n_struct):
            if time.time() > deadline:
                break
            candidates.append(structured_init(level_n, rng))

        for h0 in candidates:
            if time.time() > deadline:
                break
            h_opt, sc = slsqp_optimize(h0, target, alpha, level_n, maxiter, 1e-10, deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_opt

    return best_h, best_score


def projected_gradient(h, target, alpha, n_iter, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    y = h_cur.copy()
    t = 1.0
    h_prev = h_cur.copy()
    step = 1.0

    for it in range(n_iter):
        if time.time() > deadline:
            break
        _, grad_y = lse_obj_and_grad(y, alpha)
        h_new = y - step * grad_y
        h_new = project_feasible(h_new, target)

        obj_new, _ = lse_obj_and_grad(h_new, alpha)
        obj_y, _ = lse_obj_and_grad(y, alpha)
        diff = h_new - y
        quad = obj_y + np.dot(grad_y, diff) + 0.5 * step * np.dot(diff, diff)
        if obj_new > quad + 1e-10:
            step *= 0.5
            h_new = y - step * grad_y
            h_new = project_feasible(h_new, target)
        else:
            if it % 50 == 49:
                step = min(step * 1.2, 10.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_new + momentum * (h_new - h_prev)
        y = np.clip(y, 0.0, 1.0)
        t = t_new
        h_prev = h_cur
        h_cur = h_new

        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()

    return best_h, best_score
