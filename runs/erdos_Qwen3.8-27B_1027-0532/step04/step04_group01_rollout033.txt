from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr * dx
    a_max = np.max(a)
    exp_a = np.exp(a - a_max)
    w = exp_a / np.sum(exp_a)
    
    grad = np.zeros(n)
    for L in range(0, n):
        wk = w[n-1+L]
        if wk < 1e-10:
            continue
        # For lag L >= 0:
        # ∂corr/∂h[j] = (1-h[j+L]) for j in [0, n-1-L]
        #              - h[j-L] for j in [L, n-1]
        if L == 0:
            grad[:n] += wk * (1.0 - h[:n])  # (1-h[j]) for j in [0,n-1]
            # second term: -h[j] for j in [0, n-1], so:
            grad[:n] -= wk * h[:n]
        else:
            grad[:n-L] += wk * (1.0 - h[L:n])
            grad[L:n] -= wk * h[:n-L]

        grad[L:n] += wk * (1.0 - h[:n-L])

def project_to_polytope(a, target_sum):
    n = len(a)
    # Find lambda such that sum(clip(a_i - lambda, 0, 1)) = target_sum
    # clip(a_i - lambda, 0, 1) = max(0, min(1, a_i - lambda))
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(a - lam, 0, 1))
    
    # Binary search for lambda
    lo = np.min(a) - 1
    hi = np.max(a) + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.clip(a - lam, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_to_polytope(a, target_sum):
    n = len(a)
    lo = np.min(a) - 1.0
    hi = np.max(a) + 1.0
    for _ in range(80):

        s = np.sum(np.clip(a - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(a - lam, 0.0, 1.0)

def compute_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr * dx
    a_max = np.max(a)
    exp_a = np.exp(a - a_max)
    w = exp_a / np.sum(exp_a)
    
    obj = (a_max + np.log(np.sum(exp_a))) / alpha * dx
    
    grad = np.zeros(n)
    # Lags 0 to n-1 (indices n-1 to 2n-2 in corr array)
    for L in range(0, n):
        wk = w[n-1+L]
        if wk < 1e-12:
            continue
        if L == 0:
            grad[:] += wk * (1.0 - h[:])
            grad[:] -= wk * h[:]
        else:
            grad[:n-L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n-L]
    
    # Lags -1 to -(n-1) (indices 0 to n-2 in corr array)
    for L in range(1, n):
        wk = w[n-1-L]
        if wk < 1e-12:
            continue
        grad[L:] += wk * (1.0 - h[:n-L])
        grad[:n-L] -= wk * h[L:]
    
    grad *= dx
    return obj, grad

def smooth_projected_grad(h_init, alpha, n_iter, lr, target):
    n = len(h_init)
    h = project_to_polytope(h_init.copy(), target)
    best_h = h.copy()
    best_obj = compute_c5(h)
    
    for it in range(n_iter):
        obj, grad = compute_c5_and_grad(h, alpha)
        h_new = h - lr * grad
        h = project_to_polytope(h_new, target)
        c5 = compute_c5(h)
        if c5 < best_obj:
            best_obj = c5
            best_h = h.copy()
    
    return best_h, best_obj
