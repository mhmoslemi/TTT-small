**New idea: Use the duality gap**

The problem min max_k f_k(h) can be written as:
min_t t subject to f_k(h) ≤ t for all k, 0 ≤ h ≤ 1, sum h = n/2

The Lagrangian is:
L = t + sum_k lambda_k (f_k(h) - t) + ...

At optimality, sum_k lambda_k = 1 and lambda_k ≥ 0, with complementary slackness: lambda_k * (t - f_k(h)) = 0.

This suggests that at the optimum, the distribution of lambda weights tells us which constraints are "tight". If we could compute the optimal lambda, we could solve a simpler problem:
min_h sum_k lambda_k * f_k(h)

This is still non-convex but is a weighted sum instead of a max. We can alternate:
1. Given h, compute the "optimal" lambda (uniform over active constraints)
2. Given lambda, optimize the weighted sum

This is essentially what the alternating LP does, but with a different schedule.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    c_max = np.max(corr)

def gradient_single_shift(h, n, k_shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
    v2 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
    grad[idxs[v1]] += (1.0 - h[idxs[v1] - k_shift])
    grad[idxs[v2]] -= h[idxs[v2] + k_shift]
    grad *= dx
    return grad


def multi_shift_gradient(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_half = n - 1
    top_indices = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    idxs = np.arange(n)
    for ti in top_indices:
        k = int(ti) - n_half
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[idxs[v1]] += (1.0 - h[idxs[v1] - k])
        grad[idxs[v2]] -= h[idxs[v2] + k]
    grad *= dx
    return grad
