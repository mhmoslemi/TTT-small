def project_feasible(h, target_sum):

        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = slsqp_round(best_h, n_use, target_use,
                                  [5000, 20000, 100000], 800, 1e-16)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 12: Final subgradient
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=5000, time_limit=min(15, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
