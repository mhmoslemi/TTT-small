        h_gls2, sc_gls2 = greedy_local_search(best_h, n, dl, rng)
        if sc_gls2 < best_score:
            best_score = sc_gls2
            best_h = h_gls2

    if remaining() > 5:
        dl = time.time() + remaining() * 0.7
        h_ts2, sc_ts2 = targeted_search(best_h, n, dl, rng)
        if sc_ts2 < best_score:
            best_score = sc_ts2
            best_h = h_ts2

    return best_h, best_score, n
