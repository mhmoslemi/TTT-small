3. Solve a least-squares problem: find h that minimizes the sum of (corr_i - C)^2 for the top-k shifts, subject to the constraints
4. This is a quadratic program (QP) which can be solved efficiently with cvxpy

Let me think about this. The correlation at shift s is:
corr(s) = sum_i h[i] * (1 - h[i+s]) = sum_i h[i] - sum_i h[i]*h[i+s]

This is quadratic in h! So minimizing max corr(s) is a QCQP (quadratically constrained QP) or can be formulated as a QP with a parameter.

Using cvxpy, I can formulate:
minimize t
subject to: corr(s) <= t for all s (or top-k s)
           0 <= h[i] <= 1
           sum(h) = target

But corr(s) = sum_i h[i]*(1-h[i+s]) is NOT convex in h (it's -sum_i h[i]*h[i+s] + sum_i h[i], and the quadratic term is not convex).

So it's a non-convex QP. However, if we linearize around the current h, we can use a sequential quadratic programming approach.

Linearization of corr(s) around h0:
corr(s; h) ≈ corr(s; h0) + sum_i [((1-h0[i+s])*δh[i]) + (-h0[i]*δh[i+s])]

where δh = h - h0.

This gives a linear constraint for each shift, so we can solve:
minimize t
subject to: corr(s; h0) + grad_corr(s)·δh <= t for s in top_k
           0 <= h0 + δh <= 1
           sum(δh) = 0

This is a LP! (Linear program with variable t.)

We can solve this with cvxpy or even scipy's linprog.

This is essentially a **SQP** step for the minimax problem. Let me implement this:

    corr = compute_corr(h0)
    c_max = float(np.max(corr))

    top_idx = sorted(top_idx.tolist())

    # Variables: t, δh[0], ..., δh[n-1]
    # Minimize: t
    # Subject to:
    #   For each s in top_idx: sum_i [(1-h0[i+s])*δh[i] - h0[i]*δh[i+s]] <= t - corr[s]
    #   -h0[i] <= δh[i] <= 1-h0[i] for all i
    #   sum(δh) = 0
    #   t >= 0
    
    # This is a LP with n+1 variables and (top_k + 2n + 1) constraints
    # Can solve with scipy.optimize.linprog
    
    # Actually, for n=400 and top_k=20, this is manageable
    # But building the constraint matrix might be slow
    
    # Alternative: use cvxpy
    import cvxpy as cp
    
    delta_h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(delta_h) == 0]
    constraints.append(delta_h >= -h0)
    constraints.append(delta_h <= 1 - h0)
    
    for s_idx in top_idx:
        s = s_idx - (n - 1)
        # corr(s) = sum_i h[i]*(1-h[i+s])
        # Linearize: corr(s) ≈ corr(s;h0) + sum_i [(1-h0[i+s])*dh[i] - h0[i]*dh[i+s]]
        if s >= 0:
            # i ranges from 0 to n-1-s
            a = (1.0 - h0[s:])  # coefficient of delta_h[i] for i in [0, n-s)
            b = -h0[:n-s]       # coefficient of delta_h[i+s] for i in [0, n-s)
            # Linearized corr = corr0 + sum_{i=0}^{n-1-s} a[i]*dh[i] + sum_{i=0}^{n-1-s} b[i]*dh[i+s]
            # = corr0 + sum_j [a[j]*dh[j] (if j < n-s) + b[j-s]*dh[j] (if j >= s)]
            # Let's just build it directly
            grad_s = np.zeros(n)
            grad_s[:n-s] += a  # (1-h0[s]) for i=0..n-1-s → dh[0..n-1-s]
            grad_s[s:] += b    # -h0[0..n-1-s-1] for dh[s..n-1]
        else:

            r = -s
            # i ranges from 0 to n-1-r
            # h[i]*(1-h[i-r]) for i >= r
            # grad wrt h[i]: (1-h[i-r]) for i >= r
            # grad wrt h[i-r]: -h[i] for i >= r, i.e., for j < n-r: -h[j+r]

            grad_s[:n-r] += -h0[r:]          # -h0[i] for i=r..n-1, contributes to dh[i-r]=dh[0..n-1-r]

    problem = cp.Problem(cp.Minimize(t), constraints)

            h_new = project_feasible(h0 + delta_h.value, target)
            return h_new
    except Exception:
        pass
    
    return h0

_g_h_block = None
_g_bstart = 0
_g_bend = 0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_obj_func(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq_func(x):
    return np.sum(x) - _g_target


def block_soft_obj(x):
    h_full = _g_h_block.copy()
    h_full[_g_bstart:_g_bend] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    corr = compute_corr(h_full)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def gradient_soft_max(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    wt = np.exp(shifted)
    wsum = wt.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    wt = wt / wsum
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = wt[s + n - 1]
        if w < 1e-12:
            continue
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            grad[:n + s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:n + s]
    return grad


def gradient_top(h, top_idx):
    n = len(h)
    grad = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - (n - 1)
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        else:
            grad[:n + s] += (1.0 - h[-s:])
            grad[-s:] -= h[:n + s]
    grad /= float(len(top_idx))
    return grad


def sqp_lp_step(h0, n, target, top_k, deadline):
    try:
        import cvxpy as cp
    except ImportError:
        return h0

    if time.time() > deadline:
        return h0

    corr = compute_corr(h0)
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = sorted(top_idx.tolist())

    delta_h = cp.Variable(n)
    t = cp.Variable()

    constraints = [
        cp.sum(delta_h) == 0,
        delta_h >= -h0 + 1e-12,
        delta_h <= 1.0 - h0 - 1e-12
    ]

    for s_idx in top_idx:
        s = s_idx - (n - 1)
        grad_s = np.zeros(n)
        if s >= 0:
            grad_s[:n - s] += (1.0 - h0[s:])
            grad_s[s:] += -h0[:n - s]
        else:
            r = -s
            grad_s[r:] += (1.0 - h0[:n - r])
            grad_s[:n - r] += -h0[r:]
        constraints.append(cp.dot(grad_s, delta_h) <= t - corr[s_idx])

    problem = cp.Problem(cp.Minimize(t), constraints)
    try:
        problem.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
        if delta_h.value is not None:
            h_new = project_feasible(h0 + delta_h.value, target)
            return h_new
    except Exception:
        pass
    return h0


def momentum_opt(h, n, target, alpha, lr, momentum, n_steps, deadline, top_idx=None):
    h_cur = h.copy()
    v = np.zeros(n)
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr_cur = lr
    stall = 0

    for _ in range(n_steps):
        if time.time() > deadline:
            break
        h_look = project_feasible(h_cur + momentum * v, target)
        if top_idx is not None:
            g = gradient_top(h_look, top_idx)
        else:
            g = gradient_soft_max(h_look, alpha)
        v = momentum * v - lr_cur * g
        h_new = project_feasible(h_cur + v, target)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            stall = 0
        else:
            lr_cur *= 0.7
            stall += 1
            if lr_cur < 1e-10:
                lr_cur = lr
                v *= 0.5
            if stall > 40:
                break
    return best_h, best_score


def slsqp_phase(h, n, target, alpha, maxiter, ftol, deadline):
    if time.time() > deadline:
        return h, compute_c5(h)
    _g_alpha = alpha
    _g_target = target
    try:
        res = minimize(soft_obj_func, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq_func}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)
