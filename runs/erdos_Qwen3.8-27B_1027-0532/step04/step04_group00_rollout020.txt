   **Let me try a different angle: Direct QP formulation**

   At a given point h, the active set A is known. If we assume the active set stays the same (which is true near a local optimum), the problem becomes:

   Σh' = n/2
   0 ≤ h' ≤ 1

   However, we can linearize O_k around the current h:
   O_k(h') ≈ O_k(h) + ∇O_k(h) · (h' - h)

   min t
   subject to O_k(h) + ∇O_k(h) · (h' - h) ≤ t for all k∈A
   Σh' = n/2
   0 ≤ h' ≤ 1
   
   This is a linear program! (The objective is linear in t, constraints are linear in h' and t.)
   
   We can solve this with cvxpy! Then use the solution as a new point, and iterate.

   ```python
   def linearization_step(h, n, target, active_lags, n_lines):
       import cvxpy as cp
       dx = 2.0 / n
       
       # Compute current correlations and identify active set
       corr = compute_corr(h) * dx
       max_val = np.max(corr)
       
       # Active set: top n_lines correlations
       sorted_idx = np.argsort(corr)[::-1]
       active_idx = sorted_idx[:n_lines]
       
       # Compute gradients for active lags
       grads = np.zeros((len(active_idx), n))
       for i, idx in enumerate(active_idx):
           k = idx - (n-1)  # actual lag
           # ∂c_k/∂h[j] = (1-h[j-k])*1_{0≤j-k≤n-1} - h[j+k]*1_{0≤j+k≤n-1}
           for j in range(n):
               g = 0.0
               jk = j - k
               if 0 <= jk <= n-1:
                   g += (1 - h[jk])
               jkp = j + k
               if 0 <= jkp <= n-1:
                   g -= h[jkp]
               grads[i, j] = g * dx
       
       # Solve LP:
       # min t
       # s.t. corr[idx] + grads[i,:] @ (h_new - h) <= t  for each active i
       #      sum(h_new) = target
       #      0 <= h_new <= 1
       h_var = cp.Variable(n)
       t_var = cp.Variable()
       
       constraints = [
           cp.sum(h_var) == target,
           h_var >= 0,
           h_var <= 1
       ]
       for i in range(len(active_idx)):
           # Linearization: corr_i + grad_i · (h_new - h) <= t
           linearized = corr[active_idx[i]] + grads[i] @ (h_var - h)
           constraints.append(linearized <= t_var)
       
       prob = cp.Problem(cp.Minimize(t_var), constraints)
       prob.solve(solver=cp.GLOP)  # or CBC, GLPK
       
       if h_var.value is not None and t_var.value is not None:
           h_new = np.clip(h_var.value, 0, 1)
           h_new = project_feasible(h_new, target)
           return h_new, t_var.value
       return h, np.inf
   ```

   ```python
   import numpy as np
   from scipy.optimize import minimize
   import time
   import cvxpy as cp

   def compute_corr(h):
       return np.correlate(h, 1.0 - h, mode='full')

   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       return float(np.max(compute_corr(h))) * dx

   def project_feasible(h, target_sum):
       h = np.clip(h.astype(float), 0.0, 1.0)
       for _ in range(2000):
           diff = target_sum - np.sum(h)
           if abs(diff) < 1e-14:
               break
           if diff > 0:
               mask = h < 1.0 - 1e-15
           else:
               mask = h > 1e-15
           cnt = np.sum(mask)
           if cnt == 0:
               break
           h = h.copy()
           h[mask] += diff / cnt
           h = np.clip(h, 0.0, 1.0)
       return h

   def grad_single_lag(h, k, n):
       """Compute gradient of c_k with respect to h."""
       dx = 2.0 / n
       term1 = np.zeros(n)
       term2 = np.zeros(n)
       if k >= 0:
           term1[k:] = 1.0 - h[:n-k]
           term2[:n-k] = h[k:]
       else:
           m = -k
           term1[:n-m] = 1.0 - h[m:]
           term2[m:] = h[:n-m]
       return (term1 - term2) * dx

   def subgradient_c5(h, n, temp=0.002, n_active=None):
       """Compute a smoothed subgradient of C5."""
       dx = 2.0 / n
       corr = compute_corr(h) * dx
       max_val = float(np.max(corr))
       
       if n_active is not None:
           sorted_idx = np.argsort(corr)[::-1][:n_active]
           active = sorted_idx
       else:
           tol = temp * 2
           active = np.where(corr >= max_val - tol)[0]
       
       # Softmax weights
       diffs = corr[active] - max_val
       w = np.exp(diffs / temp)
       w /= np.sum(w)
       
       grad = np.zeros(n)
       for idx, wi in zip(active, w):
           k = int(idx) - (n - 1)
           grad += wi * grad_single_lag(h, k, n)
       
       return grad, max_val

   def momentum_subgradient(h, n, target, n_steps, lr, momentum, n_active=5, temp=0.002):
       """Subgradient method with momentum."""
       h_cur = h.copy()
       v = np.zeros(n)
       best_h = h_cur.copy()
       best_score = compute_c5(h_cur)
       
       for step in range(n_steps):
           grad, max_val = subgradient_c5(h_cur, n, temp=temp, n_active=n_active)
           v = momentum * v - lr * grad

               best_score = sc

           if sc < compute_c5(h_cur):
               h_cur = h_new
           else:

                   best_h = h_cur.copy()

           if step > 0 and step % 50 == 0:

       dx = 2.0 / n
       corr = compute_corr(h) * dx
       max_val = float(np.max(corr))
       
       # Get top n_lines lags
       sorted_idx = np.argsort(corr)[::-1][:n_lines]
       
       # Compute gradients for these lags
       grads = np.zeros((n_lines, n))
       for i, idx in enumerate(sorted_idx):
           k = int(idx) - (n - 1)
           grads[i] = grad_single_lag(h, k, n)
       
       # Solve LP:
       # min t
       # s.t. corr[idx_i] + grads[i] @ (h_new - h) <= t
       #      sum(h_new) = target
       #      0 <= h_new <= 1
       h_var = cp.Variable(n)
       t_var = cp.Variable()

           cp.sum(h_var) == target,
           h_var >= 0,
           h_var <= 1
       ]
       for i in range(n_lines):
           lin = corr[sorted_idx[i]] + grads[i] @ (h_var - h)
           constraints.append(lin <= t_var)
       
       prob = cp.Problem(cp.Minimize(t_var), constraints)
       try:
           prob.solve(solver=cp.CBC, time_limit=30)
       except:
           try:
               prob.solve(solver=cp.GLPK)
           except:
               try:
                   prob.solve(solver=cp.GLOP)
               except:
                   return h, max_val
       
       if h_var.value is None:
           return h, max_val
       
       h_new = np.clip(h_var.value, 0, 1)
       h_new = project_feasible(h_new, target)
       sc = compute_c5(h_new)
       return h_new, sc

   def kkt_iteration(h, n, target, n_iters, n_lines=8, deadline=None):
       """Iterate KKT linearization steps."""
       h_cur = h.copy()
       best_h = h_cur.copy()
       best_score = compute_c5(h_cur)
       no_improve = 0
       
       for it in range(n_iters):
           if deadline is not None and time.time() > deadline:
               break
           
           h_new, sc = kkt_lp_step(h_cur, n, target, n_lines)
           if sc < best_score:
               best_score = sc
               best_h = h_new.copy()
               no_improve = 0
           else:
               no_improve += 1
           
           # Damped move
           h_try = 0.5 * h_cur + 0.5 * h_new
           h_try = project_feasible(h_try, target)
           sc_try = compute_c5(h_try)
           if sc_try < compute_c5(h_cur):
               h_cur = h_try
           else:
               h_cur = h_new
           
           if no_improve > 10:
               break
       
       return best_h, best_score
   ```

```python
def compute_corr(h):

def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:
            mask = h > 1e-15

            break

def grad_single_lag(h, k, n):
    dx = 2.0 / n
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            term1[k:n] = 1.0 - h[0:n-k]
            term2[0:n-k] = h[k:n]
    else:
        m = -k
        if n - m > 0:
            term1[0:n-m] = 1.0 - h[m:n]
            term2[m:n] = h[0:n-m]
    return (term1 - term2) * dx

        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-9:
                break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def flatness_regularization_step(h, n, target, n_active=5, patience=5):
    """Minimize variance of the active-set gradient to flatten it."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    n_fail = 0
    for it in range(80):
        if n_fail > patience:
            break
        dx = 2.0 / n
        corr = compute_corr(h_cur) * dx
        max_val = float(np.max(corr))
        sorted_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[sorted_idx] - max_val
        w = np.exp(diffs / 0.003)
        w = w / w.sum()
        grad = np.zeros(n)
        for idx, wi in zip(sorted_idx, w):
            k = int(idx) - (n - 1)
            grad += wi * grad_single_lag(h_cur, k, n)
        mask = (h_cur > 0.05) & (h_cur < 0.95)
        if np.sum(mask) < 10:
            mask = (h_cur > 0.01) & (h_cur < 0.99)
        if np.sum(mask) < 5:
            break
        gm = np.mean(grad[mask])
        direction = grad - gm
        direction[mask] = -direction[mask]
        for scale in [0.005, 0.003, 0.001, 0.0005]:
            h_try = project_feasible(h_cur + scale * direction, target)
            sc = compute_c5(h_try)
            if sc < compute_c5(h_cur) - 1e-10:
                h_cur = h_try
                n_fail = 0
                if sc < best_score:
                    best_score = sc
                    best_h = h_cur.copy()
                break
        else:
            n_fail += 1
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_block_full, _g_block_start, _g_block_end
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: KKT LP iteration
    deadline_p1 = start_time + (budget_s - 30.0) * 0.15
    h_kkt, sc_kkt = kkt_iteration(best_h, n, target, n_iters=15, n_lines=10, deadline=deadline_p1)
    if sc_kkt < best_score:
        best_score = sc_kkt
        best_h = h_kkt
    h_ref = slsqp_round(best_h, n, target, [200, 800, 3000], 3000, 1e-16, deadline=deadline_p1)
    sc = compute_c5(h_ref)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 2: Momentum subgradient
    if tleft() > 150:
        deadline_p2 = start_time + (budget_s - 30.0) * 0.30
        h_sg, sc_sg = momentum_subgradient(best_h, n, target, n_steps=600, lr=0.003,
                                            momentum=0.85, n_active=5, temp=0.002,
                                            deadline=deadline_p2)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_ref = slsqp_round(best_h, n, target, [500, 2000, 8000], 3000, 1e-16, deadline=start_time + 30)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: More KKT LP with different active set sizes
    if tleft() > 200:
        deadline_p3 = start_time + (budget_s - 30.0) * 0.45
        for nl in [6, 12, 20]:
            if tleft() < 60:
                break
            h_kkt, sc_kkt = kkt_iteration(best_h, n, target, n_iters=10, n_lines=nl, deadline=deadline_p3)
            if sc_kkt < best_score:
                best_score = sc_kkt
                best_h = h_kkt

    # Phase 4: Flatness regularization
    if tleft() > 150:
        h_flat, sc_flat = flatness_regularization_step(best_h, n, target, n_active=5, patience=5)
        if sc_flat < best_score:
            best_score = sc_flat
            best_h = h_flat
        h_ref = slsqp_round(best_h, n, target, [1000, 5000], 2000, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Worst-lag surgery
    if tleft() > 120:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, max_steps=300, initial_epsilon=0.002)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_ref = slsqp_round(best_h, n, target, [2000, 8000, 20000], 3000, 1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Block descent
    if tleft() > 150:
        deadline_p6 = start_time + (budget_s - 30.0) * 0.65
        for bs in [3, 4, 5]:
            if tleft() < 60:
                break
            h_bd = block_descent(best_h, n, target, bs, 6, 500.0, deadline_p6)
            sc = compute_c5(h_bd)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    # Phase 7: Multi-scale
    if tleft() > 250:
        n_coarse = n // 2
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        h_coarse = kkt_iteration(h_coarse, n_coarse, target_coarse, n_iters=8, n_lines=6)
        h_coarse = slsqp_round(h_coarse, n_coarse, target_coarse, [100, 500, 2000], 2000, 1e-15)
        h_coarse = block_descent(h_coarse, n_coarse, target_coarse, 3, 5, 500.0, start_time + 30)
        h_up = interpolate_to(h_coarse, n, target)
        h_up = slsqp_round(h_up, n, target, [500, 2000, 8000, 20000], 2000, 1e-16)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 8: Random restarts with structured perturbations
    deadline_p8 = start_time + (budget_s - 30.0) * 0.85
    for attempt in range(5):
        if tleft() < 80:
            break
        noise_scale = [0.04, 0.08, 0.12, 0.06, 0.10][attempt]
        h_pert = best_h.copy()
        h_pert += rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref = slsqp_round(h_pert, n, target, [200, 1000, 5000], 2000, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final KKT + polish
    if tleft() > 80:
        h_kkt, sc_kkt = kkt_iteration(best_h, n, target, n_iters=10, n_lines=15)
        if sc_kkt < best_score:
            best_score = sc_kkt
            best_h = h_kkt
        h_flat, sc_flat = flatness_regularization_step(best_h, n, target, n_active=8, patience=3)
        if sc_flat < best_score:
            best_score = sc_flat
            best_h = h_flat

    # Phase 10: Final SLSQP polish
    if tleft() > 40:
        h_final = slsqp_round(best_h, n, target, [5000, 20000, 50000], 2000, 1e-17)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 11: Final subgradient
    if tleft() > 20:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=200, n_active=3,
                                         lr_init=0.001, lr_decay=0.998)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
