def compute_subgradient_fast(h, active_indices):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h

    for k in active_indices:
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += one_minus_h[:n - m]
            grad[:n - m] -= h[m:]
        else:
            mm = -m
            grad[:n - mm] += one_minus_h[mm:]
            grad[mm:] -= h[:n - mm]

    return grad / len(active_indices)

def lsq_smoothing_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    grad = np.zeros(n)
