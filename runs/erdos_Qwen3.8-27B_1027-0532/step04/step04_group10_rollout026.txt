import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        space = 1.0 - h[idx]
        total_space = np.sum(space)
        if total_space < 1e-15:
            return h
        h[idx] += diff * space / total_space
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return obj, grad


def lbfgs_step(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol):
    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f = f + penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                               'gtol': 1e-10, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def slsqp_step(h_start, target, alpha, topk, maxiter, ftol):
    n = len(h_start)

    def obj_func(x):
        f, _ = lse_obj_grad(x, alpha, topk)
        return f

    def obj_jac(x):
        _, g = lse_obj_grad(x, alpha, topk)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_start, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def annealed_search(h, target, rng, n_iter, base_sigma):
    best = h.copy()
    best_score = compute_c5(best)
    sigma = base_sigma
    n = len(h)
    for i in range(n_iter):
        if i % 200 == 199:
            sigma = max(base_sigma * 0.5, sigma * 0.7)
        h_try = best + rng.randn(n) * sigma
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best = h_try
            sigma = min(sigma * 1.15, base_sigma * 3.0)
        else:
            sigma *= 0.97
            sigma = max(sigma, base_sigma * 0.05)
    return best, best_score


def diverse_starts(n, rng):
    inits = []
    x = np.linspace(0, 1, n)
    for frac in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        h = (x < frac).astype(float)
        h = project_feasible(h, n / 2.0)
        inits.append(h)
    for period in [3, 4, 5, 6, 8, 10, 12, 16]:
        h = np.zeros(n)
        for i in range(0, n, period):
            end = min(i + period // 2, n)
            h[i:end] = 1.0
        h = project_feasible(h, n / 2.0)
        inits.append(h)
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        vals = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        h = (vals > 0.5).astype(float)
        h = project_feasible(h, n / 2.0)
        inits.append(h)
    for _ in range(5):
        h = rng.rand(n)
        h = (h > 0.5).astype(float)
        h = project_feasible(h, n / 2.0)
        inits.append(h)
    return inits
