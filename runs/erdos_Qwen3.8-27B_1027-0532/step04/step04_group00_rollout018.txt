    n = len(h)

    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(2 * n - 1)  # we need to accumulate over all (i, d) pairs
    # Actually, grad should be of length n (for h[i])
    grad = np.zeros(n)
    
    # For each lag d, accumulate the gradient contribution
    # d ranges from -(n-1) to n-1, corresponding to idx = d + n - 1
    for d in range(-(n-1), n):
        idx = d + n - 1
        wd = w[idx]
        if wd < 1e-15:
            continue
        if d >= 0:
            # First term: (1-h[i-d]) for i from d to n-1
            grad[d:] += wd * (1.0 - h[:n-d])
            # Second term: -h[i+d] for i from 0 to n-1-d
            grad[:n-d] -= wd * h[d:]
        else:

            grad[:n-dp] += wd * (1.0 - h[dp:])
            # Second term: -h[i-dp] for i from dp to n-1
            grad[dp:] -= wd * h[:n-dp]
    
    grad *= (1.0 / alpha)
