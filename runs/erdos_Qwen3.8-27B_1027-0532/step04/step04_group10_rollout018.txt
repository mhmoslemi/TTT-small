    weights = exp_vals / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = idx[i] - (n - 1)
        if k >= 0:
            lo = max(0, k)
            hi = min(n, n - k)
            shift = np.arange(lo, hi)
            grad[shift + k] += w * one_minus_h[shift]
            grad[shift] -= w * h[shift + k]
        else:
            m = -k
            lo = m
            hi = n
            shift = np.arange(lo, hi)
            grad[shift] += w * one_minus_h[shift + k]
            grad[shift + k] -= w * h[shift]
    return obj, grad


def lbfgs_optimize(h_start, target, alpha, topk, maxiter, bounds, ftol):
    n = len(h_start)
    penalty = 10000.0

    def obj_func(x):
        f, g = lse_grad(x, alpha, topk)
        s = np.sum(x) - target
        f = f + penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g

    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def swap_search(h, target, n_active=20, max_rounds=5000, delta_frac=1.0, rng=None):
    n = len(h)
    corr = compute_corr(h)
    corr_max = np.max(corr)
    best_h = h.copy()
    best_corr_max = corr_max
    best_score = corr_max * (2.0 / n)
    
    for round_id in range(max_rounds):
        G = compute_G(best_h)
        if n_active < 2 * n - 1:
            active = np.argpartition(corr, -n_active)[-n_active:]
        else:
            active = np.arange(2 * n - 1)
        G_active = G[:, active]
        corr_active = corr[active]
        
        high_set = np.where(best_h > 0.6)[0]
        low_set = np.where(best_h < 0.4)[0]
        
        if len(high_set) == 0 or len(low_set) == 0:
            mid_high = np.where(best_h > 0.55)[0]
            mid_low = np.where(best_h < 0.45)[0]
            high_set = mid_high
            low_set = mid_low
        if len(high_set) == 0 or len(low_set) == 0:
            break
        
        Gi = G_active[high_set]
        Gj = G_active[low_set]
        
        delta_matrix = Gj[:, None, :] - Gi[None, :, :]
        max_delta = delta_matrix.max(axis=2)
        
        valid_mask = max_delta < -1e-10
        valid_indices = np.where(valid_mask)
        
        if len(valid_indices[0]) == 0:
            break
        
        if rng is not None and len(valid_indices[0]) > 1:
            choice = rng.randint(len(valid_indices[0]))
        else:
            choice = np.argmax(-max_delta[valid_indices])
        
        ji_idx = valid_indices[0][choice]
        ii_idx = valid_indices[1][choice]
        j_pos = low_set[ji_idx]
        i_pos = high_set[ii_idx]
        
        delta = delta_frac * min(best_h[i_pos], 1.0 - best_h[j_pos])
        if delta < 1e-10:
            break
        
        best_h[i_pos] -= delta
        best_h[j_pos] += delta
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, target)
        
        corr = compute_corr(best_h)
        new_corr_max = np.max(corr)
        
        if new_corr_max < best_corr_max:
            best_corr_max = new_corr_max
            best_score = new_corr_max * (2.0 / n)
        else:
            break
    
    return best_h, best_score


def anneal_swap_search(h, target, n_active=15, n_iters=50000, T_start=0.05, T_end=0.001, rng=None):
    n = len(h)
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_score = np.max(corr) * (2.0 / n)
    curr_score = best_score
    curr_h = best_h.copy()
    
    dx = 2.0 / n
    
    for it in range(n_iters):
        T = T_start * (T_end / T_start) ** (it / max(n_iters - 1, 1))
        
        i_pos = rng.randint(n)
        j_pos = rng.randint(n)
        if i_pos == j_pos:
            continue
        
        if curr_h[i_pos] < curr_h[j_pos]:
            i_pos, j_pos = j_pos, i_pos
        
        delta = min(curr_h[i_pos], 1.0 - curr_h[j_pos])
        if delta < 1e-10:
            continue
        delta = delta * (0.5 + 0.5 * rng.random())
        
        trial = curr_h.copy()
        trial[i_pos] -= delta
        trial[j_pos] += delta
        trial = project_feasible(trial, target)
        
        corr_trial = compute_corr(trial)
        new_score = np.max(corr_trial) * dx
        
        delta_obj = new_score - curr_score
        if delta_obj <= 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            curr_h = trial
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = trial.copy()
    
    return best_h, best_score


def multi_res_optimize(h, target, n_new, alpha, topk, maxiter, bounds_new):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    h_interp = project_feasible(h_interp, target)
    h_new = lbfgs_optimize(h_interp, target, alpha, topk, maxiter, bounds_new, 1e-14)
    score = compute_c5(h_new)
    return h_new, score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 30.0 - (time.time() - start_time)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Quick L-BFGS warm-up
    for alpha in [200, 500, 1000]:
        if time_left() < 50:
            break
        h_start = best_h + rng.randn(n) * 0.003
        h_start = project_feasible(h_start, target)
        h_new = lbfgs_optimize(h_start, target, alpha, 30, 300, bounds, 1e-15)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: 2-swap local search (main new contribution)
    if time_left() > 200:
        h_swapped, score_swap = swap_search(best_h, target, n_active=20, max_rounds=10000, delta_frac=1.0, rng=rng)
        if score_swap < best_score:
            best_score = score_swap
            best_h = h_swapped
    
    # Phase 3: Gradient refinement after swaps
    if time_left() > 100:
        for alpha in [200, 500, 1000, 2000]:
            if time_left() < 40:
                break
            h_start = best_h + rng.randn(n) * 0.002
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_optimize(h_start, target, alpha, 20, 200, bounds, 1e-16)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 4: More swap rounds after gradient refinement
    if time_left() > 150:
        h_swapped, score_swap = swap_search(best_h, target, n_active=15, max_rounds=8000, delta_frac=1.0, rng=rng)
        if score_swap < best_score:
            best_score = score_swap
            best_h = h_swapped

    # Phase 5: Simulated annealing with random swaps
    if time_left() > 200:
        h_anneal, score_ann = anneal_swap_search(best_h, target, n_active=15, n_iters=30000, T_start=0.03, T_end=0.0005, rng=rng)
        if score_ann < best_score:
            best_score = score_ann
            best_h = h_anneal

    # Phase 6: Final L-BFGS polish
    if time_left() > 80:
        for alpha in [500, 1000, 2000, 5000]:
            if time_left() < 30:
                break
            h_start = best_h + rng.randn(n) * 0.001
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_optimize(h_start, target, alpha, 15, 150, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 7: Final swap polish
    if time_left() > 60:
        h_swapped, score_swap = swap_search(best_h, target, n_active=10, max_rounds=5000, delta_frac=1.0, rng=rng)
        if score_swap < best_score:
            best_score = score_swap
            best_h = h_swapped

    # Phase 8: Multi-resolution attempt
    if time_left() > 150:
        for n_try in [500, 600]:
            if time_left() < 80:
                break
            target_try = n_try / 2.0
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, target_try)
            bounds_try = [(0.0, 1.0)] * n_try
            for alpha in [200, 500]:
                if time_left() < 40:
                    break
                h_new = lbfgs_optimize(h_interp, target_try, alpha, 25, 200, bounds_try, 1e-14)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    x_back = np.linspace(0, 2, n)
                    best_h = np.interp(x_back, x_new, h_new)
                    best_h = project_feasible(best_h, target)
                    best_score = compute_c5(best_h)
                    h_interp = np.interp(x_new, x_back, best_h)
                    h_interp = project_feasible(h_interp, target_try)

    # Phase 9: Coordinate descent polish
    if time_left() > 40:
        for rd in range(2):
            if time_left() < 15:
                break
            for coord in range(n):
                if time_left() < 5:
                    break
                orig = best_h[coord]
                for d in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                    h_try = best_h.copy()
                    h_try[coord] = np.clip(orig + d, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-9:
                        best_score = sc
                        best_h = h_try
                        break

    # Phase 10: Final gradient step
    if time_left() > 20:
        h_start = best_h.copy()
        h_new = lbfgs_optimize(h_start, target, 2000, 10, 100, bounds, 1e-17)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
