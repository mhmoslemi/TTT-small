```python
h_bm = h[np.clip(b_minus, 0, n-1)]
```

```python
    best_h = h.copy()

            break
```

```python
        h_nest = h_prev + ((it+1)/it) * (h - h_prev) if it > 0 else h
        h_nest = np.clip(h_nest, 0, 1)
        h_nest = project_feasible(h_nest, target)
        
        # Compute gradient at h_nest
        grad = smooth_gradient(h_nest, alpha)
        
        # Update
        h_new = h_nest - eta * grad
        h_new = project_feasible(h_new, target)
        
        obj = softmax_smooth(get_corr(h_new), alpha)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
        
        # Adaptive step size
        if not np.isfinite(obj):
            eta *= 0.5
            h_new = h  # revert
        else:
            h_prev = h
            h = h_new
    
    return best_h, best_obj
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(float)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            o2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[o2[:extra]] = 0
        else:
            z2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[z2[:extra]] = 1
    return h_r

def update_corr_for_swap(corr, h, a, b, n):
    lag_range = np.arange(-(n-1), n)
    d = np.zeros(2*n-1)
    a_plus = a + lag_range
    a_minus = a - lag_range
    b_minus = b - lag_range
    b_plus = b + lag_range
    va_p = (a_plus >= 0) & (a_plus < n)
    va_m = (a_minus >= 0) & (a_minus < n)
    vb_m = (b_minus >= 0) & (b_minus < n)
    vb_p = (b_plus >= 0) & (b_plus < n)
    d[va_p] += h[np.clip(a_plus, 0, n-1)][va_p]
    d[va_m] += h[np.clip(a_minus, 0, n-1)][va_m]
    d[vb_m] -= h[np.clip(b_minus, 0, n-1)][vb_m]
    d[vb_p] -= h[np.clip(b_plus, 0, n-1)][vb_p]
    corr += d
    return corr

def remax_ils(h_init, target_sum, rng, time_limit, start_time, top_k=5, kick_size=4):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = float(np.max(get_corr(h))) * (2.0/n)
    corr = get_corr(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    lag_range = np.arange(-(n-1), n)
    stall = 0
    max_stall = 80
    it = 0
    while time.time() - start_time < time_limit:
        it += 1
        worst_idx = np.argsort(corr)[::-1][:top_k]
        lags = worst_idx - (n-1)
        curr_max = corr.max()
        best_score = np.inf
        best_a = -1
        best_b = -1
        for a in ones:
            a_p = a + lags
            a_m = a - lags
            vp = (a_p >= 0) & (a_p < n)

            ca = np.where(vp, h[np.clip(a_p, 0, n-1)], 0.0) + \
                 np.where(vm, h[np.clip(a_m, 0, n-1)], 0.0) - 1.0
            bp = zeros[:, None] + lags[None, :]
            bm = zeros[:, None] - lags[None, :]
            vbp = (bp >= 0) & (bp < n)

def smooth_obj(h, alpha):
    corr = get_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_vals / total
    grad = np.zeros(n)
    lag_range = np.arange(-(n-1), n)
    for k_idx in range(2*n-1):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        lag = lag_range[k_idx]
        if lag >= 0:
            if lag == 0:
                grad += w * (1.0 - h)
                grad -= w * h
            else:
                grad[lag:] += w * (1.0 - h[:-lag])
                grad[:n-lag] -= w * h[lag:]
        else:
            ll = -lag
            if ll == 0:
                grad += w * (1.0 - h)
                grad -= w * h
            else:
                grad[:n-ll] += w * (1.0 - h[ll:])
                grad[ll:] -= w * h[:-ll]
    return grad

def projected_grad_descent(h_init, target_sum, alpha, eta, max_iters, time_limit, start_time, momentum=0.9):
    n = len(h_init)
    h = project_feasible(h_init.copy(), target_sum)
    h_prev = h.copy()
    best_h = h.copy()
    best_obj = smooth_obj(h, alpha)
    v = np.zeros(n)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        grad = smooth_grad(h, alpha)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        obj = smooth_obj(h_new, alpha)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h_new.copy()
        if not np.isfinite(obj):
            eta *= 0.5
            v *= 0.5
            continue
        h_prev = h
        h = h_new
        if it % 100 == 0:
            eta *= 0.995
    return best_h, best_obj

def slsqp_with_grad(h_start, target_sum, alpha, maxiter, time_limit, start_time):
    n = len(h_start)
    def obj_func(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    def obj_grad(h_in):
        return smooth_grad(h_in, alpha)
    def cons_f(h_in):
        return np.sum(h_in) - target_sum
    def cons_j(h_in):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    try:
        res = minimize(obj_func, h_cur, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target_sum)
        sc = compute_c5(h_new)
        if sc < curr_score:
            return h_new, sc
    except Exception:
        pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary ILS on initial resolution
    h_bin = project_binary(h0, target0)
    c5_bin_init = compute_c5(h_bin)
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin.copy()

    if tleft() > 100:
        allowed = min(300.0, tleft() * 0.35)
        h_ils, c5_ils = remax_ils(h_bin, target0, rng, time_limit=allowed,
                                   start_time=start_time, top_k=5, kick_size=4)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()

    # Phase 2: Second ILS round with different params
    if tleft() > 80:
        h_bin2 = project_binary(best_h, target0)
        allowed2 = min(200.0, tleft() * 0.25)
        h_ils2, c5_ils2 = remax_ils(h_bin2, target0, rng, time_limit=allowed2,
                                     start_time=start_time, top_k=8, kick_size=3)
        if c5_ils2 < best_c5:
            best_c5 = c5_ils2
            best_h = h_ils2.copy()

    # Phase 3: Continuous optimization with projected gradient
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 100:
        h_cont_init = project_feasible(best_h.copy(), target_use)
        alphas_seq = [50, 150, 400, 1000]
        h_cur = h_cont_init
        for alpha in alphas_seq:
            if tleft() < 30:
                break
            eta_init = 2.0 / (alpha * n_use) * 10
            h_pg, obj_pg = projected_grad_descent(h_cur, target_use, alpha,
                                                   eta_init, 3000,
                                                   time_limit=min(40.0, tleft() - 10),
                                                   start_time=start_time)
            c5_pg = compute_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
            h_cur = h_pg

    # Phase 4: SLSQP refinement with analytical gradient
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slq, sc_slq = slsqp_with_grad(best_h, target_use, 500, 300,
                                         time_limit=min(50.0, tleft() - 10),
                                         start_time=start_time)
        if sc_slq < best_c5:
            best_c5 = sc_slq
            best_h = h_slq

    # Phase 5: Higher resolution
    for mult in [2, 3]:
        n_try = n0 * mult
        if n_try > 3000:
            break
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(100.0, tleft() * 0.2)
        h_hires, c5_hires = remax_ils(h_bin_try, target_try, rng,
                                       time_limit=allowed, start_time=start_time,
                                       top_k=5, kick_size=3)
        if c5_hires < best_c5:
            best_c5 = c5_hires
            best_h = h_hires.copy()
        # Continuous refinement at higher resolution
        if tleft() > 40 and len(best_h) == n_try:
            n_cu = len(best_h)
            t_cu = n_cu / 2.0
            h_pg2, _ = projected_grad_descent(best_h, t_cu, 200,
                                               2.0/(200*n_cu)*10, 2000,
                                               time_limit=min(30.0, tleft()-10),
                                               start_time=start_time)
            c5_pg2 = compute_c5(h_pg2)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2

    # Phase 6: Random restarts at base resolution
    if tleft() > 60:
        for rs in range(3):
            if tleft() < 30:
                break
            h_rand = np.zeros(n0)
            perm = rng.permutation(n0)
            h_rand[perm[:n0//2]] = 1.0
            allowed_rs = min(50.0, tleft() * 0.2)
            h_rs, c5_rs = remax_ils(h_rand, target0, rng,
                                     time_limit=allowed_rs, start_time=start_time,
                                     top_k=5, kick_size=4)
            if c5_rs < best_c5:
                best_c5 = c5_rs
                best_h = h_rs.copy()

    # Phase 7: Final polish - binary ILS on best
    if tleft() > 30:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_bin_fin = project_binary(best_h, t_fin)
        h_fin, c5_fin = remax_ils(h_bin_fin, t_fin, rng,
                                   time_limit=min(20.0, tleft()-5),
                                   start_time=start_time, top_k=10, kick_size=2)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 8: Final SLSQP
    if tleft() > 15:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_slq2, sc2 = slsqp_with_grad(best_h, t_fin, 1000, 200,
                                       time_limit=max(5.0, tleft()-3),
                                       start_time=start_time)
        if sc2 < best_c5:
            best_c5 = sc2
            best_h = h_slq2

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_c5 + 1e-10:
        h_final = best_h
        final_score = best_c5
    return h_final, final_score, n_final
```
