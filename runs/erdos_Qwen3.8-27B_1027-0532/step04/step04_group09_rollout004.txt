            h_pert, sc_pert = perturb_reoptimize(best_h, target, rng, n_pert, 0.2, 5000.0, deadline_p2b)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert

    n_hi = 800
    target_hi = n_hi / 2.0
    h_interp = interpolate_to(best_h, n_hi, target_hi)
    sc_interp = compute_c5(h_interp)

    if sc_interp < best_score:
        best_score = sc_interp
        best_h = h_interp

    deadline_p4 = start_time + total_budget * 0.72
    h_sg_hi, sc = reweight_subgrad(best_h, target_hi, 20000, 0.02, 1000.0, 100000.0, 0.93, deadline_p4)
    if sc < best_score:
        best_score = sc
        best_h = h_sg_hi

    if time.time() < deadline_final - 200:
        deadline_p5 = start_time + total_budget * 0.88
        alpha_p5 = [3000, 10000, 30000, 80000]
        h_ref_hi, sc = slsqp_refine(best_h, target_hi, alpha_p5, 500, 1e-15, deadline_p5)
        if sc < best_score:
            best_score = sc
            best_h = h_ref_hi

    if time.time() < deadline_final - 30:
        h_sg_final, sc = reweight_subgrad(best_h, target_hi, 5000, 0.01, 50000.0, 300000.0, 0.95, deadline_final - 10)
        if sc < best_score:
            best_score = sc
            best_h = h_sg_final

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
