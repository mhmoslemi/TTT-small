def get_corr(h):

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

    order = np.argsort(h[::-1])[::-1]

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_fun_target = target
    cons_fun_n = n
    
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n))
    
    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target
    
    def cons_j(h_in):
        return np.ones(cons_fun_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):

    cons_fun_n = n
    dx = 2.0 / n

    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target

    def cons_j(h_in):
        return np.ones(cons_fun_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:

            break

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})

                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score - 1e-15:
                    curr_score = sc
                    h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

    if tleft() < 80:
        n_final = len(best_h)
        h_final = project_feasible(best_h, n_final / 2.0)
        return h_final, compute_c5(h_final), n_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    n_restarts = min(6, int(max(2, tleft() / 80)))
    for ri in range(n_restarts):
        if tleft() < 60:
            break
        noise_level = 0.015 + 0.01 * ri
        h_pert = best_h + rng.randn(n_use) * noise_level
        h_pert = project_feasible(h_pert, target_use)
        alphas_restart = [100, 400, 1500, 5000]
        h_ref2, sc2 = slsqp_refine(h_pert, n_use, target_use,
                                   alphas_restart, 250, 1e-14,
                                   time_limit=min(80.0, tleft() * 0.25), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    if tleft() < 60:
        n_final = len(best_h)
        h_final = project_feasible(best_h, n_final / 2.0)
        return h_final, compute_c5(h_final), n_final

    n_alt_list = [900, 1100, 1200, 800]
    for n_alt in n_alt_list:
        if tleft() < 60:
            break
        target_alt = n_alt / 2.0
        h_alt = interpolate_to(best_h, n_alt, target_alt)
        h_alt_bin = project_binary(h_alt, target_alt)
        allowed = min(60.0, tleft() * 0.3)
        h_sa_alt, c5_alt = binary_sa(h_alt_bin, target_alt, 80000, 0.012, 0.00005,
                                     rng, time_limit=allowed, start_time=start_time)
        alphas_alt = [50, 200, 800, 3000]
        h_ref_alt, sc_alt = slsqp_refine(h_sa_alt, n_alt, target_alt,
                                         alphas_alt, 200, 1e-14,
                                         time_limit=min(50.0, tleft() * 0.25), start_time=start_time)
        if sc_alt < best_score:
            best_score = sc_alt
            best_h = h_ref_alt.copy()

    if tleft() < 30:
        n_final = len(best_h)
        h_final = project_feasible(best_h, n_final / 2.0)
        return h_final, compute_c5(h_final), n_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    alphas_fine = [500, 2000, 8000, 20000]
    h_ref_final, sc_final = slsqp_refine(best_h, n_use, target_use,
                                         alphas_fine, 300, 1e-15,
                                         time_limit=max(10.0, tleft() - 10), start_time=start_time)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h_ref_final.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
