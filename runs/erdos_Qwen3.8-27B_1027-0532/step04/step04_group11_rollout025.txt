import numpy as np
from scipy.optimize import minimize
import time

_g_n = 400
_g_target = 200.0
_g_alpha = 500.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    return float(np.max(corr)) * dx, corr


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def two_interval_h(n, L1):
    L2 = n // 2 - L1
    if L2 < 0 or L1 < 0:
        return None
    h = np.zeros(n)
    h[:L1] = 1.0
    if L2 > 0:
        h[n - L2:] = 1.0
    return h


def graded_h(n, L1, transition):
    h = np.zeros(n)
    L2 = n // 2 - L1
    if L2 < 0 or L1 < 0:
        return None
    t = min(transition, L1)
    if L1 - t > 0:
        h[:L1 - t] = 1.0
    if t > 0:
        h[L1 - t:L1] = np.linspace(1.0, 0.0, t)
    if L2 > 0:
        h[n - L2:] = 1.0
        t2 = min(transition, L2)
        if t2 > 0:
            h[n - L2:n - L2 + t2] = np.linspace(0.0, 1.0, t2)
    return project_feasible(h, n // 2.0)


def periodic_h(n, period, duty):
    pattern = np.zeros(period)
    pattern[:duty] = 1.0
    h = np.tile(pattern, n // period + 1)[:n].astype(float)
    h = project_feasible(h, n // 2.0)
    return h


def lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def con_eq(x):
    return np.sum(x) - _g_target


def con_jac(x):
    return np.ones(_g_n)


def block_con_eq(x):
    return np.sum(x) - _g_block_target


def block_con_jac(x):
    return np.ones(_g_e - _g_s)


def block_obj(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur - 1e-17:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_target, _g_alpha, _g_block_target
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        block_order = np.random.permutation(n_blocks)
        for bi in block_order:
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            if e > n:
                continue
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            rest_sum = target - np.sum(h_cur[s:e])
            _g_block_target = rest_sum
            if _g_block_target < -1e-10 or _g_block_target > block_size + 1e-10:
                continue
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 500, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur - 1e-17:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur


def binary_local_search(h, n, target, n_swaps, t_start, t_end, deadline, rng):
    h_cur = h.copy()
    h_cur[h_cur > 0.5] = 1.0
    h_cur[h_cur <= 0.5] = 0.0
    h_cur = project_feasible(h_cur, target)
    cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur
    n_iter = n_swaps
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / max(t_start, 1e-15)) ** (it / max(n_iter, 1))
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        ii = ones[i]
        jj = zeros[j]
        h_new = h_cur.copy()
        h_new[ii] = 0.0
        h_new[jj] = 1.0
        sc_new = compute_c5(h_new)
        delta = sc_new - cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-15)):
            h_cur = h_new
            cur = sc_new
            if cur < best_sc:
                best_sc = cur
                best_h = h_cur.copy()
    return best_h, best_sc


def continuous_annealing(h, n, target, n_iter, t_start, t_end, deadline, rng):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_sc = cur
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / max(t_start, 1e-15)) ** (it / max(n_iter, 1))
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            j = (j + 1) % n
        delta1 = (rng.rand() - 0.5) * 0.4
        delta2 = -delta1
        h_new = h_cur.copy()
        h_new[i] = np.clip(h_new[i] + delta1, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] + delta2, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        delta_sc = sc_new - cur
        if delta_sc < 0 or rng.rand() < np.exp(-delta_sc / max(t, 1e-15)):
            h_cur = h_new
            cur = sc_new
            if cur < best_sc:
                best_sc = cur
                best_h = h_cur.copy()
    return best_h, best_sc


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old + 1)
    x_old = x_old[:-1] + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new + 1)
    x_new = x_new[:-1] + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new // 2.0)
    return h_new
