def de_coarse_search(n, target, pop_size, max_generations, rng, deadline=None):
    bounds = [(0.0, 1.0)] * n
    # Initialize population
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    
    # Sort by fitness
    population.sort(key=lambda h: compute_c5(h))
    best = population[0].copy()
    best_score = compute_c5(best)
    
    F = 0.8  # scale factor
    CR = 0.9  # crossover rate
    
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            # Select three distinct random individuals
            idxs = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            
            # Mutation
            mutant = population[r1] + F * (population[r2] - population[r3])
            
            # Crossover
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            
            if trial_score <= compute_c5(population[i]):
                population[i] = trial
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        
        population.sort(key=lambda h: compute_c5(h))
    
    return best, best_score

            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]

        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)

            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]

        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)

        else:

def worst_lag_surgery(h, n, target, max_steps=200, initial_epsilon=0.001, deadline=None):
    h_cur = h.copy()

    dx = 2.0 / n
    eps = initial_epsilon

        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        j_pos = np.arange(0, n - k_star)
        grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
        j_neg = np.arange(k_star, n)
        grad[j_neg] -= h_cur[j_neg - k_star]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:

            cur_score = sc_new
            eps = min(eps * 1.05, 0.01)
        else:
            eps *= 0.7
            if eps < 1e-7:
                break
    return h_cur, cur_score

def de_search(n, target, pop_size, max_generations, rng, deadline=None, F=0.8, CR=0.9):
    """Differential evolution on small grid for global search."""
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    
    # Evaluate and sort
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            # DE/rand/1
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            
            # Binary crossover
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        
        # Sort
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    
    return best, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

_g_block_full = None

_g_block_end = 0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = compute_corr(x)

    shifted = _g_alpha * (corr - c_max)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target_sum - np.sum(h)

        else:
            mask = h > 0.0

def de_search(n, target, pop_size, max_generations, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 3: Iterative reweighted (main optimization)
    if tleft() > 200:
        deadline_p3 = start_time + 400
        h_irew = slsqp_weighted_round(best_h, n, target,
                                      n_outer=25, beta_init=100, beta_growth=1.3,
                                      inner_grad_iters=100, lr=0.001,
                                      deadline=deadline_p3)
        sc = compute_c5(h_irew)
        if sc < best_score:
            best_score = sc
            best_h = h_irew

    # Phase 4: Subgradient polish + SLSQP
    if tleft() > 150:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=500, n_active=3,
                                         lr_init=0.003, lr_decay=0.997,
                                         deadline=start_time + 500)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        deadline_p4 = start_time + 550
        h_ref = slsqp_round(best_h, n, target, [500, 2000, 8000], 3000, 1e-16, deadline=deadline_p4)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Momentum weighted gradient descent
    if tleft() > 150:
        deadline_p5 = start_time + 650
        h_mom, sc_mom = momentum_wgd(best_h, n, target, n_steps=800, beta=500,
                                     lr=0.0008, moment=0.85, deadline=deadline_p5)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom
        h_ref = slsqp_round(best_h, n, target, [2000, 8000], 2000, 1e-16,
                            deadline=start_time + 700)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Block descent
    if tleft() > 120:
        for bs in [2, 3]:
            if tleft() < 50:
                break
            deadline_bd = start_time + (budget_s - 100) * 0.7
            h_bd = block_descent(best_h, n, target, bs, 6, 500.0, deadline=deadline_bd)
            sc = compute_c5(h_bd)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    # Phase 7: Worst-lag surgery
    if tleft() > 80:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, max_steps=300,
                                            initial_epsilon=0.002,
                                            deadline=start_time + budget_s - 100)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_ref = slsqp_round(best_h, n, target, [5000, 20000], 2500, 1e-17,
                            deadline=start_time + budget_s - 70)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Second iterative reweighted pass (sharper)
    if tleft() > 120:
        deadline_p8 = start_time + budget_s - 80
        h_irew2 = slsqp_weighted_round(best_h, n, target,
                                       n_outer=20, beta_init=500, beta_growth=1.5,
                                       inner_grad_iters=80, lr=0.0005,
                                       deadline=deadline_p8)
        sc = compute_c5(h_irew2)
        if sc < best_score:
            best_score = sc
            best_h = h_irew2

    # Phase 9: Final SLSQP polish with very high alpha
    if tleft() > 40:
        h_final = slsqp_round(best_h, n, target, [10000, 50000], 3000, 1e-17,
                              deadline=start_time + budget_s - 20)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 10: Final subgradient micro-polish
    if tleft() > 15:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=300, n_active=5,
                                         lr_init=0.001, lr_decay=0.998,
                                         deadline=start_time + budget_s - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
