        # Focused move: target the bottleneck
        if rng.random() < 0.6:
            max_k = int(np.argmax(corr)) - (n - 1)
            i_pool = [i for i in ones if 0 <= i + max_k < n and h[i + max_k] == 0]
            j_pool = [j for j in zeros if 0 <= j + max_k < n]
            if not i_pool:
                i_pool = list(ones)
            if not j_pool:
                j_pool = list(zeros)
            i = i_pool[rng.randint(len(i_pool))]
            j = j_pool[rng.randint(len(j_pool))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        corr_new = compute_corr(h)
        new_max = float(corr_new.max())
        delta = new_max - current_max
        
        if delta <= 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-15))):
            corr = corr_new
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1
            h[j] = 0
            stall += 1
    
    return best_h, best_max * dx

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

        return budget_s - 20.0 - (time.time() - start_time)
    
    # Start from initial values

    target_init = int(n_init / 2.0)
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    best_n = n_init

    h_bin = project_binary(h_init, target_init)
