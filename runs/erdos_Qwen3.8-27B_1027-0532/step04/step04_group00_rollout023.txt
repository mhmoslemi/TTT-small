**Approach: Alternating Minimization with "Block Matching"**

The key insight is that a_k = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k]) couples adjacent elements. We can think of this as a chain of interactions.

For a given h, the "worst" lag k* has the largest a_k*. To reduce a_k*, we want to modify h to decrease the products h[j]*(1-h[j+k*]).

One effective strategy: for the worst lag k*, try to make h[j] and h[j+k*] "agree" (both 0 or both 1) for as many j as possible. This is like a 2-SAT / Ising problem on a chain.

Specifically, for lag k*, we want to minimize sum h[j]*(1-h[j+k*]).
This is equivalent to maximizing sum h[j]*h[j+k*] + sum h[j] - sum h[j+k*] ... wait:
h[j]*(1-h[j+k*]) = h[j] - h[j]*h[j+k*]
sum_j h[j]*(1-h[j+k*]) = sum_j h[j] - sum_j h[j]*h[j+k*]

To minimize this, we want to maximize sum_j h[j]*h[j+k*] (while keeping sum_j h[j] roughly constant).

This is like a maximum correlation problem between h[0:n-k*] and h[k*:n]. If we could make h[0:n-k*] = h[k*:n] (perfectly correlated), we'd get a_{k*} = sum h[j] - sum h[j]^2 ≤ sum h[j] - (sum h[j])^2/(n-k*) by Cauchy-Schwarz.

This suggests a "copying" strategy: try to make h periodic with period k* (or a divisor of k*). But this conflicts with other lags.

            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def swap_descent(h, n, target, n_passes=5, pairs_per_pass=2000, seed=42, deadline=None):
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for p in range(n_passes):
        if deadline is not None and time.time() > deadline:
            break
        C = precompute_C(h_cur)
        over = compute_over(h_cur)
        improved = 0
        
        for _ in range(pairs_per_pass):
            if deadline is not None and time.time() > deadline:
                break
            j = rng.randint(n)
            l = rng.randint(n)
            if j == l:
                continue
            
            B = C[j] - C[l]
            delta_min = max(-h_cur[j], h_cur[l] - 1.0)
            delta_max = min(1.0 - h_cur[j], h_cur[l])
            
            if delta_min >= delta_max - 1e-12:
                continue
            
            # Quick check: can this swap help?
            # Estimate best possible improvement
            pos_mask = B > 1e-10
            neg_mask = B < -1e-10
            
            if not np.any(pos_mask) or not np.any(neg_mask):
                continue
            
            # Find optimal delta via binary search
            lo, hi = delta_min, delta_max
            for _ in range(30):
                mid = (lo + hi) / 2.0
                f_inc = float(np.max(over[pos_mask] + mid * B[pos_mask]))
                f_dec = float(np.max(over[neg_mask] + mid * B[neg_mask]))
                if f_inc > f_dec:
                    lo = mid
                else:
                    hi = mid
            
            candidates = [delta_min, delta_max, (lo + hi) / 2.0]
            best_delta = 0.0
            best_val = float(np.max(over))
            for d in candidates:
                d = float(np.clip(d, delta_min, delta_max))
                v = float(np.max(over + d * B))
                if v < best_val - 1e-12:
                    best_val = v
                    best_delta = d
            
            if abs(best_delta) > 1e-10:
                h_cur[j] += best_delta
                h_cur[l] -= best_delta
                h_cur = np.clip(h_cur, 0.0, 1.0)
                over = over + best_delta * B
                C = precompute_C(h_cur)
                new_score = float(np.max(over)) * dx
                if new_score < cur_score:
                    cur_score = new_score
                    improved += 1
                else:
                    # Revert
                    h_cur[j] -= best_delta
                    h_cur[l] += best_delta
                    over = over - best_delta * B
                    C = precompute_C(h_cur)
        
        if improved == 0:
            break
    
    return h_cur, cur_score


def block_flip(h, n, target):
    """Try flipping blocks of h to 1-h"""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    dx = 2.0 / n
    
    lengths = [5, 10, 20, 40, 50, 100]
    for blen in lengths:
        n_starts = n - blen
        for start in range(0, n_starts, max(1, n_starts // 50)):
            h_test = h_cur.copy()
            h_test[start:start+blen] = 1.0 - h_test[start:start+blen]
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < best_score - 1e-10:
                best_score = sc
                best_h = h_test.copy()
    
    return best_h, best_score


def parametric_init(n, target, mode='bump'):
    """Generate a parametric initial solution"""
    x = np.linspace(0, 2, n)
    if mode == 'bump':
        h = 0.5 + 0.5 * np.sin(np.pi * x)
    elif mode == 'dip':
        h = 0.5 - 0.5 * np.sin(np.pi * x)
    elif mode == 'double_bump':
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    elif mode == 'triple':
        h = 0.5 + 0.3 * np.sin(3 * np.pi * x)
    elif mode == 'cos':
        h = 0.5 + 0.4 * np.cos(np.pi * x)
    h = project_feasible(h, target)
    return h
