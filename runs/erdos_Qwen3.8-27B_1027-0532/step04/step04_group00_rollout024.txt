    n = len(h)
    corr = compute_corr_valid(h)
    c_max = np.max(corr)

    obj = c_max + np.log(sum_exp) / alpha + mu * (np.sum(h) - target_sum) ** 2

    for k in range(n):
        w = weights[k]
        if w < 1e-12:
            continue
        # d(corr_k)/d(h_j) for j = 0..n-1-k: (1 - h[j+k])

        grad[js] += w * (1.0 - h[js + k])

        js2 = np.arange(k, n)
        grad[js2] -= w * h[js2 - k]
    
    # Penalty gradient
    grad += 2.0 * mu * (np.sum(h) - target_sum)
    
    return obj, grad

    """Run L-BFGS-B optimization."""
    def func(h):
        obj, _ = smooth_max_and_grad(h, alpha, mu, target_sum)
        return obj
    
    def grad_func(h):
        _, g = smooth_max_and_grad(h, alpha, mu, target_sum)
        return g
