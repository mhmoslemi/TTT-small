For the piecewise linear part (without penalty):

This is the upper envelope of lines plus a parabola. The minimum can be found by:

2. The overall minimum is the minimum over k of [b_k + a_k*y_k + μ(y_k-target)²] where y_k is the minimizer of line k's contribution

Let me also think about whether I should use a fine grid or a smarter 1D optimization. Since g(y) is convex (max of linear + convex quadratic), I can use:

2. The minimum is bracketed
3. Use ternary search or golden section to find the exact minimum

Actually, for a convex piecewise-quadratic function (which is what we have: max of lines + quadratic), the minimum is at a point where either:
- The gradient of the active line + quadratic = 0
- Or at a breakpoint (where the active line changes)
- Or at the boundary y=0 or y=1

The breakpoints of the upper envelope are at most O(n) in number. So I can:
1. Find all breakpoints in [0,1] (O(n²) or O(n log n))
2. For each interval between consecutive breakpoints, the active line is fixed, so g(y) is a quadratic - find its minimum
3. Check boundaries

This gives the exact optimum in O(n²) per coordinate (or O(n log n) with careful implementation).

            s_vals = np.arange(-(n - 1), n)
            i_plus_s = i + s_vals
            i_minus_s = i - s_vals
            valid1 = (i_plus_s >= 0) & (i_plus_s < n)
            valid2 = (i_minus_s >= 0) & (i_minus_s < n) & (s_vals != 0)

            h_is = np.zeros(2 * n - 1)
            h_is[valid1] = h[i_plus_s[valid1]]
            h_ims = np.zeros(2 * n - 1)
            h_ims[valid2] = h[i_minus_s[valid2]]

            a = valid1.astype(float) * (1.0 - h_is) - valid2.astype(float) * h_ims
            a[s_zero_idx] = 0.0

            S_rest = np.sum(h) - old
            target_y = target - S_rest

            delta = y_grid - old
            corr_not0 = corr[mask_not0]
            a_not0 = a[mask_not0]
            corr_0 = corr[s_zero_idx]

            new_corr_not0 = corr_not0[None, :] + delta[:, None] * a_not0[None, :]
            max_not0 = np.max(new_corr_not0, axis=1)
            new_corr_0 = corr_0 + delta * (1.0 - y_grid - old)
            max_all = np.maximum(max_not0, new_corr_0)

            obj = max_all * dx + mu * (y_grid - target_y) ** 2
            best_idx = np.argmin(obj)
            new_val = y_grid[best_idx]

            if abs(new_val - old) > 1e-10:
                delta_actual = new_val - old
                corr[mask_not0] = corr[mask_not0] + delta_actual * a[mask_not0]
                corr[s_zero_idx] = corr[s_zero_idx] + delta_actual * (1.0 - new_val - old)
                h[i] = new_val

        h = project_feasible(h, target)
        corr = compute_corr(h)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    return best_h, best_score


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break

        def fun(x):
            return lse_obj_and_grad(x, alpha)[0]

        def jac(x):
            return lse_obj_and_grad(x, alpha)[1]

        def cfun(x):
            return np.sum(x) - target

        def cjac(x):
            return np.ones(n)

        try:
            res = minimize(fun, h_cur, jac=jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        if c_max - c_min < 1e-12:
            break
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= max(len(active), 1)
        step = step_size / (1.0 + 0.01 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline_final = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent at initial resolution (n=400)
    deadline_p1 = start_time + (budget_s - 15.0) * 0.15
    h_cd1, sc1 = coord_descent(best_h, n, target, mu=50.0, n_sweeps=200,
                               grid_size=100, deadline=deadline_p1, rng=rng)
    if sc1 < best_score:
        best_score = sc1
        best_h = h_cd1

    # Phase 2: SLSQP at n=400 to polish
    deadline_p2 = start_time + (budget_s - 15.0) * 0.25
    h_sl, sc2 = slsqp_refine(best_h, target,
                              [500, 2000, 8000, 30000], 2000, 1e-16, deadline_p2)
    if sc2 < best_score:
        best_score = sc2
        best_h = h_sl

    # Phase 3: Interpolate to n=800 and coordinate descent
    n_new = 800
    target_new = n_new / 2.0
    h_interp = interpolate_to(best_h, n_new, target_new)
    sc_interp = compute_c5(h_interp)
    if sc_interp < best_score:
        best_score = sc_interp
        best_h = h_interp

    deadline_p3 = start_time + (budget_s - 15.0) * 0.55
    h_cd2, sc3 = coord_descent(h_interp, n_new, target_new, mu=100.0, n_sweeps=150,
                               grid_size=100, deadline=deadline_p3, rng=rng)
    if sc3 < best_score:
        best_score = sc3
        best_h = h_cd2

    # Phase 4: SLSQP at n=800
    deadline_p4 = start_time + (budget_s - 15.0) * 0.72
    h_sl2, sc4 = slsqp_refine(best_h, target_new,
                               [2000, 10000, 50000, 200000], 3000, 1e-16, deadline_p4)
    if sc4 < best_score:
        best_score = sc4
        best_h = h_sl2

    # Phase 5: Fine coordinate descent at n=800
    deadline_p5 = start_time + (budget_s - 15.0) * 0.85
    h_cd3, sc5 = coord_descent(best_h, n_new, target_new, mu=500.0, n_sweeps=80,
                               grid_size=300, deadline=deadline_p5, rng=rng)
    if sc5 < best_score:
        best_score = sc5
        best_h = h_cd3

    # Phase 6: Subgradient polish + final SLSQP
    deadline_p6 = deadline_final
    h_sg, sc6 = subgrad_polish(best_h, target_new, 2000, 0.01, 0.01, deadline_p6)
    if sc6 < best_score:
        best_score = sc6
        best_h = h_sg

    # Try interpolation to n=600 if it helps (sometimes lower res is better)
    if time.time() < deadline_final - 60:
        n_alt = 600
        target_alt = n_alt / 2.0
        h_alt = interpolate_to(best_h, n_alt, target_alt)
        sc_alt = compute_c5(h_alt)
        dl = min(deadline_final - 30, time.time() + 60)
        h_cd_alt, sc_alt2 = coord_descent(h_alt, n_alt, target_alt, mu=80.0,
                                          n_sweeps=50, grid_size=150,
                                          deadline=dl, rng=rng)
        if sc_alt2 < best_score:
            best_score = sc_alt2
            best_h = h_cd_alt

    # Final projection and score
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
