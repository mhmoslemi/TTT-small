import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    tol = 1e-13
    if diff > 0:
        idx = np.where(h < 1.0 - tol)[0]
        if len(idx) == 0:
            return h
        slack = 1.0 - h[idx]
        total_slack = np.sum(slack)
        if total_slack < diff:
            h[idx] = 1.0
            remaining = diff - total_slack
            other = np.where(h < 1.0 - tol)[0]
            if len(other) > 0:
                h[other] += remaining / len(other)
        else:
            frac = diff / total_slack
            h[idx] += frac * slack
    else:
        idx = np.where(h > tol)[0]
        if len(idx) == 0:
            return h
        total = np.sum(h[idx])
        if total < -diff:
            h[idx] = 0.0
            remaining = -diff - total
            other = np.where(h > tol)[0]
            if len(other) > 0:
                h[other] -= remaining / len(other)
        else:
            frac = (-diff) / total
            h[idx] *= (1.0 - frac)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad


def lbfgs_step(h_start, target, alpha, topk, penalty, maxiter, bounds, ftol):
    def obj_func(x):
        f, g = lse_obj_grad(x, alpha, topk)
        s = np.sum(x) - target
        f_pen = f + penalty * s * s
        g_pen = g + 2.0 * penalty * s
        return f_pen, g_pen
    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def slsqp_active(h_start, target, top_lags, maxiter, alpha):
    n = len(h_start)
    n_corr = 2 * n - 1

    def smooth_max_obj(x, alpha_local):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr[top_lags]
        c_max = np.max(vals)
        shifted = alpha_local * (vals - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        return c_max + np.log(total) / alpha_local

    def smooth_max_grad(x, alpha_local):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr[top_lags]
        c_max = np.max(vals)
        shifted = np.clip(alpha_local * (vals - c_max), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        grad = np.zeros(n)
        one_minus_h = 1.0 - x
        for i, li in enumerate(top_lags):
            w = weights[i]
            if w < 1e-15:
                continue
            lag = li - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[lag:] += w * one_minus_h[:n - lag]
                    grad[:n - lag] -= w * h[lag:] if False else w * x[lag:]
                    grad[:n - lag] -= w * x[lag:]
            else:
                m = -lag
                if m < n:
                    grad[:n - m] += w * one_minus_h[m:]
                    grad[m:] -= w * x[:n - m]
        return grad

    def obj_func(x):
        return smooth_max_obj(x, alpha)

    def obj_jac(x):
        return smooth_max_grad(x, alpha)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_start, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def subgradient_line_search(h, target, max_steps=200):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    step_size = 0.02
    for step in range(max_steps):
        corr = get_correlations(best_h)
        c_max = np.max(corr)
        top_idx = np.where(corr >= c_max - 1e-3)[0]
        if len(top_idx) > 10:
            top_idx = np.argpartition(corr, -10)[-10:]
        avg_grad = np.zeros(n)
        for li in top_idx:
            lag = li - (n - 1)
            g = np.zeros(n)
            one_minus_h = 1.0 - best_h
            if lag >= 0:
                if lag < n:
                    g[lag:] += one_minus_h[:n - lag]
                    g[:n - lag] -= best_h[lag:]
            else:
                m = -lag
                if m < n:
                    g[:n - m] += one_minus_h[m:]
                    g[m:] -= best_h[:n - m]
            avg_grad += g
        avg_grad /= len(top_idx)
        norm_g = np.linalg.norm(avg_grad)
        if norm_g < 1e-10:
            break
        avg_grad /= norm_g
        found = False
        for shrink in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            s = step_size * shrink
            h_try = best_h - s * avg_grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-10:
                best_h = h_try
                best_score = sc
                found = True
                break
        if not found:
            step_size *= 0.5
            if step_size < 1e-6:
                break
    return best_h, best_score


def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: LSE optimization with increasing alpha, tighter topk
    alphas_phase1 = [100, 300, 1000, 3000, 8000, 15000]
    topk1 = 15
    for alpha in alphas_phase1:
        if time_left() < 50:
            break
        for restart in range(3):
            if time_left() < 25:
                break
            noise_scale = 0.002 if restart > 0 else 0.003
            h_start = best_h + rng.randn(n) * noise_scale
            h_start = project_feasible(h_start, target)
            maxiter = 400 if alpha <= 1000 else 300
            h_new = lbfgs_step(h_start, target, alpha, topk1, 20000, maxiter, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 2: Very sharp LSE (topk=5, high alpha)
    if time_left() > 60:
        for alpha in [20000, 50000]:
            if time_left() < 30:
                break
            h_start = best_h + rng.randn(n) * 0.001
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_step(h_start, target, alpha, 5, 50000, 500, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 3: Active-set SLSQP (minimize smooth max of top lags)
    if time_left() > 80:
        for n_active in [3, 5, 8]:
            if time_left() < 40:
                break
            corr = get_correlations(best_h)
            top_lags = np.argsort(corr)[-n_active:]
            for alpha in [1000, 5000]:
                if time_left() < 30:
                    break
                h_new = slsqp_active(best_h, target, top_lags, 500, alpha)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new

    # Phase 4: Subgradient with line search
    if time_left() > 40:
        for _ in range(5):
            if time_left() < 20:
                break
            best_h, best_score = subgradient_line_search(best_h, target, max_steps=150)

    # Phase 5: Multi-resolution exploration
    res_list = [500, 600, 700]
    for n_try in res_list:
        if time_left() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to_n(best_h, n, n_try)
        h_interp = project_feasible(h_interp, target_try)
        bounds_try = [(0.0, 1.0)] * n_try

        for alpha in [500, 2000, 8000]:
            if time_left() < 40:
                break
            for restart in range(2):
                if time_left() < 20:
                    break
                h_start = h_interp + rng.randn(n_try) * 0.004
                h_start = project_feasible(h_start, target_try)
                h_new = lbfgs_step(h_start, target_try, alpha, 15, 20000, 300, bounds_try, 1e-17)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    n = n_try
                    target = target_try
                    bounds = bounds_try
                    best_h = h_new
                    h_interp = interpolate_to_n(best_h, n, n_try)
                    h_interp = project_feasible(h_interp, n_try / 2.0)

    # Phase 6: Return to best resolution and polish
    if time_left() > 60:
        # If we changed resolution, re-interpolate back to best
        pass
        # Subgradient polish
        best_h, best_score = subgradient_line_search(best_h, target, max_steps=200)

    # Phase 7: Final SLSQP on active set
    if time_left() > 40:
        corr = get_correlations(best_h)
        top_lags = np.argsort(corr)[-4:]
        h_new = slsqp_active(best_h, target, top_lags, 800, 10000)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 8: Final L-BFGS polish
    if time_left() > 25:
        h_new = lbfgs_step(best_h, target, 50000, 3, 50000, 300, bounds, 1e-17)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: Very fine subgradient
    if time_left() > 15:
        best_h, best_score = subgradient_line_search(best_h, target, max_steps=100)

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
