def slsqp_obj_active(x, alpha, k_top):

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(float)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r


def compute_subgrad_single_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        mm = -m
        grad[mm:] += (1.0 - h[:n - mm])
        grad[:n - mm] -= h[mm:]
    return grad


def compute_active_subgrad(h, active_indices):
    n = len(h)
    grad = np.zeros(n)
    for k in active_indices:
        m = int(k) - (n - 1)
        grad += compute_subgrad_single_lag(h, m)
    grad /= len(active_indices)
    return grad

def _slsqp_obj(x):
    n = _g_n
    target = _g_target
    alpha = _g_alpha
    active = _g_active
    h = project_feasible(x, target)
    corr = compute_corr_scaled(h)
    top_vals = corr[active]
    c_top = float(top_vals.max())
    shifted = alpha * (top_vals - c_top)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_top + float(np.log(np.sum(np.exp(shifted)))) / alpha


def _slsqp_jac(x):
    n = _g_n
    target = _g_target
    alpha = _g_alpha
    active = _g_active
    h = project_feasible(x, target)
    corr = compute_corr_scaled(h)
    top_vals = corr[active]
    c_top = float(top_vals.max())
    shifted = alpha * (top_vals - c_top)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    grad = np.zeros(n)
    for idx in range(len(active)):
        w = weights[idx] / w_sum
        if w < 1e-12:
            continue
        m = int(active[idx]) - (n - 1)
        grad += w * compute_subgrad_single_lag(h, m)
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)

def interpolate_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    t1 = min(40.0, tleft() * 0.1)
    h_sa, c5_sa = binary_sa_optimize(h_bin, target, t1, start_time, rng)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 2: Active-lag guided hill climb
    if tleft() > 60:
        t2 = min(80.0, tleft() * 0.12)
        h_hc, c5_hc = active_lag_hill_climb(best_h, target, t2, start_time, rng)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 3: Main SLSQP with active-set continuation
    if tleft() > 100:
        t3 = min(350.0, tleft() * 0.35)
        h_slq, c5_slq = slsqp_active_set(best_h, target, t3, start_time)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    # Phase 4: Subgradient with momentum
    if tleft() > 80:
        t4 = min(120.0, tleft() * 0.2)
        h_sg, c5_sg = subgradient_momentum(best_h, target, t4, start_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg.copy()

    # Phase 5: Multi-start SLSQP (different perturbations and parameters)
    if tleft() > 100:
        t5 = min(250.0, tleft() * 0.3)
        h_ms, c5_ms = multi_start_slsqp(best_h, target, t5, start_time, rng)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 6: Hill climb again on refined solution
    if tleft() > 50:
        h_rounded = project_binary(best_h, target)
        t6 = min(50.0, tleft() * 0.15)
        h_hc2, c5_hc2 = active_lag_hill_climb(h_rounded, target, t6, start_time, rng)
        if c5_hc2 < best_score:
            best_score = c5_hc2
            best_h = h_hc2.copy()

    # Phase 7: Another SLSQP round from best
    if tleft() > 80:
        t7 = min(200.0, tleft() * 0.3)
        h_slq2, c5_slq2 = slsqp_active_set(best_h, target, t7, start_time)
        if c5_slq2 < best_score:
            best_score = c5_slq2
            best_h = h_slq2.copy()

    # Phase 8: Fractional polish
    if tleft() > 30:
        t8 = min(60.0, tleft() * 0.2)
        h_fp, c5_fp = fractional_polish(best_h, target, t8, start_time)
        if c5_fp < best_score:
            best_score = c5_fp
            best_h = h_fp.copy()

    # Phase 9: Try different n values
    n_candidates = []
    for factor in [1.2, 0.85, 1.5]:
        nc = int(n * factor)
        if nc % 2 != 0:
            nc += 1
        if 200 <= nc <= 2000 and nc != n:
            n_candidates.append(nc)

    for n_try in n_candidates:
        if tleft() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to_n(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        # Quick SA
        t_sa = min(25.0, tleft() * 0.15)
        h_sa_r, c5_sa_r = binary_sa_optimize(h_bin_try, target_try, t_sa, start_time, rng)

        # SLSQP
        if tleft() > 40:
            t_slq = min(80.0, tleft() * 0.2)
            h_slq_r, c5_slq_r = slsqp_active_set(h_sa_r, target_try, t_slq, start_time)
            if c5_slq_r < c5_sa_r:
                h_sa_r = h_slq_r
                c5_sa_r = c5_slq_r

        if c5_sa_r < best_score:
            best_score = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 10: Final SLSQP polish with very high alpha
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        _g_n = n_use
        _g_target = target_use
        bounds = [(0.0, 1.0)] * n_use
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        h_cur = best_h.copy()
        for alpha in [15000, 25000, 50000]:
            if tleft() < 15:
                break
            _g_alpha = alpha
            corr = compute_corr_scaled(h_cur)
            k_act = min(50, len(corr) - 1)
            _g_active = np.argsort(corr)[-k_act:]
            try:
                res = minimize(
                    _slsqp_obj, h_cur, jac=_slsqp_jac,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 150, 'ftol': 1e-17}
                )
                h_new = project_feasible(res.x, target_use)
                c5_new = compute_c5(h_new)
                if c5_new < best_score:
                    best_score = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new
            except Exception:
                pass

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if abs(final_score - best_score) < 1e-10:
        best_h = h_final
        best_score = final_score
    else:
        # If projection changed things, use the projected version
        best_h = h_final
        best_score = final_score

    return best_h, float(best_score), n_final
