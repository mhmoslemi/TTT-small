```python
    # grad[i] = dx * sum_{k=0}^{2n-2} p[k] * [(1-h[i+k-(n-1)]) * 1_{0<=i+k-(n-1)<n} - h[i-k+(n-1)] * 1_{0<=i-k+(n-1)<n}]
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        weight = dx * p[k]
        if weight < 1e-15:
            continue
        # (1-h[i+lag]) for valid i+lag
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        if i_start < i_end:
            grad[i_start:i_end] += weight * (1.0 - h[i_start + lag:i_end + lag])
        # -h[i-lag] for valid i-lag
        i_start2 = max(0, lag)
        i_end2 = min(n, n + lag)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= weight * h[i_start2 - lag:i_end2 - lag]
    return grad
```

This is O(n * 2n) = O(n²) with numpy vectorization, which is fine.

But actually, most of the p[k] values will be very small (since it's a softmax over 2n-1 values, and only the few largest ones will have significant weight). So in practice, the computation is much faster.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def is_prime_check(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True

def make_qr_sequence(n, rng):
    target = n // 2
    p = n + 1
    while not is_prime_check(p):
        p += 1
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

def smooth_obj_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    s_max = np.max(scaled)
    exp_s = np.exp(scaled - s_max)
    sum_exp = np.sum(exp_s)
    f = (s_max + np.log(sum_exp)) / alpha
    
    p = exp_s / sum_exp  # softmax weights
    
    # Compute gradient
    # grad[i] = dx * sum_k p[k] * [(1-h[i+k-(n-1)]) * 1_{valid} - h[i-k+(n-1)] * 1_{valid}]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = dx * p[k]
        if w < 1e-18:
            continue
        lag = k - (n - 1)
        # (1-h[i+lag]) for valid i
        i_s = max(0, -lag)
        i_e = min(n, n - lag)
        if i_s < i_e:
            grad[i_s:i_e] += w * (1.0 - h[i_s + lag:i_e + lag])
        # -h[i-lag] for valid i
        i_s2 = max(0, lag)
        i_e2 = min(n, n + lag)
        if i_s2 < i_e2:
            grad[i_s2:i_e2] -= w * h[i_s2 - lag:i_e2 - lag]
    
    return f, grad

def continuous_optimize(h_init, n, target, alphas, maxiter_each, time_limit, t_start):
    h = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                lambda x: smooth_obj_and_grad(x, alpha, n)[0],
                h,
                jac=lambda x: smooth_obj_and_grad(x, alpha, n)[1],
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter_each, 'ftol': 1e-16, 'gtol': 1e-8}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h = h_new
        except Exception:
            pass
    
    return h, curr_score
