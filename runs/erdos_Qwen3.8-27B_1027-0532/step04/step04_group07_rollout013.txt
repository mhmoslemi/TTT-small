```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            if free.sum() == 0:
                break
            per = diff / free.sum()
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            if free.sum() == 0:
                break
            per = diff / free.sum()
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def benders_lp(h_start, target_sum, max_iter, time_limit, max_constraints=500):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cuts = []  # list of (grad_array, rhs_scalar)
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        # Generate new cuts from current h
        corr = overlap_corrs(h)
        # Add top-5 correlations as cuts
        top5 = np.argsort(corr)[-5:]
        for idx in top5:
            g = compute_grad_single_lag(h, idx)
            rhs = corr[idx] - g @ h
            cuts.append((g, rhs))
        
        # Prune if too many constraints
        if len(cuts) > max_constraints:
            # Keep the most "active" cuts (those most violated at current h)
            violations = np.array([g @ h - rhs for g, rhs in cuts])
            keep = np.argsort(violations)[-max_constraints:]
            cuts = [cuts[i] for i in keep]
        
        # Build and solve LP
        m_cuts = len(cuts)
        # Variables: h[0..n-1], t (index n)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((m_cuts, n + 1))
        b_ub = np.zeros(m_cuts)
        for i, (g, rhs) in enumerate(cuts):
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0  # -t
            b_ub[i] = rhs
        
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        
        bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if not res.success:
            break
        
        h_new = res.x[:n]
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_new.copy()
            h = h_new
        else:
            # Try blending
            h_blend = project_feasible(0.7 * h + 0.3 * h_new, target_sum)
            sc_blend = compute_c5(h_blend)
            if sc_blend < best_score - 1e-14:
                best_score = sc_blend
                best_h = h_blend.copy()
                h = h_blend
            else:
                # No improvement, try smaller step
                h_blend2 = project_feasible(0.9 * h + 0.1 * h_new, target_sum)
                sc_blend2 = compute_c5(h_blend2)
                if sc_blend2 < best_score - 1e-14:
                    best_score = sc_blend2
                    best_h = h_blend2.copy()
                    h = h_blend2
                else:
                    break
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(200):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def lp_step(h_cur, target_sum, topk, lams=1.0):
    """One LP linearization step."""
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target_sum)
    return None

def lp_refine(h_start, target_sum, topk_schedule, max_iter, time_limit):
    """LP refinement with adaptive topk."""
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # Determine current topk
        frac = it / max(1, max_iter)
        # Interpolate in log space
        log_k0 = np.log(topk_schedule[0])
        log_k1 = np.log(topk_schedule[1])
        k_cur = int(np.exp(log_k0 + frac * (log_k1 - log_k0)))
        k_cur = max(1, k_cur)
        
        h_lp = lp_step(h, target_sum, k_cur)
        if h_lp is None:
            break
        
        score = compute_c5(h_lp)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stagnation = 0
        else:
            # Try blending
            improved = False
            for frac_blend in [0.5, 0.3, 0.1]:
                h_b = project_feasible((1-frac_blend)*h + frac_blend*h_lp, target_sum)
                sc_b = compute_c5(h_b)
                if sc_b < best_score - 1e-14:
                    best_score = sc_b
                    best_h = h_b.copy()
                    h = h_b
                    stagnation = 0
                    improved = True
                    break
            if not improved:
                stagnation += 1
                if stagnation > 30:
                    break
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgrad_smoothmax(h_start, target_sum, max_iter, alpha, topk, lr, momentum, time_limit, seed):
```
