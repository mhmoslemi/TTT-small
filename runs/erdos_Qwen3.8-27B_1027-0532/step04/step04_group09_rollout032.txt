        if abs(diff) < 1e-15:

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h = h.copy()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha

    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_single(h0, target, alpha, maxiter, ftol):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)


def slsqp_sweep(h0, target, alphas, deadline, maxiter=3000, ftol=1e-18):
    h_cur = h0.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_new, sc = slsqp_single(h_cur, target, alpha, maxiter, ftol)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
    return h_cur, best_score


def fista_optimize(h0, target, n_steps, alpha, deadline):
    n = len(h0)
    h = h0.copy()
    z = h0.copy()
    t_k = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)

    obj_z, grad_z = lse_obj_and_grad(z, alpha)
    s = 1.0 / (alpha * 4.0 * n)

    for it in range(n_steps):
        if time.time() > deadline:
            break

        h_new = z - s * grad_z
        h_new = project_feasible(h_new, target)

        obj_new = lse_obj_and_grad(h_new, alpha)[0]
        if obj_new > obj_z + 0.01 * s * np.dot(grad_z, grad_z):
            s *= 0.5
            h_new = z - s * grad_z
            h_new = project_feasible(h_new, target)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        z = project_feasible(z, target)
        t_k = t_new

        obj_z, grad_z = lse_obj_and_grad(z, alpha)

        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def active_subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        range_c = c_max - c_min
        if range_c < 1e-12:
            break
        threshold = c_max - active_frac * range_c
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])

        grad /= len(active)
        step = step_size / (1.0 + 0.005 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    return best_h, best_score


def perturb_structured(h, mode, amplitude, n):
    x = np.linspace(0.0, 2.0, n)
    perturbation = amplitude * np.sin(mode * np.pi * x / 2.0)
    h_new = h + perturbation
    return project_feasible(h_new, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    total_work_time = budget_s - 30.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Initial SLSQP refinement with moderate alphas
    deadline_p1 = start_time + total_work_time * 0.12
    h_cur, sc = slsqp_sweep(best_h, target, [30, 100, 500, 2000], deadline_p1, 2500, 1e-17)
    if sc < best_score:
        best_score = sc
        best_h = h_cur

    # Phase 2: FISTA with high alpha
    deadline_p2 = start_time + total_work_time * 0.22
    h_f, sc = fista_optimize(best_h, target, 8000, 20000.0, deadline_p2)
    if sc < best_score:
        best_score = sc
        best_h = h_f

    # Phase 3: Try complement (1-h) as starting point
    deadline_p3 = start_time + total_work_time * 0.32
    h_comp = project_feasible(1.0 - h_init, target)
    h_c, sc = slsqp_sweep(h_comp, target, [100, 500, 3000], deadline_p3, 2000, 1e-17)
    if sc < best_score:
        best_score = sc
        best_h = h_c

    # Phase 4: Multi-start with structured perturbations
    deadline_p4 = start_time + total_work_time * 0.52
    n_starts = 5
    time_per_start = (deadline_p4 - time.time()) / n_starts
    for i in range(n_starts):
        if time.time() > deadline_p4 - time_per_start * 0.3:
            break
        mode = i + 1
        amp = 0.04 * (0.75 ** i)
        h_pert = perturb_structured(best_h, mode, amp, n)
        sub_dl = min(deadline_p4 - 5.0, time.time() + time_per_start * 0.8)
        h_opt, sc = slsqp_sweep(h_pert, target, [500, 2000, 10000], sub_dl, 2000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 5: Active subgradient polishing
    deadline_p5 = start_time + total_work_time * 0.62
    h_sg, sc = active_subgrad_polish(best_h, target, 8000, 0.025, 0.008, deadline_p5)
    if sc < best_score:
        best_score = sc
        best_h = h_sg

    # Phase 6: High-alpha SLSQP for precision
    deadline_p6 = start_time + total_work_time * 0.78
    h_cur, sc = slsqp_sweep(best_h, target, [20000, 60000, 150000, 300000], deadline_p6, 4000, 1e-18)
    if sc < best_score:
        best_score = sc
        best_h = h_cur

    # Phase 7: Try higher resolution
    if time.time() < start_time + total_work_time * 0.90:
        deadline_p7 = start_time + total_work_time * 0.90
        for n_new in [600, 800, 1000]:
            if time.time() > deadline_p7:
                break
            target_new = n_new / 2.0
            h_new = interpolate_to(best_h, n_new, target_new)
            sub_dl = min(deadline_p7 - 10.0, time.time() + 120.0)
            h_ref, sc = slsqp_sweep(h_new, target_new, [500, 3000, 15000, 60000], sub_dl, 2500, 1e-17)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: Final polishing
    deadline_final = start_time + budget_s - 5.0
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_cur, sc = slsqp_sweep(best_h, target_final, [100000, 300000, 500000],
                            deadline_final, 3000, 1e-18)
    if sc < best_score:
        best_score = sc
        best_h = h_cur

    # Final subgradient micro-polish
    if time.time() < start_time + budget_s - 3.0:
        h_sg, sc = active_subgrad_polish(best_h, target_final, 5000, 0.005, 0.003,
                                         start_time + budget_s - 2.0)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
