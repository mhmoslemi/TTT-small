def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

So result[k] = sum_i a[i] * v[i + k - n + 1]

With a = h, v = 1-h:
result[k] = sum_i h[i] * (1-h[i + k - n + 1]) for valid i

Let me define the "shift" as s = k - (n-1), so s ranges from -(n-1) to (n-1).
result[s + n - 1] = sum_i h[i] * (1-h[i+s]) for 0 ≤ i < n, 0 ≤ i+s < n

OK so this matches what I had before. The "shift" in the problem is s (the actual displacement).

Now, if I change h[m] from 1 to 0:
- The term i=m in the sum: h[m]*(1-h[m+s]) = 1*(1-h[m+s]) is removed
- The term i=m-s (if valid): h[m-s]*(1-h[m-s+s]) = h[m-s]*(1-h[m]) changes from h[m-s]*0 to h[m-s]*1

So the change is: -(1-h[m+s]) + h[m-s] = h[m+s] + h[m-s] - 1

For s = k_star (the max shift):
Change = h[m+k_star] + h[m-k_star] - 1

If I also change h[n] from 0 to 1:
- The term i=n: h[n]*(1-h[n+s]) = 0 changes to 1*(1-h[n+s]) = 1-h[n+s]: gain (1-h[n+s])
- The term i=n-s (if valid): h[n-s]*(1-h[n-s+s]) = h[n-s]*(1-h[n]) changes from h[n-s]*1 to h[n-s]*0: lose h[n-s]

Change = (1-h[n+s]) - h[n-s] = 1 - h[n+s] - h[n-s]

Total change at shift s:
(h[m+s] + h[m-s] - 1) + (1 - h[n+s] - h[n-s])
= h[m+s] + h[m-s] - h[n+s] - h[n-s]

Great, this confirms my formula: Δ_s = h[i+s] + h[i-s] - h[j+s] - h[j-s] where i is the 1→0 position and j is the 0→1 position.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h2 = h.copy()
    h2[idx] += diff / len(idx)
    return np.clip(h2, 0.0, 1.0)

def project_binary(h, target_sum):
    h2 = np.clip(np.round(h), 0, 1)
    cs = int(h2.sum())
    if cs == target_sum:
        return h2.astype(float)
    if cs > target_sum:
        need = cs - target_sum
        ones = np.where(h2 == 1)[0]
        order = np.argsort(h[ones])[:need]
        h2[ones[order]] = 0
    else:
        need = target_sum - cs
        zeros = np.where(h2 == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h2[zeros[order]] = 1
    return h2.astype(float)

Wait, the objective is max_k (corr_k * dx) = dx * max_k corr_k. So:
f(h, α) = dx * (1/α) * log(Σ_k exp(α * corr_k))

The gradient w.r.t. h[m]:
∂corr_k/∂h[m] = (1-h[m+k]) * [0≤m+k<n] - h[m-k] * [0≤m-k<n]

where softmax_k = exp(α * corr_k) / Σ_j exp(α * corr_j).

This is a weighted sum of the individual shift gradients. For large α, the weights concentrate on the max shift (or few max shifts).

                max_c = np.max(corr)
                if max_c < 1e-10:
                    return 0.0

                shifted = np.clip(shifted, -500, 0)

                max_c = np.max(corr)

                    if len(m_range_2) > 0:
                        grad[m_range_2] -= w * hv[m_range_2 - k]

            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target,

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

    h2 = h.copy()
    h2[idx] += diff / len(idx)
    return np.clip(h2, 0.0, 1.0)
