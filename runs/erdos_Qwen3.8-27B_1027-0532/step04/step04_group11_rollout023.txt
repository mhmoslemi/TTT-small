        if abs(diff) < 1e-12:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_from_continuous(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])  # lowest h values first
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])  # highest h values first
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin

def sa_binary(h, n, n_iter, t_start, t_end, deadline, seed=42):
    """Simulated annealing on binary space."""
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    best = h_cur.copy()
    best_score = compute_c5(h_cur, n)
    cur_score = best_score
    target = n // 2
    
    # Precompute positions
    ones = np.where(h_cur == 1)[0]
    zeros = np.where(h_cur == 0)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h_cur[a] = 0
        h_cur[b] = 1
        new_score = compute_c5(h_cur, n)
        
        delta = new_score - cur_score
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-12)):
            cur_score = new_score
            # Update ones and zeros
            a_pos = np.where(ones == a)[0]
            b_pos = np.where(zeros == b)[0]
            if len(a_pos) > 0:
                ones = np.delete(ones, a_pos[0])
            if len(b_pos) > 0:
                zeros = np.delete(zeros, b_pos[0])
            ones = np.append(ones, b)
            zeros = np.append(zeros, a)
            if cur_score < best_score:
                best_score = cur_score
                best = h_cur.copy()
        else:
            h_cur[a] = 1
            h_cur[b] = 0
    
    return best, best_score

def sa_binary_simple(h, n, n_iter, t_start, t_end, deadline, seed=42):

    for it in range(n_iter):

        t = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))
        
        # Random swap
        a = rng.randint(n)
        b = rng.randint(n)
        if h_cur[a] == h_cur[b]:
            continue  # Need one 1 and one 0
        
        # Ensure a is the 1 and b is the 0
        if h_cur[a] == 0:
            a, b = b, a
        
        h_cur[a] = 0
        h_cur[b] = 1
        new_score = compute_c5(h_cur, n)
        
        delta = new_score - cur_score
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-12)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = h_cur.copy()
        else:
            h_cur[a] = 1
            h_cur[b] = 0
    
    return best, best_score

This is simpler but has a 50% chance of rejecting the move due to h[a]==h[b]. That's fine, it just means we need about 2x the iterations.

For n=400, compute_c5 takes about 10-20 μs (np.correlate for length 400). So each iteration is about 20-40 μs. For 5 million iterations, that's 100-200 seconds. That's feasible within the 1300s budget.

- Pick a random position a and block size s
- Pick a random position b (not overlapping with a)
- Swap h[a:a+s] with h[b:b+s]

This allows for larger jumps in the solution space.

But block swaps are harder to evaluate incrementally. For now, let me stick with single swaps and add block moves as a secondary exploration.
