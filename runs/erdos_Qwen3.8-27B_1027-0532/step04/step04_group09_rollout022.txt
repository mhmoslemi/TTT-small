import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    """Log-sum-exp objective and gradient for the max of correlations."""
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad

def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    """SLSQP refinement with log-sum-exp smoothing."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        def fun(x):
            return lse_obj_and_grad(x, alpha)[0]
        def jac(x):
            return lse_obj_and_grad(x, alpha)[1]
        try:
            res = minimize(fun, h_cur, jac=jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x)-target,
                                        'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def coord_descent_optimize(h, n, n_passes, rng, deadline, refine_grid=True):
    """Coordinate descent optimization."""
    h_cur = h.copy()
    target = n / 2.0
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    x_coarse = np.array([0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45,
                         0.5, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0])
    
    for pass_i in range(n_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        corr = compute_corr(h_cur)
        
        for j in order:
            hj = h_cur[j]
            a_k = np.zeros(n-1)
            max_k1 = n-1-j
            if max_k1 >= 1:
                a_k[:max_k1] = 1.0 - h_cur[j+1:j+max_k1+1]
            max_k2 = j
            if max_k2 >= 1:
                a_k[:max_k2] -= h_cur[j-1:j-max_k2-1:-1]
            
            B_k = corr[n:2*n-1] - a_k * hj
            A = corr[n-1] - hj * (1.0 - hj)
            
            # Coarse grid evaluation
            quad_vals = -x_coarse**2 + x_coarse + A
            lin_max = np.max(a_k[None, :] * x_coarse[:, None] + B_k[None, :], axis=1)
            f_vals = np.maximum(quad_vals, lin_max)
            idx_best = np.argmin(f_vals)
            x_opt = x_coarse[idx_best]
            
            # Fine refinement around best
            if refine_grid and 0 < idx_best < len(x_coarse)-1:
                x_lo = x_coarse[max(0, idx_best-1)]
                x_hi = x_coarse[min(len(x_coarse)-1, idx_best+1)]
                x_fine = np.linspace(x_lo, x_hi, 20)
                quad_f = -x_fine**2 + x_fine + A
                lin_f = np.max(a_k[None, :] * x_fine[:, None] + B_k[None, :], axis=1)
                f_f = np.maximum(quad_f, lin_f)
                idx_f = np.argmin(f_f)
                if f_f[idx_f] < f_vals[idx_best]:
                    x_opt = x_fine[idx_f]
            
            h_cur[j] = x_opt
        
        h_cur = project_feasible(h_cur, target)
        score = compute_c5(h_cur)
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
    
    return best_h, best_score

def subgrad_polish(h, target, n_steps, step_size, deadline):
    """Subgradient polishing focused on the max term."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        # Use top 5 terms for a smoother gradient
        top_idx = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n-1)
            if k >= 0:
                if k < n:
                    grad[:n-k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n-k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n-kk]
                    grad[:n-kk] += (1.0 - h_cur[kk:])
        grad /= len(top_idx)
        step = step_size / (1.0 + 0.01 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score

Or even better: use the fact that f(x) = max(quad, linear_envelope) and solve it more directly.

The function max(quad, g(x)) where quad is a concave parabola and g is a convex piecewise-linear function (upper envelope of lines) has the following structure:
- quad(x) = -x² + x + A (decreasing for x > 0.5, increasing for x < 0.5)
- g(x) = max_k(a_k x + B_k) (convex, piecewise linear)

The minimum of max(quad, g) on [0,1]:
- If g(x) ≤ quad(x) for all x, then min is at x=0.5 (max of quad) → min of max is at endpoints or where they cross
  Actually no, if g ≤ quad everywhere, then max(quad, g) = quad, and min of quad on [0,1] is at x=0 or x=1 (whichever gives smaller quad value). quad(0) = A, quad(1) = A. So min is A.
  
- If quad(x) ≤ g(x) for all x, then max(quad, g) = g, and min is min of g on [0,1].

- In the mixed case, the minimum occurs where quad(x) = g(x) (at a point where quad is decreasing, i.e., x > 0.5), or at x=0, or at x=1.

To find where quad(x) = g(x) for x > 0.5:
quad(x) = -x² + x + A
g(x) = max_k(a_k x + B_k)

At the optimal x*, we need -x*² + x* + A = g(x*) and the subgradient condition.

This is equivalent to: at x*, the quadratic is tangent to (or equal to) the upper envelope g, and x* > 0.5.

For a convex function g and a concave function quad, the equation quad = g has at most 2 solutions (since quad - g is concave - convex = concave, and a concave function crosses zero at most twice).

So I can find the solutions by:
1. Check if quad(0.5) ≤ g(0.5). If yes, the minimum is in the "g-dominated" region and is at min(g on [0,1]).
2. If quad(0.5) > g(0.5), there might be a crossing in (0.5, 1]. Find it by bisection on h(x) = quad(x) - g(x) (which is concave).

Step 2: h(0.5) > 0 and h(1) = -1 + 1 + A - g(1) = A - g(1). If A ≤ g(1), then h(1) ≤ 0, so there's a root in (0.5, 1]. Find it by bisection (O(log(1/eps)) evaluations of g, each O(n)).

This gives an exact (up to bisection tolerance) solution in O(n log(1/eps)) time per coordinate! Much faster than the grid approach.

def find_optimal_x(A, a_k, B_k, hj):
    """Find x in [0,1] that minimizes max(-x^2+x+A, max_k(a_k*x+B_k))."""
    n = len(a_k)
    
    def g(x):
        """Upper envelope of lines at x."""
        return np.max(a_k * x + B_k)
    
    def quad(x):
        return -x*x + x + A
    
    # Check endpoints
    f0 = max(quad(0), g(0))  # quad(0) = A
    f1 = max(quad(1), g(1))  # quad(1) = A
    
    # Check if quad dominates at x=0.5
    q_mid = quad(0.5)  # A + 0.25
    g_mid = g(0.5)
    
    if q_mid <= g_mid:
        # g dominates at midpoint, so min is likely where g is minimal
        # g is convex (max of lines), so its min on [0,1] is where its subgradient contains 0
        # Or just check: min of g on [0,1]
        # g min is at the "kink" where the slope changes sign
        # For simplicity, evaluate at a few points
        # Actually, for the max of lines, the minimum on [0,1] is at x=0 or x=1
        # or at a kink point.
        # Since g is convex, its minimum is unique (or on a flat region).
        # The minimum of a convex piecewise-linear function on [0,1] is at:
        # - x=0 if the right derivative at 0 is >= 0
        # - x=1 if the left derivative at 1 is <= 0
        # - at a kink where the slope changes from negative to positive
        # For efficiency, just check x=0 and x=1:
        best = min(f0, f1)
        best_x = 0 if f0 <= f1 else 1
        # Also check x=0.5
        f_mid = max(q_mid, g_mid)
        if f_mid < best:
            best = f_mid
            best_x = 0.5
        return best_x
    
    # quad dominates at midpoint: look for crossing in (0.5, 1]
    # h(x) = quad(x) - g(x) is concave (quad concave - g convex)
    # h(0.5) > 0, h(1) = A - g(1)
    if A >= g(1):
        # h(1) >= 0, so quad dominates everywhere on [0.5, 1]
        # Min of max(quad, g) on [0.5, 1] is min of quad = quad(1) = A (at x=1)
        # But we should also check [0, 0.5]
        # On [0, 0.5], quad is increasing from A to A+0.25
        # g could be above or below
        # If g dominates somewhere on [0, 0.5], the min might be there
        # For simplicity, check x=0:
        return min(f0, f1) and (0 if f0 <= f1 else 1)
    
    # h(1) < 0 < h(0.5), so there's a root in (0.5, 1)
    # Find it by bisection
    lo, hi = 0.5, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if quad(mid) > g(mid):
            lo = mid
        else:
            hi = mid
    x_cross = (lo + hi) / 2
    
    # At the crossing, f = quad = g
    f_cross = quad(x_cross)
    
    # The minimum of max(quad, g) on [0.5, 1] is at x_cross (since quad is decreasing there
    # and g is increasing (convex), the max transitions from quad to g)
    # Actually, for x < x_cross: quad > g, so f = quad (decreasing)
    # For x > x_cross: g > quad, so f = g (increasing, since g is convex and we're past its min)
    # Wait, g is convex but not necessarily increasing for all x.
    # Let me just evaluate at x_cross and compare with endpoints.
    
    candidates = [(f0, 0.0), (f1, 1.0), (f_cross, x_cross)]
    candidates.append((max(quad(0.5), g_mid), 0.5))
    best = min(candidates, key=lambda t: t[0])
    return best[1]
