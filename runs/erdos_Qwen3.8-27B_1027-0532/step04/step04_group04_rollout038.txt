```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())

    ones = np.where(h_r == 1)[0]
```

```python
def l_bfgs_refine(h_start, n, target, alpha, maxiter, time_limit, start_time):
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        log_sum = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
        # Penalty for constraint violation
        penalty = 100.0 * (np.sum(h) - target) ** 2
        return log_sum + penalty
    
    def gradient(h):
        # Numerical gradient (expensive but works)
        eps = 1e-7
        grad = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
```

```python
    # T2[j] = sum_m w[m] * h[j-k_m] where k_m = m-(n-1)
    # = sum_m w[m] * h[j-m+(n-1)]
    # Let p = j-m+(n-1), so m = j+(n-1)-p
    # = sum_p h[p] * w[j+(n-1)-p]
    # Compare with np.correlate(h, w, 'full')[i] = sum_k h[k] * w[k + i - (2n-2)]
    # Setting k+p: I want w[j+(n-1)-p] = w[p + i - (2n-2)] where k=p
    # j+(n-1)-p = p + i - (2n-2) => 2p = j+(n-1) + (2n-2) - i => this doesn't work cleanly.
    
    # Let me try a different approach. np.correlate(h, w, 'full')[i] = sum_k h[k] * w[k+i-(2n-2)]
    # I want: sum_p h[p] * w[j+n-1-p]
    # So I need: k+i-(2n-2) = j+n-1-p => with k=p: i-(2n-2) = j+n-1-2p... no, p is the index in both.
    # Hmm, this isn't a simple correlation.
    
    # Let me try: sum_p h[p] * w[j+n-1-p] = sum_p h[p] * (w_reversed)[2n-2-(j+n-1-p)]
    # = sum_p h[p] * (w_reversed)[n-1-j+p]
    # = np.correlate(h, w_reversed, 'full')[n-1-j]  ... let me check.
    # np.correlate(h, w_rev, 'full')[i] = sum_k h[k] * w_rev[k+i-(2n-2)]
    # = sum_k h[k] * w[2n-2-k-i+(2n-2)]  ... no, w_rev[q] = w[2n-2-q]
    # = sum_k h[k] * w[2n-2-(k+i-(2n-2))] = sum_k h[k] * w[4n-4-k-i]
    # That's not right either. w_rev has length 2n-1, so w_rev[q] = w[2n-2-q] for q=0..2n-2.
    # np.correlate(h, w_rev, 'full')[i] = sum_k h[k] * w_rev[k+i-(2n-2)]
    # For this to be valid: 0 <= k+i-(2n-2) <= 2n-2, i.e., 2n-2-k <= i <= 4n-4-k
    # Since k is in [0, n-1]: i ranges from 2n-2-(n-1)=n-1 to 4n-4-0=4n-4, but also from 2n-2 to 2n-2+(n-1)=3n-3.
    # So i ranges from n-1 to 3n-3.
    # Value: sum_k h[k] * w[2n-2-(k+i-(2n-2))] = sum_k h[k] * w[4n-4-k-i]
    # For 4n-4-k-i to be in [0, 2n-2]: 2n-2-k <= i <= 4n-4-k. Same as above.
    
    # I want: sum_p h[p] * w[j+n-1-p]
    # Set 4n-4-k-i = j+n-1-p with k=p: 4n-4-p-i = j+n-1-p => 4n-4-i = j+n-1 => i = 3n-3-j
    # So T2[j] = np.correlate(h, w[::-1], 'full')[3n-3-j]
    
    # Let me verify T1:
    # np.correlate(1-h, w, 'full')[i] = sum_k (1-h[k]) * w[k+i-(2n-2)]
    # I want: sum_p (1-h[p]) * w[p-j+(n-1)]
    # Set k=p: w[p+i-(2n-2)] = w[p-j+(n-1)] => i-(2n-2) = -j+(n-1) => i = 2n-2-j+n-1 = 3n-3-j
    # So T1[j] = np.correlate(1-h, w, 'full')[3n-3-j] ✓
    
    # Wait, but both T1 and T2 give index 3n-3-j? Let me recheck T2.
    # T2[j] = sum_m w[m] * h[j+(n-1)-m]
    # Let p = j+(n-1)-m, so m = j+(n-1)-p
    # = sum_p h[p] * w[j+n-1-p]
    # Using np.correlate(h, w_rev, 'full')[i] = sum_k h[k] * w_rev[k+i-(2n-2)] = sum_k h[k] * w[2n-2-k-i+(2n-2)]
    # Wait, w_rev[q] = w[2n-2-q], so w_rev[k+i-(2n-2)] = w[2n-2-(k+i-(2n-2))] = w[4n-4-k-i]
    # I want: sum_p h[p] * w[j+n-1-p]
    # Set k=p: w[4n-4-p-i] = w[j+n-1-p] => 4n-4-p-i = j+n-1-p => 4n-4-i = j+n-1 => i = 3n-3-j
    # So T2[j] = np.correlate(h, w[::-1], 'full')[3n-3-j]
    
    # But wait, the validity conditions are different for the two!
    # For T1: np.correlate(1-h, w, 'full')[3n-3-j]
    #   1-h has length n, w has length 2n-1
    #   Index 3n-3-j must be in [0, n+2n-2-1] = [0, 3n-3] ✓ (since j in [0,n-1])
    #   At index 3n-3-j: sum_k (1-h[k]) * w[k+(3n-3-j)-(2n-2)] = sum_k (1-h[k]) * w[k+n-1-j]
    #   For this to be nonzero: 0 <= k+n-1-j <= 2n-2, i.e., j-(n-1) <= k <= j+n-1
    #   Since k in [0, n-1]: max(0, j-n+1) <= k <= min(n-1, j+n-1) = n-1
    #   
    #   And I wanted: sum_p (1-h[p]) * w[p-j+(n-1)]
    #   Setting k=p: w[k-j+(n-1)] ✓ Same thing!
    
    # For T2: np.correlate(h, w[::-1], 'full')[3n-3-j]
    #   h has length n, w[::-1] has length 2n-1
    #   At index 3n-3-j: sum_k h[k] * w_rev[k+(3n-3-j)-(2n-2)] = sum_k h[k] * w_rev[k+n-1-j]
    #   w_rev[k+n-1-j] = w[2n-2-(k+n-1-j)] = w[n-1-k+j]
    #   So T2[j] = sum_k h[k] * w[n-1-k+j]
    #   
    #   I wanted: sum_p h[p] * w[j+n-1-p]
    #   Setting k=p: w[n-1-p+j] = w[j+n-1-p] ✓ Same thing!
    
    # Great! So:
    # grad[j] = T1[j] - T2[j]
    # = np.correlate(1-h, w, 'full')[3n-3-j] - np.correlate(h, w[::-1], 'full')[3n-3-j]
    
    # I can compute both correlations at once (each takes O(n log n) with FFT):
    
    corr1 = np.correlate(1.0 - h, w, mode='full')
    corr2 = np.correlate(h, w[::-1], mode='full')
    
    idx = 3*n - 3 - np.arange(n)
    grad = corr1[idx] - corr2[idx]
    
    return grad
```

This is O(n log n) per gradient evaluation, which is much better than O(n²). For n=600, this should be very fast (a few microseconds).

Now let me put together the full algorithm:

1. Start from initial_h_values
2. Binary projection + local search
3. L-BFGS-B with analytical gradient (using the log-sum-exp)
4. Try at multiple alpha values
5. Try at n=600 and n=1200 (upsampled)
6. Return best

Let me also add a tabu-like component to the local search to avoid cycling.

Let me now write the complete code:

```python
import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum-target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum-final_sum]] = 1
    return h_r.astype(float)

def compute_smooth_gradient(h, alpha):
    """Compute gradient of log-sum-exp approximation using FFT-based correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(corr - c_max, -100.0, 0.0)
    exp_shifted = np.exp(alpha * shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    corr1 = np.correlate(1.0 - h, w, mode='full')
    corr2 = np.correlate(h, w[::-1], mode='full')
    
    idx = 3 * n - 3 - np.arange(n)
    grad = corr1[idx] - corr2[idx]
    return grad

def compute_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -100.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def lbfgs_solve(h_start, alpha, maxiter, time_limit, start_time):
    """Solve using L-BFGS-B with penalty for sum constraint."""
    n = len(h_start)
    target = n / 2.0
    
    def obj(h):
        val = compute_smooth_obj(h, alpha)
        penalty = 1e4 * (np.sum(h) - target) ** 2
        return val + penalty
    
    def grad(h):
        g = compute_smooth_gradient(h, alpha)
        g += 2e4 * (np.sum(h) - target) * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    
    def callback(res):
        if time.time() - start_time > time_limit:
            res['status'] = 8  # signal to stop
            res['message'] = 'Time limit'
    
    try:
        res = minimize(obj, h_start, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                              'gtol': 1e-8, 'callback': callback,
                                              'maxfun': 50000})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

def global_vars_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_cons_alpha * (corr - c_max), -100.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / _cons_alpha)

def global_vars_grad(h):
    return compute_smooth_gradient(h, _cons_alpha)

def slsqp_refine(h_start, alpha, maxiter, time_limit, start_time):
    global _cons_n, _cons_target, _cons_alpha
    n = len(h_start)
    target = n / 2.0
    _cons_n = n
    _cons_target = target
    _cons_alpha = alpha
    
    bounds = [(0.0, 1.0)] * n
    
    def con_fun(h):
        return np.sum(h) - target
    
    def con_jac(h):
        return np.ones(n)
    
    try:
        res = minimize(global_vars_obj, h_start, jac=global_vars_grad,
                      method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)

def local_search_binary(h, max_iters, rng, time_limit, start_time, tabu_len=200):
    """Binary local search with tabu list."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    tabu = {}  # position -> iteration when it becomes non-tabu
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates for the critical shift
        i_cand = [i for i in ones if 0 <= i + k_star < n]
        j_cand = [j for j in zeros if 0 <= j + k_star < n]
        
        # Also consider all positions for diversity
        all_i = list(ones)
        all_j = list(zeros)
        
        # Try several random swaps, keep best
        best_delta = 0
        best_i, best_j = -1, -1
        best_new_corr = None
        
        n_try = 50
        for _ in range(n_try):
            if it % 3 == 0 and i_cand and j_cand:
                i = i_cand[rng.randint(len(i_cand))]
                j = j_cand[rng.randint(len(j_cand))]
            else:
                i = all_i[rng.randint(len(all_i))]
                j = all_j[rng.randint(len(all_j))]
            
            if i == j:
                continue
            # Tabu check
            if (it - tabu.get(i, -10000)) < tabu_len or (it - tabu.get(j, -10000)) < tabu_len:
                continue
            
            h[i] = 0
            h[j] = 1
            new_corr = compute_corr(h)
            new_max = float(new_corr.max())
            delta = max_val - new_max
            
            if delta > best_delta:
                best_delta = delta
                best_i, best_j = i, j
                best_new_corr = new_corr.copy()
            
            h[i] = 1
            h[j] = 0
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            tabu[best_i] = it
            tabu[best_j] = it
            corr = best_new_corr
            new_c5 = float(corr.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Perturbation
            n_pert = rng.randint(2, 6)
            for _ in range(n_pert):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = compute_corr(h)
    
    return best_h, best_c5

def sa_binary(h, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    """Simulated annealing for binary h."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 10000:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Focused move: target the maximum correlation
        if rng.random() < 0.7:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if i_pool and j_pool:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        corr_new = compute_corr(h)
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    target = n_dst / 2.0
    return project_feasible(h_out, target)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target, _cons_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2
    
    # Start from best known
    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)
    
    # Phase 1: Binary local search with tabu
    t1 = min(120, tleft() * 0.15)
    h_ls, c5_ls = local_search_binary(best_h, 100000, rng, t1, start_time, tabu_len=100)
    if c5_ls < best_score:
        best_score = c5_ls
        best_h = h_ls.copy()
    
    # Phase 2: SA
    if tleft() > 100:
        t2 = min(200, tleft() * 0.15)
        max_it = int(t2 * 5000)
        h_sa, c5_sa = sa_binary(best_h, target, max_it, 0.012, 0.00002, rng, t2, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 3: L-BFGS-B refinement at current n
    if tleft() > 80:
        h_cont = best_h.copy()
        # Add small noise to escape binary
        h_cont = np.clip(h_cont + rng.randn(n) * 0.01, 0, 1)
        h_cont = project_feasible(h_cont, target)
        
        for alpha in [20, 50, 100, 300, 1000, 3000]:
            if tleft() < 30:
                break
            alpha_time = min(60, tleft() * 0.3)
            h_new = lbfgs_solve(h_cont, alpha, maxiter=1000, time_limit=alpha_time, start_time=start_time)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
            h_cont = h_new.copy()
    
    # Phase 4: ILS on best binary solution
    for ils_iter in range(6):
        if tleft() < 60:
            break
        # Perturb
        h_pert = project_binary(best_h, target)
        n_pert = max(5, 8 + ils_iter * 3)
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
        
        t_ils = min(40, tleft() * 0.1)
        h_ig, c5_ig = local_search_binary(h_pert, 50000, rng, t_ils, start_time, tabu_len=50)
        if c5_ig < best_score:
            best_score = c5_ig
            best_h = h_ig.copy()
    
    # Phase 5: SLSQP refinement
    if tleft() > 50:
        h_cont = best_h.copy()
        h_cont = np.clip(h_cont + rng.randn(len(h_cont)) * 0.005, 0, 1)
        h_cont = project_feasible(h_cont, len(h_cont) / 2.0)
        
        for alpha in [200, 500, 1500, 5000]:
            if tleft() < 20:
                break
            slsqp_time = min(40, tleft() * 0.25)
            h_slsqp = slsqp_refine(h_cont, alpha, maxiter=2000, time_limit=slsqp_time, start_time=start_time)
            score = compute_c5(h_slsqp)
            if score < best_score:
                best_score = score
                best_h = h_slsqp.copy()
            h_cont = h_slsqp.copy()
    
    # Phase 6: Try upsampled version
    if tleft() > 80:
        n2 = 2 * n  # double resolution
        h_up = interpolate_to(best_h, n2)
        target2 = n2 // 2
        h_up_bin = project_binary(h_up, target2)
        
        # Local search at higher resolution
        t_up = min(60, tleft() * 0.2)
        h_up_ls, c5_up = local_search_binary(h_up_bin, 30000, rng, t_up, start_time, tabu_len=100)
        
        # L-BFGS-B at higher resolution
        if tleft() > 40:
            h_up_cont = np.clip(h_up_ls + rng.randn(n2) * 0.005, 0, 1)
            h_up_cont = project_feasible(h_up_cont, target2)
            for alpha in [50, 200, 1000]:
                if tleft() < 30:
                    break
                h_up_new = lbfgs_solve(h_up_cont, alpha, maxiter=500, time_limit=min(30, tleft()*0.3), start_time=start_time)
                score = compute_c5(h_up_new)
                if score < best_score:
                    best_score = score
                    best_h = h_up_new.copy()
                h_up_cont = h_up_new.copy()
    
    # Final: ensure feasibility
```
