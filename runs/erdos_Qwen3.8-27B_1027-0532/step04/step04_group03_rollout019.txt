                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

max_k (A_k * v² + B_k * v + C_k)  (quadratic case, when j and l are k apart)

For the linear case, the minimum of max of lines is at a breakpoint (where two lines cross). We can find it by checking all pairs of lines.

But with up to 2n-1 lags, checking all pairs is O(n²) per coordinate move, which is too expensive.

Instead, let me just use a simple approach: evaluate at a few candidate v values (e.g., the current value ± small perturbations) and keep the best. This is essentially a heuristic coordinate descent.

Actually, the simplest effective approach is:
1. For each coordinate j, try moving it to 0, 0.5, or 1 (adjusting one other coordinate l to maintain the sum)
2. Keep the move if it improves the objective

This is what `local_fractional_perturb` does, but I can make it more efficient by:
- Being smarter about which pairs (j,l) to try
- Using a priority queue based on the gradient

3. SLSQP log-sum-exp (300s) → continuous refinement
4. Coordinate descent with pair moves (200s) → further refinement
5. SLSQP again from new starting point (100s)
6. Multi-resolution (200s) → try different n
7. Final SLSQP polish (100s)
8. Coordinate descent polish (50s)
9. Return best

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == int(target_sum):
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    target_int = int(target_sum)
    if current_sum > target_int:
        need = current_sum - target_int
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_int - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_int:
        if final_sum > target_int:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_int
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_int - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)


def sa_optimize(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(2000, max_iters // 15)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]

    for i in range(max_iters):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.03:
                no_improve = 0
            else:
                break

        ratio = max(i, 1) / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        move_type = rng.randint(6)
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h[i1] = 0
            h[i0] = 1
            moved = True
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            h[ones_cache[idx1[0]]] = 0
            h[ones_cache[idx1[1]]] = 0
            h[zeros_cache[idx0[0]]] = 1
            h[zeros_cache[idx0[1]]] = 1
            moved = True
        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            r = rng.randint(1, max(2, n // 30))
            i0_pos = (i1 + r) % n
            if h[i0_pos] == 1:
                i0_pos = (i1 - r) % n
                if h[i0_pos] == 1:
                    i0_pos = zeros_cache[rng.randint(len(zeros_cache))]
            h[i1] = 0
            h[i0_pos] = 1
            moved = True
        elif move_type == 3 and len(ones_cache) >= 3 and len(zeros_cache) >= 3:
            idx1 = rng.choice(len(ones_cache), 3, replace=False)
            idx0 = rng.choice(len(zeros_cache), 3, replace=False)
            for idx in idx1:
                h[ones_cache[idx]] = 0
            for idx in idx0:
                h[zeros_cache[idx]] = 1
            moved = True
        elif move_type == 4 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            block_size = rng.randint(2, 6)
            i0_pos = zeros_cache[rng.randint(len(zeros_cache))]
            h[i1] = 0
            h[i0_pos] = 1
            for ext in range(1, block_size):
                ni1 = (i1 + ext) % n
                ni0 = (i0_pos + ext) % n
                if h[ni1] == 1 and h[ni0] == 0:
                    h[ni1] = 0
                    h[ni0] = 1
            moved = True
        elif move_type == 5 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            i1a = ones_cache[rng.randint(len(ones_cache))]
            i1b = (i1a + rng.randint(1, max(2, n // 20))) % n
            if h[i1b] != 1:
                i1b = ones_cache[rng.randint(len(ones_cache))]
            i0a = zeros_cache[rng.randint(len(zeros_cache))]
            i0b = (i0a + rng.randint(1, max(2, n // 20))) % n
            if h[i0b] != 0:
                i0b = zeros_cache[rng.randint(len(zeros_cache))]
            h[i1a] = 0
            h[i1b] = 0
            h[i0a] = 1
            h[i0b] = 1
            if h[i1b] != 0:
                h[i1b] = 0
                h[i0a] = 0
                h[i1a] = 1
                moved = False
            else:
                moved = True

        if not moved:
            continue

        c5_new = compute_c5(h)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h = best_h.copy() if rng.random() < 0.1 else h
            h[ones_cache[rng.randint(len(ones_cache))]] = 0
            h[zeros_cache[rng.randint(len(zeros_cache))]] = 1
            h = project_binary(h, target_sum)
            no_improve += 1

    return best_h, best_c5


def hill_climb_binary(h, target_sum, max_iter, rng):
    n = len(h)
    h = project_binary(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 500

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0
        h[i0] = 1
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[i1] = 1
            h[i0] = 0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5


def cons_fun(x):
    return np.sum(x) - _cons_target


def cons_jac(x):
    return np.ones(_cons_n)


def soft_max_obj(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def soft_max_grad(x, alpha, n, target):
    h = project_feasible(x, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / w_sum
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return grad


def slsqp_refine(h_start, target, time_limit, start_time, alpha_list=None, maxiter_per_alpha=400):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = target
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    if alpha_list is None:
        alpha_list = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 15000]

    for alpha in alpha_list:
        if (time.time() - start_time) > time_limit:
            break
        time_per_alpha = max(5.0, time_limit / len(alpha_list))
        try:
            res = minimize(
                soft_max_obj, h, args=(alpha, n, target),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter_per_alpha, 'ftol': 1e-18}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5


def subgradient_refine(h_start, target, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.03
    no_improve = 0
    max_no_improve = 1000
    best_eval = best_c5

    for it in range(500000):
        if (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = compute_corr_scaled(h)
        c_max = float(np.max(corr))
        threshold = c_max * 0.99
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            if m >= 0:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
            else:
                mm = -m
                grad[:n - mm] += (1.0 - h[mm:])
                grad[mm:] -= h[:n - mm]
        grad /= len(active)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        grad_normed = grad / grad_norm
        h_try = project_feasible(h - lr * grad_normed, target)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            no_improve = 0
            lr = min(lr * 1.03, 0.3)
        else:
            h_try2 = project_feasible(h - lr * 0.3 * grad_normed, target)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
                h = h_try2
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.97
                if lr < 1e-8:
                    break

    return best_h, best_c5


def coord_descent_pair(h_start, target, time_limit, start_time, n_rounds=50):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng_local = np.random.RandomState(12345)

    for rnd in range(n_rounds):
        if (time.time() - start_time) > time_limit:
            break
        improved = False
        n_trials = max(200, n)
        for _ in range(n_trials):
            if (time.time() - start_time) > time_limit:
                break
            j = rng_local.randint(n)
            l = rng_local.randint(n)
            if j == l:
                continue
            hj = h[j]
            hl = h[l]
            v_lo = max(0.0, hl + hj - 1.0)
            v_hi = min(1.0, hj)
            if v_hi - v_lo < 1e-8:
                continue
            n_samples = 5
            candidates = np.linspace(v_lo, v_hi, n_samples + 2)[1:-1]
            best_v = hj
            best_val = compute_c5(h)
            for v in candidates:
                h[j] = v
                h[l] = hl + (hj - v)
                h[j] = max(0.0, min(1.0, h[j]))
                h[l] = max(0.0, min(1.0, h[l]))
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_v = v
            if best_v != hj and best_val < best_c5 - 1e-12:
                h[j] = best_v
                h[l] = np.clip(hl + (hj - best_v), 0.0, 1.0)
                h = project_feasible(h, target)
                c5_now = compute_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                    improved = True
            else:
                h[j] = hj
                h[l] = hl
        if not improved:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def random_perturb_restart(h, target, rng, n_restarts=3):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    for _ in range(n_restarts):
        h_pert = h.copy()
        n_flip = rng.randint(n // 20, n // 5)
        indices = rng.choice(n, n_flip, replace=False)
        h_pert[indices] = 1.0 - h_pert[indices]
        h_pert = project_feasible(h_pert, target)
        c5_pert = compute_c5(h_pert)
        h_bin = project_binary(h_pert, target)
        c5_bin = compute_c5(h_bin)
        if c5_bin < c5_pert:
            h_pert = h_bin
            c5_pert = c5_bin
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    return best_h, best_c5

                                                 maxiter_per_alpha=200)
                if c5_slq_r < c5_sa_r:
                    h_sa_r = h_slq_r
                    c5_sa_r = c5_slq_r

            if c5_sa_r < best_score:
                best_score = c5_sa_r
                best_h = h_sa_r.copy()

    # Phase 8: Random restarts + SLSQP (80s)
    if tleft() > 40:
        h_rst, c5_rst = random_perturb_restart(best_h, target, rng, n_restarts=5)
        if c5_rst < best_score:
            best_score = c5_rst
            best_h = h_rst.copy()

        if tleft() > 30:
            time_allow_rst = min(50.0, max(10.0, tleft() * 0.2))
            h_slq_rst, c5_slq_rst = slsqp_refine(best_h, target, time_allow_rst, start_time,
                                                 alpha_list=[200, 500, 1000, 2000, 5000],
                                                 maxiter_per_alpha=200)
            if c5_slq_rst < best_score:
                best_score = c5_slq_rst
                best_h = h_slq_rst.copy()

    # Phase 9: Final coordinate descent polish (40s)
    if tleft() > 20:
        time_allow_final = min(40.0, max(5.0, tleft() * 0.2))
        h_cd2, c5_cd2 = coord_descent_pair(best_h, target, time_allow_final, start_time, n_rounds=20)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2.copy()

    # Phase 10: Final SLSQP with very high alpha (30s)
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        time_allow_fin = min(30.0, max(5.0, tleft() * 0.3))
        h_fin, c5_fin = slsqp_refine(best_h, target_use, time_allow_fin, start_time,
                                     alpha_list=[2000, 5000, 10000, 20000, 40000],
                                     maxiter_per_alpha=150)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-12:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n_final
