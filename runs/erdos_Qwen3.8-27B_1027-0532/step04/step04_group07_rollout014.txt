The "kinks" occur when two lags have equal correlation values. At a kink, the subgradient is a convex combination of the individual gradients. import numpy as np
from scipy.optimize import minimize, linprog
import time

_SLP_N = 0
_SLP_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def smooth_max_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    c_ref = float(np.max(top_vals))
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    return c_ref + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_max_grad(x, alpha, topk):

    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]

        top_vals = corr

    total = float(np.sum(exp_shifted))

        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = exp_shifted / total
    return compute_grad_multi_lag(x, top_idx, weights)

def _slp_eq_fun(x):

def _slp_eq_jac(x):
    return np.ones(_SLP_N)

def slsqp_optimize(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _SLP_N, _SLP_TARGET
    _SLP_N = len(h_start)
    _SLP_TARGET = target_sum

        try:

                smooth_max_obj, h_cur, args=(alpha, topk),

def lp_linearized_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]

        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None

    lr = lr_init
    topk = int(topk_init)
    alpha = alpha_init
    stagnation = 0
    t0 = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        if it > 0 and it % 150 == 149:
            topk = max(3, int(topk * topk_decay))

        corr = overlap_corrs(h)
