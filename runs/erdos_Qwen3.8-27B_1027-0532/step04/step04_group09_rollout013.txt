```python
import numpy as np
from scipy.optimize import minimize
import time

_PM_PARAMS = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _pm_fun(x):
    p = _PM_PARAMS['p']
    lam = _PM_PARAMS['lam']
    target = _PM_PARAMS['target']
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    g = np.maximum(c * dx, 1e-30)
    Sp = np.sum(g ** p)
    return Sp ** (1.0 / p) + lam * (np.sum(x) - target) ** 2

def _pm_jac(x):
    p = _PM_PARAMS['p']
    lam = _PM_PARAMS['lam']
    target = _PM_PARAMS['target']
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')

    for idx in range(2 * n - 1):

        if wi < 1e-25:
            continue
        k = idx - (n - 1)
        if k >= 0:

            else:
                grad[k:] += wi * (1.0 - x[:n - k])
                grad[:n - k] -= wi * x[k:]

            grad[:n - kk] += wi * (1.0 - x[kk:])
            grad[kk:] -= x[:n - kk]

def subgrad_polish(h, target, n_steps, lr, active_frac, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        g = c * dx
        gmax = float(np.max(g))
        gmin = float(np.min(g))
        rng_ = gmax - gmin
        if rng_ < 1e-15:
            break
        thr = gmax - active_frac * rng_
        active = np.where(g >= thr)[0]
        na = len(active)
        if na == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] += (1.0 - 2.0 * h_cur)
                else:
                    grad[k:] += (1.0 - h_cur[:n - k])
                    grad[:n - k] -= h_cur[k:]
            else:
                kk = -k
                grad[:n - kk] += (1.0 - h_cur[kk:])
                grad[kk:] -= h_cur[:n - kk]
        grad /= na
        step = lr / (1.0 + 0.005 * it)
        h_new = project_feasible(h_cur - step * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
        else:
            # backtracking
            for _ in range(8):
                step *= 0.5
                h_new = project_feasible(h_cur - step * grad, target)
                sc = compute_c5(h_new)
                if sc < compute_c5(h_cur):
                    if sc < best_score:
                        best_score = sc
                        best_h = h_new.copy()
                    h_cur = h_new
                    break
    return best_h, best_score

    h_cur = h.copy()
    best_score = compute_c5(h_cur)

        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        g = c * dx
        gmax = float(np.max(g))
        gmin = float(np.min(g))
        thr = gmax - active_frac * (gmax - gmin)
        active_set = set(np.where(g >= thr)[0].tolist())
        for i in range(n):
            if time.time() > deadline:
                break
            old = h_cur[i]
            best_val = old
            best_sc = best_score
            dvals = np.array([0.01, 0.02, 0.05, 0.1])

            if best_val != old:
                h_cur[i] = best_val
                h_cur = project_feasible(h_cur, target)
                if best_sc < best_score:
                    best_score = best_sc

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
