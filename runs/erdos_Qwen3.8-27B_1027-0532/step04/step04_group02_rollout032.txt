    zeros_idx = np.where(h_cur == 0)[0]

    for _ in range(num_perturb):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break

        n_swap = rng.randint(2, 5)
        if len(ones_idx) >= n_swap and len(zeros_idx) >= n_swap:
            swap_ones = rng.choice(ones_idx, n_swap, replace=False)
            swap_zeros = rng.choice(zeros_idx, n_swap, replace=False)
            h_cur[swap_ones] = 0.0
            h_cur[swap_zeros] = 1.0
            h_cur = project_binary(h_cur, target)
            ones_idx = np.where(h_cur == 1)[0]
            zeros_idx = np.where(h_cur == 0)[0]

        h_new, c5_new = vectorized_best_swap(h_cur, max_passes=80,
                                             time_limit=time_limit, start_time=start_time)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h_cur = h_new
            ones_idx = np.where(h_cur == 1)[0]
            zeros_idx = np.where(h_cur == 0)[0]

    return h_cur, best_c5


def lse_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_raw(h) * dx
    shifted = alpha * (corr - np.max(corr))
    return float(np.max(corr) + np.log(np.sum(np.exp(shifted))) / alpha)


def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    s_range = np.arange(-(n - 1), n)
    corr_raw = compute_corr_raw(h)
    corr = corr_raw * dx
    shifted = alpha * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)

    grad = np.zeros(n)
    for i in range(n):
        ip = i + s_range
        im = i - s_range
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        d_pos = np.zeros(2 * n - 1)
        d_pos[vp] = 1.0 - h[ip[vp]]
        d_neg = np.zeros(2 * n - 1)
        d_neg[vm] = -h[im[vm]]
        grad[i] = np.dot(weights, d_pos + d_neg) * dx
    return grad


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_lse(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(lambda x: lse_objective(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def top_p_objective(h, p, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_raw(h) * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_top_p(h_start, n, target, p_vals, alphas, maxiter, ftol,
                time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(lambda x: top_p_objective(x, p, alpha), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_pattern(n, block_size, rng):
    h = np.zeros(n)
    for i in range(0, n, block_size):
        if (i // block_size) % 2 == 0:
            h[i:min(i + block_size, n)] = 1.0
    h = project_binary(h, n // 2)
    return h


def thue_morse_binary(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = float(np.binary_repr(i).count('1') % 2)
    return project_binary(t, n // 2)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    h_ls, c5_ls = vectorized_best_swap(best_h, max_passes=100,
                                       time_limit=min(60.0, tleft() * 0.3),
                                       start_time=start_time)
    if c5_ls < best_score:
        best_score = c5_ls
        best_h = h_ls.copy()

    # Phase 1: Iterated perturbation at initial resolution
    num_iters = max(5, int(tleft() * 0.12))
    for it in range(num_iters):
        if tleft() < 60:
            break
        h_pert, c5_pert = perturb_and_optimize(best_h, rng, num_perturb=1,
                                               time_limit=min(40.0, tleft() * 0.1),
                                               start_time=start_time)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert.copy()

    # SLSQP refinement at current resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0

    if tleft() > 80:
        alphas = [300, 800, 2000, 5000]
        p_vals = [max(3, n_cur // 60), max(5, n_cur // 30), max(8, n_cur // 15)]
        h_ref, sc_ref = slsqp_top_p(best_h, n_cur, target_cur, p_vals, alphas,
                                    400, 1e-14,
                                    time_limit=min(120.0, tleft() * 0.25),
                                    start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

        if tleft() > 40:
            h_bin = project_binary(best_h, target_cur)
            h_ls2, c5_ls2 = vectorized_best_swap(h_bin, max_passes=60,
                                                 time_limit=min(40.0, tleft() * 0.3),
                                                 start_time=start_time)
            if c5_ls2 < best_score:
                best_score = c5_ls2
                best_h = h_ls2.copy()

    # Multi-resolution search
    n_candidates = []
    for nf in [1.25, 1.5, 2.0, 0.75]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try != n_init and n_try >= 200:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        h_sa, c5_sa = vectorized_best_swap(h_bin_try, max_passes=60,
                                           time_limit=min(50.0, tleft() * 0.3),
                                           start_time=start_time)
        if tleft() > 30:
            h_pert2, c5_pert2 = perturb_and_optimize(h_sa, rng, num_perturb=3,
                                                     time_limit=min(30.0, tleft() * 0.2),
                                                     start_time=start_time)
            if c5_pert2 < c5_sa:
                h_sa = h_pert2
                c5_sa = c5_pert2
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Additional SLSQP at best resolution
    n_best = len(best_h)
    target_best = n_best / 2.0

    if tleft() > 60:
        alphas_fine = [1000, 3000, 8000]
        p_fine = [max(3, n_best // 50), max(6, n_best // 20)]
        h_ref3, sc3 = slsqp_top_p(best_h, n_best, target_best, p_fine, alphas_fine,
                                  300, 1e-15,
                                  time_limit=min(50.0, tleft() * 0.3),
                                  start_time=start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()
            h_final_bin = project_binary(best_h, target_best)
            h_ls3, c5_ls3 = vectorized_best_swap(h_final_bin, max_passes=40,
                                                 time_limit=min(20.0, tleft() * 0.2),
                                                 start_time=start_time)
            if c5_ls3 < best_score:
                best_score = c5_ls3
                best_h = h_ls3.copy()

    # Final perturbation rounds at best resolution
    if tleft() > 30:
        num_final = max(2, int(tleft() * 0.06))
        for _ in range(num_final):
            if tleft() < 15:
                break
            h_pert_f, c5_f = perturb_and_optimize(best_h, rng, num_perturb=1,
                                                  time_limit=min(20.0, tleft() * 0.5),
                                                  start_time=start_time)
            if c5_f < best_score:
                best_score = c5_f
                best_h = h_pert_f.copy()

    # Final local search
    if tleft() > 15:
        h_bin_final = project_binary(best_h, len(best_h) / 2.0)
        h_ls_final, c5_final = vectorized_best_swap(h_bin_final, max_passes=50,
                                                    time_limit=max(5.0, tleft() - 5),
                                                    start_time=start_time)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_ls_final.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_out = project_binary(best_h, target_final)
    final_score = compute_c5(h_out)

    if final_score > best_score + 1e-10:
        h_out = best_h
        final_score = compute_c5(h_out)

    return h_out, final_score, n_final
