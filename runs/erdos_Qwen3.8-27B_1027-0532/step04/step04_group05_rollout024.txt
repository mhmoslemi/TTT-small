def alternating_lp_cutting(h_init, n, target, max_iter, time_limit, M=50, damping_init=0.6):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute current correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        
        # Select top M lags
        sorted_idx = np.argsort(corr)[::-1]
        active_lags = sorted_idx[:M]
        
        # Set up LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        g = 1.0 - h  # fixed g
        n_half = n - 1
        
        for lag_idx in active_lags:
            k = lag_idx - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Adaptive damping
        if it % 10 == 0:
            damping = max(0.3, damping * 0.98)
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Second pass for residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp(h_current, n, target, active_lags, reg, h_ref):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    g = 1.0 - h_current
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    n_half = n - 1
    for k_idx in active_lags:
        k = k_idx - n_half
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h_ref)
    else:
        objective = t_var
    
    prob = cp.Problem(cp.Minimize(objective), constraints)
    
    solvers = [cp.GLOP, cp.CBC, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.SCS:
                kwargs = {'max_iters': 50000}
            prob.solve(solver=solver, verbose=False, **kwargs)
            if h_var.value is not None:
                return np.asarray(h_var.value).ravel()
        except:
            continue
    return None

def solve_lp_direct(h_current, n, target, active_lags, reg, h_ref):
    dx = 2.0 / n
    g = 1.0 - h_current
    n_half = n - 1
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t + reg * sum((h_i - h_ref_i)^2)
    # For the linear part: minimize t (the quadratic is handled by the proximal)
    # Actually, for linprog, we can only do linear objectives.
    # The proximal term makes it quadratic, so we can't use linprog with reg > 0.
    # Let's just do the LP (reg = 0) and handle the proximal step separately.
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # h_i >= 0: -h_i <= 0
    # h_i <= 1: h_i <= 1
    # For each active lag k: sum_i coeff_i * h_i - t <= 0
    
    n_constraints = 2 * n + len(active_lags)
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    # -h_i <= 0
    A_ub[:n, :n] = -np.eye(n)
    b_ub[:n] = 0
    
    # h_i <= 1
    A_ub[n:2*n, :n] = np.eye(n)
    b_ub[n:2*n] = 1.0
    
    # Correlation constraints
    for i, k_idx in enumerate(active_lags):
        k = k_idx - n_half
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            row = 2*n + i
            A_ub[row, lo:hi] = g[lo-k:hi-k] * dx
            A_ub[row, n] = -1.0  # -t
            b_ub[row] = 0.0
    
    # Equality constraint: sum(h_i) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(None, None)]  # t is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

    # 2. h_i <= 1 (n constraints)
    # 3. sum(coeff_i * h_i) - t <= 0 (n_active constraints)
