        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_func_and_grad(x, alpha, target):
    """Log-sum-exp objective with equality constraint via augmented Lagrangian."""
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)  # softmax weights
    f_val = c_max + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient of LSE with respect to x
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        i_start = max(0, -m)
        i_end = min(n - 1, n - 1 - m)
        if i_start > i_end:
            continue
        if m == 0:
            vals = 1.0 - 2.0 * x[i_start:i_end+1]
        else:
            vals = np.full(i_end - i_start + 1, 1.0)
            idx = np.arange(i_start, i_end + 1)
            im = idx - m
            valid_m = (im >= 0) & (im < n)
            vals[valid_m] -= x[im[valid_m]]
            ip = idx + m
            valid_p = (ip >= 0) & (ip < n)
            vals[valid_p] -= x[ip[valid_p]]
        grad[i_start:i_end+1] += wk * vals
    return f_val, grad

def aug_lagrangian_minimize(x0, n, target, alpha, max_iter=200, rho=1.0, lambda_init=0.0):
    """Augmented Lagrangian with L-BFGS-B."""
    x = x0.copy()
    lam = lambda_init
    for it in range(max_iter):
        def obj(x):
            f, _ = lse_func_and_grad(x, alpha, target)
            c = np.sum(x) - target
            return f + lam * c + 0.5 * rho * c * c
        
        def grad(x):
            _, g = lse_func_and_grad(x, alpha, target)
            c = np.sum(x) - target
            return g + lam * np.ones(n) + rho * c * np.ones(n)
        
        res = minimize(obj, x, jac=grad, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': 50, 'gtol': 1e-8})
        x = res.x
        c = np.sum(x) - target
        lam = lam + rho * c
        if abs(c) < 1e-8:
            break
        rho = min(rho * 2, 1e6)
    return x

def slsqp_optimize(x0, n, target, alpha, maxiter=2000):
    """SLSQP optimization with smooth max."""
    def obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def con_eq(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, target)

def block_descent(h, n, target, block_size, alpha, n_rounds, deadline):
    """Block coordinate descent targeting high-correlation regions."""
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        # Identify which blocks contribute most to the max correlation
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        # For each block, estimate its contribution to the max
        block_scores = np.zeros(n_blocks)
        for bi in range(n_blocks):
            s = bi * block_size
            e = s + block_size
            # Contributions of block [s,e] to corr
            # A variable h[i] contributes to corr[k] when i or i+m is in [s,e]
            # Simplification: just use the current correlation values
            block_scores[bi] = np.max(np.abs(h_cur[s:e] - 0.5)) * (c_max - np.mean(corr) + 1e-10)
        
        # Process blocks in order of score
        order = np.argsort(-block_scores)
        for bi in order:
            if time.time() > deadline:
                break
            s = int(bi) * block_size
            e = s + block_size
            block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            
            def block_obj(x):
                hf = h_cur.copy()
                hf[s:e] = np.clip(x, 0, 1)
                corr = compute_corr(hf)
                cm = float(np.max(corr))
                shifted = np.clip(alpha * (corr - cm), -700.0, 0.0)
                return cm + float(np.log(np.sum(np.exp(shifted)))) / alpha
            
            def block_con(x):
                return np.sum(x) - block_target
            
            bounds = [(0, 1)] * block_size
            cons = [{'type': 'eq', 'fun': block_con, 'jac': lambda x: np.ones(block_size)}]
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0, 1)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def periodic_h(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        h[:total - n//2] = 0.0

        deficit = n // 2 - total
        indices = np.where(h == 0)[0]
        for j in range(min(deficit, len(indices))):
            h[indices[j]] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value_and_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    f_val = c_max + float(np.log(np.sum(exp_vals))) / alpha
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        m = k - (n - 1)
        n_lo = max(0, -m)
        n_hi = min(n - 1, n - 1 - m)
        if n_lo > n_hi:
            continue
        n_vals = np.arange(n_lo, n_hi + 1)
        j_vals = n_vals + m
        grad[j_vals] += wk * (1.0 - x[n_vals])
        grad[n_vals] -= wk * x[j_vals]
    return f_val, grad

def aug_lagrangian_minimize(x0, n, target, alpha, max_outer=150, rho_init=1.0):
    x = x0.copy()
    lam = 0.0
    rho = rho_init
    best_x = x.copy()
    best_f = compute_c5(x)
    for outer in range(max_outer):
        def obj(x):
            f, _ = lse_value_and_grad(x, alpha)
            c = np.sum(x) - target
            return f + lam * c + 0.5 * rho * c * c

        def grad(x):
            _, g = lse_value_and_grad(x, alpha)
            c = np.sum(x) - target
            return g + (lam + rho * c)

        try:
            res = minimize(obj, x, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 80, 'gtol': 1e-9, 'ftol': 1e-14})
            x = res.x
        except Exception:
            break
        c = np.sum(x) - target
        lam = lam + rho * c
        if abs(c) < 1e-10:
            break
        rho = min(rho * 1.5, 1e5)
        x = project_feasible(x, target)
        sc = compute_c5(x)
        if sc < best_f:
            best_f = sc
            best_x = x.copy()
    return best_x, best_f

def slsqp_optimize(x0, n, target, alpha, maxiter=2000):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def con_eq(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return x0

def block_descent_targeted(h, n, target, block_size, alpha, n_rounds, deadline):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        block_scores = np.zeros(n_blocks)
        for bi in range(n_blocks):
            s = bi * block_size
            e = s + block_size
            block_scores[bi] = np.sum(np.abs(h_cur[s:e] - 0.5))
        order = np.argsort(-block_scores)
        for bi in order:
            if time.time() > deadline:
                break
            s = int(bi) * block_size
            e = s + block_size
            block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])

            def bobj(x):
                hf = h_cur.copy()
                hf[s:e] = np.clip(x, 0, 1)
                corr_b = compute_corr(hf)
                cm = float(np.max(corr_b))
                shifted_b = np.clip(alpha * (corr_b - cm), -700.0, 0.0)
                return cm + float(np.log(np.sum(np.exp(shifted_b)))) / alpha

            def bcon(x):
                return np.sum(x) - block_target

            bounds_b = [(0, 1)] * block_size
            cons_b = [{'type': 'eq', 'fun': bcon, 'jac': lambda xx: np.ones(block_size)}]
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(bobj, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 400, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0, 1)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

def periodic_h(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        h[:total - n // 2] = 0.0
    elif total < n // 2:
        deficit = n // 2 - total
        indices = np.where(h == 0)[0]
        for j in range(min(deficit, len(indices))):
            h[indices[j]] = 1.0
    return h

def scale_up(h_init, target_n):
    n_prev = len(h_init)
    if target_n == n_prev:
        return h_init.copy()
    x_old = np.linspace(0.0, 2.0, n_prev, endpoint=False) + 1.0 / n_prev
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False) + 1.0 / target_n
    h_scaled = np.interp(x_new, x_old, h_init)
    return project_feasible(h_scaled, target_n // 2.0)

                best_h = h_opt

    # Phase 9: Final block descent very high alpha
    deadline9 = t0 + budget_s - 50
    if elapsed() < budget_s - 50:
        h_bd, sc = block_descent_targeted(best_h, n, target, 4, 5000, 6, deadline9)
        if sc < best_score:
            best_score = sc
            best_h = h_bd

    # Phase 10: Final AL polish
    if elapsed() < budget_s - 30:
        x, sc = aug_lagrangian_minimize(best_h, n, target, 10000, max_outer=50)
        if sc < best_score:
            best_score = sc
            best_h = x

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, n
