import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fft_correlate(h):
    n = len(h)
    comp = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(comp, N)
    result = np.fft.irfft(H * np.conj(C), N)
    return result[:2 * n - 1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h)
    return float(np.max(corr)) * dx


def smooth_c5_obj(h, alpha):
    n = len(h)
    corr = fft_correlate(h)
    cmax = corr.max()
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -500.0, 500.0)
    lse = cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha
    return lse


def smooth_c5_obj_grad(h, alpha):
    n = len(h)
    corr = fft_correlate(h)
    cmax = corr.max()
    shifted = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    exp_terms = np.exp(shifted)
    w = exp_terms / exp_terms.sum()
    obj = cmax + float(np.log(np.sum(exp_terms))) / alpha

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wv = w[k]
        if wv < 1e-15:
            continue
        if s >= 0:
            j_end = n - s
            if j_end > 0:
                grad[:j_end] += wv * (1.0 - h[s:s + j_end])
            if s < n:
                grad[s:] -= wv * h[:n - s]
        else:
            sp = -s
            if sp < n:
                grad[sp:] += wv * (1.0 - h[:n - sp])
            j_end = n - sp
            if j_end > 0:
                grad[:j_end] -= wv * h[sp:sp + j_end]

    return obj, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def slsqp_refine_fast(h_start, n, target, alphas, maxiter_per_alpha, ftol,
                      time_limit=None, start_time=None):
    dx = 2.0 / n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > start_time + time_limit:
                break

        def obj_fn(x):
            return smooth_c5_obj(x, alpha) * dx

        def jac_fn(x):
            _, g = smooth_c5_obj_grad(x, alpha)
            return g * dx

        def cons_fun(x):
            return np.sum(x) - target

        def cons_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

        try:
            res = minimize(obj_fn, h_cur, jac=jac_fn, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def sgd_smooth(h_init, target, alpha, lr, max_iter, rng,
               time_limit=None, start_time=None):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n

    for it in range(max_iter):

            best_score = c5_sa2
            best_h = h_sa2.copy()

        if tleft() > 30:
            alphas_f2 = [1000, 10000, 50000]
            allowed3 = min(25, tleft() * 0.2)
            h_f2, sc_f2 = slsqp_refine_fast(best_h, n, target, alphas_f2, 3000,
                                            1e-15, time_limit=allowed3,
                                            start_time=start_time)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_h = h_f2.copy()

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
