And O_0(t) = O_0(current) + (t-h_old) - (t²-h_old²) = (O_0(current) - h_old + h_old²) + t - t²

So the 1D problem is: minimize max(O_0(t), max_{k>0} (a_k·t + b_k)) subject to 0≤t≤1 (and approximately maintaining the sum constraint).

The O_0(t) part is a concave quadratic (coefficient of t² is -1), and the rest are affine. The max of a concave function and affine functions is not necessarily convex, but it is "nice" enough that we can find its minimum numerically.

Strategy: 
- The minimum of O_0(t) alone is at t=0 or t=1 (where it equals O_0(current) - h_old + h_old² + 0 or O_0(current) - h_old + h_old² + 1 - 1 = O_0(current) - h_old + h_old²).

Wait: O_0(t) = (O_0(current) - h_old + h_old²) + t - t²
At t=0: O_0(current) - h_old + h_old²
At t=1: O_0(current) - h_old + h_old² + 1 - 1 = O_0(current) - h_old + h_old²
At t=0.5: O_0(current) - h_old + h_old² + 0.5 - 0.25 = O_0(current) - h_old + h_old² + 0.25

So O_0(t) is maximized at t=0.5 and minimized at t=0 or t=1. This makes sense: h(1-h) is maximized at 0.5.

The overall objective max(O_0(t), max_k O_k(t)) will be minimized at a point where either:
- O_0(t) is at its minimum (t=0 or t=1) and the other terms are small, OR
- The max of the affine terms is balanced with O_0(t)

Since O_0(t) ≤ O_0(max) and the affine terms can be larger, the practical minimum is likely near t=0 or t=1 (pushing h[j] to the boundary).

This gives a strong heuristic: **push h values toward 0 or 1** (binary structure) while balancing the other k terms.

Algorithm for coordinate j:
1. Evaluate the objective at t=0 and t=1
2. If both are worse than current, try t=0.5 and find the minimum by scanning
3. Otherwise, pick the better of t=0 and t=1

Actually, a cleaner approach: just evaluate the objective at a few candidate t values (0, 1, 0.5, and the "balanced" point where the top affine term equals O_0(t)), and pick the best.

```
def bpcd(h, n, target, n_passes, m_active):
    h_cur = h.copy()
    for pass in range(n_passes):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        # Get active lags
        active = np.argsort(over)[::-1][:m_active]
        for j in range(n):
            # Compute a_k and b_k for active k's
            # and the O_0 offset
            best_t = h_cur[j]
            best_val = infinity
            # Candidate t values: 0, 1, and the balanced point
            candidates = [0.0, 1.0]
            for k in active:
                k = int(k)
                if k == 0:
                    continue
                hk_plus = h_cur[j+k] if j+k < n else 0.0
                hk_minus = h_cur[j-k] if j >= k else 0.0
                a_k = 1.0 - hk_plus - hk_minus
                if abs(a_k) > 1e-12:
                    # Point where O_k(t) = O_0(t)
                    # O_k(t) = b_k + a_k*t
                    # O_0(t) = c0 + t - t^2
                    # b_k + a_k*t = c0 + t - t^2
                    # t^2 + (a_k-1)*t + (b_k-c0) = 0
                    b_k = over[k] - h_cur[j]*a_k
                    c0 = over[0] - h_cur[j] + h_cur[j]**2
                    # t^2 + (a_k-1)*t + (b_k-c0) = 0
                    disc = (a_k-1)**2 - 4*(b_k-c0)
                    if disc >= 0:
                        t1 = (-(a_k-1) + sqrt(disc)) / 2
                        t2 = (-(a_k-1) - sqrt(disc)) / 2
                        if 0 <= t1 <= 1: candidates.append(t1)
                        if 0 <= t2 <= 1: candidates.append(t2)
            # Evaluate at each candidate
            for t in candidates:
                val = eval_1d(h_cur, j, t, over, active)
                if val < best_val:
                    best_val = val
                    best_t = t
            h_cur[j] = best_t
        # Renormalize
        h_cur = project_feasible(h_cur, target)
    return h_cur
```

The eval_1d function computes max_k O_k(t) for the active lags:
```
def eval_1d(h, j, t, over, active):

    max_val = 0.0
    for k in active:
        k = int(k)
        if k == 0:
            val = over[0] + (t - h_old) - (t*t - h_old*h_old)

            val = over[k] + (t - h_old) * a_k
        max_val = max(max_val, val)
    return max_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)[n-1:2*n-1])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_1d_active(h, j, t, over, active, n):
    """Evaluate max over active lags of O_k(t) where h[j]=t"""
    h_old = h[j]
    max_val = -np.inf
    for k in active:
        k = int(k)
        if k == 0:
            val = over[0] + (t - h_old) - (t*t - h_old*h_old)
        else:
            hk_plus = h[j+k] if j+k < n else 0.0
            hk_minus = h[j-k] if j >= k else 0.0
            a_k = 1.0 - hk_plus - hk_minus
            val = over[k] + (t - h_old) * a_k
        if val > max_val:
            max_val = val
    return max_val

def bpcd_step(h, n, target, over, active, rng):
    """One full pass of binary push coordinate descent"""
    h_cur = h.copy()
    order = np.random.permutation(n)  # but we should use rng
    for j in order:
        j = int(j)
        best_t = h_cur[j]
        best_val = eval_1d_active(h_cur, j, h_cur[j], over, active, n)
        
        # Try t=0 and t=1
        for t_cand in [0.0, 1.0]:
            v = eval_1d_active(h_cur, j, t_cand, over, active, n)
            if v < best_val:
                best_val = v
                best_t = t_cand
        
        # Try balanced points
        h_old = h_cur[j]
        for k in active:
            k = int(k)
            if k == 0:
                continue
            hk_plus = h_cur[j+k] if j+k < n else 0.0
            hk_minus = h_cur[j-k] if j >= k else 0.0
            a_k = 1.0 - hk_plus - hk_minus
            if abs(a_k) < 1e-10:
                continue
            b_k = over[k] - h_old * a_k
            c0 = over[0] - h_old + h_old*h_old
            # Solve t^2 + (a_k - 1)*t + (b_k - c0) = 0
            A_coef = 1.0
            B_coef = a_k - 1.0
            C_coef = b_k - c0
            disc = B_coef*B_coef - 4*A_coef*C_coef
            if disc >= 0:
                sqrt_disc = math.sqrt(disc)
                for t_cand in [(-B_coef + sqrt_disc) / (2*A_coef),
                               (-B_coef - sqrt_disc) / (2*A_coef)]:
                    if 0.0 <= t_cand <= 1.0:
                        v = eval_1d_active(h_cur, j, t_cand, over, active, n)
                        if v < best_val:
                            best_val = v
                            best_t = t_cand
        
        h_cur[j] = best_t
    return h_cur

def bpcd(h, n, target, n_passes, m_active, deadline, rng):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        over = compute_corr(h_cur)[n-1:2*n-1]
        active = np.argsort(over)[::-1][:m_active]
        
        h_next = bpcd_pass(h_cur, n, over, active, rng)
        h_next = project_feasible(h_next, target)
        
        sc = compute_c5(h_next)
        if sc < best_score:
            best_score = sc
            best_h = h_next.copy()
        h_cur = h_next
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import math

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_over(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:2*n-1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def bpcd_pass(h, over, active, n, rng):
    """One pass of binary push coordinate descent with exact over updates."""
    h_cur = h.copy()
    over_cur = over.copy()
    active_list = [int(k) for k in active]
    
    for j_idx in range(n):
        j = int(j_idx)
        h_old = h_cur[j]
        
        # Evaluate objective at candidate t values
        def eval_at(t):
            max_v = -1e10
            for k in active_list:
                if k == 0:
                    val = over_cur[0] + t*(1.0-t) - h_old*(1.0-h_old)
                else:
                    h_jk = h_cur[j-k] if j >= k else 0.0
                    h_jkp = h_cur[j+k] if j+k < n else 0.0
                    a_k = 1.0 - h_jk - h_jkp
                    val = over_cur[k] + (t - h_old) * a_k
                if val > max_v:
                    max_v = val
            return max_v
        
        best_t = h_old
        best_val = eval_at(h_old)
        
        # Try 0 and 1
        for t_cand in [0.0, 1.0]:
            v = eval_at(t_cand)
            if v < best_val - 1e-15:
                best_val = v
                best_t = t_cand
        
        # Try balanced points with O_0
        c0 = over_cur[0] - h_old*(1.0-h_old)
        for k in active_list:
            if k == 0:
                continue
            h_jk = h_cur[j-k] if j >= k else 0.0
            h_jkp = h_cur[j+k] if j+k < n else 0.0
            a_k = 1.0 - h_jk - h_jkp
            if abs(a_k) < 1e-12:
                continue
            b_k = over_cur[k] - h_old * a_k
            # Solve: t - t^2 + c0 = b_k + a_k * t
            # t^2 + (a_k - 1)*t + (b_k - c0) = 0
            B_coef = a_k - 1.0
            C_coef = b_k - c0
            disc = B_coef*B_coef - 4.0*C_coef
            if disc >= 0:
                sq = math.sqrt(disc)
                for t_cand in [(-B_coef + sq) / 2.0, (-B_coef - sq) / 2.0]:
                    if -0.01 <= t_cand <= 1.01:
                        t_clip = max(0.0, min(1.0, t_cand))
                        v = eval_at(t_clip)
                        if v < best_val - 1e-15:
                            best_val = v
                            best_t = t_clip
        
        # Apply update
        if best_t != h_old:
            h_cur[j] = best_t
            # Update over for all k
            for k in range(n):
                if k == 0:
                    over_cur[0] += best_t*(1.0-best_t) - h_old*(1.0-h_old)
                else:
                    h_jk = h_cur[j-k] if j >= k else 0.0
                    h_jkp = h_cur[j+k] if j+k < n else 0.0
                    a_k = 1.0 - h_jk - h_jkp
                    over_cur[k] += (best_t - h_old) * a_k
    
    return h_cur, over_cur


def bpcd(h, n, target, n_passes, m_active, deadline, rng):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        over = compute_over(h_cur)
        active = np.argsort(over)[::-1][:m_active]
        
        h_next, over_next = bpcd_pass(h_cur, over, active, n, rng)
        h_next = project_feasible(h_next, target)
        
        sc = compute_c5(h_next)
        if sc < best_score:
            best_score = sc
            best_h = h_next.copy()
        h_cur = h_next
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import math

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_over(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:2*n-1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def bpcd_pass(h, over, active, n, rng):
    h_cur = h.copy()
    over_cur = over.copy()
    active_list = [int(k) for k in active]
    order = rng.permutation(n)
    k_vals = np.arange(n)
    m_active = len(active_list)

    for j in order:
        j = int(j)
        h_old = h_cur[j]

        candidates = [0.0, 1.0]
        if h_old > 0.01:
            candidates.append(h_old * 0.3)
        if h_old < 0.99:
            candidates.append(h_old + 0.3 * (1.0 - h_old))

        c0_offset = over_cur[0] - h_old * (1.0 - h_old)
        for ak in active_list:
            if ak == 0:
                continue
            h_jk = h_cur[j - ak] if j >= ak else 0.0
            h_jkp = h_cur[j + ak] if j + ak < n else 0.0
            a_k = 1.0 - h_jk - h_jkp
            if abs(a_k) < 1e-10:
                continue
            b_k = over_cur[ak] - h_old * a_k
            Bc = a_k - 1.0
            Cc = b_k - c0_offset
            disc = Bc * Bc - 4.0 * Cc
            if disc >= 0:
                sq = math.sqrt(disc)
                t1 = max(0.0, min(1.0, (-Bc + sq) / 2.0))
                t2 = max(0.0, min(1.0, (-Bc - sq) / 2.0))
                candidates.append(t1)
                candidates.append(t2)

        best_t = h_old
        best_val = 1e10

        for t_cand in candidates:
            max_v = -1e10
            for ak in active_list:
                if ak == 0:
                    val = over_cur[0] + t_cand * (1.0 - t_cand) - h_old * (1.0 - h_old)
                else:
                    h_jk = h_cur[j - ak] if j >= ak else 0.0
                    h_jkp = h_cur[j + ak] if j + ak < n else 0.0
                    a_k = 1.0 - h_jk - h_jkp
                    val = over_cur[ak] + (t_cand - h_old) * a_k
                if val > max_v:
                    max_v = val
            if max_v < best_val:
                best_val = max_v
                best_t = t_cand

        if best_t != h_old:
            h_cur[j] = best_t
            delta = best_t - h_old

            idx_jk = j - k_vals
            h_jk_arr = np.zeros(n)
            v = idx_jk >= 0
            h_jk_arr[v] = h_cur[idx_jk[v]]

            idx_jkp = j + k_vals
            h_jkp_arr = np.zeros(n)
            v2 = idx_jkp < n
            h_jkp_arr[v2] = h_cur[idx_jkp[v2]]

            a_all = 1.0 - h_jk_arr - h_jkp_arr
            over_cur[1:] += delta * a_all[1:]
            over_cur[0] += best_t * (1.0 - best_t) - h_old * (1.0 - h_old)

    return h_cur, over_cur


def bpcd(h, n, target, n_passes, m_active, deadline, rng):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    no_improve = 0
    max_no_improve = 200

    for p in range(n_passes):
        if time.time() > deadline:
            break
        if p > 0 and p % 50 == 0 and no_improve > max_no_improve:
            break

        over = compute_over(h_cur)
        active = np.argsort(over)[::-1][:m_active]

        h_next, _ = bpcd_pass(h_cur, over, active, n, rng)
        h_next = project_feasible(h_next, target)

        sc = compute_c5(h_next)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_next.copy()
            no_improve = 0
        else:
            no_improve += 1
        h_cur = h_next

    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    start = time.time()

    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        if start + (time.time() - start) > 10 * n_steps and step > n_steps // 2:
            break
        corr = compute_over(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        temp = 0.002 + 0.003 * (step / max(n_steps, 1))
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos + k] += wk * (1.0 - h_cur[j_pos])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def smooth_pass(h, n, target, alpha):
    h_s = h.copy()
    for i in range(1, n - 1):
        h_s[i] = (1 - alpha) * h[i] + alpha * 0.5 * (h[i-1] + h[i+1])
    h_s[0] = (1 - alpha) * h[0] + alpha * h[1]
    h_s[-1] = (1 - alpha) * h[-1] + alpha * h[-2]
    return project_feasible(h_s, target)
```
