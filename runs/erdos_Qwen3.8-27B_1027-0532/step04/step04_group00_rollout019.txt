            x_new = np.linspace(0.0, 1.0, nc)
            x_old = np.linspace(0.0, 1.0, n)
            hc = np.interp(x_new, x_old, bh)
            hc = project_feasible(hc, tc)
            hc = run_slq(hc, 3000, tc, 5000, 1e-15)
            hc = project_feasible(hc, tc)
            hc = run_slq(hc, 15000, tc, 5000, 1e-15)
            hc = project_feasible(hc, tc)
            xu = np.linspace(0.0, 1.0, n)
            hu = np.interp(xu, x_new, hc)
            hu = project_feasible(hu, tgt)
            hu = run_slq(hu, 15000, tgt, 5000, 1e-16)
            hu = project_feasible(hu, tgt)
            su = compute_c5(hu)
            if su < bs:
                bs = su
                bh = hu.copy()

    # Phase 7: Top-k subgradient (focus on multiple worst lags)
    if _rem() > 120:
        bh, bs = _topk_subgrad(bh, n, tgt, 8000, 30000.0, 4, 0.003, 0.998)

    # Phase 8: Final high-precision SLSQP
    if _rem() > 60:
        hr = run_slq(bh, 100000, tgt, 10000, 1e-18)
        hr = project_feasible(hr, tgt)
        s = compute_c5(hr)
        if s < bs:
            bs = s
            bh = hr.copy()

    # Phase 9: Final subgradient with very small steps
    if _rem() > 40:
        bh, bs = _sg_polish(bh, n, tgt, 5000, 50000.0, 0.001, 0.999, 1e-9)

    # Phase 10: One last SLSQP
    if _rem() > 20:
        hr = run_slq(bh, 50000, tgt, 5000, 1e-18)
        hr = project_feasible(hr, tgt)
        s = compute_c5(hr)
        if s < bs:
            bs = s
            bh = hr.copy()

    bh = project_feasible(bh, tgt)
    final_score = compute_c5(bh)
    return bh, final_score, len(bh)
