        m = -lag
        # sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # d/dh[i] = (1-h[i-m]) for i >= m
        #          - h[i+m] for i < n-m
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def weighted_lag_grad(h, lags, weights):
    """Weighted sum of gradients over multiple lags"""
    n = len(h)
    g = np.zeros(n)
    for idx, w in zip(lags, weights):
        if w < 1e-15:
            continue
        g += w * lag_grad(h, idx)
    return g

def interpolate_to_n(h, n_new, target):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), target)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return project_feasible(np.interp(x_new, x_old, h), target)

def nesterov_smooth_max(h_init, target, alphas, topk_list, max_iter, lr, momentum, time_limit):
    """Nesterov-accelerated gradient descent on smooth approximation of max"""
    n = len(h_init)
    h = h_init.copy()
    y = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    t0 = time.time()
    m = 1.0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # Nesterov momentum
        m_new = (1.0 + np.sqrt(1.0 + 4.0 * m * m)) / 2.0
        beta = (1.0 - m) / m_new
        y = h + beta * (h - y)
        y = project_feasible(y, target)
        
        # Compute smooth objective gradient at y
        corr = corr_full(y)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
            top_vals = corr[top_idx]
        else:
            top_idx = np.arange(n_corr)
            top_vals = corr
        
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500, 0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        grad = weighted_lag_grad(y, top_idx, weights)
        
        # Gradient step
        h = y - lr * grad
        h = project_feasible(h, target)
        m = m_new
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score

def coordinate_descent(h, target, max_rounds, time_limit):
    """Coordinate descent: try adjusting each coordinate to improve C5"""
    n = len(h)
    h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    
    for rd in range(max_rounds):

            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h[i] = np.clip(orig + delta, 0, 1)
                # Adjust another coordinate to maintain sum
                # Actually, let's just try it and project
                h_proj = project_feasible(h, target)
                score = compute_c5(h_proj)
                if score < best_score - 1e-14:
                    best_score = score
                    h = h_proj.copy()
                    improved = True
                    break
            h[i] = orig
        if not improved:
            # Try smaller steps

def lp_refine(h, target, topk, max_iter, time_limit):
    """LP-based refinement: linearize at current point, solve LP for top-k lags"""
    n = len(h)
    h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        corr = corr_full(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_corr)
        
        m = len(top_idx)
        # LP: minimize t subject to:
        #   for each lag in top_idx: grad_lag(h) @ (h' - h) + corr[lag] <= t
        #   i.e., grad_lag(h) @ h' - t <= corr[lag] - grad_lag(h) @ h
        #   sum(h') = target
        #   0 <= h' <= 1
        #   t >= 0 (implicit)
        
        A = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, idx in enumerate(top_idx):
            lag = idx - (n - 1)
            g = lag_grad(h, lag)
            A[i, :n] = g
            A[i, n] = -1.0
            b_ub[i] = g @ h - corr[idx]
        
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target])
        
        bounds_lp = [(0, 1)] * n + [(0, 1e6)]
        
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if not res.success:
            break
        
        h_new = project_feasible(res.x[:n], target)
        score = compute_c5(h_new)
        
        if score < best_score - 1e-14:
            best_score = score
            h = h_new
            stagnation = 0
        else:
            # Try line search
            for alpha_ls in [0.5, 0.25, 0.1]:
                h_try = project_feasible((1-alpha_ls)*h + alpha_ls*res.x[:n], target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    h = h_try
                    stagnation = 0
                    break
            else:
                stagnation += 1
                if stagnation > 15:
                    break
    
    return h, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    float(np.max(corr)) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-15
            nf = free.sum()
            if nf == 0: break
            per = diff / nf
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-15
            nf = free.sum()
            if nf == 0: break
            per = diff / nf
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12: break

        g[:n-m] -= h[m:]

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    for _ in range(100):
        if diff > 0:
            free = h < 1.0 - 1e-15
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-15
            nf = free.sum()
            if nf == 0:
                break
            per = diff / nf
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        m = -lag
        if m < n:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
    return g

def lp_single_step(h, target, topk):
    n = len(h)
    corr = corr_full(h)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, idx in enumerate(top_idx):
        lag = idx - (n - 1)
        g = lag_grad(h, lag)
        A[i, :n] = g
        A[i, n] = -1.0
        b_ub[i] = g @ h - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return project_feasible(res.x[:n], target)
    return None

def lp_loop(h, target, topk_seq, max_iter_per, time_limit):
    t0 = time.time()
    best = h.copy()
    best_sc = compute_c5(h)
    total_iter = 0
    for topk in topk_seq:
        stall = 0
        for it in range(max_iter_per):
            if time.time() - t0 > time_limit:
                return best, best_sc
            total_iter += 1
            h_new = lp_single_step(h, target, topk)
            if h_new is None:
                break
            sc = compute_c5(h_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best = h_new.copy()
                h = h_new
                stall = 0
            else:
                found = False
                for alpha in [0.5, 0.25, 0.1, 0.05, 0.01]:
                    h_frac = project_feasible((1.0 - alpha) * h + alpha * h_new, target)
                    sc_f = compute_c5(h_frac)
                    if sc_f < best_sc - 1e-14:
                        best_sc = sc_f
                        best = h_frac.copy()
                        h = h_frac
                        found = True
                        stall = 0
                        break
                if not found:

    return best, best_sc

def fista_grad(h, alpha, topk):
    n = len(h)
    corr = corr_full(h)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        top_vals = corr
    c_ref = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
    exp_v = np.exp(shifted)
    total = np.sum(exp_v)
    if total < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)

        weights = exp_v / total
    g = np.zeros(n)

        if w > 1e-14:
            g += w * lag_grad(h, idx - (n - 1))
    return g

    return project_feasible(np.interp(x_n, x_o, h), n_new / 2.0)
