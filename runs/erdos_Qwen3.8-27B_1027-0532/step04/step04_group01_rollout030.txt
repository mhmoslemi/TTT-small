def continuous_coordinate_descent(h, target, time_limit, t_start, step_size=0.1):
    n = len(h)
    h = h.astype(np.float64).copy()
    dx = 2.0 / n
    target_f = float(target)
    
    for it in range(1000000):
        if time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        k_star = int(np.argmax(corr)) - (n - 1)  # the lag
        
        # Compute gradient at worst lag
        # d/dh[i] of sum_j h[j]*(1-h[j+k_star]) = (1-h[i+k_star]) - h[i-k_star]
        # (with appropriate boundary handling)
        grad = np.zeros(n)
        if k_star >= 0:
            # h[i+k_star] exists for i < n-k_star
            grad[:n-k_star] += (1.0 - h[k_star:])  # contribution from h[i] term
            grad[k_star:] -= h[:n-k_star]  # contribution from h[i-k_star] term (i-k_star >= 0 means i >= k_star)
        else:
            k = -k_star
            # h[i-k] exists for i >= k
            grad[k:] += (1.0 - h[:n-k])  # h[i] term: i from k to n-1, h[i-k] from 0 to n-1-k
            grad[:n-k] -= h[k:]  # h[i+k] term: i from 0 to n-1-k, h[i+k] from k to n-1
        
        # Projected gradient descent
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum constraint
        diff = target_f - np.sum(h_new)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h_new < 1.0)[0]
            else:
                idx = np.where(h_new > 0.0)[0]

                h_new = np.clip(h_new, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    
    # Compute all correlations
    corr = np.correlate(h, one_minus_h, mode='full')  # length 2n-1
    shifted = alpha * corr * dx - np.max(alpha * corr * dx)  # for numerical stability
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    # For each lag k (from -(n-1) to n-1), the weight is weights[k + n - 1]
    # d/dh[i] of g_k = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
    
    for k_offset in range(2*n-1):
        k = k_offset - (n - 1)  # actual lag
        w = weights[k_offset]
        if w < 1e-15:
            continue
        if k >= 0:
            # (1-h[i+k]) for i in [0, n-1-k)
            if n - 1 - k >= 0:
                grad[:n-k] += w * (1.0 - h[k:n])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Water-filling: add diff to all elements, then clip
    # To do this properly, we need to find the right threshold
    # Simple approach: distribute evenly among non-saturated elements
    for _ in range(10):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def smooth_gradient(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Stable log-sum-exp
    max_val = np.max(corr) * alpha * dx
    shifted = alpha * corr * dx - max_val
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = weights[m]
        if w < 1e-14:
            continue
        if k >= 0:
            # (1-h[i+k]) for i in [0, n-1-k]
            n_val = n - k
            if n_val > 0:
                grad[:n_val] += w * (1.0 - h[k:k+n_val])
            # -h[i-k] for i in [k, n-1]
            if k < n:
                grad[k:] -= w * h[:n_val]
        else:
            k_abs = -k
            n_val = n - k_abs
            if n_val > 0:
                # (1-h[i-k_abs]) for i in [k_abs, n-1]
                grad[k_abs:] += w * (1.0 - h[:n_val])
                # -h[i+k_abs] for i in [0, n-1-k_abs]
                grad[:n_val] -= w * h[k_abs:]
    
    c5 = float(np.max(corr)) * dx
    return grad, c5, weights

def smooth_objective_value(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr) * alpha * dx
    return float((max_val + np.log(np.sum(np.exp(alpha * corr * dx - max_val)))) / alpha)

    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = alpha_init
    lr = lr_init
    
    for it in range(max_iters):
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        
        # Gradually increase alpha
        alpha = alpha_init * (alpha_final / alpha_init) ** (it / max(max_iters - 1, 1))
        
        grad, c5_cur, _ = smooth_gradient(h, alpha)
        
        # Gradient descent step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            # Increase learning rate slightly
            lr = min(lr * 1.01, lr_init * 5)
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
        
        # Periodic binary check
        if it % 500 == 0:

            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

def make_bipolar_sequence(n, rng):
    """Create a sequence with good low-autocorrelation properties."""
    h = np.zeros(n)
    # Use a structured pattern: alternating blocks of different lengths
    # This creates a "staircase" pattern that might have good correlation properties
    target = n // 2
    pos = 0
    block_size = 1
    while pos < n - target:
        # Place a block of 1s
        end = min(pos + block_size, target)
        h[pos:end] = 1.0
        pos = end
        # Skip a block of 0s
        pos = min(pos + block_size, n)
        block_size += 1
    
    # Fix the sum
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h

One idea: instead of random swaps, use a "best-improvement" strategy where we try all possible single swaps and pick the best one. For n=600, there are C(300,1)*C(300,1) = 90,000 possible swaps, each requiring an O(n) correlation update. That's 90,000 * 600 = 54 million operations per full sweep, which might be too slow.

But we can be smarter: instead of computing the full correlation after each swap, we can compute just the change in the correlation. For a swap (i: 1→0, j: 0→1):
- The correlation changes at specific lags
- We only need to check if the max changes

Actually, the change in correlation at lag k due to swapping i and j:
- h[i] changes from 1 to 0: affects lag k terms where i is involved
- h[j] changes from 0 to 1: affects lag k terms where j is involved

For lag k (in the output array, index m = k + n - 1):
- Original contribution from h[i]: h[i]*(1-h[i+k]) = 1*(1-h[i+k]) if 0<=i+k<n
- New contribution: 0*(1-h[i+k]) = 0
- Change: -(1-h[i+k]) = h[i+k] - 1 (if 0<=i+k<n)

- Original contribution from h[j]: h[j]*(1-h[j+k]) = 0*(1-h[j+k]) = 0
- New contribution: 1*(1-h[j+k]) = 1-h[j+k] (if 0<=j+k<n)
- Change: 1-h[j+k] (if 0<=j+k<n)

Also, h[i] appears as the "other" term:
- h[j']*(1-h[i]) where j'+k = i, i.e., j' = i-k
- Original: h[i-k]*(1-1) = 0 (if 0<=i-k<n)
- New: h[i-k]*(1-0) = h[i-k] (if 0<=i-k<n)
- Change: +h[i-k] (if 0<=i-k<n)

Similarly for j:
- h[j']*(1-h[j]) where j'+k = j, i.e., j' = j-k
- Original: h[j-k]*(1-0) = h[j-k] (if 0<=j-k<n)
- New: h[j-k]*(1-1) = 0
- Change: -h[j-k] (if 0<=j-k<n)

So the total change at lag k is:
delta_k = [h[i+k]-1 if 0<=i+k<n else 0] + [1-h[j+k] if 0<=j+k<n else 0] + [h[i-k] if 0<=i-k<n else 0] + [-h[j-k] if 0<=j-k<n else 0]

This is O(1) per lag! So for a given swap (i,j), we can compute the delta at all lags in O(1) per lag, totaling O(n) per swap.

But we still need to check all O(n^2) swaps, which is O(n^3) total. Too slow.

However, we can be smarter: only try swaps where i is a "1" that contributes to the worst lag, and j is a "0" that would reduce some other high lag. This reduces the number of candidates.

        elif move_type == 3 and len(ones) >= 3 and len(zeros) >= 3:
            idx1 = rng.choice(len(ones), 3, replace=False)
            idx0 = rng.choice(len(zeros), 3, replace=False)
            saved_change = []
            for idx in idx1:
                pos = int(ones[idx])
                h[pos] = 0
                saved_change.append((pos, 1))
            for idx in idx0:
                pos = int(zeros[idx])
                h[pos] = 1
                saved_change.append((pos, 0))
            valid = True
        elif move_type == 4 and len(ones) >= 2 and len(zeros) >= 2:
            worst_m = int(np.argmax(corr))
            worst_lag = worst_m - (n - 1)
            second_m = int(np.argsort(corr)[-2])
            second_lag = second_m - (n - 1)
            contrib1 = []
            contrib2 = []
            for ci in range(min(len(ones), 2000)):
                oi = int(ones[ci])
                pos1 = oi + worst_lag
                pos2 = oi + second_lag
                if 0 <= pos1 < n and h[pos1] == 0:
                    contrib1.append(ci)
                if 0 <= pos2 < n and h[pos2] == 0:
                    contrib2.append(ci)
            if len(contrib1) > 0 and len(contrib2) > 0:
                common = set(contrib1) & set(contrib2)
                if len(common) > 0:
                    ci = int(list(common)[rng.randint(len(common))])
                else:
                    ci = contrib1[rng.randint(len(contrib1))]
                ii1 = int(ones[ci])
            else:
                ii1 = int(ones[rng.randint(len(ones))])
            ii2 = int(ones[rng.randint(len(ones))])
            if ii2 == ii1:
                continue
            jj1 = int(zeros[rng.randint(len(zeros))])
            jj2 = int(zeros[rng.randint(len(zeros))])
            if jj1 == jj2:
                continue
            h[ii1] = 0
            h[ii2] = 0
            h[jj1] = 1
            h[jj2] = 1
            saved_change = [(ii1, 1), (ii2, 1), (jj1, 0), (jj2, 0)]
            valid = True

        if not valid:
            continue

        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        delta = new_c5 - cur_c5

        if delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-delta / temp)):
            corr = new_corr.astype(np.float64)
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            for pos, old_val in saved_change:
                h[pos] = old_val
            no_imp += 1

    return best_h, best_c5


def greedy_binary_refine(h, target, time_limit, t_start, max_sweeps=50):
    n = len(h)
    target_int = int(target)
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_tries = min(500, len(ones) * len(zeros))
        for _ in range(n_tries):
            if time.time() - t_start > time_limit:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(new_corr)) * 2.0 / n
            if new_c5 < cur_c5 - 1e-15:
                corr = new_corr
                cur_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i] = 1
                h[j] = 0
        if not improved:
            break
    return best_h, best_c5


def make_structured_seq(n, rng):
    h = np.zeros(n)
    target = n // 2
    phase = 0.0
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        phase += 1.0 / golden
        h[i] = 1.0 if (np.sin(2.0 * np.pi * phase) > 0.0) else 0.0
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        order = rng.permutation(len(zeros))[:target - current]
        h[zeros[order]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        order = rng.permutation(len(ones))[:current - target]
        h[ones[order]] = 0.0
    return h
