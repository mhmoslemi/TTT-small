```python
def project_feasible_v2(h, target_sum):
    h = h.copy().astype(float)
    n = len(h)
    # Water-filling
    for _ in range(10):  # iterate until convergence
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if len(free) == 0:
            break
        h[free] += diff / len(free)
    return np.clip(h, 0.0, 1.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = h.copy().astype(float)
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-14)[0]
        else:
            free = np.where(h > 1e-14)[0]
        if len(free) == 0:
            break
        h[free] += diff / len(free)
    return np.clip(h, 0.0, 1.0)


def corr_gradient(h, idx, n):
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(idx)):
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += one_minus_h[:n - lag]
                grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += one_minus_h[m:]
                grad[m:] -= h[:n - m]
    return grad


def lse_obj_grad(h, alpha, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if topk < n_corr:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(n_corr)
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return obj, grad


def lbfgs_optimize(h_start, target, alpha, topk, maxiter, penalty, bounds, ftol=1e-16):
    n = len(h_start)

    def obj_func(x):

        s = np.sum(x) - target
        f_pen = f + penalty * s * s
        g_pen = g + 2.0 * penalty * s
        return f_pen, g_pen

    try:
        res = minimize(obj_func, h_start, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12,
                                               'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def slsqp_optimize(h_start, target, alpha, topk, maxiter, ftol=1e-14):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        f, _ = lse_obj_grad(x, alpha, topk)
        return f

    def obj_jac(x):
        _, g = lse_obj_grad(x, alpha, topk)
        return g

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
```
