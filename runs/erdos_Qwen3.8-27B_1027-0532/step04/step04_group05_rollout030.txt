import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_gradient(h, n_top=5):
    """Compute subgradient of max-k correlation, using top-k averaging."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top k indices
    top_idx = np.argsort(corr)[-n_top:]
    weights = np.zeros_like(corr)
    max_corr = np.max(corr)
    for idx in top_idx:
        weights[idx] = 1.0
    # Average gradient over top k
    grad = np.zeros(n)
    n_half = n - 1
    idxs = np.arange(n)
    for k_idx in top_idx:
        k = k_idx - n_half
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h[idxs[idx_valid2] + k]
    grad = grad / n_top * dx
    return grad, max_corr * dx

def truncated_alternating_lp(h_init, n, target, max_iter, time_limit, top_k=20, damping=0.5, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_reg = reg
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        n_half = n - 1
        # Get top_k lags
        top_positions = np.argsort(corr)[-top_k:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k_pos in top_positions:
            k = k_pos - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=10000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 15:
            damping = max(0.2, damping * 0.9)
            stuck = 0
        if it % 200 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.8)
    
    return best_h, best_c5

def fista_smooth(h_init, n, target, alpha_start, alpha_end, max_iter, time_limit, n_top=10):
    """FISTA (accelerated projected gradient) on smooth approximation."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    y = h.copy()
    t_k = 1.0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(1, max_iter))
        
        # Compute smooth objective gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        n_half = n - 1
        max_val = np.max(corr)
        # Soft-max gradient
        shifted = alpha * (corr - max_val)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        softmax = exp_shifted / np.sum(exp_shifted)
        
        # Gradient of soft-max of correlations w.r.t. h
        grad = np.zeros(n)
        idxs = np.arange(n)
        n_corr = len(corr)
        for w_idx in range(n_corr):
            if softmax[w_idx] > 1e-10:
                k = w_idx - n_half
                idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
                idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
                grad[idx_valid1] += softmax[w_idx] * (1.0 - y[idxs[idx_valid1] - k])
                grad[idx_valid2] -= softmax[w_idx] * y[idxs[idx_valid2] + k]
        grad *= dx
        
        # Step size (Lipschitz constant estimate)
        # The Hessian of the correlation is bounded, so L ~ 2/dx * something
        L = 2.0 / dx  # conservative
        step = 1.0 / L
        
        # Nesterov step
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = project_feasible(y, target)
        t_k = t_new
        h = h_new
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def fast_correlation_grad(h, n, top_k=10, softmax_alpha=100.0):
    """Fast computation of gradient using only top-k correlations."""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    n_half = n - 1
    max_val = np.max(corr)
    
    # Get top k positions
    top_positions = np.argsort(corr)[-top_k:]
    
    # Softmax weights over top k
    top_corr = corr[top_positions]
    shifted = softmax_alpha * (top_corr - max_val)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for i, pos in enumerate(top_positions):
        if weights[i] < 1e-8:
            continue
        k = pos - n_half
        idx_v1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_v2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_v1] += weights[i] * g[idxs[idx_v1] - k]
        grad[idx_v2] -= weights[i] * h[idxs[idx_v2] + k]
    grad *= dx
    return grad

def accelerated_subgrad(h_init, n, target, max_steps, time_limit, 
                        top_k=10, alpha_sm=50.0, lr_start=0.05, lr_end=0.001):
    """Accelerated subgradient with momentum."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        lr = lr_start * (lr_end / lr_start) ** (step / max(1, max_steps))
        grad = fast_correlation_grad(h, n, top_k, alpha_sm)
        
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_try = h - 0.3 * lr * momentum
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_c5

def coordinate_descent(h_init, n, target, max_passes, time_limit, top_k=15):
    """Optimize individual coordinates (or small blocks) to reduce C5."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        # For each position, try adjusting it slightly
        # Use the fact that changing h[i] by delta changes the sum by delta
        # So we need to compensate by changing another position
        n_changed = 0
        for i in range(n):
            if time.time() - start > time_limit:
                break
            # Try small adjustments
            for delta in [-0.05, -0.02, 0.02, 0.05]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                # Find j to compensate
                if delta > 0:
                    # Need to decrease some other h[j]
                    candidates = np.where(h < 1.0 - abs(delta))[0]
                    candidates = candidates[candidates != i]
                else:
                    candidates = np.where(h > abs(delta))[0]
                    candidates = candidates[candidates != i]
                if len(candidates) == 0:
                    continue
                j = candidates[np.argmax(candidates)]  # arbitrary choice
                
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                h_try = np.clip(h_try, 0.0, 1.0)
                if abs(np.sum(h_try) - target) > 1e-10:
                    continue
                
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    n_changed += 1
                    break
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    _n = n
    _target = target
    bounds = [(0.0, 1.0)] * n
    
    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(h):
        return np.sum(h) - _target
    
    def cons_jac(h):
        return np.ones(_n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        obj = t_var + cur_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - cur_damp) * h + cur_damp * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 12:
            cur_damp = max(0.15, cur_damp * 0.85)
            stuck = 0
        if it % 80 == 0 and cur_reg > 1e-9:
            cur_reg = max(1e-9, cur_reg * 0.7)

    return best_h, best_c5

def cutting_plane_lp(h_init, n, target, time_limit, max_constraints=40, reg=1e-4, damping=0.5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    active_set = []

    for outer in range(50):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        sorted_pos = np.argsort(corr)[::-1]
        added = 0
        for pos in sorted_pos:
            if int(pos) not in active_set:
                active_set.append(int(pos))
                added += 1
                if len(active_set) >= max_constraints:
                    break
            if added >= 15:
                break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for pos in active_set:
            k = int(pos) - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        obj = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=8000, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5

def slsqp_opt(h_start, n, target, alphas, maxiter, ftol):
    _n = n
    _target = target
    bounds = [(0.0, 1.0)] * n

    def obj(h, a):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        sh = np.clip(a * (corr - cmax), -500.0, 0.0)
        return float(cmax + np.log(np.sum(np.exp(sh))) / a)

    def fun(h):
        return np.sum(h) - _target

    def jac(h):
        return np.ones(_n)

    cons = [{'type': 'eq', 'fun': fun, 'jac': jac}]
    h_cur = h_start.copy()
    for a in alphas:
        try:
            res = minimize(obj, h_cur, args=(a,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def subgrad_fast(h_init, n, target, max_steps, time_limit, top_k, sm_alpha, lr0, lr1):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    n_half = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        lr = lr0 * (lr1 / lr0) ** (step / max(1, max_steps))
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_v = np.max(corr)
        top_pos = np.argsort(corr)[-top_k:]
        top_corr = corr[top_pos]
        sh = sm_alpha * (top_corr - max_v)
        sh = np.clip(sh, -500.0, 0.0)
        ws = np.exp(sh) / np.sum(np.exp(sh))

        grad = np.zeros(n)
        for i in range(len(top_pos)):
            if ws[i] < 1e-10:
                continue
            k = int(top_pos[i]) - n_half
            v1 = (0 <= idxs - k) & ((idxs - k) < n)
            v2 = (0 <= idxs + k) & ((idxs + k) < n)
            grad[v1] += ws[i] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= ws[i] * h[idxs[v2] + k]
        grad *= dx

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_perturb(h_init, n, target, max_tries, time_limit, seed_val):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    n_half = n - 1

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = rng.choice([2, 3, 5, 8, 10])
        n_pert = min(n_pert, n)
        idxs_p = rng.choice(n, size=n_pert, replace=False)
        scale = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3])
        h_try = h.copy()
        h_try[idxs_p] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h_try.copy()
            h = h_try

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Coarse grid for better global structure
    if tleft() > 200:
        for n_coarse in [80, 160]:
            if tleft() < 120:
                break
            target_c = n_coarse / 2.0
            h_c = interpolate_to(best_h, n_coarse, target_c)
            h_c, c5_c = truncated_lp(h_c, n_coarse, target_c,
                                      max_iter=2000, time_limit=min(50, tleft() * 0.15),
                                      top_k=10, damping=0.6, reg=1e-3)
            h_back = interpolate_to(h_c, n, target)
            sc = compute_c5(h_back)
            if sc < best_score:
                best_score = sc
                best_h = h_back

    # Phase 2: Main truncated LP at full resolution
    if tleft() > 180:
        h_lp, c5_lp = truncated_lp(best_h, n, target,
                                    max_iter=4000, time_limit=min(200, tleft() * 0.28),
                                    top_k=15, damping=0.5, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: SLSQP medium alpha
    if tleft() > 120:
        h_ref = slsqp_opt(best_h, n, target,
                           [20, 100, 500, 2000, 10000],
                           800, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Subgradient fast pass
    if tleft() > 80:
        h_sg, c5_sg = subgrad_fast(best_h, n, target,
                                     max_steps=15000, time_limit=min(50, tleft() * 0.25),
                                     top_k=8, sm_alpha=80.0,
                                     lr0=0.04, lr1=0.002)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Cutting plane LP (tighter, fewer constraints)
    if tleft() > 120:
        h_cp, c5_cp = cutting_plane_lp(best_h, n, target,
                                        time_limit=min(100, tleft() * 0.25),
                                        max_constraints=30, reg=1e-5, damping=0.4)
        if c5_cp < best_score:
            best_score = c5_cp
            best_h = h_cp

    # Phase 6: SLSQP high alpha
    if tleft() > 80:
        h_ref = slsqp_opt(best_h, n, target,
                           [5000, 20000, 80000, 300000],
                           1500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Block perturbation search
    if tleft() > 60:
        h_bp, c5_bp = block_perturb(best_h, n, target,
                                      max_tries=3000, time_limit=min(45, tleft() * 0.35),
                                      seed_val=seed + 7)
        if c5_bp < best_score:
            best_score = c5_bp
            best_h = h_bp
        if tleft() > 40:
            h_ref = slsqp_opt(best_h, n, target,
                               [2000, 10000, 50000],
                               1000, 1e-15)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: Multi-start with different inits
    for rs in range(4):
        if tleft() < 60:
            break
        rng_local = np.random.RandomState(seed + 200 + rs)
        h_rnd = np.zeros(n)
        ones_idx = rng_local.choice(n, size=n // 2, replace=False)
        h_rnd[ones_idx] = 1.0
        h_rnd = project_feasible(h_rnd, target)
        # Quick SLSQP to refine random init
        h_rnd = slsqp_opt(h_rnd, n, target, [10, 50, 200, 1000], 500, 1e-12)
        sc = compute_c5(h_rnd)
        if sc < best_score:
            best_score = sc
            best_h = h_rnd

    # Phase 9: Final LP polish
    if tleft() > 80:
        h_lp, c5_lp = truncated_lp(best_h, n, target,
                                    max_iter=3000, time_limit=min(60, tleft() * 0.35),
                                    top_k=5, damping=0.25, reg=1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Extreme alpha SLSQP
    if tleft() > 50:
        h_ref = slsqp_opt(best_h, n, target,
                           [100000, 500000, 2000000, 10000000],
                           3000, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final subgradient
    if tleft() > 25:
        h_sg, c5_sg = subgrad_fast(best_h, n, target,
                                     max_steps=8000, time_limit=min(18, tleft() * 0.55),
                                     top_k=3, sm_alpha=500.0,
                                     lr0=0.01, lr1=0.0001)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
