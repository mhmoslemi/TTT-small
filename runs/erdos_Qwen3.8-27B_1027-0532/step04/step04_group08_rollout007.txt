def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, shift_idx):
    """Compute gradient of correlation at a specific shift index."""
    n = len(h)
    j = shift_idx - (n - 1)  # actual shift
    grad = np.zeros(n)
    if j == 0:
        # c[0] = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        grad[:] = 1.0 - 2.0 * h
    elif j > 0:
        # c[j] = sum_{i=0}^{n-1-j} h[i]*(1-h[i+j])
        # = sum_{i=0}^{n-1-j} h[i] - sum_{i=0}^{n-1-j} h[i]*h[i+j]
        # d/dh[m]: 
        # term 1: contributes 1 if m in [0, n-1-j]
        # term 2: -h[m+j] if m in [0, n-1-j], -h[m-j] if m in [j, n-1]
        for i in range(n - j):
            grad[i] += 1.0 - h[i + j]
            grad[i + j] += -h[i]
    else:
        # j < 0, let k = -j
        # c[j] = sum_{i=-j}^{n-1} h[i]*(1-h[i+j]) = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        k = -j
        for i in range(k, n):
            grad[i] += 1.0 - h[i - k]
            grad[i - k] += -h[i]
    return grad

def top_k_gradient(h, k, target):
    """Compute gradient weighted by top-k correlations."""
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for idx in top_idx:
        grad += corr[idx] * corr_gradient(h, idx)
    grad /= sum(corr[top_idx])  # normalize
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_gradient_vec(h, j):
    n = len(h)
    grad = np.zeros(n)
    if j == 0:
        grad[:] = 1.0 - 2.0 * h
    elif j > 0:
        grad[:n - j] += (1.0 - h[j:])
        grad[j:] += -h[:n - j]
    else:
        k = -j
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] += -h[k:]
    return grad


def top_k_gradient(h, k, n):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-k:]
    top_vals = corr[top_idx]
    grad = np.zeros(n)
    for idx in top_idx:
        j = idx - (n - 1)
        grad += top_vals[idx] * corr_gradient_vec(h, j)
    total = top_vals.sum()
    if total > 1e-30:
        grad /= total
    return grad

    h_cur = h0.copy()

    best_h = h_cur.copy()

            h_cur = h_try
    return h_cur, cur_sc


def block_swap_search(h0, n, target, block_sizes, deadline, max_iters=300):

    sc = compute_c5(h_smooth)
    if sc < compute_c5(h):
        return h_smooth, sc
    return h, sc


def lbfgs_optimize(h0, n, target, top_k, deadline, max_iter=300):

        h_x = project_feasible(x, target)
        corr = compute_corr(h_x)

        idx = n - 1
        total_terms = 2 * n - 1
        alpha = 200.0
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        obj = c_max + float(np.log(np.sum(np.exp(shifted[total_terms // 2 - idx - 1:total_terms // 2 + idx + 1]))) ) / alpha
        wt = np.exp(shifted) / (np.exp(shifted).sum() + 1e-30)
        top_idx = np.argsort(wt)[-top_k:]
        grad = np.zeros(n)
        for tidx in top_idx:
            j = tidx - (n - 1)
            grad += wt[tidx] * corr_gradient_vec(h_x, j)
        return obj, grad

    best_h = h0.copy()
    best_sc = compute_c5(h0)

                       bounds=[(0.0, 1.0)] * n,

        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_h = h_new
            best_sc = sc

        pass
