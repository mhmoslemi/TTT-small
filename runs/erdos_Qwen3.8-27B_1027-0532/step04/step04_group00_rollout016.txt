import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(h, alpha):
    """Compute log-sum-exp objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    obj = c_max + float(np.log(sum_exp)) / alpha
    weights = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[0:n-lag] += w * (1.0 - h[lag:n])
            grad[lag:n] -= w * h[0:n-lag]
        else:
            l = -lag
            grad[l:n] += w * (1.0 - h[0:n-l])
            grad[0:n-l] -= w * h[l:n]
    
    return obj, grad

def lsbfgs_optimize(h, n, target, alphas, maxiter_per_alpha=500):
    """L-BFGS-B with penalty for equality constraint."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for a in alphas:
        # Use augmented Lagrangian / penalty
        mu = 100.0
        for _ in range(20):
            def pen_obj(x):
                base, _ = lse_obj_grad(x, a)
                viol = np.sum(x) - target
                return base + 0.5 * mu * viol * viol
            
            def pen_grad(x):
                base, g = lse_obj_grad(x, a)
                viol = np.sum(x) - target
                return g + mu * viol * np.ones(n)
            
            try:
                res = minimize(pen_obj, h_cur, jac=pen_grad, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': maxiter_per_alpha, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
                mu *= 2
                if abs(np.sum(h_cur) - target) < 1e-10:
                    break
            except Exception:
                pass
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps=500, n_active=3, lr_init=0.002, lr_decay=0.995):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:2*n-1]  # lags 0 to n-1
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                grad[0:n-k] += wk * (1.0 - h_cur[k:n])
                grad[k:n] -= wk * h_cur[0:n-k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_n = 0
_g_target = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_obj(h):
    return lse_obj_val(h, _g_alpha)


def lse_grad(h):
    n = len(h)
    alpha = _g_alpha
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[0:n - lag] += w * (1.0 - h[lag:n])
            grad[lag:n] -= w * h[0:n - lag]
        else:
            l = -lag
            grad[l:n] += w * (1.0 - h[0:n - l])
            grad[0:n - l] -= w * h[l:n]
    return grad


def cons_eq(h):
    return np.sum(h) - _g_target


def cons_jac(h):
    return np.ones(_g_n)


def slsqp_with_grad(h, n, target, alphas, maxiter=2000, ftol=1e-15):

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, jac=lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def lbfgs_penalty(h, n, target, alphas, maxiter=500, mu_max=5000.0):
    global _g_alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        _g_alpha = a
        mu = 50.0
        for _ in range(30):
            def pen_obj(x):
                base = lse_obj_val(x, a)
                viol = np.sum(x) - target
                return base + 0.5 * mu * viol * viol

            def pen_grad(x):
                g = lse_grad(x)
                viol = np.sum(x) - target
                return g + mu * viol * np.ones(n)

            try:

                               bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9})

                    h_cur = h_new
                    cur_score = sc
                if abs(np.sum(h_cur) - target) < 1e-10:
                    break
                mu = min(mu * 3.0, mu_max)
            except Exception:
                break
    return h_cur, cur_score


def subgradient_polish(h, n, target, n_steps=500, n_active=3, lr_init=0.003, lr_decay=0.996):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n - 1:2 * n - 1]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        temp = 0.002 + 0.003 * (step / n_steps)
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                grad[0:n - k] += wk * (1.0 - h_cur[k:n])
                grad[k:n] -= wk * h_cur[0:n - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, lr_init * 2)
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score

    _g_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

        else:

def minimize_scalar_wrapped(f, bounds):
    from scipy.optimize import minimize_scalar
    return minimize_scalar(f, bounds=bounds, method='bounded', options={'xatol': 1e-10})

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
