```python
    # S1[i] = sum_j x[j] * q[j-i]
    # = (x * q)[i] where * is full convolution (if we define q for negative indices)
    
    # In numpy: if we create an array Q of length 2n-1 where Q[k] = q[k-(n-1)] = p[k]
    # Then S1[i] = sum_j x[j] * Q[j-i+(n-1)]
    # This is the "full" convolution of x with Q, evaluated at index i+n-1:
    # np.convolve(x, Q, mode='full')[i+n-1] = sum_j x[j] * Q[i+n-1-j]
    # But I want sum_j x[j] * Q[j-i+(n-1)] = sum_j x[j] * Q[(n-1)-(i-j)]
    # = sum_j x[j] * Q_rev[i-j] where Q_rev[k] = Q[2n-2-k]
    # = np.convolve(x, Q_rev, mode='full')[i+n-1]
    
    # Let me verify: np.convolve(x, Q_rev, mode='full')[i+n-1]
    # = sum_j x[j] * Q_rev[i+n-1-j]
    # Q_rev[i+n-1-j] = Q[2n-2-(i+n-1-j)] = Q[n-1-i+j]
    # So = sum_j x[j] * Q[n-1-i+j] = sum_j x[j] * p[n-1-i+j] ✓
    
    # Similarly, S2[i] = sum_j x[j] * p[i-j+(n-1)] = sum_j x[j] * Q[i-j+n-1]
    # = np.convolve(x, Q, mode='full')[i+n-1]
    # Verify: = sum_j x[j] * Q[i+n-1-j] ... wait that's the same as above!
    
    # Hmm, let me recheck.
    # np.convolve(x, Q, mode='full')[m] = sum_j x[j] * Q[m-j]
    # So np.convolve(x, Q, mode='full')[i+n-1] = sum_j x[j] * Q[i+n-1-j]
    
    # I want S2[i] = sum_j x[j] * p[i-j+n-1] = sum_j x[j] * Q[i-j+n-1]
    # Q[i-j+n-1] is at index i-j+n-1 in Q.
    # np.convolve(x, Q, mode='full')[i+n-1] = sum_j x[j] * Q[i+n-1-j]
    # I need Q[i-j+n-1] = Q[(i+n-1)-j] ✓!
    
    # So S2[i] = np.convolve(x, Q, mode='full')[i+n-1] ✓
    
    # And S1[i] = sum_j x[j] * p[n-1-i+j] = sum_j x[j] * Q[n-1-i+j]
    # np.convolve(x, Q_rev, mode='full')[i+n-1] = sum_j x[j] * Q_rev[i+n-1-j]
    # Q_rev[i+n-1-j] = Q[2n-2-(i+n-1-j)] = Q[n-1-i+j] ✓!
    
    # So S1[i] = np.convolve(x, Q_rev, mode='full')[i+n-1] ✓
    
    Q = p  # length 2n-1
    Q_rev = p[::-1]  # reversed
    # These are the same thing! Q_rev[k] = Q[2n-2-k] = p[2n-2-k]
    
    # Wait, Q = p (length 2n-1), Q_rev = p[::-1] (length 2n-1)
    # S2[i] = np.convolve(x, Q, 'full')[i+n-1]
    # S1[i] = np.convolve(x, Q_rev, 'full')[i+n-1]
    
    conv_Q = np.convolve(x, Q, mode='full')  # length 3n-2
    conv_Qr = np.convolve(x, Q_rev, mode='full')  # length 3n-2
    
    S2 = conv_Q[n-1 : 2*n-1]  # indices n-1 to 2n-2, total n values
    S1 = conv_Qr[n-1 : 2*n-1]
    
    return 1.0 - S1 - S2
```

Wait, but this is O(n * (2n-1)) = O(n^2) for np.convolve when one input is length n and the other is length 2n-1. That's 3n^2 operations. For n=400, that's 480K. For n=800, that's 1.9M. That's fine!

But actually, I can use FFT to make it O(n log n):
```python
def convolve_fast(a, b):
    n = len(a) + len(b) - 1
    fa = np.fft.rfft(a, n=n)
    fb = np.fft.rfft(b, n=n)
    return np.fft.irfft(fa * fb, n=n)
```

For n=800 with b of length 1599, the FFT length would be about 2399, which is fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0
_g_alpha = 100.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha, corr

def smooth_max_grad(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    f = c_max + float(np.log(np.sum(exp_s))) / alpha
    p = exp_s / np.sum(exp_s)
    
    # Gradient computation
    Q = p  # length 2n-1
    Q_rev = p[::-1]
    
    # Use FFT for speed
    conv_len = n + (2*n-1) - 1  # = 3n-2
    fa = np.fft.rfft(x, n=conv_len)
    fq = np.fft.rfft(Q, n=conv_len)
    fq_rev = np.fft.rfft(Q_rev, n=conv_len)
    
    conv_Q = np.fft.irfft(fa * fq, n=conv_len)
    conv_Qr = np.fft.irfft(fa * fq_rev, n=conv_len)
    
    S2 = conv_Q[n-1 : 2*n-1]
    S1 = conv_Qr[n-1 : 2*n-1]
    
    grad = 1.0 - S1 - S2
    return f, grad

# For SLSQP with full problem
def full_obj(x):
    f, _ = smooth_max_value(x, _g_alpha)
    return f

def full_obj_jac(x):
    _, g = smooth_max_grad(x, _g_alpha)
    return g

def full_con_eq(x):
    return np.sum(x) - _g_target

def full_con_jac(x):
    return np.ones(_g_n)

# For block SLSQP
def block_obj(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = compute_corr(hf)
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def block_con_eq(x):
    return np.sum(x) - _g_block_target

def block_con_jac(x):
    return np.ones(_g_e - _g_s)

def slsqp_full(h, n, target, alphas, maxiter=2000, ftol=1e-16, provide_jac=True):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        cons = [{'type': 'eq', 'fun': full_con_eq, 'jac': full_con_jac}]
        kwargs = {'maxiter': maxiter, 'ftol': ftol}
        try:
            if provide_jac:
                res = minimize(full_obj, h_cur, method='SLSQP', bounds=bounds,
                              constraints=cons, jac=full_obj_jac, options=kwargs)
            else:
                res = minimize(full_obj, h_cur, method='SLSQP', bounds=bounds,
                              constraints=cons, options=kwargs)
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_alpha, _g_block_target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 300, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h

    t_new = np.linspace(0, 2, new_n, endpoint=False)
```

1. Start from initial_h_values (n=400)
2. Do a quick SLSQP with analytic gradient (alpha=100, 500) → gets us to a good local opt
3. Try block descent on n=400
4. Upscale to n=800

8. Return best

The key innovation is using the **analytic gradient** for the full SLSQP, which makes it feasible at n=800.

Wait, but SLSQP with analytic gradient on n=800 variables... SLSQP uses a QP subproblem at each iteration, which is O(n^3) in the worst case. For n=800, that might be slow. Let me think...

Actually, SLSQP's QP subproblem involves a matrix of size (n + number_of_constraints) x (n + number_of_constraints). With 1 equality constraint and 2n bound constraints (active ones), the QP could be large. But in practice, SLSQP only includes active constraints in the QP, so it's usually manageable.

Alternatively, I could use a simpler method: **projected gradient descent** or **FISTA** (accelerated proximal gradient). These scale well to high dimensions.
