                best_c5 = c5_new
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h = best_h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            no_imp += 1
    return best_h, best_c5

def focused_hill_climb(h0, target, max_steps, time_limit, t_start, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    C = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    stall = 0
    for step in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        if stall > 100:
            break
        k_max = np.argmax(C)
        ell = k_max - (n - 1)
        sens = np.zeros(n)
        if ell >= 0:
            sens[ell:] += (1.0 - h[:n - ell])
            if ell > 0:
                sens[:n - ell] -= h[ell:]
        else:
            m = -ell
            sens[:n - m] += (1.0 - h[m:])
            sens[m:] -= h[:n - m]
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ones_s = sens[ones]
        zeros_s = sens[zeros]
        top_i = ones[np.argsort(-ones_s)][:30]
        top_j = zeros[np.argsort(zeros_s)][:30]
        found = False
        for i in top_i:
            if found:
                break
            for j in top_j:
                h[i] = 0.0
                h[j] = 1.0
                C_new = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                c5_new = float(np.max(C_new))
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    C = C_new
                    found = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not found:
            r = rng.randint(min(20, len(ones)))
            s = rng.randint(min(20, len(zeros)))
            i = ones[r]
            j = zeros[s]
            h[i] = 0.0
            h[j] = 1.0
            C = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            c5_now = float(np.max(C))
            if c5_now >= best_c5:
                h[i] = 1.0
                h[j] = 0.0
                C = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                stall += 1
            else:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
    return best_h, best_c5

def slsqp_lse(h0, target, alpha_list, time_limit, t_start):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                xx = project_box_sum(x, target)
                corr = np.correlate(xx, 1.0 - xx, mode='full') * dx
                cmax = np.max(corr)
                shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
                return cmax + np.log(np.sum(np.exp(shifted))) / alpha
            res = minimize(
                obj, h, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                             'jac': lambda x: np.ones(n)}],
                options={'maxiter': 200, 'ftol': 1e-15}
            )
            h_new = project_box_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start)

    # Phase 1: LSE continuation with projected gradient
    alpha_sched = [20, 40, 80, 160, 320, 640, 1280, 2560, 5120]
    for alpha in alpha_sched:
        if tleft() < 120:
            break
        tlim = min(60.0, tleft() * 0.25)
        h_new, c5_new = projected_grad_lse(best_h, target, alpha, 20000, 0.1, tlim, start)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Focused binary hill climb
    if tleft() > 60:
        h_bin = project_box_sum(best_h, target)
        h_bin = np.where(h_bin > 0.5, 1.0, 0.0)
        h_bin = project_box_sum(h_bin, target)
        tlim = min(80.0, tleft() * 0.2)
        h_fh, c5_fh = focused_hill_climb(h_bin, target, 5000, tlim, start, rng)
        if c5_fh < best_c5:
            best_c5 = c5_fh
            best_h = h_fh.copy()

    # Phase 3: SA (targeted, shorter)
    if tleft() > 80:
        tlim = min(100.0, tleft() * 0.2)
        h_sa, c5_sa = target_sa(best_h, target, 80000, 0.012, 0.00002, rng, tlim, start)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_fh2, c5_fh2 = focused_hill_climb(best_h, target, 3000, min(30.0, tleft() * 0.3), start, rng)
        if c5_fh2 < best_c5:
            best_c5 = c5_fh2
            best_h = h_fh2.copy()

    # Phase 4: Refine with LSE at high alpha
    if tleft() > 100:
        for alpha in [2000, 4000, 8000]:
            if tleft() < 60:
                break
            tlim = min(50.0, tleft() * 0.3)
            h_new, c5_new = projected_grad_lse(best_h, target, alpha, 30000, 0.05, tlim, start)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: SLSQP refinement
    if tleft() > 80:
        tlim = min(150.0, tleft() * 0.3)
        h_sl, c5_sl = slsqp_lse(best_h, target, [2000, 4000, 8000], tlim, start)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 6: Multi-n exploration
    n_candidates = []
    for factor in [0.75, 0.85, 1.1, 1.25, 1.5, 2.0]:
        nc = int(n * factor)
        if nc % 2 != 0:
            nc += 1
        if nc >= 200 and nc != n:
            n_candidates.append(nc)
    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = np.where(h_interp > 0.5, 1.0, 0.0)
        h_bin_try = project_box_sum(h_bin_try, target_try)
        tlim_sa = min(40.0, tleft() * 0.2)
        h_sa_r, c5_sa_r = target_sa(h_bin_try, target_try, 30000, 0.01, 0.00005, rng, tlim_sa, start)
        if tleft() > 40:
            tlim_pg = min(40.0, tleft() * 0.3)
            h_pg_r, c5_pg_r = projected_grad_lse(h_sa_r, target_try, 500, 5000, 0.05, tlim_pg, start)
            if c5_pg_r < c5_sa_r:
                h_sa_r = h_pg_r
                c5_sa_r = c5_pg_r
        if c5_sa_r < best_c5:
            best_c5 = c5_sa_r
            best_h = h_sa_r.copy()

    # Phase 7: Final intensive refinement on best solution
    n_final = len(best_h)
    target_final = n_final / 2.0
    if tleft() > 40:
        tlim = min(40.0, tleft() * 0.5)
        h_fh3, c5_fh3 = focused_hill_climb(best_h, target_final, 3000, tlim, start, rng)
        if c5_fh3 < best_c5:
            best_c5 = c5_fh3
            best_h = h_fh3.copy()
    if tleft() > 30:
        tlim = min(30.0, tleft() * 0.6)
        h_new, c5_new = projected_grad_lse(best_h, target_final, 5000, 20000, 0.03, tlim, start)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility check
    n_out = len(best_h)
    target_out = n_out / 2.0
    h_out = project_box_sum(best_h, target_out)
    final_score = compute_c5(h_out)
    if final_score <= best_c5 + 1e-12:
        best_h = h_out
        best_c5 = final_score
    return best_h, best_c5, n_out
