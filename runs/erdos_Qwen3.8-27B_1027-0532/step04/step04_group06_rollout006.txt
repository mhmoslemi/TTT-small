def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

    if diff > 0:

    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:

    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]

        if need <= len(ones):

    # Let m = n+k: = sum_m f[m] conj(g[m-k])
    # This is the cross-correlation.
    # 
    # In frequency domain: FFT(f ⋆ g) = conj(FFT(f)) * FFT(g)
    # So: f ⋆ g = IFFT(conj(FFT(f)) * FFT(g))
    
    size = 1
    Nh = 2*n - 1
    while size < Nh:
        size *= 2
    H = np.fft.fft(h.astype(float), size)
    Oh = np.fft.fft(1.0 - h, size)
    corr_full = np.real(np.fft.ifft(np.conj(H) * Oh)[:Nh])
