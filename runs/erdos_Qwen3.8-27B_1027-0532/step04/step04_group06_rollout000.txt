def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]

        return h

    h[idx] += diff / len(idx)

            if current_c5 < best_c5 - 1e-12:

                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                kick_size = rng.randint(3, 8)
                for _ in range(kick_size):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                current_c5 = compute_c5(h)
                stall = 0
                max_stall = max(max_stall, 30)
    
    return best_h, best_c5

def sa_local_search(h_start, target_sum, time_limit, start_time, rng, t_init=0.015, t_final=0.0001):
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    center = n - 1
    stall = 0
    max_stall = 80
    iter_count = 0
    start_t = time.time()
    est_iters = max(1, int((time_limit) / max(0.001, (time.time() - start_t) + 1e-6)))
    
    while time.time() - start_t < time_limit:
        iter_count += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_masked = corr.copy()
        c_masked[center] = -1e18
        top_idx = np.argsort(c_masked)[::-1][:8]
        top_lags = (top_idx - center).astype(int)
        top_lags = top_lags[top_lags != 0]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        elapsed = time.time() - start_t
        progress = min(1.0, elapsed / time_limit)
        t = t_init * (t_final / t_init) ** progress
        
        candidates = find_best_swaps(h, n, corr, center, top_lags, ones, zeros, n_candidates=8)
        
        if len(candidates) > 0 and t < t_init * 0.5:
            a_idx, b_idx = candidates[rng.randint(len(candidates))]
        else:
            if rng.random() < 0.5 and len(candidates) > 0:
                a_idx, b_idx = candidates[rng.randint(len(candidates))]
            else:
                a_idx = int(ones[rng.randint(len(ones))])
                b_idx = int(zeros[rng.randint(len(zeros))])
        
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))
        delta = new_max - float(np.max(corr))
        
        if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / (t * 2.0 / n))):
            current_c5 = new_max * (2.0 / n)
            stall = 0
            if current_c5 < best_c5 - 1e-12:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
        
        if stall > max_stall:
            kick_size = rng.randint(4, 10)
            for _ in range(kick_size):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a] = 0.0
                h[b] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            current_c5 = compute_c5(h)
            stall = 0
    
    return best_h, best_c5

def fractional_refine(h_bin, target_sum, n_iterations=500, time_limit=30, start_time=0):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    center = n - 1
    step = 0.05
    
    for i in range(n_iterations):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_masked = corr.copy()
        c_masked[center] = -1e18
        top_idx = np.argsort(c_masked)[::-1][:5]
        top_lags = (top_idx - center).astype(int)
        top_lags = top_lags[top_lags != 0]
        
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            for idx in range(n):
                am = idx - k
                ap = idx + k
                val = 0.0
                if 0 <= am < n:
                    val += (1.0 - h[am])
                if 0 <= ap < n:
                    val -= h[ap]
                grad[idx] += val
        grad /= max(len(top_lags), 1)
        
        h_new = h - step * grad * dx
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        step *= 0.995
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_target = target
    cons_n = n
    
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - cons_target
    
    def cons_j(h_in):
        return np.ones(cons_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    h_bin_start = project_binary(h_init, target_init)
    
    # Phase 1: Exact local search from initial (most important)
    if tleft() > 100:
        allowed = min(300.0, tleft() * 0.35)
        h_es, c5_es = exact_local_search(h_bin_start, target_init, allowed, start_time, rng, n_top=15)
        if c5_es < best_score:
            best_score = c5_es
            best_h = h_es.copy()
    
    # Phase 2: More exact local search with different parameters
    if tleft() > 80:
        allowed = min(200.0, tleft() * 0.3)
        h_es2, c5_es2 = exact_local_search(best_h, target_init, allowed, start_time, rng, n_top=20)
        if c5_es2 < best_score:
            best_score = c5_es2
            best_h = h_es2.copy()
    
    # Phase 3: SA with targeted moves for diversity
    if tleft() > 80:
        allowed = min(150.0, tleft() * 0.25)
        h_sa, c5_sa = sa_local_search(best_h, target_init, allowed, start_time, rng, t_init=0.01, t_final=0.00005)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 4: Random restarts
    n_randoms = 4
    for rs in range(n_randoms):
        if tleft() < 60:
            break
        h_rand = np.zeros(n_init)
        perm = rng.permutation(n_init)
        h_rand[perm[:n_init // 2]] = 1.0
        allowed = min(100.0, tleft() * 0.2)
        h_r, c5_r = exact_local_search(h_rand, target_init, allowed, start_time, rng, n_top=12)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Multi-resolution
    n_candidates = [500, 650, 720, 800, 900, 1000]
    n_candidates = [x for x in n_candidates if x != n_init and x >= 200 and x % 2 == 0]
    rng.shuffle(n_candidates)
    
    for n_try in n_candidates[:4]:
        if tleft() < 60:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(120.0, tleft() * 0.25)
        h_r, c5_r = exact_local_search(h_bin_try, float(target_try), allowed, start_time, rng, n_top=12)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r.copy()
    
    # Phase 6: Fractional refinement on best solution
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bin = project_binary(best_h, target_cur)
        allowed = min(60.0, tleft() * 0.3)
        h_frac, c5_frac = fractional_refine(h_bin, target_cur, n_iterations=300,
                                             time_limit=allowed, start_time=start_time)
        if c5_frac < best_score:
            best_score = c5_frac
            best_h = h_frac
    
    # Phase 7: SLSQP polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        alphas = [50, 150, 400, 1000]
        allowed = min(80.0, tleft() * 0.4)
        h_ref, sc = slsqp_refine(best_h, n_cur, target_cur, alphas, 300, 1e-12,
                                  time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 8: Final exact search polish
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bin_final = project_binary(best_h, target_cur)
        allowed = max(5.0, min(40.0, tleft() * 0.5))
        h_f, c5_f = exact_local_search(h_bin_final, target_cur, allowed, start_time, rng, n_top=18)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f
    
    # Phase 9: Try a few more exact searches from perturbed versions
    for fa in range(3):
        if tleft() < 20:
            break
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bin = project_binary(best_h, target_cur)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        for _ in range(rng.randint(2, 5)):
            if len(ones) > 0 and len(zeros) > 0:
                h_bin[ones[rng.randint(len(ones))]] = 0.0
                h_bin[zeros[rng.randint(len(zeros))]] = 1.0
        allowed = max(5.0, min(30.0, tleft() * 0.3))
        h_p, c5_p = exact_local_search(h_bin, target_cur, allowed, start_time, rng, n_top=15)
        if c5_p < best_score:
            best_score = c5_p
            best_h = h_p
    
    # Final check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score
    
    return h_final, final_score, n_final
