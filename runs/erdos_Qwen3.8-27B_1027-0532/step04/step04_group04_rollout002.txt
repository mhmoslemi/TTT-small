    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]

        if need <= len(ones):

        if need <= len(zeros):

def enhanced_greedy(h, max_iters, rng, time_limit=None, start_time=None, top_k=5):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 500
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        # Find top worst lags
        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = top_indices - (n - 1)
        max_val = float(corr.max())
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        found = False
        n_attempts = 500
        for _ in range(n_attempts):
            # Pick a random worst lag
            k_idx = top_indices[rng.randint(top_k)]
            k_star = k_idx - (n - 1)
            
            # Find i with h[i]=1 and h[i+k_star]=0 (contributes to this lag)
            # and j with h[j]=0 and h[j+k_star]=1 (also in the overlap)
            i_cand = np.array([i for i in ones if 0 <= i + k_star < n and h[int(i + k_star)] == 0])
            j_cand = np.array([j for j in zeros if 0 <= j + k_star < n and h[int(j + k_star)] == 1])
            
            if len(i_cand) == 0 or len(j_cand) == 0:
                continue
            
            i = i_cand[rng.randint(len(i_cand))]
            j = j_cand[rng.randint(len(j_cand))]
            
            # Check if this swap reduces the maximum
            # Quick check: does it reduce c[k_star]?
            # c[k_star] = sum n h[n+k_star]*(1-h[n])
            # After swap: h[i] 1->0, h[j] 0->1
            # Change in c[k_star]:
            #   term at n=i: was h[i+k_star]*(1-1)=0, now h[i+k_star]*(1-0)=h[i+k_star]=0 (since i+k_star has h=0 by construction)
            #   Wait, we selected i such that h[i+k_star]=0, so the change at n=i is 0.
            #   term at n=j: was h[j+k_star]*(1-0)=h[j+k_star]=1 (since we selected j with h[j+k_star]=1), now h[j+k_star]*(1-1)=0
            #   So c[k_star] decreases by 1.
            #   But we also need to check other lags.
            
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            corr_new = compute_corr(h_new)
            new_max = float(corr_new.max())
            
            if new_max < max_val:
                h = h_new
                corr = corr_new
                max_val = new_max
                new_c5 = max_val * (2.0 / n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                stall = 0
                found = True
                break
        
        if not found:
            stall += 1
            # Try a random swap to escape
            if rng.random() < 0.3:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = compute_corr(h)
                
    return best_h, best_c5


def diversity_sa(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    """SA with diverse move operators including block swaps."""
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 15000
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        move_type = rng.randint(8)
        h_new = h.copy()
        moved = False
        
        if move_type <= 2:
            # Focal: swap near worst lag
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n]
            j_pool = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif move_type == 3:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif move_type == 4:
            # Block swap of length 2
            if len(ones) >= 2 and len(zeros) >= 2:
                i1_idx = rng.randint(len(ones))
                i2_idx = rng.randint(len(ones))
                j1_idx = rng.randint(len(zeros))
                j2_idx = rng.randint(len(zeros))
                i1, i2 = ones[i1_idx], ones[i2_idx]
                j1, j2 = zeros[j1_idx], zeros[j2_idx]
                if len(set([i1, i2, j1, j2])) == 4:
                    h_new[i1] = 0
                    h_new[i2] = 0
                    h_new[j1] = 1
                    h_new[j2] = 1
                    moved = True
        elif move_type == 5:
            # Adjacent block swap: swap a block of ones with adjacent block of zeros
            if len(ones) >= 1 and len(zeros) >= 1:
                i = ones[rng.randint(len(ones))]
                # Try to swap with a nearby zero
                offset = rng.randint(1, 20) * rng.choice([1, -1])
                j = i + offset
                if 0 <= j < n and h[j] == 0:
                    h_new[i] = 0
                    h_new[j] = 1
                    moved = True
                elif 0 <= i - offset < n and h[i - offset] == 0:
                    j = i - offset
                    h_new[i] = 0
                    h_new[j] = 1
                    moved = True
        elif move_type == 6:
            # 3-swap: move 2 ones to 2 zeros (keeping count)
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                i1v, i2v = ones[i1], ones[i2]
                j1v, j2v = zeros[j1], zeros[j2]
                if len(set([i1v, i2v, j1v, j2v])) == 4:
                    h_new[i1v] = 0
                    h_new[i2v] = 0
                    h_new[j1v] = 1
                    h_new[j2v] = 1
                    moved = True
        else:
            # Reverse a segment
            if n >= 4:
                a = rng.randint(0, n - 3)
                b = rng.randint(a + 2, n)
                seg = h_new[a:b+1].copy()
                # Reverse the segment
                h_new[a:b+1] = seg[::-1]
                # Check sum is preserved (it is, since we just reversed)
                moved = True
        
        if not moved:
            continue
        
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5


def ils_search(h_init, target_sum, rng, time_limit, start_time):
    """Iterated Local Search with adaptive perturbation."""
    n = len(h)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    """SLSQP refinement with smooth max approximation."""
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        # Smooth approximation of max
        alpha = alphas[0]  # Will be overridden
        return c_max
    
    # Use a different approach: minimize sum of top-k squared correlations
    # This is differentiable and encourages all top correlations to be small
    
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        def smooth_obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
        
        def cons_fun(h):
            return np.sum(h) - target
        
        def cons_jac(h):
            return np.ones(n)
        
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_score
