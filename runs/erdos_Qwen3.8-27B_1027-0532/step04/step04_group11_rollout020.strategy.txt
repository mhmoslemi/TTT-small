    h = x  # step function
    corr = np.correlate(h, 1.0 - h, mode='full')  # returns a vector of length 2*n-1

    # To differentiate, we use: f(x) = max_k corr[k] + (1/alpha) * log(sum(exp(alpha*(corr - max_corr))))

```python
def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):   # iterate until closer to target or can't adjust
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = (h < 1.0)
        else:
            mask = (h > 0.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h
```

This function adjusts the vector h to have sum(h) = target by adding small steps and clipping.
