        for topk in [40, 25, 15, 10, 8, 5, 3, 2, 1]:
            if time.time() > sub_end or remaining() < 80:
                break
            h_now, sc_now = lp_refine(best_h, target_now, topk, max_iter=60, time_limit=sub_end, rng=rng)
            if sc_now < best_score:
                best_score = sc_now
                best_h = h_now.copy()

    # Phase 5: Ultra-fine final polish
    if remaining() > 40:
        n_f = len(best_h)
        target_f = n_f / 2.0
        final_end = time.time() + remaining() - 20.0
        for topk in [3, 2, 1]:
            if time.time() > final_end:
                break
            h_f, sc_f = lp_refine(best_h, target_f, topk, max_iter=200, time_limit=final_end, rng=rng)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f.copy()

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    init_score = compute_c5(project_feasible(h_init, target))
    if final_score > init_score:
        h_final = project_feasible(h_init, target)
        final_score = init_score

    return h_final, final_score, len(h_final)
