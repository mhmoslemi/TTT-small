3. If feasible, C is achievable (at least approximately)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def round_to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin


def binary_sa(h_start, n, max_iters, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = round_to_binary(h_start, n).copy()
    current_cost = compute_c5(h)
    best_cost = current_cost
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        progress = it / max(max_iters, 1)
        T = 0.03 * (1.0 - progress) ** 2 + 0.002

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        new_cost = compute_c5(h)

        delta = new_cost - current_cost
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

        if it % 3000 == 2999:
            for _ in range(3):
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            current_cost = compute_c5(h)
            if current_cost < best_cost:
                best_cost = current_cost
                best_h = h.copy()

    return best_h, best_cost


def binary_sa_multi(h_start, n, time_budget, n_starts=4, seed=42):
    start = time.time()
    best_h = h_start.copy()
    best_cost = compute_c5(h_start)
    per_start = time_budget / n_starts

    for s in range(n_starts):
        if time.time() - start > time_budget - 5:
            break
        rng_seed = seed + s * 1000
        if s == 0:
            h_init = round_to_binary(h_start, n)
        else:
            rng = np.random.RandomState(rng_seed)
            h_init = round_to_binary(h_start + rng.randn(n) * 0.15, n)
        h_sa, c5_sa = binary_sa(h_init, n, max_iters=500000, time_limit=per_start, seed=rng_seed)
        if c5_sa < best_cost:
            best_cost = c5_sa
            best_h = h_sa

    return best_h, best_cost


def alternating_lp_v2(h_init, n, target, max_iter, time_limit, solver='ECOS', top_k=None):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    h_prev = h.copy()
    momentum = np.zeros(n)
    beta = 0.7

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h_prev

        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        if top_k is None or top_k >= 2 * n - 1:
            k_range = range(-(n - 1), n)
        else:
            ov = all_overlaps(h_prev)
            top_indices = np.argsort(-ov)[:top_k]
            k_list = [int(idx) - (n - 1) for idx in top_indices]
            k_range = k_list

        for k in k_range:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            if solver == 'ECOS':
                prob.solve(solver=cp.ECOS, max_iter=10000, verbose=False)
            else:
                prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        velocity = h_new - h
        momentum = beta * momentum + (1.0 - beta) * velocity
        h = h + momentum
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        h_prev = h.copy()

        if it % 50 == 49:
            beta = max(0.4, beta * 0.99)

    return best_h, best_c5


def smooth_max_obj_grad(h_flat, alpha, n, target):
    h = np.clip(h_flat, 0.0, 1.0)
    dx = 2.0 / n
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    w = exp_s / (total + 1e-30)
    val = float((c_max + np.log(total + 1e-30) / alpha) * dx)

    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-14:
            continue
        k = m - n_half
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if hi1 > lo1:
            grad[lo1:hi1] += w[m] * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, -k + n)
        if hi2 > lo2:
            grad[lo2:hi2] -= w[m] * h[lo2 + k:hi2 + k]
    grad *= dx
    return val, grad


def slsqp_smooth(h_init, n, target, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    h_cur = h_init.copy()
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break

        cons_fun_holder = {'target': target, 'n': n}

        def con_fun(h_flat):
            return np.sum(h_flat) - cons_fun_holder['target']

        def con_jac(h_flat):
            return np.ones(cons_fun_holder['n'])

        def objfunc(h_flat):
            return smooth_max_obj_grad(h_flat, alpha, n, target)[0]

        objfun = lambda xf: smooth_max_obj_grad(xf, alpha, n, target)[0]
        objjac = lambda xf: smooth_max_obj_grad(xf, alpha, n, target)[1]

        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        try:
            res = minimize(objfun, h_cur, jac=objjac, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            sc_old = compute_c5(h_cur)
            if sc < sc_old + 1e-14:
                h_cur = h_new
        except Exception:
            pass

    return h_cur


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half

        grad = np.zeros(n)
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 50:
            if improved:
                break
            break

    return best_h, best_c5


def multi_active_subgradient(h, n, target, max_steps, time_limit, top_k=5):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_indices = np.argsort(-corr)[:top_k]
        grad = np.zeros(n)

        for ti in top_indices:
            k = int(ti) - n_half
            idx1 = (0 <= idxs - k) & ((idxs - k) < n)
            idx2 = (0 <= idxs + k) & ((idxs + k) < n)
            grad[idx1] += (1.0 - h_cur[idxs[idx1] - k])
            grad[idx2] -= h_cur[idxs[idx2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved:
            break

    return best_h, best_c5
