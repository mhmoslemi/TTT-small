import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(np.round(h_r.sum()))
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r >= 0.5)[0]
    zeros = np.where(h_r < 0.5)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(np.round(h_r.sum()))
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            for i in range(final_sum - target_sum):
                h_r[ones2[i]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            for i in range(target_sum - final_sum):
                h_r[zeros2[i]] = 1
    return h_r


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def sa_focal(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time, top_k_lags=3):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 12000

    for it in range(max_iters):
        if time.time() - start_time > time_limit + start_time:
            break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** (ratio ** 0.6)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        h_new = h.copy()
        moved = False
        r = rng.random()

        if r < 0.45:
            top_indices = np.argsort(corr)[-top_k_lags:]
            max_idx = top_indices[rng.randint(len(top_indices))]
            k_star = int(max_idx) - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
            j_pool = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
            if len(i_pool) > 0 and len(j_pool) > 0:
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif r < 0.70:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif r < 0.85:
            block_len = rng.randint(2, 5)
            if len(ones) >= block_len:
                si = rng.randint(len(ones) - block_len + 1)
                block = ones[si:si + block_len]
                ts = rng.randint(0, n - block_len + 1)
                tp = np.arange(ts, ts + block_len)
                if not np.any(np.isin(tp, block)) and np.all(h[tp] == 0):
                    h_new[block] = 0
                    h_new[tp] = 1
                    moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def fast_polish(h, target_sum, max_rounds=50, time_limit=30, start_time=None):
    n = len(h)
    h = h.copy()
    current_max = float(np.max(compute_corr(h)))

    for _ in range(max_rounds):
        if start_time is not None and time.time() - start_time > time_limit + start_time:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        zero_order = np.random.permutation(len(zeros))

        for i in ones:
            for zi in zero_order:
                j = zeros[zi]
                h[i] = 0
                h[j] = 1
                nm = float(np.max(compute_corr(h)))
                if nm < current_max - 1e-12:
                    current_max = nm
                    improved = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            if improved:
                break
        if not improved:
            break

    return h, float(np.max(compute_corr(h))) * (2.0 / n)


def soft_polish(h, target_sum, time_limit=60, start_time=None):
    n = len(h)
    h = h.copy()
    current_max = float(np.max(compute_corr(h)))

    for _ in range(200):
        if start_time is not None and time.time() - start_time > time_limit + start_time:
            break
        best_improve = 0
        best_change = None

        for i in range(n):
            old_val = h[i]
            if old_val <= 0.05:
                cands = [0.02, 0.05, 0.10, 0.15]
            elif old_val >= 0.95:
                cands = [0.85, 0.90, 0.95, 0.98]
            else:
                cands = [max(0.0, old_val - 0.05), min(1.0, old_val + 0.05)]

            for nv in cands:
                if nv == old_val:
                    continue
                h[i] = nv
                nm = float(np.max(compute_corr(h)))
                if nm < current_max - 1e-10:
                    imp = current_max - nm
                    if imp > best_improve:
                        best_improve = imp
                        best_change = (i, nv)
                h[i] = old_val

        if best_change is None:
            break

        i, nv = best_change
        h[i] = nv
        diff = float(target_sum) - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                cid = np.where(h < 1.0 - 1e-6)[0]
            else:
                cid = np.where(h > 1e-6)[0]
            if len(cid) > 0:
                h[cid] += diff / len(cid)
                h = np.clip(h, 0, 1)
        current_max = float(np.max(compute_corr(h)))

    return h, current_max * (2.0 / n)


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _g_target


def _cons_jac(h):
    return np.ones(_g_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit + start_time:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
