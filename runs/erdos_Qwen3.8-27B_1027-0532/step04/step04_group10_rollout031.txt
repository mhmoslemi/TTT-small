        remaining = budget_s - 30.0 - (time.time() - start_time)
        if remaining < 120:
            break
        target_c = n_c / 2.0
        x_c = np.linspace(0, 2, n_c)
        h_c = np.interp(x_c, x_full, best_h)
        h_c = project_feasible(h_c, target_c)
        bounds_c = [(0.0, 1.0)] * n_c
        pen_c = penalty * (n_c / n)

        for alpha in [50, 200, 600, 1500]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 50:
                break
            h_start = h_c + rng.randn(n_c) * 0.025
            h_start = project_feasible(h_start, target_c)
            topk_c = max(5, min(30, n_c // 6))
            h_new = lbfgs_optimize(h_start, target_c, alpha, topk_c, pen_c, 350, bounds_c, 1e-14)
            h_full_try = np.interp(x_full, x_c, h_new)
            h_full_try = project_feasible(h_full_try, target)
            score = compute_c5(h_full_try)
            if score < best_score:
                best_score = score
                best_h = h_full_try
                h_c = np.interp(x_c, x_full, best_h)
                h_c = project_feasible(h_c, target_c)

    # Phase 2: Fine optimization with increasing alpha and decreasing topk
    phase2_configs = [(80, 50), (200, 35), (500, 25), (1000, 18), (2500, 12), (5000, 8)]
    for alpha, topk in phase2_configs:
        remaining = budget_s - 30.0 - (time.time() - start_time)
        if remaining < 55:
            break
        n_attempts = 2 if remaining > 250 else 1
        for att in range(n_attempts):
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 25:
                break
            noise = 0.005 / (1.0 + 0.7 * att)
            h_start = best_h + rng.randn(n) * noise
            h_start = project_feasible(h_start, target)
            maxiter = 500 if alpha < 500 else 350
            h_new = lbfgs_optimize(h_start, target, alpha, topk, penalty, maxiter, bounds, 1e-16)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 3: Basin hopping with diverse perturbations
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 130:
        n_hops = min(10, int(remaining / 25))
        for hop in range(n_hops):
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 40:
                break
            if hop % 3 == 0:
                pert = make_structured_pert(n, rng)
            elif hop % 3 == 1:
                pert = rng.randn(n) * 0.035
            else:
                x = np.linspace(0, 2, n)
                f = rng.randint(2, 8)
                pert = 0.04 * np.cos(np.pi * f * x + rng.uniform(0, 3.0))
            h_jumped = project_feasible(best_h + pert, target)
            h_new = lbfgs_optimize(h_jumped, target, 400, 20, penalty, 300, bounds, 1e-15)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 4: Subgradient polish
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 35:
        h_sg, sc_sg = subgradient_polish(best_h, target, 60, 0.015, 8)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 5: Very high alpha, tiny topk for precise max reduction
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        for alpha, topk in [(5000, 5), (12000, 4), (25000, 3)]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 18:
                break
            h_new = lbfgs_optimize(best_h, target, alpha, topk, penalty, 300, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 6: Second round of subgradient + L-BFGS
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 35:
        h_sg, sc_sg = subgradient_polish(best_h, target, 40, 0.01, 5)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        remaining = budget_s - 30.0 - (time.time() - start_time)
        if remaining > 20:
            h_new = lbfgs_optimize(best_h, target, 8000, 4, penalty, 400, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 7: Coordinate descent
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 25:
        for coord in range(n):
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 5:
                break
            orig = best_h[coord]
            for delta in [0.006, -0.006, 0.003, -0.003, 0.0015, -0.0015, 0.0008, -0.0008]:
                h_try = best_h.copy()
                h_try[coord] = np.clip(orig + delta, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_h = h_try
                    break

    # Phase 8: Final polish
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 12:
        h_new = lbfgs_optimize(best_h, target, 15000, 3, penalty, 400, bounds, 1e-18)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: One more subgradient pass if time allows
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 15:
        h_sg, sc_sg = subgradient_polish(best_h, target, 30, 0.008, 4)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
