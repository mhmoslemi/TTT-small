def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):

    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):

        if wi < 1e-14:
            continue

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha, active_indices=None):
    n = len(h)
    corr = compute_corr(h)
    if active_indices is not None:
        work_corr = corr[active_indices]
    else:
        work_corr = corr
    c_max = float(np.max(work_corr))
    shifted = alpha * (work_corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    w_local = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    if active_indices is not None:
        indices = active_indices
    else:
        indices = range(2 * n - 1)
    for pos, idx in enumerate(indices):
        wi = w_local[pos]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def slsqp_one(h, target, alpha, maxiter, ftol, active_frac=None):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    active = None
    if active_frac is not None:
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        thr = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= thr)[0]
        if len(active) > n:
            active = None

    def fun(x):
        return lse_obj_and_grad(x, alpha, active)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha, active)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline, active_frac=None):
    h_cur = h.copy()
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_cur = slsqp_one(h_cur, target, alpha, maxiter, ftol, active_frac)
    return project_feasible(h_cur, target)


def weighted_subgrad(h, target, n_steps, base_step, alpha_w, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        shifted = alpha_w * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        w = np.exp(shifted)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            break
        w /= wsum
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:

            k = idx - (n - 1)

                    grad[k:] += wi * (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= wi * h_cur[:n - kk]
                    grad[:n - kk] += wi * (1.0 - h_cur[kk:])
        step = base_step / (1.0 + 0.04 * float(it) ** 0.6)

        h_new = project_feasible(h_new, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.arange(n_src) * (2.0 / n_src)
    x_new = np.arange(n_dst) * (2.0 / n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def random_feasible(n, rng):

def volume_iterate(h, target, n_iters, n_active, deadline):
    """Volume algorithm: maintain weights, solve smooth subproblem"""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    weights = np.ones(2 * n - 1) / (2 * n - 1)
    T = 50.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        # Update weights (multiplicative weight)
        weights = weights * np.exp(-corr / T)
        wsum = float(np.sum(weights))
        if wsum < 1e-30:
            break
        weights /= wsum
        # Find top-n_active indices
        top_k = np.argsort(weights)[-n_active:]
        top_k = np.sort(top_k)
        # Solve: min weighted sum over top lags (smooth, use SLSQP)
        alpha = 100.0
        h_cur = slsqp_one(h_cur, target, alpha, 200, 1e-12, n_active)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        # Decay temperature
        T *= 0.98
    return best_h, best_score


def structured_perturb(h, target, rng, n_tries, deadline):
    """Try structured perturbations: shifts, width changes, gate changes"""
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for i in range(n_tries):
        if time.time() > deadline:
            break
        h_test = h.copy()
        pert_type = rng.randint(0, 3)
        if pert_type == 0:
            # Circular shift
            shift = rng.randint(2, max(3, n // 20))
            if rng.random() < 0.5:
                shift = -shift
            h_test = np.roll(h_test, shift)
        elif pert_type == 1:
            # Gate perturbation: add/subtract a smooth bump
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.02, n * 0.1)
            amp = rng.uniform(0.01, 0.05) * (1 if rng.random() < 0.5 else -1)
            x = np.arange(n)
            h_test += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        else:
            # Random smooth perturbation
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.005, 0.02)
            x = np.arange(n, dtype=float)
            h_test += amp * np.sin(2 * np.pi * freq * x / n + phase)
        h_test = project_feasible(h_test, target)
        sc = compute_c5(h_test)
        if sc < best_score:
            best_score = sc
            best_h = h_test.copy()
        elif sc < best_score + 0.001:
            # Even if not better, try optimizing from here
            pass
    return best_h, best_score
