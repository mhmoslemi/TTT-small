import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).copy()
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

def reconstruct(h, n_ones, n, rng, n_remove_frac=0.3):
    """Ruin and recreate: remove some 1s and greedily add back."""
    h = h.copy()
    ones = np.where(h == 1)[0]
    n_remove = int(len(ones) * n_remove_frac)
    remove_idx = rng.choice(ones, n_remove, replace=False)
    h[remove_idx] = 0
    zeros = np.where(h == 0)[0]
    
    # Greedily add back
    to_add = n_ones - int(h.sum())
    for _ in range(to_add):
        # Try each zero position, pick the one that minimizes max corr
        best_pos = -1
        best_max = float('inf')
        # Sample a subset for speed
        candidates = rng.choice(zeros, min(len(zeros), 50), replace=False)
        for pos in candidates:
            h[pos] = 1
            corr = np.correlate(h, 1-h, mode='full')
            m = np.max(corr)
            h[pos] = 0
            if m < best_max:
                best_max = m
                best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1
            zeros = np.where(h == 0)[0]
    return h

def best_improvement_search(h_init, target, rng, t_start, time_limit):

    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)

    cur_c5 = float(np.max(corr)) * 2.0 / n
    
    best_h = h.copy()
    best_c5 = cur_c5
    
    while time.time() - t_start < time_limit:
        # Find worst lag
        worst_idx = int(np.argmax(corr))
        worst_val = corr[worst_idx]
        actual_lag = worst_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find positions contributing to worst lag
        if actual_lag > 0:
            mask = (ones + actual_lag < n) & (h[ones + actual_lag] == 0)
            contrib = ones[mask]
        elif actual_lag < 0:
            mask = (ones + actual_lag >= 0) & (h[ones + actual_lag] == 0)
            contrib = ones[mask]
        else:
            contrib = ones
        
        if len(contrib) == 0:
            contrib = ones
        
        # Try swapping each contributing 1 with each zero (limited sample)
        # For speed, sample at most 20 contributing positions and 20 zeros
        contrib_sample = contrib[:min(len(contrib), 20)]
        zeros_sample = np.random.choice(zeros, min(len(zeros), 20), replace=False)
        
        best_delta = 0.0
        best_move = None
        
        for ii in contrib_sample:
            for jj in zeros_sample:
                h[ii] = 0
                h[jj] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(np.max(new_corr)) * 2.0 / n
                h[ii] = 1
                h[jj] = 0
                delta = new_c5 - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = (ii, jj)
        
        if best_move is not None and best_delta < -1e-15:
            ii, jj = best_move
            h[ii] = 0
            h[jj] = 1
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Stuck, perturb
            n_pert = rng.randint(5, max(6, n // 50))

                p2 = zeros[rng.randint(len(zeros))]

                h[p2] = 1
            corr = np.correlate(h, 1-h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).copy()
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)


def targeted_sa(h_init, target, rng, t_start, time_limit,
                t0=0.008, t_min=0.0000005, alpha=0.99995,
                no_imp_limit=6000):
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n

    best_h = h.copy()
    best_c5 = cur_c5
    no_imp = 0
    temp = t0

    while time.time() - t_start < time_limit:
        if no_imp > no_imp_limit:
            h = best_h.copy()
            n_pert = rng.randint(4, max(5, n // 50))
            ones_t = np.where(h == 1)[0]
            zeros_t = np.where(h == 0)[0]

                p1 = ones_t[rng.randint(len(ones_t))]
                p2 = zeros_t[rng.randint(len(zeros_t))]
                h[p1] = 0
                h[p2] = 1
            corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            no_imp = 0
            temp = max(t0 * 0.3, t_min * 10)
            continue

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        mc = rng.random()
        saved = ()

        if mc < 0.55:
            worst_idx = int(np.argmax(corr))
            actual_lag = worst_idx - (n - 1)
            if actual_lag > 0:
                mask = (ones + actual_lag < n) & (h[ones + actual_lag] == 0)
                contrib = ones[mask]
            elif actual_lag < 0:

            else:

            if len(contrib) == 0:

            ii = int(contrib[rng.randint(len(contrib))])
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0

def steep_descent(h_init, target, t_start, time_limit, max_rounds=200):
    n = len(h_init)
    target_int = int(target)
    h = project_binary(h_init, target_int).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * 2.0 / n
    best_h = h.copy()
    best_c5 = cur_c5

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        worst_idx = int(np.argmax(corr))
        actual_lag = worst_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if actual_lag > 0:
            mask = (ones + actual_lag < n) & (h[ones + actual_lag] == 0)
            contrib = ones[mask]
        elif actual_lag < 0:
            mask = (ones + actual_lag >= 0) & (h[ones + actual_lag] == 0)
            contrib = ones[mask]
        else:
            contrib = ones
        if len(contrib) == 0:
            contrib = ones

        best_delta = 0.0
        best_move = None
        cs = contrib[:min(len(contrib), 15)]
        zs = zeros[:min(len(zeros), 15)]
        for ii in cs:
            for jj in zs:
                h[ii] = 0
                h[jj] = 1
                nc = np.correlate(h, 1.0 - h, mode='full')
                nv = float(np.max(nc)) * 2.0 / n
                h[ii] = 1
                h[jj] = 0
                d = nv - cur_c5
                if d < best_delta:
                    best_delta = d
                    best_move = (int(ii), int(jj))

        if best_move is not None and best_delta < -1e-15:
            ii, jj = best_move
            h[ii] = 0
            h[jj] = 1
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_c5 = float(np.max(corr)) * 2.0 / n
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break

    return best_h, best_c5
