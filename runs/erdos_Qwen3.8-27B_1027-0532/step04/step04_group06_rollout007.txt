import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(corr, alpha):
    """Smooth approximation of max using log-sum-exp"""
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            break

            # d corr[k] / d h[i] involves terms from both h and 1-h

            return None  # Let SLSQP use numerical gradient
        
        def cons_f(h_in):
            return np.sum(h_in) - target_sum

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_val(corr, alpha):
    c_max = corr.max()
    shifted = np.clip(alpha * (corr - c_max), -500.0, 500.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    """Compute gradient of smooth_max(corr(h), alpha) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 500.0))
    total = weights.sum()
    weights /= total
    
    # Only process significant weights
    significant = np.where(weights > 1e-8)[0]
    
    grad = np.zeros(n)
    for idx in significant:
        m = idx - (n - 1)
        w_val = weights[idx]
        # dC[m]/dh[j] = (1-h[j-m])*I[j-m in [0,n-1]] - h[j+m]*I[j+m in [0,n-1]]
        # First part: (1-h[j-m]) for j-m in [0,n-1] → j in [m, m+n-1] ∩ [0, n-1]
        lo1 = max(0, m)
        hi1 = min(n, n + m)
        if lo1 < hi1:
            grad[lo1:hi1] += w_val * (1.0 - h[lo1-m:hi1-m])
        # Second part: -h[j+m] for j+m in [0,n-1] → j in [-m, n-1-m] ∩ [0, n-1]
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= w_val * h[lo2+m:hi2+m]
    
    return grad * dx

def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.clip(np.round(h), 0, 1).astype(int)
    diff = int(target_sum) - h_r.sum()
    if diff > 0:
        zeros = np.where(h_r == 0)[0]
        if diff > 0 and len(zeros) >= diff:
            order = np.argsort(-h[zeros])[:diff]
            h_r[zeros[order]] = 1
    elif diff < 0:
        ones = np.where(h_r == 1)[0]
        if -diff <= len(ones):
            order = np.argsort(h[ones])[:(-diff)]
            h_r[ones[order]] = 0
    return h_r.astype(float)

def interp_up(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_dst)
    return np.interp(x_new, x_old, h_src)

def slsqp_opt(h0, target, alphas, maxiter, time_limit, start_time):
    n = len(h0)
    dx = 2.0 / n
    h = project_feasible(h0.copy(), target)
    best = h.copy()
    best_sc = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def f(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max_val(c, alpha) * dx
        
        def g(x):
            return smooth_max_grad(x, alpha)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        
        try:
            res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best, best_sc

def cont_sa(h0, target, n_iters, t0, t1, rng, limit, t_start):
    n = len(h0)
    h = h0.astype(float).copy()
    best = h.copy()
    best_sc = compute_c5(h)
    cur_sc = best_sc
    stall = 0
    max_stall = max(300, n_iters // 10)
    
    for i in range(n_iters):
        if time.time() - t_start > limit:
            break
        if stall > max_stall:
            if rng.random() < 0.1:
                stall = 0
                h = project_feasible(h + rng.randn(n) * 0.08, target)
                cur_sc = compute_c5(h)
            else:
                break
        
        r_i = i / max(n_iters - 1, 1)
        t = t0 * (t1 / t0) ** r_i
        
        h_new = h.copy()
        # Perturb a random subset
        n_pert = rng.randint(1, max(2, n // 50))
        idxs = rng.choice(n, n_pert, replace=False)
        h_new[idxs] += rng.randn(n_pert) * 0.05
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        d = sc_new - cur_sc
        if d < 0 or (t > 1e-12 and rng.random() < np.exp(-d / t)):
            h = h_new
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best, best_sc

def hill_climb_bin(h0, target, n_trials, rng):
    h = project_binary(h0, target).copy()
    best_sc = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    
    for _ in range(n_trials):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0; h[b] = 1.0
        sc = compute_c5(h)
        if sc < best_sc:
            best_sc = sc
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a] = 1.0; h[b] = 0.0
            stall += 1
            if stall > 100:
                break
    return h, best_sc

            if sc_new < best_sc:
                best_sc = sc_new
                best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best, best_sc

def hill_climb_bin(h0, target, n_trials, rng):
    h = project_binary(h0, target).copy()
    best_sc = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    for _ in range(n_trials):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        sc = compute_c5(h)
        if sc < best_sc:
            best_sc = sc
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
            if stall > 120:
                break
    return h, best_sc

def bin_sa(h0, target, n_iters, t0, t1, rng, limit, t_start):
    n = len(h0)
    h = project_binary(h0, target).copy()
    best = h.copy()
    best_sc = compute_c5(h)
    cur_sc = best_sc
    stall = 0
    max_stall = max(500, n_iters // 8)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for i in range(n_iters):
        if time.time() - t_start > limit:
            break
        if stall > max_stall:
            if rng.random() < 0.1:
                stall = 0
                for _ in range(rng.randint(3, 8)):
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                cur_sc = compute_c5(h)
            else:
                break
        r_i = i / max(n_iters - 1, 1)
        t = t0 * (t1 / t0) ** r_i
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        sc = compute_c5(h)
        d = sc - cur_sc
        if d < 0 or (t > 1e-12 and rng.random() < np.exp(-d / t)):
            cur_sc = sc
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if sc < best_sc:
                best_sc = sc
                best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    return best, best_sc

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    tgt0 = n0 / 2.0
    best_h = h_init.copy()
    best_sc = compute_c5(best_h)

    def tl():
        return budget_s - 15.0 - (time.time() - t_start)

    # Phase 1: Binary hill climb + SA at native resolution
    h_bin = project_binary(h_init, tgt0)
    h_hc, sc_hc = hill_climb_bin(h_bin, tgt0, 15000, rng)
    if sc_hc < best_sc:
        best_sc = sc_hc
        best_h = h_hc.copy()

    if tl() > 80:
        h_sa, sc_sa = bin_sa(h_hc, tgt0, 80000, 0.015, 0.0005,
                             rng, limit=min(80.0, tl() * 0.5), t_start=t_start)
        if sc_sa < best_sc:
            best_sc = sc_sa
            best_h = h_sa.copy()
        if sc_sa < best_sc or True:
            h_hc2, sc_hc2 = hill_climb_bin(h_sa, tgt0, 10000, rng)
            if sc_hc2 < best_sc:
                best_sc = sc_hc2
                best_h = h_hc2.copy()

    # Phase 2: Multi-resolution with analytical-gradient SLSQP
    resolutions = sorted(set([n0, 800, 1024, 1280, 1536, 2048, 2560]))
    resolutions = [r for r in resolutions if r % 2 == 0 and 400 <= r <= 4096]

    for n_res in resolutions:
        if tl() < 55:
            break
        tgt = n_res / 2.0
        h_up = project_feasible(interp_up(best_h, n_res), tgt)

        # SLSQP round 1: alpha continuation
        if tl() > 45:
            alphas = [20, 60, 200, 600, 2000]
            mi = max(40, int(min(250, tl() * 0.35 / 3)))
            sl_budget = min(55.0, tl() * 0.4)
            h_slq, sc_slq = slsqp_opt(h_up, tgt, alphas, mi,
                                      time_limit=sl_budget, start_time=t_start)
            if sc_slq < best_sc:
                best_sc = sc_slq
                best_h = h_slq.copy()

        # Continuous SA for exploration at this resolution
        if tl() > 30:
            h_work = project_feasible(interp_up(best_h, n_res), tgt)
            n_it = int(min(25000, tl() * 28))
            h_csa, sc_csa = cont_sa(h_work, tgt, n_it, 0.04, 0.0008,
                                     rng, limit=min(35.0, tl() * 0.3), t_start=t_start)
            if sc_csa < best_sc:
                best_sc = sc_csa
                best_h = h_csa.copy()

        # SLSQP round 2: high alpha sharpening
        if tl() > 30:
            h_work2 = project_feasible(interp_up(best_h, n_res), tgt)
            alphas2 = [800, 3000, 10000]
            mi2 = max(50, int(min(200, tl() * 0.3 / 2)))
            h_slq2, sc_slq2 = slsqp_opt(h_work2, tgt, alphas2, mi2,
                                         time_limit=min(35.0, tl() * 0.35), start_time=t_start)
            if sc_slq2 < best_sc:
                best_sc = sc_slq2
                best_h = h_slq2.copy()

    # Phase 3: Random restart at a high resolution
    if tl() > 80:
        n_rr = 2048
        tgt_rr = n_rr / 2.0
        h_rand = np.zeros(n_rr)
        perm = rng.permutation(n_rr)
        h_rand[perm[:n_rr // 2]] = 1.0
        h_rand = project_feasible(h_rand, tgt_rr)
        # Quick SA warmup
        n_it_rr = int(min(15000, tl() * 20))
        h_rr_sa, sc_rr_sa = cont_sa(h_rand, tgt_rr, n_it_rr, 0.1, 0.005,
                                     rng, limit=min(40.0, tl() * 0.35), t_start=t_start)
        # SLSQP from SA result
        if tl() > 30 and sc_rr_sa < best_sc + 0.02:
            alphas_rr = [100, 500, 2000, 8000]
            mi_rr = max(50, int(min(200, tl() * 0.3 / 2)))
            h_rr_slq, sc_rr_slq = slsqp_opt(h_rr_sa, tgt_rr, alphas_rr, mi_rr,
                                             time_limit=min(40.0, tl() * 0.4), start_time=t_start)
            if sc_rr_slq < best_sc:
                best_sc = sc_rr_slq
                best_h = h_rr_slq.copy()

    # Phase 4: Final high-precision SLSQP on best solution
    if tl() > 30:
        n_f = len(best_h)
        tgt_f = n_f / 2.0
        alphas_f = [3000, 10000, 30000, 80000]
        mi_f = max(100, int(min(500, tl() * 0.45)))
        h_fin, sc_fin = slsqp_opt(best_h, tgt_f, alphas_f, mi_f,
                                   time_limit=max(15, tl() - 12), start_time=t_start)
        if sc_fin < best_sc:
            best_sc = sc_fin
            best_h = h_fin.copy()

    # Phase 5: Final binary polish + one more SLSQP
    if tl() > 15:
        n_f = len(best_h)
        tgt_f = n_f / 2.0
        h_bf = project_binary(best_h, tgt_f)
        h_hcf, sc_hcf = hill_climb_bin(h_bf, tgt_f, int(min(10000, tl() * 600)), rng)
        if sc_hcf < best_sc:
            best_sc = sc_hcf
            best_h = h_hcf.copy()
        if tl() > 12:
            h_slqf, sc_slqf = slsqp_opt(best_h, tgt_f, [5000, 20000, 60000],
                                         max(50, int(min(200, tl() * 3))),
                                         time_limit=max(5, tl() - 5), start_time=t_start)
            if sc_slqf < best_sc:
                best_sc = sc_slqf
                best_h = h_slqf.copy()

    # Final feasibility check
    n_out = len(best_h)
    tgt_out = n_out / 2.0
    h_out = project_feasible(best_h, tgt_out)
    sc_out = compute_c5(h_out)
    if sc_out > best_sc + 1e-8:
        h_out = best_h
        sc_out = best_sc
    return h_out, sc_out, n_out
