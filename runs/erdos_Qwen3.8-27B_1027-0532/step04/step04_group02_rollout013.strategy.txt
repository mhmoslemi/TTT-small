```python
def project_feasible(h, target_sum, max_iters=100):
    # Enforce 0<=h<=1 and sum(h)≈target_sum via successive clipping/refinement
    h0 = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - sum(h0)
    
    if abs(diff) < 1e-5: return h0
    
    # Iterative adjustment
    for _ in range(max_iters):
        if abs(diff) < 1e-5: break
        
        if diff > 0:  # Add more 1's
            # Find indices to flip
            candidates = where(h0 < 1)
            rem = diff
            for idx in sorted(candidates, reverse=True):
                if rem < 1e-5: break
                h0[idx] += min(1 - h0[idx], rem)
                rem -= min(1 - h0[idx], rem)
                
        else:  # Remove some 1's
            candidates = where(h0 > 0)
            rem = -diff
            for idx in sorted(candidates):
                if rem < 1e-5: break
                h0[idx] -= min(h0[idx], rem)
                rem -= min(h0[idx], rem)
                
    return np.clip(h0, 0, 1)
```
