    """Convert continuous h to binary with exactly target ones."""
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    # Take top 'target' as 1
    threshold = h_sorted[target - 1] if target < n else 0
    b = (h >= threshold).astype(float)
    # If we have too many or too few, adjust
    while b.sum() > target:
        # Find a position that is 1 but has lowest h value
        ones = np.where(b == 1)[0]
        vals = h[ones]
        idx = ones[np.argmin(vals)]
        b[idx] = 0
    while b.sum() < target:
        zeros = np.where(b == 0)[0]
        vals = h[zeros]
        idx = zeros[np.argmax(vals)]
        b[idx] = 1
    return b

    Returns (new_max, changed) where changed indicates if we need to recompute."""

    # result[m] = sum_k h[k] * (1-h[k + m - (n-1)])

def proj_grad_refine(h, n, target, n_iters, deadline, alpha_start=10.0, alpha_end=10000.0):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(n_iters, 1))
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        # Log-sum-exp gradient
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_v = np.exp(shifted)
        sum_exp = np.sum(exp_v)
        w = exp_v / sum_exp
        
        # Gradient of the log-sum-exp objective
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= wi * h[k:]
                    grad[k:] += wi * (1.0 - h[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= wi * h[:n - kk]
                    grad[:n - kk] += wi * (1.0 - h[kk:])
        
        # Step size (Polyak or fixed)
        step = 0.5 / (alpha + 1.0)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

The gradient of corr[m] with respect to h is:

(where the second term is valid when i - m + (n-1) is in [0, n-1])

So: dcrr[m]/dh[i] = (1 - h[i + m - (n-1)]) * I(0 ≤ i+m-(n-1) ≤ n-1) - h[i - m + (n-1)] * I(0 ≤ i-m+(n-1) ≤ n-1)

    # Let's use a different approach.

    #   For i = 0 to n-1-|p|: 
    #     if p ≥ 0: grad[i] += w[n-1+p] * (1-h[i+p])  and  grad[i+p] -= w[n-1+p] * h[i] ...

        if wp < 1e-14:

            # Term 2: -h[i-(-pp)] = -h[i+pp] for i = 0 to n-1-pp
            grad[:n-pp] -= wp * h[pp:]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    n = len(h)
    b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    b[idx[:target]] = 1.0
    return b

def compute_lse_grad(h, w, n):
    grad = np.zeros(n)
    for p in range(-(n-1), n):
        wp = w[n-1+p]
        if wp < 1e-14:
            continue
        if p >= 0:
            grad[:n-p] += wp * (1.0 - h[p:])
            grad[p:] -= wp * h[:n-p]
        else:
            pp = -p
            grad[pp:] += wp * (1.0 - h[:n-pp])
            grad[:n-pp] -= wp * h[pp:]
    return grad

def proj_grad_refine(h, n, target, n_iters, deadline, alpha_start=5.0, alpha_end=50000.0):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(n_iters, 1))
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_v = np.exp(shifted)
        sum_exp = float(np.sum(exp_v))
        w = exp_v / sum_exp
        
        grad = compute_lse_grad(h, w, n)
        
        # Adaptive step size
        step = 1.0 / (alpha * np.linalg.norm(grad) + 1e-10)
        step = min(step, 0.1)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

def eval_swap_max(h, corr, i, j, n):
    new_corr = corr.copy()
    mi_lo = max(0, n - 1 - i)
    mi_hi = min(2 * n - 2, 2 * n - 2 - i)
    if mi_hi >= mi_lo:
        m_i = np.arange(mi_lo, mi_hi + 1)
        offsets_i = m_i - (n - 1) + i
        new_corr[m_i] -= (1.0 - h[offsets_i])
    mj_lo = max(0, n - 1 - j)
    mj_hi = min(2 * n - 2, 2 * n - 2 - j)
    if mj_hi >= mj_lo:
        m_j = np.arange(mj_lo, mj_hi + 1)
        offsets_j = m_j - (n - 1) + j
        new_corr[m_j] += (1.0 - h[offsets_j])
    return float(np.max(new_corr)), new_corr

def simulated_annealing(h, n, iters, T_start, T_end, seed, deadline):
    rng = np.random.RandomState(seed)
    corr = compute_corr(h).astype(float)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    ones_pos = list(np.where(h == 1)[0])
    zeros_pos = list(np.where(h == 0)[0])
    
    for it in range(iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(iters, 1))
        
        idx_i = rng.randint(len(ones_pos))
        idx_j = rng.randint(len(zeros_pos))
        i = ones_pos[idx_i]
        j = zeros_pos[idx_j]
        
        new_max, new_corr = eval_swap_max(h, corr, i, j, n)
        delta = new_max - cur_max
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            ones_pos[idx_i] = j
            zeros_pos[idx_j] = i
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max

def greedy_construct(n, seed, deadline):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    corr = np.zeros(2 * n - 1)
    remaining = list(range(n))
    rng.shuffle(remaining)
    
    for step in range(n // 2):
        if time.time() > deadline:
            break
        best_pos = -1
        best_max = float('inf')
        best_corr = None
        for p in remaining:
            mi_lo = max(0, n - 1 - p)
            mi_hi = min(2 * n - 2, 2 * n - 2 - p)
            new_corr = corr.copy()
            m = np.arange(mi_lo, mi_hi + 1)
            offsets = m - (n - 1) + p
            new_corr[m] += (1.0 - h[offsets])
            new_max = float(np.max(new_corr))
            if new_max < best_max:
                best_max = new_max
                best_pos = p
                best_corr = new_corr
        h[best_pos] = 1.0
        corr = best_corr
        remaining.remove(best_pos)
    
    return h

    target = n / 2.0
