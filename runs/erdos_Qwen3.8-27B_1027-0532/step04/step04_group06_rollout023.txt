import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.float64)
    h_r = np.clip(h_r, 0.0, 1.0)
    current_sum = int(np.round(h_r.sum()))
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r >= 0.5)[0]
        if need > len(ones):
            need = len(ones)
        order = np.argsort(h_r[ones])[:need]
        h_r[ones[order]] = 0.0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r < 0.5)[0]
        if need > len(zeros):
            need = len(zeros)
        order = np.argsort(-h_r[zeros])[:need]
        h_r[zeros[order]] = 1.0
    final_sum = int(np.round(h_r.sum()))
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = min(final_sum - target_sum, len(ones2))
            h_r[ones2[:extra]] = 0.0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = min(target_sum - final_sum, len(zeros2))
            h_r[zeros2[:extra]] = 1.0
    return h_r

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def fast_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time, targeted_frac=0.5):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(np.float64).copy()
    corr = get_corr(h)
    curr_max = float(np.max(corr))
    best_c5 = curr_max * dx
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    max_no_improve = 80000
    center = n - 1
    delta = np.zeros(2 * n - 1, dtype=np.float64)
    worst_lags = np.array([1, -1])
    lag_refresh = 0
    count = 0
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            break
        count += 1
        t = t_init * (t_final / t_init) ** (i / max(max_iters - 1, 1))
        r = rng.random()
        if r < targeted_frac:
            lag_refresh += 1
            if lag_refresh >= 20:
                top_idx = np.argsort(corr)[-6:][::-1]
                wl = top_idx - center
                wl = wl[wl != 0]
                if len(wl) == 0:
                    wl = np.array([1])
                worst_lags = wl
                lag_refresh = 0
            k_star = int(worst_lags[rng.randint(len(worst_lags))])
            if k_star == 0:
                k_star = 1
            if k_star > 0:
                a_cand = ones[ones + k_star < n]
                if len(a_cand) > 0:
                    a_mask = h[a_cand + k_star] == 0
                    a_valid = a_cand[a_mask]
                else:
                    a_valid = a_cand
                b_cand = zeros[zeros - k_star >= 0]
                if len(b_cand) > 0:
                    b_mask = h[b_cand - k_star] == 1
                    b_valid = b_cand[b_mask]
                else:
                    b_valid = b_cand
                if len(a_valid) > 0 and len(b_valid) > 0:
                    a = a_valid[rng.randint(len(a_valid))]
                    b = b_valid[rng.randint(len(b_valid))]
                elif len(a_valid) > 0:
                    a = a_valid[rng.randint(len(a_valid))]
                    b = zeros[rng.randint(len(zeros))]
                elif len(b_valid) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = b_valid[rng.randint(len(b_valid))]
                else:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
            else:
                ks = -k_star
                a_cand = ones[ones >= ks]
                if len(a_cand) > 0:
                    a_mask = h[a_cand - ks] == 0
                    a_valid = a_cand[a_mask]
                else:
                    a_valid = a_cand
                b_cand = zeros[zeros + ks < n]
                if len(b_cand) > 0:
                    b_mask = h[b_cand + ks] == 1
                    b_valid = b_cand[b_mask]
                else:
                    b_valid = b_cand
                if len(a_valid) > 0 and len(b_valid) > 0:
                    a = a_valid[rng.randint(len(a_valid))]
                    b = b_valid[rng.randint(len(b_valid))]
                elif len(a_valid) > 0:
                    a = a_valid[rng.randint(len(a_valid))]
                    b = zeros[rng.randint(len(zeros))]
                elif len(b_valid) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = b_valid[rng.randint(len(b_valid))]
                else:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
        elif r < targeted_frac + 0.35:
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
            else:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        delta.fill(0.0)
        i1s = max(0, n - 1 - a)
        i1e = min(2 * n - 2, 2 * n - 2 - a)
        if i1s <= i1e:
            h_idx = np.arange(i1s, i1e + 1) - (n - 1 - a)
            delta[i1s:i1e + 1] -= (1.0 - h[h_idx])
        i2s = max(0, a)
        i2e = min(2 * n - 2, a + n - 1)
        if i2s <= i2e:
            h_idx = a - np.arange(i2s, i2e + 1) + (n - 1)
            delta[i2s:i2e + 1] += h[h_idx]
        i3s = max(0, n - 1 - b)
        i3e = min(2 * n - 2, 2 * n - 2 - b)
        if i3s <= i3e:
            h_idx = np.arange(i3s, i3e + 1) - (n - 1 - b)
            delta[i3s:i3e + 1] += (1.0 - h[h_idx])
        i4s = max(0, b)
        i4e = min(2 * n - 2, b + n - 1)
        if i4s <= i4e:
            h_idx = b - np.arange(i4s, i4e + 1) + (n - 1)
            delta[i4s:i4e + 1] -= h[h_idx]
        new_max_val = np.max(corr + delta)
        delta_c5 = (new_max_val - curr_max) * dx
        if delta_c5 < 0 or (t > 1e-12 and rng.random() < np.exp(-delta_c5 / t)):
            corr += delta
            h[a] = 0.0
            h[b] = 1.0
            curr_max = new_max_val
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            new_c5 = curr_max * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5

def slsqp_refine(h_start, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - target

    def cons_j(h_in):
        return np.ones(n)

    h_cur = h_start.astype(np.float64).copy()
    h_cur = project_feasible(h_cur, target)
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_f, 'jac': cons_j}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
