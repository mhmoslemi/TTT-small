def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    # Distribute evenly, then clip
    per = diff / len(idx)
    h[idx] += per
    return np.clip(h, 0.0, 1.0)

    n = len(h)

    lags: array of lag values (integers, can be negative)
    Returns:

                 if we flip h[q] from 1 to 0

    delta_q = np.zeros((n, K))
    
    for j, lag in enumerate(lags):
        L = abs(int(lag))
        if L == 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == int(target_sum):
        return h_r
    if current_sum > int(target_sum):
        need = current_sum - int(target_sum)
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum) - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r

def get_worst_lags(corr, n, top_k=8):
    center = n - 1
    idx = np.argsort(corr)[::-1][:top_k]
    lags = (idx - center).astype(int)
    return lags

def compute_deltas_fast(h, lags):
    n = len(h)
    K = len(lags)
    pos = np.arange(n)
    delta_p = np.zeros((n, K))
    delta_q = np.zeros((n, K))
    for j in range(K):
        lag = int(lags[j])
        L = abs(lag)
        if L == 0:
            continue
        if lag > 0:
            gm = (pos <= n - 1 - L)
            lm = (pos >= L)
            delta_p[gm, j] += (1.0 - h[pos[gm] + L])
            delta_p[lm, j] -= h[pos[lm] - L]
            delta_q[gm, j] -= (1.0 - h[pos[gm] + L])
            delta_q[lm, j] += h[pos[lm] - L]
        else:
            gm = (pos >= L)
            lm = (pos <= n - 1 - L)
            delta_p[gm, j] += (1.0 - h[pos[gm] - L])
            delta_p[lm, j] -= h[pos[lm] + L]
            delta_q[gm, j] -= (1.0 - h[pos[gm] - L])
            delta_q[lm, j] += h[pos[lm] + L]
    return delta_p, delta_q

def find_best_move_all(h, corr, worst_lags, rng):
    n = len(h)
    K = len(worst_lags)
    center = n - 1
    c_worst = corr[center + worst_lags]
    delta_p, delta_q = compute_deltas_fast(h, worst_lags)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    if len(zeros) == 0 or len(ones) == 0:
        return None, None, 0.0
    dp_z = delta_p[zeros]
    dq_o = delta_q[ones]
    new_c = c_worst[np.newaxis, np.newaxis, :] + dp_z[:, np.newaxis, :] + dq_o[np.newaxis, :, :]
    new_max = new_c.max(axis=2)
    current_max = float(c_worst.max())
    flat_idx = int(np.argmin(new_max))
    pi = flat_idx // len(ones)
    qi = flat_idx % len(ones)
    return int(zeros[pi]), int(ones[qi]), float(new_max[pi, qi])

def find_best_move_sampled(h, corr, worst_lags, rng, sample_size=50):
    n = len(h)
    K = len(worst_lags)
    center = n - 1
    c_worst = corr[center + worst_lags]
    delta_p, delta_q = compute_deltas_fast(h, worst_lags)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    if len(zeros) == 0 or len(ones) == 0:
        return None, None, 0.0
    ns = min(sample_size, len(zeros))
    no = min(sample_size, len(ones))
    z_sample = rng.choice(zeros, ns, replace=False)
    o_sample = rng.choice(ones, no, replace=False)
    dp_z = delta_p[z_sample]
    dq_o = delta_q[o_sample]
    new_c = c_worst[np.newaxis, np.newaxis, :] + dp_z[:, np.newaxis, :] + dq_o[np.newaxis, :, :]
    new_max = new_c.max(axis=2)
    current_max = float(c_worst.max())
    flat_idx = int(np.argmin(new_max))
    pi = flat_idx // no
    qi = flat_idx % no
    return int(z_sample[pi]), int(o_sample[qi]), float(new_max[pi, qi])

def fast_delta_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time, full_eval_every=50):
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(float)
    best_h = h.copy()
    corr = get_full_corr(h)
    best_c5 = float(corr.max())
    current_max_corr = float(corr.max())
    worst_lags = get_worst_lags(corr, n, top_k=8)
    no_improve = 0
    max_no_improve = max(5000, max_iters // 10)
    full_eval_count = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
                for _ in range(rng.randint(3, 8)):
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        a = ones[rng.randint(len(ones))]
                        b = zeros[rng.randint(len(zeros))]
                        h[a] = 0.0
                        h[b] = 1.0
                corr = get_full_corr(h)
                current_max_corr = float(corr.max())
                worst_lags = get_worst_lags(corr, n, top_k=8)
            else:
                break

        ratio = float(it) / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        use_full = (it % full_eval_every == 0)
        if use_full:
            p, q, est_new = find_best_move_all(h, corr, worst_lags, rng)
        else:
            p, q, est_new = find_best_move_sampled(h, corr, worst_lags, rng, sample_size=80)

        if p is None:
            break

        h[p] = 0.0
        h[q] = 1.0
        corr = get_full_corr(h)
        new_max = float(corr.max())
        delta = new_max - current_max_corr

        if delta < -1e-12 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            current_max_corr = new_max
            if new_max < best_c5 - 1e-12:
                best_c5 = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            worst_lags = get_worst_lags(corr, n, top_k=8)
        else:
            h[p] = 1.0
            h[q] = 0.0
            no_improve += 1

    dx = 2.0 / n
    return best_h, float(best_c5) * dx

def thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm.astype(float)

def greedy_init(n):

    target = n // 2

            sc = float(get_full_corr(h).max())

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_target = target
    cons_n = n

    def obj_fn(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_f(h_in):
        return np.sum(h_in) - cons_target

    def cons_j(h_in):
        return np.ones(cons_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_fn, h_cur, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
