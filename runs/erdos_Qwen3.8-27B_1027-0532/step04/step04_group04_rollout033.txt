                h_new[zeros[j2]] = 1
                moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _slsqp_target


def _cons_jac(h):
    return np.ones(_slsqp_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_n, _slsqp_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def _tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: Binary greedy
    h_bin = project_binary(h_init, target_init)
    tl = _tleft()
    if tl > 30:
        h_g, c5_g = fast_greedy(h_bin, max_iters=25000, rng=rng,
                                time_limit=min(50, tl * 0.12), start_time=start_time)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Continuous coordinate descent (primary improvement)
    tl = _tleft()
    if tl > 120:
        h_cd, c5_cd = coord_descent(best_h, n_init, target_init,
                                    max_sweeps=50000, rng=rng,
                                    time_limit=min(350, tl * 0.5), start_time=start_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd.copy()

    # Phase 3: SLSQP refinement from continuous CD result
    tl = _tleft()
    if tl > 60:
        alphas = [30, 100, 500, 2000, 10000, 50000]
        h_ref, sc = slsqp_refine(best_h, n_init, float(n_init) / 2.0, alphas,
                                 3000, 1e-14,
                                 time_limit=min(120, tl * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: ILS with coord descent + SLSQP
    for ils_iter in range(15):
        tl = _tleft()
        if tl < 80:
            break

        n_pert = max(3, int(4 + ils_iter * 1.5))
        h_pert = best_h.copy()
        ones = np.where(h_pert > 0.5)[0]
        zeros = np.where(h_pert < 0.5)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j_idx = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j_idx] = 1.0
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]

        tl = _tleft()
        if tl > 50:
            h_cd2, c5_cd2 = coord_descent(h_pert, n_init, target_init,
                                          max_sweeps=8000, rng=rng,
                                          time_limit=min(80, tl * 0.3), start_time=start_time)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2.copy()

        tl = _tleft()
        if tl > 40:
            h_ref2, sc2 = slsqp_refine(best_h, n_init, float(n_init) / 2.0,
                                       [500, 5000, 30000], 1500, 1e-14,
                                       time_limit=min(30, tl * 0.25), start_time=start_time)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2

    # Phase 5: Try slightly larger n
    tl = _tleft()
    if tl > 80:
        for n_try in [n_init + 2, n_init + 4, n_init + 6]:
            tl = _tleft()
            if tl < 40:
                break
            target_try = int(n_try / 2)
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, n_init)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), float(target_try))

            h_cd3, c5_cd3 = coord_descent(h_interp, n_try, target_try,
                                          max_sweeps=6000, rng=rng,
                                          time_limit=min(35, tl * 0.25), start_time=start_time)
            tl = _tleft()
            if tl > 20:
                h_ref3, sc3 = slsqp_refine(h_cd3, n_try, float(n_try) / 2.0,
                                           [1000, 10000, 50000], 1000, 1e-14,
                                           time_limit=min(15, tl * 0.3), start_time=start_time)
                if sc3 < c5_cd3:
                    c5_cd3 = sc3
                    h_cd3 = h_ref3
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3.copy()

    # Phase 6: Final SLSQP polish with very high alpha
    tl = _tleft()
    if tl > 25:
        n_use = len(best_h)
        h_ref4, sc4 = slsqp_refine(best_h, n_use, float(n_use) / 2.0,
                                   [10000, 50000, 100000], 3000, 1e-15,
                                   time_limit=max(5.0, tl - 8), start_time=start_time)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_ref4

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
