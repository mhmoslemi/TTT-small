def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_power(h, p):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Scale to be centered around 1 for numerical stability
    c = np.max(corr)
    normalized = corr / c
    return c * float(np.sum(np.power(normalized, p)) ** (1.0/p))

    shifted = alpha * (corr - c_max)

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)

        if p == 1:

        else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def obj_lse(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    n_full = 2 * n - 1
    n_half = n - 1
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    A = 1.0 - h
    grad = np.zeros(n)
    for ki in range(n_full):
        w = weights[ki]
        if w < 1e-14:
            continue
        k = ki - n_half
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * A[lo1 - k:hi1 - k]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + k:hi2 + k]
    grad *= dx
    return grad


def obj_power(h, p):
    n = len(h)
    corr = compute_corr(h)
    corr_safe = np.maximum(corr, 1e-30)
    return float(np.sum(np.power(corr_safe, p)) ** (1.0 / p))


def grad_power(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    n_full = 2 * n - 1
    n_half = n - 1
    corr_safe = np.maximum(corr, 1e-30)
    if p == 1:
        weights = np.ones(n_full) / n_full
    else:
        weights = np.power(corr_safe, p - 1)
        w_sum = np.sum(weights)
        weights = weights / w_sum if w_sum > 1e-30 else np.ones(n_full) / n_full
    A = 1.0 - h
    grad = np.zeros(n)
    for ki in range(n_full):
        w = weights[ki]
        if w < 1e-14:
            continue
        k = ki - n_half
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * A[lo1 - k:hi1 - k]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + k:hi2 + k]
    grad *= dx
    return grad

def slsqp_with_grad(h_start, n, target, config_list, maxiter_per, ftol=1e-15):

    n_global = n
    target_global = target
    bounds = [(0.0, 1.0)] * n

    best_c5 = compute_c5(h_cur)

    for mode, param in config_list:

            jac = lambda x: grad_power(x, param)

def alternate_lp_active(h_init, n, target, max_iter, time_limit, eps=0.005):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c_max = float(np.max(corr))
        active_mask = corr >= c_max - eps
        active_lags = np.where(active_mask)[0] - n_half

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in active_lags:
            ki = int(k)
            lo = max(0, ki)
            hi = min(n, ki + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - ki:hi - ki] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def subgradient_active(h, n, target, max_steps, time_limit, n_active=3):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
