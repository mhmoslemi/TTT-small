def compute_c5(h):

    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_A(h, n):
    """Precompute flip effect matrix A where A[i,k] = change in corr[k] when h[i] increases by 1."""
    corr_len = 2 * n - 1
    A = np.zeros((n, corr_len))
    
    for i in range(n):
        for k in range(corr_len):
            s = k - (n - 1)
            # h[n+s] = h[i] when n = i-s
            n1 = i - s
            # (1-h[n]) where n=i: h[i+s]

            if 0 <= n1 < n:
                val += (1.0 - h[n1])
            if 0 <= n2 < n:
                val -= h[n2]
            A[i, k] = val
    return A
