def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def lse_obj_and_grad(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)

    # The constraint penalty
    s = float(np.sum(x) - target)
    obj += 1e6 * s * s
    return obj

            return np.sum(x) - target
        
        def con_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n

    return h_cur, cur

```python
def compute_grad(h, softmax, n):
    """Compute gradient of smooth max correlation w.r.t. h"""
    # softmax has length 2n-1, indexed by k = 0, ..., 2n-2
    # lag_k = k - (n-1)
    # grad[i] = sum_k softmax[k] * [(1-h[i+lag_k]) - h[i-lag_k]]
    
    # Let's split into positive and negative lags
    # For lag = 0 (k = n-1): grad[i] += softmax[n-1] * [(1-h[i]) - h[i]] = softmax[n-1] * (1-2*h[i])
    # For lag > 0 (k > n-1): 
    #   (1-h[i+lag]) is valid for i < n-lag
    #   h[i-lag] is valid for i >= lag
    # For lag < 0 (k < n-1), let lag' = -lag > 0:
    #   (1-h[i-lag']) is valid for i >= lag'
    #   h[i+lag'] is valid for i < n-lag'
    
    grad = np.zeros(n)
    
    # lag = 0
    grad += softmax[n-1] * (1.0 - 2.0 * h)
    
    for lag in range(1, n):
        w_pos = softmax[n-1+lag]  # positive lag
        w_neg = softmax[n-1-lag]  # negative lag (lag in original is -lag)
        
        # Positive lag contribution: grad[i] += w_pos * [(1-h[i+lag]) - h[i-lag]]
        # Valid for i < n-lag (for i+lag) and i >= lag (for i-lag)
        if w_pos > 1e-15:
            # term1: (1-h[i+lag]) for i = 0, ..., n-1-lag
            t1 = np.zeros(n)
            t1[:n-lag] = 1.0 - h[lag:]
            # term2: h[i-lag] for i = lag, ..., n-1
            t2 = np.zeros(n)
            t2[lag:] = h[:n-lag]
            grad += w_pos * (t1 - t2)
        
        # Negative lag contribution (lag_orig = -lag): grad[i] += w_neg * [(1-h[i-lag]) - h[i+lag]]
        if w_neg > 1e-15:
            # term1: (1-h[i-lag]) for i = lag, ..., n-1
            t1 = np.zeros(n)
            t1[lag:] = 1.0 - h[:n-lag]
            # term2: h[i+lag] for i = 0, ..., n-1-lag
            t2 = np.zeros(n)
            t2[:n-lag] = h[lag:]
            grad += w_neg * (t1 - t2)
    
    return grad
```

This is O(n²) but vectorized in numpy, so each iteration is O(n) numpy operations of length n. For n=400, that's 400 iterations of vector ops of length 400, which is fast in numpy.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha

def compute_grad_smooth_max(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    softmax = exp_shifted / float(np.sum(exp_shifted))
    grad = np.zeros(n)
    for lag in range(n):
        if lag == 0:
            w = softmax[n - 1]
            if w > 1e-15:
                grad += w * (1.0 - 2.0 * h)
        else:
            w_pos = softmax[n - 1 + lag]
            w_neg = softmax[n - 1 - lag]
            if w_pos > 1e-15:
                t1 = np.zeros(n)
                t1[:n - lag] = 1.0 - h[lag:]
                t2 = np.zeros(n)
                t2[lag:] = h[:n - lag]
                grad += w_pos * (t1 - t2)
            if w_neg > 1e-15:
                t1 = np.zeros(n)
                t1[lag:] = 1.0 - h[:n - lag]
                t2 = np.zeros(n)
                t2[:n - lag] = h[lag:]
                grad += w_neg * (t1 - t2)
    return grad

def adam_optimize(h, n, target, alpha, lr=0.05, penalty=5000.0,
                  beta1=0.9, beta2=0.999, max_iter=50000, deadline=None):
    h_cur = h.astype(float).copy()
    h_cur = project_feasible(h_cur, target)
    cur = compute_c5(h_cur)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h_cur.copy()
    best_sc = cur
    patience = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        t += 1
        grad = compute_grad_smooth_max(h_cur, alpha, n)
        s = float(np.sum(h_cur) - target)
        grad += 2.0 * penalty * s
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = np.clip(h_cur - lr * m_hat / (np.sqrt(v_hat) + 1e-8), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_cur = h_new
        if patience > 20000:
            break
    return best_h, best_sc

def slsqp_optimize(h, n, target, alphas, maxiter=2000, ftol=1e-16, deadline=None):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(a * (corr - c_max), -700.0, 0.0)
            return c_max + float(np.sum(np.exp(shifted))) / a
        def con_eq(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

def block_optimize(h, n, target, block_size, alpha, n_rounds, deadline):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            rest_sum = np.sum(h_cur[:s]) + np.sum(h_cur[e:])
            block_target = target - rest_sum
            base = h_cur.copy()
            def block_obj(x, s=s, e=e, base=base, bt=block_target):
                hf = base.copy()
                hf[s:e] = np.clip(x, 0.0, 1.0)
                corr = np.correlate(hf, 1.0 - hf, mode='full')
                c_max = float(np.max(corr))
                shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
                return c_max + float(np.sum(np.exp(shifted))) / alpha
            def block_con(x, bt=block_target):
                return np.sum(x) - bt
            def block_jac(x, bs=block_size):
                return np.ones(bs)
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con, 'jac': block_jac}]
            try:
                res = minimize(block_obj, h_cur[s:e].copy(), method='SLSQP',
                               bounds=bounds_b, constraints=cons_b,
                               options={'maxiter': 300, 'ftol': 1e-13})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def anneal_step(h, n, target, t, rng):
    i = rng.randint(0, n)
    j = rng.randint(0, n)
    if i == j:
        j = (j + 1) % n
    delta = (rng.rand() - 0.5) * 0.3
    h_new = h.copy()
    h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
    h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    return h_new

def annealing_descent(h, n, target, n_iter, t_start, t_end, deadline, seed=0):
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_local = h_cur.copy()
    best_local_sc = cur
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))
        h_new = anneal_step(h_cur, n, target, t, rng)
        sc = compute_c5(h_new)
        d = sc - cur
        if d < 0 or rng.rand() < np.exp(-d / max(t, 1e-12)):
            h_cur = h_new
            cur = sc
            if cur < best_local_sc:
                best_local_sc = cur
                best_local = h_cur.copy()
    return best_local, best_local_sc

def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h
