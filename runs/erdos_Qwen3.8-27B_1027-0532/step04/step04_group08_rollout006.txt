```python
        active_set = np.argsort(corr_m)[-30:]

    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    
    max_violation = 0
    worst_idx = -1
    
    for s in active_set:
        shift = int(s) - (n - 1)
        if shift >= 0:
            i_idx = np.arange(n - shift)
            j_idx = i_idx + shift
        else:
            d = -shift
            i_idx = np.arange(d, n)
            j_idx = i_idx - d

        a_j = -h_m[i_idx]

    constraints.append(t >= 0)

        solver_list = ['PDLP', 'CBC', 'GLPK', 'SCS']
    
    sol_h = None
    for solver in solver_list:
        try:
            problem.solve(solver=solver, verbose=False)
            if problem.status in ['optimal', 'optimal_inaccurate']:
                sol_h = np.clip(h.value, 0.0, 1.0)
                break
        except Exception:
            continue

        return h_m.copy(), active_set
    
    # Check if any non-active constraint is violated
    corr_new = compute_corr(sol_h)
    for s in range(2 * n - 1):
        if s not in active_set and corr_new[s] * dx > problem.value + 1e-6:
            # Add this constraint
            active_set = np.concatenate([active_set, [s]])
            break
```

The LP is:
minimize t
subject to:
  sum(h) = target (equality)
  0 <= h[i] <= 1 for all i (bounds)
  A @ [h; t] <= b (inequalities, one per shift)

Where A is a (2n-1) x (2n+1) matrix:
For shift s with offset d:
  Row s of A: [coeffs for h[0..n-1], 1/dx]
  Where coeffs[i] = 0 for non-overlapping i
  For overlapping pairs (i, i+d): A[s,i] += (1-h_m[i+d]), A[s,i+d] += -h_m[i]

sum_i [(1-h_m[i+d])*h[i] - h_m[i]*h[i+d]] <= t/dx - sum_i h_m[i]*h_m[i+d]

Rearranging:
sum_i [(1-h_m[i+d])*h[i] - h_m[i]*h[i+d]] - t/dx <= -sum_i h_m[i]*h_m[i+d]

```python
from scipy.optimize import linprog

def slp_step_scipy(h_m, n, target):
    dx = 2.0 / n
    num_constraints = 2 * n - 1
    
    # Build sparse A_ub matrix
    # Row s corresponds to shift d = s - (n-1)
    # A_ub[s, i] for h[i], A_ub[s, n] for t
    
    rows = []
    cols = []
    vals = []
    b_ub = np.zeros(num_constraints)
    
    for s in range(num_constraints):
        d = s - (n - 1)
        if d >= 0:
            i_range = np.arange(n - d)
        else:
            i_range = np.arange(-d, n)
        
        rhs = 0.0
        for i in i_range:
            j = i + d
            a1 = 1.0 - h_m[j]  # coefficient of h[i]
            a2 = -h_m[i]       # coefficient of h[j]
            rhs += h_m[i] * h_m[j]
            
            if a1 != 0:
                rows.append(s)
                cols.append(i)
                vals.append(a1)
            if a2 != 0:
                rows.append(s)
                cols.append(j)
                vals.append(a2)
        
        # t coefficient: -1/dx
        rows.append(s)
        cols.append(n)
        vals.append(-1.0 / dx)
        b_ub[s] = -rhs
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n + 1))
    
    # Objective: minimize t, so c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality constraint: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                   bounds=bounds, method='highs')
    
    if res.success:
        return np.clip(res.x[:n], 0.0, 1.0)
    return h_m.copy()
```

```python
from scipy.sparse import csr_matrix
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_step(h_m, n, target):
    dx = 2.0 / n
    nc = 2 * n - 1
    
    rows = np.zeros(nc * (n + 1), dtype=np.int32)
    cols = np.zeros(nc * (n + 1), dtype=np.int32)
    vals = np.zeros(nc * (n + 1), dtype=float)
    b_ub = np.zeros(nc)

    r_list = []
    c_list = []
    v_list = []
    
    for s in range(nc):
        d = s - (n - 1)
        if d >= 0:
            i_range = range(n - d)
        else:
            i_range = range(-d, n)
        
        rhs = 0.0
        for i in i_range:
            j = i + d
            prod = h_m[i] * h_m[j]
            rhs += prod
            a1 = 1.0 - h_m[j]
            a2 = -h_m[i]

                c_list.append(i)
                v_list.append(a1)
            if abs(a2) > 1e-15:
                r_list.append(s)
                c_list.append(j)
                v_list.append(a2)
        
        r_list.append(s)
        c_list.append(n)
        v_list.append(-1.0 / dx)
```
