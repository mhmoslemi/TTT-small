- h[p] appears in h[i] when i = p. Valid when 0 ≤ p ≤ n-1-m.

- h[p] appears in (1-h[i+m]) when i+m = p, i.e., i = p-m. Valid when 0 ≤ p-m ≤ n-1-m, i.e., m ≤ p ≤ n-1.

- For p in [0, m-1]: only h[i] term → ∂f_{-m}/∂h[p] = (1-h[p+m])
- For p in [m, n-1-m]: both → ∂f_{-m}/∂h[p] = (1-h[p+m]) - h[p-m]
- For p in [n-m, n-1]: only (1-h[i+m]) term → ∂f_{-m}/∂h[p] = -h[p-m]

        if lag >= 0:
            grad[:n - m] -= wk * h[m:n]
            grad[m:] += wk * (1.0 - h[:n - m])
        else:
            grad[:n - m] += wk * (1.0 - h[m:n])
            grad[m:] -= wk * h[:n - m]
    grad += 2.0 * mu * s
    return f, grad


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_waterfill(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    def time_left():
        return budget_s - 25.0 - (time.time() - start_time)

    def try_update(h_new, n_new):
        nonlocal best_h, best_score, best_n
        h_new = project_waterfill(h_new, n_new / 2.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            best_n = n_new
        return sc

    # Phase 1: L-BFGS-B with increasing alpha at original resolution
    bounds0 = [(0.0, 1.0)] * n0
    for alpha in [50, 100, 200, 400]:
        if time_left() < 50:
            break
        h_start = best_h + rng.randn(n0) * 0.003
        h_start = project_waterfill(h_start, target0)
        h_new = run_lbfgs(h_start, target0, alpha, 50, 5000.0, bounds0, 400, 1e-18, 1e-12)
        try_update(h_new, n0)

    # Phase 2: Subgradient on actual max
    if time_left() > 80:
        h_sg, sc_sg = subgrad_descent(best_h, target0, 500, 80, 0.0008)
        try_update(h_sg, n0)

    # Phase 3: Higher alpha L-BFGS-B
    if time_left() > 60:
        for alpha in [600, 1200, 2500]:
            if time_left() < 40:
                break
            h_start = best_h + rng.randn(n0) * 0.001
            h_start = project_waterfill(h_start, target0)
            h_new = run_lbfgs(h_start, target0, alpha, 30, 20000.0, bounds0, 300, 1e-18, 1e-13)
            try_update(h_new, n0)

    # Phase 4: Coordinate descent
    if time_left() > 30:
        best_h, best_score = coord_descent(best_h, target0, max_rounds=2)
        best_n = n0

    # Phase 5: Barrier method - push below current best
    if time_left() > 60:
        C_target = best_score / (2.0 / n0) - 0.0001
        h_start = best_h.copy()
        bounds_bar = [(0.0, 1.0)] * n0
        try:
            res = minimize(lambda x: barrier_obj_grad(x, target0, C_target, 10000.0),
                           h_start, jac=True, method='L-BFGS-B', bounds=bounds_bar,
                           options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_waterfill(res.x, target0)
            try_update(h_new, n0)
        except Exception:
            pass

    # Phase 6: Multi-resolution exploration
    for n_try in [500, 650, 800, 1000]:
        if time_left() < 120:
            break
        target_try = n_try / 2.0
        x_src = np.linspace(0, 2.0, best_n)
        x_dst = np.linspace(0, 2.0, n_try)
        h_interp = np.interp(x_dst, x_src, best_h)
        h_interp = project_waterfill(h_interp, target_try)
        bounds_try = [(0.0, 1.0)] * n_try

        improved_at_res = False
        for alpha in [80, 200, 500, 1200]:
            if time_left() < 60:
                break
            h_start = h_interp + rng.randn(n_try) * 0.004
            h_start = project_waterfill(h_start, target_try)
            h_new = run_lbfgs(h_start, target_try, alpha, min(60, 2 * n_try // 10),
                              10000.0, bounds_try, 350, 1e-18, 1e-12)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                best_n = n_try
                h_interp = h_new.copy()
                improved_at_res = True
            else:
                h_interp = h_new

        if improved_at_res and time_left() > 40:
            h_start = best_h + rng.randn(best_n) * 0.002
            h_start = project_waterfill(h_start, best_n / 2.0)
            h_new = run_lbfgs(h_start, best_n / 2.0, 1500, 30, 30000.0,
                              [(0.0, 1.0)] * best_n, 200, 1e-18, 1e-14)
            try_update(h_new, best_n)

    # Phase 7: Final coordinate descent at best resolution
    if time_left() > 25:
        best_h, best_score = coord_descent(best_h, best_n / 2.0, max_rounds=1,
                                           deltas=[0.005, -0.005, 0.002, -0.002, 0.001, -0.001])

    # Phase 8: Final L-BFGS polish
    if time_left() > 20:
        h_start = best_h.copy()
        h_new = run_lbfgs(h_start, best_n / 2.0, 2000, 20, 50000.0,
                          [(0.0, 1.0)] * best_n, 150, 1e-19, 1e-14)
        try_update(h_new, best_n)

    # Final projection and evaluation
    best_h = project_waterfill(best_h, best_n / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_n
