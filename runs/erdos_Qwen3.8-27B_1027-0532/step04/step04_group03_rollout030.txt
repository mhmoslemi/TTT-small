import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_gradient_active(h, active_k):
    """Compute gradient of average correlation over active shifts."""
    n = len(h)
    grad = np.zeros(n)
    # For each active shift k (in full mode, index = k + n - 1),
    # grad[j] += (1 - h[(j+k)%n]) - h[(j-k)%n]
    for k_idx in active_k:
        k = k_idx - (n - 1)
        # (j+k) mod n for all j
        shifted_plus = np.roll(h, -k)  # h[(j+k)%n]
        shifted_minus = np.roll(h, k)   # h[(j-k)%n] = h[(-k+j)%n]
        grad += (1.0 - shifted_plus) - shifted_minus
    grad /= len(active_k)
    return grad

def project_sum(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 <= x <= 1}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    # Need to distribute diff among elements not at bounds
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    # Water-filling
    n_free = np.sum(mask)
    step = diff / n_free
    h[mask] += step
    return np.clip(h, 0.0, 1.0)

def active_set_refine(h_init, target_sum, time_limit, start_time, min_alpha=0.99, max_alpha=0.999):
    """Iterative active-set proximal gradient method."""
    n = len(h_init)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = 0.01
    no_improve = 0
    max_no_improve = 200
    
    for it in range(100000):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        corr = compute_corr_scaled(h)
        c_max = float(np.max(corr))
        active_thresh = min_alpha * c_max
        active_arr = np.where(corr >= active_thresh)[0]
        
        if len(active_arr) == 0:
            active_arr = np.array([np.argmax(corr)])
        
        grad = compute_gradient_active(h, active_arr)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        improved = False
        for bt in range(20):
            h_try = project_sum(h - step * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                no_improve = 0
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.9
            if lr < 1e-8:
                break
        else:
            lr = min(lr * 1.1, 0.1)
    
    return best_h, best_c5

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
    else:

    if not np.any(mask):
        return h
    h_new = h.copy()
    h_new[mask] += diff / np.sum(mask)

def project_binary(h, target_sum):
    h_r = np.clip(np.round(h), 0, 1).astype(int)
    current = h_r.sum()
    if current == target_sum:
        return h_r.astype(float)
    if current > target_sum:
        need = int(current - target_sum)
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum - current)
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def sa_binary(h_start, target_sum, n_iters, t_init, t_final, rng, time_limit, start_time):

            else:

        except Exception:
            pass
    return best_h, best_c5

def continuous_sa(h_init, target, time_limit, start_time, rng):
    n = len(h_init)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    max_no_improve = 400
    n_iter = 3000
    for i in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.04:
                no_improve = 0
            else:
                break
        ratio = i / max(n_iter - 1, 1)
        temp = 0.05 * (1e-5) ** ratio
        blen = rng.randint(1, 5)
        bs = rng.randint(0, n - blen)
        h_save = h[bs:bs + blen].copy()
        h[bs:bs + blen] = np.clip(h[bs:bs + blen] + rng.uniform(-0.2, 0.2, blen), 0, 1)
        h_proj = project_sum(h, target)
        c5_try = compute_c5(h_proj)
        delta = c5_try - cur_c5
        if delta < 0 or (temp > 1e-15 and rng.random() < np.exp(-delta / temp)):
            cur_c5 = c5_try
            h = h_proj
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                no_improve = 0
        else:
            h[bs:bs + blen] = h_save
            no_improve += 1
    return best_h, best_c5

def local_perturb(h_init, target, time_limit, start_time):
    n = len(h_init)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    fracs = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95]
    for frac in fracs:
        if time.time() - start_time > time_limit:
            break
        for pos in range(n):
            if time.time() - start_time > time_limit:
                break
            h_save = h[pos]
            h[pos] = frac
            h_proj = project_sum(h, target)
            c5_try = compute_c5(h_proj)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_proj.copy()
                h = h_proj
            else:
                h[pos] = h_save
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Binary SA from initial (35s)
    h_bin = project_binary(h_init, int(target))
    t1 = min(35.0, tleft() * 0.08)
    h_sa, c5_sa = sa_binary_fast(h_bin, int(target), 100000, 0.018, 0.00008, rng, t1, start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 2: Active-set GD (80s)
    if tleft() > 60:
        t2 = min(80.0, tleft() * 0.12)
        h_gd, c5_gd = active_set_gd(best_h, target, t2, start_time)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd.copy()

    # Phase 3: SLSQP medium alpha (120s)
    if tleft() > 100:
        t3 = min(120.0, tleft() * 0.15)
        h_slq, c5_slq = slsqp_refine(best_h, target, t3, start_time,
                                     alphas=[50, 200, 800, 3000, 10000])
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq.copy()

    # Phase 4: Nesterov GD (80s)
    if tleft() > 60:
        t4 = min(80.0, tleft() * 0.12)
        h_ne, c5_ne = nesterov_gd(best_h, target, t4, start_time)
        if c5_ne < best_score:
            best_score = c5_ne
            best_h = h_ne.copy()

    # Phase 5: Continuous SA for escape (80s)
    if tleft() > 60:
        t5 = min(80.0, tleft() * 0.1)
        h_csa, c5_csa = continuous_sa(best_h, target, t5, start_time, rng)
        if c5_csa < best_score:
            best_score = c5_csa
            best_h = h_csa.copy()
        if tleft() > 40:
            t5b = min(40.0, tleft() * 0.1)
            h_gd2, c5_gd2 = active_set_gd(best_h, target, t5b, start_time)
            if c5_gd2 < best_score:
                best_score = c5_gd2
                best_h = h_gd2.copy()

    # Phase 6: SLSQP high alpha (150s)
    if tleft() > 100:
        t6 = min(150.0, tleft() * 0.18)
        h_slq2, c5_slq2 = slsqp_refine(best_h, target, t6, start_time,
                                       alphas=[2000, 5000, 15000, 40000, 80000])
        if c5_slq2 < best_score:
            best_score = c5_slq2
            best_h = h_slq2.copy()

    # Phase 7: Final GD polish (50s)
    if tleft() > 30:
        t7 = min(50.0, tleft() * 0.2)
        h_gd3, c5_gd3 = active_set_gd(best_h, target, t7, start_time)
        if c5_gd3 < best_score:
            best_score = c5_gd3
            best_h = h_gd3.copy()

    # Phase 8: Local perturbation (40s)
    if tleft() > 25:
        t8 = min(40.0, tleft() * 0.3)
        h_lp, c5_lp = local_perturb(best_h, target, t8, start_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp.copy()

    # Phase 9: Final SLSQP if time remains
    if tleft() > 30:
        t9 = min(60.0, tleft() * 0.4)
        h_slq3, c5_slq3 = slsqp_refine(best_h, target, t9, start_time,
                                       alphas=[10000, 50000, 100000])
        if c5_slq3 < best_score:
            best_score = c5_slq3
            best_h = h_slq3.copy()

    h_final = project_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-10:
        best_h = h_final
        best_score = final_score

    return best_h, best_score, n
