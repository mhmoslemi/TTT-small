    n = len(h_init)
    h = h_init.copy()
    
    # Compute initial correlation
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full').astype(np.float64)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    no_imp = 0
    max_no_imp = 200 * n
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        if no_imp > max_no_imp:
            # Perturbation
            n_pert = max(3, n // 100)
            for _ in range(n_pert):
                ia = rng.randint(len(ones_list))
                ib = rng.randint(len(zeros_list))
                a = ones_list[ia]
                b = zeros_list[ib]
                h[a], h[b] = h[b], h[a]
                ones_list[ia] = b
                zeros_list[ib] = a
            omh = 1.0 - h
            corr = np.correlate(h, omh, mode='full').astype(np.float64)
            cur_max = float(np.max(corr))
            no_imp = 0
            continue
        
        # Pick a 1 to flip
        a = ones_list[rng.randint(len(ones_list))]
        # Pick a 0 to flip
        b = zeros_list[rng.randint(len(zeros_list))]
        
        # Compute change in correlation
        # ΔC(k) for each k = m-(n-1):
        # -[(1-h[a+k])*[valid] - h[a-k]*[valid]] + [(1-h[b+k])*[valid] - h[b-k]*[valid]]
        
        # For a (flipping 1 to 0):
        #   k ranges from -(a) to (n-1-a) for the first term (a+k valid: 0<=a+k<n)
        #   k ranges from a-(n-1) to a for the second term (a-k valid: 0<=a-k<n)

        # k = m - (n-1), m in [0, 2n-2]
        
        delta = np.zeros(2*n-1)

        # -(1-h[a+k]) for k in range where a+k in [0,n-1]
        # +h[a-k] for k in range where a-k in [0,n-1]
        k_start = max(-(n-1), -a)  # a+k >= 0
        k_end = min(n-1, n-1-a)    # a+k <= n-1
        for k in range(k_start, k_end+1):
            delta[k+(n-1)] += -(1.0 - h[a+k])
        
        k_start2 = max(-(n-1), a-(n-1))  # a-k >= 0 → k <= a
        k_end2 = min(n-1, a)              # a-k <= n-1 → k >= a-(n-1)
        for k in range(k_start2, k_end2+1):
            delta[k+(n-1)] += h[a-k]

        # +(1-h[b+k]) for k where b+k in [0,n-1]
        # -h[b-k] for k where b-k in [0,n-1]
        k_start = max(-(n-1), -b)
        k_end = min(n-1, n-1-b)

```python
def compute_swap_delta(h, a, b, n):
    """Compute change in correlation when h[a]: 1→0 and h[b]: 0→1"""
    delta = np.zeros(2*n-1)
    
    # Flipping a from 1 to 0:
    # -(1-h[a+lag]) for valid lag → indices m where n-1-a <= m <= 2n-2-a
    m1_start = max(0, n-1-a)
    m1_end = min(2*n-2, 2*n-2-a)
    if m1_start <= m1_end:
        lags = np.arange(m1_start, m1_end+1) - (n-1)
        delta[m1_start:m1_end+1] += -(1.0 - h[a + lags])
    
    # +h[a-lag] for valid lag → indices m where a <= m <= n-1+a
    m2_start = max(0, a)
    m2_end = min(2*n-2, n-1+a)
    if m2_start <= m2_end:
        lags = np.arange(m2_start, m2_end+1) - (n-1)
        delta[m2_start:m2_end+1] += h[a - lags]
    
    # Flipping b from 0 to 1:
    # +(1-h[b+lag]) for valid lag
    m3_start = max(0, n-1-b)
    m3_end = min(2*n-2, 2*n-2-b)
    if m3_start <= m3_end:
        lags = np.arange(m3_start, m3_end+1) - (n-1)
        delta[m3_start:m3_end+1] += (1.0 - h[b + lags])
    
    # -h[b-lag] for valid lag
    m4_start = max(0, b)
    m4_end = min(2*n-2, n-1+b)
    if m4_start <= m4_end:
        lags = np.arange(m4_start, m4_end+1) - (n-1)
        delta[m4_start:m4_end+1] -= h[b - lags]
    
    return delta
```

This is O(n) with numpy vectorization, so each swap delta computation is very fast.

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())

        order = np.argsort(h[ones])[:need]

        h = np.clip(h, 0.0, 1.0)
    return h

        pass

            a = int(ones_arr[rng.randint(len(ones_arr))])
            b = int(zeros_arr[rng.randint(len(zeros_arr))])

        if a == b:
            continue

        delta = np.zeros(2 * n - 1)
        ns1 = n - 1

        m1s = max(0, ns1 - a)
        m1e = min(2 * n - 2, 2 * n - 2 - a)
        if m1s <= m1e:
            lags = np.arange(m1s, m1e + 1) - ns1
            delta[m1s:m1e + 1] -= (1.0 - h[a + lags])

        m2s = max(0, a)
        m2e = min(2 * n - 2, ns1 + a)
        if m2s <= m2e:
            lags = np.arange(m2s, m2e + 1) - ns1
            delta[m2s:m2e + 1] += h[a - lags]

        m3s = max(0, ns1 - b)
        m3e = min(2 * n - 2, 2 * n - 2 - b)
        if m3s <= m3e:
            lags = np.arange(m3s, m3e + 1) - ns1
            delta[m3s:m3e + 1] += (1.0 - h[b + lags])

        m4s = max(0, b)
        m4e = min(2 * n - 2, ns1 + b)
        if m4s <= m4e:
            lags = np.arange(m4s, m4e + 1) - ns1
            delta[m4s:m4e + 1] -= h[b - lags]

        new_max = float(np.max(corr + delta))
        accept = False
        if new_max <= cur_max + 1e-14:
            accept = True
        else:
            d = (new_max - cur_max) * 2.0 / n
            if d < 0.02 and rng.random() < np.exp(-d / cur_t):
                accept = True

        if accept:
            h[a] = 0.0
            h[b] = 1.0
            corr += delta
            cur_max = new_max
            ia = pos_o[a]
            ib = pos_z[b]
            ones_arr[ia] = b
            zeros_arr[ib] = a
            pos_o[b] = ia
            pos_z[a] = ib
            pos_o[a] = -1
            pos_z[b] = -1
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if it % 5000 == 0:
            cur_t *= 0.95

    best_c5 = best_max * 2.0 / n
    return best_h, best_c5


def smooth_max_hessian_free(h, target, alpha, lr, top_k=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return h
    weights /= w_sum

    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    ns1 = n - 1
    for m in top_idx:
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - ns1
        js = max(0, -lag)
        je = min(n, n - lag)
        if js < je:
            grad[js:je] += w * (1.0 - h[js + lag:je + lag])
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 - lag:je2 - lag]

    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new


def continuous_refine(h_init, target, time_limit, t_start):
    n = len(h_init)
    h_cur = project_feasible(h_init.copy(), target)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    alphas = [5, 15, 50, 150, 500, 1500, 5000]
    for alpha in alphas:
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        lr = 2.0 / alpha
        n_steps = 200
        for step in range(n_steps):
            if time.time() - t_start > time_limit:
                break
            h_new = smooth_max_hessian_free(h_cur, target, alpha, lr)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
            h_cur = h_new
            lr *= 0.995
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    h_init_arr = np.array(initial_h_values, dtype=float)
    n_init = len(h_init_arr)
    target_init = n_init / 2.0

    best_h = h_init_arr.copy()
    best_c5 = compute_c5(best_h)

    remaining = budget_s - 40.0

    h_bin = project_binary(h_init_arr, target_init)
    h_ls, c5_ls = fast_local_search(h_bin, target_init, 500000, rng,
                                     min(200.0, remaining * 0.3), start,
                                     focus_weight=0.6, t_sa=0.008)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    remaining = budget_s - 40.0 - (time.time() - start)

    for restart in range(8):
        if remaining < 30.0:
            break
        noise_scale = 0.15 + 0.1 * restart
        h_pert = best_h + rng.randn(n_init) * noise_scale
        h_pert_bin = project_binary(h_pert, target_init)
        alloc = min(120.0, remaining * 0.25)
        h_ls, c5_ls = fast_local_search(h_pert_bin, target_init, 300000, rng,
                                         alloc, start,
                                         focus_weight=0.7, t_sa=0.004)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - 40.0 - (time.time() - start)

    remaining = budget_s - 40.0 - (time.time() - start)

    if remaining > 60.0:
        h_cont = project_feasible(best_h.copy(), target_init)
        h_cr, c5_cr = continuous_refine(h_cont, target_init,
                                         min(120.0, remaining * 0.5), start)
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = h_cr.copy()
        remaining = budget_s - 40.0 - (time.time() - start)

    remaining = budget_s - 40.0 - (time.time() - start)

    if remaining > 50.0:
        n_use = len(best_h)
        h_bin2 = project_binary(best_h, n_use / 2.0)
        h_ls2, c5_ls2 = fast_local_search(h_bin2, n_use / 2.0, 200000, rng,
                                           min(100.0, remaining * 0.6), start,
                                           focus_weight=0.8, t_sa=0.003)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
        remaining = budget_s - 40.0 - (time.time() - start)

    remaining = budget_s - 40.0 - (time.time() - start)

    if remaining > 40.0:
        for nt in [n_init // 2, n_init * 3 // 2, n_init + 40, n_init - 40]:
            if remaining < 30.0:
                break
            if nt < 100 or nt > 2000:
                continue
            target_nt = nt / 2.0
            x_old = np.linspace(0.0, 1.0, n_init)
            x_new = np.linspace(0.0, 1.0, nt)
            h_interp = np.interp(x_new, x_old, best_h)
            h_bin_try = project_binary(h_interp, target_nt)
            alloc = min(60.0, remaining * 0.4)
            h_ls, c5_ls = fast_local_search(h_bin_try, target_nt, 150000, rng,
                                             alloc, start,
                                             focus_weight=0.6, t_sa=0.005)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                n_init = nt
            remaining = budget_s - 40.0 - (time.time() - start)

    remaining = budget_s - 40.0 - (time.time() - start)

    if remaining > 30.0:
        n_use = len(best_h)
        h_final_cont = project_feasible(best_h.copy(), n_use / 2.0)
        h_cr2, c5_cr2 = continuous_refine(h_final_cont, n_use / 2.0,
                                           min(80.0, remaining * 0.7), start)
        if c5_cr2 < best_c5:
            best_c5 = c5_cr2
            best_h = h_cr2.copy()

    remaining = budget_s - 40.0 - (time.time() - start)

    if remaining > 10.0:
        n_use = len(best_h)
        h_bin_final = project_binary(best_h, n_use / 2.0)
        h_ls_f, c5_ls_f = fast_local_search(h_bin_final, n_use / 2.0, 100000, rng,
                                             max(5.0, remaining - 5.0), start,
                                             focus_weight=0.8, t_sa=0.002)
        if c5_ls_f < best_c5:
            best_c5 = c5_ls_f
            best_h = h_ls_f

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(h_final)
    if final_c5 > best_c5 + 1e-10:
        h_final = best_h
        final_c5 = best_c5

    return h_final, final_c5, n_final
