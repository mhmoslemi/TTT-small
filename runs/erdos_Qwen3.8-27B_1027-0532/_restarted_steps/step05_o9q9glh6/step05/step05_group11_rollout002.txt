def compute_overlaps(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha, n):
    """Compute log-sum-exp smoothed objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 500.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones_like(corr) / len(corr)
    else:
        weights /= wsum
    
    n_half = n - 1
    grad = np.zeros(n)
    
    for i in range(len(corr)):
        w = weights[i]
        if w < 1e-10:
            continue
        s = i - n_half
        if s >= 0:
            ns = n - s
            grad[:ns] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:ns])
        else:
            sa = -s
            ns = n - sa
            grad[:ns] += w * (1.0 - h[sa:])
            grad[sa:] += w * (-h[:ns])
    
    grad *= dx
    return obj, grad

def augmented_lagrangian_lbfgsb(h_init, n, target, alphas, time_limit, 
                                  max_lbfgs_iter=200, rho_start=1.0, rho_factor=2.0):
    """Augmented Lagrangian method with L-BFGS-B for the inner problem."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lam = 0.0
    rho = rho_start
    
    for alpha_idx, alpha in enumerate(alphas):
        if time.time() - start > time_limit - 10:
            break
        
        # Inner loop: solve augmented Lagrangian subproblem
        for inner_it in range(3):
            if time.time() - start > time_limit - 5:
                break
            
            def obj_func(x):
                o, _ = smooth_obj_grad(x, alpha, n)
                diff = np.sum(x) - target
                return o + lam * diff + 0.5 * rho * diff * diff
            
            def grad_func(x):
                _, g = smooth_obj_grad(x, alpha, n)
                diff = np.sum(x) - target
                g = g + lam * np.ones(n) + rho * diff * np.ones(n)
                return g
            
            try:
                res = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': max_lbfgs_iter, 'ftol': 1e-16,
                                       'gtol': 1e-10, 'maxls': 50})
                h_new = res.x
            except Exception:
                break

            diff = np.sum(h) - target
            lam = lam + rho * diff
            lam = np.clip(lam, -100.0, 100.0)

            if c5_val < best_c5 - 1e-14:

    return best_h, best_c5

def fista_optimize(h_init, n, target, alphas, time_limit, max_iter=5000):
    """FISTA (accelerated proximal gradient) with smooth max."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit - 10:
            break
        
        # Compute Lipschitz constant estimate
        # ||∇f_α||_L ≤ α * ||correlation operator||_L * ... 
        # For safety, use a conservative estimate
        # The gradient magnitude is bounded by about 2*dx*n = 4 for our problem
        # Actually let's estimate it numerically
        _, g0 = smooth_obj_grad(h, alpha, n)
        L = np.linalg.norm(g0) * 2 + alpha * np.max(np.abs(
            np.correlate(h, 1.0-h, mode='full'))) * 2.0 / n + 1.0
        L = max(L, 1.0)
        
        t_k = 1.0
        y = h.copy()
        
        for it in range(max_iter):
            if time.time() - start > time_limit - 5:

            _, grad_y = smooth_obj_grad(y, alpha, n)
            
            # Gradient step
            z = y - grad_y / L
            
            # Proximal step: project onto feasible set
            x_new = project_feasible(z, target)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

            best_h = h.copy()

        t_k = 1.0
        y = h.copy()
    
    return best_h, best_c5

def lp_refine(h_init, n, target, max_iter, time_limit, n_top=150, 
              damping=0.9, reg=1e-5):
    """Alternating LP refinement."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        overlaps = np.correlate(g, h, mode='full') * dx
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for idx in top_idx:
            k = idx - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas, maxiter=1000):

            o, _ = smooth_obj_grad(x, alpha, n)
            return o

            _, g = smooth_obj_grad(x, alpha, n)
            return g
        
        try:

                          bounds=bounds,

                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def coordinate_descent_step(h, n, j, target, dx):
    """Find optimal h[j] given all other h[i]."""
    # Compute current overlaps
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    n_half = n - 1
    
    # For each lag s, compute the slope with respect to h[j]
    # and the intercept (value at t=0)
    a_vals = np.zeros(2*n - 1)
    b_vals = np.zeros(2*n - 1)
    
    for i in range(2*n - 1):
        s = i - n_half
        O_s = corr[i]
        slope = 0.0
        base = O_s  # O_s at current h[j]
        
        if s >= 0:
            ns = n - s
            if j < ns:  # h[j] appears as h[j]*(1-h[j+s])
                slope += (1.0 - h[j+s])
                base -= h[j] * (1.0 - h[j+s])
            if j >= s:  # h[j] appears in (1-h[j]) = 1-h[j]
                slope -= h[j-s]
                base += h[j-s] * h[j]  # was h[j-s]*(1-h[j]) = h[j-s] - h[j-s]*h[j], now removing -h[j-s]*h[j] part
        
        # a_s = base, b_s = slope (both scaled by dx)
        a_vals[i] = base
        b_vals[i] = slope
    
    a_vals *= dx
    b_vals *= dx
    
    # Now minimize max_s (a_s + b_s * t) for t in [0, 1]
    # Binary search for where subdifferential contains 0
    lo = 0.0
    hi = 1.0
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        # Find which lines are active at t=mid
        vals = a_vals + b_vals * mid
        active = np.where(vals == vals.max())[0]
        slopes_active = b_vals[active]
        
        if np.all(slopes_active > 1e-14):
            # Function increasing at mid, minimum is to the left
            hi = mid
        elif np.all(slopes_active < -1e-14):
            # Function decreasing at mid, minimum is to the right
            lo = mid
        else:
            # 0 in subdifferential
            lo = mid
            hi = mid
            break
    
    t_opt = (lo + hi) / 2.0
    t_opt = np.clip(t_opt, 0.0, 1.0)
    return t_opt
