                       T_init, T_min, cooling_rate, block_size_range, rng):

    best_h = h_init.copy()

    cur_score = best_score
    
    T = T_init
    n_steps = 0
    accept_count = 0

            if acc_rate > 0.6:
                cooling_rate *= 1.05
            elif acc_rate < 0.1:

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_weights = None

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def sa_block_perturb(h_cur, h_cand, n, target, block_size, noise_scale, rng):
    """Perturb a random block in h_cand based on h_cur."""
    start = rng.randint(0, n - block_size + 1)
    h_cand[start:start+block_size] = h_cur[start:start+block_size] + rng.randn(block_size) * noise_scale
    h_cand = project_feasible(h_cand, target)
    return h_cand

def sa_swap_perturb(h_cur, n, target, size, rng):
    """Move mass from one region to another."""
    h_new = h_cur.copy()
    i1 = rng.randint(0, n - size)
    i2 = rng.randint(0, n - size)
    if abs(i1 - i2) < size:
        return h_new
    delta = rng.uniform(-0.1, 0.1)
    h_new[i1:i1+size] += delta
    h_new[i2:i2+size] -= delta
    h_new = project_feasible(h_new, target)
    return h_new

def simulated_annealing(h_init, n, target, rng, deadline, 
                        T_init=0.01, T_min=1e-6, cooling=0.9995,
                        block_sizes=(3, 20), noise_scale=0.05,
                        polish_interval=2000, polish_alphas=(100, 500)):
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    
    T = T_init
    step = 0
    accept_count = 0
    window = 500
    
    while time.time() < deadline:
        # Adaptive cooling
        if step > 0 and step % window == 0:
            acc_rate = accept_count / window
            if acc_rate > 0.6:
                cooling = min(cooling * 1.01, 0.999)
            elif acc_rate < 0.05:
                cooling = max(cooling * 0.99, 0.9998)
            accept_count = 0
        
        # Choose perturbation type
        r = rng.rand()
        h_new = h_cur.copy()
        if r < 0.6:
            # Block perturbation
            bs = rng.randint(block_sizes[0], block_sizes[1] + 1)
            bstart = rng.randint(0, n - bs + 1)
            h_new[bstart:bstart+bs] += rng.randn(bs) * noise_scale * np.sqrt(T / T_init)
        elif r < 0.8:
            # Swap perturbation
            sz = rng.randint(2, 10)
            i1 = rng.randint(0, n - sz)
            i2 = rng.randint(0, n - sz)
            if abs(i1 - i2) >= sz:
                delta = rng.uniform(-0.05, 0.05)
                h_new[i1:i1+sz] += delta
                h_new[i2:i2+sz] -= delta
        else:
            # Single-point perturbation
            idx = rng.randint(n)
            h_new[idx] += rng.randn() * noise_scale * 2
        
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)
        
        # Metropolis acceptance
        delta = new_score - cur_score
        if delta < 0:
            h_cur = h_new
            cur_score = new_score
            accept_count += 1
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
        elif T > T_min and rng.rand() < np.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score
            accept_count += 1
        
        T = max(T * cooling, T_min)
        step += 1
        
        # Periodic SLSQP polish
        if step % polish_interval == 0 and time.time() < deadline - 60:
            h_polished = slsqp_round(h_cur, n, target, polish_alphas,

                best_score = sc
                best_h = h_polished.copy()
            cur_score = max(sc, cur_score - 0.001)

def targeted_sa(h_init, n, target, rng, deadline, 
                n_worst=3, noise_scale=0.03, T_init=0.005, T_min=1e-6, cooling=0.999):
    """SA that focuses perturbations on regions affecting the worst shifts."""
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    T = T_init
    dx = 2.0 / n
    
    while time.time() < deadline:
        # Find worst shifts
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        worst_shifts = np.argsort(over)[::-1][:n_worst]
        
        # Pick a worst shift and perturb the overlapping region
        k = int(rng.choice(worst_shifts))
        
        # Regions that contribute to shift k: h[i] and h[i+k] for i in [0, n-1-k]
        # To reduce overlap, we want to make h[i] smaller where h[i+k] is large,
        # or vice versa
        h_new = h_cur.copy()
        
        # Perturb in the "overlap region" for this shift
        overlap_len = n - k
        # Focus on a sub-region of the overlap
        sub_size = rng.randint(5, min(30, overlap_len))
        sub_start = rng.randint(0, max(1, overlap_len - sub_size))
        
        # Add noise to the first part and subtract from the shifted part
        noise = rng.randn(sub_size) * noise_scale * np.sqrt(T / T_init)
        h_new[sub_start:sub_start+sub_size] += noise
        if sub_start + k + sub_size <= n:
            h_new[sub_start+k:sub_start+k+sub_size] -= noise * 0.5
        
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)
        
        delta = new_score - cur_score
        if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
        
        T = max(T * cooling, T_min)
    
    return best_h, best_score


def de_search(n, target, pop_size, max_generations, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = list(range(pop_size))
            candidates.remove(i)
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_polish(h, n, target, n_steps=500, n_active=3, lr_init=0.002,
                       lr_decay=0.995, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-9:
                break
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def de_search(n, target, pop_size, max_generations, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = list(range(pop_size))
            candidates.remove(i)
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        elif r < 0.7:

        elif r < 0.85:

        else:
            bs = rng.randint(2, block_sizes[1] + 1)

        h_new = project_feasible(h_new, target)

def targeted_sa(h_init, n, target, rng, deadline,
                n_worst=3, noise_scale=0.025, T_init=0.004,
                T_min=1e-7, cooling=0.999):
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    T = T_init

    while time.time() < deadline:
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        worst_shifts = np.argsort(over)[::-1][:n_worst]
        k = int(rng.choice(worst_shifts))

        h_new = h_cur.copy()
        overlap_len = n - k
        if overlap_len > 10:
            sub_size = rng.randint(5, min(30, overlap_len))
            sub_start = rng.randint(0, max(1, overlap_len - sub_size))
            scale = noise_scale * np.sqrt(T / T_init)
            noise = rng.randn(sub_size) * scale
            h_new[sub_start:sub_start + sub_size] += noise
            end2 = sub_start + k + sub_size
            if end2 <= n:
                h_new[sub_start + k:end2] -= noise * 0.5

        r = rng.rand()
        if r < 0.4:
            bs = rng.randint(3, 15)
            bstart = rng.randint(0, n - bs + 1)
            h_new[bstart:bstart + bs] += rng.randn(bs) * noise_scale * 0.5 * np.sqrt(T / T_init)

        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)

        delta = new_score - cur_score
        if delta < 0:
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
        elif T > T_min and rng.rand() < np.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score

        T = max(T * cooling, T_min)

    return best_h, best_score


def subgradient_polish(h, n, target, n_steps=500, n_active=3, lr_init=0.002,
                       lr_decay=0.995, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-9:
                break
    return best_h, best_score


def momentum_wgd(h, n, target, n_steps=500, beta=500, lr=0.0008,
                 moment=0.85, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-8:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            v *= 0.8
    return h_cur, cur_score

        best_h, n, target, rng, deadline_p3,
        T_init=0.008, T_min=1e-7, cooling=0.9996,
        block_sizes=(3, 25), noise_scale=0.04,
        polish_every_s=70, reheat_threshold=40000)
    if sc_sa < best_score:
        best_score = sc_sa
        best_h = h_sa.copy()

    if time.time() < start_time + budget_s - 250:
        deadline_p4 = start_time + 780
        h_ts, sc_ts = targeted_sa(
            best_h, n, target, rng, deadline_p4,
            n_worst=3, noise_scale=0.02, T_init=0.003,
            T_min=1e-7, cooling=0.999)
        if sc_ts < best_score:
            best_score = sc_ts
            best_h = h_ts.copy()
        h_mom, sc_mom = momentum_wgd(best_h, n, target, n_steps=400, beta=800,
                                     lr=0.0006, moment=0.85,
                                     deadline=start_time + budget_s - 220)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom.copy()

    if time.time() < start_time + budget_s - 150:
        deadline_p5 = start_time + 950
        h_p5 = slsqp_round(best_h, n, target, [500, 2000, 8000], 3000, 1e-16, deadline=deadline_p5)
        sc = compute_c5(h_p5)
        if sc < best_score:
            best_score = sc
            best_h = h_p5.copy()

    if time.time() < start_time + budget_s - 120:
        deadline_p6 = start_time + 1120
        h_fine, sc_fine = simulated_annealing(
            best_h, n, target, rng, deadline_p6,
            T_init=0.003, T_min=1e-8, cooling=0.9997,
            block_sizes=(2, 12), noise_scale=0.015,
            polish_every_s=50, reheat_threshold=30000)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine.copy()

    if time.time() < start_time + budget_s - 60:
        deadline_p7 = start_time + budget_s - 25
        h_final = slsqp_round(best_h, n, target, [5000, 20000], 3000, 1e-17, deadline=deadline_p7)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final.copy()

    if time.time() < start_time + budget_s - 5:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=300, n_active=5,
                                         lr_init=0.0008, lr_decay=0.998,
                                         deadline=start_time + budget_s - 3)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
