```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
```

To address this, I can add a simple heuristic: if the sum is too high (sum > target), bias the coordinate updates downward (add a small penalty μ*δ to the objective). If sum < target, bias upward.

This is the Lagrange multiplier idea. Let me implement it:

```python
def fast_cd_with_lambda(h, n, target, deadline, rng, precomp):
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    B = compute_B_matrix(h_cur, precomp)
    dx = 2.0 / n
    best_score = float(np.max(corr)) * dx
    best_h = h_cur.copy()
    
    mu = 0.0  # Lagrange multiplier for sum constraint
    sweep = 0
    stall = 0
    
    while time.time() < deadline:
        sweep += 1
        
        # Adjust mu based on current sum
        cur_sum = h_cur.sum()
        mu = 0.1 * (cur_sum - target)  # simple PI controller-ish
        
        order = np.random.permutation(n, random_state=rng)  # or use rng
        rng.shuffle(order)
        
        for j in order:
            hj = h_cur[j]
            if hj <= 1e-13 or hj >= 1.0 - 1e-13:
                continue
            
            lo = -hj
            hi = 1.0 - hj
            B_row = B[j]
            
            for _ in range(30):
                mid = (lo + hi) * 0.5
                vals = corr + mid * B_row + mid * 0  # + mu effect
                # With mu: minimize max_k(corr[k] + mid*B_row[k]) + mu*mid
                # Subgradient: B_row[k*] + mu for active k*
                mx = vals.max()
                active = vals > mx - 1e-9
                gmin = B_row[active].min() + mu
                gmax = B_row[active].max() + mu
                if gmin > 1e-11:
                    hi = mid
                elif gmax < -1e-11:
                    lo = mid
                else:
                    break
            
            delta = (lo + hi) * 0.5
            new_v = max(0.0, min(1.0, hj + delta))
            d = new_v - hj
            if abs(d) > 1e-13:
                h_cur[j] = new_v
                corr += d * B_row
        
        # Recompute B
        B = compute_B_matrix(h_cur, precomp)
        
        # Project (mild)
        if abs(h_cur.sum() - target) > 0.5:
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            B = compute_B_matrix(h_cur, precomp)
        
        score = float(np.max(corr)) * dx
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 30:
            h_cur = project_feasible(h_cur + rng.randn(n) * 0.02, target)
            corr = compute_corr(h_cur)
            B = compute_B_matrix(h_cur, precomp)
            stall = 0
    
    h_final = project_feasible(best_h, target)
    return h_final, compute_c5(h_final)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_precomp = None


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def precompute_B_masks(n):
    j_idx = np.arange(n)[:, None]
    l_idx = np.arange(-(n - 1), n)[None, :]
    idx1 = j_idx - l_idx
    idx2 = j_idx + l_idx
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    return idx1, idx2, mask1, mask2


def compute_B_matrix(h, precomp):
    idx1, idx2, mask1, mask2 = precomp
    B = np.zeros((len(h), 2 * len(h) - 1), dtype=float)
    B[mask1] = 1.0 - h[idx1[mask1]]
    B[mask2] -= h[idx2[mask2]]
    return B


def fast_coord_descent(h, n, target, deadline, rng, precomp, recompute_every=1):
    h_cur = h.astype(float).copy()
    corr = compute_corr(h_cur)
    B = compute_B_matrix(h_cur, precomp)
    best_score = float(np.max(corr))
    best_h = h_cur.copy()
    sweep = 0
    stall = 0
    order = np.arange(n)

    while time.time() < deadline:
        sweep += 1
        rng.shuffle(order)

        for idx in order:
            j = int(idx)
            hj = h_cur[j]
            if hj <= 1e-13 or hj >= 1.0 - 1e-13:
                continue
            lo = -hj
            hi = 1.0 - hj
            B_row = B[j]
            for _ in range(32):
                mid = (lo + hi) * 0.5
                vals = corr + mid * B_row
                mx = vals.max()
                active = vals > mx - 1e-8
                if not np.any(active):
                    active = vals == mx
                gmin = B_row[active].min()
                gmax = B_row[active].max()
                if gmin > 1e-10:
                    hi = mid
                elif gmax < -1e-10:
                    lo = mid
                else:
                    break
            delta = (lo + hi) * 0.5
            new_v = hj + delta
            new_v = max(0.0, min(1.0, new_v))
            d = new_v - hj
            if abs(d) > 1e-12:
                h_cur[j] = new_v
                corr = corr + d * B_row

        if sweep % recompute_every == 0:
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            B = compute_B_matrix(h_cur, precomp)

        score = float(np.max(corr))
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1

        if stall > 50 and sweep > 30:
            scale = 0.02 + 0.01 * min(stall // 25, 5)
            h_cur = project_feasible(h_cur + rng.randn(n) * scale, target)
            corr = compute_corr(h_cur)
            B = compute_B_matrix(h_cur, precomp)
            stall = 0

    h_final = project_feasible(best_h, target)
    return h_final, compute_c5(h_final)


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_polish(h, n, target, deadline, precomp):
    h_cur = project_feasible(h.copy(), target)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step = 0.005
    stagnant = 0

    for _ in range(50000):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        active = corr > c_max - 0.01 * c_max
        B = compute_B_matrix(h_cur, precomp)
        sens = B[:, active].mean(axis=1)
        h_new = project_feasible(h_cur - step * sens, target)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 20 == 0:
                step *= 0.9
        if step < 1e-12:
            break

    return best_h, compute_c5(best_h)

    global _g_n, _g_target, _g_alpha
    start = time.time()
    rng = np.random.RandomState(seed)
    effective_budget = budget_s - 25.0

    n = len(initial_h_values)

    # Phase 4: Subgradient polish

            best_score = s4
            best_h = h4

        deadline5 = start + 0.93 * effective_budget

    final_score = compute_c5(h_out)
```
