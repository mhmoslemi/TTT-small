            grad[:n - mk] += (1.0 - h[mk:])
            grad[mk:] -= h[:n - mk]
    return grad


def active_set_descent(h, n, target, deadline, max_iter=3000, active_frac=0.15):
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * (2.0 / n)
    dx = 2.0 / n

    for it in range(max_iter):
        if time.time() > deadline:
            break

        max_c = float(np.max(corr))
        threshold = max_c - active_frac * max_c
        active_mask = corr >= threshold
        active_idx = np.where(active_mask)[0]

        if len(active_idx) < 3:
            active_idx = np.argsort(corr)[::-1][:8]

        grad = compute_active_gradient(h_cur, active_idx, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        scale = 1.0 / (max(len(active_idx), 1) * n) * 10.0
        best_obj = float(np.max(corr))
        best_h = None
        best_corr = None

        for step_mult in [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]:
            alpha = step_mult * scale
            h_try = project_feasible(h_cur - alpha * grad, target)
            corr_try = compute_corr(h_try)
            obj_try = float(np.max(corr_try))
            if obj_try < best_obj:
                best_obj = obj_try
                best_h = h_try
                best_corr = corr_try

        if best_h is None:
            break

        h_cur = best_h
        corr = best_corr
        score = best_obj * dx
        if score < best_score:
            best_score = score
        elif it > 50:
            break

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=30, bi_search_iters=55):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0

    for sweep in range(n_sweeps if n_sweeps else 200000):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            lo = 0.0
            hi = float(np.max(A + C * t_max))

            for _ in range(bi_search_iters):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                zero = ~(pos | neg)
                if np.any(zero) and float(np.max(A[zero])) > mid + 1e-12:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-10:
                    hi = mid
                else:
                    lo = mid

            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 400 and sweep > 300:
            n_pert = max(n // 8, 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            h_cur[idx] = np.clip(h_cur[idx] + rng.randn(n_pert) * 0.08, 0, 1)
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=3000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def fourier_perturb(h, n, rng, num_modes=5, amp=0.03):
    h_new = h.copy()
    x = np.arange(n)
    for _ in range(num_modes):
        freq = rng.randint(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(-1, 1) * amp
        h_new += a * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h_new, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Paired coordinate descent (main workhorse)
    deadline_p1 = start_time + (budget_s - 15.0) * 0.28
    h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p1, rng,
                                    recompute_interval=30, bi_search_iters=55)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: Active-set gradient descent
    if tleft() > 120:
        deadline_p2 = start_time + (budget_s - 15.0) * 0.38
        h_as, sc_as = active_set_descent(best_h, n, target, deadline_p2,
                                         max_iter=2000, active_frac=0.1)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as
        # Run again with tighter active set
        if tleft() > 100:
            deadline_p2b = start_time + (budget_s - 15.0) * 0.45
            h_as2, sc_as2 = active_set_descent(best_h, n, target, deadline_p2b,
                                               max_iter=2000, active_frac=0.05)
            if sc_as2 < best_score:
                best_score = sc_as2
                best_h = h_as2

    # Phase 3: SLSQP polish (aggressive)
    if tleft() > 150:
        h_ref = slsqp_polish(best_h, n, target, [100, 500, 2000, 10000, 30000],
                             maxiter=4000, ftol=1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: More coordinate descent after SLSQP
    if tleft() > 200:
        deadline_p4 = start_time + (budget_s - 15.0) * 0.62
        h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p4, rng,
                                        recompute_interval=25, bi_search_iters=60)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: Active-set + coordinate descent interleaved
    if tleft() > 180:
        for round_num in range(3):
            if tleft() < 80:
                break
            dl = start_time + (budget_s - 15.0) * (0.65 + 0.05 * round_num)
            h_as, sc_as = active_set_descent(best_h, n, target, dl,
                                             max_iter=1000, active_frac=0.08)
            if sc_as < best_score:
                best_score = sc_as
                best_h = h_as
            if tleft() < 60:
                break
            dl2 = start_time + (budget_s - 15.0) * (0.67 + 0.05 * round_num)
            h_cd, sc_cd = paired_coord_desc(best_h, n, target, dl2, rng,
                                            recompute_interval=25, bi_search_iters=55)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

    # Phase 6: Multi-scale approach
    if tleft() > 200:
        n_coarse = max(n // 3, 60)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        deadline_coarse = start_time + (budget_s - 15.0) * 0.78
        h_coarse, sc_coarse = paired_coord_desc(h_coarse, n_coarse, target_coarse,
                                                deadline_coarse, rng,
                                                recompute_interval=20, bi_search_iters=55)
        h_up = interpolate_to(h_coarse, n, target)
        h_up = slsqp_polish(h_up, n, target, [500, 3000, 15000], 3000, 1e-16)
        deadline_up = start_time + (budget_s - 15.0) * 0.85
        h_up, sc_up = paired_coord_desc(h_up, n, target, deadline_up, rng,
                                        recompute_interval=25, bi_search_iters=55)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 7: Fourier-based restarts
    if tleft() > 100:
        for attempt in range(6):
            if tleft() < 50:
                break
            h_pert = fourier_perturb(best_h, n, rng, num_modes=3 + attempt,
                                     amp=0.02 + 0.01 * attempt)
            dl = start_time + (budget_s - 15.0) * (0.87 + 0.015 * attempt)
            h_res, sc_res = paired_coord_desc(h_pert, n, target, dl, rng,
                                              recompute_interval=25, bi_search_iters=50)
            if sc_res < best_score:
                best_score = sc_res
                best_h = h_res

    # Phase 8: Final SLSQP + active set + coordinate descent squeeze
    if tleft() > 60:
        h_final = slsqp_polish(best_h, n, target, [2000, 10000, 50000], 5000, 1e-17)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 30:
        dl_final = start_time + budget_s - 12.0
        h_final, sc_final = active_set_descent(best_h, n, target, dl_final,
                                               max_iter=3000, active_frac=0.03)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    if tleft() > 15:
        h_final2, sc_final2 = paired_coord_desc(best_h, n, target,
                                                start_time + budget_s - 10.0, rng,
                                                recompute_interval=20, bi_search_iters=55)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_h = h_final2

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
