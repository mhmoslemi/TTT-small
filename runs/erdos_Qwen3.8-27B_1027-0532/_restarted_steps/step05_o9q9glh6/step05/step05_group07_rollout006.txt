import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective_and_grad(h, alpha):
    """Compute log-sum-exp objective and its gradient."""
    n = len(h)
    corr = compute_corr(h)
    # Normalize for numerical stability
    corr_shifted = alpha * (corr - np.max(corr))
    log_sum = np.max(corr) + np.log(np.sum(np.exp(corr_shifted))) / alpha
    # softmax weights
    weights = np.exp(corr_shifted) / np.sum(np.exp(corr_shifted))
    # Gradient: sum_k weights[k] * dcorr[k]/dh
    # dcorr[m]/dh[j] for m = n-1+k (k>=0): (1-h[j-k]) [k<=j<=n-1] - h[j+k] [0<=j<=n-1-k]
    # dcorr[m]/dh[j] for m = n-1-k (k>0): (1-h[j+k]) [0<=j<=n-1-k] - h[j-k] [k<=j<=n-1]
    
    grad = np.zeros(n)
    # For m = n-1 (k=0): corr = sum h[i](1-h[i]), grad = 1-2h
    # This is included in both formulas with k=0
    
    # Let's compute for each m from 0 to 2n-2
    # m = n-1+k for k = 0,...,n-1
    for k in range(n):
        w = weights[n-1+k]
        if w < 1e-15:
            continue
        # dcorr[n-1+k]/dh[j] = (1-h[j-k]) [k<=j<=n-1] - h[j+k] [0<=j<=n-1-k]
        if k == 0:
            # (1-h[j]) - h[j] = 1-2h[j] for all j
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            # (1-h[j-k]) for j=k,...,n-1
            grad[k:] += w * (1.0 - h[:n-k])
            # -h[j+k] for j=0,...,n-1-k
            grad[:n-k] -= w * h[k:]
    
    # m = n-1-k for k = 1,...,n-1
    for k in range(1, n):
        w = weights[n-1-k]
        if w < 1e-15:
            continue
        # dcorr[n-1-k]/dh[j] = (1-h[j+k]) [0<=j<=n-1-k] - h[j-k] [k<=j<=n-1]
        grad[:n-k] += w * (1.0 - h[k:])
        grad[k:] -= w * h[:n-k]
    
    return log_sum, grad
