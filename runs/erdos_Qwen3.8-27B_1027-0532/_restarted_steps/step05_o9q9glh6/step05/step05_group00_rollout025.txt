def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def sa_mixed(h, target_sum, max_iters, t_init, cool_rate, rng,
             time_limit=None, start_time=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 25000
    t = t_init

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > start_time + time_limit:
                break
        if stall > max_stall:
            break

        t *= cool_rate

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.35:
            worst_idx = np.argsort(corr)[-12:]
            k = worst_idx[rng.randint(len(worst_idx))] - (n - 1)
            ip = np.array([i for i in ones if 0 <= i + k < n])
            jp = np.array([j for j in zeros if 0 <= j + k < n])
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
        elif r < 0.55:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif r < 0.78:
            blen = int(rng.lognormal(mean=1.8, sigma=1.2))
            blen = max(4, min(blen, n - 10))
            a = rng.randint(0, n - blen)
            b = a + blen
            h_new[a:b] = h_new[b - 1:a - 1:-1]
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True

        if not moved:
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-abs(delta) / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def greedy_improve(h, max_iters, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > start_time + time_limit:
                break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = np.array([i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0])
        j_cand = np.array([j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1])

        if len(i_cand) == 0 or len(j_cand) == 0:
            break

        top_threshold = max_val * 0.88
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])

        best_improve = 0.0
        best_i = -1
        best_j = -1

        n_i = min(len(i_cand), 25)
        n_j = min(len(j_cand), 25)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                new_max = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > new_max:
                        new_max = v
                        break
                if new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
        else:
            break

    return best_h, best_c5


def legendre_sequence(p):
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
    return (leg >= 0).astype(float)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy
    allowed = min(25.0, tleft() * 0.06)
    h_g, c5_g = greedy_improve(best_h, max_iters=50000, time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Long SA with mixed moves (main workhorse)
    if tleft() > 30:
        allowed = min(350.0, tleft() * 0.4)
        max_it = int(allowed * 2200)
        h_sa, c5_sa = sa_mixed(best_h, target, max_it, 0.007, 0.99975, rng,
                               time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Greedy polish
    allowed = min(20.0, tleft() * 0.04)
    h_g2, c5_g2 = greedy_improve(best_h, max_iters=30000, time_limit=allowed, start_time=start_time)
    if c5_g2 < best_score:
        best_score = c5_g2
        best_h = h_g2.copy()

    # Phase 4: Perturb + SA
    if tleft() > 200:
        allowed = min(280.0, tleft() * 0.35)
        max_it2 = int(allowed * 2200)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(6):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
        h_sa2, c5_sa2 = sa_mixed(h_pert, target, max_it2, 0.005, 0.9998, rng,
                                 time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Try different n values
    if tleft() > 100:
        n_tries = [n + 2, n + 4, n - 2, n + 8]
        for n_try in n_tries:
            if tleft() < 60:
                break
            if n_try < 100 or n_try % 2 != 0:
                continue
            target_try = n_try // 2
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, n)
            h_interp = np.interp(x_new, x_old, best_h)
            h_bin = project_binary(h_interp, target_try)

            allowed = min(15.0, tleft() * 0.03)
            h_gt, c5_gt = greedy_improve(h_bin, max_iters=20000, time_limit=allowed, start_time=start_time)

            if tleft() < 40:
                continue
            allowed2 = min(40.0, tleft() * 0.06)
            max_it3 = int(allowed2 * 2200)
            h_sat, c5_sat = sa_mixed(h_gt, target_try, max_it3, 0.006, 0.9997, rng,
                                     time_limit=allowed2, start_time=start_time)
            if c5_sat < best_score:
                best_score = c5_sat
                best_h = h_sat.copy()

    # Phase 6: Legendre-based construction
    if tleft() > 80:
        # Find prime near n that is 3 mod 4
        p = 607
        h_leg = legendre_sequence(p)
        # Truncate or pad to n
        if p > n:
            h_leg_n = h_leg[:n].copy()
        elif p < n:
            h_leg_n = np.zeros(n)
            h_leg_n[:p] = h_leg
            h_leg_n[p:] = 0.5
        else:
            h_leg_n = h_leg.copy()
        h_leg_bin = project_binary(h_leg_n, target)
        c5_leg = compute_c5(h_leg_bin)
        if c5_leg < best_score:
            best_score = c5_leg
            best_h = h_leg_bin.copy()

        allowed = min(30.0, tleft() * 0.05)
        h_lg, c5_lg = greedy_improve(best_h, max_iters=20000, time_limit=allowed, start_time=start_time)
        if c5_lg < best_score:
            best_score = c5_lg
            best_h = h_lg.copy()

        if tleft() > 40:
            allowed2 = min(40.0, tleft() * 0.06)
            max_it4 = int(allowed2 * 2200)
            h_ls, c5_ls = sa_mixed(best_h, target, max_it4, 0.005, 0.9997, rng,
                                   time_limit=allowed2, start_time=start_time)
            if c5_ls < best_score:
                best_score = c5_ls
                best_h = h_ls.copy()

    # Phase 7: ILS with diverse perturbations
    if tleft() > 80:
        for pert_level in [3, 5, 8, 12]:
            if tleft() < 30:
                break
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(pert_level):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(15.0, tleft() * 0.03)
            h_ir, c5_ir = greedy_improve(h_pert, max_iters=15000, time_limit=allowed, start_time=start_time)

            if tleft() < 20:
                continue
            allowed2 = min(25.0, tleft() * 0.04)
            max_it5 = int(allowed2 * 2200)
            h_sr, c5_sr = sa_mixed(h_ir, int(len(h_ir) // 2), max_it5, 0.004, 0.9998, rng,
                                   time_limit=allowed2, start_time=start_time)
            if c5_sr < best_score:
                best_score = c5_sr
                best_h = h_sr.copy()

    # Phase 8: Final long SA with very slow cooling
    if tleft() > 30:
        allowed = max(5.0, min(tleft() - 10.0, 100.0))
        max_it6 = int(allowed * 2500)
        h_sf, c5_sf = sa_mixed(best_h, int(len(best_h) // 2), max_it6, 0.002, 0.99992, rng,
                               time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Final greedy polish
    if tleft() > 5:
        allowed = max(1.0, tleft() - 3.0)
        h_fg, c5_fg = greedy_improve(best_h, max_iters=20000, time_limit=allowed, start_time=start_time)
        if c5_fg < best_score:
            best_score = c5_fg
            best_h = h_fg.copy()

    # Ensure final feasibility
    n_final = len(best_h)
    target_final = n_final // 2
    h_final = project_binary(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
