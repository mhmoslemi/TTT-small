    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    deadline = start + budget_s - 10.0

    # Phase 1: Fast coordinate descent with exact min-max solver (40%)
    d1 = start + (budget_s - 20.0) * 0.40
    h1, s1 = coord_desc_exact(best_h, n, target, d1, rng)
    if s1 < best_score:
        best_score, best_h = s1, h1

    # Phase 2: SLSQP polish (15%)
    d2 = start + (budget_s - 20.0) * 0.55
    if time.time() < d2:
        h2 = slsqp_polish(best_h, n, target, [100, 500, 2000, 8000], 3000)
        s2 = compute_c5(h2)
        if s2 < best_score:
            best_score, best_h = s2, h2

    # Phase 3: More coordinate descent (20%)
    d3 = start + (budget_s - 20.0) * 0.75
    if time.time() < d3:
        h3, s3 = coord_desc_exact(best_h, n, target, d3, rng, perturb_interval=100, perturb_scale=0.015)
        if s3 < best_score:
            best_score, best_h = s3, h3

    # Phase 4: High-alpha SLSQP (10%)
    d4 = start + budget_s - 60.0
    if time.time() < d4:
        h4 = slsqp_polish(best_h, n, target, [5000, 20000, 80000], 3000)
        s4 = compute_c5(h4)
        if s4 < best_score:
            best_score, best_h = s4, h4

    # Phase 5: Final coordinate descent (15%)
    if time.time() < start + budget_s - 20.0:
        h5, s5 = coord_desc_exact(best_h, n, target, start + budget_s - 15.0, rng,
                                  perturb_interval=80, perturb_scale=0.01)
        if s5 < best_score:
            best_score, best_h = s5, h5

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, n
