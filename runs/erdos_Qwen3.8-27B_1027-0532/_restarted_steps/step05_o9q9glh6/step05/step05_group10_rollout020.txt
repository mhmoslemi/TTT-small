def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def time_remaining(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)


def alternating_lp(h_init, n, target, max_iter, time_limit,
                   damping_init=0.85, reg=1e-4, n_active=60):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=100000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())

        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-5:
                    active_k.add(k_idx)
                    if len(active_k) > n_active + 30:
                        break

        active_k = top_k | active_k
        if len(active_k) > 2 * n_active:
            active_k = set(np.argsort(corr)[-2 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.94)
        else:
            damping = min(0.95, damping + 0.01)

        if it % 40 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.6)

    return best_h, best_c5


def coord_descent_polish(h, n, target, max_sweeps=150, time_limit=60):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit - 3:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        current_max = float(np.max(g))

        improved = False
        for i in range(n):
            beta = np.zeros(2 * n - 1)

            lo1 = -i
            hi1 = n - 1 - i
            if hi1 >= lo1:
                idx = np.arange(lo1, hi1 + 1)
                beta[idx + half_n] += (1.0 - h[i + idx])

            lo2 = i - (n - 1)
            hi2 = i
            if hi2 >= lo2:
                idx = np.arange(lo2, hi2 + 1)
                beta[idx + half_n] += (-h[i - idx])

            beta *= dx

            lo_d = -h[i]
            hi_d = 1.0 - h[i]

            if hi_d - lo_d < 1e-8:
                continue

            f_lo = float(np.max(g + lo_d * beta))
            f_hi = float(np.max(g + hi_d * beta))
            mid_d = (lo_d + hi_d) * 0.5
            f_mid = float(np.max(g + mid_d * beta))

            best_f = min(f_lo, f_hi, f_mid)
            if best_f >= current_max - 1e-11:
                continue

            if f_lo <= f_hi and f_lo <= f_mid:
                a, b = lo_d, hi_d
            elif f_hi <= f_lo and f_hi <= f_mid:
                a, b = lo_d, hi_d
            else:
                a, b = lo_d, hi_d

            c1 = b - phi * (b - a)
            c2 = a + phi * (b - a)
            f_c1 = float(np.max(g + c1 * beta))
            f_c2 = float(np.max(g + c2 * beta))

            for _ in range(35):
                if f_c1 < f_c2:
                    b = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = b - phi * (b - a)
                    f_c1 = float(np.max(g + c1 * beta))
                else:
                    a = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = a + phi * (b - a)
                    f_c2 = float(np.max(g + c2 * beta))

            best_delta = (a + b) * 0.5
            best_val = float(np.max(g + best_delta * beta))

            if best_val < current_max - 1e-11:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                current_max = best_val
                improved = True

        h = project_feasible(h, target)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now

        if not improved and sweep > 10:
            break

    h = project_feasible(h, target)
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -300.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.2]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 80:
                break

    return best_h, best_c5


def circular_shift(h, shift):
    return np.roll(h, shift)
