import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_full_lp(g, dx, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in range(-(n-1), n):
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo + k:hi + k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints

def alternating_lp_full(h_init, n, target, max_iter, time_limit,
                        damping_start, damping_end, eps=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 20 and stuck > 100:
            break
        
        g = 1.0 - h
        progress = min(1.0, it / max(1, max_iter * 0.5))
        damping = damping_start + (damping_end - damping_start) * progress
        
        h_var, t_var, constraints = build_full_lp(g, dx, n, target)
        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=eps, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):

    c_max = np.max(corr)

    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n)
    
    grad = np.zeros(n)
    n_half = n - 1
    for idx in range(2 * n - 1):

        if w < 1e-10:

        k = idx - n_half
        if k >= 0:

def slsqp_polish(h_init, n, target, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
              'fun': lambda x: np.sum(x) - target,
              'jac': lambda x: np.ones(n)})
    
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    alphas = [100, 500, 2000, 10000, 50000]

                           jac=lambda x, a: smooth_max_grad(x, a),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    
    return best_h

                        damping_start, damping_end, eps=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 20 and stuck > 150:
            break
        
        g = 1.0 - h
        progress = min(1.0, it / max(1, max_iter * 0.6))
        damping = damping_start + (damping_end - damping_start) * progress
        
        h_var, t_var, constraints = build_full_lp(g, dx, n, target)
        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=eps, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=200000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n)
    
    grad = np.zeros(n)
    n_half = n - 1
    for idx in range(2 * n - 1):
        w = exp_vals[idx] / total
        if w < 1e-10:
            continue
        k = idx - n_half
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:]) * dx
            grad[k:] -= w * h[:n-k] * dx
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk]) * dx
            grad[:n-kk] -= w * h[kk:] * dx
    
    return grad

def _cons_fun(x):
    return np.sum(x) - _global_target

def _cons_jac(x):
    return np.ones(_global_n)

def _obj_wrapper(x):
    return smooth_max_obj(x, _global_alpha)

def _grad_wrapper(x):
    return smooth_max_grad(x, _global_alpha)

def slsqp_polish(h_init, n, target, time_limit):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac})
    
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _global_alpha = alpha
        try:
            res = minimize(_obj_wrapper, h_cur, jac=_grad_wrapper,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    
    return best_h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_structured(h, n, target, rng, mode):
    h_p = h.copy()
    if mode == 0:
        # Block swap
        block_size = n // 8
        i = rng.randint(0, n - 2 * block_size)
        j = rng.randint(0, n - 2 * block_size)
        if abs(i - j) > block_size:
            temp = h_p[i:i+block_size].copy()
            h_p[i:i+block_size] = h_p[j:j+block_size]
