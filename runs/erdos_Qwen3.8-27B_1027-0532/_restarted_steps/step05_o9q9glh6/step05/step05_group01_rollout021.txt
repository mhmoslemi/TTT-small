import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective_and_grad(h, alpha):
    """Compute smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Log-sum-exp for numerical stability
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    logsum = np.log(np.sum(exp_shifted))
    obj = (corr_max + logsum / alpha) * dx
    
    # Weights
    w = exp_shifted / np.sum(exp_shifted)
    
    # Build v[k] = w[mid+k] + w[mid-k]
    v = np.zeros(n)
    for k in range(n):
        v[k] = w[mid + k]
        if mid - k >= 0:
            v[k] += w[mid - k]
    
    # Convolve to get W @ h
    v_sym = np.concatenate([v[:0:-1], v])
    conv = np.convolve(h, v_sym, mode='full')
    Wh = conv[mid:mid+n]
    
    grad = dx * (1.0 - Wh)
    return obj, grad

def lbfgsb_optimize(h_init, n, target, alphas, maxiter, time_limit):
    """L-BFGS-B optimization with geometric smoothing schedule."""
    start = time.time()
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj_fn(x):
            val, _ = smooth_objective_and_grad(x, alpha)
            return val
        
        def grad_fn(x):
            _, g = smooth_objective_and_grad(x, alpha)
            return g
        
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': maxiter * 3})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        except Exception:
            pass
    
    return h

    best_h = h.copy()

    idxs = np.arange(n)

                g[mj] += (1.0 - h[j + s[mj]])
                mj2 = (j - s >= 0) & (j - s < n)

                g[mi2] += h[i - s[mi2]]

                a, b = 0.0, delta_max

                hj = h[j]
                delta_max = min(hi, 1.0 - hj)
                if delta_max < 1e-10:
                    continue

                g = np.zeros(2 * n - 1)
                s = shifts_arr
                mj = (j + s >= 0) & (j + s < n)
                g[mj] += (1.0 - h[j + s[mj]])
                mj2 = (j - s >= 0) & (j - s < n)
                g[mj2] -= h[j - s[mj2]]
                mi = (i + s >= 0) & (i + s < n)
                g[mi] -= (1.0 - h[i + s[mi]])
                mi2 = (i - s >= 0) & (i - s < n)
                g[mi2] += h[i - s[mi2]]

                a_d, b_d = 0.0, delta_max
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-13
                    f_prime = np.max(g[active_m]) if active_m.any() else 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                d = 0.5 * (a_d + b_d)
                if d < 1e-12:
                    continue
                trial = np.max(corr + d * g) * dx
                if trial < best_trial - 1e-15:
                    best_trial = trial
                    best_move = (i, j, d)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
                if len(free) > 0:
                    h[free] += diff / len(free)
                    h = np.clip(h, 0.0, 1.0)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def perturbation_search(h_init, n, target, max_tries, time_limit, seed=77):
    start = time.time()
    rng = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 13, 21]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.02, 0.05, 0.1, 0.15, 0.2]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
        elif step > 500 and rng.rand() < 0.05:
            h_cur = best_h.copy()
    return best_h, best_c5


def asgd_smoothed(h_init, n, target, time_limit, max_steps=100000, base_lr=0.01, mu_mom=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        c_max = np.max(corr)
        top_k = np.argsort(corr)[-8:]
        w = np.exp(80.0 * (corr[top_k] - c_max))
        w /= w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        lr = base_lr / (1.0 + 0.01 * step)
        v = mu_mom * v + grad
        h_new = project_feasible(h - lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_now - 1e-15:
            h = h_new
        else:
            h = project_feasible(h - lr * 0.5 * v, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B fast convergence
    if tleft() > 250:
        alphas = np.geomspace(30, 20000, 18)
        h_new = lbfgsb_phase(best_h, n, target, alphas, 300, tleft() - 30, mu=1000.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: SLSQP polish
    if tleft() > 150:
        alphas = np.geomspace(5000, 100000, 5)
        h_new = slsqp_phase(best_h, n, target, alphas, 200, tleft() - 20)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Smart pair CD (main workhorse)
    if tleft() > 200:
        cd_time = min(350, tleft() * 0.4)
        h_cd, c5_cd = smart_pair_cd(best_h, n, target, time_limit=cd_time,
                                    n_cand=25, n_active_w=80.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: L-BFGS after pair CD
    if tleft() > 120:
        alphas = np.geomspace(10000, 300000, 10)
        h_new = lbfgsb_phase(best_h, n, target, alphas, 400, tleft() - 20, mu=2000.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Second round pair CD
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.4)
        h_cd, c5_cd = smart_pair_cd(best_h, n, target, time_limit=cd_time,
                                    n_cand=20, n_active_w=150.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Try higher resolution
    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            alphas = np.geomsp

ace(50, 10000, 12)
            h_hires = lbfgsb_phase(h_interp, n_try, target_try, alphas, 250,
                                   min(120, tleft() - 50), mu=1000.0)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try
                # Continue optimizing at this resolution
                if tleft() > 100:
                    h_hires2 = lbfgsb_phase(best_h, n, target,
                                            np.geomspace(5000, 200000, 8),
                                            300, min(80, tleft() - 40), mu=2000.0)
                    sc2 = compute_c5(h_hires2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_h = h_hires2

    # Phase 7: ASGD refinement
    if tleft() > 80:
        h_sg, c5_sg = asgd_smoothed(best_h, len(best_h), len(best_h) / 2.0,
                                    time_limit=min(50, tleft() * 0.4),
                                    max_steps=50000, base_lr=0.008, mu_mom=0.9)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Multi-restart perturbation
    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.12, 0.2]:
            if tleft() < 60:
                break
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            alphas = np.geomspace(200, 20000, 8)
            h_ref = lbfgsb_phase(h_pert, n_use, target_use, alphas, 200,
                                 min(40, tleft() - 20), mu=1000.0)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 9: Final pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = smart_pair_cd(best_h, n_use, target_use,
                                    time_limit=cd_time, n_cand=20, n_active_w=200.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_phase(best_h, n_use, target_use,
                              np.array([20000.0, 80000.0, 200000.0]),
                              300, min(25, tleft() - 5))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    best_score = compute_c5(best_h)

    return best_h, best_score, len(best_h)
