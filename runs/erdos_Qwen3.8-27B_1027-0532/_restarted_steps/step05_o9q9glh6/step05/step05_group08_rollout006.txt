def coordinate_descent(h, n, target, deadline, n_passes=5):
    """Coordinate descent exploiting the gradient structure."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    rng = np.random.RandomState(123)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        # Randomize order
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            # Try small perturbations of h[i]
            best_val = compute_c5(h_cur)
            # The gradient at i tells us the direction
            # ∂corr[m]/∂h[i] = 1 - 2*h[m-i] (for valid m-i)
            # For the max, we use the subgradient
            corr = compute_corr(h_cur)
            m_max = int(np.argmax(corr))
            k = m_max - (n - 1)
            if k >= 0:
                # h[i] appears at position i in lag k
                # ∂corr/∂h[i] = (1 - h[i+k]) - h[i-k] (if both valid)
                grad_i = 0.0
                if i + k < n:
                    grad_i += (1.0 - h_cur[i + k])
                if i - k >= 0:
                    grad_i -= h_cur[i - k]
            else:
                ell = -k
                grad_i = 0.0
                if i - ell >= 0:
                    grad_i += (1.0 - h_cur[i - ell])
                if i + ell < n:
                    grad_i -= h_cur[i + ell]
            
            # Move in negative gradient direction
            if grad_i > 1e-6:
                delta = min(0.01, 0.5 / (abs(grad_i) + 1))
            elif grad_i < -1e-6:
                delta = min(0.01, 0.5 / (abs(grad_i) + 1))
            else:
                continue
            
            h_try = h_cur.copy()
            h_try[i] += -np.sign(grad_i) * delta
            h_try[i] = np.clip(h_try[i], 0, 1)
            
            # Project to feasible
            h_try_proj = project_feasible(h_try, target)
            sc = compute_c5(h_try_proj)
            if sc < best_val:
                h_cur = h_try_proj
                best_score = min(best_score, sc)
    
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

    return lse_value_active(x, _g_alpha, _g_active)

def _grad_active_cb(x):
    return lse_grad_active(x, _g_alpha, _g_active)

def _obj_aug_cb(x):
    base = lse_value_full(x, _g_alpha)
    penalty = _g_mu * (np.sum(x) - _g_target) ** 2
    return base + penalty

def _grad_aug_cb(x):
    base_g = lse_grad_full(x, _g_alpha)
    pen_g = 2.0 * _g_mu * (np.sum(x) - _g_target) * np.ones_like(x)
    return base_g + pen_g

def _obj_aug_act_cb(x):
    base = lse_value_active(x, _g_alpha, _g_active)
    penalty = _g_mu * (np.sum(x) - _g_target) ** 2
    return base + penalty

def _grad_aug_act_cb(x):
    base_g = lse_grad_active(x, _g_alpha, _g_active)
    pen_g = 2.0 * _g_mu * (np.sum(x) - _g_target) * np.ones_like(x)
    return base_g + pen_g

def _cons_fun(x):
    return np.sum(x) - _g_n / 2.0

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_solve(h, n, alpha, active_idx, maxiter, ftol):
    global _g_alpha, _g_active, _g_n
    _g_alpha = alpha
    _g_active = active_idx
    _g_n = n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    if active_idx is not None:
        obj = _obj_active_cb
        jac = _grad_active_cb
    else:
        obj = _obj_full_cb
        jac = _grad_full_cb
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def lbfgsb_solve(h, n, alpha, active_idx, mu, maxiter):
    global _g_alpha, _g_active, _g_n, _g_mu, _g_target
    _g_alpha = alpha
    _g_active = active_idx
    _g_n = n
    _g_mu = mu
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    if active_idx is not None:
        obj = _obj_aug_act_cb
        jac = _grad_aug_act_cb
    else:
        obj = _obj_aug_cb
        jac = _grad_aug_cb
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return h

def subgradient_momentum(h, n, target, n_steps, step_init, step_decay, deadline, momentum, top_k):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    v = np.zeros(n)
    step = step_init
    for s in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        k_use = min(top_k, 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:k_use]
        c_top = corr[top_idx]
        c_max_top = c_top[0]
        diffs = c_top - c_max_top
        temp = max(dx * 0.005, 1e-10)
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        v = momentum * v - step * g
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            step = min(step * 1.01, 0.02)
        else:
            step *= step_decay
            if step < 1e-12:
                break
    return best_h, best_score

def active_set_loop(h, n, target, deadline, n_outer, top_m):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_m].astype(int)
        alphas = [500, 2000, 8000, 20000]
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            h_new = slsqp_solve(h_cur, n, alpha, top_idx, 1000, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def optimize_at_resolution(h_start, n, target, deadline):
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    alphas = [30, 100, 300, 1000, 3000, 10000, 30000, 80000]
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        for mu in [20.0, 100.0, 500.0]:
            if time.time() > deadline - 25:
                break
            h_new = lbfgsb_solve(best_h, n, alpha, None, mu, 200)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
    if time.time() < deadline - 20:
        h_new = slsqp_solve(best_h, n, 5000, None, 2000, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if time.time() < deadline - 15:
        h_act, sc_act = active_set_loop(best_h, n, target, deadline - 10, 10, 8)
        if sc_act < best_score:
            best_score = sc_act
            best_h = h_act
    if time.time() < deadline - 10:
        h_sg, sc_sg = subgradient_momentum(best_h, n, target, 1000, 0.004, 0.997,
                                           deadline - 5, 0.5, 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Fine L-BFGS-B continuation
    alphas_p1 = [15, 30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 16000, 32000, 64000]
    for alpha in alphas_p1:
        if time.time() > deadline - 500:
            break
        for mu in [10.0, 50.0, 200.0, 500.0]:
            if time.time() > deadline - 495:
                break
            h_new = lbfgsb_solve(best_h, n, alpha, None, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP at moderate alpha
    for alpha in [500, 2000, 8000, 20000]:
        if time.time() > deadline - 350:
            break
        h_new = slsqp_solve(best_h, n, alpha, None, 3000, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Active-set SLSQP
    if time.time() < deadline - 250:
        h_act, sc_act = active_set_loop(best_h, n, target, deadline - 200, 20, 10)
        if sc_act < best_score:
            best_score = sc_act
            best_h = h_act

    # Phase 4: Subgradient with various momenta
    if time.time() < deadline - 200:
        for mom in [0.0, 0.3, 0.6, 0.8]:
            if time.time() > deadline - 170:
                break
            h_sg, sc_sg = subgradient_momentum(best_h, n, target, 2000, 0.005, 0.997,
                                               deadline - 160, mom, 8)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 5: Multi-resolution n=600
    if time.time() < deadline - 250:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h2_opt, sc2_opt = optimize_at_resolution(h2, n2, target2, deadline - 120)
        if sc2_opt < best_score:
            best_score = sc2_opt
            best_h = h2_opt

    # Phase 6: Multi-resolution n=800
    if time.time() < deadline - 180:
        n3 = 800
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        h3_opt, sc3_opt = optimize_at_resolution(h3, n3, target3, deadline - 80)
        if sc3_opt < best_score:
            best_score = sc3_opt
            best_h = h3_opt

    # Phase 7: Random restarts
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(5):
            if time.time() > deadline - 50:
                break
            for scale in [0.03, 0.06, 0.1, 0.15]:
                if time.time() > deadline - 40:
                    break
                h_pert = best_h + rng.randn(n_cur) * scale
                h_pert = project_feasible(h_pert, target_cur)
                h_ref = lbfgsb_solve(h_pert, n_cur, 500, None, 200.0, 200)
                sc = compute_c5(h_ref)
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref
                    h_ref2 = slsqp_solve(best_h, n_cur, 10000, None, 1500, 1e-15)
                    sc2 = compute_c5(h_ref2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_h = h_ref2

    # Phase 8: Tug-of-war between top-2 lags
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for iteration in range(10):
            if time.time() > deadline - 30:
                break
            corr = compute_corr(best_h)
            top2 = np.argsort(corr)[::-1][:2].astype(int)
            for alpha in [2000, 10000]:
                if time.time() > deadline - 25:
                    break
                h_new = slsqp_solve(best_h, n_cur, alpha, top2, 1000, 1e-15)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new

    # Phase 9: Final subgradient polish
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_momentum(best_h, n_cur, target_cur, 1500, 0.003, 0.998,
                                           deadline - 10, 0.5, 12)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 10: Final SLSQP
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        for alpha in [20000, 50000, 100000]:
            if time.time() > deadline - 3:
                break
            h_new = slsqp_solve(best_h, n_cur, alpha, None, 1500, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
