def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection."""
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.clip(h - tau, 0.0, 1.0)
    # Fix numerical issues
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_obj_grad(h, alpha):
    """Compute smooth objective value and gradient."""
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    ov_max = np.max(ov)
    ov_shifted = ov - ov_max
    exp_vals = np.exp(alpha * ov_shifted)
    p = exp_vals / exp_vals.sum()
    val = ov_max + np.log(exp_vals.sum()) / alpha
    
    grad = np.zeros(n)
    n_half = n - 1
    for k_idx in range(2 * n - 1):
        k = k_idx - n_half
        w = p[k_idx]
        if w < 1e-14:

        lo = max(0, -k)

    return val, grad

def slsqp_smooth(h_start, n, target, alpha, maxiter=3000, ftol=1e-15):
    """SLSQP on smooth objective with exact gradient and constraint."""
    def obj(h):
        v, _ = smooth_obj_grad(h, alpha)
        return v
    
    def obj_grad(h):
        _, g = smooth_obj_grad(h, alpha)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
             'jac': lambda h: np.ones_like(h)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def all_lags_lp(h_current, n, target, reg):
    """LP with all lag constraints linearized at h_current."""
    dx = 2.0 / n
    g = 1.0 - h_current
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
        t_var >= 0.0
    ]
    n_half = n - 1
    for k_idx in range(2 * n - 1):
        k = k_idx - n_half
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo+k:hi+k] * dx
            constraints.append(cp.multiply(coeff, h_var).sum() <= t_var)
    
    objective = t_var + reg * cp.sum_squares(h_var - h_current)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def nesterov_subgrad(h_start, n, target, max_steps, time_limit, start_time, alpha_smooth=100):
    """Nesterov-accelerated subgradient with smooth approximation."""
    h = h_start.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        
        # Nesterov lookahead
        if step > 0:
            t = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * (step-1)**2))
            y = h + (t - 1.0) / t * (h - h_prev)
        else:
            y = h.copy()
        y = np.clip(y, 0.0, 1.0)
        
        # Compute gradient of smooth objective
        _, grad = smooth_obj_grad(y, alpha_smooth)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-16:

        # Polyak step size
        y_score = compute_c5(y)  # Actually use the smooth value

        lr = (y_val - best_score) / (grad_norm ** 2 + 1e-20)
        lr = min(max(lr, 0.001), 0.5)
        
        h_new = h - lr * grad

        sc = compute_c5(h_new)
        if sc < best_score - 1e-15:

            best_h = h_new.copy()
        h_prev = h.copy()
        h = h_new

One approach: parameterize h = base + D @ z where:
- base is the center of the polytope (all entries = 0.5)
- D is an (n x (n-1)) matrix whose columns form an orthonormal basis for the null space of 1^T
- z ∈ R^{n-1} is the free variable

- Box constraints: 0 ≤ base[i] + D[i,:] @ z ≤ 1 for all i

            coeff[lo:hi] = g[lo+k:hi+k] * dx
            constraints.append(cp.multiply(coeff, h_var).sum() <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h_current)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
