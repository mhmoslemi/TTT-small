import time

    N = 1 << (2 * n - 1).bit_length()

    r = np.fft.irfft(fa * np.conj(fb), N)

    dx = 2.0 / n

```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(a, b):
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * np.conj(fb), N)
    return np.concatenate((r[N-n+1:], r[:n]))

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha, top_k=20):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = fft_correlate(h, one_minus_h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    val = c_max + np.log(w_sum) / alpha
    w /= w_sum
    
    # Gradient: only use top_k significant lags
    top_idx = np.argsort(w)[::-1][:top_k]
    top_w = w[top_idx]
    
    g = np.zeros(n)
    for wk, m in zip(top_w, top_idx):
        if wk < 1e-15:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    g *= dx
    return val, g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr)) * dx

def lbfgsb_round(h, n, target, alpha, mu, maxiter, deadline):
    def obj(x):
        v, _ = lse_obj_and_grad(x, alpha)
        return v + mu * (np.sum(x) - target) ** 2
    def objg(x):
        v, g = lse_obj_and_grad(x, alpha)
        return g + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def slsqp_round(h, n, target, alpha, maxiter, ftol, deadline):
    def obj(x):
        v, _ = lse_obj_and_grad(x, alpha)
        return v
    def grad(x):
        _, g = lse_obj_and_grad(x, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = fft_correlate(h_cur, 1.0 - h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n-k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n-k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n-ell])
                g[:n-ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = fft_correlate(h_cur, 1.0 - h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n-k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n-k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n-ell])
            g[:n-ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.01)
        else:
            eps *= 0.9
            if eps < 1e-10:
                break
    return h_cur, best_score

def multi_lag_slsqp(h, n, target, m_active, alpha, maxiter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for outer in range(10):
        if time.time() > deadline:
            break
        corr = fft_correlate(h_cur, 1.0 - h_cur)
        top_m = np.argsort(corr)[::-1][:m_active]
        def obj_ml(x):
            c = fft_correlate(x, 1.0 - x)
            c_active = c[top_m]
            am = float(np.max(c_active))
            shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
            val = am + np.log(np.sum(np.exp(shifted))) / alpha
            # Add small penalty for non-active lags exceeding am
            c_other = np.delete(c, top_m)
            penalty = 1e-4 * float(np.max(c_other))
            return val + penalty
        def grad_ml(x):
            c = fft_correlate(x, 1.0 - x)
            c_active = c[top_m]
            am = float(np.max(c_active))
            shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
            w = np.exp(shifted)
            w /= w.sum()
            dx = 2.0 / n
            g = np.zeros(n)
            for wk, m in zip(w, top_m):
                m = int(m)
                k = m - (n - 1)
                if k >= 0:
                    g[:n-k] += wk * (1.0 - x[k:])
                    g[k:] -= wk * x[:n-k]
                else:
                    ell = -k
                    g[ell:] += wk * (1.0 - x[:n-ell])
                    g[:n-ell] -= wk * x[ell:]
            g *= dx
            return g
        bounds = [(0.0, 1.0)] * n
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_ml, h_cur, method='SLSQP', jac=grad_ml,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_candidate = project_feasible(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < best_score:
                best_score = sc
                h_cur = h_candidate
        except Exception:
            break
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
