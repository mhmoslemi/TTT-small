For k=0: value = sum_i h[i](1-h[i]) = n/2 - sum h[i]²
For k=1: value = sum_{i=0}^{n-2} h[i+1](1-h[i])

If h is binary (0 or 1), then h[i](1-h[i]) = 0 for all i, so the k=0 value is 0!
But for k=1: sum h[i+1](1-h[i]) = number of transitions from 0 to 1.

So for a binary h with n/2 ones and n/2 zeros, the k=0 value is 0, and we want to minimize the maximum over other k.

For a binary sequence, the value at lag k is:
# of positions where h[i+k]=1 and h[i]=0 (for k>0)
= # of 1s in the shifted positions that overlap with 0s in the original positions

This is the number of "1→0" transitions at distance k... no, it's the number of positions where the shifted version has 1 and the original has 0.

For a binary sequence, this is related to the "crossover" or "error" pattern. If we think of the sequence as a binary string, the value at lag k is the Hamming distance between the sequence and its k-shift, counted only in the "1 in shifted, 0 in original" direction.

Actually, for a binary sequence with exactly n/2 ones:
- At lag k: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) = number of i where h[i+k]=1 and h[i]=0

This is the number of "ascending" transitions at distance k.

If the sequence is "balanced" (like a de Bruijn sequence or a Hadamard row), the autocorrelation is as flat as possible.

So the problem for binary sequences is: find a binary string of length n with n/2 ones that minimizes the maximum "ascending" autocorrelation.

This is related to the theory of "low autocorrelation binary sequences" or "Barker sequences" (though Barker is about the full autocorrelation, not just one side).

For the continuous case (h ∈ [0,1]), we have more freedom, so the optimal value should be lower than the best binary solution.

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_run(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-16, 'maxls': 100, 'maxfun': 50000})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgrad_polish(h, target, n_iters, top_k, deadline, lr0=None):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    if lr0 is None:
        lr0 = 0.3
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        threshold = c_max - 0.3 / (it + 1)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for ki in active:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= len(active)
        lr = lr0 * dx / (np.sqrt(it + 1) * 5.0)
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_cur = h_new.copy()
        else:
            h_new2 = project_feasible(h_cur - lr * 0.3 * grad, target)
            sc2 = compute_c5(h_new2)
            if sc2 < best_score:
                best_score = sc2
                h_cur = h_new2.copy()
    return h_cur, best_score


def coord_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            a = 0.0
            b = 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-12:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_score - 1e-14:
                    best_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, best_score


def slsqp_polish(h, target, alpha, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h


def upsample(h_coarse, n_target):
    n_old = len(h_coarse)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_target)
    h_new = np.interp(x_new, x_old, h_coarse)
    return h_new


def multi_start_optimize(h_base, n, target, deadline, rng, mu=100.0):

    best_score = compute_c5(best_h)

    alphas = [10, 50, 200, 1000, 5000, 50000]

        if time.time() > deadline - 60:
