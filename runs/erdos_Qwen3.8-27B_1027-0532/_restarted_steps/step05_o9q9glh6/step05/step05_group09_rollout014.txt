import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-boundary elements
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
        else:
            result += diff / len(result)
        result = np.clip(result, 0.0, 1.0)
    return result

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return grad

def fista_v2(h0, target, alpha, deadline, max_iters, step=None):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if step is None:
        step = 2.0 / n  # Conservative default
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        grad_y = lse_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def lse_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
