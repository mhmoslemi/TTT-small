```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def lag_gradient(h, k, n, dx):
    """Compute gradient of overlap at lag k with respect to h."""
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1 - 2*h)
    elif k > 0:
        # f_k = dx * sum_{j=0}^{n-k-1} h[j]*(1-h[j+k])
        # First factor: for m in [0, n-k): dx*(1-h[m+k])
        grad[:n-k] += dx * (1 - h[k:])
        # Second factor: for m in [k, n): -dx*h[m-k]
        grad[k:] -= dx * h[:n-k]
    else:
        k_abs = -k
        # f_k = dx * sum_{j=k_abs}^{n-1} h[j]*(1-h[j-k_abs])
        # First factor: for m in [k_abs, n): dx*(1-h[m-k_abs])
        grad[k_abs:] += dx * (1 - h[:n-k_abs])
        # Second factor: for m in [0, n-k_abs): -dx*h[m+k_abs]
        grad[:n-k_abs] -= dx * h[k_abs:]
    return grad

def soft_max_gradient(h, alpha, n_top=50):
    """Compute gradient of alpha-smoothed max of overlaps."""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h)
    
    # Use top lags for efficiency
    top_idx = np.argsort(overlaps)[::-1][:n_top]
    top_vals = overlaps[top_idx]
    
    # Softmax weights
    shifted = alpha * (top_vals - top_vals[0])
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        k = idx - (n - 1)
        grad += weights[i] * lag_gradient(h, k, n, dx)
    
    # The smooth max value
    smooth_val = top_vals[0] + np.log(weights.sum() * (top_vals[0] - top_vals[0])) / alpha
    # Actually: smooth_max = max + log(sum(exp(alpha*(f_k - max))))/alpha
    smooth_val = top_vals[0] + np.log(np.sum(np.exp(alpha * (top_vals - top_vals[0])))) / alpha
    
    return smooth_val, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_method(h_init, n, target, max_steps, time_limit, 
                       n_top=30, momentum=0.9):
    """Direct subgradient method on the exact max of overlaps."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        overlaps = compute_all_overlaps(y)
        n_half = n - 1
        max_val = np.max(overlaps)
        
        # Active set: lags within 5% of max
        threshold = max_val * 0.95
        active_mask = overlaps >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(overlaps)])
        
        # Cap active set size
        if len(active_idx) > n_top:
            top_weights = overlaps[active_idx]
            order = np.argsort(top_weights)[::-1][:n_top]
            active_idx = active_idx[order]
        
        # Compute subgradient (average of gradients at active lags)
        grad = np.zeros(n)
        for idx in active_idx:
            k = idx - n_half
            grad += lag_gradient(y, k, n, dx)
        grad /= len(active_idx)
        
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        
        # Polyak-like step size
        gap = max_val - best_c5
        if gap <= 0:
            break
        lr = min(1.0, gap / (norm_grad ** 2 + 1e-10))
        lr = max(lr, 0.01)
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            # Backtracking
            for factor in [0.5, 0.25, 0.1]:
                h_try2 = h - lr * factor * grad
                h_try2 = project_feasible(h_try2, target)
                if compute_c5(h_try2) < best_c5 - 1e-15:
                    best_c5 = compute_c5(h_try2)
                    best_h = h_try2.copy()
                    h_prev = h.copy()
                    h = h_try2
                    break
            else:
                h_prev = h.copy()
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def lag_gradient(h, k, n, dx):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        grad[:n - k] += dx * (1.0 - h[k:])
        grad[k:] -= dx * h[:n - k]
    else:
        ka = -k
        grad[ka:] += dx * (1.0 - h[:n - ka])
        grad[:n - ka] -= dx * h[ka:]
    return grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def alternating_lp(h_init, n, target, max_iter, time_limit, n_top, solver='SCS'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 200:
            break
        
        g = 1.0 - h
        overlaps = compute_all_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        reg = max(1e-10, 1e-4 * (0.9 ** (it / 10.0)))
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            if solver == 'SCS':
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
            elif solver == 'ECOS':
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            else:
                prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        # Damping
        progress = min(1.0, it / max(1, max_iter * 0.5))
        damping = 0.95 - 0.15 * progress  # 0.95 to 0.80
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5


def true_subgradient(h_init, n, target, max_steps, time_limit, n_top=50, momentum=0.8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)
        
        overlaps = compute_all_overlaps(y)
        n_half = n - 1
        max_val = np.max(overlaps)
        
        # Active set
        threshold = max_val * 0.98
        active_mask = overlaps >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) > n_top:
            order = np.argsort(overlaps[active_idx])[::-1][:n_top]
            active_idx = active_idx[order]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(overlaps)])
        
        grad = np.zeros(n)
        for idx in active_idx:
            k = idx - n_half
            grad += lag_gradient(y, k, n, dx)

        if norm_grad < 1e-15:

        if gap <= 0:
            break

        lr = max(lr, 0.05)

        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
