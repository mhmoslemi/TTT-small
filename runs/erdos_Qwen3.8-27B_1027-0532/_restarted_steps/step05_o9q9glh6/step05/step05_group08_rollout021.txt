```python
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n - k]
        else:
            ell = -k
            g[ell:] += wm * (1.0 - h[:n - ell])
            g[:n - ell] -= wm * h[ell:]
    g *= dx
    return g
```
