import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def build_basis(n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(k * np.pi * x)
        B[:, 2*(k-1) + 1] = np.sin(k * np.pi * x)
    return B


def lse_h(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad_h(h, alpha, dx):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def sigmoid_fourier_obj(v, K, n, alpha, mu_int):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    g = np.full(n, v[0])
    for k in range(1, K + 1):
        g += v[2*(k-1)+1] * np.cos(k * np.pi * x)
        g += v[2*(k-1)+2] * np.sin(k * np.pi * x)
    h = 1.0 / (1.0 + np.exp(-np.clip(g, -50.0, 50.0)))
    dx = 2.0 / n
    val = lse_h(h, alpha) * dx
    int_viol = np.sum(h) * dx - 1.0
    val += mu_int * int_viol * int_viol
    return val


def sigmoid_fourier_grad(v, K, n, alpha, mu_int):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    g = np.full(n, v[0])
    for k in range(1, K + 1):
        g += v[2*(k-1)+1] * np.cos(k * np.pi * x)
        g += v[2*(k-1)+2] * np.sin(k * np.pi * x)
    h = 1.0 / (1.0 + np.exp(-np.clip(g, -50.0, 50.0)))
    dx = 2.0 / n
    g_h = lse_grad_h(h, alpha, dx)
    int_viol = np.sum(h) * dx - 1.0
    g_h += 2.0 * mu_int * int_viol * dx
    sigma_factor = h * (1.0 - h)
    grad = np.zeros(2 * K + 1)
    grad[0] = np.sum(g_h * sigma_factor)
    for k in range(1, K + 1):
        grad[2*(k-1)+1] = np.sum(g_h * sigma_factor * np.cos(k * np.pi * x))
        grad[2*(k-1)+2] = np.sum(g_h * sigma_factor * np.sin(k * np.pi * x))
    return grad


def sigmoid_to_h(v, K, n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    g = np.full(n, v[0])
    for k in range(1, K + 1):

        g += v[2*(k-1)+2] * np.sin(k * np.pi * x)
    h = 1.0 / (1.0 + np.exp(-np.clip(g, -50.0, 50.0)))
    return h


def h_to_fourier_coeff(h, K, n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    coeffs = np.zeros(2 * K)
    dx = 2.0 / n

        coeffs[2*(k-1) + 1] = 2.0 * np.sum((h - 0.5) * np.sin(k * np.pi * x)) * dx
    return coeffs


def lbfgsb_direct(h, n, target, alpha, mu, maxiter):
    dx = 2.0 / n
    def obj(x):
        return lse_h(x, alpha) * dx + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad_h(x, alpha, dx) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_direct(h, n, target, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        return lse_h(x, alpha) * dx
    def objg(x):
        return lse_grad_h(x, alpha, dx)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h

        h_sg, sc_sg = subgradient_polish(best_h, n_cur, target_cur, 1000, 0.001, 0.002, 0.998, deadline - 15)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if time.time() < deadline - 10:
            h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 300, 0.001, deadline - 8)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg
        if time.time() < deadline - 5:
            h_new = slsqp_direct(best_h, n_cur, target_cur, 50000, 1000)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
