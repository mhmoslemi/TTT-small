import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def fista_lse(h0, target, alpha, deadline, max_iters=200000, step_init=None):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0.astype(float), target)
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    
    if step_init is None:
        step_init = 2.0 / (alpha * 0.05 + 1.0)
    step = step_init
    max_step = 5.0 / (alpha * 0.01 + 1.0)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # LSE objective and gradient via FFT
        corr = np.correlate(y, 1.0 - y, mode='full')[n-1:]
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        
        w_flipped = weights[::-1].copy()
        v = 1.0 - y
        fsize = 2 * n
        Wf = np.fft.rfft(w_flipped, fsize)
        V = np.fft.rfft(v, fsize)
        W = np.fft.rfft(weights, fsize)
        H = np.fft.rfft(y, fsize)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad = dx * (corr_pos - corr_neg)
        
        # Backtracking
        obj_y = (c_max + np.log(sum_exp) / alpha) * dx
        for _ in range(40):
            x_new = project_feasible(y - step * grad, target)
            corr_new = np.correlate(x_new, 1.0 - x_new, mode='full')[n-1:]
            cm = float(np.max(corr_new))
            sh = np.clip(alpha * (corr_new - cm), -700.0, 0.0)
            se = float(np.exp(sh).sum())
            obj_new = (cm + np.log(se) / alpha) * dx
            quad = 0.5 / step * np.dot(x_new - y, x_new - y)
            if obj_new <= obj_y + quad + 1e-16:
                break
            step *= 0.5
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y - step * grad, target)
        
        step = min(step * 1.5, max_step)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    
    return best_x, best_score

def subgradient_polish(h0, target, deadline, top_k=5, max_rounds=100000):
    """Subgradient descent on the max with top-k active lags."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0.astype(float), target)
    best_x = x.copy()
    best_score = compute_c5(x)
    step = 0.01
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        
        corr = get_corr_array(x)
        current_max = float(np.max(corr))
        
        # Top-k lags
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        
        # Compute weighted gradient (equal weights for simplicity, or exponential)
        diffs = corr[sorted_idx] - current_max
        w = np.exp(diffs / 0.01)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, sorted_idx):
            ki = int(ki)
            if ki == 0:
                grad[:] += wi * (1.0 - 2.0 * x[:])
            else:
                grad[ki:] += wi * (1.0 - x[:n-ki])
                grad[:n-ki] -= wi * x[ki:]
        grad *= dx
        
        # Try decreasing steps
        for eps in [step, step*0.5, step*0.25, step*0.1]:
            x_new = project_feasible(x - eps * grad, target)
            sc = compute_c5(x_new)
            if sc < best_score - 1e-15:
                x = x_new
                best_score = sc
                best_x = x.copy()
                step = min(step * 1.1, 0.1)
                break
        else:
            step *= 0.7
            if step < 1e-5:
                break
    
    return best_x, best_score

def binary_pair_swaps(h0, target, deadline, max_iters=500000):
    """Local search via binary pair swaps."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0.astype(float), target)
    best_x = x.copy()
    best_score = compute_c5(x)

    ones = np.where(x > 0.99)[0]
    zeros = np.where(x < 0.01)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return best_x, best_score
    
    rng = np.random.RandomState(123)
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Random swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        x[a] = x[a] - 1.0 + 0.0  # 1 -> 0
        x[b] = x[b] + 1.0 - 0.0  # 0 -> 1
        # Actually just set them
        x[a] = 0.0
        x[b] = 1.0
        x = project_feasible(x, target)
        sc = compute_c5(x)
        
        if sc < best_score - 1e-15:
            best_score = sc
            best_x = x.copy()
            no_improve = 0
            # Update ones and zeros
            ones = np.where(x > 0.99)[0]
            zeros = np.where(x < 0.01)[0]
        else:
            # Undo
            x[a] = 1.0
            x[b] = 0.0
            x = project_feasible(x, target)
            no_improve += 1
            if no_improve > 10000:
                break
    
    return best_x, best_score

def coord_optimize(h0, target, deadline, max_sweeps=50):
    """For each non-boundary coordinate, find optimal value via bisection."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0.astype(float), target)
    best_score = compute_c5(x)
    
    # Identify free coordinates
    free = np.where((x > 0.01) & (x < 0.99))[0]
    
    if len(free) == 0:
        return x, best_score
    
    rng = np.random.RandomState(99)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(len(free))
        for idx in order:
            j = free[idx]
            old_val = x[j]
            
            # Try to find better value for x[j]
            # The sum constraint means changing x[j] by δ changes the sum by δ
            # We need to redistribute: project handles this
            
            # Try several candidate values
            best_val = old_val
            best_sc = best_score
            
            # Binary search style: try values
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                candidate = old_val * (1 - frac) if old_val > 0.5 else old_val * (1 + frac)
                # Also try moving towards 0 or 1
                candidate = np.clip(candidate, 0.0, 1.0)
                if abs(candidate - old_val) < 1e-6:
                    continue
                x[j] = candidate
                x = project_feasible(x, target)
                sc = compute_c5(x)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_val = candidate
                x[j] = old_val
                x = project_feasible(x, target)
            
            if best_val != old_val:
                x[j] = best_val
                x = project_feasible(x, target)
                best_score = compute_c5(x)
                improved = True
                # Update free set
                free = np.where((x > 0.01) & (x < 0.99))[0]
        
        if not improved:
            break
    
    return x, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)
