import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_slsqp(h):
    n = _global_n
    alpha = _global_alpha
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(h):
    n = _global_n
    alpha = _global_alpha
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of corr[k] with respect to h
    # corr[k] = sum_i h[i] * (1-h[i+k]) for appropriate bounds
    # d(corr[k])/d(h[j]) = (1-h[j+k]) - h[j-k] (with bounds)
    
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s_idx in np.where(weights > 1e-10 / n)[0]:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[s_idx] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[s_idx] * h[idxs[v2] + k]
    return grad * dx_factor(n)


def dx_factor(n):
    return 1.0  # The objective doesn't include dx factor


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        _global_alpha = float(alpha)
        try:
            res = minimize(smooth_obj_slsqp, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def block_cd(h, n, target, time_limit, max_sweeps=2000, block_size=3, n_cand=30):
    """Block coordinate descent: pick block_size coordinates and optimize them jointly."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        # Select top candidates by |gradient|
        sorted_idx = np.argsort(np.abs(grad))
        high_set = sorted_idx[-n_cand:]
        low_set = sorted_idx[:n_cand]
        
        # Try all pairs still (but from the candidate sets)
        best_trial_c5 = c5_val
        best_move = None

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                # Compute the effect on correlations
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                # Binary search for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def topk_equalize(h, n, target, time_limit, k_values=(3, 5, 10, 20)):

    start = time.time()

    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    _global_n = n
    _global_target = target

            break

                corr = np.correlate(h_in, 1.0 - h_in, mode='full')
                top_k = np.sort(corr)[-k_val:]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturbation(h, n, target, rng, mode='sinusoidal'):
    """Create structured perturbations of h."""
    if mode == 'sinusoidal':
        freq = rng.choice([1, 2, 3, 5, 7, 11, 13])
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.15)
        x = np.arange(n) / n * 2 * np.pi
        pert = amp * np.sin(freq * x + phase)
    elif mode == 'localization':
        center = rng.integers(0, n)
        width = rng.choice([3, 5, 8, 12])
        amp = rng.uniform(0.05, 0.2) * (1 if rng.uniform() > 0.5 else -1)
        x = np.arange(n)
        pert = amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    elif mode == 'random_block':
        n_blocks = rng.choice([2, 3, 5])
        pert = np.zeros(n)
        for _ in range(n_blocks):
            start = rng.integers(0, n - 5)
            length = rng.integers(3, 15)
            length = min(length, n - start)
            amp = rng.uniform(-0.15, 0.15)
            pert[start:start+length] += amp
    else:
        pert = rng.randn(n) * 0.05
    
    h_new = h + pert
    return project_feasible(h_new, target)

                                        time_limit=min(60, tleft() * 0.3),
                                        max_steps=30000, top_k=5, base_lr=0.015)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
            if tleft() > 80:
                h_pol2 = slsqp_round(best_h, n, target,
                                     [1000, 5000, 20000],
                                     800, 1e-14)
                sc2 = compute_c5(h_pol2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_pol2

    if tleft() > 200:
        cd_time2 = min(180, tleft() * 0.3)
        h_cd2, c5_cd2 = pair_cd(best_h, n, target,
                                time_limit=cd_time2,
                                max_sweeps=5000, n_cand=100)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    for mode in ['sinusoidal', 'gaussian_bump', 'mixed_frequencies', 'block']:
        if tleft() < 100:
            break
        for trial in range(2):
            if tleft() < 60:
                break
            h_pert = structured_perturbation(best_h, n, target, rng, mode)
            h_ref = slsqp_round(h_pert, n, target,
                                [200, 1000, 5000, 20000],
                                800, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            if tleft() > 60:
                h_cd3, c5_cd3 = pair_cd(h_ref, n, target,
                                        time_limit=min(35, tleft() * 0.25),
                                        max_sweeps=1500, n_cand=50)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3

    if tleft() > 100:
        h_bin, c5_bin = binary_local_search(best_h, n, target,
                                            time_limit=min(60, tleft() * 0.4))
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_bin
        if tleft() > 60:
            h_pol3 = slsqp_round(best_h, n, target,
                                 [1000, 5000, 20000, 100000],
                                 1000, 1e-14)
            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 140:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [100, 500, 2000, 10000, 50000],
                                  1000, 1e-14)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
            if tleft() > 100:
                h_cd4, c5_cd4 = pair_cd(h_hires, n_try, target_try,
                                         time_limit=min(50, tleft() * 0.3),
                                         max_sweeps=2000, n_cand=50)
                if c5_cd4 < best_score:
                    best_score = c5_cd4
                    best_h = h_cd4
                if tleft() > 60:
                    h_pol4 = slsqp_round(best_h, n_try, target_try,
                                         [5000, 20000, 100000],
                                         800, 1e-14)
                    sc4 = compute_c5(h_pol4)
                    if sc4 < best_score:
                        best_score = sc4
                        best_h = h_pol4

    for round_num in range(3):
        if tleft() < 80:
            break
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_pol = slsqp_round(best_h, n_use, target_use,
                            [5000, 20000, 100000, 300000],
                            1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
        if tleft() > 50:
            h_cd5, c5_cd5 = pair_cd(best_h, n_use, target_use,
                                     time_limit=min(40, tleft() * 0.4),
                                     max_sweeps=1500, n_cand=60)
            if c5_cd5 < best_score:
                best_score = c5_cd5
                best_h = h_cd5

    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [50000, 200000, 500000],
                              600, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = float(n_use) / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
