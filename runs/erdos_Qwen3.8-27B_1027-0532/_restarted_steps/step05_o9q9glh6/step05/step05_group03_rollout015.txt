    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h0, alphas, maxiter=3000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h0)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h0.copy()
    best = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(obj_smooth, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < best:
                h_cur = h_new
                best = sc
        except Exception:
            pass
    return h_cur, best

def grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    else:
        upper = n - 1 - k
        g[0:upper+1] -= h[k:n] * dx
        g[k:n] += (1.0 - h[0:n-k]) * dx
    return g

def lag_surgery(h0, n_steps, eps_init, n_active, eps_max=0.008):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_sc = compute_c5(h)
    eps = eps_init
    stagnation = 0
    for step in range(n_steps):
        corr = compute_corr(h)
        top = np.argsort(corr)[::-1][:n_active]
        grad = np.zeros(n)
        for k in top:
            grad += grad_lag(h, int(k))
        grad /= n_active
        h_new = h - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_sc - 1e-14:
            h = h_new
            best_h = h.copy()
            best_sc = sc_new
            eps = min(eps * 1.04, eps_max)
            stagnation = 0
        else:
            eps *= 0.75
            stagnation += 1
            if stagnation > 30 or eps < 1e-11:
                break
    return best_h, best_sc

def soft_max_subgrad(h0, n_steps, lr_init, lr_decay, momentum, temp, n_top=5):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_sc = compute_c5(h)
    lr = lr_init
    vel = np.zeros(n)
    for step in range(n_steps):
        corr = compute_corr(h)
        c_max = np.max(corr)
        # Soft weights over top lags
        top = np.argsort(corr)[::-1][:n_top]
        diffs = corr[top] - c_max
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top):
            grad += w * grad_lag(h, int(k))
        vel = momentum * vel + (1 - momentum) * grad
        h_new = h - lr * vel
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_sc - 1e-14:
            h = h_new
            best_h = h.copy()
            best_sc = sc_new
            lr = min(lr * 1.002, 0.005)
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_sc

def try_bangbang(h0, target):
    """Try rounding to bang-bang and re-optimizing."""
    n = len(h0)
    best = h0.copy()
    best_sc = compute_c5(h0)
    for thresh in [0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65]:
        h_r = h0.copy()
        h_r[h_r < thresh] = 0.0
        h_r[h_r > 1.0 - thresh] = 1.0
        h_r = project_feasible(h_r, target)
        sc = compute_c5(h
_r)
        if sc < best_sc:
            best_sc = sc
            best = h_r.copy()
    return best, best_sc
