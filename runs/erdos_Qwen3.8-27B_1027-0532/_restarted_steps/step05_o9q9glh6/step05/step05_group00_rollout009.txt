import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP constraints
_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def smooth_obj(h, alpha, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Focus on the top M correlations
    if top_m < len(corr):
        idx = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[idx]
        c_max = float(top_vals.max())
        shifted = alpha * (top_vals - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    else:
        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + float(np.sum(np.exp(shifted)) * 0 + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None, top_m=50):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, top_m),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

To reduce c(k*), we want this to be negative:
term1(j,k*) - term2(j,k*) < term1(i,k*) - term2(i,k*)

So we want i where (term1(i,k*) - term2(i,k*)) is large (positive), and j where (term1(j,k*) - term2(j,k*)) is small (negative).

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def smooth_obj(h, alpha, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    m = min(top_m, n_corr)
    idx = np.argpartition(corr, -m)[-m:]
    top_vals = corr[idx]
    c_max = float(top_vals.max())
    shifted = np.clip(alpha * (top_vals - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None, top_m=100):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, top_m),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def greedy_binary(h, max_iters, time_limit=None, start_time=None, scan_width=25):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 500

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Candidates that directly affect the worst shift
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            # Try random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = compute_corr(h)
                sc = float(corr.max()) * (2.0 / n)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h.copy()
                    stall = 0
                else:
                    h[i] = 1
                    h[j] = 0
                    corr = compute_corr(h)
            continue

        # Consider top shifts for evaluation
        top_threshold = max_val * 0.8
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])

        best_improve = 0.0
        best_i = -1
        best_j = -1

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = compute_corr(h)
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 8000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        h_new = h.copy()
        moved = False
        r = rng.random()

        if r < 0.6:
            # Targeted: focus on worst shifts
            top_k = 5
            worst = np.argsort(corr)[-top_k:]
            k = worst[rng.randint(top_k)] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
        elif r < 0.9:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            # Double swap
            if len(ones) >= 2 and len(zeros) >= 2:
                idx1 = rng.choice(len(ones), 2, replace=False)
                idx2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[idx1[0]]] = 0
                h_new[ones[idx1[1]]] = 0
                h_new[zeros[idx2[0]]] = 1
                h_new[zeros[idx2[1]]] = 1
                moved = True

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)

            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                if h[i] >= 0.99:

                                              replace=False):
