Actually, let me try a simpler approach: **iterative linearization**.

At each iteration:
1. Compute corr = correlate(h, 1-h)
2. Find the top M shifts S
3. Linearize: for each k in S, approximate corr_k(h_new) ≈ corr_k(h) + (h_new - h) · grad_k
   where grad_k = d(corr_k)/dh (a vector of length n)
4. Solve: min t s.t. corr_k(h) + (h_new-h)·grad_k <= t for k in S, 0<=h_new<=1, sum(h_new)=n/2
   This is a linear program!
5. Update h = h_new (or a fraction of the way)
6. Repeat

Step 4 can be solved with scipy's linprog or cvxpy. With M=10 constraints and n=400 variables (plus the sum constraint and bounds), it's a small LP.

This is essentially a cutting-plane method, but with linearization of the bilinear terms. It should converge to a local optimum.

The advantage is that each iteration is fast (solving a small LP), and we can do many iterations.

def linearized_iteration(h, n, target, M=10, step_size=1.0):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-M:][::-1]
    
    # Compute gradients for top M shifts
    # d(corr[k])/dh[i] = (1-h[i+k-(n-1)]) * 1[valid] - h[i-k+(n-1)] * 1[valid]
    
    A_rows = []
    b_vals = []
    for k in top_idx:
        grad = np.zeros(n)
        # Part 1: i such that 0 <= i+k-(n-1) <= n-1
        # n-1-k <= i <= 2n-2-k, combined with 0<=i<=n-1
        i_lo = max(0, n-1-k)
        i_hi = min(n-1, 2*n-2-k)
        if i_lo <= i_hi:
            js = np.arange(i_lo, i_hi+1) + k - (n-1)
            grad[i_lo:i_hi+1] += (1.0 - h[js])
        # Part 2: i such that 0 <= i-k+(n-1) <= n-1
        # k <= i <= k+n-1, combined with 0<=i<=n-1
        i_lo2 = max(0, k)
        i_hi2 = min(n-1, k+n-1)
        if i_lo2 <= i_hi2:
            js2 = np.arange(i_lo2, i_hi2+1) - k + (n-1)
            grad[i_lo2:i_hi2+1] -= h[js2]
        A_rows.append(grad)
        b_vals.append(corr[k])
    
    A_mat = np.array(A_rows)  # M x n
    # Solve: min t s.t. A_mat @ h_new <= t + ... 
    
    # Actually, we want: min t
    # s.t. corr_k(h) + (h_new - h) · grad_k <= t
    #      0 <= h_new <= 1
    #      sum(h_new) = n/2
    #
    # => t >= corr_k(h) + (h_new - h) · grad_k for each k
    # => t - (h_new) · grad_k >= corr_k(h) - h · grad_k
    #
    # Variables: h_new (n vars) + t (1 var) = n+1 vars
    # Constraints: M inequality + 1 equality + 2n bounds
    
    # Using cvxpy would be cleaner, but let's use scipy linprog
    # min c · x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = [h_new(0), ..., h_new(n-1), t]
    # c = [0, ..., 0, 1]  (minimize t)
    # A_ub: for each k: -grad_k · h_new + t >= corr_k(h) - grad_k · h
    #   => grad_k · h_new - t <= corr_k(h) - grad_k · h
    #   => [grad_k, -1] · x <= corr_k(h) - grad_k · h
    # A_eq: [1, 1, ..., 1, 0] · x = n/2
    # bounds: 0 <= h_new[i] <= 1, t >= 0 (or t >= min possible)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    for idx, k in enumerate(top_idx):
        A_ub[idx, :n] = A_rows[idx]
        A_ub[idx, n] = -1.0
        b_ub[idx] = b_vals[idx] - A_rows[idx] @ h
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds_lin = [(0, 1)] * n + [(None, None)]  # t unbounded above
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds_lin, method='highs')
    
    if res.status == 0:
        h_new = res.x[:n]
        t_new = res.x[n]
        # Take a step (for stability)
        h_step = h + step_size * (h_new - h)
        h_step = project_feasible(h_step, target)
        return h_step
    return h

def compute_B(h, i, n):

    B = np.zeros(nl1)
    k_lo1 = max(0, n - 1 - i)
    k_hi1 = min(2 * n - 2, 2 * n - 2 - i)
    if k_lo1 <= k_hi1:
        ks = np.arange(k_lo1, k_hi1 + 1)
        js = ks + i - (n - 1)
        B[k_lo1:k_hi1 + 1] += (1.0 - h[js])

    k_hi2 = min(2 * n - 2, i + n - 1)

        ks2 = np.arange(k_lo2, k_hi2 + 1)

    return B

def optimize_pair(h, i, j, n, corr, target):
    """Optimize the split of h[i]+h[j] = c. Returns (t_star, C_vector, improvement)."""
    c = h[i] + h[j]
    if c < 1e-14 or c > 2.0 - 1e-14:
        return None, None, 0.0
    
    t_min = max(0.0, c - 1.0)
    t_max = min(1.0, c)
    if t_max - t_min < 1e-13:
        return None, None, 0.0
    
    Bi = compute_B(h, i, n)
    Bj = compute_B(h, j, n)
    C = Bi - Bj
    # corr_new[k] = corr[k] + (t - t_old) * C[k] where t_old = h[i]
    # We want to minimize max_k corr_new[k]
    # This is equivalent to finding the smallest v such that
    # corr[k] + (t - h[i]) * C[k] <= v for all k, with t in [t_min, t_max]
    
    t_old = h[i]
    # For given v, feasible t satisfies:
    # C[k] > 0: t <= (v - corr[k]) / C[k] + t_old
    # C[k] < 0: t >= (v - corr[k]) / C[k] + t_old
    # t in [t_min, t_max]
    
    pos = C > 1e-14
    neg = C < -1e-14
    zero = ~(pos | neg)
    
    # Bisection on v
    # Lower bound: max(corr) (can't do better than current max, actually can)
    # Upper bound: max(corr) + max(|C|) * (t_max - t_old)
    
    # Actually, the optimal v is at least min_k corr[k] (trivially) and at most max_k corr[k]
    # (since t=t_old gives v=max(corr))

            lo = mid
            continue

        lb = t_min

            lb = max(lb, float(np.max((mid - corr[neg]) / C[neg]))) + t_old
        if lb <= ub + 1e-11:
            hi = mid
        else:
            lo = mid
    
    v = hi
    # Find t_star
    pos = C > 1e-14
    neg = C < -1e-14
    ub = t_max
    if np.any(pos):
        ub = min(ub, float(np.min((v - corr[pos]) / C[pos]))) + t_old
    lb = t_min
    if np.any(neg):

    lb = max(lb, t_min)
    if lb > ub:
        t_star = (t_min + t_max) * 0.5
    else:
        t_star = lb + (ub - lb) * 0.5

def linearized_subgrad(h, n, target, deadline, M=10):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[-M:][::-1]
        
        # Build LP
        c_lp = np.zeros(n + 1)
        c_lp[-1] = 1.0
        
        A_ub = np.zeros((M, n + 1))
        b_ub = np.zeros(M)
        
        for idx, k in enumerate(top_idx):
            d = k - (n - 1)
            grad = np.zeros(n)
            # Part 1: i such that 0 <= i+d <= n-1, i.e., max(0,-d) <= i <= min(n-1, n-1-d)
            i_lo = max(0, -d)
            i_hi = min(n-1, n-1-d)
            if i_lo <= i_hi:
                js = np.arange(i_lo, i_hi+1) + d
                grad[i_lo:i_hi+1] += (1.0 - h_cur[js])
            # Part 2: i such that 0 <= i-d <= n-1 (i appears as h[j] where j=i-d)
            # i.e., d <= i <= d+n-1, combined with 0<=i<=n-1
            i_lo2 = max(0, d)
            i_hi2 = min(n-1, d+n-1)
            if i_lo2 <= i_hi2:
                js2 = np.arange(i_lo2, i_hi2+1) - d
                grad[i_lo2:i_hi2+1] -= h_cur[js2]
            
            A_ub[idx, :n] = grad
            A_ub[idx, n] = -1.0
            b_ub[idx] = corr[k] - grad @ h_cur
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lin = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lin, method='highs')
        
        if res.status == 0:
            h_new = res.x[:n]
            # Damped step
            step = 0.5
            h_next = h_cur + step * (h_new - h_cur)
            h_next = project_feasible(h_next, target)

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:

        cnt = int(mask.sum())

        h = np.clip(h, 0.0, 1.0)

        if i_lo2 <= i_hi2:
            js2 = np.arange(i_lo2, i_hi2 + 1) - d
            grad[i_lo2:i_hi2 + 1] -= h[js2]
        A_ub[idx, :n] = grad
        A_ub[idx, n] = -1.0
        b_ub[idx] = corr[k] - grad @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds_lin = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lin, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None


def linearized_descent(h, n, target, deadline, M=8, max_iters=5000):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = linprog_direction(h_cur, n, target, M)
        if h_new is None:
            break
        for step in [1.0, 0.5, 0.25, 0.1]:
            h_try = h_cur + step * (h_new - h_cur)
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                h_cur = h_try
                break
    return h_cur, best_score
