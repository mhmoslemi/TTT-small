from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_gradient(h, w):
    """Compute gradient of smoothed max.
    w: weight vector of length 2n-1 (softmax of alpha*corr)
    Returns: gradient vector of length n
    """
    n = len(h)
    i_idx = np.arange(n)
    j_idx = np.arange(n)
    idx_mat = i_idx[None, :] - j_idx[:, None] + (n - 1)
    valid = (idx_mat >= 0) & (idx_mat <= 2*n-2)
    idx_clipped = np.clip(idx_mat, 0, 2*n-2)
    M1 = np.where(valid, w[idx_clipped], 0.0)
    g = M1.sum(axis=1) - M1.dot(h) - M1.T.dot(h)
    return g

def gradient_descent(h_init, n_steps, alpha_schedule, lr, momentum, 
                     time_limit, start_time):
    n = len(h_init)
    target = float(n) / 2.0
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    
    best_h = h.copy()
    best_score = compute_c5(h)
    stall = 0
    max_stall = 2000
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        if step % 500 == 0 and time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = step / max(n_steps - 1, 1)
        # alpha increases over time
        alpha = alpha_schedule[min(step, len(alpha_schedule)-1)]
        
        corr = full_corr_fast(h)
        corr_shifted = alpha * corr
        corr_shifted -= corr_shifted.max()
        exp_vals = np.exp(np.clip(corr_shifted, -500, 0))
        w = exp_vals / exp_vals.sum()
        
        g = compute_gradient(h, w)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * g
        h_new = h - lr * v
        
        # Project
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h_new < 1.0)[0]
            else:
                idx = np.where(h_new > 0.0)[0]
            if len(idx) > 0:
                h_new[idx] += diff / len(idx)
                h_new = np.clip(h_new, 0.0, 1.0)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
    
    return best_h, best_score

def fast_greedy(h, max_iters, rng, time_limit, start_time):

    max_stall = 500

        if time.time() - start_time > time_limit:

            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if k_star == 0:
            stall += 1

        # Compute A and B for candidate selection
        A = np.zeros(n)
        idx = positions + k_star

        idx = positions - k_star

        B = np.zeros(n)
        idx = positions - k_star

        idx = positions + k_star

    """Compute subgradient of max correlation at the known worst shift d_star."""
    n = len(h)
    g = np.zeros(n)
    # For shift d_star: ∂c(d)/∂h[j] = (1-h[j+d*])·[0≤j+d*<n] - h[j-d*]·[0≤j-d*<n]
    # First term: (1-h[j+d*]) for j such that 0 ≤ j+d* < n
    if d_star >= 0:

            g[j] -= h[j - d_star]
    else:
        dd = -d_star
        for j in range(dd, n):
            g[j] += (1 - h[j - dd])
        for j in range(n - dd):
            g[j] -= h[j + dd]
