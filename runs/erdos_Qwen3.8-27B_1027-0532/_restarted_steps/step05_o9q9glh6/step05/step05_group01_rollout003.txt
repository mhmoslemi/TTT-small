def smooth_obj_and_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    obj = c_max + np.log(Z) / alpha
    
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    
    # For efficiency, only process non-negligible weights
    thresh = 1e-10 * Z
    active = weights > thresh
    for s_idx in np.where(active)[0]:
        w = weights[s_idx] / Z
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(h, alpha):
    """Compute smooth objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    obj = c_max + np.log(Z) / alpha
    
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    
    thresh = 1e-12 * Z
    active = np.where(weights > thresh)[0]
    for s_idx in active:
        w = weights[s_idx] / Z
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    return obj, grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_with_grad(h_start, n, target, alpha, maxiter, ftol):
    """SLSQP with analytic gradient of smoothed objective."""
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    
    def obj_fun(h):
        val, _ = smooth_obj_grad(h, alpha)
        return val
    
    def obj_grad(h):
        _, g = smooth_obj_grad(h, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj_fun, h_start, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        def obj_fun(h):
            val, _ = smooth_obj_grad(h, alpha)
            return val
        def obj_grad(h):
            _, g = smooth_obj_grad(h, alpha)
            return g
        try:
            res = minimize(obj_fun, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def bb_projected_grad(h, n, target, time_limit, max_iters, alpha_init=100.0, 
                      alpha_growth=1.5, alpha_max=100000.0):
    """Barzilai-Borwein projected gradient with adaptive alpha."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    _, grad_prev = smooth_obj_grad(h, alpha_init)
    alpha = alpha_init
    s_prev = None
    y_prev = None
    h_prev = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = np.dot(s_prev, y_prev)
            if sy > 1e-12:
                ss = np.dot(s_prev, s_prev)
                yy = np.dot(y_prev, y_prev)
                lr = ss / sy  # BB1
                lr = max(1e-6, min(lr, 0.5))
            else:
                lr = 0.01
        else:
            lr = 0.01
        
        # Take step
        h_new = project_feasible(h - lr * grad_prev, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Update
        s_curr = h_new - h
        _, grad_curr = smooth_obj_grad(h_new, alpha)
        y_curr = grad_curr - grad_prev
        
        s_prev = s_curr
        y_prev = y_curr
        grad_prev = grad_curr
        h_prev = h
        h = h_new
        
        # Occasionally increase alpha
        if it % 50 == 0:
            alpha = min(alpha * alpha_growth, alpha_max)
            _, grad_prev = smooth_obj_grad(h, alpha)
    
    return best_h, best_c5


def accelerated_smooth_grad(h, n, target, time_limit, max_iters, alpha=1000.0):
    """Accelerated (Nesterov) projected gradient for smoothed objective."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    z = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    L = 2.0 * n  # Lipschitz constant estimate (rough)
    lr = 1.0 / L
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        _, grad_z = smooth_obj_grad(z, alpha)
        # Project gradient to constraint manifold
        grad_proj = grad_z - np.mean(grad_z)
        
        # Nesterov step
        h_new = project_feasible(z - lr * grad_proj, target)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z_new = h_new + (t - 1.0) / t_new * (h_new - h)
        z_new = np.clip(z_new, 0.0, 1.0)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        z = z_new
        t = t_new
    
    return best_h, best_c5

        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                # Compute the "direction" vector g for this pair
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                # Binary search for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def boundary_reduce_optimize(h, n, target, time_limit, tol=0.005):
    """Identify boundary variables and optimize only free ones."""
    start = time.time()
    h = h.copy()
    
    # Identify boundary variables
    fixed_zero = h < tol
    fixed_one = h > 1.0 - tol
    free = ~(fixed_zero | fixed_one)
    
    if np.sum(free) < 5:
        return h, compute_c5(h)
    
    free_idx = np.where(free)[0]
    n_free = len(free_idx)
    
    # The target sum for free variables
    fixed_sum = np.sum(h[fixed_zero]) + np.sum(h[fixed_one])
    free_target = target - fixed_sum
    
    # Optimize only free variables using SLSQP
    _global_n = n_free
    _global_target = free_target
    
    def free_to_full(h_free):
        h_full = h.copy()
        h_full[free_idx] = h_free
        return h_full
    
    def full_to_free(h_full):
        return h_full[free_idx].copy()
    
    def obj_fun(h_free):
        h_full = free_to_full(h_free)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        c_max = np.max(corr)
        alpha = 5000.0
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    bounds = [(0.0, 1.0)] * n_free
    cons = [{'type': 'eq', 'fun': lambda h_f: np.sum(h_f) - free_target,
             'jac': lambda h_f: np.ones(n_free)}]
    
    h_free_start = full_to_free(h)
    try:
        res = minimize(obj_fun, h_free_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        h_free_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = free_to_full(h_free_opt)
        h_opt = project_feasible(h_opt, target)
        
        if compute_c5(h_opt) < compute_c5(h):
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    
    return h, compute_c5(h)
