import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.array(h, dtype=float)
    n = len(h)
    # Euclidean projection onto {x : 0 <= x <= 1, sum x = target}
    # Using water-filling (tau method)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k == n:
            if tau >= 0 and tau <= h_sorted[k-1]:
                return np.clip(h - tau, 0.0, 1.0)
        else:
            if h_sorted[k-1] >= tau >= h_sorted[k]:
                return np.clip(h - tau, 0.0, 1.0)
    # Bisection fallback
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    # Adaptive initial step size
    obj0, grad0 = lse_obj_and_grad(x, alpha)
    gnorm = np.linalg.norm(grad0)
    if gnorm > 1e-12:
        step = 1.0 / (alpha * 0.1 + 1.0)
    else:
        step = 0.01
        
    lr_max = 1.0 / max(alpha * 0.01, 0.01)
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        
        # Backtracking line search
        for _ in range(60):
            x_new = project_feasible(y_clipped - step * grad_y, target)
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            # Decreasing step if not sufficient decrease
            if obj_new <= obj_y - 1e-4 * step * np.dot(grad_y, x_new - y_clipped):
                break
            step *= 0.5
        else:

            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step * 1.2, lr_max)  # Slowly increase step
        
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:

    return best_x, best_obj

def active_set_polish(h, target, deadline, n_active=5, use_momentum=True):
    """Polish using subgradient on active set with optional momentum."""
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    
    velocity = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for iteration in range(100000):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_max = float(np.max(corr))
        cur_score = c_max * dx
        
        # Select active set (top correlations)
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        # Compute weights (temperature-based)
        diffs = corr[top_idx] - corr[top_idx[0]]
        T = max(0.001, (c_max - corr[top_idx[-1]]) * 0.5)
        w = np.exp(diffs / T)
        w /= w.sum()
        
        # Compute gradient of weighted sum
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            if ki == 0:
                # d/dh[i] of h[i]*(1-h[i]) = 1 - 2*h[i]
                grad[:] += wi * (1.0 - 2.0 * h_cur)
            else:
                jp = np.arange(0, n - ki)
                jn = np.arange(ki, n)
                grad[jp] += wi * (1.0 - h_cur[jp + ki])
                grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        # Momentum step
        velocity = mu * velocity + (1 - mu) * grad
        effective_grad = velocity
        
        # Adaptive step size
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h_cur - eps * effective_grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                break
        else:
            # No improvement, reduce momentum and try smaller step
            mu = max(0.5, mu * 0.95)
            velocity *= 0.5
    
    return h_cur, cur_score

        c_max = float(np.max(corr))
        cur_score = c_max * dx
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        w = np.exp(diffs / T_adapt)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            if ki == 0:
                grad[:] += wi * (1.0 - 2.0 * h_cur)
            else:
                jp = np.arange(0, n - ki)
                jn = np.arange(ki, n)
                grad[jp] += wi * (1.0 - h_cur[jp + ki])
                grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        velocity = mu * velocity + (1.0 - mu) * grad
        improved = False
        for eps in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = project_feasible(h_cur - eps * velocity, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                improved = True
                break
        if not improved:
            velocity *= 0.5
            mu = max(0.3, mu * 0.98)
            T_adapt = max(0.002, T_adapt * 0.95)
        else:
            T_adapt = max(0.002, T_adapt * 0.995)
    return h_cur, cur_score

def structured_restart(base_h, n, target, seed):
    rng = np.random.RandomState(seed)
    starts = []
    h_noise = project_feasible(base_h + rng.randn(n) * 0.03, target)
    starts.append(h_noise)
    h_noise2 = project_feasible(base_h + rng.randn(n) * 0.08, target)
    starts.append(h_noise2)
    x = np.linspace(0, 2, n)
    h_sine = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2.0 + rng.uniform(0, 2*np.pi))
    h_sine = project_feasible(np.clip(h_sine, 0, 1), target)
    starts.append(h_sine)
    block_size = n // 20
    h_blocks = np.zeros(n)
    for i in range(0, n, 2 * block_size):
        h_blocks[i:i+block_size] = 1.0
    h_blocks = project_feasible(h_blocks + rng.randn(n) * 0.1, target)
    starts.append(h_blocks)
    return starts

def two_lag_polish(h, target, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    for _ in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        top2 = np.argsort(corr)[::-1][:2]
        k1, k2 = int(top2[0]), int(top2[1])
        def obj_2d(params):
            delta1 = params[0]
            delta2 = params[1]
            h_try = h_cur.copy()
            if k1 < n:
                mask1 = np.arange(n - k1)
                h_try[mask1] -= delta1 * (1.0 - h_cur[mask1 + k1])
                h_try[k1:] -= delta1 * h_cur[:-k1] if k1 > 0 else 0
            if k2 < n and k2 != k1:
                mask2 = np.arange(n - k2)
                h_try[mask2] -= delta2 * (1.0 - h_cur[mask2 + k2])
                if k2 > 0:
                    h_try[k2:] -= delta2 * h_cur[:-k2]
            h_try = project_feasible(h_try, target)
            return compute_c5(h_try)
        for scale in [0.001, 0.0005, 0.0002]:
            result = minimize(obj_2d, [0, 0], method='Nelder-Mead',
                            options={'xatol': scale*0.01, 'fatol': 1e-8, 'maxiter': 200})
            if result.fun < cur_score - 1e-14:
                delta1, delta2 = result.x
                h_try = h_cur.copy()
                if k1 < n:
                    mask1 = np.arange(n - k1)
                    h_try[mask1] -= delta1 * (1.0 - h_cur[mask1 + k1])
                    if k1 > 0:
                        h_try[k1:] -= delta1 * h_cur[:-k1]
                if k2 < n and k2 != k1:
                    mask2 = np.arange(n - k2)
                    h_try[mask2] -= delta2 * (1.0 - h_cur[mask2 + k2])
                    if k2 > 0:
                        h_try[k2:] -= delta2 * h_cur[:-k2]
                h_cur = project_feasible(h_try, target)
                cur_score = compute_c5(h_cur)
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    alphas_cont = [100, 400, 2000, 8000, 30000]
    times_cont = [60, 50, 60, 80, 80]
    for a, td in zip(alphas_cont, times_cont):
        if time.time() > deadline - td - 30:
            break
        h_new, sc_new = fista_optimize(best_h, target, a, 300000, start_time + td)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 50:
        h_new, sc_new = active_set_polish(best_h, target, start_time + 50, n_active=5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 60:
        h_new, sc_new = fista_optimize(best_h, target, 50000, 300000, start_time + 60)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 40:
        h_new, sc_new = active_set_polish(best_h, target, start_time + 40, n_active=8)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 50:
        h_new, sc_new = active_set_polish(best_h, target, start_time + 50, n_active=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 40:
        h_new, sc_new = two_lag_polish(best_h, target, start_time + 40)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 60:
        restarts = structured_restart(best_h, n, target, seed)
        for i, h_r in enumerate(restarts):
            if time.time() > deadline - 20:
                break
            h_new, sc_new = fista_optimize(h_r, target, 5000, 100000, start_time + 15)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            if time.time() < deadline - 15:
                h_new2, sc_new2 = active_set_polish(h_new, target, start_time + 12, n_active=5)
                if sc_new2 < best_score:
                    best_score = sc_new2
                    best_h = h_new2

    if time.time() < deadline - 30:
        h_new, sc_new = active_set_polish(best_h, target, start_time + 30, n_active=5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if time.time() < deadline - 20:
        h_new, sc_new = fista_optimize(best_h, target, 80000, 300000, start_time + 20)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
