import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h, -100.0, 100.0)
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def lse_grad_fft(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective using FFT.
    Returns (objective_value, gradient)."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    fsize = 2 * n
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    
    # B[i] = (w * h)[i] = sum_k w[k] h[i-k] for valid k
    B = np.fft.irfft(W * H, fsize)[:n]
    
    # A[i] = (w ⋆ h)[i] = sum_k w[k] h[i+k] for valid k
    # (w ⋆ h)[i] = (w_rev * h)[i + n - 1]
    w_rev = weights[::-1]
    W_rev = np.fft.rfft(w_rev, fsize)
    conv_wh_rev = np.fft.irfft(W_rev * H, fsize)
    A = conv_wh_rev[n-1:2*n-1]
    
    # P[i] = sum_{k=0}^{n-1-i} w[k] (prefix sum)
    P = np.cumsum(weights)[::-1]
    
    grad = dx * (P - A - B)
    return obj, grad

def fista_fast(h0, target, alpha, max_iters, deadline, step_init=None):
    """FISTA with FFT-based gradient and fixed (or warm-started) step size."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0.astype(float).copy(), target)
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if step_init is None:
        step = 1.0 / (alpha * dx + 0.02)  # conservative initial step
    else:
        step = step_init
    
    max_step = 5.0 / (alpha * dx + 0.02)
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        obj_y, grad_y = lse_grad_fft(y_clipped, alpha)
        
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Light backtracking (only 3 iterations max)
        for _ in range(3):
            obj_new, _ = lse_grad_fft(x_new, alpha)
            if obj_new <= obj_y + 0.5 * step * np.dot(x_new - y_clipped, x_new - y_clipped) + 1e-12:
                break
            step *= 0.5
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-10)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step * 1.05, max_step)  # slowly grow step
        
        x_prev = x
        x = x_new
        t = t_new
        
        if k % 50 == 0:
            sc = compute_c5(x)
            if sc < best_obj:
                best_obj = sc
                best_x = x.copy()
    
    sc = compute_c5(x)
    if sc < best_obj:
        best_obj = sc
        best_x = x.copy()
    
    return best_x, best_obj

def topk_polish(h, target, deadline, k_top=3, max_rounds=500):
    """Polish by directly minimizing the top-k correlations using subgradient."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        # Subgradient: average gradient of top-k lags
        grad = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            for i in range(n - ki):
                grad[i] += (1.0 - h_cur[i + ki])
            for i in range(ki, n):
                grad[i] -= h_cur[i - ki]
        grad *= dx / k_top
        
        # Try decreasing step sizes
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                break
    
    return h_cur, cur_score

def structured_perturb(h, n, rng):
    """Create structured perturbations: shifting, inverting, smoothing."""
    choice = rng.randint(4)
    if choice == 0:
        # Circular shift
        shift = rng.randint(1, n // 4)
        return np.roll(h, shift)
    elif choice == 1:
        # Invert in a random window
        start = rng.randint(0, n // 2)
        length = rng.randint(n // 8, n // 3)
        h_new = h.copy()
        h_new[start:start+length] = 1.0 - h[start:start+length]
        return h_new
    elif choice == 2:
        # Smooth (blur)
        kernel = np.ones(rng.randint(3, 11)) / 5.0
        h_new = np.convolve(h, kernel, mode='same')
        return h_new
    else:
        # Gaussian noise
        sigma = rng.choice([0.02, 0.05, 0.1, 0.15])
        return h + rng.randn(n) * sigma

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h, -100.0, 100.0)
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def lse_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    fsize = 2 * n
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    B = np.fft.irfft(W * H, fsize)[:n]
    w_rev = weights[::-1]
    W_rev = np.fft.rfft(w_rev, fsize)
    conv_rev = np.fft.irfft(W_rev * H, fsize)
    A = conv_rev[n-1:2*n-1]
    P = np.cumsum(weights)[::-1]
    grad = dx * (P - A - B)
    return obj, grad

def topk_polish(h, target, deadline, k_top=3, max_rounds=500):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            for i in range(n - ki):
                grad[i] += (1.0 - h_cur[i + ki])
            for i in range(ki, n):
                grad[i] -= h_cur[i - ki]
        grad *= dx / float(k_top)
        improved = False
        for eps in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                improved = True
                break

            break
    return h_cur, cur_score

def structured_perturb(h, n, rng):
    choice = rng.randint(5)
    if choice == 0:
        shift = rng.randint(1, max(2, n // 5))
        return np.roll(h, shift)
    elif choice == 1:
        start = rng.randint(0, max(1, n // 3))
        length = rng.randint(max(2, n // 10), max(3, n // 3))
        h_new = h.copy()
        h_new[start:start+length] = 1.0 - h[start:start+length]
        return h_new
    elif choice == 2:
        klen = rng.choice([3, 5, 7])
        kernel = np.ones(klen) / klen
        h_new = np.convolve(h, kernel, mode='same')
        return h_new
    elif choice == 3:
        sigma = rng.choice([0.03, 0.06, 0.1, 0.15, 0.2])
        return h + rng.randn(n) * sigma
    else:
        mask = rng.rand(n) < 0.3
        h_new = h.copy()
        h_new[mask] = rng.rand(mask.sum()) * 0.5
        return h_new
