        if n_mask == 0:

        h = np.clip(h, 0.0, 1.0)

    vm = (j - k >= 0) & (j - k < n)

    sm = np.clip(j - k, 0, n-1)
    mp = vp.astype(np.float64)
    mm = vm.astype(np.float64)
    return (dx, sp, sm, mp, mm)

    mc = float(np.max(corr))

    se = float(es.sum())
    fv = mc + np.log(se) / alpha
    p = es / se
    pr = p[None, :]
    hp = h[sp]
    hm = h[sm]
    t1 = (mp * pr).sum(axis=1)
    t2 = (hp * mp * pr).sum(axis=1)

        g = g + 2.0 * penalty * sv
    return fv, g

            self.alpha = alpha
            self.pre = pre

            self.pre = pre
