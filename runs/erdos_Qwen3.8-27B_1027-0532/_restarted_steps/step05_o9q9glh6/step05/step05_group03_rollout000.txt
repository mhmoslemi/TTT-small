import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def get_corr_over(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_bang_bang(h, target, threshold=0.5):
    n = len(h)
    h_bb = (h > threshold).astype(float)
    target_ones = int(target)
    n_ones = int(h_bb.sum())
    if n_ones > target_ones:
        ones_idx = np.where(h_bb == 1)[0]
        order = np.argsort(h[ones_idx])[:n_ones - target_ones]
        h_bb[ones_idx[order]] = 0

        zeros_idx = np.where(h_bb == 0)[0]

def swap_based_search(h_bb, n_passes=50, n_tries_per_pass=5000, seed=42, time_limit=None, start_time=None):

            j = zeros_idx[j_pos]

def slsqp_optimize(h, alphas, maxiter, ftol, target):
    n = len(h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(x, alpha):
        h_proj = project_feasible(x, target)
        corr = get_corr_over(h_proj)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def cons_eq(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    
    for a in alphas:
        try:
            res = minimize(lambda x: obj_func(x, a), h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp, target):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    
    for step in range(n_steps):
        corr = get_corr_over(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        curr_score = float(np.max(get_corr_over(h_cur))) * dx
        if sc_new < curr_score - 1e-15:

            lr = min(lr * 1.005, 0.005)

            lr *= lr_decay

                break
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_cur.copy()
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
