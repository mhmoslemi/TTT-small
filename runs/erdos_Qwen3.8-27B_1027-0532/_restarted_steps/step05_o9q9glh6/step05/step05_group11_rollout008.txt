def project_simplex_box(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        # Water-filling: find threshold where we add equally
        h_work = h.copy()
        remaining = diff
        for _ in range(len(idx)):
            per = remaining / len(idx)
            if per <= 1.0 - h_work[idx].max():
                h_work[idx] += per
                break
            else:
                # Some elements hit 1
                overflow = (1.0 - h_work[idx]) 
                h_work[idx] = 1.0
                remaining -= overflow.sum()
                idx = idx[h_work[idx] < 1.0 - 1e-15]
                if len(idx) == 0:
                    break

        idx = np.where(h > 0.0)[0]

        f_cur = smooth_max(corr, alpha)
        while step > 1e-10:

            corr_try = np.correlate(h_try, 1-h_try, 'full') * (2.0/n)

            best_val = val
            best_h = h.copy()

def compute_smooth_grad(h, alpha):
    """Compute gradient of smooth max of overlaps w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    n_half = n - 1
    g = 1.0 - h
    corr_raw = np.correlate(h, g, mode='full')

    total = np.sum(exp_s)
    p = exp_s / total  # p[m] for m = 0..2n-2
