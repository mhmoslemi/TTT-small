def binarize(h, n):
    # Round to binary while maintaining exactly n/2 ones
    # Sort by value, keep top n/2 as 1
    thresholds = np.sort(h)[::-1]
    # The threshold is the n/2-th largest value
    # Actually, we want exactly n/2 to be 1
    # Use the median as a starting point
    median_val = np.median(h)
    h_bin = (h >= median_val).astype(float)
    # Fix if count isn't exactly n/2
    count = int(h_bin.sum())
    if count > n // 2:
        # Need to flip some 1s to 0s
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])
        for idx in order[:count - n//2]:
            h_bin[ones[idx]] = 0
    elif count < n // 2:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:n//2 - count]:
            h_bin[zeros[idx]] = 1
    return h_bin

def multiplicative_weights(h_init, n, target, n_iter, eta):
    h = h_init.copy()
    dx = 2.0 / n
    # Weights over lags (2n-1 lags)
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    
    for it in range(n_iter):
        # Inner: minimize sum_k w_k * corr(k) subject to 0<=h<=1, sum(h)=target
        # This is a QP: minimize sum_k w_k * sum_i h[i]*(1-h[i+k])
        # = sum_i h[i] * sum_k w_k * (1-h[i+k])
        # = sum_i h[i] * (sum_k w_k - sum_k w_k * h[i+k])
        # = (sum_k w_k) * sum_i h[i] - sum_i h[i] * (w * h shifted)
        # This is quadratic in h.
        # For simplicity, use projected gradient:
        for pg_iter in range(50):
            corr = compute_corr(h)
            g = np.zeros(n)
            for k in range(n):
                wk = w[n-1+k]
                if wk < 1e-15: continue
                g[:n-k] += wk * (1.0 - h[k:])
                g[k:] -= wk * h[:n-k]
            for ell in range(1, n):
                wk = w[n-1-ell]
                if wk < 1e-15: continue
                g[ell:] += wk * (1.0 - h[:n-ell])
                g[:n-ell] -= wk * h[ell:]
            g *= dx
            # Projected gradient step
            lr = 0.01
            h_new = h - lr * g
            h_new = project_feasible(h_new, target)
            if np.max(np.abs(h_new - h)) < 1e-10:
                h = h_new
                break
            h = h_new
        
        # Outer: update weights
        corr = compute_corr(h) * dx
        max_corr = np.max(corr)
        w = w * np.exp(eta * corr / max(1e-10, max_corr))
        w /= np.sum(w)
    
    return h, np.max(compute_corr(h)) * dx

def binary_local_search(h_init, n, max_passes=500, verbose=False):
    # Binarize: keep top n/2 values as 1
    h = (h_init >= np.sort(h_init)[::-1][n//2 - 1]).astype(float)
    # Ensure exactly n/2 ones
    if h.sum() != n // 2:
        # Adjust
        sorted_idx = np.argsort(h_init)[::-1]
        h = np.zeros(n)
        h[sorted_idx[:n//2]] = 1
    
    best_score = np.max(compute_corr(h))
    best_h = h.copy()
    
    for pass_num in range(max_passes):
        corr = compute_corr(h).astype(float)
        curr_max = np.max(corr)
        
        ones = np.where(h == 1)[0].astype(int)
        zeros = np.where(h == 0)[0].astype(int)
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_lags = 2 * n - 1
        
        # Compute effects (vectorized)
        # effect_remove[i, m] = change in corr[m] when h[ones[i]] goes 1->0
        effect_remove = np.zeros((n_ones, n_lags))
        for idx, a in enumerate(ones):
            # Positive lags k=0..n-1-a: first factor
            k_max = n - 1 - a
            if k_max >= 0:
                ks = np.arange(0, k_max + 1)
                effect_remove[idx, n-1+ks] += h[a + ks] - 1
            # Positive lags k=1..a: second factor
            if a >= 1:
                ks = np.arange(1, a + 1)
                effect_remove[idx, n-1+ks] += h[a - ks]
            # Negative lags ell=1..a: first factor
            if a >= 1:
                ells = np.arange(1, a + 1)
                effect_remove[idx, n-1-ells] += h[a - ells] - 1
            # Negative lags ell=1..n-1-a: second factor
            if n - 1 - a >= 1:
                ells = np.arange(1, n - a)
                effect_remove[idx, n-1-ells] += h[a + ells]
        
        # effect_add[j, m] = change in corr[m] when h[zeros[j]] goes 0->1
        effect_add = np.zeros((n_zeros, n_lags))
        for idx, b in enumerate(zeros):
            # Positive lags k=0..n-1-b: first factor
            k_max = n - 1 - b
            if k_max >= 0:
                ks = np.arange(0, k_max + 1)
                effect_add[idx, n-1+ks] += 1 - h[b + ks]
            # Positive lags k=1..b: second factor
            if b >= 1:
                ks = np.arange(1, b + 1)
                effect_add[idx, n-1+ks] -= h[b - ks]
            # Negative lags ell=1..b: first factor
            if b >= 1:
                ells = np.arange(1, b + 1)
                effect_add[idx, n-1-ells] += 1 - h[b - ells]
            # Negative lags ell=1..n-1-b: second factor
            if n - 1 - b >= 1:
                ells = np.arange(1, n - b)
                effect_add[idx, n-1-ells] -= h[b + ells]
        
        # Find best pair
        best_improvement = 0
        best_ii = -1
        best_jj = -1
        
        # Process in chunks to save memory
        chunk_size = 50
        for ii_start in range(0, n_ones, chunk_size):
            ii_end = min(ii_start + chunk_size, n_ones)
            # deltas shape: (chunk, n_zeros, n_lags)
            deltas = effect_remove[ii_start:ii_end][:, None, :] + effect_add[None, :, :]
            new_corrs = corr[None, None, :] + deltas  # (chunk, n_zeros, n_lags)
            new_maxes = np.max(new_corrs, axis=2)  # (chunk, n_zeros)
            improvements = curr_max - new_maxes  # (chunk, n_zeros)
            # Find the best in this chunk
            flat_imp = np.max(improvements)
            if flat_imp > best_improvement:
                best_improvement = flat_imp
                local_best = np.unravel_index(np.argmax(improvements), improvements.shape)
                best_ii = ii_start + local_best[0]
                best_jj = local_best[1]
        
        if best_improvement <= 1e-10:
            if verbose:
                print(f"  Pass {pass_num}: no improvement (best_improvement={best_improvement:.2e})")
            break
        
        # Apply flip
        a = ones[best_ii]
        b = zeros[best_jj]
        h[a] = 0
        h[b] = 1
        
        new_score = np.max(compute_corr(h))
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            if verbose:
                print(f"  Pass {pass_num}: {curr_max:.1f} -> {new_score:.1f}")
    
    return best_h, best_score
