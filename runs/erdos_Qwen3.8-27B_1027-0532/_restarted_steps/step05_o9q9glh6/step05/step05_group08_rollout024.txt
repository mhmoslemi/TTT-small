Actually, I realize I should think more carefully about the game-theoretic approach. The key question is: what is the inner optimization solving?

The inner problem is: min_h Σ_k p_k * corr_k(h)

where corr_k(h) is the correlation at shift k.

For a fixed p, this is a quadratic function of h:
Σ_k p_k * Σ_i h_i(1-h_{i±k}) = Σ_k p_k * [Σ_i h_i - Σ_i h_i * h_{i±k}]
= (Σ_k p_k) * Σ_i h_i - Σ_k p_k * Σ_i h_i * h_{i±k}
= (n/2) * (Σ_k p_k) - Σ_k p_k * Σ_i h_i * h_{i±k}  [since Σ h_i = n/2]

Since Σ_k p_k = 1 (p is a probability distribution), this simplifies to:
n/2 - Σ_k p_k * Σ_i h_i * h_{i±k}

So minimizing this is equivalent to maximizing:
Σ_k p_k * Σ_i h_i * h_{i±k}

Which is h^T Q h where Q is the weighted sum of shift matrices.

Now, Q is a symmetric matrix (we can show this), and it's banded (with bandwidth determined by the support of p). If p has full support, Q is dense but has a Toeplitz-like structure.

The key insight: maximizing h^T Q h subject to 0 ≤ h ≤ 1, Σ h = n/2.

If Q is PSD, this is a convex problem (well, maximizing a convex function is not convex... wait).

Actually, h^T Q h is a quadratic form. If Q is PSD, then h^T Q h is convex, and we're maximizing a convex function over a convex set. This is a non-convex problem in general (the maximum of a convex function over a convex polytope is at a vertex).

If Q is indefinite, it's even harder.

So the inner problem is non-convex regardless. L-BFGS-B will find a local optimum. That's fine for the heuristic approach.

Let me also think about whether there's a better way to handle the inner problem. Since we're maximizing a quadratic form, the Karush-Kuhn-Tucker (KKT) conditions are:
2Qh = λ * 1 + μ (where μ is the vector of dual variables for the box constraints)

At the optimum, for components where 0 < h_i < 1, we have (Qh)_i = λ/2.
For h_i = 0, we have (Qh)_i ≤ λ/2.
For h_i = 1, we have (Qh)_i ≥ λ/2.

This is similar to the conditions for a thresholded eigenvector or a "shrinkage" solution.

One approach: use the power iteration or Lanczos method to find the top eigenvector of Q, then project and threshold. But this might not give the best solution due to the constraints.

I think L-BFGS-B is the most practical approach for the inner problem. Let me proceed with that.

    return np.ones(_g_n)

def game_theoretic_optimize(h, n, target, n_rounds, base_temp, cool_rate, mix_rate, deadline):
    global _g_p, _g_dx
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    _g_dx = dx

    corr = compute_corr(h_cur)
    c_max = float(np.max(corr))
    temp = base_temp
    _g_p = np.exp((corr - c_max) / temp)
    _g_p /= _g_p.sum()

    bounds = [(0.0, 1.0)] * n

    for r in range(n_rounds):
        if time.time() > deadline:
            break

        try:
            res = minimize(_wcorr_obj, h_cur, method='L-BFGS-B', jac=_wcorr_grad,
                          bounds=bounds, maxiter=200, gtol=1e-10)
            h_new = project_feasible(res.x, target)
        except Exception:
            h_new = h_cur

        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new.copy()

        corr_new = compute_corr(h_cur)
        c_max_new = float(np.max(corr_new))
        temp = max(temp * cool_rate, 3.0)
        p_new = np.exp((corr_new - c_max_new) / temp)
        p_new /= p_new.sum()
        _g_p = mix_rate * p_new + (1.0 - mix_rate) * _g_p

    return best_h, best_score

def lsb_round(h, n, target, alpha, mu, maxiter):
    global _g_alpha, _g_mu, _g_target, _g_n
    _g_alpha = alpha
    _g_mu = mu
    _g_target = target
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_pen_obj, h, method='L-BFGS-B', jac=_lse_pen_grad,
                      bounds=bounds, maxiter=maxiter, gtol=1e-12, factr=1e7)
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    global _g_alpha, _g_target, _g_n
    _g_alpha = alpha
    _g_target = target
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', jac=_slsqp_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def top_k_subgradient(h, n, target, n_steps, top_k, temp, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        weights = np.exp((top_vals - top_vals[0]) / temp)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(top_k)
            weights[0] = 1.0
        else:
            weights /= w_sum
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[k:] += wk * (1.0 - h_cur[:n-k])
                g[:n-k] -= wk * h_cur[k:]
            else:
                ell = -k
                g[:n-ell] += wk * (1.0 - h_cur[ell:])
                g[ell:] -= wk * h_cur[:n-ell]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.005, 0.008)
        else:
            lr *= 0.93
            if lr < 1e-13:
                break
    return best_h, best_score

def subgradient_single(h, n, target, n_steps, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        m_star = int(np.argmax(corr))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h_cur[:n-k])
            g[:n-k] -= h_cur[k:]
        else:
            ell = -k
            g[:n-ell] += (1.0 - h_cur[ell:])
            g[ell:] -= h_cur[:n-ell]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, 0.01)
        else:
            lr *= 0.88
            if lr < 1e-12:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_mu, _g_target, _g_n
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Game-theoretic optimization (multiplicative weights)
    if time.time() < deadline - 400:
        h_gt, sc_gt = game_theoretic_optimize(best_h, n, target, 60, 40.0, 0.96, 0.6, deadline - 400)
        if sc_gt < best_score:
            best_score = sc_gt
            best_h = h_gt.copy()

    # Phase 2: L-BFGS-B with LSE continuation
    alphas_p2 = [30, 100, 400, 1500, 5000, 15000]
    for alpha in alphas_p2:
        if time.time() > deadline - 300:
            break
        for mu in [5.0, 50.0]:
            h_new = lsb_round(best_h, n, target, alpha, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()

    # Phase 3: SLSQP refinement
    for alpha in [300, 1500, 8000, 30000]:
        if time.time() > deadline - 200:
            break
        h_new = slsqp_round(best_h, n, target, alpha, 2500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 4: Top-k subgradient with various settings
    for top_k in [3, 5, 10, 20, 35]:
        if time.time() > deadline - 120:
            break
        for temp in [0.01, 0.005, 0.002]:
            if time.time() > deadline - 100:
                break
            h_sg, sc_sg = top_k_subgradient(best_h, n, target, 2000, top_k, temp, 0.004, deadline - 90)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg.copy()

    # Phase 5: Single-worst subgradient
    if time.time() < deadline - 80:
        h_sg, sc_sg = subgradient_single(best_h, n, target, 3000, 0.005, deadline - 70)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg.copy()

    # Phase 6: Higher resolution (n=600)
    if time.time() < deadline - 250:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)

        if time.time() < deadline - 180:
            h_gt2, sc_gt2 = game_theoretic_optimize(best_h2, n2, target2, 30, 50.0, 0.95, 0.5, deadline - 180)
            if sc_gt2 < best_score2:
                best_score2 = sc_gt2
                best_h2 = h_gt2.copy()

        for alpha in [50, 300, 2000, 10000]:
            if time.time() > deadline - 100:
                break
            h2_new = lsb_round(best_h2, n2, target2, alpha, 20.0, 200)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new.copy()

        if time.time() < deadline - 80:
            h2_new = slsqp_round(best_h2, n2, target2, 5000, 1500, 1e-16)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new.copy()

        if time.time() < deadline - 60:
            h_sg2, sc_sg2 = top_k_subgradient(best_h2, n2, target2, 1500, 5, 0.005, 0.003, deadline - 50)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2.copy()

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2.copy()

    # Phase 7: Second game-theoretic round at best resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if time.time() < deadline - 60:
        h_gt3, sc_gt3 = game_theoretic_optimize(best_h, n_cur, target_cur, 40, 25.0, 0.94, 0.7, deadline - 40)
        if sc_gt3 < best_score:
            best_score = sc_gt3
            best_h = h_gt3.copy()

    # Phase 8: Final SLSQP + subgradient
    if time.time() < deadline - 20:
        h_new = slsqp_round(best_h, n_cur, target_cur, 20000, 1500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    if time.time() < deadline - 8:
        h_fin, sc_fin = subgradient_single(best_h, n_cur, target_cur, 1000, 0.002, deadline - 5)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin.copy()

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)

    h_parent = project_feasible(np.array(initial_h_values, dtype=float), n / 2.0)
    parent_score = compute_c5(h_parent)
    if parent_score < final_score:
        return h_parent, parent_score, n

    return h_final, final_score, n_final
