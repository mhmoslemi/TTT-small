    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    lse_val = c_max + np.log(np.sum(exp_s)) / alpha
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    n1 = n - 1
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        lag = m - n1
        if lag >= 0:
            # first: (1-h[i+lag]) for i in [0, n-1-lag]
            grad[:n-lag] += w * (1.0 - h[lag:])
            # second: -h[i-lag] for i in [lag, n-1]
            grad[lag:] -= w * h[:n-lag]
        else:
            kp = -lag
            # first: (1-h[i-kp]) for i in [kp, n-1]
            grad[kp:] += w * (1.0 - h[:n-kp])
            # second: -h[i+kp] for i in [0, n-1-kp]
            grad[:n-kp] -= w * h[kp:]
    
    return lse_val, grad

def lbfgsb_with_penalty(h, target, alphas, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            xc = np.clip(x, 0.0, 1.0)
            val, _ = lse_obj_and_grad(xc, alpha)
            pen = 50.0 * (np.sum(xc) - target) ** 2
            return val + pen
        
        def grad(x):
            xc = np.clip(x, 0.0, 1.0)
            _, g = lse_obj_and_grad(xc, alpha)
            g = g + 100.0 * (np.sum(xc) - target)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 500, 'ftol': 1e-16, 'maxfun': 5000})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def slsqp_optimize(h, target, alphas, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            val, _ = lse_obj_and_grad(x, alpha)
            return val
        
        def obj_grad(x):
            val, g = lse_obj_and_grad(x, alpha)
            return val, g
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,

def subgradient_optimize(h, target, n_steps, n_active, lr_init, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        c_max = np.max(corr)
        
        # Find active lags
        sorted_idx = np.argsort(corr)[::-1]
        top_idx = sorted_idx[:n_active]
        
        # Compute weights (softmax over active lags)
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        n1 = n - 1
        for wk, m in zip(weights, top_idx):
            m = int(m)
            lag = m - n1
            if lag >= 0:
                grad[:n-lag] += wk * (1.0 - h_cur[lag:])
                grad[lag:] -= wk * h_cur[:n-lag]
            else:
                kp = -lag
                grad[kp:] += wk * (1.0 - h_cur[:n-kp])
                grad[:n-kp] -= wk * h_cur[kp:]
        
        grad *= dx
        
        # Step
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        
        # Adaptive learning rate
        if sc_new < compute_c5(h_cur) - 1e-12:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    
    return best_h, best_score

def coordinate_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Try a few values for h[i]
            old_val = h_cur[i]
            # Golden section search on [0, 1]
            best_v = old_val
            best_sc = compute_c5(h_cur)
            
            # Sample a few points
            candidates = np.array([0.0, 0.25, 0.5, 0.75, 1.0, old_val])
            for cv in candidates:
                h_test = h_cur.copy()
                h_test[i] = cv
                h_test = project_feasible(h_test, target)

            if best_v != old_val:
                h_cur[i] = best_v
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
                improved = True
        
        if not improved:
            break
    
    return h_cur, cur_score

def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            extra = np.zeros((pop_size - pop.shape[0], n))
            for i in range(extra.shape[0]):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        for i in range(pop_size):
            pop[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
    
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = np.ones(pop_size, dtype=bool)
            mask[i] = False
            avail = np.where(mask)[0]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
