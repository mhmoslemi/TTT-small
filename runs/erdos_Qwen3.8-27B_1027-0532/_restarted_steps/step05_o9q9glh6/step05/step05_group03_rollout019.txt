def make_3periodic(n, w1, gap):

    # Period = 2/3, so each period has n/3 points
    # In each period, active interval of length w1 at start, gap, then w2 at end

def compute_effect_vectors(h):
    """
    For each position m, compute the effect of flipping h[m] on O(k) for all k.
    Returns array of shape (n, n) where effect[m, k] is the change in O(k)
    when h[m] flips from 0 to 1.
    """
    n = len(h)
    effect = np.zeros((n, n))
    for m in range(n):
        # First factor: h[m] appears in O(k) for k=0,...,n-1-m
        # O(k) includes h[m]*(1-h[m+k]) for k <= n-1-m
        # Flipping h[m] from 0 to 1: O(k) increases by (1-h[m+k])
        for k in range(n - m):
            effect[m, k] += (1.0 - h[m+k])
        # Second factor: h[m] appears as h[i+k] when i+k=m, i.e., i=m-k
        # O(k) includes h[m-k]*(1-h[m]) for k=1,...,m (and m-k >= 0, m-k <= n-1-k)
        # Flipping h[m] from 0 to 1: (1-h[m]) goes from 1 to 0
        # So O(k) changes by -h[m-k] for each valid k
        for k in range(1, min(m, n-1) + 1):
            if m - k <= n - 1 - k:  # i = m-k must be valid
                effect[m, k] -= h[m-k]
    return effect
```

    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

    n = len(h)
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_effect_vectors(h):
    n = len(h)
    effect = np.zeros((n, n))
    # First factor: effect[m, k] = (1-h[m+k]) for m+k < n, i.e., k < n-m
    # This is: for all (m, k) with m+k < n, effect[m,k] = 1-h[m+k]
    # Let's use a different indexing: let s = m+k (the sum), then for each s,
    # m can be 0..s and k can be s-m. But that's not quite right either.
    
    # Alternative: create a matrix where entry (m, k) is m+k
    M, K = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    S = M + K  # S[m, k] = m + k
    valid1 = S < n
    # effect[m, k] = 1 - h[m+k] = 1 - h[S[m,k]]
    effect[valid1] = 1.0 - h[S[valid1]]
    
    # Second factor: effect[m, k] -= h[m-k] for 1 <= k <= m, and m-k <= n-1-k
    # m-k <= n-1-k => m <= n-1, which is always true for m < n
    # So the condition is just 1 <= k <= m
    valid2 = (K >= 1) & (K <= M) & (M - K >= 0)
    # m-k >= 0 is K <= M which we already have
    # m-k is the index
    if valid2.any():
        effect[valid2] -= h[(M - K)[valid2]]
    
    return effect
```

    best_max = cur_max

        ones_idx = np.where(h == 1.0)[0]

            tmp = O[None, None, :] + E0[start:end, None, :] - E1[None, :, :]
