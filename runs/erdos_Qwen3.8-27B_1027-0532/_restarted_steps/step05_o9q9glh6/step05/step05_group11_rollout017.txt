def alt_lp_step(h, n, target, n_top, reg, solver='SCS'):
    """Single LP step: fix g=1-h, optimize h to minimize max overlap on top lags + reg."""
    dx = 2.0 / n
    g = 1.0 - h
    overlaps = np.correlate(h, g, mode='full') * dx
    n_half = n - 1
    top_idx = np.argsort(overlaps)[::-1][:n_top]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for j in top_idx:
        k = j - n_half
        if k >= 0:
            lo = 0
            hi = n - k
            if hi > lo:
                coeff = g[k:k+hi] * dx
                # sum_{i=0}^{n-k-1} h[i+k] * g[i] * dx
                # = sum_{i=0}^{n-k-1} h[i+k] * g[i] * dx
                # shift: let j' = i+k, so j' from k to n-1
                # = sum_{j'=k}^{n-1} h[j'] * g[j'-k] * dx
                coeff = np.zeros(n)
                coeff[k:n] = g[0:hi] * dx
                constraints.append(coeff @ h_var <= t_var)
        else:
            k_neg = -k  # k is negative, k_neg = -k > 0
            # sum_{i=k_neg}^{n-1} h[i-k_neg] * g[i] * dx
            # let j' = i - k_neg, j' from 0 to n-k_neg-1
            # = sum_{j'=0}^{n-k_neg-1} h[j'] * g[j'+k_neg] * dx
            lo = 0
            hi = n - k_neg
            coeff = np.zeros(n)
            coeff[0:hi] = g[k_neg:k_neg+hi] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        return h_var.value
    except:
        return None
