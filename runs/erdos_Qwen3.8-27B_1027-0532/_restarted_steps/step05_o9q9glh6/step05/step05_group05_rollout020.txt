The gradient of corr[k] with respect to h:
For the overlap correlation at lag k:
corr[k] = sum_{j=0}^{n-1-k} h[j] * (1-h[j+k])

d(corr[k])/dh[j]:
- From the first factor: if j <= n-1-k, contributes (1-h[j+k])
- From the second factor: if j >= k, contributes -h[j-k]

So d(corr[k])/dh[j] = (1-h[j+k]) * [0 <= j <= n-1-k] - h[j-k] * [k <= j <= n-1]

For the log-sum-exp of the top-k correlations:
Let S be the set of top-k lag indices, and let c_i = corr[i] for i in S.
LSE = c_max + (1/alpha) * log(sum_{i in S} exp(alpha * (c_i - c_max)))

d(LSE)/dh = (1/alpha) * sum_{i in S} (alpha * exp(alpha*(c_i - c_max)) / sum_j exp(alpha*(c_j - c_max))) * d(corr[i])/dh
           = sum_{i in S} w_i * d(corr[i])/dh

where w_i = exp(alpha*(c_i - c_max)) / sum_j exp(alpha*(c_j - c_max))

This is a convex combination of the gradients of the top-k correlations.

```python
def topk_lse_grad(h, alpha, k):
    n = len(h)
    corr = compute_corr(h)
    overlap = corr[n-1:n-1+n]
    top_idx = np.argpartition(overlap, -k)[-k:]
    top_vals = overlap[top_idx]
    c_max = float(np.max(top_vals))
    shifted = alpha * (top_vals - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for w, ki in zip(weights, top_idx):
        ki = int(ki)
        # d(corr[ki])/dh[j]
        # For j in [0, n-1-ki]: (1-h[j+ki])
        # For j in [ki, n-1]: -h[j-ki]
        j_pos = np.arange(0, n - ki)
        grad[j_pos] += w * (1.0 - h[j_pos + ki])
        j_neg = np.arange(ki, n)
        grad[j_neg] -= w * h[j_neg - ki]
    
    return grad
```

Let me also add a "coordinate descent on worst lag" method that exploits the linearity of corr[k] in each individual h[i]:

For fixed k and all h[j] with j ≠ i, corr[k] as a function of h[i] is:
corr[k](h[i]) = h[i] * (1 - h[i+k]) * [i < n-k] + h[i-k] * (1-h[i]) * [i >= k] + const
= h[i] * (1 - h[i+k] - h[i-k]) + h[i-k] + const  (when both terms apply)
= h[i] * (1 - h[i+k]) + const  (when only first term)
= h[i] * (-h[i-k]) + h[i-k] + const  (when only second term)

So it's linear in h[i]! The optimal h[i] is:
- If coefficient > 0: h[i] = 1
- If coefficient < 0: h[i] = 0
- If coefficient = 0: any value

But we have the sum constraint, so we can't just set each h[i] independently. We need to do it in a way that maintains sum(h) = n/2.

One approach: for each i, compute the "desirability" of setting h[i] to 0 or 1 (the change in corr[k]). Then, find the set of indices where flipping improves the objective the most, subject to the sum constraint.

This is essentially a greedy approach:
1. For each i, compute delta_i = corr[k] with h[i] flipped (0→1 or 1→0)
2. Sort by delta_i (most negative first)
3. Flip the indices with most negative delta, subject to maintaining sum = n/2

This is like a "lazy coordinate descent" for the quadratic. Let me implement this:

```python
def coord_descent_worst(h, n, target, max_rounds, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        overlap = corr[n-1:n-1+n]
        k_star = int(np.argmax(overlap))
        
        if k_star == 0:
            # corr[0] = sum h[j]*(1-h[j]), each term is h[j] - h[j]^2
            # This is maximized at h[j]=0.5, minimized at h[j]=0 or 1
            # To minimize, push h toward 0 or 1
            # Gradient: 1 - 2*h[j]
            # If h[j] > 0.5, gradient > 0, so increasing h[j] increases corr[0]
            # If h[j] < 0.5, gradient < 0, so increasing h[j] decreases corr[0]
            # So we want to push h[j] > 0.5 toward 1 and h[j] < 0.5 toward 0
            # But this doesn't change the sum!
            # Actually for k=0, corr[0] = sum h[j] - sum h[j]^2
            # To minimize this with fixed sum, we want to maximize sum h[j]^2
            # which means making h more binary (0 or 1)
            # So push each h[j] away from 0.5
            pass
        else:
            # Compute the coefficient for each h[i] in corr[k_star]
            # d(corr[k])/dh[i] = (1-h[i+k])*[i<n-k] - h[i-k]*[i>=k]
            coeff = np.zeros(n)
            j_pos = np.arange(0, n - k_star)
            coeff[j_pos] += (1.0 - h_cur[j_pos + k_star])
            j_neg = np.arange(k_star, n)
            coeff[j_neg] -= h_cur[j_neg - k_star]
            
            # To minimize corr[k], we want to decrease h[i] where coeff[i] > 0
            # and increase h[i] where coeff[i] < 0
            # But we must maintain sum = target
            # So: find i with coeff[i] > 0 (want to decrease) and j with coeff[j] < 0 (want to increase)
            # Transfer mass from i to j: change in corr[k] ≈ coeff[i]*(-dt) + coeff[j]*(dt) = (coeff[j] - coeff[i])*dt
            # We want this negative, so coeff[j] < coeff[i] (which is guaranteed: positive < negative is wrong)
            # Wait: we want to decrease h[i] (coeff[i] > 0) and increase h[j] (coeff[j] < 0)
            # Change = -coeff[i]*dt + coeff[j]*dt = (coeff[j] - coeff[i])*dt
            # Since coeff[i] > 0 and coeff[j] < 0, coeff[j] - coeff[i] < 0, so change < 0. Good!
            
            # Greedy: find the pair (i, j) that gives the most negative change
            # i.e., minimize coeff[j] - coeff[i] = max coeff[i] - min coeff[j] (among appropriate signs)
            # Actually, we can do multiple transfers at once
            
            pos_indices = np.where(coeff > 0.01)[0]  # want to decrease
            neg_indices = np.where(coeff < -0.01)[0]  # want to increase
            zero_indices = np.where(np.abs(coeff) <= 0.01)[0]  # can go either way
            
            if len(pos_indices) > 0 and len(neg_indices) > 0:
                # Compute the improvement for each pair
                # Best: max coeff among pos, min coeff among neg
                best_i = pos_indices[np.argmax(coeff[pos_indices])]
                best_j = neg_indices[np.argmin(coeff[neg_indices])]
                improvement = coeff[best_i] - coeff[best_j]  # positive
                # Transfer mass from best_i to best_j
                amount = min(h_cur[best_i], 1.0 - h_cur[best_j])
                h_cur[best_i] -= amount
                h_cur[best_j] += amount
                h_cur = np.clip(h_cur, 0.0, 1.0)
        
        new_score = compute_c5(h_cur)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # Revert
            h_cur = project_feasible(h, target)  # Hmm, this isn't right
            break
    
    return h_cur, cur_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _slsqp_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _slsqp_grad(x):
    h = np.clip(x, 0.0, 1.0)
    n = _g_n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total

    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m in range(n_lags):
        w = weights[m]
        if w < 1e-10:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                idx1 = np.arange(k, n)
                grad[idx1] += w * (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                grad[idx2] -= w * h[idx2 + k]
        else:
            kp = -k
            idx1 = np.arange(0, n - kp)
            grad[idx1] += w * (1.0 - h[idx1 + kp])
            idx2 = np.arange(kp, n)
            grad[idx2] -= w * h[idx2 - kp]
    return grad


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_analytic(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgradient_momentum(h, n, target, n_steps, lr_init, momentum, k_top, tau, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    v = np.zeros(n)
    lr = lr_init

    for step in range(n_steps):
        if time.time() > deadline:
            break

        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_curr = c_max * dx

        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()

        n_lags = len(corr)
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        diffs = corr[top_idx] - c_max
        weights = np.exp(diffs / tau)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                if k == 0:
                    grad += w * (1.0 - 2.0 * h_cur)
                else:
                    idx1 = np.arange(k, n)
                    grad[idx1] += w * (1.0 - h_cur[idx1 - k])
                    idx2 = np.arange(0, n - k)
                    grad[idx2] -= w * h_cur[idx2 + k]
            else:
                kp = -k
                idx1 = np.arange(0, n - kp)
                grad[idx1] += w * (1.0 - h_cur[idx1 + kp])
                idx2 = np.arange(kp, n)
                grad[idx2] -= w * h_cur[idx2 - kp]
        grad *= dx
        grad = grad - np.mean(grad)

        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)

        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.01, lr_init * 5.0)
        else:
            lr *= 0.92
            v *= 0.5
            if lr < 1e-13:
                break

    return best_h, best_score


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 6
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 6)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.2, target)
            elif r == 4:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.35, 0.05, 0.95), target)
            else:
                blocks = rng.randint(3, 8)
                edges = np.sort(rng.randint(1, n, blocks - 1))
                val = 0.5
                seg = np.zeros(n)
                for bi in range(len(edges) + 1):
                    s = 0 if bi == 0 else edges[bi - 1]
                    e = n if bi == len(edges) else edges[bi]
                    seg[s:e] = val
                    val = 1.0 - val
                pop[i] = project_feasible(seg, target)

    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    idx_range = np.arange(pop_size)
    stagnant = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        gen_best = best_score
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
        if best_score < gen_best - 1e-10:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k = m_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            if k == 0:
                grad = 1.0 - 2.0 * h_cur
            else:
                idx1 = np.arange(k, n)
                grad[idx1] += (1.0 - h_cur[idx1 - k])
                idx2 = np.arange(0, n - k)
                grad[idx2] -= h_cur[idx2 + k]
        else:
            kp = -k
            idx1 = np.arange(0, n - kp)
            grad[idx1] += (1.0 - h_cur[idx1 + kp])
            idx2 = np.arange(kp, n)
            grad[idx2] -= h_cur[idx2 - kp]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.02)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break
    return h_cur, cur_score
```
