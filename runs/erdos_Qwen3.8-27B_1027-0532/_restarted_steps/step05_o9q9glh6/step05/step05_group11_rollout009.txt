def build_overlap_matrix(g, n):
    """Build matrix A where A[k, i] * h[i] gives the i-th term of overlap at lag k."""
    # k ranges from 0 to 2n-2, corresponding to lag k-(n-1)
    A = np.zeros((2*n-1, n))

        lo = max(0, lag)
        hi = min(n, n + lag)
        if hi > lo:

    return A

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        residual = target_sum - np.sum(h)
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += residual / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_overlap coeffs(g, n):
    """Build matrix where row k (corresponding to lag k-(n-1)) dotted with h gives sum h[i]*g[i-lag]."""
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for m in range(n_lags):
        k = m - (n - 1)
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            A[m, lo:hi] = g[lo-k:hi-k]
    return A

def lp_solve_block(h_current, n, target, solver='GLOP'):
    """Solve one LP block: fix g=1-h, optimize h to minimize max overlap."""
    dx = 2.0 / n
    g = 1.0 - h_current
    A = build_overlap_coeffs(g, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        (A @ h_var) * dx <= t_var
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0, 1)

def alternating_lp(h_init, n, target, max_iter=100, time_limit=100, solver='GLOP'):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        h_new = lp_solve_block(h, n, target, solver)
        if h_new is None:
            break
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # Still update to potentially escape
    
    return best_h, best_c5

1]

    stalled = 0

        best_score = current_max
        best_j = -1
        best_dir = 0

            hi_lag = min(j, n - 1 - j)
            if hi_lag < lo_lag:

            lags = np.arange(lo_lag, hi_lag + 1)
