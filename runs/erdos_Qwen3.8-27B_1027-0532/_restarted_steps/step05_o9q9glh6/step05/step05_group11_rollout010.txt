        if time.time() - start > time_limit:
            break

        overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        top_vals = overlaps[top_idx]

        w = np.exp(-0.1 * np.arange(n_top))

                grad[idx2] += w[i] * (-h[idx2 + k])

def all_lags_lp_fast(h_old, n, target):
    """Solve LP with all-lag linearized constraints using scipy linprog."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    
    # Variables: h[0], ..., h[n-1], t
    # Minimize t
    # Subject to:
    #   sum(h) = target
    #   0 <= h[i] <= 1
    #   For each lag k: coeff_k @ h + const_k <= t
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Inequality constraints: coeff_k @ h - t <= -const_k
    # i.e., coeff_k @ h - t + const_k <= 0
    # In linprog form: A_ub @ x <= b_ub
    # So: [coeff_k, -1] @ [h, t] <= -const_k
    
    A_ub_rows = []
    b_ub_vals = []
    
    for k in range(-(n-1), n):
        if k == 0:
            coeff = dx * (1.0 - 2.0 * h_old)
            const = dx * np.sum(h_old ** 2)
        elif k > 0:
            coeff = np.zeros(n)
            idx1 = np.arange(n - k)
            coeff[idx1] += dx * (1.0 - h_old[idx1 + k])
            idx2 = np.arange(k, n)
            coeff[idx2] -= dx * h_old[idx2 - k]
            const = dx * np.sum(h_old[:n-k] * h_old[k:])
        else:
            m = -k
            coeff = np.zeros(n)
            idx1 = np.arange(m, n)
            coeff[idx1] += dx * (1.0 - h_old[idx1 - m])
            idx2 = np.arange(n - m)
            coeff[idx2] -= dx * h_old[idx2 + m]
            const = dx * np.sum(h_old[m:] * h_old[:n-m])
        
        # Constraint: coeff @ h + const <= t  =>  coeff @ h - t <= -const
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(-const)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return np.clip(res.x[:n], 0.0, 1.0)
    return h_old

Let me also think about whether the LP approach will actually converge well. The issue is that the linearization at h_old is only a lower bound on the true bilinear form (or upper bound? let me check).

For L_k(h) = dx * sum h[i](1-h[i+k]):
Linearized at h_old: L_k_lin(h) = dx * sum [(1-h_old[i+k])*h[i] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k]]

Is L_k_lin(h) ≤ L_k(h) for all h? Let's check:
L_k(h) - L_k_lin(h) = dx * sum [h[i]*h[i+k] - (1-h_old[i+k])*h[i] + h_old[i]*h[i+k] - h_old[i]*h_old[i+k]]
= dx * sum [(h[i]-h_old[i])(h[i+k]-h_old[i+k])]

So L_k(h) - L_k_lin(h) = dx * sum Δh[i]*Δh[i+k] where Δh = h - h_old.

This can be positive or negative depending on the correlation of Δh with its shifts. So the linearization is NOT a lower bound in general. This means the LP might give solutions where the true C5 is higher than the LP value.

This is why we need the damping (to stay close to h_old where the linearization is accurate) and the SLSQP polishing (to handle the true non-linear objective).

            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Top-lags sweep LP (fast, builds up from few to all lags)
    if tleft() > 200:
        h1, c5_1 = top_lags_sweep_lp(best_h, n, target,
                                     max_iter=1000, time_limit=min(120, tleft() * 0.35),
                                     k_schedule=[30, 60, 100, 200, None, None, None],
                                     damping=0.65)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h1

    # Phase 2: SLSQP with increasing alpha
    if tleft() > 150:
        h2 = slsqp_optimize(best_h, n, target,
                            [50, 200, 1000, 5000, 20000],
                            maxiter=3000, ftol=1e-15)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Full all-lags LP
    if tleft() > 200:
        h3, c5_3 = alternating_all_lags(best_h, n, target,
                                        max_iter=500, time_limit=min(150, tleft() * 0.4),
                                        damping=0.7)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h3

    # Phase 4: Subgradient polish
    if tleft() > 100:
        h4, c5_4 = subgradient_hedge(best_h, n, target,
                                     max_steps=50000, time_limit=min(80, tleft() * 0.3),
                                     n_top=50, momentum=0.9, lr_init=0.15)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4

    # Phase 5: SLSQP with very high alpha
    if tleft() > 100:
        h5 = slsqp_optimize(best_h, n, target,
                            [2000, 10000, 50000, 200000],
                            maxiter=2000, ftol=1e-15)
        sc5 = compute_c5(h5)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Random restarts with all-lags LP
    for attempt in range(10):
        if tleft() < 50:
            break
        noise_scale = 0.04 + attempt * 0.025
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h6, c5_6 = alternating_all_lags(h_pert, n, target,
                                         max_iter=150, time_limit=25,
                                         damping=0.6)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6
        elif tleft() > 40:
            h6b = slsqp_optimize(h_pert, n, target,
                                 [100, 1000, 5000],
                                 maxiter=1000, ftol=1e-14)
            sc6b = compute_c5(h6b)
            if sc6b < best_score:
                best_score = sc6b
                best_h = h6b

    # Phase 7: Final all-lags LP polish
    if tleft() > 50:
        h7, c5_7 = alternating_all_lags(best_h, n, target,
                                         max_iter=300, time_limit=min(40, tleft() * 0.7),
                                         damping=0.85)
        if c5_7 < best_score:
            best_score = c5_7
            best_h = h7

    # Phase 8: Final SLSQP
    if tleft() > 40:
        h8 = slsqp_optimize(best_h, n, target,
                            [5000, 20000, 100000],
                            maxiter=1500, ftol=1e-15)
        sc8 = compute_c5(h8)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
