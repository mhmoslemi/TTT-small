```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, alpha):
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    logsum = float(np.log(np.sum(np.exp(shifted))))
    return cmax + logsum / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)  # gradient w.r.t. correlation values
    # Now compute gradient w.r.t. h
    g_h = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m]
        if wk < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            # c(k) = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            # dc/dh[j] = (1-h[j+k]) if j <= n-1-k, minus h[j-k] if j >= k
            idx1 = np.arange(0, n - k)
            g_h[idx1] += wk * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            g_h[idx2] -= wk * h[idx2 - k]
        else:
            ell = -k
            # c(-ell) = sum_{i=ell}^{n-1} h[i](1-h[i-ell])
            # dc/dh[j] = (1-h[j-ell]) if j >= ell, minus h[j+ell] if j <= n-1-ell

            g_h[idx1] += wk * (1.0 - h[idx1 - ell])
            idx2 = np.arange(0, n - ell)
            g_h[idx2] -= wk * h[idx2 + ell]
    g_h *= dx
```

```python
def adam_optimize(h_init, n, target, alpha, max_iters, deadline, lr=0.01, beta1=0.9, beta2=0.999):
    h = h_init.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for i in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        g = lse_grad(h, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, target)
        h = h_new
        if i % 50 == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_val(h, alpha):
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m]
        if wk < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lbfgsb_lse(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_val(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_lse(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lse_val, h, args=(alpha,), method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def nesterov_gs(h_init, n, target, alpha, max_iters, deadline, lr_init=0.003, momentum=0.9):
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    prev_obj = lse_val(h, alpha)
    
    for i in range(max_iters):
        if time.time() > deadline:
            break
        g = lse_grad(h, alpha)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, target)
        obj_new = lse_val(h_new, alpha)
        if obj_new <= prev_obj + 1e-13:
            h = h_new
            prev_obj = obj_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.7
            v *= 0.3
        if i % 30 == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    
    sc = compute_c5(h)
    if sc < best_score:
        best_score = sc
        best_h = h.copy()
    return best_h, best_score


def subgradient_opt(h_init, n, target, n_steps, temp, lr_init, deadline):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        c_curr = float(np.max(corr))
        if c_curr * dx < best_score:
            best_score = c_curr * dx
            best_h = h.copy()
        top_k = min(10, len(corr))
        top_idx = np.argsort(corr)[::-1][:top_k]
        vals = corr[top_idx]
        weights = np.exp((vals - vals[0]) / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h[k:])
                g[k:] -= wk * h[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h[:n - ell])
                g[:n - ell] -= wk * h[ell:]
        g *= dx
        h_new = h - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score * 0.9999 + 1e-10:  # accept if not too bad
            h = h_new
        elif sc_new < c_curr * dx:
            h = h_new
        else:
            lr *= 0.95
    
    return best_h, best_score


def surgery(h_init, n, target, max_steps, eps_init, deadline):
    h = h_init.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        g = np.zeros(n)
        k = m_star - (n - 1)
        if k >= 0:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        else:
            ell = -k
            g[ell:] += (1.0 - h[:n - ell])
            g[:n - ell] -= h[ell:]
        g *= dx
        h_new = h - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.02)
        else:
            eps *= 0.9
            if eps < 1e-10:
                break
    return h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_val(h, alpha):
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m]
        if wk < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def _obj_lse_pen(x, alpha, mu, target):
    return lse_val(x, alpha) + mu * (float(np.sum(x)) - target) ** 2

    s = float(np.sum(x))
    return lse_grad(x, alpha) + 2.0 * mu * (s - target)

    return lse_val(x, alpha)


def _grad_lse(x, alpha):

def _cons_fun(x, target):

def _cons_jac(x, target):

    except Exception:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h2 = h2

    # Phase 4: L-BFGS-B refinement
    if time.time() < deadline - 400:
        for alpha in [5000, 20000, 60000]:
            if time.time() > deadline - 380:
                break
            h2 = lbfgsb_round(h2, n2, target2, alpha, 200.0, 200)
            sc = compute_c5(h2)
            if sc < best_score2:
                best_score2 = sc
                best_h2 = h2

    # Phase 5: SLSQP rounds
    if time.time() < deadline - 300:
        for alpha in [8000, 30000, 100000]:
            if time.time() > deadline - 280:
                break
            h2 = slsqp_round(h2, n2, target2, alpha, 2000, 1e-15)
            sc = compute_c5(h2)
            if sc < best_score2:
                best_score2 = sc
                best_h2 = h2

    if best_score2 < best_score:
        best_score = best_score2
        best_h = best_h2

    # Phase 6: Subgradient with multiple temperatures
    if time.time() < deadline - 220:
        for temp in [0.02, 0.008, 0.003, 0.001]:
            if time.time() > deadline - 200:
                break
            h2, sc2 = subgradient_opt(best_h2, n2, target2, 3000, temp, 0.004,
                                       deadline=time.time() + 40)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2
                best_score = best_score2
                best_h = best_h2

    # Phase 7: Surgery on worst lags
    if time.time() < deadline - 140:
        h2, sc2 = surgery_opt(best_h2, n2, target2, 600, 0.004, deadline=time.time() + 50)
        if sc2 < best_score2:
            best_score2 = sc2
            best_h2 = h2
            best_score = best_score2
            best_h = best_h2

    # Phase 8: Final Nesterov with very high alpha
    if time.time() < deadline - 80:
        h2, sc2 = nesterov_opt(best_h2, n2, target2, 80000, max_iters=50000,
                                deadline=time.time() + 45, lr_init=0.001, momentum=0.95)
        if sc2 < best_score2:
            best_score2 = sc2
            best_h2 = h2
            best_score = best_score2
            best_h = best_h2

    # Phase 9: SLSQP final polish
    if time.time() < deadline - 35:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_round(best_h, n_cur, target_cur, 100000, 1500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 10: Try n=1200 with Nesterov
    if time.time() < deadline - 250:
        n3 = 1200
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)
        for alpha in [100, 500, 2000, 8000, 30000]:
            if time.time() > deadline - 100:
                break
            h3, sc3 = nesterov_opt(h3, n3, target3, alpha, max_iters=50000,
                                    deadline=time.time() + 35, lr_init=0.002, momentum=0.93)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3
        if time.time() < deadline - 80:
            h3 = slsqp_round(h3, n3, target3, 20000, 1500, 1e-15)
            sc3 = compute_c5(h3)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3
        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 11: Random perturbations at best resolution
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(3):
            if time.time() > deadline - 30:
                break
            scale = 0.06 + 0.03 * attempt
            h_pert = best_h + rng.randn(n_cur) * scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = nesterov_opt(h_pert, n_cur, target_cur, 5000,
                                          max_iters=30000, deadline=time.time() + 15,
                                          lr_init=0.003, momentum=0.92)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 12: Final surgery + Nesterov
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h2, sc2 = surgery_opt(best_h, n_cur, target_cur, 300, 0.002, deadline=time.time() + 10)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n_final
```
