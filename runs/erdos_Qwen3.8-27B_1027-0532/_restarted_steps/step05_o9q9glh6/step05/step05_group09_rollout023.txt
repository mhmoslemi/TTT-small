import numpy as np
import time

def compute_c5_full(h):
    """Compute true C5 = max over all lags of correlate(h, 1-h, 'full') * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    """Project to [0,1]^n with sum = target."""
    h = np.clip(h, 0.0, 1.0).astype(float)
    diff = h.sum() - target
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad_full(h, alpha):
    """Compute log-sum-exp smoothed objective and gradient for full C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Compute gradient: sum_m weights[m] * dI[k_m]/dh
    # where k_m = n-1-m (lag for full array index m)
    # dI(k)/dh[i] = (1-h[i+k])*1[i+k<n] - h[i-k]*1[i-k>=0]
    grad = np.zeros(n)
    for m in range(len(weights)):
        if weights[m] < 1e-10:
            continue
        k = n - 1 - m  # lag
        w = weights[m]
        if k >= 0:
            # (1-h[i+k]) for i=0..n-1-k
            grad[:n-k] += w * (1.0 - h[k:])
            # -h[i-k] for i=k..n-1
            grad[k:] -= w * h[:-k]
        else:
            mk = -k  # positive
            # (1-h[i-k]) for i=mk..n-1 (i.e., i+mk >= mk, and i-k = i+mk)
            # Wait, k < 0, so i+k = i-mk
            # (1-h[i+k]) = (1-h[i-mk]) for i >= mk (i-mk >= 0) and i-mk < n (always true since i < n)
            # So: (1-h[i-mk]) for i=mk..n-1
            grad[mk:] += w * (1.0 - h[:n-mk])
            # -h[i-k] = -h[i+mk] for i+mk < n, i.e., i < n-mk
            grad[:n-mk] -= w * h[mk:]
    
    grad *= dx
    return obj, grad

def compute_c5_full(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    diff = h.sum() - target
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(exp_vals.sum())
    weights = exp_vals / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 1e-8
    active = np.where(weights > threshold)[0]

        if k >= 0:

        else:
            mk = -k
            grad[mk:] += w * (1.0 - h[:n-mk])
            grad[:n-mk] -= w * h[mk:]
    
    grad *= dx
    return obj, grad

import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(exp_vals.sum())
    weights = exp_vals / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = weights[m]
        if wm < 1e-10:
            continue
        k = n - 1 - m
        if k >= 0:
            nk = n - k
            grad[:nk] += wm * (1.0 - h[k:n])
            if k > 0:
                grad[k:n] -= wm * h[0:nk]
        else:
            mk = -k
            grad[mk:n] += wm * (1.0 - h[0:n-mk])
            grad[0:n-mk] -= wm * h[mk:n]
    grad *= dx
    return obj, grad

def fista_opt(h0, target, alpha, deadline, max_iters=100000):
    n = len(h0)
    x = project_feasible(h0, target)
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5_full(x)
    dx = 2.0 / n
    L = 1.0 / dx
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_grad(y, alpha)

                break

        x = x_try
        t = t_new

def subgrad_active(h0, target, n_active, deadline, lr_init=0.5):

        top_idx = np.argsort(corr)[::-1][:n_active]

        diffs = np.clip(corr[top_idx] - c_top, -100.0, 0.0)
        w = np.exp(diffs / 0.01)
        w /= w.sum()

        for wi, m in zip(w, top_idx):

                nk = n - k
                grad[:nk] += wi * (1.0 - x[k:n])

                mk = -k
                grad[mk:n] += wi * (1.0 - x[0:n-mk])
                grad[0:n-mk] -= wi * x[mk:n]

        improved = False
        for _ in range(5):
            x_new = project_feasible(x - step * grad, target)

def coord_descend(h0, target, deadline, max_sweeps=200):
    n = len(h0)
    x = project_feasible(h0.copy(), target)
    best_score = compute_c5_full(x)
    rng = np.random.RandomState(42)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        any_improve = False
        for i in order:
            if time.time() > deadline:
                break
            old = x[i]
            found = False
            for trial in [0.0, 1.0, 0.5, old + 0.15, old - 0.15, old + 0.05, old - 0.05]:
                trial = float(np.clip(trial, 0.0, 1.0))
                if abs(trial - old) < 1e-12:
                    continue
                x[i] = trial
                x = project_feasible(x, target)
                sc = compute_c5_full(x)
                if sc < best_score - 1e-13:
                    best_score = sc
                    found = True
                    any_improve = True
                    break
            if not found:
                x[i] = old
                x = project_feasible(x, target)
        if not any_improve:
            break
    return x, best_score
