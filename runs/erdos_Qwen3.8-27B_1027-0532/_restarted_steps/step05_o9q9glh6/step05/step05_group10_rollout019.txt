    # Where 1_valid is 1 for i in [max(0,k), min(n, n+k)) and 0 otherwise
    
    # This is O(n * num_k) in the worst case. For n=400 and ~20 significant k's, that's 8000 ops per k, very fast.
    
    significant = np.where(weights > 1e-8)[0]
    for k_idx in significant:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, n + k)
        if hi <= lo:
            continue
        w = weights[k_idx]
        if w < 1e-12:
            continue
        # g[i] = dx for i in [lo, hi)
        # g[i] -= h[i-k] * dx if 0 <= i-k < n
        # g[i] -= h[i+k] * dx if 0 <= i+k < n
        g = np.full(hi - lo, dx)
        # h[i-k] for i in [lo, hi): i-k in [lo-k, hi-1-k]
        lo_ik = lo - k
        hi_ik = hi - k
        valid_ik = (lo_ik >= 0) & (hi_ik <= n)
        if lo_ik >= 0 and hi_ik <= n:
            g -= h[lo_ik:hi_ik] * dx
        elif lo_ik < 0:
            g[-lo_ik:] -= h[0:hi_ik+lo_ik] * dx
        elif hi_ik > n:
            g[:n-lo_ik] -= h[lo_ik:n] * dx
        
        # h[i+k] for i in [lo, hi): i+k in [lo+k, hi-1+k]
        lo_ipk = lo + k
        hi_ipk = hi + k
        if lo_ipk >= 0 and hi_ipk <= n:
            g -= h[lo_ipk:hi_ipk] * dx
        elif lo_ipk < 0:
            g[-lo_ipk:] -= h[0:hi_ipk+lo_ipk] * dx
        elif hi_ipk > n:
            g[:n-lo_ipk] -= h[lo_ipk:n] * dx
        
        grad[lo:hi] += w * g
    
    return obj, grad


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    cur_score = compute_c5(h_cur)

        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})

            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:

                cur_score = sc
        except Exception:

    return h_cur

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        if c5_val >= best_c5 - 1e-12:
            damping = max(0.3, damping * 0.96)
        else:
            damping = min(0.95, damping + 0.04)

        if it % 40 == 0 and current_reg > reg_floor:
            current_reg = max(reg_floor, current_reg * 0.65)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_idx_star = int(np.argmax(corr))
        k = k_idx_star - half_n

        lo = max(0, k)
        hi = min(n, n + k)
        grad = np.zeros(n)
        if hi > lo:
            g = np.full(hi - lo, dx)
            lo_ik = lo - k
            hi_ik = hi - k
            if lo_ik >= 0 and hi_ik <= n:
                g -= h_cur[lo_ik:hi_ik] * dx
            elif lo_ik < 0:
                g[-lo_ik:] -= h_cur[0:hi_ik + lo_ik] * dx
            elif hi_ik > n:
                g[:n - lo_ik] -= h_cur[lo_ik:n] * dx

            lo_ipk = lo + k
            hi_ipk = hi + k
            if lo_ipk >= 0 and hi_ipk <= n:
                g -= h_cur[lo_ipk:hi_ipk] * dx
            elif lo_ipk < 0:
                g[-lo_ipk:] -= h_cur[0:hi_ipk + lo_ipk] * dx
            elif hi_ipk > n:
                g[:n - lo_ipk] -= h_cur[lo_ipk:n] * dx

            grad[lo:hi] = g

        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5, 1.0]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 200:
            if step > 1000:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    if tleft() > 150:
        lp_time = min(300, tleft() * 0.35)
        h_lp, c5_lp = alternating_lp_full(h_init, n, target,
                                          max_iter=50000, time_limit=lp_time,
                                          damping_init=0.5, reg_init=1e-3)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 80:
        h_refined = slsqp_round(best_h, n, target,
                                [100, 500, 2000], 3000, 1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    if tleft() > 120:
        lp_time = min(200, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                          max_iter=50000, time_limit=lp_time,
                                          damping_init=0.7, reg_init=1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 50:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                           max_steps=10000, time_limit=min(50, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 80:
        h_refined = slsqp_round(best_h, n, target,
                                [2000, 10000, 30000], 2000, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    if tleft() > 250:
        n_try = n * 2
        if n_try % 2 == 0 and n_try <= 1600:
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(150, tleft() * 0.4)
            h_lp, c5_lp = alternating_lp_full(h_interp, n_try, target_try,
                                              max_iter=30000, time_limit=lp_time,
                                              damping_init=0.4, reg_init=1e-4)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try

            if tleft() > 60:
                h_refined = slsqp_round(best_h, n, target,
                                        [200, 1000, 5000], 1500, 1e-14)
                sc = compute_c5(h_refined)
                if sc < best_score:
                    best_score = sc
                    best_h = h_refined

    if tleft() > 120:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(150, tleft() * 0.4)
        h_lp, c5_lp = alternating_lp_full(best_h, n_use, target_use,
                                          max_iter=50000, time_limit=lp_time,
                                          damping_init=0.85, reg_init=1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    for noise in [0.05, 0.1, 0.15, 0.25]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        if tleft() > 50:
            lp_time = min(40, tleft() * 0.5)
            h_lp, c5_lp = alternating_lp_full(h_pert, n, target,
                                              max_iter=10000, time_limit=lp_time,
                                              damping_init=0.5, reg_init=1e-3)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = slsqp_round(best_h, n_use, target_use,
                                  [5000, 20000, 50000], 1000, 1e-16)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                           max_steps=5000, time_limit=min(20, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
