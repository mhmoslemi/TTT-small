    except Exception:
        return h


def subgradient_polish_nesterov(h, n, target, n_steps, top_k, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    g_sq_sum = np.zeros(n)
    rho = 0.9
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * (2.0 / n)
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        dx = 2.0 / n
        for m in top_idx:
            m = int(m)
            k = m - (n - 1)
            if k == 0:
                g[:] += (1.0 - 2.0 * h_cur)
            elif k > 0:
                g[:n - k] += (1.0 - h_cur[k:])
                g[k:] -= h_cur[:n - k]
            else:
                ell = -k
                g[:n - ell] -= h_cur[ell:]
                g[ell:] += (1.0 - h_cur[:n - ell])
        g *= dx
        g_sq_sum += g ** 2
        lr_adaptive = lr / np.sqrt(1.0 + g_sq_sum)
        v = rho * v + (1.0 - rho) * g
        v_adaptive = v / np.sqrt(1.0 + g_sq_sum)
        h_new = h_cur - lr * v_adaptive
        h_new = project_bisect(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def active_set_slsqp(h, n, target, n_active, alpha, maxiter):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for outer in range(10):
        if time.time() > deadline_global:
            break
        corr = compute_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:n_active]
        corr_active = corr[top_m]
        c_act_max = float(np.max(corr_active))

        def obj_active(x, alpha, mu, target, top_m, c_act_max):
            c = compute_corr(x)
            c_a = c[top_m]
            am = float(np.max(c_a))
            shifted = np.clip(alpha * (c_a - am), -700.0, 0.0)
            return am + float(np.log(np.sum(np.exp(shifted)))) / alpha + 0.01 * float(np.max(c))

        def grad_active(x, alpha, mu, target, top_m, c_act_max):
            c = compute_corr(x)
            c_a = c[top_m]
            am = float(np.max(c_a))
            shifted = np.clip(alpha * (c_a - am), -700.0, 0.0)
            w_a = np.exp(shifted)
            w_a /= w_a.sum()
            dx = 2.0 / n
            g = np.zeros(n)
            for wk, m in zip(w_a, top_m):
                m = int(m)
                k = m - (n - 1)
                if k == 0:
                    g[:] += wk * (1.0 - 2.0 * x)
                elif k > 0:
                    g[:n - k] += wk * (1.0 - x[k:])
                    g[k:] -= wk * x[:n - k]
                else:
                    ell = -k
                    g[:n - ell] -= wk * x[ell:]
                    g[ell:] += wk * (1.0 - x[:n - ell])
            g *= dx
            return g

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac, 'args': (target,)}]
        try:
            res = minimize(obj_active, h_cur, method='SLSQP', jac=grad_active,
                           args=(alpha, 0.0, target, top_m, c_act_max),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_cand = project_bisect(res.x, target)
            sc = compute_c5(h_cand)
            if sc < best_score:
                best_score = sc
                h_cur = h_cand
        except Exception:
            break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_bisect(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_new = np.interp(x_new, x_old, h_src)
    return project_bisect(h_new, target_dst)


def fourier_perturb(h, n, strength, rng):
    h_f = np.fft.rfft(h - 0.5)
    n_freq = len(h_f)
    for i in range(1, n_freq):
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength * rng.uniform(0.5, 1.5) / (1.0 + i * 0.1)
        h_f[i] += amp * np.exp(1j * phase)
        if i < n_freq - 1:
            h_f[n_freq - 1 - i] = h_f[i].conjugate()
    h_pert = np.fft.irfft(h_f, n)
    return project_bisect(h_pert, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    global deadline_global
    start_time = time.time()
    deadline_global = start_time + budget_s - 18.0
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 400
        h = np.full(n, 0.5)

    n = len(h)
    target = n / 2.0
    h = project_bisect(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def time_left():
        return deadline_global - time.time()

    # Phase 1: Multi-scale coarse-to-fine
    if time_left() > 500:
        n_coarse = 100
        h_c = interpolate_to(best_h, n_coarse, n_coarse / 2.0)
        for alpha in [30, 100, 400]:
            if time_left() < 400:
                break
            h_c = lbfgsb_optimize(h_c, n_coarse, n_coarse / 2.0, alpha, 200.0, 300)
        h_up = interpolate_to(h_c, n, target)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up

    # Phase 2: L-BFGS-B continuation (fine-grained alpha schedule)
    alphas = [20, 35, 55, 90, 150, 250, 450, 800, 1500, 3000, 6000, 12000, 25000]
    for alpha in alphas:
        if time_left() < 300:
            break
        for mu in [30.0, 150.0]:
            if time_left() < 250:
                break
            h_new = lbfgsb_optimize(best_h, n, target, alpha, mu, 400)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new

    # Phase 3: SLSQP refinement
    for alpha in [500, 2000, 8000, 20000, 50000]:
        if time_left() < 150:
            break
        h_new = slsqp_optimize(best_h, n, target, alpha, 5000)
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new

    # Phase 4: Active-set SLSQP
    if time_left() > 200:
        for n_act in [3, 5, 8]:
            if time_left() < 150:
                break
            h_act, sc_act = active_set_slsqp(best_h, n, target, n_act, 3000, 3000)
            if sc_act < best_c5:
                best_c5 = sc_act
                best_h = h_act

    # Phase 5: Nesterov subgradient polish
    if time_left() > 150:
        h_sg, sc_sg = subgradient_polish_nesterov(best_h, n, target, 3000, 5, 0.008, 0.998, deadline_global - 100)
        if sc_sg < best_c5:
            best_c5 = sc_sg
            best_h = h_sg

    # Phase 6: Higher resolution n=600
    if time_left() > 350:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_c5_2 = compute_c5(h2)
        for alpha in [25, 50, 100, 250, 600, 1500, 4000, 10000]:
            if time_left() < 120:
                break
            for mu in [30.0, 150.0]:
                if time_left() < 100:
                    break
                h2_new = lbfgsb_optimize(best_h2, n2, target2, alpha, mu, 250)
                sc2 = compute_c5(h2_new)
                if sc2 < best_c5_2:
                    best_c5_2 = sc2
                    best_h2 = h2_new
        if time_left() > 100:
            h2_new = slsqp_optimize(best_h2, n2, target2, 5000, 3000)
            sc2 = compute_c5(h2_new)
            if sc2 < best_c5_2:
                best_c5_2 = sc2
                best_h2 = h2_new
        if time_left() > 80:
            h_sg2, sc_sg2 = subgradient_polish_nesterov(best_h2, n2, target2, 1500, 5, 0.005, 0.998, deadline_global - 60)
            if sc_sg2 < best_c5_2:
                best_c5_2 = sc_sg2
                best_h2 = h_sg2
        if best_c5_2 < best_c5:
            best_c5 = best_c5_2
            best_h = best_h2

    # Phase 7: Try n=800 if time permits
    if time_left() > 300:
        n3 = 800
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_c5_3 = compute_c5(h3)
        for alpha in [20, 50, 150, 400, 1000, 3000, 8000]:
            if time_left() < 100:
                break
            for mu in [50.0, 200.0]:
                if time_left() < 80:
                    break
                h3_new = lbfgsb_optimize(best_h3, n3, target3, alpha, mu, 200)
                sc3 = compute_c5(h3_new)
                if sc3 < best_c5_3:
                    best_c5_3 = sc3
                    best_h3 = h3_new
        if time_left() > 60:
            h3_new = slsqp_optimize(best_h3, n3, target3, 5000, 2000)
            sc3 = compute_c5(h3_new)
            if sc3 < best_c5_3:
                best_c5_3 = sc3
                best_h3 = h3_new
        if best_c5_3 < best_c5:
            best_c5 = best_c5_3
            best_h = best_h3

    # Phase 8: Fourier-based perturbations
    if time_left() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(5):
            if time_left() < 60:
                break
            strength = 0.15 * (1.0 + 0.3 * attempt)
            h_pert = fourier_perturb(best_h, n_cur, strength, rng)
            h_new = lbfgsb_optimize(h_pert, n_cur, target_cur, 500, 100.0, 300)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new
                if time_left() > 80:
                    h_ref = slsqp_optimize(best_h, n_cur, target_cur, 5000, 2000)
                    sc_ref = compute_c5(h_ref)
                    if sc_ref < best_c5:
                        best_c5 = sc_ref
                        best_h = h_ref

    # Phase 9: Final subgradient + SLSQP polish
    if time_left() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_polish_nesterov(best_h, n_cur, target_cur, 1000, 3, 0.003, 0.999, deadline_global - 30)
        if sc_sg < best_c5:
            best_c5 = sc_sg
            best_h = h_sg
        if time_left() > 20:
            h_new = slsqp_optimize(best_h, n_cur, target_cur, 20000, 2000)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new

    # Phase 10: Ultimate final polish
    if time_left() > 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_polish_nesterov(best_h, n_cur, target_cur, 500, 2, 0.001, 0.999, deadline_global - 5)
        if sc_sg < best_c5:
            best_c5 = sc_sg
            best_h = h_sg

    final_h = project_bisect(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
