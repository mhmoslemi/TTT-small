def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):

    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]

        grad[valid_j2] -= h[valid_j2 - k_star]
    else:
        # k_star < 0, let m = -k_star
        m = -k_star
        # j+k_star = j-m valid means j >= m
        valid_j = np.arange(m, n)
        grad[valid_j] += (1 - h[valid_j - m])
        # j-k_star = j+m valid means j < n - m
        valid_j2 = np.arange(n - m)
        grad[valid_j2] -= h[valid_j2 + m]
    
    # Projected subgradient step
    h_new = h - step_size * grad
    return project_feasible(h_new, target), k_star

Wait, I need to double-check the gradient computation. Let me be very careful.

g_k(h) = Σ_{i ∈ S_k} h[i] · (1 - h[i+k])

where S_k = {i : 0 ≤ i < n, 0 ≤ i+k < n}

∂g_k/∂h[j] = Σ_{i ∈ S_k} [δ_{i,j} · (1 - h[i+k]) + h[i] · (-δ_{i+k,j})]
= (1 - h[j+k]) · 1[j ∈ S_k, j+k valid] + Σ_{i ∈ S_k, i+k=j} (-h[i])
= (1 - h[j+k]) · 1[0 ≤ j < n, 0 ≤ j+k < n] - h[j-k] · 1[0 ≤ j-k < n, 0 ≤ j < n, j-k+k valid]

Wait, the second term: i+k = j means i = j-k. We need i ∈ S_k, so 0 ≤ j-k < n and 0 ≤ j < n. Also j must be in [0, n-1].

So: ∂g_k/∂h[j] = (1 - h[j+k]) · 1[0 ≤ j+k < n] - h[j-k] · 1[0 ≤ j-k < n]

This matches what I had. Good.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def subgradient_search(h_start, target, time_limit, start_time, n_inner=200000):
    n = len(h)
    h_cur = h_start.copy().astype(float)
    h_cur = project_feasible(h_cur, target)
    full = full_corr_fast(h_cur)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h_cur.copy()
    positions = np.arange(n)
    stall = 0
    max_stall = 20000
    step_init = 0.1
    step_min = 0.0001

    for it in range(n_inner):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        k_star = int(np.argmax(full)) - (n - 1)
        if k_star == 0:
            break

        grad = np.zeros(n)
        if k_star >= 0:
            valid_j = np.arange(n - k_star)
            grad[valid_j] += (1.0 - h_cur[valid_j + k_star])
            valid_j2 = np.arange(k_star, n)
            grad[valid_j2] -= h_cur[valid_j2 - k_star]
        else:
            m = -k_star
            valid_j = np.arange(m, n)
            grad[valid_j] += (1.0 - h_cur[valid_j - m])
            valid_j2 = np.arange(n - m)
            grad[valid_j2] -= h_cur[valid_j2 + m]

        step = step_init * max(step_min / step_init, 0.9999 ** it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)

        full_new = full_corr_fast(h_new)
        c5_new = float(full_new.max()) * dx

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            full = full_new
            stall = 0
        else:
            h_cur = h_cur * 0.5 + h_new * 0.5
            h_cur = project_feasible(h_cur, target)
            full = full_corr_fast(h_cur)
            stall += 1

    return best_h, best_c5

def smooth_obj(h, alpha, target):
    corr = full_corr_fast(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    penalty = 100.0 * (np.sum(h) - target) ** 2
    return lse + penalty

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    h_cur = h_start.copy().astype(float)
    h_cur = project_feasible(h_cur, target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def slsqp_lse(h_start, n, target, time_limit, start_time):
