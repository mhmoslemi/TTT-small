    mask = (idx >= 0) & (idx < n)
    delta[mask] += (1.0 - h[idx[mask]])
    # + term in (1-h[...]): h[l]*(1-h[l+k]) where l+k=j, l=j-k: h[l] goes from h[l]*1 to h[l]*0
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    
    return delta

def multi_swap_eval(h, positions_a, positions_b, n):
    """Evaluate the new max correlation after swapping positions_a (1->0) and positions_b (0->1)."""
    h_new = h.copy()
    h_new[positions_a] = 0.0
    h_new[positions_b] = 1.0
    corr_new = full_corr(h_new)
    return float(corr_new.max())

def adam_continuous(h_start, n, target, time_limit, start_time, 
                    lr=0.1, alpha_start=10, alpha_end=5000, n_phases=5):
    """ADAM optimizer for the continuous relaxation with increasing alpha."""
    h = h_start.astype(float).copy()
    h = np.clip(h, 0.01, 0.99)
    h = project_feasible(h, float(target))
    
    dx = 2.0 / n
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    total_time = time_limit
    n_iters = int(time_limit * 500)  # estimate
    iters_per_phase = max(100, n_iters // n_phases)
    
    for phase in range(n_phases):
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        
        phase_start = time.time()
        phase_time = total_time / n_phases
        
        alpha = alpha_start * (alpha_end / alpha_start) ** (phase / max(1, n_phases-1))
        
        for it in range(iters_per_phase):
            if time.time() - start_time > time_limit - 5:
                return best_h, best_c5
            if time.time() - phase_start > phase_time:
                break
            t += 1
            
            corr = full_corr(h)
            max_c = float(corr.max())
            
            # Smooth max weights
            scaled = alpha * dx * (corr - max_c)
            scaled = np.clip(scaled, -50, 0)
            exp_c = np.exp(scaled)
            p = exp_c / exp_c.sum()
            
            # Gradient
            f_arr = 1.0 - h
            t1 = np.correlate(p, f_arr, mode='full')[n-1:2*n-1][::-1]
            t2 = np.convolve(h, p, mode='full')[n-1:2*n-1]
            grad = (t1 - t2) * dx
            
            # ADAM update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h_new, float(target))
            
            # Check improvement
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            h = h_new
    
    return best_h, best_c5
