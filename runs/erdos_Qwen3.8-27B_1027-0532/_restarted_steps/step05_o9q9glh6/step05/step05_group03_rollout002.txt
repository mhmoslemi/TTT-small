def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_slsqp(x):
    h = project_feasible(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[_g_n - 1:]

    """n_periods-periodic bang-bang on [0,2].
    Period = 2/n_periods. Active set in each period: [0,a] ∪ [a+g, a+g+1/n_periods-a].

    period = 2.0 / n_periods
    x = np.arange(n) * 2.0 / n
    x_adj = x % period
    h = ((x_adj < a) | ((x_adj >= a + g) & (x_adj < a + g + b))).astype(float)
    return h

def optimize_parametric(n, n_periods, n_grid=25):
    """Grid search + Nelder-Mead for n_periods-periodic family."""
    b_max = 1.0 / n_periods
    best_score = float('inf')
    best_params = None
    
    # Grid search
    a_vals = np.linspace(0.001, b_max - 0.001, n_grid)
    g_vals = np.linspace(0.0, b_max - 0.001, n_grid)
    
    for a in a_vals:
        for g in g_vals:
            if a + g + (b_max - a) > 2.0/n_periods + 1e-9:
                continue
            h = build_nperiodic(n, n_periods, a, g)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_params = (a, g)

    if best_params is not None:
        def obj_pm(params):
            a, g = params
            a = np.clip(a, 0.0, b_max)
            g = np.clip(g, 0.0, b_max)
            if a + g > 2.0/n_periods:
                return 1.0
            h = build_nperiodic(n, n_periods, a, g)
            return compute_c5(h)

            best_params = res.x

def subgradient_step(h, n_active, lr, momentum, temp, velocity):
    """One subgradient step with momentum on h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:]
    top_idx = np.argsort(over)[::-1][:n_active]
    diffs = over[top_idx] - over[top_idx[0]]
    weights = np.exp(diffs / temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    for wk, k in zip(weights, top_idx):
        k = int(k)
        if k == 0:
            grad[:] += wk * dx * (1.0 - 2.0 * h)
        else:
            upper = n - 1 - k
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += wk * dx * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= wk * dx * h[idx2 - k]
    velocity = momentum * velocity + (1.0 - momentum) * grad
    return h - lr * velocity, velocity

    lr = lr_init
