def _qp_matvec(x):
    # Compute Q @ x where Q = sum_k lambda_k * S_k
    # S_k @ x: (S_k x)[i] = 0.5*(x[i+k] + x[i-k]) for valid indices
    n = len(x)
    result = np.zeros(n)
    for k, lam in _g_lag_weights:
        k = int(k)
        if k >= n:
            continue
        # Superdiagonal: result[i] += 0.5 * lam * x[i+k] for i = 0..n-1-k
        result[:n-k] += 0.5 * lam * x[k:]
        # Subdiagonal: result[i] += 0.5 * lam * x[i-k] for i = k..n-1
        result[k:] += 0.5 * lam * x[:-k]
    return result

_g_target = 0.0
_g_alpha = 0.0
_g_Q = None
_g_lag_weights = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_slsqp(x):
    h_p = project_feasible(x, _g_target)
    corr = np.correlate(h_p, 1.0 - h_p, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _qp_obj(x):
    return float(x @ _g_Q @ x)

def _qp_grad(x):
    return 2.0 * (_g_Q @ x)

def build_Q_from_lags(lags, lambdas, n):
    Q = np.zeros((n, n))
    for k, lam in zip(lags, lambdas):
        k = int(k)
        if k <= 0 or k >= n or lam <= 1e-15:
            continue
        for i in range(n - k):
            Q[i, i + k] += 0.5 * lam
            Q[i + k, i] += 0.5 * lam
    return Q

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _cons_eq}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(_obj_slsqp, h_cur, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def dual_qp_one_step(h, n_lags, reg):
    global _g_Q
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr[n - 1:]
    top_idx = np.argsort(f)[::-1][:n_lags]
    top_val = f[top_idx]
    span = max(top_val[0] - top_val[-1], 1e-10)
    weights = np.exp(-(top_val - top_val[0]) / span)
    weights += 1e-8
    weights /= weights.sum()
    Q = build_Q_from_lags(top_idx, weights, n)
    if reg > 0:
        Q = Q - reg * np.eye(n)
    _g_Q = -Q
    try:

                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 300, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
    except Exception:
        h_new = h.copy()
    return h_new

def dual_iterations(h, n_outer, n_lags, reg, mix):

def targeted_lag_reduction(h, n_steps, lr_init, n_top):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n - 1:]
        top_lags = np.argsort(over)[::-1][:n_top]
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k == 0:

                grad[idx2] -= dx * h_cur[idx2 - k]
