**Key insight I haven't exploited**: The problem has a special structure. The objective depends on h only through its autocorrelation. This means that different h's with the same autocorrelation have the same objective. This is a huge symmetry that we can exploit!

Specifically, if we think of h in the Fourier domain: H_k = FFT(h). Then the autocorrelation is IFFT(|H_k|²). So the objective depends only on |H_k|² (the power spectrum), not on the phases of H_k!

1. Optimize the power spectrum |H_k|² directly (a lower-dimensional problem in some sense).
2. Use phase retrieval ideas to reconstruct h from a desired spectrum.

Let me think about what the "ideal" power spectrum looks like:

    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    if lr_init is None:
        lr_init = 1.0 / max(alpha * 0.05, 1.0)
    step = lr_init
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        for _ in range(40):
            obj_new, _ = lse_obj_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * 0.9:
                break
            step *= 0.9
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / 0.9, lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def spectral_phaserecovery(h0, target, deadline, n_iters=500):
    """Alternating spectrum shaping and constraint projection."""
    n = len(h0)
    dx = 2.0 / n
    h_cur = project_feasible(h0, target).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    # Target autocorrelation: r[0] = n/2, r[k] = n/4 for k>0
    # But we can soften this: r[k] = n/4 * (1 + eps) for small k, n/4*(1-eps) for large k
    # Let's use a parameter to control the "flatness"
    ideal_r = np.full(n, n / 4.0)
    ideal_r[0] = n / 2.0
    target_power = np.abs(np.fft.fft(ideal_r)) ** 2
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        H = np.fft.fft(h_cur)
        current_power = np.abs(H) ** 2
        
        # Blend current and target power
        # Use a schedule: start with more current, end with more target
        blend = 0.3 + 0.5 * (it / n_iters)  # 0.3 to 0.8
        new_power = blend * current_power + (1 - blend) * target_power
        
        # Ensure DC is correct (maintains sum)
        new_power[0] = current_power[0]
        
        # Ensure conjugate symmetry for real output
        # For real h: H[n-k] = conj(H[k]), so |H[n-k]|² = |H[k]|²
        # new_power should already be symmetric if current_power and target_power are
        # But let's enforce it
        for k in range(1, n // 2 + 1):
            if k != n - k:
                avg = (new_power[k] + new_power[n - k]) / 2.0
                new_power[k] = avg
                new_power[n - k] = avg
        
        # New spectrum with current phase
        phase = np.angle(H)
        H_new = np.sqrt(np.maximum(new_power, 0)) * np.exp(1j * phase)
        h_new = np.real(np.fft.ifft(H_new))
        
        # Project to feasible set
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h_cur = h_new
    
    return best_h, best_score
