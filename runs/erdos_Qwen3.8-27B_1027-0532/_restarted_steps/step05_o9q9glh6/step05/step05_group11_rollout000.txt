import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_gn = 0
_gt = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def get_top_lag_indices(h, K):
    overlaps = compute_overlaps(h)
    n = len(h)
    n_half = n - 1
    K = min(K, len(overlaps))
    top_idx = np.argsort(overlaps)[::-1][:K]
    lags = top_idx - n_half
    return lags


def build_lp_constraints(h, g, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints


def progressive_lp(h_init, n, target, max_iter, time_limit,
                   n_top_start=50, n_top_end=399,
                   damping_start=0.95, damping_end=0.98,
                   reg_start=1e-3, reg_end=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 500 and stuck > 300:
            break

        g = 1.0 - h
        progress = min(1.0, it / max(1, max_iter * 0.4))
        n_top = int(n_top_start + (n_top_end - n_top_start) * progress)
        n_top = min(n_top, 2 * n - 1)
        lags = get_top_lag_indices(h, n_top)

        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress

        h_var, t_var, constraints = build_lp_constraints(h, g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5


def _cons_eq(v):
    return np.sum(v) - _gt


def _cons_eq_jac(v):
    return np.ones(_gn)


def _smooth_obj(v, alpha):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_refine(h_start, n, target, alphas, maxiter=3000):
    global _gn, _gt
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    best_sc = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_sc


def subgradient_momentum(h_init, n, target, max_steps, time_limit,
                         n_top=30, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half

        w = 0.82 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            if k >= 0:
                idx1 = idxs[idxs + k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 + k])
                idx2 = idxs[idxs >= k]
                grad[idx2] += w[i] * (-y[idx2 - k])
            else:
                idx1 = idxs[idxs - k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 - k])
                idx2 = idxs[idxs + k >= 0]
                grad[idx2] += w[i] * (-y[idx2 + k])

        grad *= dx
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break

        c5_y = float(np.max(overlaps))
        polyak_lr = (c5_y - best_c5) / (norm_grad ** 2 + 1e-20)
        lr = min(max(0.005, polyak_lr), 0.25)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, rng, n_modes=5):
    h_new = h.copy()
    x = np.linspace(0.0, 2.0, n)
    for _ in range(n_modes):
        freq = rng.randint(2, 25)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.01, 0.06)
        h_new += amp * np.sin(np.pi * freq * x + phase)
    h_new = project_feasible(h_new, target)
    return h_new


def random_perturb(h, n, target, rng, noise_level=0.04, mask_frac=0.3):
    h_new = h.copy()
    mask = rng.random(n) < mask_frac
    if mask.sum() > 0:
        h_new[mask] += rng.randn(int(mask.sum())) * noise_level
    h_new = project_feasible(h_new, target)
    return h_new
