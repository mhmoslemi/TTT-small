```python
def compute_all_gradients(h, n):
    """Compute gradient of all c_k with respect to h.
    Returns matrix G of shape (n, 2n-1) where G[i, k_idx] = ∂c_{k_idx}/∂h[i]."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    G = np.zeros((n, n_corr))
    idxs = np.arange(n)
    
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)
        # First term: (1-h[i+k]) for valid i+k
        valid1 = (idxs + k >= 0) & (idxs + k < n)
        G[valid1, k_idx] += (1.0 - h[idxs[valid1] + k]) * dx
        # Second term: -h[i-k] for valid i-k
        valid2 = (idxs - k >= 0) & (idxs - k < n)
        G[valid2, k_idx] -= h[idxs[valid2] - k] * dx
    
    return G
```

This is O(n²) but with a Python loop over n_corr values. For n=600, that's 1199 iterations, each doing O(n) numpy operations. Total ~1 million operations, should take ~5-10ms.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def build_A(n, h, dx):
    n_corr = 2 * n - 1
    A = np.zeros((n_corr, n))
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            A[k_idx, lo:hi] = (1.0 - h[lo + k:hi + k]) * dx
    return A


def alternating_lp_full(h_init, n, target, max_iter, time_limit,
                        reg_init=1e-4, damping_init=0.8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    reg = reg_init
    n_corr = 2 * n - 1
    stagnant = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        A = build_A(n, h, dx)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0,
            A @ h_var <= t_var * cp.ones(n_corr)
        ]
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=300000, verbose=False)
            solved = (h_var.value is not None)
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=300000)
                solved = (h_var.value is not None)
            except Exception:
                pass
        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 0:
            damping = max(0.25, damping * 0.96)
        else:
            damping = min(0.98, damping + 0.02)

        if stagnant > 100:
            reg = max(1e-9, reg * 0.6)
            if reg < 1e-8:
                break

        if it % 20 == 0:
            reg = max(1e-9, reg * 0.85)

    return best_h, best_c5


def alternating_lp_top(h_init, n, target, max_iter, time_limit,
                       reg_init=1e-5, damping_init=0.9, n_top=80):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    reg = reg_init
    half_n = n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = set(np.argsort(corr)[-n_top:].tolist())
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k_idx in sorted(top_idx):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=300000, verbose=False)
            solved = (h_var.value is not None)
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=300000)
                solved = (h_var.value is not None)
            except Exception:
                pass
        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.98, damping + 0.02)

        if it % 25 == 0:
            reg = max(1e-9, reg * 0.75)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def nesterov_lse(h_init, n, target, time_limit, alpha_start=100, alpha_max=50000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    half_n = n - 1
    idxs = np.arange(n)
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    t_aux = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_start
    no_improve = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break

        t_prev = t_aux
        t_aux = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        beta = (t_prev - 1.0) / t_aux
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        scaled = alpha * corr
        scaled_max = np.max(scaled)
        exp_scaled = np.exp(scaled - scaled_max)
        p = exp_scaled / np.sum(exp_scaled)

        grad = np.zeros(n)
        active_k = np.where(p > 1e-8)[0]
        for k_idx in active_k:
            k = int(k_idx) - half_n
            w = p[k_idx]
            valid1 = (idxs + k >= 0) & (idxs + k < n)
            grad[valid1] += w * (1.0 - y[idxs[valid1] + k]) * dx
            valid2 = (idxs - k >= 0) & (idxs - k < n)
            grad[valid2] -= w * y[idxs[valid2] - k] * dx

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        L_est = alpha * 4.0 * n * dx * dx
        lr = 1.0 / max(L_est, 1e-6)
        lr = min(lr, 1.0 / (grad_norm + 1e-10))

        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        attempts = 0
        while c5_new > best_c5 - 1e-14 and attempts < 6:
            lr *= 0.5
            h_new = y - lr * grad
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            attempts += 1

        h_prev = h
        h = h_new

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            alpha = min(alpha_max, alpha * 2.0)
            no_improve = 0
            h_prev = h.copy()
            t_aux = 1.0

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def binary_push_and_refine(h, n, target):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    sc_bin = compute_c5(h_bin)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if sc_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = sc_bin
    for blend in [0.7, 0.5, 0.3, 0.8, 0.2]:
        h_blend = blend * best_h + (1.0 - blend) * project_feasible(1.0 - best_h, target)
        h_blend = project_feasible(h_blend, target)
        sc = compute_c5(h_blend)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_blend
    return best_h, best_c5


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100:
            break

    return best_h, best_c5
