```python
    Must be called BEFORE updating h."""

    #           + (1-h[j+k])*[0<=j+k<n] - h[j-k]*[0<=j-k<n]
```

```python
def find_ideal_swaps(h, k, n):
    """Find swaps (i,j) that would most reduce corr[k]."""
    # i: h[i]=1, h[i+k]=0, h[i-k]=0 (when valid)
    # j: h[j]=0, h[j+k]=1, h[j-k]=1 (when valid)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    i_cands = []
    for i in ones:
        ok = True
        if 0 <= i+k < n:
            if h[i+k] != 0:
                ok = False
        # If i+k is out of range, no contribution from that term, so it's "neutral"
        if 0 <= i-k < n:
            if h[i-k] != 0:
                ok = False
        if ok:
            i_cands.append(i)
    
    j_cands = []
    for j in zeros:
        ok = True
        if 0 <= j+k < n:
            if h[j+k] != 1:
                ok = False
        if 0 <= j-k < n:
            if h[j-k] != 1:
                ok = False
        if ok:
            j_cands.append(j)
    
    return i_cands, j_cands
```
