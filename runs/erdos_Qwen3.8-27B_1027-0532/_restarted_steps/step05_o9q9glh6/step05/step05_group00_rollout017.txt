import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.clip(np.round(h), 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def greedy_full(h, max_iters, time_limit=None, start_time=None, scan_width=20):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 500
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i_cand = np.array([i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0])
        j_cand = np.array([j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1])
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            break
        
        top_threshold = max_val * 0.9
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_vals = corr[top_indices].astype(float)
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in range(len(top_indices)):
                    k = top_indices[t_idx] - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_fft_N = 2048
_cons_n = 0
_cons_target = 0.0
_pen_target = 0.0
_lam = 0.0


def fast_corr(h):
    n = len(h)
    size = 2 * n - 1
    H = np.fft.rfft(h, _fft_N)
    ac = np.fft.irfft(H * np.conj(H), _fft_N)[:size]
    ac = np.maximum(ac, 0.0)
    prefix = np.concatenate([[0.0], np.cumsum(h)])
    total = prefix[n]
    s = np.arange(-(n - 1), n)
    sum_h = np.empty(size)
    pos = s >= 0
    neg = ~pos
    sum_h[pos] = prefix[n - s[pos]]
    sum_h[neg] = total - prefix[-s[neg]]
    return sum_h - ac


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.clip(np.round(h), 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def greedy_full(h, max_iters, time_limit, start_time, scan_width):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = fast_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 600
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = np.array([i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0])
        j_cand = np.array([j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1])
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            break
        top_threshold = max_val * 0.88
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_vals = corr[top_indices].astype(float)
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in range(len(top_indices)):
                    k = top_indices[t_idx] - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = fast_corr(h)
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def sa_targeted(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = fast_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 10000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.5:
            worst = np.argsort(corr)[-10:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
        elif r < 0.82:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                idx1, idx2 = rng.choice(len(ones), 2, replace=False)
                idx3, idx4 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[idx1]] = 0
                h_new[ones[idx2]] = 0
                h_new[zeros[idx3]] = 1
                h_new[zeros[idx4]] = 1
                moved = True
        if not moved:
            continue
        corr_new = fast_corr(h_new)
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        accept = delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30)))
        if accept:
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def _smooth_obj(h, alpha):
    corr = fast_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha


def _smooth_jac(h, alpha):
    n = len(h)
    corr = fast_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    S = float(np.sum(exp_s))
    if S < 1e-300:
        return np.zeros(n)
    p = exp_s / S
    q = np.zeros(2 * _fft_N)
    q[_fft_N - (n - 1):_fft_N] = p[n - 1:n * 2 - 1]
    q[0:n] = p[0:n]
    qn = np.fft.irfft(np.fft.rfft(q, _fft_N) * np.conj(np.fft.rfft(1.0 - h, _fft_N)), _fft_N)
    qrev = np.zeros(2 * _fft_N)
    qrev[_fft_N - (n - 1):_fft_N] = p[:n - 1]
    qrev[0:n] = p[n - 1:]
    qnr = np.fft.irfft(np.fft.rfft(qrev, _fft_N) * np.conj(np.fft.rfft(h, _fft_N)), _fft_N)
    return qn[:n] - qnr[:n]


def slsqp_stage(h_start, n, target, alpha, maxiter, ftol):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start.copy()


def _lbgfs_obj(h, alpha):
    corr = fast_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    sm = c_max + float(np.sum(np.exp(shifted))) / alpha
    sv = np.sum(h) - _pen_target
    return sm + _lam * sv * sv


def _lbgfs_jac(h, alpha):
    corr = fast_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    S = float(np.sum(exp_s))
    if S < 1e-300:
        return np.zeros(len(h))
    p = exp_s / S
    n = len(h)
    q = np.zeros(2 * _fft_N)
    q[_fft_N - (n - 1):_fft_N] = p[n - 1:n * 2 - 1]
    q[0:n] = p[0:n]
    qn = np.fft.irfft(np.fft.rfft(q, _fft_N) * np.conj(np.fft.rfft(1.0 - h, _fft_N)), _fft_N)
    qrev = np.zeros(2 * _fft_N)
    qrev[_fft_N - (n - 1):_fft_N] = p[:n - 1]
    qrev[0:n] = p[n - 1:]
    qnr = np.fft.irfft(np.fft.rfft(qrev, _fft_N) * np.conj(np.fft.rfft(h, _fft_N)), _fft_N)
    g = qn[:n] - qnr[:n]
    sv = np.sum(h) - _pen_target
    g = g + 2.0 * _lam * sv * np.ones(n)
    return g


def lbgfs_stage(h_start, n, target, alphas, maxiter, start_time, time_limit):
    global _pen_target, _lam
    _pen_target = target
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for _ in range(3):
            if time.time() - start_time > time_limit:
                break
            try:
                res = minimize(_lbgfs_obj, h_cur, args=(alpha,),
                               jac=lambda x, a: _lbgfs_jac(x, a),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-16,
                                        'gtol': 1e-10, 'maxls': 40, 'maxfun': 50000})
                h_new = project_feasible(res.x, target)
                if abs(np.sum(h_new) - target) < 1e-6:
                    h_cur = h_new
                    break
                else:
                    _lam = _lam * 10.0
            except Exception:
                _lam = _lam * 10.0
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h = h_sa.copy()

    # Phase 3: SLSQP with analytical gradient - progressive alpha
    if time.time() < budget_cutoff:
        h_slsqp = best_h.copy()
        alphas = [5, 15, 50, 150, 500, 1500, 5000]
        for alpha in alphas:
            if time.time() > budget_cutoff - 200:
                break
            h_new = slsqp_stage(h_slsqp, n, target, alpha, 8000, 1e-14)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_slsqp = h_new

    # Phase 4: L-BFGS-B with high alpha
    remaining = budget_s - 20 - (time.time() - start_time)
    if remaining > 120:
        _lam = 1e5
        alphas_lbg = [500, 2000, 8000, 30000]
        lbg_budget = min(remaining * 0.5, 500)
        h_lbg = lbgfs_stage(best_h, n, target, alphas_lbg, 8000,
                            start_time, lbg_budget)
        sc = compute_c5(h_lbg)
        if sc < best_score:
            best_score = sc
            best_h = h_lbg.copy()

    # Phase 5: SLSQP high alpha polish
    if time.time() < budget_cutoff - 30:
        h_pol = best_h.copy()
        alphas_high = [15000, 50000, 150000]
        for alpha in alphas_high:
            if time.time() > budget_cutoff - 15:
                break
            h_new = slsqp_stage(h_pol, n, target, alpha, 5000, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_pol = h_new

    # Phase 6: L-BFGS-B final push
    remaining = budget_s - 20 - (time.time() - start_time)
    if remaining > 30:
        _lam = 1e6
        h_lbg2 = lbgfs_stage(best_h, n, target, [20000, 100000], 5000,
                             start_time, max(5, remaining - 5))
        sc = compute_c5(h_lbg2)
        if sc < best_score:
            best_score = sc
            best_h = h_lbg2.copy()

    # Phase 7: Final SLSQP
    if time.time() < budget_cutoff - 5:
        h_new = slsqp_stage(best_h, n, target, 300000, 3000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Ensure feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
