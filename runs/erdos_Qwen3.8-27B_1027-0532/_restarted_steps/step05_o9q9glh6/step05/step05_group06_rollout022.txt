def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu, beta):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    val += beta * float(np.sum(x * (1.0 - x)))
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    grad += beta * (1.0 - 2.0 * x)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, beta, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu, beta)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu, beta)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def binary_local_search(h_bin, max_sweeps=200, deadline=None):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    
    # Verify it's binary
    assert np.allclose(h, np.round(h))
    
    corr = compute_corr(h).astype(float)
    cur_max = float(np.max(corr))
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    # Precompute effect matrix
    # effect[p, k] = (1 - h[p+k-(n-1)]) if 0 <= p+k-(n-1) < n else 0
    effect = np.zeros((n, 2*n-1), dtype=float)
    for p in range(n):
        k_start = max(0, n-1-p)
        k_end = min(2*n-2, n-1+p)
        idx = np.arange(k_start, k_end+1)
        src = p + idx - (n-1)
        effect[p, idx] = 1.0 - h[src]
    
    def recompute_effect():
        for p in range(n):
            k_start = max(0, n-1-p)
            k_end = min(2*n-2, n-1+p)
            idx = np.arange(k_start, k_end+1)
            src = p + idx - (n-1)
            effect[p, idx] = 1.0 - h[src]
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 10:
            break
        
        best_gain = 1e-10
        best_p = -1
        best_q = -1
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        for ip in range(n_ones):
            p = ones[ip]
            base = corr - effect[p]
            # For each zero q, new_corr = base + effect[q]
            # We want argmin_q max(base + effect[q])
            # Vectorized: (n_zeros, 2n-1) matrix, take max along axis 1
            new_maxes = (base[None, :] + effect[zeros]).max(axis=1)
            best_local = new_maxes.argmin()
            candidate_max = new_maxes[best_local]
            
            if candidate_max < cur_max - best_gain:
                best_gain = cur_max - candidate_max
                best_p = p
                best_q = zeros[best_local]
        
        if best_p < 0:
            break
        
        # Apply swap
        h[best_p] = 0.0
        h[best_q] = 1.0
        corr = compute_corr(h).astype(float)
        cur_max = float(np.max(corr))
        
        # Update effect
        recompute_effect()
        
        # Update ones and zeros
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    
    return h, cur_max
