def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

def interpolate_to(h_src, n_dst, target_dst):

def _sl_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _sl_target

    return np.ones(_sl_n)

def slsqp_refine(h_start, n, target, time_limit, start_time):
    global _sl_n, _sl_target
    _sl_n = n
    _sl_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sl_cons_fun, 'jac': _sl_cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    alphas = [50, 150, 500, 2000, 8000, 30000, 100000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_sl_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-15})

            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
