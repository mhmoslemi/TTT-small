    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

At convergence (h_new ≈ h_current), the gap vanishes and t* ≈ C₅(h*).

So the LP gives a certificate: if the LP is infeasible with t < C, then C₅ ≥ C for any h.

This means: if the LP optimum is 0.3808, then we've found h with C₅ ≤ 0.3808 + (convergence gap).

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    c_max = np.max(vals)
    shifted = alpha * (vals - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)

    grad = np.zeros(n)
    n_half = n - 1
    for ki in range(2 * n - 1):
        if weights[ki] < 1e-12:
            continue
        k = ki - n_half
        if k >= 0:

            grad[idx1] += weights[ki] * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= weights[ki] * h[idx2 - k]
        else:

            idx1 = np.arange(kk, n)
            grad[idx1] += weights[ki] * (1.0 - h[idx1 - kk])
            idx2 = np.arange(0, n - kk)
            grad[idx2] -= weights[ki] * h[idx2 + kk]

    return grad

                best_h = h_hi
                n = n_hi
                target = n_hi / 2.0

    # Phase 8: Final gradient polish
    if tleft() > 30:
        h_pol, c5_pol = projected_gradient(best_h, len(best_h), len(best_h) / 2.0,
                                            max_steps=50000, time_limit=min(25, tleft() * 0.6),
                                            alpha=5000, momentum=0.8)
        if c5_pol < best_score:
            best_score = c5_pol
            best_h = h_pol

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
