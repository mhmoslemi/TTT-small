```python
def compute_B_matrix(h, n):
    """Compute full B matrix: B[m, p] = d(corr[m])/d(h[p])"""
    nl1 = 2*n - 1
    B = np.zeros((nl1, n))
    for p in range(n):
        for lag in range(-(n-1), n):
            m = lag + (n-1)
            val = 0.0
            if 0 <= p - lag < n:
                val += (1.0 - h[p - lag])
            if 0 <= p + lag < n:
                val -= h[p + lag]
            B[m, p] = val
    return B
```

```python
    # p_idx[None,:] shape (1, n)

    B[mask2] -= h[p_plus_lag[mask2]]
    
    return B
```

```python
def compute_B_matrix(h, n):
    nl1 = 2 * n - 1
    lags = np.arange(nl1) - (n - 1)
    p_idx = np.arange(n)
    p_minus_lag = p_idx[None, :] - lags[:, None]
    p_plus_lag = p_idx[None, :] + lags[:, None]
    B = np.zeros((nl1, n))
    mask1 = (p_minus_lag >= 0) & (p_minus_lag < n)
    mask2 = (p_plus_lag >= 0) & (p_plus_lag < n)
    B[mask1] += (1.0 - h[p_minus_lag[mask1]])
    B[mask2] -= h[p_plus_lag[mask2]]
    return B

def update_B_single(B, idx, delta, n):

    nl1 = 2 * n - 1

    B[np.where(valid2)[0], p2[valid2]] -= delta

def find_optimal_delta(corr, C, delta_min, delta_max):
    """Find delta in [delta_min, delta_max] that minimizes max(corr + delta*C)."""
    # Binary search on the optimal value v
    # For given v: need corr[m] + delta*C[m] <= v for all m
    #   C[m] > 0: delta <= (v - corr[m]) / C[m]
    #   C[m] < 0: delta >= (v - corr[m]) / C[m]
    #   C[m] = 0: need corr[m] <= v
    
    pos = C > 1e-14
    neg = C < -1e-14
    zero = ~(pos | neg)
    
    # Lower bound on v: max(corr[zero]) if any zero, else -inf
    if np.any(zero):
        v_zero_max = float(np.max(corr[zero]))
    else:
        v_zero_max = -np.inf
    
    # Upper bound on v: max over feasible delta
    # At delta_min: max(corr + delta_min * C)
    # At delta_max: max(corr + delta_max * C)
    v_at_min = float(np.max(corr + delta_min * C))
    v_at_max = float(np.max(corr + delta_max * C))
    
    lo = max(v_zero_max, -1e10)
    hi = max(v_at_min, v_at_max)
    
    for _ in range(40):
        mid = (lo + hi) * 0.5
        # Check if mid is feasible
        ub = delta_max
        if np.any(pos):
            ub = min(ub, float(np.min((mid - corr[pos]) / C[pos])))
        lb = delta_min
        if np.any(neg):
            lb = max(lb, float(np.max((mid - corr[neg]) / C[neg])))
        if lb <= ub + 1e-12:
            hi = mid
        else:
            lo = mid
    
    v = hi
    # Find the feasible delta range at value v
    ub = delta_max
    if np.any(pos):

    lb = delta_min
    if np.any(neg):
        lb = max(lb, float(np.max((v - corr[neg]) / C[neg])))
    ub = min(ub, delta_max)
    lb = max(lb, delta_min)
    
    if lb > ub:
        return (delta_min + delta_max) * 0.5
    return (lb + ub) * 0.5
```

```python
        if score_new < best_score - 1e-15:
            best_score = score_new
            h_cur = h_new
            corr = corr_new
            stall = 0
            lr = max(lr * 0.995, 0.002)
        else:
            stall += 1
            if stall > 25:
                lr = min(lr * 1.4, 0.12)
                momentum_vec *= 0.4
                if stall > 60:
                    h_cur = project_feasible(h_cur + rng.randn(n) * 0.01, target)
                    corr = compute_corr(h_cur)
                    B = compute_B_matrix(h_cur, n)
                    beta = max(beta * 0.6, 2.0)
                    stall = 0
    return h_cur, best_score


def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=1200, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    dl1 = start_time + (budget_s - 10.0) * 0.38
    h1, s1 = fast_paired_cd(best_h, n, target, dl1, rng)
    if s1 < best_score:
        best_score = s1
        best_h = h1

    if tleft() > 70:
        dl2 = start_time + (budget_s - 10.0) * 0.53
        h2, s2 = gradient_polish(best_h, n, target, dl2, rng)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    if tleft() > 130:
        rng2 = np.random.RandomState(seed + 999983)
        dl3 = start_time + (budget_s - 10.0) * 0.78
        h3, s3 = fast_paired_cd(best_h, n, target, dl3, rng2)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    if tleft() > 50:
        dl4 = start_time + (budget_s - 10.0) * 0.87
        h4, s4 = gradient_polish(best_h, n, target, dl4, rng)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    if tleft() > 20:
        dl5 = start_time + budget_s - 8.0
        h5, s5 = fast_paired_cd(best_h, n, target, dl5, rng)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    if time.time() < start_time + budget_s - 30 and best_score > 0.38084:
        h_slq = slsqp_polish(best_h, n, target, [200, 800, 2500], 1000, 1e-15)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
