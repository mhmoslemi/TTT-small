def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_corr = corr.max()
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    smooth_max_val = max_corr + np.log(exp_shifted.sum()) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    
    # For each shift k, add p[m] * d corr_k / d h
    # m = k + (n-1), so k = m - (n-1)
    for m in range(2*n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        k = m - (n - 1)
        
        if k == 0:
            grad += pm * (1.0 - 2.0 * h)
        elif k > 0:
            # d corr_k / d h[j] = (1-h[j+k]) for j < n-k, minus h[j-k] for j >= k
            n_valid = n - k
            if n_valid > 0:
                grad[:n_valid] += pm * (1.0 - h[k:])
                if k < n:
                    grad[k:] -= pm * h[:n_valid]
        else:
            kp = -k
            n_valid = n - kp
            if n_valid > 0:
                grad[:n_valid] -= pm * h[kp:]
                if kp < n:
                    grad[kp:] += pm * (1.0 - h[:n_valid])
    
    return smooth_max_val, grad

def proj_grad_descent(h_init, n, target, alpha, steps, time_limit, start_time):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(steps):
        if time.time() - start_time > time_limit:
            break
        
        val, grad = smooth_max_and_grad(h, alpha)
        
        # Line search
        lr = 0.1
        for _ in range(20):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            val_new, _ = smooth_max_and_grad(h_new, alpha)
            if val_new < val:
                break
            lr *= 0.5
        
        h = h_new
        if lr < 1e-10:
            break
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    smooth_val = max_corr + np.log(exp_shifted.sum()) / alpha
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        k = m - (n - 1)
        if k == 0:
            grad += pm * (1.0 - 2.0 * h)
        elif k > 0:
            n_valid = n - k
            if n_valid > 0:
                grad[:n_valid] += pm * (1.0 - h[k:])
                if k < n:
                    grad[k:] -= pm * h[:n_valid]
        else:
            kp = -k
            n_valid = n - kp
            if n_valid > 0:
                grad[:n_valid] -= pm * h[kp:]
                if kp < n:
                    grad[kp:] += pm * (1.0 - h[:n_valid])
    
    return smooth_val, grad
