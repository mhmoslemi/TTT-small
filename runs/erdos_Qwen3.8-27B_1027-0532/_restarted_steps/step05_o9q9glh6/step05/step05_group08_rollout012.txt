```python
            h[pos_0] = 0
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-12:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-12:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def slsqp_optimize(h, n, target, alpha, maxiter, ftol, deadline):
    if time.time() > deadline:
        return h
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgs_optimize(h, n, target, alpha, mu, maxiter, deadline):
    if time.time() > deadline:
        return h
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def to_binary(h, n, target):
    """Convert to binary with exactly target ones"""
    h = h.copy()
    # Sort by value, take top n/2 as 1s
    threshold_idx = int(np.partition(h, n - n // 2)[n - n // 2])
    # Use threshold
    b = (h >= threshold_idx).astype(float)
    # Adjust if needed
    n_ones = int(b.sum())
    if n_ones != int(target):
        # Sort by value, pick exactly target
        sorted_idx = np.argsort(h)[::-1]
        b = np.zeros(n)
        b[sorted_idx[:int(target)]] = 1.0
    return b


def binary_local_search(h_bin, n, deadline, rng, max_stagnation=200):
    """Local search on binary vector using swaps"""
    h = h_bin.copy()
    target = n / 2.0
    best_score = compute_c5(h)
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    no_improve = 0
    stagnation_limit = max_stagnation
    
    while time.time() < deadline:
        if no_improve > stagnation_limit:
            break
        
        # Try random swap
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        p1 = int(ones[i])
        p0 = int(zeros[j])
        
        # Do swap
        h[p1] = 0.0
        h[p0] = 1.0
        
        score = compute_c5(h)
        
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
            # Update sets
            ones[i] = p0
            zeros[j] = p1
        else:
            # Revert
            h[p1] = 1.0
            h[p0] = 0.0
            no_improve += 1
    
    return best_h, best_score


def gradient_guided_binary_search(h_bin, n, deadline, rng):
    """Binary local search guided by the subgradient of C5"""
    h = h_bin.copy()
    target = n / 2.0
    best_score = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    no_improve = 0
    max_stagnation = 500
    
    while time.time() < deadline and no_improve < max_stagnation:
        # Compute subgradient
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        
        # Subgradient of the max overlap
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h[k_star:])
            g[k_star:] -= h[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h[:n - ell])
            g[:n - ell] -= h[ell:]
        g *= dx
        
        # We want to decrease C5, so move in direction -g
        # For binary: a 1 at position i should be moved if g[i] > 0 (it's contributing positively to overlap)
        # A 0 at position j should receive a 1 if g[j] < 0
        
        # Find candidate swaps: 1s with high g, 0s with low g
        g_ones = g[ones]
        g_zeros = g[zeros]
        
        # Pick the 1 with highest g and 0 with lowest g (with some randomness)
        top_1_idx = int(np.argmax(g_ones))
        bottom_0_idx = int(np.argmin(g_zeros))
        
        # Add some randomness: mix top and random
        if rng.random() < 0.5:
            p1 = int(ones[top_1_idx])
            p0 = int(zeros[bottom_0_idx])
        else:
            p1 = int(ones[rng.randint(len(ones))])
            p0 = int(zeros[rng.randint(len(zeros))])
        
        # Do swap
        h[p1] = 0.0
        h[p0] = 1.0
        score = compute_c5(h)
        
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
            # Update sets
            idx_in_ones = np.where(ones == p1)[0]
            if len(idx_in_ones) > 0:
                ones[idx_in_ones[0]] = p0
            idx_in_zeros = np.where(zeros == p0)[0]
            if len(idx_in_zeros) > 0:
                zeros[idx_in_zeros[0]] = p1
        else:
            h[p1] = 1.0
            h[p0] = 0.0
            no_improve += 1
    
    return best_h, best_score


def spectral_smooth(h, n, target, n_iter, deadline, lr):
    """Minimize the variance of the autocorrelation of g = h - 0.5"""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        g = h_cur - 0.5
        # Compute autocorrelation of g
        ac = np.correlate(g, g, mode='full')
        ac_mean = np.mean(ac)
        ac_dev = ac - ac_mean
        
        # Gradient of var(ac) w.r.t. g[m]
        # d(var)/d(g[m]) = (2/(2n-1)) * sum_k ac_dev[k] * (g[m-k] + g[m+k])
        # Using FFT for efficiency:
        # sum_k ac_dev[k] * g[m-k] is a convolution
        # But for n=400, direct computation is fine
        grad = np.zeros(n)
        for k in range(n):
            # For positive lag k: ac[n-1+k] involves g[i]*g[i+k]
            # d/dg[m] contributes when m=i or m=i+k
            val = ac_dev[n - 1 + k]
            if val < 1e-14:
                continue
            # g[m] * g[m+k] -> derivative w.r.t. g[m] is g[m+k]
            # g[m-k] * g[m] -> derivative w.r.t. g[m] is g[m-k]
            if k == 0:
                grad += 2.0 * val * g
            else:
                grad[:n-k] += val * g[k:]
                grad[k:] += val * g[:n-k]
        
        # Also handle negative lags (they give the same contribution due to symmetry)
        for ell in range(1, n):
            val = ac_dev[n - 1 - ell]
            if abs(val - ac_dev[n - 1 + ell]) < 1e-14:
                continue  # symmetric, already counted
            if val < 1e-14:
                continue
            grad[ell:] += val * g[:n-ell]
            grad[:n-ell] += val * g[ell:]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Gradient step (we want to DECREASE variance, so move in -grad direction)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
            h_cur = h_new
        else:
            lr *= 0.95
            if lr < 1e-8:
                break
    
    return best_h, best_score


def subgradient_step(h, n, target, lr, deadline):
    """Single subgradient step on C5"""
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    m_star = int(np.argmax(corr))
    k = m_star - (n - 1)
    dx = 2.0 / n
    
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    g *= dx
    
    h_new = h - lr * g
    return project_feasible(h_new, target)


def multi_subgradient(h, n, target, n_top, lr, n_steps, deadline):
    """Subgradient using top-n lags with softmax weighting"""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    dx = 2.0 / n
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        # Top n_top lags
        top_idx = np.argsort(corr)[::-1][:n_top]
        corr_top = corr[top_idx]
        c_max = corr[top_idx[0]]
        
        # Softmax weights
        w = np.exp((corr_top - c_max) / 0.01)
        w /= w.sum()
        
        g = np.zeros(n)
        for wk, m in zip(w, top_idx):
            kk = int(m) - (n - 1)
            if kk >= 0:
                g[:n - kk] += wk * (1.0 - h_cur[kk:])
                g[kk:] -= wk * h_cur[:n - kk]
            else:
                ell = -kk
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        
        if score < best_score:
            best_score = score
            best_h = h_new
            h_cur = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_h, best_score


def create_staircase(n):
    """Create a staircase pattern"""
    # Create a pattern that alternates in blocks
    block_size = max(1, n // 8)
    h = np.zeros(n)
    for i in range(n):
        block = i // block_size
        if block % 2 == 0:
            h[i] = 1.0
    return h


def create_sine_pattern(n, freq):
    """Create a sine-based pattern thresholded to binary"""
    x = np.linspace(0, 1, n)
    h = (np.sin(2 * np.pi * freq * x) > 0).astype(float)
    return h


def create_random_binary(n, rng):
    """Random binary with n/2 ones"""
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h
```

```python
def to_binary(h, n):
    """Convert to binary with exactly n//2 ones"""
    target_ones = n // 2
    # Get the threshold: the n//2-th largest value
    threshold = np.partition(h, n - target_ones)[n - target_ones]
    b = (h >= threshold).astype(float)
    # Handle ties: if we have too many or too few ones, adjust
    n_ones = int(b.sum())
    if n_ones > target_ones:
        # Demote some of the ones that are exactly at threshold
        at_thresh = np.where((b == 1) & (h == threshold))[0]
        excess = n_ones - target_ones
        to_demote = at_thresh[:excess]
        b[to_demote] = 0
    elif n_ones < target_ones:
        # Promote some zeros that are exactly at threshold
        at_thresh = np.where((b == 0) & (h == threshold))[0]
        deficit = target_ones - n_ones
        to_promote = at_thresh[:deficit]
        b[to_promote] = 1
    return b
```

```python
def binary_sas(h_bin, n, deadline, rng, T_init=2.0, T_final=0.1, max_steps=50000):
    """Simulated Annealing on binary representation"""
    h = h_bin.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    T = T_init
    # Exponential cooling
    n_total = max_steps
    for step in range(n_total):
        if time.time() > deadline:
            break
        
        # Acceptance probability
        current_score = compute_c5(h)  # This is expensive!
        
        # Try a random swap
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        p1 = ones[i]
        p0 = zeros[j]
        
        h[p1] = 0.0
        h[p0] = 1.0
        new_score = compute_c5(h)
        
        if new_score < current_score:
            # Accept (improvement)
            ones[i] = p0
            zeros[j] = p1
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # Maybe accept (worsening)
            delta = new_score - current_score
            if delta < 0.05 and rng.random() < np.exp(-delta / max(T, 1e-10)):
                # Accept
                ones[i] = p0
                zeros[j] = p1
                current_score = new_score
            else:
                # Reject
                h[p1] = 1.0
                h[p0] = 0.0
        
        # Cool down
        T = max(T_final, T_init * (T_final / T_init) ** (step / n_total))
    
    return best_h, best_score
```

```python
from scipy.optimize import minimize
import time

_OPT_TARGET = 0.0
_OPT_N = 0
_OPT_ALPHA = 1.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk > 1e-12:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk > 1e-12:
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

    return lse_value(x, _OPT_ALPHA)

    return lse_grad(x, _OPT_ALPHA)

    return np.sum(x) - _OPT_TARGET

    return np.ones(_OPT_N)

    global _OPT_TARGET, _OPT_N, _OPT_ALPHA
    if time.time() > deadline:
        return h
    _OPT_TARGET = target
    _OPT_N = n
    _OPT_ALPHA = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
```

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk > 1e-12:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk > 1e-12:
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def binary_sa(h_bin, n, deadline, rng, T_init, T_final, max_stall):
    h = h_bin.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    current_score = best_score
    T = T_init
    stall = 0
    n_ones = len(ones)
    n_zeros = len(zeros)
    while time.time() < deadline:
        T = max(T_final, T * 0.99995)
        i = rng.randint(n_ones)
        j = rng.randint(n_zeros)
        p1 = int(ones[i])
        p0 = int(zeros[j])
        h[p1] = 0.0
        h[p0] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta < -1e-12:
            ones[i] = p0
            zeros[j] = p1
            current_score = new_score
            stall = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            accept_prob = np.exp(-delta / max(T, 1e-10))
            if rng.random() < accept_prob:
                ones[i] = p0
                zeros[j] = p1
                current_score = new_score
            else:
                h[p1] = 1.0
                h[p0] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return best_h, best_score

            else:
