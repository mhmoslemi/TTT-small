def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Shift so that the max is at the center for numerical stability
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    # Softmax
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / alpha
    
    # Gradient: weighted sum of gradients of each R_k
    weights = np.exp(shifted - log_sum)  # softmax weights
    
    # Compute gradient of each R_k wrt h
    grad = np.zeros(n)
    # For efficiency, we can vectorize
    # For shift k (0-indexed in the full array, corresponding to lag k-(n-1)):
    # lag = idx - (n-1)
    for idx in range(2*n-1):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag > 0:
            # R = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # dR/dh[m] = (1-h[m-lag]) if lag <= m <= n-1, minus h[m+lag] if 0 <= m <= n-1-lag
            mask1 = (lag <= np.arange(n)) & (np.arange(n) <= n-1)
            mask2 = (np.arange(n) >= 0) & (np.arange(n) <= n-1-lag)
            grad[mask1] += w * (1 - h[np.arange(n)[mask1] - lag])
            grad[mask2] -= w * h[np.arange(n)[mask2] + lag]
        elif lag < 0:
            # R = sum_{j=0}^{n-1+lag} h[j]*(1-h[j-lag]) where lag is negative, so j-lag = j+|lag|
            # Actually: c[idx] = sum_{i=-lag}^{n-1} h[i+lag]*(1-h[i])
            # Let j = i+lag, then i = j-lag = j+|lag|
            # = sum_{j=0}^{n-1+lag} h[j]*(1-h[j+|lag|])
            abs_lag = -lag
            # R = sum_{j=0}^{n-1-abs_lag} h[j]*(1-h[j+abs_lag])
            # dR/dh[m] = (1-h[m+abs_lag]) if 0 <= m <= n-1-abs_lag, minus h[m-abs_lag] if abs_lag <= m <= n-1
            mask1 = (np.arange(n) >= 0) & (np.arange(n) <= n-1-abs_lag)
            mask2 = (abs_lag <= np.arange(n)) & (np.arange(n) <= n-1)
            grad[mask1] += w * (1 - h[np.arange(n)[mask1] + abs_lag])
            grad[mask2] -= w * h[np.arange(n)[mask2] - abs_lag]
        else:
            # lag = 0: R = sum h[i]*(1-h[i]), dR/dh[m] = 1-2h[m]

    return obj, grad

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        else:

            break

        h[mask] += adj

        h = np.clip(h, 0.0, 1.0)
        if abs(diff) < 1e-12:

    """Compute smooth max of correlations and its gradient"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[ti] + d
                    if v > new_max:
                        new_max = v
                        break
                if new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def sa_block(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 8000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h_new = h.copy()
        moved = False
        r = rng.random()
        if r < 0.45:
            worst = np.argsort(corr)[-8:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n and h[i + k] == 0]
            jp = [j for j in zeros if 0 <= j - k < n and h[j - k] == 1]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
            elif len(ones) > 0 and len(zeros) > 0:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.75:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif r < 0.90:
            block_len = rng.randint(2, 5)
            if len(ones) >= block_len and len(zeros) >= block_len:
                i_start = ones[rng.randint(len(ones))]
                j_start = zeros[rng.randint(len(zeros))]
                i_block = [i_start + d for d in range(block_len) if 0 <= i_start + d < n]
                j_block = [j_start + d for d in range(block_len) if 0 <= j_start + d < n]
                i_block = [x for x in i_block if h[x] == 1]
                j_block = [x for x in j_block if h[x] == 0]
                if len(i_block) > 0 and len(j_block) > 0:
                    h_new[i_block[:block_len]] = 0
                    h_new[j_block[:block_len]] = 1
                    moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def thue_morse_init(n, target):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] + (i & 1)
    h = (t % 2 == 0).astype(float)
    return project_binary(h, target)

def m_sequence_init(n, target, seed_poly=0x3B8):
    m = n
    state = 1
    seq = np.zeros(m, dtype=int)
    for i in range(m):
        seq[i] = state & 1
        feedback = 0
        for b in range(16):
            if (seed_poly >> b) & 1:
                feedback ^= (state >> b) & 1
        feedback ^= state & 1
        state = (state >> 1) | (feedback << 15)
    h = (1 - seq).astype(float)
    return project_binary(h, target)

def greedy_block(h, max_iters, time_limit, start_time, scan_width=12, top_k=4):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 400
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        top_idx = np.argsort(corr)[-top_k:]
        top_shifts = [int(x) - (n - 1) for x in top_idx]
        top_vals = np.array([float(corr[x]) for x in top_idx])
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i_cand = []
        for i in ones:
            for k in top_shifts:
                if 0 <= i + k < n and h[i + k] == 0:
                    i_cand.append(i)
                    break
        i_cand = list(dict.fromkeys(i_cand))[:scan_width]
        j_cand = []
        for j in zeros:
            for k in top_shifts:
                if 0 <= j + k < n and h[j + k] == 1:
                    j_cand.append(j)
                    break
        j_cand = list(dict.fromkeys(j_cand))[:scan_width]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            break
        best_improve = 0.0
        best_move = None
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                new_max = max_val
                for ti in range(len(top_idx)):
                    k = top_shifts[ti]
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[ti] + d
                    if v > new_max:
                        new_max = v
                        break
                if new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
