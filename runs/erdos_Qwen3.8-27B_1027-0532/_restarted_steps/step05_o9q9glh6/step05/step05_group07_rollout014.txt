    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = float(c.max())
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    p = np.exp(shifted)
    p_sum = float(p.sum())
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / p_sum
    grad = np.zeros(n)
    omh = 1.0 - h
    for m in range(2 * n - 1):
        w = p[m] * dx
        if abs(w) < 1e-14:
            continue
        k = m - (n - 1)
        ak = abs(k)
        if ak == 0:
            grad = grad + w * omh - w * h
        else:
            grad[ak:] = grad[ak:] + w * omh[:n - ak]
            grad[:n - ak] = grad[:n - ak] - w * h[ak:]
    return grad


def gd_smooth(h_init, n, target, alpha, max_iter, deadline):
    h = h_init.copy()
    step = 1.0 / (3.0 * alpha)
    prev_val = smooth_max_val(h, alpha)
    n_grow = 0
    for it in range(max_iter):
        if it % 50 == 0 and time.time() > deadline:
            break
        grad = smooth_max_grad(h, alpha)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        x_new = project_sum_interval(h - step * grad, target)
        val_new = smooth_max_val(x_new, alpha)
        n_shrink = 0
        while val_new > prev_val - 1e-14 and n_shrink < 20:
            step *= 0.5
            x_new = project_sum_interval(h - step * grad, target)
            val_new = smooth_max_val(x_new, alpha)
            n_shrink += 1
        if n_shrink >= 20:
            break
        h = x_new
        prev_val = val_new
        n_grow += 1
        if n_grow >= 10 and step < 1.0 / (3.0 * alpha):
            step = min(step * 1.2, 1.0 / (3.0 * alpha))
            n_grow = 0
    return h


def obj_slsqp(x):
    return smooth_max_val(x, _g_alpha)


def obj_slsqp_grad(x):
    return smooth_max_grad(x, _g_alpha)


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter,
                 deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac,
             'direction': -1.0}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline - 15:
            break
        _g_alpha = a
        try:
            res = minimize(obj_slsqp, h_cur, jac=obj_slsqp_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter,
                                    'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def paired_coord_desc(h, n, target, deadline, rng,
                      recompute_interval=30):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep = 0
    while time.time() < deadline:
        sweep += 1
        rng.shuffle(pair_positions)
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c_val = h_cur[i] + h_cur[j]
            if c_val < 1e-14 or c_val > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c_val - 1.0)
            t_max = min(1.0, c_val)
            if t_max - t_min < 1e-13:
                continue
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c_val
            lo = float(np.max(A + C * t_min))
            hi = float(np.max(A + C * t_max))
            for _ in range(30):
                mid = 0.5 * (lo + hi)
                pos = C > 1e-14
                neg = C < -1e-14
                zero = ~(pos | neg)
                vmid = float(np.max(A[zero])) if np.any(zero) else -1e30
                if np.any(pos):
                    vmid2 = float(np.max(A[pos] + C[pos] * t_max))
                    can = mid <= float(np.min((mid - A[pos]) / C[pos])) + 1e-12
                    feasible = (mid >= float(np.max((mid - A[pos]) / C[pos])) - 1e12)
                    v_at_tmax = float(np.max(A[pos] + C[pos] * t_max))
                    v_at_tmin = float(np.max(A[pos] + C[pos] * t_min))
                    vlin = np.interp(mid, [float(np.max(A+C*t_min)), float(np.max(A+C*t_max))],
                                     [float(np.max(A+C*t_min)), float(np.max(A+C*t_max))])
                    vpos = float(np.max(np.where(pos, A + C * mid, -1e30))) if np.any(pos) else -1e30
                else:
                    vpos = -1e30
                vneg = float(np.max(np.where(neg, A + C * mid, -1e30))) if np.any(neg) else -1e30
                vz = float(np.max(np.where(zero, A, -1e30))) if np.any(zero) else -1e30
                vval = max(vpos, vneg, vz)
                if vval <= mid + 1e-12:
                    hi = mid
                else:
                    lo = mid
            v = hi
            t_star = h_cur[i]
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = 0.5 * (t_min + t_max)
            else:
                t_star = 0.5 * (lb + ub) + 0.5 * h_cur[i]
                t_star = max(t_min, min(t_max, t_star))
            old_i = h_cur[i]
            delta_i = t_star - old_i
            if abs(delta_i) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c_val - t_star
                corr = corr + delta_i * C
        if sweep % recompute_interval == 0:
            corr = compute_corr(h_cur)
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-14:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 150 and sweep > 50:
            break
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Gradient descent with smooth max (find basin)
    deadline1 = start_time + (budget_s - 20.0) * 0.18
    h1 = gd_smooth(best_h, n, target, 50.0, 50000, deadline1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: GD with sharper alpha
    if tleft() > 60:
        deadline2 = start_time + (budget_s - 20.0) * 0.30
        h2 = gd_smooth(best_h, n, target, 300.0, 30000, deadline2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: SLSQP polish
    if tleft() > 80:
        deadline3 = start_time + (budget_s - 20.0) * 0.40
        h3, sc3 = slsqp_polish(best_h, n, target,
                               [500, 2000], 2000, deadline3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Coordinate descent (fine-tuning)
    if tleft() > 100:
        deadline4 = start_time + (budget_s - 20.0) * 0.58
        h4, sc4 = paired_coord_desc(best_h, n, target,
                                    deadline4, rng, recompute_interval=25)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Multi-scale (downsample, optimize, upscale)
    if tleft() > 200:
        n_coarse = max(n // 3, 60)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        deadline_c1 = start_time + (budget_s - 20.0) * 0.65
        h_coarse = gd_smooth(h_coarse, n_coarse, target_coarse,
                             100.0, 20000, deadline_c1)
        deadline_c2 = start_time + (budget_s - 20.0) * 0.68
        h_coarse, _ = paired_coord_desc(h_coarse, n_coarse,
                                        target_coarse, deadline_c2, rng)
        h_up = interpolate_to(h_coarse, n, target)
        if tleft() > 100:
            deadline_up1 = start_time + (budget_s - 20.0) * 0.72
            h_up = gd_smooth(h_up, n, target, 200.0, 20000, deadline_up1)
        if tleft() > 80:
            deadline_up2 = start_time + (budget_s - 20.0) * 0.76
            h_up, sc_up = paired_coord_desc(h_up, n, target,
                                            deadline_up2, rng)
            if sc_up < best_score:
                best_score = sc_up
                best_h = h_up

    # Phase 6: SLSQP with high alpha
    if tleft() > 60:
        deadline6 = start_time + (budget_s - 20.0) * 0.82
        h6, sc6 = slsqp_polish(best_h, n, target,
                               [5000, 15000], 2000, deadline6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: More coordinate descent
    if tleft() > 50:
        deadline7 = start_time + (budget_s - 20.0) * 0.90
        h7, sc7 = paired_coord_desc(best_h, n, target,
                                    deadline7, rng, recompute_interval=20)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Structured restarts
    if tleft() > 80:
        patterns = []
        # DCT-based perturbation
        for amp in [0.03, 0.06, 0.10]:
            h_pert = best_h.copy()
            for mode in range(1, 4):
                k = mode * 3 + rng.randint(0, 2)
                phase = rng.uniform(0, 2.0 * np.pi)
                basis = np.cos(np.pi * k * (np.arange(n) + 0.5) / n)
                h_pert = h_pert + amp * np.cos(phase) * basis
            h_pert = project_feasible(h_pert, target)
            patterns.append(h_pert)
        # Sine perturbation
        for freq in [2, 3, 5]:
            h_pert = best_h + 0.04 * np.sin(
                2.0 * np.pi * freq * np.arange(n) / n)
            h_pert = project_feasible(h_pert, target)
            patterns.append(h_pert)
        for h_pert in patterns:
            if tleft() < 40:
                break
            deadline_r = min(start_time + (budget_s - 20.0) * 0.93,
                             time.time() + 35.0)
            hr, scr = paired_coord_desc(h_pert, n, target,
                                        deadline_r, rng, recompute_interval=20)
            if scr < best_score:
                best_score = scr
                best_h = hr

    # Phase 9: Final SLSQP polish
    if tleft() > 30:
        h_fin, sc_fin = slsqp_polish(best_h, n, target,
                                     [10000, 30000], 1500,
                                     start_time + budget_s - 8.0)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
