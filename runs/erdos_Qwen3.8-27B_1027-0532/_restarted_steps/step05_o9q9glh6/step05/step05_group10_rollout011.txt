import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_vals = np.exp(np.clip(shifted, -500, 0))
    return float(c_max + np.log(np.sum(exp_vals)) / alpha)

def smooth_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_vals = np.exp(np.clip(shifted, -500, 0))
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_vals / total
    g = 1.0 - h
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wv = weights[k_idx]
        if wv < 1e-16:
            continue
        if k >= 0:
            grad[:n-k] += wv * g[k:]
            grad[k:] -= wv * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wv * g[:n-kk]
            grad[:n-kk] -= wv * h[kk:]
    grad *= dx
    return grad

def slsqp_analytic(h_start, n, target, alpha, maxiter, ftol):
    """SLSQP with analytic gradient for smooth C5."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h_cur = h_start.copy()
    try:
        res = minimize(
            lambda x: smooth_c5(x, alpha),
            h_cur,
            jac=lambda x: smooth_c5_grad(x, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur

def coord_descent_worst(h, n, target, max_iter, time_limit):
    """Coordinate descent targeting the worst shift."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        
        # Gradient of c_k at h_cur
        grad = np.zeros(n)
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        
        # Projected gradient step
        improved = False
        for lr in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        
        if not improved:
            # Try perturbing a random subset
            mask = np.random.random(n) < 0.3
            h_try = h_cur.copy()
            h_try[mask] += 0.05 * np.random.randn(int(mask.sum()))
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
        
        if not improved and it > 20:
            break
    
    return best_h, best_c5

def lp_constraint_generation(h_init, n, target, max_iter, time_limit, n_active=50):
    """Alternating LP with constraint generation."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = 0.7
    current_reg = 1e-3
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = True
            except Exception:
                pass
        
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        
        t_val = t_var.value if t_var.value is not None else best_c5
        # Add violated constraints
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-5:
                    active_k.add(k_idx)
                    break
        
        active_k = top_k | active_k
        if len(active_k) > 2 * n_active:
            active_k = set(np.argsort(corr)[-2 * n_active:].tolist())
        
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.9, damping + 0.03)
        
        if it % 50 == 0 and current_reg > 1e-6:
            current_reg = max(1e-6, current_reg * 0.7)
    
    return best_h, best_c5

def pairwise_swap(h, n, target, max_iter, time_limit):
    """Try swapping small ranges to improve."""
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(42)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Pick random position and try to adjust a small segment
        pos = rng.randint(0, n - 5)
        length = rng.randint(3, 10)
        
        # Try flipping the segment (1 - h)
        h_try = h_cur.copy()
        h_try[pos:pos+length] = 1.0 - h_try[pos:pos+length]
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
        
        # Try shifting the segment
        if pos + length < n - 2:
            h_try2 = h_cur.copy()
            segment = h_try2[pos:pos+length].copy()
            h_try2[pos:pos+length] = h_try2[pos+1:pos+length+1]
            h_try2[pos+length] = segment[0]
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < best_c5 - 1e-14:
                best_c5 = sc2
                best_h = h_try2.copy()
                h_cur = h_try2
    
    return best_h, best_c5

from scipy.optimize import minimize
import cvxpy as cp
import time

_global_alpha = 1.0
_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_vals = np.exp(np.clip(shifted, -500, 0))
    return float(c_max + np.log(np.sum(exp_vals)) / alpha)


def smooth_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_vals = np.exp(np.clip(shifted, -500, 0))
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_vals / total
    g = 1.0 - h
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wv = weights[k_idx]
        if wv < 1e-16:
            continue
        if k >= 0:
            grad[:n - k] += wv * g[k:]
            grad[k:] -= wv * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wv * g[:n - kk]
            grad[:n - kk] -= wv * h[kk:]
    grad *= dx
    return grad

    return smooth_c5(x, _global_alpha)


def _grad_func(x):
    return smooth_c5_grad(x, _global_alpha)


def _cons_fun(x):
    return np.sum(x) - _global_target


def _cons_jac(x):
    return np.ones(_global_n)


def slsqp_analytic(h_start, n, target, alpha, maxiter, ftol):
    global _global_alpha, _global_n, _global_target
    _global_alpha = alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    try:
        res = minimize(_obj_func, h_cur, jac=_grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur

    if tleft() > 50:
        h_try = slsqp_analytic(best_h, n, target, alpha=100000, maxiter=3000, ftol=1e-16)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Phase 9: Final coordinate descent
    if tleft() > 30:
        h_cd, c5_cd = coord_descent_worst(best_h, n, target,
                                          max_iter=3000, time_limit=min(20, tleft() * 0.6))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Ultra-high alpha polish
    if tleft() > 40:
        h_try = slsqp_analytic(best_h, n, target, alpha=500000, maxiter=2000, ftol=1e-16)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Phase 11: Try higher resolution
    if tleft() > 200:
        n_try = n + 100
        if n_try % 2 == 0:
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_try = slsqp_analytic(h_interp, n_try, target_try, alpha=500, maxiter=2000, ftol=1e-14)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
                n = n_try
                target = target_try
            if tleft() > 100:
                h_try2 = slsqp_analytic(best_h, n, target, alpha=5000, maxiter=2000, ftol=1e-15)
                sc2 = compute_c5(h_try2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_try2
            if tleft() > 60:
                h_try3 = slsqp_analytic(best_h, n, target, alpha=50000, maxiter=2000, ftol=1e-15)
                sc3 = compute_c5(h_try3)
                if sc3 < best_score:
                    best_score = sc3
                    best_h = h_try3

    # Final feasibility and evaluation
    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_use
