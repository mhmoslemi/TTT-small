def steepest_descent_direction(h, n, target, C_max):
    """Find direction that reduces max overlap while not increasing others."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    half_n = n - 1
    
    # Compute gradients of each overlap w.r.t. h
    # d O_k / d h[i] = (1-h[i-k]) * I(valid) - h[i+k] * I(valid)
    
    d_var = cp.Variable(n)
    constraints = [
        cp.sum(d_var) == 0,
        cp.norm(d_var, 2) <= 1.0  # This makes it non-convex!
    ]
    
    # Actually, let's use ||d||_inf <= 1 instead (box constraint)
    # Or just find the direction and normalize later.
    
    # Simpler: minimize dO_{k*}/dh * d subject to dO_k/dh * d <= 0 for all k, sum(d) = 0
    # This is a LP if we use ||d||_1 or ||d||_inf as the objective scaling.
    
    # Even simpler: just minimize dO_{k*}/dh * d subject to sum(d)=0 and |d_i| <= 1
    # This gives a feasible descent direction for the active constraint.
    
    k_star_idx = int(np.argmax(corr))
    k_star = k_star_idx - half_n

    prob = cp.Problem(cp.Minimize(g @ d_var), constraints)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, mu, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = c_max + np.log(sum_exp) / alpha + mu * (np.sum(h) - target) ** 2
    
    weights = exp_s / sum_exp
    half_n = n - 1
    grad = np.zeros(n)
    
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-16:
            continue
        k = m - half_n
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            kp = -k
            if kp < n:
                grad[:n - kp] += w * (1.0 - h[kp:])
                grad[kp:] -= w * h[:n - kp]
    
    grad += 2.0 * mu * (np.sum(h) - target)
    return obj, grad

def lbfgsb_run(h_start, alpha, mu, target, maxiter=500):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj_and_grad(x, alpha, mu, target)[0]
    
    def grad(x):
        return smooth_obj_and_grad(x, alpha, mu, target)[1]
    
    try:
        res = minimize(obj, h_start, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': maxiter * 3})
        return project_feasible(res.x, target)
    except Exception:
        return h_start.copy()

def subgradient_step(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star_idx = int(np.argmax(corr))
        k = k_star_idx - half_n
        
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h_cur[:n - k])
            grad[:n - k] -= h_cur[k:]
        else:
            kp = -k
            grad[:n - kp] += (1.0 - h_cur[kp:])
            grad[kp:] -= h_cur[:n - kp]
        grad *= dx
        
        improved = False
        for lr in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        
        if not improved and step > 30:
            break
    
    return best_h, best_c5

def alternating_lp(h_init, n, target, max_iter, time_limit, n_active=40):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    damping = 0.8
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, mu, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = c_max + np.log(sum_exp) / alpha + mu * (np.sum(h) - target) ** 2

    weights = exp_s / sum_exp
    half_n = n - 1
    grad = np.zeros(n)

    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-16:
            continue
        k = m - half_n
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            kp = -k
            if kp < n:
                grad[:n - kp] += w * (1.0 - h[kp:])
                grad[kp:] -= w * h[:n - kp]

    grad += 2.0 * mu * (np.sum(h) - target)
    return obj, grad

def lbfgsb_run(h_start, alpha, mu, target, maxiter=500):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj_and_grad(x, alpha, mu, target)[0]

    def grad(x):
        return smooth_obj_and_grad(x, alpha, mu, target)[1]

    try:
        res = minimize(obj, h_start, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': maxiter * 4})
        result = project_feasible(res.x, target)
        return result
    except Exception:
        return h_start.copy()

def subgradient_step(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star_idx = int(np.argmax(corr))
        k = k_star_idx - half_n

        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h_cur[:n - k])
            grad[:n - k] -= h_cur[k:]
        else:
            kp = -k
            grad[:n - kp] += (1.0 - h_cur[kp:])
            grad[kp:] -= h_cur[:n - kp]
        grad *= dx

        improved = False
        for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)

                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True

def alternating_lp(h_init, n, target, max_iter, time_limit, n_active=40):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    damping = 0.8
    reg = 1e-4

    corr = np.correlate(h, 1.0 - h, mode='full')

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        active_k = set(np.argsort(corr)[-n_active:].tolist())
        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
            solved = True
        except Exception:
            pass

        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if tleft() > 50:
        h_sg, c5_sg = subgradient_step(best_h, n, target, max_steps=3000, time_limit=min(40, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 40:
        h_tr, c5_tr = targeted_reduction(best_h, n, target, max_steps=300, time_limit=min(30, tleft() * 0.5))
        if c5_tr < best_score:
            best_score = c5_tr
            best_h = h_tr

    if tleft() > 200:
        h_ms = multiscale_optimize(best_h, n, target, min(100, tleft() * 0.3))
        sc = compute_c5(h_ms)
        if sc < best_score:
            best_score = sc
            best_h = h_ms
        if tleft() > 60:
            h_new = lbfgsb_run(best_h, 50000, 5000.0, target, 400)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    for noise in [0.005, 0.01, 0.015, 0.02]:
        if tleft() < 60:
            break
        h_pert = project_feasible(best_h + rng.randn(n) * noise, target)
        h_new = lbfgsb_run(h_pert, 100000, 10000.0, target, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 40:
        h_sg, c5_sg = subgradient_step(best_h, n, target, max_steps=2000, time_limit=min(30, tleft() * 0.6))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 40:
        h_tr, c5_tr = targeted_reduction(best_h, n, target, max_steps=300, time_limit=min(30, tleft() * 0.6))
        if c5_tr < best_score:
            best_score = c5_tr
            best_h = h_tr

    if tleft() > 30:
        h_new = lbfgsb_run(best_h, 500000, 20000.0, target, 400)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
