import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def greedy_binary(h, max_iters, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > 500:
            break
            
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates that would reduce the worst shift
        i_cand = [i for i in ones if 0 <= i + k_star < n]
        j_cand = [j for j in zeros if 0 <= j + k_star < n]
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            break
        
        # Top shifts to consider
        top_threshold = max_val * 0.80
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        # Sample candidates
        n_i = min(len(i_cand), 25)
        n_j = min(len(j_cand), 25)
        
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def smooth_max_obj(h, alpha, target_sum):
    """Smooth max approximation of the C5 objective."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Soft max: (1/alpha) * log(sum(exp(alpha * (corr - c_max))))
    shifted = np.clip(alpha * (corr - c_max), -100.0, 0.0)
    soft_max = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    return soft_max * (2.0 / len(h))

# Global variables for constraint
_g_n = 0
_g_target = 0.0

def _constraint_fun(h):
    return np.sum(h) - _g_target

def _constraint_jac(h):
    return np.ones(_g_n)

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    
    h_cur = h_start.astype(float).copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(lambda x: smooth_max_obj(x, alpha, target), 
                         h_cur, method='SLSQP', bounds=bounds, 
                         constraints=cons,
                         options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def binary_sa(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > 10000:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        h_new = h.copy()
        r = rng.random()
        
        if r < 0.7:
            # Targeted swap: pick from worst shifts
            worst = np.argsort(corr)[-8:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        else:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def randomized_rounding(h_frac, target_sum, n_trials, rng):
    """Try multiple randomized roundings and keep the best."""
    n = len(h_frac)
    best_h = project_binary(h_frac, target_sum)
    best_score = compute_c5(best_h)
    
    for trial in range(n_trials):
        # Adjust to get exactly target_sum ones
        h_r = np.zeros(n)
        # Probabilistic assignment
        frac = h_frac.copy()
        # Ensure sum of frac equals target (approximately)
        diff = target_sum - np.sum(frac)
        if abs(diff) > 1e-10:
            # Adjust uniformly
            frac = frac + diff / n
            frac = np.clip(frac, 0.0, 1.0)
        
        # Randomized rounding
        h_r = (rng.rand(n) < frac).astype(float)
        current_sum = int(h_r.sum())
        
        if current_sum > target_sum:
            # Remove excess: pick from those with lowest frac values
            excess = current_sum - target_sum
            ones_idx = np.where(h_r == 1)[0]
            order = np.argsort(frac[ones_idx])[:excess]
            h_r[ones_idx[order]] = 0
        elif current_sum < target_sum:
            # Add needed: pick from those with highest frac values
            needed = target_sum - current_sum
            zeros_idx = np.where(h_r == 0)[0]
            order = np.argsort(-frac[zeros_idx])[:needed]
            h_r[zeros_idx[order]] = 1
        
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best_h = h_r.copy()
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _con_fun(h):
    return np.sum(h) - _g_target

def _con_jac(h):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.astype(float).copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    for _ in range(n_trials):
        frac = np.clip(h_frac, 0.0, 1.0)
        h_r = (rng.rand(n) < frac).astype(float)
        current_sum = int(h_r.sum())
        
        if current_sum > target_sum:
            excess = current_sum - target_sum
            ones_idx = np.where(h_r == 1)[0]
            order = np.argsort(frac[ones_idx])[:excess]
            h_r[ones_idx[order]] = 0
        elif current_sum < target_sum:
            needed = target_sum - current_sum
            zeros_idx = np.where(h_r == 0)[0]
            order = np.argsort(-frac[zeros_idx])[:needed]
            h_r[zeros_idx[order]] = 1
        
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best_h = h_r.copy()
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
