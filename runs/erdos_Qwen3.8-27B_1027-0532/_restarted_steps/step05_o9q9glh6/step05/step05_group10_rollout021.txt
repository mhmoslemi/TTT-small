def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()

        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

        else:

    active_set = set()
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_set = set(np.argsort(corr)[-n_active:].tolist())
    
    consecutive_no_improve = 0
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1

        try:

        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_blend = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h_blend, target)

    idxs = np.arange(n)
    
    for alpha in alphas:
        for it in range(max_iter):
            if time.time() - start > time_limit:
                return best_h, best_score
            
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
            exps = np.exp(shifted)
            weights = exps / np.sum(exps)
            
            # Compute gradient
            grad = np.zeros(n)
            for k_idx in range(len(corr)):
                if weights[k_idx] < 1e-10:
                    continue
                k = k_idx - half_n
                w = weights[k_idx] * dx
                m_lo = max(0, k)
                m_hi = min(n, n + k)
                if m_hi > m_lo:
                    grad[m_lo:m_hi] += w * (1.0 - h[m_lo - k:m_hi - k])

                    h = h_try
                    improved = True
                    break
            
            if not improved:
                break
    
    return best_h, best_score

The problem is:
min_{h} max_k f_k(h)
s.t. 0 ≤ h ≤ 1, sum(h) = target

where f_k(h) = sum_m h[m] * (1-h[m-k]) * dx (for the correlation definition)

This is a min-max problem with bilinear objectives. We can formulate it as:
min_{h,t} t
s.t. f_k(h) ≤ t for all k
     0 ≤ h ≤ 1
     sum(h) = target

This is a non-convex feasible set (because f_k is not concave in h). But we can solve it with a general nonlinear solver like IPOPT or SNOPT (if available).

Actually, let me check: is f_k concave or convex in h?
f_k(h) = sum_m h[m] * (1-h[m-k]) * dx = sum_m h[m] * dx - sum_m h[m]*h[m-k] * dx

The first term is linear (convex and concave). The second term is -sum h[m]*h[m-k], which is a quadratic form. The Hessian of -h^T A h is -2A, where A is a banded matrix. If A is positive semidefinite, then -h^T A h is concave, and f_k is concave. But A is not necessarily PSD.

So f_k is generally non-convex, and the problem is non-convex. We can't use a convex solver directly.

The alternating approach (fix g, solve LP, update) is essentially a coordinate descent on this non-convex problem, which is the standard approach.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_active_constraints(h, n, top_k=40):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return set(np.argsort(corr)[-top_k:].tolist())

def solve_lp(h_current, n, target, active_set):
    """Solve the LP with given active constraints. Returns h_new, t_val."""
    g = 1.0 - h_current
    half_n = n - 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    
    for k_idx in sorted(active_set):
        k = k_idx - half_n
        m_lo = max(0, k)
        m_hi = min(n, n + k)
        if m_hi > m_lo:
            coeff = np.zeros(n)
            coeff[m_lo:m_hi] = g[m_lo - k:m_hi - k]
            if coeff.sum() > 1e-15:
                constraints.append(coeff @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-12, max_iters=80000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=80000)
        except Exception:
            return h_current, float('inf')
    
    if h_var.value is None:
        return h_current, float('inf')
    
    h_new = np.clip(h_var.value, 0.0, 1.0)
    t_val = float(t_var.value) if t_var.value is not None else float('inf')
    return h_new, t_val

def alternating_lp_improve(h_init, n, target, time_limit, n_active=40):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = 0.5
    stall_count = 0
    
    active_set = get_active_constraints(h, n, n_active)
    
    for it in range(50000):
        if time.time() - start > time_limit:
            break
        
        h_lp, t_val = solve_lp(h, n, target, active_set)
        
        h_blend = (1.0 - damping) * h + damping * h_lp
        h = project_feasible(h_blend, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 30:
            if damping < 0.9:
                damping = min(0.95, damping + 0.1)
            else:
                # Try to break out: randomize active set
                active_set = get_active_constraints(h, n, n_active)
                damping = 0.4
            stall_count = 0
        
        # Update active set every 10 iterations
        if it % 10 == 0:
            new_active = get_active_constraints(h, n, n_active)
            # Keep some old constraints for stability
            active_set = (active_set & get_active_constraints(h, n, 2 * n_active)) | new_active
        
        # Occasionally add a random constraint
        if it % 20 == 0 and len(active_set) < 3 * n_active:
            corr = np.correlate(h, 1.0 - h, mode='full')
            random_idx = np.random.randint(n - 1, 2 * n - 1)
            if corr[random_idx] > np.max(corr) * 0.8:
                active_set.add(random_idx)
    
    return best_h, best_c5

def subgradient_analytical(h, n, alpha):
    """Compute the gradient of the smooth max approximation."""
    dx = 2.0 / n
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    if total < 1e-300:
        # Fallback: use the max element
        k_max = int(np.argmax(corr)) - half_n
        grad = np.zeros(n)
        m_lo = max(0, k_max)
        m_hi = min(n, n + k_max)
        if m_hi > m_lo:
            grad[m_lo:m_hi] = (1.0 - h[m_lo - k_max:m_hi - k_max])
        m_lo2 = max(0, -k_max)
        m_hi2 = min(n, n - k_max)
        if m_hi2 > m_lo2:
            grad[m_lo2:m_hi2] -= h[m_lo2 + k_max:m_hi2 + k_max]
        return grad * dx, float(c_max)
    
    weights = exps / total
    grad = np.zeros(n)
    
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - half_n
        wdx = w * dx
        m_lo = max(0, k)
        m_hi = min(n, n + k)
        if m_hi > m_lo:
            grad[m_lo:m_hi] += wdx * (1.0 - h[m_lo - k:m_hi - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n, n - k)
        if m_hi2 > m_lo2:
            grad[m_lo2:m_hi2] -= wdx * h[m_lo2 + k:m_hi2 + k]
    
    obj_val = float(c_max + np.log(total) / alpha)
    return grad, obj_val

def projected_subgradient_improve(h_init, n, target, alphas, steps_per_alpha, time_limit):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        for step in range(steps_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_score
            
            grad, smooth_val = subgradient_analytical(h, n, alpha)
            
            improved = False
            for lr in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                h_try = h - lr * grad
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
                elif sc < smooth_val:  # Accept if it reduces the smooth objective
                    h = h_try
                    improved = True
                    break
            
            if not improved:
                break
    
    return best_h, best_score
