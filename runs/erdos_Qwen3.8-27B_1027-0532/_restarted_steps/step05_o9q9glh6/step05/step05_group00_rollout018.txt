def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

    else:
        idx = np.where(h > 0.0)[0]

        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):

def incremental_swap_delta(h, i, j):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    lo_i = i
    hi_i = min(i + n - 1, 2 * n - 2)
    if lo_i <= hi_i:
        offset = np.arange(lo_i, hi_i + 1) - i
        d_i = 2.0 * h[offset] - 1.0
        ij = i + j
        if lo_i <= ij <= hi_i:
            d_i[ij - lo_i] = -1.0
        delta[lo_i:hi_i + 1] += d_i
    lo_j = j
    hi_j = min(j + n - 1, 2 * n - 2)
    if lo_j <= hi_j:
        offset = np.arange(lo_j, hi_j + 1) - j
        d_j = 1.0 - 2.0 * h[offset]
        ij = i + j
        if lo_j <= ij <= hi_j:
            d_j[ij - lo_j] = 0.0
        delta[lo_j:hi_j + 1] += d_j
    return delta

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    n_options = [n_init + 2, n_init - 2, n_init + 4, n_init - 4]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n_init]
    for n_try in n_options:
        if tleft() < 80:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin = project_binary(h_interp, target_try)

        allowed = min(40, tleft() * 0.08)
        h_g3, c5_g3 = greedy_incremental(h_bin, max_iters=50000, time_limit=allowed,
                                         start_time=start_time, scan_width=30, use_double=True)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

        if tleft() < 50:
            continue
        allowed2 = min(60, tleft() * 0.10)
        max_it2 = int(min(200000, allowed2 * 2000))
        h_sa2, c5_sa2 = sa_incremental(h_g3, max_it2, 0.006, 0.00002, rng,
                                       time_limit=allowed2, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    for n_sym in [604, 608, 612]:
        if tleft() < 100:
            break
        h_sym = symmetric_init(n_sym, rng)
        target_sym = n_sym // 2
        h_sym = project_binary(h_sym, target_sym)

        allowed = min(60, tleft() * 0.10)
        h_gs, c5_gs = symmetric_local_search(h_sym, max_iters=50000, time_limit=allowed,
                                             start_time=start_time, scan_width=30)
        if c5_gs < best_score:
            best_score = c5_gs
            best_h = h_gs.copy()

        if tleft() < 50:
            continue
        allowed2 = min(60, tleft() * 0.08)
        max_it3 = int(min(150000, allowed2 * 2000))
        h_ss, c5_ss = sa_incremental(h_gs, max_it3, 0.005, 0.00001, rng,
                                     time_limit=allowed2, start_time=start_time)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss.copy()

    if tleft() > 80:
        n_p = len(best_h)
        h_p_bin = project_binary(best_h, n_p // 2)
        for pert_n in [2, 4, 6, 8, 12]:
            if tleft() < 40:
                break
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(pert_n):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(30, tleft() * 0.05)
            h_pg, c5_pg = greedy_incremental(h_pert, max_iters=30000, time_limit=allowed,
                                             start_time=start_time, scan_width=25, use_double=True)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

            if tleft() < 30:
                break
            allowed2 = min(25, tleft() * 0.04)
            max_it4 = int(min(100000, allowed2 * 2500))
            h_ps, c5_ps = sa_incremental(h_pg, max_it4, 0.004, 0.00001, rng,
                                         time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 500, 1500, 5000, 15000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-15,
                                 time_limit=min(tleft() - 20, 250), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(60, tleft() - 10)
        max_it5 = int(min(300000, allowed * 2500))
        h_sf, c5_sf = sa_incremental(h_bin_f, max_it5, 0.004, 0.00001, rng,
                                     time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [8000, 20000, 50000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 2000, 1e-16,
                                   time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
