```python
def compute_soft_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    w = np.exp(shifted)
    w_norm = w.sum()
    
    # Combine weights for same |k|
    # w[k] with index: k=0 → m=n-1; k>0 → m=n-1+k and m=n-1-k
    W = np.zeros(n)  # W[k] = combined weight for |lag|=k
    W[0] = w[n-1]
    for k in range(1, n):
        W[k] = w[n-1+k] + w[n-1-k]
    W /= w_norm
    
    grad = np.zeros(n)
    for k in range(n):
        if W[k] < 1e-15:
            continue
        grad[k:] += W[k] * (1.0 - h[:n-k])
        grad[:n-k] -= W[k] * h[k:]
    
    return grad
```

```python
def project_box_and_sum(h, target):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target}."""
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target - h_proj.sum()
    if abs(diff) > 1e-14:
        mask = (h_proj > 0) & (h_proj < 1)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

```python
                lr_init *= 0.5
```

```python
    ms = np.arange(1, n)  # ms = k' = 1, ..., n-1

            b_neg[idx] -= h[i + k_val]
```

```python
                lr *= 0.8

        h_cur = h_try
        alpha = min(alpha * 1.002, 30000.0)

    return best_h, best_score


def compute_B_single(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    for k in range(n):
        if k == 0:
            B[n - 1] = 1.0 - 2.0 * h[i]
        else:
            if k <= i <= n - 1:
                B[n - 1 + k] += (1.0 - h[i - k])
                B[n - 1 - k] += (1.0 - h[i - k])
            if 0 <= i <= n - 1 - k:
                B[n - 1 + k] -= h[i + k]
                B[n - 1 - k] -= h[i + k]
    return B


def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=30):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep = 0

    while time.time() < deadline:
        sweep += 1
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B_single(h_cur, i, n)
            Bj = compute_B_single(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            if not np.any(C > 1e-14) and not np.any(C < -1e-14):
                t_star = (t_min + t_max) * 0.5
            else:
                pos = C > 1e-14
                neg = C < -1e-14

                if np.any(pos):
                    hi = float(np.max(A[pos] + C[pos] * t_max))
                elif np.any(neg):
                    hi = float(np.max(A[neg] + C[neg] * t_max))
                else:
                    hi = float(np.max(A))

                lo = 0.0
                for _ in range(40):
                    mid = (lo + hi) * 0.5
                    ub = t_max
                    lb = t_min
                    if np.any(pos):
                        ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                    if np.any(neg):
                        lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                    if lb <= ub + 1e-11:
                        hi = mid
                    else:
                        lo = mid

                v = hi
                ub = t_max
                lb = t_min
                if np.any(pos):
                    ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
                if np.any(neg):
                    lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
                ub = min(ub, t_max)
                lb = max(lb, t_min)
                if lb > ub:
                    t_star = (t_min + t_max) * 0.5
                else:
                    t_star = lb + (ub - lb) * 0.5

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-14:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 150 and sweep > 50:
            pert = rng.randn(n) * 0.015
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            best_score = float(np.max(corr)) * dx
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: SLSQP with analytic gradient (fast convergence)
    deadline_p1 = start_time + 120
    h1, s1 = slsqp_sequence(best_h, n, target, [100, 500, 2000, 10000], 4000, deadline_p1)
    if s1 < best_score:
        best_score = s1
        best_h = h1

    # Phase 2: Trust-constr with higher alphas
    if tleft() > 200:
        deadline_p2 = start_time + 280
        h2, s2 = tc_sequence(best_h, n, target, [5000, 20000], 2500, deadline_p2)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Projected gradient with momentum (explore)
    if tleft() > 300:
        deadline_p3 = start_time + 550
        h3, s3 = projected_gradient_opt(best_h, n, target, deadline_p3, rng)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Paired coordinate descent (fine-tune)
    if tleft() > 250:
        deadline_p4 = start_time + 900
        h4, s4 = paired_coord_desc(best_h, n, target, deadline_p4, rng, recompute_interval=25)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: SLSQP polish after CD
    if tleft() > 120:
        h5, s5 = slsqp_sequence(best_h, n, target, [10000, 50000], 5000,
                                start_time + budget_s - 100)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Final coordinate descent
    if tleft() > 40:
        deadline_final = start_time + budget_s - 15
        h6, s6 = paired_coord_desc(best_h, n, target, deadline_final, rng, recompute_interval=15)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Final SLSQP squeeze
    if tleft() > 30:
        h7, s7 = slsqp_sequence(best_h, n, target, [100000], 8000,
                                start_time + budget_s - 10)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
