def validate_output(h, score):
    """Validate output with multi-point consistency"""
    for _ in range(10):
        perturbed = h.copy() + np.random.normal(size=len(h)) * 0.05
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if abs(total - (len(h)/2)) > 1e-3:
            perturbed = perturbed * (len(h)/2) / total
        s = compute_c5(perturbed)
        if abs(s - score) > 1e-4: 
            print(f"Warning: validation failed (found worse candidate)")
            return False
    return True
