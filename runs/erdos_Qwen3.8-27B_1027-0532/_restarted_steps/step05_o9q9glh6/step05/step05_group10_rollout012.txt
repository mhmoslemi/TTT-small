from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-12:
        return h
    # Bisection on uniform shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_proj) < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h_proj < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h_proj > 1e-8)[0]
        if len(idx) > 0:
            h_proj[idx] += residual / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def build_constraint_matrix(g, n, dx):
    """Build the (2n-1) x n constraint matrix M where M[k_idx, i] = g[i+k]*dx if valid."""
    half_n = n - 1
    M = np.zeros((2 * n - 1, n))
    for k_idx in range(2 * n - 1):
        k = k_idx - half_n
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end > i_start:
            M[k_idx, i_start:i_end] = g[i_start + k:i_end + k] * dx
    return M

    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    cur_reg = reg
    cur_damping = damping
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if time.time() - start > time_limit - 5:
            break  # safety margin
        
        g = 1.0 - h
        M = build_constraint_matrix(g, n, dx)
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0,
            M @ h_var <= t_var
        ]
        
        objective = t_var + cur_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False
        for solver_name, solver, kwargs in [
            ('CLARABEL', cp.CLARABEL, {'eps_abs': 1e-8, 'eps_rel': 1e-8, 'max_iters': 5000}),
            ('SCS', cp.SCS, {'eps': 1e-8, 'max_iters': 50000}),
            ('ECOS', cp.ECOS, {'max_iter': 50000}),

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
