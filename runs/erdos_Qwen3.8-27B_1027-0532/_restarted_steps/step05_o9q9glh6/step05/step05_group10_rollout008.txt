import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_vector(h):
    """Return the full vector of c_k * dx values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, alpha, target):
    """Log-sum-exp approximation of max_k c_k * dx."""
    c_vals = compute_c5_vector(h)
    c_max = np.max(c_vals)
    # Shift for numerical stability
    shifted = alpha * (c_vals - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_gradient(h, alpha):
    """Gradient of the smooth objective."""
    n = len(h)
    c_vals = compute_c5_vector(h)
    c_max = np.max(c_vals)
    shifted = alpha * (c_vals - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    probs = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Now compute gradient
    # grad[j] = sum_k probs[k] * dc_k/dh[j] * dx
    # dc_k/dh[j] = (1-h[j+k]) * 1[0<=j+k<n] - h[j-k] * 1[0<=j-k<n]
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k_idx in range(len(c_vals)):
        k = k_idx - (n - 1)
        if abs(probs[k_idx]) < 1e-15:
            continue
        # Contribution to grad from this k
        # dc_k/dh[j] = (1-h[j+k]) for j in [0, n-1-k] (if k>=0) or [max(0,-k), n-1] (if k<0)
        #               - h[j-k] for j in [max(0,k), n-1] (if k>=0) or [0, n-1+|k|] (if k<0)
        
        # First term: (1-h[j+k]) for valid j
        if k >= 0:
            j_range1 = range(0, n - k)  # j where j+k < n
            if len(j_range1) > 0:
                grad[:n-k] += probs[k_idx] * dx * (1.0 - h[k:n])
        else:
            j_range1 = range(-k, n)  # j where j+k >= 0
            if len(j_range1) > 0:
                grad[-k:] += probs[k_idx] * dx * (1.0 - h[:n+k])
        
        # Second term: -h[j-k] for valid j
        if k >= 0:
            j_range2 = range(k, n)  # j where j-k >= 0
            if len(j_range2) > 0:
                grad[k:] -= probs[k_idx] * dx * h[:n-k]
        else:
            j_range2 = range(0, n + k)  # j where j-k < n, i.e., j < n+k
            if len(j_range2) > 0:
                grad[:n+k] -= probs[k_idx] * dx * h[-k:]
    
    return grad
