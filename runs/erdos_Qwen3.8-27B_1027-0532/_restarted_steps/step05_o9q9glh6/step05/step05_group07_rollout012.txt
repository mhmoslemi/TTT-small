```python
def solve_1d(A, C, t_lo, t_hi):
    """Minimize max_k (A[k] + C[k]*t) for t in [t_lo, t_hi]."""
    # Precompute
    pos = C > 1e-15
    neg = C < -1e-15
    zero = ~(pos | neg)
    
    # Initial bounds
    M_hi = max(float(np.max(A + C * t_lo)), float(np.max(A + C * t_hi)))
    if np.any(zero):
        M_lo = float(np.max(A[zero]))
    else:
        M_lo = 0.0  # or some lower bound
    
    # Actually, a better lower bound:
    # The optimal M* ≥ max of the minimum over t for each k... no that's not right.
    # M* = min_t max_k (A_k + C_k*t)
    # M* ≥ max_k min_t (A_k + C_k*t) for t in [t_lo, t_hi]
    # For C_k > 0: min_t (A_k + C_k*t) = A_k + C_k*t_lo
    # For C_k < 0: min_t (A_k + C_k*t) = A_k + C_k*t_hi
    # For C_k = 0: min_t (A_k + C_k*t) = A_k
    if np.any(pos):
        lb_vals = A[pos] + C[pos] * t_lo
    else:
        lb_vals = np.array([])
    if np.any(neg):
        lb_vals2 = A[neg] + C[neg] * t_hi
    else:
        lb_vals2 = np.array([])
    all_lb = np.concatenate([lb_vals, lb_vals2, A[zero]]) if (np.any(pos) or np.any(neg)) else A[zero]
    if len(all_lb) > 0:
        M_lo = float(np.max(all_lb))
    else:
        M_lo = 0.0
    
    M_hi = max(float(np.max(A + C * t_lo)), float(np.max(A + C * t_hi)))
    
    # Bisection
    for _ in range(50):
        M_mid = (M_lo + M_hi) * 0.5
        # Check feasibility: exists t in [t_lo, t_hi] such that A_k + C_k*t ≤ M_mid for all k
        lb = t_lo
        ub = t_hi
        if np.any(pos):
            ub = min(ub, float(np.min((M_mid - A[pos]) / C[pos])))
        if np.any(neg):
            lb = max(lb, float(np.max((M_mid - A[neg]) / C[neg])))
        if np.any(zero) and float(np.max(A[zero])) > M_mid:
            M_lo = M_mid
            continue
        if lb <= ub + 1e-14:
            M_hi = M_mid
        else:
            M_lo = M_mid
    
    # Now find t* at M = M_hi
    M = M_hi
    lb = t_lo
    ub = t_hi
    if np.any(pos):
        ub = min(ub, float(np.min((M - A[pos]) / C[pos])))
    if np.any(neg):
        lb = max(lb, float(np.max((M - A[neg]) / C[neg])))
    ub = min(ub, t_hi)
    lb = max(lb, t_lo)
    if lb > ub:
        t_star = (t_lo + t_hi) * 0.5
    else:
        t_star = (lb + ub) * 0.5
    
    return t_star
```

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
