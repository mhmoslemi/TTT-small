            # delta * A_k where delta = v_new - v_old
            # A_k = partial derivative of corr_k with respect to h[j] (which is constant, not depending on v)

            ncorr = 2 * n - 1

                    a = 0.0

                        a += (1.0 - h_cur[j - m])
                    if j <= n - 1 - m:
                        a -= h_cur[j + m]
                    A[ki] = a

            # We want to minimize max_k (corr_old[k] + (v - v_old) * A[k]) subject to 0 <= v <= 1
            # This is minimize max_k (B_k + v * A_k) where B_k = corr_old[k] - v_old * A[k]
            # which is convex in v (max of linear functions)
            # The minimum is at v* where the derivative changes sign or at boundary
            
            B = corr_old - v_old * A
            # f(v) = max_k (B[k] + A[k] * v)
            # f'(v) = max_k (A[k]) at the active k
            # We want f'(v) = 0, i.e., the max A[k] among active k equals 0
            # At v=0: f'(0) = max(A[k] for k where B[k] is max)
            # At v=1: f'(1) = max(A[k] for k where B[k] + A[k] is max)
            
            # Simple bisection on v in [0,1]:
            v_lo, v_hi = 0.0, 1.0
            for _ in range(40):
                v_mid = (v_lo + v_hi) / 2
                fval = np.max(B + A * v_mid)
                # Subgradient at v_mid
                active = np.where(B + A * v_mid >= fval - 1e-14)[0]
                sg = A[active].max()  # max subgradient
                if sg > 1e-14:
                    v_hi = v_mid
                elif sg < -1e-14:
                    v_lo = v_mid
                else:
                    break
            
            v_new = (v_lo + v_hi) / 2
            if abs(v_new - v_old) > 1e-12:
                h_cur[j] = v_new
                # Re-project for sum constraint
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score:
                    cur_score = sc
                    improved = True
        
        if not improved:
            break
    
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]
    
    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def nesterov_optimize(h, target, n_iters=5000, k_top=3, eta_init=0.01):
    """Nesterov accelerated gradient with top-k subgradient and backtracking."""
    n = len(h)
    h_x = h.copy()
    h_y = h.copy()
    cur_score = compute_c5(h_x)
    
    for t in range(n_iters):
        # Compute subgradient at y
        corr_y = np.correlate(h_y, 1.0 - h_y, mode='full')
        top_idx = np.argsort(corr_y)[::-1][:k_top]
        top_vals = corr_y[top_idx]
        # Soft weights
        shifted = 10.0 * (top_vals - top_vals[0])
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                if lag == 0:
                    grad += wi * (1.0 - 2.0 * h_y)
                else:
                    j1 = np.arange(lag, n)
                    grad[j1] += wi * (1.0 - h_y[j1 - lag])
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_y[j2 + lag]
            else:
                m = -lag
                if m == 0:
                    grad += wi * (1.0 - 2.0 * h_y)
                else:
                    j1 = np.arange(0, n - m)
                    grad[j1] += wi * (1.0 - h_y[j1 + m])
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h_y[j2 - m]
        
        # Add constraint gradient
        s = float(np.sum(h_y))
        grad += 2.0 * 50.0 * (s - target)
        
        # Backtracking line search
        eta = eta_init / (1.0 + t * 0.01)
        f_y = compute_c5(h_y)
        for ls_iter in range(20):
            h_trial = h_y - eta * grad
            h_trial = project_feasible(h_trial, target)
            f_trial = compute_c5(h_trial)
            # Armijo condition (loose)
            if f_trial < f_y + 0.01 * eta * np.dot(grad, h_trial - h_y):
                break
            eta *= 0.5
        
        h_new = h_trial
        # Nesterov momentum
        rho = (1.0 + t) / (2.0 + t)
        h_y = h_new + rho * (h_new - h_x)
        h_y = np.clip(h_y, 0.0, 1.0)
        h_x = h_new
        
        sc = compute_c5(h_x)
        if sc < cur_score:
            cur_score = sc
    
    return h_x, cur_score

def fast_coord_descent(h, target, n_sweeps=20, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                break
            # Compute A: sensitivity of each corr to h[j]
            A = np.zeros(ncorr)
            v_old = h_cur[j]
            # For each lag, compute d(corr)/d(h[j])
            # Positive lags L=0..n-1, index = n-1+L
            for L in range(n):
                ki = n - 1 + L
                a = 0.0
                if L < n - j:
                    a += (1.0 - h_cur[j + L])
                if L <= j:
                    a -= h_cur[j - L]
                A[ki] = a
            # Negative lags m=1..n-1, index = n-1-m
            for m in range(1, n):
                ki = n - 1 - m
                a = 0.0
                if m <= j:
                    a += (1.0 - h_cur[j - m])
                if m < n - j:
                    a -= h_cur[j + m]
                A[ki] = a

            corr_prev = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            B = corr_prev - v_old * A

            # Bisection for 1D convex minimization
            v_lo, v_hi = 0.0, 1.0
            for _ in range(25):
                v_mid = 0.5 * (v_lo + v_hi)
                vals = B + A * v_mid
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-13
                sg = A[active].max()
                if sg > 1e-14:
                    v_hi = v_mid
                elif sg < -1e-14:
                    v_lo = v_mid
                else:
                    v_lo = v_hi = v_mid
                    break

            v_new = 0.5 * (v_lo + v_hi)
            if abs(v_new - v_old) > 1e-10:
                h_cur[j] = v_new
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-15:
                    cur_score = sc
                    improved = True

        if not improved:
            break

    return h_cur, cur_score

This is much faster than the previous coordinate descent (which was O(n³) per sweep due to golden section search with O(n) function evaluations each costing O(n²)).

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


class LSEObjective:
    def __init__(self, alpha, target, mu):
        self.alpha = alpha
        self.target = target
        self.mu = mu

    def __call__(self, x):
        n = len(x)
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = self.alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / self.alpha
        s = float(np.sum(x))
        val += self.mu * (s - self.target) ** 2
        return val

    def gradient(self, x):
        n = len(x)
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = self.alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum > 1e-300:
            weights = weights / wsum
        else:
            weights = np.ones(len(corr)) / len(corr)
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for ki in active:
            w = weights[ki]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * x[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - x[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * x[j2 - m]
        s = float(np.sum(x))
        grad += 2.0 * self.mu * (s - self.target)
        return grad


def lbfgs_step(h, target, alpha, mu, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    obj = LSEObjective(alpha, target, mu)
    try:
        res = minimize(obj, h_cur, jac=obj.gradient, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def upsample_cubic(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    cs = CubicSpline(x_old, h_old)
    h_new = cs(x_new)
    return project_feasible(h_new, n_new / 2.0)
