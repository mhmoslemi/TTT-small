```python
def smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    w = weights / Z  # shape (2n-1,)
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if w[m] < 1e-16:
            continue
        k = m - (n-1)
        # ∂f(k)/∂h[j] = [k ≤ j < k+n]*(1-h[j-k]) - [max(0,-k) ≤ j < min(n,n-k)]*h[j+k]
        # First term: j from max(0,k) to min(n, k+n)-1
        j_start1 = max(0, k)
        j_end1 = min(n, k + n)
        if j_start1 < j_end1:
            # j-k ranges from j_start1-k to j_end1-k-1
            # = max(0,k)-k to min(n,k+n)-k-1
            # = max(-k, 0) to min(n-k, n)-1
            idx1 = np.arange(j_start1, j_end1)
            grad[j_start1:j_end1] += w[m] * (1.0 - h[idx1 - k])
        
        # Second term: j from max(0,-k) to min(n, n-k)-1
        j_start2 = max(0, -k)
        j_end2 = min(n, n - k)
        if j_start2 < j_end2:
            idx2 = np.arange(j_start2, j_end2)
            grad[j_start2:j_end2] -= w[m] * h[idx2 + k]
    
    return grad
```

```python
            break
```

```python
            break

        h = np.clip(h, 0.0, 1.0)

    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha
```
