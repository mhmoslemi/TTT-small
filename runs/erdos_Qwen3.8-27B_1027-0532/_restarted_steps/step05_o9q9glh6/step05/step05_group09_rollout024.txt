    n = len(h)

    return float(np.max(corr)) * dx

def project_feasible(h, target):
    # Water-filling projection onto {x: 0<=x<=1, sum(x)=target}
    h = h.astype(float).copy()
    n = len(h)
    # Binary search for tau such that sum(clip(h - tau, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    """Compute log-sum-exp objective and gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))

    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)

    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)

    corr_neg = np.fft.irfft(W * H, fsize)[:n]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(exp_vals.sum())
    weights = exp_vals / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # FFT-based gradient
    fsize = 2 * n
    w_flipped = weights[::-1]
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def subgrad_polish(h, target, deadline, n_top=10, max_rounds=500):
    """Subgradient method on the top lags."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_sc = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_max = float(np.max(corr))
        
        # Find lags within a small tolerance of max
        tol = 0.001 * c_max + 1e-8
        active = np.where(corr >= c_max - tol)[0]
        if len(active) > n_top:
            # Take the top n_top
            top_idx = np.argsort(corr)[::-1][:n_top]
            active = top_idx
        
        # Compute weighted subgradient
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / 0.01)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            if ki > 0:
                jn = np.arange(ki, n)
                grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        # Try decreasing step sizes
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_sc - 1e-15:
                h_cur = h_new
                cur_sc = sc
                break
    
    return h_cur, cur_sc
