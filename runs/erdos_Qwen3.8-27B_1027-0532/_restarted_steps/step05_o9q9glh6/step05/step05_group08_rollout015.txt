    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)

        g[k:] -= wk * h[:n-k]

        if wk < 1e-12:
            continue
        g[ell:] += wk * (1.0 - h[:n-ell])
        g[:n-ell] -= wk * h[ell:]
    g *= dx

    def obj(x):
        return lse_value(x, alpha)
    def objg(x):
        return lse_grad(x, alpha, n)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def lbfgsb_optimize(h, n, target, alpha, mu, maxiter=300):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha, n) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

    top_ix = np.argsort(corr)[::-1][:m]
    c_top = corr[top_ix]
    c_max = float(c_top.max())
    shifted = np.clip(alpha * (c_top - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def top_m_lse_grad(h, alpha, m, n):
    corr = compute_corr(h)
    top_ix = np.argsort(corr)[::-1][:m]
    c_top = corr[top_ix]
    c_max = float(c_top.max())
    shifted = np.clip(alpha * (c_top - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m_idx in zip(w, top_ix):
        m_val = int(m_idx)
        k = m_val - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    g *= dx
    return g
