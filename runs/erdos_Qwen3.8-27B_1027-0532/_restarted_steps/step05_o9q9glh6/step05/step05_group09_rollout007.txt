import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    """Compute all correlation values c(k) for k=0..n-1"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr

def project_feasible(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    h = h.astype(float).copy()
    n = len(h)
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Then adjust to meet sum constraint using bisection on the shift
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Bisection for the water-filling level
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h - mid, 0.0, 1.0)
        s_try = np.sum(h_try)
        if s_try > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def gradient_topk(h, corr, top_indices):
    """Compute gradient of max over top lags (as weighted average)"""
    n = len(h)
    dx = 2.0 / n
    c_max = corr[top_indices[0]]
    # Soft weights
    diffs = corr[top_indices] - c_max
    w = np.exp(diffs / 0.01)  # sigma=0.01 for sharp peak selection
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, ki in zip(w, top_indices):
        ki = int(ki)
        if ki == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            # ∂c(k)/∂h[j] = (1-h[j+k])*I(j<=n-1-k) - h[j-k]*I(j>=k)
            # Vectorized:
            # First term: (1-h[j+k]) for j=0..n-1-k
            j_pos = np.arange(n - ki)
            grad[j_pos] += wi * (1.0 - h[j_pos + ki])
            # Second term: -h[j-k] for j=k..n-1
            j_neg = np.arange(ki, n)
            grad[j_neg] -= wi * h[j_neg - ki]
    grad *= dx
    return grad

def compute_corr(h):
