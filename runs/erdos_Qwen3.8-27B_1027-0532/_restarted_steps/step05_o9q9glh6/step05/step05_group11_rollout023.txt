import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    """Build the constraint matrix A (2n-1 × n) for all lags.
    Row k (0-indexed) corresponds to lag k-(n-1).
    A[k, i] = g[i + lag] * dx if valid, else 0.
    """
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for lag in range(-(n-1), n):
        row_idx = lag + (n - 1)
        if lag >= 0:
            # A[row_idx, i] = g[i+lag] * dx for i = 0, ..., n-1-lag
            A[row_idx, :n-lag] = g[lag:] * dx
        else:
            # A[row_idx, i] = g[i+lag] * dx for i = -lag, ..., n-1
            A[row_idx, -lag:] = g[:n+lag] * dx
    return A

def full_lp_iteration(h, n, target, dx, reg, h_current):
    """Solve one LP iteration with all lags."""
    g = 1.0 - h
    A = build_full_lp_matrix(g, dx, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
        A @ h_var <= t_var
    ]
    
    objective = t_var + reg * cp.sum_squares(h_var - h_current)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
    
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def full_lp_solve_csc(g, dx, n, target):
    """Solve the full LP using scipy linprog.
    min t
    s.t. A @ h <= t  (for all lags)
         sum(h) = target
         0 <= h <= 1
    """
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ h - t <= 0 for each lag
    # A is (2n-1) x n
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    # A_ub[i, :n] = row i of A (coefficient of h)
    # A_ub[i, n] = -1 (coefficient of t)
    
    g_row = g  # 1 - h
    for lag in range(-(n-1), n):
        row_idx = lag + (n - 1)
        A_ub[row_idx, -1] = -1.0
        if lag >= 0:
            A_ub[row_idx, :n-lag] = g_row[lag:] * dx
        else:
            A_ub[row_idx, -lag:] = g_row[:n+lag] * dx
    
    # h <= 1
    A_ub_h = np.hstack([np.eye(n), np.zeros((n, 1))])
    b_ub_h = np.ones(n)
    
    A_ub_full = np.vstack([A_ub, A_ub_h])
    b_ub_full = np.concatenate([np.zeros(n_lags), b_ub_h])
    
    # h >= 0 -> -h <= 0
    A_ub_neg = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_ub_neg = np.zeros(n)
    
    A_ub_final = np.vstack([A_ub_full, A_ub_neg])
    b_ub_final = np.concatenate([b_ub_full, b_ub_neg])
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub_final, b_ub=b_ub_final, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
