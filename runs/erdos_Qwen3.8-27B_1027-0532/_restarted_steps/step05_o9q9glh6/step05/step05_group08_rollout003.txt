```python
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_fourier_matrix(n, K):
    dx = 2.0 / n
    x = np.arange(n) * dx
    F = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        F[:, 2*(k-1)] = np.cos(k * np.pi * x)
        F[:, 2*(k-1)+1] = np.sin(k * np.pi * x)
    return F

def h_from_coeffs(v, F):
    return 0.5 + F @ v

def lse_corr_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_corr_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

        wk = w[n - 1 + k]

        if wk < 1e-13:

        g[ell:] += wk * (1.0 - h[:n - ell])

    g *= dx
    return g

def fourier_obj(v, F, alpha):
    h = h_from_coeffs(v, F)
    return lse_corr_value(h, alpha)

def fourier_grad(v, F, alpha):
    h = h_from_coeffs(v, F)
    gh = lse_corr_grad(h, alpha)
    return F.T @ gh

def fourier_optimize(v_init, F, n, K, alpha, deadline):

    # i.e., F@v >= -0.5 and -F@v >= -0.5
    A = np.vstack([F, -F])
    b = np.hstack([0.5 * np.ones(n), 0.5 * np.ones(n)])
    
    # For SLSQP, we need to be careful with many constraints
    # Let's use a subset of constraints (sample)
    if n > 200:
        indices = np.random.choice(n, 200, replace=False)
        A_sub = A[indices]
        b_sub = b[indices]
    else:
        A_sub = A
        b_sub = b
    
    try:
        res = minimize(
            lambda v: fourier_obj(v, F, alpha),
            v_init,
            method='SLSQP',
            jac=lambda v: fourier_grad(v, F, alpha),
            bounds=[(None, None)] * len(v_init),
            constraints=[{'type': 'ineq', 'fun': lambda v: A_sub @ v - b_sub + 1e-10, 'jac': lambda v: A_sub}],
            options={'maxiter': 3000, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return v_init
```

```python
def box_penalty(h, mu, alpha_pen):
    # Smooth approximation of box constraint violation
    upper = np.logaddexp(0, alpha_pen * (h - 1.0))  # softplus(h-1)
    lower = np.logaddexp(0, -alpha_pen * h)  # softplus(-h)
    return mu * np.sum(upper + lower)

def box_penalty_grad(h, mu, alpha_pen):
    upper_grad = alpha_pen / (1.0 + np.exp(-alpha_pen * (h - 1.0)))  # sigmoid(h-1) * alpha
    lower_grad = -alpha_pen / (1.0 + np.exp(alpha_pen * h))  # -sigmoid(-h) * alpha... wait
    # d/dh log(1 + exp(alpha*(h-1))) = alpha * exp(alpha*(h-1)) / (1 + exp(alpha*(h-1))) = alpha / (1 + exp(-alpha*(h-1)))
    # d/dh log(1 + exp(-alpha*h)) = -alpha * exp(-alpha*h) / (1 + exp(-alpha*h)) = -alpha / (1 + exp(alpha*h))
    return mu * (upper_grad + lower_grad)
```

Wait, let me be more careful:
- f(x) = log(1 + exp(x)), f'(x) = exp(x)/(1+exp(x)) = 1/(1+exp(-x)) = sigmoid(x)
- g(h) = log(1 + exp(α(h-1))), g'(h) = α * sigmoid(α(h-1))
- k(h) = log(1 + exp(-αh)), k'(h) = -α * exp(-αh)/(1+exp(-αh)) = -α / (1+exp(αh)) = -α * (1 - sigmoid(αh))

So:
- grad penalty = μ * (α * sigmoid(α(h-1)) - α * (1 - sigmoid(αh)))

This is smooth and bounded. Good.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_fourier_matrix(n, K):
    dx = 2.0 / n
    x = np.arange(n) * dx
    F = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        F[:, 2*(k-1)] = np.cos(k * np.pi * x)
        F[:, 2*(k-1)+1] = np.sin(k * np.pi * x)
    return F

def lse_value(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def corr_grad_from_weights(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def softplus(x):
    return np.logaddexp(0.0, x)

def softplus_grad(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -500, 500)))

def box_pen_value(h, mu, ap):
    upper = softplus(ap * (h - 1.0))
    lower = softplus(-ap * h)
    return mu * float(np.sum(upper + lower))

def box_pen_grad(h, mu, ap):
    gu = ap * softplus_grad(ap * (h - 1.0))
    gl = -ap * softplus_grad(-ap * h)
    return mu * (gu + gl)

def fourier_eval(v, F, alpha, mu, ap):
    h = 0.5 + F.dot(v)
    corr = compute_corr(h)
    val = lse_value(corr, alpha) + box_pen_value(h, mu, ap)
    return val

def fourier_grad(v, F, alpha, mu, ap):
    h = 0.5 + F.dot(v)
    corr = compute_corr(h)
    w = lse_weights(corr, alpha)
    gh = corr_grad_from_weights(h, w) + box_pen_grad(h, mu, ap)
    return F.T.dot(gh)

def fourier_optimize(v_init, F, alpha, mu, ap, maxiter):
    def obj(v):
        return fourier_eval(v, F, alpha, mu, ap)
    def objg(v):
        return fourier_grad(v, F, alpha, mu, ap)
    try:
        res = minimize(obj, v_init, method='L-BFGS-B', jac=objg,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e10})
        return res.x
    except Exception:
        return v_init

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 16
        top_idx = np.argsort(corr)[::-1][:top_k]
        base = corr[top_idx[0]]
        diffs = corr[top_idx] - base
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:

    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.012)
        else:
            eps *= 0.85
            if eps < 1e-10:

    return h_cur, best_score

def grid_lbfbs(h, n, target, alpha, mu, maxiter, deadline):
    def obj(x):
        return lse_value(compute_corr(x), alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        corr = compute_corr(x)
        w = lse_weights(corr, alpha)
        gh = corr_grad_from_weights(x, w)
        return gh + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_fourier_matrix(n, K):
    dx = 2.0 / n
    x = np.arange(n) * dx
    F = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        F[:, 2 * (k - 1)] = np.cos(k * np.pi * x)
        F[:, 2 * (k - 1) + 1] = np.sin(k * np.pi * x)
    return F

def lse_value(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def corr_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def softplus_grad(x):
    x = np.clip(x, -500.0, 500.0)
    return 1.0 / (1.0 + np.exp(-x))

def fourier_eval(v, F, alpha, mu, ap):
    h = 0.5 + F.dot(v)
    corr = compute_corr(h)
    val = lse_value(corr, alpha)
    if mu > 0:
        upper = np.logaddexp(0.0, ap * (h - 1.0))
        lower = np.logaddexp(0.0, -ap * h)
        val += mu * float(np.sum(upper + lower))
    return val

def fourier_grad(v, F, alpha, mu, ap):
    h = 0.5 + F.dot(v)
    corr = compute_corr(h)
    w = lse_weights(corr, alpha)
    gh = corr_grad(h, w)
    if mu > 0:
        gu = ap * softplus_grad(ap * (h - 1.0))
        gl = -ap * softplus_grad(-ap * h)
        gh += mu * (gu + gl)
    return F.T.dot(gh)

def grid_eval(x, alpha, mu, target):

    return lse_value(corr, alpha) + mu * (np.sum(x) - target) ** 2

def grid_grad(x, alpha, mu, target):

    w = lse_weights(corr, alpha)
    gh = corr_grad(x, w)
    return gh + 2.0 * mu * (np.sum(x) - target)
