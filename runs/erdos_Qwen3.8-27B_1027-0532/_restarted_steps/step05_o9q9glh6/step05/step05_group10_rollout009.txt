def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()

        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]

        return h

    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_minimax(A, B, lo=0.0, hi=1.0):
    active_lo_vals = A * lo + B
    f_lo = np.max(active_lo_vals)
    active_lo = active_lo_vals >= f_lo - 1e-12
    active_hi_vals = A * hi + B
    f_hi = np.max(active_hi_vals)
    active_hi = active_hi_vals >= f_hi - 1e-12
    
    if np.min(A[active_lo]) >= -1e-12:
        return lo
    if np.max(A[active_hi]) <= 1e-12:
        return hi
    
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = A * mid + B
        fval = np.max(vals)
        active = vals >= fval - 1e-12
        A_act = A[active]
        if np.max(A_act) < -1e-14:
            hi = mid
        elif np.min(A_act) > 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

    n = len(h_init)
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    result = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    
    if start_time is None:
        start_time = time.time()
    
    # Precompute valid index patterns for each coordinate
    # For coordinate i, the valid shifts j are those where i+j or i-j is in [0,n)
    # valid1[i] = bool array of length 2*half_n+1, valid1[i][k] = (0 <= i + (k-half_n) < n)
    # valid2[i] = bool array, valid2[i][k] = (0 <= i - (k-half_n) < n)
    # js[k] = k - half_n for k in range(2*half_n+1)
    
    js = np.arange(-half_n, half_n + 1)
    
    no_improve = 0
    patience = 50
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        for i in range(0, n - 1, 2):
            j = i + 1
            s = h[i] + h[j]
            
            # Compute A^i and A^j
            valid1_i = (i + js >= 0) & (i + js < n)
            valid2_i = (i - js >= 0) & (i - js < n)
            valid1_j = (j + js >= 0) & (j + js < n)
            valid2_j = (j - js >= 0) & (j - js < n)
            
            Ai = np.zeros(2 * half_n + 1)
            Ai[valid1_i] = 1.0 - h[i + js[valid1_i]]
            Ai[valid2_i] -= h[i - js[valid2_i]]
            
            Aj = np.zeros(2 * half_n + 1)
            Aj[valid1_j] = 1.0 - h[j + js[valid1_j]]
            Aj[valid2_j] -= h[j - js[valid2_j]]
            
            # Combined: result[k] = (Ai - Aj) * x + Aj*s + (result - h[i]*Ai - h[j]*Aj)
            # where x = h[i], h[j] = s - x
            combined_A = Ai - Aj
            B_rest = result - h[i] * Ai - h[j] * Aj
            combined_B = Aj * s + B_rest
            
            # Bounds: max(0, s-1) <= x <= min(1, s)
            x_lo = max(0.0, s - 1.0)
            x_hi = min(1.0, s)
            
            if x_lo >= x_hi:
                x_star = x_lo
            else:
                x_star = solve_1d_minimax(combined_A, combined_B, x_lo, x_hi)
            
            delta_i = x_star - h[i]
            delta_j = (s - x_star) - h[j]
            
            if abs(delta_i) > 1e-15 or abs(delta_j) > 1e-15:

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            break

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_minimax(A, B, lo=0.0, hi=1.0):
    vals_lo = A * lo + B
    f_lo = np.max(vals_lo)
    active_lo = vals_lo >= f_lo - 1e-12
    vals_hi = A * hi + B
    f_hi = np.max(vals_hi)
    active_hi = vals_hi >= f_hi - 1e-12
    if np.min(A[active_lo]) >= -1e-12:
        return lo
    if np.max(A[active_hi]) <= 1e-12:
        return hi
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = A * mid + B
        fval = np.max(vals)
        active = vals >= fval - 1e-12
        A_act = A[active]
        if np.max(A_act) < -1e-14:
            hi = mid
        elif np.min(A_act) > 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5


def ebcd_pairwise(h_init, target_sum, max_sweeps, time_limit, start_time, offset=0):
    n = len(h_init)
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    result = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    js = np.arange(-half_n, half_n + 1)
    size = 2 * half_n + 1

    valid1_cache = np.zeros((n, size), dtype=np.int32)
    valid2_cache = np.zeros((n, size), dtype=np.int32)
    for i in range(n):
        v1 = (i + js >= 0) & (i + js < n)
        v2 = (i - js >= 0) & (i - js < n)
        valid1_cache[i] = np.where(v1, i + js, -1).astype(np.int32)
        valid2_cache[i] = np.where(v2, i - js, -1).astype(np.int32)

    no_improve = 0
    patience = 80

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        order = list(range(offset % 2, n - 1, 2))
        for pi, i in enumerate(order):
            j = i + 1
            if j >= n:
                break
            s = h[i] + h[j]
            idx1_i = valid1_cache[i]
            idx2_i = valid2_cache[i]
            idx1_j = valid1_cache[j]
            idx2_j = valid2_cache[j]

            Ai = np.zeros(size)
            m1 = idx1_i >= 0
            if np.any(m1):
                Ai[m1] = 1.0 - h[idx1_i[m1]]
            m2 = idx2_i >= 0
            if np.any(m2):
                Ai[m2] -= h[idx2_i[m2]]

            Aj = np.zeros(size)
            m1j = idx1_j >= 0
            if np.any(m1j):
                Aj[m1j] = 1.0 - h[idx1_j[m1j]]
            m2j = idx2_j >= 0
            if np.any(m2j):
                Aj[m2j] -= h[idx2_j[m2j]]

            combined_A = Ai - Aj
            B_rest = result - h[i] * Ai - h[j] * Aj
            combined_B = Aj * s + B_rest

            x_lo = max(0.0, s - 1.0)
            x_hi = min(1.0, s)
            if x_lo >= x_hi - 1e-15:
                x_star = x_lo
            else:
                x_star = solve_1d_minimax(combined_A, combined_B, x_lo, x_hi)

            delta_i = x_star - h[i]
            delta_j = (s - x_star) - h[j]
            if abs(delta_i) > 1e-15 or abs(delta_j) > 1e-15:
                h[i] = x_star
                h[j] = s - x_star
                result += delta_i * Ai + delta_j * Aj

        if sweep % 50 == 49:
            result = np.correlate(h, 1.0 - h, mode='full')

        current_c5 = float(np.max(result)) * dx
        if current_c5 < best_c5 - 1e-15:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            break

    return best_h, best_c5


def ebcd_single(h_init, target_sum, max_sweeps, time_limit, start_time):
    n = len(h_init)
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    result = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    js = np.arange(-half_n, half_n + 1)
    size = 2 * half_n + 1

    valid1_cache = np.zeros((n, size), dtype=np.int32)
    valid2_cache = np.zeros((n, size), dtype=np.int32)
    for i in range(n):
        v1 = (i + js >= 0) & (i + js < n)
        v2 = (i - js >= 0) & (i - js < n)
        valid1_cache[i] = np.where(v1, i + js, -1).astype(np.int32)
        valid2_cache[i] = np.where(v2, i - js, -1).astype(np.int32)

    no_improve = 0
    patience = 60

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        for i in range(n):
            idx1_i = valid1_cache[i]
            idx2_i = valid2_cache[i]
            Ai = np.zeros(size)
            m1 = idx1_i >= 0
            if np.any(m1):
                Ai[m1] = 1.0 - h[idx1_i[m1]]

        h = project_feasible(h, target_sum)
        if sweep % 30 == 29:

        current_c5 = float(np.max(result)) * dx

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.4]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 100:
            break
    return best_h, best_c5
