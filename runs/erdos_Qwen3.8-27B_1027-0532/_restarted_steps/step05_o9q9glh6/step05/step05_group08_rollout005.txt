import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for kk in range(n):
        wk = w[n - 1 + kk]
        if wk < 1e-14:
            continue
        g[:n - kk] += wk * (1.0 - h[kk:])
        g[kk:] -= wk * h[:n - kk]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def slsqp_optimize(h, n, target, alpha, maxiter, ftol):
    def obj(x):
        return lse_value(x, alpha)

    def grad(x):
        return lse_grad(x, alpha)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgsb_optimize(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def nesterov_subgrad(h, n, target, n_steps, top_k, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        max_val = corr[top_idx[0]]
        diffs = corr[top_idx] - max_val
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        v = 0.9 * v + (1.0 - 0.9) * g
        h_new = h_cur - lr * v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
        else:
            lr *= lr_decay
            v = 0.5 * v
            if lr < 1e-12:
                lr = lr_init * 0.1
        if step % 50 == 0 and lr < lr_init * 0.01:
            lr = lr_init * 0.5
    return best_h, best_score


def line_search_surgery(h, n, target, max_iters, deadline):

    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > deadline:

        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))

            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:

        g = g / g_norm
        best_alpha = 0.0
        best_sc = compute_c5(h_cur)
        alphas = [0.01, 0.005, 0.002, 0.001, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]

            h_try = project_feasible(h_cur - a * g, target)
            sc_try = compute_c5(h_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_alpha = a
        if best_alpha > 0:
            h_cur = project_feasible(h_cur - best_alpha * g, target)
            if best_sc < best_score:
                best_score = best_sc
        else:
            pos_mask = g < 0
            neg_mask = g > 0
            h_try2 = project_feasible(h_cur + 0.001 * g, target)
            sc_try2 = compute_c5(h_try2)
            if sc_try2 < best_score - 1e-14:
                h_cur = h_try2
                best_score = sc_try2
            else:
                break
    return h_cur, best_score


def local_search(h, n, target, max_passes, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    rng_local = np.random.RandomState(12345)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        improved = False
        n_trials = min(5000, max(1000, n * n // 40))
        for _ in range(n_trials):
            a = rng_local.randint(n)
            b = rng_local.randint(n)
            if a == b or abs(h_cur[a] - h_cur[b]) < 1e-8:
                continue
            pa = h_cur[a]
            pb = h_cur[b]
            h_cur[a] = pb
            h_cur[b] = pa
            sc_new = compute_c5(h_cur)
            if sc_new < best_score - 1e-14:
                best_score = sc_new
                improved = True
            else:
                h_cur[a] = pa
                h_cur[b] = pb
        if not improved:
            break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def optimize_at_resolution(h, n, target, deadline, budget_frac):
    best_h = h.copy()
    best_score = compute_c5(h)
    phase_deadline = deadline - budget_frac * 60 if budget_frac > 0 else deadline

    alphas_lbfgs = [20, 80, 300, 1000, 3000, 10000]
    for alpha in alphas_lbfgs:
        if time.time() > phase_deadline - 120:
            break
        for mu in [10.0, 100.0]:
            h_new = lbfgsb_optimize(best_h, n, target, alpha, mu, 200)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new

    alphas_slsqp = [200, 1000, 5000, 20000]
    for alpha in alphas_slsqp:
        if time.time() > phase_deadline - 80:
            break
        h_new = slsqp_optimize(best_h, n, target, alpha, 3000, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new

    if time.time() < phase_deadline - 40:
        for temp in [0.01, 0.003, 0.001]:
            if time.time() > phase_deadline - 20:
                break
            h_sg, sc_sg = nesterov_subgrad(best_h, n, target, 800, 5, temp, 0.003, 0.998, phase_deadline - 15)
            if sc_sg < best_score - 1e-14:
                best_score = sc_sg
                best_h = h_sg

    if time.time() < phase_deadline - 10:
        h_surg, sc_surg = line_search_surgery(best_h, n, target, 300, phase_deadline - 5)
        if sc_surg < best_score - 1e-14:
            best_score = sc_surg
            best_h = h_surg

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Optimize at original resolution
    remaining = deadline - time.time()
    h_opt, sc_opt = optimize_at_resolution(best_h, n, target, deadline, max(0, remaining / 300 - 1))
    if sc_opt < best_score:
        best_score = sc_opt
        best_h = h_opt

    # Phase 2: Try n=600
    if time.time() < deadline - 200:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h2_opt, sc2 = optimize_at_resolution(h2, n2, target2, deadline, max(0, (deadline - time.time()) / 300 - 1))
        if sc2 < best_score:
            best_score = sc2
            best_h = h2_opt

    # Phase 3: Try n=800 if still time
    if time.time() < deadline - 180:
        n3 = 800
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        remaining3 = deadline - time.time() - 30
        h3_opt, sc3 = optimize_at_resolution(h3, n3, target3, deadline, max(0, remaining3 / 300 - 1))
        if sc3 < best_score:
            best_score = sc3
            best_h = h3_opt

    # Phase 4: Local search at best resolution
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_ls, sc_ls = local_search(best_h, n_cur, target_cur, 10, deadline - 15)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin = slsqp_optimize(best_h, n_cur, target_cur, 20000, 1500, 1e-16)
        sc_fin = compute_c5(h_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 6: Final surgery
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_surg, sc_surg = line_search_surgery(best_h, n_cur, target_cur, 100, deadline - 2)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
