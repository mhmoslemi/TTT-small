import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_value(h, T):
    corr = compute_corr(h)
    max_c = float(corr.max())
    shifted = (corr - max_c) / T
    exps = np.exp(shifted)
    return float(max_c + T * np.log(np.sum(exps)))

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = float(corr.max())
    shifted = (corr - max_c) / T
    exps = np.exp(shifted)
    p = exps / exps.sum()
    first = np.zeros(n)
    second = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j_s = max(0, -k)
        j_e = min(n - 1, n - 1 - k)
        if j_s <= j_e:

        j_s2 = max(0, k)
        j_e2 = min(n - 1, n - 1 + k)
        if j_s2 <= j_e2:
            second[j_s2:j_e2 + 1] += p[m] * h[j_s2 - k:j_e2 - k + 1]
    return dx * (first - second)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff2 = target_sum - np.sum(result)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result = result.copy()
            result[idx] += diff2 / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_swap_delta(h, p, q, n):
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    m_s = max(0, n - 1 - p)
    m_e = min(2 * n - 2, 2 * n - 2 - p)
    if m_s <= m_e:

        delta[m_s:m_e + 1] -= (1.0 - h[idx])
    m_s = max(0, p)
    m_e = min(2 * n - 2, p + n - 1)
    if m_s <= m_e:

        delta[m_s:m_e + 1] += h[idx]
    m_s = max(0, n - 1 - q)
    m_e = min(2 * n - 2, 2 * n - 2 - q)
    if m_s <= m_e:

    if m_s <= m_e:
        idx = q - np.arange(m_s, m_e + 1) + (n - 1)
        delta[m_s:m_e + 1] -= h[idx]

def lbfgs_continuous(h0, n, target, T, penalty, maxiter, time_limit, start_time):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        val = softmax_value(h, T)
        s = np.sum(h) - target
        val += penalty * s * s
        return val

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        val = softmax_value(h, T)
        s = np.sum(h) - target
        val += penalty * s * s
        g = softmax_gradient(h, T)
        g += 2.0 * penalty * s * np.ones(n)
        return val, g

    try:
        res = minimize(obj_grad, h0, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': maxiter * 2})
        h_new = project_feasible(res.x, target)
        score = compute_c5(h_new)
        return h_new, score
    except Exception:
        return h0.copy(), compute_c5(h0)

def binary_greedy_fast(h, rng, time_limit, start_time, T=0.01, max_iters=200000):

    best_h = h.copy()
    stall = 0
    max_stall = 500
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        idx_max = int(np.argmax(corr))
        if idx_max == n - 1:
            stall += 1
            continue
        i_scores = -g[ones]
        j_scores = g[zeros]
        top_i = ones[np.argsort(i_scores)[::-1][:min(40, len(ones))]]
        top_j = zeros[np.argsort(j_scores)[:min(40, len(zeros))]]

def binary_sa(h, rng, t_init, t_final, n_iters, time_limit, start_time, T_grad=0.01):

        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 7 == 0:

            if len(i_cands) == 0:
                i_cands = ones
            if len(j_cands) == 0:
                j_cands = zeros
            i = i_cands[rng.randint(len(i_cands))]
            j = j_cands[rng.randint(len(j_cands))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_swap_delta(h, i, j, n)

        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def binary_2swap(h, rng, time_limit, start_time, T_grad=0.01, max_iters=50000):
    n = len(h)
    h = h.copy().astype(float)
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        g = softmax_gradient(h, T_grad)
        i1_cands = ones[np.argsort(-g[ones])[:min(10, len(ones))]]
        i2_cands = ones[np.argsort(-g[ones])[:min(10, len(ones))]]
        j1_cands = zeros[np.argsort(g[zeros])[:min(10, len(zeros))]]
        j2_cands = zeros[np.argsort(g[zeros])[:min(10, len(zeros))]]
        found = False
        for i1 in i1_cands:
            for j1 in j1_cands[:min(5, len(j1_cands))]:
                if i1 == j1:
                    continue
                for i2 in i2_cands[:min(5, len(i2_cands))]:
                    if i2 == i1 or i2 == j1:
                        continue
                    for j2 in j2_cands[:min(5, len(j2_cands))]:
                        if j2 == i1 or j2 == j1 or j2 == i2:
                            continue
                        delta = (compute_swap_delta(h, i1, j1, n) +
                                 compute_swap_delta(h, i2, j2, n))
                        new_max = float((corr + delta).max())
                        if new_max < float(corr.max()):
                            h[i1] = 0.0
                            h[i2] = 0.0
                            h[j1] = 1.0
                            h[j2] = 1.0
                            corr += delta
                            stall = 0
                            c5_new = float(corr.max()) * dx
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h.copy()
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def lsqp_smooth(h_start, n, target, T, penalty, maxiter, time_limit, start_time):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        val = softmax_value(h, T)
        s = np.sum(h) - target
        val += penalty * s * s
        return val

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        val = softmax_value(h, T)
        s = np.sum(h) - target
        val += penalty * s * s
        g = softmax_gradient(h, T)
        g += 2.0 * penalty * s * np.ones(n)
        return val, g

    try:
        res = minimize(obj_grad, h_start, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init / 2.0)
    target_init_int = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init_int)
    best_score = compute_c5(best_h)

    phase1_end = start_time + budget_s * 0.20
    phase2_end = start_time + budget_s * 0.55
    phase3_end = start_time + budget_s * 0.80

    # Phase 1: Continuous optimization with L-BFGS-B
    T_list = [0.05, 0.02, 0.01, 0.005, 0.002]
    penalty = 500.0
    h_cont = h_init.copy()
    for T in T_list:
        if time.time() > phase1_end:
            break
        remaining = phase1_end - time.time()
        if remaining < 10:
            break
        maxiter = int(min(500, remaining * 30))
        h_new, sc = lbfgs_continuous(h_cont, n_init, target_init, T, penalty,
                                     maxiter, remaining * 0.8, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cont = h_new

    # Binary local search from continuous solution
    h_bin = project_binary(best_h, target_init_int)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin.copy()

    # Phase 2: Binary local search with multiple restarts
    n_restarts = 8
    restart_time = max(5.0, (phase2_end - time.time()) / n_restarts)
    for r in range(n_restarts):
        if time.time() > phase2_end:
            break
        if r == 0:
            h_cur = h_bin.copy()
        else:
            h_cur = best_h.copy()
            n_pert = rng.randint(2, 5)
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                ones = np.where(h_cur == 1.0)[0]
                zeros = np.where(h_cur == 0.0)[0]

        # Greedy
        t_alloc = restart_time * 0.4
        h_g, c5_g = binary_greedy_fast(h_cur, rng, t_alloc, time.time(), T_grad=0.01)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

        # SA
        t_alloc = restart_time * 0.35
        n_sa = int(min(2000000, t_alloc * 30000))
        h_s, c5_s = binary_sa(h_g, rng, 0.015, 0.00005, n_sa, t_alloc, time.time())
        if c5_s < best_score:
            best_score = c5_s
            best_h = h_s.copy()

        # 2-swap
        if time.time() < phase2_end - 10:
            t_alloc = restart_time * 0.2
            h_2s, c5_2s = binary_2swap(h_s, rng, t_alloc, time.time(), T_grad=0.01)
            if c5_2s < best_score:
                best_score = c5_2s
                best_h = h_2s.copy()

    # Phase 3: Different resolutions
    n_tries = [n_init - 8, n_init + 8, n_init + 16, 400, 800]
    for n_try in n_tries:
        if time.time() > phase3_end:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = float(n_try / 2.0)
        target_try_int = int(n_try / 2.0)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try_int)
        sc_try = compute_c5(h_bin_try)

        t_alloc = max(5.0, (phase3_end - time.time()) * 0.3)
        h_g3, c5_g3 = binary_greedy_fast(h_bin_try, rng, t_alloc, time.time(), T_grad=0.01)
        if c5_g3 < sc_try:
            sc_try = c5_g3
            h_bin_try = h_g3

        t_alloc = max(5.0, (phase3_end - time.time()) * 0.2)
        n_sa3 = int(min(500000, t_alloc * 20000))
        h_s3, c5_s3 = binary_sa(h_bin_try, rng, 0.01, 0.00005, n_sa3, t_alloc, time.time())
        if c5_s3 < sc_try:
            sc_try = c5_s3
            h_bin_try = h_s3

        if sc_try < best_score:
            best_score = sc_try
            best_h = h_bin_try.copy()

    # Phase 4: Final refinement
    # L-BFGS-B on best solution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use / 2.0)
        T_list2 = [0.01, 0.005, 0.002]
        for T in T_list2:
            if tleft() < 15:
                break
            h_ref, sc = lsqp_smooth(best_h, n_use, target_use, T, 1000.0,
                                    300, tleft() * 0.3, start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref.copy()

    # Final binary local search burst
    if tleft() > 20:
        n_use = len(best_h)
        target_use_int = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use_int)
        t_alloc = max(5.0, tleft() * 0.4)
        h_gf, c5_gf = binary_greedy_fast(h_bin_final, rng, t_alloc, time.time(), T_grad=0.005)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gf.copy()

    if tleft() > 15:
        n_use = len(best_h)
        target_use_int = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use_int)
        t_alloc = max(5.0, tleft() * 0.5)
        n_saf = int(min(1000000, t_alloc * 20000))
        h_sf, c5_sf = binary_sa(h_bin_final, rng, 0.008, 0.00002, n_saf, t_alloc, time.time())
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = project_binary(best_h, int(n_final / 2.0))
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
