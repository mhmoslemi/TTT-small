    # Find x in [x_min, x_max] minimizing max_k (c_k * x + d_k)
    lo = x_min

    for _ in range(n_iter):

            hi = mid

def compute_a_coeffs(h, i):

    # k ranges from 0 to 2n-2, representing lag k-(n-1)
    k = np.arange(2*n-1) - (n-1)  # actual lags

    a[mask1] += (1 - h[i_minus_k[mask1]])
    a[mask2] -= h[i_plus_k[mask2]]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_a_vec(h, i):
    """Compute a_k(i) for all k (full mode indices 0..2n-2)."""
    n = len(h)
    k_lags = np.arange(2*n-1) - (n-1)
    i_minus_k = i - k_lags
    i_plus_k = i + k_lags
    mask1 = (i_minus_k >= 0) & (i_minus_k < n)
    mask2 = (i_plus_k >= 0) & (i_plus_k < n)
    a = np.zeros(2*n-1)
    a[mask1] = 1.0 - h[i_minus_k[mask1]]
    # subtract h[i+k] where valid
    a[mask2] -= h[i_plus_k[mask2]]
    return a

def solve_2coord(h, i, j, corr_curr):
    """Solve the 2-coordinate subproblem for h[i] and h[j]."""
    n = len(h)
    hi = h[i]
    hj = h[j]
    
    a_i = compute_a_vec(h, i)
    a_j = compute_a_vec(h, j)
    c = a_i - a_j  # coefficient of (x - hi) in new_corr
    
    # Feasible range for y = x - hi:
    # x = hi + y in [0,1] → y in [-hi, 1-hi]
    # hj - y in [0,1] → y in [hj-1, hj]
    y_min = max(-hi, hj - 1)
    y_max = min(1 - hi, hj)
    
    if y_min >= y_max:
        return hi, hj  # no feasible movement
    
    # Bisection to find y* minimizing max_k (corr_curr[k] + y * c[k])
    lo = y_min
    hi_b = y_max
    for _ in range(60):
        mid = (lo + hi_b) / 2
        vals = corr_curr + mid * c
        k_max = np.argmax(vals)
        if c[k_max] > 1e-15:
            hi_b = mid
        elif c[k_max] < -1e-15:
            lo = mid
        else:
            break
    
    y_star = (lo + hi_b) / 2
    new_hi = hi + y_star
    new_hj = hj - y_star
    new_hi = np.clip(new_hi, 0.0, 1.0)
    new_hj = np.clip(new_hj, 0.0, 1.0)
    return new_hi, new_hj

def coord_descent_2(h, target, n_passes, deadline):
    """2-coordinate descent method."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(0, n - 1, 2):
            j = i + 1
            new_hi, new_hj = solve_2coord(h, i, j, corr)
            if abs(new_hi - h[i]) > 1e-14 or abs(new_hj - h[j]) > 1e-14:
                # Update correlation
                dy_i = new_hi - h[i]
                a_i = compute_a_vec(h, i)  # use old h for coefficients
                a_j = compute_a_vec(h, j)
                c = a_i - a_j
                corr = corr + dy_i * c
                h[i] = new_hi
                h[j] = new_hj
                any_change = True
        
        if not any_change:
            break
    
    h = project_feasible(h, target)
    return h

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    """SLSQP with log-sum-exp smoothing."""
    h_cur = h.copy()
    dx = 2.0 / n
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for a in alphas:
        if time.time() > deadline:
            break
        alpha = float(a)
        
        def _lse_obj(x, alpha=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * dx
        
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    """Subgradient method focusing on worst lags."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):

            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Focus on top lags
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            lag = k - (n - 1)
            # d(corr[k])/dh[i] = (1-h[i-lag])*[0<=i-lag<n] - h[i+lag]*[0<=i+lag<n]
            i_range = np.arange(n)
            i_minus_lag = i_range - lag
            i_plus_lag = i_range + lag
            m1 = (i_minus_lag >= 0) & (i_minus_lag < n)
            m2 = (i_plus_lag >= 0) & (i_plus_lag < n)
            g = np.zeros(n)
            g[m1] += (1.0 - h_cur[i_minus_lag[m1]])
            g[m2] -= h_cur[i_plus_lag[m2]]
            grad += wk * g
        
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_a_vec(h, i):
    n = len(h)
    k_lags = np.arange(2 * n - 1) - (n - 1)
    i_minus_k = i - k_lags
    i_plus_k = i + k_lags
    mask1 = (i_minus_k >= 0) & (i_minus_k < n)
    mask2 = (i_plus_k >= 0) & (i_plus_k < n)
    a = np.zeros(2 * n - 1)
    a[mask1] = 1.0 - h[i_minus_k[mask1]]
    a[mask2] -= h[i_plus_k[mask2]]
    return a

def solve_2coord(h, i, j, corr_curr):

    hi = h[i]
    hj = h[j]

        return hi, hj
    lo = y_min

    lr = lr_init

        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.01)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)

            i_range = np.arange(n)

            m2 = (i_plus_lag >= 0) & (i_plus_lag < n)
