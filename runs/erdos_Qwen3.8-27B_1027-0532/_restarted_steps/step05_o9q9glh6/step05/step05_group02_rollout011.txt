import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def multi_k_greedy(h, max_iters, rng, time_limit, start_time, top_k=5, candidate_size=15):
    """Greedy that considers top-k correlations, not just argmax."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 500
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        
        # Get top-k shifts
        top_indices = np.argsort(full)[::-1][:top_k]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        for max_idx in top_indices:
            k_star = max_idx - (n - 1)
            if k_star == 0:
                continue

            # Compute A (contribution to correlation if we move i from 1 to 0)
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]

            # Compute B (contribution if we move j from 0 to 1)
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]

            # We want to decrease correlation at k_star
            # Moving i (1->0) decreases corr by A[i] (the h values at i±k)
            # Moving j (0->1) decreases corr by B[j]
            # So we want high A[i] and high B[j]
            top_i = ones[np.argsort(-A[ones])[:min(candidate_size, len(ones))]]
            top_j = zeros[np.argsort(-B[zeros])[:min(candidate_size, len(zeros))]]

            for i in top_i:
                for j in top_j:
                    delta = compute_delta(h, i, j)
                    new_max = float((full + delta).max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        stall = 0
                        new_c5 = float(full.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            # Random try
            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5


def paired_swaps_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):

    full = full_corr_fast(h)

    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        use_paired = (rng.random() < 0.2)  # 20% of moves are paired

        if use_paired and len(ones) >= 2 and len(zeros) >= 2:
            # Paired swap: move two 1s to two 0s
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                i2 = ones[(i2 + 1) % len(ones)]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                j2 = zeros[(j2 + 1) % len(zeros)]
            
            delta = compute_delta(h, i1, j1) + compute_delta(h, i2, j2)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5

            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:

            j = zeros[rng.randint(len(zeros))]

            c5_new = new_max * dx

                    best_h = h.copy()
                    stall = 0

def adaptive_sa(h, n_iters, rng, time_limit, start_time):
    """SA with adaptive temperature based on acceptance rate."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    positions = np.arange(n)
    
    t = 0.01
    accept_count = 0
    total_count = 0
    window = 500
    
    stall = 0
    max_stall = 50000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        accepted = delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t))
        
        if accepted:
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            accept_count += 1
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        total_count += 1
        if total_count % window == 0:
            rate = accept_count / total_count
            if rate < 0.1:
                t *= 1.5
            elif rate > 0.4:
                t *= 0.7
            t = max(t, 0.00001)
            t = min(t, 0.05)
            accept_count = 0
            total_count = 0

    return best_h, best_c5


def continuation_gradient(h_binary, n_steps, lr, time_limit, start_time):
    """Continuation method: solve continuous relaxed problem with log-sum-exp."""
    n = len(h_binary)
    target = float(n) / 2.0
    h = h_binary.copy().astype(float)
    
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    
    best_h = h.copy()
    best_score = compute_c5(best_h)
    
    alphas = [10, 30, 100, 300, 1000]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        # Projected subgradient for log-sum-exp objective
        for step in range(n_steps):
            if time.time() - start_time > time_limit:
                break
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            # Compute subgradient of log-sum-exp
            shifted = alpha * (corr - c_max)
            weights = np.exp(np.clip(shifted, -50, 0))
            weights /= weights.sum()
            
            # Identify which indices are in the top correlations
            top_weight_indices = np.where(weights > 0.01)[0]
            
            # Compute gradient w.r.t. h
            # corr[m] = sum_i h[i] * (1-h[i+k]) where k = m - (n-1)
            # d(corr[m])/d(h[i]) = (1-h[i+k]) - h[i-k] (where k = m-(n-1))
            
            grad = np.zeros(n)
            for m in top_weight_indices:
                w = weights[m]
                k = m - (n - 1)
                if k == 0:
                    # corr[0] = sum h[i](1-h[i]) -> d/dh[i] = 1-2h[i]
                    grad += w * (1.0 - 2.0 * h)
                else:
                    # Terms: h[i]*(1-h[i+k]) and h[i-k]*(1-h[i])
                    # d/dh[i] = (1-h[i+k]) - h[i-k]
                    for pos in range(n):
                        term = 0.0
                        if 0 <= pos + k < n:
                            term += (1.0 - h[pos + k])
                        if 0 <= pos - k < n:
                            term -= h[pos - k]
                        grad[pos] += w * term
            
            # Project: h_new = clip(h - lr * grad, 0, 1) then adjust sum
            h_new = h - lr * grad / max(1.0, np.max(np.abs(grad)))
            h_new = np.clip(h_new, 0.0, 1.0)
            # Adjust sum
            diff = target - np.sum(h_new)
            if diff > 0:
                idx = np.where(h_new < 1.0)[0]
                if len(idx) > 0:
                    h_new[idx] += diff / len(idx)
                    h_new = np.clip(h_new, 0.0, 1.0)
            elif diff < 0:
                idx = np.where(h_new > 0.0)[0]
                if len(idx) > 0:
                    h_new[idx] += diff / len(idx)
                    h_new = np.clip(h_new, 0.0, 1.0)
            
            h = h_new
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    
    return best_h, best_score

def compute_c5(h):

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:

    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k

    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

                               time_limit=allowed, start_time=start_time, top_k=5, candidate_size=14)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Long adaptive SA
    allowed = min(180, tleft() * 0.22)
    n_sa = int(min(4000000, allowed * 35000))
    h_sa, c5_sa = adaptive_sa(best_h, n_sa, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Focused SA
    allowed = min(150, tleft() * 0.18)
    n_fs = int(min(3000000, allowed * 30000))
    h_fs, c5_fs = focused_sa(best_h, n_fs, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_fs < best_score:
        best_score = c5_fs
        best_h = h_fs.copy()

    # Phase 4: Block move SA
    allowed = min(120, tleft() * 0.15)
    n_bs = int(min(2000000, allowed * 25000))
    h_bs, c5_bs = block_move_sa(best_h, n_bs, rng,
                                 time_limit=allowed, start_time=start_time)
    if c5_bs < best_score:
        best_score = c5_bs
        best_h = h_bs.copy()

    # Phase 5: Iterated improvement loop
    for it in range(20):
        if tleft() < 40:
            break

        allowed = min(15, tleft() * 0.05)
        h_g2, c5_g2 = multi_k_greedy(best_h, max_iters=20000, rng=rng,
                                      time_limit=allowed, start_time=start_time,
                                      top_k=5, candidate_size=12)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 25:
            break

        n_pert = rng.randint(3, 8)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        allowed = min(50, tleft() * 0.10)
        n_sa2 = int(min(1500000, allowed * 30000))
        h_sa2, c5_sa2 = adaptive_sa(h_pert, n_sa2, rng,
                                     time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Different resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init + 12, n_init + 16, n_init + 20]:
        if tleft() < 35:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        x_new = np.linspace(0, 1, n_try)
        x_old = np.linspace(0, 1, n_init)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(15, tleft() * 0.07)
        h_g3, c5_g3 = multi_k_greedy(h_bin_try, max_iters=10000, rng=rng,
                                       time_limit=allowed * 0.5, start_time=start_time,
                                       top_k=4, candidate_size=10)
        allowed2 = min(12, tleft() * 0.05)
        n_sa3 = int(min(300000, allowed2 * 30000))
        h_s3, c5_s3 = adaptive_sa(h_g3, n_sa3, rng,
                                   time_limit=allowed2, start_time=start_time)
        if c5_s3 < c5_g3:
            h_try = h_s3
            c5_try = c5_s3
        else:
            h_try = h_g3
            c5_try = c5_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 7: Final focused SA burst
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_final_sa = int(min(2000000, max(30000, (tleft() - 10) * 35000)))
        h_fl, c5_fl = focused_sa(h_bin_final, n_final_sa, rng,
                                  time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if c5_fl < best_score:
            best_score = c5_fl
            best_h = h_fl

    # Final: quick greedy polish
    if tleft() > 10:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_fin = project_binary(best_h, target_use)
        h_gp, c5_gp = multi_k_greedy(h_bin_fin, max_iters=5000, rng=rng,
                                      time_limit=min(5.0, tleft() - 3), start_time=start_time,
                                      top_k=4, candidate_size=10)
        if c5_gp < best_score:
            best_score = c5_gp
            best_h = h_gp

    n_final = len(best_h)
    target_final = int(n_final / 2.0)
    h_final = project_binary(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
