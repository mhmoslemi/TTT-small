            g_h[:n - ell] -= wm * h[ell:]
    g_h *= dx
    g_h += 2.0 * mu * (np.sum(h) - n / 2.0)
    return g_h


def active_set_obj(x, n, active_lags, mu):
    h = x[:n]
    corr = compute_corr(h)
    dx = 2.0 / n
    f_active = corr[active_lags] * dx
    c_max_all = float(np.max(corr)) * dx
    obj = np.sum(f_active * f_active) + 1e-5 * c_max_all + mu * (np.sum(h) - n / 2.0) ** 2
    return obj


def active_set_grad(x, n, active_lags, mu):
    h = x[:n]
    corr = compute_corr(h)
    dx = 2.0 / n
    g_h = np.zeros(n)
    for m in active_lags:
        c_m = corr[m] * dx
        k = m - (n - 1)
        if k >= 0:
            g_h[:n - k] += 2.0 * c_m * (1.0 - h[k:])
            g_h[k:] -= 2.0 * c_m * h[:n - k]
        else:
            ell = -k
            g_h[ell:] += 2.0 * c_m * (1.0 - h[:n - ell])
            g_h[:n - ell] -= 2.0 * c_m * h[ell:]
    c_max = float(np.max(corr)) * dx
    m_star = int(np.argmax(corr))
    ws = 1e-5
    k_star = m_star - (n - 1)
    if k_star >= 0:
        g_h[:n - k_star] += ws * (1.0 - h[k_star:])
        g_h[k_star:] -= ws * h[:n - k_star]
    else:
        ell = -k_star
        g_h[ell:] += ws * (1.0 - h[:n - ell])
        g_h[:n - ell] -= ws * h[ell:]
    g_h += 2.0 * mu * (np.sum(h) - n / 2.0)
    return g_h


def lbfgsb_level(h, n, target, rho, mu, maxiter, deadline):
    x0 = np.concatenate([h, [compute_c5(h)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.6)]
    best_x = x0.copy()
    best_f = level_obj(x0, n, rho, mu)

    def obj_wrapped(x):
        return level_obj(x, n, rho, mu)

    def grad_wrapped(x):
        return level_grad(x, n, rho, mu)

    try:
        res = minimize(obj_wrapped, x0, method='L-BFGS-B', jac=grad_wrapped,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], target)
            sc = compute_c5(h_new)
            if sc < best_f:
                best_f = sc
                best_x = np.concatenate([h_new, [sc]])
    except Exception:
        pass
    h_out = project_feasible(best_x[:n], target)
    return h_out, compute_c5(h_out)


def lbfgsb_softmax(h, n, target, alpha, mu, maxiter, deadline):
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_wrapped(x):
        return softmax_obj(x, n, alpha, mu)

    def grad_wrapped(x):
        return softmax_grad(x, n, alpha, mu)

    try:
        res = minimize(obj_wrapped, x0, method='L-BFGS-B', jac=grad_wrapped,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)


def lbfgsb_active(h, n, target, active_lags, mu, maxiter, deadline):
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_wrapped(x):
        return active_set_obj(x, n, active_lags, mu)

    def grad_wrapped(x):
        return active_set_grad(x, n, active_lags, mu)

    try:
        res = minimize(obj_wrapped, x0, method='L-BFGS-B', jac=grad_wrapped,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)


def fourier_smooth(h, n_keep):
    n = len(h)
    h_centered = h - np.mean(h)
    spec = np.fft.rfft(h_centered)
    spec[n_keep:] = 0.0
    h_smooth = np.fft.irfft(spec, n=n) + np.mean(h)
    return h_smooth


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 12
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.02)
        else:
            eps *= 0.82
            if eps < 1e-10:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_lag_refine(h, n, target, m_active, mu, maxiter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for outer in range(20):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:m_active]
        h_candidate, sc = lbfgsb_active(h_cur, n, target, top_m, mu, maxiter, deadline)
        if sc < best_score:
            best_score = sc
            h_cur = h_candidate
        else:
            break
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Leveling objective with continuation on rho
    rhos_p1 = [5, 15, 50, 150, 500, 1500, 5000, 15000, 50000]
    for rho in rhos_p1:
        if time.time() > deadline - 200:
            break
        for mu in [10.0, 100.0]:
            if time.time() > deadline - 50:
                break
            h_new, sc = lbfgsb_level(best_h, n, target, rho, mu, 300, deadline)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new

    # Phase 2: Softmax continuation
    alphas_p2 = [50, 200, 800, 3000, 10000, 30000]
    for alpha in alphas_p2:
        if time.time() > deadline - 150:
            break
        for mu in [10.0, 100.0, 500.0]:
            if time.time() > deadline - 40:
                break
            h_new, sc = lbfgsb_softmax(best_h, n, target, alpha, mu, 300, deadline)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new

    # Phase 3: Multi-lag active set refinement
    if time.time() < deadline - 120:
        for m_act in [4, 8, 12, 16]:
            if time.time() > deadline - 40:
                break
            h_ml, sc_ml = multi_lag_refine(best_h, n, target, m_act, 50.0, 200, deadline - 30)
            if sc_ml < best_score - 1e-14:
                best_score = sc_ml
                best_h = h_ml

    # Phase 4: Fourier smoothing + re-optimize
    if time.time() < deadline - 200:
        for n_keep in [8, 12, 20, 30]:
            if time.time() > deadline - 60:
                break
            h_smooth = fourier_smooth(best_h, n_keep)
            h_smooth = project_feasible(h_smooth, target)
            for rho in [500, 5000]:
                if time.time() > deadline - 40:
                    break
                h_ref, sc_ref = lbfgsb_level(h_smooth, n, target, rho, 100.0, 200, deadline - 20)
                if sc_ref < best_score - 1e-14:
                    best_score = sc_ref
                    best_h = h_ref

    # Phase 5: Subgradient polish
    if time.time() < deadline - 100:
        for temp in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline - 40:
                break
            h_sg, sc_sg = subgradient_polish(best_h, n, target, 2000, temp, 0.005, 0.995, deadline - 30)
            if sc_sg < best_score - 1e-14:
                best_score = sc_sg
                best_h = h_sg

    # Phase 6: Higher resolution n=500
    if time.time() < deadline - 250:
        n2 = 500
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for rho in [20, 100, 500, 2000, 8000, 30000]:
            if time.time() > deadline - 80:
                break
            for mu in [50.0, 200.0]:
                if time.time() > deadline - 40:
                    break
                h2_new, sc2 = lbfgsb_level(best_h2, n2, target2, rho, mu, 200, deadline - 20)
                if sc2 < best_score2 - 1e-14:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 60:
            h2_new, sc2 = lbfgsb_softmax(best_h2, n2, target2, 10000, 200.0, 200, deadline - 30)
            if sc2 < best_score2 - 1e-14:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 50:
            h2_ml, sc2_ml = multi_lag_refine(best_h2, n2, target2, 8, 200.0, 150, deadline - 30)
            if sc2_ml < best_score2 - 1e-14:
                best_score2 = sc2_ml
                best_h2 = h2_ml
        if time.time() < deadline - 40:
            h2_sg, sc2_sg = subgradient_polish(best_h2, n2, target2, 1000, 0.002, 0.003, 0.995, deadline - 25)
            if sc2_sg < best_score2 - 1e-14:
                best_score2 = sc2_sg
                best_h2 = h2_sg
        if time.time() < deadline - 25:
            h2_surg, sc2_surg = worst_lag_surgery(best_h2, n2, target2, 300, 0.002, deadline - 15)
            if sc2_surg < best_score2 - 1e-14:
                best_score2 = sc2_surg
                best_h2 = h2_surg
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 7: Even higher resolution n=600
    if time.time() < deadline - 200:
        n3 = 600
        target3 = n3 / 2.0
        n_cur = len(best_h)
        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)
        for rho in [20, 100, 500, 2000, 10000]:
            if time.time() > deadline - 60:
                break

                if time.time() > deadline - 30:
                    break
                h3_new, sc3 = lbfgsb_level(best_h3, n3, target3, rho, mu, 150, deadline - 20)
                if sc3 < best_score3 - 1e-14:
                    best_score3 = sc3
                    best_h3 = h3_new
        if time.time() < deadline - 40:
            h3_new, sc3 = lbfgsb_softmax(best_h3, n3, target3, 5000, 200.0, 150, deadline - 20)
            if sc3 < best_score3 - 1e-14:
                best_score3 = sc3
                best_h3 = h3_new
        if time.time() < deadline - 30:
            h3_sg, sc3_sg = subgradient_polish(best_h3, n3, target3, 800, 0.002, 0.003, 0.995, deadline - 15)
            if sc3_sg < best_score3 - 1e-14:

            if sc3_surg < best_score3 - 1e-14:
                best_score3 = sc3_surg
                best_h3 = h3_surg
        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 8: Random perturbations + re-optimize
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(5):
            if time.time() > deadline - 30:
                break
            scales = [0.05, 0.08, 0.12, 0.03, 0.15]
            h_pert = best_h + rng.randn(n_cur) * scales[attempt]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = lbfgsb_level(h_pert, n_cur, target_cur, 1000, 200.0, 200, deadline - 20)
            if sc_ref < best_score - 1e-14:
                best_score = sc_ref
                best_h = h_ref
                h_ref2, sc2 = lbfgsb_level(best_h, n_cur, target_cur, 10000, 500.0, 200, deadline - 10)
                if sc2 < best_score - 1e-14:
                    best_score = sc2
                    best_h = h_ref2

    # Phase 9: Final refinement
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for rho in [10000, 50000, 200000]:
            if time.time() > deadline - 5:
                break
            h_new, sc = lbfgsb_level(best_h, n_cur, target_cur, rho, 500.0, 300, deadline - 3)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new
        if time.time() < deadline - 5:
            h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 300, 0.001, deadline - 2)
            if sc_surg < best_score - 1e-15:
                best_score = sc_surg
                best_h = h_surg
        if time.time() < deadline - 2:
            h_sg, sc_sg = subgradient_polish(best_h, n_cur, target_cur, 500, 0.001, 0.002, 0.997, deadline - 1)
            if sc_sg < best_score - 1e-15:
                best_score = sc_sg
                best_h = h_sg

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
