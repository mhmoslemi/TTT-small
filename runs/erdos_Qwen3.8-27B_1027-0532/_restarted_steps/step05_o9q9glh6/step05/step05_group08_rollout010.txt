import numpy as np
from scipy.optimize import minimize
import time


_target_global = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def obj_with_args(x, alpha):
    return lse_value(x, alpha)


def grad_with_args(x, alpha):
    return lse_grad(x, alpha)


def cons_fun_with_args(x, target):
    return np.sum(x) - target


def cons_jac_with_args(x, target):
    return np.ones(len(x))


def slsqp_solve(h, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun_with_args, 'jac': cons_jac_with_args, 'args': (target,)}]
    try:
        res = minimize(obj_with_args, h, method='SLSQP', jac=grad_with_args,
                       args=(alpha, alpha), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgsb_solve(h, n, target, alpha, mu, maxiter):
    def penalized_obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

    def penalized_grad(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target) * np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalized_obj, h, method='L-BFGS-B', jac=penalized_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h

        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_step(h, n, target, top_k, temp):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    top_idx = np.argsort(corr)[::-1][:top_k]
    diffs = corr[top_idx] - corr[top_idx[0]]
    weights = np.exp(diffs / temp)
    weights /= weights.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(weights, top_idx):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = subgradient_step(h_cur, n, target, 6, temp)
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        c_curr = compute_c5(h_cur)
        if sc_new < c_curr:
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.02)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    phase1_deadline = start_time + 0.20 * budget_s
    h1, sc1 = continuous_refine(best_h, n, target, phase1_deadline, n_rounds=5)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    h1b = slsqp_solve(best_h, n, target, 5000, 3000, 1e-15)
    sc1b = compute_c5(h1b)
    if sc1b < best_score:
        best_score = sc1b
        best_h = h1b

    phase2_deadline = start_time + 0.45 * budget_s
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    h2, sc2 = binary_swap_search(h_bin, n, phase2_deadline, max_rounds=500)
    if sc2 < best_score:
        best_score = sc2
        best_h = h2

    h2b, sc2b = continuous_refine(best_h, n, target, min(phase2_deadline + 10, deadline), n_rounds=3)
    if sc2b < best_score:
        best_score = sc2b
        best_h = h2b

    phase3_deadline = start_time + 0.65 * budget_s
    h3, sc3 = hybrid_alternate(best_h, n, target, phase3_deadline)
    if sc3 < best_score:
        best_score = sc3
        best_h = h3

    phase4_deadline = start_time + 0.80 * budget_s
    h4, sc4 = subgradient_polish(best_h, n, target, 2000, 0.003, 0.005, 0.997, phase4_deadline)
    if sc4 < best_score:
        best_score = sc4
        best_h = h4

    h4b, sc4b = worst_lag_surgery(best_h, n, target, 500, 0.003, min(phase4_deadline + 10, deadline))
    if sc4b < best_score:
        best_score = sc4b
        best_h = h4b

    phase5_deadline = start_time + 0.92 * budget_s
    h5, sc5 = slsqp_solve(best_h, n, target, 20000, 3000, 1e-16)
    if sc5 < best_score:
        best_score = sc5
        best_h = h5

    if time.time() < deadline - 120:
        n2 = 600
        target2 = n2 / 2.0
        h2_start = interpolate_to(best_h, n2, target2)
        best_h2 = h2_start.copy()
        best_score2 = compute_c5(h2_start)

        h2a, sc2a = continuous_refine(best_h2, n2, target2, min(time.time() + 60, deadline - 60), n_rounds=4)
        if sc2a < best_score2:
            best_score2 = sc2a
            best_h2 = h2a

        h2b_bin = (best_h2 > 0.5).astype(float)
        h2b_bin = project_feasible(h2b_bin, target2)
        h2b, sc2b2 = binary_swap_search(h2b_bin, n2, min(time.time() + 30, deadline - 30), max_rounds=300)
        if sc2b2 < best_score2:
            best_score2 = sc2b2
            best_h2 = h2b

        h2c, sc2c = continuous_refine(best_h2, n2, target2, min(time.time() + 20, deadline - 20), n_rounds=2)
        if sc2c < best_score2:
            best_score2 = sc2c
            best_h2 = h2c

        h2d, sc2d = subgradient_polish(best_h2, n2, target2, 1000, 0.002, 0.004, 0.997, deadline - 10)
        if sc2d < best_score2:
            best_score2 = sc2d
            best_h2 = h2d

        h2e, sc2e = worst_lag_surgery(best_h2, n2, target2, 300, 0.002, deadline - 5)
        if sc2e < best_score2:
            best_score2 = sc2e
            best_h2 = h2e

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    if time.time() < deadline - 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_surg, sc_final = worst_lag_surgery(best_h, n_cur, target_cur, 200, 0.001, deadline - 5)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_surg

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
