def project_feasible(h, target_sum):

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def compute_c5(h):
    return float(np.max(get_overlaps(h)))

def project_box_sum(h, target, n_iter=80):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() - 0.0 + 1.0
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - mid, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            proj = proj.copy()
            proj[mask] += diff / cnt
    return np.clip(proj, 0.0, 1.0)

def softmax_weights(over, alpha):
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def grad_overlaps(h, weights):
    """Compute sum_k weights[k] * grad(O_k)/grad(h) using FFT"""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    # a_j = sum_{k=0}^{n-1-j} weights[k] * (1-h[j+k])
    # = correlate(q, weights, 'full')[n-1+j] for j=0,...,n-1
    full_a = np.correlate(q, weights, mode='full')
    a = full_a[n-1:n-1+n]
    # b_j = sum_{k=0}^{j} weights[k] * h[j-k]
    # = convolve(weights, h, 'full')[j] for j=0,...,n-1
    full_b = np.convolve(weights, h, mode='full')
    b = full_b[:n]
    return (a - b) * dx

def projected_gradient_softmax(h, n, target, alpha, n_steps, deadline, momentum=0.9):
    """Nesterov-accelerated projected gradient for soft-max objective"""
    h_cur = h.copy()
    y_cur = h.copy()  # Nesterov momentum variable
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    t_k = 1.0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (1.0 - t_k) / t_new
        y_new = y_cur + beta * (y_cur - h_cur)
        
        over_y = get_overlaps(y_new)
        w = softmax_weights(over_y, alpha)
        grad = grad_overlaps(y_new, w)
        
        # BB step size (or fixed)
        # For now, use a simple step size
        lr = 2.0 / (alpha * np.max(get_overlaps(y_new)) + 1.0)
        # Actually, let's use a fixed small step
        lr = 0.01
        
        h_new = y_new - lr * grad
        h_new = project_box_sum(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        h_cur = h_new
        y_cur = y_new
        t_k = t_new
    
    return best_h, best_score

    def obj(x):

        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

                scores[i] = sc

                    best_h = trial.copy()

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def subgradient_max(h, n, target, n_steps, deadline, top_m=3):
    """Subgradient method on exact max with adaptive step size"""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = 0.01
    dx = 2.0 / n
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        over = get_overlaps(h_cur)
        c_max = float(np.max(over))
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        
        # Find top-m lags
        top_idx = np.argsort(over)[::-1][:top_m]
        top_vals = over[top_idx]
        # Exponential weights among top-m
        w = np.exp(top_vals / 0.003)
        w /= w.sum()
        
        # Compute weighted gradient
        all_w = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            all_w[int(ki)] = wi
        grad = grad_overlaps(h_cur, all_w)
        
        h_new = h_cur - lr * grad
        h_new = project_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < c_max:
            h_cur = h_new
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
    
    return best_h, best_score

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

    hi = float(h.max()) + 1.5

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:

        else:
            hi = mid
