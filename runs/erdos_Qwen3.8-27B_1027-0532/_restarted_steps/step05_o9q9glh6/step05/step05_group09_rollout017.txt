import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def corr_and_weights(h, alpha, top_k=None):
    """Compute correlations and weights for LSE or active set."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    if top_k is not None:
        idx = np.argsort(corr)[::-1][:top_k]
        weights = np.zeros(n)
        weights[idx] = 1.0 / top_k
    else:
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        c_smoothed = c_max + np.log(sum_exp) / alpha
    return c_max, c_smoothed if top_k is None else c_max, weights, corr

def grad_from_weights(weights, h, dx):
    """Compute gradient of weighted sum of correlations using FFT."""
    n = len(h)
    fsize = 2 * n
    Wf = np.fft.rfft(weights, fsize)
    Hf = np.fft.rfft(h, fsize)
    conv_wh = np.fft.irfft(Wf * Hf, fsize)[:n]
    h_rev = h[::-1]
    Hrevf = np.fft.rfft(h_rev, fsize)
    conv_whrev = np.fft.irfft(Wf * Hrevf, fsize)[:n]
    prefix_w = np.cumsum(weights)
    grad = prefix_w - conv_wh - conv_whrev[::-1]
    grad *= dx
    return grad

def fista_optimize(h0, target, alpha, max_iters, deadline):

    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    L = alpha * n * dx * 2.0
    step = 1.0 / L
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y_clipped, 1.0 - y_clipped, mode='full')[n-1:]
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        obj = (c_max + np.log(sum_exp) / alpha) * dx
        
        grad = grad_from_weights(weights, y_clipped, dx)
        x_new = project_feasible(y_clipped - step * grad, target)
        
        # Backtracking
        for bt in range(30):
            corr_new = np.correlate(x_new, 1.0 - x_new, mode='full')[n-1:]
            c_max_new = float(np.max(corr_new))
            shifted_new = alpha * (corr_new - c_max_new)
            shifted_new = np.clip(shifted_new, -500.0, 0.0)
            exp_new = np.exp(shifted_new)
            sum_new = float(exp_new.sum())
            obj_new = (c_max_new + np.log(sum_new) / alpha) * dx
            if obj_new <= obj + 1e-12 * np.dot(x_new - y_clipped, x_new - y_clipped):
                break
            step *= 0.5
            x_new = project_feasible(y_clipped - step * grad, target)
        
        step = min(step * 1.5, 2.0 / L)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def subgradient_anderson(h0, target, max_iters, deadline, top_k=3, damp=0.5, m=5):
    """Subgradient method with Anderson acceleration for the exact max."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    # Anderson history
    y_hist = [x.copy()]
    g_hist = [np.zeros(n)]
    
    # Initial step size
    t = 0.001
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute current correlations and active set
        corr = np.correlate(x, 1.0 - x, mode='full')[n-1:]

        sc = c_max * dx
        
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()

        idx = np.argsort(corr)[::-1][:top_k]

        weights[idx] = 1.0 / top_k

        grad = grad_from_weights(weights, x, dx)

            Y = np.array([y_hist[-m_use+i] for i in range(m_use)])
            G = np.array([g_hist[-m_use+i] for i in range(m_use)])

            dG = np.diff(G, axis=0)

                # Let me use a simpler form

            pass

        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-10:

        x_new_raw = x - t * grad

def coord_descent(h, target, deadline, max_sweeps=100):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(777)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            # Try moving to boundary or to a few interior values
            for trial in [0.0, 1.0, 0.5, old_val * 0.5, old_val * 1.5]:
                if trial < 0 or trial > 1:
                    continue
                if abs(trial - old_val) < 1e-12:
                    continue
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
                    break
                else:
                    h_cur[idx] = old_val
                    h_cur = project_feasible(h_cur, target)
        if not improved and sweep > 20:
            break
    
    return h_cur, cur_score

def generate_diverse_starts(n, target, seed, h_ref):

    rng = np.random.RandomState(seed)
    starts = []

    for scale in [0.1, 0.2, 0.3, 0.5]:

    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 4, 6, 8]:
        for phase in [0, 0.5]:

            starts.append(h)
    
    # Block patterns
    for block_size in [4, 8, 16, 32]:
        n_blocks = n // block_size
        h = np.zeros(n)

Actually, I realize I should also try a different approach: **Nesterov's accelerated gradient for non-smooth functions**. For f(x) = max_k g_k(x) where g_k are smooth, the accelerated proximal gradient method (NAG) with prox = projection onto the feasible set should work well.

The iteration is:
y_k = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
x_{k+1} = prox_{t_k * L}(y_k - t_k * ∇g_{i_k}(y_k))

where i_k is the index of the max (or a random one from the active set), and ∇g_{i_k} is the gradient of that specific correlation.

For a single active k, the gradient is:
∇g_k[j] = (1-h[j-k])*1_{j≥k} - h[j+k]*1_{j+k<n}

This is even simpler to compute (just two shifts of h).

Let me also think about the **Anderson acceleration** more carefully. The standard Anderson acceleration for fixed-point iterations:
- We have a mapping F: x → T(x) (the proximal step)
- We want to find x such that x = T(x)
- Anderson: x_{k+1} = (1-μ)T(x_k) + μ ∑_{i=0}^{m-1} γ_i x_{k-i}
- The γ_i are chosen to minimize ||T(x_k) - ∑ γ_i (T(x_{k-i}) - (x_{k-i}))||

For our problem, T(x) = project_feasible(x - step * grad(x)) where grad is the subgradient.

Actually, for non-smooth problems, Anderson acceleration can be tricky. Let me use a simpler approach: **heavy ball method** (momentum) for the subgradient.

x_{k+1} = project_feasible(x_k - t_k * g_k + β * (x_k - x_{k-1}))

where β is the momentum parameter (typically 0.9).

Or even simpler: **accelerated subgradient** (Nesterov's method for non-smooth):
y_k = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
f_k = subgradient at y_k
x_{k+1} = project_feasible(y_k - t_{k+1} * f_k)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def grad_from_weights(weights, h, dx):
    n = len(h)
    fsize = 2 * n
    Wf = np.fft.rfft(weights, fsize)
    Hf = np.fft.rfft(h, fsize)
    conv_wh = np.fft.irfft(Wf * Hf, fsize)[:n]
    h_rev = h[::-1]
    Hrevf = np.fft.rfft(h_rev, fsize)
    conv_whrev = np.fft.irfft(Wf * Hrevf, fsize)[:n]
    prefix_w = np.cumsum(weights)
    grad = prefix_w - conv_wh - conv_whrev[::-1]
    grad *= dx
    return grad

def fista_optimize(h0, target, alpha, max_iters, deadline):

    y = x.copy()

    best_x = x.copy()
    best_obj = compute_c5(x)
    L = alpha * 2.0 * n * dx
    step = 1.0 / L

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y_clipped, 1.0 - y_clipped, mode='full')[n-1:]
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        obj = (c_max + np.log(sum_exp) / alpha) * dx
        grad = grad_from_weights(weights, y_clipped, dx)
        x_new = project_feasible(y_clipped - step * grad, target)
        for _bt in range(25):
            corr_new = np.correlate(x_new, 1.0 - x_new, mode='full')[n-1:]
            c_max_new = float(np.max(corr_new))
            shifted_new = alpha * (corr_new - c_max_new)
            shifted_new = np.clip(shifted_new, -500.0, 0.0)
            exp_new = np.exp(shifted_new)
            sum_new = float(exp_new.sum())
            obj_new = (c_max_new + np.log(sum_new) / alpha) * dx
            if obj_new <= obj + 1e-12 * np.dot(x_new - y_clipped, x_new - y_clipped):

            x_new = project_feasible(y_clipped - step * grad, target)

        x = x_new
        t = t_new

def coord_descent(h, target, deadline, max_sweeps=200):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            for trial in [0.0, 1.0, 0.5, max(0.0, old_val - 0.2), min(1.0, old_val + 0.2)]:
                if abs(trial - old_val) < 1e-12:
                    continue
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
                    break
                else:
                    h_cur[idx] = old_val
                    h_cur = project_feasible(h_cur, target)
        if not improved and sweep > 30:
            break
    return h_cur, cur_score

        h_new, sc_new = exact_subgrad_linsearch(best_h, target, start_time + min(30, tleft() - 10), top_k=4)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 10: Coordinate descent final polish
    if tleft() > 25:
        h_new, sc_new = coord_descent(best_h, target, start_time + max(tleft() - 5, 10), max_sweeps=300)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Final: tiny perturbation search
    if tleft() > 15:
        rng_final = np.random.RandomState(seed + 200)
        for _ in range(20):
            if tleft() < 3:
                break
            h_try = best_h + rng_final.randn(n) * 0.01
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try
                h_new, sc_new = exact_subgrad_linsearch(best_h, target, start_time + min(8, tleft() - 3), top_k=2)
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
