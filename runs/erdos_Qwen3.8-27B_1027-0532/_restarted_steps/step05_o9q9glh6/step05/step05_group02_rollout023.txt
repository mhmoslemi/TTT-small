def compute_smooth_objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx  # these are the f_k values
    
    # Log-sum-exp for numerical stability
    f_max = f.max()
    log_sum_exp = f_max + np.log(np.exp(f - f_max).sum()) / alpha
    weights = np.exp(alpha * (f - f_max)) / np.exp(alpha * (f - f_max)).sum()
    
    # Gradient: weighted average of individual gradients
    grad = np.zeros(n)
    for k_weighted, w in enumerate(weights):

    #     After: 1*(1-h[j+k]). So delta += (1-h[j+k]) for valid j+k
    #   - As h[pos+k] with pos+k=j, i.e., pos=j-k: contributes h[j-k]*(1-h[j]) = h[j-k]*1
    #     After: h[j-k]*(1-1) = 0. So delta -= h[j-k] for valid j-k
    #
    # Total delta[k]:
    #   -= (1-h[i+k]) * [0<=i+k<n]
    #   += h[i-k] * [0<=i-k<n]
    #   += (1-h[j+k]) * [0<=j+k<n]
    #   -= h[j-k] * [0<=j-k<n]
    
    delta = np.zeros(2*n-1)
    k_range = np.arange(2*n-1) - (n-1)  # actual lag values
    
    # Term 1: -(1-h[i+k]) for valid i+k
    idx = i + k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= (1 - h[idx[mask]])
    
    # Term 2: +h[i-k] for valid i-k
    idx = i - k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]
    
    # Term 3: +(1-h[j+k]) for valid j+k
    idx = j + k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] += (1 - h[idx[mask]])
    
    # Term 4: -h[j-k] for valid j-k
    idx = j - k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    
    return delta

Now let me think about the smooth gradient approach more carefully.

For the smooth objective with log-sum-exp:
f_smooth(h, α) = (1/α) log(Σ_k exp(α * f_k(h)))

where f_k(h) = (2/n) * Σ_i h[i](1-h[i+k]) for valid i.

The gradient is:
∇f_smooth = Σ_k p_k * ∇f_k

where p_k = exp(α*f_k) / Σ_j exp(α*f_j) (softmax weights)

And ∇f_k[j] = (2/n) * [(1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]]

So:
∇f_smooth[j] = (2/n) * Σ_k p_k * [(1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]]

Let me define:
S1[j] = Σ_k p_k * [0≤j+k<n]  (sum of p_k over valid k for position j as the "first" index)
S2[j] = Σ_k p_k * h[j+k] * [0≤j+k<n]  (weighted sum of h[j+k])
S3[j] = Σ_k p_k * h[j-k] * [0≤j-k<n]  (weighted sum of h[j-k])

Then ∇f_smooth[j] = (2/n) * (S1[j] - S2[j] - S3[j])

S1[j] = Σ_{k: 0≤j+k<n} p_k = sum of p_k for k in range [-j, n-1-j]
This is just a cumulative sum of p, easy to compute.

S2[j] = Σ_{k: 0≤j+k<n} p_k * h[j+k] = Σ_{i: 0≤i<n} p_{i-j} * h[i]
This is a cross-correlation of p and h!

S3[j] = Σ_{k: 0≤j-k<n} p_k * h[j-k] = Σ_{i: 0≤i<n} p_{j-i} * h[i]
This is also a cross-correlation (with p reversed).

Both S2 and S3 can be computed via FFT in O(n log n).

Let me implement this:

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    f = corr * dx  # f_k values
    
    # Softmax weights (numerically stable)
    f_max = f.max()
    shifted = alpha * (f - f_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()  # softmax weights, length 2n-1
    
    # Compute gradient
    # S1[j] = sum of p[k] for k in range [-j, n-1-j] (where k = index - (n-1))
    # p is indexed 0..2n-2, corresponding to lag -(n-1)..(n-1)
    # For position j: valid k range is [-j, n-1-j]
    # In p-index: [n-1-j, n-1-j+n-1] = [n-1-j, 2n-2-j]
    
    # S2[j] = sum over k of p[k] * h[j+k] * [0<=j+k<n]
    # = sum over i of p[k=i-j] * h[i] where 0<=i<n and 0<=k+n-1 < 2n-1 (always) and k = i-j
    # p-index for k = i-j is: i-j+n-1
    # So S2[j] = sum_i h[i] * p[i-j+n-1] for 0<=i<n
    # This is cross-correlation: (h ⊛ p)[j] where we define appropriately
    
    # Actually, S2[j] = sum_i h[i] * p[i-j+n-1]
    # Let q[m] = p[m] for m in 0..2n-2
    # S2[j] = sum_i h[i] * q[i-j+n-1]
    # This is the cross-correlation of h and q, evaluated at shift (n-1-j):
    # (h ⊛ q)(s) = sum_i h[i] * q[i-s]
    # So S2[j] = (h ⊛ q)(n-1-j)
    
    # Similarly, S3[j] = sum_i p[j-i] * h[i] (with bounds 0<=j-i+n-1 < 2n-1, i.e., -(j-1) <= ... )
    # S3[j] = sum_i h[i] * p[j-i+n-1] (using p-index = lag + n-1)
    # = sum_i h[i] * q[j-i+n-1]
    # Let r[m] = q[2n-2-m] (reversed p)
    # S3[j] = sum_i h[i] * r[i-(j-(n-1))]
    #        = (h ⊛ r)(i-(j-(n-1)))... 
    
    # This is getting complicated. Let me just use direct computation for n=600.
    
    # Direct computation:
    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - (n-1)  # lag values
    
    for k_idx in range(2*n-1):
        if p[k_idx] < 1e-12:
            continue
        k = k_vals[k_idx]
        w = p[k_idx]
        j = np.arange(n)
        mask_pos = (j + k >= 0) & (j + k < n)
        mask_neg = (j - k >= 0) & (j - k < n)
        grad[mask_pos] += w * (1 - h[j[mask_pos] + k])
        grad[mask_neg] -= w * h[j[mask_neg] - k]
    
    grad *= dx
    return grad, p

def smooth_gradient_fft(h, alpha):

    corr = full_corr_fast(h)  # length 2n-1

    f_max = f.max()

    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # S2[j] = sum_i h[i] * p[i - j + n - 1] for valid indices
    # This is cross-correlation of h and p
    # Using FFT: 
    N = 1
    while N < 2*n + 2*n - 1:  # need enough for the convolution
        N <<= 1
    # Actually, cross-correlation of length-n and length-(2n-1) gives length 3n-2
    # Let's use: S2[j] = sum_i h[i] * p[i-j+n-1]
    # = sum_{i=0}^{n-1} h[i] * p[i-j+n-1] where p is defined on [0, 2n-2]
    # Let a = h (length n), b = p (length 2n-1)
    # We want: for each j in [0, n-1], sum over i of a[i] * b[i-j+n-1]
    # = (a cross-correlated with b) evaluated at (j - (n-1))

    # S2[j] = sum_i h[i] * p[i-j+n-1]
    # Let's define c[m] = p[m] for m = 0..2n-2
    # S2[j] = sum_{i=0}^{n-1} h[i] * c[i-j+n-1]
    # This is (h * c_flipped)[j+n-1] where * is convolution and c_flipped[m] = c[2n-2-m]
    # No wait, let me think again.
    # 
    # Convolution: (a * b)[k] = sum_i a[i] * b[k-i]
    # Cross-correlation: (a ⋆ b)[k] = sum_i a[i] * b[i+k]
    # 
    # S2[j] = sum_i h[i] * c[i-j+n-1] = (h ⋆ c)[n-1-j]
    # 
    # Cross-correlation can be computed as: (h ⋆ c)(k) = (h * c_flipped)(k + len(c) - 1)
    # where c_flipped[m] = c[len(c)-1-m]
    
    # Actually, let me just use scipy.signal or direct FFT.
    # For n=600, the direct approach (O(n * 2n)) = O(n^2) = 360,000 ops is fine.
    
    # Let me just use the direct approach for clarity and correctness.
    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - (n-1)
    
    for k_idx in range(2*n-1):
        w = p[k_idx]
        if w < 1e-14:
            continue
        k = k_vals[k_idx]
        j = np.arange(n)

        mask_neg = (j - k >= 0) & (j - k < n)
        grad[mask_pos] += w * (1.0 - h[j[mask_pos] + k])
        grad[mask_neg] -= w * h[j[mask_neg] - k]
    
    grad *= dx
    return grad, p

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta_swap(h, i, j):
    n = len(h)
    delta = np.zeros(2*n-1)
    k_range = np.arange(2*n-1) - (n-1)
    idx = i + k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= (1.0 - h[idx[mask]])
    idx = i - k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]
    idx = j + k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] += (1.0 - h[idx[mask]])
    idx = j - k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    return delta

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            h_r[np.where(h_r == 1)[0][:final_sum - target_sum]] = 0
        else:
            h_r[np.where(h_r == 0)[0][:target_sum - final_sum]] = 1
    return h_r.astype(float)

    target = n / 2.0
    best_score = compute_c5(h)
    best_h = h.copy()
    
    alphas_schedule = [10, 30, 100, 300, 1000, 3000, 10000]
    iters_per_alpha = n_iters_max // len(alphas_schedule)
    
    for alpha_idx, alpha in enumerate(alphas_schedule):
        if time.time() - start_time > time_limit:
            break
        
        lr = 0.1 / (1.0 + alpha * 0.01)
        n_iters = min(iters_per_alpha, max(100, int((time_limit - (time.time() - start_time)) * 50)))
        
        for it in range(n_iters):
            if time.time() - start_time > time_limit:
                break
            
            corr = full_corr_fast(h)
            f = corr * (2.0 / n)
            
            # Softmax weights
            f_max = f.max()
            shifted = alpha * (f - f_max)
            exp_shifted = np.exp(shifted - shifted.max())
            p = exp_shifted / exp_shifted.sum()
            
            # Compute gradient
            dx = 2.0 / n
            grad = np.zeros(n)
            k_vals = np.arange(2*n-1) - (n-1)

            significant = p > 1e-8
            k_sig = k_vals[significant]
            p_sig = p[significant]
            
            for k_i in range(len(k_sig)):
                k = k_sig[k_i]
                w = p_sig[k_i]
                j = np.arange(n)
                mp = (j + k >= 0) & (j + k < n)
                mn = (j - k >= 0) & (j - k < n)
                grad[mp] += w * (1.0 - h[j[mp] + k])
                grad[mn] -= w * h[j[mn] - k]
            
            grad *= dx
            
            # Gradient step
            h = h - lr * grad
            h = project_feasible(h, target)
            
            # Evaluate

            if score < best_score:
                best_score = score
                best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta_swap(h, i, j):
    n = len(h)
    delta = np.zeros(2*n-1)
    k_range = np.arange(2*n-1) - (n-1)
    idx = i + k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= (1.0 - h[idx[mask]])
    idx = i - k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]
    idx = j + k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] += (1.0 - h[idx[mask]])
    idx = j - k_range
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    return delta

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            h_r[np.where(h_r == 1)[0][:final_sum - target_sum]] = 0
        else:
            h_r[np.where(h_r == 0)[0][:target_sum - final_sum]] = 1
    return h_r.astype(float)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def smooth_gradient_step(h, alpha, n):
    dx = 2.0 / n
    corr = full_corr_fast(h)
    f = corr * dx
    f_max = f.max()
    shifted = alpha * (f - f_max)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()

    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - (n-1)
    sig = p > 1e-8
    sig_idx = np.where(sig)[0]
    for idx in sig_idx:
        k = k_vals[idx]
        w = p[idx]
        j = np.arange(n)
        mp = (j + k >= 0) & (j + k < n)
        mn = (j - k >= 0) & (j - k < n)
        grad[mp] += w * (1.0 - h[j[mp] + k])
        grad[mn] -= w * h[j[mn] - k]
    grad *= dx
    return grad, p

def smooth_slsqp(h, time_limit, start_time):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    best_score = compute_c5(h)
    best_h = h.copy()

    alphas = [20, 80, 300, 1000, 5000, 20000]
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break

        def obj(x):
            corr = full_corr_fast(x)
            f_max = corr.max()
            shifted = np.clip(alpha * (corr - f_max), -500.0, 0.0)
            return f_max / alpha + np.log(np.sum(np.exp(shifted))) / alpha

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': 500, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h = h_new

    return best_h, best_score
