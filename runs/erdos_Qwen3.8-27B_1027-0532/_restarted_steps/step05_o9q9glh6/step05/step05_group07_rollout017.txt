- It uses a "memetic" approach (EA + local search hybrid)
- The local search is more targeted (focus on worst lags)

Let me also think about a **dual decomposition** approach:

Since we want min max_k r_k, we can use the fact that:
max_k r_k = max_{p ∈ Δ} sum_k p_k r_k (where Δ is the simplex)

So we're looking at:
min_h max_{p ∈ Δ} sum_k p_k r_k(h)

By minimax theorem (if it applies), this equals:
max_{p ∈ Δ} min_h sum_k p_k r_k(h)

For fixed p, min_h sum_k p_k r_k(h) is a (non-convex) quadratic program. But we can try to solve it approximately using projected gradient descent or SLSQP.

The outer problem is then a max over the simplex, which can be solved by a multiplicative weights / exponentiated gradient method:
p_k^{t+1} ∝ p_k^t * exp(η * r_k(h^t))

This is a multiplicative weights approach for the dual problem!

**Algorithm: Multiplicative Weights + Inner QP**

1. Initialize p = uniform
2. For t = 1, 2, ...:
   a. Solve (approximately) h* = argmin_h sum_k p_k r_k(h) subject to constraints
   b. Update p_k ∝ p_k * exp(η * r_k(h*))
   
This is a primal-dual method. The dual variable p represents which lags are "active" (contributing to the max), and the inner problem tries to minimize the weighted sum.

The advantage is that the inner problem is a single quadratic (not a max of quadratics), which is easier to optimize. And the outer multiplicative weights automatically focuses on the worst lags.

Let me implement this:

```python
def multiplicative_weights(h_init, n, target, budget, rng):
    dx = 2.0 / n
    nlags = 2 * n - 1
    p = np.ones(nlags) / nlags  # uniform initialization
    eta = 0.5  # learning rate
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start = time.time()
    while time.time() - start < budget:
        # Inner: minimize sum_k p_k * r_k(h)
        # Use projected gradient descent or SLSQP
        h_new = inner_solve(h, p, n, target, rng)
        
        # Evaluate
        corr = compute_corr(h_new)
        c5 = float(np.max(corr)) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        # Outer: multiplicative weights update
        # p_k ∝ p_k * exp(eta * r_k)
        logits = np.log(p) + eta * corr
        logits -= np.max(logits)  # for numerical stability
        p = np.exp(logits)
        p /= np.sum(p)
        
        h = h_new
    
    return best_h, best_c5
```

For the inner solve, I'll use projected gradient descent with the analytic gradient of the weighted sum.

The gradient of sum_k p_k r_k(h) with respect to h[i]:
= sum_k p_k * [(1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]]

This is a convolution and can be computed efficiently.

                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': tol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < score:
            return h_new, sc
    except Exception:
        pass
    
    return h_cur, score

def transfer_anneal(h0, n, target, deadline, rng, T_init=0.01, T_min=1e-5, cooling=0.999):
    dx = 2.0 / n
    h_cur = h0.copy()
    best_h = h0.copy()
    best_score = compute_c5(h_cur)
    corr = compute_corr(h_cur)
    T = T_init
    
    n_stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        
        if T < T_min:
            # Reset with restart
            h_cur = best_h + rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            T = T_init * 0.5
            n_stall = 0
            continue
        
        # Pick random i, j
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        
        # Transfer amount
        max_pos = min(1.0 - h_cur[j], h_cur[i])
        max_neg = min(h_cur[j], 1.0 - h_cur[i])
        # Choose δ uniformly in [-max_neg, max_pos]
        delta = rng.uniform(-max_neg, max_pos)
        if abs(delta) < 1e-14:
            continue
        
        # Apply
        h_cur[i] += delta
        h_cur[j] -= delta
        
        # Compute new correlation (only need to update affected entries)
        # For efficiency, just recompute fully (n=400 is fast)
        corr_new = compute_corr(h_cur)
        score_new = float(np.max(corr_new)) * dx
        delta_score = score_new - best_score
        
        if delta_score <= 0 or rng.rand() < np.exp(-delta_score / T):
            corr = corr_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_cur.copy()
                n_stall = 0
            else:
                n_stall += 1
        else:
            # Revert
            h_cur[i] -= delta
            h_cur[j] += delta
            n_stall += 1
        
        T *= cooling
    
    return best_h, best_score

def transfer_anneal_smoothed(h0, n, target, deadline, rng, 
                              T_init=0.01, T_min=1e-5, cooling=0.9995,
                              alpha_init=100, alpha_final=10000):
    dx = 2.0 / n
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h_cur)
    corr = compute_corr(h_cur)
    
    def smooth_energy(c):
        c_max = np.max(c)
        shifted = alpha * (c - c_max)
        shifted = np.clip(shifted, -700, 0)
        return (c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    T = T_init
    alpha = alpha_init
    total_iters = 0
    start_t = time.time()
    
    while time.time() < deadline:
        total_iters += 1
        if T < T_min:
            h_cur = best_h + rng.randn(n) * 0.01
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            T = T_init * 0.5
            continue
        
        # Update alpha (increase over time)
        frac = (time.time() - start_t) / max(deadline - start_t, 1)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        
        max_pos = min(1.0 - h_cur[j], h_cur[i])
        max_neg = min(h_cur[j], 1.0 - h_cur[i])
        if max_pos < 1e-10 and max_neg < 1e-10:
            continue
        delta = rng.uniform(-max_neg, max_pos)
        if abs(delta) < 1e-12:
            continue
        
        h_cur[i] += delta
        h_cur[j] -= delta
        
        corr_new = compute_corr(h_cur)
        e_old = smooth_energy(corr)
        e_new = smooth_energy(corr_new)
        de = e_new - e_old
        
        c5_new = float(np.max(corr_new)) * dx
        if de <= 0 or rng.rand() < np.exp(-de / T):
            corr = corr_new
            if c5_new < best_c5:

                best_h = h_cur.copy()
        else:
            h_cur[i] -= delta
            h_cur[j] += delta
        
        T *= cooling
    
    return best_h, best_c5
