import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_weights = None


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_opt(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def nesterov_wgd(h, n, target, n_steps, beta_init, beta_growth, lr_init, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.85
    stalled = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        total = weights.sum()
        if total < 1e-30:
            weights = np.ones(n) / n
            total = n
        weights /= total

        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx

        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            stalled = 0
        else:
            stalled += 1
            v *= 0.5
            lr = max(lr * 0.95, 1e-7)
            if stalled > 50:
                lr = max(lr * 0.5, 1e-8)
                v *= 0.3
                stalled = 0
        if (step + 1) % 40 == 0:
            beta = min(beta * beta_growth, 2e6)
    return h_cur, cur_score


def top_k_surgery(h, n, target, top_k, max_steps, eps_init, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    no_improve = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        top = np.argsort(over)[::-1][:top_k]
        improved = False
        for k_idx in top:
            k = int(k_idx)
            grad = np.zeros(n)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= h_cur[j_neg - k]
            grad *= dx
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc_new
                improved = True
        if improved:
            eps = min(eps * 1.015, 0.008)
            no_improve = 0
        else:
            eps *= 0.85
            no_improve += 1
            if no_improve > 30 or eps < 1e-8:

    return h_cur, cur_score

        h = project_feasible(h, target)
        pop.append(h)
    scores = [compute_c5(h) for h in pop]
    order = np.argsort(scores)
    pop = [pop[i] for i in order]
    scores = [scores[i] for i in order]
    best = pop[0].copy()
    best_sc = scores[0]
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial = project_feasible(trial, target)
            sc = compute_c5(trial)
            if sc <= scores[i]:
                pop[i] = trial
                scores[i] = sc
                if sc < best_sc:
                    best_sc = sc
                    best = trial.copy()
        order = np.argsort(scores)
        pop = [pop[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_sc

    n_src = len(h_src)

            if time.time() > deadline_p5:
                break
            best_h, sc = top_k_surgery(best_h, n, target, top_k=tk, max_steps=400,
                                       eps_init=0.003, deadline=deadline_p5)
            if sc < best_score:
                best_score = sc
                best_h = project_feasible(best_h, target)

    # Phase 6: SLSQP with high alpha
    if remaining() > 120:
        deadline_p6 = min(start_time + 900, start_time + budget_s * 0.82)
        best_h, sc = slsqp_opt(best_h, n, target, [20000, 60000, 150000], 3000, 1e-17,
                               deadline_p6)
        if sc < best_score:
            best_score = sc
            best_h = project_feasible(best_h, target)

    # Phase 7: Multi-restart with perturbation
    if remaining() > 150:
        deadline_p7 = min(start_time + budget_s - 120, start_time + budget_s * 0.92)
        rng_restart = np.random.RandomState(seed + 777)
        best_h, sc = multi_restart(best_h, n, target, n_restarts=2,
                                   scales=[0.03, 0.01], deadline=deadline_p7,
                                   rng=rng_restart)
        if sc < best_score:
            best_score = sc
            best_h = project_feasible(best_h, target)

    # Phase 8: Final surgery + SLSQP
    if remaining() > 60:
        deadline_p8 = start_time + budget_s - 50
        best_h, sc = top_k_surgery(best_h, n, target, top_k=2, max_steps=300,
                                   eps_init=0.001, deadline=min(deadline_p8 - 40, time.time() + 60))
        if sc < best_score:
            best_score = sc
            best_h = project_feasible(best_h, target)

    if remaining() > 40:
        deadline_final = start_time + budget_s - 15
        best_h, sc = slsqp_opt(best_h, n, target, [200000, 500000], 4000, 1e-17,
                               deadline_final)
        if sc < best_score:
            best_score = sc

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
