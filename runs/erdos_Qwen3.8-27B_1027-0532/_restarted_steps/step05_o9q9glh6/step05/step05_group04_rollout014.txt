    order = np.argsort(scores)
    pop_h = [pop_h[i] for i in order]
    scores = [scores[i] for i in order]
    best = pop_h[0].copy()
    best_sc = scores[0]
    for g in range(gens):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop):
            cands = [j for j in range(pop) if j != i]
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            mutant = pop_h[r1] + F * (pop_h[r2] - pop_h[r3])
            trial = np.zeros(n)

                    trial[j] = pop_h[i][j]

                scores[i] = sc
                if sc < best_sc:
                    best_sc = sc
                    best = trial.copy()
        order = np.argsort(scores)
        pop_h = [pop_h[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_sc

        _g_best_val = target_val
        _g_alpha = alpha
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}]
        try:
            res = minimize(_obj_target, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-18})
            h_new = _project(res.x, target)
            sc = _eval_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new.copy()
            h_cur = h_new
            if sc < target_val - 1e-6:
                target_val = sc + 0.5e-6
        except Exception:
            target_val += 5e-5
            break
    return best_h, best_sc


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_lags, _g_best_val
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = _project(np.array(initial_h_values, dtype=np.float64), target)
    best_h = h_init.copy()
    best_sc = _eval_c5(best_h)
    dx = 2.0 / n

    # Phase 1: Multi-start DE on full grid
    deadline1 = start_time + 80.0
    if _tleft() > 100:
        de_results = _de_multi(n, target, pop=40, gens=400, rng=rng,
                               n_variants=4, deadline=deadline1)
        for sc, h in de_results:
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()

    # Phase 2: SLSQP with alpha annealing on best candidate
    deadline2 = start_time + 250.0
    if _tleft() > 150:
        h_ref = _slsqp_solve(best_h, n, target, 20.0, maxiter=3000, ftol=1e-15)
        sc = _eval_c5(h_ref)
        if sc < best_sc:
            best_sc = sc
            best_h = h_ref
        for a in [50.0, 100.0, 300.0, 800.0]:
            if _tleft() < 60:
                break
            h_ref = _slsqp_solve(best_h, n, target, a, maxiter=3000, ftol=1e-16)
            sc = _eval_c5(h_ref)
            if sc < best_sc:
                best_sc = sc
                best_h = h_ref
            h_ref = _slsqp_solve(best_h, n, target, a, maxiter=2000, ftol=1e-16)
            sc = _eval_c5(h_ref)
            if sc < best_sc:
                best_sc = sc
                best_h = h_ref

    # Phase 3: Nesterov accelerated projected gradient
    deadline3 = start_time + 450.0
    if _tleft() > 200:
        h_nes, sc_nes = _bb_proj_grad(best_h, n, target, max_steps=800,
                                      alpha_init=50.0, alpha_growth=1.05,
                                      lr_init=0.002, use_nesterov=True,
                                      deadline=deadline3)
        if sc_nes < best_sc:
            best_sc = sc_nes
            best_h = h_nes
        h_ref = _slsqp_solve(best_h, n, target, 500.0, maxiter=3000, ftol=1e-16)
        sc = _eval_c5(h_ref)
        if sc < best_sc:
            best_sc = sc
            best_h = h_ref

    # Phase 4: Active set optimization (top K lags)
    deadline4 = start_time + 700.0
    if _tleft() > 200:
        for k_act in [3, 5, 8]:
            if _tleft() < 80:
                break
            for a in [200.0, 500.0, 1500.0]:
                if _tleft() < 40:
                    break
                h_act, sc_act = _active_set_opt(best_h, n, target,
                                                n_rounds=4, k_active=k_act,
                                                alpha=a, deadline=deadline4)
                if sc_act < best_sc:
                    best_sc = sc_act
                    best_h = h_act

    # Phase 5: Full SLSQP with high alpha
    deadline5 = start_time + 900.0
    if _tleft() > 150:
        for a in [3000.0, 8000.0, 15000.0]:
            if _tleft() < 50:
                break
            h_ref = _slsqp_solve(best_h, n, target, a, maxiter=4000, ftol=1e-17)
            sc = _eval_c5(h_ref)
            if sc < best_sc:
                best_sc = sc
                best_h = h_ref

    # Phase 6: Alternating single-lag minimization
    deadline6 = start_time + 1050.0
    if _tleft() > 120:
        for a in [2000.0, 5000.0, 10000.0]:
            if _tleft() < 40:
                break
            h_alt, sc_alt = _alternating_min(best_h, n, target,
                                             n_rounds=8, alpha=a,
                                             deadline=deadline6)
            if sc_alt < best_sc:
                best_sc = sc_alt
                best_h = h_alt

    # Phase 7: Target descent (try to push below current best)
    deadline7 = start_time + 1180.0
    if _tleft() > 80:
        h_tgt, sc_tgt = _target_descent(best_h, n, target, n_steps=30,
                                        alpha=500.0, lr_init=0.001,
                                        deadline=deadline7)
        if sc_tgt < best_sc:
            best_sc = sc_tgt
            best_h = h_tgt

    # Phase 8: Final micro-polish
    deadline8 = start_time + budget_s - 10.0
    if _tleft() > 20:
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        top3 = np.argsort(c)[::-1][:3]
        lags3 = np.array(sorted(top3.tolist()))
        h_fin = _slsqp_solve(best_h, n, target, 20000.0, maxiter=3000,
                             ftol=1e-18, use_active=True, lags=lags3)
        sc = _eval_c5(h_fin)
        if sc < best_sc:
            best_sc = sc
            best_h = h_fin
        worst = int(np.argmax(np.correlate(best_h, 1.0 - best_h, mode='full')))
        h_fin2 = _slsqp_solve(best_h, n, target, 30000.0, maxiter=2000,
                              ftol=1e-18, use_active=True, lags=np.array([worst]))
        sc = _eval_c5(h_fin2)
        if sc < best_sc:
            best_sc = sc
            best_h = h_fin2

    final_h = _project(best_h, target)
    final_sc = _eval_c5(final_h)
    return final_h, final_sc, n
