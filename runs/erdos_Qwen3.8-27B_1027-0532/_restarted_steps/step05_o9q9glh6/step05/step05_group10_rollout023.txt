def compute_grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        k_abs = abs(k)
        if k_abs >= n:
            continue
        if k >= 0:
            grad[:n-k_abs] += w * (1.0 - h)[k_abs:]
            grad[k_abs:] -= w * h[:n-k_abs]
        else:
            grad[k_abs:] += w * (1.0 - h)[:n-k_abs]
            grad[:n-k_abs] -= w * h[k_abs:]
    
    grad *= dx
    return grad

Actually, let me think about a different approach: **alternating between the "h" variables and the "k" selection**.

The problem is: min_h max_k g_k(h)

By Sion's minimax theorem (if applicable), this equals max_{p in simplex} min_h sum_k p_k * g_k(h)

For fixed p, the inner problem is min_h sum_k p_k * g_k(h) subject to constraints. This is a QP (quadratic program)!

sum_k p_k * g_k(h) = sum_k p_k * [sum_i h[i](1-h[i+k])]
= sum_i h[i] * sum_k p_k * (1-h[i+k])
= sum_i h[i] - sum_{i,k} p_k * h[i] * h[i+k]

This is a quadratic function of h. Minimizing a (non-convex) quadratic over a polytope is hard in general, but for specific structure it might be tractable.

Actually, the quadratic form is:
f(h) = sum_i h[i] - sum_{i,k} p_k * h[i] * h[i+k]

The Hessian has the form: H[i,j] = -sum_{k: j-i=k} p_k

So H is a (weighted) centrosymmetric matrix. This is a Toeplitz-like matrix!

This is interesting because it means the quadratic form has a specific structure that might be exploitable.

For the inner QP: min f(h) s.t. 0 ≤ h ≤ 1, sum(h) = n/2

This is a non-convex QP (the Hessian can have positive eigenvalues), so it's NP-hard in general. But we can use:

Let me also think about whether I can speed up the gradient computation by vectorizing the loop over k.

Actually, I can! The gradient is:
grad = sum_k w_k * (contribution of shift k)

For k ≥ 0:
contribution[j] = (1-h[j+k]) for j in [0, n-1-k] minus h[j-k] for j in [k, n-1]

This can be written as:
contribution = (1-h) with a shift, minus h with a shift

Specifically:
contribution[:n-k] += (1-h)[k:]  (which is (1-h[k:n]))
contribution[k:] -= h[:n-k]  (which is h[0:n-k])

For k < 0 (let k' = -k):
contribution[k':] += (1-h)[:n-k']  (which is (1-h[0:n-k']))
contribution[:n-k'] -= h[k':]  (which is h[k':n])

I can precompute the "shifted" arrays and use the weights to combine them. But since n is only 400 and 2n-1 is 799, the loop is fine.

However, I can use a trick: note that the weights w_k are concentrated around the argmax of corr (for large alpha). So I only need to process the top ~50 shifts.

Let me add this optimization:

```python
def compute_grad_smooth(h, alpha, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
    weights = np.exp(shifted)
    total = weights.sum()
    
    if total < 1e-300:
        # Only the max matters
        top_indices = [np.argmax(corr)]
        weights = np.zeros(len(corr))
        weights[top_indices[0]] = 1.0
    else:
        weights /= total
        # Only keep top-k
        if len(corr) > top_k:
            top_indices = np.argsort(weights)[-top_k:]
            mask = np.zeros(len(corr), dtype=bool)
            mask[top_indices] = True
            weights = np.where(mask, weights, 0.0)
            # Renormalize
            weights /= weights.sum()
    
    grad = np.zeros(n)
    half = n - 1
    for k_idx in np.where(weights > 0)[0]:
        w = weights[k_idx]
        k = k_idx - half
        ka = abs(k)
        if ka >= n:
            continue
        if k >= 0:
            grad[:n-ka] += w * (1.0 - h[ka:])
            grad[ka:] -= w * h[:n-ka]
        else:
            grad[ka:] += w * (1.0 - h[:n-ka])
            grad[:n-ka] -= w * h[ka:]
    grad *= dx
    return grad
```

This reduces the loop to at most 50 iterations, making it 5x faster.

where `g = 1 - h` (current iterate).

This constraint is: sum_i g[i-k] * h_new[i] * dx ≤ t

Which is: dx * sum_{valid i} (1-h[i-k]) * h_new[i] ≤ t

This is NOT corr_k(h_new). It's a linear approximation/upper bound on corr_k(h_new).

Actually, let me think about when this is a valid upper bound:
corr_k(h_new) = sum_{valid i} h_new[i] * (1-h_new[i+k])

If h_new[i+k] ≥ 0, then 1-h_new[i+k] ≤ 1, so:
corr_k(h_new) ≤ sum_{valid i} h_new[i]

That's just a trivial bound. The LP constraint is tighter because it uses (1-h[i+k]) instead of 1.

But it's not an upper bound in general! If h_new[i+k] < h[i+k], then 1-h_new[i+k] > 1-h[i+k], so the actual corr could be larger.

So the LP is solving a relaxed problem. The solution might not be feasible for the original problem. But with the damping and projection, it still makes progress in practice.

import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
    return (max_corr + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def compute_grad_smooth(h, alpha, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)

        if top_k < len(corr):

            mask[top_indices] = True
            weights = np.where(mask, weights, 0.0)

        else:
            if n - ka > 0:

            grad[:n-ka] -= w * h[ka:]
    grad *= dx
    return grad

def nesterov_round(h0, target, alpha, max_iter, start_time, deadline):
    """One round of Nesterov-accelerated projected gradient at fixed alpha."""
    h = h0.copy()
    y = h0.copy()
    t_mom = 1.0
    step = 10.0 / alpha
    best_h = h0.copy()
    best_score = compute_c5(h0)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        g = compute_grad_smooth(y, alpha)
        f_y = smooth_obj_val(y, alpha)
        
        # Backtracking line search
        s = step
        h_try = project_feasible(y - s * g, target)
        f_try = smooth_obj_val(h_try, alpha)
        bt = 0
        while f_try > f_y - 0.4 * s * np.dot(g, h_try - y) and bt < 20:
            s *= 0.5
            h_try = project_feasible(y - s * g, target)
            f_try = smooth_obj_val(h_try, alpha)
            bt += 1
        step = s
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = h_try + ((t_mom - 1.0) / t_new) * (h_try - h)
        y = project_feasible(y, target)
        t_mom = t_new
        h = h_try
        
        score = compute_c5(h)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            break
    
    return best_h, best_score
