import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    # Water-filling projection onto {0 <= h <= 1, sum(h) = target}
    h = np.asarray(h, dtype=float)
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # FFT-based gradient
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        # Backtracking
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
            shrink = 0.5
        else:
            step = max(step, 1e-14)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink) if shrink < 1 else step*2, lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def multi_lag_polish(h, target, max_rounds, deadline):
    """Polish by targeting multiple top lags simultaneously with adaptive temperature."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        # Get top m lags with adaptive weights
        sorted_idx = np.argsort(corr)[::-1]
        m = min(5, n)
        active = sorted_idx[:m]
        max_val = corr[active[0]]
        temperature = 0.01  # Controls how sharply weights concentrate
        diffs = (corr[active] - max_val) / temperature
        w = np.exp(diffs)
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            if ki == 0:
                # corr[0] = sum h[i](1-h[i]), gradient is (1-2h[i])
                grad += wi * (1.0 - 2.0 * h_cur)
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += wi * (1.0 - h_cur[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        # Line search
        for eps_try in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps_try * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                # Update temperature to be more aggressive
                temperature = max(temperature * 0.7, 0.001)
                break
        else:
            # Try increasing temperature (more spread out)
            temperature = min(temperature * 1.5, 0.1)
    
    return h_cur, cur_score

def block_optimize(h, target, deadline, n_blocks=50, block_size_range=(5, 25)):
    """Optimize random blocks using Nelder-Mead."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(777)
    
    attempts = 0
    while time.time() < deadline and attempts < n_blocks:
        block_size = rng.randint(block_size_range[0], block_size_range[1] + 1)
        start = rng.randint(0, max(1, n - block_size))
        indices = np.arange(start, start + block_size)
        
        # Define objective for the block
        def block_obj(block_vals):
            h_trial = h_cur.copy()
            h_trial[indices] = block_vals
            h_trial = project_feasible(h_trial, target)
            return compute_c5(h_trial)
        
        x0 = h_cur[indices].copy()
        res = minimize(block_obj, x0, method='Nelder-Mead',
                      options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-10})

            h_trial = h_cur.copy()
            h_trial[indices] = res.x
            h_cur = project_feasible(h_trial, target)
            cur_score = compute_c5(h_cur)
        
        attempts += 1
    
    return h_cur, cur_score

                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
                else:
                    h_cur[idx] = old_val
                    h_cur = project_feasible(h_cur, target)
            else:
                for fr in [0.2, 0.8, 0.4, 0.6, 0.0, 1.0]:
                    trial = old_val * fr
                    if abs(trial - old_val) < 1e-10:
                        continue
                    h_cur[idx] = trial
                    h_cur = project_feasible(h_cur, target)
                    sc = compute_c5(h_cur)
                    if sc < cur_score - 1e-14:
                        cur_score = sc
                        improved = True
                        break
                    h_cur[idx] = old_val
                h_cur = project_feasible(h_cur, target)
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    n = len(initial_h_values)
    target = n / 2.0

    h_init = np.array(initial_h_values, dtype=float)
    h_init = project_feasible(h_init, target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA continuation with increasing alpha
    alphas = [80, 400, 1500, 6000, 20000, 80000]
    time_alloc = [70, 60, 70, 80, 90, 100]
    for i, (alpha, td) in enumerate(zip(alphas, time_alloc)):
        remaining = deadline - time.time()
        if remaining < 300:
            break
        t_end = time.time() + min(td, remaining - 300)
        h_new, sc = fista_optimize(best_h, target, alpha, 200000, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Multi-lag subgradient polish
    remaining = deadline - time.time()
    if remaining > 120:
        t_end = time.time() + min(90, remaining - 110)
        h_new, sc = multi_lag_polish(best_h, target, 1000, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: High-alpha FISTA
    remaining = deadline - time.time()
    if remaining > 100:
        t_end = time.time() + min(80, remaining - 90)
        h_new, sc = fista_optimize(best_h, target, 150000, 200000, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Block optimization
    remaining = deadline - time.time()
    if remaining > 130:
        t_end = time.time() + min(110, remaining - 100)
        h_new, sc = block_optimize(best_h, target, t_end, max_blocks=35)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Frequency perturbations with re-optimization
    remaining = deadline - time.time()
    if remaining > 80:
        t_end = time.time() + min(60, remaining - 60)
        h_new, sc = freq_perturb(best_h, target, t_end, seed)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Another FISTA pass after perturbations
    remaining = deadline - time.time()
    if remaining > 60:
        t_end = time.time() + min(40, remaining - 40)
        h_new, sc = fista_optimize(best_h, target, 50000, 100000, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Multi-lag polish again
    remaining = deadline - time.time()
    if remaining > 50:
        t_end = time.time() + min(35, remaining - 30)
        h_new, sc = multi_lag_polish(best_h, target, 500, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Coordinate polish
    remaining = deadline - time.time()
    if remaining > 30:
        t_end = time.time() + min(25, remaining - 10)
        h_new, sc = coord_polish(best_h, target, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final: one more multi-lag pass with remaining time
    remaining = deadline - time.time()
    if remaining > 15:
        t_end = time.time() + remaining - 5
        h_new, sc = multi_lag_polish(best_h, target, 500, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
