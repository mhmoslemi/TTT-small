import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h = h.copy()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    n = len(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def gradient_at_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        upper = n - 1 - k
        idx1 = np.arange(0, upper + 1)
        grad[idx1] += (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= h[idx2 - k]
    return grad

def top_lag_gradient_descent(h, n_steps, n_active, lr_init, lr_min, temp_init, temp_decay, backtracking=5):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    temp = temp_init
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx_val
        dx_val = 2.0 / n
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(-diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            grad += w * gradient_at_lag(h_cur, int(k))
        improved = False
        for bt in range(backtracking):
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < c_curr - 1e-14:
                h_cur = h_new
                improved = True
                break
            lr *= 0.5
        if improved:
            lr = min(lr * 1.05, 0.08)
        else:
            lr = max(lr * 0.7, lr_min)
            temp *= temp_decay
            if lr < lr_min:
                break
    return best_h, best_score

def multi_start_gradient(h, n_starts, n_steps_each, n_active, lr_init, temp_init, seed_val):
    rng = np.random.RandomState(seed_val)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()

    dx_val = 2.0 / n
    for s in range(n_starts):

        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    return best_h, best_score

def block_coordinate_descent(h, block_size, n_rounds, n_active, lr_init):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    h_best = h_cur.copy()
    lr = lr_init
    for rnd in range(n_rounds):
        n_blocks = n // block_size
        for b in range(n_blocks):
            sl = b * block_size
            sr = sl + block_size
            corr = compute_corr(h_cur)
            over = corr[n-1:n-1+n]
            top_idx = np.argsort(over)[::-1][:n_active]
            diffs = over[top_idx] - over[top_idx[0]]
            weights = np.exp(-diffs / 0.01)
            weights /= weights.sum()
            grad_full = np.zeros(n)
            for w, k in zip(weights, top_idx):
                grad_full += w * gradient_at_lag(h_cur, int(k))
            block_grad = grad_full[sl:sr]
            h_block = h_cur[sl:sr]
            for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1]:
                h_block_new = h_block - lr_try * block_grad
                h_block_new = np.clip(h_block_new, 0.0, 1.0)
                h_trial = h_cur.copy()
                h_trial[sl:sr] = h_block_new
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_score:
                    h_cur = h_trial
                    h_best = h_trial.copy()
                    best_score = sc
                    break
        lr = max(lr * 0.98, 1e-6)
    return h_best, best_score

                best_h = h_fine
                n = n_fine
                target = n / 2.0

    # Phase 9: Final SLSQP polish
    if tleft() > 60:
        h_ref, sc = slsqp_optimize(best_h, [50000, 200000, 500000], 4000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final gradient descent
    if tleft() > 30:
        h_wl, sc_wl = top_lag_gradient_descent(best_h, 1000, 5, 0.003, 1e-9, 0.004, 0.999)
        if sc_wl < best_score:
            best_score = sc_wl
            best_h = h_wl

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
