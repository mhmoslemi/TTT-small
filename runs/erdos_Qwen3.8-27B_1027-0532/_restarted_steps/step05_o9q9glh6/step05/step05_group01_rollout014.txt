def slp_cutting_plane(h_init, n, target, t_target, max_iter=50, trust_radius=0.2):
    h = h_init.copy()
    dx = 2.0 / n
    t_limit = t_target / dx
    mid = n - 1
    
    active_constraints = list(range(2*n - 1))  # Start with all
    
    for it in range(max_iter):
        # Compute current correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Update active set: constraints where linearized value is close to t_limit
        h_new_var = cp.Variable(n)
        constraints = [h_new_var >= 0, h_new_var <= 1, cp.sum(h_new_var) == target]
        
        # Trust region
        constraints.append(cp.norm(h_new_var - h, np.inf) <= trust_radius)
        
        # Add active constraints (linearized)
        for p in active_constraints:
            k = p - mid
            # Compute gradient of c_p w.r.t. h
            grad = np.zeros(n)
            j = np.arange(n)
            v1 = (j - k >= 0) & (j - k < n)
            v2 = (j + k >= 0) & (j + k < n)
            grad[v1] += (1.0 - h[j[v1] - k])
            grad[v2] -= h[j[v2] + k]
            
            # Linearized: corr[p] + grad @ (h_new - h) <= t_limit
            constraints.append(corr[p] + grad @ (h_new_var - h) <= t_limit)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                break
        
        if h_new_var.value is None:
            # Infeasible - need to increase t or reduce trust region
            break
        
        h_new = np.clip(h_new_var.value, 0, 1)
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        if np.max(corr_new) * dx < np.max(corr) * dx:
            h = h_new
            # Update active set
            tol = 0.01
            active_constraints = list(np.where(corr_new >= np.max(corr_new) - tol * dx)[0])
        
        # Check convergence
        if np.max(corr) * dx <= t_target + 1e-8:
            break
    
    return h, np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

import numpy as np

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):

    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_smooth(h_start, n, target, alpha, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    def cons_fun(v):
        return np.sum(v) - target

    def cons_jac(v):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLP after pair CD
    if tleft() > 100:
        h_slp, c5_slp = slp_optimize(best_h, n, target, time_limit=min(80, tleft() * 0.4))
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    # Phase 4: Basin hopping with SLP
    rng = np.random.RandomState(seed)
    n_hops = 8
    hop_budget = min(250, tleft() * 0.25)
    hop_per = hop_budget / n_hops if n_hops > 0 else 0

    for hop in range(n_hops):
        if tleft() < 100 or hop_per <= 0:
            break
        scale = rng.choice([0.02, 0.03, 0.05, 0.07, 0.1])
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        h_hop, c5_hop = slp_optimize(h_pert, n, target, time_limit=hop_per, trust_init=0.25)
        if c5_hop < best_score:
            best_score = c5_hop
            best_h = h_hop

        if tleft() > 40:
            h_hop2, c5_hop2 = pair_cd(h_hop, n, target, time_limit=min(20, tleft() * 0.1),
                                      max_sweeps=1500, n_cand=40)
            if c5_hop2 < best_score:
                best_score = c5_hop2
                best_h = h_hop2

    # Phase 5: Final pair CD + SLP
    if tleft() > 100:
        cd_time = min(100, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time, max_sweeps=4000, n_cand=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_slp, c5_slp = slp_optimize(best_h, n, target, time_limit=min(45, tleft() * 0.4))
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    # Phase 6: SLSQP polish
    if tleft() > 50:
        for alpha in [500, 2000, 8000]:
            if tleft() < 20:
                break
            h_pol = slsqp_smooth(best_h, n, target, alpha, 800, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 7: Higher resolution
    for n_try in [600, 800]:
        if tleft() < 180:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        h_hires, c5_hires = slp_optimize(h_interp, n_try, target_try,
                                         time_limit=min(70, tleft() * 0.3), trust_init=0.25)
        if c5_hires < best_score:
            best_score = c5_hires
            best_h = h_hires

        if tleft() > 60:
            cd_time = min(50, tleft() * 0.25)
            h_cd, c5_cd = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                  time_limit=cd_time, max_sweeps=2500, n_cand=40)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        if tleft() > 40:
            h_slp, c5_slp = slp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                         time_limit=min(30, tleft() * 0.3))
            if c5_slp < best_score:
                best_score = c5_slp
                best_h = h_slp

    # Phase 8: Final polish round
    if tleft() > 40:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_cd, c5_cd = pair_cd(best_h, n_use, t_use,
                              time_limit=min(25, tleft() * 0.4), max_sweeps=2000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 25:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_slp, c5_slp = slp_optimize(best_h, n_use, t_use,
                                     time_limit=min(20, tleft() * 0.5))
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    # Final feasibility check
    n_use = len(best_h)
    t_use = n_use / 2.0
    h_final = project_feasible(best_h, t_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
