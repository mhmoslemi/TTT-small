import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        nfree = float(np.sum(free))
        if nfree == 0:
            break
        h[free] += diff / nfree
    return h


def project_simplex_bounds(x, target_sum):
    lo = np.min(x) - 2.0
    hi = np.max(x) + 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-14:
            break
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def lse_obj_grad(h, alpha, target_sum, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = c_max + np.log(sum_exp) / alpha + penalty * (np.sum(h) - target_sum) ** 2
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wm * (1.0 - h[i_lo + k:i_hi + k])
        i_lo2 = max(0, k)
        i_hi2 = min(n, n + k)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wm * (-h[i_lo2 - k:i_hi2 - k])
    grad *= dx
    grad += 2.0 * penalty * (np.sum(h) - target_sum)
    return obj, grad


def lse_optimize(h_init, alpha, target_sum, penalty=50.0, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj_grad, h_init, args=(alpha, target_sum, penalty),
                       method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': 50000})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h_init.copy(), target_sum)


def get_top_lags(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(corr)[::-1][:K]
    return top_idx - (n - 1)


def alternating_lp(h_init, n, target_sum, max_iter, time_limit, n_top=150,
                   damping_start=0.9, damping_end=0.75, start_time=None):
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if it > 100 and stuck > 200:
            break
        progress = min(1.0, it / max(1, max_iter * 0.5))
        damping = damping_start + (damping_end - damping_start) * progress
        g = 1.0 - h
        lags = get_top_lags(h, n_top)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        for k in lags:
            i_lo = max(0, -k)
            i_hi = min(n, n - k)
            if i_hi <= i_lo:
                continue
            coeff = np.zeros(n)
            coeff[i_lo:i_hi] = g[i_lo + k:i_hi + k] * dx
            constraints.append(coeff @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target_sum)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def proximal_gradient(h_init, n, target_sum, time_limit, alpha_start=100,
                      alpha_end=5000, n_iters=5000, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, it / max(1, n_iters))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        _, grad = lse_obj_grad(h, alpha, target_sum, penalty=0.0)
        y = h + momentum * (h - h_prev)
        lr = 1.0 / (np.linalg.norm(grad) + 1e-10)
        lr = min(lr, 0.1)
        h_new = y - lr * grad
        h_new = project_simplex_bounds(h_new, target_sum)
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_new.copy()
        h_prev = h.copy()
        h = h_new
    return best_h, best_c5

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    return h_final, final_score, n_final
