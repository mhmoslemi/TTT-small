import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    weights = alpha * (corr - c_max)
    weights = np.clip(weights, -700.0, 0.0)
    exp_w = np.exp(weights)
    exp_w /= exp_w.sum()
    grad = np.zeros(n)
    over = corr[n-1:n-1+n]
    over_weights = exp_w[n-1:n-1+n]
    for k in range(n):
        wk = over_weights[k]
        if wk < 1e-12:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    return grad

def slsqp_smooth(h, n, target, alphas, maxiter, ftol, deadline=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h.copy()

            break

            return lse_objective(x, a)

                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def fourier_init(n, n_harmonics=3):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for m in range(1, n_harmonics + 1):
        amp = 0.15 / m
        phase = m * 0.7
        h += amp * np.sin(np.pi * m * x + phase)
    h = project_feasible(h, n / 2.0)
    return h

def thue_morse_smooth(n, smooth_bw=5):
    tm = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = 1.0 if c % 2 == 0 else 0.0
    kernel = np.exp(-0.5 * (np.arange(-smooth_bw, smooth_bw + 1) / (smooth_bw / 2.0))**2)
    kernel /= kernel.sum()
    h = np.convolve(tm, kernel, mode='same')
    h = (h - h.min()) / (h.max() - h.min() + 1e-12)
    h = 0.5 + 0.5 * (h - 0.5) * 2.0
    h = project_feasible(h, n / 2.0)
    return h
