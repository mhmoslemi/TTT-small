def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu, topk=None):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    if topk is not None and topk < len(corr):
        # Only use top K in LSE
        idx = np.argsort(corr)[::-1][:topk]
        corr_sel = corr[idx]
    else:
        corr_sel = corr
        idx = np.arange(len(corr))
    shifted = alpha * (corr_sel - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse_val = float(np.log(np.sum(np.exp(shifted)))) / alpha
    val = c_max + lse_val
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu, topk=None):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    if topk is not None and 0 < topk < len(corr):
        idx = np.argsort(corr)[::-1][:topk]
        corr_sel = corr[idx]
    else:
        corr_sel = corr
        idx = np.arange(len(corr))
    shifted = alpha * (corr_sel - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse_val = float(np.log(np.sum(np.exp(shifted)))) / alpha
    val = c_max + lse_val
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr_sel)) / len(corr_sel)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ai in active:
        w = weights[ai]
        ki = int(idx[ai])
        lag = ki - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline, topk=None):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        return lse_obj_and_grad(x, alpha, target, mu, topk)[0]

    def grad_fn(x):
        return lse_obj_and_grad(x, alpha, target, mu, topk)[1]

    try:
        res = minimize(obj_fn, h_cur, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def top_lag_equalize(h, target, n_top=5, n_steps=200, deadline=None):

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.01
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return h_cur, cur_score


def direct_max_reduction(h, target, n_steps=200, deadline=None, n_top=3):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.02
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        grad = np.zeros(n)
        for k_star in top_idx:
            lag = int(k_star) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= n_top
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.02, 0.08)
        else:
            lr *= 0.8
            if lr < 1e-9:
                break
    return h_cur, cur_score


def binary_local_search(h, target, max_rounds=3, deadline=None):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    n_needed = int(round(target))
    for _ in range(10):
        n_ones = int(h_bin.sum())
        if n_ones == n_needed:
            break
        if n_ones > n_needed:
            candidates = np.where(h_bin == 1)[0]
        else:
            candidates = np.where(h_bin == 0)[0]
        best_flip = None
        best_sc = compute_c5(h_bin)
        if len(candidates) > 80:
            candidates = np.random.choice(candidates, 80, replace=False)
        for j in candidates:
            h_bin[j] = 1.0 - h_bin[j]
            sc = compute_c5(h_bin)
            if sc < best_sc:
                best_sc = sc
                best_flip = j
            else:
                h_bin[j] = 1.0 - h_bin[j]
        if best_flip is not None:
            h_bin[best_flip] = 1.0 - h_bin[best_flip]
        else:
            break
    cur_score = compute_c5(h_bin)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            h_bin[j] = 1.0 - h_bin[j]
            sc = compute_c5(h_bin)
            if sc < cur_score - 1e-12:
                cur_score = sc
                improved = True
            else:
                h_bin[j] = 1.0 - h_bin[j]
        if not improved:
            break
    return h_bin, cur_score


def fista_optimize(h, target, alpha, mu, step_size, maxiter, deadline):
    n = len(h)
    x = h.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    for it in range(maxiter):
        if time.time() > deadline:
            break
        _, grad = lse_obj_and_grad(y, alpha, target, mu)
        x_new = y - step_size * grad
        x_new = project_feasible(x_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    return project_feasible(best_x, target)


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 30.0

    def tleft():
        return deadline - time.time()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    alphas_p1 = [20, 100, 500, 2000, 10000, 50000, 200000]
    for a in alphas_p1:
        if tleft() < 50:
            break
        h_new = lbfgs_optimize(best_h, target, a, 50.0, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 80:
        for n_top in [10, 5, 3]:
            if tleft() < 30:
                break
            h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=n_top,
                                           n_steps=200, deadline=deadline)
            if sc_eq < best_score:
                best_score = sc_eq
                best_h = h_eq
            h_new = lbfgs_optimize(best_h, target, 50000, 50.0, 300, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if tleft() > 60:
        h_dr, sc_dr = direct_max_reduction(best_h, target, n_steps=200,
                                           deadline=deadline, n_top=3)
        if sc_dr < best_score:
            best_score = sc_dr
            best_h = h_dr
        h_new = lbfgs_optimize(best_h, target, 100000, 50.0, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 60:
        h_bin, sc_bin = binary_local_search(best_h, target, max_rounds=3,
                                            deadline=deadline)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
            h_new = lbfgs_optimize(best_h, target, 10000, 50.0, 300, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    n_current = len(best_h)
    target_current = n_current / 2.0
    restarts = [
        ('gauss', 0.1), ('gauss', 0.2), ('sparse', 0.15),
        ('binary_perturb', 0.2), ('gauss_small', 0.05)
    ]
    for strat, param in restarts:
        if tleft() < 50:
            break
        h_pert = best_h.copy()
        if strat == 'gauss' or strat == 'gauss_small':
            h_pert += rng.randn(n_current) * param
        elif strat == 'sparse':
            n_flip = int(n_current * param)
            idx = rng.choice(n_current, size=n_flip, replace=False)
            h_pert[idx] = rng.uniform(0, 1, n_flip)
        elif strat == 'binary_perturb':
            h_bin = (h_pert > 0.5).astype(float)
            n_flip = int(n_current * param)
            idx = rng.choice(n_current, size=n_flip, replace=False)
            h_bin[idx] = 1.0 - h_bin[idx]
            diff = int(target_current) - int(h_bin.sum())
            if diff != 0:
                if diff > 0:
                    candidates = np.where(h_bin == 0)[0]
                else:
                    candidates = np.where(h_bin == 1)[0]
                flip_n = min(abs(diff), len(candidates))
                if flip_n > 0:
                    flip_idx = rng.choice(candidates, flip_n, replace=False)
                    h_bin[flip_idx] = 1.0 - h_bin[flip_idx]
            h_pert = h_bin + rng.randn(n_current) * 0.05
        h_pert = project_feasible(h_pert, target_current)
        h_new = lbfgs_optimize(h_pert, target_current, 1000, 50.0, 200, deadline)
        h_new = lbfgs_optimize(h_new, target_current, 10000, 50.0, 200, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 50:
        h_fista = fista_optimize(best_h, target_current, 5000, 50.0,
                                 step_size=0.005, maxiter=500, deadline=deadline)
        sc = compute_c5(h_fista)
        if sc < best_score:
            best_score = sc
            best_h = h_fista
            h_new = lbfgs_optimize(best_h, target_current, 50000, 50.0, 300, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 200:
        x_old = np.linspace(0.0, 1.0, n)
        x_new_pts = np.linspace(0.0, 1.0, n2)
        h_fine = np.interp(x_new_pts, x_old, best_h)
        h_fine = project_feasible(h_fine, target2)
        for a in [50, 500, 5000]:
            if tleft() < 50:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, 50.0, 300, deadline)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n_current = n2
            target_current = target2

    if tleft() > 50:
        h_eq, sc_eq = top_lag_equalize(best_h, target_current, n_top=3,
                                       n_steps=150, deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    if tleft() > 30:
        h_new = lbfgs_optimize(best_h, target_current, 200000, 100.0, 400, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 25:
        h_bin, sc_bin = binary_local_search(best_h, target_current,
                                            max_rounds=2, deadline=deadline)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()

    if tleft() > 20:
        h_slq = slsqp_polish(best_h, target_current, 50000, 500)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    h_final = project_feasible(best_h, target_current)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
