        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

def softmax_gradient(h, n, alpha):
    """Compute gradient of softmax-approximated max overlap."""
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    n_lags = 2 * n - 1
    center = n - 1
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    # Gradient: grad[i] = -2*dx/alpha * sum_j h[j] * weights[center + i - j]
    # Vectorized:
    idx = center + np.arange(n)[:, None] - np.arange(n)[None, :]
    grad = -2.0 * dx / alpha * (h[None, :] * weights[idx]).sum(axis=1)
    return grad, overlaps

def softmax_pg_descent(h_init, n, target, max_steps, time_limit,
                        alpha_schedule, lr, momentum=0.9):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    h_mom = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        alpha = alpha_schedule[step] if step < len(alpha_schedule) else alpha_schedule[-1]
        grad, overlaps = softmax_gradient(h, n, alpha)
        
        # Nesterov momentum
        h_mom = h + momentum * h_mom
        h_try = project_feasible(h_mom - lr * grad, target)
        
        # Backtracking line search
        c5_val = compute_c5(h_try)
        lr_local = lr
        for ls in range(5):
            if c5_val < best_c5 + 1e-8:  # allow slight increase for momentum
                break
            lr_local *= 0.5
            h_try = project_feasible(h_mom - lr_local * grad, target)
            c5_val = compute_c5(h_try)
        
        h_mom = h_try - h
        h = h_try
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_pure(h_fixed, g_fixed, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g_fixed[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-7, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000, verbose=False)
            except Exception:
                return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def solve_lp_reg(h_fixed, g_fixed, dx, lags, n, target, h_ref, reg):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g_fixed[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    obj = t_var + reg * cp.sum_squares(h_var - h_ref)
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def lp_alternating(h_init, n, target, max_iter, time_limit, n_lags_use,
                   damping_start, damping_end, reg_start, reg_end, use_pure=False):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, it / max(1, max_iter * 0.5))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress

        g = 1.0 - h
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:n_lags_use]
        lags = top_idx - n_half

        if use_pure:
            h_new = solve_lp_pure(h, g, dx, lags, n, target)
        else:
            h_new = solve_lp_reg(h, g, dx, lags, n, target, h, reg)

        if h_new is None:
            break
        h_new = project_feasible(h_new, target)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5

def softmax_gradient(h, n, alpha):
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    n_lags = 2 * n - 1
    center = n - 1
    shifted = alpha * (overlaps - np.max(overlaps))
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    idx = center + np.arange(n)[:, None] - np.arange(n)[None, :]
    grad = -(2.0 * dx / alpha) * (h[None, :] * weights[idx]).sum(axis=1)
    return grad, overlaps, weights
