```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_round(h, n, target, max_k, time_limit, solver_str='SCS'):
    """One round of alternating LP with constraint generation"""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for it in range(10000):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h_cur
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in range(1, max_k + 1):
            # Positive shift
            coeff = np.zeros(n)
            coeff[:n-k] = g[k:] * dx
            constraints.append(coeff @ h_var <= t_var)
            # Negative shift
            coeff_neg = np.zeros(n)
            coeff_neg[k:] = g[:n-k] * dx
            constraints.append(coeff_neg @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Damped update
        lam = 0.7
        h_try = (1-lam) * h_cur + lam * h_new
        h_try = project_feasible(h_try, target)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        
        h_cur = h_try
    
    return best_h, best_c5
```

```python
def compute_L(h, n):
    """Compute L[i, j] = h[i+k_j] + h[i-k_j] for all i, j"""
    half_n = n - 1
    k_vals = np.arange(-half_n, half_n + 1)
    L = np.zeros((n, 2*n-1))
    for j, k in enumerate(k_vals):
        # h[i+k] is valid for 0 <= i+k < n, i.e., -k <= i < n-k
        # h[i-k] is valid for 0 <= i-k < n, i.e., k <= i < n+k
        lo1 = max(0, -k)
        hi1 = min(n, n-k)
        if hi1 > lo1:
            L[lo1:hi1, j] += h[lo1+k:hi1+k]
        lo2 = max(0, k)
        hi2 = min(n, n+k)
        if hi2 > lo2:
            L[lo2:hi2, j] += h[lo2-k:hi2-k]
    return L
```

```python
            for zi in range(0, len(zeros), chunk_b):
                b_chunk = zeros[zi:zi + chunk_b]
                diff = V[:, None, :] - L[b_chunk][None, :, :]
                new_maxes = np.max(diff, axis=2)
                flat_idx = int(np.argmin(new_maxes))
                idx = np.unravel_index(flat_idx, new_maxes.shape)
                val = float(new_maxes[idx])
                if val < best_new_max:
                    best_new_max = val
                    best_a = a_chunk[idx[0]]
                    best_b = b_chunk[idx[1]]

        accept = best_new_max < current_max - 1e-12
        if not accept and use_sa and temp > 0.01:
            delta = best_new_max - current_max
            if delta < 5.0:
                prob = np.exp(-delta / max(temp, 1e-10))
                if np.random.random() < prob:
                    accept = True

        if accept and best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_max = float(np.max(corr))
            L = compute_L(h, n)
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if use_sa:
                temp *= 0.995
        else:
            if use_sa:
                temp *= 0.98
                if temp < 0.01:
                    break
            else:
                break

    return best_h, best_max * dx

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def cons_fun(h):
    return np.array([np.sum(h) - _g_target])

def cons_jac(h):
    return np.array([np.ones(_g_n)])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=120):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 100:
            break
    return best_h, best_c5

def soft_refine(h_bin, n, target, time_limit):
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for _ in range(200):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        half_n = n - 1
        idxs = np.arange(n)
        idx_valid1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
        idx_valid2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h[idxs[idx_valid1] - k_star])
        grad[idx_valid2] -= h[idxs[idx_valid2] + k_star]

        for lr in [0.002, 0.005, 0.01, 0.02, 0.05]:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: LP with all constraints and line search
    if tleft() > 150:
        lp_time = min(140, tleft() * 0.35)
        h_lp, c5_lp = lp_with_linesearch(best_h, n, target, max_k=100, time_limit=lp_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Binarize and local search (greedy)
    if tleft() > 150:
        h_bin = binarize(best_h, n, target)
        best_score_bin = compute_c5(h_bin)
        # Try binary local search on the binarized LP solution
        ls_time = min(120, tleft() * 0.3)
        h_bl, c5_bl = binary_local_search(h_bin, n, target, time_limit=ls_time, use_sa=False)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl
        # Also try binarizing with different thresholds
        for thr in [0.4, 0.6]:
            h_b2 = (best_h > thr).astype(float)
            h_b2 = project_feasible(h_b2, target)
            h_b2 = binarize(h_b2, n, target)
            if compute_c5(h_b2) < best_score:
                h_bl2, c5_bl2 = binary_local_search(h_b2, n, target, time_limit=min(60, tleft() * 0.2))
                if c5_bl2 < best_score:
                    best_score = c5_bl2
                    best_h = h_bl2

    # Phase 3: Simulated annealing on binary space
    if tleft() > 80:
        h_bin_cur = binarize(best_h, n, target)
        sa_time = min(60, tleft() * 0.3)
        h_sa, c5_sa = binary_local_search(h_bin_cur, n, target, time_limit=sa_time, use_sa=True, sa_start_temp=2.0)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 4: SLSQP on best solution
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n, target, [100, 500, 2000], 2000, 1e-14, time_limit=min(80, tleft() * 0.4))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Another LP round
    if tleft() > 100:
        lp_time = min(80, tleft() * 0.3)
        h_lp, c5_lp = lp_with_linesearch(best_h, n, target, max_k=100, time_limit=lp_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: Binary local search again on new LP solution
    if tleft() > 100:
        h_bin = binarize(best_h, n, target)
        ls_time = min(80, tleft() * 0.3)
        h_bl, c5_bl = binary_local_search(h_bin, n, target, time_limit=ls_time, use_sa=False)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl

    # Phase 7: SLSQP with high alpha
    if tleft() > 80:
        h_ref = slsqp_round(best_h, n, target, [2000, 10000, 30000], 1500, 1e-15, time_limit=min(60, tleft() * 0.5))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Soft refinement (subgradient on fractional)
    if tleft() > 60:
        h_soft, c5_soft = soft_refine(best_h, n, target, time_limit=min(40, tleft() * 0.4))
        if c5_soft < best_score:
            best_score = c5_soft
            best_h = h_soft

    # Phase 9: Final LP polish
    if tleft() > 50:
        lp_time = min(40, tleft() * 0.6)
        h_lp, c5_lp = lp_with_linesearch(best_h, n, target, max_k=100, time_limit=lp_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Final binary local search
    if tleft() > 50:
        h_bin = binarize(best_h, n, target)
        h_bl, c5_bl = binary_local_search(h_bin, n, target, time_limit=min(30, tleft() * 0.5), use_sa=False)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl

    # Phase 11: Final SLSQP
    if tleft() > 30:
        h_ref = slsqp_round(best_h, n, target, [5000, 20000], 500, 1e-16, time_limit=min(20, tleft() * 0.6))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 12: Subgradient final
    if tleft() > 15:
        h_sg, c5_sg = subgradient_improve(best_h, n, target, max_steps=2000, time_limit=min(10, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
