    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish
    if tleft() > 200:
        h_pol = slsqp_round(best_h, n, target,
                            [100, 500, 2000, 10000, 50000],
                            2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Simulated annealing pair CD (main workhorse)
    if tleft() > 300:
        sa_time = min(500, tleft() * 0.4)
        h_sa, c5_sa = sa_pair_cd(best_h, n, target,
                                  time_limit=sa_time,
                                  T_start=0.003, T_end=0.00005,
                                  max_steps=500000, n_cand=80, n_lags=3,
                                  n_trials_per_step=30)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 3: Deterministic CD polish
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = fast_deterministic_cd(best_h, n, target,
                                             time_limit=cd_time,
                                             max_sweeps=5000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP after CD
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target,
                            [1000, 5000, 20000, 100000],
                            2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Second SA round with different parameters
    if tleft() > 200:
        sa_time = min(200, tleft() * 0.3)
        h_sa, c5_sa = sa_pair_cd(best_h, n, target,
                                  time_limit=sa_time,
                                  T_start=0.001, T_end=0.00002,
                                  max_steps=300000, n_cand=60, n_lags=5,
                                  n_trials_per_step=50)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 6: Multi-restart with perturbations
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.05, 0.08, 0.12]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        # Quick SA + SLSQP
        if tleft() > 80:
            h_ref, c5_ref = sa_pair_cd(h_pert, n, target,
                                        time_limit=min(40, tleft() * 0.3),
                                        T_start=0.002, T_end=0.0001,
                                        max_steps=100000, n_cand=50, n_lags=3,
                                        n_trials_per_step=20)
            if c5_ref < best_score:
                best_score = c5_ref
                best_h = h_ref
            if tleft() > 50:
                h_pol2 = slsqp_round(h_ref, n, target,
                                      [500, 2000, 10000, 50000],
                                      1000, 1e-15)
                sc2 = compute_c5(h_pol2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_pol2

    # Phase 7: Try different resolution
    if tleft() > 250:
        for n_try in [300, 500, 600]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires, c5_hires = sa_pair_cd(h_interp, n_try, target_try,
                                            time_limit=min(80, tleft() * 0.25),
                                            T_start=0.002, T_end=0.0001,
                                            max_steps=100000, n_cand=60, n_lags=3,
                                            n_trials_per_step=20)
            if tleft() > 60:
                h_hires = slsqp_round(h_hires, n_try, target_try,
                                       [500, 2000, 10000, 50000],
                                       1000, 1e-15)
            sc_hires = compute_c5(h_hires)
            if sc_hires < best_score:
                best_score = sc_hires
                best_h = h_hires

    # Phase 8: Final deterministic CD + SLSQP
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(40, tleft() * 0.5)
        h_cd, c5_cd = fast_deterministic_cd(best_h, n_use, target_use,
                                             time_limit=cd_time,
                                             max_sweeps=3000, n_cand=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000, 100000],
                              1000, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
