```python
import time
from itertools import combinations

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lns_solve(h, S, base_corr, deltas, k_size, target_frac):
    """Solve LNS subproblem: find best assignment of positions S."""
    n = len(h)
    num_ones = k_size // 2

    indices = list(range(k_size))
    all_assignments = []
    for combo in combinations(indices, num_ones):
        a = np.zeros(k_size)
        a[list(combo)] = 1.0
        all_assignments.append(a)
    all_assignments = np.array(all_assignments)  # (num_assign, k)
    
    # Compute new correlations: base_corr + deltas.T @ all_assignments.T
    # deltas shape: (k, 2n-1)
    # all_assignments shape: (num_assign, k)
    # Result: (num_assign, 2n-1)
    new_corrs = base_corr[None, :] + (deltas.T @ all_assignments.T).T  # (num_assign, 2n-1)
    c5_vals = new_corrs.max(axis=1)
    best_idx = int(np.argmin(c5_vals))
```

```python
def compute_delta(h, i, n):
    delta = np.zeros(2 * n - 1)
    # For k != 0:
    # Part A: add (1-h[i-k]) for valid k where 0 <= i-k < n
    #   i-k in [0, n-1] means k in [i-n+1, i]
    #   but k != 0, and k in [-(n-1), n-1]
    k_vals = np.arange(-(n-1), n)  # all valid shifts
    k_vals = k_vals[k_vals != 0]
    
    # Part A: valid when 0 <= i-k < n
    valid_a = (i - k_vals >= 0) & (i - k_vals < n)
    delta[k_vals[valid_a] + (n-1)] += (1.0 - h[i - k_vals[valid_a]])
    
    # Part B: subtract h[i+k] where 0 <= i+k < n
    valid_b = (i + k_vals >= 0) & (i + k_vals < n)
    delta[k_vals[valid_b] + (n-1)] -= h[i + k_vals[valid_b]]
    
    return delta
```

**Conflict-based selection:**
- Compute the top M shifts (say M=20)
- For each top shift k, the "conflict" positions are those i where h[i]=1 and h[i-k]=0 (they contribute (1-0)=1 to corr[k])
- Also positions j where h[j]=0 and h[j-k]=1 (they "should" contribute but don't because h[j]=0)
- Select positions from these conflict sets

Actually, for a shift k, corr[k] = sum_i h[i](1-h[i-k]). The "bad" terms are where h[i]=1 and h[i-k]=0. To reduce corr[k], we want to either:
- Set h[i]=0 for some bad i (reduce the term)
- Set h[i-k]=1 for some bad i (make the term zero)

So the conflict positions for shift k are: {i : h[i]=1, h[i-k]=0} ∪ {i-k : h[i]=1, h[i-k]=0}

```python
def select_positions(h, corr, n, k_size, rng, top_m=20):
    # Get top M shifts
    top_indices = np.argsort(corr)[-top_m:]
    top_shifts = top_indices - (n - 1)
    
    # Collect conflict positions
    conflict_set = set()
    for k in top_shifts:
        if k >= 0:
            # i ranges from k to n-1 (so i-k >= 0)
            for i in range(k, n):
                if h[i] == 1 and h[i-k] == 0:
                    conflict_set.add(i)
                    conflict_set.add(i-k)
        else:
            # k < 0, so i-k = i+|k|, need i+|k| < n, i.e., i < n+|k|
            abs_k = -k
            for i in range(0, n - abs_k):
                if h[i] == 1 and h[i+abs_k] == 0:
                    conflict_set.add(i)
                    conflict_set.add(i+abs_k)
    
    conflict_list = list(conflict_set)
    
    # Mix: 60% from conflicts, 40% random
    n_conf = int(0.6 * k_size)
    n_rand = k_size - n_conf
    n_conf = min(n_conf, len(conflict_list))
    
    if n_conf > 0:
        selected_conf = rng.choice(conflict_list, size=n_conf, replace=False)
    else:
        selected_conf = np.array([], dtype=int)
    
    remaining = np.setdiff1d(np.arange(n), selected_conf)
    if n_rand > 0 and len(remaining) > 0:
        selected_rand = rng.choice(remaining, size=min(n_rand, len(remaining)), replace=False)
    else:
        selected_rand = np.array([], dtype=int)
    
    selected = np.concatenate([selected_conf, selected_rand])
    if len(selected) > k_size:
        selected = selected[:k_size]
    elif len(selected) < k_size:
        # Fill with random
        used = set(selected.tolist())
        avail = np.array([x for x in range(n) if x not in used])
        need = k_size - len(selected)
        if len(avail) > 0:
            extra = rng.choice(avail, size=min(need, len(avail)), replace=False)
            selected = np.concatenate([selected, extra])
    
    return selected.astype(int)
```

```python
    shifted = np.clip(shifted, -500, 0)
```

```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def select_lns_positions(h, n, k_size, rng, corr):
    half = k_size // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) < half or len(zeros) < half:
        return None
    top_m = min(20, 2 * n - 1)
    top_indices = np.argsort(corr)[-top_m:]
    top_shifts = (top_indices - (n - 1)).astype(int)
    conflict_ones = set()
    conflict_zeros = set()
    for k in top_shifts:
        if k >= 0:
            for i in range(k, n):
                if h[i] == 1 and h[i - k] == 0:
                    conflict_ones.add(i)
                    conflict_zeros.add(i - k)
        else:
            ak = -k
            for i in range(0, n - ak):
                if h[i] == 1 and h[i + ak] == 0:
                    conflict_ones.add(i)
                    conflict_zeros.add(i + ak)
    c_ones = np.array([x for x in conflict_ones if h[x] == 1], dtype=int)
    c_zeros = np.array([x for x in conflict_zeros if h[x] == 0], dtype=int)
    sel_ones = np.array([], dtype=int)
    sel_zeros = np.array([], dtype=int)
    n_c_o = min(half, len(c_ones), max(1, int(0.7 * half)))
    if n_c_o > 0:
        sel_ones = np.sort(rng.choice(c_ones, size=n_c_o, replace=False))
    n_c_z = min(half, len(c_zeros), max(1, int(0.7 * half)))
    if n_c_z > 0:
        sel_zeros = np.sort(rng.choice(c_zeros, size=n_c_z, replace=False))
    rem_ones = np.setdiff1d(ones, sel_ones)
    rem_zeros = np.setdiff1d(zeros, sel_zeros)
    need_o = half - len(sel_ones)
    need_z = half - len(sel_zeros)
    if need_o > 0 and len(rem_ones) > 0:
        sel_ones = np.concatenate([sel_ones, rng.choice(rem_ones, size=need_o, replace=False)])
    if need_z > 0 and len(rem_zeros) > 0:
        sel_zeros = np.concatenate([sel_zeros, rng.choice(rem_zeros, size=need_z, replace=False)])
    positions = np.concatenate([sel_ones, sel_zeros]).astype(int)
    rng.shuffle(positions)
    return positions


def lns_solve(h, positions, n):
    k_size = len(positions)
    half = k_size // 2
    h_base = h.copy()
    h_base[positions] = 0.0
    base_corr = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_vals = np.arange(-(n - 1), n)
    mask_nz = k_vals != 0
    k_nz = k_vals[mask_nz]
    deltas = np.zeros((k_size, 2 * n - 1))
    for idx in range(k_size):
        pos = int(positions[idx])
        valid_a = (pos - k_nz >= 0) & (pos - k_nz < n)
        deltas[idx, k_nz[valid_a] + (n - 1)] += (1.0 - h_base[pos - k_nz[valid_a]])
        valid_b = (pos + k_nz >= 0) & (pos + k_nz < n)
        deltas[idx, k_nz[valid_b] + (n - 1)] -= h_base[pos + k_nz[valid_b]]
    combos = list(combinations(range(k_size), half))
    num_assigns = len(combos)
    all_assigns = np.zeros((num_assigns, k_size), dtype=np.float32)
    for ai, combo in enumerate(combos):
        for c in combo:
            all_assigns[ai, c] = 1.0
    dx = 2.0 / n
    best_c5_val = float('inf')
    best_assign = all_assigns[0]
    batch_size = min(num_assigns, 4000)
    for start in range(0, num_assigns, batch_size):
        end = min(start + batch_size, num_assigns)
        batch = all_assigns[start:end]
        batch_corrs = base_corr[None, :].astype(np.float32) + (batch @ deltas.T).astype(np.float32)
        batch_max = batch_corrs.max(axis=1)
        local_best = int(np.argmin(batch_max))
        if batch_max[local_best] < best_c5_val:
            best_c5_val = float(batch_max[local_best])
            best_assign = batch[local_best]
    new_h = h_base.copy()
    for j in range(k_size):
        if best_assign[j] > 0.5:
            new_h[positions[j]] = 1.0
    return new_h, best_c5_val * dx


def greedy_polish(h, n, max_iters, time_limit=None, start_time=None):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 500
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = np.array([i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0])
        j_cand = np.array([j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1])
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 10
            continue
        top_threshold = max_val * 0.8
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_shifts_arr = (top_indices - (n - 1)).astype(int)
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), 30)
        n_j = min(len(j_cand), 30)
        for ii in range(n_i):
            i = int(i_cand[ii])
            for jj in range(n_j):
                j = int(j_cand[jj])
                nm = max_val
                for tk in top_shifts_arr:
                    d = 0.0
                    if 0 <= i + tk < n:
                        d -= (1.0 - h[i + tk])
                    if 0 <= i - tk < n:
                        d += h[i - tk]
                    if 0 <= j + tk < n:
                        d += (1.0 - h[j + tk])
                    if 0 <= j - tk < n:
                        d -= h[j - tk]
                    v = corr[tk + (n - 1)] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

            break
        if stall > max_stall:
            break

def projected_gradient(h, n, target, alpha, lr, max_iters, time_limit=None, start_time=None):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_s = np.exp(shifted)
        sum_exp = float(np.sum(exp_s))
        weights = exp_s / sum_exp
        obj = (c_max + np.log(sum_exp) / alpha) * (2.0 / n)
        k_vals = np.arange(-(n - 1), n)
        grad = np.zeros(n)
        w0 = weights[n - 1]
        for k_idx in range(2 * n - 1):
            k = int(k_vals[k_idx])
            w = float(weights[k_idx])
            if w < 1e-12:
                continue
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                i_lo = max(0, k)
                i_hi = min(n - 1, n - 1 + k)
                if i_lo <= i_hi:
                    idx = np.arange(i_lo, i_hi + 1)
                    j = idx - k
                    grad[idx] += w * (1.0 - h[j])
                i_lo2 = max(0, -k)
                i_hi2 = min(n - 1, n - 1 - k)
                if i_lo2 <= i_hi2:
                    idx2 = np.arange(i_lo2, i_hi2 + 1)
                    j2 = idx2 + k
                    grad[idx2] -= w * h[j2]
        h_new = h - lr * grad
        h_new = project_feasible(h_new, float(target))
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, float(target))
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)

                if c5_l < best_score:
                    best_score = c5_l
                    best_h = h_l.copy()
                    corr_t = np.correlate(best_h, 1.0 - best_h, mode='full')

    # Phase 6: Legendre init + LNS
    if tleft() > 80:
        h_leg = legendary_init(n_init, rng)
        allowed_leg = min(30, tleft() * 0.10)
        h_g4, c5_g4 = greedy_polish(h_leg, n_init, max_iters=20000,
                                    time_limit=allowed_leg, start_time=start_time)
        corr_leg = np.correlate(h_g4, 1.0 - h_g4, mode='full')
        for _ in range(300):
            if time.time() - start_time > 1300 - 25:
                break
            k_s = 14
            pos = select_lns_positions(h_g4, n_init, k_s, rng, corr_leg)
            if pos is None:
                break
            h_l2, c5_l2 = lns_solve(h_g4, pos, n_init)
            if c5_l2 < c5_g4:
                h_g4 = h_l2
                c5_g4 = c5_l2
                corr_leg = np.correlate(h_g4, 1.0 - h_g4, mode='full')
        if c5_g4 < best_score:
            best_score = c5_g4
            best_h = h_g4.copy()

    # Phase 7: ILS - large perturbations + LNS restart
    if tleft() > 60:
        for pert_iter in range(8):
            if tleft() < 20:
                break
            n_pert = 5 + pert_iter * 2
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = int(ones[rng.randint(len(ones))])
                jj = int(zeros[rng.randint(len(zeros))])
                h_pert[ii] = 0
                h_pert[jj] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            corr_p = np.correlate(h_pert, 1.0 - h_pert, mode='full')
            best_pert = h_pert.copy()
            best_pert_c5 = compute_c5(h_pert)
            for _ in range(200):
                if tleft() < 15:
                    break
                k_s = 14
                pos = select_lns_positions(best_pert, n_init, k_s, rng, corr_p)
                if pos is None:
                    break
                h_pl, c5_pl = lns_solve(best_pert, pos, n_init)
                if c5_pl < best_pert_c5:
                    best_pert = h_pl
                    best_pert_c5 = c5_pl
                    corr_p = np.correlate(h_pert, 1.0 - h_pert, mode='full') if False else np.correlate(best_pert, 1.0 - best_pert, mode='full')
            if best_pert_c5 < best_score:
                best_score = best_pert_c5
                best_h = best_pert.copy()

    # Phase 8: Projected gradient continuous refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use // 2
        alphas = [20, 50, 100, 300, 1000, 3000]
        lrs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]
        h_cur = best_h.copy().astype(float)
        h_cur = project_feasible(h_cur, float(target_use))
        for ai in range(len(alphas)):
            if tleft() < 10:
                break
            iters = min(200, int(tleft() / max(1, len(alphas) - ai) / 0.02))
            h_pg, c5_pg = projected_gradient(h_cur, n_use, target_use,
                                             alphas[ai], lrs[ai], iters,
                                             time_limit=min(tleft() - 5, 30),
                                             start_time=start_time)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg
            h_cur = h_pg

    # Phase 9: SLSQP final polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [500, 2000, 8000, 30000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas2, 2000, 1e-14,
                                 time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Binary SA on refined solution
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(20, tleft() - 5)
        max_it5 = int(min(80000, allowed * 1500))
        h_sf, c5_sf = sa_refine(h_bin_f, n_use, target_use, max_it5,
                                0.004, 0.00001, rng,
                                time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Final: continuous refinement on best found
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cur = best_h.copy().astype(float)
        h_cur = project_feasible(h_cur, target_use)
        for alpha in [2000, 8000, 30000, 100000]:
            if tleft() < 5:
                break
            h_pg, c5_pg = projected_gradient(h_cur, n_use, int(target_use),
                                             alpha, 0.01, 100,
                                             time_limit=min(tleft() - 3, 10),
                                             start_time=start_time)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg
            h_cur = h_pg

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score != best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n_final
```
