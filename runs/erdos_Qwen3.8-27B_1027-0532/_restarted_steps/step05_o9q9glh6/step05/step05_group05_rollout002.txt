                ka = -k
                grad[:N - ka] += w * (1.0 - h_cur[ka:])
                grad[ka:] += -w * h_cur[:N - ka]
        grad = grad - np.mean(grad)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        corr_new = compute_corr(h_new)
        spread_new = float(np.max(corr_new[top_idx]) - np.min(corr_new[top_idx]))
        if spread_new < spread and sc < cur_score + 0.5 * spread * (2.0 / N):
            h_cur = h_new
            if sc < cur_score:
                cur_score = sc
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.6
            if lr < 1e-10:
                break
    return h_cur, cur_score


def worst_lag_grad(h, target, deadline, max_iters, eps_init):
    N = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / N
    eps = eps_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        k_star = int(np.argmax(corr))
        k = k_star - (N - 1)
        grad = np.zeros(N)
        if k >= 0:
            if k == 0:
                grad = 1.0 - 2.0 * h_cur
            else:
                grad[k:] += 1.0 - h_cur[:N - k]
                grad[:N - k] += -h_cur[k:]
        else:
            ka = -k
            grad[:N - ka] += 1.0 - h_cur[ka:]
            grad[ka:] += -h_cur[:N - ka]
        grad = dx * (grad - np.mean(grad))
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc
            eps = min(eps * 1.05, 0.03)
        else:
            eps *= 0.5
            if eps < 1e-10:
                break
    return h_cur, cur_score


def multi_start_adam(h_base, target, n_starts, deadline):
    N = len(h_base)
    rng = np.random.RandomState(12345)
    best_h = h_base.copy()
    best_score = compute_c5(best_h)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        scale = 0.03 * (s + 1)
        h_pert = h_base + rng.randn(N) * scale
        h_pert = project_feasible(h_pert, target)
        alphas = [200, 1000, 5000]
        h_try, sc_try = adam_lse(h_pert, target, alphas, 150, 0.008, 15, deadline)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try
    return best_h, best_score


def coarse_fine(h_parent, target_final, n_final, deadline):
    for n_c in [50, 100, 200]:
        if time.time() > deadline:
            break
        t_c = n_c / 2.0
        h_c = interpolate_to(h_parent, n_c, t_c)
        h_c = lbfgsb_lse(h_c, t_c, [100, 500], 300, 10, deadline)
        h_c = adam_lse(h_c, t_c, [500, 2000, 8000], 200, 0.01, 10, deadline)[0]
        h_up = interpolate_to(h_c, n_final, target_final)
        return h_up
    return h_parent


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start + budget_s - 12.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B on parent
    if tleft() > 100:
        d1 = min(start + 80, hard_deadline)
        h1 = lbfgsb_lse(best_h, target_final, [200, 1000], 500, 15, d1)
        sc = compute_c5(h1)
        if sc < best_score:
            best_score = sc
            best_h = h1

    # Phase 2: Coarse-to-fine at lower res
    if tleft() > 150:
        d2 = min(start + 180, hard_deadline)
        h_cf = coarse_fine(best_h, target_final, n_final, d2)
        h_cf = lbfgsb_lse(h_cf, target_final, [500, 2000], 300, 15, d2)
        sc = compute_c5(h_cf)
        if sc < best_score:
            best_score = sc
            best_h = h_cf

    # Phase 3: Adam main optimization
    if tleft() > 200:
        d3 = min(start + 400, hard_deadline)
        h3, sc3 = adam_lse(best_h, target_final,
                           [200, 500, 2000, 8000], 300, 0.012, 20, d3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: L-BFGS-B after Adam
    if tleft() > 150:
        d4 = min(start + 550, hard_deadline)
        h4 = lbfgsb_lse(best_h, target_final, [2000, 8000, 20000], 500, 12, d4)
        sc = compute_c5(h4)
        if sc < best_score:
            best_score = sc
            best_h = h4

    # Phase 5: Active-set equalization
    if tleft() > 120:
        d5 = min(start + 680, hard_deadline)
        for k_act in [3, 5, 8]:
            if tleft() < 40:
                break
            h_eq, sc_eq = equalize_top(best_h, target_final, k_act, d5, 200, 0.015)
            if sc_eq < best_score:
                best_score = sc_eq
                best_h = h_eq
            h_l = lbfgsb_lse(h_eq, target_final, [5000, 20000], 300, 10, d5)
            sc2 = compute_c5(h_l)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_l

    # Phase 6: Adam with high alpha
    if tleft() > 120:
        d6 = min(start + 800, hard_deadline)
        h6, sc6 = adam_lse(best_h, target_final,
                           [5000, 15000, 40000], 300, 0.005, 8, d6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Worst-lag surgery
    if tleft() > 100:
        d7 = min(start + 900, hard_deadline)
        h7, sc7 = worst_lag_grad(best_h, target_final, d7, 300, 0.003)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
        h7b = lbfgsb_lse(h7, target_final, [10000, 30000], 300, 8, d7)
        sc7b = compute_c5(h7b)
        if sc7b < best_score:
            best_score = sc7b
            best_h = h7b

    # Phase 8: Multi-start Adam
    if tleft() > 120:
        d8 = min(start + 1050, hard_deadline)
        h8, sc8 = multi_start_adam(best_h, target_final, 4, d8)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final equalization + L-BFGS-B
    if tleft() > 80:
        d9 = min(start + 1150, hard_deadline)
        h9, sc9 = equalize_top(best_h, target_final, 5, d9, 150, 0.008)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9
        h9b = lbfgsb_lse(best_h, target_final, [20000, 50000], 300, 8, d9)
        sc9b = compute_c5(h9b)
        if sc9b < best_score:
            best_score = sc9b
            best_h = h9b

    # Phase 10: Final Adam polish
    if tleft() > 40:
        d10 = hard_deadline
        h10, sc10 = adam_lse(best_h, target_final,
                             [30000, 60000, 100000], 150, 0.003, 5, d10)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
