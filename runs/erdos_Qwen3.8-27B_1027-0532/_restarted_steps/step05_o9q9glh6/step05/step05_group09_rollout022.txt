def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target}"""
    h = h.astype(float).copy()
    n = len(h)
    # Bisection on threshold tau such that sum(clip(h - tau, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_grad(h, alpha):
    """Compute log-sum-exp objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    # d/dh_i [corr_k] where corr_k = sum_j h[j](1-h[j+k])
    # = sum_j [delta_{j,i}(1-h[j+k]) - h[j]*delta_{j+k,i}]
    # = (1-h[i+k]) - h[i-k] (with appropriate bounds)
    # 
    # grad_i = dx * sum_k weights_k * [(1-h[i+k]) - h[i-k]]
    #        = dx * (sum_k weights_k * (1-h[i+k]) - sum_k weights_k * h[i-k])
    #
    # First term: sum_k w_k (1-h[i+k]) = (1-h) convolved with w (reversed)
    # Second term: sum_k w_k h[i-k] = h convolved with w (reversed)
    
    w_flipped = weights[::-1].copy()
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    H = np.fft.rfft(h, fsize)

    corr_neg = np.fft.irfft(Wf * H, fsize)[n-1:2*n-1]  # sum_k w_k h[i-k]
    
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def adam_optimize(h0, target, alpha, max_iters, deadline, lr=0.01, beta1=0.9, beta2=0.999):
    """Adam optimizer with projection."""
    n = len(h0)
    x = project_feasible(h0, target).copy()
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        obj, grad = lse_grad(x, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x_new = project_feasible(x_new, target)
        x = x_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def arc_polish(h, target, max_rounds, deadline, temp=0.005, num_active=5):
    """Subgradient descent on top correlations."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:num_active]
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            if ki > 0:
                jn = np.arange(ki, n)
                grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        for eps_try in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps_try * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]

    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 0.5

            lo = mid
        else:

    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s_exp = float(exp_s.sum())
    w = exp_s / s_exp
    obj = (c_max + np.log(s_exp) / alpha) * dx
    wf = w[::-1].copy()
    fs = 2 * n
    Wf = np.fft.rfft(wf, fs)
    V = np.fft.rfft(1.0 - h, fs)
    H = np.fft.rfft(h, fs)
    A = np.fft.irfft(Wf * V, fs)[n-1:2*n-1]
    B = np.fft.irfft(Wf * H, fs)[n-1:2*n-1]
    grad = dx * (A - B)
    return obj, grad

    hc = h.copy()
    cur = compute_c5(hc)
    for rd in range(max_rounds):

            break
        corr = np.correlate(hc, 1.0-hc, mode='full')[n-1:]
        top = np.argsort(corr)[::-1][:n_act]

            if time.time() > deadline:
                break
            ov = h[idx]
            if ov < 1e-10:
                trials = [0.01, 0.05, 0.1, 0.2]
            elif ov > 1.0 - 1e-10:
                trials = [0.9, 0.95, 0.8, 0.7]
            else:
                trials = [ov * 0.5, ov * 1.5, 1.0 - ov, min(ov + 0.1, 1.0), max(ov - 0.1, 0.0)]
            for tr in trials:
                if tr < -0.5 or tr > 1.5:
                    continue
                h[idx] = tr
                h = project_feasible(h, target)
                sc = compute_c5(h)
                if sc < cur - 1e-15:
                    cur = sc
                    break
            h = project_feasible(h, target)
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20.0

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_sc = compute_c5(h)

    def tl():
        return deadline - time.time()

    # Phase 1: Adam with alpha continuation
    stages = [
        (50, 0.03, 60),
        (200, 0.02, 60),
        (800, 0.015, 70),
        (4000, 0.01, 80),
        (15000, 0.006, 80),
        (60000, 0.004, 80),
    ]
    for a, lr, td in stages:
        if tl() < td + 15:
            break
        h_new, sc = adam_opt(best_h, target, a, t0 + td, lr=lr)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    # Phase 2: Arc polish
    if tl() > 50:
        h_new, sc = arc_polish(best_h, target, t0 + 50, temp=0.003, n_act=5)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    # Phase 3: Very high alpha Adam
    if tl() > 80:
        h_new, sc = adam_opt(best_h, target, 300000, t0 + 80, lr=0.001)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    # Phase 4: Arc polish with smaller temperature
    if tl() > 40:
        h_new, sc = arc_polish(best_h, target, t0 + 40, temp=0.001, n_act=3)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    # Phase 5: Random restarts with diverse scales
    rng = np.random.RandomState(seed)
    if tl() > 180:
        scales = [0.02, 0.04, 0.07, 0.12, 0.03, 0.06, 0.1, 0.2]
        for s in scales:
            if tl() < 35:
                break
            hp = project_feasible(best_h + rng.randn(n) * s, target)
            h_new, sc = adam_opt(hp, target, 3000, t0 + 22, lr=0.01)
            h_new, sc2 = arc_polish(h_new, target, t0 + 10, temp=0.002, n_act=4)
            if sc2 < best_sc:
                best_sc = sc2
                best_h = h_new

    # Phase 6: Structured perturbations (sinusoidal)
    if tl() > 80:
        x = np.linspace(0, 2, n)
        for freq in [3, 5, 7, 11, 13]:
            if tl() < 20:
                break
            ph = rng.random() * 2 * np.pi
            hp = project_feasible(best_h + 0.03 * np.sin(2 * np.pi * freq * x / 2.0 + ph), target)
            h_new, sc = adam_opt(hp, target, 5000, t0 + 12, lr=0.008)
            h_new, sc2 = arc_polish(h_new, target, t0 + 6, temp=0.002, n_act=4)
            if sc2 < best_sc:
                best_sc = sc2
                best_h = h_new

    # Phase 7: Coordinate descent polish
    if tl() > 40:
        h_new, sc = coord_polish(best_h, target, t0 + min(35, tl() - 5))
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    # Phase 8: Final arc polish
    if tl() > 15:
        h_new, sc = arc_polish(best_h, target, t0 + min(12, tl() - 5), temp=0.001, n_act=3)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    # Phase 9: Tiny final Adam steps
    if tl() > 15:
        h_new, sc = adam_opt(best_h, target, 1000000, t0 + min(12, tl() - 5), lr=0.0005)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_sc = compute_c5(h_final)
    if final_sc > best_sc:
        h_final = best_h
        final_sc = best_sc
    return h_final, final_sc, n
