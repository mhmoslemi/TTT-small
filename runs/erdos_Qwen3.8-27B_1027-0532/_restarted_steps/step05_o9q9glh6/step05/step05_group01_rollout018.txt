def fast_cd_v2(h, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    k_vals = np.arange(-(n-1), n)  # shift values
    mid = n - 1
    
    no_improve = 0
    n_cand = 10  # smaller candidate set for speed
    
    while time.time() - start < time_limit and no_improve < 100:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = np.max(corr) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active shifts (top few)
        n_top = 5
        top_shifts = np.argsort(corr)[-n_top:]
        
        # Compute gradient for active shifts (weighted)
        c_max = np.max(corr)
        weights = np.exp(100 * (corr[top_shifts] - c_max))
        weights /= weights.sum()
        
        idxs = np.arange(n)
        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Pick candidate set
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Try pairs: top 5 high × top 5 low = 25 pairs
        n_try = min(5, n_cand)
        high_try = high_set[-n_try:]  # most positive gradient
        low_try = low_set[:n_try]  # most negative gradient
        
        best_trial = c5_now
        best_pair = None
        best_delta = 0
        
        for i in high_try:
            for j in low_try:
                if i == j: continue
                # Compute g vector
                delta_lo = -h[j]
                delta_hi = min(1.0 - h[j], h[i])
                if delta_hi - delta_lo < 1e-12: continue
                
                g = np.zeros(2*n-1)
                # h[j+k] term
                m = j + k_vals
                valid = (m >= 0) & (m < n)
                g[valid] += h[m[valid]]
                # h[j-k] term
                m = j - k_vals
                valid = (m >= 0) & (m < n)
                g[valid] += h[m[valid]]
                # h[i+k] term
                m = i + k_vals
                valid = (m >= 0) & (m < n)
                g[valid] -= h[m[valid]]
                # h[i-k] term
                m = i - k_vals
                valid = (m >= 0) & (m < n)
                g[valid] -= h[m[valid]]
                
                # Binary search for optimal delta
                a, b = delta_lo, delta_hi
                for _ in range(20):
                    mid_d = 0.5*(a+b)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-12
                    f_prime = np.max(g[active]) if active.any() else 0
                    if f_prime > 0: b = mid_d
                    else: a = mid_d
                
                delta_opt = 0.5*(a+b)
                trial_max = np.max(corr + delta_opt * g) * dx
                if trial_max < best_trial:
                    best_trial = trial_max
                    best_pair = (i, j)
                    best_delta = delta_opt
        
        if best_pair:
            i, j = best_pair
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0, 1)
            h = project_feasible(h, target)
        else:
            no_improve += 10  # encourage break
    
    return best_h, best_c5

            no_improve += 1

        if no_improve > 300:
            break

        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved:
            no_improve += 5

    return best_h, best_c5


def momentum_subgradient(h, n, target, time_limit, max_steps, mu, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation_search(h_init, n, target, max_tries, time_limit, rng_seed):
    start = time.time()
    rng_local = np.random.RandomState(rng_seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.03, 0.05, 0.1, 0.15, 0.2]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def time_remaining():
    return _global_budget - 30.0 - (time.time() - _global_start_time)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_start_time, _global_budget
    _global_start_time = time.time()
    _global_budget = budget_s

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with analytic gradient - initial polish
    if time_remaining() > 180:
        h_pol = slsqp_round(best_h, n, target,
                            [50, 200, 1000, 5000, 20000],
                            2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main coordinate descent (fast, many iterations)
    if time_remaining() > 300:
        cd_time = min(400, time_remaining() * 0.35)
        h_cd, c5_cd = fast_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=30000, n_cand=10, n_side=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP after CD
    if time_remaining() > 150:
        h_pol = slsqp_round(best_h, n, target,
                            [500, 2000, 10000, 50000],
                            2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Second CD round with wider search
    if time_remaining() > 250:
        cd_time = min(250, time_remaining() * 0.35)
        h_cd, c5_cd = fast_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=20000, n_cand=14, n_side=7)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if time_remaining() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [2000, 10000, 50000, 200000],
                            2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Soft subgradient for fine-tuning
    if time_remaining() > 80:
        h_sg, c5_sg = soft_subgradient(best_h, n, target,
                                       time_limit=min(40, time_remaining() * 0.3),
                                       max_steps=50000, n_top=7, base_lr=0.02)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Multi-restart with SLSQP
    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.1, 0.2]:
        if time_remaining() < 60:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref = slsqp_round(h_pert, n, target,
                            [100, 500, 2000, 10000],
                            1200, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Local perturbation search
    if time_remaining() > 50:
        h_pert, c5_pert = local_perturbation_search(best_h, n, target,
                                                    max_tries=5000,
                                                    time_limit=min(30, time_remaining() * 0.4),
                                                    rng_seed=12345)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 9: Try higher resolution
    if time_remaining() > 350:
        for n_try in [500, 600]:
            if time_remaining() < 200:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [50, 200, 1000, 5000],
                                  1500, 1e-14)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try
                if time_remaining() > 150:
                    h_cd2, c5_cd2 = fast_cd(best_h, n, target,
                                            time_limit=min(60, time_remaining() * 0.3),
                                            max_sweeps=8000, n_cand=10, n_side=5)
                    if c5_cd2 < best_score:
                        best_score = c5_cd2
                        best_h = h_cd2

    # Phase 10: Final CD round
    if time_remaining() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, time_remaining() * 0.4)
        h_cd, c5_cd = fast_cd(best_h, n_use, target_use,
                              time_limit=cd_time,
                              max_sweeps=15000, n_cand=10, n_side=6)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP
    if time_remaining() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000, 100000, 500000],
                              1500, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
