def compute_c5(h):
    n = len(h)

    return float(np.max(corr)) * dx

    # Project h to [0,1]^n with sum(h) = target

    lo, hi = h.min() - 1.0, h.max() + 1.0

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))

    weights = exp_shifted / sum_exp

    # Gradient via FFT

    v = 1.0 - h

    H = np.fft.rfft(h, fsize)

    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

        h_new, sc_new = active_lag_refine(best_h, target, start_time + alloc, num_active=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if tleft() > 60:
        perturbations = structured_perturbations(best_h, n, target, seed)
        for p in perturbations:
            if tleft() < 20:
                break
            alloc = min(15, max(8, tleft() // 4))
            h_new, sc_new = fista_optimize(p, target, 3000, 15000, start_time + alloc)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    if tleft() > 40:
        alloc = min(30, tleft() - 10)
        h_new, sc_new = fista_optimize(best_h, target, 200000, 100000, start_time + alloc)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if tleft() > 25:
        alloc = min(20, tleft() - 5)
        h_new, sc_new = active_lag_refine(best_h, target, start_time + alloc, num_active=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if tleft() > 20:
        alloc = min(15, tleft() - 5)
        h_new, sc_new = fista_optimize(best_h, target, 500000, 50000, start_time + alloc)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    if tleft() > 10:
        alloc = min(8, tleft() - 3)
        h_new, sc_new = active_lag_refine(best_h, target, start_time + alloc, num_active=2)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
