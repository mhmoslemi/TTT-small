import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    fsize = 2 * n
    v = 1.0 - h
    w_flipped = weights[::-1].copy()
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, verbose=False):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    # Estimate Lipschitz constant
    lr_init = 1.0 / max(alpha * 0.1, 0.5)
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        
        # Backtracking line search
        step_local = step
        x_new = project_feasible(y_clipped - step_local * grad_y, target)
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            if obj_new <= obj_y + 0.5 * step_local * np.dot(grad_y, x_new - y_clipped):
                break
            step_local *= 0.5
            x_new = project_feasible(y_clipped - step_local * grad_y, target)
        
        step = step_local
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def subgrad_nesterov(h0, target, max_iters, deadline, top_k=5, temp=0.01):
    """Nesterov-accelerated subgradient method on the max of correlations."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    # Initial step size
    step = 0.01
    # Decay rate
    decay = 0.999
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        corr = get_corr(y_clipped)
        c_max = float(np.max(corr))
        
        # Top-k lags with softmax weights
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:top_k]
        diffs = corr[active] - c_max
        w = np.exp(diffs / temp)
        w /= w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            if ki == 0:
                # corr[0] = dx * sum h[i](1-h[i])
                # d/dh[j] = dx * (1 - 2*h[j])
                grad += wi * dx * (1.0 - 2.0 * y_clipped)
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += wi * dx * (1.0 - y_clipped[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= wi * dx * y_clipped[jn - ki]
        
        x_new = project_feasible(y_clipped - step * grad, target)
        x_prev = x
        x = x_new
        t = t_new
        step *= decay
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def arc_polish_v2(h, target, max_rounds, deadline, top_k=5):
    """Enhanced arc polishing with adaptive temperature."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    temps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for temp in temps:
        if time.time() > deadline:
            break
        for rd in range(max_rounds // len(temps)):
            if time.time() > deadline:
                break
            corr = get_corr(h_cur)
            sorted_idx = np.argsort(corr)[::-1]
            active = sorted_idx[:top_k]
            diffs = corr[active] - corr[active[0]]
            w = np.exp(diffs / temp)
            w /= w.sum()
            
            grad = np.zeros(n)
            for wi, ki in zip(w, active):
                ki = int(ki)
                if ki == 0:
                    grad += wi * (1.0 - 2.0 * h_cur)
                else:
                    jp = np.arange(0, n - ki)
                    grad[jp] += wi * (1.0 - h_cur[jp + ki])
                    jn = np.arange(ki, n)
                    grad[jn] -= wi * h_cur[jn - ki]
            grad *= dx
            
            # Line search
            for eps_try in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                h_new = project_feasible(h_cur - eps_try * grad, target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
                    break
    
    return h_cur, cur_score

def coord_descent_v2(h, target, deadline, max_sweeps=100):
    """Coordinate descent with smart trial values."""
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    corr_cur = get_corr(h_cur)
    active_lag = int(np.argmax(corr_cur))
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for idx in range(n):
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            
            # Generate trial values
            trials = [0.0, 1.0, old_val * 0.5, old_val * 1.5, 0.5]
            # Add values related to the active lag
            if active_lag > 0 and idx + active_lag < n:
                trials.append(h_cur[idx + active_lag])
            if active_lag > 0 and idx >= active_lag:
                trials.append(1.0 - h_cur[idx - active_lag])
            
            best_trial = old_val
            best_trial_score = cur_score
            
            for trial in trials:
                if trial < -0.01 or trial > 1.01:
                    continue
                trial = np.clip(trial, 0.0, 1.0)
                if abs(trial - old_val) < 1e-12:
                    continue
                h_trial = h_cur.copy()
                h_trial[idx] = trial
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_trial_score - 1e-15:
                    best_trial = trial
                    best_trial_score = sc
            
            if best_trial_score < cur_score - 1e-15:
                h_cur[idx] = best_trial
                h_cur = project_feasible(h_cur, target)
                cur_score = best_trial_score
                improved = True
                # Update active lag
                corr_cur = get_corr(h_cur)
                active_lag = int(np.argmax(corr_cur))
        
        if not improved:
            break
    
    return h_cur, cur_score
