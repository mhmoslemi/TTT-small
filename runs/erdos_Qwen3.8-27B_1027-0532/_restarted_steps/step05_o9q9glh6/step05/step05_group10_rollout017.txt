```python
def nesterov_subgradient(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_score = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    
    mu = 0.9  # momentum
    lr = 0.01
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        
        # Subgradient
        grad = np.zeros(n)
        mask1 = (idxs >= k) & (idxs - k < n) & (idxs - k >= 0)
        mask2 = (idxs + k >= 0) & (idxs + k < n)
        
        if k >= 0:
            grad[mask1] += (1.0 - h[idxs[mask1] - k]) * dx
            grad[mask2] -= h[idxs[mask2] + k] * dx
        else:
            # k is negative, so i-k = i+|k|
            grad[mask1] += (1.0 - h[idxs[mask1] - k]) * dx  # i - k = i + |k|
            grad[mask2] -= h[idxs[mask2] + k] * dx  # i + k = i - |k|
        
        # Nesterov step
        v = mu * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        # Adaptive step size
        if score > best_score:
            lr *= 0.9
        else:
            lr = min(lr * 1.05, 0.1)
        
        h = h_new
    
    return best_h, best_score
```

```python
def smooth_obj(h, alpha):

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def nesterov_subgradient(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    mu = 0.85
    lr = 0.02

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        grad = np.zeros(n)
        if k >= 0:
            mask1 = (idxs >= k) & (idxs < n)
            actual1 = idxs - k
            grad[actual1] += (1.0 - h[idxs[mask1]]) * dx
            mask2 = (idxs + k < n)
            actual2 = idxs
            grad[actual2[mask2]] -= h[idxs[mask2] + k] * dx
        else:
            kk = -k
            mask2 = (idxs >= kk) & (idxs < n)
            actual2 = idxs - kk
            grad[actual2] += (1.0 - h[idxs[mask2]]) * dx
            mask1 = (idxs < n - kk)
            actual1 = idxs
            grad[actual1[mask1]] -= h[idxs[mask1] + kk] * dx

        v = mu * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_new.copy()
        if score > best_score + 0.001:
            lr = max(lr * 0.8, 0.001)
        else:
            lr = min(lr * 1.02, 0.08)
        h = h_new
    return best_h, best_score

            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.95, damping + 0.02)
        if it % 60 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.7)
    return best_h, best_c5


def directed_swap_search(h, n, target, max_rounds, time_limit, rng):
    start = time.time()
    h_bin = (h >= 0.5).astype(float)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    tgt = int(target)
    while len(ones) > tgt:
        idx = ones.pop()
        h_bin[idx] = 0
        zeros.append(idx)
    while len(ones) < tgt:
        idx = zeros.pop()
        h_bin[idx] = 1
        ones.append(idx)
    best_h = h_bin.copy()
    best_score = compute_c5(h_bin)
    dx = 2.0 / n
    half_n = n - 1
    no_improve = 0

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        k_star_idx = int(np.argmax(corr))
        k = k_star_idx - half_n
        improved = False
        candidate_swaps = []
        if k >= 0:
            for i in range(n - k):
                if h_bin[i] == 1 and h_bin[i + k] == 0:
                    for j in zeros:
                        inc = 0
                        if j < n - k and h_bin[j + k] == 0:
                            inc = 1
                        dec = 1
                        net = inc - dec
                        if net < 0:
                            candidate_swaps.append((net, i, j))
        else:
            kk = -k
            for i in range(kk, n):
                if h_bin[i] == 1 and h_bin[i - kk] == 0:
                    for j in zeros:
                        inc = 0
                        if j >= kk and h_bin[j - kk] == 0:
                            inc = 1
                        dec = 1
                        net = inc - dec
                        if net < 0:
                            candidate_swaps.append((net, i, j))
        if not candidate_swaps:
            no_improve += 100
            continue
        candidate_swaps.sort()
        for net, i, j in candidate_swaps[:20]:
            if time.time() - start > time_limit:
                break
            i_idx = ones.index(i)
            j_idx = zeros.index(j)
            h_bin[i] = 0
            h_bin[j] = 1
            sc = compute_c5(h_bin)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_bin.copy()
                ones[i_idx] = j
                zeros[j_idx] = i
                improved = True
                no_improve = 0
                break
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        if not improved:
            no_improve += 1
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binary push + local search
    if tleft() > 80:
        h_bin = (best_h >= 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_bin
        h_bl, sc_bl = binary_local_search(best_h, n, target, 3000, min(50, tleft() * 0.4), rng)
        if sc_bl < best_score:
            best_score = sc_bl
            best_h = h_bl

    # Phase 2: Directed swap search
    if tleft() > 100:
        h_ds, sc_ds = directed_swap_search(best_h, n, target, 2000, min(80, tleft() * 0.4), rng)
        if sc_ds < best_score:
            best_score = sc_ds
            best_h = h_ds

    # Phase 3: Simulated annealing
    if tleft() > 100:
        h_sa, sc_sa = simulated_annealing_binary(best_h, n, target, 50000, min(80, tleft() * 0.4), rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 4: Coordinate descent (continuous)
    if tleft() > 150:
        h_cd, sc_cd = coordinate_descent(best_h, n, target, 50, min(120, tleft() * 0.5))
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: LP with constraint generation
    if tleft() > 200:
        lp_time = min(180, tleft() * 0.35)
        h_lp, c5_lp = alternating_lp_sparse(best_h, n, target,
                                            max_iter=15000, time_limit=lp_time,
                                            damping_init=0.8, reg=1e-4, n_active=20)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: SLSQP moderate
    if tleft() > 120:
        _global_n = n
        _global_target = target
        h_ref = slsqp_round(best_h, n, target, [100, 500, 2000], 1500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Nesterov subgradient
    if tleft() > 80:
        h_ns, sc_ns = nesterov_subgradient(best_h, n, target, 3000, min(60, tleft() * 0.5))
        if sc_ns < best_score:
            best_score = sc_ns
            best_h = h_ns

    # Phase 8: Coordinate descent again (finer)
    if tleft() > 100:
        h_cd, sc_cd = coordinate_descent(best_h, n, target, 80, min(80, tleft() * 0.5))
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: LP tighter
    if tleft() > 150:
        lp_time = min(120, tleft() * 0.35)
        h_lp, c5_lp = alternating_lp_sparse(best_h, n, target,
                                            max_iter=15000, time_limit=lp_time,
                                            damping_init=0.9, reg=1e-5, n_active=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: SLSQP high alpha
    if tleft() > 100:
        _global_n = n
        _global_target = target
        h_ref = slsqp_round(best_h, n, target, [2000, 10000, 30000], 2000, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final coordinate descent
    if tleft() > 60:
        h_cd, sc_cd = coordinate_descent(best_h, n, target, 100, min(50, tleft() * 0.6))
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 12: Final SLSQP polish
    if tleft() > 50:
        _global_n = n
        _global_target = target
        h_ref = slsqp_round(best_h, n, target, [5000, 20000, 50000], 800, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 13: Final Nesterov
    if tleft() > 30:
        h_ns, sc_ns = nesterov_subgradient(best_h, n, target, 2000, min(20, tleft() * 0.7))
        if sc_ns < best_score:
            best_score = sc_ns
            best_h = h_ns

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
