def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):

        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:

def multi_peak_score(full, dx, m=5, gamma=2.0):
    """Score based on top-m peaks with exponential weighting"""
    # Sort in descending order
    sorted_idx = np.argsort(full)[::-1]
    top_m = full[sorted_idx[:m]]
    # Weighted sum: max + (1/gamma) * sum of remaining top peaks
    score = top_m[0] + (1.0/gamma) * np.sum(top_m[1:])

# New: Water-filling coordinate descent
def waterfill_cd(h, max_iters, rng, time_limit, start_time, top_k=3):
    """Coordinate descent that targets the top-k peaks"""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Get top k peaks
        top_idx = np.argsort(full)[::-1][:top_k]
        k_stars = top_idx - (n-1)
        
        # For each peak, find the best single swap to reduce it
        best_delta = None
        best_new_max = float(full.max())
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        for k_star in k_stars:
            if k_star == 0:
                continue
            # Find ones that contribute to this peak (h[i]=1, h[i+k*]=0)
            # and zeros that would benefit (h[j]=0, h[j-k*]=1)
            
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            # ones where A[i]==0 means h[i+k*]=0, so this one contributes
            i_cands = ones[A[ones] == 0.0]
            
            B = np.zeros(n)
            idx2 = positions - k_star
            valid2 = (idx2 >= 0) & (idx2 < n)
            B[valid2] += h[idx2[valid2]]
            # zeros where B[j]==1 means h[j-k*]=1, so making j=1 helps
            j_cands = zeros[B[zeros] == 1.0]
            
            if len(i_cands) == 0 or len(j_cands) == 0:
                continue
            
            # Try top candidates
            n_try = min(5, len(i_cands), len(j_cands))
            i_pick = i_cands[rng.permutation(len(i_cands))[:n_try]]
            j_pick = j_cands[rng.permutation(len(j_cands))[:n_try]]
            
            for i in i_pick:
                for j in j_pick:
                    delta = compute_delta(h, i, j)
                    new_max = float((full + delta).max())
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = (i, j, delta)
        
        if best_delta is not None:
            i, j, delta = best_delta
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5

# New: SLSQP with better smoothing
def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return c_max + log_sum / alpha

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

_cons_target = 0.0

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    return best_h, best_c5

def sa_adaptive(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    positions = np.arange(n)
    accept_count = 0
    window_size = 500
    accepts = np.zeros(window_size)
    accept_idx = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        avg_acc = np.mean(accepts)
        if avg_acc < 0.05 and it > window_size:
            t *= 2.0
        elif avg_acc > 0.5 and it > window_size:
            t *= 0.5

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if it % 3 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] == 0.0]
                B = np.zeros(n)
                idx2 = positions - k_star
                valid2 = (idx2 >= 0) & (idx2 < n)
                B[valid2] += h[idx2[valid2]]
                j_cands = zeros[B[zeros] == 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_full = full + delta
        new_max = float(new_full.max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        accepted = False
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full = new_full
            current_c5 = c5_new
            accepted = True
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        accepts[accept_idx] = 1.0 if accepted else 0.0
        accept_idx = (accept_idx + 1) % window_size

    return best_h, best_c5

def pair_swap_sa(h, max_iters, rng, time_limit, start_time, t_init=0.01, t_final=0.0001):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        i1 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        rem_ones = np.setdiff1d(ones, [i1])
        rem_zeros = np.setdiff1d(zeros, [j1])
        i2 = rem_ones[rng.randint(len(rem_ones))]
        j2 = rem_zeros[rng.randint(len(rem_zeros))]

        delta = compute_delta_2(h, i1, j1, i2, j2)
        new_full = full + delta
        new_max = float(new_full.max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i1] = 0.0
            h[j1] = 1.0
            h[i2] = 0.0
            h[j2] = 1.0
            full = new_full
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
