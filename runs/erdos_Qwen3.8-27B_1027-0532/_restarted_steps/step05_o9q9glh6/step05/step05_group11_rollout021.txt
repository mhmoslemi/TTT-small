def compute_c5(h):

    return float(np.max(corr)) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]

        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

            break
        g = 1.0 - h
        lags = get_top_lag_indices(h, n_top)
        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = 0.95 - 0.10 * progress
        reg = 1e-4 * (1e-6 / 1e-4) ** progress
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=solver, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 200:
        h_cd, score_cd = coord_desc(best_h, n, target, max_sweeps=20000, time_limit=min(400, tleft() * 0.45))
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd

    if tleft() > 120:
        h_lp, score_lp = lp_refine(best_h, n, target, max_iter=5000, time_limit=min(200, tleft() * 0.4), n_top=220)
        if score_lp < best_score:
            best_score = score_lp
            best_h = h_lp

    if tleft() > 80:
        h_cd2, score_cd2 = coord_desc(best_h, n, target, max_sweeps=20000, time_limit=min(200, tleft() * 0.5))
        if score_cd2 < best_score:
            best_score = score_cd2
            best_h = h_cd2

    if tleft() > 100:
        for attempt in range(8):
            if tleft() < 30:
                break
            noise = 0.04 + attempt * 0.025
            h_rand = best_h + rng.randn(n) * noise
            h_rand = project_feasible(h_rand, target)
            h_r, score_r = coord_desc(h_rand, n, target, max_sweeps=8000, time_limit=min(15, tleft() * 0.15))
            if score_r < best_score:
                best_score = score_r
                best_h = h_r

    if tleft() > 120:
        for n_res in [200, 300, 500]:
            if tleft() < 80:
                break
            target_r = n_res / 2.0
            h_r = interpolate_to(best_h, n_res, target_r)
            h_r, score_r = coord_desc(h_r, n_res, target_r, max_sweeps=10000, time_limit=min(50, tleft() * 0.3))
            if n_res != n:
                h_fine = interpolate_to(h_r, n, target)
                h_fine, score_fine = coord_desc(h_fine, n, target, max_sweeps=8000, time_limit=min(30, tleft() * 0.2))
                if score_fine < best_score:
                    best_score = score_fine
                    best_h = h_fine
            else:
                if score_r < best_score:
                    best_score = score_r
                    best_h = h_r

    if tleft() > 60:
        h_lp2, score_lp2 = lp_refine(best_h, n, target, max_iter=3000, time_limit=min(80, tleft() * 0.5), n_top=180)
        if score_lp2 < best_score:
            best_score = score_lp2
            best_h = h_lp2

    if tleft() > 30:
        h_final, score_final = coord_desc(best_h, n, target, max_sweeps=20000, time_limit=min(40, tleft() * 0.8))
        if score_final < best_score:
            best_score = score_final
            best_h = h_final

    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)
    return best_h, best_score, len(best_h)
