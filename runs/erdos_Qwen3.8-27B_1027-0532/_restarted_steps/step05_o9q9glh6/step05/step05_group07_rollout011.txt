    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)

        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]

    mask4 = ms <= i

    if np.any(mask4):

def find_1d_min(A, C, t_min, t_max, n_iter=30):
    """Find t in [t_min, t_max] minimizing max_k(A[k] + C[k]*t)."""
    lo = t_min
    hi = t_max
    # Initial evaluation to get a better range
    f_lo = float(np.max(A + C * lo))
    f_hi = float(np.max(A + C * hi))
    
    # Identify the active constraint at each endpoint
    active_lo = np.argmax(A + C * lo)
    active_hi = np.argmax(A + C * hi)
    
    # If the same constraint is active at both ends, the min is at the interior
    # where the derivative changes sign (or at a breakpoint)
    
    for _ in range(n_iter):
        mid = (lo + hi) * 0.5
        f_mid = A + C * mid
        pos = C > 1e-14
        neg = C < -1e-14
        zero = ~(pos | neg)
        
        # Check if f_mid <= max(A) (i.e., if mid is feasible for the "free" part)
        current_max = float(np.max(f_mid))
        
        # Find the tightest upper bound on t from positive C constraints
        ub = t_max
        if np.any(pos):
            # A[k] + C[k]*t <= current_max => t <= (current_max - A[k]) / C[k]
            candidate_ub = (current_max - A[pos]) / C[pos]
            ub = min(ub, float(np.min(candidate_ub)))
        
        # Find the tightest lower bound from negative C constraints
        lb = t_min
        if np.any(neg):
            # A[k] + C[k]*t <= current_max => t >= (current_max - A[k]) / C[k]
            # Note: C[k] < 0, so dividing flips inequality
            candidate_lb = (current_max - A[neg]) / C[neg]
            lb = max(lb, float(np.max(candidate_lb)))
        
        ub = min(ub, t_max)
        lb = max(lb, t_min)
        
        if lb <= ub + 1e-12:
            hi = mid
        else:
            lo = mid
    
    return (lo + hi) * 0.5

def paired_cd_fast(h, n, target, deadline, rng, n_sweeps=None, n_bisect=30):
    """Optimized paired coordinate descent."""
    dx = 2.0 / n
    h_cur = h.copy()
    nl1 = 2 * n - 1
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    best_h = h_cur.copy()
    pair_positions = np.arange(n)
    stall_count = 0
    
    for sweep in range(n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break
        
        if sweep % 50 == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        
        rng.shuffle(pair_positions)
        
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            
            t_star = find_1d_min(A, C, t_min, t_max, n_bisect)
            
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_h = h_cur.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 300 and sweep > 200:
            # Perturb to escape local minimum
            pert_size = 0.005 + 0.01 * min(stall_count / 1000.0, 1.0)
            h_cur = project_feasible(h_cur + rng.randn(n) * pert_size, target)
            corr = compute_corr(h_cur)
            stall_count = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

    # grad of g_k at h = [1-2*h shifted appropriately]

    # This is exactly -2*h shifted + 1 (with boundary corrections)

    # grad[i] = sum_k weights[k] * dcorr[k]/dh[i]

def compute_gradient(h, weights, n):
    """Compute the weighted average gradient of the correlation."""
    nl1 = 2 * n - 1
    grad = np.zeros(n)
    for k in range(nl1):
        w = weights[k]
        if w < 1e-15:
            continue
        # dcorr[k]/dh[p] = (1-h[p+k-n+1])*I(0<=p+k-n+1<n) - h[p-k+n-1]*I(0<=p-k+n-1<n)
        # First term: valid p where 0 <= p+k-n+1 < n => n-1-k <= p <= n-1-k+n-1
        # i.e., max(0, n-1-k) <= p <= min(n-1, 2n-2-k)
        lo1 = max(0, n-1-k)
        hi1 = min(n-1, 2n-2-k)
        if lo1 <= hi1:
            j = np.arange(lo1, hi1+1) + k - n + 1
            grad[lo1:hi1+1] += w * (1.0 - h[j])
        # Second term: valid p where 0 <= p-k+n-1 < n => k-n+1 <= p <= k
        # i.e., max(0, k-n+1) <= p <= min(n-1, k)
        lo2 = max(0, k-n+1)
        hi2 = min(n-1, k)
        if lo2 <= hi2:
            j = np.arange(lo2, hi2+1) - k + n - 1
            grad[lo2:hi2+1] -= w * h[j]
    return grad

Wait, this is O(n * 2n) = O(2n²) per call, which for n=400 is 320K operations. In a Python loop over k (799 iterations), each doing a few numpy operations on small arrays, it should take maybe 1-2ms. That's acceptable.

But actually, I realize this subgradient approach might not be the most effective. The issue is that the subgradient of max at a non-differentiable point is the convex hull of the active gradients, and using a single subgradient (even the average) can be slow.

A better approach might be to use the **Dantzig-Wolfe / column generation** idea:
1. At the current point h, find the active constraint k* = argmax_k g_k(h)
2. Solve the LP: min g_{k*}(h) s.t. feasible (this is a QP since g_{k*} is quadratic)
3. Check if the solution still has k* as the max
4. If not, add the new active constraint and repeat

But step 2 is a QP with a non-convex objective (g_{k*} = 1 - h^T A_{k*} h is concave), so minimizing it is non-convex.

I think the most practical approach is to stick with what works (coordinate descent + SLSQP) but add:
1. A subgradient-based "diversification" step
2. Better hyperparameter tuning
3. More aggressive SLSQP with very high alpha

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def find_1d_min(A, C, t_min, t_max, n_iter=35):
    lo = t_min
    hi = t_max
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        pos = C > 1e-14
        neg = C < -1e-14
        current_max = float(np.max(A + C * mid))
        ub = t_max
        if np.any(pos):
            cand = (current_max - A[pos]) / C[pos]
            ub = min(ub, float(np.min(cand)))
        lb = t_min
        if np.any(neg):
            cand = (current_max - A[neg]) / C[neg]
            lb = max(lb, float(np.max(cand)))
        ub = min(ub, t_max)
        lb = max(lb, t_min)
        if lb <= ub + 1e-12:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)


def paired_cd(h, n, target, deadline, rng, n_sweeps=None, n_bisect=35, recompute=40):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    best_h = h_cur.copy()
    pair_positions = np.arange(n)
    stall = 0
    ns = n_sweeps if n_sweeps else 200000
    for sweep in range(ns):
        if time.time() > deadline:
            break
        if sweep % recompute == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        rng.shuffle(pair_positions)
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            t_star = find_1d_min(A, C, t_min, t_max, n_bisect)
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400 and sweep > 300:
            ps = 0.005 + 0.01 * min(stall / 2000.0, 1.0)
            h_cur = project_feasible(h_cur + rng.randn(n) * ps, target)
            corr = compute_corr(h_cur)
            stall = 0
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=5000, ftol=1e-17):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_step(h, corr, n, target, eta, alpha=5000):
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    sw = float(np.sum(weights))
    if sw < 1e-300:
        return h
    weights = weights / sw
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1) + k - n + 1
            grad[lo1:hi1 + 1] += w * (1.0 - h[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1) - k + n - 1
            grad[lo2:hi2 + 1] -= w * h[idx]
    grad = grad - np.mean(grad)
    z = h - eta * grad
    return project_feasible(z, target)


def subgradient_phase(h, n, target, deadline, rng, alpha_start=2000, n_iter=500):
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    dx = 2.0 / n
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    eta = 0.01
    mu = 0.5
    y_prev = h_cur.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        alpha = alpha_start * (1.0 + 0.1 * it)
        h_new = subgradient_step(h_cur, corr, n, target, eta, alpha)
        corr_new = compute_corr(h_new)
        # Nesterov update
        mu_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu))
        y_new = h_new + (mu_new / mu - 1.0) * (h_new - y_prev)
        y_new = project_feasible(y_new, target)
        y_prev = h_cur
        h_cur = y_new
        eta = 0.01 / (1.0 + 0.005 * it)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
