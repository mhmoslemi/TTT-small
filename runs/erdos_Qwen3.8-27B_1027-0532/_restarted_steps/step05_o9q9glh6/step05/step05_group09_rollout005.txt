    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    top_idx = np.argsort(corr)[-top_m:]

    # Weight by how close to max (softmax)
    diffs = corr[top_idx] - max_val
    weights = np.exp(diffs / 0.01)  # Sharp softmax

        for j in range(k, n):

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full')[n-1:] * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    sorted_h = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(sorted_h)])
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k < n:
            if tau <= sorted_h[k-1] + 1e-12 and tau >= sorted_h[k] - 1e-12:
                return np.clip(h - tau, 0.0, 1.0)
        else:
            if tau >= -1e-12:
                return np.clip(h - tau, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def grad_top_lags(h, top_m, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    # Get top_m lags
    top_idx = np.argsort(corr)[-top_m:]
    max_val = corr[top_idx[0]]
    diffs = corr[top_idx] - max_val
    weights = np.exp(diffs / temp)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        if n - k > 0:
            grad[:n-k] += w * (1.0 - h[k:n]) * dx
        if k > 0:
            grad[k:] += w * (-h[:n-k]) * dx
    return grad

def gradient_descent(h0, target, deadline, top_m=5, temp_init=0.01, lr_init=0.5):
    """Nesterov-accelerated projected gradient with backtracking"""
    n = len(h0)
    h = project_feasible(h0, target).copy()
    prev = h.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    
    while time.time() < deadline:
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + ((t - 1.0) / t_new) * (h - prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad = grad_top_lags(y, top_m, temp_init)
        
        # Backtracking line search
        h_trial = project_feasible(y - lr * grad, target)
        obj_y = compute_c5(y)
        obj_trial = compute_c5(h_trial)
        
        if obj_trial <= obj_y - 0.5 * lr * np.dot(grad, h_trial - y):
            pass  # Accept
        else:
            # Reduce step size
            for _ in range(10):
                lr = lr * 0.5
                h_trial = project_feasible(y - lr * grad, target)
                obj_trial = compute_c5(h_trial)
                if obj_trial <= obj_y - 0.5 * lr * np.dot(grad, h_trial - y):
                    break
            else:
                lr = max(lr, 1e-8)
                h_trial = project_feasible(y - lr * grad, target)
        
        prev = h
        h = h_trial
        t = t_new
        
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    sh = np.sort(h)[::-1]
    cs = np.concatenate([[0.0], np.cumsum(sh)])
    for k in range(1, n + 1):
        tau = (cs[k] - target) / k
        if k < n:
            if sh[k] - 1e-12 <= tau <= sh[k-1] + 1e-12:
                return np.clip(h - tau, 0.0, 1.0)
        else:
            if tau >= -1e-12:
                return np.clip(h - tau, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_top(h, m, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    idx = np.argsort(corr)[-m:]
    mx = corr[idx[0]]
    w = np.exp((corr[idx] - mx) / temp)
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        k = int(ki)
        if n > k:
            g[:n-k] += wi * (1.0 - h[k:]) * dx
        if k > 0:
            g[k:] -= wi * h[:n-k] * dx
    return g
