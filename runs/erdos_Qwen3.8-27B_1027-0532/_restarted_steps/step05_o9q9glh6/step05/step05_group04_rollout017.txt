    h: current iterate

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_grad_weighted(h, weights, n):
    """Compute weighted sum of gradients of g_k."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-10:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    return grad

def lse_objective(x, n, alpha, target):
    """Log-sum-exp smoothed max of g_k, with penalty for sum constraint violation."""
    corr = compute_corr(x)
    over = corr[n-1:n-1+n]
    c_ref = float(np.max(over))
    shifted = alpha * (over - c_ref)
    shifted = np.clip(shifted, -500.0, 0.0)
    lse = c_ref + float(np.log(np.sum(np.exp(shifted)))) / alpha
    # Penalty for sum constraint
    sum_viol = np.sum(x) - target
    penalty = 1e4 * sum_viol ** 2
    return lse + penalty

                       lr_init=0.003, deadline=None):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    beta = beta_init

        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]

        if wsum < 1e-30:

        weights /= wsum

                break

    return best_h, best_score

    best_score = compute_c5(h_cur)
    
    # Track active set stability
    recent_active = []
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
            
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_max_corr = float(np.max(over))
        
        # Active set: top n_active lags
        top_idx = np.argsort(over)[::-1][:n_active]
        active_lags = sorted(top_idx.tolist())
        
        # Track stability
        recent_active.append(set(active_lags))
        if len(recent_active) > 10:
            recent_active.pop(0)
        
        # If the active set is stable (same for last 5 iterations), use it
        # Otherwise, use the current top
        if len(recent_active) >= 5:
            last5 = recent_active[-5:]
            if all(s == last5[0] for s in last5):
                stable_set = last5[0]
                # Use stable set
                active_lags = sorted(stable_set)
            else:
                active_lags = sorted(set().union(*last5))
                if len(active_lags) > n_active + 2:
                    active_lags = sorted(top_idx.tolist())
        
        # Weights: softmax on active lags
        a_vals = over[active_lags]
        a_max = float(np.max(a_vals))
        temp = max(a_max * 0.001, 1e-8)
        weights = np.exp((a_vals - a_max) / temp)
        weights /= weights.sum()
        
        # Interior mask
        interior_mask = (h_cur > 0.01) & (h_cur < 0.99)
        interior_idx = np.where(interior_mask)[0]
        n_int = len(interior_idx)
        
        if n_int < 5:
            break
        
        # Build and solve
        boundary_sum = np.sum(h_cur[~interior_mask])
        target_int = target - boundary_sum
        
        size = n_int + 1
        A_mat = np.zeros((size, size))
        b_vec = np.zeros(size)
        
        int_pos = np.full(n, -1, dtype=np.int32)
        int_pos[interior_idx] = np.arange(n_int)
        
        for kk, k in enumerate(active_lags):
            wk = float(weights[kk])
            if wk < 1e-12:
                continue
            k = int(k)
            for pp in range(n_int):
                i = int(interior_idx[pp])
                if i <= n - 1 - k:
                    jk = i + k
                    pos_jk = int_pos[jk]
                    if pos_jk >= 0:
                        A_mat[pp, pos_jk] -= wk * dx
                    b_vec[pp] += wk * dx
                if i >= k:
                    jk2 = i - k
                    pos_jk2 = int_pos[jk2]
                    if pos_jk2 >= 0:
                        A_mat[pp, pos_jk2] -= wk * dx
                A_mat[pp, n_int] = 1.0
        
        A_mat[n_int, :n_int] = 1.0
        b_vec[n_int] = target_int
        
        try:
            sol = np.linalg.solve(A_mat, b_vec)
        except:
            try:
                sol = np.linalg.lstsq(A_mat, b_vec, rcond=None)[0]
            except:
                break
        
        h_sol = h_cur.copy()
        h_sol[interior_idx] = sol[:n_int]
        
        # Check bounds
        if float(np.min(sol[:n_int])) < -0.05 or float(np.max(sol[:n_int])) > 1.05:
            # Significant out-of-bounds: expand interior mask or break
            if it > 100:
                break
            # Try with looser interior mask
            continue
        
        # Damped update (higher damping for stability)
        damping = 0.5 if it < 30 else 0.8
        h_new = (1 - damping) * h_cur + damping * h_sol
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        if np.max(np.abs(h_new - h_cur)) < tol:
            h_cur = h_new
            break
        h_cur = h_new
    
    return best_h, best_score
