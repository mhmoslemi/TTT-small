import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_all_lag_constraints(h_lin, g, dx, n):
    """Build constraints for all lags."""
    constraints = []
    n_half = n - 1
    for lag in range(-(n-1), n):
        lo = max(0, lag)
        hi = min(n, lag + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - lag:hi - lag] * dx
            constraints.append(coeff)
    return constraints

def alternating_lp_all_lags(h_init, n, target, max_iter, time_limit, 
                            damping=0.85, reg_start=1e-3, reg_end=1e-10):
    """Alternating linearization with ALL lags."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        progress = min(1.0, it / max(1, max_iter))
        reg = reg_start * (reg_end / reg_start) ** progress

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0

        prob = cp.Problem(cp.Minimize(objective), constraints)

        c5_val = compute_c5(h)

    return best_h, best_c5

def smooth_obj_grad(h, alpha, target):
    """Compute the smooth objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute all overlaps
    overlaps = np.zeros(2*n - 1)
    for lag in range(-(n-1), n):
        if lag >= 0:
            idx = np.arange(lag, n)
            overlaps[lag + n - 1] = np.sum(h[idx] * (1 - h[idx - lag])) * dx
        else:
            idx = np.arange(0, n + lag)
            overlaps[lag + n - 1] = np.sum(h[idx] * (1 - h[idx - lag])) * dx
    
    # Soft max
    c_max = np.max(overlaps)
    shifted = alpha * (overlaps - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / alpha
    
    # Gradient
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)

            idx2 = np.arange(k, n)

def smooth_obj_grad(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    
    # overlaps[k + n - 1] = sum_i h[i] * (1-h[i+k]) * dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    
    # Soft max
    c_max = np.max(overlaps)
    shifted = alpha * (overlaps - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / alpha
    
    # Soft max weights
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: sum_k weights[k] * grad_k
    # grad_k[i] = dx * [(1-h[i+k]) if i+k < n else 0] - dx * [h[i-k] if i-k >= 0 else 0]
    grad = np.zeros(n)

    # For i where i-k is valid: -w[k] * h[i-k] * dx

    # First term: for each i, sum over k where i+k < n of w[k+n-1] * (1-h[i+k]) * dx
    # This is a cross-correlation of w with (1-h)
    # Second term: for each i, sum over k where i-k >= 0 of w[k+n-1] * h[i-k] * dx
    # This is a cross-correlation of w with h
    
    # Actually, let me think about this more carefully.
    # c_k = sum_{i: i,i+k in [0,n)} h[i] * (1-h[i+k]) * dx
    # dc_k/dh[j] = sum over k of [contribution of h[j] to c_k]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def overlapping_lp(h_init, n, target, max_iter, time_limit, n_lags,
                   damping=0.85, reg_start=1e-3, reg_end=1e-10):
    """Alternating linearization LP with top n_lags."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        # Get top lags
        overlaps = compute_overlaps(h)
        n_half = n - 1
        if n_lags < len(overlaps):
            top_idx = np.argsort(overlaps)[::-1][:n_lags]
            lags = top_idx - n_half
        else:
            lags = np.arange(-(n-1), n)
        
        progress = min(1.0, it / max(1, max_iter * 0.7))
        reg = reg_start * (reg_end / reg_start) ** progress
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=100000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_softmax_obj(h, alpha, target):
    """Compute soft-max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    
    c_max = np.max(overlaps)
    shifted = alpha * (overlaps - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    obj = c_max + log_sum / alpha
    
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient
    grad = np.zeros(n)
    n_half = n - 1
    # Only process significant weights
    sig = np.where(weights > 1e-10)[0]
    for idx in sig:
        k = idx - n_half
        w = weights[idx]
        if k >= 0:
            idx1 = np.arange(0, n - k)
            grad[idx1] += w * (1.0 - h[idx1 + k]) * dx
            idx2 = np.arange(k, n)
            grad[idx2] += w * (-h[idx2 - k]) * dx
        else:
            kk = -k
            idx1 = np.arange(0, n - kk)
            grad[idx1] += w * (1.0 - h[idx1 + kk]) * dx
            idx2 = np.arange(kk, n)
            grad[idx2] += w * (-h[idx2 - kk]) * dx
    
    return obj, grad

def smooth_slsqp(h_start, n, target, alphas, maxiter=2000, ftol=1e-15):
    """Optimize using SLSQP with soft-max smooth objective."""
    h_cur = h_start.copy()
    
    for alpha in alphas:
        def fun(h):
            obj, _ = smooth_softmax_obj(h, alpha, target)
            return obj
        
        def grad(h):
            _, g = smooth_softmax_obj(h, alpha, target)
            return g
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                 'jac': lambda h: np.ones(n)}]
        
        try:
            res = minimize(fun, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def lag_correction_step(h, lag, n, strength=0.03):
    """Apply a single lag correction step."""
    h_new = h.copy()
    k = abs(lag)
    if k == 0 or k >= n:
        return h_new
    idx = np.arange(n - k)
    a = h_new[idx]
    b = h_new[idx + k]
    valid = (a > 0.005) & (a < 0.995) & (b > 0.005) & (b < 0.995)
    if not np.any(valid):
        return h_new
    diffs = a - b
    transfers = strength * diffs
    h_new[idx[valid]] -= transfers[valid]
    h_new[idx[valid] + k] += transfers[valid]
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

def lag_correction_optimize(h_init, n, target, time_limit, n_worst=5, strength=0.03):
    """Iteratively correct the worst lags."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(5000):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break
        
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_worst]
        top_lags = top_idx - n_half
        
        improved = False
