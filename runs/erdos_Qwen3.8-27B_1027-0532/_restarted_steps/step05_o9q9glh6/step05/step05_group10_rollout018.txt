    corr = np.correlate(h, 1.0 - h, mode='full')

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

    """Simulated annealing on binary string with sum = target."""
    start = time.time()
    best = b.copy()
    best_score = compute_c5(best)
    cur_score = best_score
    T = T0
    half_n = n - 1
    
    # Precompute indices
    ones_list = [i for i in range(n) if b[i] == 1]
    zeros_list = [i for i in range(n) if b[i] == 0]

        if time.time() - start > time_limit:
            break
        if it % 100 == 0:

            T = T0 * (T_end / T0) ** progress

            b[j] = 0

def cutting_plane_lp(h_start, n, target, time_limit, max_constraints=100):
    """Cutting plane LP for the Erdős problem."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Active constraints: list of k values
    corr = np.correlate(h, 1.0 - h, mode='full')
    half_n = n - 1
    active_k = list(np.argsort(corr)[-20:])  # Start with top 20
    
    max_iter = 5000
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Build LP: min t + reg*||h - h_prev||^2
        # s.t. for each k in active_k: sum_i (1-h[i-k]) * h_new[i] * dx <= t
        #      sum(h) = target, 0 <= h <= 1
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        
        for k_idx in active_k:
            k = k_idx - half_n  # Actual lag
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                # coeff[i] = (1 - h[i-k]) * dx for valid i
                coeff = np.zeros(n)
                for i in range(lo, hi):
                    j = i - k
                    if 0 <= j < n:
                        coeff[i] = (1.0 - h[j]) * dx
                constraints.append(coeff @ h_var <= t_var)
        
        # Small regularization for stability
        reg = 0.01 * (1.0 - it / max_iter)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000)
        except:
            try:
                prob.solve(solver=cp.SCS, eps=1e-8, max_iters=50000)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0, 1)
        h = project_feasible(h_new, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Update active constraints
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = t_var.value if t_var.value is not None else best_c5
        # Find violated constraints
        violated = []
        for k_idx in range(1, len(corr)-1):
            if k_idx not in active_k:
                if corr[k_idx] * dx > t_val + 1e-6:
                    violated.append(k_idx)
        
        if violated:
            # Add most violated constraints
            violated.sort(key=lambda k: corr[k]*dx, reverse=True)
            active_k.extend(violated[:5])
            if len(active_k) > max_constraints:
                # Keep only the most binding
                active_k = list(np.argsort(corr)[-max_constraints:])
        else:
            # No violations, add constraint near current max
            top = np.argsort(corr)[-10:]
            new_ks = [k for k in top if k not in active_k]
            active_k.extend(new_ks[:3])
            if len(active_k) > max_constraints:
                active_k = list(np.argsort(corr)[-max_constraints:])
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_seq(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        t[i] = float(bin(i).count('1') % 2)
    return t

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def binary_best_improve(b, n, target, max_scans, time_limit, rng):
    start = time.time()
    best = b.copy()
    best_score = compute_c5(best)
    b_cur = b.copy()
    cur_score = best_score
    ones = set(np.where(b_cur == 1.0)[0].tolist())
    zeros = set(np.where(b_cur == 0.0)[0].tolist())
    half_n = n - 1

    for scan in range(max_scans):
        if time.time() - start > time_limit:
            break
        ones_list = sorted(ones)
        zeros_list = sorted(zeros)
        best_gain = 0.0
        best_pair = None
        for i in ones_list:
            for j in zeros_list:
                iak = 0.0
                iak_p = 0.0
                jk = 0.0
                jk_p = 0.0
                for k_offset in range(-half_n, half_n + 1):
                    ia = i + k_offset
                    ib = i - k_offset
                    ja = j + k_offset
                    jb = j - k_offset
                    if 0 <= ia < n:
                        iak += b_cur[ia]
                    if 0 <= ib < n:
                        iak_p += b_cur[ib]
                    if 0 <= ja < n:
                        jk += b_cur[ja]
                    if 0 <= jb < n:
                        jk_p += b_cur[jb]
                delta = (iak + iak_p - jk - jk_p)
                if delta < -best_gain:
                    best_gain = -delta
                    best_pair = (i, j)
        if best_pair is None or best_gain < 1e-12:
            break
        i, j = best_pair
        b_cur[i] = 0.0
        b_cur[j] = 1.0
        ones.discard(i)
        ones.add(j)
        zeros.discard(j)
        zeros.add(i)
        cur_score = compute_c5(b_cur)
        if cur_score < best_score:
            best_score = cur_score
            best = b_cur.copy()
        else:
            b_cur[i] = 1.0
            b_cur[j] = 0.0
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
            break
    return best, best_score

def binary_sa(b, n, target, max_iters, time_limit, rng, T0=0.5, Tend=0.005):

    best = b.copy()
    best_score = compute_c5(best)
    b_cur = b.copy()
    cur_score = best_score

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, it / max(1, max_iters))
        T = T0 * (Tend / T0) ** progress
        ones_list = np.where(b_cur == 1.0)[0]
        zeros_list = np.where(b_cur == 0.0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = int(ones_list[rng.randint(len(ones_list))])
        j = int(zeros_list[rng.randint(len(zeros_list))])
        b_cur[i] = 0.0
        b_cur[j] = 1.0
        new_score = compute_c5(b_cur)
        delta = new_score - cur_score
        if delta < 0 or (T > Tend * 0.5 and rng.rand() < np.exp(-delta / max(T, 1e-10))):

                best = b_cur.copy()
        else:

            b_cur[j] = 0.0

def slsqp_refine(h_start, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smooth_max_obj(x, alpha), h_start.copy(),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def lp_refine(h_start, n, target, time_limit, max_active=60, damping_init=0.85):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-min(25, max_active)].tolist())
    damping = damping_init
    reg = 0.05
    it = 0
    max_it = 20000

    for it in range(max_it):
        if time.time() - start > time_limit:
            break
        if it % 50 == 0 and reg > 1e-5:
            reg *= 0.9

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]

        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[max(0, -k):hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.SCS, eps=1e-9, max_iters=50000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            damping = min(0.95, damping + 0.01)
        else:
            damping = max(0.4, damping * 0.95)

        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = t_var.value if t_var.value is not None else best_c5 / dx
        violated = []
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and corr[k_idx] > t_val + 0.5:
                violated.append(k_idx)
        if violated:
            violated.sort(key=lambda x: corr[x], reverse=True)
            active_k.update(violated[:10])
        top_k = set(np.argsort(corr)[-max_active:].tolist())
        active_k = active_k | top_k
        if len(active_k) > max_active * 2:
            active_k = set(np.argsort(corr)[-max_active * 2:].tolist())

    return best_h, best_c5

def subgradient_refine(h_start, n, target, max_steps, time_limit):
    start = time.time()
    dx = 300.0 / n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_v1 = (idxs - k >= 0) & (idxs - k < n)
        idx_v2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[idx_v1] += (1.0 - h_cur[idxs[idx_v1] - k])
        grad[idx_v2] -= h_cur[idxs[idx_v2] - k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:

                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

            break

                    h_ref, sc_ref = slsqp_refine(best_h, len(best_h), len(best_h)/2.0,
                                                 1000, 500, 1e-12)
                    if sc_ref < best_score:
                        best_score = sc_ref
                        best_h = h_ref.copy()

    # Phase 11: Final binary push + refinement cycle
    if tleft() > 60:
        h_bin_final = (best_h > 0.5).astype(float)
        h_bin_final = project_feasible(h_bin_final, target)
        sc_bin_f = compute_c5(h_bin_final)
        if sc_bin_f < best_score:
            best_score = sc_bin_f
            best_h = h_bin_final.copy()
        bi_time = min(40, tleft() * 0.5)
        h_bi3, sc_bi3 = binary_best_improve(best_h, n, target,
                                            max_scans=15, time_limit=bi_time, rng=rng)
        if sc_bi3 < best_score:
            best_score = sc_bi3
            best_h = h_bi3.copy()

    # Phase 12: Final continuous polish
    if tleft() > 40:
        h_ref, sc_ref = slsqp_refine(best_h, n, target, 10000, 500, 1e-14)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    if tleft() > 30:
        sg_time = min(20, tleft() * 0.6)
        h_sg2, sc_sg2 = subgradient_refine(best_h, n, target,
                                           max_steps=3000, time_limit=sg_time)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2.copy()

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
