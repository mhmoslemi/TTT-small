    g += 2.0 * mu * (np.sum(x) - target)
    return g


def lbfgsb_active(h, n, target, alpha, m_active, mu, maxiter, deadline):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:m_active]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, method='L-BFGS-B', jac=lambda x: lse_objg(x, n, target, alpha, top_idx, mu),
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e6})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_active(h, n, target, alpha, m_active, maxiter, ftol, deadline):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:m_active]
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_obj(x, n, target, alpha, top_idx, 0.0), h, method='SLSQP',
                       jac=lambda x: lse_objg(x, n, target, alpha, top_idx, 0.0),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_full_softmax(h, n, target, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n

    def obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def objg(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        g = np.zeros(n)
        for wk, m in zip(w, range(len(corr))):
            if wk < 1e-14:
                continue
            k = int(m) - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - x[k:])
                g[k:] -= wk * x[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - x[:n - ell])
                g[:n - ell] -= wk * x[ell:]
        g *= dx
        return g

    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, target)
    except Exception:
        return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coarse_to_fine(h_fine, n_coarse, alpha, m_active, deadline):
    n_fine = len(h_fine)
    target_coarse = n_coarse / 2.0
    x_fine = np.linspace(0.0, 1.0, n_fine)
    x_coarse = np.linspace(0.0, 1.0, n_coarse)
    h_c = np.interp(x_coarse, x_fine, h_fine)
    h_c = project_feasible(h_c, target_coarse)
    best_c = h_c.copy()
    best_sc = compute_c5(h_c)
    h_a, sc_a = adam_optimize(h_c, n_coarse, target_coarse, alpha, m_active,
                               min(3000, max(100, int((deadline - time.time()) * 20))),
                               0.05, 0.9, 0.999, deadline)
    if sc_a < best_sc:
        best_sc = sc_a
        best_c = h_a
    h_out = np.interp(x_fine, x_coarse, best_c)
    return project_feasible(h_out, n_fine / 2.0)


def worst_lag_step(h, n, target, eps, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for _ in range(200):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B with active set (fast global improvement)
    if time.time() < deadline - 400:
        alphas_p1 = [50, 200, 800, 3000]
        for alpha in alphas_p1:
            if time.time() > deadline - 300:
                break
            for m_act in [5, 8]:
                if time.time() > deadline - 200:
                    break
                h_new = lbfgsb_active(best_h, n, target, alpha, m_act, 100.0, 150, deadline)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new

    # Phase 2: Adam subgradient - main workhorse
    if time.time() < deadline - 300:
        configs = [
            (500, 6, 0.03, 0.9, 0.999),
            (1000, 5, 0.02, 0.9, 0.999),
            (3000, 4, 0.015, 0.9, 0.999),
            (5000, 3, 0.01, 0.9, 0.999),
            (8000, 4, 0.008, 0.9, 0.998),
        ]
        for alpha, m_act, lr, b1, b2 in configs:
            if time.time() > deadline - 100:
                break
            h_a, sc_a = adam_optimize(best_h, n, target, alpha, m_act, 50000, lr, b1, b2, deadline - 60)
            if sc_a < best_score:
                best_score = sc_a
                best_h = h_a

    # Phase 3: SLSQP refinement on active set
    if time.time() < deadline - 150:
        for alpha in [2000, 8000, 20000]:
            if time.time() > deadline - 80:
                break
            h_new = slsqp_active(best_h, n, target, alpha, 5, 2000, 1e-15, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Coarse-to-fine multi-resolution
    if time.time() < deadline - 250:
        h_c2f = coarse_to_fine(best_h, 100, 500, 5, deadline - 150)
        sc_c2f = compute_c5(h_c2f)
        if sc_c2f < best_score:
            best_score = sc_c2f
            best_h = h_c2f
        # Now optimize at full resolution from the coarse-guided solution
        if time.time() < deadline - 100:
            h_a, sc_a = adam_optimize(best_h, n, target, 2000, 4, 20000, 0.02, 0.9, 0.999, deadline - 60)
            if sc_a < best_score:
                best_score = sc_a
                best_h = h_a
            h_new = slsqp_active(best_h, n, target, 10000, 4, 1000, 1e-15, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 5: Higher resolution attempt (n=600)
    if time.time() < deadline - 300:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        if time.time() < deadline - 200:
            for alpha in [200, 1000, 5000]:
                if time.time() > deadline - 100:
                    break
                h2_new = lbfgsb_active(best_h2, n2, target2, alpha, 6, 50.0, 150, deadline)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 100:
            h2_a, sc2_a = adam_optimize(best_h2, n2, target2, 3000, 4, 10000, 0.02, 0.9, 0.999, deadline - 60)
            if sc2_a < best_score2:
                best_score2 = sc2_a
                best_h2 = h2_a
        if time.time() < deadline - 60:
            h2_new = slsqp_active(best_h2, n2, target2, 10000, 4, 1000, 1e-15, deadline)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        best_score = min(best_score, best_score2)
        if best_score2 < compute_c5(best_h):
            best_score = best_score2
            best_h = best_h2

    # Phase 6: Random restarts with structured perturbations
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(6):
            if time.time() > deadline - 40:
                break
            if attempt % 3 == 0:
                scale = 0.1
                h_pert = best_h + rng.randn(n_cur) * scale
            elif attempt % 3 == 1:
                # Sinusoidal perturbation
                x = np.linspace(0, 2, n_cur)
                freq = rng.uniform(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.05
                h_pert = best_h + amp * np.sin(2 * np.pi * freq * x + phase)
            else:
                # Block perturbation
                h_pert = best_h.copy()
                block = n_cur // 4
                start = rng.randint(0, max(1, n_cur - block))
                h_pert[start:start + block] += rng.randn(block) * 0.15
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = adam_optimize(h_pert, n_cur, target_cur, 1000, 5, 5000, 0.02, 0.9, 0.999, deadline - 20)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_new = slsqp_active(best_h, n_cur, target_cur, 5000, 4, 1000, 1e-15, deadline)
                sc2 = compute_c5(h_new)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_new

    # Phase 7: Final Adam polish with high alpha
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_a, sc_a = adam_optimize(best_h, n_cur, target_cur, 20000, 3, 10000, 0.005, 0.9, 0.999, deadline - 10)
        if sc_a < best_score:
            best_score = sc_a
            best_h = h_a

    # Phase 8: Worst-lag surgery
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_surg, sc_surg = worst_lag_step(best_h, n_cur, target_cur, 0.003, deadline - 5)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 9: Final SLSQP
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_active(best_h, n_cur, target_cur, 30000, 3, 500, 1e-16, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
