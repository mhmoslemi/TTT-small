            h = proj_grad_step(h, weights, dx, target_sum, lr)
            h = proj_onto_set(h, target_sum)
    return best_h, best_score

def binary_greedy(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = np.clip(h, 0, 1).astype(float)
    dx = 2.0 / n
    full = compute_corr_full(h)
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = np.argsort(A)[:min(30, n)]
        top_j = np.argsort(-B)[:min(30, n)]
        found = False
        for i in top_i:
            for j in top_j:
                if h[i] <= 0.01 or h[j] >= 0.99:
                    continue
                di = h[i]
                dj = 1.0 - h[j]
                delta = np.zeros(2 * n - 1)
                k_vals = np.arange(2 * n - 1) - (n - 1)
                idx_a = i + k_vals
                m1 = (idx_a >= 0) & (idx_a < n)
                delta[m1] += di * (1.0 - h[idx_a[m1]])
                idx_b = i - k_vals
                m2 = (idx_b >= 0) & (idx_b < n)
                delta[m2] += di * h[idx_b[m2]]
                idx_c = j - k_vals
                m3 = (idx_c >= 0) & (idx_c < n)
                delta[m3] -= dj * (1.0 - h[idx_c[m3]])
                idx_d = j + k_vals
                m4 = (idx_d >= 0) & (idx_d < n)
                delta[m4] -= dj * h[idx_d[m4]]
                new_max = float((full + delta).max())
                if new_max < max_val - 1e-12:
                    h[i] -= di
                    h[j] += dj
                    h[i] = max(0.0, h[i])
                    h[j] = min(1.0, h[j])
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def sa_binary(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = np.round(h).astype(float)
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    dx = 2.0 / n
    full = compute_corr_full(h)
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 100000:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 7 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                i_cands = ones[A[ones] < 0.5]
                j_cands = zeros[B[zeros] > 0.5]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = np.zeros(2 * n - 1)
        k_vals = np.arange(2 * n - 1) - (n - 1)
        idx_a = i + k_vals
        m1 = (idx_a >= 0) & (idx_a < n)
        delta[m1] += (1.0 - h[idx_a[m1]])
        idx_b = i - k_vals
        m2 = (idx_b >= 0) & (idx_b < n)
        delta[m2] += h[idx_b[m2]]
        idx_c = j - k_vals
        m3 = (idx_c >= 0) & (idx_c < n)
        delta[m3] -= (1.0 - h[idx_c[m3]])
        idx_d = j + k_vals
        m4 = (idx_d >= 0) & (idx_d < n)
        delta[m4] -= h[idx_d[m4]]
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def interpolate_to(src, n_dst, target_dst):
    n_src = len(src)
    if n_src == n_dst:
        return project_feasible(src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Continuous gradient descent at original resolution
    if tleft() > 100:
        alphas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
        allowed = min(200, tleft() * 0.25)
        h_gd, c5_gd = smooth_grad_descent(h_init, n_init, target_init,
                                          alphas, 500, allowed, start_time)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd.copy()

    # Phase 2: Binary refinement
    if tleft() > 60:
        h_bin = np.round(best_h).astype(float)
        h_bin = project_feasible(h_bin, target_init)
        allowed = min(60, tleft() * 0.15)
        h_bg, c5_bg = binary_greedy(h_bin, 30000, rng, allowed, start_time)
        h_sa, c5_sa = sa_binary(h_bg, int(min(3000000, allowed * 30000)),
                                0.02, 0.0001, rng, allowed, start_time)
        if c5_sa < c5_bg:
            h_bin_best = h_sa
            c5_bin_best = c5_sa
        else:
            h_bin_best = h_bg
            c5_bin_best = c5_bg
        if c5_bin_best < best_score:
            best_score = c5_bin_best
            best_h = h_bin_best.copy()

    # Phase 3: Higher resolution
    for n_try in [n_init * 2, n_init * 3]:
        if tleft() < 120:
            break
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_h = [10, 20, 50, 100, 200, 500, 1000]
        allowed = min(180, tleft() * 0.35)
        h_gd2, c5_gd2 = smooth_grad_descent(h_interp, n_try, target_try,
                                            alphas_h, 400, allowed, start_time)
        if c5_gd2 < best_score:
            best_score = c5_gd2
            best_h = h_gd2.copy()

    # Phase 4: More gradient descent on best (refined)
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_r = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
        allowed = min(150, tleft() * 0.3)
        h_ref, c5_ref = smooth_grad_descent(best_h, n_use, target_use,
                                            alphas_r, 800, allowed, start_time)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref.copy()

    # Phase 5: Binary search from continuous
    if tleft() > 50:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin2 = np.round(best_h).astype(float)
        h_bin2 = project_feasible(h_bin2, float(target_use))
        allowed = min(40, tleft() * 0.15)
        h_bg2, c5_bg2 = binary_greedy(h_bin2, 20000, rng, allowed, start_time)
        h_sa2, c5_sa2 = sa_binary(h_bg2, int(min(2000000, allowed * 40000)),
                                  0.015, 0.00005, rng, allowed, start_time)
        if c5_sa2 < c5_bg2:
            h_b2 = h_sa2
            c5_b2 = c5_sa2
        else:
            h_b2 = h_bg2
            c5_b2 = c5_bg2
        if c5_b2 < best_score:
            best_score = c5_b2
            best_h = h_b2.copy()

    # Phase 6: Multi-start perturbation + gradient
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        for trial in range(5):
            if tleft() < 20:
                break
            h_pert = best_h.copy()
            noise = np.random.RandomState(seed + 1000 * trial).randn(n_use) * 0.1
            h_pert = np.clip(h_pert + noise, 0, 1)
            h_pert = project_feasible(h_pert, target_use)
            allowed = min(30, tleft() * 0.15)
            alphas_p = [50, 200, 500, 2000]
            h_p, c5_p = smooth_grad_descent(h_pert, n_use, target_use,
                                            alphas_p, 600, allowed, start_time)
            if c5_p < best_score:
                best_score = c5_p
                best_h = h_p.copy()

    # Phase 7: Final fine-tune gradient
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_f = [500, 1000, 5000, 10000, 20000]
        allowed = max(10, tleft() - 10)
        h_fin, c5_fin = smooth_grad_descent(best_h, n_use, target_use,
                                            alphas_f, 1000, allowed, start_time)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = proj_onto_set(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = project_feasible(best_h, target_final)
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
