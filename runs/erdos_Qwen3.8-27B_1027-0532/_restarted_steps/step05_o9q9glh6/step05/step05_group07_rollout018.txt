```python
def project_box_sum(v, target_sum):
    """Project v onto {h : 0<=h_i<=1, sum(h)=target_sum}"""
    # Find t such that sum(clip(v - t, 0, 1)) = target_sum
    lo = v.min() - target_sum  # lower bound for t
    hi = v.max()               # upper bound for t
    # Actually: sum(clip(v-t, 0, 1)) is decreasing in t
    # At t = min(v): sum = sum(max(v - min(v), 0)) >= 0
    # At t = max(v): sum = sum(min(v - max(v), 1)) = sum(0) = 0... 
    # Let me think more carefully.
    # clip(v-t, 0, 1) = max(0, min(1, v-t))
    # As t increases, v-t decreases, so clip decreases.
    # f(t) = sum(clip(v-t, 0, 1)) is decreasing in t.
    # f(min(v)) = sum(max(0, min(1, v-min(v)))) which could be up to n
    # f(max(v)+1) = 0
    # We want f(t) = target_sum
    
    lo = float(v.min()) - 1.0  # f(lo) >= n (all clip to 1 or close)
    hi = float(v.max()) + 1.0  # f(hi) <= 0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(v - t, 0, 1)
```

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_D(h, n):
    h_pad = np.zeros(3 * n - 1)
    h_pad[n-1:2*n-1] = h
    i = np.arange(n)[:, None]
    k = np.arange(-(n-1), n)[None, :]

    mask1 = ((i - k) >= 0) & ((i - k) < n)
    idx2 = (n-1) + i + k
    mask2 = ((i + k) >= 0) & ((i + k) < n)
    return (1.0 - h_pad[idx1]) * mask1 - h_pad[idx2] * mask2

            break
        
        D = compute_D(h_cur, n)
        perm = rng.permutation(n)

            j = int(perm[2 * p + 1])
            c_sum = h_cur[i] + h_cur[j]

            t_min = max(0.0, c_sum - 1.0)

            hi = t_max - h_cur[i]

                k_star = np.argmax(corr + mid * B)
                bs = B[k_star]
                if bs > 1e-11:
                    hi = mid
                elif bs < -1e-11:
                    lo = mid
                else:
                    break

            if abs(delta_opt) > 1e-14:
                h_cur[i] += delta_opt
                h_cur[j] -= delta_opt
                corr += delta_opt * B

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_D(h, n):
    h_pad = np.zeros(3 * n - 1)
    h_pad[n-1:2*n-1] = h
    i = np.arange(n)[:, None]
    k = np.arange(-(n-1), n)[None, :]
    idx1 = (n-1) + i - k
    mask1 = ((i - k) >= 0) & ((i - k) < n)
    idx2 = (n-1) + i + k
    mask2 = ((i + k) >= 0) & ((i + k) < n)
    return (1.0 - h_pad[idx1]) * mask1 - h_pad[idx2] * mask2

def paired_cd(h, n, target, deadline, rng):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    n_half = n // 2
    stall = 0
    pert_count = 0
    
    for sweep in range(10000000):
        if time.time() > deadline:
            break
        
        D = compute_D(h_cur, n)
        perm = rng.permutation(n)
        
        for p in range(n_half):
            i = int(perm[2 * p])
            j = int(perm[2 * p + 1])
            c_sum = h_cur[i] + h_cur[j]
            if c_sum < 1e-14 or c_sum > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c_sum - 1.0)
            t_max = min(1.0, c_sum)
            if t_max - t_min < 1e-13:
                continue
            
            B = D[i] - D[j]
            lo = t_min - h_cur[i]
            hi = t_max - h_cur[i]
            
            for _ in range(45):
                mid = (lo + hi) * 0.5
                k_star = np.argmax(corr + mid * B)
                bs = B[k_star]
                if bs > 1e-11:
                    hi = mid
                elif bs < -1e-11:
                    lo = mid
                else:
                    break
            
            delta_opt = (lo + hi) * 0.5
            if abs(delta_opt) > 1e-14:
                h_cur[i] += delta_opt
                h_cur[j] -= delta_opt
                corr += delta_opt * B
        
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-14:
            best_score = new_score
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            scale = min(0.01 + 0.005 * pert_count, 0.06)
            h_cur = project_feasible(h_cur + rng.randn(n) * scale, target)
            corr = compute_corr(h_cur)
            stall = 0
            pert_count += 1
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_deadline = 0.0

def _lse_obj(x):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _lse_grad(x):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    p = np.exp(shifted)
    p /= np.sum(p)
    D = compute_D(x, _g_n)
    return D.T @ p / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h, n, target, alphas, maxiter=2000):
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if time.time() > _g_deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
