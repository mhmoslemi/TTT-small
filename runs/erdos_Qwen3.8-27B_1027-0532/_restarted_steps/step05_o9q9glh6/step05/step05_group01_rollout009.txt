_global_n = 0
_global_target = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    for _ in range(3):
        if abs(residual) < 1e-13:
            break
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) == 0:
            break
        h[idx2] += residual / len(idx2)
        h = np.clip(h, 0.0, 1.0)
        residual = target_sum - np.sum(h)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _global_target

    return np.ones(_global_n)

    _global_n = n
    _global_target = target
    start_t = time.time()

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break

        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + (1.0 - mu) * grad
        h_new = project_feasible(h - lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_now - 1e-15:
            h = h_new
            if c5_new < best_c5:
                lr = min(lr * 1.05, base_lr * 2)
        else:
            lr *= 0.92
        if lr < 1e-7:
            lr = base_lr * 0.5
            v *= 0.3

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_sa(h_init, n, target, time_limit, max_iters=500000, seed=777):
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = np.where(h_init > 0.5, 1.0, 0.0)
    h = project_feasible(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    T = 0.08
    T_min = 0.0005
    cooling = 0.9998

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)

        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if rng.rand() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
            else:
                h[i] = 1.0
                h[j] = 0.0

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        T = max(T_min, T * cooling)

    return best_h, best_c5

def block_swap_cd(h, n, target, time_limit, max_sweeps=2000, block_sizes=(2, 3, 5, 8)):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(42)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break

        bs = block_sizes[rng.randint(len(block_sizes))]
        if bs >= n:
            bs = max(1, n // 4)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue

        h_try = h.copy()
        h_try[i:i+bs], h_try[j:j+bs] = h_try[j:j+bs].copy(), h_try[i:i+bs].copy()
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_val - 1e-14:
            h = h_try

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_cur = len(h_init)
    target_cur = n_cur / 2.0

    best_h = project_feasible(h_init, target_cur)
    best_score = compute_c5(best_h)
    n_best = n_cur

    # Phase 1: SLSQP with gradual alpha increase
    if tleft() > 120:
        h1 = slsqp_round(best_h, n_best, target_cur,
                         [30, 100, 500, 2000, 10000],
                         1200, 1e-15, time_limit=100)
        sc = compute_c5(h1)
        if sc < best_score:
            best_score = sc
            best_h = h1

    # Phase 2: Pair coordinate descent
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.35)
        h2, sc2 = pair_cd(best_h, n_best, target_cur,
                          time_limit=cd_time, max_sweeps=5000, n_cand=18, bs_iters=14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: SLSQP after CD
    if tleft() > 80:
        h3 = slsqp_round(best_h, n_best, target_cur,
                         [5000, 20000, 80000, 200000],
                         2000, 1e-15, time_limit=min(70, tleft() * 0.5))
        sc = compute_c5(h3)
        if sc < best_score:
            best_score = sc
            best_h = h3

    # Phase 4: Another pair CD round with different params
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.3)
        h4, sc4 = pair_cd(best_h, n_best, target_cur,
                          time_limit=cd_time, max_sweeps=4000, n_cand=15, bs_iters=18)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Momentum subgradient for global mixing
    if tleft() > 80:
        h5, sc5 = momentum_sg(best_h, n_best, target_cur,
                              time_limit=min(50, tleft() * 0.4),
                              max_steps=20000, n_top=15, base_lr=0.01, mu=0.9)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: SLSQP polish
    if tleft() > 60:
        h6 = slsqp_round(best_h, n_best, target_cur,
                         [10000, 50000, 200000],
                         1500, 1e-15, time_limit=min(50, tleft() * 0.5))
        sc = compute_c5(h6)
        if sc < best_score:
            best_score = sc
            best_h = h6

    # Phase 7: Binary simulated annealing for diversification
    if tleft() > 80:
        h7, sc7 = binary_sa(best_h, n_best, target_cur,
                            time_limit=min(40, tleft() * 0.4),
                            max_iters=300000, seed=seed)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
        # Polish the SA result
        if tleft() > 40:
            h7p = slsqp_round(h7, n_best, target_cur,
                             [1000, 5000, 20000],
                             1000, 1e-15, time_limit=min(30, tleft() * 0.4))
            sc7p = compute_c5(h7p)
            if sc7p < best_score:
                best_score = sc7p
                best_h = h7p

    # Phase 8: Multi-resolution attempt
    if tleft() > 150:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            if n_try <= n_best:
                continue
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            sc_in = compute_c5(h_interp)
            h8 = slsqp_round(h_interp, n_try, target_try,
                            [50, 200, 1000, 5000, 20000],
                            1500, 1e-15, time_limit=min(60, tleft() * 0.35))
            sc8 = compute_c5(h8)
            if tleft() > 50:
                h8cd, sc8cd = pair_cd(h8, n_try, target_try,
                                      time_limit=min(30, tleft() * 0.25),
                                      max_sweeps=2000, n_cand=15, bs_iters=12)
                if sc8cd < sc8:
                    sc8 = sc8cd
                    h8 = h8cd
            if sc8 < best_score:
                best_score = sc8
                best_h = h8
                n_best = n_try
                target_cur = target_try

    # Phase 9: Block swap diversification
    if tleft() > 60:
        h9, sc9 = block_swap_cd(best_h, n_best, target_cur,
                                time_limit=min(30, tleft() * 0.4),
                                max_sweeps=3000)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Final SLSQP + pair CD
    if tleft() > 40:
        h10 = slsqp_round(best_h, n_best, target_cur,
                          [20000, 100000, 500000],
                          2000, 1e-15, time_limit=min(25, tleft() * 0.5))
        sc = compute_c5(h10)
        if sc < best_score:
            best_score = sc
            best_h = h10

    if tleft() > 20:
        h10b, sc10b = pair_cd(best_h, n_best, target_cur,
                              time_limit=min(15, tleft() * 0.6),
                              max_sweeps=1000, n_cand=12, bs_iters=14)
        if sc10b < best_score:
            best_score = sc10b
            best_h = h10b

    # Final projection
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_use
