    return best_h, best_f


def subgrad_phase(h, n, target, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_f = compute_c5(h_cur)
    step = 0.004
    momentum = np.zeros(n)
    beta_m = 0.85
    temp = 0.008
    for it in range(20000):
        if time.time() > deadline:
            break
        c = compute_corr(h_cur)
        top = np.argsort(c)[::-1][:6]
        vt = c[top]
        vm = float(vt.max())
        w = np.exp((vt - vm) / temp)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(6):
            idx = int(top[ii])
            k = idx - (n - 1)
            if k >= 0:
                g[:n - k] += w[ii] * (1.0 - h_cur[k:])
                g[k:] -= w[ii] * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += w[ii] * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= w[ii] * h_cur[ell:]
        g *= dx
        momentum = beta_m * momentum + g
        h_try = proj(h_cur - step * momentum, target)
        f_try = compute_c5(h_try)
        f_cur = compute_c5(h_cur)
        if f_try < f_cur - 1e-14:
            h_cur = h_try
            if f_try < best_f:
                best_f = f_try
                best_h = h_cur.copy()
            step = min(step * 1.02, 0.02)
        else:
            step *= 0.9
            if step < 1e-10:
                break
        if it % 200 == 0:
            temp *= 0.8
    return best_h, best_f


def swap_local_search(h, n, target, n_trials, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_f = compute_c5(h_cur)
    rng = np.random.RandomState(12345)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_try = h_cur.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        d = target - h_try.sum()
        if abs(d) > 1e-12:
            h_try = proj(h_try, target)
        f_try = compute_c5(h_try)
        if f_try < best_f - 1e-14:
            best_f = f_try
            best_h = h_try
            h_cur = h_try
    return best_h, best_f


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return proj(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return proj(h_out, target_dst)


def full_optimize(h0, n, target, deadline):
    h = proj(h0.copy(), target)
    best_h = h.copy()
    best_f = compute_c5(h)

    t1 = deadline * 0.35
    h, f = lbfgs_phase(h, n, target, t1)
    if f < best_f:
        best_f = f
        best_h = h

    t2 = deadline * 0.55
    h, f = active_set_phase(best_h, n, target, t2)
    if f < best_f:
        best_f = f
        best_h = h

    t3 = deadline * 0.7
    h, f = slsqp_phase(best_h, n, target, t3)
    if f < best_f:
        best_f = f
        best_h = h

    t4 = deadline * 0.85
    h, f = subgrad_phase(best_h, n, target, t4)
    if f < best_f:
        best_f = f
        best_h = h

    h, f = slsqp_phase(best_h, n, target, deadline)
    if f < best_f:
        best_f = f
        best_h = h

    return best_h, best_f


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20.0
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    target0 = n0 / 2.0
    h0 = proj(np.array(initial_h_values, dtype=float), target0)

    best_h = h0.copy()
    best_f = compute_c5(h0)

    remaining = deadline - time.time()
    alloc_400 = min(remaining * 0.45, 550.0)
    alloc_600 = min(remaining * 0.30, 380.0)
    alloc_800 = min(remaining * 0.20, 250.0)

    t_end_1 = t0 + alloc_400
    h1, f1 = full_optimize(h0, n0, target0, t_end_1)
    if f1 < best_f:
        best_f = f1
        best_h = h1

    n1 = 600
    if n1 > n0 and time.time() < deadline - alloc_800 - 20:
        target1 = n1 / 2.0
        h1_init = interpolate_to(best_h, n1, target1)
        t_end_2 = min(t_end_1 + alloc_600, deadline - alloc_800 - 20)
        h2, f2 = full_optimize(h1_init, n1, target1, t_end_2)
        if f2 < best_f:
            best_f = f2
            best_h = h2
    elif time.time() < deadline - alloc_800 - 20:
        t_end_2 = min(t_end_1 + 200, deadline - alloc_800 - 20)
        h2, f2 = full_optimize(best_h, n0, target0, t_end_2)
        if f2 < best_f:
            best_f = f2
            best_h = h2

    n2 = 800
    if len(best_h) < n2 and time.time() < deadline - 30:
        target2 = n2 / 2.0
        h2_init = interpolate_to(best_h, n2, target2)
        h3, f3 = full_optimize(h2_init, n2, target2, deadline - 10)
        if f3 < best_f:
            best_f = f3
            best_h = h3

    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        t_fin = deadline - 3
        h_fin, f_fin = slsqp_phase(best_h, n_cur, target_cur, t_fin)
        if f_fin < best_f:
            best_f = f_fin
            best_h = h_fin
        h_sg, f_sg = subgrad_phase(best_h, n_cur, target_cur, deadline - 2)
        if f_sg < best_f:
            best_f = f_sg
            best_h = h_sg

    h_final = proj(best_h, len(best_h) / 2.0)
    final_f = compute_c5(h_final)
    return h_final, final_f, len(h_final)
