        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj - 1e-14:
            best_obj = sc
            best_x = x.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 200:
            t = 1.0
            stagnant = 0
    return best_x, best_obj

def active_equalize(h, target, deadline, n_active=5, n_rounds=500):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        corr_dx = corr * (2.0 / n)
        sorted_idx = np.argsort(corr_dx)[::-1][:n_active]
        c_top = corr_dx[sorted_idx]
        c_max = c_top[0]
        c_min_top = c_top[-1]
        spread = c_max - c_min_top
        if spread < 1e-8:
            break
        w = np.exp((c_top - c_max) / 0.002)
        w /= w.sum()
        grad = np.zeros(n)
        dx = 2.0 / n
        for wi, ki in zip(w, sorted_idx):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            if ki > 0:
                jn = np.arange(ki, n)
                grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        for eps_try in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h_cur - eps_try * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

def perturb_and_optimize(h, target, deadline, alpha, rng, scale):
    n = len(h)
    h_pert = h + rng.standard_normal(n) * scale
    h_pert = project_feasible(h_pert, target)
    h_opt, sc = fista_optimize(h_pert, target, alpha, 10000, deadline)
    return h_opt, sc

def flip_search(h, target, deadline, n_trials=5000):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(99)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        idx = rng.randint(n)
        if h_cur[idx] < 0.5:
            h_new = h_cur.copy()
            h_new[idx] = 1.0
        else:
            h_new = h_cur.copy()
            h_new[idx] = 0.0
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc
    return h_cur, cur_score

def multi_scale_perturb(h, target, deadline, alpha, rng):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    scales = [0.02, 0.04, 0.08, 0.12, 0.06, 0.1, 0.15, 0.05]
    for s in scales:
        if time.time() > deadline:
            break
        h_pert = h_cur + rng.standard_normal(n) * s
        h_pert = project_feasible(h_pert, target)
        h_opt, sc = fista_optimize(h_pert, target, alpha, 8000, min(deadline, time.time() + 20))
        sc_actual = compute_c5(h_opt)
        if sc_actual < cur_score:
            h_cur = h_opt
            cur_score = sc_actual
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline_final = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline_final - time.time()

    # Phase 1: FISTA continuation (moderate alpha)
    alphas_cont = [100, 500, 2000, 8000]
    times_cont = [40, 50, 60, 70]
    for i, (a, td) in enumerate(zip(alphas_cont, times_cont)):
        if tleft() < td + 20:
            break
        dl = time.time() + min(td, tleft() - 20)
        h_new, sc_new = fista_optimize(best_h, target, a, 50000, dl)
        sc_actual = compute_c5(h_new)
        if sc_actual < best_score:
            best_score = sc_actual
            best_h = h_new

    # Phase 2: Active set equalization
    if tleft() > 60:
        h_new, sc_new = active_equalize(best_h, target, time.time() + min(60, tleft() - 20), n_active=5, n_rounds=800)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: High alpha FISTA
    if tleft() > 80:
        h_new, sc_new = fista_optimize(best_h, target, 20000, 50000, time.time() + min(50, tleft() - 30))
        sc_actual = compute_c5(h_new)
        if sc_actual < best_score:
            best_score = sc_actual
            best_h = h_new

    # Phase 4: More equalization
    if tleft() > 50:
        h_new, sc_new = active_equalize(best_h, target, time.time() + min(40, tleft() - 20), n_active=7, n_rounds=600)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 5: Very high alpha FISTA
    if tleft() > 80:
        h_new, sc_new = fista_optimize(best_h, target, 50000, 50000, time.time() + min(50, tleft() - 30))
        sc_actual = compute_c5(h_new)
        if sc_actual < best_score:
            best_score = sc_actual
            best_h = h_new

    # Phase 6: Multi-scale perturbation with high alpha
    if tleft() > 100:
        h_new, sc_new = multi_scale_perturb(best_h, target, time.time() + min(80, tleft() - 30), 10000, rng)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Flip search
    if tleft() > 60:
        h_new, sc_new = flip_search(best_h, target, time.time() + min(50, tleft() - 20), n_trials=8000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 8: Final high-alpha FISTA + equalization
    if tleft() > 60:
        h_new, sc_new = fista_optimize(best_h, target, 100000, 50000, time.time() + min(35, tleft() - 25))
        sc_actual = compute_c5(h_new)
        if sc_actual < best_score:
            best_score = sc_actual
            best_h = h_new
        if tleft() > 25:
            h_new, sc_new = active_equalize(best_h, target, time.time() + min(20, tleft() - 15), n_active=10, n_rounds=400)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 9: Final flip polish
    if tleft() > 20:
        h_new, sc_new = flip_search(best_h, target, time.time() + min(15, tleft() - 10), n_trials=3000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
