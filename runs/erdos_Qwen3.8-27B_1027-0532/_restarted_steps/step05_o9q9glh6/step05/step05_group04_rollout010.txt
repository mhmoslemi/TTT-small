- From h[m+k] term: m+k = j, so m = j-k. Valid for 0 <= j-k <= n-1-k, i.e., k <= j <= n-1.

        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final SLSQP with very high alpha
    if remaining() > 40:
        deadline = start_time + budget_s - 20
        h9, sc9 = slsqp_round(best_h, n, target, [10000, 30000], 3000, 1e-17, deadline)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
