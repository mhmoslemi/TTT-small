from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

        else:
            mask = h > 0.0

            break

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

    return np.sum(x) - _g_target

def constraint_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(soft_max_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:

    return h_cur, cur_score

def subgradient_all_lags(h, n_steps, n_top, lr_init, lr_decay, momentum, temp):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Get top lags (all positive and negative)
        top_idx = np.argsort(corr)[::-1][:n_top]
        max_val = corr[top_idx[0]]
        diffs = corr[top_idx] - max_val
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k_idx in zip(weights, top_idx):
            k_idx = int(k_idx)
            # k_idx is index in full correlation, shift = k_idx - (n-1)
            shift = k_idx - (n - 1)
            if shift == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            elif shift > 0:
                idx1 = np.arange(0, n - shift)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + shift])
                idx2 = np.arange(shift, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - shift]
            else:
                m = -shift
                idx1 = np.arange(0, n - m)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + m])
                idx2 = np.arange(m, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - m]
        
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

These are different because the linear terms S_k and S'_k are different!

So for the alternating minimization, I need to handle positive and negative lags differently.

For lag k > 0:
Maximize: ∑_{j=k}^{n-1} h[j] + ∑_{j=0}^{n-k-1} h[j]*h[j+k]
