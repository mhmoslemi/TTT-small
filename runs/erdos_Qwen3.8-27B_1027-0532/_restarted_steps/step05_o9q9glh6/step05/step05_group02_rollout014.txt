def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = np.where(h < 1.0 - 1e-15)[0]
    else:
        free = np.where(h > 1e-15)[0]
    if len(free) == 0:
        return h
    h[free] += diff / len(free)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    """Round h to binary with exact target_sum ones."""
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current > target_sum:
        need = current - target_sum
        order = np.argsort(h[ones])
        h_r[ones[order[:need]]] = 0
    elif current < target_sum:
        need = target_sum - current
        order = np.argsort(-h[zeros])
        h_r[zeros[order[:need]]] = 1
    return h_r.astype(float)

    m_idx = np.arange(n)

    term2 = M1.T @ h

def project_feasible(h, target_sum):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = np.where(h < 1.0 - 1e-14)[0]
    else:
        free = np.where(h > 1e-14)[0]
    if len(free) == 0:
        return h
    h[free] += diff / len(free)
    return np.clip(h, 0.0, 1.0)

    ones_idx = np.where(h_r == 1)[0]
    zeros_idx = np.where(h_r == 0)[0]
    if current > target_sum:
        need = current - target_sum
        order = np.argsort(h[ones_idx])
        h_r[ones_idx[order[:need]]] = 0
    elif current < target_sum:
