def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def leveling_step(h_cur, corr, top_idxs, n, target, epsilon):
    """Find a direction that decreases all top shifts."""
    dx = 2.0 / n
    n_active = len(top_idxs)
    
    # Compute gradients for each active shift
    grads = []
    for m_idx in top_idxs:
        m_idx = int(m_idx)
        k = m_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h_cur[k:])
            g[k:] -= h_cur[:n-k]
        else:
            ell = -k
            g[ell:] += (1.0 - h_cur[:n-ell])
            g[:n-ell] -= h_cur[ell:]
        g *= dx
        grads.append(g)
    
    # Find direction that minimizes max inner product
    # Simple approach: negative average gradient
    avg_grad = np.mean(grads, axis=0)
    d = -avg_grad
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return None
    d = d / d_norm
    
    # Line search
    for step in [epsilon, epsilon*0.5, epsilon*0.25, epsilon*0.1, epsilon*0.05]:
        h_new = h_cur + step * d
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < compute_c5(h_cur):
            return h_new, sc
    
    return None, compute_c5(h_cur)

Let me also consider using a **barrier method** instead of penalty: add -μ*sum(log(h[i])) - μ*sum(log(1-h[i])) to handle the box constraints. This keeps h strictly in (0,1) and can be more effective. But it's more complex to implement.

I'll stick with L-BFGS-B + penalty for simplicity.
