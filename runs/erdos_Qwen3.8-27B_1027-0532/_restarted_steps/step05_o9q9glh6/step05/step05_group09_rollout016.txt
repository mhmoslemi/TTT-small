```python
def project_feasible(h, target):
    h = np.array(h, dtype=float)
    n = len(h)
    # Sort in descending order
    order = np.argsort(h)[::-1]
    h_sorted = h[order]
    # For the top k elements above tau, and bottom n-k below tau:
    # sum(top_k) - k*tau = target => tau = (sum(top_k) - target) / k
    # Need: h_sorted[k-1] >= tau >= h_sorted[k] (with h_sorted[n] = -inf, h_sorted[-1] = +inf)
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    for k in range(1, n+1):
        tau = (cumsum[k] - target) / k
        if k < n and tau < h_sorted[k] - 1e-12:
            continue
        if tau > h_sorted[k-1] + 1e-12:

        result = np.clip(h - tau, 0.0, 1.0)

    # Fallback: bisection
    lo, hi = h.min() - 1.0, h.max() + 1.0

            hi = mid
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    order = np.argsort(h)[::-1]
    hs = h[order]
    cs = np.empty(n + 1)
    cs[0] = 0.0
    np.cumsum(hs, out=cs[1:])
    for k in range(1, n + 1):
        tau = (cs[k] - target) / k
        if k < n and tau < hs[k] - 1e-13:
            continue
        if tau > hs[k - 1] + 1e-13:
            continue
        return np.clip(h - tau, 0.0, 1.0)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = np.max(corr)
    sh = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    es = np.exp(sh)
    se = float(es.sum())
    w = es / se
    obj = (c_max + np.log(se) / alpha) * dx
    wf = w[::-1]
    fs = 2 * n
    A = np.fft.rfft(wf, fs) * np.fft.rfft(1.0 - h, fs)
    B = np.fft.rfft(w, fs) * np.fft.rfft(h, fs)
    cp = np.fft.irfft(A, fs)[n - 1:2 * n - 1]
    cn = np.fft.irfft(B, fs)[:n]
    grad = dx * (cp - cn)
    return obj, grad
```
