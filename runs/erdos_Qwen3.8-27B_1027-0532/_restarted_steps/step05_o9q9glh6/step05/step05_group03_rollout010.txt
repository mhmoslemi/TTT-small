    over = corr[n - 1:n - 1 + n]

    return h_cur, cur_score

def compute_grad_overlap(h, k, dx):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        upper = n - 1 - k
        idx1 = np.arange(0, upper + 1)
        grad[idx1] += (1.0 - h[idx1 + k]) * dx
        idx2 = np.arange(k, n)
        grad[idx2] -= h[idx2 - k] * dx
    return grad

def multi_shift_surgery(h, n_steps, n_active, lr_init, lr_decay, increase_factor, temp, momentum=0.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        over = get_overlaps(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            grad += wk * compute_grad_overlap(h_cur, int(k), dx)
        if momentum > 0:
            velocity = momentum * velocity + (1.0 - momentum) * grad
            step_vec = velocity
        else:
            step_vec = grad
        h_new = h_cur - lr * step_vec
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * increase_factor, 0.01)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def lq2_optimize(h, max_rounds=100, active_frac=0.05, project_iters=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    n_active = max(10, int(n * active_frac))
    for rd in range(max_rounds):
        over = get_overlaps(h_cur)
        c_max_raw = float(np.max(over))
        if c_max_raw * dx >= best_score - 1e-15:
            active = np.argsort(over)[::-1][:n_active]
        else:
            break
        slack = np.zeros(n)
        for k_idx in active:
            k = int(k_idx)
            g = compute_grad_overlap(h_cur, k, dx) / dx
            g = np.maximum(g, 0.0)
            slack = np.maximum(slack, g)
        slack_binary = (slack > 0.0).astype(float)
        gain = slack_binary * h_cur - (1.0 - slack_binary) * (1.0 - h_cur)
        order = np.argsort(-gain)[:n_active]
        obj = np.zeros(n)
        for k_idx in active:
            k = int(k_idx)
            g = compute_grad_overlap(h_cur, k, dx) / dx
            obj = obj + np.maximum(g, 0.0)
        c_vec = np.zeros(n)
        c_vec[n_active:] = obj[n_active:]
        A_eq = np.array([[1.0] * n])
        b_eq = np.array([target])
        A_ub = -np.eye(n)
        b_ub = np.zeros(n)
        try:
            from scipy.optimize import linprog as _lp
            res = _lp(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0.0, 1.0)] * n, method='highs')
            if res.status == 0:
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score - 1e-15:
                    h_cur = h_new
                    best_score = sc
                    best_h = h_new.copy()
        except Exception:
            break
    return best_h, best_score

def coordinate_round(h, target, thresh):
    h_r = h.copy()
    h_r[h_r < thresh] = 0.0
    h_r[h_r > 1.0 - thresh] = 1.0
    return project_feasible(h_r, target)

def try_rounding_improve(h, target):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(h)
    for thresh in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        h_r = coordinate_round(h, target, thresh)
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best = h_r.copy()
    return best, best_score

def simulate_annealing(h, target, n_steps, t_init, t_final, cooling, max_block):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    t = t_init
    for step in range(n_steps):
        t = t_init * (t_final / t_init) ** (step / max(1, n_steps - 1))
        n_block = np.random.randint(1, max_block + 1)
        start = np.random.randint(0, n - n_block)
        h_old = h_cur.copy()
        h_cur[start:start + n_block] += np.random.randn(n_block) * 0.3
        h_cur = project_feasible(h_cur, target)
        sc_new = compute_c5(h_cur)
        sc_old = compute_c5(h_old)
        delta = sc_new - sc_old
        if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
            pass
        else:
            h_cur = h_old
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_cur.copy()
    return best_h, best_score

def sparse_translation_move(h, target, k_shift):
    n = len(h)
    dx = 2.0 / n
    h_new = h.copy()
    if k_shift > 0:
        h_new[k_shift:] = h[:n - k_shift]
        h_new[:k_shift] = h[:k_shift]
    else:
        ks = -k_shift
        h_new[:n - ks] = h[ks:]
        h_new[n - ks:] = h[n - ks:]
    h_new = project_feasible(h_new, target)
    return h_new

def build_three_periodic(n, w1, gap):
    p = 2.0 / 3.0
    w2 = 1.0 / 3.0 - w1 - gap
    if w2 < 0:
        w2 = 0.0
    target = n / 2.0
    h = np.zeros(n)
    for i in range(n):
        x = i * 2.0 / n
        xp = x % p
        if xp <= w1 or (xp >= w1 + gap and xp <= w1 + gap + w2):
            h[i] = 1.0
    return project_feasible(h, target)

def tleft_calc(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP from initial
    if tleft_calc(start_time, budget_s) > 800:
        h_ref, sc = slsqp_optimize(best_h, [100, 500, 2000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Multi-shift surgery (different params than before)
    if tleft_calc(start_time, budget_s) > 600:
        # Attack top 3 with moderate temp
        h_sg, sc_sg = multi_shift_surgery(best_h, 800, 3, 0.002, 0.995, 1.005, 0.003, momentum=0.2)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        # Attack top 5 with lower temp
        h_sg2, sc_sg2 = multi_shift_surgery(best_h, 600, 5, 0.0015, 0.996, 1.004, 0.002, momentum=0.3)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2
        # Refine
        h_ref, sc = slsqp_optimize(best_h, [2000, 10000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: LQ2 block optimization
    if tleft_calc(start_time, budget_s) > 400:
        h_lq, sc_lq = lq2_optimize(best_h, max_rounds=60, active_frac=0.04, project_iters=50)
        if sc_lq < best_score:
            best_score = sc_lq
            best_h = h_lq
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Round to bang-bang then optimize
    if tleft_calc(start_time, budget_s) > 500:
        h_round, sc_round = try_rounding_improve(best_h, target)
        if sc_round < best_score:
            best_score = sc_round
            best_h = h_round
        h_ref, sc = slsqp_optimize(best_h, [1000, 5000, 20000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        h_sg3, sc_sg3 = multi_shift_surgery(best_h, 500, 3, 0.001, 0.997, 1.003, 0.002, momentum=0.3)
        if sc_sg3 < best_score:
            best_score = sc_sg3
            best_h = h_sg3

    # Phase 5: Try three-periodic structure as alternative starting point
    if tleft_calc(start_time, budget_s) > 300:
        best_3p_h = None
        best_3p_sc = best_score
        for w1_try in [0.08, 0.1, 0.12, 0.15, 0.18]:
            for gap_try in [0.0, 0.02, 0.04, 0.06, 0.08]:
                if w1_try + gap_try + 0.001 > 1.0 / 3.0:
                    continue
                h_3p = build_three_periodic(n, w1_try, gap_try)
                sc_3p = compute_c5(h_3p)
                if sc_3p < best_3p_sc:
                    best_3p_sc = sc_3p
                    best_3p_h = h_3p
        if best_3p_h is not None and best_3p_sc < best_score:
            h_ref, sc = slsqp_optimize(best_3p_h, [500, 2000, 10000], 2000, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            h_sg4, sc_sg4 = multi_shift_surgery(best_h, 400, 4, 0.002, 0.996, 1.004, 0.003, momentum=0.2)
            if sc_sg4 < best_score:
                best_score = sc_sg4
                best_h = h_sg4

    # Phase 6: Simulated annealing for global exploration
    if tleft_calc(start_time, budget_s) > 200:
        h_sa, sc_sa = simulate_annealing(best_h, target, n_steps=2000, t_init=0.01, t_final=0.0001, cooling=1.0, max_block=8)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa
        h_ref, sc = slsqp_optimize(best_h, [2000, 10000, 50000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Fine-grained multi-shift with very low temp
    if tleft_calc(start_time, budget_s) > 150:
        h_sg5, sc_sg5 = multi_shift_surgery(best_h, 1000, 2, 0.0005, 0.998, 1.002, 0.001, momentum=0.4)
        if sc_sg5 < best_score:
            best_score = sc_sg5
            best_h = h_sg5
        h_sg6, sc_sg6 = multi_shift_surgery(best_h, 800, 3, 0.0003, 0.999, 1.001, 0.0005, momentum=0.5)
        if sc_sg6 < best_score:
            best_score = sc_sg6
            best_h = h_sg6

    # Phase 8: LQ2 again with different params
    if tleft_calc(start_time, budget_s) > 120:
        h_lq2, sc_lq2 = lq2_optimize(best_h, max_rounds=80, active_frac=0.03, project_iters=50)
        if sc_lq2 < best_score:
            best_score = sc_lq2
            best_h = h_lq2

    # Phase 9: Random perturbations + SLSQP
    deadline_p9 = start_time + (budget_s - 25.0) * 0.82
    for attempt in range(8):
        if time.time() > deadline_p9 or tleft_calc(start_time, budget_s) < 60:
            break
        h_pert = best_h.copy()
        if attempt % 3 == 0:
            h_pert += rng.randn(n) * 0.03
        elif attempt % 3 == 1:
            block = max(5, n // 30)
            for b in range(n // block):
                if rng.random() < 0.3:
                    shift_val = rng.uniform(-0.08, 0.08)
                    h_pert[b * block:min((b + 1) * block, n)] += shift_val
        else:
            n_flips = max(5, n // 20)
            flip_idx = rng.choice(n, n_flips, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_optimize(h_pert, [1000, 5000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final ultra-fine surgery
    if tleft_calc(start_time, budget_s) > 60:
        h_sg7, sc_sg7 = multi_shift_surgery(best_h, 1500, 2, 0.0002, 0.999, 1.001, 0.0003, momentum=0.5)
        if sc_sg7 < best_score:
            best_score = sc_sg7
            best_h = h_sg7
        h_ref, sc = slsqp_optimize(best_h, [50000, 200000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final rounding check
    if tleft_calc(start_time, budget_s) > 30:
        h_round2, sc_r2 = try_rounding_improve(best_h, target)
        if sc_r2 < best_score:
            best_score = sc_r2
            best_h = h_round2
        h_ref, sc = slsqp_optimize(best_h, [100000], 500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final projection and evaluation
    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
