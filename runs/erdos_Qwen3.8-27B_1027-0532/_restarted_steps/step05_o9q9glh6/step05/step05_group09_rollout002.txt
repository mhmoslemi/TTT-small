```python
def compute_swap_delta(h, i, j, n):
    """Compute change in corr when h[i] goes 1->0 and h[j] goes 0->1"""
    delta = np.zeros(n)
    # i going 1->0:
    # Role as x (first position in pair): for k where i+k < n, i.e., k < n-i
    #   term h[i](1-h[i+k]) changes from (1-h[i+k]) to 0, delta = -(1-h[i+k])
    for k in range(min(n-i, n)):
        delta[k] -= (1 - h[i+k])
    # Role as x+k (second position): for k where i-k >= 0, i.e., k <= i
    #   term h[i-k](1-h[i]) changes from h[i-k]*0 to h[i-k]*1, delta = h[i-k]
    for k in range(min(i+1, n)):
        delta[k] += h[i-k]
    # j going 0->1:
    # Role as x: for k where j+k < n
    #   term h[j](1-h[j+k]) changes from 0 to (1-h[j+k]), delta = (1-h[j+k])
    for k in range(min(n-j, n)):
        delta[k] += (1 - h[j+k])
    # Role as x+k: for k where j-k >= 0
    #   term h[j-k](1-h[j]) changes from h[j-k]*1 to 0, delta = -h[j-k]
    for k in range(min(j+1, n)):
        delta[k] -= h[j-k]
    return delta
```

```python
    H = np.fft.rfft(h, fsize)

def swap_delta(h, i, j, n):
    """Delta in corr when h[i]:1->0, h[j]:0->1. Returns array of length n."""
    delta = np.zeros(n)
    # i: 1->0
    # As x (first): for k=0..n-1-i, term (1-h[i+k]) -> 0, delta -= (1-h[i+k])
    ni = n - i
    if ni > 0:
        delta[:ni] -= (1.0 - h[i:i+ni])
    # As x+k (second): for k=0..i, term 0 -> h[i-k], delta += h[i-k]
    ip1 = i + 1
    if ip1 > 0:
        delta[:ip1] += h[i::-1]  # h[i], h[i-1], ..., h[0]
    # j: 0->1
    # As x (first): for k=0..n-1-j, term 0 -> (1-h[j+k]), delta += (1-h[j+k])
    nj = n - j
    if nj > 0:
        delta[:nj] += (1.0 - h[j:j+nj])
    # As x+k (second): for k=0..j, term h[j-k] -> 0, delta -= h[j-k]
    jp1 = j + 1
    if jp1 > 0:
        delta[:jp1] -= h[j::-1]  # h[j], h[j-1], ..., h[0]
    return delta
```

delta[i,j] = f(i) + g(j)
where f(i)[k] = -(1-h[i+k]) + h[i-k] (with range checking for i)
and g(j)[k] = (1-h[j+k]) - h[j-k] (with range checking for j)

So I can precompute f(i) for all i and g(j) for all j, then for each (i,j) pair, just add f(i) + g(j) and take the max.

f(i) is a vector of length n for each i, so the total storage is n×n = 160,000 floats. That's fine.

Then the greedy search becomes:
- Precompute F[i] = f(i) for i in ones (200 vectors of length n)
- Precompute G[j] = g(j) for j in zeros (200 vectors of length n)
- For each trial: pick random i from ones, j from zeros, compute max(corr + F[i] + G[j])

But this is still O(n) per trial. The precomputation saves the O(n) swap_delta computation, replacing it with a simple addition.

Actually, let me think about an even faster approach. I want to find the (i,j) pair that minimizes max(corr + F[i] + G[j]).

This is the same as finding (i,j) that minimizes the max over k of (corr[k] + F[i][k] + G[j][k]).

For a fixed i, I want to find j that minimizes max_k (corr[k] + F[i][k] + G[j][k]).

This is still not trivially vectorizable because of the max. But I can try all j for a fixed i:
- For fixed i, compute A[i][k] = corr[k] + F[i][k] for all k (O(n))
- Then for each j, compute max_k (A[i][k] + G[j][k]) (O(n) per j, O(n²) total for all j)

So for a fixed i, finding the best j takes O(n²) = O(160,000). For all i (200 of them), that's O(200 * 160,000) = O(32,000,000). That's about 32M operations, which in numpy should take about 0.1-0.5 seconds.

But wait, I can vectorize this further. For fixed i:
- A = corr + F[i] (shape: n)
- For all j simultaneously: B[j, k] = A[k] + G[j][k] (shape: n_zeros × n)
- new_max[j] = max_k B[j, k] (shape: n_zeros)
- best_j = argmin(new_max)

This is a matrix operation that can be done in one shot:
B = A[None, :] + G  # broadcast: (n_zeros, n)
new_max = B.max(axis=1)  # (n_zeros,)

Time: O(n_zeros × n) = O(80,000). Very fast.

- For each i (200 iterations): O(n_zeros × n) = O(80,000)
- Total: 200 × 80,000 = 16,000,000 operations

In numpy with vectorization, this should be very fast (maybe 50-100ms per full greedy step).

```python
def binary_greedy_full(h, n, deadline, rng):
    """Full greedy: try all possible swaps, pick the best"""
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_score = cur_max * dx
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Precompute F[i] for all i in ones
        # F[i][k] = -(1-h[i+k]) if k < n-i else 0) + (h[i-k] if k <= i else 0)
        F_list = []
        for i in ones:
            fi = np.zeros(n)
            ni = n - i
            if ni > 0:
                fi[:ni] -= (1.0 - h[i:i+ni])
            ip1 = i + 1
            if ip1 > 0:
                fi[:ip1] += h[i::-1]
            F_list.append(fi)
        F = np.array(F_list)  # shape (len(ones), n)
        
        # Precompute G[j] for all j in zeros
        # G[j][k] = (1-h[j+k]) if k < n-j else 0) - (h[j-k] if k <= j else 0)
        G_list = []
        for j in zeros:
            gj = np.zeros(n)
            nj = n - j
            if nj > 0:
                gj[:nj] += (1.0 - h[j:j+nj])
            jp1 = j + 1
            if jp1 > 0:
                gj[:jp1] -= h[j::-1]
            G_list.append(gj)
        G = np.array(G_list)  # shape (len(zeros), n)
        
        # For each i, find best j
        best_new_max = cur_max
        best_ii, best_jj = -1, -1
        
        for ii in range(len(ones)):
            A = corr + F[ii]  # shape (n,)
            B = A[None, :] + G  # shape (len(zeros), n)
            new_maxes = B.max(axis=1)  # shape (len(zeros),)
            min_idx = np.argmin(new_maxes)
            if new_maxes[min_idx] < best_new_max:
                best_new_max = new_maxes[min_idx]
                best_ii = ii
                best_jj = min_idx
        
        if best_ii >= 0 and best_new_max < cur_max - 1e-10:
            i = ones[best_ii]
            j = zeros[best_jj]
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            cur_max = best_new_max
            if cur_max * dx < best_score - 1e-15:
                best_score = cur_max * dx
                best_h = h.copy()
        else:
            # Stuck, perturb
            for _ in range(5):
                ii = rng.randint(len(ones))
                jj = rng.randint(len(zeros))
                i = ones[ii]
                j = zeros[jj]
                h[i], h[j] = h[j], h[i]
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            # Rebuild ones/zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    
    return best_h, best_score
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:].astype(float)

def project_feasible(h, target):
    n = len(h)
    h = h.astype(float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(exp_vals.sum())
    weights = exp_vals / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1]
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    cp = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    cn = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (cp - cn)
    return obj, grad

        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)

            quad = 0.5 / step * np.dot(x_new - y_clip, x_new - y_clip)

            step *= 0.5
            x_new = project_feasible(y_clip - step * grad_y, target)

        t = t_new
```
