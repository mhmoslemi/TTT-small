    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_objective_and_grad(x, alpha, target, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_scaled = corr * dx
    c_max = float(np.max(c_scaled))
    shifted = alpha * (c_scaled - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * dx * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * dx * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * dx * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * dx * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_objective_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_objective_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_max = top_vals.max()

        new_score = compute_c5(h_cur)
        if new_score < cur_score:
            cur_score = new_score
        else:

Instead of Nelder-Mead (which is derivative-free and slow for >5 dimensions), let me use L-BFGS-B with bounds for the block. But L-BFGS-B needs a gradient. I can compute the gradient of the block objective analytically.

The block objective is: max_{k in S} f_k + λ * (sum(h) - target)²

where S is the set of active lags.

The gradient w.r.t. h[j] (for j in [start, end)) of f_k is:
- For lag m: dx * [(1-h[j+m]) if j+m < n + (-h[j-m]) if j-m ≥ 0] (with appropriate i ranges)

So the gradient of the block objective w.r.t. the block variables is the same as the full gradient restricted to the block (for the active lags part), plus 2λ*(sum-target) for each component.

I can compute this efficiently!

def block_grad(x, h_full, start, end, top_idx, target, n_lags_active):

    c = np.correlate(h_new, 1.0 - h_new, mode='full')
    max_c = np.max(c)
    # For the max, the subgradient is the gradient of the argmax lag
    # Use all top lags with equal weight for a smoother approximation
    grad = np.zeros(n)
    for ki in top_idx:
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += dx * (1.0 - h_new[j1 - lag])

                grad[j2] -= dx * h_new[j2 + lag]

            if m > 0:

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_scaled = corr * dx
    c_max = float(np.max(c_scaled))
    shifted = alpha * (c_scaled - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * dx * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * dx * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * dx * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * dx * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lse_obj_wrapper(x, alpha, target, mu):
    return lse_obj_and_grad(x, alpha, target, mu)[0]

def lse_grad_wrapper(x, alpha, target, mu):
    return lse_obj_and_grad(x, alpha, target, mu)[1]

def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj_wrapper, h.copy(), jac=lse_grad_wrapper,
                       args=(alpha, target, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def block_obj_func(x, h_full, start, end, top_idx, target, alpha):
    n = len(h_full)
    dx = 2.0 / n
    h_new = h_full.copy()
    h_new[start:end] = np.clip(x, 0.0, 1.0)
    c = np.correlate(h_new, 1.0 - h_new, mode='full')
    top_c = c[top_idx].astype(float) * dx
    c_max = float(np.max(top_c))
    shifted = alpha * (top_c - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(h_new))
    val += 1e4 * (s - target) ** 2
    return val

def block_cd(h, target, block_size, n_sweeps, deadline, n_top=8):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    alpha = 100.0
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        h_sweep_start = h_cur.copy()
        
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block = h_cur[start:end].copy()
            try:
                res = minimize(block_obj_func, block,
                              args=(h_cur, start, end, top_idx, target, alpha),
                              method='Nelder-Mead',
                              options={'maxiter': 20, 'xatol': 1e-8,
                                       'fatol': 1e-12, 'adaptive': True})
                new_block = np.clip(res.x, 0.0, 1.0)
            except Exception:
                new_block = block
            h_cur[start:end] = new_block
            h_cur = project_feasible(h_cur, target)
        
        new_score = compute_c5(h_cur)
        if new_score < cur_score:
            cur_score = new_score
        else:
            h_cur = h_sweep_start
    
    return h_cur, cur_score

def top_lag_equalize(h, target, n_top=5, n_steps=200, deadline=None):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.005
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * dx * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * dx * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * dx * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * dx * h_cur[j2 - m]
        h_new = project_feasible(h_cur - lr * grad, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.85
            if lr < 1e-9:
                break
    return h_cur, cur_score

def golden_section_1d(h, j, target, n_iter=30):
    n = len(h)
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    def f1d(v):
        h_new = h.copy()
        h_new[j] = v
        h_new = project_feasible(h_new, target)
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        return float(np.max(corr))
    
    fc = f1d(c)
    fd = f1d(d)
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f1d(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f1d(d)
    best_v = (a + b) / 2.0
    return best_v

def coord_descent(h, target, n_sweeps=3, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            best_v = golden_section_1d(h_cur, j, target, n_iter=25)
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score

def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(lse_obj_wrapper, h.copy(), jac=lse_grad_wrapper,
                       args=(alpha, target, 1e8),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_scaled = corr * dx
    c_max = float(np.max(c_scaled))
    shifted = alpha * (c_scaled - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * dx * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * dx * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * dx * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * dx * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lse_obj_wrapper(x, alpha, target, mu):
    return lse_obj_and_grad(x, alpha, target, mu)[0]


def lse_grad_wrapper(x, alpha, target, mu):
    return lse_obj_and_grad(x, alpha, target, mu)[1]


def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj_wrapper, h.copy(), jac=lse_grad_wrapper,
                       args=(alpha, target, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def block_obj_func(x, h_full, start, end, top_idx, target, alpha):
    n = len(h_full)
    dx = 2.0 / n
    h_new = h_full.copy()
    h_new[start:end] = np.clip(x, 0.0, 1.0)
    c = np.correlate(h_new, 1.0 - h_new, mode='full')
    top_c = c[top_idx].astype(float) * dx
    c_max = float(np.max(top_c))
    shifted = alpha * (top_c - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(h_new))
    val += 1e4 * (s - target) ** 2
    return val


def block_cd(h, target, block_size, n_sweeps, deadline, n_top=8):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    alpha = 100.0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        h_sweep_start = h_cur.copy()

        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block = h_cur[start:end].copy()
            try:
                res = minimize(block_obj_func, block,
                              args=(h_cur, start, end, top_idx, target, alpha),
                              method='Nelder-Mead',
                              options={'maxiter': 15, 'xatol': 1e-8,
                                       'fatol': 1e-12, 'adaptive': True})
                new_block = np.clip(res.x, 0.0, 1.0)
            except Exception:
                new_block = block
            h_cur[start:end] = new_block
            h_cur = project_feasible(h_cur, target)

        new_score = compute_c5(h_cur)
        if new_score < cur_score:
            cur_score = new_score
        else:
            h_cur = h_sweep_start

def top_lag_equalize(h, target, n_top, n_steps, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.005

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * dx * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * dx * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * dx * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * dx * h_cur[j2 - m]
        h_new = project_feasible(h_cur - lr * grad, target)

def coord_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            try:
                res = minimize_scalar(f1d_eval, args=(h_cur, j, target),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 30})
                best_v = float(res.x)
            except Exception:
                best_v = h_cur[j]
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score
