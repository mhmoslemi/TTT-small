But projected gradient descent might be slow to converge to high precision. However, it's good for getting close to the optimum, and then we can use a few SLSQP iterations to polish.

Let me implement a hybrid:
1. L-BFGS-B (with penalty for equality constraint) for the bulk of the optimization
2. SLSQP for final polishing

Actually, scipy's L-BFGS-B doesn't handle equality constraints. Let me use a different approach.

**Approach: L-BFGS-B with reparameterization**

Parameterize h as:
h[i] = sigmoid(z[i]) for i = 0, ..., n-2
h[n-1] = n/2 - Σ_{i=0}^{n-2} h[i]

Then optimize over z ∈ R^(n-1) using L-BFGS-B.

But this doesn't respect the bounds easily. The last component h[n-1] might go outside [0,1].

**Alternative: Use the penalty method with L-BFGS-B**

Objective: g(h) + μ * (Σh - n/2)²

Use L-BFGS-B with bounds [0,1]. Start with μ=1, then increase to μ=10, 100, 1000, etc.

This should work well because:
- L-BFGS-B is efficient for large n
- The penalty method handles the equality constraint
- The bounds are naturally handled by L-BFGS-B

Let me implement this.

Actually, I just realized that for the log-sum-exp, I need to be careful about numerical stability. When α is large, exp(α * f_k) can overflow. I should subtract the max:

g(h) = f_max + (1/α) * log(Σ_k exp(α * (f_k - f_max)))

where f_max = max_k f_k.

This is numerically stable.

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    # Water-filling projection
    for _ in range(100):
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def lns_exp_obj(h, alpha, n, target):
    """Log-sum-exp smoothed objective + penalty for sum constraint"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Numerically stable log-sum-exp
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    # Penalty for sum constraint
    sum_diff = np.sum(h) - target
    penalty = 100.0 * sum_diff * sum_diff
    return lse * dx + penalty

def lns_exp_grad(h, alpha, n, target):
    """Gradient of the log-sum-exp objective"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    
    # corr has length 2n-1, index m corresponds to shift k = m - (n-1)
    # f_k = dx * corr[m]
    # ∂f_k/∂h[j] = dx * [(1-h[j+k]) * I(0≤j+k<n) - h[j-k] * I(0≤j-k<n)]
    
    grad = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)
        for j in range(n):
            d = 0.0
            if 0 <= j + k < n:
                d += (1.0 - h[j + k])
            if 0 <= j - k < n:
                d -= h[j - k]
            grad[j] += weights[m] * d
    grad *= dx
    # Penalty gradient
    sum_diff = np.sum(h) - target
    grad += 200.0 * sum_diff
    return grad

def continuous_refine(h, n, target, time_limit, start_time):
    """Refine using L-BFGS-B with penalty method"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return lns_exp_obj(x, 500.0, n, target)
    
    def grad(x):
        return lns_exp_grad(x, 500.0, n, target)
    
    h_cur = h.copy()
    curr_score = compute_c5(h_cur)
    
    alphas = [100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:

        def obj_a(x):
            return lns_exp_obj(x, alpha, n, target)
        
        def grad_a(x):
            return lns_exp_grad(x, alpha, n, target)
        
        try:
            res = minimize(obj_a, h_cur, jac=grad_a, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def lse_obj(h, alpha, n, target, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))

    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    sum_diff = np.sum(h) - target
    return lse * dx + mu * sum_diff * sum_diff

def continuous_refine_lbfgsb(h, n, target, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    curr_score = compute_c5(h_cur)

    stages = [(50, 10.0), (150, 50.0), (500, 200.0), (1500, 1000.0), (5000, 5000.0)]
    for alpha, mu in stages:
        if time.time() - start_time > time_limit:
            break

        def obj_a(x):
            return lse_obj(x, alpha, n, target, mu)

        def grad_a(x):
            return lse_grad(x, alpha, n, target, mu)

        try:
            res = minimize(obj_a, h_cur, jac=grad_a, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def continuous_refine_slsqp(h, n, target, time_limit, start_time):
    global _csl_n, _csl_target
    _csl_n = n
    _csl_target = float(target)

    def _csl_obj(x):
        return lse_obj(x, 3000.0, n, float(target), 0.0)

    def _csl_cons_fun(x):
        return np.sum(x) - float(target)

    def _csl_cons_jac(x):
        return np.ones(_csl_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csl_cons_fun, 'jac': _csl_cons_jac}]

    h_cur = h.copy()
    curr_score = compute_c5(h_cur)

    try:
        res = minimize(_csl_obj, h_cur, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < curr_score:
            curr_score = sc
            h_cur = h_new
    except Exception:
        pass

    return h_cur, curr_score


_csl_n = 0
_csl_target = 0.0


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h
_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy from initial
    allowed = min(40, max(5, tleft() * 0.06))
    h_g, c5_g = greedy_optimize(best_h, max_iters=50000, time_limit=allowed,
                                start_time=start_time, scan_width=30)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA round 1
    if tleft() > 50:
        allowed = min(250, max(20, tleft() * 0.25))
        max_it = max(5000, int(allowed * 1000))
        h_sa, c5_sa = sa_optimize(best_h, target_init, max_it, 0.015, 0.00002, rng,
                                  allowed, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: SA round 2 (different params, focused)
    if tleft() > 80:
        allowed = min(200, max(20, tleft() * 0.22))
        max_it = max(5000, int(allowed * 1000))
        h_sa2, c5_sa2 = sa_optimize(best_h, target_init, max_it, 0.007, 0.000003, rng,
                                    allowed, start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: ILS perturbations
    if tleft() > 120:
        for pert_iter in range(4):
            if tleft() < 50:
                break
            n_pert = 3 + pert_iter * 2
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(20, max(5, tleft() * 0.04))
            h_pg, c5_pg = greedy_optimize(h_pert, max_iters=15000, time_limit=allowed,
                                          start_time=start_time, scan_width=20)
            if tleft() < 30:
                if c5_pg < best_score:
                    best_score = c5_pg
                    best_h = h_pg.copy()
                break

            allowed2 = min(60, max(10, tleft() * 0.07))
            max_it2 = max(3000, int(allowed2 * 1000))
            h_ps, c5_ps = sa_optimize(h_pg, target_init, max_it2, 0.01, 0.00003, rng,
                                      allowed2, start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 5: Try larger n values
    n_opts = [n_init + 10, n_init + 20, n_init + 40, n_init + 60]
    for n_try in n_opts:
        if tleft() < 70:
            break
        if n_try % 2 != 0:
            n_try += 1
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(15, max(3, tleft() * 0.03))
        h_g2, c5_g2 = greedy_optimize(h_bin, max_iters=10000, time_limit=allowed,
                                      start_time=start_time, scan_width=20)
        if tleft() < 40:
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()
            continue

        allowed2 = min(60, max(10, tleft() * 0.09))
        max_it2 = max(3000, int(allowed2 * 1000))
        h_s2, c5_s2 = sa_optimize(h_g2, target_try, max_it2, 0.012, 0.00005, rng,
                                  allowed2, start_time)
        if c5_s2 < best_score:
            best_score = c5_s2
            best_h = h_s2.copy()

    # Phase 6: Continuous L-BFGS-B refinement (key improvement)
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(250, max(30, tleft() * 0.30))
        h_cont, sc_cont = continuous_refine_lbfgsb(best_h, n_use, target_use,
                                                   allowed, start_time)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()

    # Phase 7: SLSQP polish on continuous solution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(60, max(10, tleft() * 0.12))
        h_slq, sc_slq = continuous_refine_slsqp(best_h, n_use, target_use,
                                                allowed, start_time)
        if sc_slq < best_score:
            best_score = sc_slq
            best_h = h_slq.copy()

    # Phase 8: Binary SA from best (in case binary is better than continuous)
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        sc_bin = compute_c5(h_bin_f)
        allowed = min(80, max(10, tleft() * 0.15))
        max_it = max(3000, int(allowed * 1000))
        h_sf, c5_sf = sa_optimize(h_bin_f, target_use, max_it, 0.005, 0.000008, rng,
                                  allowed, start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Phase 9: Final L-BFGS-B polish on the overall best
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = max(5.0, min(50, tleft() * 0.15))
        h_final_cont, sc_final_cont = continuous_refine_lbfgsb(
            project_feasible(best_h, target_use), n_use, target_use,
            allowed, start_time)
        if sc_final_cont < best_score:
            best_score = sc_final_cont
            best_h = h_final_cont.copy()

    # Final: ensure feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
