```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(h):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    N = 1 << (2 * n - 1).bit_length()
    v = 1.0 - h
    r = np.fft.irfft(np.fft.rfft(h, N) * np.conj(np.fft.rfft(v, N)), N)
    return np.roll(r, n - 1)[:2 * n - 1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h)
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, alpha, target, mu):
    """Log-sum-exp smooth approximation of max(corr) + penalty on sum."""
    corr = fft_correlate(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    s = np.sum(h)
    return lse + mu * (s - target) ** 2

def lse_gradient(h, alpha, target, mu):
    """Gradient of LSE objective."""
    n = len(h)
    corr = fft_correlate(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    
    # Compute gradient using FFT-based approach
    # grad[j] = sum_k w_k * [(1-h[j+lag_k])*I(valid) - h[j-lag_k]*I(valid)]
    # = C[j] - (B * W_neg)[j] - (A * W)[j]
    # where C[j] = sum_k w_k * I(0<=j+lag_k<n)
    
    # Construct W vector (padded to length 2n-1, indexed by lag from -(n-1) to n-1)
    # W[lag + (n-1)] = weights[lag + (n-1)] for lag in [-(n-1), n-1]
    # So W = weights (already in the right order for np.correlate full mode)
    # lag_k = k - (n-1) for k = 0, ..., 2n-2
    
    N = 1 << (2 * n - 1).bit_length()
    
    # W as circular vector of length N
    W_circ = np.zeros(N)
    # W[k] in correlate indexing corresponds to lag = k-(n-1)
    # In circular indexing with N points, lag m is at position m mod N
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        W_circ[lag % N] = weights[k]
    
    # W_neg[k] = W[-k]
    W_neg_circ = np.zeros(N)
    for k in range(N):
        W_neg_circ[k] = W_circ[(-k) % N]
    
    # A = h padded, B = (1-h) padded
    A = np.zeros(N)
    A[:n] = h
    B = np.zeros(N)
    B[:n] = 1.0 - h
    
    # (B * W_neg) via FFT
    BWneg = np.fft.irfft(np.fft.rfft(B) * np.fft.rfft(W_neg_circ), N)
    # (A * W) via FFT  
    AW = np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(W_circ), N)
    
    # C[j] = sum_lag W[lag] * I(0 <= j+lag < n)
    # C[j] = sum_k weights[k] * I(0 <= j + k - (n-1) < n)
    # = sum_k weights[k] * I(n-1-j <= k <= n-1-j+n-1)
    # = sum_k weights[k] * I(max(0, n-1-j) <= k <= min(2n-2, 2n-2-j))
    # For j in [0, n-1]:
    #   lower = max(0, n-1-j)
    #   upper = min(2n-2, 2n-2-j) = 2n-2-j (since j>=0)
    #   But also upper <= 2n-2 always
    #   And lower >= 0
    # So C[j] = sum_{k=lower}^{upper} weights[k]
    
    # Compute C via cumulative sum
    cumsum = np.zeros(2 * n)
    cumsum[1:] = np.cumsum(weights)
    C = np.zeros(n)
    for j in range(n):
        lower = max(0, n - 1 - j)
        upper = min(2 * n - 2, 2 * n - 2 - j)
        C[j] = cumsum[upper + 1] - cumsum[lower]
    
    grad = C[:n] - BWneg[:n] - AW[:n]
    s = np.sum(h)
    grad += 2.0 * mu * (s - target)
    return grad
```
