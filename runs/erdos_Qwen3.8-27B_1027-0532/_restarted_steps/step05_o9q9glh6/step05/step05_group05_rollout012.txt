def active_lse_obj(h_full):
    """Active-set log-sum-exp objective."""
    corr = compute_corr(h_full)
    dx = 2.0 / len(h_full)
    # Get the top m lags
    m = _g_top_m
    top_idx = np.argsort(corr)[::-1][:m]
    top_corr = corr[top_idx]
    c_max = float(np.max(top_corr))
    alpha = _g_alpha
    shifted = alpha * (top_corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    return lse * dx

```python
def active_subgradient(h, n, target, n_steps, top_m, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    dx = 2.0 / n
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr <= best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Get top m lags
        top_idx = np.argsort(corr)[::-1][:top_m]
        
        # Compute weights (softmax)
        top_vals = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(top_vals / 0.005)
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for w, m in zip(weights, top_idx):
            k = int(m - (n - 1))  # convert to lag
            if k > 0:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += w * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= w * h_cur[j_neg - k]
            elif k < 0:
                ak = -k
                j_pos = np.arange(ak, n)
                grad[j_pos] += w * (1.0 - h_cur[j_pos - ak])
                j_neg = np.arange(0, n - ak)
                grad[j_neg] -= w * h_cur[j_neg + ak]
            else:
                grad += w * (1.0 - 2.0 * h_cur)
        
        grad *= dx
        
        # Line search
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    
    return best_h, best_score
```

def periodic_objective(h0, T, n):
    h = np.tile(h0, n // T)
    corr = compute_corr(h)
    return float(np.max(corr)) * (2.0 / n)

def optimize_periodic(h0_init, T, n, target_T, alphas, maxiter, deadline):
    """Optimize h0 for periodic structure."""
    global _g_alpha
    R = n // T
    
    def obj(h0):
        h = np.tile(h0, R)[:n]
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        alpha = _g_alpha
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        return lse * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_T, 
             'jac': lambda x: np.ones(T)}]
    
    h0_cur = h0_init.copy()
    cur_score = periodic_objective(h0_cur, T, n)
    
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(obj, h0_cur, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h0_new = project_feasible(res.x, target_T)
            sc = periodic_objective(h0_new, T, n)
            if sc < cur_score:
                h0_cur = h0_new
                cur_score = sc
        except Exception:
            pass
    return h0_cur
