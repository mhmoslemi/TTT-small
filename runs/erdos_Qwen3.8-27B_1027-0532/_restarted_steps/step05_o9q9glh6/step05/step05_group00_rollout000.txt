                break
        if stall > max_stall:
            if reheat_count < 5:
                t = t_init * 0.5
                stall = 0
                reheat_count += 1
            else:
                break

        ratio = min(it / max(max_iters - 1, 1), 1.0)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.50:
            worst = np.argsort(corr)[-top_m:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n or 0 <= i - k < n]
            jp = [j for j in zeros if 0 <= j + k < n or 0 <= j - k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
        elif r < 0.80:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True

        if not moved:
            continue

        corr_new = full_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-14 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = full_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def iln_search(h, n, target, rng, time_limit, start_time):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    pert_size = 2
    no_improve = 0
    max_no_improve = 8

    while time.time() - start_time < time_limit:
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        n_pert = min(pert_size, len(ones), len(zeros))
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        corr_pert = full_corr(h_pert)
        rem_greedy = min(60, (time_limit - (time.time() - start_time)) * 0.3)
        if rem_greedy > 5:
            h_g, c5_g, corr_g = greedy_lag_aware(h_pert, corr_pert, n, max_iters=8000,
                                                  time_limit=rem_greedy, start_time=start_time,
                                                  top_m=15, scan_width=25)
        else:
            c5_g = compute_c5(h_pert)
            h_g = h_pert
            corr_g = full_corr(h_pert)

        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            no_improve = 0
            if pert_size > 1:
                pert_size = max(1, pert_size - 1)
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                pert_size = min(pert_size + 2, 15)
                no_improve = 0
            else:
                pert_size = min(pert_size + 1, 12)

        rem_sa = min(30, (time_limit - (time.time() - start_time)) * 0.2)
        if rem_sa > 10:
            h_s, c5_s = sa_with_2swap(h_g, target, max_iters=50000,
                                       t_init=0.008, t_final=0.00003, rng=rng,
                                       time_limit=rem_sa, start_time=start_time, top_m=10)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                no_improve = 0
                if pert_size > 1:
                    pert_size = max(1, pert_size - 1)

    return best_h, best_c5


def thue_morse_init(n, rng):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    target = n // 2
    current = int(tm.sum())
    if current > target:
        excess = current - target
        ones = np.where(tm == 1)[0]
        to_flip = ones[:excess]
        tm[to_flip] = 0
    else:
        deficit = target - current
        zeros = np.where(tm == 0)[0]
        to_flip = zeros[:deficit]
        tm[to_flip] = 1
    return tm.astype(float)


def random_init(n, rng, seed_offset=0):
    target = n // 2
    h = (rng.random(n) < 0.5).astype(float)
    current = int(h.sum())
    if current > target:
        excess = current - target
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:excess]] = 0
    else:
        deficit = target - current
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:deficit]] = 1
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy from parent (quick)
    corr_init = full_corr(best_h)
    rem = tleft()
    if rem > 30:
        allowed = min(40, rem * 0.10)
        h_g, c5_g, corr_g = greedy_lag_aware(best_h, corr_init, n_init,
                                              max_iters=20000, time_limit=allowed,
                                              start_time=t0, top_m=15, scan_width=25)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()
            corr_init = corr_g

    # Phase 2: SA refinement
    rem = tleft()
    if rem > 60:
        allowed = min(180, rem * 0.25)
        max_it = int(min(400000, allowed * 1200))
        h_sa, c5_sa = sa_with_2swap(best_h, target_init, max_it, 0.01, 0.00002, rng,
                                    time_limit=allowed, start_time=t0, top_m=10)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Second SA with different params
    rem = tleft()
    if rem > 80:
        allowed = min(120, rem * 0.15)
        max_it = int(min(250000, allowed * 1200))
        h_sa2, c5_sa2 = sa_with_2swap(best_h, target_init, max_it, 0.005, 0.00001, rng,
                                      time_limit=allowed, start_time=t0, top_m=12)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: ILS (iterated local neighborhood search)
    rem = tleft()
    if rem > 120:
        allowed = min(300, rem * 0.30)
        h_il, c5_il = iln_search(best_h, n_init, target_init, rng, allowed, t0)
        if c5_il < best_score:
            best_score = c5_il
            best_h = h_il.copy()

    # Phase 5: Thue-Morse initialization + optimization
    rem = tleft()
    if rem > 150:
        h_tm = thue_morse_init(n_init, rng)
        corr_tm = full_corr(h_tm)
        allowed = min(40, rem * 0.08)
        h_tm_g, c5_tm_g, _ = greedy_lag_aware(h_tm, corr_tm, n_init,
                                               max_iters=15000, time_limit=allowed,
                                               start_time=t0, top_m=15, scan_width=20)
        if c5_tm_g < best_score:
            best_score = c5_tm_g
            best_h = h_tm_g.copy()
        else:
            rem2 = tleft()
            if rem2 > 60:
                allowed2 = min(60, rem2 * 0.10)
                max_it2 = int(min(100000, allowed2 * 1500))
                h_tm_sa, c5_tm_sa = sa_with_2swap(h_tm_g, target_init, max_it2,
                                                   0.008, 0.00003, rng,
                                                   time_limit=allowed2, start_time=t0, top_m=10)
                if c5_tm_sa < best_score:
                    best_score = c5_tm_sa
                    best_h = h_tm_sa.copy()

    # Phase 6: Random initializations
    rem = tleft()
    for rseed in range(3):
        if rem < 80:
            break
        h_rand = random_init(n_init, rng, seed_offset=rseed * 1000)
        corr_rand = full_corr(h_rand)
        allowed = min(25, rem * 0.05)
        h_r_g, c5_r_g, _ = greedy_lag_aware(h_rand, corr_rand, n_init,
                                             max_iters=10000, time_limit=allowed,
                                             start_time=t0, top_m=15, scan_width=20)
        if c5_r_g < best_score:
            best_score = c5_r_g
            best_h = h_r_g.copy()
            rem = tleft()
            if rem > 40:
                allowed2 = min(40, rem * 0.06)
                max_it3 = int(min(60000, allowed2 * 1500))
                h_r_sa, c5_r_sa = sa_with_2swap(h_r_g, target_init, max_it3,
                                                 0.008, 0.00003, rng,
                                                 time_limit=allowed2, start_time=t0, top_m=10)
                if c5_r_sa < best_score:
                    best_score = c5_r_sa
                    best_h = h_r_sa.copy()
        rem = tleft()

    # Phase 7: Try different n values
    rem = tleft()
    n_options = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 16, 20, 24, 30, 40, 50]:
        nn = n_init + dn
        if nn >= 200 and nn % 2 == 0 and nn != n_init:
            n_options.append(nn)
    n_options = sorted(n_options, key=lambda x: abs(x - n_init))

    for n_try in n_options[:6]:
        if rem < 50:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)
        corr_try = full_corr(h_bin)
        allowed = min(20, rem * 0.06)
        h_ng, c5_ng, corr_ng = greedy_lag_aware(h_bin, corr_try, n_try,
                                                 max_iters=10000, time_limit=allowed,
                                                 start_time=t0, top_m=15, scan_width=20)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng.copy()
        rem = tleft()
        if c5_ng < best_score + 0.001 and rem > 30:
            allowed2 = min(25, rem * 0.05)
            max_it4 = int(min(50000, allowed2 * 1500))
            h_ns, c5_ns = sa_with_2swap(h_ng, target_try, max_it4,
                                         0.006, 0.00002, rng,
                                         time_limit=allowed2, start_time=t0, top_m=10)
            if c5_ns < best_score:
                best_score = c5_ns
                best_h = h_ns.copy()
        rem = tleft()

    # Phase 8: SLSQP continuous refinement
    rem = tleft()
    if rem > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 600, 1500, 4000, 10000, 25000]
        max_iter_sl = 3000
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, max_iter_sl, 1e-15,
                                  time_limit=min(rem - 10, 200), start_time=t0)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Binary projection + final SA
    rem = tleft()
    if rem > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(50, rem - 5)
        max_it5 = int(min(150000, allowed * 1800))
        h_sf, c5_sf = sa_with_2swap(h_bin_f, target_use, max_it5, 0.004, 0.00001, rng,
                                     time_limit=allowed, start_time=t0, top_m=12)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 10: Final SLSQP polish
    rem = tleft()
    if rem > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [8000, 20000, 50000, 100000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 2000, 1e-16,
                                    time_limit=max(5.0, rem - 5), start_time=t0)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
