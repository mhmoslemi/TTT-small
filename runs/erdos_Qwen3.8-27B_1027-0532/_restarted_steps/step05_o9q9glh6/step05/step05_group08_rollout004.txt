def active_slsqp(h, n, target, alpha, active_set, maxiter, ftol):
    """SLSQP on active set."""
    if len(active_set) < 1:
        active_set = np.array([np.argmax(compute_corr_fast(h))])
    
    def obj(x):
        val, _ = lse_value_and_grad(x, alpha, active_set)
        return val
    
    def objg(x):
        _, g = lse_value_and_grad(x, alpha, active_set)
        return g
    
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def active_lbfgsb(h, n, target, alpha, active_set, mu, maxiter):
    """L-BFGS-B with active set and penalty for constraint."""
    if len(active_set) < 1:
        active_set = np.array([np.argmax(compute_corr_fast(h))])
    
    def obj(x):
        val, _ = lse_value_and_grad(x, alpha, active_set)
        return val + mu * (np.sum(x) - target) ** 2
    
    def objg(x):
        _, g = lse_value_and_grad(x, alpha, active_set)
        return g + 2.0 * mu * (np.sum(x) - target)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

def update_active_set(h, n, current_active, threshold_frac, min_count, max_count):
    """Update active set based on current correlations."""
    corr = compute_corr_fast(h)
    c_max = float(np.max(corr))
    threshold = c_max * threshold_frac
    active = np.where(corr >= threshold)[0]
    
    # Ensure at least min_count and at most max_count
    if len(active) < min_count:
        top = np.argsort(corr)[::-1][:min_count]
        active = np.unique(np.concatenate([active, top]))
    if len(active) > max_count:
        active = np.argsort(corr[active])[::-1][:max_count] + active[np.argsort(corr[active])[::-1][:max_count]]
        # Simpler: just take top max_count
        all_sorted = np.argsort(corr)[::-1][:max_count]
        active = all_sorted
    
    return active

def subgradient_step(h, n, target, corr, weights, lr, temp):
    """Single subgradient step."""
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(weights, range(len(corr))):
        if wk < 1e-13:
            continue
        k = int(m) - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    g *= dx
    h_new = h - lr * g
    return project_feasible(h_new, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
