import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    # Bisection on the watermark
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Backtracking
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink) - 1e-12:
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step / (1.0 - shrink), lr_init * 3.0)
        step = max(step, 1e-12)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def best_shift_descent(h, target, deadline, max_rounds=5000):
    """Iteratively focus on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        overlaps = get_all_overlaps(h_cur)
        k_star = int(np.argmax(overlaps))
        k_star = min(k_star, n - 1)
        
        # Gradient of shift k_star's overlap
        # f_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
        # df_k/dh[i] = (1-h[i+k]) for i < n-k (from first term)
        #              - h[i-k] for i >= k (from second term where h[i] appears as h[j+k])
        grad = np.zeros(n)
        jp = np.arange(0, n - k_star)
        grad[jp] += (1.0 - h_cur[jp + k_star])
        jn = np.arange(k_star, n)
        grad[jn] -= h_cur[jn - k_star]
        grad *= dx
        
        # Take a step
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005]:
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                break
    
    return h_cur, cur_score

def binary_local_search(h, target, deadline):
    """Quantize to binary and do local search by flipping bits."""
    n = len(h)
    dx = 2.0 / n
    # Quantize: take top n/2 values as 1
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    cur_score = compute_c5(h_bin)
    
    # Local search: try flipping pairs (one 1->0, one 0->1)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for iteration in range(200):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle for diversity
        rng = np.random.RandomState(iteration + 42)
        ones_shuffled = ones[rng.permutation(len(ones))]
        zeros_shuffled = zeros[rng.permutation(len(zeros))]
        
        for i in range(min(len(ones_shuffled), 50)):
            for j in range(min(len(zeros_shuffled), 50)):
                oi = ones_shuffled[i]
                zo = zeros_shuffled[j]
                if oi == zo:
                    continue
                h_bin[oi] = 0.0
                h_bin[zo] = 1.0
                sc = compute_c5(h_bin)
                if sc < cur_score - 1e-12:
                    cur_score = sc
                    # Update lists
                    ones = np.where(h_bin == 1.0)[0]
                    zeros = np.where(h_bin == 0.0)[0]
                    improved = True
                    break
                else:
                    h_bin[oi] = 1.0
                    h_bin[zo] = 0.0
            if improved:
                break
        if not improved:
            break
    
    return h_bin, cur_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)

    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

        if time.time() > deadline:
            break
        overlaps = get_all_overlaps(h_cur)
        k_star = int(np.argmax(overlaps))
        k_star = min(k_star, n - 1)
        if k_star == 0:
            k_star = 1  # skip shift 0 as it's always 0 for binary
        
        grad = np.zeros(n)
        jp = np.arange(0, n - k_star)
        grad[jp] += (1.0 - h_cur[jp + k_star])
        jn = np.arange(k_star, n)
        grad[jn] -= h_cur[jn - k_star]
        grad *= dx

        x_prev = x
        x = x_new
        t_val = t_new
        sc = compute_c5(x)

            best_x = x.copy()

def top_k_focus(h, target, deadline, k_focus=3):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(10000):
        if time.time() > deadline:
            break
        overlaps = get_all_overlaps(h_cur)
        top_k = np.argsort(overlaps)[::-1][:k_focus]
        k_max = overlaps[top_k[0]]
        if k_max < 1e-10:
            break
        diffs = overlaps[top_k] - k_max
        w = np.exp(diffs / 0.01)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, top_k):
            ki = int(ki)
            if ki >= n:
                continue
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        for lr in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

def shifted_gradient_descent(h, target, deadline):
    """Momentum-based descent on rolling top shifts."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    momentum = np.zeros(n)
    
    window = []
    window_size = 5
    
    for rd in range(10000):
        if time.time() > deadline:
            break
        overlaps = get_all_overlaps(h_cur)
        top_k = np.argsort(overlaps)[::-1][:3]
        window.append(overlaps[top_k].copy())
        if len(window) > window_size:
            window.pop(0)
        
        # Average gradient over recent worst shifts
        grad = np.zeros(n)
        for overlaps_snap in window:
            for ki in np.argsort(overlaps_snap)[::-1][:2]:
                ki = int(ki)
                if ki >= n or ki == 0:
                    continue
                jp = np.arange(0, n - ki)
                grad[jp] += (1.0 - h_cur[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= h_cur[jn - ki]
        grad *= dx / (window_size * 2)
        
        # Momentum
        momentum = 0.9 * momentum + 0.1 * grad
        
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005]:
            h_new = h_cur - lr * momentum
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo = -1.0
    hi = 2.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t_val = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x + ((t_val - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(25):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink) - 1e-12:
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 3.0)
        step = max(step, 1e-12)
        x_prev = x
        x = x_new
        t_val = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def top_k_focus(h, target, deadline, k_focus=3):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(20000):
        if time.time() > deadline:
            break
        overlaps = get_all_overlaps(h_cur)
        top_k = np.argsort(overlaps)[::-1][:k_focus]
        k_max = overlaps[top_k[0]]
        if k_max < 1e-10:
            break
        diffs = overlaps[top_k] - k_max
        temp = max(0.005, np.std(diffs) * 0.5)
        w = np.exp(diffs / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_k):
            ki = int(ki)
            if ki >= n or ki == 0:
                continue
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        for lr in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:

    return h_cur, cur_score

def shifted_grad_descent(h, target, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    momentum = np.zeros(n)
    window = []
    window_size = 5
    for rd in range(20000):
        if time.time() > deadline:
            break
        overlaps = get_all_overlaps(h_cur)
        top_k = np.argsort(overlaps)[::-1][:4]
        window.append(top_k.copy())
        if len(window) > window_size:
            window.pop(0)
        grad = np.zeros(n)
        count = 0
        for snap in window:
            for ki in snap[:2]:
                ki = int(ki)
                if ki >= n or ki == 0:
                    continue
                jp = np.arange(0, n - ki)
                grad[jp] += (1.0 - h_cur[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= h_cur[jn - ki]
                count += 1
        if count > 0:
            grad *= dx / count
        momentum = 0.85 * momentum + 0.15 * grad
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            h_new = h_cur - lr * momentum
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

def binary_refine(h, target, deadline):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    cur_score = compute_c5(h_bin)
    rng = np.random.RandomState(777)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for sweep in range(80):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(400):
            if time.time() > deadline:
                break
            oi = ones[rng.randint(len(ones))]
            zo = zeros[rng.randint(len(zeros))]
            if oi == zo:
                continue
            h_bin[oi] = 0.0
            h_bin[zo] = 1.0
            sc = compute_c5(h_bin)
            if sc < cur_score - 1e-12:
                cur_score = sc
                ones = np.where(h_bin == 1.0)[0]
                zeros = np.where(h_bin == 0.0)[0]
                improved = True
            else:
                h_bin[oi] = 1.0
                h_bin[zo] = 0.0
        if not improved:
            break
    if time.time() < deadline:
        remaining = deadline - time.time()
        h_polished, sc_polished = fista_optimize(h_bin, target, 5000, 15000, time.time() + remaining * 0.7)
        if sc_polished < cur_score:
            h_bin = h_polished
            cur_score = sc_polished
    return h_bin, cur_score

def coarse_to_fine(h_fine, target_fine, deadline):
    n_fine = len(h_fine)
    best_h = h_fine.copy()
    best_score = compute_c5(best_h)
    for ratio in [4, 8]:
        if time.time() > deadline:
            break
        n_coarse = n_fine // ratio
        if n_coarse < 50:
            continue
        target_coarse = n_coarse / 2.0
        x_fine = np.linspace(0, 1, n_fine)
        x_coarse = np.linspace(0, 1, n_coarse)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        h_coarse = project_feasible(h_coarse, target_coarse)
        remaining = deadline - time.time()
        td1 = remaining * 0.3
        td2 = remaining * 0.3
        h_coarse, _ = fista_optimize(h_coarse, target_coarse, 5000, 15000, time.time() + td1)
        h_coarse, _ = top_k_focus(h_coarse, target_coarse, time.time() + td2, k_focus=3)
        h_up = np.interp(x_fine, x_coarse, h_coarse)
        h_up = project_feasible(h_up, target_fine)
        sc_up = compute_c5(h_up)
        if time.time() < deadline:
            h_up, sc_up = fista_optimize(h_up, target_fine, 10000, 10000, deadline)
        if sc_up < best_score:
            best_h = h_up
            best_score = sc_up
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline_final = start_time + budget_s - 20.0
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA continuation
    for alpha, td in [(200, 40), (1000, 40), (5000, 50), (20000, 60)]:
        remaining = deadline_final - time.time()
        if remaining < td + 5:
            td = max(remaining - 5, 10)
        if td < 10:
            break
        h_new, sc_new = fista_optimize(best_h, target, alpha, 50000, time.time() + td)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 2: Top-k focus
    remaining = deadline_final - time.time()
    if remaining > 40:
        h_new, sc_new = top_k_focus(best_h, target, time.time() + 35, k_focus=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: Binary refine
    remaining = deadline_final - time.time()
    if remaining > 70:
        h_bin, sc_bin = binary_refine(best_h, target, time.time() + 60)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 4: Alternating optimization loop
    methods = [
        ('fista', 10000, 35),
        ('topk', 3, 30),
        ('fista', 50000, 35),
        ('shifted', 0, 30),
        ('fista', 200000, 35),
        ('topk', 5, 25),
    ]
    idx = 0
    while deadline_final - time.time() > 50:
        method, param, td = methods[idx % len(methods)]
        remaining = deadline_final - time.time()
        if remaining < td + 5:
            td = max(remaining - 5, 10)
        if td < 10:
            break
        if method == 'fista':
            h_new, sc_new = fista_optimize(best_h, target, param, 50000, time.time() + td)
        elif method == 'topk':
            h_new, sc_new = top_k_focus(best_h, target, time.time() + td, k_focus=param)
        elif method == 'shifted':
            h_new, sc_new = shifted_grad_descent(best_h, target, time.time() + td)
        else:
            idx += 1
            continue
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        idx += 1

    # Phase 5: Coarse-to-fine
    remaining = deadline_final - time.time()
    if remaining > 80:
        h_c2f, sc_c2f = coarse_to_fine(best_h, target, time.time() + 70)
        if sc_c2f < best_score:
            best_score = sc_c2f
            best_h = h_c2f

    # Phase 6: Perturbation restarts
    remaining = deadline_final - time.time()
    if remaining > 60:
        rng = np.random.RandomState(seed)
        for s in [0.03, 0.06, 0.1, 0.05]:
            if deadline_final - time.time() < 25:
                break
            h_pert = best_h + rng.randn(n) * s
            h_pert = project_feasible(h_pert, target)
            h_new, sc_new = fista_optimize(h_pert, target, 10000, 20000, time.time() + 18)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            if deadline_final - time.time() > 15:
                h_new2, sc_new2 = top_k_focus(best_h, target, time.time() + 10, k_focus=3)
                if sc_new2 < best_score:
                    best_score = sc_new2
                    best_h = h_new2

    # Phase 7: Final intensive polish
    remaining = deadline_final - time.time()
    if remaining > 30:
        h_new, sc_new = top_k_focus(best_h, target, time.time() + 20, k_focus=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    remaining = deadline_final - time.time()
    if remaining > 20:
        h_new, sc_new = fista_optimize(best_h, target, 500000, 30000, time.time() + max(remaining - 5, 10))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    remaining = deadline_final - time.time()
    if remaining > 15:
        h_new, sc_new = top_k_focus(best_h, target, time.time() + max(remaining - 5, 5), k_focus=5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
