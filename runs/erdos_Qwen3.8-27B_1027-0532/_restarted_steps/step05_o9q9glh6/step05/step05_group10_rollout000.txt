import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def alternating_lp(h_init, n, target, max_iter, time_limit,
                   damping_init=0.85, reg_init=1e-4, n_active=50,
                   margin_factor=1.5):

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg_init
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    h_prev = h.copy()

        if time.time() - start > time_limit:

        g = 1.0 - h

        t_var = cp.Variable()

            cp.sum(h_var) == target,
            h_var >= 0.0,

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-16:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stagnation = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coordinate_descent(h, n, target, max_passes, time_limit):
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h_cur[i]
            for new_val in [0.0, 1.0, max(0, old_val - 0.1), min(1, old_val + 0.1),
                            max(0, old_val - 0.3), min(1, old_val + 0.3)]:
                if abs(new_val - old_val) < 1e-10:
                    continue
                h_try = h_cur.copy()
                h_try[i] = new_val
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = h_try.copy()
                    h_cur = h_try
                    improved = True
                    break
        if not improved and p > 2:
            break

    return best_h, best_c5


def multi_shift_subgradient(h, n, target, max_steps, time_limit, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stagnation = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(corr)[-n_shifts:]
        grad = np.zeros(n)
        for k_idx in top_k:
            k = int(k_idx) - half_n
            idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
            idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
            grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
            grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        grad_norm = grad / norm

        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stagnation = 0
                break

        if not improved:
            stagnation += 1
            if stagnation > 100:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_h = project_feasible(best_h, target_init)
    best_score = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Quick LP at original resolution
    if tleft() > 250:
        lp_time = min(250, tleft() * 0.25)
        h_lp, c5_lp = alternating_lp(best_h, n_best, target_init,
                                     max_iter=15000, time_limit=lp_time,
                                     damping_init=0.85, reg_init=1e-4,
                                     n_active=50, margin_factor=1.5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Interpolate to higher resolution
    n_upgrade = 640
    if n_upgrade % 2 == 0:
        target_upgrade = n_upgrade / 2.0
        h_up = interpolate_to(best_h, n_upgrade, target_upgrade)
        best_h = h_up
        n_best = n_upgrade
        best_score = compute_c5(best_h)

    # Phase 3: LP at upgraded resolution
    if tleft() > 300:
        lp_time = min(400, tleft() * 0.35)
        h_lp, c5_lp = alternating_lp(best_h, n_best, n_best / 2.0,
                                     max_iter=15000, time_limit=lp_time,
                                     damping_init=0.85, reg_init=1e-4,
                                     n_active=60, margin_factor=1.4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP at upgraded resolution
    if tleft() > 150:
        h_ref = slsqp_round(best_h, n_best, n_best / 2.0,
                            [100, 500, 2000, 10000], 1500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Multi-shift subgradient
    if tleft() > 100:
        h_sg, c5_sg = multi_shift_subgradient(best_h, n_best, n_best / 2.0,
                                              max_steps=5000, time_limit=min(80, tleft() * 0.4),
                                              n_shifts=5)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: LP polish at upgraded resolution
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.25)
        h_lp, c5_lp = alternating_lp(best_h, n_best, n_best / 2.0,
                                     max_iter=15000, time_limit=lp_time,
                                     damping_init=0.92, reg_init=1e-5,
                                     n_active=80, margin_factor=1.2)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: High-alpha SLSQP
    if tleft() > 120:
        h_ref = slsqp_round(best_h, n_best, n_best / 2.0,
                            [5000, 20000, 80000], 1200, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Single-shift subgradient (very fine)
    if tleft() > 80:
        h_sg, c5_sg = subgradient_improve(best_h, n_best, n_best / 2.0,
                                          max_steps=10000, time_limit=min(60, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 9: Coordinate descent
    if tleft() > 100:
        h_cd, c5_cd = coordinate_descent(best_h, n_best, n_best / 2.0,
                                         max_passes=20, time_limit=min(80, tleft() * 0.5))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Try even higher resolution
    if tleft() > 300:
        n_hires = 800
        if n_hires % 2 == 0:
            target_hires = n_hires / 2.0
            h_hi = interpolate_to(best_h, n_hires, target_hires)
            lp_time = min(200, tleft() * 0.4)
            h_lp, c5_lp = alternating_lp(h_hi, n_hires, target_hires,
                                         max_iter=10000, time_limit=lp_time,
                                         damping_init=0.85, reg_init=1e-4,
                                         n_active=60, margin_factor=1.4)
            if tleft() > 100:
                h_ref = slsqp_round(h_lp, n_hires, target_hires,
                                    [200, 1000, 5000], 1000, 1e-14)
                sc_hi = compute_c5(h_ref)
                if sc_hi < c5_lp:
                    h_lp = h_ref
                    c5_lp = sc_hi
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n_best = n_hires

    # Phase 11: Final LP polish
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(60, tleft() * 0.6)
        h_lp, c5_lp = alternating_lp(best_h, n_use, target_use,
                                     max_iter=10000, time_limit=lp_time,
                                     damping_init=0.95, reg_init=1e-6,
                                     n_active=80, margin_factor=1.1)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = slsqp_round(best_h, n_use, target_use,
                                  [10000, 50000, 200000], 600, 1e-16)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 13: Final subgradient
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=3000, time_limit=min(15, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
