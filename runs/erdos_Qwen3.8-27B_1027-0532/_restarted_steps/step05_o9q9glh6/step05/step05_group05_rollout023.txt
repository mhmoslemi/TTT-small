        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    over = corr[_g_n - 1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_lse(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def coord_descent(h, n, target, max_sweeps, deadline):
    """Coordinate descent for the Erdős overlap problem."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improved = False
        for j in range(n):
            t0 = h_cur[j]
            over = compute_over(h_cur)
            
            # Compute coefficients a[k] for k >= 1
            # a[k] = (1-h[j+k]) if j+k < n, minus h[j-k] if j-k >= 0
            a = np.zeros(n)
            if j + 1 < n:
                a[1:n-j] += (1.0 - h_cur[j+1:n])
            if j >= 1:
                a[1:j+1] -= h_cur[0:j][::-1]
            
            # Dense search + refinement
            t_grid = np.linspace(0.0, 1.0, 40)
            # Vectorized evaluation
            # f0(t) = over[0] + dx*(t-t0)*(1-t-t0)
            # fk(t) = over[k] + dx*a[k]*(t-t0)
            
            f0_vals = over[0] + dx * (t_grid - t0) * (1.0 - t_grid - t0)
            fk_vals = over[1:] + dx * a[1:] * (t_grid - t0)  # shape (n-1, 40)
            g_vals = np.maximum(f0_vals, fk_vals.max(axis=0))
            
            best_idx = np.argmin(g_vals)
            best_t = t_grid[best_idx]
            best_g = g_vals[best_idx]
            
            # Golden section refinement
            if best_t > 0.001 and best_t < 0.999:
                lo = t_grid[best_idx - 1] if best_idx > 0 else max(0, best_t - 0.025)
                hi = t_grid[best_idx + 1] if best_idx < 39 else min(1.0, best_t + 0.025)
                for _ in range(25):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f0_1 = over[0] + dx * (m1 - t0) * (1.0 - m1 - t0)
                    g1 = max(f0_1, np.max(over[1:] + dx * a[1:] * (m1 - t0)))
                    f0_2 = over[0] + dx * (m2 - t0) * (1.0 - m2 - t0)
                    g2 = max(f0_2, np.max(over[1:] + dx * a[1:] * (m2 - t0)))
                    if g1 < g2:
                        hi = m2
                    else:
                        lo = m1
                best_t = (lo + hi) / 2.0
            
            if best_t != t0:
                h_cur[j] = best_t
                any_improved = True
        
        # Project after each sweep
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < cur_score:
            cur_score = sc
        elif sweep > 5:
            # Check for convergence
            if sweep > 20 and not any_improved:
                break
    
    return h_cur, cur_score
