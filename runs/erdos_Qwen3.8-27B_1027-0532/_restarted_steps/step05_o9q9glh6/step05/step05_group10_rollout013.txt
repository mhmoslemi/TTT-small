    # h[j] appears in h[k + i - (n-1)] when k + i - (n-1) = j, i.e., k = j - i + (n-1)
    #   Contribution: (1-h[k]) = (1-h[j-i+(n-1)]) if 0 <= j-i+(n-1) <= n-1
    # h[j] appears in (1-h[k]) when k = j
    #   Contribution: -h[j + i - (n-1)] if 0 <= j+i-(n-1) <= n-1

    # grad[j] = sum_i weights[i] * dx * [(1-h[j-i+half_n]) * I1 - h[j+i-half_n] * I2]
    # where I1 = [0 <= j-i+half_n <= n-1] and I2 = [0 <= j+i-half_n <= n-1]

    significant = np.where(weights > 1e-6)[0]
    
    for i in significant:
        w = weights[i]
        if w < 1e-10:
            continue
        for j in range(n):
            k1 = j - i + half_n  # for (1-h[k1]) term
            if 0 <= k1 <= n - 1:
                grad[j] += w * (1.0 - h[k1]) * dx
            k2 = j + i - half_n  # for h[k2] term  
            if 0 <= k2 <= n - 1:
                grad[j] -= w * h[k2] * dx

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

def lp_all_constraints(h, n, target, reg, solver='SCS', eps=1e-10):
    """Single LP iteration with ALL constraints."""
    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1
    for k_idx in range(1, 2 * n - 1):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var[lo:hi] <= t_var)
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h)
    else:
        objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    solved = False
    try:

            prob.solve(solver=cp.SCS, eps=eps, max_iters=100000, verbose=False)
            solved = True
        elif solver == 'CLARABEL':
            prob.solve(solver=cp.CLARABEL, verbose=False)

        elif solver == 'ECOS':

                    actual_new_max = np.max(new_overlaps)
                    if actual_new_max < best_c5 - 1e-12 and actual_new_max < current_max + delta_max + 1e-10:
                        # Check if this is the best so far
                        if actual_new_max < current_max + best_delta:
                            best_delta = actual_new_max - current_max
                            best_a, best_b = a, b
        
        if best_a >= 0:
            h_bin[best_a] = 0.0
            h_bin[best_b] = 1.0
        else:
            break
    
    return best_h, best_c5

def subgradient_step(h, n, target, lr, dx):
    """One subgradient step minimizing the max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    half_n = n - 1
    k = k_star - half_n
    idxs = np.arange(n)
    
    grad = np.zeros(n)
    idx_v1 = (0 <= idxs - k) & ((idxs - k) < n)
    idx_v2 = (0 <= idxs + k) & ((idxs + k) < n)
    grad[idx_v1] += (1.0 - h[idxs[idx_v1] - k])
    grad[idx_v2] -= h[idxs[idx_v2] + k]
    grad *= dx
    
    h_new = h - lr * grad
    return project_feasible(h_new, target)

def subgradient_descent(h, n, target, time_limit=30, max_steps=10000):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1]:
            h_try = subgradient_step(h_cur, n, target, lr, dx)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                break
        else:
            if step > 200:
                break
    
    return best_h, best_c5

def coordinate_descent(h, n, target, time_limit=60):
    """Coordinate descent: optimize one coordinate at a time."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for sweep in range(1000):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            # Try to optimize h[j] while keeping others fixed
            # The objective is max_k overlap_k, and each overlap_k is linear in h[j]
            # So for a fixed h[j]=v, the objective as a function of v is:
            # max_k (a_k * v + b_k) where a_k and b_k depend on h[j]
            
            # For simplicity, try a few values: 0, 1, and the current value +/-
            h_save = h_cur[j]
            best_v = h_save
            best_sc = compute_c5(h_cur)
            
            for v in [0.0, 1.0, h_save + 0.1, h_save - 0.1, h_save + 0.3, h_save - 0.3]:
                if v < 0 or v > 1:
                    continue
                h_cur[j] = v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_v = v
                    best_h_try = h_cur.copy()
            
            if best_v != h_save:
                h_cur = best_h_try if 'best_h_try' in dir() else h_cur
                if best_sc < best_c5:
                    best_c5 = best_sc
                    best_h = h_cur.copy()
                improved = True
            else:
                h_cur[j] = h_save
                h_cur = project_feasible(h_cur, target)
        
        if not improved and sweep > 10:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)

    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1

        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var[lo:hi] <= t_var)
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h)
    else:
        objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    solved = False
    try:

            prob.solve(solver=cp.GLPK, verbose=False)
            solved = True

                    k1a = a - i + half_n
                    if 0 <= k1a < n:
                        d -= (1.0 - h_bin[k1a]) * dx
                    k2a = a + i - half_n
                    if 0 <= k2a < n:
                        d += h_bin[k2a] * dx
                    k1b = b - i + half_n
                    if 0 <= k1b < n:
                        d += (1.0 - h_bin[k1b]) * dx
                    k2b = b + i - half_n
                    if 0 <= k2b < n:
                        d -= h_bin[k2b] * dx
                    new_overlaps[i] += d
                actual_max = float(np.max(new_overlaps))
                improvement = current_max - actual_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_a, best_b = a, b
        if best_a >= 0 and best_improvement > 1e-11:
            h_bin[best_a] = 0.0
            h_bin[best_b] = 1.0
        else:
            break
    return best_h, best_c5

def subgradient_descent(h, n, target, time_limit=30):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    for step in range(20000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_v1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_v2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_v1] += (1.0 - h_cur[idxs[idx_v1] - k])
        grad[idx_v2] -= h_cur[idxs[idx_v2] + k]
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 300:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
