```python
    h = np.array(h, dtype=float)
    if abs(h.sum() - target_sum) < 1e-14:
        return np.clip(h, 0, 1)
    # Bisection on shift
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) * 0.5, 0, 1)
    # Fine-tune
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-13:
        mask = (h_proj > 1e-12) & (h_proj < 1 - 1e-12)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def lp_serve_all_lags(h, n, target, time_limit, n_top=None):
    """LP that linearizes all (or top-n_top) overlaps at current h"""
    start = time.time()
    dx = 2.0 / n
    g = 1.0 - h
    overlaps = compute_overlaps(h)
    n_full = len(overlaps)
    n_half = n - 1
    
    if n_top is not None and n_top < n_full:
        top_idx = np.argsort(overlaps)[::-1][:n_top]
    else:
        top_idx = np.arange(n_full)
    
    constraints = [cp.sum(cp.Variable(n)) == target]  # placeholder
    
    # Build LP: minimize t s.t. for each selected lag, linearized_overlap <= t
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    
    for idx in top_idx:
        k = idx - n_half  # actual lag
        if k >= 0:
            # overlap_k = sum_{i=0}^{n-1-k} h[i] * g[i+k] * dx
            # linearized: sum_{i=0}^{n-1-k} h_var[i] * g[i+k] * dx <= t
            if n - 1 - k >= 0:
                coeff = np.zeros(n)
                coeff[:n-k] = g[k:n] * dx
                cons.append(coeff @ h_var <= t_var)
        else:
            # k < 0, let m = -k
            m = -k
            # overlap at lag -m: sum_{i=m}^{n-1} h[i] * g[i-m] * dx
            # linearized: sum_{i=m}^{n-1} h_var[i] * g[i-m] * dx <= t
            coeff = np.zeros(n)
            coeff[m:] = g[:n-m] * dx
            cons.append(coeff @ h_var <= t_var)
    
    # Add regularization to stay close to h
    obj = t_var + 1e-6 * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(obj), cons)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iter=100000)
        except:
            return h, compute_c5(h)
    
    if h_var.value is None:
        return h, compute_c5(h)
    
    h_new = np.clip(np.asarray(h_var.value).ravel(), 0, 1)
    h_new = project_feasible(h_new, target)
    return h_new, compute_c5(h_new)
```

```python
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score

    overlaps = compute_overlaps(h)  # shape (2n-1,)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        T = T_start * (T_end / T_start) ** (step / max_steps)

        j = rng.randint(n)
        delta = rng.randn() * 0.03

        #   Δ = actual_delta * dx * (1-h[j+k]) - actual_delta * dx * h[j-k] (with bounds)

        k_pos = np.arange(n_half + 1)  # 0 to n-1

        valid2 = k_pos <= j

        deltas[n_half + k_pos[valid2]] += -actual_delta * dx * h[j - k_pos[valid2]]
```

Wait, there's an issue. When I change h[j], the overlaps change, but I also need to make sure the sum constraint is maintained. If I change h[j] by δ, the sum changes by δ. I need to compensate.

One approach: change two coordinates (j and l) by opposite amounts. But then the change in overlaps is more complex.

Another approach: After SA, project to feasibility. The projection might change the score slightly, but it should be close.

Actually, a better approach: maintain the sum constraint during SA by always making paired moves:
- Change h[j] by +δ and h[l] by -δ (for j ≠ l)
- Or change h[j] by +δ and h[j] by -δ (which is a no-op)

For a paired move (j gets +δ, l gets -δ):
The change in overlap_k is:
δ * ∂overlap_k/∂h[j] - δ * ∂overlap_k/∂h[l]

This maintains the sum constraint exactly!

Let me implement this:

```python
def sa_optimize_paired(h, n, target, time_limit, T_start=0.01, T_end=1e-8, rng=None):
    start = time.time()
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    
    overlaps = compute_overlaps(h)
    n_half = n - 1
    
    max_steps = 200000
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        T = T_start * (T_end / T_start) ** (step / max_steps)
        
        # Pick two distinct indices
        j = rng.randint(n)
        l = rng.randint(n)
        while l == j:
            l = rng.randint(n)
        
        delta = rng.randn() * 0.02
        
        # Make sure both stay in [0,1]
        if h[j] + delta < 0 or h[j] + delta > 1 or h[l] - delta < 0 or h[l] - delta > 1:
            continue
        
        # Compute change in overlaps
        # Δoverlap_k = delta * (∂overlap_k/∂h[j] - ∂overlap_k/∂h[l])
        
        deltas = np.zeros(2*n - 1)
        
        # For index j (gets +delta):
        k_pos = np.arange(n_half + 1)
        valid1_j = k_pos < (n - j)
        valid2_j = k_pos <= j
        deltas[n_half + k_pos[valid1_j]] += delta * dx * (1 - h[j + k_pos[valid1_j]])
        deltas[n_half + k_pos[valid2_j]] += -delta * dx * h[j - k_pos[valid2_j]]
        
        m_vals = np.arange(1, n_half + 1)
        k_idx_neg = n_half - m_vals
        valid1_j_neg = (j >= m_vals)
        valid2_j_neg = (j + m_vals < n)
        deltas[k_idx_neg[valid1_j_neg]] += delta * dx * (1 - h[j - m_vals[valid1_j_neg]])
        deltas[k_idx_neg[valid2_j_neg]] += -delta * dx * h[j + m_vals[valid2_j_neg]]
        
        # For index l (gets -delta):
        valid1_l = k_pos < (n - l)
        valid2_l = k_pos <= l
        deltas[n_half + k_pos[valid1_l]] -= delta * dx * (1 - h[l + k_pos[valid1_l]])
        deltas[n_half + k_pos[valid2_l]] += delta * dx * h[l - k_pos[valid2_l]]
        
        valid1_l_neg = (l >= m_vals)
        valid2_l_neg = (l + m_vals < n)
        deltas[k_idx_neg[valid1_l_neg]] -= delta * dx * (1 - h[l - m_vals[valid1_l_neg]])
        deltas[k_idx_neg[valid2_l_neg]] += delta * dx * h[l + m_vals[valid2_l_neg]]
        
        new_overlaps = overlaps + deltas
        new_score = np.max(new_overlaps)
        delta_f = new_score - current_score
        
        if delta_f < 0 or rng.random() < np.exp(-max(delta_f, -50) / max(T, 1e-15)):
            h[j] += delta
            h[l] -= delta
            overlaps = new_overlaps
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
    
    h_proj = project_feasible(best_h, target)
    return h_proj, compute_c5(h_proj)
```

```python
import numpy as np
from scipy.optimize import minimize

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    if abs(h.sum() - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-13:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_smoothed_grad(h, overlaps, alpha, dx):
    n = len(h)
    n_half = n - 1
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(n)
    k_pos = np.arange(n_half + 1)
    m_vals = np.arange(1, n_half + 1)
    k_idx_neg = n_half - m_vals
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - n_half
        if k >= 0:
            if k < n:
                valid1 = k_pos < (n - k)
                valid2 = k_pos <= k
                # For each j: contribution is w*dx*(1-h[j+k]) for j < n-k
                # and -w*dx*h[j-k] for j >= k
                j_range = np.arange(n)
                v1 = np.zeros(n)
                if k < n:
                    v1[:n - k] = 1.0 - h[k:]
                v2 = np.zeros(n)
                if k > 0:
                    v2[k:] = -h[:n - k]
                else:
                    v2[:] = -h[:]
                grad += w * dx * (v1 + v2)
        else:
            m = -k
            if m < n:
                v1 = np.zeros(n)
                v1[m:] = 1.0 - h[:n - m]
                v2 = np.zeros(n)
                v2[:n - m] = -h[m:]
                grad += w * dx * (v1 + v2)
    return grad, max_ov + np.log(np.sum(exp_shifted)) / alpha

def lbfgs_smoothed(h_start, n, target, alpha_list, time_limit, ftol=1e-15):
    start_time = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n

    for alpha in alpha_list:
        if time.time() - start_time > time_limit:
            break

        def obj_and_grad(h_flat):
            h_c = np.clip(h_flat, 0.0, 1.0)
            overlaps = compute_overlaps(h_c)
            max_ov = np.max(overlaps)
            shifted = alpha * (overlaps - max_ov)
            shifted = np.clip(shifted, -700.0, 0.0)
            exp_s = np.exp(shifted)
            weights = exp_s / np.sum(exp_s)
            grad = np.zeros(n)
            for k_idx in range(2 * n - 1):
                w = weights[k_idx]
                if w < 1e-14:
                    continue
                k = k_idx - (n - 1)
                if k >= 0:
                    if k < n:
                        v1 = np.zeros(n)
                        v1[:n - k] = 1.0 - h_c[k:]
                        v2 = np.zeros(n)
                        if k > 0:
                            v2[k:] = -h_c[:n - k]
                        else:
                            v2[:] = -h_c[:]
                        grad += w * dx * (v1 + v2)
                else:
                    m = -k
                    if m < n:
                        v1 = np.zeros(n)
                        v1[m:] = 1.0 - h_c[:n - m]
                        v2 = np.zeros(n)
                        v2[:n - m] = -h_c[m:]
                        grad += w * dx * (v1 + v2)
            penalty = 100.0 * (np.sum(h_c) - target)
            grad += penalty
            obj_val = max_ov + np.log(np.sum(exp_s)) / alpha + 50.0 * (np.sum(h_c) - target) ** 2
            return obj_val, grad

        try:
            res = minimize(obj_and_grad, h, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 2000, 'ftol': ftol,
                                                  'gtol': 1e-10, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass

    return best_h, best_score

def lp_all_lags(h, n, target, n_iter=50, time_limit=60.0, reg=1e-8):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n_half = n - 1
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        for k in range(-n_half, n_half + 1):
            if k >= 0:
                coeff = np.zeros(n)
                if k < n:
                    coeff[:n - k] = g[k:] * dx
            else:
                coeff = np.zeros(n)
                m = -k
                if m < n:
                    coeff[m:] = g[:n - m] * dx
            if np.any(coeff != 0):
                cons.append(coeff @ h_var <= t_var)
        obj = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(obj), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = project_feasible(h_new, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score

def lp_top_k(h, n, target, n_top, n_iter=100, time_limit=60.0, reg=1e-8, damping=0.9):
```

def project_feasible(h, target_sum):

    if abs(h.sum() - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)

        # l contributions (positive lags)

        mask1_l = kd < (n - l)

        mask2_l = kd <= l
        if mask2_l.any():
            dl[mask2_l] += -h[l - kd[mask2_l]]
        # For negative lags m=1..n-1:
        #   contribution from j: delta*dx*(1-h[j-m]) if j>=m, and -delta*dx*h[j+m] if j+m<n
        #   contribution from l: -delta*dx*(1-h[l-m]) if l>=m, and +delta*dx*h[l+m] if l+m<n
