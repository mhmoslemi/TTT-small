def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj_and_grad(x, alpha, target, mu):

    shifted = np.clip(shifted, -700.0, 0.0)

    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    n_corr = len(corr)
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]
    
    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-16, 'maxls': 50, 'maxfun': maxiter * 5})
        return project_feasible(res.x, target)
    except Exception:
        return h

def projected_grad_descent(h, target, alpha, mu, n_steps, deadline):
    """Fast projected gradient descent with backtracking line search."""
    n = len(h)
    h_cur = h.copy()
    cur_val, cur_grad = lse_obj_and_grad(h_cur, alpha, target, mu)
    lr = 0.01
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Backtracking line search
        best_h = h_cur
        best_val = cur_val
        for trial in range(10):
            h_new = h_cur - lr * cur_grad
            h_new = project_feasible(h_new, target)
            val_new, _ = lse_obj_and_grad(h_new, alpha, target, mu)
            if val_new < best_val:
                best_h = h_new
                best_val = val_new
            lr *= 0.5
            if lr < 1e-12:
                break
        if val_new < cur_val - 1e-15:
            h_cur = best_h
            cur_val, cur_grad = lse_obj_and_grad(h_cur, alpha, target, mu)
            lr = min(lr * 2, 0.05)
        else:
            break
    return h_cur

def active_lag_refine(h, target, n_top, n_steps, deadline):
    """Refine by focusing on the top n_top lags with adaptive step."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.01
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_max = top_vals.max()
        
        # Weight by exponential of (val - max) * beta
        beta = 50.0
        w = np.exp(beta * (top_vals - top_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for ki, wi in zip(top_idx, w):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += wi * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += wi * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h_cur[j2 - m]
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h_cur, cur_score

def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[0]
    
    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h

def fast_coord_descent(h, target, n_sweeps, deadline):
    """Fast coordinate descent using gradient-directed 1D search."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Get current gradients to determine direction
        _, grad = lse_obj_and_grad(h_cur, 1000.0, target, 1e8)
        
        for j in range(n):
            # Direction: -sign of gradient component
            if grad[j] > 0:
                direction = -1.0  # decrease h[j]
            elif grad[j] < 0:
                direction = 1.0  # increase h[j]
            else:
                continue
            
            # Try a few step sizes
            best_v = h_cur[j]
            best_sc = cur_score
            for step_frac in [0.1, 0.3, 0.5, 0.8]:
                trial_v = h_cur[j] + direction * step_frac * 0.1
                trial_v = max(0.0, min(1.0, trial_v))
                if abs(trial_v - h_cur[j]) < 1e-10:
                    continue
                h_new = h_cur.copy()
                h_new[j] = trial_v
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_sc:
                    best_sc = sc
                    best_v = trial_v
            
            if best_v != h_cur[j]:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
                improved = True
        if not improved:
            break
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

        lr_trial = lr
        for _ in range(12):
            h_new = h_cur - lr_trial * cur_grad
            h_new = project_feasible(h_new, target)
            val_new, _ = lse_obj_and_grad(h_new, alpha, target, mu)
            if val_new < best_val:
                best_h = h_new
                best_val = val_new
            lr_trial *= 0.5
            if lr_trial < 1e-14:
                break
        if best_val < cur_val - 1e-16:
            h_cur = best_h
            cur_val, cur_grad = lse_obj_and_grad(h_cur, alpha, target, mu)
            lr = min(lr * 1.2, 0.05)
        else:
            lr *= 0.6
            if lr < 1e-12:
                break
    return h_cur


def active_lag_refine(h, target, n_top, n_steps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.008
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_max = top_vals.max()
        beta = 100.0
        w = np.exp(beta * (top_vals - top_max))
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.ones(n_top) / n_top
        else:
            w = w / wsum
        grad = np.zeros(n)
        for ki, wi in zip(top_idx, w):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += wi * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += wi * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h_cur[j2 - m]
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-16:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.05, 0.04)
        else:
            lr *= 0.7
            if lr < 1e-11:
                break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    prob = SLSQPProb(alpha, target, n)
    cons = [{'type': 'eq', 'fun': prob.c_fun, 'jac': prob.c_jac}]
    try:
        res = minimize(prob, h, jac=prob.grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h


def fast_coord_descent(h, target, n_sweeps, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_score = float(np.max(corr)) * dx
    deltas = np.array([-0.08, -0.04, -0.02, 0.02, 0.04, 0.08])
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            best_v = h_cur[j]
            best_sc = cur_score
            for d in deltas:
                trial_v = h_cur[j] + d
                if trial_v < 0.0 or trial_v > 1.0:
                    continue
                h_new = h_cur.copy()
                h_new[j] = trial_v
                h_new = project_feasible(h_new, target)
                corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
                sc = float(np.max(corr_new)) * dx
                if sc < best_sc:
                    best_sc = sc
                    best_v = trial_v
            if best_v != h_cur[j]:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
                corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                improved = True
        if not improved:
            break
    return h_cur, cur_score


def multitask_restarts(h_best, target, n, mu, rng, deadline, best_score):
    n_restarts = 8
    strategies = []
    for p in [0.03, 0.06, 0.12, 0.2, 0.04, 0.09]:
        strategies.append(('gauss', p))
    strategies.append(('antithetic', 0.0))
    strategies.append(('flip', 0.03))
    results = []
    for strat, param in strategies[:n_restarts]:
        if time.time() > deadline:
            break
        h_pert = h_best.copy()
        if strat == 'gauss':
            h_pert += rng.randn(n) * param
        elif strat == 'antithetic':
            h_pert = 1.0 - h_best
        elif strat == 'flip':
            n_flip = max(1, int(n * param))
            idx = rng.choice(n, size=n_flip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        h_pert = project_feasible(h_pert, target)
        h_new = lbfgs_optimize(h_pert, target, 500, mu, 400, deadline)
        h_new = lbfgs_optimize(h_new, target, 50000, mu, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_best = h_new
    return h_best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 25.0

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 50.0

    # Phase 1: L-BFGS annealing with extended alpha schedule
    alphas_p1 = [10, 50, 200, 1000, 5000, 20000, 80000]
    maxiters_p1 = [400, 500, 500, 700, 800, 1000, 1200]
    for a, mi in zip(alphas_p1, maxiters_p1):
        if time.time() > deadline:
            break
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        alloc = min(mi, int(remaining * 0.15 * 100))
        if alloc < 50:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, alloc, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Projected gradient descent for fast refinement
    if time.time() < deadline - 60:
        h_pgd = projected_grad_descent(best_h, target, 50000, mu, 8000, deadline)
        sc = compute_c5(h_pgd)
        if sc < best_score:
            best_score = sc
            best_h = h_pgd

    # Phase 3: Active lag equalization
    if time.time() < deadline - 60:
        h_eq, sc_eq = active_lag_refine(best_h, target, n_top=10, n_steps=800,
                                        deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 100000, mu, 800, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Diverse restarts
    if time.time() < deadline - 120:
        best_h, best_score = multitask_restarts(best_h, target, n, mu, rng,
                                                deadline, best_score)

    # Phase 5: Second round of active lag refinement + L-BFGS
    if time.time() < deadline - 60:
        h_eq, sc_eq = active_lag_refine(best_h, target, n_top=5, n_steps=600,
                                        deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 300000, mu, 600, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Fast coordinate descent
    if time.time() < deadline - 80:
        h_cd, sc_cd = fast_coord_descent(best_h, target, n_sweeps=4,
                                         deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 7: SLSQP polish
    if time.time() < deadline - 40:
        h_slq = slsqp_polish(best_h, target, 50000, 2000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 8: Final active lag + high-alpha L-BFGS
    if time.time() < deadline - 30:
        h_eq, sc_eq = active_lag_refine(best_h, target, n_top=3, n_steps=400,
                                        deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 1000000, mu, 400, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 9: Final coordinate descent
    if time.time() < deadline - 30:
        h_cd, sc_cd = fast_coord_descent(best_h, target, n_sweeps=3,
                                         deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 10: Final SLSQP
    if time.time() < deadline - 10:
        h_slq = slsqp_polish(best_h, target, 200000, 1000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
