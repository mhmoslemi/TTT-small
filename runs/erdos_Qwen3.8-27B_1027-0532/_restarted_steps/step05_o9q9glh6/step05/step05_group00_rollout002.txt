import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def vec_swap_delta(h, a, b, k_arr):
    n = len(h)
    a_mk = a - k_arr
    a_pk = a + k_arr
    b_mk = b - k_arr
    b_pk = b + k_arr
    d = np.zeros(len(k_arr))
    m1 = (a_mk >= 0) & (a_mk < n)
    if m1.any():
        d[m1] -= (1.0 - h[a_mk[m1]])
    m2 = (a_pk >= 0) & (a_pk < n)
    if m2.any():
        d[m2] += h[a_pk[m2]]
    m3 = (b_mk >= 0) & (b_mk < n)
    if m3.any():
        d[m3] += (1.0 - h[b_mk[m3]])
    m4 = (b_pk >= 0) & (b_pk < n)
    if m4.any():
        d[m4] -= h[b_pk[m4]]
    return d

    b_pk = b + k_range

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(h):
                return smooth_obj(h, alpha, n, target)
            def cons_fun(h):
                return np.sum(h) - target
            def cons_jac(h):
                return np.ones(n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_sf, c5_sf = sa_diverse(h_bin, target_use, max_it_final, 0.005, 0.00001, rng,
                                 time_limit=sa_time, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

        if tleft() > 20:
            g_time = min(40, tleft() * 0.5)
            h_fg, c5_fg = fast_greedy(best_h, max_iters=200000, time_limit=g_time,
                                      start_time=start_time, top_m=25, scan_width=35)
            if c5_fg < best_score:
                best_score = c5_fg
                best_h = h_fg.copy()

    # Final SLSQP polish if time remains
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [5000, 20000, 100000]
        h_ref2, sc2 = slsqp_polish(best_h, n_use, target_use, alphas2, 2000, 1e-16,
                                   time_limit=max(5.0, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score and len(best_h) == n_final:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
