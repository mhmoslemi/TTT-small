    # FISTA for the smooth approximation
    h_cur = h.copy()
    y = h_cur.copy()
    t = 1.0
    step = step_init
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for i in range(n_iter):
        if time.time() > deadline:
            break
        # Compute gradient at y
        g = lse_grad(y, alpha)
        # Gradient step
        z = y - step * g
        # Project onto feasible set
        x_new = project_feasible(z, target)
        # Update momentum
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = x_new + (t - 1) / t_new * (x_new - h_cur)
        h_cur = x_new
        t = t_new
        # Check score
        score = compute_c5(h_cur)
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
    
    return best_h, best_score

        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            if lr < 0.01:
                lr *= 1.01
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.015, 0.012)
        else:
            eps *= 0.88
            if eps < 1e-10:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_fista(h, n, target, alpha, max_iter, step_init, deadline, rng, n_starts):
    best_h = h.copy()
    best_score = compute_c5(h)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        h_start = h.copy()
        if s > 0:
            scale = 0.05 * (s + 1) / n_starts
            h_start = h_start + rng.randn(n) * scale
            h_start = project_feasible(h_start, target)
        h_new, sc = fista_optimize(h_start, n, target, alpha, max_iter, step_init, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 8.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick FISTA at original resolution
    for alpha in [50, 200, 1000]:
        if time.time() > deadline - 700:
            break
        step = 0.8 / alpha
        h_new, sc = fista_optimize(best_h, n, target, alpha, 3000, step, min(time.time() + 40, deadline - 700))
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: L-BFGS-B refinement at original resolution
    for alpha in [500, 3000]:
        if time.time() > deadline - 600:
            break
        for mu in [10.0, 100.0]:
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 3: Interpolate to higher resolution
    n2 = 800
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)
    best_h2 = h2.copy()
    best_score2 = compute_c5(h2)

    if best_score2 < best_score:
        best_score = best_score2
        best_h = best_h2

    # Phase 4: FISTA at high resolution with increasing alpha
    alphas_high = [20, 60, 200, 600, 2000, 5000, 10000]
    for alpha in alphas_high:
        if time.time() > deadline - 200:
            break
        step = 0.6 / alpha
        n_iter = max(500, int(50000.0 / alpha))
        t_rem = min(150, deadline - time.time() - 100)
        if t_rem < 20:
            break
        h2_new, sc2 = fista_optimize(best_h2, n2, target2, alpha, n_iter, step, time.time() + t_rem)
        sc2 = compute_c5(h2_new)
        if sc2 < best_score2:
            best_score2 = sc2
            best_h2 = h2_new

    # Phase 5: Multi-start FISTA at high resolution
    if time.time() < deadline - 120:
        for alpha in [2000, 5000]:
            if time.time() > deadline - 100:
                break
            h2_ms, sc2_ms = multi_start_fista(best_h2, n2, target2, alpha, 2000, 0.3 / alpha,
                                               deadline - 80, rng, 3)
            if sc2_ms < best_score2:
                best_score2 = sc2_ms
                best_h2 = h2_ms

    # Phase 6: SLSQP at high resolution
    for alpha in [5000, 15000, 40000]:
        if time.time() > deadline - 60:
            break
        h2_new = slsqp_round(best_h2, n2, target2, alpha, 2000, 1e-15)
        sc2 = compute_c5(h2_new)
        if sc2 < best_score2:
            best_score2 = sc2
            best_h2 = h2_new

    # Phase 7: Subgradient polish
    if time.time() < deadline - 40:
        for temp in [0.003, 0.001, 0.0005]:
            if time.time() > deadline - 30:
                break
            h2_sg, sc2_sg = subgradient_polish(best_h2, n2, target2, 3000, temp, 0.002, 0.998, deadline - 30)
            if sc2_sg < best_score2:
                best_score2 = sc2_sg
                best_h2 = h2_sg

    # Phase 8: Worst-lag surgery
    if time.time() < deadline - 25:
        h2_surg, sc2_surg = worst_lag_surgery(best_h2, n2, target2, 500, 0.002, deadline - 25)
        if sc2_surg < best_score2:
            best_score2 = sc2_surg
            best_h2 = h2_surg

    # Phase 9: Final SLSQP
    if time.time() < deadline - 10:
        for alpha in [30000, 80000]:
            if time.time() > deadline - 6:
                break
            h2_new = slsqp_round(best_h2, n2, target2, alpha, 1000, 1e-16)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

    # Phase 10: Try n=1000 if time permits
    if time.time() < deadline - 80:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h2, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)
        for alpha in [50, 200, 1000, 5000]:
            if time.time() > deadline - 20:
                break
            step = 0.5 / alpha
            n_iter = max(300, int(30000.0 / alpha))
            h3_new, sc3 = fista_optimize(best_h3, n3, target3, alpha, n_iter, step,
                                          min(time.time() + 40, deadline - 20))
            sc3 = compute_c5(h3_new)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3_new
        if time.time() < deadline - 10:
            h3_new = slsqp_round(best_h3, n3, target3, 10000, 1000, 1e-15)
            sc3 = compute_c5(h3_new)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3_new
        if best_score3 < best_score2:
            best_score2 = best_score3
            best_h2 = best_h3

    if best_score2 < best_score:
        best_score = best_score2
        best_h = best_h2

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
