import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_periodic(base, n):
    """Build a periodic signal by tiling the base pattern."""
    p = len(base)
    m = n // p
    h = np.tile(base, m)
    return h[:n]

def periodic_obj(base, p, n, alpha):
    """Objective for periodic optimization."""
    h = build_periodic(np.clip(base, 0, 1), n)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def optimize_periodic(p, n, alpha, maxiter, rng, n_restarts):
    """Optimize a periodic pattern of period p."""
    target = p / 2.0
    best_base = None
    best_score = np.inf
    
    for r in range(n_restarts):
        # Random binary initialization
        idx = rng.choice(p, size=p//2, replace=False)
        base = np.zeros(p)
        base[idx] = 1.0
        base = project_feasible(base, target)
        
        # SLSQP optimization
        try:
            res = minimize(
                lambda b: periodic_obj(b, p, n, alpha),
                base,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * p,
                constraints=[{'type': 'eq', 'fun': lambda b: np.sum(b) - target,
                              'jac': lambda b: np.ones(p)}],
                options={'maxiter': maxiter, 'ftol': 1e-14}
            )
            h_test = project_feasible(build_periodic(np.clip(res.x, 0, 1), n), n/2.0)
            sc = compute_c5(h_test)
            if sc < best_score:
                best_score = sc
                best_base = res.x.copy()
        except Exception:
            pass
    
    return best_base, best_score
