def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def binary_sa(h_bin, n, target, time_limit, rng):
    """Simulated annealing on binary sequences."""
    start = time.time()
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    
    T = 0.005  # initial temperature (scaled for C5 values around 0.38)
    n_iter = 0
    stuck_count = 0
    
    while time.time() - start < time_limit:
        n_iter += 1
        T *= 0.99995
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stuck_count += 1
        
        if stuck_count > 50000:
            break
    
    return best_h, best_c5

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Try targeted swaps
        mid = n - 1
        worst_k = int(np.argmax(corr)) - mid
        improved = False
        
        # Find positions contributing to worst shift
        # c[worst_idx] = sum_i h[i] * (1-h[i-worst_k])
        # Bad positions: h[i]=1, h[i-worst_k]=0
        bad_ones = []
        for i in ones:
            ik = i - worst_k
            if 0 <= ik < n and h[ik] == 0:
                bad_ones.append(i)
        
        # Try swapping bad ones with random zeros
        rng.shuffle(bad_ones)
        n_try = min(100, len(bad_ones))
        
        for i in bad_ones[:n_try]:
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5_val - 1e-15:
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        # If no targeted improvement, try random swaps
        if not improved:
            n_random = 50
            for _ in range(n_random):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < c5_val - 1e-15:
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    
    return best_h, best_c5

def binary_sa(h_bin, n, target, time_limit, rng, T0=0.003, cooling=0.99998):
    """Simulated annealing on binary sequences."""
    start = time.time()
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    T = T0
    dx = 2.0 / n
    no_improve = 0
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    while time.time() - start < time_limit and T > 1e-10:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        T *= cooling
        if no_improve > 200000:
            break
    
    return best_h, best_c5

def refine_binary(h_best, n, target, time_limit):
    """Try small continuous perturbations around a binary solution."""
    start = time.time()
    rng = np.random.RandomState(12345)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    
    # Try various perturbation scales
    for scale in [0.1, 0.2, 0.3]:
        if time.time() - start > time_limit:
            break
        h_try = h_best + rng.randn(n) * scale
        h_try = project_feasible(h_try, target)
        # SLSQP polish
        h_pol, sc = slsqp_round(h_try, n, target, [1000, 10000], 500)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pol
    
    return best_h, best_c5

def make_structured_binary(n, pattern_type, rng):
    """Create structured binary sequences as initial guesses."""
    h = np.zeros(n)
    target = n / 2.0
    
    if pattern_type == 'random':
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
    elif pattern_type == 'blocks':
        # Alternating blocks of varying sizes
        pos = 0
        ones_count = 0
        while pos < n:
            block_len = max(1, int(rng.exponential(10)))
            if ones_count < target:
                fill = min(block_len, int(target) - ones_count, n - pos)
                h[pos:pos+fill] = 1.0
                ones_count += fill
            pos += block_len
    elif pattern_type == 'periodic':
        # Periodic pattern with period p
        p = int(n / 10)
        for i in range(n):
            if (i % p) < p // 2:
                h[i] = 1.0
        # Adjust to have exactly n/2 ones
        h = project_feasible((h > 0.5).astype(float), target)
    elif pattern_type == 'golay_like':
        # Golay-like: use quadratic residues
        p = n
        h = np.zeros(n)
        for i in range(n):
            # Use a pseudo-random but structured pattern
            if (i * i) % (n + 1) < n // 2:
                h[i] = 1.0
        # Adjust
        ones_count = int(h.sum())
        if ones_count != int(target):
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_count > target:
                remove = rng.choice(ones_idx, size=ones_count - int(target), replace=False)
                h[remove] = 0
            else:
                add = rng.choice(zeros_idx, size=int(target) - ones_count, replace=False)
                h[add] = 1
    return h
