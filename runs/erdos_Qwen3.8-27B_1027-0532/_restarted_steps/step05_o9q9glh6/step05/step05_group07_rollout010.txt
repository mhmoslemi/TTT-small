    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_sum(h_out, target_dst)

def make_step_init(n, target, frac_ones):
    h = np.zeros(n)
    n_ones = int(n * frac_ones)
    h[:n_ones] = 1.0
    h = h + (target - np.sum(h)) / n
    return project_sum(h, target)

def make_sinusoidal_init(n, target, freq, amp):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x)
    return project_sum(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_sum(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Momentum PG with moderate alpha
    deadline1 = start_time + (budget_s - 25.0) * 0.12
    h_mom, sc_mom = momentum_pg(best_h, n, target, alpha=200, max_iter=500,
                                  deadline=deadline1, lr=0.004, momentum=0.85)
    if sc_mom < best_score:
        best_score = sc_mom
        best_h = h_mom

    # Phase 2: Coordinate descent
    deadline2 = start_time + (budget_s - 25.0) * 0.30
    h_cd, sc_cd = paired_cd(best_h, n, target, deadline2, rng, n_sweeps=8000, recompute_every=30)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 3: SLSQP with increasing alpha
    if tleft() > 150:
        h_slq, sc_slq = slsqp_polish(best_h, n, target, [100, 500, 2000], 1500, 1e-16)
        if sc_slq < best_score:
            best_score = sc_slq
            best_h = h_slq

    # Phase 4: More coordinate descent
    deadline4 = start_time + (budget_s - 25.0) * 0.48
    h_cd, sc_cd = paired_cd(best_h, n, target, deadline4, rng, n_sweeps=8000, recompute_every=25)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 5: Multi-scale
    if tleft() > 300:
        n_coarse = max(n // 4, 50)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        deadline_coarse = start_time + (budget_s - 25.0) * 0.60
        h_coarse, sc_coarse = paired_cd(h_coarse, n_coarse, target_coarse,
                                         deadline_coarse, rng, n_sweeps=5000, recompute_every=20)
        h_up = interpolate_to(h_coarse, n, target)
        h_up, sc_up = slsqp_polish(h_up, n, target, [200, 1000], 1500, 1e-15)
        deadline_up = start_time + (budget_s - 25.0) * 0.68
        h_up, sc_up = paired_cd(h_up, n, target, deadline_up, rng, n_sweeps=5000, recompute_every=25)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 6: Diverse restarts
    deadline6 = start_time + (budget_s - 25.0) * 0.85
    inits = [
        project_sum(best_h + rng.randn(n) * 0.04, target),
        project_sum(best_h + rng.randn(n) * 0.08, target),
        project_sum(best_h + rng.randn(n) * 0.15, target),
        make_step_init(n, target, 0.5),
        make_sinusoidal_init(n, target, 3, 0.3),
        make_sinusoidal_init(n, target, 5, 0.4),
        project_sum(1.0 - best_h, target),
    ]
    for init_h in inits:
        if tleft() < 100:
            break
        h_sub = init_h.copy()
        sub_deadline = min(deadline6, time.time() + 100)
        h_sub, sc_sub = momentum_pg(h_sub, n, target, alpha=300, max_iter=200,
                                     deadline=sub_deadline, lr=0.003, momentum=0.85)
        if tleft() < 60:
            break
        sub_deadline2 = min(deadline6, time.time() + 80)
        h_sub, sc_sub = paired_cd(h_sub, n, target, sub_deadline2, rng, n_sweeps=4000, recompute_every=25)
        if sc_sub < best_score:
            best_score = sc_sub
            best_h = h_sub

    # Phase 7: Final SLSQP
    if tleft() > 80:
        h_fin, sc_fin = slsqp_polish(best_h, n, target, [500, 2000, 8000], 2000, 1e-17)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: Final CD
    if tleft() > 30:
        h_fin, sc_fin = paired_cd(best_h, n, target, start_time + budget_s - 20.0,
                                   rng, n_sweeps=5000, recompute_every=20)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    h_out = project_sum(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
