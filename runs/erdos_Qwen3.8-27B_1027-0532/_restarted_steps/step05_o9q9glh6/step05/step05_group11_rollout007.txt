from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, n_top=None, eps=None):
    """Get lags that are near the maximum."""
    overlaps = compute_overlaps(h)
    n = len(h)
    n_half = n - 1
    c_max = np.max(overlaps)
    if n_top is not None:
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        return top_idx - n_half
    if eps is not None:
        active = np.where(overlaps >= c_max - eps)[0]
        return active - n_half
    return np.arange(len(overlaps)) - n_half

def solve_active_lp(h_fixed, lags, n, target, dx, reg=0.0):
    """Solve LP: minimize t s.t. overlap_k(h_new) <= t for k in lags, 0<=h<=1, sum(h)=target
    where overlap_k is computed with h_fixed on the (1-h) side."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h_fixed
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h_fixed)
    else:
        objective = t_var
    
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
        except:
            return None, None
    
    if h_var.value is None:
        return None, None
    return np.asarray(h_var.value).ravel(), t_var.value

def alternating_lp_active_set(h_init, n, target, max_iter, time_limit,
                               n_top_init=50, n_top_max=300, 
                               damping=0.9, reg_decay=True,
                               reg_start=1e-4, reg_end=1e-9):
    """Alternating LP with adaptive active set."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        progress = min(1.0, it / max(1, max_iter))
        # Adaptive number of active lags
        n_top = min(n_top_max, n_top_init + int(progress * (n_top_max - n_top_init)))
        # Adaptive damping
        damp = damping + (1.0 - damping) * progress * 0.3
        # Adaptive regularization
        reg = reg_start * (reg_end / reg_start) ** progress if reg_decay else reg_start
        
        # Get active lags (top K)
        lags = get_active_lags(h, n_top=n_top)
        
        # Solve LP
        h_new, t_val = solve_active_lp(h, lags, n, target, dx, reg)
        if h_new is None:
            break
        
        h_new = np.clip(h_new, 0.0, 1.0)
        h = (1.0 - damp) * h + damp * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5
