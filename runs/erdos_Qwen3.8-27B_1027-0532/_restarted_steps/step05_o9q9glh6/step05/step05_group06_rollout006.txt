- Term (1-h[i+k]) with i+k=j, i=j-k: contributes -h[j-k] if k ≤ j ≤ n-1 (and i=j-k must satisfy 0≤i≤n-1-k, i.e., k≤j≤n-1)

- Second term: for j = k to n-1, subtract h[j-k] → this is h[0:n-k] subtracted from grad[k:n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_gradient_for_lag(h, lag, n):
    """Compute gradient of corr_lag with respect to h."""
    grad = np.zeros(n)
    if lag > 0:
        # corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[j]: (1-h[j+lag]) for j in [0, n-1-lag]
        #          -h[j-lag] for j in [lag, n-1]
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n-lag]
    elif lag < 0:
        m = -lag
        # corr = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # d/dh[j]: (1-h[j-m]) for j in [m, n-1]
        #          -h[j+m] for j in [0, n-1-m]
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    else:
        # corr = sum h[i]*(1-h[i])
        grad[:] = 1.0 - 2.0 * h
    return grad


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    # Log-sum-exp
    log_sum_exp = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    val = log_sum_exp
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    
    # Gradient
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        grad += w * lse_gradient_for_lag(x, lag, n)
    
    grad += 2.0 * mu * (s - target) * np.ones(n)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]
    
    def grad(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-14, 'maxls': 50,
                                'maxfun': maxiter * 3})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, target)
    except Exception:
        return h

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def grad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    elif lag < 0:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        grad += w * grad_for_lag(x, lag, n)
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h0 = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-12, 'maxls': 50,
                                'maxfun': maxiter * 5})
        return project_feasible(res.x, target)
    except Exception:
        return h0


def top_lag_momentum(h, target, n_top, n_steps, deadline):
    n = len(h)
    h_cur = h.copy()
    mom = np.zeros(n)
    cur_score = compute_c5(h_cur)
    lr = 0.005
    beta = 0.85
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-13:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = corr[ki] - top_mean
            if abs(w) < 1e-15:
                continue
            lag = int(ki) - (n - 1)
            grad += w * grad_for_lag(h_cur, lag, n)
        mom = beta * mom + grad
        h_new = h_cur - lr * mom
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            lr = min(lr * 1.02, 0.08)
        else:
            lr *= 0.7
            mom *= 0.5
            if lr < 1e-11:
                break
    return h_cur, cur_score


def coord_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h_cur, cur_score
            a, b = 0.0, 1.0
            c_val = b - gr * (b - a)
            d_val = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                return float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))

            fc = f1d(c_val)
            fd = f1d(d_val)
            for _ in range(25):
                if fc < fd:
                    b, d_val, fd = d_val, c_val, fc
                    c_val = b - gr * (b - a)
                    fc = f1d(c_val)
                else:
                    a, c_val, fc = c_val, d_val, fc
                    d_val = a + gr * (b - a)
                    fd = f1d(d_val)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-9:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-15:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    mu = 1e10

    def obj(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_j}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h0


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 30.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_n = n
    cur_target = target
    mu = 100.0

    # Phase 1: L-BFGS-B at original resolution
    alphas_p1 = [20, 200, 2000, 20000, 100000, 500000]
    for a in alphas_p1:
        if time.time() > deadline - 550:
            break
        h = lbfgs_optimize(best_h, cur_target, a, mu, 500, deadline - 550)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 2: Top-lag equalization at original resolution
    if time.time() < deadline - 450:
        h_eq, sc = top_lag_momentum(best_h, cur_target, n_top=12, n_steps=600, deadline=deadline - 450)
        if sc < best_score:
            best_score = sc
            best_h = h_eq
            cur_n = len(best_h)
            cur_target = cur_n / 2.0

    # Phase 3: Final L-BFGS at original resolution
    if time.time() < deadline - 420:
        h = lbfgs_optimize(best_h, cur_target, 1000000, mu, 300, deadline - 420)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 4: Upsample to 2x resolution
    n2 = 2 * cur_n
    target2 = n2 / 2.0
    x_old = np.linspace(0.0, 2.0, cur_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    try:
        cs = CubicSpline(x_old, best_h, bc_type='not-a-knot')
        h2 = cs(x_new)
    except Exception:
        h2 = np.interp(x_new, x_old, best_h)
    h2 = np.clip(h2, 0.0, 1.0)
    h2 = project_feasible(h2, target2)

    # Phase 5: L-BFGS-B at fine resolution
    alphas_fine = [50, 500, 5000, 50000, 200000, 1000000]
    for a in alphas_fine:
        if time.time() > deadline - 120:
            break
        h2 = lbfgs_optimize(h2, target2, a, mu, 400, deadline - 120)
        sc = compute_c5(h2)
        if sc < best_score:
            best_score = sc
            best_h = h2.copy()
            cur_n = n2
            cur_target = target2

    # Phase 6: Top-lag equalization at fine resolution
    if time.time() < deadline - 100:
        h_eq, sc = top_lag_momentum(best_h, cur_target, n_top=15, n_steps=500, deadline=deadline - 100)
        if sc < best_score:
            best_score = sc
            best_h = h_eq

    # Phase 7: Another L-BFGS at fine resolution
    if time.time() < deadline - 80:
        h = lbfgs_optimize(best_h, cur_target, 5000000, mu, 400, deadline - 80)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 8: Coordinate descent
    if time.time() < deadline - 50:
        h_cd, sc = coord_descent(best_h, cur_target, n_sweeps=2, deadline=deadline - 50)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 9: SLSQP polish
    if time.time() < deadline - 20:
        h_sl = slsqp_polish(best_h, cur_target, 50000, 5000)
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    # Phase 10: Final coordinate descent
    if time.time() < deadline - 10:
        h_cd, sc = coord_descent(best_h, cur_target, n_sweeps=1, deadline=deadline - 10)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    best_h = project_feasible(best_h, cur_target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
