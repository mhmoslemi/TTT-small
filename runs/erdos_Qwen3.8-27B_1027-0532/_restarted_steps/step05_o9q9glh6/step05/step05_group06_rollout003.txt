```python
def project_box_sum(h, target):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target}"""
    h = h.copy()
    for _ in range(100):
        excess = np.sum(h) - target
        if abs(excess) < 1e-15:
            break
        if excess > 0:
            # Need to decrease sum by excess
            # Find threshold: reduce all h_i > threshold by (h_i - threshold)
            # such that sum of reductions = excess
            free = h > 0
            if not free.any():
                break
            # Binary search for threshold
            lo, hi = 0.0, h[free].max()
            for _ in range(50):
                mid = (lo + hi) / 2
                reduction = np.sum(np.maximum(h[free] - mid, 0))
                if reduction > excess:
                    lo = mid
                else:
                    hi = mid
            threshold = (lo + hi) / 2
            h[h > threshold] -= (h[h > threshold] - threshold)
            h = np.clip(h, 0, 1)
        else:
            # Need to increase sum by -excess
            free = h < 1
            if not free.any():
                break
            lo, hi = h[free].min(), 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                increase = np.sum(np.maximum(mid - h[free], 0))
                if increase > -excess:
                    hi = mid
                else:
                    lo = mid
            threshold = (lo + hi) / 2
            h[h < threshold] += (threshold - h[h < threshold])
            h = np.clip(h, 0, 1)
    return h
```
