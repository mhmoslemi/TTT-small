def compute_shift_delta(h, idx, n):
    """Compute the change in corr vector when h[idx] flips from current to 1-current"""
    # Returns delta vector of length 2n-1
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    half_n = n - 1
    
    # If h[idx] goes from 0 to 1:
    # delta[k] = (1-h[idx+s]) * [idx+s valid] - h[idx-s] * [idx-s valid] where s = k-half_n
    # If h[idx] goes from 1 to 0:
    # delta[k] = -(1-h[idx+s]) * [idx+s valid] + h[idx-s] * [idx-s valid]
    
    current_val = h[idx]
    sign = 1.0 if current_val == 0 else -1.0  # +1 for 0->1, -1 for 1->0
    
    # (1-h[idx+s]) for s in [-(n-1), n-1] where idx+s in [0, n-1]
    # idx+s in [0, n-1] means s in [-idx, n-1-idx]
    s_range = np.arange(max(-idx, -(n-1)), min(n-1-idx, n-1) + 1)

    vals2 = h[idx - s_range2]
    delta[k_idx2] -= sign * vals2

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target):
    """Convert h to binary with exactly target ones"""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target:
        return h_bin
    if n_ones > target:
        # Need to flip (n_ones - target) ones to zeros
        # Flip the ones with lowest "importance" (e.g., lowest h value)
        ones = np.where(h_bin == 1.0)[0]
        # Sort by h value (ascending) - flip the smallest first
        order = np.argsort(h[ones])
        to_flip = ones[order[:n_ones - target]]
        h_bin[to_flip] = 0.0
    else:
        # Need to flip (target - n_ones) zeros to ones
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])  # Flip the largest zeros first
        to_flip = zeros[order[:target - n_ones]]
        h_bin[to_flip] = 1.0
    return h_bin

def compute_shift_delta(h, idx, n):
    """Compute delta in corr when h[idx] flips. Returns delta for the flip direction."""
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    half_n = n - 1
    current_val = h[idx]
    sign = 1.0 if current_val < 0.5 else -1.0  # +1 for 0->1, -1 for 1->0
    
    # (1-h[idx+s]) for valid s where idx+s in [0, n-1]
    s_min = max(-idx, -(n-1))
    s_max = min(n-1-idx, n-1)
    if s_max >= s_min:
        s_vals = np.arange(s_min, s_max + 1)
        k_vals = s_vals + half_n
        vals = 1.0 - h[idx + s_vals]
        delta[k_vals] += sign * vals
    
    # -h[idx-s] for valid s where idx-s in [0, n-1]
    s_min2 = max(idx - (n-1), -(n-1))
    s_max2 = min(idx, n-1)
    if s_max2 >= s_min2:
        s_vals2 = np.arange(s_min2, s_max2 + 1)
        k_vals2 = s_vals2 + half_n
        vals2 = h[idx - s_vals2]
        delta[k_vals2] -= sign * vals2
    
    return delta

def fast_binary_search(h_bin, n, target, max_iter, time_limit, rng):
    """Fast local search on binary sequences using O(n) per move"""
    start = time.time()
    half_n = n - 1
    corr_len = 2 * n - 1
    
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_corr = corr.copy()
    best_c5_raw = float(np.max(best_corr))
    best_h = h.copy()
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    no_improve_count = 0
    max_no_improve = 500
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve_count > max_no_improve:
            break
        
        improved = False
        # Try multiple random moves
        n_tries = 20
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            
            delta_i = compute_shift_delta(h, i, n)  # 0->1
            delta_j = compute_shift_delta(h, j, n)  # 1->0
            
            new_corr_max = np.max(corr + delta_i + delta_j)
            if new_corr_max < best_c5_raw - 1e-12:
                # Accept
                corr = corr + delta_i + delta_j
                h[i] = 1.0
                h[j] = 0.0
                best_c5_raw = new_corr_max
                best_h = h.copy()
                best_corr = corr.copy()
                
                # Update lists
                zeros.remove(i)
                ones.append(i)
                ones.remove(j)
                zeros.append(j)
                
                improved = True
                no_improve_count = 0
                break
            # Otherwise revert (we didn't change h)
        
        if not improved:
            no_improve_count += 1
    
    return best_h, best_c5_raw * (2.0 / n)

def lbfgs_smooth_optimize(h_init, n, target, alphas, time_limit):
    """L-BFGS-B optimization with log-sum-exp smoothing and penalty"""
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:

            c_max = np.max(corr)

            lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            penalty = 50.0 * (np.sum(x) - target) ** 2
            return float(lse * (2.0/n) + penalty)
        
        def grad(x):
            x = np.clip(x, 0.0, 1.0)

            c_max = np.max(corr)

            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            if w_sum < 1e-300:
                weights = np.ones(len(corr)) / len(corr)
            else:
                weights /= w_sum
            
            g = np.zeros(n)
            half_n = n - 1

                g[valid_ms] -= weights[k_idx] * x[jms[valid_ms]]
            
            g *= (2.0 / n)
            g += 100.0 * (np.sum(x) - target)
            return g

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target:
        return h_bin
    if n_ones > target:
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones])
        to_flip = ones[order[:n_ones - target]]
        h_bin[to_flip] = 0.0
    else:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])
        to_flip = zeros[order[:target - n_ones]]
        h_bin[to_flip] = 1.0
    return h_bin

def compute_shift_delta(h, idx, n):
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    half_n = n - 1
    sign = 1.0 if h[idx] < 0.5 else -1.0
    s_vals = np.arange(-(n - 1), n)
    mask = s_vals != 0
    s_vals = s_vals[mask]
    valid1 = (s_vals >= -idx) & (s_vals <= n - 1 - idx)
    if np.any(valid1):
        sv = s_vals[valid1]
        kv = sv + half_n
        delta[kv] += sign * (1.0 - h[idx + sv])
    valid2 = (s_vals >= idx - (n - 1)) & (s_vals <= idx)
    if np.any(valid2):
        sv2 = s_vals[valid2]
        kv2 = sv2 + half_n
        delta[kv2] -= sign * h[idx - sv2]
    return delta

    best_h = h.copy()
    no_improve = 0
    max_no_improve = 3000
    total_iters = max(500000, int(time_limit * 2000))
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        if h[i] == 1.0:
            i, j = j, i
        delta_i = compute_shift_delta(h, i, n)
        delta_j = compute_shift_delta(h, j, n)
        new_max = float(np.max(corr + delta_i + delta_j))
        delta_obj = new_max - best_corr_max
        accept = False
        if delta_obj < -1e-12:

            if new_max < best_corr_max:
                best_corr_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_corr_max * (2.0 / n)

def exhaustive_binary_search(h_bin, n, time_limit, rng):
    start = time.time()
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_corr_max = float(np.max(corr))
    best_h = h.copy()
    ones_arr = np.where(h == 1.0)[0]
    zeros_arr = np.where(h == 0.0)[0]
    for epoch in range(200):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in zeros_arr:
            if time.time() - start > time_limit:
                break
            delta_i = compute_shift_delta(h, int(i), n)
            base = corr + delta_i
            for j in ones_arr:
                if int(j) == int(i):
                    continue
                delta_j = compute_shift_delta(h, int(j), n)
                new_max = float(np.max(base + delta_j))
                if new_max < best_corr_max - 1e-12:
                    corr = base + delta_j
                    h[int(i)] = 1.0
                    h[int(j)] = 0.0
                    best_corr_max = new_max
                    best_h = h.copy()
                    ones_arr = np.where(h == 1.0)[0]
                    zeros_arr = np.where(h == 0.0)[0]
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_corr_max * (2.0 / n)

def alternating_lp(h_init, n, target, time_limit, n_active=50, eps=1e-10, damping_init=0.8):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    damping = damping_init
    reg = 1e-4
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            s = k_idx - half_n
            lo = max(0, s)
            hi = min(n, n + s)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - s:hi - s] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        active_k = set(np.argsort(corr)[-n_active:].tolist())
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.95, damping + 0.01)
        if it % 100 == 0 and reg > 1e-7:
            reg = max(1e-7, reg * 0.85)
    return best_h, best_c5

def lbfgs_smooth(h_init, n, target, alphas, time_limit):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def obj(x):
            x = np.clip(x, 0.0, 1.0)
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            sh = np.clip(alpha * (c - cm), -400.0, 0.0)
            lse = cm + np.log(np.sum(np.exp(sh))) / alpha
            pen = 50.0 * (np.sum(x) - target) ** 2
            return float(lse * dx + pen)
        def grad(x):
            x = np.clip(x, 0.0, 1.0)
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            sh = np.clip(alpha * (c - cm), -400.0, 0.0)
            w = np.exp(sh)
            ws = np.sum(w)
            if ws < 1e-300:
                w = np.ones(len(c)) / len(c)
            else:
                w = w / ws
            g = np.zeros(n)
            hn = n - 1
            idxs = np.arange(n)
            for ki in range(len(c)):
                if w[ki] < 1e-14:
                    continue
                s = ki - hn
                js = idxs + s
                vp = (js >= 0) & (js < n)
                g[vp] += w[ki] * (1.0 - x[js[vp]])
                jms = idxs - s
                vm = (jms >= 0) & (jms < n)
                g[vm] -= w[ki] * x[jms[vm]]
            g *= dx
            g += 100.0 * (np.sum(x) - target)
            return g
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def subgradient_refine(h_init, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        iv1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
        iv2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
        grad = np.zeros(n)
        grad[iv1] += (1.0 - h_cur[idxs[iv1] - k_star])
        grad[iv2] -= h_cur[idxs[iv2] + k_star]
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1]:
            h_try = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 200:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_lp, c5_lp = alternating_lp(best_h, n, target, time_limit=min(100, tleft() * 0.3), n_active=50, eps=1e-10)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 150:
        h_bin = make_binary(best_h, target)
        h_bs, c5_bs = fast_binary_search(h_bin, n, time_limit=min(100, tleft() * 0.35), rng=rng)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs

    if tleft() > 100:
        h_bin = make_binary(best_h, target)
        h_ex, c5_ex = exhaustive_binary_search(h_bin, n, time_limit=min(60, tleft() * 0.3), rng=rng)
        if c5_ex < best_score:
            best_score = c5_ex
            best_h = h_ex

    if tleft() > 100:
        h_ref, sc = lbfgs_smooth(best_h, n, target, alphas=[100, 500, 2000, 10000], time_limit=min(80, tleft() * 0.3))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 80:
        h_lp, c5_lp = alternating_lp(best_h, n, target, time_limit=min(60, tleft() * 0.3), n_active=60, eps=1e-11)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 100:
        h_bin = make_binary(best_h, target)
        h_bs, c5_bs = fast_binary_search(h_bin, n, time_limit=min(80, tleft() * 0.4), rng=rng, sa_schedule=False)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs

    if tleft() > 60:
        h_sg, c5_sg = subgradient_refine(best_h, n, target, time_limit=min(40, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 80:
        h_ref, sc = lbfgs_smooth(best_h, n, target, alphas=[5000, 20000, 100000], time_limit=min(60, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    for noise in [0.15, 0.3, 0.5]:
        if tleft() < 70:
            break
        h_pert = project_feasible(best_h + rng.randn(n) * noise, target)
        h_bin = make_binary(h_pert, target)
        h_bs, c5_bs = fast_binary_search(h_bin, n, time_limit=min(50, tleft() * 0.3), rng=rng)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs

    if tleft() > 50:
        h_bin = make_binary(best_h, target)
        h_ex, c5_ex = exhaustive_binary_search(h_bin, n, time_limit=min(40, tleft() * 0.5), rng=rng)
        if c5_ex < best_score:
            best_score = c5_ex
            best_h = h_ex

    if tleft() > 40:
        h_lp, c5_lp = alternating_lp(best_h, n, target, time_limit=min(35, tleft() * 0.7), n_active=60, eps=1e-11)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 30:
        h_bin = make_binary(best_h, target)
        h_bs, c5_bs = fast_binary_search(h_bin, n, time_limit=min(25, tleft() * 0.8), rng=rng, sa_schedule=False)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
