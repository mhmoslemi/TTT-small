import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_lag(h, k, n):
    """Gradient of f_k = dx * sum_j h[j+k]*(1-h[j]) with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g = (1.0 - 2.0 * h) * dx
        return g
    g[k:] += (1.0 - h[:n-k])
    g[:n-k] -= h[k:]
    g *= dx
    return g

def projected_gradient_step(h, n, target, k_active, lr):
    """Take a projected gradient step to reduce the sum of active lag overlaps."""
    g = np.zeros(n)
    for k in k_active:
        g += gradient_lag(h, k, n)
    h_new = h - lr * g
    h_new = project_feasible(h_new, target)
    return h_new

def slsqp_refine(h, n, target, alpha, maxiter=2000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def grad(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        w = np.exp(shifted)
        w = w / w.sum()
        # Weighted gradient
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            wk = w[k_idx]
            if wk < 1e-10:
                continue
            lag = k_idx - (n - 1)
            if lag >= 0:
                g[lag:] += wk * (1.0 - x[:n-lag])
                g[:n-lag] -= wk * x[lag:]
            else:
                lag = -lag
                g[lag:] += wk * (1.0 - x[:n-lag])
                g[:n-lag] -= wk * x[lag:]
        g *= 2.0 / n
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

1. Compute the overlap for each lag

3. For each worst lag k*, compute the "contribution map": c[i] = h[i] * h[i+k*] (this shows where h contributes most to the overlap at lag k*)
4. Perturb h at positions with high contribution (decrease h there) and at positions with low contribution (increase h)

```python
def directed_perturbation(h, n, target, k_worst, strength, rng):
    """Perturb h to reduce the overlap at lag k_worst."""
    k = int(k_worst)
    h_new = h.copy()
    # Contribution map: where h[i] and h[i+k] are both high
    if k == 0:
        contrib = h * h  # h[i]*(1-h[i]) is what we minimize, so high h*h means high contribution to -log term
        # Actually for k=0: f_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # To reduce f_0, we want to increase sum h[i]^2, i.e., make h more 0/1
        # So decrease h where 0 < h < 1 (make it more extreme)
        mask = (h > 0.1) & (h < 0.9)
        h_new[mask] -= strength * np.abs(h[mask] - 0.5)
    else:
        contrib = np.zeros(n)
        # For lag k: f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # To reduce f_k, we want to reduce h[i] where h[i+k] is high, or increase h[i+k] where h[i] is high
        # Contribution of position i to f_k:
        #   as h[i]: contributes h[i] when h[i+k] is high → reduce h[i]
        #   as h[i+k]: contributes -h[i+k] when h[i] is high → increase h[i+k]
        # Net: for i in [0, n-1-k): reduce h[i] by amount proportional to h[i+k]
        #      for i in [k, n-1]: increase h[i] by amount proportional to h[i-k]
        # But we need to maintain the sum constraint, so we need to remove mass where we add it elsewhere
        
        # Simpler: just reduce h where the "product" is high
        # For i in [0, n-1-k): product = h[i] if we think of it as h[i]*(1-h[i+k])
        #   Actually the overlap at lag k is sum h[i]*(1-h[i+k]), so high h[i] and low h[i+k] contributes
        #   To reduce: either decrease h[i] or increase h[i+k]
        
        # Let's use a simple approach: subtract strength * h[i+k] from h[i] for i in [0, n-1-k)
        # and add it to h[i+k] for i in [k, n-1)
        # This is like shifting mass from i to i+k
        
        shift_amount = strength * h[:n-k].copy()
        shift_amount = np.minimum(shift_amount, h[:n-k] * 0.5)  # don't make negative
        h_new[:n-k] -= shift_amount
        h_new[k:n] += np.concatenate([shift_amount, np.zeros(k)])
    
    h_new = project_feasible(h_new, target)
    return h_new
```

    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    shifted = np.clip(shifted, -700.0, 0.0)

    _g_target = target
    _g_alpha = alpha

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def projected_gradient(h, n, target, k_active, lr):
    """Take a projected gradient descent step on the sum of active lag overlaps."""
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_active:
        k = int(k)
        if k == 0:
            g += dx * (1.0 - 2.0 * h)
        elif k > 0:
            g[:n - k] += dx * (1.0 - h[k:n])
            g[k:] += dx * (-h[:n - k])
        else:
            lp = -k
            g[lp:] += dx * (1.0 - h[:n - lp])
            g[:n - lp] += dx * (-h[lp:n])
    h_new = h - lr * g
    h_new = project_feasible(h_new, target)
    return h_new

def mgd_refine(h, n, target, n_iters=5000, lr_init=0.05, lr_decay=0.999, n_active=5, deadline=None):
    """Multiplicative/gradient descent refinement focusing on active lags."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    dx = 2.0 / n
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr  # all lags
        top_idx = np.argsort(over)[::-1][:n_active]
        h_new = projected_gradient(h_cur, n, target, top_idx, lr)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
            # Keep lr or increase slightly
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= lr_decay
            if lr < 1e-8:

    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _lse_grad(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    w = w / wsum
    n = _g_n
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        lag = m - (n - 1)
        if lag == 0:
            g += wm * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - lag] += wm * dx * (1.0 - x[lag:n])
            g[lag:] += wm * dx * (-x[:n - lag])
        else:
            lp = -lag
            g[lp:] += wm * dx * (1.0 - x[:n - lp])
            g[:n - lp] += wm * dx * (-x[lp:n])
    return g

def _cons_eq_fun(x):
    return np.sum(x) - _g_target

def _cons_eq_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h, n, target, alpha, maxiter=2000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def gradient_active_lags(h, n, k_list):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_list:
        k = int(k)
        if k == 0:
            g += dx * (1.0 - 2.0 * h)
        elif k > 0:
            g[:n - k] += dx * (1.0 - h[k:n])
            g[k:] += dx * (-h[:n - k])
        else:
            lp = -k
            g[lp:] += dx * (1.0 - h[:n - lp])
            g[:n - lp] += dx * (-h[lp:n])
    return g

def mgd_refine(h, n, target, n_iters, lr_init, lr_decay, n_active, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    stale = 0
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = gradient_active_lags(h_cur, n, top_idx)
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
            lr = min(lr * 1.005, 0.15)
        else:
            lr *= lr_decay
            if lr < 1e-9:
                break
        if stale > 500:
            break
    return best_h, best_score

def sa_search(h_init, n, target, n_iters, t_init, t_min, rng, deadline=None,
