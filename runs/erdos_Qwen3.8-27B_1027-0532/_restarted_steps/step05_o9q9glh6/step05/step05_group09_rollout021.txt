def arc_subgradient(h0, target, K, max_iters, deadline):

    dx = 2.0 / n

    best_x = x.copy()
    best_score = compute_c5_score(x)

        if time.time() > deadline:
            break
        
        corr = compute_corr(x)
        top_K = np.argsort(corr)[::-1][:K]
        corr_top = corr[top_K]
        
        # Smoothed max gradient (equal weights on top K)
        grad = np.zeros(n)
        for k in top_K:
            k = int(k)
            if k == 0:
                grad += (1.0 - 2.0 * x)
            else:
                grad[:n-k] += (1.0 - x[k:n])
                grad[k:] -= x[:n-k]
        grad *= dx / K  # average gradient
        
        # Step size: use a decreasing sequence
        step = 0.01 / np.sqrt(it + 1)
        
        x_new = project_feasible(x - step * grad, target)
        sc = compute_c5_score(x_new)
        
        if sc < best_score:
            best_score = sc
            best_x = x_new.copy()
            x = x_new
        # else: keep x unchanged (no improvement)
    
    return best_x, best_score

def coord_descent(h0, target, deadline, max_sweeps=500):
    n = len(h0)
    x = project_feasible(h0.copy(), target)
    best_score = compute_c5_score(x)
    best_x = x.copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for idx in order:
            if time.time() > deadline:
                break
            old_val = x[idx]
            # Try a few candidate values
            candidates = [0.0, 1.0, old_val * 0.5, old_val * 1.5, 
                         min(1.0, old_val + 0.1), max(0.0, old_val - 0.1)]
            candidates = [c for c in candidates if 0.0 <= c <= 1.0 and abs(c - old_val) > 1e-10]

import numpy as np
import time

def compute_c5_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:].astype(float)

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def gradient_corr_k(h, k):
    """Gradient of corr_k with respect to h."""
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[:n-k] += (1.0 - h[k:n])
    g[k:] -= h[:n-k]
    return g

            # g[k] = (1-h[idx+k])*[idx < n-k] - h[idx-k]*[idx >= k]
            # Vectorized:
            # For k in 1..min(idx, n-1): idx >= k, so -h[idx-k] contributes
            # For k in 1..n-1-idx: idx < n-k, so (1-h[idx+k]) contributes
            # For k where both conditions hold: both terms contribute
            
            # idx < n-k means k < n-idx, i.e., k <= n-1-idx
            # idx >= k means k <= idx
            
            max_k_plus = min(n - 1 - idx, n - 1)  # max k where idx < n-k
            max_k_minus = min(idx, n - 1)  # max k where idx >= k
            
            if max_k_plus >= 1:
                ks = np.arange(1, max_k_plus + 1)
                g[ks] += (1.0 - x[idx + ks])
            if max_k_minus >= 1:
                ks = np.arange(1, max_k_minus + 1)
                g[ks] -= x[idx - ks]
            
            # For k=0: special handling
            # corr[0] = sum h[i]*(1-h[i])
            # new_corr0 = corr[0] + new*(1-new) - old*(1-old)
            # = corr[0] + (new-old) - (new² - old²)
            # = corr[0] + (new-old)(1 - new - old)
            
            # Evaluate all trials
            deltas = grid - old_val  # shape (n_grid,)
            
            # For k >= 1: new_corr[k] = corr[k] + deltas * g[k]
            # candidate_max_k = max_k (corr[k] + delta * g[k]) for each delta
            # Vectorized: candidates = corr[1:][None, :] + deltas[:, None] * g[1:][None, :]
            candidates = corr[1:][None, :] + deltas[:, None] * g[1:][None, :]
            candidate_maxes_k = candidates.max(axis=1)  # shape (n_grid,)
            
            # For k=0: new_corr0 = corr[0] + deltas * (1 - grid - old_val)
            candidate_max0 = corr[0] + deltas * (1.0 - grid - old_val)
            
            # Total candidate max
            total_maxes = np.maximum(candidate_maxes_k, candidate_max0)
            
            # Find best trial
            best_i = np.argmin(total_maxes)
            best_trial = grid[best_i]
            best_new_max = total_maxes[best_i]
            
            if best_new_max < current_max - 1e-12:
                x[idx] = best_trial
                x = project_feasible(x, target)
                corr = compute_corr(x)
                current_max = float(np.max(corr))
                if current_max * dx < best_score:
                    best_score = current_max * dx
                    best_x = x.copy()
                any_improved = True
        
        if not any_improved:
            break
    
    return best_x, best_score
