```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Fix if still not exact
    while int(h_r.sum()) != target_sum:
        if int(h_r.sum()) > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
    return h_r.astype(float)
```

```python
def coord_descent_step(h, i, corr):
    """Find optimal h[i] given current h and correlation."""
    n = len(h)
    old = h[i]
    
    # a_m = (1-h[i+k_m]) - h[i-k_m] where k_m = m - (n-1)
    # For m in 0..2n-2, k = m - (n-1) ranges from -(n-1) to n-1
    # i+k is valid if 0 <= i+k < n, i.e., -i <= k < n-i
    # i-k is valid if 0 <= i-k < n, i.e., i-n < k <= i, i.e., k > i-n and k <= i
    
    a = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        val = 0.0
        if 0 <= i + k < n:
            val += (1.0 - h[i + k])
        if 0 <= i - k < n:
            val -= h[i - k]
        a[m] = val
    
    # Now new_corr[m] = corr[m] + t * a[m] where t = new_h_i - old
    # Minimize max_m new_corr[m] over t in [-old, 1-old]
    
    t_lo = -old
    t_hi = 1.0 - old
    
    # Evaluate at boundaries
    val_lo = np.max(corr + t_lo * a)
    val_hi = np.max(corr + t_hi * a)
    
    # Find the top few m at t=0
    top_m = np.argsort(corr)[-15:]
    
    best_t = 0.0
    best_val = float(np.max(corr))
    
    candidates = [t_lo, t_hi, 0.0]
    
    # Check intersections of top lines
    for p in range(len(top_m)):
        for q in range(p+1, len(top_m)):
            mp, mq = top_m[p], top_m[q]
            ap, aq = a[mp], a[mq]
            if abs(ap - aq) < 1e-12:
                continue
            t_int = (corr[mq] - corr[mp]) / (ap - aq)
            if t_lo <= t_int <= t_hi:
                candidates.append(t_int)
    
    for t in candidates:
        t = np.clip(t, t_lo, t_hi)
        v = float(np.max(corr + t * a))
        if v < best_val:
            best_val = v
            best_t = t
    
    return best_t, best_val
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    while int(h_r.sum()) != target_sum:
        if int(h_r.sum()) > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
    return h_r.astype(float)

def coord_descent_continuous(h, max_passes, time_limit, start_time, target_sum):
    n = len(h)
    h = h.copy().astype(float)
    h = project_feasible(h, float(target_sum))
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    n2m1 = 2 * n - 1
    h_rev = h[::-1]
    
    for p in range(max_passes):
        if time.time() - start_time > time_limit:
            break
        
        improved = False
        old_max = float(np.max(corr))
        
        for i in range(n):
            old = h[i]
            t_lo = -old
            t_hi = 1.0 - old
            if t_hi - t_lo < 1e-12:
                continue
            
            # Compute a vector: a[m] = (1-h[i+k_m]) - h[i-k_m] with out-of-range = 0
            a = np.zeros(n2m1)
            a[n-1-i : 2*n-1-i] += 1.0 - h
            h_rev = h[::-1]
            a[i : i+n] -= h_rev
            
            # Ternary search for optimal t
            lo = t_lo
            hi = t_hi
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = float(np.max(corr + m1 * a))
                f2 = float(np.max(corr + m2 * a))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            t_opt = (lo + hi) / 2.0
            new_val = float(np.max(corr + t_opt * a))
            
            if new_val < old_max - 1e-8:
                h[i] = old + t_opt
                h_rev = h[::-1]
                corr += t_opt * a
                old_max = new_val
                improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, float(target_sum))
    final_corr = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(final_corr)) * (2.0 / n)
```

```python
def greedy_binary(h, target, max_iters, time_limit, start_time, scan_width):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')

    best_h = h.copy()

        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(np.max(corr))
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates for i (ones that could go to 0) and j (zeros that could go to 1)
        # Prioritize those at positions related to k_star
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        
        if not i_cand or not j_cand:
            # Try random
            i_cand = list(rng.choice(ones, min(10, len(ones)), replace=False))
            j_cand = list(rng.choice(zeros, min(10, len(zeros)), replace=False))
        
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        
        best_improve = 0
        best_i = -1
        best_j = -1
        
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                # Estimate new max (approximate: just check top few shifts)
                # delta at shift k:
                # - (1-h[i+k]) + h[i-k] + (1-h[j+k]) - h[j-k]
                # For k = k_star:
                dk = 0.0
                if 0 <= i + k_star < n:
                    dk -= (1.0 - h[i + k_star])
                if 0 <= i - k_star < n:
                    dk += h[i - k_star]
                if 0 <= j + k_star < n:
                    dk += (1.0 - h[j + k_star])
                if 0 <= j - k_star < n:
                    dk -= h[j - k_star]
                
                new_max_est = max_val + dk
                # Also check a few other top shifts
                improve = max_val - new_max_est
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(np.max(corr))
            if sc < best_max:
                best_max = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_max * (2.0 / n)
```

                improved_count += 1
        
        if improved_count == 0:
            break
        if pass_old_max - old_max < 1e-8:
            break
    
    h = project_feasible(h, float(target_sum))
    final_corr = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(final_corr)) * (2.0 / n)

def greedy_binary_fast(h, target, max_iters, time_limit, start_time, scan_width, rng):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(np.max(corr))
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 500:
            break
        
        max_val = float(np.max(corr))
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        
        if not i_cand:
            i_cand = [int(x) for x in rng.choice(ones, min(10, len(ones)), replace=False)]
        if not j_cand:
            j_cand = [int(x) for x in rng.choice(zeros, min(10, len(zeros)), replace=False)]
        
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                dk = 0.0
                if 0 <= i + k_star < n:
                    dk -= (1.0 - h[i + k_star])
                if 0 <= i - k_star < n:
                    dk += h[i - k_star]
                if 0 <= j + k_star < n:
                    dk += (1.0 - h[j + k_star])
                if 0 <= j - k_star < n:
                    dk -= h[j - k_star]
                new_max_est = max_val + dk
                improve = max_val - new_max_est
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(np.max(corr))
            if sc < best_max:
                best_max = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_max * (2.0 / n)

def sa_binary(h, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        h_new = h.copy()
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new[i] = 0
        h_new[j] = 1
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(np.max(corr_new)) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
