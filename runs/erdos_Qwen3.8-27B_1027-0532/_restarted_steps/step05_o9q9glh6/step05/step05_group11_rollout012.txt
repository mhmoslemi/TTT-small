def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, threshold_frac=0.95):
    """Get lags where overlap is above threshold_frac * max."""
    overlaps = compute_overlaps(h)
    n = len(h)
    n_half = n - 1
    max_ov = np.max(overlaps)
    active = np.where(overlaps >= threshold_frac * max_ov)[0]
    lags = active - n_half
    return lags

def build_and_solve_lp(h_current, n, target, lags, reg, dx, solver='SCS'):
    """Build and solve the LP subproblem."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h_current
    n_half = n - 1
    for k in lags:
        lag_idx = k + n_half
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h_current)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
        except Exception:
            return h_current.copy()
    if h_var.value is None:
        return h_current.copy()
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def smooth_logsumexp_obj(h, alpha):
    """Compute smooth approximation to max overlap."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    corr = corr * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return smooth_max

def smooth_grad(h, alpha):
    """Compute gradient of smooth log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    
    # Gradient of overlap at lag k with respect to h[j]
    # O(k) = dx * sum_i h[i]*(1-h[i+k])
    # dO(k)/d(h[j]) = dx * [(1-h[j+k])*1_{j+k valid} - h[j-k]*1_{j-k valid}]
    
    grad = np.zeros(n)
    n_half = n - 1
    for i, k in enumerate(range(-n_half, n_half+1)):
        if weights[i] < 1e-10:
            continue
        # Term 1: h[j] appears as h[i] in the sum, contribution = (1-h[j+k])*dx
        # Term 2: h[j] appears as h[i+k] in the sum, contribution = -h[j-k]*dx
        if k >= 0:
            # j+k < n for term 1
            idx1 = np.arange(0, n - k)
            grad[idx1] += weights[i] * dx * (1.0 - h[idx1 + k])
            # j-k >= 0 for term 2
            idx2 = np.arange(k, n)
            grad[idx2] -= weights[i] * dx * h[idx2 - k]
        else:
            # j+k >= 0 for term 1 (k is negative, so j >= -k)
            idx1 = np.arange(-k, n)
            grad[idx1] += weights[i] * dx * (1.0 - h[idx1 + k])
            # j-k < n for term 2 (j < n+k = n-k_abs)
            idx2 = np.arange(0, n + k)
            grad[idx2] -= weights[i] * dx * h[idx2 - k]
    
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, threshold_frac=0.98, max_lags=None):
    overlaps = compute_overlaps(h)
    n = len(h)
    n_half = n - 1
    max_ov = np.max(overlaps)
    active = np.where(overlaps >= threshold_frac * max_ov)[0]
    if max_lags is not None and len(active) > max_lags:
        top_idx = np.argsort(overlaps)[::-1][:max_lags]
        active = top_idx
    lags = active - n_half
    return lags

def build_and_solve_lp(h_current, n, target, lags, reg, dx):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h_current
    n_half = n - 1
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(cp.multiply(coeff, h_var).sum() <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h_current)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=200000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except Exception:
                return h_current.copy(), None
    if h_var.value is None:
        return h_current.copy(), None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), t_var.value

def alternating_lp_refined(h_init, n, target, max_iter, time_limit,
                           regret_frac=0.98, damping=0.93, reg=1e-4,
                           max_lags=None, reg_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_reg = reg
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break

        lags = get_active_lags(h, threshold_frac=regret_frac, max_lags=max_lags)
        if len(lags) == 0:
            overlaps = compute_overlaps(h)
            n_half = n - 1
            top_idx = np.argsort(overlaps)[::-1][:100]
            lags = top_idx - n_half

        h_new, t_val = build_and_solve_lp(h, n, target, lags, current_reg, dx)
        h_update = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h_update, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        current_reg *= reg_decay
        current_reg = max(current_reg, 1e-10)

    return best_h, best_c5

def smooth_grad_fast(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights_sum = weights.sum()
    if weights_sum < 1e-300:
        return np.zeros(n)
    weights = weights / weights_sum

    n_half = n - 1
    sig_thresh = max(1e-10, 1e-15 * len(h))
    significant = np.where(weights > sig_thresh)[0]

    grad = np.zeros(n)
    for idx in significant:
        k = idx - n_half
        w = weights[idx]
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[:n-k]
        else:
            absk = -k
            if n - absk > 0:
                grad[absk:] += w * (1.0 - h[:n-absk])
                grad[:n-absk] -= w * h[absk:]
    grad *= dx
    return grad

def nesterov_smooth_descent(h_init, n, target, time_limit,
                            alpha_start=20.0, alpha_end=20000.0,
                            n_alphas=8, steps_per_alpha=2000):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    log_alphas = np.linspace(np.log(alpha_start), np.log(alpha_end), n_alphas)

    for ai in range(n_alphas):
        if time.time() - start > time_limit:
            break
        alpha = np.exp(log_alphas[ai])
        x = h.copy()
        x_prev = h.copy()
        v = np.zeros(n)

        for step in range(steps_per_alpha):
            if time.time() - start > time_limit:
                break
            grad = smooth_grad_fast(x, alpha)
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-15:
                break

            f_x = np.max(compute_overlaps(x))
            f_best = np.max(compute_overlaps(best_h))
            lr = min(1.0, (f_x - f_best + 0.05) / (g_norm ** 2 + 1e-12))
            lr = max(lr, 1e-6)

            v = 0.9 * v + lr * grad
            x_try = x - v
            x_try = project_feasible(x_try, target)
            c5_try = compute_c5(x_try)

            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = x_try.copy()
                x = x_try
            else:
                v = 0.5 * v

    return best_h, best_c5

def spectral_refine(h, n, target, n_modes=15, n_phases=12):
    dx = 2.0 / n
    x = np.linspace(0, 2, n, endpoint=False)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for mode in range(1, n_modes + 1):
        freq = mode * np.pi
        for phase in np.linspace(0, 2.0 * np.pi, n_phases, endpoint=False):
            for amp in [0.005, 0.002, 0.001]:
                pert = amp * np.cos(freq * x + phase)
                h_try = project_feasible(h + pert, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try
                    h = h_try
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
