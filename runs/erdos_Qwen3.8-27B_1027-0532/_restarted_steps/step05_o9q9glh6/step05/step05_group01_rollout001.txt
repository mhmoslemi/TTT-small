def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

def lp_round(h, n, target, time_limit, max_iters=50):
    import cvxpy as cp
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)

        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        threshold = c_max - 0.02 * c_max
        active_k = np.where(corr >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([np.argmax(corr)])
        if len(active_k) > 20:
            top_idx = np.argsort(corr)[-20:]
            active_k = top_idx

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

        for s_idx in active_k:
            k = int(s_idx) - mid
            grad_shift = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_shift[v1] += (1.0 - h[idxs[v1] - k])
            grad_shift[v2] -= h[idxs[v2] + k]
            linear_corr = corr[s_idx] + float(np.dot(grad_shift, h_var - h))
            constraints.append(linear_corr <= t_var)

        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break

        if h_var.value is None:
            break
        h_new = project_feasible(h_var.value, target)
        sc = compute_c5(h_new)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_new.copy()
            h = h_new
        else:
            break

    return best_h, best_c5


def spectral_perturbation(h, n, target, rng, n_modes=5, scale=0.05):
    idxs = np.arange(n)
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = rng.randint(1, max(2, n // (2 * m)))
        phase = rng.uniform(0, 2 * np.pi)
        amp = scale / np.sqrt(m)
        pert += amp * np.sin(2 * np.pi * freq * idxs / n + phase)
    h_new = h + pert
    return project_feasible(h_new, target)
