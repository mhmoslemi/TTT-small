def build_full_from_gamma(gamma):
    """Expand gamma (303 points) to full 606-point function with reflection"""
    full = np.zeros(606)
    for i in range(303):
        full[i] = gamma[i]
    for i in range(303,606):
        full[i] = gamma[605 - i]
    return full

def compute_c5_full(gamma):
    """Compute overlap integral using symmetry"""
    full = build_full_from_gamma(gamma)
    corr = np.correlate(full, 1.0 - full, mode='full')
    return float(np.max(corr)) * (2.0/606)

def project_to_feasible(gamma, target_sum):
    """Ensure gamma satisfies gamma constraint"""
    n = len(gamma)
    diff = target_sum - sum(gamma)
    if abs(diff) < 1e-4:
        return gamma
    # Distribute adjustment proportionally
    new_gamma = gamma + 1.0 / n * diff
    # Clip and recompute if necessary
    new_gamma = np.clip(new_gamma, 0.0, 1.0)
    # Ensure l1 constraint is met
    new_sum = sum(new_gamma)
    while abs(new_sum - target_sum) > 1e-4:
        # Moving target sums shifts about the sum constraint
        move_amount = (target_sum - new_sum) / n
        new_gamma += move_amount
        new_gamma = np.clip(new_gamma, 0.0, 1.0)
        new_sum = sum(new_gamma)
    return new_gamma
