import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def active_subgrad_step(h, target, top_m, temperature, deadline):
    """Take a step using subgradient of the active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    
    # Exponential weighting on top-m lags
    sorted_idx = np.argsort(corr)[::-1][:top_m]
    diffs = corr[sorted_idx] - c_max
    weights = np.exp(diffs / temperature)
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for wi, ki in zip(weights, sorted_idx):
        ki = int(ki)
        jp = np.arange(0, n - ki)
        grad[jp] += wi * (1.0 - h[jp + ki])
        jn = np.arange(ki, n)
        grad[jn] -= wi * h[jn - ki]
    grad *= dx
    
    return grad, c_max

def fista_exact(h0, target, top_m, temperature, max_iters, deadline, lr_base=0.1):
    """FISTA with active-set subgradient."""
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = lr_base
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        grad, c_max = active_subgrad_step(y_clipped, target, top_m, temperature, deadline)
        x_new = project_feasible(y_clipped - step * grad, target)
        
        # Simple backtracking
        for _ in range(30):
            obj_new = compute_c5(x_new)
            obj_y = compute_c5(y_clipped)
            if obj_new <= obj_y + 1e-15:
                break
            step *= 0.7
            x_new = project_feasible(y_clipped - step * grad, target)
        step = min(step / 0.7, lr_base * 3.0)
        
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:

    n = len(h0)

    x_prev = x.copy()
    y = x.copy()
    t = 1.0

def simulated_annealing(h0, target, deadline, rng, n_rounds=5000):
    """Simulated annealing with binary-like moves."""
    n = len(h0)
    h_cur = project_feasible(h0, target).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    T = 0.02
    T_min = 0.001
    cooling = (T_min / T) ** (1.0 / n_rounds)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        T *= cooling
        
        # Propose a random move: perturb a random subset of entries
        n_perturb = rng.randint(1, 6)
        idx = rng.choice(n, n_perturb, replace=False)
        h_new = h_cur.copy()
        for i in idx:
            h_new[i] = rng.uniform(0, 1)
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        delta = sc_new - best_score
        
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-8)):
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
    
    return best_h, best_score

def coord_refine(h, target, deadline, max_sweeps=50):
    """Coordinate descent with more trial values."""
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            # Try several trial values
            best_trial = old_val
            best_trial_score = cur_score
            trials = [0.0, 1.0, 0.5, old_val * 0.5, old_val * 1.5, 
                      old_val - 0.1, old_val + 0.1, old_val - 0.05, old_val + 0.05]
            trials = [t for t in trials if 0.0 <= t <= 1.0 and abs(t - old_val) > 1e-10]
            for trial in trials:
                h_cur[idx] = trial
                h_tmp = project_feasible(h_cur, target)
                sc = compute_c5(h_tmp)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = trial
                h_cur[idx] = old_val
            
            if best_trial != old_val:
                h_cur[idx] = best_trial
                h_cur = project_feasible(h_cur, target)
                cur_score = best_trial_score
                improved = True
        
        if not improved:
            break
    
    return h_cur, cur_score

def structured_init(n, mode):
    """Generate structured initializations."""
    target = n / 2.0
    x = np.linspace(0, 2, n)
    
    if mode == 0:  # Alternating

    elif mode == 1:  # Sine wave

    elif mode == 2:  # Multiple frequencies
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.2 * np.sin(2 * np.pi * 7 * x / 2)
    elif mode == 3:  # Triangle wave
        h = 1.0 - np.abs(2 * (x / 2 - 0.5) * 4 - 1)
        h = np.clip(h, 0, 1)
    elif mode == 4:  # Block structure
        h = np.zeros(n)
        block = n // 4
        for i in range(0, n, block * 2):
            h[i:i+block] = 1.0
    elif mode == 5:  # Random
        h = np.random.RandomState(mode).uniform(0, 1, n)
    else:
        h = np.ones(n) * 0.5
    
    return project_feasible(h, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_grad_fft(y, alpha, n):
    dx = 2.0 / n
    corr = compute_corr(y)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    w_flipped = weights[::-1].copy()
    v = 1.0 - y
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)

def topm_grad(h, n, top_m, temperature):
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    sorted_idx = np.argsort(corr)[::-1][:top_m]
    diffs = corr[sorted_idx] - c_max
    weights = np.exp(diffs / temperature)
    weights /= weights.sum()
    grad = np.zeros(n)
    for wi, ki in zip(weights, sorted_idx):
        ki = int(ki)
        jp = np.arange(0, n - ki)
        grad[jp] += wi * (1.0 - h[jp + ki])
        jn = np.arange(ki, n)
        grad[jn] -= wi * h[jn - ki]
    grad *= dx
    return grad, c_max

def coord_descent_v2(h, target, deadline, max_sweeps=100):

    h_cur = project_feasible(h.copy(), target)

    rng = np.random.RandomState(777)

            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            if time.time() - sweep * 0.001 > deadline - 1:
                break
            old_val = h_cur[idx]
            best_score_local = cur_score
            best_val_local = old_val

            if old_val < 1e-8:
                trials = [0.01, 0.05, 0.1, 0.2, 0.5]
            elif old_val > 1.0 - 1e-8:
                trials = [0.99, 0.95, 0.9, 0.8, 0.5]

                trials = [0.0, 1.0, 0.5,

                          max(0, old_val - 0.07), max(1, old_val + 0.07),
                          max(0, old_val * 0.6), min(1, old_val * 1.4)]
                trials = [t for t in trials if abs(t - old_val) > 1e-7]

            for tv in trials:
                h_cur[idx] = tv
                h_tmp = project_feasible(h_cur, target)
                sc = compute_c5(h_tmp)

                    best_val_local = tv

                h_cur[idx] = best_val_local
                h_cur = project_feasible(h_cur, target)
                cur_score = best_score_local
                improved = True

        if not improved:
            break

    return h_cur, cur_score

    # Phase 3: Top-m subgradient method (200s)
    if tleft() > 200:
        configs = [(2, 0.005, 0.06), (3, 0.01, 0.05), (5, 0.02, 0.04), (3, 0.003, 0.05)]
        for top_m, temp, lr in configs:
            if tleft() < 30:
                break
            h_new, sc_new = fista_topm(best_h, target, top_m, temp, 10000, start_time + 25, lr)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()

    # Phase 4: Multi-scale refinement (100s)
    if tleft() > 100:
        h_new, sc_new = multi_scale_opt(best_h, target, start_time + 70)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if tleft() > 40:
            h_new, sc_new = lse_fista(best_h, target, 50000, 10000, start_time + 35, 0.03)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()

    # Phase 5: High-alpha LSE FISTA (200s)
    for a in [200000, 500000, 1000000]:
        if tleft() < 40:
            break
        h_new, sc_new = lse_fista(best_h, target, a, 15000, start_time + 35, 0.02)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    # Phase 6: Simulated annealing (150s)
    if tleft() > 150:
        h_new, sc_new = simulated_anneal(best_h, target, start_time + 120, seed, n_rounds=4000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if tleft() > 40:
            h_new, sc_new = lse_fista(best_h, target, 100000, 10000, start_time + 35, 0.02)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()

    # Phase 7: Coordinate descent refinement (150s)
    if tleft() > 150:
        h_new, sc_new = coord_descent_v2(best_h, target, start_time + 120, max_sweeps=80)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    # Phase 8: Final alternating FISTA + topm + coord (150s)
    if tleft() > 150:
        for round_i in range(3):
            if tleft() < 45:
                break
            h_new, sc_new = lse_fista(best_h, target, 500000, 8000, start_time + 15, 0.015)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
            if tleft() < 30:
                break
            h_new, sc_new = fista_topm(best_h, target, 2, 0.002, 5000, start_time + 12, 0.04)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
            if tleft() < 20:
                break
            h_new, sc_new = coord_descent_v2(best_h, target, start_time + 12, max_sweeps=30)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()

    # Phase 9: Final polish
    if tleft() > 30:
        h_new, sc_new = lse_fista(best_h, target, 2000000, 5000, start_time + 8, 0.01)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
    if tleft() > 15:
        h_new, sc_new = coord_descent_v2(best_h, target, start_time + 10, max_sweeps=20)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
