import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_objective(h, alpha, dx):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max * dx + np.log(np.sum(np.exp(shifted))) / alpha * dx)


def lse_gradient(h, alpha, dx, n):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        return np.zeros(n), corr
    w = weights / total
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    top_count = min(len(w), 20)
    top_idx = np.argpartition(w, -top_count)[-top_count:]
    for si in top_idx:
        wi = w[si]
        if wi < 1e-15:
            continue
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad, corr


def nesterov_projected_grad(h_init, n, target, alpha_schedule, iters_per_alpha,
                            base_lr, time_limit, momentum_decay=0.95):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    h_prev = h.copy()

    for alpha in alpha_schedule:
        if time.time() - start > time_limit:
            break
        for it in range(iters_per_alpha):
            if time.time() - start > time_limit:
                break
            grad, corr = lse_gradient(h, alpha, 2.0 / n, n)
            v = momentum_decay * v + grad
            lr = base_lr / (1.0 + 0.01 * it)
            h_new = project_feasible(h - lr * v, target)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
            v = 0.9 * v
            h_prev = h
            h = h_new
    return best_h, best_score


def lbfgsb_round(h_init, n, target, alphas, maxiter, ftol, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_init.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

        def obj(x):
            return lse_objective(x, alpha, dx)

        def obj_grad(x):
            g, _ = lse_gradient(x, alpha, dx, n)
            return g

        def con_fun(x):
            return np.sum(x) - target

        def con_jac(x):
            return np.ones(n)

        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:

    return best_h, best_score

            best_score = score
            best_h = h.copy()

        top_mask = corr >= c_max - 0.05
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(corr)])
        if len(top_idx) > n_top:
            top_idx = np.argpartition(corr, -n_top)[-n_top:]

        grad = np.zeros(n)
        for si in top_idx:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        lr = 0.05 / (1.0 + 0.001 * it)
        h_new = project_feasible(h - lr * grad, target)
        h = h_new

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def diverse_restart(h_base, n, target, rng, noise_scale, alpha_sched, iters, base_lr, time_limit):
    h_pert = h_base + rng.randn(n) * noise_scale
    h_pert = project_feasible(h_pert, target)
    h_opt, score = nesterov_projected_grad(h_pert, n, target, alpha_sched,
                                           iters, base_lr, time_limit)
    return h_opt, score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Nesterov accelerated projected gradient with geometric alpha
    if tleft() > 200:
        alpha_sched = [5, 10, 20, 40, 80, 160, 320, 640, 1280]
        h_opt, sc = nesterov_projected_grad(best_h, n, target, alpha_sched,
                                            iters_per_alpha=200,
                                            base_lr=0.04,
                                            time_limit=min(120, tleft() * 0.8))
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 2: L-BFGS-B polish
    if tleft() > 150:
        h_pol, sc = lbfgsb_round(best_h, n, target,
                                 alphas=[200, 500, 1000, 2000, 5000],
                                 maxiter=800, ftol=1e-16,
                                 time_limit=min(100, tleft() * 0.7))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Balance top shifts
    if tleft() > 150:
        h_bal, sc = balance_top_shifts(best_h, n, target,
                                       time_limit=min(80, tleft() * 0.6),
                                       n_top=5, n_iter=5000)
        if sc < best_score:
            best_score = sc
            best_h = h_bal

    # Phase 4: More Nesterov with different hyperparams
    if tleft() > 150:
        alpha_sched2 = [50, 100, 200, 400, 800, 1600, 3200]
        h_opt2, sc2 = nesterov_projected_grad(best_h, n, target, alpha_sched2,
                                              iters_per_alpha=150,
                                              base_lr=0.02,
                                              time_limit=min(100, tleft() * 0.7),
                                              momentum_decay=0.98)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_opt2

    # Phase 5: L-BFGS-B again
    if tleft() > 100:
        h_pol2, sc = lbfgsb_round(best_h, n, target,
                                  alphas=[1000, 3000, 10000],
                                  maxiter=600, ftol=1e-16,
                                  time_limit=min(80, tleft() * 0.7))
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    # Phase 6: Block coordinate descent
    if tleft() > 100:
        h_bcd, sc = block_cd(best_h, n, target,
                             time_limit=min(60, tleft() * 0.6),
                             block_size=15, max_iters=3000)
        if sc < best_score:
            best_score = sc
            best_h = h_bcd

    # Phase 7: Diverse restarts
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.05, 0.08, 0.12]:
        if tleft() < 80:
            break
        h_rs, sc = diverse_restart(best_h, n, target, rng, noise_scale,
                                   alpha_sched=[20, 50, 100, 200, 500, 1000],
                                   iters=100, base_lr=0.03,
                                   time_limit=min(40, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_rs

    # Phase 8: Final balance + polish
    if tleft() > 80:
        h_bal2, sc = balance_top_shifts(best_h, n, target,
                                        time_limit=min(40, tleft() * 0.5),
                                        n_top=8, n_iter=3000)
        if sc < best_score:
            best_score = sc
            best_h = h_bal2

    if tleft() > 60:
        h_pol3, sc = lbfgsb_round(best_h, n, target,
                                  alphas=[2000, 5000, 20000],
                                  maxiter=500, ftol=1e-16,
                                  time_limit=min(50, tleft() * 0.7))
        if sc < best_score:
            best_score = sc
            best_h = h_pol3

    # Phase 9: Try higher resolution
    if tleft() > 250:
        for n_try in [500, 600, 800]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hi, sc_hi = nesterov_projected_grad(h_interp, n_try, target_try,
                                                  alpha_sched=[20, 50, 100, 200, 500],
                                                  iters_per_alpha=100,
                                                  base_lr=0.03,
                                                  time_limit=min(60, tleft() * 0.4))
            if tleft() > 60:
                h_hi2, sc_hi2 = lbfgsb_round(h_hi, n_try, target_try,
                                             alphas=[500, 2000, 5000],
                                             maxiter=500, ftol=1e-16,
                                             time_limit=min(50, tleft() * 0.5))
                if sc_hi2 < sc_hi:
                    h_hi = h_hi2
                    sc_hi = sc_hi2
            if sc_hi < best_score:
                best_score = sc_hi
                best_h = h_hi
                n = n_try
                target = target_try

    # Phase 10: Final Nesterov + L-BFGS-B at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin, sc_fin = nesterov_projected_grad(best_h, n_use, target_use,
                                                alpha_sched=[100, 300, 1000, 3000],
                                                iters_per_alpha=100,
                                                base_lr=0.02,
                                                time_limit=min(40, tleft() * 0.5))
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc_final = lbfgsb_round(best_h, n_use, target_use,
                                         alphas=[5000, 10000, 50000],
                                         maxiter=400, ftol=1e-16,
                                         time_limit=min(25, tleft() * 0.8))
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
