    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def compute_top_grad(h, top_shifts):
    n = len(h)
    grad = np.zeros(n)
    for k in top_shifts:
        if k >= 0:
            if k < n:
                js = np.arange(k, n)
                grad[js] += (1.0 - h[js - k])
                js2 = np.arange(0, n - k)
                grad[js2] += -h[js2 + k]
        else:
            ka = -k
            if ka < n:
                js = np.arange(0, n - ka)
                grad[js] += (1.0 - h[js + ka])
                js2 = np.arange(ka, n)
                grad[js2] += -h[js2 - ka]
    return grad


def targeted_descend(h, target_sum, max_iters, lr0, decay, top_m,
                     time_limit=None, start_time=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 3000

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        top_order = np.argsort(corr[top_idx])[::-1]
        top_shifts = (top_idx[top_order]) - (n - 1)

        grad = compute_top_grad(h, top_shifts) / float(top_m)
        sum_dev = np.sum(h) - target_sum
        grad += sum_dev

        lr = lr0 * (decay ** it)
        if lr < 1e-6:
            lr = 1e-6

        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1

        h = h_new
        corr = corr_new

    return best_h, best_c5


def _glb_f(x):
    n = _glb_n
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    corr = np.correlate(xc, 1.0 - xc, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip((corr - c_max) * _glb_alpha, -500.0, 0.0)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(np.sum(exp_s)))
    obj = (c_max + log_sum / _glb_alpha) * dx
    sum_dev = np.sum(xc) - _glb_target
    obj += 0.5 * _glb_penalty * sum_dev * sum_dev
    return obj


def _glb_fg(x):
    n = _glb_n
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    corr = np.correlate(xc, 1.0 - xc, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip((corr - c_max) * _glb_alpha, -500.0, 0.0)
    exp_s = np.exp(shifted)
    sums = float(np.sum(exp_s))
    weights = exp_s / sums
    obj = (c_max + np.log(sums) / _glb_alpha) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-7)[0]
    for ki in active:
        k = int(ki) - (n - 1)
        w = weights[int(ki)] * dx / _glb_alpha
        if w < 1e-12:
            continue
        if k >= 0:
            if k < n:
                js = np.arange(k, n)
                grad[js] += w * (1.0 - xc[js - k])
                js2 = np.arange(0, n - k)
                grad[js2] += -w * xc[js2 + k]
        else:
            ka = -k
            if ka < n:
                js = np.arange(0, n - ka)
                grad[js] += w * (1.0 - xc[js + ka])
                js2 = np.arange(ka, n)
                grad[js2] += -w * xc[js2 - ka]
    sum_dev = np.sum(xc) - _glb_target
    grad += _glb_penalty * sum_dev
    obj += 0.5 * _glb_penalty * sum_dev * sum_dev
    return obj, grad


def lbfgsb_smooth(h_start, target_sum, alpha, maxiter,
                  time_limit=None, start_time=None):
    global _glb_n, _glb_target, _glb_alpha, _glb_penalty
    n = len(h_start)
    _glb_n = n
    _glb_target = target_sum
    _glb_alpha = alpha
    _glb_penalty = 500.0
    h0 = project_feasible(h_start.copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    if time_limit is not None and start_time is not None:
        if time.time() - start_time > time_limit:
            return h0, compute_c5(h0)
    try:
        res = minimize(_glb_f, h0, jac=_glb_fg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-9, 'maxfun': maxiter * 3})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)


def greedy_full(h, max_iters, time_limit=None, start_time=None, scan_width=20):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 600

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] > 0.5]

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue

        top_threshold = max_val * 0.75
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        n_tops = min(len(top_indices), 20)
        top_indices = top_indices[np.argsort(corr[top_indices])[::-1]][:n_tops]
        top_vals = corr[top_indices].astype(float)

        best_improve = 0.0
        best_i = -1
        best_j = -1

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for ti in range(n_tops):
                    t_idx = top_indices[ti]
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[ti] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _glb_n, _glb_target, _glb_alpha, _glb_penalty
    start_time = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0
    target_int = int(n_init / 2)

    best_h = project_binary(h_init, target_int)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy
    allowed = min(40, max(10, tleft() * 0.05))
    h_g, c5_g = greedy_full(best_h, max_iters=30000, time_limit=allowed,
                            start_time=start_time, scan_width=25)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Targeted gradient descent - multiple configs
    td_configs = [
        (0.08, 0.9995, 3, 40000),
        (0.05, 0.9997, 5, 35000),
        (0.03, 0.9998, 5, 30000),
        (0.02, 0.9999, 3, 25000),
        (0.01, 0.99995, 5, 20000),
    ]

    for lr0, decay, top_m, max_it in td_configs:
        if tleft() < 50:
            break
        allowed = min(100, max(15, tleft() * 0.18))
        h_td, c5_td = targeted_descend(best_h, target_init, max_it, lr0, decay,
                                        top_m, time_limit=allowed,
                                        start_time=start_time)
        if c5_td < best_score:
            best_score = c5_td
            best_h = h_td.copy()

        if tleft() < 25:
            break
        h_bin = project_binary(best_h, target_int)
        allowed2 = min(15, max(5, tleft() * 0.04))
        h_g2, c5_g2 = greedy_full(h_bin, max_iters=10000, time_limit=allowed2,
                                   start_time=start_time, scan_width=20)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    # Phase 3: L-BFGS-B smooth optimization
    if tleft() > 80:
        for alpha in [50, 200, 1000, 5000, 20000]:
            if tleft() < 50:
                break
            allowed = min(100, max(20, tleft() * 0.2))
            h_ref, sc = lbfgsb_smooth(best_h, target_init, alpha, 250,
                                       time_limit=allowed, start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref.copy()

    # Phase 4: More targeted descent from improved continuous solution
    if tleft() > 100:
        for lr0, decay, top_m, max_it in [
            (0.04, 0.9998, 5, 30000),
            (0.02, 0.9999, 5, 25000),
            (0.01, 0.99995, 3, 20000),
        ]:
            if tleft() < 50:
                break
            allowed = min(100, max(15, tleft() * 0.2))
            h_td2, c5_td2 = targeted_descend(best_h, target_init, max_it, lr0,
                                              decay, top_m, time_limit=allowed,
                                              start_time=start_time)
            if c5_td2 < best_score:
                best_score = c5_td2
                best_h = h_td2.copy()

    # Phase 5: ILS perturbations
    if tleft() > 120:
        for pert_iter in range(10):
            if tleft() < 60:
                break
            n_pert = 2 + pert_iter
            h_pert = project_binary(best_h, target_int)
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(50, max(15, tleft() * 0.12))
            h_pt, c5_pt = targeted_descend(h_pert, target_init, 15000, 0.04,
                                            0.9998, 5, time_limit=allowed,
                                            start_time=start_time)
            if c5_pt < best_score:
                best_score = c5_pt
                best_h = h_pt.copy()

    # Phase 6: Try slightly different n
    if tleft() > 100:
        n_options = [n_init + 4, n_init - 4, n_init + 8]
        n_options = [x for x in n_options if x >= 400 and x % 2 == 0 and x != n_init]
        for n_try in n_options:
            if tleft() < 60:
                break
            target_try = float(n_try) / 2.0
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, n_init)
            h_interp = np.interp(x_new, x_old, best_h)
            h_bin = project_binary(h_interp, int(n_try / 2))

            allowed = min(20, max(8, tleft() * 0.06))
            h_g3, c5_g3 = greedy_full(h_bin, max_iters=10000, time_limit=allowed,
                                       start_time=start_time, scan_width=20)
            if tleft() < 40:
                if c5_g3 < best_score:
                    best_score = c5_g3
                    best_h = h_g3.copy()
                continue

            allowed2 = min(60, max(15, tleft() * 0.12))
            h_t3, c5_t3 = targeted_descend(h_g3, target_try, 20000, 0.04,
                                            0.9998, 5, time_limit=allowed2,
                                            start_time=start_time)
            if c5_t3 < best_score:
                best_score = c5_t3
                best_h = h_t3.copy()
            elif c5_g3 < best_score:
                best_score = c5_g3
                best_h = h_g3.copy()

    # Phase 7: Final high-alpha L-BFGS-B polish
    if tleft() > 30:
        for alpha in [50000, 200000]:
            if tleft() < 20:
                break
            n_use = len(best_h)
            t_use = float(n_use) / 2.0
            allowed = max(10, min(60, tleft() * 0.3))
            h_ref, sc = lbfgsb_smooth(best_h, t_use, alpha, 150,
                                       time_limit=allowed, start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref.copy()

    # Phase 8: Final targeted descent polish
    if tleft() > 15:
        n_use = len(best_h)
        t_use = float(n_use) / 2.0
        allowed = max(5, tleft() - 10)
        h_tf, c5_tf = targeted_descend(best_h, t_use, 20000, 0.005,
                                        0.99999, 3, time_limit=allowed,
                                        start_time=start_time)
        if c5_tf < best_score:
            best_score = c5_tf
            best_h = h_tf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
