import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_cd(h, n, target, time_limit, max_sweeps=500000):
    """Fast single-pair coordinate descent with incremental understanding."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    consecutive_no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > 500:
                break
        
        k_star = int(np.argmax(corr)) - mid
        
        # Gradient at worst shift
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        
        # Compute g_diff for pair (i, j)
        # We decrease h[i] by δ and increase h[j] by δ
        # ΔC(s) = dx * δ * [g_j(s) - g_i(s)]
        # where g_p(s) = (1-h[p+s])*1_{valid} - h[p-s]*1_{valid}
        g_diff = np.zeros(2*n-1)
        for s_idx, s in enumerate(shifts):
            gj = 0.0
            if 0 <= j + s < n:
                gj += (1.0 - h[j + s])
            if 0 <= j - s < n:
                gj -= h[j - s]
            gi = 0.0
            if 0 <= i + s < n:
                gi += (1.0 - h[i + s])
            if 0 <= i - s < n:
                gi -= h[i - s]
            g_diff[s_idx] = gj - gi
        
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-14:
            continue
        
        # Bisection for optimal δ
        a, b = delta_lo, delta_hi
        for _ in range(40):
            mid_d = 0.5 * (a + b)
            vals = corr + dx * mid_d * g_diff
            f_mid = float(np.max(vals))
            active = vals >= f_mid - 1e-14
            if active.any():
                f_prime = dx * float(np.max(g_diff[active]))
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b = mid_d
            else:
                a = mid_d
        
        delta_star = 0.5 * (a + b)
        if abs(delta_star) < 1e-12:
            continue
        
        new_c5 = float(np.max(corr + dx * delta_star * g_diff)) * dx
        if new_c5 < c5_val - 1e-14:
            h[i] -= delta_star
            h[j] += delta_star
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
    
    return best_h, best_c5

        mask_j1 = (j + shifts >= 0) & (j + shifts < n)
        mask_j2 = (j - shifts >= 0) & (j - shifts < n)
        mask_i1 = (i + shifts >= 0) & (i + shifts < n)
        mask_i2 = (i - shifts >= 0) & (i - shifts < n)
        
        g_j = np.where(mask_j1, 1.0 - h[j + shifts[mask_j1]], 0.0) - np.where(mask_j2, h[j - shifts[mask_j2]], 0.0)
        g_i = np.where(mask_i1, 1.0 - h[i + shifts[mask_i1]], 0.0) - np.where(mask_i2, h[i - shifts[mask_i2]], 0.0)
        g_diff = g_j - g_i
        
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-14:
            continue
        
        # Bisection
        a, b = delta_lo, delta_hi
        for _ in range(35):
            mid_d = 0.5 * (a + b)
            vals = corr + dx * mid_d * g_diff
            f_mid = float(np.max(vals))
            active = vals >= f_mid - 1e-15
            if active.any():
                f_prime = dx * float(np.max(g_diff[active]))
            else:
                f_prime = 0.0
            if f_prime > 1e-13:
                b = mid_d
            else:
                a = mid_d
        
        delta_star = 0.5 * (a + b)
        if abs(delta_star) < 1e-13:
            continue
        
        new_c5 = float(np.max(corr + dx * delta_star * g_diff)) * dx
        if new_c5 < c5_val - 1e-15:
            h[i] -= delta_star
            h[j] += delta_star
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
    
    return best_h, best_c5
