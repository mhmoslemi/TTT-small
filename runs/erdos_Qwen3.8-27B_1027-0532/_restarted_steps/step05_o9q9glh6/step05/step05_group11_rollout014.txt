```python
def lp_epsilon_band(h_init, n, target, dx, time_limit):
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start = time.time()
    
    for phase in range(3):  # 3 phases of decreasing regularization
        eps = [0.05, 0.02, 0.005][phase]  # relative epsilon band
        reg = [1e-3, 1e-5, 1e-7][phase]
        
        for it in range(2000):
            if time.time() - start > time_limit:
                break
            
            # Compute current overlaps
            overlaps = compute_overlaps(h)
            current_max = np.max(overlaps) * dx
            
            # Select lags in epsilon band
            n_half = n - 1
            threshold = current_max - eps * current_max
            active_mask = overlaps * dx >= threshold
            active_lags = np.where(active_mask)[0] - n_half
            
            if len(active_lags) == 0:
                active_lags = np.argsort(overlaps)[::-1][0:50] - n_half
            
            # Cap the number of lags for LP tractability
            if len(active_lags) > 200:
                # Keep the top 200 by overlap value
                overlap_vals = overlaps * dx
                top_idx = np.argsort(overlap_vals)[::-1][:200]
                active_lags = top_idx - n_half
            
            # Build and solve LP
            h_new = solve_lp_step(h, n, target, dx, active_lags, reg)
            
            # Damped update
            damping = 0.9 if phase == 0 else 0.95
            h = (1-damping) * h + damping * h_new
            h = project_feasible(h, target)
            
            # Check improvement
            c5 = compute_c5(h)
```

```python
def solve_lp_step(h, n, target, dx, lags, reg):
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    
    for k in lags:
        # Compute coefficient vector for this lag
        if k >= 0:
            # overlap at lag k: sum_{i=0}^{n-1-k} h[i] * g[i+k] * dx
            coeff = np.zeros(n)
            coeff[:n-k] = g[k:] * dx
        else:
            # overlap at lag -k: sum_{i=-k}^{n-1} h[i] * g[i+k] * dx
            # which is sum_{j=0}^{n-1+k} h[j-k] * g[j] * dx (reindexing)
            # = sum_{i=0}^{n-1+k} h[i] * g[i+k] * dx
            # Wait, let me be more careful.
            # np.correlate(h, 1-h, mode='full') at index n-1+k (k<0) gives:
            # sum_i h[i] * (1-h[i+k]) where i+k >= 0 and i < n
            # = sum_{i=-k}^{n-1} h[i] * (1-h[i+k])
            # So the coefficient of h[i] for i >= -k is (1-h[i+k]) = g[i+k]
            coeff = np.zeros(n)
            coeff[-k:] = g[:n+k] * dx

    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    
    if h_var.value is None:
        return h.copy()
    
    return np.clip(np.asarray(h_var.value).ravel(), 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(overlaps, n, n_max, eps_rel=0.0):
    """Get active lag indices from the correlation array.
    If eps_rel > 0, include all lags with value >= (max - eps_rel * max).
    Otherwise, include the top n_max lags.
    """
    n_half = n - 1
    if eps_rel > 0:
        current_max = np.max(overlaps)
        threshold = current_max - eps_rel * current_max
        mask = overlaps >= threshold
        idx = np.where(mask)[0]
    else:
        idx = np.argsort(overlaps)[::-1][:n_max]
    
    if len(idx) > n_max:
        # Keep only the top n_max
        vals_at_idx = overlaps[idx]
        order = np.argsort(vals_at_idx)[::-1][:n_max]
        idx = idx[order]
    
    return idx - n_half  # Convert to actual lag values

def solve_lp_step(h, n, target, dx, lags, reg):
    """Solve one LP step of the alternating method."""
    g = 1.0 - h
    n_half = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    for k in lags:
        idx = n_half + k  # correlation index
        # coeff[j] = g[n-1-idx+j] * dx for valid j
        j_min = max(0, idx - n + 1)
        j_max = min(n - 1, idx)
        if j_max < j_min:
            continue
        js = np.arange(j_min, j_max + 1)
        g_idx = n - 1 - idx + js
        coeff = np.zeros(n)
        coeff[js] = g[g_idx] * dx
        constraints.append(coeff @ h_var <= t_var)
    
    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, verbose=False)
        except Exception:
            return h.copy()
    
    if h_var.value is None:
        return h.copy()
    
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def lp_epsilon_band(h_init, n, target, time_limit, 
                    n_max_lags=150, max_inner=5000):
    """Alternating LP with epsilon-band lag selection and progressive tightening."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phases with decreasing regularization and tightening band
    phases = [
        {'eps_rel': 0.0, 'n_lags': 80, 'reg': 1e-2, 'damping': 0.8, 'max_iter': 500},
        {'eps_rel': 0.0, 'n_lags': 120, 'reg': 1e-3, 'damping': 0.9, 'max_iter': 1000},
        {'eps_rel': 0.02, 'n_lags': n_max_lags, 'reg': 1e-4, 'damping': 0.95, 'max_iter': 1500},
        {'eps_rel': 0.01, 'n_lags': n_max_lags, 'reg': 1e-5, 'damping': 0.97, 'max_iter': 2000},
        {'eps_rel': 0.005, 'n_lags': n_max_lags, 'reg': 1e-6, 'damping': 0.98, 'max_iter': 3000},
    ]
    
    for phase in phases:
        if time.time() - start > time_limit:
            break
        
        eps_rel = phase['eps_rel']
        n_lags = phase['n_lags']
        reg = phase['reg']
        damping = phase['damping']
        max_iter = phase['max_iter']
        
        stuck = 0
        for it in range(max_iter):
            if time.time() - start > time_limit:
                break
            if stuck > 200:
                break
            
            overlaps = compute_overlaps(h)
            lags = get_active_lags(overlaps, n, n_lags, eps_rel=eps_rel)
            
            h_new = solve_lp_step(h, n, target, dx, lags, reg)
            
            h = (1.0 - damping) * h + damping * h_new
            h = project_feasible(h, target)
            
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
    
    return best_h, best_c5

def smooth_max_obj(h, alpha, n, target):
    """Compute smooth max of overlaps using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    # For numerical stability, only consider top values
    top_k = 50
    top_idx = np.argsort(c)[::-1][:top_k]
    top_c = c[top_idx]
    shifted = alpha * (top_c - c_max)
    log_sum = c_max + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / alpha
    return log_sum

_global_n = 0
_global_target = 0.0

def slsqp_smooth_max_fun(h, alpha):
    return smooth_max_obj(h, alpha, _global_n, _global_target)

def slsqp_sum_fun(h):
    return np.sum(h) - _global_target

def slsqp_sum_jac(h):
    return np.ones(_global_n)

def slsqp_polish(h_init, n, target, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    start = time.time()
    
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alphas = [20, 50, 100, 200, 500, 1000, 3000, 10000, 30000]
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 10:
            break
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_sum_fun, 'jac': slsqp_sum_jac}]
        
        try:
            res = minimize(slsqp_smooth_max_fun, h, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': int(min(3000, remaining * 5)), 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def gradient_step_minimax(h, n, target, n_top=30, lr=0.01):
    """One gradient step minimizing the smooth max of top overlaps."""
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    n_half = n - 1
    
    top_idx = np.argsort(overlaps)[::-1][:n_top]
    top_overlaps = overlaps[top_idx]
    
    # Compute gradient of smooth max w.r.t. h
    # For each active lag, d(overlap_k)/d(h[j]) is the coefficient
    alpha = 50.0
    c_max = np.max(top_overlaps)
    weights = np.exp(alpha * (top_overlaps - c_max))
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - n_half
        j_min = max(0, idx - n + 1)
        j_max = min(n - 1, idx)
        js = np.arange(j_min, j_max + 1)
        g_idx = n - 1 - idx + js
        # d(corr[idx])/d(h[j]) = g[g_idx] * dx (where g = 1-h)
        g = 1.0 - h
        grad[js] += w * g[g_idx] * dx
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new

def momentum_minimax(h_init, n, target, time_limit, n_top=30, momentum=0.9):
    """Momentum-based gradient descent on the smooth max objective."""
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(500000):
        if time.time() - start > time_limit:
            break
        
        # Nesterov-like lookahead
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute gradient at y
        dx = 2.0 / n
        overlaps_y = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps_y)[::-1][:n_top]
        top_overlaps = overlaps_y[top_idx]
        
        alpha = 30.0
        c_max = np.max(top_overlaps)
        weights = np.exp(alpha * (top_overlaps - c_max))
        weights = weights / np.sum(weights)
        
        g = 1.0 - y
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            j_min = max(0, idx - n + 1)
            j_max = min(n - 1, idx)
            js = np.arange(j_min, j_max + 1)
            g_idx = n - 1 - idx + js
            grad[js] += w * g[g_idx] * dx
        
        # Adaptive learning rate
        c5_y = float(np.max(overlaps_y))
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break
        lr = min(0.1, max(0.001, (c5_y - best_c5) / (norm_g * norm_g + 1e-10)))
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()
            # Reduce learning rate
            momentum = max(0.5, momentum * 0.999)
    
    return best_h, best_c5
```

```python
def solve_lp_fast(h, n, target, dx, lags):
    """Solve the LP without regularization using scipy linprog."""
    g = 1.0 - h
    n_half = n - 1
    num_lags = len(lags)
    
    # Variables: [h_0, ..., h_{n-1}, t]  (total n+1)
    # Minimize: t  →  c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each lag: sum_j coeff[j] * h[j] - t ≤ 0
    A_ub = np.zeros((num_lags, n + 1))
    b_ub = np.zeros(num_lags)
    
    for i, k in enumerate(lags):
        idx = n_half + k
        j_min = max(0, idx - n + 1)
        j_max = min(n - 1, idx)
        if j_max < j_min:
            # Degenerate: set constraint to 0 ≤ 0 (always satisfied)
            continue
        js = np.arange(j_min, j_max + 1)
        g_idx = n - 1 - idx + js
        A_ub[i, js] = g[g_idx] * dx
        A_ub[i, -1] = -1.0
    
    # Equality constraints: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 ≤ h[j] ≤ 1, t unbounded below (but will be ≥ 0)
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:n], 0, 1)
    except Exception:
        pass
    
    return h.copy()
```

```python
def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def get_lag_coefficients(n, k):
    """Get the coefficient vector for the linearized overlap at lag k."""
    n_half = n - 1
    idx = n_half + k
    j_min = max(0, idx - n + 1)
    j_max = min(n - 1, idx)
    if j_max < j_min:
        return None, None
    js = np.arange(j_min, j_max + 1)
    g_pos = n - 1 - idx + js
    return js, g_pos

    """Solve LP: min t s.t. linearized overlaps <= t, sum(h)=target, 0<=h<=1."""
    g = 1.0 - h
    num_lags = len(lags)

    b_ub = np.zeros(num_lags)

    for i, k in enumerate(lags):
        js, g_pos = get_lag_coefficients(n, k)
        if js is None:

        A_ub[valid_count, -1] = -1.0
        valid_count += 1

    b_ub = b_ub[:valid_count]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    bounds = [(0.0, 1.0)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
```
