import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

# Precomputation for coefficients
def make_coeff_indices(n):
    """Precompute index arrays for coefficient computation."""
    r = np.arange(2*n-1) - (n-1)
    return r

def compute_coeff(h, i, n, r):
    out = np.zeros(2*n-1)
    idx1 = i + r
    m1 = (idx1 >= 0) & (idx1 < n)
    out[m1] = 1.0 - h[idx1[m1]]
    idx2 = i - r
    m2 = (idx2 >= 0) & (idx2 < n)
    out[m2] -= h[idx2[m2]]
    return out

def line_search_1d(corr, coeff, dlo, dhi):
    """Find delta minimizing max(corr + coeff*delta) over [dlo, dhi]."""
    lo = dlo
    hi = dhi
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = corr + coeff * mid
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        cmin = np.min(coeff[active])
        cmax = np.max(coeff[active])
        if cmin <= 0.0 and cmax >= 0.0:
            break
        elif cmax < 0.0:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return 0.5 * (lo + hi)

def cd_single(h, n, target, deadline, rng, r, stall_limit=300):
    """Single-coordinate descent with exact line search."""
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best = float(np.max(corr))
    order = np.arange(n)
    stall = 0
    
    while time.time() < deadline:
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline + 5:
                break
            old = h_cur[i]
            dlo = -old
            dhi = 1.0 - old
            if dhi - dlo < 1e-14:
                continue
            coeff = compute_coeff(h_cur, i, n, r)
            d = line_search_1d(corr, coeff, dlo, dhi)
            new = old + d
            if new < 0.0:
                new = 0.0
            elif new > 1.0:
                new = 1.0
            if abs(new - old) > 1e-14:
                h_cur[i] = new
                corr += (new - old) * coeff
        if np.max(corr) < best - 1e-12:
            best = np.max(corr)
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            break
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)
