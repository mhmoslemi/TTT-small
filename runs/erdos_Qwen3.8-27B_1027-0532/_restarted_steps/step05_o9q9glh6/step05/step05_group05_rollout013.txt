from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_lags(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n - 1:]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_single_k(h, k, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        nk = n - k
        grad[0:nk] += (1.0 - h[k:]) * dx
        grad[k:] -= h[0:nk] * dx
    return grad


def momentum_gd(h, n, target, n_steps, lr_init, beta, n_active, deadline):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    stuck = 0

        weights = np.exp((raw - w_max) / 0.005)
        sw = weights.sum()

            best_h = h_new.copy()

def slsqp_obj(x):
    corr = compute_lags(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_jac(x):
    n = len(x)
    corr = compute_lags(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)

    return grad

def pair_optimize(h, n, target, k_star, max_steps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n

    for step in range(max_steps):
        if time.time() > deadline:
            break
        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = -2.0 * h_cur
        else:
            nk = n - k_star
            grad[0:nk] -= h_cur[k_star:]
            grad[k_star:] -= h_cur[0:nk]
        grad *= dx

        best_lr = 0.0
        best_sc = cur_score
        for lr in [0.03, 0.015, 0.008, 0.004, 0.002]:

            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_sc:
                best_sc = sc
                best_lr = lr

        if best_lr > 0:
            h_cur = h_cur + best_lr * grad
            h_cur = project_feasible(h_cur, target)
            cur_score = best_sc
        else:
            break
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
