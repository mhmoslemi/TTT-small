import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_constraint_lp(h_init, n, target, damping=0.7, reg=1e-4, max_iter=200, time_limit=300):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Add ALL shift constraints
        for k in range(-(n-1), n):
            if k == 0:
                # corr[n-1] = sum h[i]*(1-h[i]) - this is always 0 for binary
                # but for general h: sum h[i] - sum h[i]^2
                # Linearized: sum g[i]*h[i] where g = 1-h
                # Actually: coeff @ h = sum_i g[i] * h[i] * dx
                coeff = g * dx
                constraints.append(coeff @ h_var <= t_var)
            else:
                lo = max(0, k)
                hi = min(n, n + k)
                if hi > lo:
                    coeff = np.zeros(n)
                    coeff[lo:hi] = g[lo-k:hi-k] * dx
                    constraints.append(coeff @ h_var <= t_var)
        
        obj = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=100000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = (1 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

Condition 2 is hard to check efficiently for all k. But condition 1 is cheap (|K_max| is small).

A pragmatic approach: check condition 1 for all k in K_max, and for condition 2, just check if the maximum increase (over all k not in K_max) is less than the minimum decrease (over K_max). The maximum increase is an upper bound on how much the max could increase.

The maximum increase over all k:
max_k Δcorr[n-1+k] ≤ max_k |h[i-k] - h[j-k] + h[i+k] - h[j+k]|

For binary h, each term is in {-1, 0, 1}, so the sum is in {-2, -1, 0, 1, 2}. The maximum possible increase is 2.

So if the minimum decrease over K_max is > 2, the max must decrease. If it's ≤ 2, we need to check more carefully.

In practice, for a "flat" correlation (which is what we're aiming for), the values near the max are all close together, so the min decrease over K_max will be small (0 or 1 for binary). This means we often need to do the full check.

Let me just do the full O(n) check for each candidate swap, but limit the number of candidates per iteration.

For n=400 and 1000 candidates per iteration, each O(400) = 400,000 operations per iteration. At 100 iterations, that's 40 million operations. In numpy, this should take a few seconds.

Actually, I can vectorize: for a batch of 1000 swaps, compute all Δcorr vectors at once using broadcasting. Each Δcorr is a length-2n-1 vector. 1000 × 799 = 800,000 values. That's fine.

        h[to_flip] = 0
    elif ones < target:

        to_flip = np.random.choice(idx, int(target - ones), replace=False)

    best_h = h.copy()

    max_val = np.max(corr)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        max_val = np.max(corr)
        K_max = np.where(corr >= max_val - 0.01)[0]  # shifts near max
        min_decrease_needed = max_val - np.min(corr[K_max] + 1)  # rough
        
        one_pos = np.where(h == 1)[0]
        zero_pos = np.where(h == 0)[0]

        i_cands = one_pos[np.random.choice(len(one_pos), n_cand, replace=True)]
        j_cands = zero_pos[np.random.choice(len(zero_pos), n_cand, replace=True)]

        mask = i_cands != j_cands
        i_cands = i_cands[mask]
        j_cands = j_cands[mask]
        
        # Compute Δcorr for each candidate
        # For k shift (indexed 0 to 2n-2, where k_shift = idx - (n-1)):
        # Δcorr[idx] = h[i-k] - h[j-k] + h[i+k] - h[j+k] (with bounds checking)
        
        k_sh
        ifts = np.arange(-(n-1), n)  # length 2n-1
        
        # For candidate r with (i, j):
        # Δcorr[n-1+k] = h[i-k] - h[j-k] + h[i+k] - h[j+k] for valid indices
        # Need to handle bounds: h[x] = 0 if x < 0 or x >= n
        
        best_new_max = max_val
        best_swap = None
        
        # Process in batches for memory efficiency
        batch = min(500, len(i_cands))
        for b in range(0, len(i_cands), batch):

            j_batch = j_cands[b:b+batch]

            # i_batch[:, None] - k_shifts[None, :] gives (batch, 2n-1) matrix of i-k values
            ik = i_batch[:, None] - k_shifts[None, :]
            jk = j_batch[:, None] - k_shifts[None, :]
            ikp = i_batch[:, None] + k_shifts[None, :]
            jkp = j_batch[:, None] + k_shifts[None, :]
            
            # Get h values with bounds (out of bounds -> 0)
            def safe_index(arr, idx):
                mask = (idx >= 0) & (idx < len(arr))
                out = np.zeros(idx.shape)
                out[mask] = arr[idx[mask]]
                return out
            
            h_ik = safe_index(h, ik)
            h_jk = safe_index(h, jk)
            h_ikp = safe_index(h, ikp)
            h_jkp = safe_index(h, jkp)
            
            delta_corr = h_ik - h_jk + h_ikp - h_jkp
            # For k=0: subtract 2 (the -2δ_{k,0} term)
            delta_corr[:, n-1] -= 2
            # For k = i-j and k = j-i: handle special terms
            # -δ_{k,i-j}: for candidate r, k=i_r-j_r, so at position n-1+(i_r-j_r)
            for r in range(batch):
                k1 = i_batch[r] - j_batch[r]
                k2 = j_batch[r] - i_batch[r]
                pos1 = n - 1 + k1
                pos2 = n - 1 + k2
                if 0 <= pos1 < 2*n - 1:
                    delta_corr[r, pos1] -= 1
                if 0 <= pos2 < 2*n - 1:
                    delta_corr[r, pos2] += 1
            
            # New correlation
            new_corr = corr[None, :] + delta_corr
            new_maxes = np.max(new_corr, axis=1)
            
            for r in range(len(new_maxes)):
                if new_maxes[r] < best_new_max:
                    best_new_max = new_maxes[r]
                    best_swap = (i_batch[r], j_batch[r])

            h[j] = 1
            corr = np.correlate(h, 1-h, mode='full')
            if best_new_max < best_c5 * n / 2:
                best_c5 = best_new_max * 2.0 / n
                best_h = h.copy()
        elif it > 50:

    return best_h, best_c5

**Key mathematical insight**: The problem min max_k f_k(h) where f_k(h) = Σ h[i](1-h[i-k])dx is a minimax problem. By von Neumann's minimax theorem (for the convex-concave relaxation), the optimal value equals the max over probability distributions p on k of min_h Σ p_k f_k(h).

The inner min is:
min_h Σ_k p_k f_k(h) = min_h Σ_k p_k Σ_i h[i](1-h[i-k])dx

This is still quadratic in h. But if we further relax h to be in [0,1] (which it already is), the inner problem is a QP.

Actually, for the specific structure, let me expand:
Σ_k p_k Σ_i h[i](1-h[i-k])dx = Σ_k p_k [Σ_{valid i} h[i]dx - Σ_{valid i} h[i]h[i-k]dx]

The first term: Σ_k p_k Σ_{valid i} h[i]dx = Σ_i h[i]dx * Σ_k p_k * I(i valid for k)
= Σ_i h[i]dx * w[i] where w[i] = Σ_k p_k * I(i valid for k)

The second term: Σ_k p_k Σ_{valid i} h[i]h[i-k]dx = Σ_{i,j} h[i]h[j] * (Σ_k p_k * I(j=i-k, both valid)) * dx
= Σ_{i,j} h[i]h[j] * Q[i-j] * dx where Q[d] = p_d (if we identify k with the shift d)

So the objective becomes:
Σ_i w[i] h[i] dx - Σ_{i,j} Q[i-j] h[i]h[j] dx

This is a quadratic form in h. Minimizing a quadratic over a simplex with box constraints is a standard QP.

The overall problem is:
max_p min_h [linear term - quadratic term] s.t. sum h = n/2, 0≤h≤1

This is a max-min QP. In principle, we could solve it with a bisection on the value:
- Guess value v
- Check: is there a p such that min_h [objective with p] > v? (infeasible if yes)

But this is still complex. Let me just stick with the practical approach.
