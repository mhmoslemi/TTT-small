```python
def project_feasible(h, target_sum):

def lbfgsb_refine(h_start, n, target, alphas, penalties, maxiter, time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = curr_score
    
    for alpha in alphas:
        for pen in penalties:
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    break
            try:
                res = minimize(logsumexp_obj, h_cur, args=(alpha, target, pen),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new.copy()
            except Exception:
                pass
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def lse_obj_and_grad(h, alpha, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = c_max + float(np.log(sum_exp) / alpha) + penalty * (np.sum(h) - target) ** 2
    grad = np.zeros(n)
    # For shift s, contribution to grad:
    #   (1-h[j+s]) for valid j (j+s in [0,n-1])
    #   -h[j-s] for valid j (j-s in [0,n-1])
    for s in range(-n + 1, n):
        m = s + n - 1
        w = weights[m]
        if w < 1e-16:
            continue
        if s >= 0:
            lo1, hi1 = 0, n - s
            lo2, hi2 = s, n
        else:
            lo1, hi1 = -s, n
            lo2, hi2 = 0, n + s
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + s:hi1 + s])
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 - s:hi2 - s]
    grad += 2.0 * penalty * (np.sum(h) - target)
    return obj, grad


def lse_objective(h, alpha, target, penalty):
    return lse_obj_and_grad(h, alpha, target, penalty)[0]


def lse_gradient(h, alpha, target, penalty):
    return lse_obj_and_grad(h, alpha, target, penalty)[1]


def lbfgsb_refine(h_start, n, target, alphas, penalties, maxiter, time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = curr_score
    for alpha in alphas:
        for pen in penalties:
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    break
            try:
                def obj_wrapper(x):
                    return lse_obj_and_grad(x, alpha, target, pen)[0]

                def grad_wrapper(x):
                    return lse_obj_and_grad(x, alpha, target, pen)[1]

                res = minimize(obj_wrapper, h_cur, jac=grad_wrapper,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-18,
                                        'gtol': 1e-12, 'maxfun': maxiter * n})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                if sc <= curr_score + 1e-12:
                    curr_score = sc
                    h_cur = h_new.copy()
            except Exception:
                pass
    return best_h, best_score

            for ti in top_indices:
                k = ti - (n - 1)
                i_c = [i for i in ones if 0 <= i + k < n and h[i + k] == 0]
                j_c = [j for j in zeros if 0 <= j + k < n and h[j + k] == 1]
                if len(i_c) > 0 and len(j_c) > 0:
                    i_cand = i_c
                    j_cand = j_c
                    found = True
                    break
            if not found:
                stall += 1
                continue
        top_threshold = max_val * 0.82
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_vals = corr.astype(float)
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy
    allowed = min(25, max(5, tleft() * 0.04))
    h_g, c5_g = greedy_improved(best_h, max_iters=15000, time_limit=allowed,
                                 start_time=start_time, scan_width=15)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA diverse moves
    allowed = min(100, max(10, tleft() * 0.12))
    max_it = int(min(150000, allowed * 800))
    h_sa, c5_sa = sa_diverse(best_h, target_init, max_it, 0.012, 0.00004, rng,
                             time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: L-BFGS-B continuous with analytical gradient (key improvement)
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 60, 120, 250, 500, 1000]
        penalties = [5000.0]
        time_alloc = min(60, max(10, tleft() * 0.08))
        h_ref, sc = lbfgsb_refine(best_h, n_use, target_use, alphas, penalties,
                                   maxiter=1500, time_limit=time_alloc, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 4: Round-trip binary -> SA -> continuous
    if tleft() > 70:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin = project_binary(best_h, target_use)
        allowed = min(50, max(8, tleft() * 0.07))
        max_it = int(min(80000, allowed * 800))
        h_sa2, c5_sa2 = sa_diverse(h_bin, target_use, max_it, 0.006, 0.00002, rng,
                                    time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        if tleft() > 35:
            n_use = len(best_h)
            target_use = float(n_use) / 2.0
            alphas = [200, 500, 1200, 3000]
            penalties = [8000.0]
            time_alloc = min(30, max(5, tleft() * 0.05))
            h_ref2, sc2 = lbfgsb_refine(best_h, n_use, target_use, alphas, penalties,
                                         maxiter=2000, time_limit=time_alloc, start_time=start_time)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2.copy()

    # Phase 5: Try nearby n values
    n_options = [n_init, n_init + 2, n_init - 2, n_init + 4, n_init + 6,
                 n_init - 4, n_init + 8, n_init + 10, n_init - 6]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0]
    n_options = list(dict.fromkeys(n_options))

    for n_try in n_options:
        if tleft() < 45:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(18, max(4, tleft() * 0.04))
        h_g2, c5_g2 = greedy_improved(h_bin_try, max_iters=8000, time_limit=allowed,
                                       start_time=start_time, scan_width=12)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()
        if tleft() < 25:
            continue
        allowed2 = min(25, max(5, tleft() * 0.03))
        max_it2 = int(min(40000, allowed2 * 800))
        h_sa3, c5_sa3 = sa_diverse(h_g2, target_try, max_it2, 0.006, 0.00004, rng,
                                    time_limit=allowed2, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()
        if tleft() < 20:
            continue
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [100, 300, 800, 2000]
        penalties = [6000.0]
        time_alloc = min(15, max(5, tleft() * 0.03))
        h_ref3, sc3 = lbfgsb_refine(best_h, n_use, target_use, alphas, penalties,
                                     maxiter=1500, time_limit=time_alloc, start_time=start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    # Phase 6: ILS perturbations
    if tleft() > 50:
        for pert_iter in range(4):
            if tleft() < 25:
                break
            n_pert = 2 + pert_iter
            h_pert = project_binary(best_h, int(len(best_h) / 2))
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            allowed = min(20, max(4, tleft() * 0.04))
            h_pg, c5_pg = greedy_improved(h_pert, max_iters=8000, time_limit=allowed,
                                           start_time=start_time, scan_width=12)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()
            if tleft() < 15:
                break
            allowed2 = min(18, max(4, tleft() * 0.03))
            max_it4 = int(min(30000, allowed2 * 800))
            h_ps, c5_ps = sa_diverse(project_binary(best_h, int(len(best_h) / 2)),
                                      int(len(best_h) / 2), max_it4,
                                      0.004, 0.00002, rng,
                                      time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 7: Final L-BFGS-B polish with high alpha
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [800, 2000, 5000, 12000]
        penalties = [10000.0]
        h_ref4, sc4 = lbfgsb_refine(best_h, n_use, target_use, alphas, penalties,
                                     maxiter=3000, time_limit=max(5.0, tleft() - 4),
                                     start_time=start_time)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_ref4.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = project_binary(best_h, int(n_final / 2))
        final_score = compute_c5(h_final)
    return h_final, final_score, n_final
```
