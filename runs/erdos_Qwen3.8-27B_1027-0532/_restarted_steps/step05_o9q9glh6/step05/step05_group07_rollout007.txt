def lse_gradient(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            # First term: h[m] contribution, valid for m = 0..n-1-shift
            # grad[m] += wk * (1 - h[m+shift]) for m = 0..n-1-shift
            m_range = np.arange(n - shift)
            grad[m_range] += wk * (1.0 - h[m_range + shift])
            # Second term: h[m] as shifted, valid for m = shift..n-1
            # grad[m] -= wk * h[m-shift] for m = shift..n-1
            m_range2 = np.arange(shift, n)
            grad[m_range2] -= wk * h[m_range2 - shift]
        else:
            s = -shift
            # First term: m - s < n always (m<n), m-s >= 0 means m >= s
            # grad[m] += wk * (1 - h[m-s]) for m = s..n-1
            m_range = np.arange(s, n)
            grad[m_range] += wk * (1.0 - h[m_range - s])
            # Second term: m + s >= 0 always, m+s < n means m < n-s
            # grad[m] -= wk * h[m+s] for m = 0..n-1-s
            m_range2 = np.arange(n - s)
            grad[m_range2] -= wk * h[m_range2 + s]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    val = c_max + np.log(total) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            m1 = np.arange(n - shift)
            grad[m1] += wk * (1.0 - h[m1 + shift])
            m2 = np.arange(shift, n)
            grad[m2] -= wk * h[m2 - shift]
        else:
            s = -shift
            m1 = np.arange(s, n)
            grad[m1] += wk * (1.0 - h[m1 - s])
            m2 = np.arange(n - s)
            grad[m2] -= wk * h[m2 + s]
    return val, grad

def adam_lse(h_init, n, target, alphas, steps_per_alpha, lr=0.01, deadline=None, rng=None):
    """Adam optimizer on LSE objective with schedule of alphas"""
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for alpha in alphas:
        for step in range(steps_per_alpha):
            if deadline is not None and time.time() > deadline:
                break
            
            t += 1
            val, grad = lse_value_and_grad(h, alpha)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_feasible(h, target)
            
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    
    return best_h, best_score

def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B

def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=50):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep_count = 0
    
    for sweep in range(n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break
        sweep_count += 1
        
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        
        rng.shuffle(pair_positions)
        
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            
            lo = 0.0
            if np.any(C > 0):
                hi = float(np.max(A[C > 0] + C[C > 0] * t_max))
            else:
                hi = float(np.max(A + C * t_max))
            if not np.any(C > 0):
                hi = float(np.max(A + C * t_max))
            
            for _ in range(40):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid
            
            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5
            
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 300 and sweep > 200:
            pert = rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

                    grad[m1] += wk * (1.0 - x[m1 - s])
                    m2 = np.arange(n - s)
                    grad[m2] -= wk * x[m2 + s]
            return grad
        
        def cons_eq(x):
            return np.sum(x) - target

            return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    val = c_max + np.log(total) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            m1 = np.arange(n - shift)
            grad[m1] += wk * (1.0 - h[m1 + shift])
            m2 = np.arange(shift, n)
            grad[m2] -= wk * h[m2 - shift]
        else:
            s = -shift
            m1 = np.arange(s, n)
            grad[m1] += wk * (1.0 - h[m1 - s])
            m2 = np.arange(n - s)
            grad[m2] -= wk * h[m2 + s]
    return val, grad


def adam_optimize(h_init, n, target, alphas, steps_per_alpha, lr, deadline, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    for alpha in alphas:
        stall = 0
        for step in range(steps_per_alpha):
            if time.time() > deadline:
                return best_h, best_score
            t += 1
            val, grad = lse_value_and_grad(h, alpha)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h - step_vec, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            h = h_new
            if stall > 3000:
                h = project_feasible(h + rng.randn(n) * 0.04, target)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
    return best_h, best_score


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def paired_coord_desc(h, n, target, deadline, rng, n_sweeps, recompute_interval):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep_count = 0
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        sweep_count += 1
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        rng.shuffle(pair_positions)
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            lo = 0.0
            if np.any(C > 0):
                hi = float(np.max(A[C > 0] + C[C > 0] * t_max))
            else:
                hi = float(np.max(A + C * t_max))
            for _ in range(40):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid
            v_val = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v_val - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v_val - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 250 and sweep_count > 100:
            pert = rng.randn(n) * 0.015
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


class LSEProblem:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target

    def obj(self, x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = self.alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / self.alpha

    def grad(self, x):
        n = self.n
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = self.alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        w = np.exp(shifted)
        w /= np.sum(w)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            shift = k - (n - 1)
            if shift >= 0:
                m1 = np.arange(n - shift)
                grad[m1] += wk * (1.0 - x[m1 + shift])
                m2 = np.arange(shift, n)
                grad[m2] -= wk * x[m2 - shift]
            else:
                s = -shift
                m1 = np.arange(s, n)
                grad[m1] += wk * (1.0 - x[m1 - s])
                m2 = np.arange(n - s)
                grad[m2] -= wk * x[m2 + s]
        return grad

    def cons_eq(self, x):
        return np.sum(x) - self.target

    def cons_jac(self, x):
        return np.ones(self.n)


def slsqp_polish(h, n, target, alphas, maxiter):
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        prob = LSEProblem(a, n, target)
        cons = [{'type': 'eq', 'fun': prob.cons_eq, 'jac': prob.cons_jac}]
        try:
            res = minimize(prob.obj, h_cur, jac=prob.grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_cd(h, n, target, deadline, rng, block_size):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    for iteration in range(200000):
        if time.time() > deadline:
            break
        idx = rng.randint(0, n - block_size + 1)
        block = slice(idx, idx + block_size)
        vals = h_cur[block].copy()
        s = vals.sum()
        best_block = vals.copy()
        best_local = float(np.max(corr))
        for sub_iter in range(50):
            step = rng.randn(block_size) * 0.02
            new_vals = vals + step
            new_vals = np.clip(new_vals, 0.0, 1.0)
            diff = s - new_vals.sum()
            mask = (new_vals < 1.0) if diff > 0 else (new_vals > 0.0)
            if mask.sum() > 0:
                new_vals[mask] += diff / mask.sum()
                new_vals = np.clip(new_vals, 0.0, 1.0)
            else:
                continue
            h_test = h_cur.copy()
            h_test[block] = new_vals
            c_test = compute_corr(h_test)
            sc_test = float(np.max(c_test))
            if sc_test < best_local:
                best_local = sc_test
                best_block = new_vals.copy()
                vals = new_vals.copy()
        h_cur[block] = best_block
        corr = compute_corr(h_cur)
        cur_s = float(np.max(corr)) * dx
        if cur_s < best_score - 1e-14:
            best_score = cur_s
    return project_feasible(h_cur, target), best_score
