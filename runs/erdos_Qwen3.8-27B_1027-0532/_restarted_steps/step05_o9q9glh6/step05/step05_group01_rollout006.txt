- where g_star(j) = (1-h[j+k*]) - h[j-k*] (with boundary checks)
- We want the largest |δ| (within bounds) such that c_{k*} decreases and no other c_k increases above the new c_{k*}.

The constraint that other correlations don't increase above the new max:
c_k + δ * g_k(j) ≤ c_{k*} + δ * g_{k*}(j) for all k

This gives: δ * (g_k(j) - g_{k*}(j)) ≤ c_{k*} - c_k for all k
If g_k(j) > g_{k*}(j): δ ≤ (c_{k*} - c_k) / (g_k(j) - g_{k*}(j))
If g_k(j) < g_{k*}(j): δ ≥ (c_{k*} - c_k) / (g_k(j) - g_{k*}(j))

So the feasible δ range is:
[max of lower bounds, -h[j]] ≤ δ ≤ [min of upper bounds, 1-h[j]]

And we want to maximize |δ| in the direction that reduces c_{k*}.

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)

    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:

    h = np.clip(h, 0.0, 1.0)

        else:

        h += coeffs[m + M] * np.sin(np.pi * (m+1) * x)

import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP constraints
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    alpha = _g_alpha
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_continue(h0, n, target, alphas, maxiter, ftol=1e-18):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h0.copy()
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) <= compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_scalar_cd(h0, n, target, time_limit, max_sweeps=200000, top_k=20):
    """Very fast scalar coordinate descent."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10000:
                break
        
        top_idx = np.argsort(corr)[-top_k:]
        c_active = float(corr[top_idx[np.argmax(corr[top_idx])]])
        
        n_improved = 0
        for j in range(n):
            if time.time() - start > time_limit:
                break
            
            ks = top_idx - mid
            # Compute gradient of each top-K correlation w.r.t. h[j]
            vp = (j + ks >= 0) & (j + ks < n)
            vm = (j - ks >= 0) & (j - ks < n)
            g = np.zeros(top_k)
            if np.any(vp):
                g[vp] += 1.0 - h[j + ks[vp]]
            if np.any(vm):
                g[vm] -= h[j - ks[vm]]
            
            # Find which top-K is the maximum
            active = int(np.argmax(corr[top_idx]))
            ga = g[active]
            if abs(ga) < 1e-14:
                continue
            
            # Direction
            if ga > 0:
                d_lo, d_hi = -h[j], 0.0
            else:
                d_lo, d_hi = 0.0, 1.0 - h[j]
            
            if d_hi - d_lo < 1e-14:
                continue
            
            # Constrain by other top-K
            c_act = corr[top_idx[active]]
            for m in range(top_k):
                if m == active:
                    continue
                dg = g[m] - ga
                dc = c_act - corr[top_idx[m]]
                if dg > 1e-14:
                    d_hi = min(d_hi, dc / dg)
                elif dg < -1e-14:
                    d_lo = max(d_lo, dc / dg)
            
            if d_hi < d_lo:
                # Infeasible full step; try a fraction
                if ga > 0:
                    d = min(max(-0.01, d_lo), min(0.0, d_hi + 0.01))
                else:
                    d = max(min(0.01, d_hi), max(0.0, d_lo - 0.01))
                if abs(d) < 1e-14:
                    continue
            else:
                d = d_lo if ga > 0 else d_hi
            
            # Apply
            new_val = h[j] + d
            if 0.0 <= new_val <= 1.0:
                h[j] = new_val
                n_improved += 1
        
        if n_improved > 0:
            # Fix sum after full pass
            diff = target - np.sum(h)
            if abs(diff) > 1e-10:
                if diff > 0:
                    idx = np.where(h < 1.0 - 1e-10)[0]
                else:
                    idx = np.where(h > 1e-10)[0]
                if len(idx) > 0:
                    h[idx] += diff / len(idx)
                    h = np.clip(h, 0.0, 1.0)
        else:
            break
    
    return best_h, best_c5

def fourier_perturb(h, n, target, rng, scale=0.05, M=10):
    """Add a small Fourier perturbation to escape local optima."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    pert = np.zeros(n)
    for m in range(1, M+1):
        a = rng.randn() * scale
        b = rng.randn() * scale
        pert += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    h_new = h + pert
    return project_feasible(h_new, target)
