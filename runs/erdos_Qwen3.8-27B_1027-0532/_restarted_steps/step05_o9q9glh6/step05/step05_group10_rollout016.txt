def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10 and len(idx) > 0:
        h[idx] += residual / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return np.clip(h, 0, 1)
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_with_sparse_grad(h_init, target, alphas, maxiter, ftol, n_active=20):
    n = len(h_init)
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    h_cur = h_init.copy()
    
    for alpha in alphas:
        def obj_grad(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_val = corr * dx
            c_max = np.max(c_val)
            shifted = np.clip(alpha * (c_val - c_max), -400.0, 0.0)
            obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            
            top_idx = np.argsort(c_val)[-n_active:]
            top_vals = c_val[top_idx]
            sh = np.clip(alpha * (top_vals - c_max), -400.0, 0.0)
            ew = np.exp(sh)
            sw = np.sum(ew)
            if sw < 1e-300:
                return obj, np.zeros(n)
            weights = ew / sw
            
            grad = np.zeros(n)
            for i in range(len(top_idx)):
                k = top_idx[i] - (n - 1)
                w = weights[i] * dx
                if abs(w) < 1e-15:
                    continue
                lo1 = max(0, k)
                hi1 = min(n, n + k)
                if hi1 > lo1:
                    grad[lo1:hi1] += w * (1.0 - x[lo1 - k:hi1 - k])
                lo2 = max(0, -k)
                hi2 = min(n, n - k)
                if hi2 > lo2:
                    grad[lo2:hi2] -= w * x[lo2 + k:hi2 + k]
            return obj, grad
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj_grad, h_cur, jac=True, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def active_set_lp(h_init, n, target, time_limit, n_active_max=50, damping=0.9):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr * dx)[-n_active_max:].tolist())
    
    while time.time() - start < time_limit:
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        
        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, n + k)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.SCS, cp.ECOS, cp.GLOP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1-damping)*h + damping*h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        top = set(np.argsort(c_vals)[-n_active_max:].tolist())
        
        # Add any violated constraint
        t_val = best_c5
        for k_idx in range(1, 2*n-1):
            if k_idx not in active_k and k_idx not in top:
                if c_vals[k_idx] > t_val + 0.5 * (best_c5 - t_val) + 1e-5:
                    active_k.add(k_idx)
                    break
        
        active_k = top | active_k
        if len(active_k) > 2 * n_active_max:
            active_k = set(np.argsort(c_vals)[-n_active_max:].tolist())
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_descent(h_init, target, max_steps, time_limit):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n-1)
        grad = np.zeros(n)
        lo1 = max(0, k_star)
        hi1 = min(n, n + k_star)
        if hi1 > lo1:
            grad[lo1:hi1] += dx * (1.0 - h[lo1-k_star:hi1-k_star])
        lo2 = max(0, -k_star)
        hi2 = min(n, n - k_star)
        if hi2 > lo2:
            grad[lo2:hi2] -= dx * h[lo2+k_star:hi2+k_star]
        
        grad -= np.mean(grad)  # project to sum=0 subspace
        
        improved = False
        for lr in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            h_try = project_feasible(h - lr*grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        if not improved and step > 100:
            break
    return best_h, best_c5

The issue is that SLSQP at n=400 with 799 constraints (for the sum) + bounds + the smooth max is a difficult optimization problem. The smooth max with finite alpha doesn't perfectly capture the max, so we might converge to a point that's not quite optimal for the true max.

A better approach might be to use **iterative relinearization** more carefully:

1. Start with h = initial_h_values
2. Compute all g_k(h)
3. Let C = max g_k(h)
4. Solve the LP: minimize t subject to linearized g_k constraints for top m k's
5. Update h
6. Repeat with updated linearization

The key is to use **tight enough** tolerances and **enough iterations**.
