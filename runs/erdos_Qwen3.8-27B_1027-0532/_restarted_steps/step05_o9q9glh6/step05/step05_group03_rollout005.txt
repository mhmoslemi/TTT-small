def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return corr[n-1:n-1+n] * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single_lag(h, k):
    """Gradient of overlap at lag k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        upper = n - 1 - k
        # For j in [0, upper]: grad[j] += dx * (1 - h[j+k])
        grad[0:upper+1] += dx * (1.0 - h[k:k+upper+1])
        # For j in [k, n-1]: grad[j] -= dx * h[j-k]
        grad[k:n] -= dx * h[0:n-k]
    return grad

    n = len(h)
    grad = np.zeros(n)

            grad[k:n] -= w * (2.0/n) * h[0:n-k]

    best_score = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP
    if tleft() > 200:
        h_ref, sc = slsqp_soft_max(best_h, 200, 2000, 1e-14)
        if sc < best_score:
            best_score, best_h = sc, h_ref
        h_ref, sc = slsqp_soft_max(best_h, 2000, 2000, 1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 2: Deflation optimization
    if tleft() > 180:
        for n_top in [1, 2, 3]:
            h_def, sc = deflation_opt(best_h, 300, n_top, 0.004, 0.02, 1.01, 0.92)
            if sc < best_score:
                best_score, best_h = sc, h_def
        h_ref, sc = slsqp_soft_max(best_h, 5000, 1500, 1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 3: Bang-bang refinement
    if tleft() > 160:
        h_bb, sc_bb = try_bang_bang_refine(best_h, target)
        if sc_bb < best_score:
            best_score, best_h = sc_bb, h_bb
            h_ref, sc = slsqp_soft_max(best_h, 1000, 1000, 1e-15)
            if sc < best_score:
                best_score, best_h = sc, h_ref
        h_ref, sc = slsqp_top_lags(best_h, 3, 1000, 1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 4: Interval-based optimization
    if tleft() > 120:
        for n_int in [3, 4, 5]:
            if tleft() < 60:
                break
            h_int, sc_int = interval_optimize(n, n_int, seed + n_int * 100)
            if h_int is not None and sc_int < best_score:
                best_score, best_h = sc_int, h_int
                h_ref, sc = slsqp_soft_max(best_h, 1000, 800, 1e-14)
                if sc < best_score:
                    best_score, best_h = sc, h_ref
    
    # Phase 5: Genetic algorithm on bang-bang
    if tleft() > 120:
        h_ga, sc_ga = ga_bang_bang(best_h, 80, 40, 5, seed + 7)
        if sc_ga < best_score:
            best_score, best_h = sc_ga, h_ga
        h_ref, sc = slsqp_soft_max(best_h, 2000, 1000, 1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 6: Aggressive random restarts
    restart_budget = min(tleft(), 400.0)
    n_restarts = max(3, int(restart_budget / 40.0))
    for attempt in range(n_restarts):
        if tleft() < 50:
            break
        h_pert = best_h.copy()
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        pert_type = attempt % 5
        if pert_type == 0:
            h_pert += rng.randn(n_cur) * 0.08
        elif pert_type == 1:
            block = max(5, n_cur // 25)
            for b in range(n_cur // block):
                if rng.random() < 0.35:
                    h_pert[b*block:min((b+1)*block, n_cur)] += rng.uniform(-0.2, 0.2)
        elif pert_type == 2:
            freq = rng.uniform(0.03, 0.15)
            h_pert += np.sin(np.arange(n_cur) * freq * 2.0 * np.pi + rng.rand() * 6.28) * 0.06
        elif pert_type == 3:
            n_flips = n_cur // 8
            flip_idx = rng.choice(n_cur, n_flips, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        else:
            shift_amt = rng.randint(2, n_cur // 10)
            h_pert = np.roll(h_pert, shift_amt)
        
        h_pert = project_feasible(h_pert, target_cur)
        h_def, sc_def = deflation_opt(h_pert, 150, 3, 0.003, 0.015, 1.01, 0.93)
        h_ref, sc = slsqp_soft_max(h_def, 500, 1000, 1e-14)
        if sc < best_score:
            best_score, best_h = sc, h_ref
        h_ref2, sc2 = slsqp_top_lags(h_ref, 2, 800, 1e-15)
        if sc2 < best_score:
            best_score, best_h = sc2, h_ref2
    
    # Phase 7: Simulated annealing
    if tleft() > 60:
        h_sa, sc_sa = simulated_annealing(best_h, 3000, 0.005, 0.0001, 0.999, seed + 99)
        if sc_sa < best_score:
            best_score, best_h = sc_sa, h_sa
        h_ref, sc = slsqp_soft_max(best_h, 5000, 1000, 1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 8: Multi-lag surgery
    if tleft() > 40:
        for nl in [1, 2, 3, 5]:
            h_surg, sc_surg = multi_lag_surgery(best_h, nl, 200, 0.001)
            if sc_surg < best_score:
                best_score, best_h = sc_surg, h_surg
        h_ref, sc = slsqp_soft_max(best_h, 20000, 1000, 1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 9: Final high-alpha SLSQP
    if tleft() > 30:
        h_ref, sc = slsqp_soft_max(best_h, 50000, 800, 1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_ref
        h_ref, sc = slsqp_top_lags(best_h, 1, 500, 1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    # Phase 10: Final surgery + polish
    if tleft() > 20:
        h_surg, sc_surg = surgery_worst_lag(best_h, 100, 0.0005)
        if sc_surg < best_score:
            best_score, best_h = sc_surg, h_surg
        h_ref, sc = slsqp_soft_max(best_h, 100000, 300, 1e-17)
        if sc < best_score:
            best_score, best_h = sc, h_ref
    
    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
