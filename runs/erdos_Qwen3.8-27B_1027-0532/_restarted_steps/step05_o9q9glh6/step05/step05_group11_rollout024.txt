OK let me think of a cleaner approach. The function f(h) = max_k (a_k * h + b_k) is convex piecewise linear on [0,1]. We want to find its minimum.

The subgradient of f at h is the set {a_k : a_k*h + b_k = f(h)} (the slopes of all lines achieving the max at h).

f is minimized at h* iff 0 ∈ ∂f(h*), i.e., iff there exist lines achieving the max at h* with both positive and negative slopes (or one with zero slope).

Algorithm:
1. Start with h = 0.5
2. Compute f(h) = max(a_k*h + b_k)
3. Find all k with a_k*h + b_k = f(h) (within tolerance)
4. If min active slope ≤ 0 ≤ max active slope: done, return h
5. If all active slopes > 0: f is increasing at h, so minimum is to the left
   - Set h_upper = h, h = h/2 (or use bisection)
6. If all active slopes < 0: f is decreasing at h, so minimum is to the right
   - Set h_lower = h, h = (h+1)/2 (or use bisection)

This is bisection on the subgradient, which converges in O(log(1/eps) * n) time.

def coordinate_descent(h_init, n, target, max_sweeps, time_limit, order='forward'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)

    best_c5 = compute_c5(h)

def solve_all_lag_lp(h_current, n, target):
    """Solve the all-lag LP relaxation."""
    dx = 2.0 / n
    g = 1.0 - h_current
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    for k in range(-(n-1), n):
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end > i_start:
            coeff = np.zeros(n)
            coeff[i_start:i_end] = g[i_start+k:i_end+k] * dx
            if coeff.max() > 1e-15:
                constraints.append(coeff @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
        except Exception:
            return None
    
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def alternating_lp(h_init, n, target, max_iter, time_limit, 
                   damping_start=1.0, damping_end=0.7):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        progress = min(1.0, it / max(1, max_iter * 0.5))
        damping = damping_start + (damping_end - damping_start) * progress
        
        h_lp = solve_all_lag_lp(h, n, target)
        if h_lp is None:
            break
        
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
            h = h_new
        else:
            # Still update h for the next LP (might help escape)
            h = h_new
    
    return best_h, best_c5

def smooth_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def min_max_lines(slopes, intercepts, h_low=0.0, h_high=1.0):
    for _ in range(80):
        h_mid = 0.5 * (h_low + h_high)
        values = slopes * h_mid + intercepts
        f_max = values.max()
        active = values >= f_max - 1e-12
        if not active.any():
            idx = int(np.argmax(values))
            active = np.zeros(len(slopes), dtype=bool)
            active[idx] = True
        min_s = slopes[active].min()
        max_s = slopes[active].max()
        if min_s <= 0.0 and max_s >= 0.0:
            return h_mid
        elif min_s > 0.0:
            h_high = h_mid
        else:
            h_low = h_mid
    return 0.5 * (h_low + h_high)


def coordinate_descent(h_init, n, target, max_sweeps, time_limit, seed=0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    lags = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr.copy()

        if sweep % 3 == 0:
            indices = rng.permutation(n)
        else:
            indices = np.arange(n)

        for j in indices:
            if time.time() - start > time_limit:
                break

            j_pk = j + lags
            j_mk = j - lags

            h_jpk = np.zeros(2 * n - 1)
            v1 = (j_pk >= 0) & (j_pk < n)
            h_jpk[v1] = h[j_pk[v1]]

            h_jmk = np.zeros(2 * n - 1)
            v2 = (j_mk >= 0) & (j_mk < n)
            h_jmk[v2] = h[j_mk[v2]]

            # slope for lag k: dx*(1 - h[j+k] - h[j-k])
            # For k=0, this gives dx*(1-2*h[j]) which is the derivative approximation
            # We exclude k=0 from the optimization (handle via projection)
            slopes = dx * (1.0 - h_jpk - h_jmk)

            # Set slope for k=0 to 0 (exclude from optimization)
            idx_k0 = n - 1
            slopes[idx_k0] = 0.0

            intercepts = overlaps - slopes * h[j]
            # Fix k=0 intercept to current value (don't optimize)
            intercepts[idx_k0] = overlaps[idx_k0]

            h_new = min_max_lines(slopes, intercepts, 0.0, 1.0)

            if abs(h_new - h[j]) > 1e-15:
                delta = h_new - h[j]
                h[j] = h_new
                # Update overlaps (k=0 is handled by slopes=0 so no change there)
                overlaps += slopes * delta

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()

    return best_h, best_c5


def solve_all_lag_lp(h_current, n, target):
    dx = 2.0 / n
    g = 1.0 - h_current
    h_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]

    for k in range(-(n - 1), n):
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end > i_start:
            coeff = np.zeros(n)
            coeff[i_start:i_end] = g[i_start + k:i_end + k] * dx
            if coeff.min() > 1e-15:
                constraints.append(coeff @ h_var <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8, verbose=False)
            except Exception:
                return None

    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)


def alternating_lp(h_init, n, target, max_iter, time_limit,
                   damping_start=1.0, damping_end=0.8):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break

        progress = min(1.0, it / max(1, max_iter * 0.4))
        damping = damping_start + (damping_end - damping_start) * progress

        h_lp = solve_all_lag_lp(h, n, target)
        if h_lp is None:
            break

        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1

    return best_h, best_c5


def _global_n_slsqp = 0
_global_target_slsqp = 0.0
_global_alpha_slsqp = 100.0


def smooth_obj_slsqp(h):
    n = _global_n_slsqp
    dx = 2.0 / n
    alpha = _global_alpha_slsqp
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad_slsqp(h):
    n = _global_n_slsqp
    dx = 2.0 / n
    alpha = _global_alpha_slsqp
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    sw = weights.sum()
    if sw < 1e-30:
        return np.zeros(n)
    weights = weights / sw

    grad = np.zeros(n)
    n_half = n - 1
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - n_half
        i_start = max(0, -k)
        i_end = min(n, n - k)
        for i in range(i_start, i_end):
            ik = i + k
            grad[i] += w * (1.0 - h[ik]) * dx
            im = i - k
            if 0 <= im < n and im != i:
                grad[i] += w * (-h[im]) * dx
            elif im == i:
                grad[i] += w * (-h[i]) * dx
    return grad


def cons_fun_slsqp(h):
    return np.sum(h) - _global_target_slsqp


def cons_jac_slsqp(h):
    return np.ones(_global_n_slsqp)


def slsqp_optimize(h_start, n, target, alphas, maxiter):
    global _global_n_slsqp, _global_target_slsqp, _global_alpha_slsqp
    _global_n_slsqp = n
    _global_target_slsqp = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun_slsqp, 'jac': cons_jac_slsqp}]

    h_cur = h_start.copy()
    for alpha in alphas:
        _global_alpha_slsqp = alpha
        try:
            res = minimize(smooth_obj_slsqp, h_cur, jac=smooth_grad_slsqp,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            c5_cur = compute_c5(h_cur)
            if c5_new < c5_cur:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def push_toward_binary(h, target, strength=0.1):
    """Push values toward 0 or 1 to reduce k=0 overlap."""
    h_new = h.copy()
    mask_low = h < 0.5
    mask_high = h >= 0.5
    h_new[mask_low] -= strength * h_new[mask_low]
    h_new[mask_high] += strength * (1.0 - h_new[mask_high])
    return project_feasible(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent (exact per coordinate, k≠0)
    if tleft() > 250:
        cd_time = min(350, tleft() * 0.3)
        h_cd, c5_cd = coordinate_descent(best_h, n, target,
                                         max_sweeps=500, time_limit=cd_time, seed=seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: All-lag LP
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=500, time_limit=lp_time,
                                     damping_start=1.0, damping_end=0.85)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: More coordinate descent
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd2, c5_cd2 = coordinate_descent(best_h, n, target,
                                           max_sweeps=500, time_limit=cd_time, seed=seed + 1)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 4: SLSQP polish
    if tleft() > 150:
        h_sls = slsqp_optimize(best_h, n, target,
                                [50, 200, 1000, 5000],
                                maxiter=500)
        sc = compute_c5(h_sls)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    # Phase 5: LP after SLSQP
    if tleft() > 120:
        lp_time = min(120, tleft() * 0.4)
        h_lp2, c5_lp2 = alternating_lp(best_h, n, target,
                                       max_iter=300, time_limit=lp_time,
                                       damping_start=0.9, damping_end=0.75)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 6: Final coordinate descent
    if tleft() > 100:
        cd_time = min(120, tleft() * 0.5)
        h_cd3, c5_cd3 = coordinate_descent(best_h, n, target,
                                           max_sweeps=400, time_limit=cd_time, seed=seed + 2)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 7: Binary push + coordinate descent
    if tleft() > 100:
        h_bin = push_toward_binary(best_h, target, strength=0.05)
        h_bin, c5_bin = coordinate_descent(h_bin, n, target,
                                           max_sweeps=200, time_limit=min(60, tleft() * 0.5),
                                           seed=seed + 3)
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin

    # Phase 8: Try higher resolution
    if tleft() > 200:
        for n_hi in [600, 800]:
            if tleft() < 120:
                break
            target_hi = n_hi / 2.0
            h_hi = interpolate_to(best_h, n_hi, target_hi)
            h_hi, c5_hi = coordinate_descent(h_hi, n_hi, target_hi,
                                             max_sweeps=200, time_limit=min(80, tleft() * 0.4),
                                             seed=seed + 10)
            h_hi, c5_lp_hi = alternating_lp(h_hi, n_hi, target_hi,
                                            max_iter=150, time_limit=min(50, tleft() * 0.3),
                                            damping_start=1.0, damping_end=0.85)
            if c5_lp_hi < c5_hi:
                c5_hi = c5_lp_hi
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_hi
                n = n_hi
                target = target_hi
                break

    # Phase 9: Final LP polish
    if tleft() > 80:
        lp_time = min(60, tleft() * 0.6)
        h_final_lp, c5_final_lp = alternating_lp(best_h, n, target,
                                                 max_iter=200, time_limit=lp_time,
                                                 damping_start=0.95, damping_end=0.85)
        if c5_final_lp < best_score:
            best_score = c5_final_lp
            best_h = h_final_lp

    # Phase 10: Final coordinate descent
    if tleft() > 50:
        cd_time = min(40, tleft() * 0.7)
        h_final_cd, c5_final_cd = coordinate_descent(best_h, n, target,
                                                     max_sweeps=300, time_limit=cd_time,
                                                     seed=seed + 100)
        if c5_final_cd < best_score:
            best_score = c5_final_cd
            best_h = h_final_cd

    # Phase 11: Final SLSQP
    if tleft() > 40:
        h_final_sls = slsqp_optimize(best_h, n, target,
                                     [2000, 10000, 50000],
                                     maxiter=300)
        sc = compute_c5(h_final_sls)
        if sc < best_score:
            best_score = sc
            best_h = h_final_sls

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
