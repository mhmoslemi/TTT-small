**Pair Coordinate Descent (PCD):**

Given h, we want to find a pair (i,j) and a step t such that h[i]+t and h[j]-t (with 0<=both<=1) minimize c5.

For a given pair (i,j), the correlation values as a function of t are:
c_k(t) = c_k + t * g_k

where g_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]

We want: min_t max_k c_k(t) subject to max(-h[i], h[j]-1) <= t <= min(1-h[i], h[j])

This is a 1D convex optimization. The optimal t is where the two "tightest" constraints intersect (or at a boundary).

To find it efficiently:
1. Compute all g_k values (O(n))
2. Sort by g_k (O(n log n))
3. The optimal t is where the upper envelope changes slope to include 0 in its subgradient

Actually, a simpler approach: since the function is convex, use ternary search.

```python
        h_new = h.copy()  # Actually, we can be smarter
        h_new[i] += t
        h_new[j] -= t
        score = compute_c5(h_new)

        if f(m1) < f(m2):

        else:

    # The sum is preserved exactly since we're adding t to one and subtracting from another
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_of_lag(h, idx):
    """Gradient of correlation[idx] with respect to h."""
    n = len(h)
    s = n - 1 - idx
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    else:
        a = -s
        grad[:n-a] += (1.0 - h[a:])
        grad[a:] -= h[:n-a]
    return grad

def weighted_grad(h, weights):
    """Weighted sum of gradients: sum_k weights[k] * grad_k"""
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-15:
            continue
        s = n - 1 - idx
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            a = -s
            grad[:n-a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n-a]
    return grad

def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def gradient_line_search(h, n, target, n_active, beta, n_ls_trials=30, deadline=None):
    """Multi-lag gradient descent with exact line search."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = 0.005
    
    for it in range(2000):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr  # use all lags
        max_val = float(np.max(over))
        
        # Get top-M lags
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(beta * diffs)
        weights /= weights.sum()
        
        # Compute weighted gradient
        full_weights = np.zeros(len(over))
        full_weights[top_idx] = weights
        grad = weighted_grad(h_cur, full_weights)
        
        # Project gradient to sum=0
        grad = grad - np.mean(grad)
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in np.linspace(0.001, lr, n_ls_trials):
            h_new = h_cur - t * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            h_cur = h_cur - best_t * grad
            h_cur = project_feasible(h_cur, target)
            best_score = compute_c5(h_cur)
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.95
            if lr < 1e-6:

    return h_cur, best_score
```

```python
    n = len(h)

        grad[:n-s] -= h[s:]
    else:
```
