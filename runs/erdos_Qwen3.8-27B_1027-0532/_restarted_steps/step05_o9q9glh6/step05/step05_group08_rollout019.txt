import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def topk_weighted_grad(h, n, dx, top_idx, weights):
    g = np.zeros(n)
    for m, wk in zip(top_idx, weights):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lse_value_and_grad(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
        ws = 1.0
    w /= ws
    val = c_max + float(np.log(ws)) / alpha
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    thresh = 1e-12
    for m in range(len(corr)):
        if w[m] > thresh:
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += w[m] * (1.0 - h[k:])
                g[k:] -= w[m] * h[:n - k]
            else:
                ell = -k
                g[ell:] += w[m] * (1.0 - h[:n - ell])
                g[:n - ell] -= w[m] * h[ell:]
    g *= dx
    return val, g


def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def obj_fn(x):
        return lse_value_and_grad(x, alpha)[0]

    def objg_fn(x):
        return lse_value_and_grad(x, alpha)[1]

    bounds = [(0.0, 1.0)] * n
    def cf(x):
        return np.sum(x) - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', jac=objg_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj_fn(x):
        v, _ = lse_value_and_grad(x, alpha)
        return v + mu * (np.sum(x) - target) ** 2

    def objg_fn(x):
        _, g = lse_value_and_grad(x, alpha)
        return g + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, method='L-BFGS-B', jac=objg_fn,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-11, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def nesterov_subgrad(h, n, target, alpha, n_steps, deadline):
    h_cur = h.copy()
    h_ys = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    step = 1.0 / np.sqrt(alpha)
    for t in range(n_steps):
        if time.time() > deadline:
            break
        h_look = h_cur + 0.9 * (h_cur - h_ys)
        h_look = project_feasible(h_look, target)
        _, g = lse_value_and_grad(h_look, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        f_cur = compute_c5(h_cur)
        h_trial = h_cur - step * g
        h_trial = project_feasible(h_trial, target)
        f_trial = compute_c5(h_trial)
        if f_trial < f_cur + 1e-15:
            h_ys = h_cur.copy()
            h_cur = h_trial
            if f_trial < best_score:
                best_score = f_trial
                best_h = h_cur.copy()
        else:
            step *= 0.7
            if step < 1e-10:
                break
            h_ys = h_cur.copy()
    return best_h, best_score


def topk_subgrad(h, n, target, top_k, n_steps, lr_init, deadline):
    h_cur = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_cur = float(np.max(corr)) * dx
        if c_cur < best_score:
            best_score = c_cur
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        vals = corr[top_idx]
        diffs = vals - vals[0]
        w = np.exp(diffs * 10.0)
        w /= w.sum()
        g = topk_weighted_grad(h_cur, n, dx, top_idx, w)
        h_trial = h_cur - lr * g
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < c_cur:
            h_cur = h_trial
        else:
            lr *= 0.85
            if lr < 1e-11:
                break
    return best_h, best_score


def surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        g = topk_weighted_grad(h_cur, n, dx, [m_star], [1.0])
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.015, 0.02)
        else:
            eps *= 0.88
            if eps < 1e-10:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def balance_top_lags(h, n, target, m, max_iter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:m]
        max_val = corr[top_idx[0]]
        min_val = corr[top_idx[-1]]
        if max_val - min_val < 1e-10:
            break
        spread = max_val - min_val
        weights = np.zeros(len(top_idx))
        for i in range(m):
            weights[i] = (corr[top_idx[i]] - min_val) / spread + 0.1
        weights /= weights.sum()
        g = topk_weighted_grad(h_cur, n, dx, top_idx, weights)
        h_trial = h_cur - 0.005 * g
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < best_score:
            best_score = sc
            h_cur = h_trial
        else:
            h_trial = h_cur - 0.002 * g
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < best_score:
                best_score = sc
                h_cur = h_trial
            else:
                break
    return h_cur, best_score
