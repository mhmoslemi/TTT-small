import numpy as np
from scipy.optimize import minimize
import time

_slsqp_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target_sum]] = 1.0
    return h_bin


def greedy_enhanced(h, max_iters, time_limit, start_time, scan_width=25, top_shifts=8):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 1500

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        top_indices = np.argsort(-corr)[:top_shifts]
        top_vals = corr[top_indices].astype(float)
        top_shifts_arr = top_indices - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        k_star = top_shifts_arr[0]
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]

        if len(i_cand) == 0:
            k_alt = top_shifts_arr[1] if len(top_shifts_arr) > 1 else 0
            i_cand = [i for i in ones if 0 <= i + k_alt < n and h[i + k_alt] == 0]
            j_cand = [j for j in zeros if 0 <= j + k_alt < n and h[j + k_alt] == 1]
            if len(i_cand) == 0 or len(j_cand) == 0:
                stall += 100
                continue

        best_improve = 0.0
        best_i = -1
        best_j = -1

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_pos in range(len(top_indices)):
                    k = top_shifts_arr[t_pos]
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[t_pos] + d

                        nm = v
                        break

                    best_j = j

                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _slsqp_target


def _cons_jac(h):
    return np.ones(_slsqp_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def golden_ratio_init(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    ones_count = n // 2
    positions = np.zeros(ones_count)
    for i in range(ones_count):
        positions[i] = (i * phi) % 1.0
    positions = np.sort(positions)
    h = np.zeros(n)
    for pos in positions:
        idx = int(pos * n)
        if idx >= n:
            idx = n - 1
        if h[idx] == 0:
            h[idx] = 1.0
        else:
            for d in range(1, n):
                if idx + d < n and h[idx + d] == 0:
                    h[idx + d] = 1.0
                    break
                elif idx - d >= 0 and h[idx - d] == 0:
                    h[idx - d] = 1.0
                    break
    h = project_binary(h, n // 2)
    return h


def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v > 0:
            cnt += v & 1
            v >>= 1
        tm[i] = 1.0 if cnt % 2 == 0 else 0.0
    target = n // 2
    ones = int(tm.sum())
    if ones > target:
        order = np.where(tm == 1)[0]
        tm[order[ones - target:]] = 0.0
    elif ones < target:
        order = np.where(tm == 0)[0]
        tm[order[:(target - ones)]] = 1.0
    return tm

                best_score = c5_pg
                best_h = h_pg.copy()

            if tleft() < 25:
                break
            allowed2 = min(20, tleft() * 0.04)
            max_it_p = int(min(40000, allowed2 * 2000))
            h_ps, c5_ps = sa_diverse(h_pg, target_ils, max_it_p, 0.004, 0.00001, rng,
                                      time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 8: Final SLSQP polish at best solution
    if tleft() > 30:
        n_final_use = len(best_h)
        target_final_use = float(n_final_use) / 2.0
        alphas_final = [10000, 25000, 50000, 100000]
        allowed_final = max(5.0, min(tleft() - 10, 120))
        h_last, sc_last = slsqp_refine(best_h, n_final_use, target_final_use,
                                        alphas_final, 5000, 1e-16,
                                        time_limit=allowed_final,
                                        start_time=start_time)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    # Phase 9: Final binary SA polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_f = project_binary(best_h, target_use)
        allowed = max(3.0, min(tleft() - 5, 40))
        max_it_f = int(min(50000, allowed * 2500))
        h_sf, c5_sf = sa_diverse(h_bin_f, target_use, max_it_f, 0.003, 0.000005,
                                  rng, time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = project_feasible(best_h, target_final)
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
