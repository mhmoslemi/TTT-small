def binary_local_search(h, n, target, n_flips, deadline, rng):

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for _ in range(n_flips):
        if time.time() > deadline:
            break
        # Find "boundary" points (between 0.2 and 0.8)
        boundary = np.where((h_cur > 0.2) & (h_cur < 0.8))[0]
        if len(boundary) < 2:

        # Try flipping a random boundary point
        idx = boundary[rng.randint(len(boundary))]

        h_test = h_cur.copy()
        if h_cur[idx] > 0.5:
            h_test[idx] = max(0.0, h_cur[idx] - 0.8)
        else:
            h_test[idx] = min(1.0, h_cur[idx] + 0.8)
        h_test = project_feasible(h_test, target)
        sc = compute_c5(h_test)
        if sc < cur_score - 1e-14:
            h_cur = h_test
            cur_score = sc
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        # Focus on the overlap region (lags from n-1 to 2n-2)
        over = corr[n-1:2*n-1]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:

            best_h = h_cur.copy()
        # Get top active lags

        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.01)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k < n:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                if k > 0:
                    j_neg = np.arange(k, n)
                    grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay

            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 5
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 6)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.3, target)
            else:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.4, 0.05, 0.95), target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r_arr = rng.rand(n)
            trial[r_arr > cr] = pop[i][r_arr > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def multiscale_refine(h, n, target, n_coarse, deadline):
    """Downscale, optimize, upscale, then refine"""
    h_coarse = interpolate_to(h, n_coarse, n_coarse / 2.0)
    h_coarse_opt, sc_coarse = slsqp_optimize(h_coarse, n_coarse, n_coarse / 2.0,
                                              [50, 200, 800], 1000, deadline)
    h_up = interpolate_to(h_coarse_opt, n, target)
    h_refined, sc_ref = slsqp_optimize(h_up, n, target, [200, 1000, 4000], 1500, deadline)
    return h_refined, sc_ref


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 12.0

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP from parent (60s)
    dl1 = start_time + 55
    h1, sc1 = slsqp_optimize(best_h, n_final, target_final, [100, 500, 2000], 2500, dl1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: Sharpening cycles (60-200s)
    for cycle in range(4):
        dl2 = start_time + 65 + (cycle + 1) * 35
        if time.time() > dl2 - 5:
            break
        eta = 0.15 * (cycle + 1)
        h_sharp = sharpen(best_h, target_final, eta)
        h_opt, sc = slsqp_optimize(h_sharp, n_final, target_final,
                                    [200, 1000, 5000, 20000], 2000, dl2)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 3: Subgradient + SLSQP (200-350s)
    dl3 = start_time + 340
    if time.time() < dl3 - 20:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                         400, 6, 0.005, 0.997, dl3, rng)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_opt, sc = slsqp_optimize(best_h, n_final, target_final,
                                    [500, 2000, 8000], 2000, dl3)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 4: Binary SA + refinement (350-550s)
    dl4 = start_time + 540
    if time.time() < dl4 - 30:
        h_sa, sc_sa = binary_sa(best_h, n_final, target_final,
                                 4000, 0.002, 1e-5, start_time + 470, rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa
        h_opt, sc = slsqp_optimize(best_h, n_final, target_final,
                                    [1000, 5000, 20000, 80000], 2000, dl4)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 5: Multi-scale downscale/upscale (550-700s)
    dl5 = start_time + 690
    if time.time() < dl5 - 30:
        for n_c in [100, 64]:
            if time.time() > dl5 - 30:
                break
            h_ms, sc_ms = multiscale_refine(best_h, n_final, target_final,
                                            n_c, dl5)
            if sc_ms < best_score:
                best_score = sc_ms
                best_h = h_ms

    # Phase 6: Worst-lag surgery (700-800s)
    dl6 = start_time + 790
    if time.time() < dl6 - 20:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final,
                                             500, 0.003, dl6)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_opt, sc = slsqp_optimize(best_h, n_final, target_final,
                                    [2000, 8000, 30000], 2000, dl6)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 7: DE at full resolution (800-1000s)
    dl7 = start_time + 990
    if time.time() < dl7 - 50:
        pop = [best_h.copy()]
        for i in range(11):
            scale = rng.uniform(0.02, 0.08)
            p = best_h + rng.randn(n_final) * scale
            pop.append(project_feasible(p, target_final))
        x = np.linspace(0, 2, n_final)
        for i in range(2):
            freq = rng.uniform(2, 10)
            amp = rng.uniform(0.2, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            pop.append(project_feasible(0.5 + amp * np.sin(freq * np.pi * x + phase), target_final))
        h_de, sc_de = de_optimize(n_final, target_final, 14, 50, 0.4, 0.35,
                                   min(start_time + 920, dl7), rng, init_pop=pop)
        if sc_de < best_score:
            best_score = sc_de
            best_h = h_de
        h_opt, sc = slsqp_optimize(best_h, n_final, target_final,
                                    [1000, 5000, 20000], 2000, min(start_time + 990, dl7))
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 8: Another sharpening + SLSQP cycle (1000-1100s)
    dl8 = start_time + 1090
    if time.time() < dl8 - 20:
        h_sharp = sharpen(best_h, target_final, 0.3)
        h_opt, sc = slsqp_optimize(h_sharp, n_final, target_final,
                                    [5000, 20000, 60000], 2000, dl8)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 9: Final subgradient (1100-1200s)
    dl9 = start_time + 1190
    if time.time() < dl9 - 20:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                          300, 10, 0.002, 0.998, dl9, rng)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 10: Ultra-fine SLSQP (1200-1280s)
    dl10 = hard_deadline
    if time.time() < dl10 - 15:
        h_opt, sc = slsqp_optimize(best_h, n_final, target_final,
                                    [20000, 60000, 120000], 1500, dl10)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
