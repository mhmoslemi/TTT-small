import numpy as np
import time

def compute_corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_grad_full(h, weights, fsize):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    w_rev = weights[:][::-1]
    W_rev = np.fft.rfft(w_rev, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    term1 = np.fft.irfft(W_rev * V, fsize)[:2 * n - 1]
    term2 = np.fft.irfft(W * H, fsize)[:-1]
    grad = term1[:n] + term1[2 * n - 1:n:-1][::-1]
    grad = term1[:n]
    g2 = term2[:n]
    g2_ext = term2[2 * n - 1 - 1:n - 1:-1]
    grad_full = term1[:n] + np.zeros(n)
    t2_left = term2[1:n + 1]
    t2_right = term2[2 * n - 1:n - 1:-1][::-1]
    grad_out = np.zeros(n)
    grad_out[:n - 1] = term1[1:n]
    grad_out[1:n] += term2[:n - 1][::-1]
    grad_out[:n - 1] += term2[n - 1:]
    return grad_out * dx

def lse_grad_full_v2(h, weights, fsize):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    num_lags = 2 * n - 1
    k_arr = np.arange(-(n - 1), n)
    w_rev = weights[::-1]
    W_rev = np.fft.rfft(w_rev, fsize)
    V = np.fft.rfft(v, fsize)
    cross_wv = np.fft.irfft(W_rev * V, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    cross_wh = np.fft.irfft(W * H, fsize)
    zero_pad_pos = np.zeros(n - 1)
    v_ext = np.concatenate([v, zero_pad_pos])
    h_ext = np.concatenate([h, zero_pad_pos])
    w_ext1 = np.concatenate([w_rev, np.zeros(fsize - num_lags)])
    W_ext1 = np.fft.rfft(w_ext1, fsize)
    V_ext = np.fft.rfft(v_ext, fsize)
    cross1 = np.fft.irfft(W_ext1 * V_ext, fsize)
    grad_t1 = cross1[:n]
    w_ext2 = np.concatenate([weights, np.zeros(fsize - num_lags)])
    W_ext2 = np.fft.rfft(w_ext2, fsize)
    H_ext = np.fft.rfft(h_ext, fsize)
    cross2 = np.fft.irfft(W_ext2 * H_ext, fsize)
    grad_t2 = cross2[:n]
    return (grad_t1 - grad_t2) * dx

def lse_obj_grad(h, alpha, fsize):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    grad = lse_grad_full_v2(h, weights, fsize)
    return obj, grad

def fista_optimize(h0, target, alpha, deadline, lr_mult=1.0):
    n = len(h0)
    fsize = 2 * n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    lr_init = lr_mult / (max(alpha * 0.1, 1.0))
    step = lr_init
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_grad(y_clipped, alpha, fsize)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_grad(x_new, alpha, fsize)
            decrease = obj_y - obj_new
            dist2 = np.dot(x_new - y_clipped, x_new - y_clipped)
            quadratic = 0.5 * (1.0 / step) * dist2
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-14)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def grad_lag_full(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    abs_k = abs(k)
    if k > 0:
        grad[:n - abs_k] += dx * (1.0 - h[abs_k:])
        grad[abs_k:] += dx * (1.0 - h[:n - abs_k])
        grad[:n - abs_k] -= dx * h[abs_k:]
        grad[abs_k:] -= dx * h[:n - abs_k]
    else:
        m = abs_k
        grad[m:] += dx * (1.0 - h[:n - m])
        grad[:n - m] += dx * (1.0 - h[m:])
        grad[:n - m] -= dx * h[m:]
        grad[m:] -= dx * h[:n - m]
    return grad

def direct_subgrad_max(h0, target, deadline):
    n = len(h0)
    h_cur = project_feasible(h0, target).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    x_prev = h_cur.copy()
    y = h_cur.copy()
    t = 1.0
    step = 0.5
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_prev + ((t - 1.0) / t_new) * (h_cur - x_prev) if t > 1 else h_cur
        y = np.clip(y, 0.0, 1.0)
        corr = compute_corr_full(y)
        max_val = float(np.max(corr))
        threshold = max_val - 1e-6
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        w = np.zeros(2 * n - 1)
        active_vals = corr[active]
        w_act = active_vals - (max_val - 0.001)
        w_act = np.clip(w_act, 0.0, None)
        if w_act.sum() < 1e-10:
            w_act = np.ones(len(active))
        w_act /= w_act.sum()
        grad = np.zeros(n)
        for wi, am in zip(w_act, active):
            lag = int(am) - (n - 1)
            grad += wi * grad_lag_full(y, lag)
        x_new = project_feasible(y - step * grad, target)
        obj_new = compute_c5(x_new)
        if obj_new > compute_c5(y) + 1e-10:
            step *= 0.5
            x_new = project_feasible(y - step * grad, target)
        else:
            step = min(step * 1.05, 2.0)
        sc = compute_c5(x_new)
        x_prev = h_cur
        h_cur = x_new
        t = t_new
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score

def focal_polish(h0, target, deadline, K=3):
    n = len(h0)
    dx = 2.0 / n
    h_cur = project_feasible(h0, target).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    eps = 0.01
    while time.time() < deadline:
        corr = compute_corr_full(h_cur)
        max_val = float(np.max(corr))
        threshold = max_val - 0.002
        active = np.where(corr >= threshold)[0]
        if len(active) > K:
            top_idx = np.argsort(corr[active])[::-1][:K]
            active = active[top_idx]
        active_vals = corr[active]
        w = active_vals - (max_val - 0.001)
        w = np.clip(w, 1e-6, None)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, am in zip(w, active):
            lag = int(am) - (n - 1)
            grad += wi * grad_lag_full(h_cur, lag)
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = eps * factor
            h_new = project_feasible(h_cur - step * grad, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                h_cur = h_new
                best_score = sc
                best_h = h_cur.copy()
                improved = True
                break
        if not improved:
            eps *= 0.8
            if eps < 1e-5:
                break
    return best_h, best_score
