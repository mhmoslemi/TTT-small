import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):

        else:

            extra = target_sum - final_sum

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def lsgd_gradient(h, p):
    """Compute gradient of (sum |C[k]|^p)^(1/p) with respect to h."""
    n = len(h)
    C = full_corr_fast(h)
    F = float(np.sum(np.abs(C)**p))**(1.0/p)
    if F < 1e-30:
        return np.zeros(n), F
    
    w = np.abs(C)**(p-2) * C
    grad = np.zeros(n)
    positions = np.arange(n)
    
    for m in range(2*n-1):
        k = m - (n-1)  # lag
        # dC[m]/dh[p] = (1-h[p+k]) - h[p-k] (with boundary conditions)
        s1 = positions + k
        mask1 = (s1 >= 0) & (s1 < n)
        term1 = np.zeros(n)
        if np.any(mask1):
            term1[mask1] = 1.0 - h[s1[mask1]]
        
        s2 = positions - k
        mask2 = (s2 >= 0) & (s2 < n)
        term2 = np.zeros(n)
        if np.any(mask2):
            term2[mask2] = h[s2[mask2]]
        
        grad += w[m] * (term1 - term2)
    
    grad *= F**(1.0-p)
    return grad, F

def lsgd_refine(h, p_values, max_iters_per_p, lr_init, lr_decay, time_limit, start_time):
    """Lp-smoothed projected gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    
    best_score = compute_c5(h)
    best_h = h.copy()
    
    for p in p_values:
        lr = lr_init
        stall = 0
        max_stall = max(50, max_iters_per_p // 4)
        
        for it in range(max_iters_per_p):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            
            grad, F = lsgd_gradient(h, p)
            
            # Gradient step
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            
            # Check improvement
            new_score = compute_c5(h_new)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            
            h = h_new
            lr *= lr_decay
            
            if stall > max_stall:
                break
    
    return best_h, best_score
