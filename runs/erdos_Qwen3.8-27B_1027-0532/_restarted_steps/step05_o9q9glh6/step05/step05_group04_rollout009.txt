        corr = compute_corr(h_cur)
        over = corr[n - 1:2 * n - 1]
        top_idx = np.argsort(over)[::-1][:n_top]
        for k_star in top_idx:
            k_star = int(k_star)
            grad = np.zeros(n)
            j_pos = np.arange(0, n - k_star)
            grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
            grad[j_neg] if False else None
            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]
            grad *= dx
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < best_score - 1e-13:
                h_cur = h_new
                best_score = sc_new
                best_h = h_cur.copy()
    return best_h, best_score

def coarse_de(n, target, pop, gens, seed, deadline=None):
    rng = np.random.RandomState(seed)
    pop_arr = []
    for i in range(pop):
        hv = project_feasible(rng.rand(n), target)
        pop_arr.append(hv)
    scores = [compute_c5(p) for p in pop_arr]
    for gen in range(gens):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop):
            cands = [j for j in range(pop) if j != i]
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            mutant = pop_arr[r1] + 0.8 * (pop_arr[r2] - pop_arr[r3])
            trial = np.zeros(n)
            jr = rng.randint(n)
            cr = 0.9
            for j in range(n):
                if rng.rand() < cr or j == jr:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop_arr[i][j]
            trial = project_feasible(trial, target)
            ts = compute_c5(trial)
            if ts < scores[i]:
                pop_arr[i] = trial
                scores[i] = ts
    idx = int(np.argmin(scores))
    return pop_arr[idx], scores[idx]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - 30.0 - elapsed()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: DE on coarse grid for diverse starts (0-40s)
    deadline1 = start_time + 40
    n_c = 60
    h_de, sc_de = coarse_de(n_c, n_c / 2.0, 25, 60, seed, deadline=deadline1)
    h_de_fine = interpolate_to(h_de, n, target)
    sc_de_fine = compute_c5(h_de_fine)
    if sc_de_fine < best_score:
        best_score = sc_de_fine
        best_h = h_de_fine

    # Phase 2: Nesterov projected gradient with adaptive alpha (40-250s)
    deadline2 = start_time + 250
    alphas = [50, 100, 300, 1000, 3000]
    h_nest, sc_nest = nesterov_projected(best_h, n, target, alphas, 3000, deadline=deadline2)
    if sc_nest < best_score:
        best_score = sc_nest
        best_h = h_nest

    # Phase 3: SLSQP polish (250-350s)
    if remaining() > 80:
        deadline3 = start_time + 350
        for a in [500, 2000]:
            if time.time() > deadline3:
                break
            h_sl = slsqp_smooth(best_h, n, target, a, 3000, deadline=deadline3)
            sc = compute_c5(h_sl)
            if sc < best_score:
                best_score = sc
                best_h = h_sl

    # Phase 4: Reweighted gradient descent (350-550s)
    if remaining() > 150:
        deadline4 = start_time + 550
        h_rw, sc_rw = reweighted_gradient_descent(best_h, n, target,
                                                  n_outer=30, beta_init=200,
                                                  beta_growth=1.4,
                                                  inner_iters=80, lr_init=0.001,
                                                  deadline=deadline4)
        if sc_rw < best_score:
            best_score = sc_rw
            best_h = h_rw

    # Phase 5: Perturbation + re-optimize (550-750s)
    if remaining() > 150:
        deadline5 = start_time + 750
        for trial_scale in [0.05, 0.03, 0.01]:
            if time.time() > deadline5:
                break
            h_pert, sc_pert = perturbation_restart(best_h, n, target,
                                                   n_trials=20, pert_scale=trial_scale,
                                                   deadline=deadline5)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert
            # Quick Nesterov from perturbed point
            h_n2, sc_n2 = nesterov_projected(best_h, n, target,
                                             [500, 2000], 2000, deadline=deadline5)
            if sc_n2 < best_score:
                best_score = sc_n2
                best_h = h_n2

    # Phase 6: Top-lag push + SLSQP (750-950s)
    if remaining() > 150:
        deadline6 = start_time + 950
        for n_top in [2, 3, 5]:
            if time.time() > deadline6:
                break
            h_push, sc_push = top_lag_push(best_h, n, target, n_top=n_top,
                                           n_push=200, eps=0.003, deadline=deadline6)
            if sc_push < best_score:
                best_score = sc_push
                best_h = h_push
        h_sl6 = slsqp_smooth(best_h, n, target, 5000, 3000, deadline=deadline6)
        sc6 = compute_c5(h_sl6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h_sl6

    # Phase 7: Fine-tuned Nesterov with very high alpha (950-1100s)
    if remaining() > 100:
        deadline7 = start_time + 1100
        h_nest2, sc_nest2 = nesterov_projected(best_h, n, target,
                                               [2000, 5000, 10000], 3000, deadline=deadline7)
        if sc_nest2 < best_score:
            best_score = sc_nest2
            best_h = h_nest2

    # Phase 8: Final SLSQP with very high alpha (1100-1250s)
    if remaining() > 80:
        deadline8 = start_time + 1250
        for a in [5000, 20000]:
            if time.time() > deadline8:
                break
            h_final = slsqp_smooth(best_h, n, target, a, 5000, deadline=deadline8)
            sc = compute_c5(h_final)
            if sc < best_score:
                best_score = sc
                best_h = h_final

    # Phase 9: Micro-polish with top-lag surgery (1250-1290s)
    if remaining() > 20:
        deadline9 = start_time + budget_s - 15
        h_surg, sc_surg = top_lag_push(best_h, n, target, n_top=2,
                                       n_push=100, eps=0.001, deadline=deadline9)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n
