```python
            break

            break

        active = b == g_lo

            return hi  # g is decreasing, minimum at hi

            lo = mid
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coord_minimize(a, b):
    """Find t in [0,1] minimizing max_k(a[k]*t + b[k]) using binary search on subgradient."""
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        g = np.max(vals)
        # Find active set (within tolerance)
        tol = 1e-12 * max(1.0, abs(g))
        active = (g - vals) <= tol
        if not np.any(active):
            active = (vals == g)
        min_slope = np.min(a[active])
        max_slope = np.max(a[active])
        if min_slope >= 0:
            hi = mid
        elif max_slope <= 0:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def compute_coefficients(h, i, n, dx):
    """Compute a_ell for all lags ell in [-(n-1), ..., (n-1)] for coordinate i."""
    # corr array has indices 0 to 2n-2, corresponding to lags -(n-1) to (n-1)
    a = np.zeros(2 * n - 1)
    
    # Positive lags ell = 0, 1, ..., n-1 (indices n-1 to 2n-2)
    ell = np.arange(n)
    # term1: (1 - h[i+ell]) for i+ell < n
    valid1 = ell <= (n - 1 - i)
    t1 = np.zeros(n)
    t1[valid1] = 1.0 - h[i + ell[valid1]]
    # term2: -h[i-ell] for i >= ell
    valid2 = ell <= i
    t2 = np.zeros(n)
    t2[valid2] = -h[i - ell[valid2]]
    a[n-1:] = dx * (t1 + t2)
    
    # Negative lags: ell = -m for m = 1, ..., n-1 (indices 0 to n-2)
    m = np.arange(1, n)
    # a_{-m} = dx * [(1-h[i-m])*1_{m<=i} - h[i+m]*1_{i+m<n}]
    valid3 = m <= i
    t3 = np.zeros(n-1)
    t3[valid3] = 1.0 - h[i - m[valid3]]
    valid4 = (i + m) < n
    t4 = np.zeros(n-1)
    t4[valid4] = -h[i + m[valid4]]
    a[:n-1] = dx * (t3 + t4)
    
    return a

def coord_descent(h, n, target, n_sweeps, deadline, random_order=True, rng=None):
    """Coordinate descent with exact 1D minimization."""
    h_cur = h.copy()
    dx = 2.0 / n
    corr_vals = compute_corr(h_cur) * dx
    cur_score = float(np.max(corr_vals))
    
    if rng is not None and random_order:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        for i in order:
            i = int(i)
            a = compute_coefficients(h_cur, i, n, dx)
            b = corr_vals - a * h_cur[i]
            t_new = coord_minimize(a, b)
            if abs(t_new - h_cur[i]) > 1e-14:
                delta = t_new - h_cur[i]
                h_cur[i] = t_new
                corr_vals += a * delta
                new_score = float(np.max(corr_vals))
                if new_score < cur_score:
                    cur_score = new_score
                    sweep_improved = True
        
        # Project to feasibility
        h_cur = project_feasible(h_cur, target)
        corr_vals = compute_corr(h_cur) * dx
        new_score = float(np.max(corr_vals))
        if new_score < cur_score:
            cur_score = new_score
        
        if not sweep_improved and sweep > 10:
            # Check if we've converged
            if sweep > 50:
                # Do a few more sweeps to be safe
                if sweep > 100:
                    break
    
    return h_cur, cur_score

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj_wrapper(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha + 0.5 * (np.sum(x) - target) ** 2
        
        try:
            res = minimize(obj_wrapper, h_cur, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score
```

```python
def compute_coefficients(h, i, n, dx):
    a = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)  # lags from -(n-1) to (n-1)
    
    # i - k: valid when 0 ≤ i-k < n
    ik_minus = i - k  # = i - (idx - n + 1) = i + n - 1 - idx
    valid_ik_minus = (ik_minus >= 0) & (ik_minus < n)
    
    # i + k: valid when 0 ≤ i+k < n
    ik_plus = i + k  # = i + idx - n + 1
    valid_ik_plus = (ik_plus >= 0) & (ik_plus < n)
    
    term1 = np.zeros(2 * n - 1)
    term1[valid_ik_minus] = 1.0 - h[ik_minus[valid_ik_minus]]
    
    term2 = np.zeros(2 * n - 1)
    term2[valid_ik_plus] = h[ik_plus[valid_ik_plus]]
    
    a = dx * (term1 - term2)
    return a
```

For the 2D problem:
minimize max_l(a_l * t1 + c_l * t2 + b_l)
subject to 0 ≤ t1, t2 ≤ 1

This is equivalent to:
minimize z
subject to a_l * t1 + c_l * t2 + b_l ≤ z for all l
           0 ≤ t1, t2 ≤ 1

Using scipy.optimize.linprog:
- Variables: [t1, t2, z]
- Constraints: a_l * t1 + c_l * t2 - z ≤ -b_l for all l
           -t1 ≤ 0, -t2 ≤ 0, t1 ≤ 1, t2 ≤ 1
- Minimize: z (i.e., c = [0, 0, 1])

For 799 constraints, linprog should solve this in <1ms. We could do many pair updates.

But actually, for the 2D case, there's a simpler approach. The minimum of the upper envelope of planes over [0,1]² occurs at:
- A corner: check all 4 corners, O(n) each
- A point on an edge: 1D problem, O(n) each, 4 edges
- An interior point: where 3 planes are "active"

The interior case is the tricky one. At an interior optimum, the subgradient in both directions contains 0. This means:
- There exist active planes with positive and negative "t1-coefficients"
- There exist active planes with positive and negative "t2-coefficients"

Finding such a point requires solving a small system. In general, at an interior optimum of max of L planes over R², exactly 2 or 3 planes are active (by Carathéodory's theorem in 2D).

If 2 planes are active: their intersection gives a point, check if it's optimal.
If 3 planes are active: they might be concurrent (all pass through one point) or form a triangle. The minimum is at the vertex of the triangle that's lowest.

# Helper functions (top-level, no closures)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.99999
        else:
            mask = h > 1e-5
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_coeffs(h, i, n, dx):
    """Compute coefficient vector for coordinate i in all lag integrals."""
    k = np.arange(2 * n - 1) - (n - 1)
    ik_minus = i - k
    ik_plus = i + k
    valid_m = (ik_minus >= 0) & (ik_minus < n)
    valid_p = (ik_plus >= 0) & (ik_plus < n)
    t1 = np.zeros(2 * n - 1)
    t1[valid_m] = 1.0 - h[ik_minus[valid_m]]
    t2 = np.zeros(2 * n - 1)
    t2[valid_p] = h[ik_plus[valid_p]]
    return dx * (t1 - t2)

def coord_minimize(a, b):
    """Binary search for 1D min of max of linear functions."""
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        g = np.max(vals)
        active = (g - vals) <= 1e-11 * max(1.0, abs(g))
        if not active.any():
            active = (vals == g)
        ms = np.min(a[active])
        if ms >= 0:
            hi = mid
        elif np.max(a[active]) <= 0:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def coord_descent(h, n, target, n_sweeps, deadline, rng):
    h_cur = h.copy()
    dx = 2.0 / n
    corr_vals = compute_corr(h_cur) * dx
    cur_score = float(np.max(corr_vals))
    best_sc = cur_score
    best_h = h_cur.copy()
    order = rng.permutation(n)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        n_changes = 0
        for ii in range(n):
            i = int(order[ii])
            a = compute_coeffs(h_cur, i, n, dx)
            b = corr_vals - a * h_cur[i]
            t_new = coord_minimize(a, b)
            if abs(t_new - h_cur[i]) > 1e-14:
                delta = t_new - h_cur[i]
                h_cur[i] = t_new
                corr_vals += a * delta
                n_changes += 1
                sc = float(np.max(corr_vals))
                if sc < best_sc:
                    best_sc = sc
                    best_h = h_cur.copy()

        corr_vals = compute_corr(h_cur) * dx
        sc = float(np.max(corr_vals))
        if sc < best_sc:

            best_h = h_cur.copy()
        
        # Early stopping
        if n_changes < n * 0.01 and sweep > 20:
            if n_changes == 0 and sweep > 50:
                break
    
    return best_h, best_sc

def slsqp_polish(h, n, target, alphas, maxiter, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            break

            cmax = float(np.max(corr))
            shifted = alpha * (corr - cmax)
            shifted = np.clip(shifted, -700.0, 0.0)

            sc = compute_c5(h_new)
            if sc < cur_score:

                cur_score = sc
