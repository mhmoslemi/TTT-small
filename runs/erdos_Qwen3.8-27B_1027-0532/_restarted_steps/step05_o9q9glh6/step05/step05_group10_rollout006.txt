def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def min_max_pwl(A, B):
    f0 = np.max(A)
    active_0 = A >= f0 - 1e-12
    if np.min(B[active_0]) >= -1e-12:
        return 0.0
    f1 = np.max(A + B)
    active_1 = (A + B) >= f1 - 1e-12
    if np.max(B[active_1]) <= 1e-12:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = A + B * mid
        f_mid = np.max(vals)
        active = vals >= f_mid - 1e-10
        min_B = np.min(B[active])
        max_B = np.max(B[active])
        if min_B >= 0:
            lo = mid
        elif max_B <= 0:
            hi = mid
        else:
            return mid
    return 0.5 * (lo + hi)


def coord_descent_hp(h_init, n, target, n_passes, time_limit, start_time, order='seq'):
    dx = 2.0 / n
    h = h_init.astype(float).copy()
    h = project_feasible(h, target)
    half_n = n - 1
    lag = np.arange(-half_n, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    converged = 0
    for pass_num in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        if order == 'seq':
            orderings = [np.arange(n)]
        else:
            orderings = [np.random.permutation(n)]
        for ordering in orderings:
            for i in ordering:
                idx1 = i + lag
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = i - lag
                valid2 = (idx2 >= 0) & (idx2 < n)
                B = np.zeros(2 * n - 1)
                B[valid1] += (1.0 - h[idx1[valid1]]) * dx
                B[valid2] -= h[idx2[valid2]] * dx
                A = corr - h[i] * B
                x_opt = min_max_pwl(A, B)
                delta = x_opt - h[i]
                if abs(delta) > 1e-13:
                    h[i] = x_opt
                    corr += delta * B
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            converged = 0
        else:
            converged += 1
        if converged >= 5:
            break
    return best_h, best_c5


def lp_round_hp(h_init, n, target, time_limit, start_time, damping=0.85, n_active=60):
    dx = 2.0 / n
    h = h_init.astype(float).copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    current_damping = damping
    reg = 1e-5
    for it in range(50000):
        if time.time() - start_time > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - current_damping) * h + current_damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            current_damping = min(0.95, current_damping + 0.02)
        else:
            current_damping = max(0.3, current_damping * 0.95)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())
        if it % 100 == 0 and reg > 1e-7:
            reg = max(1e-7, reg * 0.7)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.astype(float).copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_smooth(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    n_local = n
    target_local = target
    h_cur = h_start.astype(float).copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj_alpha(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_local,
                     'jac': lambda x: np.ones(n_local)}]
            res = minimize(obj_alpha, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_local)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score
