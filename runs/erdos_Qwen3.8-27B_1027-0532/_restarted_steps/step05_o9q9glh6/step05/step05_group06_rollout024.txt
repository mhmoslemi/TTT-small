    Returns the new correlation vector."""

    old_val = h[j]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    corr_new = corr.copy()

    # corr[k] = sum_i h[i]*(1-h[i+k-shift]) for valid i

            grad[j1] += wi * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= wi * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f_obj(x):
        return obj_grad(x, alpha, target, mu)[0]

    def g_obj(x):
        return obj_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f_obj, h, jac=g_obj, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxls': 100})
        return project_feasible(res.x, target)
    except Exception:
        return h


def coord_eval_max(h, j, new_val, corr, sum_h, target, mu):
    n = len(h)
    old_val = h[j]
    delta = new_val - old_val
    if abs(delta) < 1e-16:
        return float(np.max(corr)) + mu * (sum_h - target) ** 2
    max_corr = 0.0
    for ki in range(2 * n - 1):
        s = ki - (n - 1)
        change = 0.0
        if s >= 0:
            if j < n - s:
                change += delta * (1.0 - h[j + s])
            if s <= j:
                change -= delta * h[j - s]
        else:
            m = -s
            if j >= m:
                change += delta * (1.0 - h[j - m])
            if j + m < n:
                change -= delta * h[j + m]
        val_ki = corr[ki] + change
        if val_ki > max_corr:
            max_corr = val_ki
    new_sum = sum_h + delta
    penalty = mu * (new_sum - target) ** 2
    return max_corr + penalty


def coordinate_descent(h, target, n_sweeps, deadline, mu=500.0):
    n = len(h)
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    sum_h = float(np.sum(h_cur))
    cur_score = float(np.max(corr))
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = coord_eval_max(h_cur, j, c, corr, sum_h, target, mu)
            fd = coord_eval_max(h_cur, j, d, corr, sum_h, target, mu)
            for _ in range(22):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = coord_eval_max(h_cur, j, c, corr, sum_h, target, mu)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = coord_eval_max(h_cur, j, d, corr, sum_h, target, mu)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-9:
                h_cur[j] = best_v
                delta = best_v - (h_cur[j] - (best_v - h_cur[j]))
                h_cur[j] = best_v
                new_sum = sum_h + (best_v - (best_v - (h_cur[j] - best_v) + h_cur[j] - best_v))
                old = h_cur[j]
                h_cur[j] = best_v
                delta = best_v - old
                sum_h += delta
                corr = compute_corr(h_cur)
                sc = float(np.max(corr))
                if sc < cur_score - 1e-15:
                    cur_score = sc
                    improved = True
        h_cur = project_feasible(h_cur, target)
        corr = compute_corr(h_cur)
        sum_h = float(np.sum(h_cur))
        cur_score = float(np.max(corr))
        if not improved:
            break
    final_score = coord_score(h_cur)
    return h_cur, final_score


def coord_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def top_lag_optimize(h, target, n_top, n_steps, deadline, mu=200.0):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.003

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            wi = 1.0 / n_top
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += wi * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += wi * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h_cur[j2 - m]
        s = float(np.sum(h_cur))
        grad += 2.0 * mu * (s - target)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.85
            if lr < 1e-9:
                break
    return h_cur, cur_score


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def make_binary_pattern(n, period, duty=0.5):
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h


def make_sinusoid(n, freq, amp):
    x = np.linspace(0.0, 2.0 * np.pi * freq, n)
    return 0.5 + amp * np.sin(x)


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return obj_grad(x, alpha, target, 1e6)[0]

    def g(x):
        return obj_grad(x, alpha, target, 1e6)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 30
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing alpha at base resolution
    alphas_p1 = [10, 50, 200, 1000, 5000, 20000, 100000]
    for a in alphas_p1:
        if time.time() > deadline - 500:
            break
        h_new = lbfgs_optimize(best_h, target, a, 50.0, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Top-lag equalization at base resolution
    if time.time() < deadline - 400:
        h_eq, sc_eq = top_lag_optimize(best_h, target, 5, 400, deadline - 300, mu=300.0)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 50000, 200.0, 600)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Fast coordinate descent at base resolution
    if time.time() < deadline - 350:
        best_h, sc_cd = coordinate_descent(best_h, target, 15, deadline - 300, mu=1000.0)
        if sc_cd < best_score:
            best_score = sc_cd

    # Phase 4: Upsample to 2x and optimize
    n2 = 2 * n
    target2 = n2 / 2.0
    if time.time() < deadline - 350:
        h_fine = upsample(best_h, n2)
        h_fine = project_feasible(h_fine, target2)
        alphas_fine = [50, 500, 5000, 50000]
        for a in alphas_fine:
            if time.time() > deadline - 150:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, 50.0, 400)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 5: Top-lag equalization at current resolution
    if time.time() < deadline - 250:
        h_eq, sc_eq = top_lag_optimize(best_h, target, 5, 300, deadline - 200, mu=300.0)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 100000, 500.0, 600)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Diverse restarts
    if time.time() < deadline - 150:
        n_current = len(best_h)
        target_current = n_current / 2.0
        restart_strategies = []
        for amp in [0.05, 0.1, 0.15]:
            restart_strategies.append(('gauss', amp))
        restart_strategies.append(('binary', 3))
        restart_strategies.append(('binary', 7))
        restart_strategies.append(('sinusoid', 3))

        for strat, param in restart_strategies:
            if time.time() > deadline - 60:
                break
            h_pert = best_h.copy()
            if strat == 'gauss':
                h_pert += rng.randn(n_current) * param
            elif strat == 'binary':
                h_bin = make_binary_pattern(n_current, param)
                h_pert = 0.5 * best_h + 0.3 * h_bin + 0.2 * h_pert
            elif strat == 'sinusoid':
                h_sin = make_sinusoid(n_current, param, 0.3)
                h_pert = 0.6 * best_h + 0.4 * h_sin
            h_pert = project_feasible(h_pert, target_current)
            h_new = lbfgs_optimize(h_pert, target_current, 500, 50.0, 300)
            h_new = lbfgs_optimize(h_new, target_current, 20000, 200.0, 400)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                n_current = len(best_h)
                target_current = n_current / 2.0

    # Phase 7: Aggressive coordinate descent at final resolution
    if time.time() < deadline - 40:
        best_h, sc_cd = coordinate_descent(best_h, target, 20, deadline - 30, mu=2000.0)
        if sc_cd < best_score:
            best_score = sc_cd

    # Phase 8: SLSQP polish
    if time.time() < deadline - 20:
        h_slq = slsqp_polish(best_h, target, 50000, 2000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 9: Final coordinate descent
    if time.time() < deadline - 5:
        best_h, sc_cd = coordinate_descent(best_h, target, 10, deadline - 2, mu=2000.0)
        if sc_cd < best_score:
            best_score = sc_cd

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
