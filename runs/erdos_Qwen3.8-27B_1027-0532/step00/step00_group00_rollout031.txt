                break
        
        off = N - 1
        g = np.zeros(N)
        a = idx + k - off
        m1 = (a >= 0) & (a < N)
        g[m1] += 1.0 - h[a[m1]]
        b = idx - k + off
        m2 = (b >= 0) & (b < N)
        g[m2] -= h[b[m2]]

        h2 = project_box_simplex(h2, s)
        d = np.abs(h2 - h).sum()
        if d > 1e-12:
            h = h2
            S = compute_S(h)
        else:

            k2 = int(np.argsort(S)[::-1][min(1, len(S)-1)])
            a = idx + k2 - off
            m1 = (a >= 0) & (a < N)
            g[m1] = 0.0
            g[m1] += 1.0 - h[a[m1]]
            b = idx - k2 + off
            m2 = (b >= 0) & (b < N)
            g[m2] -= h[b[m2]]
            h2 = h - alpha * g
            h2 = project_box_simplex(h2, s)
            h = h2
            S = compute_S(h)

Hmm, this complementary swap only works when h[i] + h[j] ≈ 1. That's too restrictive. Let me use a more general approach.

Actually, I think the most effective approach for this problem is:

1. Start with the initial solution
2. Use a "hill climbing" approach where at each step:
   - Compute S = correlate(h, 1-h)
   - Find k* = argmax(S)
   - For k*, the "cost" is S[k*] = sum_n h[n](1-h[n+k*-(N-1)])
   - To reduce this, we want to make h[n] and h[n+k*-(N-1)] more "aligned" (both high or both low)
   - This means: if h[n] is high and h[n+k*-(N-1)] is low, try to swap them (or reduce the difference)
   - The swap must preserve the sum constraint

Let me implement this more directly:

For a given k*, the pairs that contribute to S[k*] are (n, n+k*-(N-1)) for valid n. For each such pair where h[n] > h[n+k*-(N-1)], we can try:
- Decrease h[n] and increase h[n+k*-(N-1)] (if the second is not already at 1)
- Or: swap h[n] and h[n+k*-(N-1)] (preserves sum)
- Or: transfer mass from n to n+k*-(N-1)

The "swap" is the cleanest move. Let me implement a local search that:
1. Identifies k*
2. For all pairs in the k* lag, tries swaps
3. Accepts the best swap
4. Repeats

This is O(N) per step (since there are at most N pairs for a given k*), so very fast.

But we also need to consider that changing h might increase other lags. So we should check the full max after each swap.

import numpy as np
import time

def compute_S(h):
    return np.correlate(h, 1.0 - h, mode="full")

def eval_c5(h):
    N = len(h)
    S = compute_S(h)
    return float(np.max(S) * 2.0 / N)

def project_box_simplex(x, s):
    """Project x onto {y: 0<=y<=1, sum(y)=s}"""
    if x.sum() == s:
        return x.copy()
    lo, hi = -x.min() - 1.0, 2.0 - x.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        v = np.clip(x + mid, 0.0, 1.0)
        if v.sum() < s:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def make_binary_sequence(N, rng):
    """Create a random binary sequence with sum = N/2"""
    n_ones = N // 2
    n_zeros = N - n_ones - (N % 2)
    h = np.zeros(N)
    if N % 2 == 0:
        # Exactly N/2 ones
        positions = rng.permutation(N)
        h[positions[:n_ones]] = 1.0
    else:
        # N/2 = (N-1)/2 + 0.5
        # Use (N-1)/2 ones, one 0.5, (N-1)/2 zeros
        positions = rng.permutation(N)
        h[positions[:n_ones]] = 1.0
        h[positions[n_ones]] = 0.5
    return h

def alternating_sequence(N):
    """Create an alternating 1,0,1,0,... sequence"""
    h = np.zeros(N)
    h[0::2] = 1.0
    # Adjust sum
    current_sum = h.sum()
    target = N / 2.0
    diff = target - current_sum
    if diff > 0:
        # Need to increase sum
        zeros = np.where(h == 0)[0]
        for idx in zeros:
            amount = min(diff, 1.0 - h[idx])
            h[idx] += amount
            diff -= amount
            if diff <= 0:
                break
    elif diff < 0:
        ones = np.where(h == 1)[0]
        for idx in ones:
            amount = min(-diff, h[idx])
            h[idx] -= amount
            diff += amount
            if diff >= 0:
                break
    return h
