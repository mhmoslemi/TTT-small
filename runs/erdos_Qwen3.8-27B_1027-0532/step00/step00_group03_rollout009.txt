For the projection onto {h: sum(h) = n/2, 0 <= h <= 1}, I need to solve:

This is a quadratic program. I can solve it using a bisection on the Lagrange multiplier for the sum constraint:
h_i = clip(x_i - λ, 0, 1) for some λ such that sum(h) = n/2.

```python
def project_onto_simplex(x, target_sum):
    """Project x onto {h: sum(h) = target_sum, 0 <= h <= 1}."""
    n = len(x)
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(x - lam, 0, 1))
    
    # Bisection on lambda
    lo = x.min() - 1  # ensures sum > target
    hi = x.max() + 1  # ensures sum < target
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.clip(x - lam, 0, 1)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr * dx))

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_corr_k_vec(h, k):
    n = len(h)
    dx = 2.0 / n
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    elif k > 0:
        grad = np.zeros(n)
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
        return dx * grad
    else:
        m = -k
        grad = np.zeros(n)
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
        return dx * grad

def obj_and_grad(h, alpha, penalty):
    n = len(h)
    corr = corr_values(h)
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    s = exps.sum()
    weights = exps / s
    
    val = m + np.log(s) / alpha
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w > 1e-10:
            lag = idx - (n - 1)
            grad += w * grad_corr_k_vec(h, lag)
    
    if penalty > 0:
        dev = np.sum(h) - n / 2.0
        val += 0.5 * penalty * dev * dev
        grad += penalty * dev
    
    return val, grad

def project_sum(x, target):
    n = len(x)
    def s_at(lam):
        return np.sum(np.clip(x - lam, 0.0, 1.0))
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if s_at(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def optimize_at_resolution(h0, time_budget, start_time):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alphas = [5.0, 20.0, 50.0, 200.0, 1000.0]
    time_per_alpha = time_budget / len(alphas)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget + start_time:
            break
        try:
            result = minimize(
                obj_and_grad, h,
                args=(alpha, 1e4),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 500, 'gtol': 1e-10, 'ftol': 1e-12}
            )
            h = project_sum(result.x, n / 2.0)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass
    
    return best_h, best_c5
```
