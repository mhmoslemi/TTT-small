from scipy.optimize import minimize, Bounds, LinearConstraint

_opt = {}

def _obj_func(x):
    n = _opt['n']
    t = _opt['t']
    dx = _opt['dx']
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    if mc < -1e300:
        return mc
    return float(mc + np.log(np.sum(np.exp(corr - mc))) / t)

def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        h = np.full(len(h), target_sum / len(h))
    else:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual sum error by adjusting entries furthest from boundary
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        _obj_func, h_init, method='SLSQP',
