```python
from scipy.optimize import minimize, linprog
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute C5 for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_grad_c5(h, dx, active_lags=None):
    """Compute gradient of C5 (subgradient) using active lags."""
    n = len(h)
    # Get all correlation values
    corr = np.correlate(h, 1 - h, mode="full")
    if active_lags is None:
        active_lags = np.where(corr == np.max(corr))[0]
    # Average gradient over active lags
    grad = np.zeros(n)
    for p in active_lags:
        l = p - (n - 1)  # lag
        if l >= 0:
            # f_l = sum_{i=0}^{n-1-l} h[i+l] * (1-h[i])
            # d/dh[j]: for j = i+l (i.e., i = j-l), term is (1-h[j-l])
            #           for j = i (i.e., i = j), term is -h[j+l]
            for i in range(n - l):
                grad[i + l] += (1 - h[i])  # derivative w.r.t. h[i+l]
                grad[i] += -h[i + l]       # derivative w.r.t. h[i]
        else:
            l_abs = -l
            # f_{-l_abs} = sum_{i=l_abs}^{n-1} h[i] * (1-h[i-l_abs])
            for i in range(l_abs, n):
                grad[i] += (1 - h[i - l_abs])  # derivative w.r.t. h[i]
                grad[i - l_abs] += -h[i]       # derivative w.r.t. h[i-l_abs]
    grad *= dx
    grad /= len(active_lags)
    return grad
```

```python
def optimize(h, n, time_budget):
    dx = 2.0 / n
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    momentum = np.zeros(n)
    lr = 0.01
    step = 0
    
    while time.time() - start_time < time_budget:
        # Compute active lags (top few)
        corr = np.correlate(h, 1-h, mode="full")
        max_corr = np.max(corr)
        threshold = max_corr - 0.01 * max_corr
        active = np.where(corr >= threshold)[0]
        
        # Compute subgradient
        grad = compute_subgradient(h, dx, active)
        
        # Momentum update
        momentum = 0.9 * momentum + grad
        h_new = h - lr * momentum
        
        # Project onto feasible set
        h_new = project_onto_feasible(h_new, n)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr *= 1.01  # slightly increase
        else:
            lr *= 0.9  # decrease step size
            h = best_h.copy()
        
        step += 1
    
    return best_h, best_c5
```

```python
def project_onto_feasible(h, n):
    target = n / 2.0
    # Find lambda such that sum_i clip(h_i - lambda, 0, 1) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)
```

```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def resample_linear(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    if n_new % n_old == 0:
        return np.repeat(h, n_new // n_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def c5_smooth_and_grad(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    u = t * (c - c_max)
    u_max = np.max(u)
    w = np.exp(u - u_max)
    w_sum = w.sum()
    p = w / w_sum
    c5_s = c_max + (u_max + np.log(w_sum)) / t
    grad = np.zeros(n)
    idx0 = n - 1

        wn = p[(n - 1) - s]

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
```
