    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def smooth_max_obj(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    shifted = (c - cmax) * beta
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    s = e.sum()
    val = cmax + np.log(s) / beta
    w = e / s
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            g += w[k] * grad_corr(x, n, k)
    return val, g

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = initial_h_values.astype(float).copy()

    best_h = h0.copy()
    best_val = c5_raw(h0)

    t0 = time.time()
    end = t0 + budget_s - 10

    h = h0.copy()
    alpha = 0.06
    stall = 0
    restart_count = 0

    while time.time() < end:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = corr.max()

        if mx < best_val - 1e-12:
            best_val = mx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            restart_count += 1
            h = best_h.copy()
            if restart_count % 4 == 1:
                h = project_feasible(h + rng.randn(n) * 0.02, n)
            elif restart_count % 4 == 2:
                h = project_feasible(h + rng.randn(n) * 0.05, n)
            elif restart_count % 4 == 3:
                h = project_feasible(1.0 - h + rng.randn(n) * 0.01, n)
            else:
                h = project_feasible(h + rng.randn(n) * 0.03, n)
            stall = 0
            alpha = 0.06
            continue

        top_n = min(3, len(corr))
        top_idx = np.argsort(corr)[-top_n:]
        g = np.zeros(n)
        for i in top_idx:
            g += grad_corr(h, n, int(i))
        g /= top_n

        step_done = False
        for f in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_new = project_feasible(h - alpha * f * g, n)
            v_new = c5_raw(h_new)
            if v_new < mx - 1e-14:
                h = h_new
                step_done = True
                break
        if not step_done:
            alpha *= 0.6
            if alpha < 0.003:
                alpha = 0.06
                h = project_feasible(best_h + rng.randn(n) * 0.015, n)

    if time.time() < end - 5:
        beta = 50.0
        def penalized(x):
            v, g = smooth_max_obj(x, n, beta)
            d = x.sum() - n / 2.0
            return v + 2000.0 * d * d, g + 4000.0 * d
        try:
            r = minimize(penalized, best_h, jac=True, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 100, 'gtol': 1e-8})
            hc = project_feasible(r.x, n)
            vc = c5_raw(hc)
            if vc < best_val:
                best_val = vc
                best_h = hc
        except Exception:
            pass

    if time.time() < end - 3:
        beta2 = 80.0
        def penalized2(x):
            v, g = smooth_max_obj(x, n, beta2)
            d = x.sum() - n / 2.0
            return v + 5000.0 * d * d, g + 10000.0 * d
        try:
            r2 = minimize(penalized2, best_h, jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 50, 'gtol': 1e-9})
            hc2 = project_feasible(r2.x, n)
            vc2 = c5_raw(hc2)
            if vc2 < best_val:
                best_val = vc2
                best_h = hc2
        except Exception:
            pass

    best_c5 = best_val * 2.0 / n
    return best_h, best_c5, n
