    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = state['t']
    m = state['m'] * beta1 + grads * (1 - beta1)
    v = state['v'] * beta2 + grads ** 2 * (1 - beta2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    newState = {'lr': lr, 't': t + 1, 'm': m, 'v': v}
    return step, newState


def run_smooth_opt(h0, n_steps, T_schedule, lr, budget_end):
    N = len(h0)
    target_sum = N / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    state = {'lr': lr, 't': 1, 'm': np.zeros(N), 'v': np.zeros(N)}
    for step in range(n_steps):
        if time.time() > budget_end:
            break
        T = T_schedule(step)
        grad = smooth_max_grad(h, T)
        adam_step_val, state = adam_step(None, grad, state)
        h = h - adam_step_val
        h = project_box_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def slsqp_optimize(h0, maxiter, budget_end):
    N = len(h0)
    target_sum = N / 2.0

    def objective(x):
        return compute_c5(np.clip(x, 0.0, 1.0))

    bounds = [(0.0, 1.0)] * N
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(N)}]
    try:
        result = minimize(objective, h0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_opt = np.clip(result.x, 0.0, 1.0)
    except Exception:
        h_opt = h0.copy()
    h_opt = project_box_sum(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)


def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_box_sum(h, n / 2.0)


def make_pattern(n, variant):
    if variant == 0:
        h = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    elif variant == 1:
        x = np.arange(n) * 2.0 / n
        h = 0.5 + 0.45 * np.sin(np.pi * x)
    elif variant == 2:
        h = np.zeros(n)
        q = max(1, n // 6)
        h[0:q] = 1.0
        h[2 * q:3 * q] = 1.0
        h[4 * q:min(n, 5 * q)] = 1.0
    elif variant == 3:
        x = np.arange(n) * 2.0 / n
        h = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, n / 2.0)


def interpolate_to(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.arange(old_n) * (2.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_box_sum(h_new, new_n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_initial(80, seed)

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    t_end = start_time + budget_s * 0.12
    h_opt, c5 = slsqp_optimize(h_init, maxiter=400, budget_end=t_end)
    if c5 < best_c5:
        best_h, best_c5 = h_opt, c5

    # Phase 2: Smooth gradient descent
    t_end = start_time + budget_s * 0.25
    def t_schedule_1(step):
        return max(5.0, 50.0 * np.exp(-step / 2000))
    h_sg, c5_sg = run_smooth_opt(best_h, n_steps=50000, T_schedule=t_schedule_1, lr=0.05, budget_end=t_end)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # Phase 3: SLSQP polish again
    t_end = start_time + budget_s * 0.35
    h_opt, c5 = slsqp_optimize(best_h, maxiter=400, budget_end=t_end)
    if c5 < best_c5:
        best_h, best_c5 = h_opt, c5

    # Phase 4: More smooth gradient with lower T
    t_end = start_time + budget_s * 0.45
    def t_schedule_2(step):
        return max(2.0, 10.0 * np.exp(-step / 3000))
    h_sg, c5_sg = run_smooth_opt(best_h, n_steps=50000, T_schedule=t_schedule_2, lr=0.02, budget_end=t_end)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # Phase 5: SLSQP at higher resolution (n=160)
    if time.time() < start_time + budget_s * 0.80:
        h160 = interpolate_to(best_h, 160)
        t_end = start_time + budget_s * 0.55
        h_opt, c5 = slsqp_optimize(h160, maxiter=200, budget_end=t_end)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

        t_end = start_time + budget_s * 0.62
        def t_schedule_3(step):
            return max(3.0, 20.0 * np.exp(-step / 5000))
        h_sg, c5_sg = run_smooth_opt(best_h, n_steps=50000, T_schedule=t_schedule_3, lr=0.03, budget_end=t_end)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 6: SLSQP at n=160 polish
    if time.time() < start_time + budget_s * 0.80:
        if len(best_h) == 80:
            best_h = interpolate_to(best_h, 160)
        t_end = start_time + budget_s * 0.70
        h_opt, c5 = slsqp_optimize(best_h, maxiter=300, budget_end=t_end)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 7: Try n=320
    if time.time() < start_time + budget_s * 0.88:
        h320 = interpolate_to(best_h, 320)
        t_end = start_time + budget_s * 0.78
        h_opt, c5 = slsqp_optimize(h320, maxiter=150, budget_end=t_end)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

        t_end = start_time + budget_s * 0.85
        def t_schedule_4(step):
            return max(2.0, 15.0 * np.exp(-step / 3000))
        h_sg, c5_sg = run_smooth_opt(best_h, n_steps=30000, T_schedule=t_schedule_4, lr=0.02, budget_end=t_end)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 8: Multiple restarts at best resolution
    if time.time() < start_time + budget_s * 0.93:
        n_best = len(best_h)
        for restart in range(8):
            if time.time() > start_time + budget_s * 0.93:
                break
            rng = np.random.RandomState(seed + 7000 + restart)
            h_pert = best_h + rng.randn(n_best) * 0.05
            h_pert = project_box_sum(h_pert, n_best / 2.0)
            t_end = min(time.time() + 20, start_time + budget_s * 0.93)
            h_opt, c5 = slsqp_optimize(h_pert, maxiter=200, budget_end=t_end)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Phase 9: Final polish
    if time.time() < start_time + budget_s * 0.97:
        t_end = start_time + budget_s * 0.97
        h_opt, c5 = slsqp_optimize(best_h, maxiter=500, budget_end=t_end)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        if time.time() < start_time + budget_s * 0.99:
            def t_schedule_5(step):
                return 3.0
            h_sg, c5_sg = run_smooth_opt(best_h, n_steps=20000, T_schedule=t_schedule_5, lr=0.01, budget_end=start_time + budget_s * 0.99)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg

    # Final validation
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
