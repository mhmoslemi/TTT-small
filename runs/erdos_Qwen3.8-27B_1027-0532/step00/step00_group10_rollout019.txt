import numpy as np
import time

def compute_overlap(h):
    """Compute C5 = max_k correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(conv), conv

def project_feasible(h, target_sum):
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Bisection on lambda
    # We want sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def get_active_gradient(h, k_star, n, dx):
    """Compute gradient of g_k* at h"""
    grad = np.zeros(n)
    k = k_star
    for j in range(n):
        val = 0.0
        if 0 <= j + k < n:
            val += (1 - h[j + k])
        if 0 <= j - k < n:
            val -= h[j - k]
        grad[j] = dx * val
    return grad

    # Project to feasible
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5, _ = compute_overlap(h)
    
    # Optimization parameters
    lr = 0.1  # learning rate
    momentum = 0.5
    v = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 5:
        # Compute current objective
        c5, conv = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active shift
        k_star = np.argmax(conv)
        
        # Compute gradient of active constraint
        grad = get_active_gradient(h, k_star, n_points, dx)
        
        # Compute gradients for all top-5 shifts and average (smoothing)
        top_k = np.argsort(conv)[::-1][:5]
        total_grad = np.zeros(n_points)
        for k in top_k:
            total_grad += get_active_gradient(h, k, n_points, dx)
        total_grad /= len(top_k)

        grad = 0.5 * grad + 0.5 * total_grad

        h_new = h + v

        # Check if it improved

            h = h_new
            lr = min(lr * 1.05, 1.0)  # increase LR slightly
        else:

    grad = np.zeros(n)
    if k >= 0:
        nk = n - k
        grad[k:] += (1.0 - h[:nk])
        grad[:nk] -= h[k:]
    else:
        kp = -k
        nk = n - kp
        grad[:nk] += (1.0 - h[kp:])
        grad[kp:] -= h[:nk]

        if s > target_sum:
            lo = mid
        else:
            hi = mid

    grad = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            grad[k:] += (1.0 - h[:nk])
            grad[:nk] -= h[k:]
    else:
        kp = -k
        nk = n - kp
        if nk > 0:
            grad[:nk] += (1.0 - h[kp:])
            grad[kp:] -= h[:nk]

def subgradient_c5(h, n, dx, top_m=5):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(conv)
    sorted_idx = np.argsort(conv)[::-1]

        grad += grad_shift(h, n, k)
    grad /= len(active)
