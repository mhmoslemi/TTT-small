def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def find_1d_optimum(a, b):
    mask = np.abs(a) > 1e-14
    a_rel = a[mask]
    b_rel = b[mask]
    floor = np.max(b[~mask]) if np.any(~mask) else -1e30
    candidates = [0.0, 1.0]
    m = len(a_rel)
    for i in range(m):
        for j in range(i + 1, m):
            da = a_rel[i] - a_rel[j]
            if abs(da) < 1e-14:
                continue
            t = (b_rel[j] - b_rel[i]) / da
            if 0.0 < t < 1.0:
                candidates.append(t)
    best_t = 0.0
    best_val = max(np.max(b_rel), floor)
    for t in candidates:
        val = max(np.max(a_rel * t + b_rel), floor)
        if val < best_val:
            best_val = val
            best_t = t
    val1 = max(np.max(a_rel + b_rel), floor)
    if val1 < best_val:
        best_t = 1.0
    return best_t
