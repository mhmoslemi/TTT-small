def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def local_search_binary(h, n, max_iter=10000):
    """Local search for binary sequences with n/2 ones."""
    best_h = h.copy()
    best_score = compute_c5(best_h, n)
    
    for _ in range(max_iter):

        ones = np.where(best_h > 0.5)[0]

        j = zeros[np.random.randint(len(zeros))]
        if i == j:

        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_score = compute_c5(new_h, n)
        if new_score < best_score:

            best_score = new_score

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def compute_cross_counts(h, n):
    """For binary h, compute cross counts for each shift."""
    # cross(d) = number of i where h[i]=1 and h[i+d]=0 (valid i)
    # This is the same as correlate(h, 1-h, "full") but shifted
    corr = np.correlate(h, 1 - h, mode="full")
    return corr  # indexed by k, shift = k - (n-1)

def binary_local_search(h, n, max_iter=50000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    
    best_h = h.copy()
    best_score = compute_c5(best_h, n)
    curr_h = best_h.copy()
    curr_score = best_score
    
    ones = np.where(curr_h > 0.5)[0].tolist()
    zeros = np.where(curr_h < 0.5)[0].tolist()
    
    for it in range(max_iter):
        # Pick random 1 and 0 to swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        new_h = curr_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_score = compute_c5(new_h, n)
        
        if new_score < curr_score:
            curr_h = new_h
            curr_score = new_score
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if curr_score < best_score:
                best_h = new_h.copy()
                best_score = curr_score
        else:
            # Try another random pair
            pass
    
    return best_h, best_score

This can be computed in O(n) for each d, and O(n²) for all d. The np.correlate already does this efficiently.

For local search, the key question is: can I update the correlation efficiently after a swap? 

If I swap h[i] from 1 to 0 and h[j] from 0 to 1:
- For shift d ≥ 0: cross(d) changes by:
  - Remove contribution from pair (i, i+d) if i+d < n and h[i+d]=0: -1 (if h[i+d]=0)
  - Add contribution from pair (j, j+d) if j+d < n and h[j+d]=0: +1 (if h[j+d]=0)
  - Also, for the case where i and j are close: if d = j-i, the pair (i, j) was contributing (h[i]=1, h[j]=0) but now it's (0,1), so -1.
  - Similarly, if d = i-j (negative), the pair (j, i) was contributing but now isn't.

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), (1 - h).astype(float), mode="full")
    return float(np.max(corr) * dx)

def compute_c5_detailed(h, n):
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), (1 - h).astype(float), mode="full") * dx
    return float(np.max(corr)), corr

def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1
    return h

def swap_local_search(h, n, max_iter=100000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    
    h = h.copy()
    score = compute_c5(h, n)
    best_h = h.copy()
    best_score = score
    
    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h < 0.5)[0].tolist()
    
    for it in range(max_iter):
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h, n)
        
        if new_score < score:
            score = new_score
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_score

def simulated_annealing(h, n, start_temp=0.1, end_temp=0.001, max_iter=100000, rng=None):

        rng = np.random.default_rng()

    score = compute_c5(h, n)
    best_h = h.copy()
    best_score = score

    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h < 0.5)[0].tolist()

        h[j] = 1

            h[j] = 0

- As "target" (second element of pair):
  - For pair (i-d, i): if i-d is valid.
    - If h[i-d]=1: was (1,1)→(1,0). Δ = +1. (Now counted!)
    - If h[i-d]=0: was (0,1)→(0,0). Δ = 0.

From position j (was 0, now 1):
- As "source":
  - For pair (j, j+d): if j+d is valid.
    - If j+d = i: was (0,1)→(1,0). Δ = +1. (Now counted!)
    - If j+d ≠ i and h[j+d]=0: was (0,0)→(1,0). Δ = +1. (Now counted!)
    - If j+d ≠ i and h[j+d]=1: was (0,1)→(1,1). Δ = 0.
- As "target":
  - For pair (j-d, j): if j-d is valid.
    - If h[j-d]=1: was (1,0)→(1,1). Δ = -1. (No longer counted!)
    - If h[j-d]=0: was (0,0)→(0,1). Δ = 0.

So total:
Δcross(d) = 
  [i+d=j ? -1 : (0≤i+d<n and h[i+d]=0 and i+d≠j) ? -1 : 0]
  + [(0≤i-d<n and h[i-d]=1) ? +1 : 0]
  + [j+d=i ? +1 : (0≤j+d<n and h[j+d]=0 and j+d≠i) ? +1 : 0]
  + [(0≤j-d<n and h[j-d]=1) ? -1 : 0]

Note: when i+d=j (d=j-i), the first term gives -1.
When j+d=i (d=i-j), the third term gives +1.
