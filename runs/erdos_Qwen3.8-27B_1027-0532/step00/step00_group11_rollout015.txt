def update_corr_after_swap(corr, h, i, j, N):
    """Update correlation vector after swapping h[i]:1->0 and h[j]:0->1.
    corr[m] corresponds to shift k = m - (N-1)
    """
    # corr[k_idx] = sum_n h[n+k] * (1-h[n]) where k = k_idx - (N-1)
    # Changing h[i] from 1 to 0:
    #   - Term where n=i: h[i+k] * (1-1) = h[i+k] * 0 was the new value, old was h[i+k]*1... 
    #   Wait, let me re-derive.
    
    # c[k] = sum_n h[n+k] * (1-h[n])
    # Old: h[i] = 1, h[j] = 0
    # New: h[i] = 0, h[j] = 1
    
    # Effect of h[i]: 1 -> 0
    #   In the sum, terms involving h[i]:
    #     (a) n = i: term is h[i+k] * (1-h[i]) = h[i+k] * 0 (new) vs h[i+k] * 0 (old, since h[i]=1)
    #         Wait, h[i]=1 means (1-h[i])=0, so the term is already 0! Changing h[i] to 0 makes (1-h[i])=1, so the term becomes h[i+k]*1 = h[i+k].
    #         Change: +h[i+k]
    #     (b) n+k = i, i.e., n = i-k: term is h[i] * (1-h[i-k]) = 1*(1-h[i-k]) (old) vs 0*(1-h[i-k]) = 0 (new)
    #         Change: -(1-h[i-k])
    
    # Effect of h[j]: 0 -> 1
    #   In the sum, terms involving h[j]:
    #     (a) n = j: term is h[j+k] * (1-h[j]) = h[j+k]*1 (old, since h[j]=0) vs h[j+k]*0 = 0 (new)
    #         Change: -h[j+k]
    #     (b) n+k = j, i.e., n = j-k: term is h[j] * (1-h[j-k]) = 0 (old, since h[j]=0) vs 1*(1-h[j-k]) (new)
    #         Change: +(1-h[j-k])
    
    # Total change in c[k]:
    # Δc[k] = h[i+k] - (1-h[i-k]) - h[j+k] + (1-h[j-k])
    
    new_corr = corr.copy()
    for k in range(-(N-1), N):
        k_idx = k + (N-1)
        delta = 0.0
        # h[i+k] contribution
        if 0 <= i+k < N:
            delta += h[i+k]  # old h[i+k] before swap
        # -(1-h[i-k])
        if 0 <= i-k < N:
            delta += -(1-h[i-k])  # old h[i-k] before swap
        # -h[j+k]
        if 0 <= j+k < N:
            delta += -h[j+k]  # old h[j+k] before swap
        # +(1-h[j-k])
        if 0 <= j-k < N:
            delta += (1-h[j-k])  # old h[j-k] before swap
        new_corr[k_idx] += delta
    return new_corr

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(x, alpha, target_sum, dx, n):

    return log_sum + sum_pen
