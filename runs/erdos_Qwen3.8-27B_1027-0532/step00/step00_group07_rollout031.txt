def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def gradient_single_lag(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            grad[:n - d] += (1.0 - h[d:])
            if d > 0:
                grad[d:] -= h[:n - d]
    else:
        dp = -d
        if dp < n:
            grad[dp:] += (1.0 - h[:n - dp])
            if dp > 0:
                grad[:n - dp] -= h[dp:]
    return grad


def subgradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        grad += gradient_single_lag(h, d)
    grad /= len(active)
    return dx * grad


def smooth_obj(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = tau * (corr - mc)
    log_sum = mc + np.log(np.sum(np.exp(shifted))) / tau
    return log_sum * dx


def smooth_grad(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = tau * (corr - mc)
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for i in range(len(corr)):
        if w[i] > 1e-14:
            d = i - (n - 1)
            grad += w[i] * gradient_single_lag(x, d)
    return dx * grad


def eq_con_fun(x, n):
    return np.sum(x) - n / 2.0


def eq_con_jac(x, n):
    return np.ones(n)


def project_onto_constraint(y, n):
    target = n / 2.0
    lo = -3.0
    hi = 4.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_onto_constraint(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    def elapsed():
        return time.time() - start_time

    # Phase 1: Projected subgradient with Barzilai-Borwein step size
    phase1_end = budget_s * 0.45
    lr = 0.005
    prev_h = h.copy()
    prev_grad = subgradient_c5(h)
    stall = 0

    while elapsed() < phase1_end:
        grad = subgradient_c5(h)
        s = h - prev_h
        y = grad - prev_grad
        sy = np.dot(s, y)
        ss = np.dot(s, s)
        if sy > 1e-14 and ss > 1e-14:
            bb = ss / sy
            if 1e-6 < bb < 0.5:
                lr = 0.7 * lr + 0.3 * bb
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.8
        prev_h = h.copy()
        prev_grad = grad.copy()
        h = h_new

    # Phase 2: SLSQP with progressive tau
    for tau_val in [50.0, 200.0, 1000.0, 5000.0]:
        if elapsed() > budget_s * 0.75:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n,)}]
        try:
            res = minimize(
                smooth_obj, best_h, method='SLSQP',
                jac=smooth_grad, args=(tau_val, n),
                bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0}
            )
            h_res = project_onto_constraint(res.x, n)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        except Exception:
            pass

    # Phase 3: Multi-restart with subgradient
    if elapsed() < budget_s * 0.88:
        for restart in range(40):
            if elapsed() > budget_s * 0.88:
                break
            step = rng.choice([0.01, 0.03, 0.05, 0.08, 0.12])
            h_r = best_h + rng.randn(n) * step
            h_r = project_onto_constraint(h_r, n)
            lr_r = 0.003
            for it in range(300):
                if elapsed() > budget_s * 0.88:
                    break
                grad = subgradient_c5(h_r)
                h_r = project_onto_constraint(h_r - lr_r * grad, n)
                lr_r *= 0.995
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                # Quick SLSQP polish on the new best
                if elapsed() < budget_s * 0.92:
                    for tau_val in [500.0, 5000.0]:
                        if elapsed() > budget_s * 0.92:
                            break
                        bounds2 = [(0.0, 1.0)] * n
                        cons2 = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n,)}]
                        try:
                            res2 = minimize(
                                smooth_obj, best_h, method='SLSQP',
                                jac=smooth_grad, args=(tau_val, n),
                                bounds=bounds2, constraints=cons2,
                                options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0}
                            )
                            h_r2 = project_onto_constraint(res2.x, n)
                            c5_r2 = compute_c5(h_r2)
                            if c5_r2 < best_c5:
                                best_c5 = c5_r2
                                best_h = h_r2.copy()
                        except Exception:
                            pass

    # Phase 4: Try alternative resolutions
    if elapsed() < budget_s * 0.95:
        for trial_n in [127, 191, 255, 159]:
            if elapsed() > budget_s * 0.95:
                break
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, trial_n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_onto_constraint(h_interp, trial_n)
            c5_interp = compute_c5(h_interp)
            if c5_interp < best_c5:
                best_c5 = c5_interp
                best_h = h_interp.copy()
                tn = trial_n
                lr = 0.002
                for it in range(400):
                    if elapsed() > budget_s * 0.96:
                        break
                    grad = subgradient_c5(best_h)
                    best_h = project_onto_constraint(best_h - lr * grad, tn)
                    lr *= 0.998
                    c5_new = compute_c5(best_h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                if elapsed() < budget_s * 0.97:
                    bounds3 = [(0.0, 1.0)] * tn
                    cons3 = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (tn,)}]
                    try:
                        res3 = minimize(
                            smooth_obj, best_h, method='SLSQP',
                            jac=smooth_grad, args=(2000.0, tn),
                            bounds=bounds3, constraints=cons3,
                            options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0}
                        )
                        h_r3 = project_onto_constraint(res3.x, tn)
                        c5_r3 = compute_c5(h_r3)
                        if c5_r3 < best_c5:
                            best_c5 = c5_r3
                            best_h = h_r3.copy()
                    except Exception:
                        pass

    # Phase 5: Final high-precision SLSQP polish
    if elapsed() < budget_s * 0.98:
        n_final = len(best_h)
        for tau_val in [10000.0, 50000.0]:
            if elapsed() > budget_s * 0.98:
                break
            bounds4 = [(0.0, 1.0)] * n_final
            cons4 = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n_final,)}]
            try:
                res4 = minimize(
                    smooth_obj, best_h, method='SLSQP',
                    jac=smooth_grad, args=(tau_val, n_final),
                    bounds=bounds4, constraints=cons4,
                    options={'maxiter': 300, 'ftol': 1e-18, 'iprint': 0}
                )
                h_final = project_onto_constraint(res4.x, n_final)
                c5_final = compute_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
            except Exception:
                pass

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
