    t_old = np.linspace(0.0, 2.0, n_old)

    h_new = np.interp(t_new, t_old, h)
    return project_feasible(h_new, n_new)


def multi_beta_slsqp(h0, n_pts, time_budget):
    start = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n_pts

    beta_list = [5, 8, 12, 20, 30, 50, 80, 120, 200, 350, 600]
    for beta in beta_list:
        if time.time() - start > time_budget - 3:
            break
        cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n_pts,), 'jac': sum_eq_jac}]
        res = minimize(lse_objective, best_h, args=(n_pts, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_try = project_feasible(res.x, n_pts)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at native resolution
    alloc1 = min(300.0, 0.25 * budget_s)
    if budget_s - (time.time() - t0) > alloc1 + 50:
        h1, c5_1 = multi_beta_slsqp(best_h, n0, alloc1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Subgradient refinement
    elapsed = time.time() - t0
    alloc2 = min(250.0, 0.18 * budget_s)
    if budget_s - elapsed > alloc2 + 30:
        h2, c5_2 = subgradient_method(best_h, n0, alloc2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Coordinate descent
    elapsed = time.time() - t0
    alloc3 = min(250.0, 0.18 * budget_s)
    if budget_s - elapsed > alloc3 + 20:
        h3, c5_3 = coord_descent(best_h, n0, alloc3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: SLSQP again at native resolution (warm start from phase 3)
    elapsed = time.time() - t0
    alloc4 = min(150.0, 0.10 * budget_s)
    if budget_s - elapsed > alloc4 + 20:
        h4, c5_4 = multi_beta_slsqp(best_h, n0, alloc4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
        # Quick CD polish
        if budget_s - (time.time() - t0) > 30:
            h4b, c5_4b = coord_descent(best_h, n0, min(60.0, budget_s - (time.time() - t0) - 10))
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b

    # Phase 5: Try other resolutions
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 80:
        n_candidates = []
        for delta in [-15, -10, -5, 5, 10, 15, 20]:
            nr = n0 + delta
            if 50 <= nr <= 150 and nr != n0:
                n_candidates.append(nr)
        n_candidates = list(set(n_candidates))
        n_candidates.sort(key=lambda x: abs(x - n0))
        time_per = (remaining - 30) / max(1, len(n_candidates))
        for n_r in n_candidates:
            if time.time() - t0 > budget_s - 30:
                break
            if time_per < 20:
                continue
            h_proj = project_resolution(best_h, n_r)
            h_r, c5_r = multi_beta_slsqp(h_proj, n_r, time_per * 0.55)
            if c5_r < best_c5 and time.time() - t0 < budget_s - 30:
                h_cd, c5_cd = coord_descent(h_r, n_r, min(60.0, time_per * 0.4))
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd
                elif c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r
            elif c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 6: Final coordinate descent
    elapsed = time.time() - t0
    if budget_s - elapsed > 15:
        h_f, c5_f = coord_descent(best_h, len(best_h), budget_s - elapsed - 3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 7: Final SLSQP polish at best resolution
    elapsed = time.time() - t0
    if budget_s - elapsed > 30:
        h_p, c5_p = multi_beta_slsqp(best_h, len(best_h), budget_s - elapsed - 5)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
