        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    else:
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    return grad

def project(h, target):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-15) & (h < 1.0 - 1e-15)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample(h, n):
    if len(h) == n:
        return h.astype(np.float64).copy()
    return np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h)), h)

def anneal_optimize(start_h, target, dx, rng, budget, best_c5_init):
    n = len(start_h)
    h = project(start_h.copy(), target)
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = min(c5, best_c5_init)
    
    T = 0.001
    T_min = 1e-7
    cool = 0.99997
    t_start = time.time()
    stall = 0
    
    while time.time() - t_start < budget:
        r = rng.random()
        if r < 0.6:
            step = rng.uniform(0.003, 0.03)
            n_move = 2
        elif r < 0.9:
            step = rng.uniform(0.03, 0.1)
            n_move = rng.randint(2, 5)
        else:
            step = rng.uniform(0.1, 0.3)
            n_move = rng.randint(2, 8)
        
        idx = rng.choice(n, size=n_move, replace=False)
        deltas = rng.uniform(-step, step, size=n_move)
        deltas -= deltas.mean()
        
        trial = h.copy()
        trial[idx] += deltas
        trial = project(trial, target)
        tc = compute_c5(trial, dx)

            h = trial
            c5 = tc
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
