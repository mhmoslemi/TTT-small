def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    h = np.array(initial_h_values, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Ensure feasibility
    h = project(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    step_size = 0.1
    lr_decay = 0.999
    
    for iteration in range(1000000):
        if time.time() - start_time > budget_s - 30:
            break
        
        # Compute overlaps
        overlaps = compute_overlaps(h, dx)
        current_c5 = overlaps.max()
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient of smooth max
        # Use softmax weighting
        temp = 0.01
        weights = np.exp(overlaps / temp - overlaps.max())
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for s in range(-(n-1), n):
            if weights[s + n - 1] < 1e-10:
                continue
            # Add contribution from this shift
            grad += weights[s + n - 1] * gradient_at_shift(h, s)
        
        # Gradient descent step
        h = h - step_size * grad
        h = project(h, target_sum)
        
        step_size *= lr_decay

    overlaps = np.correlate(h, 1 - h, mode="full") * dx

    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx

    weights = exps / exps.sum()
    
    grad = np.zeros(n)
