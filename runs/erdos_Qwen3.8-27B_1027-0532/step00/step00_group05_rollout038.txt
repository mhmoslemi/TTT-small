from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        count = np.sum(mask)
        step = diff / count
        # Check bounds
        if diff > 0:
            max_step = np.min(1.0 - h[mask])
            step = min(step, max_step)
        else:
            max_step = np.min(h[mask])
            step = max(step, -max_step)
        h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    return h

def p_norm_obj(h, p):
    overlaps = compute_all_overlaps(h)
    return np.sum(overlaps ** p) ** (1.0 / p)

def gradient_p_norm(h, p, eps=1e-5):
    n = len(h)
    overlaps = compute_all_overlaps(h)
    val = np.sum(overlaps ** p) ** (1.0 / p)
    grad = np.zeros(n)
    for i in range(n):
        h2 = h.copy()
        h2[i] += eps
        h2 = project_feasible(h2, n)
        f2 = p_norm_obj(h2, p)
        grad[i] = (f2 - val) / (h2[i] - h[i] + 1e-15)
    return val, grad

def local_search(h, n, seed, time_budget, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    current_temp = 0.01
    cooling = 0.999
    
    while time.time() - t_start < time_budget:
        # Random coordinate perturbation
        i = rng.randint(n)
        delta = rng.uniform(-0.1, 0.1)

        current_temp *= cooling
        if current_temp < 1e-6:

    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break

        step = diff / count

            max_step = np.min(1.0 - h[mask])
            step = min(step, max_step)
        else:
            max_step = np.min(h[mask])
            step = max(step, -max_step)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        count = float(np.sum(mask))
        step = diff / count
        if diff > 0:
            max_step = np.min(1.0 - h[mask])
            step = min(step, max_step)
        else:
            max_step = np.min(h[mask])
            step = max(step, -max_step)
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h)
    S = float(np.sum(overlaps ** p))

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        count = float(np.sum(mask))
        step = diff / count
        if diff > 0:
            max_step = np.min(1.0 - h[mask])
            step = min(step, max_step)
        else:
            max_step = np.min(h[mask])
            step = max(step, -max_step)
        h[mask] += step
    diff = target - np.sum(h)
    if abs(diff) > 1e-16:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def pnorm_gradient(h, p):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h)
    S = float(np.sum(overlaps ** p))
    if S < 1e-30:
        return 0.0, np.zeros(n)
    factor = S ** (1.0 / p - 1.0)
    W = overlaps ** (p - 1.0)
    grad = np.zeros(n)
    for i in range(n):
        total = 0.0
        for j in range(2 * n - 1):
            k = j - (n - 1)
            im = i - k
            ip = i + k
            term = 0.0
            if 0 <= im < n:
                term += 1.0 - h[im]
            if 0 <= ip < n:
                term -= h[ip]
            if term != 0.0:
                total += W[j] * term
        grad[i] = factor * dx * total
    val = S ** (1.0 / p)
    return val, grad

def gradient_phase(h, n, time_limit, rng):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    for p in [4, 8, 16, 32, 64, 128]:
        while time.time() - t_start < time_limit:
            val, grad = pnorm_gradient(h, p)
            improved = False
            for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
                h2 = h - trial_lr * grad
                h2 = project_feasible(h2, n)
                c5 = compute_c5(h2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h2.copy()
                    h = h2
                    lr = trial_lr
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5

def local_search_phase(h, n, time_limit, rng):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = 0.004
    patience = 0
    while time.time() - t_start < time_limit:
        n_pert = 1 if rng.random() < 0.6 else (2 if rng.random() < 0.85 else 3)
        h2 = h.copy()
        for _ in range(n_pert):
            i = rng.randint(n)
            h2[i] += rng.uniform(-0.2, 0.2)
        h2 = project_feasible(h2, n)
        c5 = compute_c5(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()
            patience = 0
            temp = min(temp * 1.02, 0.01)
        elif rng.random() < np.exp(-(c5 - best_c5) / max(temp, 1e-10)):
            h = h2
            patience += 1
        else:
            h = best_h.copy()
            patience += 1
        if patience > 2000:
            h = best_h + rng.normal(0, 0.05, n)
            h = project_feasible(h, n)
            patience = 0
        if patience % 1000 == 0:
            temp *= 0.9
            if temp < 0.001:
                temp = 0.004
    return best_h, best_c5

def fine_gradient_phase(h, n, time_limit, rng):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    p = 64
    lr = 0.005

    return best_h, best_c5

def optimize_single(h, n, time_limit, rng):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    t1 = time_limit * 0.25
    h, c5 = gradient_phase(h, n, t1, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    t2 = time_limit * 0.55
    remaining = t_start + time_limit * 0.55 - time.time()
    if remaining > 1:
        h, c5 = local_search_phase(h, n, remaining, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    t3 = time_limit

    if remaining > 1:

            best_c5 = c5

        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    return best_h, best_c5

def make_structured_init(n, rng, variant=0):
    if variant == 0:
        return np.full(n, 0.5)
    elif variant == 1:
        h = np.zeros(n)
        h[n // 4:3 * n // 4] = 1.0
        return project_feasible(h, n)
    elif variant == 2:
        freq = rng.uniform(0.3, 2.0)
        h = 0.5 + 0.4 * np.cos(2.0 * np.pi * np.arange(n) * freq / n)
        return project_feasible(h, n)
    elif variant == 3:
        h = np.zeros(n)
        step = max(1, n // (2 * n // 3))
        for i in range(0, n, 2 * step):
            h[i:i + step] = 1.0
        return project_feasible(h, n)
    else:
        h = rng.uniform(0, 1, n)
        return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)

    best_h = None

    n_main = 42
    if 'initial_h_values' in globals() and initial_h_values is not None:

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
