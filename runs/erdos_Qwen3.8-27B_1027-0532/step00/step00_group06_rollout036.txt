    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_grad_c5(h, n):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr_raw)
    k = max_idx - (n - 1)
    grad = np.zeros(n)

        jk = j - k
        if 0 <= jk <= n - 1:

        jk = j + k
        if 0 <= jk <= n - 1:

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_grad_c5(h, n):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr_raw)
    max_indices = np.where(corr_raw >= max_val - 1e-10)[0]
    j = np.arange(n)
    grad = np.zeros(n)
    for mi in max_indices:
        k = mi - (n - 1)
        jm = j - k
        jp = j + k
        mask_m = (jm >= 0) & (jm <= n - 1)
        mask_p = (jp >= 0) & (jp <= n - 1)
        grad[mask_m] += 1.0 - h[jm[mask_m]]
        grad[mask_p] -= h[jp[mask_p]]
    return (grad / len(max_indices)) * dx
