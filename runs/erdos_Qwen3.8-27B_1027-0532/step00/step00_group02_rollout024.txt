```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h onto {0≤h≤1, sum(h)=target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_step(h, n, active_lags):
    """Compute subgradient of max_k g_k at h, averaged over active lags"""
    grad = np.zeros(n)
    for k in active_lags:
        L = k - (n - 1)
        if L >= 0:
            if L < n:
                grad[:n-L] += (1.0 - h[L:])
                grad[L:] -= h[:n-L]
        else:
            La = -L
            if La < n:
                grad[La:] += (1.0 - h[:n-La])
                grad[:n-La] -= h[La:]
    grad /= len(active_lags)
    return grad

def optimize_subgradient(h, n, n_iters, tol=1e-6):
    """Subgradient optimization"""
    best = h.copy()
    best_val = compute_c5(best, n)
    target_sum = n / 2.0
    
    for t in range(1, n_iters + 1):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_val - tol:
            best_val = c5
            best = h.copy()
        
        # Find active lags (within epsilon of max)
        thresh = corr.max() - max(1e-6, 0.01 * corr.max())
        active = np.where(corr >= thresh)[0]
        
        # Compute subgradient
        grad = subgradient_step(h, n, active)
        
        # Step size
        alpha = 2.0 / (np.sqrt(t) * np.linalg.norm(grad) + 1e-10)
        # Actually, let's use a fixed step size that works well
        alpha = 0.1 / (np.sqrt(t) + 1)
        
        h_new = h - alpha * grad
        h = project_h(h_new, target_sum)
    
    return best, best_val

def optimize_slsqp(h, n, maxiter=1000):
    """SLSQP optimization"""
    target_sum = n / 2.0
    
    def objective(x):
        return compute_c5(x, n)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n
    
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-8})
    if res.success:
        h_opt = np.clip(res.x, 0, 1)
        # Re-project to satisfy constraint
        h_opt = project_h(h_opt, target_sum)
        return h_opt, compute_c5(h_opt, n)
    return h, compute_c5(h, n)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex_box(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} via bisection"""

        else:
            lo = mid

                        best_c5 = c5_candidate
                        best_h = h_candidate.copy()

                    h[i] -= delta

                return compute_c5(x)

            h_candidate = project_to_simplex_box(res.x, target_sum)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
        except:
            pass
    
    return (best_h, best_c5, n)
```
