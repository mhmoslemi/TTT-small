def gradient_step(h, n, step_size=0.01):
    """One gradient step on the active constraint."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    m = k_star - (n - 1)
    
    # Compute R_k (valid range for this lag)
    i_min = max(0, n - 1 - k_star)
    i_max = min(n - 1, 2*n - 2 - k_star)
    R_k = set(range(i_min, i_max + 1))
    
    # Gradient of f_k w.r.t. h[j]
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if j in R_k:
            g += 1.0 - h[j + m] if (j + m) < n else 1.0
            # Actually, if j ∈ R_k, then j+m is valid by definition
            g += 1.0 - h[j + m]
        if (j - m) in R_k:
            g -= h[j - m]
        grad[j] = g
    
    # Projected gradient step
    h_new = h - step_size * grad
    # Enforce sum constraint
    h_new = h_new - (np.sum(h_new) - n/2) / n
    # Enforce bounds
    h_new = np.clip(h_new, 0, 1)
    # Fix sum again
    h_new = h_new - (np.sum(h_new) - n/2) / n
    h_new = np.clip(h_new, 0, 1)
    
    return h_new, k_star

**Approach: Subgradient method with projection**

1. At each iteration, compute C₅ and identify the active lag k*
2. Compute the subgradient of f_{k*} w.r.t. h
3. Take a step in the negative subgradient direction
4. Project onto the feasible set (0 ≤ h ≤ 1, sum h = n/2)
5. Decrease step size

The projection onto {0 ≤ h ≤ 1, sum h = n/2} can be done efficiently:
- Clip to [0,1]
- If sum > n/2, reduce the largest values
- If sum < n/2, increase the smallest values
(This is similar to the "water-filling" or "Dinkelbach" approach.)

Actually, the exact projection onto {h : 0 ≤ h ≤ 1, sum h = s} is:
- Find θ such that sum_i clip(h_i - θ, 0, 1) = s
- Then the projection is clip(h_i - θ, 0, 1) for each i

This can be solved by bisection on θ.

Let me code this up:

```python
def project_feasible(v, n):
    """Project v onto {h : 0 <= h <= 1, sum h = n/2}."""
    target = n / 2.0
    # Find theta such that sum clip(v_i - theta, 0, 1) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.clip(v - theta, 0, 1)

And the subgradient of f_k = sum_{i ∈ R_k} h[i] * (1 - h[i+m]) w.r.t. h[j]:

f_k = sum_{i ∈ R_k} h[i] - sum_{i ∈ R_k} h[i] * h[i+m]

∂f_k/∂h[j] = 1_{j ∈ R_k} - 1_{j ∈ R_k} * h[j+m] - 1_{j-m ∈ R_k} * h[j-m]

Let me verify: if j ∈ R_k, then the term h[j] appears in the first sum (as h[i] with i=j) and in the second sum (as h[i] with i=j, giving h[j]*h[j+m]). So the derivative contribution is 1 - h[j+m].

Also, if j-m ∈ R_k, then h[j] appears in the second sum as h[i+m] with i=j-m, giving h[j-m]*h[j]. The derivative contribution is -h[j-m].

So: ∂f_k/∂h[j] = 1_{j ∈ R_k} * (1 - h[j+m]) - 1_{j-m ∈ R_k} * h[j-m]

where m = k - (n-1).

Let me code the full subgradient approach:

```python
def subgradient_step(h, n, step_size):
    """One subgradient step."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    m = k_star - (n - 1)
    
    # Valid range for this lag
    i_min = max(0, n - 1 - k_star)
    i_max = min(n - 1, 2*n - 2 - k_star)
    
    # Compute subgradient
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if i_min <= j <= i_max:
            # j is in R_k
            jm = j + m
            if 0 <= jm < n:
                g += 1.0 - h[jm]
        jm = j - m
        if 0 <= jm < n and i_min <= jm <= i_max:
            g -= h[jm]
        grad[j] = g
    
    # Step
    v = h - step_size * grad
    h_new = project_feasible(v, n)
    return h_new, k_star

```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.clip(v - theta, 0.0, 1.0)

def make_valid(h, n):
    return project_feasible(h, n)

def compute_subgradient(h, n, k_star):
    m = k_star - (n - 1)
    i_min = max(0, n - 1 - k_star)
    i_max = min(n - 1, 2 * n - 2 - k_star)
    grad = np.zeros(n)
    R_k_indices = np.arange(i_min, i_max + 1)
    # For j in R_k: grad[j] += 1 - h[j+m]
    for j in R_k_indices:
        jm = j + m
        if 0 <= jm < n:
            grad[j] += 1.0 - h[jm]
    # For j such that j-m in R_k: grad[j] -= h[j-m]
    for j in R_k_indices:
        jm = j - m
        if 0 <= jm < n:
            grad[jm] -= h[jm]
    return grad

def subgradient_method(h, n, n_iters, step_init=0.1, step_decay=0.99):
    h = h.copy()
    best_h = h.copy()
    best_val = eval_c5(h, n)
    step = step_init
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        grad = compute_subgradient(h, n, k_star)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        v = h - step * grad / grad_norm  # normalized step
        h_new = project_feasible(v, n)
        val = eval_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        step *= step_decay
    
    return best_h, best_val
