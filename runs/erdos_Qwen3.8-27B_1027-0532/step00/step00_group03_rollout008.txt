```python
        else:
            hi = mid

        mask = (result > 0) & (result < 1)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
```

Actually, let me reconsider the whole approach. The issue is that SLSQP with a non-smooth objective (max of quadratics) might not work well. Let me use a smoothing trick:

f_smooth(h) = (1/τ) * log(Σ_k exp(τ * c_k(h)))

This is a smooth approximation to max_k c_k(h). For large τ, it's close to the max. For small τ, it's smoother and easier to optimize.

I can use this with gradient-based methods.

The gradient of f_smooth:
∂f_smooth/∂h[m] = (1/τ) * Σ_k exp(τ*c_k) * ∂c_k/∂h[m] / Σ_k exp(τ*c_k)
= Σ_k p_k * ∂c_k/∂h[m]

where p_k = exp(τ*c_k) / Σ_j exp(τ*c_j) (softmax weights)

This is a weighted average of the individual gradients, which is much better than the subgradient.

```python
def smooth_objective(h, tau):
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax smoothing
    shifted = overlaps - np.max(overlaps)
    exp_vals = np.exp(tau * shifted)
    log_sum = np.log(np.sum(exp_vals))
    return np.max(overlaps) + log_sum / tau

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = overlaps - np.max(overlaps)
    weights = np.exp(tau * shifted)
    weights /= np.sum(weights)
    
    # Weighted gradient
    grad = np.zeros(n)
    m = np.arange(n)
    for k_idx in range(len(overlaps)):
        d = k_idx - (n - 1)
        w = weights[k_idx]
        if w < 1e-10:
            continue
        idx1 = (m - d >= 0) & (m - d < n)
        grad[idx1] += w * (1.0 - h[m[idx1] - d]) * dx
        idx2 = (m + d >= 0) & (m + d < n)
        grad[idx2] += w * (-h[m[idx2] + d]) * dx
```

```python
def smooth_gradient_fast(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    weights = np.exp(tau * shifted)
    weights /= np.sum(weights)

    # (1-h[m-d]) for valid m, and -h[m+d] for valid m
    # This is essentially a weighted sum of shifted versions of h

    # (1-h[shifted]) part: for lag d, add weights for d to (1-h) shifted by d
    # -h[shifted] part: for lag d, add weights for d to (-h) shifted by -d
```
