    return np.correlate(h, 1-h, mode='full')

def local_search(h, n, dx, max_passes=50, time_limit=30):
    """Swap-based local search for binary/nearly-binary h."""
    h = h.copy()
    start_time = time.time()

                if i == j:

                h[i] = 1.0
                h[j] = 0.0

def local_search_binary(h, n, time_limit=30.0, rng=None):
    """Swap-based local search with restart."""
    if rng is None:
        rng = np.random.default_rng()
    h = h.astype(float).copy()
    start_time = time.time()
    
    no_improve_count = 0
    max_no_improve = 50
    
    while time.time() - start_time < time_limit and no_improve_count < max_no_improve:
        current_max = max_overlap(h)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        # Try random order of swaps
        idx_i = rng.permutation(len(zeros))
        idx_j = rng.permutation(len(ones))
        
        for ii in idx_i:
            if time.time() - start_time > time_limit:
                break
            i = zeros[ii]
            for jj in idx_j:
                j = ones[jj]
                h[i], h[j] = 1.0, 0.0
                new_max = max_overlap(h)
                if new_max < current_max - 1e-10:
                    current_max = new_max
                    improved = True
                    break
                h[i], h[j] = 0.0, 1.0
            if improved:
                break
        
        if improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Try a random perturbation to escape local optimum
            if no_improve_count % 10 == 0 and len(ones) > 0 and len(zeros) > 0:
                i = zeros[rng.integers(len(zeros))]
                j = ones[rng.integers(len(ones))]
                h[i], h[j] = 1.0, 0.0
    
    return h

def simulated_annealing_binary(h, n, time_limit=30.0, rng=None):
    """Simulated annealing for binary sequences."""
    if rng is None:
        rng = np.random.default_rng()
    h = h.astype(float).copy()
    start_time = time.time()
    
    T = 5.0  # Initial temperature
    T_min = 0.1
    cooling = 0.995
    
    current_max = max_overlap(h)
    best_h = h.copy()
    best_max = current_max
    
    while time.time() - start_time < time_limit and T > T_min:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        
        h[i], h[j] = 1.0, 0.0
        new_max = max_overlap(h)
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        
        T *= cooling
    
    return best_h

from scipy.optimize import minimize
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def max_overlap(h):
    return np.max(np.correlate(h, 1-h, mode='full'))

def local_search_binary(h, n, time_limit=30.0, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h = h.astype(float).copy()
    start_time = time.time()
    current_max = max_overlap(h)
    
    while time.time() - start_time < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        order_i = rng.permutation(len(zeros))
        for ii in order_i:
            if time.time() - start_time > time_limit:
                break
            i = zeros[ii]
            order_j = rng.permutation(len(ones))
            for jj in order_j:
                j = ones[jj]
                h[i] = 1.0
                h[j] = 0.0
                new_max = max_overlap(h)
                if new_max < current_max - 1e-10:
                    current_max = new_max
                    improved = True
                    break
                h[i] = 0.0
                h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    
    return h

def sa_binary(h, n, time_limit=30.0, rng=None):

        rng = np.random.default_rng(0)
    h = h.astype(float).copy()
    start_time = time.time()
    
    T = 10.0
    T_min = 0.01
    cooling = 0.999
    
    current_max = max_overlap(h)
    best_h = h.copy()
    best_max = current_max
    
    while time.time() - start_time < time_limit and T > T_min:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        h[i], h[j] = 1.0, 0.0
        new_max = max_overlap(h)
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        T *= cooling
    
    return best_h

def continuous_optimize(h, n, time_limit=20.0):
    start_time = time.time()
    dx = 2.0 / n
    
    def obj(x):
        if time.time() - start_time > time_limit:
            return 9999
        return max_overlap(x)
    
    def obj_grad(x):
        corr = np.correlate(x, 1-x, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (n-1)
        grad = np.zeros(n)
        idx = np.arange(n)
        valid1 = (idx + s >= 0) & (idx + s < n)
        grad[valid1] += (1 - x[idx[valid1] + s])
        valid2 = (idx - s >= 0) & (idx - s < n)
        grad[valid2] -= x[idx[valid2] - s]
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-10})
        if result.fun < max_overlap(h):
            return result.x
    except:
        pass
    return h

    h = np.zeros(n)
    choices = rng.choice(n, size=n//2, replace=False)
    h[choices] = 1.0

        best_h += diff / n_final
        best_h = np.clip(best_h, 0, 1)
    
    final_c5 = compute_c5_from_h(best_h)
    return best_h, final_c5, n_final
