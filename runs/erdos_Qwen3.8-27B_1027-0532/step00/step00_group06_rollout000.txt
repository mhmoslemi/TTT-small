import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, N):
    """Project h onto {0 <= h <= 1, sum(h) = N/2}"""
    target = N / 2.0
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        tau = (lo + hi) / 2
        h_proj = np.clip(h - tau, 0, 1)
        s = np.sum(h_proj)
        if s > target:
            lo = tau
        else:
            hi = tau
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune to get exact sum
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-boundary elements
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.sum(mask) > 0:
            adj = diff / np.sum(mask)
            h_proj[mask] += adj
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def gradient_c5(h, N):
    """Compute gradient of the active constraint of C5 with respect to h.
    Returns (gradient, active_lag, c5_value)"""
    dx = 2.0 / N
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    # Active lag is argmax of corr
    idx = np.argmax(corr)
    L = idx - (N - 1)  # Convert to actual lag
    
    # Compute gradient of c[L] w.r.t. h
    # c[L] = sum_{i=lo}^{hi} h[i] * (1 - h[i+L])
    # where lo = max(0, -L), hi = min(N-1, N-1-L)
    lo = max(0, -L)
    hi = min(N-1, N-1-L)
    
    grad = np.zeros(N)
    # d/dh[j] of sum h[i]*(1-h[i+L])
    # = [lo<=j<=hi] - h[j+L]*[lo<=j<=hi, valid j+L] - h[j-L]*[lo<=j-L<=hi, valid j]
    
    # First term: 1 for j in [lo, hi]
    grad[lo:hi+1] += 1.0
    
    # Second term: -h[j+L] for j in [lo, hi] where j+L is valid
    if L >= 0:
        # j+L ranges from lo+L to hi+L, all valid since hi <= N-1-L
        grad[lo:hi+1] -= h[lo+L:hi+1+L]
    else:
        # j+L ranges from lo+L to hi+L
        # lo = 0, hi = N-1+L (since L<0, hi = min(N-1, N-1-L) = N-1-L since -L > 0)
        # Wait, if L < 0: lo = max(0, -L) = -L, hi = min(N-1, N-1-L) = N-1 (since N-1-L > N-1)
        # So j ranges from -L to N-1, and j+L ranges from 0 to N-1+L
        grad[lo:hi+1] -= h[lo+L:hi+1+L]
    
    # Third term: -h[j-L] for j where j-L is in [lo, hi] and j is valid
    # j = i+L for i in [lo, hi], so j ranges from lo+L to hi+L
    # Need j in [0, N-1]
    j_lo = max(0, lo + L)
    j_hi = min(N-1, hi + L)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] -= h[j_lo - L:j_hi - L + 1]
    
    return grad, L, np.max(corr) * dx

def optimize_gradient(h, N, budget_s, start_time):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.01
    patience = 200
    no_improve = 0

def random_perturb(h, N, strength=0.1):
    """Apply a random perturbation and project back."""
    rng = np.random.default_rng()
    delta = rng.standard_normal(N) * strength
    h_new = h + delta
    h_new = project_feasible(h_new, N)
    return h_new

def optimize_sa(h, N, budget_s, start_time):
    """Simulated annealing style optimization."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(42)
    T = 0.01
    current_c5 = best_c5
    
    while time.time() - start_time < budget_s:
        # Random perturbation
        h_new = random_perturb(h, N, T)
        c5_new = compute_c5(h_new)
        
        # Acceptance criterion
        if c5_new < current_c5 or rng.random() < np.exp(-(c5_new - current_c5) / (T + 1e-10)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cool down
        T *= 0.999
        
        if T < 1e-6:
            T = 0.001
            # Reheat occasionally
            if rng.random() < 0.1:
                T = 0.01
    
    return best_h, best_c5

def interpolate_to(h_old, N_new):
    """Interpolate h from old resolution to new resolution, preserving integral."""
    N_old = len(h_old)
    # Use linear interpolation
    x_old = np.linspace(0, 2, N_old, endpoint=False) + 1.0/N_old  # cell centers
    x_new = np.linspace(0, 2, N_new, endpoint=False) + 1.0/N_new
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)
    # Fix the sum
    h_interp = project_feasible(h_interp, N_new)
    return h_interp

import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h_p = np.clip(h - tau, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-14:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
    return h_p


def compute_grad_topk(h, N, top_k=5, temperature=0.02):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[::-1][:top_k]
    top_vals = corr[top_indices]
    exp_vals = np.exp((top_vals - top_vals[0]) / (N * temperature))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(N)
    for w, idx in zip(weights, top_indices):
        L = idx - (N - 1)
        lo = max(0, -L)
        hi = min(N - 1, N - 1 - L)
        g = np.zeros(N)
        g[lo:hi + 1] += (1.0 - h[lo + L:hi + 1 + L])
        j_lo = max(0, lo + L)
        j_hi = min(N - 1, hi + L)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] -= h[j_lo - L:j_hi - L + 1]
        grad += w * g
    return np.max(corr) * dx, grad


def optimize_gradient(h, N, time_limit, seed=42):
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.08
    vel = np.zeros(N)
    momentum = 0.85
    stall = 0
    restart_count = 0

    while time.time() < time_limit:
        c5, grad = compute_grad_topk(h, N, top_k=5, temperature=0.02)
        vel = momentum * vel - lr * grad
        h_new = project_feasible(h + vel, N)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.03, 0.3)
        elif c5_new < c5 + lr * 0.1:
            h = h_new
            stall += 1
        else:
            stall += 2
            vel *= 0.5
            h = h_new
            if stall > 300:
                h = best_h + rng.standard_normal(N) * 0.08
                h = project_feasible(h, N)
                lr = 0.08
                vel = np.zeros(N)
                stall = 0
                restart_count += 1
                if restart_count > 30:
                    break

    return best_h, best_c5


def to_binary(h, N):
    target = N // 2
    threshold = np.partition(h, N - target)[N - target]
    h_bin = (h >= threshold).astype(np.float64)
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(n_ones - target):
            h_bin[one_idx[order[i]]] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[zero_idx])[::-1]
        for i in range(target - n_ones):
            h_bin[zero_idx[order[i]]] = 1
    return h_bin


def optimize_binary_swap(h_bin, N, time_limit, seed=42):
    rng = np.random.default_rng(seed)
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    h = h_bin.copy()
    T = 0.005
    stall = 0
    best_T = T
    best_best_c5 = best_c5

    while time.time() < time_limit:
        one_idx = np.where(h == 1.0)[0]
        zero_idx = np.where(h == 0.0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = one_idx[rng.integers(len(one_idx))]
        j = zero_idx[rng.integers(len(zero_idx))]

        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            if best_c5 < best_best_c5:
                best_best_c5 = best_c5
                best_T = T
        elif delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1

        T *= 0.9997
        if T < 1e-7:
            if stall > 3000:
                T = best_T
                stall = 0
            else:
                T = 1e-7

    return best_h, best_c5


def interpolate_to_resolution(h_old, N_new):
    N_old = len(h_old)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h_old, h_old, h_old])
    h_interp = np.interp(x_new, x_ext, h_ext)
    h_interp = project_feasible(h_interp, N_new)
    return h_interp


def random_binary_init(N, seed=0):
    rng = np.random.default_rng(seed)
    idx = rng.permutation(N)
    h = np.zeros(N)
    h[idx[:N // 2]] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.full(70, 0.5)

    N0 = len(h0)
    h0 = project_feasible(h0.copy(), N0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    t_phase1 = start_time + budget_s * 0.28
    t_phase2 = start_time + budget_s * 0.50
    t_phase3 = start_time + budget_s * 0.72
    t_phase4 = start_time + budget_s * 0.88
    t_phase5 = start_time + budget_s * 0.99

    # Phase 1: Gradient at N0
    h1, c5_1 = optimize_gradient(best_h.copy(), N0, t_phase1, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary search at N0
    if time.time() < t_phase2:
        h_bin = to_binary(best_h, N0)
        c5_bin_init = compute_c5(h_bin)
        if c5_bin_init < best_c5:
            best_c5 = c5_bin_init
            best_h = h_bin.copy()
        h2, c5_2 = optimize_binary_swap(h_bin, N0, t_phase2, seed=seed + 1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

        # Also try random binary init
        if time.time() < t_phase2 - 5:
            for trial in range(3):
                if time.time() >= t_phase2 - 5:
                    break
                h_rand = random_binary_init(N0, seed=seed + 100 + trial)
                h2b, c5_2b = optimize_binary_swap(h_rand, N0, min(t_phase2 - 5, time.time() + 30), seed=seed + 200 + trial)
                if c5_2b < best_c5:
                    best_c5 = c5_2b
                    best_h = h2b.copy()

    # Phase 3: Interpolate to 2*N0, gradient
    if time.time() < t_phase3:
        N2 = N0 * 2
        h_interp = interpolate_to_resolution(best_h, N2)
        h3, c5_3 = optimize_gradient(h_interp, N2, t_phase3, seed=seed + 3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Binary search at current best resolution
    if time.time() < t_phase4:
        N_best = len(best_h)
        h_bin2 = to_binary(best_h, N_best)
        c5_bin2_init = compute_c5(h_bin2)
        if c5_bin2_init < best_c5:
            best_c5 = c5_bin2_init
            best_h = h_bin2.copy()
        h4, c5_4 = optimize_binary_swap(h_bin2, N_best, t_phase4, seed=seed + 4)

            best_h = h5.copy()

    # Final projection
    N_out = len(best_h)
    best_h = project_feasible(best_h, N_out)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, N_out)
