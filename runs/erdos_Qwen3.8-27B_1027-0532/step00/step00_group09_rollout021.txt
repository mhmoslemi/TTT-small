def correlate_gradient(h, m):
    """Compute gradient of np.correlate(h, 1-h, mode='full')[m] w.r.t. h"""
    n = len(h)
    k = m - (n - 1)  # shift
    grad = np.zeros(n)
    
    # Contribution from h[i+k] term (j = i+k, so i = j-k)
    # Valid when 0 <= j < n and 0 <= j-k < n
    for j in range(n):
        i = j - k
        if 0 <= i < n:
            grad[j] += (1.0 - h[i])
    
    # Contribution from (1-h[i]) term (j = i, so we need 0 <= j+k < n)
    for j in range(n):
        if 0 <= j < n and 0 <= j + k < n:
            grad[j] += (-h[j + k])
    
    return grad

    return float(dx * np.max(c))

                if h_trial.sum() != target_sum:
                    # Find coordinate to adjust
                    diff = target_sum - h_trial.sum()
                    # Add to a coordinate that has room
                    for k in range(n):
                        if k != j:
                            if h_trial[k] + diff <= 1.0 + 1e-9 and h_trial[k] + diff >= -1e-9:
                                h_trial[k] += diff
                                break
                    if h_trial[j] < -1e-9 or h_trial[j] > 1.0 + 1e-9:
                        continue
                h_trial = np.clip(h_trial, 0.0, 1.0)
                # Re-fix sum
                remaining = target_sum - h_trial.sum()
                if abs(remaining) > 1e-9:
                    continue
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_current - 1e-12:
                    h = h_trial
                    c5_current = c5_trial
                    improved = True
        if not improved:
            break
    return h, c5_current
