  - If j >= L: -(1-h[j-L]) [the term h[j]*(1-h[j-L]) goes from (1-h[j-L]) to 0]
  - If j+L <= n-1: +h[j+L] [the term h[j+L]*(1-h[j]) goes from 0 to h[j+L]]
  - Total: -I(j>=L)*(1-h[j-L]) + I(j+L<=n-1)*h[j+L]

So the total change to c[p] from the swap (i: 0→1, j: 1→0) is:
Δc[p] = I(i>=L_p)*(1-h[i-L_p]) - I(i+L_p<=n-1)*h[i+L_p] 
       - I(j>=L_p)*(1-h[j-L_p]) + I(j+L_p<=n-1)*h[j+L_p]

where L_p = |p - (n-1)|.

This is O(1) per p, so O(n) total for all lags. 

For the local search, we want to find a swap (i,j) that minimizes the max overlap. A greedy approach:
1. Compute current max overlap and the lag(s) that achieve it
2. Try all possible swaps and find the one that most reduces the max overlap
3. This is O(n^3) per step (n^2 swaps × n lags), which might be too slow

A faster approach: only consider swaps that affect the critical lags (the ones achieving the max). If there are only a few critical lags, this is much faster.

Or: use random swaps with simulated annealing. At each step, pick a random (i,j) swap and accept if it improves or with some probability (SA).

            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return None


def binary_sa_step(overlaps, h, i, j, n):
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    L_arr = np.abs(lags)
    delta = np.zeros(n_lags)
    mask_i_ge = L_arr <= i
    mask_i_plus = L_arr <= (n - 1 - i)
    valid_i_ge = mask_i_ge & (L_arr < n)
    valid_i_plus = mask_i_plus & (L_arr < n)
    if np.any(valid_i_ge):
        delta[valid_i_ge] -= (1.0 - h[i - L_arr[valid_i_ge]])
    if np.any(valid_i_plus):
        delta[valid_i_plus] += h[i + L_arr[valid_i_plus]]
    mask_j_ge = L_arr <= j
    mask_j_plus = L_arr <= (n - 1 - j)
    valid_j_ge = mask_j_ge & (L_arr < n)
    valid_j_plus = mask_j_plus & (L_arr < n)
    if np.any(valid_j_ge):
        delta[valid_j_ge] += (1.0 - h[j - L_arr[valid_j_ge]])
    if np.any(valid_j_plus):
        delta[valid_j_plus] -= h[j + L_arr[valid_j_plus]]
    return delta


def binary_sa(h_start, n, max_iter, T_init, T_final, seed):
    rng = np.random.RandomState(seed)
    h = h_start.copy()
    if n % 2 == 0:
        n_ones = n // 2
        idx = np.argsort(h)[::-1][:n_ones]
        h = np.zeros(n)
        h[idx] = 1.0
    overlaps = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(overlaps))
    best_h = h.copy()
    best_max = current_max
    ones_pos = np.where(h == 1.0)[0]
    zeros_pos = np.where(h == 0.0)[0]
    for it in range(max_iter):
        frac = it / max(max_iter - 1, 1)
        T = T_init * (T_final / T_init) ** frac
        i = int(ones_pos[rng.randint(len(ones_pos))])
        j = int(zeros_pos[rng.randint(len(zeros_pos))])
        delta = binary_sa_step(overlaps, h, i, j, n)
        new_overlaps = overlaps + delta
        new_max = float(np.max(new_overlaps))
        accept = False
        if new_max <= current_max:
            accept = True
        else:
            diff = (new_max - current_max) / max(T, 1e-10)
            if diff < 500:
                if rng.rand() < math.exp(-diff):
                    accept = True
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            ones_pos = np.where(h == 1.0)[0]
            zeros_pos = np.where(h == 0.0)[0]
            overlaps = new_overlaps
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
    return best_h, best_max


def make_binary_from_continuous(h, n):
    if n % 2 != 0:
        return None
    n_ones = n // 2
    h_binary = np.zeros(n)
    idx = np.argsort(h)[::-1][:n_ones]
    h_binary[idx] = 1.0
    return h_binary


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is not None and len(h_init) > 0:
        n = len(h_init)
    else:
        n = 98
        h_init = np.full(n, 0.5)

    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = project_to_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP from initial with various betas
    for beta in [50, 100, 200, 500]:
        if time_left() < 30:
            break
        result = run_slsqp(h_init, n, beta, maxiter=400)
        if result is not None:
            result = project_to_feasible(result, n)
            c5 = compute_c5(result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.copy()
                h_init = result.copy()

    # Phase 2: Binary SA from multiple starts
    if n % 2 == 0 and time_left() > 90:
        h_bin = make_binary_from_continuous(best_h, n)
        if h_bin is not None:
            h_bin_best, bin_best_max = binary_sa(h_bin, n, 150000, 5.0, 0.05, seed)
            bin_c5 = bin_best_max * (2.0 / n)
            if bin_c5 < best_c5:
                best_c5 = bin_c5
                best_h = h_bin_best.copy()

        for restart in range(5):
            if time_left() < 30:
                break
            h_rand = rng.rand(n)
            h_rand_bin = make_binary_from_continuous(h_rand, n)
            if h_rand_bin is None:
                continue
            h_sa, sa_max = binary_sa(h_rand_bin, n, 50000, 3.0, 0.1, seed + restart + 100)
            sa_c5 = sa_max * (2.0 / n)
            if sa_c5 < best_c5:
                best_c5 = sa_c5
                best_h = h_sa.copy()

    # Phase 3: SLSQP refinement from best
    for beta in [200, 500, 1000]:
        if time_left() < 30:
            break
        result = run_slsqp(best_h, n, beta, maxiter=500)
        if result is not None:
            result = project_to_feasible(result, n)
            c5 = compute_c5(result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.copy()

    # Phase 4: Perturbation + SLSQP
    for p in range(10):
        if time_left() < 30:
            break
        sigma = 0.1 * (1.0 - p / 10.0) + 0.01
        h_pert = best_h + sigma * rng.randn(n)
        h_pert = project_to_feasible(h_pert, n)
        result = run_slsqp(h_pert, n, 500, maxiter=300)
        if result is not None:
            result = project_to_feasible(result, n)
            c5 = compute_c5(result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.copy()

    # Phase 5: Try finer grid
    for n_try in [2 * n, max(200, n + 50)]:
        if time_left() < 60:
            break
        if n_try > 400 or n_try == n:
            continue
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_feasible(h_interp, n_try)
        for beta in [200, 500]:
            if time_left() < 30:
                break
            result = run_slsqp(h_interp, n_try, beta, maxiter=300)
            if result is not None:
                result = project_to_feasible(result, n_try)
                c5 = compute_c5(result)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = result.copy()

    # Phase 6: Final SLSQP refinements
    for _ in range(5):
        if time_left() < 20:
            break
        result = run_slsqp(best_h, len(best_h), 1000, maxiter=400)
        if result is not None:
            result = project_to_feasible(result, len(best_h))
            c5 = compute_c5(result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.copy()

    # Phase 7: Binary SA on current best if even length
    n_cur = len(best_h)
    if n_cur % 2 == 0 and time_left() > 60:
        h_bin = make_binary_from_continuous(best_h, n_cur)
        if h_bin is not None:
            h_bin_best, bin_best_max = binary_sa(h_bin, n_cur, 100000, 3.0, 0.05, seed + 200)
            bin_c5 = bin_best_max * (2.0 / n_cur)
            if bin_c5 < best_c5:
                best_c5 = bin_c5
                best_h = h_bin_best.copy()
                # Refine continuously
                result = run_slsqp(best_h, n_cur, 500, maxiter=400)
                if result is not None:
                    result = project_to_feasible(result, n_cur)
                    c5 = compute_c5(result)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = result.copy()

    return best_h, best_c5, len(best_h)
