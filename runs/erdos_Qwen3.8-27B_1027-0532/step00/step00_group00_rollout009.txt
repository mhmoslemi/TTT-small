import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for idx in order:
            add = min(1.0 - h[idx], rem)
            h[idx] += add
            rem -= add
            if rem < 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        rem = -diff
        for idx in order:
            sub = min(h[idx], rem)
            h[idx] -= sub
            rem -= sub
            if rem < 1e-12:
                break
    return h

def aa(h, n, dx, rng, th, tl, ni, ss):
    bc = compute_c5(h, dx)
    h = h.astype(np.float64).copy()
    omh = 1.0 - h
    ratio = tl / th if th > 0 else 0.0
    for it in range(ni):
        f = it / max(1, ni - 1)
        t = th * (ratio ** f) if th > 0 else 0.0
        s = ss * (1.0 - 0.5 * f)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        d = rng.uniform(-s, s)
        oi = h[i]
        oj = h[j]
        x = oi + d
        y = oj - d
        if x < 0.0 or x > 1.0 or y < 0.0 or y > 1.0:
            continue
        h[i] = x
        h[j] = y
        omh[i] = 1.0 - x
        omh[j] = 1.0 - y
        c = float(np.max(np.correlate(h, omh, mode='full')) * dx)
        if c <= bc:
            bc = c
        elif t > 1e-15:
            u = rng.random()
            if u > 1e-15 and -np.log(1.0 - u) * t < bc - c:
                pass
            else:
                h[i] = oi
                h[j] = oj
                omh[i] = 1.0 - oi

        else:
            h[i] = oi
            h[j] = oj
