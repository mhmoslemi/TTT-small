```python
def project_onto_simplex(h, target_sum):
    """Project h onto {x ∈ [0,1]^n : sum(x) = target_sum}"""
    n = len(h)
    # Use bisection on the Lagrange multiplier
    # We want: sum(clamp(h - mu, 0, 1)) = target_sum
    lo, hi = -1, 1
    for _ in range(100):
        mu = (lo + hi) / 2
        projected = np.clip(h - mu, 0, 1)
        if projected.sum() > target_sum:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
def swap_delta(h, i, j, n):
    """Compute delta_c[m] for swapping h[i]:0->1, h[j]:1->0"""
    lag_range = np.arange(-(n-1), n)  # all possible lags
    # For each lag, compute the change
    delta = np.zeros(2*n - 1)
    for idx, lag in enumerate(lag_range):
        m = idx  # m = lag + (n-1)
        d = 0.0
        # Term from h[i] change (0->1):
        # -h[i+lag] (if valid)
        if 0 <= i + lag < n:
            d -= h[i + lag]
        # +(1-h[i-lag]) (if valid)
        if 0 <= i - lag < n:
            d += (1.0 - h[i - lag])
        # Term from h[j] change (1->0):
        # +h[j+lag] (if valid)
        if 0 <= j + lag < n:
            d += h[j + lag]
        # -(1-h[j-lag]) (if valid)
        if 0 <= j - lag < n:
            d -= (1.0 - h[j - lag])
        delta[m] = d
    return delta * (2.0 / n)  # multiply by dx
```

```python
    delta[mask_j_minus] += h[j - L[mask_j_minus]]
    mask_i_minus = (i - L >= 0) & (i - L < n)
    delta[mask_i_minus] -= h[i - L[mask_i_minus]]
```

```python
        for i in zeros:
```

```python
import numpy as np
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def project_onto_constraint(h, target_sum):
    n = h.size
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def round_to_binary(h, target_sum):
    n = len(h)
    num_ones = int(round(target_sum))
    h_sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:num_ones]] = 1.0
    return h_bin


def compute_swap_delta(h, i, j, n, L, dx):
    delta = np.zeros(2 * n - 1)
    mask = (j + L >= 0) & (j + L < n)
    delta[mask] += h[j + L[mask]]
    mask = (i + L >= 0) & (i + L < n)
    delta[mask] -= h[i + L[mask]]
    mask = (j - L >= 0) & (j - L < n)
    delta[mask] += h[j - L[mask]]
    mask = (i - L >= 0) & (i - L < n)
    delta[mask] -= h[i - L[mask]]
    return delta * dx


def binary_hill_climb(h, n, target_sum, L, dx, time_limit, rng):
    overlaps = get_overlaps(h)
    current_c5 = np.max(overlaps)
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        pairs = [(i, j) for i in zeros for j in ones]
        rng.shuffle(pairs)
        for i, j in pairs:
            delta = compute_swap_delta(h, i, j, n, L, dx)
            new_c5 = np.max(overlaps + delta)
            if new_c5 < current_c5 - 1e-12:
                h[i] = 1.0
                h[j] = 0.0
                overlaps += delta
                current_c5 = new_c5
                improved = True
                break
        if not improved:
            break
    return h, current_c5

    overlaps = get_overlaps(h)
    current_c5 = np.max(overlaps)
    best_c5 = current_c5
    best_h = h.copy()
    start = time.time()
    T0 = 0.04
    Tmin = 0.0003
    while time.time() - start < time_limit:
        frac = (time.time() - start) / time_limit
        T = T0 * (1.0 - frac) ** 2 + Tmin * frac
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        delta = compute_swap_delta(h, i, j, n, L, dx)
        new_overlaps = overlaps + delta
        new_c5 = np.max(new_overlaps)
        if new_c5 < current_c5:
```
