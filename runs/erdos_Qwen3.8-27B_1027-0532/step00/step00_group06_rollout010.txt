    starts.append(repair_h(h_rand, n))
    h_pert = h_interp + rng.normal(0, 0.15, n)
    starts.append(repair_h(h_pert, n))
    x = np.linspace(0, 2, n)
    h_periodic = 0.5 + 0.35 * np.sin(2 * np.pi * x * 3)
    starts.append(repair_h(np.clip(h_periodic, 0, 1), n))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    h_init = initial_h_values.copy().astype(float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    h1, c5_1 = slsqp_optimize(h_init, n_init, maxiter=500)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    if tl() > 60:
        h2, c5_2 = softmax_optimize(best_h, len(best_h), betas=(30, 80, 200), maxiter=200)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    n_candidates = sorted(set([n_init, 80, 100, 120, 140, 160, 200, 280, 350, 500]))

    for n in n_candidates:
        if tl() < 120:
            break

        h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(best_h)), best_h)
        h_interp = repair_h(h_interp, n)

        h_opt, c5_opt = slsqp_optimize(h_interp, n, maxiter=400)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

        if tl() > 60:
            h2, c5_2 = softmax_optimize(best_h if len(best_h) == n else h_interp, n,
                                        betas=(40, 100, 250), maxiter=100)
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2

        if tl() > 45:
            h_rand = rng.uniform(0, 1, n)
            h_rand = repair_h(h_rand, n)
            h_opt2, c5_opt2 = slsqp_optimize(h_rand, n, maxiter=200)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    for _ in range(20):
        if tl() < 30:
            break
        n = len(best_h)
        noise_scale = 0.03 + 0.02 * rng.random()
        h_pert = best_h + rng.normal(0, noise_scale, n)
        h_pert = repair_h(h_pert, n)
        h_opt, c5_opt = slsqp_optimize(h_pert, n, maxiter=300)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_1 if c5_opt < best_c5 else best_c5
            best_c5 = c5_opt

    if tl() > 60:
        h3, c5_3 = slsqp_optimize(best_h, len(best_h), maxiter=1000, ftol=1e-14)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    return best_h, best_c5, len(best_h)
