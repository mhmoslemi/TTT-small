import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -h.max() - 1, -h.min() + 1  # lambda range
    # We want: sum(clamp(h - lambda, 0, 1)) = target_sum
    
    def sum_clamped(lam):
        return np.clip(h - lam, 0.0, 1.0).sum()
    
    # Binary search
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_clamped(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_subgrad(h):
    """Compute subgradient of max_k correlate(h, 1-h, k) * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(c)
    
    # Active set
    tol = 1e-8
    active = np.where(c >= c5 - tol)[0]
    
    sg = np.zeros(n)
    for k in active:
        m = k - (n - 1)  # shift
        # ∂c[k]/∂h[j] = (1-h[j+m])*1_{0<=j+m<n} - h[j-m]*1_{0<=j-m<n}
        if m >= 0:
            # (1-h[j+m]) for j = 0 to n-1-m
            j_range = np.arange(n - m)
            sg[j_range] += (1.0 - h[j_range + m])
            # -h[j-m] for j = m to n-1
            j_range2 = np.arange(m, n)
            sg[j_range2] += (-h[j_range2 - m])
        else:
            mp = -m  # positive
            # (1-h[j-m]) for j = m' to n-1 where m' = -m... 
            # Wait, m is negative, so j+m = j-|m|
            # (1-h[j+m]) valid when 0 <= j+m < n, i.e., |m| <= j < n
            # -h[j-m] valid when 0 <= j-m < n, i.e., 0 <= j < n+|m|, always true for j < n
            # Actually j-m = j+|m|, valid when j+|m| < n, i.e., j < n-|m|
            j_range = np.arange(mp, n)
            sg[j_range] += (1.0 - h[j_range + m])  # j_range + m = j_range - mp, valid since j_range >= mp
            j_range2 = np.arange(0, n - mp)
            sg[j_range2] += (-h[j_range2 - m])  # j_range2 - m = j_range2 + mp, valid since j_range2 < n-mp
            
    sg /= max(len(active), 1)
    sg *= dx
    return c5, sg

    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP with log-sum-exp smoothing
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    for alpha in [30.0, 80.0, 200.0, 500.0]:

            max_c = np.max(corr)

            if c5_new < best_c5:

            max_c = np.max(corr)

                c5 = compute_c5(hx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hx.copy()
        except Exception:
            pass
    
    return best_h, best_c5, n

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj_and_grad(x, alpha, dx):
    n = len(x)
    c_raw = np.correlate(x, 1.0 - x, mode="full")
    c = c_raw * dx
    max_c = np.max(c)
    shifted = alpha * (c - max_c)
    exp_s = np.exp(np.clip(shifted, -100.0, 0.0))
    log_sum = np.sum(exp_s)
    f = max_c + (1.0 / alpha) * np.log(log_sum)
    p = exp_s / log_sum
    grad = np.zeros(n)
    n_half = n - 1
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-16:
            continue
        L = k - n_half
        if L == 0:
            grad[:] += w * (1.0 - 2.0 * x)
        elif L > 0:
            j1 = np.arange(L, n)
            grad[j1] += w * (1.0 - x[j1 - L])
            j2 = np.arange(0, n - L)
            grad[j2] += w * (-x[j2 + L])
        else:
            pp = -L
            j1 = np.arange(0, n - pp)
            grad[j1] += w * (1.0 - x[j1 + pp])
            j2 = np.arange(pp, n)
            grad[j2] += w * (-x[j2 - pp])
    return f, grad

def subgrad_c5(h, dx):
    n = len(h)
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    c5 = np.max(c)
    tol = 1e-10

    sg = np.zeros(n)
    n_half = n - 1
    for k in active:
        L = k - n_half
        if L == 0:
            sg[:] += (1.0 - 2.0 * h)
        elif L > 0:
            j1 = np.arange(L, n)
            sg[j1] += (1.0 - h[j1 - L])
            j2 = np.arange(0, n - L)
            sg[j2] += (-h[j2 + L])

            j1 = np.arange(0, n - pp)
            sg[j1] += (1.0 - h[j1 + pp])
            j2 = np.arange(pp, n)
            sg[j2] += (-h[j2 - pp])
    sg /= max(len(active), 1)
    sg *= dx
    return c5, sg
