import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def project_to_feasible(y, S):
    """Project y onto {x : 0 <= x <= 1, sum(x) = S}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max()
    
    for _ in range(100):
        lam = (lo + hi) / 2
        x = np.clip(y - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    return x

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def subgradient_step(h, top_k=20, alpha=0.01):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)
        if d >= 0:
            grad[:n-d] += (1 - h[d:])
            grad[d:] += (-h[:n-d])
        else:
            grad[-d:] += (1 - h[:n+d])
            grad[:n+d] += (-h[-d:])
    
    grad /= top_k
    return h - alpha * grad, corr

def optimize_subgradient(h0, max_time=60, top_k=20):
    n = len(h0)
    S = n / 2.0
    h = h0.copy()
    h = project_to_feasible(h, S)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start = time.time()
    alpha = 0.1
    min_alpha = 1e-6
    
    while time.time() - start < max_time:
        h_new, corr = subgradient_step(h, top_k=top_k, alpha=alpha)
        h_new = project_to_feasible(h_new, S)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.95
            if alpha < min_alpha:
                break
        h = h_new
    
    return best_h, best_c5

import numpy as np
import time

def project_feasible(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        x = np.clip(y - lam, 0.0, 1.0)
        if x.sum() > S:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    return x

def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n), corr

def compute_subgrad(h, top_k=30, tau=3.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    if top_k >= m:
        idxs = np.arange(m)
        w = np.full(m, 1.0 / m)
    else:
        idxs = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idxs]
        w = np.exp(tau * (vals - vals.max()))
        w = w / w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(idxs, w):
        d = idx - (n - 1)
        if d >= 0:
            nd = n - d
            if nd > 0:
                grad[d:] += wi * (1.0 - h[:nd])
                grad[:nd] += wi * (-h[d:])
        else:
            dp = -d
            nd = n - dp
            if nd > 0:
                grad[:nd] += wi * (1.0 - h[dp:])
                grad[dp:] += wi * (-h[:nd])
    return grad, corr

def optimize(h0, t_max=60.0, top_k=30, tau=3.0):
    n = len(h0)
    S = n * 0.5
    h = project_feasible(h0.astype(float).copy(), S)
    best_h = h.copy()
    best_val, _ = eval_c5(h)
    t0 = time.time()
    alpha = 0.08
    stall = 0
    stall_limit = 2000
    while time.time() - t0 < t_max:
        grad, corr = compute_subgrad(h, top_k=top_k, tau=tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_new = h - alpha * grad / gnorm
        h_new = project_feasible(h_new, S)
        val, _ = eval_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            stall = 0
            alpha = min(alpha * 1.01, 0.4)
            tau = min(tau * 1.005, 20.0)
        else:
            stall += 1
            alpha *= 0.99
            if stall > stall_limit or alpha < 1e-9:
                break
        h = h_new
    return best_h, best_val

def make_random(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.random(n)
    return project_feasible(h, n * 0.5)

def make_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h
