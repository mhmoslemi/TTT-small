import numpy as np
import time


def project_onto_box_sum(v, target_sum):
    n = len(v)
    if n == 0:
        return v.copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def precompute_grad_struct(n):
    k = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)
    J, K = np.meshgrid(j, k, indexing='ij')
    T1 = J - K
    T2 = J + K
    M1 = (T1 >= 0) & (T1 < n)
    M2 = (T2 >= 0) & (T2 < n)
    return T1[M1].copy(), M1.copy(), T2[M2].copy(), M2.copy(), 2 * n - 1


def smooth_obj_grad(h, T1v, M1, T2v, M2, ncorr, eps):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    mc = float(np.max(corr))
    sh = np.clip((corr - mc) / eps, -50.0, 50.0)
    ev = np.exp(sh)
    Z = float(ev.sum())
    w = ev / Z
    obj = mc + eps * float(np.log(Z))
    SG = np.zeros((n, ncorr))
    SG[M1] += (1.0 - h[T1v])
    SG[M2] -= h[T2v]
    grad = (SG.T.dot(w)) * dx
    return obj, grad


def grad_descent(h_init, n, max_iters, t0, budget_s, frac, seed):
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    T1v, M1, T2v, M2, nc = precompute_grad_struct(n)
    h = project_onto_box_sum(h_init.astype(float).copy(), target_sum)
    best_local = compute_c5(h)
    best_local_h = h.copy()
    eps = 0.03
    step = 0.8 / n
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > budget_s * frac:
            break
        obj, grad = smooth_obj_grad(h, T1v, M1, T2v, M2, nc, eps)
        alpha = step
        ok = False
        for _ in range(18):
            hn = project_onto_box_sum(h - alpha * grad, target_sum)
            objn, _ = smooth_obj_grad(hn, T1v, M1, T2v, M2, nc, eps)
            if objn < obj - 1e-12:
                h = hn
                step = alpha
                ok = True
                break
            alpha *= 0.5
        if not ok:
            step *= 0.88
            if step < 1e-9:
                break
        if it % 10 == 0:
            c5 = compute_c5(h)
            if c5 < best_local - 1e-7:
                best_local = c5
                best_local_h = h.copy()
                stall = 0
            else:
                stall += 1
        if it % 150 == 149:
            eps = max(eps * 0.94, 0.001)
        if stall > 30:
            eps = 0.03
            step = 0.8 / n
            stall = 0
            h = h + rng.normal(0, 0.015, n)
            h = project_onto_box_sum(h, target_sum)
    return best_local_h, best_local


def try_round(h, n):
    if n % 2 != 0:
        return h, compute_c5(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_r = np.zeros(n)
    h_r[sorted_idx[:target]] = 1.0
    c5_r = compute_c5(h_r)
    c5_orig = compute_c5(h)
    if c5_r < c5_orig:
        return h_r, c5_r
    return h, c5_orig
