    h = np.zeros(n)
    n_ones = int(target_sum)
    idx = rng.permutation(n)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 0:
        remaining = list(set(range(n)) - set(idx[:n_ones]))
        h[remaining[0]] = frac
    return project_feasible(h, target_sum)

def optimize_from(x0, n, target_sum, beta, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        sm = mc + np.log(np.sum(np.exp(beta * (c - mc)))) / beta
        return sm * dx + 100.0 * (np.sum(x) - target_sum) ** 2
    res = minimize(obj, x0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
    x_out = project_feasible(res.x, target_sum)
    return x_out, evaluate_c5(x_out)

def subgradient_polish(h, n, target_sum, max_iters, time_fn):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    dx = 2.0 / n
    alpha = 0.008
    for it in range(max_iters):
        if time_fn() < 15:
            break
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(c))
        ell = k_star - (n - 1)
        sg = np.zeros(n)
        if ell >= 0:
            sg[ell:] += 1.0 - best_h[:n - ell]
            sg[:n - ell] -= best_h[ell:]
        else:
            m = -ell
            sg[:n - m] += 1.0 - best_h[m:]
            sg[m:] -= best_h[:n - m]
        h_try = best_h - alpha * sg * dx
        h_try = project_feasible(h_try, target_sum)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            alpha = min(alpha * 1.03, 0.03)
        else:
            alpha *= 0.97
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    def time_left_fn():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    target_sum = n / 2.0

    h_init = project_feasible(h_init, target_sum)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)

    # Phase 1: L-BFGS-B from initial with increasing beta
    betas_phase1 = [10, 30, 100, 300, 500, 1000]
    for beta in betas_phase1:
        if remaining() < 60:
            break
        x_new, c5_new = optimize_from(best_h, n, target_sum, beta, 400)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 2: Multi-start with perturbations
    perturbations = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]
    for scale in perturbations:
        if remaining() < 45:
            break
        for _ in range(2):
            if remaining() < 30:
                break
            x0 = best_h + rng.randn(n) * scale
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = project_feasible(x0, target_sum)
            beta = 200 if scale < 0.1 else 50
            x_new, c5_new = optimize_from(x0, n, target_sum, beta, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = x_new

    # Phase 3: Try different initializations
    if remaining() > 90:
        # Binary-like init
        for _ in range(5):
            if remaining() < 45:
                break
            x0 = binary_init(n, rng)
            x_new, c5_new = optimize_from(x0, n, target_sum, 50, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = x_new

        # Random continuous init
        for _ in range(5):
            if remaining() < 45:
                break
            x0 = random_feasible_init(n, rng)
            x_new, c5_new = optimize_from(x0, n, target_sum, 30, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = x_new

    # Phase 4: Fine-tuning from best
    if remaining() > 60:
        for beta in [500, 1000, 3000, 5000]:
            if remaining() < 30:
                break
            x_new, c5_new = optimize_from(best_h, n, target_sum, beta, 500)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = x_new

    # Phase 5: Subgradient polishing
    if remaining() > 25:
        best_h, best_c5 = subgradient_polish(best_h, n, target_sum, 30000, time_left_fn)

    # Phase 6: Final L-BFGS-B polish
    if remaining() > 30:
        x_new, c5_new = optimize_from(best_h, n, target_sum, 5000, 500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 7: Try different n if time permits
    if remaining() > 120:
        for n_try in [100, 190, 140]:
            if remaining() < 100:
                break
            ts = n_try / 2.0
            # Scale up/down the best solution
            if n_try > n:
                best_resized = np.interp(
                    np.linspace(0, 1, n_try), np.linspace(0, 1, n), best_h
                )
            else:
                best_resized = np.interp(
                    np.linspace(0, 1, n), np.linspace(0, 1, n_try), best_h
                )
            best_resized = project_feasible(best_resized, ts)
            c5_resized = evaluate_c5(best_resized)

            # Optimize at new resolution
            x_new, c5_new = optimize_from(best_resized, n_try, ts, 100, 300)
            if remaining() > 50:
                x_new2, c5_new2 = optimize_from(x_new, n_try, ts, 500, 400)
                if c5_new2 < c5_new:
                    x_new, c5_new = x_new2, c5_new2

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = x_new

    # Final subgradient polish on best
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    if remaining() > 15:
        best_h, best_c5 = subgradient_polish(best_h, n_final, ts_final, 10000, time_left_fn)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
