    m = slopes
    
    # Find the t in [0,1] that minimizes max_k (c_k + m_k * t)
    # Approach: the minimum is either at t=0, t=1, or at an intersection point
    
    # Collect all candidate t values: 0, 1, and all pairwise intersections in [0,1]
    candidates = [0.0, 1.0, t_current]

        for j in range(i+1, n):
            if abs(m[i] - m[j]) < 1e-12:

            t_cross = (c[j] - c[i]) / (m[i] - m[j])

    best_t = t_current

        val = np.max(c + m * t)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t

    c = corr - slopes * t_current
    m = slopes

    values = np.max(c[None, :] + m[None, :] * t_grid[:, None], axis=1)
    best_idx = np.argmin(values)
    
    # Refine with ternary search
    a = t_grid[max(0, best_idx - 1)]
    b = t_grid[min(len(t_grid)-1, best_idx + 1)]
    
    for _ in range(50):
        if b - a < 1e-10:
            break
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        f1 = np.max(c + m * m1)
        f2 = np.max(c + m * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    return (a + b) / 2

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def minimize_1d_convex_fast(corr, slopes, t_current):
    n_corr = len(corr)
    c = corr - slopes * t_current
    m = slopes
    t_grid = np.linspace(0.0, 1.0, 80)
    values = np.max(c[None, :] + m[None, :] * t_grid[:, None], axis=1)
    best_idx = np.argmin(values)
    a = t_grid[max(0, best_idx - 1)]
    b = t_grid[min(len(t_grid) - 1, best_idx + 1)]
    for _ in range(60):
        if b - a < 1e-12:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c + m * m1)
        f2 = np.max(c + m * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return float((a + b) / 2.0)

def coordinate_descent(h, n_rounds=5, time_budget=None, start_time=None):
    n = len(h)
    for rnd in range(n_rounds):
        if time_budget is not None and time.time() - start_time > time_budget:
            break
        improved = False
        for j in range(n):
            shifts = np.arange(-(n - 1), n)
            idx1 = j - shifts
            valid1 = (idx1 >= 0) & (idx1 < n)
            slope1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
            idx2 = j + shifts
            valid2 = (idx2 >= 0) & (idx2 < n)
            slope2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
            slope = slope1 - slope2
            corr = get_correlations(h)
            best_t = minimize_1d_convex_fast(corr, slope, h[j])
            if abs(best_t - h[j]) > 1e-12:
                h[j] = best_t
                improved = True
        if not improved:
            break
    return h

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    for _ in range(500):
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = repair_h(h_new, n_new)
    return h_new

def make_bump(n, p=1.0):
    x = np.linspace(0.0, 2.0, n)
    t = np.sin(np.pi * x / 2.0) ** p
    integral = np.sum(t) * (2.0 / n)
    h = t / integral
    h = np.clip(h, 0.0, 1.0)
    h = repair_h(h, n)
    return h

def smooth_max_objective(h, alpha):
    corr = get_correlations(h)
    return float(logsumexp(alpha * corr) / alpha)

def smooth_max_gradient_numeric(h, alpha, eps=1e-5):
    n = len(h)
    grad = np.zeros(n)
    base = smooth_max_objective(h, alpha)
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] = min(1.0, h[j] + eps)
        h_minus = h.copy()
        h_minus[j] = max(0.0, h[j] - eps)
        fp = smooth_max_objective(h_plus, alpha)
        fm = smooth_max_objective(h_minus, alpha)
        if h[j] + eps <= 1.0 and h[j] - eps >= 0.0:
            grad[j] = (fp - fm) / (2.0 * eps)
        elif h[j] + eps <= 1.0:
            grad[j] = (fp - base) / eps
        else:
            grad[j] = (base - fm) / eps
    return grad

def scipy_polish(h, n, alpha=50.0, max_iter=200):
    h0 = h.copy()
    target_sum = n / 2.0

    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        s = np.sum(hx)
        if s < 1e-10:
            return 1e6
        penalty = 1000.0 * (s - target_sum) ** 2
        return smooth_max_objective(hx, alpha) + penalty

    res = minimize(obj, h0, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = repair_h(h_new, n)
    return h_new

def symmetric_h(h):
    n = len(h)
    return 0.5 * (h + h[::-1])
