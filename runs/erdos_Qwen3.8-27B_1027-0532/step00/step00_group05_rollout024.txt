            seg_len = rng.randint(2, min(6, n // 4))
            start = rng.randint(0, n - seg_len)
            end = start + seg_len
            seg = h[start:end]
            if np.sum(seg) == 0 or np.sum(seg) == seg_len:
                continue
            old_seg = seg.copy()
            h[start:end] = 1.0 - seg
            new_c5 = get_c5(h, dx)
            delta = new_c5 - c5
            if should_accept(delta, T, rng):
                c5 = new_c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[start:end] = old_seg
        if it_count % 500 == 0:
            T = T_init * (cool_per_iter ** (it_count // 500))
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_h = None
    best_c5 = np.inf
    best_n = 0

    if h_init is not None:
        n = len(h_init)
        if n % 2 == 1:
            n -= 1
            h_init = h_init[:n]
        dx = 2.0 / n
        h = make_binary(h_init, n, rng)
        c5 = get_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        limit = t0 + budget_s * 0.25
        h, c5 = local_search(h, dx, rng, limit, 0.003, 0.999)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    for n_try in [128, 256, 512]:
        if time.time() - t0 > budget_s * 0.55:
            break
        dx = 2.0 / n_try
        h_tm = make_binary(thue_morse(n_try), n_try, rng)
        c5 = get_c5(h_tm, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_tm.copy()
            best_n = n_try
        remaining = budget_s * 0.55 - (time.time() - t0)
        if remaining > 10:
            h_tm, c5 = local_search(h_tm, dx, rng, t0 + budget_s * 0.55, 0.002, 0.9995)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_tm.copy()
                best_n = n_try

    for attempt in range(3):
        if time.time() - t0 > budget_s * 0.65:
            break
        n_try = 200 + attempt * 100
        if n_try % 2 == 1:
            n_try -= 1
        dx = 2.0 / n_try
        h_rand = np.zeros(n_try)
        ones_idx = rng.choice(n_try, n_try // 2, replace=False)
        h_rand[ones_idx] = 1.0
        remaining = budget_s * 0.65 - (time.time() - t0)
        if remaining > 10:
            h_rand, c5 = local_search(h_rand, dx, rng, t0 + budget_s * 0.65 - attempt * 5, 0.005, 0.999)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_rand.copy()
                best_n = n_try

    if best_h is not None and time.time() - t0 < budget_s * 0.85:
        n = best_n
        dx = 2.0 / n
        h = best_h.copy()
        limit = t0 + budget_s * 0.92
        h, c5 = local_search(h, dx, rng, limit, 0.0005, 0.99999)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n

    if best_h is not None and time.time() - t0 < budget_s * 0.92:
        n = best_n
        dx = 2.0 / n
        for attempt in range(20):
            if time.time() - t0 > budget_s * 0.96:
                break
            h_nb = best_h.copy()
            n_perturb = max(1, n // 10)
            idx = rng.choice(n, n_perturb, replace=False)
            h_nb[idx] = np.clip(h_nb[idx] + rng.randn(n_perturb) * 0.15, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h_nb)
            if abs(diff) > 1e-10:
                mask = h_nb < 1.0 if diff > 0 else h_nb > 0.0
                h_nb[mask] += diff / np.count_nonzero(mask)
                h_nb = np.clip(h_nb, 0.0, 1.0)
            for _ in range(50):
                d2 = n / 2.0 - np.sum(h_nb)
                if abs(d2) < 1e-12:
                    break
                mask = h_nb < 1.0 if d2 > 0 else h_nb > 0.0
                cnt = np.count_nonzero(mask)
                if cnt == 0:
                    break
                h_nb[mask] += d2 / cnt
                h_nb = np.clip(h_nb, 0.0, 1.0)
            c5_nb = get_c5(h_nb, dx)
            if c5_nb < best_c5 - 1e-12:
                best_c5 = c5_nb
                best_h = h_nb.copy()
                best_n = n

    try:
        val = evaluate_erdos_solution(best_h)
        if val is not None and val < best_c5:
            best_c5 = val
    except Exception:
        pass

    return (best_h, best_c5, best_n)
