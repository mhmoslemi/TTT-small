```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(y, S):
    """Project y onto {x: 0<=x<=1, sum(x)=S}"""
    N = len(y)
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, N):
    """Compute subgradient of max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active shifts
    tol = max_val * 1e-6 + 1e-10
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(N)
    for m in active:
        p = m - (N - 1)
        # For this p, gradient contribution:
        # ∂corr/∂h[j] = (1-h[j-p])*I(0<=j-p<N) - h[j+p]*I(0<=j+p<N)
        # First term: (1-h[j-p]) for j in [max(0,p), min(N, N+p))
        j1_start = max(0, p)
        j1_end = min(N, N + p)
        if j1_start < j1_end:
            grad[j1_start:j1_end] += (1.0 - h[j1_start - p : j1_end - p])
        # Second term: -h[j+p] for j in [max(0,-p), min(N, N-p))
        j2_start = max(0, -p)
        j2_end = min(N, N - p)
        if j2_start < j2_end:
            grad[j2_start:j2_end] -= h[j2_start + p : j2_end + p]
    
    grad /= len(active)
    return grad, np.max(corr)

def subgradient_descent(h0, S, max_time=60.0):
    """Run subgradient descent."""
    N = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    step_size = 0.1
    min_step = 1e-8
    patience = 500
    no_improve = 0
    
    start = time.time()
    while time.time() - start < max_time:
        grad, max_corr = compute_gradient(h, N)
        
        # Project gradient onto the hyperplane sum(x) = S
        grad = grad - np.mean(grad)
        
        # Take step
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, S)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new  # still move, but don't update best
            no_improve += 1
            step_size *= 0.95
        
        if step_size < min_step or no_improve > patience:
            break
    
    return best_h, best_c5

def local_search_01(h, N, max_time=30.0):
    """Local search on near-0-1 solutions."""
    S = N / 2.0
    h = h.copy()
    
    # Round to nearest 0-1 (maintaining sum)
    # Sort by fractional part
    h_rounded = np.where(h > 0.5, 1.0, 0.0)
    # Adjust to get sum = S
    diff = S - h_rounded.sum()
    if diff > 0:
        # Need to add more ones
        frac_parts = np.where(h_rounded == 0, 1.0 - h, h)  # priority for zero->one
        for _ in range(int(diff)):
            idx = np.argmax(frac_parts)
            h_rounded[idx] = 1.0
            frac_parts[idx] = -1
    elif diff < 0:

        frac_parts = np.where(h_rounded == 1, h, 1.0 - h)  # priority for one->zero
        for _ in range(int(-diff)):
            idx = np.argmax(frac_parts)
            h_rounded[idx] = 0.0
            frac_parts[idx] = -1
    
    best_h = h_rounded.copy()
    best_c5 = evaluate_c5(h_rounded)
    
    start = time.time()
    improved = True
    while improved and time.time() - start < max_time:

        ones = np.where(h_rounded == 1.0)[0]

                    best_h = h_rounded.copy()
                    improved = True
                    break
                h_rounded[i] = 1.0
                h_rounded[j] = 0.0
    
    return best_h, best_c5
```

```python
def smooth_max_gradient(h, N, epsilon=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(shifted / epsilon)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-8:
            continue
        p = m - (N - 1)
        j1_s = max(0, p)
        j1_e = min(N, N + p)
        if j1_s < j1_e:
            grad[j1_s:j1_e] += w * (1.0 - h[j1_s - p : j1_e - p])
        j2_s = max(0, -p)
        j2_e = min(N, N - p)
        if j2_s < j2_e:
            grad[j2_s:j2_e] -= w * h[j2_s + p : j2_e + p]
    
    return grad
```
