from scipy.optimize import minimize
import time

def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, dx):
    return float(np.max(compute_corr(h, dx)))

def smoothed_max_value(corr, alpha):
    m = np.max(corr)
    if m <= 0:
        return 0.0
    shifted = np.clip(alpha * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smoothed_max_grad(corr, h, dx, alpha):
    """Compute gradient of smoothed max of corr w.r.t. h."""
    n = len(h)
    m = np.max(corr)
    if m <= 0:
        return np.zeros(n)
    shifted = np.clip(alpha * (corr - m), -500, 0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    # For each k (lag index in the full correlation):
    # l = k - (n-1)
    # dc_k/dh[j] = dx * [(1-h[j+l])*I(0<=j+l<n) - h[j-l]*I(0<=j-l<n)]
    for k in range(2*n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        # First term: (1-h[j+l]) for valid j
        js = max(0, -l)
        je = min(n - 1, n - 1 - l)
        if js <= je:
            for j in range(js, je + 1):
                grad[j] += w * dx * (1.0 - h[j + l])
        # Second term: -h[j-l] for valid j
        js2 = max(0, l)
        je2 = min(n - 1, n - 1 + l)
        if js2 <= je2:
            for j in range(js2, je2 + 1):
                grad[j] -= w * dx * h[j - l]
    
    return grad

def normalize_h(h, n):
    """Normalize h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) < 1e-10:
        return h
    if s > 0:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    # After clipping, sum might not be exact; fix with water-filling
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h
