        res = minimize(f, h0, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
        return res.x

from scipy.optimize import minimize

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    
    def c5_of(h, n):
        dx = 2.0 / n
        return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    
    def smooth_max(v, beta):
        m = np.max(v)
        return m + np.log(np.sum(np.exp((v - m) * beta))) / beta
    
    def repair(h, n):
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        s = np.sum(h)
        if s < 1e-10:
            return np.full(n, 0.5)
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
        return h
    
    def do_slsqp(h0, n, beta, max_iter=3000):
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
        
        def f(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode="full") * dx
            return smooth_max(corr, beta)
        
        res = minimize(f, h0, method='SLSQP', bounds=bounds, cons=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return res.x
    
    best_h = h_init.copy()
    best_c5 = c5_of(h_init, n0)
