def compute_subgradient(h, active_lag):
    """Compute subgradient of C5 w.r.t. h at the active lag."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # active_lag is the index in the "full" correlation output
    # which corresponds to lag = active_lag - (n-1)

        #           + (-h[j+lag]) if 0 <= j <= n-1-lag (h[j] appears as h[i] with i=j)
        for j in range(lag, n):
            grad[j] += (1 - h[j-lag])
        for j in range(0, n-lag):

        l = -lag
        for j in range(0, n-l):
            grad[j] += (1 - h[j+l])
        for j in range(l, n):
            grad[j] += (-h[j-l])

            l = -lag
            grad[:n-l] += (1.0 - h[l:])

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp(eps * (corr - max_c))
    return dx * max_c + (dx / eps) * np.log(exp_c.sum())

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp(eps * (corr - max_c))
    total = exp_c.sum()
    weights = exp_c / total
    
    grad = np.zeros(n)
    sig = weights > 1e-10
    for k in np.where(sig)[0]:
        w = weights[k]
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n-l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n-l]
    return grad * dx
