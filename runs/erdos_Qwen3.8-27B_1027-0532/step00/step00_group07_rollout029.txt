- This gives g[lag:n] += 1-h[0:n-lag] ✓

So for lag > 0:
```python
g = np.zeros(n)
g[lag:] += (1.0 - h[:n-lag])
g[:n-lag] += -h[lag:]
```

For lag < 0 (let m = -lag > 0):
- First term (1-h[j-lag]) = (1-h[j+m]): valid for 0 ≤ j+m < n, i.e., j ∈ [0, n-1-m]
- Second term -h[j+lag] = -h[j-m]: valid for 0 ≤ j-m < n, i.e., j ∈ [m, n-1]

Vectorized:
```python
g = np.zeros(n)
g[:n-m] += (1.0 - h[m:])  # first term for j in [0, n-1-m]
g[m:] += -h[:n-m]  # second term for j in [m, n-1]
```

Let me verify with the earlier example (n=3, h=[0.5,0.5,0.5], lag=1):
g = zeros(3)
g[1:] += (1-h[:2]) = (1-[0.5,0.5]) = [0.5, 0.5] → g = [0, 0.5, 0.5]
g[:2] += -h[1:] = -[0.5, 0.5] → g = [-0.5, 0, 0.5]

Expected: ∂c(1)/∂h = [-0.5, 0, 0.5] ✓

For lag=-1 (m=1):
g = zeros(3)
g[:2] += (1-h[1:]) = (1-[0.5,0.5]) = [0.5, 0.5] → g = [0.5, 0.5, 0]
g[1:] += -h[:2] = -[0.5, 0.5] → g = [0.5, 0, -0.5]

Expected: c(-1) = h[0](1-h[1]) + h[1](1-h[2])
∂c(-1)/∂h[0] = (1-h[1]) = 0.5 ✓
∂c(-1)/∂h[1] = h[0] - (1-h[2]) = 0.5 - 0.5 = 0 ✓
∂c(-1)/∂h[2] = -h[1] = -0.5 ✓



So the vectorized gradient for a single lag is:

And the full smoothed gradient:
```python
def smooth_gradient(h, alpha):
    n = len(h)
    n_corr = 2*n - 1
    corr = np.correlate(h, 1.0-h, mode="full")
    mc = np.max(corr)
    w = np.exp(alpha * (corr - mc))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(n_corr):
        if w[k] < 1e-12:
            continue
        lag = k - (n-1)
        grad += w[k] * grad_single_lag(h, lag)
    return grad
```

This is O(n * n_corr) = O(n²) which is fine.

At the optimum, typically several k's are "active" (achieving the max). The KKT conditions say that the gradient of the weighted sum of active c_k's should be -λ * 1 (for the sum constraint) + μ (for the box constraints).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def grad_single_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] += -h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] += -h[:n - m]
    return g

def project_feasible(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(p)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    n_corr = 2 * n - 1

    def time_left():
        return budget_s - (time.time() - start_time)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n

    alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0]
    mu_base = 200.0

    for alpha in alphas:
        n_restarts = 8 if alpha <= 20 else 5
        for restart in range(n_restarts):
            if time_left() < 40:
                break

            if restart == 0:
                h_init = best_h.copy()
            elif restart == 1:
                h_init = best_h + rng.randn(n) * 0.05
            elif restart == 2:
                h_init = best_h + rng.randn(n) * 0.1
            elif restart == 3:
                h_init = best_h + rng.randn(n) * 0.15
            elif restart == 4:
                h_init = best_h * 0.8 + 0.2 * (1.0 - best_h) + rng.randn(n) * 0.05
            elif restart == 5:
                perm = rng.permutation(n)
                h_init = best_h[perm].copy()
            elif restart == 6:
                h_init = best_h[::-1].copy() + rng.randn(n) * 0.03
            else:
                h_init = rng.rand(n)

            h_init = np.clip(h_init, 0.0, 1.0)

            mu = mu_base

            def obj(h):
                hc = np.clip(h, 0.0, 1.0)
                corr = np.correlate(hc, 1.0 - hc, mode="full")
                mc = np.max(corr)
                sm = mc + np.log(np.sum(np.exp(alpha * (corr - mc)))) / alpha
                pen = mu * (np.sum(hc) - target_sum) ** 2
                return sm * dx + pen

            def obj_grad(h):
                hc = np.clip(h, 0.0, 1.0)
                corr = np.correlate(hc, 1.0 - hc, mode="full")
                mc = np.max(corr)
                w = np.exp(alpha * (corr - mc))
                w /= np.sum(w)
                grad = np.zeros(n)
                for k in range(n_corr):
                    if w[k] < 1e-13:
                        continue
                    lag = k - (n - 1)
                    grad += w[k] * grad_single_lag(hc, lag)
                grad *= dx
                grad += 2.0 * mu * (np.sum(hc) - target_sum)
                return grad

            try:
                res = minimize(
                    obj, h_init, jac=obj_grad,
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 400, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
                )
                h_cand = project_feasible(res.x, target_sum)
                c_cand = compute_c5(h_cand, n)
                if c_cand < best_c5:
                    best_c5 = c_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

        if time_left() < 60:
            break

    # Fine-tuning: active set refinement with very high alpha
    alpha_fine = 300.0
    mu_fine = 500.0
    for restart in range(10):
        if time_left() < 30:
            break
        h_init = best_h + rng.randn(n) * 0.01

        def obj_fine(h):
            hc = np.clip(h, 0.0, 1.0)
            corr = np.correlate(hc, 1.0 - hc, mode="full")
            mc = np.max(corr)
            sm = mc + np.log(np.sum(np.exp(alpha_fine * (corr - mc)))) / alpha_fine
            pen = mu_fine * (np.sum(hc) - target_sum) ** 2
            return sm * dx + pen

        def obj_fine_grad(h):
            hc = np.clip(h, 0.0, 1.0)
            corr = np.correlate(hc, 1.0 - hc, mode="full")
            mc = np.max(corr)
            w = np.exp(alpha_fine * (corr - mc))
            w /= np.sum(w)
            grad = np.zeros(n)
            for k in range(n_corr):
                if w[k] < 1e-13:
                    continue
                lag = k - (n - 1)
                grad += w[k] * grad_single_lag(hc, lag)
            grad *= dx
            grad += 2.0 * mu_fine * (np.sum(hc) - target_sum)
            return grad

        try:
            res = minimize(
                obj_fine, h_init, jac=obj_fine_grad,
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
            )
            h_cand = project_feasible(res.x, target_sum)
            c_cand = compute_c5(h_cand, n)
            if c_cand < best_c5:
                best_c5 = c_cand
                best_h = h_cand.copy()
        except Exception:
            pass

    # Coordinate descent polish
    if time_left() > 20:
        for _ in range(200):
            if time_left() < 5:
                break
            h = best_h.copy()
            improved = False
            idx_order = rng.permutation(n)
            for idx in idx_order:
                corr_old = compute_c5(h, n)
                for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    h2 = h.copy()
                    h2[idx] = np.clip(h2[idx] + delta, 0.0, 1.0)
                    h2 = project_feasible(h2, target_sum)
                    c2 = compute_c5(h2, n)
                    if c2 < corr_old - 1e-8:
                        h = h2
                        improved = True
                        break
            if not improved:
                break
            if compute_c5(h, n) < best_c5:
                best_c5 = compute_c5(h, n)
                best_h = h.copy()

    best_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h, n)
    if final_c5 > best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
```
