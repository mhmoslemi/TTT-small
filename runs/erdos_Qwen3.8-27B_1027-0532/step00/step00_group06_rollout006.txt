    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    x_old = (np.arange(m) + 0.5) * 2.0 / m
    x_new = (np.arange(n) + 0.5) * 2.0 / n
    return np.interp(x_new, x_old, parent)

def fix_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(1000):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

            step = room
        if diff < 0 and step < -room:
            step = -room
        h[mask] += step

import numpy as np
from scipy.optimize import minimize
import time

_g_target_sum = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _con_fun(x):
    return float(np.sum(x)) - _g_target_sum

def _con_jac(x):
    return np.ones(len(x))

def interpolate_to_n(parent, n):
    m = len(parent)
    x_old = (np.arange(m) + 0.5) * 2.0 / m
    x_new = (np.arange(n) + 0.5) * 2.0 / n
    return np.interp(x_new, x_old, parent)

def fix_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(1000):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        cnt = float(np.sum(mask))
        if diff > 0:
            room = float((1.0 - h[mask]).min())
        else:
            room = float(h[mask].min())
        step = diff / cnt
        if diff > 0 and step > room:
            step = room
        elif diff < 0 and step < -room:
            step = -room
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

def time_remaining(budget_s, start_time):
    return budget_s - (time.time() - start_time)

def run_slsqp(h0, n, target_sum, maxiter):
    global _g_target_sum
    _g_target_sum = target_sum
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return res
    except Exception:
        return None

def local_search_swap(h, n, target_sum, rng, max_sweeps, budget_s, start_time):
    h = np.asarray(h, dtype=float).copy()
    h = fix_sum(h, target_sum)
    c5_cur = compute_c5(h)
    alpha = 0.05
    for sweep in range(max_sweeps):
        if time_remaining(budget_s, start_time) < 10:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time_remaining(budget_s, start_time) < 5:
                break
            for _ in range(3):
                k = rng.randint(n)
                if k == j:
                    continue
                amt = min(alpha, h[j], 1.0 - h[k])
                if amt > 1e-8:
                    backup = h.copy()
                    h[j] -= amt
                    h[k] += amt
                    c5_new = compute_c5(h)
                    if c5_new < c5_cur - 1e-14:
                        c5_cur = c5_new
                        improved = True
                    else:
                        h = backup
                amt = min(alpha, h[k], 1.0 - h[j])
                if amt > 1e-8:
                    backup = h.copy()
                    h[k] -= amt
                    h[j] += amt
                    c5_new = compute_c5(h)
                    if c5_new < c5_cur - 1e-14:
                        c5_cur = c5_new
                        improved = True
                    else:
                        h = backup
        if not improved:
            alpha *= 0.8
            if alpha < 0.002:
                break
    return h, c5_cur
