from scipy.optimize import minimize
import time

def evaluate_c5(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(v, n):
    target = n / 2.0
    def sum_at_lambda(lam):
        return np.sum(np.clip(v - lam, 0.0, 1.0))
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if sum_at_lambda(mid) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_gradient_at_lag(h, j, n):
    """Compute gradient of g_j(h) = sum_t h[t]*(1-h[t+j]) w.r.t. h"""
    grad = np.zeros(n)
    for s in range(n):
        if 0 <= s + j < n:
            grad[s] += (1.0 - h[s + j])
        if 0 <= s - j < n:
            grad[s] -= h[s - j]
    return grad

def find_worst_lag(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    j = k_star - (n - 1)
    return j, corr[k_star]

def subgradient_optimize(h, n, dx, max_iters=5000, time_limit=100):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h, n, dx)
    
    alpha = 0.1
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        j, _ = find_worst_lag(h, n)
        grad = compute_gradient_at_lag(h, j, n)
        
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n)
        
        val = evaluate_c5(h_new, n, dx)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        alpha *= 0.999  # Slow decay
    
    return best_h, best_val

def local_search_swap(h, n, dx, max_iters=10000, time_limit=50):
    start = time.time()

    current_val = evaluate_c5(h, n, dx)
    best_h = h.copy()
