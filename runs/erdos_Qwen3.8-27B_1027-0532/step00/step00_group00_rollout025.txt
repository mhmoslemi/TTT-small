import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_sum(h, target_sum):
    np.clip(h, 0.0, 1.0, out=h)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-9) & (h < 1.0 - 1e-9)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] -= diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    diff2 = np.sum(h) - target_sum
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask2 = h > 1e-9
        else:
            mask2 = h < 1.0 - 1e-9
        cnt2 = np.sum(mask2)
        if cnt2 > 0:
            h[mask2] -= diff2 / cnt2
            np.clip(h, 0.0, 1.0, out=h)
    return h


def local_search_continuous(h, target_sum, rng, n_iters, deadline, eps_min, eps_max):
    n = len(h)
    best_c5 = evaluate_c5(h)
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        eps = rng.uniform(eps_min, eps_max)
        si = h[i]
        sj = h[j]
        if rng.random() < 0.5:
            if si < eps or sj > 1.0 - eps:
                continue
            h[i] = si - eps
            h[j] = sj + eps
        else:
            if sj < eps or si > 1.0 - eps:
                continue
            h[j] = sj - eps
            h[i] = si + eps
        c5 = evaluate_c5(h)
        if c5 <= best_c5:
            if c5 < best_c5 - 1e-12:
                stall = 0
            else:
                stall += 1
            best_c5 = c5
        else:
            h[i] = si
            h[j] = sj
            stall += 1
    return h, best_c5


def local_search_binary(h, rng, n_iters, deadline):
