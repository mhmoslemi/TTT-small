    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(vals, mu):
    """Stable log-sum-exp based smooth max."""
    v = vals / mu
    v_max = float(np.max(v))
    return mu * (v_max + float(np.log(np.sum(np.exp(v - v_max)))))

def c5_value(h):
    """Compute the actual C5 value."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corrs(h))) * dx
