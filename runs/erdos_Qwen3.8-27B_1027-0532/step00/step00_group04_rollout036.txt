    v = np.array(v, dtype=float).copy()

        room = 1.0 - v

        else:
            v += diff / n
    else:

        room = v.copy()
        total_room = np.sum(room)
        if total_room > 1e-10:

        else:

        else:

            v[mask] += diff / np.sum(mask)
        v = np.clip(v, 0, 1)

    n = len(h)

def repair(v, n):
    v = np.array(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)

    diff = target - np.sum(v)
