import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def smooth_obj_func(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    s = corr - m
    return m + (1.0 / alpha) * float(np.log(np.sum(np.exp(alpha * s))))


def repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
            else:
                h[:] = np.minimum(h + diff / n, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
            else:
                h[:] = np.maximum(h + diff / n, 0.0)
    return h


def interpolate_to(parent, n):
    np_ = len(parent)
    if np_ == n:
        return parent.astype(np.float64).copy()
    xp = np.linspace(0.0, 2.0, np_)
    xn = np.linspace(0.0, 2.0, n)
    f = interp1d(xp, parent, kind='linear', fill_value='extrapolate', assume_sorted=True)
    return repair(f(xn), n)


class _Obj:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return compute_c5(x, self.n)


class _SmoothObj:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha
    def __call__(self, x):
        return smooth_obj_func(x, self.n, self.alpha)


class _SumCon:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return float(np.sum(x) - self.n / 2.0)
    def jac(self, x):
        return np.ones(self.n)


def _make_cons(n):
    sc = _SumCon(n)
    return [{'type': 'eq', 'fun': sc, 'jac': sc.jac}]


def _make_bounds(n):
    return [(0.0, 1.0)] * n


def optimize_grid(h0, n, time_limit, seed=42):
    t0 = time.time()
    h0 = repair(h0.copy(), n)
    best_h = h0.copy()
    best_val = compute_c5(best_h, n)
    bounds = _make_bounds(n)
    cons = _make_cons(n)
    obj = _Obj(n)

    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-14})
        h_new = repair(res.x, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_h = h_new
            best_val = val
    except Exception:
        pass

    if time.time() - t0 > time_limit:
        return best_h, best_val

    for alpha in [15, 30, 60, 120]:
        if time.time() - t0 > time_limit - 5:
            break
        so = _SmoothObj(n, alpha)
        try:
            res = minimize(so, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = repair(res.x, n)
            val = compute_c5(h_new, n)
            if val < best_val:
                best_h = h_new
                best_val = val
        except Exception:
            pass

    if time.time() - t0 > time_limit:
        return best_h, best_val

    rng = np.random.RandomState(seed)
    n_restarts = min(10, max(3, int((time_limit - (time.time() - t0)) / 10)))
    for i in range(n_restarts):
        if time.time() - t0 > time_limit:
            break
        scale = 0.02 + 0.01 * i
        hp = best_h + rng.uniform(-scale, scale, n)
        hp = repair(hp, n)
        try:
            res = minimize(obj, hp, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = repair(res.x, n)
            val = compute_c5(h_new, n)
            if val < best_val:
                best_h = h_new
                best_val = val
        except Exception:
            pass

    return best_h, best_val
