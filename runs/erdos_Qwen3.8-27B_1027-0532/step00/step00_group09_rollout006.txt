    h = np.clip(h, 0, 1)

def coordinate_step(h, i, j, n, target_sum):
    """Optimize h[i] and h[j] with h[j] = target_sum - sum(others) - h[i]"""
    others_sum = np.sum(h) - h[i] - h[j]
    # h[j] = target_sum - others_sum - h[i]
    # Let t = target_sum - others_sum, so h[j] = t - h[i]
    t = target_sum - others_sum
    # h[i] ranges over [max(0, t-1), min(1, t)]
    lo = max(0.0, t - 1.0)
    hi = min(1.0, t)
    if lo >= hi:
        return h[i], h[j]
    
    # For each shift, compute the coefficient of h[i] in out[shift]
    # out[shift] = sum_n h[n+shift]*(1-h[n])
    # Terms with h[i]: h[i]*(1-h[i-shift]) + h[i+shift]*(1-h[i])
    #                 = h[i] - h[i]*h[i-shift] + h[i+shift] - h[i]*h[i+shift]
    #                 = h[i]*(1 - h[i-shift] - h[i+shift]) + h[i+shift]
    # Terms with h[j]: h[j]*(1-h[j-shift]) + h[j+shift]*(1-h[j])
    #                 = h[j]*(1 - h[j-shift] - h[j+shift]) + h[j+shift]
    #                 = (t-h[i])*(1 - h[j-shift] - h[j+shift]) + h[j+shift]
    
    # So out[shift] = A_shift * h[i] + B_shift * (t - h[i]) + C_shift
    # where A_shift = 1 - h[i-shift] - h[i+shift] (coeff of h[i] from i's terms)
    #       B_shift = 1 - h[j-shift] - h[j+shift] (coeff of h[j] from j's terms)
    #       C_shift = (terms not involving h[i] or h[j])
    # out[shift] = (A_shift - B_shift) * h[i] + B_shift * t + C_shift
    
    # This is linear in h[i] for each shift!
    # So the objective max_shift dx * out[shift] is max of linear functions → convex in h[i]
    # Minimum is at a breakpoint or boundary.
    
    # Find all breakpoints: where (A_shift - B_shift) * h[i] + const = (A_shift' - B_shift') * h[i] + const'
    # i.e., h[i] = (const' - const) / ((A_shift - B_shift) - (A_shift' - B_shift'))
    
    # This is O(n^2) to find all breakpoints, but n is small.
    # Actually, let's just evaluate at a grid of points and also at breakpoints.
    
    # For efficiency, let's compute all coefficients first
    shifts = np.arange(-(n-1), n)  # all possible shifts
    # For each shift, compute A_shift, B_shift, C_shift
    
    # Actually, let me just sample h[i] at many points and also at 0 and 1
    # Since the function is piecewise linear, the minimum is at a vertex.
    # A practical approach: ternary search or golden section on the convex function.
    
    def obj_at(t_val):
        h_new = h.copy()
        h_new[i] = t_val
        h_new[j] = t - t_val
        return compute_c5(h_new)
    
    # Since it's convex, use golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_at(d)
    
    best_t = (a + b) / 2
    h_new = h.copy()
    h_new[i] = best_t
    h_new[j] = t - best_t
    return h_new[i], h_new[j]

def repair(h, n):
    """Ensure h is feasible: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
            else:
                h += diff / n
        else:
            mask = h > 1e-12
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
            else:
                h += diff / n
    h = np.clip(h, 0, 1)
    # Final fix if still off
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-14:
        # Distribute proportionally
        for i in range(n):
            room = 1.0 - h[i] if diff2 > 0 else h[i]
            if room > 0:
                adjust = min(abs(diff2), room) * (1 if diff2 > 0 else -1)
                h[i] += adjust
                diff2 -= adjust
                if abs(diff2) < 1e-14:

def simulated_annealing(h, time_limit=60.0, seed=42):
    """Simulated annealing with swap moves"""
    rng = np.random.RandomState(seed)
    start = time.time()
    n = len(h)

    current_val = compute_c5(current)
    best = current.copy()
    best_val = current_val

        new_val = compute_c5(current)
