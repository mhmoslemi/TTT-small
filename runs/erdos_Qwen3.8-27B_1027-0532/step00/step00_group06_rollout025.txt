def interpolate_to_n(parent, n):

    x_parent = np.linspace(0, 2, n_parent, endpoint=False)

                    sub = min(-diff, new_h[i])

    def obj(x):
        val, _ = c5_obj_and_grad(x)
        return val

    def grad(x):
        _, g = c5_obj_and_grad(x)
        return g

    try:
        result = minimize(
            obj, h_start, jac=grad, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        h_opt = fix_constraint(result.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h_start, compute_c5(h_start)

def try_softmax_slsqp(h_start, n, s=50.0, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        shifted = s * corr / dx
        shifted -= np.max(shifted)
        logsum = np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted))))
        return (dx / s) * (np.max(shifted) + logsum) / dx * dx

    def grad_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        shifted = s * corr / dx
        shifted_norm = shifted - np.max(shifted)
        weights = np.exp(shifted_norm)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-10:
                continue
            lag = k_idx - (n - 1)
            w = weights[k_idx] * s
            if lag >= 0:
                if lag < n:
                    grad[lag:] += w * (1.0 - x[:n - lag])
                if n - 1 - lag >= 0:
                    grad[:n - lag] -= w * x[lag:]
            else:
                m = -lag
                if m < n:
                    grad[:n - m] += w * (1.0 - x[m:])
                    grad[m:] -= w * x[:n - m]
        return grad

    try:
        result = minimize(
            obj, h_start, jac=grad_fn, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        h_opt = fix_constraint(result.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h_start, compute_c5(h_start)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    parent = np.array(initial_h_values, dtype=float)
    n_parent = len(parent)
    parent_c5 = compute_c5(parent)

    best_h = parent.copy()
    best_c5 = parent_c5
    best_n = n_parent

    grid_sizes = [70, 100, 140]

    for n in grid_sizes:
        if time.time() - start_time > budget_s - 120:
            break

        if n == n_parent:
            h0 = parent.copy()
        else:
            h0 = interpolate_to_n(parent, n)

        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = n

        # Strategy 1: Direct SLSQP
        h1, c5_1 = try_slsqp(h0, n, maxiter=300)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n
            h0 = h1.copy()

        # Strategy 2: Binarize then SLSQP
        if time.time() - start_time < budget_s - 120:
            h_bin = (h0 > 0.5).astype(float)
            h_bin = fix_constraint(h_bin, n)
            h2, c5_2 = try_slsqp(h_bin, n, maxiter=200)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n
                h0 = h2.copy()

        # Strategy 3: Soft-max SLSQP
        if time.time() - start_time < budget_s - 100:
            h3, c5_3 = try_softmax_slsqp(h0, n, s=30.0, maxiter=150)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n
                h0 = h3.copy()

        # Strategy 4: Perturbed restart
        if time.time() - start_time < budget_s - 80:
            h_pert = h0 + 0.05 * rng.randn(n)
            h_pert = fix_constraint(h_pert, n)
            h4, c5_4 = try_slsqp(h_pert, n, maxiter=200)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
                best_n = n

        # Strategy 5: Polishing - run SLSQP again from best
        if time.time() - start_time < budget_s - 60:
            h5, c5_5 = try_slsqp(h0, n, maxiter=200)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()
                best_n = n

        # Update parent
        if best_n == n:
            parent = best_h.copy()
            n_parent = n

    # Final polish on best solution
    if time.time() - start_time < budget_s - 10:
        h_final, c5_final = try_slsqp(best_h, best_n, maxiter=200)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5, best_n
