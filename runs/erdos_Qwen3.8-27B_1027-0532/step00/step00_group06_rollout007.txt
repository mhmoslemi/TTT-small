import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr * dx)

def c5_smooth_value_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = np.sum(exp_bc)
    c5_val = c_max + (1.0 / beta) * np.log(sum_exp)
    w = exp_bc / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k] * dx
        if abs(wk) < 1e-12:
            continue
        i1s = max(0, lag)
        i1e = min(n - 1, lag + n - 1)
        if i1s <= i1e:
            idx = np.arange(i1s, i1e + 1)
            grad[idx] += wk * (1.0 - h[idx - lag])
        i2s = max(0, -lag)
        i2e = min(n - 1, n - 1 - lag)
        if i2s <= i2e:
            idx = np.arange(i2s, i2e + 1)
            grad[idx] -= wk * h[idx + lag]
    return c5_val, grad

def project_feasible(y, target_sum):
    y = np.asarray(y, dtype=float)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def resample_to_n(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def pgd_optimize(h, beta, n_iter=1000, time_limit=None, start_time=None):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for i in range(n_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        val, grad = c5_smooth_value_and_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        alpha = 1.0 / (beta * gnorm + 1e-12)
        improved = False
        for _ in range(40):
            h_trial = project_feasible(h - alpha * grad, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < val - 1e-15:
                h = h_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return best_h, best_c5

def slsqp_optimize(h, beta, maxiter=150):
    n = len(h)
    target_sum = n / 2.0
    def obj(x):
        return c5_smooth_value_and_grad(x, beta)[0]
    def obj_grad(x):
        return c5_smooth_value_and_grad(x, beta)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_feasible(res.x, target_sum)
        return h_out, compute_c5(h_out)
    except Exception:
        return h, compute_c5(h)
