import time

def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def project_feasible(v, target_sum, n):

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def smooth_obj_grad(h, dx, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    nc = len(corr)
    mx = corr.max()
    shifted = np.minimum(beta * (corr - mx), 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    val = mx + np.log(exp_s.sum()) / beta
    grad = np.zeros(n)
    for m in range(nc):
        wm = w[m]

        lo1 = max(0, s)

        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 + s:hi2 + s]
    grad *= dx

def subgrad_topk(h, dx, n, k=5):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for m in top_idx:
        s = m - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    grad *= dx / float(k)
