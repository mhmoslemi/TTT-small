def project_onto_constraint(h, n, target_sum):

    # Solution: x[i] = clip(h[i] - lambda, 0, 1) for appropriate lambda
    
    lo, hi = -2, 2
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)

One key insight: if h is 0-1 valued (a binary step function), then h(x)(1-h(x+k)) is 1 only where h(x)=1 and h(x+k)=0. So the overlap integral counts the measure of the set where h is 1 at position x but 0 at position x+k.

For a binary h with ∫h = 1 on [0,2], h is 1 on a set of measure 1 and 0 on a set of measure 1.

The overlap for shift k is the measure of {x : h(x)=1, h(x+k)=0}.

        # i-lag in [0, n-1] → i in [lag, lag+n-1], clipped to [0, n-1]
        i_start = max(0, lag)
        i_end = min(n, lag + n)  # exclusive
        if i_start < i_end:
            j_start = 0
            j_end = i_end - lag
            grad[i_start:i_end] += pk * (1 - h[j_start:j_end])
        
        # Second term: -h[i+lag] for valid i
        # i+lag in [0, n-1] → i in [-lag, n-1-lag], clipped to [0, n-1]
        i_start2 = max(0, -lag)
        i_end2 = min(n, n - lag)  # exclusive
        if i_start2 < i_end2:
            j_start2 = lag  # i_start2 + lag
            j_end2 = i_end2 + lag
            grad[i_start2:i_end2] -= pk * h[j_start2:j_end2]
    
    return grad * dx

        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gradient_soft_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(len(corr)):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        i_start = max(0, lag)
        i_end = min(n, lag + n)
        if i_start < i_end:
            h_s = i_start - lag
            h_e = i_end - lag
            grad[i_start:i_end] += pk * (1.0 - h[h_s:h_e])
        i_s2 = max(0, -lag)
        i_e2 = min(n, n - lag)
        if i_s2 < i_e2:
            h_s2 = i_s2 + lag
            h_e2 = i_e2 + lag
            grad[i_s2:i_e2] -= pk * h[h_s2:h_e2]
    return grad * dx

def projected_gd(h0, alpha, lr, max_iters, time_limit, momentum_decay=0.9, patience=400):
    n = len(h0)
    target_sum = n / 2.0
    h = project_onto_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    momentum = np.zeros(n)
    last_imp = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if it > 200 and (it - last_imp) > patience:
            break
        grad = gradient_soft_max(h, alpha)
        momentum = momentum_decay * momentum - lr * grad
        h_new = h + momentum
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_imp = it
        h = h_new
    return best_h, best_c5

def penalty_obj(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    sm = (1.0 / alpha) * (max_s + np.log(np.sum(np.exp(scaled - max_s))))
    return sm + penalty * (np.sum(h) - target_sum) ** 2

def penalty_grad(h, alpha, penalty):
    n = len(h)
    target_sum = n / 2.0
    g = gradient_soft_max(h, alpha)
    g += 2.0 * penalty * (np.sum(h) - target_sum) * np.ones(n)
    return g

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 95
        h0 = np.ones(n0) * 0.5
    n = len(h0)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def remaining_time():
        return max(0.0, budget_s - 8.0 - elapsed())

    # Phase 1: L-BFGS-B with penalty (medium alpha)
    tl = remaining_time() * 0.20
    if tl > 30:
        res = minimize(
            penalty_obj, h0.copy(),
            args=(30.0, 50.0),
            jac=penalty_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 300, 'maxfun': 10000, 'ftol': 1e-12}
        )
        if res.x is not None:
            hc = np.clip(res.x, 0.0, 1.0)
            hc = project_onto_constraint(hc, target_sum)
            c5c = compute_c5(hc)
            if c5c < best_c5:
                best_h = hc
                best_c5 = c5c

    # Phase 2: Projected gradient descent with increasing alpha
    schedules = [
        (15.0, 0.010, 0.15),
        (40.0, 0.005, 0.12),
        (100.0, 0.003, 0.10),
        (250.0, 0.0015, 0.10),
        (600.0, 0.0008, 0.08),
        (1500.0, 0.0004, 0.08),
    ]
    for alpha, lr, frac in schedules:
        tl = remaining_time() * frac
        if tl < 2:
            break
        hb, c5b = projected_gd(best_h, alpha, lr, 200000, tl)
        if c5b < best_c5:
            best_h = hb
            best_c5 = c5b

    # Phase 3: Random restarts
    tl_total = remaining_time()
    if tl_total > 20:
        n_restarts = max(3, min(int(tl_total / 15), 15))
        tl_per = tl_total / n_restarts
        for r in range(n_restarts):
            if remaining_time() < 5:
                break
            h_start = best_h.copy()
            strat = r % 5
            if strat == 0:
                h_start += rng.randn(n) * 0.02
            elif strat == 1:
                mask = rng.rand(n) < 0.1
                h_start[mask] = 1.0 - h_start[mask]
            elif strat == 2:
                h_start += rng.randn(n) * 0.05
            elif strat == 3:
                bs = max(1, n // 12)
                for _ in range(4):
                    pos = rng.randint(0, n - bs)
                    h_start[pos:pos + bs] = 1.0 - h_start[pos:pos + bs]
            else:
                h_start += rng.randn(n) * 0.01
            h_start = project_onto_constraint(h_start, target_sum)
            alpha_r = [50.0, 150.0, 300.0, 80.0, 200.0][strat]
            lr_r = [0.003, 0.001, 0.0005, 0.002, 0.0008][strat]
            hb, c5b = projected_gd(h_start, alpha_r, lr_r, 80000, tl_per)
            if c5b < best_c5:
                best_h = hb
                best_c5 = c5b

    # Phase 4: Final refinement with very sharp soft-max
    tl = remaining_time()
    if tl > 5:
        hb, c5b = projected_gd(best_h, 2000.0, 0.0002, 100000, tl, patience=200)
        if c5b < best_c5:
            best_h = hb
            best_c5 = c5b

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
