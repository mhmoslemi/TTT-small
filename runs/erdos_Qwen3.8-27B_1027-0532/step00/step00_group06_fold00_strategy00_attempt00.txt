   def restore_integral(h, target_sum, scale=True):
       """Restore the sum constraint using minimal nearest scaling."""
       current_sum = np.sum(h)
       if abs(current_sum - target_sum) < 1e-5:
           return h.copy()
       factor = target_sum / current_sum
       h_scaled = h * factor
       if scale:
           low_clip = (1 - 1e-7)
           high_clip = (1 + 1e-7)
           # Bound to [0,1]
           h_scaled = np.clip(h_scaled, low_clip, high_clip)
           # Renormalize to target sum
           extra_sum = np.sum(h_scaled) - target_sum
           if extra_sum != 0:
               h_scaled *= (target_sum - extra_sum) / np.sum(h_scaled)
       return h_scaled.astype(np.float32)
