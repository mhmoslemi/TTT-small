    diff = target_sum - np.sum(h)

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max(x, t):
    x_scale = x / t
    mx = np.max(x_scale)
    if mx > 500:
        return mx * t
    return t * (mx + np.log(np.sum(np.exp(x_scale - mx))))

def project_feasible(h, target_sum):

    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

    return smooth_max(c, temp)
