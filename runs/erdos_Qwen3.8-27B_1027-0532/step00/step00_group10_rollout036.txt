    # For shift k, slope = (1-h[j-k])*1_{0<=j-k<n} - h[j+k]*1_{0<=j+k<n}

    k_min = max(-(n-1), j - (n-1))

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_slopes(h, j, n):
    slopes = np.zeros(2 * n - 1)
    k_min = max(-(n - 1), j - (n - 1))
    k_max = min(n - 1, j)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max + 1)
        idx1 = j - ks
        slopes[ks + (n - 1)] += (1.0 - h[idx1])
    k_min2 = max(-(n - 1), -j)
    k_max2 = min(n - 1, n - 1 - j)
    if k_min2 <= k_max2:
        ks2 = np.arange(k_min2, k_max2 + 1)
        idx2 = j + ks2
        slopes[ks2 + (n - 1)] -= h[idx2]
    return slopes

def optimize_single_coordinate(h, j, c_vals, n):
    slopes = get_slopes(h, j, n)
    v_grid = np.linspace(0.0, 1.0, 101)
    offsets = v_grid - h[j]
    new_c = c_vals[None, :] + slopes[None, :] * offsets[:, None]
    obj = np.max(new_c, axis=1)
    best_idx = np.argmin(obj)
    if best_idx == 0:
        return 0.0
    if best_idx == len(v_grid) - 1:
        return 1.0
    lo = v_grid[best_idx - 1]
    hi = v_grid[best_idx + 1]
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c_vals + slopes * (m1 - h[j]))
        f2 = np.max(c_vals + slopes * (m2 - h[j]))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def cd_pass(h, n, target_sum, n_rounds=3):
    for rnd in range(n_rounds):
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            c_vals = compute_c_vals(h)
            new_v = optimize_single_coordinate(h, j, c_vals, n)
            h[j] = new_v
        h = project_feasible(h, target_sum)
    return h

def subgradient_top_k(h, c_vals, K, n):
    if K >= len(c_vals):
        K = len(c_vals)
    top_idx = np.argpartition(c_vals, -K)[-K:]
    grad = np.zeros(n)
    for i in top_idx:
        k = i - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
        else:
            m = -k
            if m < n:
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] -= h[:n - m]
    grad /= float(K)
    grad -= np.mean(grad)
    return grad

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0

    h = initial_h_values.astype(np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, target_sum)

    best_h = h.copy()
    c_vals = compute_c_vals(h)
    best_c5 = np.max(c_vals) * dx

    def elapsed():
        return time.time() - start_time

    # Phase 1: Coordinate descent for quick local convergence
    cd_end = budget_s * 0.25
    while elapsed() < cd_end:
        h = cd_pass(h, n, target_sum, n_rounds=2)
        c_vals = compute_c_vals(h)
        c5 = np.max(c_vals) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Subgradient descent with momentum and restarts
    alpha = 0.02
    mu = 0.8
    momentum = np.zeros(n)
    no_improve = 0
    patience = 3000

    phase2_end = budget_s * 0.75
    while elapsed() < phase2_end:
        c_vals = compute_c_vals(h)
        c5 = np.max(c_vals) * dx

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            restart_mode = no_improve % 3
            if restart_mode == 0:
                h = project_feasible(best_h + 0.08 * np.random.randn(n), target_sum)
            elif restart_mode == 1:
                h = project_feasible(best_h + 0.15 * np.random.randn(n), target_sum)
            else:
                alt = np.zeros(n)
                alt[0::2] = 1.0
                alt[1::2] = 0.0
                h = project_feasible(alt + 0.3 * np.random.randn(n), target_sum)
            momentum = np.zeros(n)
            alpha = 0.02
            no_improve = 0
            continue

        K = 8
        grad = subgradient_top_k(h, c_vals, K, n)
        gn = np.linalg.norm(grad)

        if gn < 1e-12:
            if no_improve > 100:
                h = project_feasible(best_h + 0.03 * np.random.randn(n), target_sum)
                no_improve = 0
            continue

        momentum = mu * momentum - alpha * grad / gn
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        c_vals_new = compute_c_vals(h_new)
        c5_new = np.max(c_vals_new) * dx

        if c5_new <= c5 + 1e-10:
            h = h_new
            alpha = min(alpha * 1.005, 0.1)
        else:
            momentum *= 0.5
            alpha *= 0.8
            if alpha < 1e-5:
                h = project_feasible(h + 0.02 * np.random.randn(n), target_sum)
                momentum = np.zeros(n)
                alpha = 0.01

        if no_improve % 2000 == 0 and no_improve > 0:
            h = cd_pass(h, n, target_sum, n_rounds=2)
            c_vals = compute_c_vals(h)
            c5 = np.max(c_vals) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0

    # Phase 3: Multiple CD refinement passes from best
    h = best_h.copy()
    phase3_end = budget_s * 0.92
    while elapsed() < phase3_end:
        h = cd_pass(h, n, target_sum, n_rounds=3)
        c_vals = compute_c_vals(h)
        c5 = np.max(c_vals) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Try diverse restarts with CD
    diverse_end = budget_s * 0.97
    n_attempts = 0
    while elapsed() < diverse_end and n_attempts < 50:
        mode = n_attempts % 4
        if mode == 0:
            h = project_feasible(best_h + 0.05 * np.random.randn(n), target_sum)
        elif mode == 1:
            h = project_feasible(best_h + 0.1 * np.random.randn(n), target_sum)
        elif mode == 2:
            h = project_feasible(np.random.rand(n), target_sum)
        else:
            h = best_h.copy()
            idx = np.random.choice(n, size=n // 3, replace=False)
            h[idx] = np.clip(h[idx] + 0.3 * np.random.randn(len(idx)), 0, 1)
            h = project_feasible(h, target_sum)

        h = cd_pass(h, n, target_sum, n_rounds=3)
        c_vals = compute_c_vals(h)
        c5 = np.max(c_vals) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        n_attempts += 1

    final_c_vals = compute_c_vals(best_h)
    final_c5 = np.max(final_c_vals) * dx
    return best_h, final_c5, n
