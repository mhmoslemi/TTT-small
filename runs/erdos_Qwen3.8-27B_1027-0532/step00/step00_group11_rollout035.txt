        p = _proj(best_h + rng.randn(n) * sc)
        try:
            res = minimize(_obj, p, method='COBYLA',
                          options={'maxiter': 800, 'rhobnd': 1e-7},
                          constraints=[{'type': 'eq', 'fun': _ceq}],
                          tol=1e-11)
            hc = _proj(res.x)
            v = _obj(hc)
            if v < best_v:
                best_v = v
                best_h = hc.copy()
        except Exception:
            pass

    if time.time() - t0 < budget_s - 20:
        hc = best_h.copy()
        deltas = (0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2)
        for it in range(50000):
            if time.time() - t0 > budget_s - 10:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            IMPROVED = False
            for d in deltas:
                for sgn in (1.0, -1.0):
                    hi_ = hc[i] + sgn * d
                    hj_ = hc[j] - sgn * d
                    if 0.0 <= hi_ <= 1.0 and 0.0 <= hj_ <= 1.0:
                        hn = hc.copy()
                        hn[i] = hi_
                        hn[j] = hj_
                        vn = _obj(hn)
                        if vn < best_v - 1e-14:
                            best_v = vn
                            best_h = hn.copy()
                            hc = hn.copy()
                            IMPROVED = True
                            break
                if IMPROVED:
                    break

    return best_h, best_v, n
