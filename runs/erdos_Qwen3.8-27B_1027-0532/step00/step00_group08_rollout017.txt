import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, dx, t):
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    if m < 1e-300:
        return t * np.log(np.sum(np.exp(corr / t))) * dx
    shifted = corr / t - m / t
    return (m + t * np.log(np.sum(np.exp(shifted)))) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy 0<=h[i]<=1 and sum(h)=target_sum"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    # Distribute the difference
    diff = target_sum - s
    n = len(h)
    if diff > 0:
        # Need to add mass: add to elements < 1
        mask = h < 1.0
        if np.sum(mask) == 0:
            return h
        per = diff / np.sum(mask)
        h[mask] = np.minimum(h[mask] + per, 1.0)
        # If still short, do it again
        s = np.sum(h)
        if s < target_sum:
            diff2 = target_sum - s
            mask2 = h < 1.0
            if np.sum(mask2) > 0:
                per2 = diff2 / np.sum(mask2)
                h[mask2] = np.minimum(h[mask2] + per2, 1.0)
    else:
        # Need to remove mass: remove from elements > 0
        mask = h > 0.0
        if np.sum(mask) == 0:
            return h
        per = diff / np.sum(mask)
        h[mask] = np.maximum(h[mask] + per, 0.0)
        s = np.sum(h)
        if s > target_sum:
            diff2 = target_sum - s
            mask2 = h > 0.0
            if np.sum(mask2) > 0:
                per2 = diff2 / np.sum(mask2)
                h[mask2] = np.maximum(h[mask2] + per2, 0.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h0 = np.zeros(n)
        h0[::2] = 1.0  # 40 ones alternating
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0  # = 40
    
    # Ensure initial is feasible
    h0 = project_to_feasible(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Phase 1: SLSQP with decreasing temperature (continuation method)
    current_h = h0.copy()
    temperatures = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    
    for temp in temperatures:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 60:
            break
        
        def obj(x):
            return smooth_max_c5(x, dx, temp)
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n
        
        try:
            result = minimize(obj, current_h, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 500, 'ftol': 1e-14})
            new_h = project_to_feasible(result.x, target_sum)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            current_h = new_h
        except Exception:
            pass
    
    # Phase 2: Try random perturbations and local refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 30:
        for attempt in range(200):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 15:
                break
            
            # Random perturbation
            h_perturbed = best_h + np.random.randn(n) * 0.05
            h_perturbed = project_to_feasible(h_perturbed, target_sum)
            
            # Local optimization with very small temperature
            def obj2(x):
                return smooth_max_c5(x, dx, 0.001)
            
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0, 1)] * n
            
            try:
                result = minimize(obj2, h_perturbed, method='SLSQP',
                                bounds=bounds, constraints=constraints,
                                options={'maxiter': 200, 'ftol': 1e-14})
                new_h = project_to_feasible(result.x, target_sum)
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
            except Exception:
                pass
    
    # Phase 3: Try to push further with COBYLA or other methods
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        def obj_direct(x):
            return compute_c5(x, dx)
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n
        
        try:
            result = minimize(obj_direct, best_h, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 500, 'ftol': 1e-15})
            new_h = project_to_feasible(result.x, target_sum)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        except Exception:
            pass
    
    return (best_h, best_c5, n)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < 1e-300:
        return 0.0
    shifted = corr - m
    return (m + t * np.log(np.sum(np.exp(shifted / t)))) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    diff = target_sum - s
    n = len(h)
    # Binary search for the right water level
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_adj = np.clip(h + mid, 0.0, 1.0)
        s_adj = np.sum(h_adj)
        if s_adj < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    return h
