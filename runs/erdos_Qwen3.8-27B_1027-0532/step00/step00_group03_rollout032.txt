import numpy as np
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def lag_gradient(h, m, n):
    grad = np.zeros(n)
    if m >= 0:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    else:
        mp = -m
        grad[:n - mp] += (1.0 - h[mp:])
        grad[mp:] -= h[:n - mp]
    return grad

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def project_feasible_exact(h, target_sum):
    h = project_feasible(h, target_sum)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
        residual = target_sum - np.sum(h)
        if abs(residual) > 1e-12:
            idx = np.argmax((h > 1e-12) & (h < 1.0 - 1e-12))
            if h[idx] > 0 and h[idx] < 1:
                h[idx] += residual
    return h
