    corr = compute_correlation(h, n)
    # log-sum-exp for numerical stability
    max_corr = np.max(corr)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1-h, mode='full')) * 2.0 / n)

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on lambda"""
    target = n / 2.0
    lo = h.min() - target - 1
    hi = h.max() + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smoothed_c5(h, n, T):
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    return (max_c + T * np.log(np.sum(np.exp((corr - max_c) / T))) / T) * (2.0 / n)

def smoothed_c5_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / T)
    p = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    # For each lag index k (0 to 2n-2), d = k - (n-1)
    for k in range(2 * n - 1):
        if p[k] < 1e-8:
            continue
        d = k - (n - 1)
        # Term 1: (1-h[j+d]) for j where 0 <= j+d < n
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad[js:je] += p[k] * (1 - h[js+d:je+d])
        # Term 2: -h[j-d] for j where 0 <= j-d < n
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad[js2:je2] -= p[k] * h[js2-d:je2-d]
    return grad * dx

def binarize(h, n):
    """Project to nearest binary vector with sum = n/2"""
    # Sort indices by h value
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1
    return h_bin

def binary_flip_search(h, n, max_iters=5000, time_limit=60):
    """Local search: flip pairs (1->0, 0->1) to reduce C5"""
    h = h.copy()
    c5 = evaluate_c5(h)
    start = time.time()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    improved = True
    it = 0
    while improved and it < max_iters and (time.time() - start) < time_limit:
        improved = False
        it += 1
        best_c5 = c5
        best_pair = None
        # Shuffle for diversity
        rng = np.random.RandomState(it)
        ones_shuffled = ones[:]
        zeros_shuffled = zeros[:]
        rng.shuffle(ones_shuffled)
        rng.shuffle(zeros_shuffled)
        
        for i in ones_shuffled:
            h[i] = 0
            for j in zeros_shuffled:
                h[j] = 1
                c5_test = evaluate_c5(h)
                h[j] = 0
                if c5_test < best_c5 - 1e-12:
                    best_c5 = c5_test
                    best_pair = (i, j)
            h[i] = 1
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            c5 = best_c5
            improved = True
    
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - target - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_c5(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return (max_c + T * np.log(np.sum(np.exp((corr - max_c) / T))) / T) * (2.0 / n)

def smoothed_c5_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / T)
    p = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-8:
            continue
        d = k - (n - 1)
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad[js:je] += p[k] * (1.0 - h[js + d:je + d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad[js2:je2] -= p[k] * h[js2 - d:je2 - d]
    return grad * dx

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def binary_flip_search(h, n, max_iters=20000, time_limit=120):
    h = h.copy()
    c5 = evaluate_c5(h)
    start = time.time()
    ones = list(np.where(h == 1.0)[0].tolist())
    zeros = list(np.where(h == 0.0)[0].tolist())
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        it += 1
        best_c5 = c5
        best_pair = None
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c5_test = evaluate_c5(h)
                h[j] = 0.0
                if c5_test < best_c5 - 1e-12:
                    best_c5 = c5_test
                    best_pair = (i, j)
            h[i] = 1.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        zeros.append(i)
        zeros.remove(j)
        ones.append(j)
        c5 = best_c5
    return h, c5

def binary_sa_search(h, n, time_limit=60, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5 = evaluate_c5(h)
    h_best = h.copy()
    c5_best = c5
    start = time.time()
    T_start = 0.5
    T_end = 0.005
    ones = list(np.where(h == 1.0)[0].tolist())
    zeros = list(np.where(h == 0.0)[0].tolist())
    while (time.time() - start) < time_limit:
        i = int(rng.choice(len(ones)))
        j = int(rng.choice(len(zeros)))
        ii = ones[i]
        jj = zeros[j]
        h[ii] = 0.0
        h[jj] = 1.0
        c5_new = evaluate_c5(h)
        delta = c5_new - c5
        elapsed_frac = min(1.0, (time.time() - start) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = c5_new
            ones.remove(ii)
            ones.append(jj)
            zeros.remove(jj)
            zeros.append(ii)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
        else:
            h[ii] = 1.0
            h[jj] = 0.0
    return h_best, c5_best

def continuous_optimize(h, n, T_list, max_iter=300):
    h_curr = h.copy()
    for T in T_list:
        try:
            res = minimize(
                fun=lambda x: smoothed_c5(x, n, T),
                x0=h_curr.copy(),
                jac=lambda x: smoothed_c5_grad(x, n, T),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': max_iter, 'ftol': 1e-12}
            )
            if res.x is not None:
                h_candidate = project_feasible(res.x, n)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < evaluate_c5(h_curr):
                    h_curr = h_candidate
        except Exception:
            pass
    return h_curr

def try_higher_resolution(h, n, target_n, time_limit=120, seed=42):
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_interp = np.interp(x_new, x_old, h, period=2.0)
    h_interp = project_feasible(h_interp, target_n)
    h_opt = continuous_optimize(h_interp, target_n, [0.1, 0.05, 0.02], max_iter=200)
    h_bin = binarize(h_opt, target_n)
    h_bin, c5_bin = binary_flip_search(h_bin, target_n, max_iters=5000,
                                        time_limit=min(time_limit, 90))
    return h_bin, c5_bin
