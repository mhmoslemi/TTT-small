                if best_h[i] > 0.7 and best_h[j] > 0.7:
                    continue
                h_try = best_h.copy()
                h_try[i] = 1.0 - best_h[i]
                h_try[j] = 1.0 - best_h[j]
                if abs(np.sum(h_try) - n / 2.0) > 1e-6:
                    h_try[i] = best_h[i] + (best_h[j] - best_h[i])
                    h_try[j] = best_h[j] - (best_h[j] - best_h[i])
                    h_try = np.clip(h_try, 0, 1)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True

    n_values = [44, 46, 48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 78, 80, 84, 88, 92, 96, 100, 104, 108, 112, 116, 120, 128, 136, 144, 152, 160, 168, 176, 184, 192, 200]

    deadline_phase2 = start_time + budget_s * 0.90
    time_remaining = max(deadline_phase2 - time.time(), 1.0)
    time_per_n = time_remaining / len(n_values)

    seed_counter = seed + 1000
    for n in n_values:
        if time.time() > deadline_phase2:
            break
        n_deadline = time.time() + max(time_per_n, 1.5)
        n_restarts = max(2, int(time_per_n / 2.5))
        iter_per_restart = max(2000, int(2.0 * 500000))

        for restart in range(n_restarts):
            if time.time() > n_deadline:
                break
            seed_counter += 1
            rng = np.random.RandomState(seed_counter)
            if restart == 0:
                h_cur = np.zeros(n)
                h_cur[:n // 2] = 1.0
            elif restart == 1:
                h_cur = np.zeros(n)
                idx = rng.choice(n, n // 2, replace=False)
                h_cur[idx] = 1.0
            elif restart == 2:
                phi = (1.0 + np.sqrt(5.0)) / 2.0
                h_cur = np.zeros(n)
                fracs = (np.arange(n) * phi) % 1.0
                h_cur[fracs < 0.5] = 1.0
                if np.sum(h_cur) > n // 2:
                    excess = int(np.sum(h_cur)) - n // 2
                    ones = np.where(h_cur > 0.5)[0]
                    for e in range(excess):
                        h_cur[ones[(e * 7 + 3) % len(ones)]] = 0.0
                elif np.sum(h_cur) < n // 2:
                    deficit = n // 2 - int(np.sum(h_cur))
                    zeros = np.where(h_cur < 0.5)[0]
                    for e in range(deficit):
                        h_cur[zeros[(e * 11 + 5) % len(zeros)]] = 1.0
            else:
                h_cur = np.zeros(n)
                h_cur[::2] = 1.0
                for _ in range(n // 3):
                    ones = np.where(h_cur > 0.5)[0]
                    zeros = np.where(h_cur < 0.5)[0]
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    h_cur[ii], h_cur[jj] = h_cur[jj], h_cur[ii]

            current_c5 = evaluate_c5(h_cur)
            best_local = h_cur.copy()
            best_local_c5 = current_c5

            T = 0.04
            T_min = 1e-6
            cooling = (T_min / T) ** (1.0 / iter_per_restart)

            for it in range(iter_per_restart):
                if time.time() > n_deadline:
                    break
                ones = np.where(h_cur > 0.5)[0]
                zeros = np.where(h_cur < 0.5)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                new_c5 = evaluate_c5(h_cur)
                delta = new_c5 - current_c5
                if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
                    current_c5 = new_c5
                    if new_c5 < best_local_c5:
                        best_local_c5 = new_c5
                        best_local = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                T *= cooling

            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = best_local.copy()

    deadline_phase3 = start_time + budget_s * 0.97
    if time.time() < deadline_phase3:
        h = best_h.copy()
        n = len(h)
        current_c5 = best_c5
        T = 0.015
        T_min = 1e-7
        cooling = (T_min / T) ** (1.0 / 50000.0)
        rng = np.random.RandomState(seed + 99999)
        for it in range(80000):
            if time.time() > deadline_phase3:
                break
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = evaluate_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
            T *= cooling
            if T < T_min:
                T = 0.01
                cooling = (T_min / T) ** (1.0 / 30000.0)

    return (best_h, best_c5, len(best_h))
