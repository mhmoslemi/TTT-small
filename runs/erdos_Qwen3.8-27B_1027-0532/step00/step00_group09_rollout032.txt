from scipy.optimize import minimize

        m = np.max(corr)
        if m < 1e-12:
            return 0.0
        shifted = corr - m
        return m + np.log(np.sum(np.exp(self.beta * shifted))) / self.beta

    def __init__(self, target):
        self.target = float(target)
    def __call__(self, h):
        return np.sum(h) - self.target

def compute_c5(h, dx):

    return float(np.max(corr) * dx)

    """Project y onto {0 <= h_i <= 1, sum(h) = target_sum}"""
    # Binary search on lambda
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

        m = np.max(corr)
        if m < 1e-12:
            return 0.0
        shifted = corr - m
        return m + np.log(np.sum(np.exp(self.beta * shifted))) / self.beta

    def __init__(self, target):
        self.target = float(target)

    def __call__(self, h):
        return np.sum(h) - self.target


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_onto_feasible(y, target_sum):

    current = h.copy()
    current_c5 = compute_c5(current, dx)

        improved = False
        for i in range(n):

                for delta in [0.05, 0.1, 0.25, 0.5, 0.01, 0.02]:

                        trial[j] -= delta
