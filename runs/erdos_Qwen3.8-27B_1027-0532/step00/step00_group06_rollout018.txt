        idx = m_vals + N - 1
        coef[idx] += (-h[i + m_vals])

    h = h.copy()

        if time.time() - start_time > time_limit:
            break

        order = np.random.permutation(N) if p > 0 else np.arange(N)

            if old_v == 0.0 or old_v == 1.0:

            best_v = old_v

        h = project_to_feasible(h, target_sum)

def _project(h, target):
