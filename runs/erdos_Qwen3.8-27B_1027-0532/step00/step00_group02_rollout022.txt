from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, n_points, alpha):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / alpha

def repair(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-12:
            h = h + (diff / total_room) * room
    else:
        total_h = np.sum(h)
        if total_h > 1e-12:
            h = h + (diff / total_h) * h
    h = np.clip(h, 0.0, 1.0)
    # Fix any remaining discrepancy
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Sequential adjustment
        remaining = target - s
        step = remaining / n_points
        h = h + step
        h = np.clip(h, 0.0, 1.0)
    return h

    h = repair(x, n_points)
    return smooth_max_c5(h, n_points, alpha)

    """Hill climbing: flip one 1 to 0 and one 0 to 1."""
    n = n_points
    curr_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = curr_c5
    T = 0.01  # temperature for simulated annealing
    for epoch in range(max_epochs):
        h = best_h.copy()
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        h[ones[i]] = 1.0 - h[ones[i]]
        h[zeros[j]] = 1.0 - h[zeros[j]]
        new_c5 = compute_c5(h, n)
        if new_c5 < best_c5 or rng.random() < np.exp((best_c5 - new_c5) / T):

            best_c5 = new_c5

        room = 1.0 - h
        total_room = np.sum(room)

The variable vector would be [h[0], h[1], ..., h[n-1], t] of length n+1.

The number of inequality constraints is 2n-1 (one for each correlation lag), which is a lot but manageable.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, n_points, alpha):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / alpha

def repair(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-12:
            h = h + (diff / total_room) * room
        else:
            h = h + diff / n_points
    else:
        total_h = np.sum(h)
        if total_h > 1e-12:
            h = h + (diff / total_h) * h
        else:
            h = h + diff / n_points
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h = h + (target - s) / n_points
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x, n_points, alpha):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_c5(h, n_points, alpha)

def eq_con(x, n_points):
    return np.sum(x) - n_points / 2.0

def minimax_obj(x, n_points):
    return x[-1]

def minimax_ineq(x, n_points):
    h = x[:n_points]
    t = x[-1]
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr * dx

def minimax_eq(x, n_points):
    return np.sum(x[:n_points]) - n_points / 2.0

def sa_binary(h_initial, n_points, rng, n_iter, T_init, T_decay):
    n = n_points
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, n)
    T = T_init
    for _ in range(n_iter):
        h = best_h.copy()
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h[i] = 1.0
        h[j] = 0.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            best_h = h.copy()
            best_c5 = new_c5
        T *= T_decay
    return best_h, best_c5

def slsqp_smooth(h_start, n_points, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h_start
    result = minimize(obj_smooth, h_start, method='SLSQP',
                     args=(n_points, float(alpha)),
                     bounds=[(0.0, 1.0)] * n_points,
                     constraints=[{'type': 'eq', 'fun': eq_con, 'args': (n_points,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-12})
    return repair(result.x, n_points)

def slsqp_minimax(h_start, n_points, maxiter, deadline):
    if time.time() > deadline:
        return h_start
    c5_init = compute_c5(h_start, n_points)
    x0 = np.concatenate([h_start, [c5_init + 0.01]])
    cons = [
        {'type': 'ineq', 'fun': minimax_ineq, 'args': (n_points,)},
        {'type': 'eq', 'fun': minimax_eq, 'args': (n_points,)}
    ]
    bounds = [(0.0, 1.0)] * n_points + [(0.0, 1.0)]
    result = minimize(minimax_obj, x0, method='SLSQP',
                     args=(n_points,),
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
    if result.success or np.max(result.x[:n_points]) <= 1.0 + 1e-6:
        return repair(result.x[:n_points], n_points)
    return h_start
