def project_onto_feasible(y, S):
    n = len(y)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    return float(np.max(corr)) * dx, corr

def subgradient_of_active(h, j_star, n):

    g = np.zeros(n)
    if m >= 0:
        # c[j] = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
        if m == 0:

            t_range2 = np.arange(0, n - m)
            g[t_range2] += -h[t_range2 + m]

import numpy as np
import time

def _project(y, S):
    n = len(y)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

def _eval(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _grad_active(h, n, j_star):
    m = j_star - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        t1 = np.arange(m, n)
        g[t1] += (1.0 - h[t1 - m])
        t2 = np.arange(0, n - m)
        g[t2] += -h[t2 + m]
    else:
        p = -m
        t1 = np.arange(0, n - p)
        g[t1] += (1.0 - h[t1 + p])
        t2 = np.arange(p, n)
        g[t2] += -h[t2 - p]
    return g

def _subgrad_iter(h, n, S, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    j_star = int(np.argmax(corr))
    g = _grad_active(h, n, j_star)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h, _eval(h)
    step = alpha * g / gn
    h_new = _project(h - step, S)
    return h_new, _eval(h_new)

def _local_search(h, n, S, n_iters, rng, time_limit):
    best = h.copy()
    best_val = _eval(best)
    for it in range(n_iters):
        if it % 500 == 0 and time.time() > time_limit:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        delta = rng.random() * 0.05 + 0.001
        if h[i] > delta and h[j] < 1.0 - delta:
            h[i] -= delta
            h[j] += delta
            v = _eval(h)
            if v < best_val:
                best_val = v
                best = h.copy()
            else:
                h[i] += delta
                h[j] -= delta
    return best, best_val

def _pair_improve(h, i, j, n, S):
    """Try to improve by adjusting h[i] and h[j] (maintaining sum)."""
    # d_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        # ∂c_k/∂h[i]
        if m >= 0:
            di = (1.0 - h[i - m]) * (1 if m <= i <= n - 1 else 0) - h[i + m] * (1 if 0 <= i <= n - 1 - m else 0)
        else:
            p = -m
            di = (1.0 - h[i + p]) * (1 if 0 <= i <= n - 1 - p else 0) - h[i - p] * (1 if p <= i <= n - 1 else 0)
        # ∂c_k/∂h[j]
        if m >= 0:
            dj = (1.0 - h[j - m]) * (1 if m <= j <= n - 1 else 0) - h[j + m] * (1 if 0 <= j <= n - 1 - m else 0)
        else:
            p = -m
            dj = (1.0 - h[j + p]) * (1 if 0 <= j <= n - 1 - p else 0) - h[j - p] * (1 if p <= j <= n - 1 else 0)
        d[k] = di - dj
    
    # Find optimal δ
    # Feasible range: max(0, h[i]-1) ≤ -δ ≤ min(1, ... ) 
    # h[i] + δ ∈ [0,1] => δ ∈ [-h[i], 1-h[i]]
    # h[j] - δ ∈ [0,1] => δ ∈ [h[j]-1, h[j]]
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])

    # This is convex in δ. Find minimum.

    for _ in range(50):

        g2 = np.max(corr + m2 * d)
        if g1 < g2:

    h_new[i] += delta_opt
    h_new[j] -= delta_opt
