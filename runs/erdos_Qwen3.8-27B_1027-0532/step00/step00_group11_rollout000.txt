from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(x - tau, 0.0, 1.0)

def c5_objective(h, dx):
    return evaluate_c5(h, dx)

def smooth_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (dx / alpha) * (alpha * mc + np.log(np.sum(np.exp(alpha * (corr - mc)))))

def smooth_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    p = np.exp(alpha * (corr - mc))
    p /= np.sum(p)
    
    d = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        mask_a = (j + d >= 0) & (j + d < n)
        A = np.dot(p[mask_a], 1.0 - h[j + d[mask_a]])
        mask_b = (j - d >= 0) & (j - d < n)
        B = np.dot(p[mask_b], h[j - d[mask_b]])

    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    for _ in range(n_rounds):
        if time.time() - start_time > time_limit:
            break
        h_try = h.copy()
        n_perturb = int(rng.integers(2, 6))
        idx = rng.choice(n, size=n_perturb, replace=False)
        for idx_i in idx:
            delta = rng.uniform(-0.1, 0.1)
            h_try[idx_i] = np.clip(h_try[idx_i] + delta, 0.0, 1.0)
        h_try = project_feasible(h_try, target_sum)
        c5_try = evaluate_c5(h_try, dx)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    time_limit = budget_s - 15

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        n = 86
        h0 = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    rng = np.random.default_rng(seed)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)

    for alpha in [5, 10, 20, 50, 100]:
        if time.time() - start_time > time_limit:
            break
        h, c5 = projected_grad_descent(best_h, dx, target_sum, alpha, 200, start_time, time_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    if time.time() - start_time < time_limit:
        h, c5 = local_search_2coord(best_h, dx, target_sum, rng, 30000, start_time, time_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    for alpha in [200, 500, 1000]:
        if time.time() - start_time > time_limit:
            break
        h, c5 = projected_grad_descent(best_h, dx, target_sum, alpha, 150, start_time, time_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    if time.time() - start_time < time_limit:
        h, c5 = local_search_2coord(best_h, dx, target_sum, rng, 50000, start_time, time_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    if time.time() - start_time < time_limit:
        h, c5 = local_search_multi(best_h, dx, target_sum, rng, 2000, start_time, time_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    for restart in range(3):
        if time.time() - start_time > time_limit - 60:
            break
        h_pert = best_h + rng.normal(0, 0.05, size=n)
        h_pert = project_feasible(h_pert, target_sum)
        c5_pert = evaluate_c5(h_pert, dx)
        h_start = h_pert if c5_pert < best_c5 else best_h
        for alpha in [50, 100]:
            if time.time() - start_time > time_limit:
                break
            h, c5 = projected_grad_descent(h_start, dx, target_sum, alpha, 100, start_time, time_limit)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        if time.time() - start_time < time_limit:
            h, c5 = local_search_2coord(best_h, dx, target_sum, rng, 10000, start_time, time_limit)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    return best_h, best_c5, n
