import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, n, alpha=100.0):
    """Smooth approximation of max(corr*dx) using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_full = 2 * n - 1
    scaled = alpha * dx * corr
    
    max_scaled = np.max(scaled)
    shifted = scaled - max_scaled
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    f = (max_scaled + np.log(sum_exp)) / alpha
    
    grad = np.zeros(n)
    for k in range(n_full):
        lag = k - (n - 1)
        if abs(lag) >= n:
            continue
        w = weights[k] * dx
        
        # dc[k]/dh[m] = (1-h[m+lag]) for valid m+lag, minus h[m-lag] for valid m-lag
        # Term 1: (1-h[m+lag]) where 0 <= m+lag <= n-1
        s1 = max(0, -lag)
        e1 = min(n, n - lag)
        if s1 < e1:
            grad[s1:e1] += w * (1.0 - h[s1 + lag:e1 + lag])
        
        # Term 2: -h[m-lag] where 0 <= m-lag <= n-1
        s2 = max(0, lag)
        e2 = min(n, n + lag)
        if s2 < e2:
            grad[s2:e2] -= w * h[s2 - lag:e2 - lag]
    
    return f, grad

def project_to_constraint(h, n, tol=1e-10):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Bisection on lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def interpolate_to_size(h_init, n_init, n_target):
    """Interpolate h from n_init points to n_target points."""
    x_init = np.linspace(0, 2, n_init, endpoint=False) + 1.0/n_init
    x_target = np.linspace(0, 2, n_target, endpoint=False) + 1.0/n_target
    return np.interp(x_target, x_init, h_init)

def optimize_slsqp(h0, n, alpha=100.0, max_iter=500, time_limit=100.0):
    """Optimize using SLSQP with smooth max objective."""
    start_time = time.time()
    
    def obj(x):
        if time.time() - start_time > time_limit:
            raise TimeoutError()
        f, _ = smooth_obj_grad(x, n, alpha)
        return f
    
    def grad(x):
        if time.time() - start_time > time_limit:
            raise TimeoutError()
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n)}]
    
    try:
        result = minimize(obj, h0, method='SLSQP', jac=grad,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-10})
        if result.success or hasattr(result, 'x'):
            return result.x
    except (TimeoutError, Exception):
        pass
    
    return h0

    start_time = time.time()
    h = h0.copy()

    mu = 0.9  # momentum
    lr = lr_init

        f, g = smooth_obj_grad(h, n, alpha)
        v = mu * v + g
        h_new = h - lr * v
        
        # Project

        # Check improvement

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj_grad(h, n, alpha=100.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_full = 2 * n - 1
    scaled = alpha * dx * corr
    max_scaled = np.max(scaled)
    shifted = scaled - max_scaled
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    f = (max_scaled + np.log(sum_exp)) / alpha
    grad = np.zeros(n)
    for k in range(n_full):
        lag = k - (n - 1)
        if abs(lag) >= n:
            continue
        w = weights[k] * dx
        s1 = max(0, -lag)
        e1 = min(n, n - lag)
        if s1 < e1:
            grad[s1:e1] += w * (1.0 - h[s1 + lag:e1 + lag])
        s2 = max(0, lag)
        e2 = min(n, n + lag)
        if s2 < e2:
            grad[s2:e2] -= w * h[s2 - lag:e2 - lag]
    return f, grad


def project_to_constraint(h, n):

        else:

    for it in range(max_iter):
        if time.time() - start_time > time_limit:

        f, g = smooth_obj_grad(h, n, alpha)
        v = mu * v + g
        h_new = h - lr * v
        h_new = project_to_constraint(h_new, n)
