import numpy as np
from scipy import optimize
import time

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, mode='full') * dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_onto_constraint(h, target_sum):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target_sum}"""
    n = len(h)
    # Bisection on the offset
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_1d_convex_pl(p, q):
    """
    Solve min_{t in [0,1]} max_s (p[s]*t + q[s])
    p, q: arrays of slopes and intercepts
    Returns the optimal t.
    """
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = p * mid + q
        max_val = np.max(vals)
        # Find active constraints (within tolerance)
        active = np.abs(vals - max_val) < 1e-10
        if np.any(active):
            min_slope = np.min(p[active])
            max_slope = np.max(p[active])
            if min_slope > 1e-10:
                hi = mid
            elif max_slope < -1e-10:
                lo = mid
            else:
                break
        else:
            # Shouldn't happen
            break
    return (lo + hi) / 2.0

def coordinate_descent(h, dx, target_sum, max_sweeps=100, tol=1e-8, time_limit=None, start_time=None):
    """
    Coordinate descent to minimize C5 = max_s corr_s * dx
    subject to 0 <= h <= 1, sum(h) = target_sum
    """
    n = len(h)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        old_c5 = compute_c5(h, dx)

        corr = np.correlate(h, 1.0 - h, mode='full') * dx

            old_hj = h[j]

            # slope_s = [(1 - h[j+s]) * I[0<=j+s<n] - h[j-s] * I[0<=j-s<n]] * dx
            # Shift s ranges from -(n-1) to (n-1)
            # For the corr array index k = s + (n-1)

            # h[j-s] is valid when 0 <= j-s < n, i.e., j-n < s <= j
            
            # Let's compute for all s in range -(n-1) to (n-1)
            s = np.arange(-(n-1), n)
            
            # h[j+s] contribution
            js = j + s
            h_js = np.where((js >= 0) & (js < n), h[js[(js >= 0) & (js < n)]], 0.0)
            # Actually, let's be more careful
            valid_js = (js >= 0) & (js < n)
            h_js = np.zeros(2*n - 1)
            h_js[valid_js] = h[js[valid_js]]
            
            # h[j-s] contribution
            jss = j - s

            h_jss = np.zeros(2*n - 1)
            h_jss[valid_jss] = h[jss[valid_jss]]

            if t_star != old_hj:

                h[j] = t_star
                # Update corr
                delta = t_star - old_hj
                corr = corr + delta * slopes

        h = project_onto_constraint(h, target_sum)

        new_c5 = np.max(corr)

def solve_1d_convex_pl(p, q):
    """
    Solve min_{t in [0,1]} max_s (p[s]*t + q[s])
    p, q: arrays of slopes and intercepts
    """
    # Evaluate at endpoints
    val_0 = np.max(q)
    val_1 = np.max(p + q)
    
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = p * mid + q
        max_val = np.max(vals)
        # Active set: where vals is close to max
        active = vals >= max_val - 1e-12
        min_slope = np.min(p[active])
        max_slope = np.max(p[active])
        if min_slope > 0:
            hi = mid
        elif max_slope < 0:
            lo = mid
        else:
            break
    
    t_star = (lo + hi) / 2.0
    
    # Verify: also check endpoints
    val_star = np.max(p * t_star + q)
    result = min(val_0, val_1, val_star)
    if result == val_0:
        return 0.0
    elif result == val_1:
        return 1.0
    else:
        return t_star

import numpy as np
import time

def compute_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_1d(p, q):
    # min_{t in [0,1]} max_s (p[s]*t + q[s])
    val0 = np.max(q)
    val1 = np.max(p + q)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = p * mid + q
        mv = np.max(vals)
        active = vals >= mv - 1e-14
        ms = np.min(p[active])
        Mx = np.max(p[active])
        if ms > 1e-14:
            hi = mid
        elif Mx < -1e-14:
            lo = mid
        else:
            break
    t = (lo + hi) / 2.0
    vt = np.max(p * t + q)
    if val0 <= val1 and val0 <= vt:
        return 0.0
    elif val1 <= val0 and val1 <= vt:
        return 1.0
    return t

def coord_desc(h, dx, target, max_sweeps=500, tol=1e-10, time_fn=None, time_limit=None):
    n = len(h)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    prev_c5 = np.max(corr)
    n_full = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time_fn is not None and time_limit is not None:
            if time_fn() > time_limit:
                break
        
        for j in range(n):
            old = h[j]
            # Compute slopes for all shifts
            # shift s = k - (n-1), k in [0, 2n-2]
            # h[j+s] valid when 0 <= j+s < n
            # h[j-s] valid when 0 <= j-s < n
            
            # Vectorized: for shift index k (0 to 2n-2), s = k - (n-1)
            # j+s = j + k - (n-1)
            # j-s = j - k + (n-1)
            
            s = np.arange(0, n_full) - (n - 1)  # shifts, length 2n-1
            js = j + s
            jss = j - s
            
            h_js = np.zeros(n_full)
            m1 = (js >= 0) & (js < n)
            h_js[m1] = h[js[m1]]
            
            h_jss = np.zeros(n_full)
            m2 = (jss >= 0) & (jss < n)
            h_jss[m2] = h[jss[m2]]
            
            slopes = ((1.0 - h_js) - h_jss) * dx
            intercepts = corr - old * slopes
            
            t_star = solve_1d(slopes, intercepts)
            
            delta = t_star - old
            if abs(delta) > 1e-15:
                h[j] = t_star
                corr += delta * slopes
        
        # Project
        h = project_feasible(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = np.max(corr)
        
        if prev_c5 - c5 < tol:
            break
        prev_c5 = c5
    
    return h, np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

    deadline = t0 + budget_s - 10
    
    h0 = np.array(initial_h_values, dtype=float)

Actually, I just realized there might be an issue with my approach. When I do coordinate descent and then project, the projection changes h in a way that might undo the optimization. Let me think about this.

After optimizing h[j] in the coordinate descent step, I set h[j] = t*. But then at the end of the sweep, I project h onto {sum = target, 0 ≤ h ≤ 1}. This projection might change h[j] back.

To avoid this, I can:
1. Not project at the end of each sweep, but only at the end of the optimization
2. Or, incorporate the sum constraint into the 1D problem (using a Lagrange multiplier)

Option 2 is better. Let me use a Lagrange multiplier.

**Revised approach with Lagrange multiplier:**

The problem:
min C
s.t. 0 ≤ h[i] ≤ 1, sum(h) = target

Lagrangian: L(h, λ) = C + λ * (sum(h) - target)

For a fixed λ, the problem decomposes into n independent 1D problems:
For coordinate j: min_{t∈[0,1]} max_s (slope_s * t + intercept_s) + λ * t
= min_{t∈[0,1]} max_s ((slope_s + λ) * t + intercept_s)

After solving all n 1D problems, we get h(λ). Then we adjust λ so that sum(h(λ)) = target.

The function sum(h(λ)) is a decreasing function of λ (larger λ penalizes larger values, so the optimal h(λ) has smaller sum). We can use bisection on λ.

But doing a full coordinate sweep for each bisection iteration on λ is expensive. Instead, I'll use an iterative approach:
- Start with λ = 0
- Do a coordinate sweep (which finds the optimal h for that λ)
- Check if sum(h) = target
- If sum > target, increase λ; if sum < target, decrease λ
- Repeat

The step size for λ can be adaptive.

Actually, a simpler approach that works well in practice:
- Do the coordinate descent WITHOUT the sum constraint (λ = 0)
- After each sweep, project onto the constraint
- The projection is a small perturbation (if the sum is close to target)

This should work because the coordinate descent is making progress on the objective, and the projection only makes a small correction.

Let me test this approach. The key question is: does the projection significantly undo the optimization?

If the coordinate descent (without constraint) drives sum(h) to a value far from target, then the projection will be large and undoes a lot of work. But if it's close, the projection is small.

In practice, for this problem, I expect sum(h) to stay reasonably close to target during the optimization (since the objective doesn't strongly push the sum away from target). So the projection should be a small correction.
