            best_h, target_sum, beta, 3000, 0.01, phase1_limit, start_time
        )
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

    # Phase 2: Nelder-Mead refinement
    if elapsed() < budget_s * 0.75:
        h_nm, c5_nm = nelder_mead_phase(
            best_h, target_sum, 3000, budget_s * 0.75, start_time
        )
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()

    # Phase 3: Multi-start with gradient descent
    phase3_limit = budget_s * 0.93
    n_starts = 80

    for restart in range(n_starts):
        if elapsed() > phase3_limit:
            break
        scale = 0.003 + 0.004 * np.sqrt(restart / max(n_starts, 1))
        h_try = best_h + np.random.randn(n) * scale
        h_try = project_onto_feasible(h_try, target_sum)

        h_try, c5_try = gradient_descent_phase(
            h_try, target_sum, 300.0, 800, 0.003, phase3_limit, start_time
        )
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Final Nelder-Mead polish
    if elapsed() < budget_s * 0.97:
        h_nm2, c5_nm2 = nelder_mead_phase(
            best_h, target_sum, 2000, budget_s * 0.97, start_time
        )
        if c5_nm2 < best_c5:
            best_c5 = c5_nm2
            best_h = h_nm2.copy()

    # Final projection and evaluation
    best_h = project_onto_feasible(best_h, target_sum)
    best_c5 = compute_c5_fast(best_h)

    return (best_h, best_c5, n)
