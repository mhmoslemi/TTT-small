import numpy as np
import time
from scipy.optimize import minimize, Bounds
import math

def _get_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _project_box_sum(v, n, target=None):
    if target is None:
        target = n / 2.0
    lo = v.min() - 1.5
    hi = v.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _compute_subgrad(h, n, eps=1e-3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - eps * max_val)[0]
    grad = np.zeros(n)
    count = 0
    for idx in active:
        k = idx - (n - 1)
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            ka = -k
            g = np.zeros(n)
            g[:n - ka] += (1.0 - h[ka:])
            g[ka:] -= h[:n - ka]
        grad += g
        count += 1
    if count > 0:
        grad /= count
    return dx * grad

def _slsqp_optimize(h_init, n, max_iter=500, ftol=1e-10, use_analytic_grad=True):
    h = _project_box_sum(h_init.copy(), n)

    def obj(x):
        return _get_score(x)

    def obj_grad(x):
        if use_analytic_grad:
            return _compute_subgrad(x, n)
        return None

    def cons_fun(x):
        return x.sum() - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    best_h = h.copy()
    best_score = _get_score(h)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=obj_grad if use_analytic_grad else None,
                       options={'maxiter': max_iter, 'ftol': ftol})
        h_opt = _project_box_sum(res.x, n)
        s = _get_score(h_opt)
        if s < best_score:
            best_h = h_opt
            best_score = s
    except Exception:
        pass
    return best_h, best_score

def _pgd_optimize(h_init, n, max_iters=5000, initial_step=0.005):
    h = _project_box_sum(h_init.copy(), n)
    best_h = h.copy()
    best_score = _get_score(h)
    step = initial_step
    patience = 0

    for it in range(max_iters):
        grad = _compute_subgrad(h, n)
        h_new = _project_box_sum(h - step * grad, n)
        score = _get_score(h_new)
        if score < best_score:
            best_h = h_new.copy()
            best_score = score
            patience = 0
        else:
            patience += 1
            step *= 0.95
        h = h_new
        if patience > 200:
            break
        if step < 1e-8:
            break

    return best_h, best_score

def _smooth_max_optimize(h_init, n, max_iter=300):
    h = _project_box_sum(h_init.copy(), n)
    best_h = h.copy()
    best_score = _get_score(h)
    temperatures = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for T in temperatures:
        def obj_smooth(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            shifted = np.clip(corr - m, -50.0 * T, 0.0)
            lse = T * np.log(np.sum(np.exp(shifted / T)))
            return dx * (m + lse)

        def cons_fun(x):
            return x.sum() - n / 2.0

        bounds = Bounds(np.zeros(n), np.ones(n))
        constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
        per_iter = max(30, max_iter // len(temperatures))

        try:
            res = minimize(obj_smooth, h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': per_iter, 'ftol': 1e-12})
            h_opt = _project_box_sum(res.x, n)
            s = _get_score(h_opt)
            if s < best_score:
                best_h = h_opt.copy()
                best_score = s
            h = h_opt
        except Exception:
            pass

    return best_h, best_score

def _rand_structured(n, rng):
    h = np.full(n, 0.5)
    n_modes = rng.randint(2, 7)
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 2))
        phase = rng.uniform(0, 2.0 * math.pi)
        amp = rng.uniform(0.05, 0.35)
        h += amp * np.sin(2.0 * math.pi * freq * np.arange(n) / n + phase)
    return _project_box_sum(h, n)

def _interpolate_to(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return _project_box_sum(h_tgt, n_target)
