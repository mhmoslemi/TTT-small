import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_value_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    val = corr[idx] * dx
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:n] = 1.0
            if n - k > 0:
                grad[k:n] -= h[0:n-k]
                grad[0:n-k] -= h[k:n]
    else:
        p = -k
        if p < n:
            grad[0:n-p] = 1.0
            if n - p > 0:
                grad[0:n-p] -= h[p:n]
                grad[p:n] -= h[0:n-p]
    return val, grad * dx

def constraint_sum(h, n):
    return np.sum(h) - n / 2.0

    n = len(h)
    return np.ones(n)

    val, _ = c5_value_and_grad(np.array(h))
    return val

def objective_grad(h):
    _, grad = c5_value_and_grad(np.array(h))
    return grad

def optimize_slq(p, n, maxiter=200, ftol=1e-10):
    bounds = [(0.0, 1.0)] * n
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}
    ]
    res = minimize(
        objective_func, p, jac=objective_grad,
        method='SLSQP', bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return res.x, res.fun

def coordinate_descent(h, n, n_sweeps=50):
    """Do coordinate descent on individual coordinates."""
    h = np.array(h, dtype=float)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):

            # The objective as a function of h[i] is piecewise quadratic
            # For each lag k, the contribution involving h[i] is:
            #   If i in [k, n-1]: h[i] - h[i]*h[i-k] (from first sum and S2)
            #   If i in [0, n-1-k]: -h[i]*h[i+k] (from S2 second term)
            # So the total as a function of t = h[i] for a specific active lag is:
            #   (indicator of i in active range) * t - t * (sum of neighbors)
            # But since we're taking max over all lags, it's more complex.
            # Simple approach: try a few values of h[i] and pick the best.
            old_val = h[i]
            # Try values: old_val, and a few neighboring values
            candidates = [max(0, min(1, old_val + d)) for d in [-0.1, -0.05, -0.01, 0.01, 0.05, 0.1]]
            candidates.append(old_val)
            # Also try to satisfy sum constraint by adjusting
            best_candidate = old_val
            best_obj = compute_c5(h)
            for c in candidates:
                h[i] = c
                # Adjust sum: distribute the difference to other coordinates
                diff = n/2.0 - np.sum(h)
                if abs(diff) > 1e-10:
                    # Distribute to coordinates that are not at bounds
                    mask = (h > 1e-6) & (h < 1 - 1e-6)
                    if np.sum(mask) > 0:
                        h[mask] += diff / np.sum(mask)
                        h = np.clip(h, 0, 1)
                        # Re-adjust for clipping
                        diff2 = n/2.0 - np.sum(h)
                        if abs(diff2) > 1e-10:
                            h += diff2 / n
                            h = np.clip(h, 0, 1)
                obj = compute_c5(h)
                if obj < best_obj:
                    best_obj = obj
                    best_candidate = c
            h[i] = best_candidate
            # Re-normalize sum
            diff = n/2.0 - np.sum(h)
            if abs(diff) > 1e-10:
                h += diff / n
                h = np.clip(h, 0, 1)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5 - 1e-12:
            best_c5 = current_c5
            improved = True
        if not improved:
            break
    
    return h, best_c5

    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute uniformly (simple but effective)
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Second pass if needed
        diff = target - np.sum(h)

            if np.sum(mask) > 0:

                h = np.clip(h, 0.0, 1.0)
    return h
