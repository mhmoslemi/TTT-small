import time
import random

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def paired_perturbation(h, i, j, delta):
    """Transfer delta from h[i] to h[j]"""
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
        return None
    return new_h

def local_search(h, n_iters=10000, time_limit=None, t_start=None):
    """Simulated annealing style local search with paired perturbations"""
    n = len(h)
    current = h.copy()
    current_val = evaluate_c5(current)
    best = current.copy()
    best_val = current_val

        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Random delta
        magnitude = np.random.uniform(0.01, 0.1)
        sign = np.random.choice([-1, 1])
        delta = sign * magnitude
        
        new_h = paired_perturbation(current, i, j, delta)
        if new_h is None:
            continue
        
        new_val = evaluate_c5(new_h)
        
        # Accept if better or with probability exp(-(new_val - current_val)/T)

def binary_evaluate_c5(h):
    """Evaluate C5 for a binary array (exactly n/2 ones)"""
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n_half = n // 2
    indices = np.argsort(h)[::-1][:n_half]
    b = np.zeros(n)
    b[indices] = 1.0

    best_val = current_val
    T = 0.005

import time
import random

def evaluate_c5(h):

    dx = 2.0 / n

def sa_continuous(h, deadline):
    """Simulated annealing on continuous values with paired perturbations"""
    n = len(h)
    cur = h.copy()
    cur_val = evaluate_c5(cur)
    best = cur.copy()
    best_val = cur_val
    T = 0.02
    step = 0.03
    
    while time.time() < deadline:
        # Batch of iterations
        for _ in range(1000):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            d = np.random.uniform(-step, step)
            new_hi = cur[i] + d
            new_hj = cur[j] - d
            if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
                continue
            new_h = cur.copy()
            new_h[i] = new_hi
            new_h[j] = new_hj
            new_val = evaluate_c5(new_h)

                cur = new_h
                cur_val = new_val

                    best = new_h.copy()
                    best_val = new_val
        T = max(0.001, T * 0.9)
        step = max(0.005, step * 0.95)
    
    return best, best_val
