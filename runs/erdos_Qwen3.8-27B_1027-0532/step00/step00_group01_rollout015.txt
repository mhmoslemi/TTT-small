def compute_grad_smoothed(h, T):
    n = len(h)
    v = 1 - h
    dx = 2.0 / n
    corr = np.correlate(h, v, mode="full")
    
    # Smoothed max weights
    # corr has length 2n-1, indexed 0 to 2n-2
    # shift s corresponds to index s + n - 1
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))  # for numerical stability
    weights = exp_c / np.sum(exp_c)
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        s = idx - (n - 1)  # actual shift
        # v[j+s] for valid j: j+s in [0, n-1] => j in [-s, n-1-s]
        # h[j-s] for valid j: j-s in [0, n-1] => j in [s, n-1+s]
        
        # Contribution from v[j+s]:
        j_min = max(0, -s)
        j_max = min(n-1, n-1-s)
        if j_min <= j_max:
            grad[j_min:j_max+1] += w * v[j_min+s:j_max+s+1]
        
        # Contribution from -h[j-s]:
        j_min2 = max(0, s)
        j_max2 = min(n-1, n-1+s)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] -= w * h[j_min2-s:j_max2-s+1]
    
    return grad * dx  # multiply by dx because the objective is max(corr) * dx

    v = 1.0 - h

def _lpb_jac(h, n, T, pen):
    _, grad = smooth_obj_and_grad(h, T)
    return grad + 2.0 * pen * (h.sum() - n * 0.5) * np.ones(n)


def projected_gradient(h0, n, rng, time_limit, T_start=10.0, T_end=300.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        frac = (time.time() - start) / time_limit
        T = T_start + (T_end - T_start) * frac
        step = 0.08 * (1.0 - 0.9 * frac) + 0.003
        _, grad = smooth_obj_and_grad(h, T)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if rng.random() < 0.003:
            h = h + 0.04 * (1.0 - frac) * rng.randn(n)
            h = project_feasible(h, n)
    return best_h, best_c5


def lpb_optimize(h0, n, T, pen, maxiter=2000):
    fun = partial(_lpb_fun, n=n, T=T, pen=pen)
    jac = partial(_lpb_jac, n=n, T=T, pen=pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    h_init = np.array(initial_h_values, dtype=float).copy()
    h_init = project_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Projected gradient from initial (40% budget)
    t1 = 0.40 * (budget_s - 60)
    h1, c5_1 = projected_gradient(h_init, n, rng, t1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: L-BFGS-B with penalty at various T (25% budget)
    remaining = budget_s - (time.time() - t0) - 30
    if remaining > 20:
        configs = [(20, 50), (40, 200), (80, 500)]
        t_per = remaining / len(configs)
        for T_val, pen_val in configs:
            if time.time() - t0 > budget_s - 15:
                break
            t_bud = min(t_per, budget_s - (time.time() - t0) - 10)
            if t_bud < 5:
                break
            h_lpb = lpb_optimize(best_h, n, T_val, pen_val, maxiter=int(t_bud * 150))
            c5_lpb = compute_c5(h_lpb)
            if c5_lpb < best_c5:
                best_c5 = c5_lpb
                best_h = h_lpb

    # Phase 3: Continue projected gradient from best (20% budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h3, c5_3 = projected_gradient(best_h, n, rng, remaining - 10, T_start=50.0, T_end=400.0)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: Try doubled resolution (last 10% budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        n2 = n * 2
        x1 = np.linspace(0.0, 2.0, n, endpoint=False)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x2, x1, best_h)
        h_interp = project_feasible(h_interp, n2)
        h4, c5_4 = projected_gradient(h_interp, n2, rng, remaining - 10, T_start=20.0, T_end=300.0)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
            n = n2

    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
