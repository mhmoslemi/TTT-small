```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # Gradient of corr[max_idx] with respect to h
    # corr[m] = sum_i h[i] * (1-h[i + m - (n-1)])
    # For fixed m, d(corr[m])/d(h[j]) involves terms where h[j] appears
    # h[j] appears as h[i] when i=j, contributing (1-h[j+m-(n-1)])
    # h[j] appears as h[i + m - (n-1)] when i + m - (n-1) = j, i.e., i = j + (n-1-m)
    # contributing -h[j + (n-1-m)]
    
    s = max_idx - (n - 1)  # the actual shift
    n_valid = n - abs(s)
    
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            # term: h[i] * (1 - h[i+s])
            grad[i] += (1 - h[i + s])
            grad[i + s] += -h[i]
    else:
        s_abs = -s
        for i in range(n - s_abs):
            # term: h[i+s_abs] * (1 - h[i])
            grad[i + s_abs] += (1 - h[i])
            grad[i] += -h[i + s_abs]
    
    return grad * dx
```

The projection onto this set can be done efficiently. Given a point z, find h that minimizes ||h-z||² subject to sum h = n/2, 0 ≤ h ≤ 1. This is a quadratic program that can be solved by binary search on the Lagrange multiplier.

```python
def project_feasible(z, n):
    """Project z onto {h: sum h = n/2, 0 <= h <= 1}"""
    target = n / 2.0
    
    def clip_with_offset(z, offset):
        h = np.clip(z + offset, 0, 1)
        return h
    
    # Binary search for the offset
    lo = -z.max() - 1
    hi = -z.min() + 2
    for _ in range(100):
        mid = (lo + hi) / 2
        h = clip_with_offset(z, mid)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = clip_with_offset(z, (lo + hi) / 2)
    # Fine-tune
    diff = target - h.sum()
    mask = (h > 0) & (h < 1)
    if mask.sum() > 0 and abs(diff) > 1e-12:
        h[mask] += diff / mask.sum()
    return h
```

Actually, let me think about this more carefully. The gradient of the max function is only defined at the point where a unique maximum is achieved. At points where multiple correlations are tied for the maximum, the subdifferential is the convex hull of their gradients. For practical purposes, I'll just use the gradient of one of the maximum components.

But actually, a better approach for min-max is to smooth the max with a softmax (log-sum-exp trick). This gives a smooth objective that approximates the max:

f_smooth(h) = (1/τ) * log(sum_k exp(τ * corr_k))

As τ → ∞, this approaches max_k corr_k. The gradient is a weighted average of the individual gradients, which is more stable.

Let me use this approach:

```python
def smooth_max_objective(h, n, tau=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Shift to avoid overflow
    c_max = corr.max()
    return c_max + np.log(np.sum(np.exp(tau * (corr - c_max)))) / tau

def smooth_max_gradient(h, n, tau=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    weights = np.exp(tau * (corr - c_max))
    weights /= weights.sum()
    # Gradient is weighted average of individual gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i] += weights[k] * (1 - h[i + s])
                grad[i + s] += -weights[k] * h[i]
        else:
            s_abs = -s
            for i in range(n - s_abs):
                grad[i + s_abs] += weights[k] * (1 - h[i])
                grad[i] += -weights[k] * h[i + s_abs]
    return grad * dx
```
