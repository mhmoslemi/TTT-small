def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_core(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    sum_exp = np.sum(np.exp(tau * (corr - max_corr)))
    return (max_corr + np.log(sum_exp) / tau) * dx

def smooth_grad_core(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    sum_exp = np.sum(np.exp(tau * (corr - max_corr)))
    weights = np.exp(tau * (corr - max_corr)) / sum_exp
    grad = np.zeros(n)
    n_corr = len(corr)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (n - 1)
