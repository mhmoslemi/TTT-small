```python
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    return (max_c + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_grad(h, dx, beta, n):
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    shifts = np.arange(len(corr)) - (n - 1)
```

```python
    """Compute gradient using top-m active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[::-1][:top_m]
    max_val = corr[sorted_idx[0]]
    # Use lags within 1% of max
    active = sorted_idx[corr[sorted_idx] >= max_val * 0.999]
    
    shifts = active - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        dc = 0.0
        for s in shifts:
            if 0 <= j + s < n:
                dc += (1.0 - h[j + s])
            if 0 <= j - s < n:
                dc += (-h[j - s])
        grad[j] = dc / len(shifts)
    grad *= dx
    grad -= np.mean(grad)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, ts):
    h = np.clip(h, 0.0, 1.0)
    d = ts - np.sum(h)
    if abs(d) > 1e-14:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= np.sum(w)
    sh = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        vp = (j + sh >= 0) & (j + sh < n)
        vm = (j - sh >= 0) & (j - sh < n)
        dc = np.zeros(2 * n - 1)
        if np.any(vp):
            dc[vp] = 1.0 - h[j + sh[vp]]
        if np.any(vm):
            dc[vm] = dc[vm] - h[j - sh[vm]]
        g[j] = np.dot(w, dc)
    g *= dx
    g -= g.mean()
    return g

def _grad_active(h, dx, m=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:m]
    g = np.zeros(n)
    for m_idx in idx:
        s = m_idx - (n - 1)
        for j in range(n):
            if 0 <= j + s < n:
                g[j] += 1.0 - h[j + s]
            if 0 <= j - s < n:
                g[j] -= h[j - s]
    g /= len(idx)
    g *= dx
    g -= g.mean()
    return g
```
