  - Similarly for j.

- For the term where i+k = i₀, i.e., i = i₀-k: h[i+k] changes from 1 to 0, so (1-h[i+k]) changes from 0 to 1. The term h[i]*(1-h[i+k]) changes from 0 to h[i₀-k]. Change: +h[i₀-k] (if i₀-k ≥ 0)

- For the term where i = j₀: h[i] changes from 0 to 1, so the term changes from 0 to (1-h[j₀+k]). Change: +(1-h[j₀+k])
- For the term where i+k = j₀, i.e., i = j₀-k: h[i+k] changes from 0 to 1, so (1-h[i+k]) changes from 1 to 0. The term changes from h[j₀-k]*1 to 0. Change: -h[j₀-k] (if j₀-k ≥ 0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def binarize(h):
    return (h > 0.5).astype(np.float64)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def local_search(h, dx, time_limit, start_time, seed=42, restart_every=20000):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    stuck = 0
    
    while time.time() - start_time < time_limit:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_try = best_h.copy()
        h_try[i] = 1.0 - h_try[i]
        h_try[j] = 1.0 - h_try[j]
        
        c5_try = compute_c5(h_try, dx)
        
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
            stuck = 0
        else:
            stuck += 1
        
        if stuck >= restart_every:
            # Restart with perturbation
            best_h = best_h.copy()
            for _ in range(3):
                ones = np.where(best_h > 0.5)[0]
                zeros = np.where(best_h < 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    best_h[i] = 0
                    best_h[j] = 1
            best_c5 = compute_c5(best_h, dx)
            stuck = 0
    
    return best_h, best_c5

def steepest_descent(h, dx, time_limit, start_time, max_rounds=5000):
    n = len(h)
    current_c5 = compute_c5(h, dx)
    current_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        ones = list(np.where(current_h > 0.5)[0])
        zeros = list(np.where(current_h < 0.5)[0])
        
        if not ones or not zeros:
            break
        
        best_imp = 0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                current_h[i] = 0
                current_h[j] = 1
                c5_try = compute_c5(current_h, dx)
                current_h[i] = 1
                current_h[j] = 0
                
                imp = current_c5 - c5_try
                if imp > best_imp + 1e-15:
                    best_imp = imp
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        current_h[i] = 0
        current_h[j] = 1
        current_c5 -= best_imp
    
    return current_h, current_c5

def continuous_refine(h, dx, max_iter=200):
    n = len(h)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * dx)
    
    def grad(x):
        # Numerical gradient
        eps = 1e-7
        g = np.zeros(n)
        base = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - base) / eps
            x[i] -= eps
        return g
    
    # Simple projected gradient descent
    x = h.astype(float).copy()
    lr = 0.01
    
    for it in range(max_iter):
        g = grad(x)
        x_new = x - lr * g
        # Project onto feasible set: 0 <= x <= 1, sum(x) = n/2
        x_new = np.clip(x_new, 0, 1)
        # Adjust to satisfy sum constraint
        diff = n/2 - np.sum(x_new)
        if abs(diff) > 1e-10:
            # Distribute the difference
            mask_pos = x_new < 1
            mask_neg = x_new > 0
            if diff > 0:
                # Need to increase sum
                avail = np.where(mask_pos)[0]
                if len(avail) > 0:
                    add_per = diff / len(avail)
                    x_new[avail] = np.minimum(1, x_new[avail] + add_per)
            else:
                avail = np.where(mask_neg)[0]
                if len(avail) > 0:
                    sub_per = (-diff) / len(avail)
                    x_new[avail] = np.maximum(0, x_new[avail] - sub_per)
        
        if obj(x_new) < obj(x):
            x = x_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return x, obj(x)
