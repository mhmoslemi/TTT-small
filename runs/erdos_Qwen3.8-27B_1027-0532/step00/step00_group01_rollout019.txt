def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx

    y = T * scaled
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    w = exp_y / np.sum(exp_y)
    
    # w[i] corresponds to lag k = i - (n-1)
    # F = (y_max + log(sum(exp_y))) / T
    
    # Gradient: dF/dh[j] = dx * sum_k w[k_idx] * dc_k/dh[j]
    # where k_idx = k + (n-1) and k ranges from -(n-1) to n-1
    
    grad = np.zeros(n)
    for j in range(n):
        for k_idx in range(2 * n - 1):
            k = k_idx - (n - 1)  # actual lag
            # dc_k/dh[j] = (1-h[j-k]) if 0 <= j-k < n, minus h[j+k] if 0 <= j+k < n
            term = 0.0
            jm = j - k  # j-k
            if 0 <= jm < n:
                term += (1.0 - h[jm])
            jp = j + k  # j+k
            if 0 <= jp < n:
                term -= h[jp]
            grad[j] += w[k_idx] * term
        grad[j] *= dx
    
    F = (y_max + np.log(np.sum(exp_y))) / T
    return F, grad

The smooth objective: F = (1/T) * log(sum_k exp(T * c_k)) where c_k = (correlation at lag k) * dx.

dF/dh[j] = sum_k [exp(T*c_k) / sum_m exp(T*c_m)] * T * (dc_k/dh[j]) / T
         = sum_k w_k * dc_k/dh[j]

where w_k = softmax(T * c_k)

c_k = sum_{n: valid} h[n+k] * (1-h[n]) * dx

dc_k/dh[j]:
- h[j] appears in h[n+k] when n+k = j, i.e., n = j-k. Contribution: (1-h[j-k]) * dx if 0 ≤ j-k < n.
- h[j] appears in (1-h[n]) when n = j. Contribution: -h[j+k] * dx if 0 ≤ j+k < n.

So dc_k/dh[j] = dx * [(1-h[j-k]) * 1{0≤j-k<n} - h[j+k] * 1{0≤j+k<n}]

And dF/dh[j] = sum_k w_k * dx * [(1-h[j-k]) * 1{0≤j-k<n} - h[j+k] * 1{0≤j+k<n}]

    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(63, 0.5)

    n_init = len(h_init)
    h_init = project_to_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    best_n = n_init

    # Phase 1: Projected gradient with increasing T
    phase1_end = start_time + budget_s * 0.22
    h_curr = best_h.copy()
    for T in [10, 20, 50, 100, 200, 400]:
        if time.time() > phase1_end:
            break
        lr = 1.0 / (T + 5.0)
        h_curr = pd_optimize(h_curr, n_init, T, lr, 2000, phase1_end)
        c5 = evaluate_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()

    # Phase 2: High-T refinement
    phase2_end = start_time + budget_s * 0.35
    for T in [800, 1500]:
        if time.time() > phase2_end:
            break
        lr = 0.4 / (T + 10.0)
        best_h = pd_optimize(best_h, best_n, T, lr, 800, phase2_end)
        best_c5 = evaluate_c5(best_h)

    # Phase 3: Coordinate polish
    phase3_end = start_time + budget_s * 0.48
    best_h, best_c5 = coordinate_polish(best_h, best_n, max(1, phase3_end - time.time()))

    # Phase 4: Local random search (multiple restarts)
    phase4_end = start_time + budget_s * 0.75
    n_starts = 5
    time_per = max(5.0, (phase4_end - time.time()) / n_starts)
    for i in range(n_starts):
        if time.time() > phase4_end:
            break
        if i == 0:
            h_start = best_h.copy()
        else:
            sigma = 0.04 + 0.12 * (i / n_starts)
            h_start = best_h + rng.normal(0, sigma, best_n)
            h_start = project_to_feasible(h_start, best_n)
        h_local, c5_local = local_search(h_start, best_n, np.random.default_rng(seed + 7000 * (i + 1)), time_per)
        if c5_local < best_c5:
            best_h, best_c5 = h_local.copy(), c5_local

    # Phase 5: Try larger n_points
    if time.time() < start_time + budget_s * 0.82:
        for n_try in [126, 256]:
            if time.time() > start_time + budget_s * 0.86:
                break
            x_old = np.linspace(0.0, 2.0, len(best_h))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_to_feasible(h_interp, n_try)
            pd_end = start_time + budget_s * 0.89
            h_interp = pd_optimize(h_interp, n_try, 100.0, 0.008, 300, pd_end)
            c5_interp = evaluate_c5(h_interp)
            if c5_interp < best_c5:
                best_h, best_c5, best_n = h_interp.copy(), c5_interp, n_try

    # Phase 6: Final local search
    remaining = max(5.0, budget_s - (time.time() - start_time) - 8)
    if remaining > 5:
        h_local, c5_local = local_search(best_h, best_n, rng, remaining)
        if c5_local < best_c5:
            best_h, best_c5 = h_local.copy(), c5_local

    # Final validation
    best_h = project_to_feasible(best_h, best_n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, best_n
