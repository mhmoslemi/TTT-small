                grad[m] -= h[idx2]

import numpy as np
import time


def _shift_z(v, d):
    n = len(v)
    if d == 0:
        return v
    r = np.zeros(n)
    if 0 < d < n:
        r[d:] = v[:n - d]
    elif -n < d < 0:
        r[:n + d] = v[-d:]
    return r


def _proj(v, ts):
    lo = -5.0
    hi = 5.0
    for _ in range(70):
        m = (lo + hi) * 0.5
        h = np.clip(v - m, 0.0, 1.0)
        if h.sum() > ts:
            lo = m
        else:
            hi = m
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / len(h))


def _subgrad(h, tk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    tk = min(tk, 2 * n - 1)
    idx = np.argpartition(corr, -tk)[-tk:]
    g = np.zeros(n)
    omh = 1.0 - h
    nm1 = n - 1
    for k in idx:
        g += _shift_z(omh, k - nm1) - _shift_z(h, nm1 - k)
    g /= tk
    return g


def _rand_h(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return _proj(h, n * 0.5)


def _alt_h(n, rng):
    h = np.zeros(n)
    pattern = rng.choice([2, 3, 4, 5])
    for i in range(n):
        h[i] = 1.0 if (i % pattern) < pattern // 2 else 0.0
    return _proj(h, n * 0.5)


def _optimize(n, h0, tl, seed):
    rng = np.random.RandomState(seed)
    ts = n * 0.5
    h = _proj(h0.copy(), ts)
    bh = h.copy()
    bv = _c5(bh)
    t0 = time.time()
    patience = 0
    max_patience = 3000
    alpha_min = 0.001
    alpha_max = 0.3

    step_sizes = np.array([0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.1, 0.15, 0.2, 0.3, 0.5])

    while time.time() - t0 < tl:
        if rng.random() < 0.6:
            tk = int(np.random.choice([3, 5, 8, 12, 15, 20]))
            g = _subgrad(h, tk)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                d = rng.normal(0, 1, n)
            else:
                d = -g
        else:
            d = rng.normal(0, 1, n)

        d = d - d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            continue
        d /= dn

        ba = 0.0
        blv = bv
        for a in step_sizes:
            if a < alpha_min or a > alpha_max:
                continue
            ht = _proj(h + a * d, ts)
            vt = _c5(ht)
            if vt < blv:
                blv = vt
                ba = a

        if ba > 0:
            h = _proj(h + ba * d, ts)
            if blv < bv:
                bh = h.copy()
                bv = blv
                patience = 0
            else:
                patience += 1
        else:
            patience += 1

        if patience > max_patience:
            h = bh + rng.normal(0, 0.04, n)
            h = _proj(h, ts)
            patience = 0

        step_sizes *= 1.02
        step_sizes = np.clip(step_sizes, alpha_min, alpha_max)

    return bh, bv


def _polish(n, h0, tl, seed):
    rng = np.random.RandomState(seed)
    ts = n * 0.5
    h = _proj(h0.copy(), ts)
    bh = h.copy()
    bv = _c5(bh)
    t0 = time.time()

    small_steps = np.array([0.001, 0.003, 0.005, 0.01, 0.015, 0.02])

    while time.time() - t0 < tl:
        d = rng.normal(0, 1, n)
        d = d - d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            continue
        d /= dn

        ba = 0.0
        blv = bv
        for a in small_steps:
            ht = _proj(h + a * d, ts)
            vt = _c5(ht)
            if vt < blv:
                blv = vt
                ba = a
        if ba > 0:
            h = _proj(h + ba * d, ts)
            if blv < bv:
                bh = h.copy()
                bv = blv

    return bh, bv
