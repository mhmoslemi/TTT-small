    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):

        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def softmax_obj(h, beta):
    n = len(h)
    corr = compute_corr_vec(h)
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def softmax_grad(h, beta):
    n = len(h)
    corr = compute_corr_vec(h)
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        shift = m - n + 1
        if shift >= 0:
            # (1-h[i+shift]) for i = 0..n-1-shift
            if n - shift > 0:
                idx = np.arange(0, n - shift)
                grad[idx] += wm * (1.0 - h[idx + shift])
            # -h[i-shift] for i = shift..n-1
            if shift < n:
                idx2 = np.arange(shift, n)
                grad[idx2] -= wm * h[idx2 - shift]
        else:
            s = -shift
            # (1-h[i-s]) for i = s..n-1
            if n - s > 0:
                idx = np.arange(s, n)
                grad[idx] += wm * (1.0 - h[idx - s])
            # -h[i+s] for i = 0..n-1-s
            if n - s > 0:
                idx2 = np.arange(0, n - s)
                grad[idx2] -= wm * h[idx2 + s]

def _compute_corr(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _compute_c5(h):

    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _grad_shift(h, shift):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if shift >= 0:
        if n > shift:
            g[:n - shift] += 1.0 - h[shift:]
        if shift > 0:
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if n > s:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g * dx


def _project(y, target):
