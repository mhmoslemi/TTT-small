def make_thue_morse(n):
    # Thue-Morse sequence of length n
    t = np.zeros(n, dtype=float)
    for i in range(1, n):

- Old A[i, k] = 1*(1-h[i+k]) + h[i-k]*(1-1) = 1-h[i+k]
- Δc[k] = h[i-k] - (1-h[i+k])

And if we change h[j] from 0 to 1:
- New A[j, k] = 1*(1-h[j+k]) + h[j-k]*(1-1) = 1-h[j+k]
- Old A[j, k] = 0*(1-h[j+k]) + h[j-k]*(1-0) = h[j-k]
- Δc[k] = (1-h[j+k]) - h[j-k]

So for a swap (i→0, j→1):
Δc[k] = [h[i-k] - (1-h[i+k])] + [(1-h[j+k]) - h[j-k]]
     = h[i-k] + h[i+k] - 1 + 1 - h[j-k] - h[j+k]
     = h[i-k] + h[i+k] - h[j-k] - h[j+k]

(with appropriate bounds: h[x] = 0 if x < 0 or x >= n)

This is O(1) per k value! And only O(n) k values are affected (those where i±k or j±k is in range).

So for a single swap, computing Δc takes O(n) time. And we want to find the swap that minimizes max(c + Δc).

A greedy approach: focus on the k* that achieves the current max. We want to reduce c[k*]. The change in c[k*] for swap (i, j) is:
Δc[k*] = h[i-k*] + h[i+k*] - h[j-k*] - h[j+k*]

To reduce c[k*], we want Δc[k*] < 0, i.e., h[i-k*] + h[i+k*] < h[j-k*] + h[j+k*].

So we want i to be a position where its "neighbors at distance k*" are mostly 0, and j to be a position where its "neighbors at distance k*" are mostly 1.

This gives us an O(n) way to find candidate pairs:
- Score for position i (as the one to turn off): s_off(i) = h[i-k*] + h[i+k*] (want this small)
- Score for position j (as the one to turn on): s_on(j) = h[j-k*] + h[j+k*] (want this large)

Then the best swap is to pick i from ones with smallest s_off and j from zeros with largest s_on.

But we need to check that this actually reduces the max (not just c[k*]). So we should verify by computing the full Δc.

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def swap_delta(h, i, j):
    """Compute delta in correlation when h[i] goes 1->0 and h[j] goes 0->1"""
    n = len(h)
    lags = np.arange(-(n-1), n)
    # For i (1->0):
    # c[k] changes by: -(1-h[i+k]) + h[i-k] for valid k
    dk = np.zeros(2*n-1)
    valid_p = (i + lags >= 0) & (i + lags < n)
    dk[valid_p] -= (1.0 - h[i + lags[valid_p]])
    valid_m = (i - lags >= 0) & (i - lags < n)
    dk[valid_m] += h[i - lags[valid_m]]
    # For j (0->1):
    valid_p = (j + lags >= 0) & (j + lags < n)
    dk[valid_p] += (1.0 - h[j + lags[valid_p]])
    valid_m = (j - lags >= 0) & (j - lags < n)
    dk[valid_m] -= h[j - lags[valid_m]]
    return dk

def local_search(h, dx, time_limit=60, random_seed=42):
    n = len(h)
    rng = np.random.RandomState(random_seed)
    start = time.time()
    
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = np.max(c) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    no_improve_count = 0
    max_no_improve = 500
    
    while time.time() - start < time_limit and no_improve_count < max_no_improve:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a batch of random swaps
        n_tries = 100
        best_improvement = 1e-10
        best_swap = None
        
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            dk = swap_delta(h, i, j)
            new_c5 = np.max(c + dk) * dx
            improvement = current_c5 - new_c5
            if improvement > best_improvement:
                best_improvement = improvement
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = np.max(c) * dx
            no_improve_count = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            no_improve_count += 1
    
    return best_h, best_c5

Let me also think about the "steepest descent" approach more carefully. Instead of random swaps, I should try to systematically find the best swap.

**Systematic steepest descent:**
For each position i with h[i]=1, compute the "cost" of turning it off (i.e., how much it increases the max correlation).
For each position j with h[j]=0, compute the "benefit" of turning it on.

The cost of turning off position i:
- It removes the contribution h[i]*(1-h[i+k]) = 1-h[i+k] to c[k] for each valid k
- It adds the contribution h[i-k]*(1-h[i]) = h[i-k] to c[k] for each valid k
- Net: c[k] changes by h[i-k] - (1-h[i+k]) = h[i-k] + h[i+k] - 1

The benefit of turning on position j:
- It adds the contribution h[j]*(1-h[j+k]) = 1-h[j+k] to c[k] for each valid k
- It removes the contribution h[j-k]*(1-h[j]) = h[j-k] to c[k] for each valid k
- Net: c[k] changes by (1-h[j+k]) - h[j-k] = 1 - h[j+k] - h[j-k]

For a swap (i off, j on), the total delta is:
Δc[k] = (h[i-k] + h[i+k] - 1) + (1 - h[j+k] - h[j-k])
       = h[i-k] + h[i+k] - h[j-k] - h[j+k]

OK so this confirms my earlier derivation. Now, to find the best swap efficiently:

The key insight is that Δc[k] only depends on the values of h at positions i±k and j±k. For a fixed k, the "interaction" between i and j is:
Δc[k] = (h[i-k] + h[i+k]) - (h[j-k] + h[j+k])

So for a fixed k, the best swap is to minimize h[i-k] + h[i+k] (over i with h[i]=1) and maximize h[j-k] + h[j+k] (over j with h[j]=0).

But we need to minimize the MAX over all k, not just one k. So the best swap is one that simultaneously reduces c[k] for all large c[k] values.

A good heuristic: focus on the top few c[k] values (the "critical" shifts), and find a swap that reduces all of them.

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def swap_delta(h, i, j, lags):
    n = len(h)
    dk = np.zeros(len(lags))
    valid_p = (i + lags >= 0) & (i + lags < n)
    dk[valid_p] -= (1.0 - h[i + lags[valid_p]])
    valid_m = (i - lags >= 0) & (i - lags < n)
    dk[valid_m] += h[i - lags[valid_m]]
    valid_p = (j + lags >= 0) & (j + lags < n)
    dk[valid_p] += (1.0 - h[j + lags[valid_p]])
    valid_m = (j - lags >= 0) & (j - lags < n)
    dk[valid_m] -= h[j - lags[valid_m]]
    return dk

def thue_morse(n):
    h = np.zeros(n)
    m = 1
    while m < n:
        h[m:min(2*m, n)] = 1.0 - h[0:min(m, n)]
        m *= 2
    # Adjust to have exactly n//2 ones (for even n)
    target = n // 2
    ones = int(np.sum(h == 1))
    if ones > target:
        one_pos = np.where(h == 1)[0]
        h[one_pos[:ones-target]] = 0
    elif ones < target:
        zero_pos = np.where(h == 0)[0]
        h[zero_pos[:target-ones]] = 1
    return h

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

            i_minus = ones - k
            off_scores = np.where((i_plus >= 0) & (i_plus < n), h[np.clip(i_plus, 0, n-1)], 0.0) + \

                        np.where((j_minus >= 0) & (j_minus < n), h[np.clip(j_minus, 0, n-1)], 0.0)
            
            best_i = ones[np.argmin(off_scores)]

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_lags(n):
    return np.arange(-(n-1), n)

def swap_delta(h, i, j, lags):
    n = len(h)
    dk = np.zeros(len(lags))
    # i: 1 -> 0
    vp = (i + lags >= 0) & (i + lags < n)
    if np.any(vp):
        dk[vp] -= (1.0 - h[i + lags[vp]])
    vm = (i - lags >= 0) & (i - lags < n)
    if np.any(vm):
        dk[vm] += h[i - lags[vm]]
    # j: 0 -> 1
    vp = (j + lags >= 0) & (j + lags < n)
    if np.any(vp):
        dk[vp] += (1.0 - h[j + lags[vp]])
    vm = (j - lags >= 0) & (j - lags < n)
    if np.any(vm):
        dk[vm] -= h[j - lags[vm]]
    return dk

def thue_morse(n):
    h = np.zeros(n)
    m = 1
    while m < n:
        h[m:min(2*m, n)] = 1.0 - h[0:min(m, n)]
        m *= 2
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        pos = np.where(h == 1)[0]
        h[pos[:ones-target]] = 0.0
    elif ones < target:
        pos = np.where(h == 0)[0]
        h[pos[:target-ones]] = 1.0
    return h

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0

            ip = ones + k
