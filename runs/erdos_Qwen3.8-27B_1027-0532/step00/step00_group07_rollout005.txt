import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, n):
    return np.max(_corr(h)) * (2.0 / n)

def _proj(h, s):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        p = np.clip(h - m, 0.0, 1.0)
        if p.sum() > s:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def _grad_lag(k, h, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += 1.0 - h[k:]
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        j = -k
        g[j:] += 1.0 - h[:n-j]
        g[:n-j] -= h[j:]
    return g

def _optimize(h, n, ts, time_limit, rng):
    dx = 2.0 / n
    nl = 2*n - 1
    lags = np.arange(nl) - (n-1)
    t0 = time.time()

    best_c = _c5(h, n)
    
    lr = 0.01
    tau = 0.01
    bad = 0
    
    while time.time() - t0 < time_limit:
        c = _corr(h)
        mc = c.max()
        w = np.exp((c - mc) / tau)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(nl)
            w[c.argmax()] = 1.0
        else:
            w /= ws
        
        g = np.zeros(n)
        am = w > 1e-5
        idx = np.where(am)[0]
        for i in idx:
            g += w[i] * _grad_lag(lags[i], h, n)
        
        # Try a few step sizes
        best_step_c = best_c
        best_step_h = None
        for a in [lr, 0.5*lr, 2.0*lr]:
            hn = _proj(h - a*g, ts)
            cn = _c5(hn, n)
            if cn < best_step_c:
                best_step_c = cn
                best_step_h = hn

            best_h = best_step_h.copy()

        if bad > 40:

    t0 = time.time()
    best_h = h.copy()

        j = rng.integers(n)
        if i == j:

            trial[i] = new_i
            trial[j] = new_j
            c = _c5(trial, n)
            if c < best_c:
                best_c = c
                best_h = trial.copy()
                improved = True

                    if time.time() - t0 > time_limit:
                        break
                    new_i2 = best_h[i] + delta
                    new_j2 = best_h[j] - delta
                    if 0.0 <= new_i2 <= 1.0 and 0.0 <= new_j2 <= 1.0:

                        trial2[i] = new_i2
                        trial2[j] = new_j2

                        if c2 < best_c:
                            best_c = c2
                            best_h = trial2.copy()
                        else:

                    else:

        idx = np.where(am)[0]
        for i in idx:
            g += w[i] * _grad_lag(int(lags[i]), h, n)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        best_step_c = best_c + 1e-9
        best_step_h = None
        for a in [lr, 0.5 * lr, 0.25 * lr, 2.0 * lr]:
            hn = _proj(h - a * g, ts)
            cn = _c5(hn, n)
            if cn < best_step_c:
                best_step_c = cn
                best_step_h = hn
        if best_step_h is not None:
            best_c = best_step_c
            best_h = best_step_h.copy()
            h = best_step_h.copy()
            bad = 0
        else:
            bad += 1
            h = _proj(h - 0.5 * lr * g, ts)
        if bad > 60:
            break
        if bad > 20:
            lr *= 0.95
        tau = max(0.999 * tau, 0.002)
    return best_h, best_c

def _coordinate_polish(h, n, ts, time_limit, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c = _c5(h, n)
    n_iter = 0
    while time.time() - t0 < time_limit and n_iter < 50000:
        n_iter += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            j = (j + 1) % n
        delta = float(rng.uniform(-0.02, 0.02))
        old_i = best_h[i]
        old_j = best_h[j]
        new_i = old_i + delta
        new_j = old_j - delta
        if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
            trial = best_h.copy()
            trial[i] = new_i
            trial[j] = new_j
            c_try = _c5(trial, n)
            if c_try < best_c:
                best_c = c_try
                best_h = trial.copy()
    return best_h, best_c

def _swap_polish(h, n, ts, time_limit, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c = _c5(h, n)
    n_iter = 0
    while time.time() - t0 < time_limit and n_iter < 30000:
        n_iter += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            j = (j + 1) % n
        v_i = best_h[i]
        v_j = best_h[j]
        new_i = min(v_j, 1.0)
        new_j = min(v_i, 1.0)
        if abs(new_i - v_i) < 1e-10 and abs(new_j - v_j) < 1e-10:
            continue
        if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
            sum_change = (new_i + new_j) - (v_i + v_j)
            if abs(sum_change) < 1e-10:
                trial = best_h.copy()
                trial[i] = new_i
                trial[j] = new_j
                c_try = _c5(trial, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = trial.copy()
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()
    n = 95
    ts = n * 0.5
    dx = 2.0 / n

    h0 = np.array(initial_h_values, dtype=np.float64)
    h0 = _proj(h0, ts)

    best_h = h0.copy()
    best_c = _c5(h0, n)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Subgradient from initial
    if remaining() > 30:
        t_alloc = remaining() * 0.35
        best_h, best_c = _subgrad_optimize(h0, n, ts, t_alloc, rng)

    # Phase 2: Coordinate polish
    if remaining() > 10:
        t_alloc = min(remaining() * 0.2, 60.0)
        best_h, best_c = _coordinate_polish(best_h, n, ts, t_alloc, rng)

    # Phase 3: Multiple restarts
    n_restarts = 0
    while remaining() > 30 and n_restarts < 20:
        n_restarts += 1
        scale = 0.03 + 0.07 * rng.random()
        h_try = best_h + rng.standard_normal(n) * scale
        h_try = _proj(h_try, ts)
        t_alloc = min(remaining() * 0.3, 50.0)
        h_opt, c_opt = _subgrad_optimize(h_try, n, ts, t_alloc, rng)
        if remaining() > 10:
            h_opt, c_opt = _coordinate_polish(h_opt, n, ts, min(remaining() * 0.1, 20.0), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 4: More aggressive restarts with different structure
    if remaining() > 30:
        for restart in range(5):
            if remaining() < 15:
                break
            h_rand = rng.random(n)
            h_rand = _proj(h_rand, ts)
            t_alloc = min(remaining() * 0.2, 40.0)
            h_try, c_try = _subgrad_optimize(h_rand, n, ts, t_alloc, rng)
            if remaining() > 10:
                h_try, c_try = _coordinate_polish(h_try, n, ts, min(remaining() * 0.05, 15.0), rng)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()

    # Final polish
    if remaining() > 5:
        t_alloc = min(remaining() * 0.5, 30.0)
        best_h, best_c = _coordinate_polish(best_h, n, ts, t_alloc, rng)

    # Verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj(best_h, ts)
    best_c = _c5(best_h, n)

    return (best_h, best_c, n)
