import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project(h, n, max_iter=100):
    """Project h onto [0,1]^n ∩ {sum(h) = n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_method(h, n, max_iter, step_size, time_budget, start_time):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > time_budget:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        k_star = np.argmax(corr)
        # k_star is 0 to 2n-2, corresponding to lag k_star - (n-1)
        lag = k_star - (n - 1)
        
        # Subgradient: for each j, grad[j] = (1 - h[j+lag]) - h[j-lag]
        # (with appropriate boundary handling)
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            j_plus = j + lag
            if 0 <= j_plus < n:
                val += (1.0 - h[j_plus])
            j_minus = j - lag
            if 0 <= j_minus < n:
                val -= h[j_minus]
            grad[j] = val
        
        # Normalize and step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            h = h - step_size * grad / grad_norm
        
        # Project
        h = project(h, n)
        
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-7:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_f_values(h, n):
    """Compute F_k = sum_j h[j]*(1-h[j+k]) for all lags k"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # This gives F_k * dx for all k in "full" mode

def compute_c5(h, n):
    return np.max(compute_f_values(h, n))

def project_onto_constraint(h, n):
    """Project h onto [0,1]^n ∩ {sum(h) = n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_objective_and_gradient(h, n, t):
    """Compute smoothed max and its gradient"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # corr has 2n-1 entries, corresponding to lags -(n-1) to (n-1)
    
    # For numerical stability
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    obj = (max_corr + log_sum / t) * dx
    
    # Weights (softmax)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: weighted sum of dF_k/dh[j]
    grad = np.zeros(n)
    for idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            if lag > 0:
                grad[lag:] += -w * h[:n - lag]
        else:
            l = -lag
            grad[l:] += w * (1.0 - h[:n - l])
            grad[:n - l] += -w * h[l:]
    
    grad *= dx
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    max_corr = np.max(corr)

    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp

    grad = np.zeros(n)

        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)

        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]

    grad *= dx
    return obj, grad


def obj_wrapper(h, n, t):
    val, _ = smoothed_obj_and_grad(h, n, t)
    return val


def grad_wrapper(h, n, t):
    _, g = smoothed_obj_and_grad(h, n, t)
    return g


def constraint_fun(h, n):
    return np.sum(h) - n / 2.0


def constraint_jac(h, n):
    return np.ones(n)


def slsqp_optimize(h0, n, t, max_iter=500):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{
        'type': 'eq',
        'fun': constraint_fun,
        'jac': constraint_jac,
        'args': (n,)
    }]
    try:
        result = minimize(
            obj_wrapper, h0.copy(), method='SLSQP',
            jac=grad_wrapper, args=(n, t),
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}

        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()

    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.1

        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:

            grad[:n - lag] += -h[lag:]
        else:
            l = -lag

            grad[l:] += -h[:n - l]
        grad *= dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.995
    return best_h, best_c5
