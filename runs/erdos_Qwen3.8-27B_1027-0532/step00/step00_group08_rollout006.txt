    error = target_sum - np.sum(h_scaled)

    n = len(h)

def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        h = np.zeros(len(h))
        h[:int(target_sum)] = 1.0
        return h
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        err = target_sum - np.sum(h)
        if abs(err) < 1e-12:

        if err > 0:

            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            i = idx[np.argmax(h[idx])]
            h[i] = min(1.0, h[i] + err)
        else:

            if not np.any(mask):

            h[i] = max(0.0, h[i] + err)

    return float(val * dx)
