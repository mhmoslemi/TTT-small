    return float(np.max(corr))

def binary_init(n, rng):

                n_perturb = min(20, no_improve)
                for _ in range(n_perturb):
                    all_pos = np.where((h > 0.5) | (h > 0.01))[0]
                    all_neg = np.where((h < 0.5) | (h < 0.99))[0]
                    if len(all_pos) == 0 or len(all_neg) == 0:
                        break
                    ip = all_pos[rng.randint(len(all_pos))]
                    jn = all_neg[rng.randint(len(all_neg))]
                    if ip == jn:
                        continue
                    h[ip], h[jn] = h[jn], h[ip]
                    c = c + dx * (sig[jn] - sig[ip])
                    update_sig_row(sig, h, ip, s_all)
                    update_sig_row(sig, h, jn, s_all)
                no_improve = 0

        if time.time() - start > budget_s:
            break

    return best_h, best_c5

def quick_sa(h_init, budget_s, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    start = time.time()
    s_all = np.arange(-(n - 1), n)

    c = compute_auto(h)
    best_c5 = c5_from_auto(c)
    best_h = h.copy()
    sig = compute_sig(h, s_all)

    T = 0.015
    T_min = 0.0005

    while time.time() - start < budget_s:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = dx * (sig[j] - sig[i])
        c_new = c + delta
        c5_new = 1.0 - float(np.min(c_new))

        if c5_new < best_c5:
            h[i], h[j] = h[j], h[i]
            c = c_new
            best_c5 = c5_new
            best_h = h.copy()
            update_sig_row(sig, h, i, s_all)
            update_sig_row(sig, h, j, s_all)
        else:
            gap = c5_new - best_c5
            if gap < 0.5 and rng.random() < np.exp(-gap / T):
                h[i], h[j] = h[j], h[i]
                c = c_new
                update_sig_row(sig, h, i, s_all)
                update_sig_row(sig, h, j, s_all)

        T = max(T_min, T * 0.99999)

    return best_h, best_c5

def quadratic_residue_seq(p):
    n = p
    h = np.zeros(n)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == 1:
            h[a % n] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start = time.time()

    best_h = None
    best_c5 = 1.0

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        h_parent = np.clip(h_parent, 0, 1)
    except Exception:
        h_parent = None

    if h_parent is not None:
        c5_p = compute_c5(h_parent)
        best_c5 = c5_p
        best_h = h_parent.copy()

    n_values_to_try = []
    if h_parent is not None:
        n_values_to_try.append(len(h_parent))
    n_values_to_try.extend([96, 98, 128, 192, 193, 256])
    n_values_to_try = list(dict.fromkeys(n_values_to_try))

    total = budget_s
    n_experiments = len(n_values_to_try) + 3
    time_per = total / n_experiments

    for n_val in n_values_to_try:
        if time.time() - start > total - 30:
            break

        if n_val == len(h_parent) if h_parent is not None else False:
            h_start = h_parent
        else:
            h_start = binary_init(n_val, rng)

        remaining = total - (time.time() - start)
        sub_budget = min(time_per * 1.5, remaining * 0.3)
        if sub_budget < 5:
            sub_budget = 5

        h_res, c5_res = greedy_sa_optimize(h_start, sub_budget, rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Continue optimizing the best found
    if best_h is not None:
        remaining = total - (time.time() - start)
        if remaining > 20:
            h2, c5_2 = greedy_sa_optimize(best_h, remaining * 0.5, rng)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2

    # Try quadratic residue based construction
    if time.time() - start < total - 20:
        for p in [97, 89, 113]:
            if time.time() - start > total - 10:
                break
            h_qr = quadratic_residue_seq(p)
            n_qr = len(h_qr)
            target = n_qr / 2.0
            # Adjust to have correct sum
            order = np.argsort(-h_qr)
            h_qr_adj = np.zeros(n_qr)
            n_ones = int(target)
            h_qr_adj[order[:n_ones]] = 1.0
            rem = target - n_ones
            if rem > 1e-10:
                h_qr_adj[order[n_ones]] = rem

            remaining = total - (time.time() - start)
            sub_b = min(60, remaining * 0.2)
            if sub_b > 5:
                h_qr_opt, c5_qr = greedy_sa_optimize(h_qr_adj, sub_b, rng)
                if c5_qr < best_c5:
                    best_c5 = c5_qr
                    best_h = h_qr_opt

    # Final: quick SA from best to squeeze more
    if best_h is not None and time.time() - start < total - 10:
        remaining = total - (time.time() - start)
        h3, c5_3 = quick_sa(best_h, remaining, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    if best_h is None:
        n = 96
        best_h = binary_init(n, rng)
        best_c5 = compute_c5(best_h)

    n_final = len(best_h)
    best_h = np.clip(best_h, 0, 1)
    target_sum = n_final / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-10:
        best_h += diff / n_final
        best_h = np.clip(best_h, 0, 1)
        diff2 = target_sum - np.sum(best_h)
        if abs(diff2) > 1e-14:
            candidates = np.where((best_h > 1e-10) & (best_h < 1 - 1e-10))[0]
            if len(candidates) > 0:
                best_h[candidates[0]] = np.clip(best_h[candidates[0]] - diff2, 0, 1)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_final
