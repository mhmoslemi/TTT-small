    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(80)
        h0[::2] = 1.0

    n = len(h0)
    ts = n / 2.0
    _G_TARGET = ts

    best_h = _repair(h0, ts)
    best_c5 = _compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con}]

    def _elapsed():
        return time.time() - t0

    # Phase 1: SLSQP from initial
    rem = budget_s - _elapsed()
    if rem > 60:
        h_opt = _slsqp(h0, bounds, cons, 300, 1e-14)
        hc = _repair(h_opt, ts)
        c = _compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc

    # Phase 2: Log-sum-exp continuation
    rem = budget_s - _elapsed()
    for tval in [5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 500.0]:
        rem = budget_s - _elapsed()
        if rem < 20:
            break
        _G_T = tval
        try:
            res = minimize(_lse_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 150,
                                                      'ftol': 1e-12, 'disp': False})
            if res.x is not None:
                hc = _repair(res.x, ts)
                c = _compute_c5(hc)
                if c < best_c5:
                    best_c5, best_h = c, hc
        except Exception:
            pass

    # Phase 3: Random restarts
    for i in range(300):
        rem = budget_s - _elapsed()
        if rem < 8:
            break

        s = i % 9
        if s == 0:
            hs = best_h + rng.randn(n) * 0.01
        elif s == 1:
            hs = best_h + rng.randn(n) * 0.03
        elif s == 2:
            hs = best_h + rng.randn(n) * 0.08
        elif s == 3:
            hs = best_h + rng.randn(n) * 0.15
        elif s == 4:
            hs = best_h.copy()
            idx = rng.choice(n, size=rng.randint(5, 20), replace=False)
            hs[idx] = rng.rand(len(idx))
        elif s == 5:
            p = int(rng.choice([4, 5, 8, 10, 16, 20, 40]))
            period = rng.rand(p)
            hs = np.tile(period, n // p + 1)[:n]
        elif s == 6:
            hs = np.zeros(n)
            hs[rng.choice(n, size=40, replace=False)] = 1.0
        elif s == 7:
            hs = best_h.copy()
            bs = int(rng.randint(2, 6))
            nb = n // bs
            perm = rng.permutation(nb)
            for old_i, new_i in enumerate(perm):
                hs[new_i * bs:(new_i + 1) * bs] = best_h[old_i * bs:(old_i + 1) * bs]
        else:
            hs = best_h.copy()
            swap_size = int(rng.randint(3, min(15, n // 3)))
            i1 = int(rng.randint(0, n - swap_size))
            i2 = int(rng.randint(0, n - swap_size))
            if i1 != i2:
                seg1 = hs[i1:i1 + swap_size].copy()
                seg2 = hs[i2:i2 + swap_size].copy()
                hs[i1:i1 + swap_size] = seg2
                hs[i2:i2 + swap_size] = seg1

        hs = _repair(hs, ts)
        mi = min(200, max(20, int((rem - 3) / 0.12)))
        h_opt = _slsqp(hs, bounds, cons, mi, 1e-12)
        hc = _repair(h_opt, ts)
        c = _compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc

    # Phase 4: Coordinate descent refinement
    rem = budget_s - _elapsed()
    if rem > 60:
        best_h, best_c5 = _coordinate_descent(best_h, bounds, cons, ts,
                                              max_rounds=min(20, int(rem / 3)))

    # Phase 5: Final SLSQP refinement
    rem = budget_s - _elapsed()
    if rem > 30:
        h_opt = _slsqp(best_h, bounds, cons, 500, 1e-15)
        hc = _repair(h_opt, ts)
        c = _compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc

    # Final coordinate descent
    rem = budget_s - _elapsed()
    if rem > 30:
        best_h, best_c5 = _coordinate_descent(best_h, bounds, cons, ts,
                                              max_rounds=min(10, int(rem / 3)))

    return (best_h, best_c5, n)
