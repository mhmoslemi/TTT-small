import numpy as np
import time

def _project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    proj = np.clip(h - t, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _smoothed_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    return val

def _smoothed_grad(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= np.sum(weights)
    
    n = len(h)
    grad = np.zeros(n)
    # corr has 2n-1 entries, indexed 0 to 2n-2
    # corr[k] corresponds to shift s = k - (n-1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        # d(c_k)/d(h[j]) for all j
        # c_k = sum_{i in I_k} h[i]*(1-h[i+s])
        # where I_k = {i : 0<=i<n, 0<=i+s<n}
        # d/dh[j] = (1-h[j+s])*1_{j in I_k} - h[j-s]*1_{j-s in I_k}
        for j in range(n):
            term1 = 0.0
            term2 = 0.0
            if 0 <= j < n and 0 <= j + s < n:
                term1 = 1.0 - h[j + s]
            if 0 <= j - s < n and 0 <= j < n:
                term2 = h[j - s]
            grad[j] += w * (term1 - term2)
    
    return grad * dx
