```python
import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {x: 0≤x[i]≤1, sum(x)=target_sum}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

    overlaps = np.correlate(h, 1-h, mode='full') * dx
```

```python
        return np.clip(h - lam, 0.0, 1.0)

    lo, hi = -2.0, 2.0
    # Make sure the range is wide enough

        lo = -5.0

        hi = 5.0

        s = np.sum(do_project(mid))

        else:

def compute_c5(h):
    """Compute C5 = max_k overlap(k)."""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h, T=50.0):
    """Compute gradient of smooth max of overlaps using softmax approximation."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    overlaps = np.correlate(h, one_minus_h, mode='full') * dx
    
    # Softmax weights
    max_ov = np.max(overlaps)
    exp_ov = np.exp(T * (overlaps - max_ov))
    p = exp_ov / np.sum(exp_ov)
    
    # Gradient computation (vectorized over j, loop over s)
    grad = np.zeros(n)
    js = np.arange(n)
    for s in range(-(n-1), n):
        w = p[s + n - 1] * dx
        if w < 1e-10:
            continue
        js1 = js - s
        mask1 = (js1 >= 0) & (js1 < n)
        if np.any(mask1):
            grad[mask1] += w * one_minus_h[js1[mask1]]
        js2 = js + s
        mask2 = (js2 >= 0) & (js2 < n)
        if np.any(mask2):
            grad[mask2] -= w * h[js2[mask2]]
    
    return grad
```

**Improvement 1: Better gradient**
Instead of using a fixed T, I can use the exact subgradient of the max function. The subgradient of max(f_1, ..., f_m) at a point is the convex hull of the gradients of the active functions. A simple choice is the average gradient of the active functions (those with O(s) close to the max).

```python
def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    overlaps = np.correlate(h, one_minus_h, mode='full') * dx
    max_ov = np.max(overlaps)
    
    # Active set: shifts where overlap is within epsilon of max
    epsilon = 1e-4
    active = overlaps >= max_ov - epsilon
    
    # Average gradient over active set
    grad = np.zeros(n)
    js = np.arange(n)
    count = 0
    for s in range(-(n-1), n):
        if not active[s + n - 1]:
            continue
        w = dx
        js1 = js - s
        mask1 = (js1 >= 0) & (js1 < n)
        if np.any(mask1):
            grad[mask1] += w * one_minus_h[js1[mask1]]
        js2 = js + s
        mask2 = (js2 >= 0) & (js2 < n)
        if np.any(mask2):
            grad[mask2] -= w * h[js2[mask2]]
        count += 1
    
    if count > 0:
        grad /= count
    return grad
```
