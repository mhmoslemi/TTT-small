When I swap h[i] from 1 to 0 and h[j] from 0 to 1:
- The change in c_k for each k can be computed from the change in h.

Let's see: c_k = sum_m h[m] * (1-h[m+k-n+1]) (valid).

When h changes at positions i and j, only the terms involving i or j change.

For position i (h[i] changes from 1 to 0):
- Terms where m = i: h[i] changes from 1 to 0, so the term h[i]*(1-h[i+k-n+1]) changes from (1-h[i+k-n+1]) to 0. Change: -(1-h[i+k-n+1]) = h[i+k-n+1] - 1.
- Terms where m+k-n+1 = i, i.e., m = i-k+n-1: the term h[m]*(1-h[i]) changes from 0 (since h[i] was 1, so 1-h[i]=0) to h[m] (since h[i] is now 0, so 1-h[i]=1). Change: h[m] = h[i-k+n-1].

def _evaluate_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _make_binary(h, n, rng):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        rng.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target]:
            h_bin[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        rng.shuffle(zeros_idx)
        for idx in zeros_idx[:target - n_ones]:
            h_bin[idx] = 1
    return h_bin

def _steepest_descent(h, n, dx, time_limit, t_start):
    """Steepest descent local search on binary vectors."""
    h = h.copy()
    current_c5 = _evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    while time.time() - t_start < time_limit:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        
        best_gain = 1e-15
        best_i, best_j = -1, -1
        
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                c5 = _evaluate_c5(h, dx)
                gain = current_c5 - c5
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
                h[j] = 0
            h[i] = 1
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            current_c5 -= best_gain
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break

def _simulated_annealing(h, n, dx, time_limit, t_start, rng):
    """Simulated annealing on binary vectors."""
    h = h.copy()
    current_c5 = _evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    t_current = 1.0
    t_min = 0.01
    cooling = 0.999
    
    step = 0
    while time.time() - t_start < time_limit:
        step += 1
        # Random swap
        i = rng.choice(len(ones))
        j = rng.choice(len(zeros))
        pos_i = ones[i]
        pos_j = zeros[j]
        
        h[pos_i] = 0
        h[pos_j] = 1
        # Update ones and zeros lists
        ones.remove(pos_i)
        zeros.remove(pos_j)
        ones.append(pos_j)
        zeros.append(pos_i)
        
        new_c5 = _evaluate_c5(h, dx)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / (t_current * dx)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Undo
            h[pos_i] = 1
            h[pos_j] = 0
            ones.remove(pos_j)
            zeros.remove(pos_i)
            ones.append(pos_i)
            zeros.append(pos_j)
        
        if step % 100 == 0:
            t_current = max(t_min, t_current * cooling)
