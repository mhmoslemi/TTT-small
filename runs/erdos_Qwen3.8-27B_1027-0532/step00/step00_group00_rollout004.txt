    dx = 2.0 / n

    n = len(h)

        delta = (rng.random() - 0.5) * 0.15
        h[i] = old_i + delta
        h[j] = old_j - delta
        if h[i] < 0.0:
            h[j] += h[i]
            h[i] = 0.0
        elif h[i] > 1.0:
            h[j] -= (h[i] - 1.0)
            h[i] = 1.0
        if h[j] < 0.0:
            h[i] += h[j]
            h[j] = 0.0
        elif h[j] > 1.0:
            h[i] -= (h[j] - 1.0)
            h[j] = 1.0
        new_c5 = evaluate_c5_fast(h)
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
            no_improve = 0
        else:
            h[i] = old_i
            h[j] = old_j
            no_improve += 1
            if no_improve > 50000:
                break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = evaluate_c5_fast(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Continuous local search on initial
    if time_left() > 60:
        h, c5 = local_search_continuous(best_h.copy(), rng, 100000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Continuous SA on initial with multiple restarts
    for restart in range(5):
        if time_left() < 30:
            break
        h_perturbed = best_h + rng.normal(0, 0.02, size=n_init)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Fix sum
        current_sum = np.sum(h_perturbed)
        target_sum = n_init / 2.0
        if current_sum != target_sum:
            diff = current_sum - target_sum
            h_perturbed -= diff / n_init
            h_perturbed = np.clip(h_perturbed, 0, 1)
        n_sa = min(100000, max(5000, int(time_left() * 5000)))
        h, c5 = sa_continuous(h_perturbed, rng, n_sa, 0.005, 1e-7)
        h, c5 = local_search_continuous(h, rng, 30000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Binary SA for even n values
    n_values = [44, 46, 48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 78, 80, 82, 84, 86, 88, 90, 92, 94, 96, 98, 100]
    for n in n_values:
        if time_left() < 20:
            break
        # Create initial by interpolation
        x_init = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_init, h_init)
        target = n // 2
        h_bin = make_binary_from(h_interp, target, rng)

        n_sa = min(150000, max(5000, int(time_left() * 8000)))
        if n_sa < 1000:
            continue
        h, c5 = sa_binary(h_bin, rng, n_sa, 0.008, 1e-7)
        if time_left() > 10:
            h, c5 = local_search_binary(h, rng, 50000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Try binary SA on n_init if even, or n_init±1 if odd
    if n_init % 2 == 1:
        for n_adj in [n_init - 1, n_init + 1]:
            if time_left() < 15:
                break
            x_init = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n_adj, endpoint=False)
            h_interp = np.interp(x_new, x_init, h_init)
            target = n_adj // 2
            h_bin = make_binary_from(h_interp, target, rng)
            n_sa = min(200000, max(10000, int(time_left() * 10000)))
            h, c5 = sa_binary(h_bin, rng, n_sa, 0.008, 1e-7)
            if time_left() > 5:
                h, c5 = local_search_binary(h, rng, 50000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 5: Final polish
    if time_left() > 10:
        if len(best_h) % 2 == 0:
            h, c5 = local_search_binary(best_h.copy(), rng, 100000)
        else:
            h, c5 = local_search_continuous(best_h.copy(), rng, 100000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    n_final = len(best_h)
    return (best_h, best_c5, n_final)
