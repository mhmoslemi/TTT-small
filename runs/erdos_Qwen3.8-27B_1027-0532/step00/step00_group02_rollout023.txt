            h = np.clip(h, 0, 1)

import time

def calc_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_objective(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(p * (corr - max_c)))) / p

def smooth_gradient(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(p * (corr - max_c))
    w = exp_c / np.sum(exp_c)
    
    g = np.zeros(n)

            if 0 <= t + k < n:
                g[t] -= wi * h[t + k]

import time

def calc_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def pair_optimize(h, i, j, dx, n_iter=20):
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-13:
        return h, calc_c5(h, dx)
    
    a, b = t_lo, t_hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        ha = h.copy()
        ha[i] += m1
        ha[j] -= m1
        c1 = calc_c5(ha, dx)
        hb = h.copy()
        hb[i] += m2
        hb[j] -= m2
        c2 = calc_c5(hb, dx)
        if c1 < c2:
            b = m2
        else:
            a = m1
    
    t_opt = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] += t_opt
    h_new[j] -= t_opt
    return h_new, calc_c5(h_new, dx)
