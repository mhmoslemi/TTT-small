import numpy as np
import time

def compute_all_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_max_overlap(h, dx):
    return float(np.max(compute_all_overlaps(h, dx)))

def project_feasible(y, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, dx, temperature):
    N = len(h)
    c = compute_all_overlaps(h, dx)
    c_scaled = c * temperature
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    w_sum = exp_c.sum()
    weights = exp_c / w_sum
    smooth_val = (c_max + np.log(w_sum)) / temperature
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        s = idx - (N - 1)
        w = weights[idx] * dx
        if w < 1e-15:
            continue
        j_min = max(0, -s)
        j_max = min(N - 1, N - 1 - s)
        n = j_max - j_min + 1
        if n <= 0:
            continue
        grad[j_min:j_min+n] += w * (1.0 - h[j_min+s:j_min+s+n])
        ps = j_min + s
        grad[ps:ps+n] += w * (-h[j_min:j_min+n])
    return smooth_val, grad

def optimize_inner(h_init, dx, time_limit, target_sum, seed=0):

    N = len(h_init)
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c = compute_max_overlap(h, dx)
    temperature = 8.0
    lr = 0.008
    momentum = np.zeros(N)
    beta = 0.85
    stall = 0
    t_start = time.time()
    while time.time() - t_start < time_limit:
        smooth_val, grad = smooth_max_and_grad(h, dx, temperature)
        momentum = beta * momentum + (1.0 - beta) * grad
        gn = np.linalg.norm(momentum)
        if gn < 1e-14:
            break
        h_new = h - lr * (momentum / gn) * np.max(np.abs(momentum))
        h_new = project_feasible(h_new, target_sum)
        c_new = compute_max_overlap(h_new, dx)
        if c_new < best_c - 1e-14:
            h = h_new
            best_h = h.copy()
            best_c = c_new
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                lr *= 0.92
            if stall > 150:
                temperature = max(temperature * 0.88, 1.0)
                lr = 0.008
                h = best_h + 0.003 * rng.randn(N)
                h = project_feasible(h, target_sum)
                momentum = np.zeros(N)
                stall = 0
        if best_c < 0.385:
            temperature = max(temperature * 0.998, 1.0)
    return best_h, best_c

def upsample_to(h, N_new):
    N_old = len(h)
    if N_new == N_old:
        return h.copy()
    x_old = np.arange(N_old) * (2.0 / N_old)
    x_new = np.arange(N_new) * (2.0 / N_new)
    return np.interp(x_new, x_old, h)

def local_improve(h, dx, target_sum, time_limit, seed=0):
    rng = np.random.RandomState(seed)
    N = len(h)
    best_h = h.copy()

        if c_try < best_c - 1e-14:

            best_c = c_try
    return best_h, best_c
