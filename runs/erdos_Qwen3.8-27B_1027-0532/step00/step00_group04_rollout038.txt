def evaluate_c5(h):

def pair_swap_search(h, n_iter, rng, temperature=0.0, cooling=1.0):
    """Perform pair-swap local search"""
    n = len(h)
    h = h.copy()
    best_c5 = evaluate_c5(h)
    curr_c5 = best_c5
    dx = 2.0 / n
    target_sum = n / 2.0

        a = rng.integers(n)
        b = rng.integers(n)
        if a == b:
            continue
        
        # Determine max delta
        max_delta_to_b = 1.0 - h[b]  # max we can add to b
        max_delta_from_a = h[a]  # max we can take from a
        max_delta = min(max_delta_to_b, max_delta_from_a)
        
        if max_delta < 1e-10:
            continue

        delta = rng.uniform(0.01, 0.5) * max_delta
        
        # Try the swap
        old_a = h[a]
        old_b = h[b]
        h[a] = old_a - delta
        h[b] = old_b + delta
        
        new_c5 = evaluate_c5(h)
        
        # Accept or reject
        if new_c5 < curr_c5 or (temperature > 0 and rng.random() < np.exp((curr_c5 - new_c5) / temperature)):
            curr_c5 = new_c5
            if new_c5 < best_c5:

                best_h = h.copy()
        else:
            h[a] = old_a
            h[b] = old_b
        
        # Cool temperature
        temperature *= cooling
    
    return h, best_c5

def projected_gradient(h, n_iter, lr=0.01):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find active set (correlations near max)
        threshold = np.max(corr) - 0.01
        active = np.where(corr >= threshold)[0]
        
        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            shift = k - (n - 1)
            # For each i, compute contribution
            for i in range(n):
                j1 = i + shift  # for first factor
                j2 = i - shift  # for second factor
                if 0 <= j1 < n:
                    grad[i] += (1.0 - h[j1])
                if 0 <= j2 < n:
                    grad[i] -= h[j2]
        
        # Project gradient onto sum=0 subspace
        grad -= np.mean(grad)
        
        # Line search
        for _ in range(20):
            h_new = h - lr * grad
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            # Adjust sum
            diff = n/2.0 - np.sum(h_new)
            if abs(diff) > 1e-10:
                # Distribute adjustment
                if diff > 0:
                    room = 1.0 - h_new
                    total_room = np.sum(room)
                    if total_room > 0:
                        h_new += room * (diff / total_room)
                        h_new = np.clip(h_new, 0, 1)
                else:
                    room = h_new
                    total_room = np.sum(room)
                    if total_room > 0:
                        h_new += room * (diff / total_room)
                        h_new = np.clip(h_new, 0, 1)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < c5:
                h = h_new
                c5 = new_c5
                break
            lr *= 0.5
    
    return h

Given h of length n, corr = np.correlate(h, 1-h, 'full') has length 2n-1.
The k-th element (0-indexed) corresponds to shift s = k - (n-1).

The gradient of corr[k] with respect to h[i]:
- h[i] appears as first factor: contribution = (1-h[i+s]) if 0 ≤ i+s < n
- h[i] appears as second factor (inside 1-h): contribution = -h[i-s] if 0 ≤ i-s < n

So grad[k, i] = (1-h[i+s]) * 1[0≤i+s<n] - h[i-s] * 1[0≤i-s<n]

∂f/∂h[i] = dx * (1/|A|) * Σ_{k∈A} grad[k,i]
where A is the active set {k : corr[k] = max(corr)}.

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradient(h, top_k=5):
    """Compute subgradient of the max correlation objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Find active set (top k correlations)
    max_corr = np.max(corr)
    # Use softmax weighting for smoothness
    weights = np.exp(-10.0 * (max_corr - corr))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)  # shift
        # First factor: h[i] * (1-h[i+s])
        # grad contribution: (1-h[i+s]) for valid i+s
        if s >= 0:
            # i ranges 0 to n-1-s
            grad[:n-s] += weights[k] * (1.0 - h[s:])
        else:
            # i ranges -s to n-1
            grad[-s:] += weights[k] * (1.0 - h[:n+s])
        
        # Second factor: h[i] appears in (1-h[i+s]), i.e., i+s = j means i = j-s
        # grad contribution: -h[i-s] for valid i-s
        # Wait, let me re-derive.
        # corr[k] = sum_i h[i] * (1-h[i+s])
        # d/dh[j] of this = (1-h[j+s]) * 1[j+s valid] - h[j-s] * 1[j-s valid]
        # The first term is already handled above (it's the derivative w.r.t. h[j] when j is in the first factor)
        # The second term: when h[j] appears inside (1-h[...]), we need j = i+s for some i
        # i.e., i = j-s, and the term is h[j-s] * (1-h[j]), derivative w.r.t. h[j] is -h[j-s]
        # So for position j, the second factor contribution is -h[j-s] if 0 ≤ j-s < n
        
        if s >= 0:
            # j-s valid when j >= s, j ranges s to n-1
            grad[s:] -= weights[k] * h[:n-s]
        else:
            # j-s = j+|s| valid when j+|s| < n, j ranges 0 to n-1-|s|
            grad[:n+s] -= weights[k] * h[-s:]
    
    return grad * dx, max_corr * dx

def project_feasible(h, target_sum):
    """Project h onto {0≤h≤1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    if diff > 0:
        # Need to add 'diff' total
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room < 1e-10:
            return h
        alpha = diff / total_room
        h = np.clip(h + alpha * room, 0.0, 1.0)
    else:
        # Need to remove |diff| total
        room = h
        total_room = np.sum(room)
        if total_room < 1e-10:
            return h
        alpha = -diff / total_room
        h = np.clip(h - alpha * room, 0.0, 1.0)
    
    # Final adjustment for any residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        n = len(h)
        if residual > 0:
            mask = h < 1.0
            if np.any(mask):
                per = residual / np.sum(mask)
                h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
        else:
            mask = h > 0.0
            if np.any(mask):
                per = -residual / np.sum(mask)
                h[mask] = np.clip(h[mask] - per, 0.0, 1.0)
    
    return h

def gradient_descent(h, n_iter, lr_init=0.1, lr_decay=0.999, time_limit=None, t_start=None):
    """Projected gradient descent with line search"""
    n = len(h)
    target_sum = n / 2.0
    h = project_feasible(h.copy(), target_sum)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        grad, curr_c5 = compute_gradient(h)
        
        # Project gradient onto sum=0 subspace
        grad -= np.mean(grad)
        
        # Gradient step with line search
        improved = False
        for ls in range(15):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < curr_c5:
                h = h_new
                curr_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                break
            lr = max(lr * 0.5, 1e-6)
        
        if not improved:
            lr = max(lr * 0.5, 1e-6)
        
        lr = lr_init * (lr_decay ** it)
    
    return best_h, best_c5

def pair_swap_local(h, n_iter, rng, time_limit=None, t_start=None):
    """Pair-swap local search"""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    
    for it in range(n_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        a = rng.integers(n)
        b = rng.integers(n)
        if a == b:
            continue
        
        # Determine valid delta range
        max_add_b = 1.0 - h[b]
        max_sub_a = h[a]
        max_delta = min(max_add_b, max_sub_a)
        
        if max_delta < 1e-8:
            continue
        
        # Try a random delta
        delta = rng.uniform(0.05, 0.8) * max_delta
        
        old_a, old_b = h[a], h[b]
        h[a] -= delta
        h[b] += delta
        
        new_c5 = evaluate_c5(h)
        
        if new_c5 < curr_c5:
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = old_a
            h[b] = old_b
    
    return best_h, best_c5

def upscale(h, new_n):
    """Upscale h from current resolution to new_n using linear interpolation"""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n  # cell centers
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(x_new, x_old, h)
    return h_new

        total = np.sum(room)

"C₅ = max_k ∫ h(x)(1 - h(x+k)) dx"

And the evaluation is: "C₅ = max(np.correlate(h, 1-h, mode="full") * dx)"

So the correlation is between h and (1-h), and we take the max over all lags. The max includes both positive and negative lags (the full correlation).

One subtlety: at lag 0, corr[0] = Σ h[i] * (1-h[i]) = Σ h[i] - Σ h[i]². This is the "self-overlap" which is related to the L² norm.

For a 0-1 valued function with sum = n/2, corr[0] = n/2 (since h[i](1-h[i]) = 0 for 0-1 values). So C₅ ≥ 0 for 0-1 functions at lag 0.

For continuous values, corr[0] = Σ h[i] - Σ h[i]². To minimize this, we want h[i]² to be as large as possible, which means h should be concentrated (0-1 valued). But this conflicts with minimizing other lags.

    # Phase 4: Upscale
    time_left = budget_s - (time.time() - t_start)
    for new_n in [140, 280]:
        time_left = budget_s - (time.time() - t_start)
        if time_left < 50:
            break
        h_up = upscale(best_h, new_n)
        h_up = project_feasible(h_up, new_n / 2.0)
        mt = min(60.0, (budget_s - (time.time() - t_start)) * 0.4)
        h_up, c5_up = local_search_random(h_up, mt, t_start, rng)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = new_n

    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = evaluate_c5(best_h)
    return tuple(best_h.tolist()), final_c5, best_n
