def project_feasible(h, n):
    """Project h onto {x : 0 ≤ x_i ≤ 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection on tau
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Soft max
    scaled = corr / temperature
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    weights = exp_s / np.sum(exp_s)
    value = temperature * (max_s + np.log(np.sum(exp_s))) * dx
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-12:
            continue
        l = m - (n - 1)
        for j in range(n):
            g = 0.0
            jl = j - l
            if 0 <= jl < n:
                g += (1.0 - h[jl])
            jl2 = j + l
            if 0 <= jl2 < n:
                g -= h[jl2]
            grad[j] += weights[m] * g * dx
    
    return value, grad

import numpy as np
from scipy.optimize import minimize
import time

_params = {'n': 0, 'dx': 0.0, 'target_sum': 0.0, 'temp': 1.0}


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def soft_max_objective(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr / temp
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    return temp * (max_s + np.log(np.sum(exp_s))) * dx


def soft_max_grad(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr / temp
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        l = m - (n - 1)
        wm_dx = wm * dx
        for j in range(n):
            g = 0.0
            jl = j - l
            if 0 <= jl < n:
                g = 1.0 - h[jl]
            jl2 = j + l
            if 0 <= jl2 < n:
                g -= h[jl2]
            if g != 0.0:
                grad[j] += wm_dx * g
    return grad


def _soft_obj(h):
    return soft_max_objective(h, _params['n'], _params['dx'], _params['temp'])
