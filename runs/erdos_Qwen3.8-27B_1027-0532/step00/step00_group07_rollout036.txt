```python
def smooth_max_gradient(h, beta):
    N = len(h)
    dx = 2.0 / N
    corr = compute_correlations(h)
    max_corr = corr.max()
    shifted = np.clip(beta * (corr - max_corr), -400, 400)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = N - 1 - k
        if s >= 0:
            grad[:N-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:N-s]
        else:
            t = -s
            grad[t:] += w * (1.0 - h[:N-t])
            grad[:N-t] -= w * h[t:]
    grad *= dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1 - h, mode='full')

def c5_value(h):
    return float(np.max(compute_correlations(h)) * (2.0 / len(h)))

def smooth_max_gradient(h, beta):
    N = len(h)
    dx = 2.0 / N
    corr = compute_correlations(h)
    max_corr = corr.max()
    shifted = np.clip(beta * (corr - max_corr), -400.0, 400.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = N - 1 - k
        if s >= 0:
            grad[:N - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:N - s]
        else:
            t = -s
            grad[t:] += w * (1.0 - h[:N - t])
            grad[:N - t] -= w * h[t:]
    grad *= dx
    return grad

def project_feasible(p, target_sum):
    N = len(p)
    if target_sum <= 0:
        return np.zeros(N)
    if target_sum >= N:
        return np.ones(N)
    lo = p.min() - 1.0
    hi = 1.0 - p.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(p + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(p + (lo + hi) * 0.5, 0.0, 1.0)

def pgd_optimize(h, target_sum, rng, max_iters, deadline):
    N = len(h)
    best_h = h.copy()
    best_c5 = c5_value(h)
    beta = 50.0
    step_size = 0.01
    stagnation = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = smooth_max_gradient(h, beta)
        direction = -grad
        dn = np.linalg.norm(direction)
        if dn < 1e-13:
            break
        f_curr = c5_value(h)
        improved = False
        step = step_size
        for _ in range(30):
            h_new = project_feasible(h + step * direction, target_sum)
            f_new = c5_value(h_new)
            if f_new < f_curr - 1e-15:
                h = h_new
                f_curr = f_new
                improved = True
                break
            step *= 0.5
        if improved:
            if f_curr < best_c5 - 1e-15:
                best_c5 = f_curr
                best_h = h.copy()
            stagnation = 0
            step_size = min(step_size * 1.02, 0.05)
        else:
            step_size *= 0.5
            stagnation += 1
        if it % 200 == 0 and it > 0:
            corr = compute_correlations(h)
            top5 = np.sort(corr)[::-1][:5]
            gap = top5[0] - top5[-1]
            if gap < 0.002:
                beta = min(beta * 1.4, 500.0)
            elif gap > 0.02:
                beta = max(beta * 0.9, 15.0)
        if stagnation > 400:
            h = best_h + 0.01 * rng.standard_normal(N)
            h = project_feasible(h, target_sum)
            step_size = 0.02
            stagnation = 0
    return best_h, best_c5

def lbfgs_obj(x, N, target_sum, beta, M):
    corr = compute_correlations(x)
    mc = corr.max()
    sh = np.clip(beta * (corr - mc), -400.0, 400.0)
    e = np.exp(sh)
    return (2.0 / N) * (mc + np.log(e.sum()) / beta) + M * (x.sum() - target_sum) ** 2

def lbfgs_grad(x, N, target_sum, beta, M):
    g = smooth_max_gradient(x, beta)
    g += 2.0 * M * (x.sum() - target_sum)
    return g

def try_lbfgs(h, N, target_sum, deadline):
    beta = 120.0
    M = 500.0
    def obj(x):
        return lbfgs_obj(x, N, target_sum, beta, M)
    def grad(x):
        return lbfgs_grad(x, N, target_sum, beta, M)
    res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                   bounds=[(0.0, 1.0)] * N,
                   options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-10})
    if res.x is not None:
        h_res = project_feasible(res.x, target_sum)
        c5_res = c5_value(h_res)
        return h_res, c5_res
    return h, c5_value(h)

def snap_and_reoptimize(h, N, target_sum, rng, deadline, threshold=0.05):
    h_snap = h.copy()
    mask_zero = h_snap < threshold
    mask_one = h_snap > 1.0 - threshold
    h_snap[mask_zero] = 0.0
    h_snap[mask_one] = 1.0
    free = ~(mask_zero | mask_one)
    n_free = free.sum()
    if n_free == 0:
        return h, c5_value(h)
    deficit = target_sum - h_snap.sum()
    if deficit > 0:
        h_snap[free] = np.minimum(1.0, h_snap[free] + deficit / n_free)
    elif deficit < 0:
        h_snap[free] = np.maximum(0.0, h_snap[free] + deficit / n_free)
    h_snap = project_feasible(h_snap, target_sum)
    best_h, best_c5 = pgd_optimize(h_snap, target_sum, rng, 5000, deadline)
    return best_h, best_c5

    start_time = time.time()
    deadline = start_time + budget_s - 20
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=np.float64)
    N = len(h0)

    h = project_feasible(h0, target_sum)
    best_h, best_c5 = pgd_optimize(h, target_sum, rng, 80000, deadline)

    if time.time() < deadline:
        h_lb, c5_lb = try_lbfgs(best_h, N, target_sum, deadline)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    if time.time() < deadline:
        h_snap, c5_snap = snap_and_reoptimize(best_h, N, target_sum, rng, deadline)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()

    n_restarts = 20
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        if r < 5:
            h_start = best_h + 0.02 * rng.standard_normal(N)
        elif r < 10:
            h_start = (rng.random(N) > 0.5).astype(np.float64)
        elif r < 15:
            x = np.linspace(0.0, 2.0, N, endpoint=False)
            a1 = rng.random() * 5.0 + 1.0
            p1 = rng.random() * 2.0 * np.pi
            a2 = rng.random() * 3.0 + 0.5
            p2 = rng.random() * 2.0 * np.pi
            h_start = 0.5 + 0.35 * np.sin(a1 * x + p1) + 0.15 * np.cos(a2 * x + p2)
        else:
            h_start = best_h.copy()
            n_flip = max(1, N // 4)
            idx = rng.choice(N, size=n_flip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
        h_start = project_feasible(h_start, target_sum)
        local_h, local_c5 = pgd_optimize(h_start, target_sum, rng, 6000, deadline)
        if local_c5 < best_c5:
            best_c5 = local_c5
            best_h = local_h.copy()

    if time.time() < deadline - 15:
        h_lb2, c5_lb2 = try_lbfgs(best_h, N, target_sum, deadline)
        if c5_lb2 < best_c5:
            best_c5 = c5_lb2
            best_h = h_lb2.copy()

    return best_h, best_c5, N
```
