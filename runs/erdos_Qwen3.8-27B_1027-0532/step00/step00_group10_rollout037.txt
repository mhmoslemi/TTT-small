def compute_grad_vec(h, c, T, dx):
    n = len(h)
    c_max = np.max(c)
    exp_vals = np.exp(T * (c - c_max))
    weights = exp_vals / np.sum(exp_vals)
    
    # For each m, compute sum_i weights[i] * (1-h[m+k_i]) over valid i
    # and sum_i weights[i] * h[m-k_i] over valid i
    
    grad = np.zeros(n)
    for m in range(n):
        # First term: i from n-1-m to 2n-2-m (clamped to [0, 2n-2])
        i_lo = max(0, n - 1 - m)
        i_hi = min(2 * n - 2, 2 * n - 2 - m)
        if i_lo <= i_hi:
            # h indices: m + (i - (n-1)) for i in [i_lo, i_hi]
            # = m - n + 1 + i
            h_idx = m - n + 1 + i_lo  # starting index
            # h[h_idx : h_idx + (i_hi - i_lo + 1)]
            term1 = np.dot(weights[i_lo:i_hi+1], 1.0 - h[h_idx:h_idx + (i_hi - i_lo + 1)])
        else:
            term1 = 0.0
        
        # Second term: i from max(0, m-n+1) to min(2n-2, m)
        i_lo2 = max(0, m - n + 1)
        i_hi2 = min(2 * n - 2, m)
        if i_lo2 <= i_hi2:
            # h indices: m - (i - (n-1)) = m - i + n - 1 for i in [i_lo2, i_hi2]
            # When i = i_lo2: m - i_lo2 + n - 1
            # When i = i_hi2: m - i_hi2 + n - 1
            # These are in decreasing order as i increases
            h_idx_start = m - i_lo2 + n - 1
            h_idx_end = m - i_hi2 + n - 1
            # h[h_idx_end : h_idx_start + 1] is in the right order (decreasing i means increasing h index)
            # Wait: when i increases, h index decreases. So h_idx for i_lo2 is the largest, h_idx for i_hi2 is the smallest.
            # We need to reverse: h[h_idx_end : h_idx_start+1] gives indices h_idx_end, h_idx_end+1, ..., h_idx_start
            # which corresponds to i_hi2, i_hi2-1, ..., i_lo2
            # But we need weights[i_lo2], weights[i_lo2+1], ..., weights[i_hi2] to match h indices
            # weights[i_lo2] should match h[m - i_lo2 + n - 1] = h[h_idx_start]
            # weights[i_hi2] should match h[m - i_hi2 + n - 1] = h[h_idx_end]
            # So we need: dot(weights[i_lo2:i_hi2+1], h[h_idx_start:h_idx_end-1:-1])
            # Or equivalently: dot(weights[i_lo2:i_hi2+1], h[h_idx_end:h_idx_start+1][::-1])
            term2 = np.dot(weights[i_lo2:i_hi2+1], h[h_idx_end:h_idx_start+1][::-1])
        else:
            term2 = 0.0
        
        grad[m] = (term1 - term2) * dx / T
    
    return grad

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c), c
