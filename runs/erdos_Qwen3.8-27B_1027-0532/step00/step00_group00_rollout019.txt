def project_to_constraint(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            room = 1.0 - h[idx]
            add = min(room, remaining)
            h[idx] += add
            remaining -= add
            if remaining < 1e-12:
                break
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            sub = min(h[idx], remaining)
            h[idx] -= sub
            remaining -= sub
            if remaining < 1e-12:

        best_h = h
