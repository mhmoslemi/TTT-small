    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)

    active = np.where(corr >= np.max(corr) - 1e-10)[0]

    grad = np.zeros(n)
    for m in active:

        if remaining() < 40:
            break
        n_cur = best_n
        scale = 0.005 + 0.04 * rng.random()
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_feasible(h_pert, n_cur)
        mit = max(100, min(500, int(800000.0 / n_cur)))
        h_opt = do_slsqp(h_pert, n_cur, mit, 1e-14)
        c5_opt = compute_c5(h_opt, n_cur)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_cur

    # Phase 6: Try smooth max at best resolution if time allows
    if remaining() > 120:
        n_cur = best_n
        h_opt = do_smooth_slsqp(best_h, n_cur, 300, 80.0)
        c5_opt = compute_c5(h_opt, n_cur)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_cur

    return best_h, best_c5, best_n
