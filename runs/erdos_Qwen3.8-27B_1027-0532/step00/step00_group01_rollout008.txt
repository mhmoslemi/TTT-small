import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)

def smooth_max_value(c, beta):
    scaled = beta * c
    mx = np.max(scaled)
    return (mx + np.log(np.sum(np.exp(scaled - mx)))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c = corr * dx
    scaled = beta * c
    mx = np.max(scaled)
    w = np.exp(scaled - mx)
    s = np.sum(w)
    w = w / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wi = w[k] * dx
        for j in range(n):
            term = 0.0
            i1 = j - lag
            if 0 <= i1 < n:
                term += (1.0 - h[i1])
            i2 = j + lag
            if 0 <= i2 < n:
                term -= h[i2]
            grad[j] += wi * term
    return grad

def project_to_feasible(h, n):
    """Project h onto {0 <= h[i] <= 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum
        candidates = np.where(h < 1.0)[0]
    else:
        candidates = np.where(h > 0.0)[0]
    if len(candidates) == 0:
        return h
    adj = diff / len(candidates)
    h = h.copy()
    h[candidates] = np.clip(h[candidates] + adj, 0, 1)
    # May need another iteration
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
        if len(candidates) > 0:
            adj2 = diff2 / len(candidates)
            h[candidates] = np.clip(h[candidates] + adj2, 0, 1)
    return h

    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)

    def optimize_at_n(h_init, n, time_limit):

        h = h_init.copy()

                break
            
            def obj(x):
                corr = np.correlate(x, 1 - x, mode="full")
                return smooth_max_value(corr * (2.0 / n), beta)
            
            def grad(x):
                return smooth_max_grad(x, beta)
            
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0, 1)] * n,
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8})

            c5_new = compute_c5(h_new)
            if c5_new < best_local:
                h = h_new
                best_local = c5_new

    if c5_opt < c5_best:
        h_best = h_opt
        c5_best = c5_opt

    for n_try in [90, 126, 180, 252]:

                                  np.linspace(0, 2, n_best, endpoint=False), h_best)
            h_init = project_to_feasible(h_interp, n_try)
        else:
            h_init = np.array(initial_h_values, dtype=float).copy()

        if c5_opt2 < c5_best:

    def local_search(h, n, time_limit):
