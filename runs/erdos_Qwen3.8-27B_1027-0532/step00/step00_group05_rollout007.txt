import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def gradient_smoothed(h, beta):
    n = len(h)
    corr = compute_corr(h)
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    total = np.sum(exp_vals)
    if total < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = exp_vals / total
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-10:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            ln = -lag
            grad[:n-ln] += w * (1.0 - h[ln:])
            grad[ln:] -= w * h[:n-ln]
    return grad

def project_onto_constraint(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_proj) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def binary_local_search(h, n, iterations=10000, rng=None):
    """Local search on binary sequences (maintaining sum = n/2)"""
    # Convert to binary if not already
    h_bin = (h >= 0.5).astype(float)
    # Ensure sum = n/2
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    if len(ones) != n // 2:
        # Fix by swapping
        if len(ones) > n // 2:
            for i in range(len(ones) - n // 2):
                h_bin[ones[-1-i]] = 0.0
        else:
            for i in range(n // 2 - len(ones)):
                h_bin[zeros[-1-i]] = 1.0
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    for _ in range(iterations):
        if rng is None:
            i, j = np.random.randint(n, size=2)
        else:
            i, j = rng.randint(n, size=2)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        # Swap
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5 = compute_c5(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best_h, best_c5

def interpolate_h(h, n_new):
    """Interpolate h from length n to length n_new"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h)
    return h_new
