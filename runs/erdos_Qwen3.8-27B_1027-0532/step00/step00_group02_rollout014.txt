def _softmax_obj_and_grad(h, T):
    """Softmax approximation of max correlation, with gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = _compute_correlations(h) * dx  # length 2n-1

    max_c = np.max(corr)
    shifted = T * (corr - max_c)

    weights = exp_shifted / Z

        k = i - (n-1)  # lag
        grad += weights[i] * dx * _grad_corr_lag(h, k)
    return obj, grad

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

    n = len(h)
    return float(np.max(_corr(h)) * (2.0 / n))

def _grad_lag(h, k):
    """Gradient of corr at lag k w.r.t. h. Lag k means index (n-1+k) in full corr."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            # m in [k, n-1]: (1-h[m-k])
            g[k:] += 1.0 - h[:n-k]
            # m in [0, n-1-k]: -h[m+k]
            g[:n-k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            # m in [0, n-1-kk]: (1-h[m+kk])
            g[:n-kk] += 1.0 - h[kk:]
            # m in [kk, n-1]: -h[m-kk]
            g[kk:] -= h[:n-kk]
    return g

    """Project onto feasible set {0<=h<=1, sum=n/2}."""
    h = h.copy()

        h = np.clip(h, 0.0, 1.0)

            h[m] += d / m.sum()

            if not m.any():
                break
            h[m] += d / m.sum()

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * (2.0 / n))

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += 1.0 - h[:n-k]
            g[:n-k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            g[:n-kk] += 1.0 - h[kk:]
            g[kk:] -= h[:n-kk]
    return g

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            h[m] += d / m.sum()
        else:
            m = h > 1e-14
            if not m.any():
                break
            h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _soft_solve(h0, n, T_list, steps_per_T):
    h = h0.copy()
    best_c = _c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    for T in T_list:
        lr = 2.0 / (T + 1.0)
        for step in range(steps_per_T):
            c = _corr(h) * dx
            mc = c.max()
            e = np.exp(T * (c - mc))
            w = e / e.sum()
            g = np.zeros(n)
            # Only consider significant weights
            for i in range(n_corr):
                wi = w[i]
                if wi > 1e-8:
                    g += wi * _grad_lag(h, i - (n - 1))
            g *= dx
            h_new = _project(h - lr * g, n)
            v = _c5(h_new)
            if v < best_c:
                best_c = v
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def _slsqp(h0, n, maxiter=5000):
    dx = 2.0 / n
    def f(x):
        return _c5(x)
    def gf(x):
        c = _corr(x)
        im = int(c.argmax())
        return dx * _grad_lag(x, im - (n - 1))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,

                     options={'maxiter': maxiter, 'ftol': 1e-15})

        if c_new < _c5(h0):

    except Exception:
        pass
    return h0

def _interp_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)
