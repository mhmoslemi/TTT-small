            h_i = population[i]
            if is_feasible(h_i):
                corr = np.correlate(h_i, 1 - h_i, mode="full")
                c5_val_i = np.max(corr) * dx
                fitness_vals.append((i, c5_val_i))
            else:
                # If not feasible, score higher
                h_i_repaired = repair_solution(h_i)
                corr = np.correlate(h_i_repaired, 1 - h_i_repaired, mode="full")
                c5_val_i = np.max(corr) * dx * 2  # Penalize by doubling
                fitness_vals.append((i, c5_val_i))
        
        # Update best solution if improved
        for i, val in fitness_vals:
            if val < best_fitness:
                best_fitness = val
                best_h = population[i]
        
        # Local search on frequently occurring solutions
        freq_solutions = min(5, n_individuals)
        indices = np.random.choice(n_individuals, min(freq_solutions, n_individuals), replace=False)
        
        # Remove one worst quality winner to maintain diversity
        if best_fitness < 0.38:
            threshold = best_fitness * 1.05
            max_val = 0.5
            while max_val < threshold:
                idx_max = np.argmax(population[idx_max])
                max_val = np.random.rand() * 0.3 + 0.2
                # Scale if necessary
                population[idx_max] = local_search(population[idx_max])
                if is_feasible(population[idx_max]):
                    corr = np.correlate(population[idx_max], 1-population[idx_max], mode="full")
                    current_val = np.max(corr) * dx
                    if current_val < best_fitness:
                        best_fitness = current_val
                        best_h = population[idx_max]
        
        # Tournament selection for crossover
        new_population = []
        tournament_size = 3
        num_keep = 5  # Number elitists to keep
        
        # Select top 'num_keep' solutions
        fitness_argsort = np.argsort([val for _, val in fitness_vals])
        for i in range(num_keep):
            new_population.append(population[fitness_argsort[i]])
        
        # Apply tournament selection for offspring
        while len(new_population) < population_size:
            # Select two parents
            a_idx = np.random.randint(0, n_individuals)
            b_idx = np.random.randint(0, n_individuals)
            # Select better parent more likely
            prob_a = 0.5 * (1 + fitness_vals[a_idx][1] - np.min([fitness_vals[a_idx][1], fitness_vals[b_idx][1]])) / max(0.5, fitness_vals[a_idx][1])
            if prob_a < np.random.rand():
                par1 = population[a_idx]
            else:
                par1 = population[b_idx]
            
            par2 = population[np.random.randint(0, n_individuals)]
            # Create offspring via crossover
            offsprings = simplex_crossover(par1, par2)
            # Apply mutation with probability μ
            if np.random.rand() < 0.3:  # Mutation probability
                n_dim = n_points
                should_i_m = np.random.rand(n_dim)
                mutation = (should_i_m < 0.1).astype(int)
                mutation_strength = np.random.randn(n_dim) * 0.1
                mutation_strength = np.clip(mutation_strength, -0.5, 0.5)
                mutation_strength[mutation] = np.random.randn(np.sum(mutation))
                offsprings += mutation_strength * (47.5 / n_points)
                # Repair bounds
                offsprings = np.clip(offsprites, 0, 1)
                total_off = np.sum(offsprites)
                if abs(total_off - 47.5) > 1e-4:
                    offsprings *= (47.5 / total_off)
            
            new_population.append(offsprites)
        
        # Update population for next iteration
        population = new_population
        time_remaining -= (time.time() - start_time)
    
    # Final solution cleanup
    best_h_clean = repair_solution(best_h)
    corr_final = np.correlate(best_h_clean, 1-best_h_clean, mode="full")
    c5_final = np.max(corr_final) * dx
    
    return (best_h_clean, c5_final, n_points)
