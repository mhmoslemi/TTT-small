```python
import time

def compute_c5_correlation(h):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5, corr * dx

    """Repair constraints: 0 <= h <= 1, sum(h) = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s == 0:
        return np.full(n, 0.5)
    # Try to adjust
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # If still not matching, do iterative adjustment
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase: add to elements not at 1
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / np.sum(mask), 1.0)
        else:
            # Need to decrease: subtract from elements not at 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / np.sum(mask), 0.0)
    return h
```

3. Project back to the feasible set (0 ≤ h ≤ 1, sum(h) = n/2)

The projection is the tricky part. The feasible set is the intersection of the hypercube [0,1]^n with the hyperplane {h: sum(h) = n/2}. The projection onto this set can be done iteratively:
- Clip to [0,1]
- Adjust to satisfy sum constraint (add/subtract uniformly from free coordinates)
- Repeat

This is essentially what my `repair` function does.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_c5(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def smooth_c5_grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k_idx in range(2*n - 1):
            k = k_idx - (n - 1)
            hjk = h[j+k] if 0 <= j+k < n else 0.0
            hjmk = h[j-k] if 0 <= j-k < n else 0.0
            s += w[k_idx] * ((1.0 - hjk) - hjmk)
        grad[j] = s * dx
    return grad

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=np.float64)
    else:
        h_best = np.full(70, 0.5)
    
    n = len(h_best)
    best_c5 = compute_c5(h_best)
    
    # Phase 1: Gradient-based optimization
    deadline = t0 + 0.35 * budget_s
    h = h_best.copy()
    lr = 0.01
    T = 50.0
    
    while time.time() < deadline:
        g = smooth_c5_grad(h, T)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            h_best = h.copy()
            # Decrease T to make smoother
        else:
            lr *= 0.95
            if lr < 1e-8:
                break
    
    # Phase 2: Swap-based local search
    deadline = t0 + 0.6 * budget_s
    lr = 0.01
    while time.time() < deadline:
        h = h_best.copy()
        improved = False
        # Try random swaps
        for _ in range(2000):
            i, j = np.random.choice(n, 2, replace=False)
            if abs(h[i] - h[j]) < 1e-10:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                h_best = h.copy()
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    
    # Phase 3: Fine-tuning with small transfers
    deadline = t0 + 0.8 * budget_s
    while time.time() < deadline:
        h = h_best.copy()
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.uniform(0.005, 0.05) * np.sign(h[i] - h[j])
        if delta > 0:
            if h[i] < delta or h[j] > 1.0:
                continue
        else:
            if h[j] < -delta or h[i] > 1.0:
                continue
        h[i] -= delta
        h[j] += delta
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h_best = h.copy()
    
    # Phase 4: Try higher resolution
    deadline = t0 + 0.95 * budget_s
    for n_target in [140, 200]:
        if time.time() > deadline:
            break
        # Interpolate
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_target, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_feasible(h_interp, n_target)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp > best_c5:
            # The interpolation increased C5, try to fix it
            # Run gradient optimization at higher resolution
            h_high = h_interp.copy()
            lr = 0.005
            T = 30.0
            local_best = c5_interp
            phase_deadline = min(deadline, time.time() + 0.1 * budget_s)
            
            while time.time() < phase_deadline:
                g = smooth_c5_grad(h_high, T)
                h_new = h_high - lr * g
                h_new = project_feasible(h_new, n_target)
                c5_new = compute_c5(h_new)
                if c5_new < local_best:
                    h_high = h_new
                    local_best = c5_new
                    if local_best < best_c5:
                        best_c5 = local_best
                        h_best = h_high.copy()
                        n = n_target
                    lr *= 0.99
                else:
                    lr *= 0.9
    
    return h_best, best_c5, len(h_best)
```
