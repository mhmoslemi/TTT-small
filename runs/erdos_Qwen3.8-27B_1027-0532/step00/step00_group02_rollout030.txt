import numpy as np
from scipy.optimize import minimize, minimize_scalar
from functools import partial
import time

def obj_smoothed(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < 1e-30:
        return 0.0
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def obj_max(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def constraint_sum(h, target_sum):
    return np.sum(h) - target_sum

def constraint_jac(h, target_sum):
    return np.ones(len(h))

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def pair_obj(h_save, i, j, orig_i, orig_j, dx, d):
    h_save[i] = orig_i + d
    h_save[j] = orig_j - d
    return obj_max(h_save, dx)

def coord_descent_pair(h, dx, max_passes=200):
    n = len(h)
    best = h.copy()
    best_c5 = obj_max(h, dx)
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            j = (i + 1) % n
            orig_i = h[i]
            orig_j = h[j]
            lo = max(-orig_i, orig_j - 1.0)
            hi = min(1.0 - orig_i, orig_j)
            if hi - lo < 1e-14:
                continue
            pf = partial(pair_obj, h_save=h, i=i, j=j, orig_i=orig_i, orig_j=orig_j, dx=dx)
            res = minimize_scalar(pf, bounds=(lo, hi), method='bounded', options={'xatol': 1e-12})
            h[i] = np.clip(orig_i + res.x, 0.0, 1.0)
            h[j] = np.clip(orig_j - res.x, 0.0, 1.0)
            c5 = obj_max(h, dx)
            if c5 < best_c5 - 1e-13:
                best = h.copy()
                best_c5 = c5
                improved = True
            else:
                h[i] = orig_i
                h[j] = orig_j
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(47) * 0.5

    best_h = h_init.copy()
    best_c5 = obj_max(h_init, 2.0 / len(h_init))

    n_base = len(h_init)
    n_list = sorted(set(n for n in range(n_base - 8, n_base + 12) if n >= 16))

    for n in n_list:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 25:
            break

        dx = 2.0 / n
        target_sum = n / 2.0

        if n == len(best_h):
            h_start = best_h.astype(float).copy()
        else:
            x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            h_start = np.interp(x_new, x_old, best_h)

        h_start = fix_sum(h_start, target_sum)
        bounds = [(0.0, 1.0)] * n
        con_sum = partial(constraint_sum, target_sum=target_sum)
        jac_sum = partial(constraint_jac, target_sum=target_sum)

        for beta in [15, 30, 60, 120, 250]:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 10:
                break
            try:
                pf = partial(obj_smoothed, dx=dx, beta=beta)
                res = minimize(pf, h_start, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': con_sum, 'jac': jac_sum}],
                               options={'maxiter': 2000, 'ftol': 1e-16})
                if res.x is not None:
                    h_try = fix_sum(res.x, target_sum)
                    c5_try = obj_max(h_try, dx)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h_start = h_try.copy()
            except Exception:
                pass

        elapsed = time.time() - start_time
        if elapsed > budget_s - 15:
            break
        try:
            h_cd, c5_cd = coord_descent_pair(h_start, dx, max_passes=300)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h_start = h_cd.copy()
        except Exception:
            pass

        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        try:
            pm = partial(obj_max, dx=dx)
            res = minimize(pm, h_start, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.x is not None:
                h_try = fix_sum(res.x, target_sum)
                c5_try = obj_max(h_try, dx)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        except Exception:
            pass

        # Random perturbation restarts for current n
        for restart in range(3):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 15:
                break
            try:
                h_pert = best_h.astype(float).copy()
                if len(h_pert) != n:
                    x_old = (np.arange(len(h_pert)) + 0.5) * (2.0 / len(h_pert))
                    x_new = (np.arange(n) + 0.5) * (2.0 / n)
                    h_pert = np.interp(x_new, x_old, h_pert)
                h_pert += np.random.normal(0, 0.05, n)
                h_pert = fix_sum(h_pert, target_sum)
                for beta in [50, 150]:
                    elapsed = time.time() - start_time
                    if elapsed > budget_s - 8:
                        break
                    pf = partial(obj_smoothed, dx=dx, beta=beta)
                    res = minimize(pf, h_pert, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': con_sum, 'jac': jac_sum}],
                                   options={'maxiter': 1000, 'ftol': 1e-16})
                    if res.x is not None:
                        h_try = fix_sum(res.x, target_sum)
                        c5_try = obj_max(h_try, dx)
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h_start = h_try.copy()
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
