    return float(np.max(c) * _DX)

def _soft_c5(h, beta=60.0):
    c = np.correlate(h, 1.0 - h, mode="full") * _DX
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(beta * (c - m)))) / beta)

def _project(h):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = _SUM - np.sum(h)
    for _ in range(50):
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            if diff > 0:
                mask = np.ones(len(h), dtype=bool)
            else:
                mask = np.ones(len(h), dtype=bool)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
        diff = _SUM - np.sum(h)
    return h

def _g_sum(h):
    return np.sum(h) - _SUM

def _g_sum_jac(h):
    return np.ones_like(h)

def _pen_soft(h, beta=60.0):
    c = np.correlate(h, 1.0 - h, mode="full") * _DX
    m = np.max(c)
    obj = m + np.log(np.sum(np.exp(beta * (c - m)))) / beta
    obj += 20.0 * (np.sum(h) - _SUM) ** 2
    return obj

def run(seed=42, budget_s=1300, **kwargs):
    global _N, _DX, _SUM
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = None
    if 'initial_h_values' in kwargs:
        h_init = np.array(kwargs['initial_h_values'], dtype=float)
    else:
        try:
            h_init = np.array(initial_h_values, dtype=float)
        except NameError:
            pass

    if h_init is not None and len(h_init) > 0:
        _N = len(h_init)
        h0 = h_init.copy()
    else:
        _N = 86
        h0 = 0.5 * np.ones(_N)

    _DX = 2.0 / _N
    _SUM = _N / 2.0

    h0 = _project(h0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    constraints = [{'type': 'eq', 'fun': _g_sum, 'jac': _g_sum_jac}]
    bounds = [(0.0, 1.0)] * _N

    trial = 0
    while time.time() - t0 < budget_s - 40:
        trial += 1

        if trial == 1:
            x0 = best_h.copy()
        elif trial % 30 == 0:
            x0 = rng.uniform(0, 1, _N)
        elif trial % 10 == 0:
            s = rng.uniform(0.2, 0.6)
            x0 = best_h + s * rng.randn(_N)
        elif trial % 3 == 0:
            s = rng.uniform(0.05, 0.2)
            x0 = best_h + s * rng.randn(_N)
        else:
            s = rng.uniform(0.005, 0.08)
            x0 = best_h + s * rng.randn(_N)

        x0 = _project(x0)
        c5_x0 = _compute_c5(x0)
        if c5_x0 < best_c5:
            best_h = x0.copy()
            best_c5 = c5_x0

        # SLSQP
        try:
            res = minimize(_compute_c5, x0, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-12})
            if hasattr(res, 'fun') and res.fun < best_c5 - 1e-12:
                cand = _project(res.x)
                cv = _compute_c5(cand)
                if cv < best_c5:
                    best_h, best_c5 = cand, cv
        except Exception:
            pass

        # L-BFGS-B with penalized soft max
        if trial % 3 == 1:
            try:
                res2 = minimize(_pen_soft, x0, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 300, 'gtol': 1e-8})
                cand = _project(res2.x)
                cv = _compute_c5(cand)
                if cv < best_c5:
                    best_h, best_c5 = cand, cv
            except Exception:
                pass

        # Binarization attempt
        if trial % 12 == 6:
            for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
                bin_h = (best_h > thresh).astype(float)
                deficit = _SUM - np.sum(bin_h)
                if abs(deficit) < 1e-6:
                    pass
                elif deficit > 0:
                    zeros = np.where(bin_h == 0)[0]
                    if len(zeros) > 0:
                        order = zeros[np.argsort(-best_h[zeros])]
                        n_add = int(np.floor(deficit))
                        for i in range(min(n_add, len(order))):
                            bin_h[order[i]] = 1.0
                        frac = deficit - n_add
                        if frac > 0.01:
                            remaining = np.where(bin_h == 0)[0]
                            if len(remaining) > 0:
                                bin_h[remaining[0]] = frac
                elif deficit < 0:
                    ones = np.where(bin_h == 1)[0]
                    if len(ones) > 0:
                        order = ones[np.argsort(best_h[ones])]
                        n_rem = int(np.floor(-deficit))
                        for i in range(min(n_rem, len(order))):
                            bin_h[order[i]] = 0.0
                        frac = -deficit - n_rem
                        if frac > 0.01:
                            remaining = np.where(bin_h == 1)[0]
                            if len(remaining) > 0:
                                bin_h[remaining[0]] = 1.0 - frac
                bin_h = _project(bin_h)
                cv = _compute_c5(bin_h)
                if cv < best_c5:
                    best_h, best_c5 = bin_h, cv

        # Local polish with SLSQP from best
        if trial % 8 == 4 and time.time() - t0 < budget_s - 60:
            try:
                res3 = minimize(_compute_c5, best_h, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 1000, 'ftol': 1e-14})
                if hasattr(res3, 'fun') and res3.fun < best_c5 - 1e-14:
                    cand = _project(res3.x)
                    cv = _compute_c5(cand)
                    if cv < best_c5:
                        best_h, best_c5 = cand, cv
            except Exception:
                pass

    best_h = _project(best_h)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, _N
