from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    g = 1.0 - h
    return np.correlate(h, g, mode="full")

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        mask = (projected > 1e-10) & (projected < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            projected[mask] += diff / cnt
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def smoothed_grad(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    scaled = corr / temperature
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / np.sum(exp_scaled)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-12:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            k = n - lag
            grad[lag:] += p[m] * (1.0 - h[:k])
            grad[:k] -= p[m] * h[lag:]
        else:
            neg = -lag
            k = n - neg
            grad[:k] += p[m] * (1.0 - h[neg:])
            grad[neg:] -= p[m] * h[:k]
    
    return grad / temperature

    h_new = h - lr * grad
    h_new = project_feasible(h_new, target_sum)

            best_h = h_new.copy()
