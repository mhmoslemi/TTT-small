Now, the gradient of the soft max:
∇ soft_max(corr*dx, β) = ∑_{k=0}^{2N-2} p_k * ∇(corr[k] * dx)

where p_k = exp(β*(corr[k]*dx - m)) / ∑_j exp(β*(corr[j]*dx - m)), m = max(corr*dx)

And ∇(corr[k]*dx) is the vector with i-th component:
dx * [(1-h[i-s_k]) * valid1(i, s_k) - h[i+s_k] * valid2(i, s_k)]

where s_k = k - (N-1).

This is all O(N^2) to compute the gradient (for each of 2N-1 k values, we update N h values). Actually it's O(N * (2N-1)) = O(N^2) which is fine.

    shifted = np.minimum(vals - m, 0.0)
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

    N = len(h)

    # Softmax weights
    m = np.max(corr)
    shifted = np.minimum(corr - m, 0.0)
    exp_shifted = np.exp(beta * shifted)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_corr(h)) * dx

def soft_max(vals, beta):
    m = np.max(vals)
    shifted = np.minimum(vals - m, 0.0)
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def corr_gradient(h, dx, beta):
    N = len(h)
    n_corr = 2 * N - 1
    corr = compute_corr(h) * dx
    m = np.max(corr)
    shifted = np.minimum(corr - m, 0.0)
    exp_vals = np.exp(beta * shifted)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(N)
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-13:
            continue
        s = k - (N - 1)
        n_lo = max(0, k - N + 1)
        n_hi = min(N - 1, k)
        for n in range(n_lo, n_hi + 1):
            i = n
            j = n - s
            grad[i] += pk * dx * (1.0 - h[j])
            if j != i:
                grad[j] -= pk * dx * h[i]
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, dx, beta, target_sum):
    corr = compute_corr(h) * dx
    sm = soft_max(corr, beta)
    penalty = 100.0 * (np.sum(h) - target_sum) ** 2
    return sm + penalty

def smooth_grad(h, dx, beta, target_sum):
    g = corr_gradient(h, dx, beta)
    g += 200.0 * (np.sum(h) - target_sum)
    return g

def local_search(h, dx, target_sum, max_time):
    N = len(h)
    best_c5 = compute_c5(h, dx)
    start = time.time()
    improved = True
    while improved and time.time() - start < max_time:
        improved = False
        for i in range(N):
            for j in range(i + 1, N):
                for delta in [0.02, 0.01, 0.005, 0.001]:
                    if h[i] >= delta and h[j] + delta <= 1.0:
                        h_try = h.copy()
                        h_try[i] -= delta
                        h_try[j] += delta
                        c5_try = compute_c5(h_try, dx)
                        if c5_try < best_c5 - 1e-12:
                            h = h_try
                            best_c5 = c5_try

                    if h[j] >= delta and h[i] + delta <= 1.0:

                        h_try[j] -= delta
                        h_try[i] += delta

                        if c5_try < best_c5 - 1e-12:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
    return h, best_c5
