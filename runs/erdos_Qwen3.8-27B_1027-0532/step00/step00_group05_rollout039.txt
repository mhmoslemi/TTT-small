        grad = np.zeros(n)

def project_feasible(x, target):
    n = len(x)

    lo, hi = -5.0, 5.0

        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
