    return np.max(corr), corr

def project_feasible(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum"""
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust: add/subtract uniformly, then clip, iterate
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum: add to elements that are < 1
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to decrease sum: subtract from elements that are > 0
            mask = h > 1e-10
            if not np.any(mask):

            h[mask] += diff / np.sum(mask)

    max_val = np.max(corr)

    active_mask = corr >= threshold

    # Convert correlation indices to shifts

    active_shifts = active_indices - (n - 1)

    center = n

    h_ext[center:center+n] = h

    for s in active_shifts:

        j_range = np.arange(n)
        idx_minus = j_range + center - s

        h_plus[valid_plus] = h_ext[idx_plus[valid_plus]]

For the projection, I need to handle both the box constraint (0 ≤ h ≤ 1) and the equality constraint (sum = 48.5).

A good projection algorithm:
1. Clip to [0, 1]
2. While sum ≠ target:
   a. If sum > target: find elements that can be decreased (h > 0) and decrease them
   b. If sum < target: find elements that can be increased (h < 1) and increase them
   c. Use a "water filling" approach: decrease/increase by the min(amount needed, headroom)

Actually, a cleaner way: use the Euclidean projection onto the intersection of [0,1]^n and {x: sum(x) = target}.

The projection can be computed as:
h_proj = clip(h - lambda, 0, 1) for some lambda chosen so that sum(clip(h - lambda, 0, 1)) = target.

This is a 1D root-finding problem! We can use bisection:
- If h >= target for all + sum of clipped = target... 
- Actually: let f(lambda) = sum(clip(h_i - lambda, 0, 1))
- f is a decreasing function of lambda
- f(+inf) = 0, f(-inf) = n
- We want f(lambda) = target

Since target = n/2, and h is in [0,1], we can always find such a lambda.

def project_sum(h, target_sum):
    """Project h onto {x in [0,1]^n : sum(x) = target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo = np.min(h) - 1
    hi = np.max(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Final adjustment for numerical precision
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Adjust one element
        if diff > 0:
            idx = np.argmax(1 - h_proj)
            h_proj[idx] += min(diff, 1 - h_proj[idx])
        else:
            idx = np.argmin(h_proj)
            h_proj[idx] += max(diff, -h_proj[idx])
    return h_proj

    weights = np.exp(scaled)

        if w < 1e-8:

        idx_p = j_range + center + s

        h_p[vp] = h_ext[idx_p[vp]]
        grad += w * ((1.0 - h_m) - h_p)
    
    grad *= dx

        if c5_try < c5_cur:

        alpha *= 0.5
        if alpha < 1e-8:

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr


def project_sum(h, target_sum):
    n = len(h)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = float(np.sum(h_proj))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - float(np.sum(h_proj))
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h_proj))
            h_proj[idx] = min(1.0, h_proj[idx] + diff)
        else:
            idx = int(np.argmin(h_proj))
            h_proj[idx] = max(0.0, h_proj[idx] + diff)
    return h_proj


def compute_smooth_grad(h, corr, dx, tau):
    n = len(h)
    scaled = corr / tau
    scaled -= float(np.max(scaled))
    weights = np.exp(scaled)
    weights /= float(np.sum(weights))

    threshold = 1e-8
    active = weights > threshold
    if not np.any(active):
        return np.zeros(n)

    active_weights = weights[active]
    active_indices = np.where(active)[0]
    shifts = active_indices - (n - 1)

    ext_size = 3 * n
    center = n
    h_ext = np.zeros(ext_size)
    h_ext[center:center + n] = h

    grad = np.zeros(n)
    j_range = np.arange(n)

    for w, s in zip(active_weights, shifts):
        idx_m = j_range + center - s
        idx_p = j_range + center + s
        h_m = np.zeros(n)
        h_p = np.zeros(n)
        vm = (idx_m >= 0) & (idx_m < ext_size)
        vp = (idx_p >= 0) & (idx_p < ext_size)
        h_m[vm] = h_ext[idx_m[vm]]
        h_p[vp] = h_ext[idx_p[vp]]
        grad += w * ((1.0 - h_m) - h_p)

    grad *= dx
    return grad


def binary_rounding(h_cont, target_sum, n_points):
    best = h_cont.copy()
    best_c5, _ = compute_c5(h_cont)
    order = np.argsort(h_cont)

    k_ones = 48
    h_bin = np.zeros(n_points)
    h_bin[order[n_points - k_ones:]] = 1.0
    h_bin[order[n_points - k_ones - 1]] = 0.5
    c5_bin, _ = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best = h_bin.copy()

    for offset in range(1, 6):
        if n_points - k_ones - 1 - offset < 0:
            break
        h_try = h_bin.copy()
        h_try[order[n_points - k_ones - 1]] = 0.0
        h_try[order[n_points - k_ones - 1 - offset]] = 0.5
        h_try[order[n_points - k_ones - offset]] = 0.0
        h_try[order[n_points - k_ones - 1 - offset]] = 0.5
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best = h_try.copy()

    for offset in range(1, 6):
        idx_top = n_points - k_ones + offset
        if idx_top >= n_points:
            break
        h_try = h_bin.copy()
        h_try[order[n_points - k_ones - 1]] = 0.0
        h_try[order[idx_top]] = 0.0
        h_try[order[n_points - k_ones - 1]] = 0.5
        h_try[order[idx_top - 1]] = 0.5
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best = h_try.copy()

    return best, best_c5


def local_refine(h, target_sum, n_points, dx, budget_sec=30.0):
    start = time.time()
    best = h.copy()
    best_c5, _ = compute_c5(h)
    tau = 0.002
    lr = 0.005

    while time.time() - start < budget_sec:
        c5, corr = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best = h.copy()

        grad = compute_smooth_grad(h, corr, dx, tau)
        grad -= np.mean(grad)

        h_new = project_sum(h - lr * grad, target_sum)
        c5_new, _ = compute_c5(h_new)

        if c5_new < c5:
            h = h_new
            lr = min(lr * 1.05, 0.02)
        else:
            lr *= 0.8
            if lr < 1e-4:
                break

    return best, best_c5
