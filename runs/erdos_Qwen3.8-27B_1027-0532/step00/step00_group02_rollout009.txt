```python
def project_to_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - target_sum - 1.0
    hi = h.max() + target_sum + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-10
            if mask.any():
                idx = np.where(mask)[0]
                room = 1.0 - x[idx]
                order = np.argsort(-room)
                remaining = diff
                for i in order:
                    add = min(remaining, room[i])
                    x[idx[i]] += add
                    remaining -= add
                    if remaining <= 1e-15:
```

Let me rewrite the code with this:

```python
def c5_and_gradient(h, dx):
    """Compute C5 and its gradient with respect to h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    c5 = float(corr[idx] * dx)
    k = idx - (n - 1)
    
    grad = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            grad[j] += (1.0 - h[j - k])
        for j in range(0, n - k):
            grad[j] += (-h[j + k])
    else:
        kp = -k
        for j in range(0, n - kp):
            grad[j] += (1.0 - h[j + kp])
        for j in range(kp, n):
            grad[j] += (-h[j - kp])
    
    return c5, grad * dx
```

This is O(n) to compute the gradient (plus O(n^2) for the correlation itself to find the argmax). Actually, the correlation computation is O(n^2) for the full mode, but numpy optimizes it, so it's fast.

```python
        c5, _ = c5_and_gradient(x, dx)
        return c5

        return g

        result = minimize(
            obj, h, jac=grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.fun < obj(h):
            h_opt = project_to_feasible(result.x, target_sum)
            c5_opt = compute_c5(h_opt, dx)
            return h_opt, c5_opt
    except Exception:
        pass
    return h, compute_c5(h, dx)
```

def project_to_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - target_sum - 1.0
    hi = h.max() + target_sum + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-10
            if mask.any():
                idx = np.where(mask)[0]
                room = 1.0 - x[idx]
                order = np.argsort(-room)
                remaining = diff
                for i in order:
                    add = min(remaining, room[i])
                    x[idx[i]] += add
                    remaining -= add
                    if remaining <= 1e-15:
                        break
        else:
            mask = x > 1e-10
            if mask.any():

                remaining = -diff
                for i in order:
                    rem = min(remaining, space[i])
                    x[idx[i]] -= rem
                    remaining -= rem
                    if remaining <= 1e-15:
                        break
    return x

def local_search_continuous(h, target_sum, dx, n_iter, rng, T_init=0.01):
