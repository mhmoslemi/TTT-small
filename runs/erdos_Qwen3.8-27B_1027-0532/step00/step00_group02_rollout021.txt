def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(2000):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask].sum()
            if room < 1e-15:
                break
            scale = diff / room
            h[mask] *= (1.0 - scale)
        else:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = (1.0 - h[mask]).sum()
            if room < 1e-15:

            scale = (-diff) / room
            h[mask] += scale * (1.0 - h[mask])
    h = np.clip(h, 0.0, 1.0)
