    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def _project_simplex(x, target):
    lo, hi = -5.0, 6.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute difference
    if diff > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room < diff:
            return h  # infeasible
        h = h + diff * room / total_room
    else:
        room = h
        total_room = room.sum()
        if total_room < -diff:
            return h
        h = h + diff * room / total_room

    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(h < 1.0)) if diff > 0 else int(np.argmax(h > 0.0))
        h[idx] += diff if abs(diff) < min(h[idx] if diff < 0 else 1.0-h[idx], abs(diff)) else 0
