    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        req = n_init / 2.0
        h_init = project_onto_constraints(h_init, req)
        c5 = evaluate_c5(h_init)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_init.copy()
    except Exception:
        h_init = None

    elapsed = time.time() - start_time
    t_phase1 = min(budget_s * 0.45, max(30, budget_s - 60))
    remaining_p1 = max(5, t_phase1 - elapsed)

    if h_init is not None:
        h_opt, c5_opt = optimize_phase(h_init, n_init / 2.0, remaining_p1, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    t_phase2_budget = budget_s * 0.25
    remaining_p2 = max(5, t_phase2_budget - (elapsed - start_time) + (start_time - start_time))
    remaining_p2 = max(5, (t_phase1 + t_phase2_budget) - (time.time() - start_time))

    n_trials = [63, 62, 64, 61, 65, 66, 68, 72, 80, 60]
    if h_init is not None and n_init not in n_trials:
        n_trials.insert(0, n_init)

    for n in n_trials:
        elapsed = time.time() - start_time
        if elapsed > t_phase1 + t_phase2_budget:
            break
        remaining = (t_phase1 + t_phase2_budget) - elapsed
        if remaining < 10:
            break

        n_starts = 2
        time_per_start = remaining / n_starts
        if time_per_start < 5:
            time_per_start = 5
            n_starts = 1

        for attempt in range(n_starts):
            elapsed = time.time() - start_time
            if elapsed > t_phase1 + t_phase2_budget:
                break

            if n == (n_init if h_init is not None else -1) and attempt == 0:
                h_try = h_init.copy()
            elif attempt == 0:
                h_try = make_binary_h(n)
            elif attempt == 1:
                h_try = make_alternating_h(n)
            else:
                h_try = rng.uniform(0, 1, size=n)

            req = n / 2.0
            h_try = project_onto_constraints(h_try, req)
            c5_try = evaluate_c5(h_try)

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

            time_limit = min(time_per_start, remaining - 5)
            if time_limit > 3:
                h_opt, c5_opt = optimize_phase(h_try, req, time_limit, np.random.RandomState(seed + n * 100 + attempt))
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    elapsed = time.time() - start_time
    remaining = max(5, budget_s - 10 - elapsed)

    if remaining > 20:
        n_best = len(best_h)
        req = n_best / 2.0
        h_final, c5_final = optimize_phase(best_h, req, remaining * 0.5, np.random.RandomState(seed + 9999))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

        elapsed = time.time() - start_time
        remaining2 = max(5, budget_s - 10 - elapsed)
        if remaining2 > 10:
            h_cd, c5_cd = coordinate_descent(best_h, req, remaining2, np.random.RandomState(seed + 7777))
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    n_final = len(best_h)
    best_h = project_onto_constraints(best_h, n_final / 2.0)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_final)
