def project_h(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_h(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    js = np.arange(n)
    for k in active:
        l = k - (n - 1)
        idx1 = js - l
        m1 = (idx1 >= 0) & (idx1 < n)
        grad[m1] += (1.0 - h[idx1[m1]])
        idx2 = js + l
        m2 = (idx2 >= 0) & (idx2 < n)
        grad[m2] += -h[idx2[m2]]
    return grad / len(active)

def cd_sweep(h, target_sum, order=None):
    n = len(h)
    lags = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    
    if order is None:
        order = list(range(n))
    
    for j in order:
        jl = j - lags
        m1 = (jl >= 0) & (jl < n)
        jpl = j + lags
        m2 = (jpl >= 0) & (jpl < n)
        
        a = np.zeros(2 * n - 1)
        if np.any(m1):
            a[m1] += (1.0 - h[jl[m1]])
        if np.any(m2):
            a[m2] += -h[jpl[m2]]
