        return h

    if abs(diff) < 1e-10:

    h = np.clip(h, 0, 1)

    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        if not np.any(mask):

        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)

            break
