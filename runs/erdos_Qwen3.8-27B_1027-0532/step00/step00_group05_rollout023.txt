def binary_local_search(h_init, dx, target_sum, budget_s, start_time):
    n = len(h_init)
    # Convert to binary
    h = (h_init >= 0.5).astype(float)
    # Adjust to get exactly target_sum ones
    ones = h.sum()
    if ones > target_sum:
        # Remove excess ones (from positions where h is closest to 0.5)
        indices = np.where(h == 1)[0]
        # Sort by h_init value (closest to 0 first)
        order = np.argsort(h_init[indices])
        for i in range(int(ones - target_sum)):
            h[indices[order[i]]] = 0
    elif ones < target_sum:
        indices = np.where(h == 0)[0]
        order = np.argsort(-h_init[indices])
        for i in range(int(target_sum - ones)):
            h[indices[order[i]]] = 1
    
    dx = 2.0 / n
    best_val = compute_c5(h, dx)
    best_h = h.copy()
    
    # Hill climbing with random restarts
    np.random.seed(42)
    for it in range(1000000):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Try a random swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h[i] == h[j]:

        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h, dx)
        if new_val >= best_val:
            h[i], h[j] = h[j], h[i]  # Undo
        else:
            best_val = new_val
            best_h = h.copy()
    
    return best_h, best_val

1. Fix h, compute the "effective" linear terms in the bilinear products
2. Solve the resulting linear program
3. Iterate

- We have h^t
- The constraint f_k(h) ≤ T becomes: sum_i h[i+k] - sum_i h[i+k]*h[i] ≤ T
- Linearize: sum_i h[i+k] - sum_i (h[i+k]*h^t[i] + h^t[i+k]*h[i] - h^t[i+k]*h^t[i]) ≤ T
