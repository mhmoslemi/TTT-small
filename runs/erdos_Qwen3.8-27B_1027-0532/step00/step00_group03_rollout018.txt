```python
def find_1d_optimum(a, b, t_min=0.0, t_max=1.0):
    """Find t in [t_min, t_max] that minimizes max_m (a[m]*t + b[m])"""
    n_lines = len(a)
    
    # Evaluate at boundaries
    best_t = t_min
    best_val = np.max(a * t_min + b)
    
    val_at_max = np.max(a * t_max + b)
    if val_at_max < best_val:
        best_t = t_max
        best_val = val_at_max
    
    # Find all intersection points in (t_min, t_max)
    # Line m1 and m2 intersect where a[m1]*t + b[m1] = a[m2]*t + b[m2]
    # t = (b[m2] - b[m1]) / (a[m1] - a[m2])
    
    # For efficiency, only consider intersections where both lines could be on the upper envelope
    # But for correctness, let's check all pairs (might be slow for large n)
    
    # Actually, for n up to ~400 lags, O(n^2) is fine
    for i in range(n_lines):
        for j in range(i+1, n_lines):
            da = a[i] - a[j]
            if abs(da) < 1e-15:
                continue
            t = (b[j] - b[i]) / da
            if t_min < t < t_max:
                val = np.max(a * t + b)
                if val < best_val:
                    best_val = val
                    best_t = t
    
    return best_t
```

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_box_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    if h.min() >= 0 and h.max() <= 1 and abs(h.sum() - target_sum) < 1e-12:
        return h.copy()
    # Bisection on the Lagrange multiplier
    lo = -1.0
    hi = 1.0
    for _ in range(200):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def solve_1d_subproblem(a, b):
    """Find t in [0,1] minimizing max_m(a[m]*t + b[m])"""
    n = len(a)
    
    # Evaluate at boundaries
    val0 = np.max(b)
    val1 = np.max(a + b)
    
    best_t = 0.0 if val0 <= val1 else 1.0
    best_val = min(val0, val1)
    
    # Check if the function is decreasing at 0 or increasing at 1
    # If min(b) < max(b), the function might have an interior minimum
    
    # For efficiency with large n, sample a few points
    # and use those as starting points for refinement
    t_samples = np.linspace(0, 1, 50)
    vals = np.max(a[:, None] * t_samples[None, :] + b[:, None], axis=0)
    idx = np.argmin(vals)
    
    if vals[idx] < best_val:
        best_val = vals[idx]
        best_t = t_samples[idx]
    
    # Refine around the best point
    if 0 < idx < 49:
        t_lo = t_samples[idx-1]
        t_hi = t_samples[idx+1]
        # Fine grid
        t_fine = np.linspace(t_lo, t_hi, 200)
        vals_fine = np.max(a[:, None] * t_fine[None, :] + b[:, None], axis=0)
        idx_fine = np.argmin(vals_fine)
        if vals_fine[idx_fine] < best_val:
            best_val = vals_fine[idx_fine]
            best_t = t_fine[idx_fine]
    
    return best_t

def coordinate_descent(h, max_passes=200, tol=1e-12):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    
    for p in range(max_passes):
        max_improvement = 0.0
        for i in range(n):
            # Save current value
            old_val = h[i]
            
            # Compute a and b for the 1D subproblem
            # corr[m] = sum_j h[j+L]*(1-h[j]) where L = m-(n-1)
            # Terms with h[i]:
            #   h[i]*(1-h[i-L]) if i-L in [0,n): coefficient (1-h[i-L])
            #   h[i+L]*(1-h[i]) if i+L in [0,n): coefficient -h[i+L]
            
            # Full correlation has indices m = 0..2n-2, lag L = m-(n-1)
            # So L ranges from -(n-1) to n-1
            
            # For each lag L:
            # a_L = (1-h[i-L]) * I(0<=i-L<n) - h[i+L] * I(0<=i+L<n)
            # b_L = corr[L + (n-1)] - a_L * old_val
            
            lags = np.arange(-(n-1), n)
            
            valid_left = (i - lags >= 0) & (i - lags < n)
            valid_right = (i + lags >= 0) & (i + lags < n)
            
            # a_L
            a = np.zeros(2*n-1)
            a[valid_left] += (1 - h[i - lags[valid_left]])
            a[valid_right] -= h[i + lags[valid_right]]
            
            # Current correlations
            corr = np.correlate(h, 1-h, mode='full')
            
            # b_L = corr - a * old_val
            b = corr - a * old_val
            
            # Solve 1D problem
            t_new = solve_1d_subproblem(a, b)
            
            h[i] = t_new
            
            max_improvement = max(max_improvement, abs(t_new - old_val))
        
        # Project to constraint
        h = project_box_sum(h, target_sum)
        
        # Check convergence
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        
        if max_improvement < tol:
            break
    
    return h, best_c5
```

Alternative: compute b_L directly.

corr[m] where m = L + (n-1):
corr[m] = ∑_j h[j+L]*(1-h[j]) for valid j

= ∑_j h[j+L] - ∑_j h[j+L]*h[j]

The first term: ∑_j h[j+L] for valid j. If L ≥ 0, j ranges from 0 to n-1-L, so it's ∑_{j=0}^{n-1-L} h[j+L] = ∑_{j=L}^{n-1} h[j]. If L < 0, j ranges from -L to n-1, so it's ∑_{j=-L}^{n-1} h[j+L] = ∑_{j=0}^{n-1+L} h[j].

The second term: ∑_j h[j+L]*h[j] for valid j. This is the autocorrelation of h at lag L.

So corr[m] = (sum of h over a subset) - (autocorrelation of h at lag L)

I can precompute:
1. Prefix sums of h for the first term
2. Autocorrelation of h for the second term

The autocorrelation can be computed via FFT in O(n log n).

```python
def coordinate_descent_v2(h, max_passes=200, tol=1e-12):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    
    # Precompute prefix sums
    prefix = np.zeros(n+1)
    prefix[1:] = np.cumsum(h)
    
    def get_autocorr(h):
        # Autocorrelation at all lags using FFT
        N = 2*n
        H = np.fft.rfft(h, N)
        ac = np.fft.irfft(H * np.conj(H), N)[:n]
        # ac[L] = sum_j h[j]*h[j+L] for valid j (for L >= 0)
        # For L < 0, ac[L] = ac[-L] by symmetry
        return ac
    
    autocorr = get_autocorr(h)
    
    for p in range(max_passes):
        max_improvement = 0.0
        for i in range(n):
            old_val = h[i]
            
            lags = np.arange(-(n-1), n)  # L = -(n-1), ..., n-1
            m = lags + (n-1)  # index in full correlation: 0..2n-2
            
            valid_left = (i - lags >= 0) & (i - lags < n)
            valid_right = (i + lags >= 0) & (i + lags < n)
            
            # a_L
            a = np.zeros(2*n-1)
            a[valid_left] += (1 - h[i - lags[valid_left]])
            a[valid_right] -= h[i + lags[valid_right]]
            
            # corr[m] = sum_{valid j} h[j+L] - autocorr_at_lag_L
            # For L >= 0: sum_{j=0}^{n-1-L} h[j+L] = prefix[n] - prefix[L]
            # For L < 0: sum_{j=-L}^{n-1} h[j+L] = prefix[n-1+L] - prefix[0] = prefix[n+L]
            
            # Actually let me be more careful:
            # corr_full(m) where m = L + (n-1):
            # = sum_j h[j+L] * (1-h[j]) where j and j+L are both in [0,n)
            # = sum_j h[j+L] - sum_j h[j+L]*h[j]
            
            # First part: sum of h[j+L] for valid j
            # If L >= 0: j from 0 to n-1-L, so j+L from L to n-1
            #   sum = prefix[n] - prefix[L]
            # If L < 0: j from -L to n-1, so j+L from 0 to n-1+L
            #   sum = prefix[n+L] - prefix[0] = prefix[n+L]
            
            pos_lags = lags >= 0
            neg_lags = lags < 0
            
            sum_h_shifted = np.zeros(2*n-1)
            L_pos = lags[pos_lags]
            sum_h_shifted[pos_lags] = prefix[n] - prefix[L_pos]
            
            L_neg = lags[neg_lags]
            sum_h_shifted[neg_lags] = prefix[n + L_neg]
            
            # Second part: autocorrelation of h at lag L
            # autocorr[L] for L >= 0 = sum_j h[j]*h[j+L]
            # For L < 0, autocorr[L] = autocorr[-L] (by symmetry)
            
            # Our autocorr array: autocorr[L] for L = 0..n-1
            # For L in [-(n-1), n-1]:
            #   if L >= 0: ac_L = autocorr[L]
            #   if L < 0: ac_L = autocorr[-L]
            
            ac_all = np.zeros(2*n-1)
            ac_all[pos_lags] = autocorr[L_pos]
            ac_all[neg_lags] = autocorr[-L_neg]
            
            # corr[m] = sum_h_shifted - ac_all
            corr = sum_h_shifted - ac_all
            
            # b = corr - a * old_val
            b = corr - a * old_val
            
            # Solve 1D
            t_new = solve_1d_subproblem(a, b)
            h[i] = t_new
            max_improvement = max(max_improvement, abs(t_new - old_val))
        
        h = project_box_sum(h, target_sum)
        
        # Update autocorrelation
        autocorr = get_autocorr(h)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        
        if max_improvement < tol:
            break
    
    return h, best_c5
```

```python
def solve_1d_subproblem_fast(a, b):
    """Find t in [0,1] minimizing max_m(a[m]*t + b[m]) using binary search on slope"""
    n = len(a)
    
    def eval_max(t):
        vals = a * t + b
        idx = np.argmax(vals)
        return vals[idx], a[idx]
    
    # Check boundaries
    val0, slope0 = eval_max(0.0)
    val1, slope1 = eval_max(1.0)
    
    if slope0 >= 0:
        return 0.0
    if slope1 <= 0:
        return 1.0
    
    # Binary search
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        val_mid, slope_mid = eval_max(mid)
        if slope_mid > 0:
            hi = mid
        elif slope_mid < 0:
            lo = mid
        else:
            return mid
    
    return (lo + hi) / 2
```

```python
    n = len(h)
    if abs(h.sum() - target_sum) < 1e-14 and h.min() >= -1e-14 and h.max() <= 1 + 1e-14:

        if s > target_sum:

        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_1d_sub(a, b):
    """Find t in [0,1] minimizing max_m(a[m]*t + b[m])"""
    # Coarse grid
    t_grid = np.linspace(0.0, 1.0, 100)
    vals = np.max(a[:, None] * t_grid[None, :] + b[:, None], axis=0)
    idx = np.argmin(vals)
    if idx == 0:
        return 0.0
    if idx == len(t_grid) - 1:
        return 1.0
    # Fine grid
    tl, th = t_grid[idx - 1], t_grid[idx + 1]
    t_fine = np.linspace(tl, th, 2000)
    vf = np.max(a[:, None] * t_fine[None, :] + b[:, None], axis=0)
    return t_fine[np.argmin(vf)]

    best_c5 = compute_c5(h)

    prefix = np.concatenate([[0.0], prefix])

    while Nf < 2 * n:
        Nf <<= 1
    H = np.fft.rfft(h, Nf)
    ac = np.fft.irfft(H * H.conj(), Nf)[:n]
    
    lags = np.arange(-(n-1), n)
    pos_mask = lags >= 0
    
    for p in range(max_passes):
        if time_limit is not None and time.time() - t0 > time_limit:
```
