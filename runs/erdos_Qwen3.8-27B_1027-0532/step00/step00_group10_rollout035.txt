import time

def compute_correlations(h):
    """Compute all correlation values: c[m] = dx * correlate(h, 1-h, mode='full')[m]"""
    return np.correlate(h, 1-h, mode='full')  # We'll multiply by dx separately

def project_to_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def compute_gradient(h, c, T, n):
    """Compute gradient of smoothed max of c"""
    # c is the raw correlation (without dx factor)
    # We want gradient of T * log(sum(exp(c * dx / T))) which is proportional to the gradient of T * log(sum(exp(c/T')))
    # where T' = T/dx. Let's just work with c directly and use temperature T.
    
    # Softmax weights
    c_scaled = c / T
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)  # length 2n-1
    
    # Gradient computation
    c1 = np.correlate(w, (1-h), mode='full')
    c2 = np.correlate(w, h, mode='full')
    grad = c1[n-1::-1] - c2[n-1::-1]
    
    return grad, w

```python
def compute_gradient_direct(h, w, n, dx):
    """Compute gradient of smoothed max objective."""
    grad = np.zeros(n)
    for t in range(-(n-1), n):
        if w[t + (n-1)] < 1e-10:
            continue
        wt = w[t + (n-1)]
        for j in range(n):
            # (1-h[j+t])*[0<=j+t<n] - h[j-t]*[0<=j-t<n]
            val = 0.0
            if 0 <= j+t < n:
                val += (1 - h[j+t])
            if 0 <= j-t < n:
                val -= h[j-t]
            grad[j] += wt * val
    grad *= dx
    return grad
```

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_objective(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def compute_gradient(h, w, n, dx):
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wt = w[idx]
        if wt < 1e-12:
            continue
        t = idx - (n - 1)
        if t >= 0:
            grad[:n - t] += wt * (1.0 - h[t:])
            grad[t:] -= wt * h[:n - t]
        else:
            grad[-t:] += wt * (1.0 - h[:n + t])
            grad[:n + t] -= wt * h[-t:]
    grad *= dx
    return grad

def optimize_loop(h_start, n, dx, target_sum, T_init, lr_init, deadline, state):
    h = h_start.copy()
    h = project_to_feasible(h, target_sum)
    T = T_init
    lr = lr_init
    stale = 0
    best_val = state['best_val']
    best_h = state['best_h']

    cur_obj = compute_objective(h, dx)
    if cur_obj < best_val:
        state['best_val'] = cur_obj
        state['best_h'] = h.copy()

    while time.time() < deadline:
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c_scaled = c_raw * dx
        cur_val = np.max(c_scaled)

        if cur_val < state['best_val'] - 1e-10:
            state['best_val'] = cur_val
            state['best_h'] = h.copy()
            stale = 0
        else:
            stale += 1

        if stale > 150:
            noise_amt = 0.08 + 0.04 * (stale // 150)
            h = state['best_h'] + np.random.randn(n) * noise_amt
            h = project_to_feasible(h, target_sum)
            stale = 0
            T = min(T * 2.5, 0.08)
            lr = min(lr * 1.5, 0.06)

        c_centered = c_scaled - np.max(c_scaled)
        exp_c = np.exp(c_centered / T)
        w = exp_c / np.sum(exp_c)
        grad = compute_gradient(h, w, n, dx)

        h_new = project_to_feasible(h - lr * grad, target_sum)
        new_val = compute_objective(h_new, dx)

        if new_val <= cur_val + 1e-12:
            h = h_new
            lr = min(lr * 1.02, 0.06)
        else:
            h_half = project_to_feasible(h - 0.5 * lr * grad, target_sum)
            hv = compute_objective(h_half, dx)
            if hv <= cur_val + 1e-12:
                h = h_half
            lr = max(lr * 0.75, 0.002)

        T = max(T * 0.9994, 0.002)

        if np.random.random() < 0.015:
            h = project_to_feasible(h + np.random.randn(n) * 0.02, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    n = 97
    dx = 2.0 / n
    target_sum = n / 2.0

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(n, 0.5)

    if len(h_init) != n:
        if len(h_init) > n:
            h_init = h_init[:n]
        else:
            h_init = np.pad(h_init, (0, n - len(h_init)), mode='edge')

    h_init = project_to_feasible(h_init, target_sum)
    init_val = compute_objective(h_init, dx)

    state = {'best_val': init_val, 'best_h': h_init.copy()}

    start_time = time.time()
    total_time = budget_s - 15

    t1 = start_time + 0.50 * total_time
    t2 = start_time + 0.72 * total_time
    t3 = start_time + 0.84 * total_time
    t4 = start_time + 0.93 * total_time
    t5 = start_time + total_time

    optimize_loop(h_init, n, dx, target_sum, 0.04, 0.04, t1, state)

    optimize_loop(state['best_h'].copy(), n, dx, target_sum, 0.02, 0.02, t2, state)

    h_comp = project_to_feasible(1.0 - h_init, target_sum)
    opt_val_comp = compute_objective(h_comp, dx)
    if opt_val_comp < state['best_val']:
        state['best_val'] = opt_val_comp
        state['best_h'] = h_comp.copy()
    optimize_loop(h_comp, n, dx, target_sum, 0.03, 0.03, t3, state)

    for attempt in range(4):
        if time.time() > t4:
            break
        h_rand = np.zeros(n)
        idx = np.random.permutation(n)
        h_rand[idx[:48]] = 1.0
        h_rand[idx[48]] = 0.5
        h_rand = project_to_feasible(h_rand, target_sum)
        per_attempt_end = t3 + (t4 - t3) * (attempt + 1) / 4.0
        optimize_loop(h_rand, n, dx, target_sum, 0.03, 0.03, per_attempt_end, state)

    optimize_loop(state['best_h'].copy(), n, dx, target_sum, 0.008, 0.008, t5, state)

    optimize_loop(state['best_h'].copy(), n, dx, target_sum, 0.003, 0.003, start_time + budget_s - 2, state)

    final_val = compute_objective(state['best_h'], dx)
    return state['best_h'], final_val, n
