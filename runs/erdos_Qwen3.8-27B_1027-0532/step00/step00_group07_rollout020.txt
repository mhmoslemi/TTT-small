    log_sum = max_corr + t * np.log(np.sum(np.exp((corr - max_corr) / t)))

    corr_shifted = corr - np.max(corr)
    weights = np.exp(corr_shifted / t)
    weights = weights / np.sum(weights)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    w = np.exp((corr - max_c) / t)

    # corr[m] = sum_i h[i] * (1 - h[i+s]) over valid i
    # dcorr[m]/dh[j] = (1-h[j+s]) - h[j-s] (valid terms only)

        j_lo1 = max(0, -s)

            grad[j_lo2:j_hi2] -= w_m * h[jms]

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_objective(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    return dx * (max_c + t * np.log(np.sum(np.exp((corr - max_c) / t))))

def smooth_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / t)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        w_m = w[m]
        if w_m < 1e-14:
            continue
        j_lo1 = max(0, s)
        j_hi1 = min(n, s + n)
        if j_lo1 < j_hi1:

            grad[j_lo1:j_hi1] += w_m * (1.0 - h[js])
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_lo2 < j_hi2:
            jps = np.arange(j_lo2, j_hi2) + s
            grad[j_lo2:j_hi2] -= w_m * h[jps]
    return dx * grad
