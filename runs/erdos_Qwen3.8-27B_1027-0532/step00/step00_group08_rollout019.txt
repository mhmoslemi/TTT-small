import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_smoothed(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-16:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            if n + lag > 0:
                grad[:n + lag] += w * (1.0 - h[-lag:])
                grad[-lag:] -= w * h[:n + lag]
    return grad

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def slsqp_objective(x, n, T, dx, target_sum):
    xc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(xc, 1.0 - xc, mode='full')
    scaled = corr / T
    ms = np.max(scaled)
    ev = np.exp(scaled - ms)
    f = float(T * (ms + np.log(np.sum(ev))) * dx)
    return f

def slsqp_objective_grad(x, n, T, dx, target_sum):
    xc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(xc, 1.0 - xc, mode='full')
    scaled = corr / T
    ms = np.max(scaled)
    ev = np.exp(scaled - ms)
    w = ev / np.sum(ev)
    f = float(T * (ms + np.log(np.sum(ev))) * dx)
    g = compute_grad_smoothed(xc, w) * dx
    return f, g

def do_slsqp(h, n, T, dx, target_sum, maxiter=500):
    def obj(x):
        return slsqp_objective(x, n, T, dx, target_sum)
    
    def obj_grad(x):
        return slsqp_objective_grad(x, n, T, dx, target_sum)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_grad, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
    if res.x is not None and not np.any(np.isnan(res.x)):
        return project_feasible(res.x, target_sum)
    return h

def gradient_descent_refine(h, n, target_sum, dx, max_steps=5000, T=0.001):
    current = h.copy()
    lr = 0.02
    best_local = compute_c5(current)
    for step in range(max_steps):
        corr = np.correlate(current, 1.0 - current, mode='full')
        scaled = corr / T
        ms = np.max(scaled)
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
        g = compute_grad_smoothed(current, w) * dx
        g = g - np.mean(g)
        h_new = current - lr * g
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_local - 1e-13:
            current = h_new
            best_local = c5_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return current, best_local

def binary_round_and_optimize(h, n, target_sum, dx, rng):
    n_ones = int(round(target_sum))
    best_bin_c5 = float('inf')
    best_bin = None
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    best_bin_c5 = compute_c5(h_bin)
    best_bin = h_bin.copy()
    for _ in range(2000):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        if h_bin[i] == h_bin[j]:

        h_try = best_bin.copy()
        h_try[i] = 0.0

        c5_try = compute_c5(h_try)
        if c5_try < best_bin_c5 - 1e-12:
            best_bin = h_try
            best_bin_c5 = c5_try
    return best_bin, best_bin_c5
