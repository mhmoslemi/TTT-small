```python
def project_onto_feasible(h, target_sum):
    """Project h onto {x : 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    # Define f(tau) = sum(clip(h_i - tau, 0, 1)) - target_sum
    # f is decreasing in tau
    # f(-inf) = n - target_sum > 0 (assuming target_sum < n)
    # f(inf) = -target_sum < 0
    
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0, 1)
    # Adjust for numerical error
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference equally among elements not at boundary
        mask = (projected > 0) & (projected < 1)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0, 1)
    return projected
```

```python
def compute_weighted_gradient(h, weights, dx):
    """
    h: array of length n
    weights: array of length 2n-1, weights[j] corresponds to lag L = j-(n-1)
    Returns gradient of sum_j weights[j] * corr[j] w.r.t. h
    """
    n = len(h)
    grad = np.zeros(n)
    
    for j in range(2*n-1):
        if weights[j] < 1e-10:
            continue
        k = j - (n-1)  # lag
        if k >= 0:
            # corr[j] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # grad contribution:
            # for m = i+k (i.e., m from k to n-1): (1-h[m-k])
            # for m = i (i.e., m from 0 to n-1-k): -h[m+k]
            valid_i = np.arange(0, n-k)
            m1 = valid_i + k  # indices where h appears as h[i+k]
            m2 = valid_i      # indices where h appears in (1-h[i])
            grad[m1] += weights[j] * (1 - h[m2]) * dx
            grad[m2] += weights[j] * (-h[m1]) * dx
        else:
            # k < 0, let k' = -k > 0
            # corr[j] = sum_{i=-k}^{n-1} h[i+k]*(1-h[i]) = sum_{i=k'}^{n-1} h[i-k']*(1-h[i])
            k_prime = -k
            valid_i = np.arange(k_prime, n)
            m1 = valid_i + k  # = valid_i - k_prime, indices where h appears as h[i+k]
            m2 = valid_i      # indices where h appears in (1-h[i])
            grad[m1] += weights[j] * (1 - h[m2]) * dx
            grad[m2] += weights[j] * (-h[m1]) * dx
    
    return grad
```
