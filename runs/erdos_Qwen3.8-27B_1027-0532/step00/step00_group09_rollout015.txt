import numpy as np
from scipy.optimize import minimize
import time

_CURRENT_T = 1.0


def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / _CURRENT_T
    return dx * (cmax + _CURRENT_T * np.log(np.sum(np.exp(scaled))))


def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    return result


def sum_con(x):
    return np.sum(x) - len(x) / 2.0


def sum_con_jac(x):
    return np.ones(len(x))


def local_swap_search(h, n, time_limit, t_start):
    best = h.copy()
    best_score = score_c5(best)
    step_size = 0.05
    stall_count = 0
    max_stall = 50
    while time.time() - t_start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            can_dec = mask & (best > 1e-8)
            can_inc = mask & (best < 1.0 - 1e-8)
            idx_dec = np.where(can_dec)[0]
            idx_inc = np.where(can_inc)[0]
            if len(idx_dec) == 0 or len(idx_inc) == 0:
                continue
            j_dec = idx_dec[np.argmin(best[idx_dec])]
            j_inc = idx_inc[np.argmax(best[idx_inc])]
            trial1 = best.copy()
            amt1 = min(step_size, best[i], 1.0 - best[j_inc])
            trial1[i] -= amt1
            trial1[j_inc] += amt1
            trial1 = project_feasible(trial1, n)
            s1 = score_c5(trial1)
            if s1 < best_score - 1e-12:
                best = trial1
                best_score = s1
                improved = True
                continue
            trial2 = best.copy()
            amt2 = min(step_size, best[j_dec], 1.0 - best[i])
            trial2[j_dec] -= amt2
            trial2[i] += amt2
            trial2 = project_feasible(trial2, n)
            s2 = score_c5(trial2)
            if s2 < best_score - 1e-12:
                best = trial2
                best_score = s2
                improved = True
        if not improved:
            stall_count += 1
            step_size *= 0.7
            if stall_count > max_stall or step_size < 1e-7:
                break
        else:
            stall_count = 0
    return best, best_score


def try_slsqp(h, n, T, max_iter):
    global _CURRENT_T
    _CURRENT_T = T
    try:
        res = minimize(lse_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)


def random_binary_init(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _CURRENT_T
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = rng.random(85)

    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = score_c5(best_h)

    temps = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for T in temps:
        if time.time() - t0 > budget_s * 0.35:
            break
        _CURRENT_T = T
        x_new, s_new = try_slsqp(best_h, n, T, 3000)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = x_new.copy()

    if time.time() - t0 < budget_s * 0.65:
        tl = budget_s * 0.65 - (time.time() - t0)
        best_h, best_c5 = local_swap_search(best_h, n, tl, t0)

    if time.time() - t0 < budget_s * 0.85:
        for n_try in [64, 96, 128, 160]:
            if time.time() - t0 > budget_s * 0.85:
                break
            tl = budget_s * 0.85 - (time.time() - t0)
            if tl < 20:

            for attempt in range(3):
                if time.time() - t0 > budget_s * 0.85:

                h_try = project_feasible(rng.random(n_try), n_try)
                _CURRENT_T = 0.5
                x_new, s_new = try_slsqp(h_try, n_try, 0.5, 1000)
                if s_new < best_c5:
                    _CURRENT_T = 0.1
                    x_new, s_new = try_slsqp(x_new, n_try, 0.1, 2000)
                if s_new < best_c5:
                    best_c5 = s_new
                    best_h = x_new.copy()
                    n = n_try

    if time.time() - t0 < budget_s - 30:

        if tl > 10:
            best_h, best_c5 = local_swap_search(best_h, n, tl, t0)
            _CURRENT_T = 0.005
            n = len(best_h)
            x_new, s_new = try_slsqp(best_h, n, 0.005, 2000)
            if s_new < best_c5:
                best_c5 = s_new
                best_h = x_new.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = score_c5(best_h)
    return best_h, best_c5, n
