corr[k] = sum_i h[i+s] * (1-h[i]) where the sum is over valid i (0≤i<n, 0≤i+s<n).

So for a given k (shift s = k-(n-1)):
d(corr[k])/d(h[m]) = (1-h[m-s]) * [0≤m-s<n] - h[m+s] * [0≤m+s<n]

Let me implement the projected subgradient method:

```python
def subgradient_step(h, dx, step_size):
    corr = np.correlate(h, 1-h, mode='full') * dx
    k_star = np.argmax(corr)
    s = k_star - (n-1)
    # Compute subgradient
    grad = np.zeros(n)
    for m in range(n):
        g = 0
        if 0 <= m-s < n:
            g += (1 - h[m-s])
        if 0 <= m+s < n:
            g -= h[m+s]
        grad[m] = g * dx
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new)
    return h_new, k_star
```

This is O(n) per step (plus the O(n) for the projection), which is very fast.

        return h_start, compute_c5(h_start, dx)

def try_binary(h, n, dx):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin, compute_c5(h_bin, dx)

def run(seed=42, budget_s=1300, **kwargs):
    global _P_STATE, _TARGET_SUM, _DX
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        h0 = np.full(86, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    for p in [4, 8, 16, 32, 64, 128, 256]:
        if remaining() < 15:
            break
        x_new, c5_new = slsqp_optimize(best_h.copy(), dx, n, p, 2000)
        if c5_new < best_c5 - 1e-12:
            best_h = x_new
            best_c5 = c5_new

    if remaining() > 10:
        h_sub = best_h.copy()
        step_size = 0.1
        for it in range(200000):
            if remaining() < 5:
                break
            h_sub, _ = subgradient_step(h_sub, dx, n, step_size)
            if it % 200 == 0:
                c5_now = compute_c5(h_sub, dx)
                if c5_now < best_c5 - 1e-12:
                    best_h = h_sub.copy()
                    best_c5 = c5_now
            if it % 5000 == 0 and it > 0:
                step_size *= 0.85

    if remaining() > 10:
        h_new, c5_new = targeted_search(best_h.copy(), dx, n, 200000, remaining)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    if remaining() > 10:
        h_new, c5_new = random_pair_search(best_h.copy(), dx, n, 500000, remaining)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    if remaining() > 15:
        h_bin, c5_bin = try_binary(best_h, n, dx)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin
        x_new, c5_new = slsqp_optimize(h_bin, dx, n, 32, 1000)
        if c5_new < best_c5 - 1e-12:
            best_h = x_new
            best_c5 = c5_new

    if remaining() > 15:
        for restart in range(15):
            if remaining() < 10:
                break
            perm = np.random.permutation(n)
            h_start = np.zeros(n)
            h_start[perm[:int(target)]] = 1.0
            x_new, c5_new = slsqp_optimize(h_start, dx, n, 16, 500)
            if c5_new < best_c5 - 1e-12:
                best_h = x_new
                best_c5 = c5_new

    if remaining() > 10:
        h_new, c5_new = targeted_search(best_h.copy(), dx, n, 200000, remaining)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    if remaining() > 5:
        h_new, c5_new = random_pair_search(best_h.copy(), dx, n, 200000, remaining)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n)
