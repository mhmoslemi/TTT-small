```python
def project_onto_feasible(h, target_sum):
    """Project h onto {x: 0≤x[i]≤1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    
    if s > target_sum:
        # Need to reduce
        excess = s - target_sum
        # Reduce from largest first
        order = np.argsort(-h)
        h = h.copy()
        for i in order:
            reduction = min(h[i], excess)
            h[i] -= reduction
            excess -= reduction
            if excess <= 1e-10:
                break
    else:
        # Need to increase
        deficit = target_sum - s
        order = np.argsort(h)
        h = h.copy()
        for i in order:
            increase = min(1 - h[i], deficit)
            h[i] += increase
            deficit -= increase
            if deficit <= 1e-10:
                break
    return h
```

```python
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_gradients(h):
    """Compute ∂c[k]/∂h[j] for all k, j. Returns array of shape (2n-1, n)"""
    n = len(h)
    num_shifts = 2 * n - 1
    grads = np.zeros((num_shifts, n))
    for idx in range(num_shifts):
        k = idx - (n - 1)
        for j in range(n):
            g = 0.0
            jm = j - k  # j - k
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jp = j + k  # j + k
            if 0 <= jp < n:
                g -= h[jp]
            grads[idx, j] = g
    return grads

def smooth_objective(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    vals = tau * c * dx
    max_val = np.max(vals)
    log_sum = max_val + np.log(np.sum(np.exp(vals - max_val)))
    return log_sum / tau

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    vals = tau * c * dx
    max_val = np.max(vals)
    w = np.exp(vals - max_val)
    w /= np.sum(w)
    
    # Compute weighted gradient
    grads = compute_gradients(h)
    grad = np.dot(w, grads) * dx
    return grad
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c

def smooth_grad_vec(h, w, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    num_shifts = 2 * n - 1
    for idx in range(num_shifts):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += wi * (1.0 - h[:n - k])
                grad[:n - k] -= wi * h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += wi * (1.0 - h[m:])
                grad[m:] -= wi * h[:n - m]
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        excess = s - target
        idx = np.argsort(-h)
        for i in idx:
            red = min(h[i], excess)
            h[i] -= red
            excess -= red
            if excess <= 1e-12:
                break
    else:
        deficit = target - s
        idx = np.argsort(h)
        for i in idx:
            inc = min(1.0 - h[i], deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    return h

def slsqp_obj(x, n):
    dx = 2.0 / n
    tau = 50.0
    x = np.clip(x, 0, 1)
    c = np.correlate(x, 1.0 - x, mode='full')
    vals = tau * c * dx
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(vals - m)))) / tau

def slsqp_grad(x, n):
    dx = 2.0 / n
    tau = 50.0
    x = np.clip(x, 0, 1)
    c = np.correlate(x, 1.0 - x, mode='full')
    vals = tau * c * dx
    m = np.max(vals)
    w = np.exp(vals - m)
    w /= np.sum(w)
    grad = smooth_grad_vec(x, w, n)
    return grad

def constraint_fun(x):
    return np.sum(x) - 40.0

def constraint_jac(x):
    return np.ones(len(x))

def gradient_descent_main(h, n, step, max_iters, tau_start, tau_end, time_limit_fn):

            else:
                patience += 1
        else:
            patience += 1

def local_coord_search(h, n, time_limit_fn, max_rounds=50):

    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
```
