import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx, corr

def compute_gradients(h):
    """Compute gradients of each correlation value w.r.t. h"""
    n = len(h)
    # For shift s (where s = m - (n-1)), m is the index in full correlation
    # corr[m] = sum_{valid i} h[i] * (1 - h[i+s]) where s = m - (n-1)
    # d corr[m] / d h[j] = (1-h[j+s])*[valid j+s] - h[j-s]*[valid j-s]
    corr = np.correlate(h, 1-h, mode="full")
    shifts = np.arange(len(corr)) - (n-1)
    
    # For each shift, compute gradient
    # grad_s[j] = (1-h[j+s]) if 0<=j+s<n, 0 otherwise
    #           - h[j-s] if 0<=j-s<n, 0 otherwise
    return corr, shifts

def soft_subgradient(h, beta=10.0):
    """Compute softmax-weighted gradient of max correlation"""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    shifts = np.arange(len(corr)) - (n-1)
    
    # Softmax weights
    max_corr = np.max(corr)
    weights = np.exp(beta * (corr - max_corr))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-10:
            continue
        s = shifts[idx]
        # (1-h[j+s]) for valid j
        if s >= 0:
            # j+s in [0,n-1] => j in [0, n-1-s]
            t1 = np.zeros(n)
            t1[:n-s] = (1 - h[s:]).copy()
            # h[j-s] for valid j: j-s in [0,n-1] => j in [s, n-1]
            t2 = np.zeros(n)
            t2[s:] = -h[:n-s].copy()
        else:
            ss = -s
            # j+s in [0,n-1] => j-ss in [0,n-1] => j in [ss, n-1]
            t1 = np.zeros(n)
            t1[ss:] = (1 - h[:n-ss]).copy()
            # j-s in [0,n-1] => j+ss in [0,n-1] => j in [0, n-1-ss]
            t2 = np.zeros(n)
            t2[:n-ss] = -h[ss:].copy()
        grad += weights[idx] * (t1 + t2)
    
    return grad, corr, max_corr
