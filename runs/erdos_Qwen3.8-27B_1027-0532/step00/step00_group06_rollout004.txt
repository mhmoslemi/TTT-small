    result = minimize(obj, h0, method='SLSQP', jac=obj_g,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
    h_opt = np.clip(result.x, 0.0, 1.0)
    h_opt = project_sum(h_opt, n)
    return h_opt, compute_c5(h_opt, n)


def projected_gradient(h0, n, n_iter=500, lr=None):
    h = h0.copy()
    if lr is None:
        lr = 0.5 * n / n
    lr = 2.0 / (4.0 * 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(n_iter):
        g = c5_subgrad(h, n)
        h = h - lr * g
        h = project_sum(h, n)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        parent = np.ones(70) * 0.5

    n_parent = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(best_h, n_parent)

    def elapsed():
        return time.time() - t0

    # Phase 1: Optimize parent grid with SLSQP (hard max)
    if elapsed() < budget_s * 0.1:
        h_opt, c5_opt = slsqp_optimize(best_h, n_parent, maxiter=300)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 2: Soft-max optimization on parent grid
    if elapsed() < budget_s * 0.2:
        h_opt, c5_opt = soft_slsqp_optimize(best_h, n_parent, maxiter=300, T=15.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Try larger grids
    larger_grids = []
    for gn in range(n_parent + 10, n_parent + 260, 10):
        larger_grids.append(gn)

    for target_n in larger_grids:
        if elapsed() > budget_s * 0.8:
            break

        h0 = interpolate_parent(best_h, target_n)
        h0 = project_sum(h0, target_n)

        # Soft optimization first (smoother)
        h_opt, c5_opt = soft_slsqp_optimize(h0, target_n, maxiter=200, T=10.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

        # Then hard max refinement
        if elapsed() < budget_s * 0.8:
            h_opt2, c5_opt2 = slsqp_optimize(h_opt if c5_opt < best_c5 else h0,
                                             target_n, maxiter=200)
            if c5_opt2 < best_c5:
                best_h = h_opt2
                best_c5 = c5_opt2

    # Phase 4: Fine-tune on best grid
    if elapsed() < budget_s * 0.9:
        n_best = len(best_h)
        h_opt, c5_opt = soft_slsqp_optimize(best_h, n_best, maxiter=400, T=8.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    if elapsed() < budget_s * 0.95:
        n_best = len(best_h)
        h_opt, c5_opt = slsqp_optimize(best_h, n_best, maxiter=400, ftol=1e-14)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: Random restarts on best grid
    n_best = len(best_h)
    restart_i = 0
    while elapsed() < budget_s - 5:
        restart_i += 1
        scale = max(0.02, 0.15 / (1 + restart_i * 0.3))
        h0 = best_h + np.random.randn(n_best) * scale
        h0 = project_sum(h0, n_best)
        try:
            h_opt, c5_opt = slsqp_optimize(h0, n_best, maxiter=150)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        except Exception:
            pass

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_sum(best_h, n_final)
    best_c5 = compute_c5(best_h, n_final)

    return best_h, best_c5, n_final
