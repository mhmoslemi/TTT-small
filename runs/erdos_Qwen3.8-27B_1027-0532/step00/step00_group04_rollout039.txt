def coordinate_update(h, j, n, dx):
    """Find optimal h[j] given other components fixed"""
    # Compute A_s and B_s for all shifts s
    # c_s = dx * (B_s + h[j] * A_s)
    n_shifts = 2*n - 1
    A = np.zeros(n_shifts)
    B = np.zeros(n_shifts)
    
    for si in range(n_shifts):
        s = si - (n - 1)  # shift value
        a_val = 0.0
        # h[j] appears as h[i] when i=s: contributes (1-h[j+s])
        if 0 <= j + s < n:
            a_val += (1 - h[j+s])
        # h[j] appears as h[i+s] when i+s=j, i=j-s: contributes -h[j-s]
        if 0 <= j - s < n:
            a_val += (-h[j-s])
        A[si] = a_val
    
    # Compute B from current c_s
    corr = np.correlate(h, 1-h, mode='full')  # without dx
    for si in range(n_shifts):
        B[si] = corr[si] - h[j] * A[si]
    
    # Now minimize max_s (B_s + t * A_s) over t in [0,1]
    # Convex piecewise linear in t
    # Bisection on the subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = B + mid * A
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        slopes = A[active]
        left_deriv = np.min(slopes)
        right_deriv = np.max(slopes)
        if left_deriv > 0:
            hi = mid
        elif right_deriv < 0:
            lo = mid
        else:
            break
    
    return (lo + hi) / 2

import numpy as np
import time

def project_onto_feasible(h, n):
    """Project h onto {x: 0 <= x[i] <= 1, sum(x) = n/2} using bisection"""
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h, n):
    """Compute C5 value"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def compute_c5_gradient(h, n):
    """Compute subgradient of C5 (max of correlation) at h"""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')  # without dx
    c_max = np.max(corr)
    active = corr >= c_max - 1e-10
    active_indices = np.where(active)[0]
    
    grad = np.zeros(n)
    for idx in active_indices:
        s = idx - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            sp = -s
            grad[sp:] += (1.0 - h[:n-sp])
            grad[:n-sp] += (-h[sp:])
    
    grad *= dx
    return grad

def subgradient_method(h, n, max_iter=5000, initial_step=0.05):
    """Projected subgradient method"""
    h = project_onto_feasible(h.copy(), n)
    best_c = evaluate_c5(h, n)
    best_h = h.copy()
    step = initial_step
    
    for it in range(max_iter):
        c_current = evaluate_c5(h, n)
        if c_current < best_c - 1e-12:
            best_c = c_current
            best_h = h.copy()
        
        grad = compute_c5_gradient(h, n)
        
        # Backtracking line search
        for _ in range(30):
            h_new = project_onto_feasible(h - step * grad, n)
            c_new = evaluate_c5(h_new, n)
            if c_new < c_current:
                h = h_new
                break
            step *= 0.5
        else:
            # Can't improve, try smaller step
            step *= 0.9
    
    return best_h, best_c

def coordinate_sweep(h, n, max_sweeps=100):
    """Coordinate descent with projection"""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    h = h.copy()
    best_c = evaluate_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')  # cache
        
        for j in range(n):
            # Compute A vector for coordinate j
            A = np.zeros(n_shifts)
            for si in range(n_shifts):
                s = si - (n - 1)
                a_val = 0.0
                if 0 <= j + s < n:
                    a_val += (1.0 - h[j+s])
                if 0 <= j - s < n:
                    a_val += (-h[j-s])
                A[si] = a_val
            
            B = corr - h[j] * A
            
            # Minimize max(B + t*A) over t in [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = B + mid * A
                max_val = np.max(vals)
                active = vals >= max_val - 1e-12
                if np.min(A[active]) > 0:
                    hi = mid
                elif np.max(A[active]) < 0:
                    lo = mid
                else:
                    break
            t_new = (lo + hi) / 2.0
            
            delta = t_new - h[j]
            if abs(delta) > 1e-12:
                corr += delta * A
                h[j] = t_new
        
        # Project to maintain sum constraint
        h = project_onto_feasible(h, n)
        c = evaluate_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

Key observation: the coordinate descent finds the exact optimal value of h[j] given all other coordinates fixed. This is powerful. The issue is the sum constraint.

What if I formulate the problem differently? Instead of optimizing h directly, I optimize h subject to the sum constraint by using a Lagrange multiplier.

minimize max_s c_s(h)
subject to h ∈ [0,1]^n, ∑h = n/2

Lagrangian: max_s c_s(h) + λ(∑h - n/2)

For a fixed λ, the inner problem is:
minimize max_s c_s(h) + λ∑h
subject to h ∈ [0,1]^n

This is still non-convex (max of quadratics). But we can use a similar subgradient approach:
subgradient of c_s + λ∑h = subgrad(c_s) + λ

And we can do coordinate descent on this modified objective (which now doesn't have the sum constraint, just box constraints).

In the coordinate descent for the Lagrangian:

import numpy as np
import time

def project_onto_feasible(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h, n):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_subgradient(h, n):
    """Compute subgradient of C5 at h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    active = np.where(corr >= c_max - 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        s = idx - (n-1)
        if s == 0:
            grad += (1.0 - 2.0*h)
        elif s > 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            sp = -s
            grad[sp:] += (1.0 - h[:n-sp])
            grad[:n-sp] += (-h[sp:])
    return grad * dx

def coord_descent(h0, n, max_sweeps=200, perturb_every=10, perturb_size=0.01):

    n_sh = 2*n - 1
    h = np.array(h0, dtype=float).copy()

        corr = np.correlate(h, 1.0-h, mode='full')
        
        for j in range(n):
            A = np.zeros(n_sh)
            A[n-1-j : 2*n-1-j] += (1.0 - h)
            A[j : n+j] += (-h[::-1])

                elif np.max(A[act]) < -1e-13:

                else:

            t_new = (lo+hi) / 2.0
