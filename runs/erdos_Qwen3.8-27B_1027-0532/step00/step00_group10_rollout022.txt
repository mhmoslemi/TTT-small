def compute_effect_single(h, p, dh, n):
    """Compute change in corr vector when h[p] changes by dh (from old to old+dh)."""
    delta = np.zeros(2*n-1)

    # Valid when 0 <= p+shift < n, i.e., -p <= shift <= n-1-p

    valid = (k_indices >= 0) & (k_indices < 2*n-1) & (q_indices >= 0) & (q_indices < n)

    delta[k_indices2[valid2]] += -dh * h[q_indices2[valid2]]

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}."""
    # Binary search for alpha
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target_sum:

        else:

    alpha = (lo + hi) / 2.0
    h_proj = np.clip(h - alpha, 0.0, 1.0)
    # Fine adjustment
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:

        if diff > 0:

        else:

    """Compute smooth max of corr * dx."""
    scaled = tau * corr * dx
    # Shift for numerical stability

Now, for the overall gradient using smooth max:
∇F_τ = sum_k w_k * ∇f_k where w_k = softmax(τ * f_k)

This requires computing the gradient of f_k for all k and taking a weighted sum. For n=97, there are 2n-1 = 193 shifts. Computing all gradients would be O(n²) total, which is fine.

But actually, we can be smarter. The gradient of the smooth max is:
∇F_τ[j] = dx * sum_k w_k * [(1-h[j+k])*I(valid) - h[j-k]*I(valid)]

This can be computed as a weighted correlation of w with h.

Let me define: for each j, ∇F_τ[j] = dx * [(W * (1-h))[j] - (W_rev * h)[j]] where W is the softmax weight vector and the convolution is appropriate.

Actually, let me think about it differently. The correlation vector is:
corr[k] = sum_i h[i] * (1-h[i+shift_k]) where shift_k = (n-1-k)

The gradient of corr[k] with respect to h[j]:
= dx * [(1-h[j+shift_k])*I(valid) + (-h[j-shift_k])*I(valid)]

So ∇F_τ[j] = dx * sum_k w_k * [(1-h[j+shift_k])*I(j+shift_k valid) - h[j-shift_k]*I(j-shift_k valid)]

= dx * sum_k w_k * (1-h[j+shift_k])*I(j+shift_k valid) - dx * sum_k w_k * h[j-shift_k]*I(j-shift_k valid)

The first sum: for fixed j, sum over k where j+shift_k is valid.
shift_k = n-1-k, so shift ranges over {-(n-1), ..., n-1}.
j+shift_k valid means 0 ≤ j+shift_k < n, i.e., -j ≤ shift_k < n-j.

So the first sum = sum_{shift: -j ≤ shift < n-j} w_{n-1-shift} * (1-h[j+shift])

This is a cross-correlation of w (reversed) with (1-h), evaluated at position j, but with a window.

Similarly for the second sum.

This is essentially a convolution/correlation operation, which can be done with FFT in O(n log n).

But for n=97, even a direct O(n²) computation is fast (97² ≈ 10^4 per evaluation).

Let me just implement it directly:

```python
def compute_smooth_max_gradient(h, n, dx, tau):
    corr = np.correlate(h, 1-h, mode='full')
    f = corr * dx
    # Softmax weights
    shifted_f = tau * f - (tau * f).max()
    w = np.exp(shifted_f)
    w /= w.sum()
    
    grad = np.zeros(n)
    # For each shift (index k in corr), compute contribution to grad
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        shift = (n-1) - k
        w_k = w[k]
        if shift == 0:
            grad += w_k * (1 - 2*h) * dx
        elif shift > 0:
            grad[:n-shift] += w_k * (1 - h[shift:]) * dx
            grad[shift:] += w_k * (-h[:n-shift]) * dx
        else:
            s = -shift
            grad[s:] += w_k * (1 - h[:n-s]) * dx
            grad[:n-s] += w_k * (-h[s:]) * dx
    
    return grad
```

This is O(n²) per evaluation. For n=97, that's about 2*97*97/2 ≈ 10^4 operations. Very fast.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h_proj = np.clip(h - alpha, 0.0, 1.0)
    # Fine-tune to get exact sum
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            candidates = np.where(h_proj < 1.0 - 1e-10)[0]
        else:
            candidates = np.where(h_proj > 1e-10)[0]
        if len(candidates) > 0:
            h_proj[candidates] += diff / len(candidates)
            h_proj = np.clip(h_proj, 0.0, 1.0)

def grad_smooth_max(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    tf = tau * f

    w = np.exp(tf_shifted)
