def compute_overlap(h):
    """Compute C5 = max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    diff = target_sum - np.sum(h)
    if diff > 0:
        # Need to increase sum; add to elements not at 1
        mask = h < 1.0
        if np.sum(mask) == 0:
            return h
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    elif diff < 0:
        # Need to decrease sum; subtract from elements not at 0
        mask = h > 0.0
        if np.sum(mask) == 0:
            return h
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

    max_val = np.max(corr)

def project_box_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:

    dx = 2.0 / n

    n = len(h)

        return g
    if k >= 0:
        m = n - k
        g[:m] += (1.0 - h[k:])
        g[k:] -= h[:m]
    else:
        k2 = -k
        m = n - k2
        g[k2:] += (1.0 - h[:m])
        g[:m] -= h[k2:]

def subgradient(h, top_m=5):
    """Compute subgradient of max_k g_k(h) using top_m active shifts"""

def project_box_sum(h, target_sum):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo = -1.0
    hi = 1.0
    # Ensure brackets
    if np.sum(np.clip(h - lo, 0, 1)) < target_sum:
        lo = -2.0
    if np.sum(np.clip(h - hi, 0, 1)) > target_sum:
        hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(h - lam, 0, 1)
    # Final fix for floating point
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        # Distribute to non-boundary elements
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0, 1)
    return result

    """Compute C5 value"""
