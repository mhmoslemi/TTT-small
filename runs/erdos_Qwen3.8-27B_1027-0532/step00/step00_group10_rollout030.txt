import numpy as np
import time

def compute_objective(h, n):
    """Compute C5 value and all shift values"""
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode='full') * dx
    return np.max(conv), conv

def compute_gradient(h, n, T, conv):
    """Compute gradient of softmax-smoothed objective"""
    dx = 2.0 / n
    # conv has length 2n-1, index k corresponds to shift s = k - (n-1)
    shifts = np.arange(-(n-1), n)
    g = conv  # g_s values
    
    # Softmax weights
    max_g = np.max(g)
    exp_g = np.exp(T * (g - max_g))
    w = exp_g / np.sum(exp_g)
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, s in enumerate(shifts):
        if w[idx] < 1e-10:
            continue
        if s == 0:
            grad += w[idx] * (1 - 2*h) * dx
        else:
            # ∂g_s/∂h[j] = (1-h[j-s]) * I[0≤j-s<n] - h[j+s] * I[0≤j+s<n]

                    val += (1 - h[j-s])
                if 0 <= j+s < n:
                    val -= h[j+s]
                grad[j] += w[idx] * val * dx
    
    return grad

```python
def compute_smoothed_gradient(h, n, T, conv):
    dx = 2.0 / n
    # conv has length 2n-1
    g = conv
    max_g = np.max(g)
    exp_g = np.exp(T * (g - max_g))
    w = exp_g / np.sum(exp_g)
    
    grad = np.zeros(n)
    # shift k in conv corresponds to s = k - (n-1)
    for k in range(2*n-1):
        s = k - (n-1)
        if w[k] < 1e-12:
            continue
        if s == 0:
            grad += w[k] * (1 - 2*h) * dx
        elif s > 0:
            grad[s:] += w[k] * (1 - h[:n-s]) * dx
            grad[:n-s] += w[k] * (-h[s:]) * dx
        else:
            t = -s
            grad[:n-t] += w[k] * (1 - h[t:]) * dx
            grad[t:] += w[k] * (-h[:n-t]) * dx
    
    return grad
```

```python
def project(y, n, target_sum):
    # Binary search for lambda
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)
```

    stuck_count = 0

        for _ in range(20):  # backtracking line search

def compute_conv(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(conv):
    return np.max(conv)

def compute_smoothed_gradient(h, n, T, conv):
    dx = 2.0 / n
    g = conv
    g_max = np.max(g)
    exp_g = np.exp(T * (g - g_max))
    w = exp_g / np.sum(exp_g)
    
    grad = np.zeros(n)
    n_half = n - 1  # offset for correl
    for k in range(2 * n - 1):
        s = k - n_half
        wk = w[k]
        if wk < 1e-14:
            continue
        if s == 0:
            grad += wk * (1.0 - 2.0 * h) * dx
        elif s > 0:
            grad[s:] += wk * (1.0 - h[:n - s]) * dx
            grad[:n - s] += -wk * h[s:] * dx
        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:]) * dx
            grad[t:] += -wk * h[:n - t] * dx
    
    return grad

def project_to_feasible(y, n, target_sum):
    # Project y onto {h: 0 <= h[i] <= 1, sum(h) = target_sum}
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    # Final correction (should be very small)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference to elements not at bounds
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h
