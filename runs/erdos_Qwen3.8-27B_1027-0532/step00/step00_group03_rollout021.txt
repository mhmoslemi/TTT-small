import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_objective_function(h, n, temperature, penalty_weight, binary_weight):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlap = corr * dx
    m = np.max(overlap)
    exp_vals = np.exp(temperature * (overlap - m))
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / temperature
    obj += penalty_weight * (np.sum(h) - n / 2.0) ** 2
    if binary_weight > 0:
        obj += binary_weight * np.sum(h * (1.0 - h))
    return obj

def get_gradient_function(h, n, temperature, penalty_weight, binary_weight):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlap = corr * dx
    m = np.max(overlap)
    exp_vals = np.exp(temperature * (overlap - m))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        s = l - (n - 1)
        w = weights[l]
        if w < 1e-14:
            continue
        js1 = max(0, -s)
        je1 = min(n - 1, n - 1 - s)
        if js1 <= je1:
            grad[js1:je1 + 1] += w * (1.0 - h[js1 + s:je1 + s + 1])
        js2 = max(0, s)
        je2 = min(n - 1, n - 1 + s)
        if js2 <= je2:
            grad[js2:je2 + 1] -= w * h[js2 - s:je2 - s + 1]
    grad *= dx
    grad += 2.0 * penalty_weight * (np.sum(h) - n / 2.0)
    if binary_weight > 0:
        grad += 2.0 * binary_weight * (0.5 - h)
    
    return grad

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Need to increase sum, add to elements < 1
            mask = h < 1.0
            if np.sum(mask) > 0:
                per_element = diff / np.sum(mask)
                h[mask] = np.minimum(1.0, h[mask] + per_element)
                # If still not at target, adjust further
                while abs(np.sum(h) - target) > 1e-10 and np.sum(h) < target:
                    mask = h < 1.0
                    if np.sum(mask) == 0:
                        break
                    per_element = (target - np.sum(h)) / np.sum(mask)
                    h[mask] = np.minimum(1.0, h[mask] + per_element)
        else:
            mask = h > 0.0
            if np.sum(mask) > 0:
                per_element = -diff / np.sum(mask)
                h[mask] = np.maximum(0.0, h[mask] - per_element)
                while abs(np.sum(h) - target) > 1e-10 and np.sum(h) > target:
                    mask = h > 0.0
                    if np.sum(mask) == 0:
                        break
                    per_element = (np.sum(h) - target) / np.sum(mask)
                    h[mask] = np.maximum(0.0, h[mask] - per_element)
    return h

def optimize_stage(h_init, n, temperature, max_iter, time_limit):
    """Run one stage of optimization."""
    start_time = time.time()
    h = h_init.copy()
    penalty_weight = 100.0
    binary_weight = 0.01
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n

        def obj_func(x):
            return get_objective_function(x, n, temperature, penalty_weight, binary_weight)
        
        def grad_func(x):
            return get_gradient_function(x, n, temperature, penalty_weight, binary_weight)
        
        result = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
        
        h = result.x

        c5 = compute_c5(h)
        if c5 < best_c5:

            best_h = h.copy()
        
        # Increase penalty
        penalty_weight *= 10.0
        temperature *= 2.0
        
        if time.time() - start_time > time_limit:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_gs = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_objective(x):
    n = _gs['n']
    t = _gs['temp']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    ov = corr * dx
    m = np.max(ov)
    ev = np.exp(t * (ov - m))
    return m + np.log(np.sum(ev)) / t

def softmax_gradient(x):
    n = _gs['n']
    t = _gs['temp']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    ov = corr * dx
    m = np.max(ov)
    ev = np.exp(t * (ov - m))
    w = ev / np.sum(ev)
    g = np.zeros(n)
    for l in range(2 * n - 1):
        s = l - (n - 1)
        wl = w[l]
        if wl < 1e-15:
            continue
        js1 = max(0, -s)
        je1 = min(n - 1, n - 1 - s)
        if js1 <= je1:
            g[js1:je1 + 1] += wl * (1.0 - x[js1 + s:je1 + s + 1])
        js2 = max(0, s)
        je2 = min(n - 1, n - 1 + s)
        if js2 <= je2:
            g[js2:je2 + 1] -= wl * x[js2 - s:je2 - s + 1]
    return g * dx

def obj_func(x):
    n = _gs['n']
    p = _gs['pen']
    val = softmax_objective(x)
    d = np.sum(x) - n / 2.0
    val += p * d * d
    return val

def grad_func(x):
    n = _gs['n']
    p = _gs['pen']
    g = softmax_gradient(x)
    g += 2.0 * p * (np.sum(x) - n / 2.0)
    return g

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not np.any(mask):
                break
            adj = diff / np.sum(mask)
            h[mask] = np.minimum(1.0, h[mask] + adj)
        else:
            mask = h > 1e-13
            if not np.any(mask):
                break
            adj = -diff / np.sum(mask)
            h[mask] = np.maximum(0.0, h[mask] - adj)
    return h

def interp_h(h, n_from, n_to):
    xf = np.linspace(0.0, 2.0, n_from, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def binary_round(h, n):
    hb = (h > 0.5).astype(float)
    hb = project_feasible(hb, n)
    c5b = compute_c5(hb)
    c5o = compute_c5(h)
    if c5b < c5o:
        return hb, c5b
    return h, c5o

def optimize_stage(h_init, n, tl, t0):
    ts = time.time()
    h = project_feasible(h_init.copy(), n)
    bh = h.copy()
    bc5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    temp = t0
    pen = 100.0
    for _ in range(12):
        if time.time() - ts > tl:
            break
        _gs['n'] = n
        _gs['temp'] = temp
        _gs['pen'] = pen
        res = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 60})
        h = project_feasible(res.x, n)
        c5 = compute_c5(h)
        if c5 < bc5:
            bc5 = c5
            bh = h.copy()
        h2, c52 = binary_round(h, n)
        if c52 < bc5:
            bc5 = c52
            bh = h2.copy()
        temp *= 3.0
        pen *= 4.0
    return bh, bc5

def local_search(h, n, tlimit):
    ts = time.time()
    bc5 = compute_c5(h)
    bh = h.copy()
    for _ in range(20000):

                    improved = True
                    break
                bh[i] = 0.0
                bh[j] = 1.0

                break
        if not improved:
            break
    return bh, bc5
