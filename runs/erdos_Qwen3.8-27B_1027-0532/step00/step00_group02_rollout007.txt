    corr = np.correlate(h, 1-h, mode="full")
    return dx * tau * np.log(np.sum(np.exp(corr / tau)))

def smooth_grad(h, dx, tau):
    corr = np.correlate(h, 1-h, mode="full")
    weights = np.exp(corr / tau)
    weights /= np.sum(weights)
    # grad of corr_k w.r.t. h[m]
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        d = n - 1 - k
        for m in range(n):
            j = m - d
            if 0 <= j < n:
                grad[m] += weights[k] * (1-h[j])
            j2 = m + d
            if 0 <= j2 < n:
                grad[m] -= weights[k] * h[j2]
    return dx * grad
