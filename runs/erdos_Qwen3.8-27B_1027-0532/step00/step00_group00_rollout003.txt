    target = n / 2.0
    total = 0.0
    for _ in range(n):
        best_val = 0.0
        best_c5 = float('inf')
        for v in [0.0, 0.25, 0.5, 0.75, 1.0]:
            if abs(total + v - target) > 1.0 + 1e-9:
                continue
            h[-1] = v
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_val = v
        h[-1] = best_val
        total += best_val
    h = fix_sum(h, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_h = None
    best_c5 = 1.0

    def try_update(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 1: Optimize initial solution
    if h_init is not None:
        h_work = h_init.copy()
        h_work, c5_work = sa_optimize(h_work, t_start + budget_s * 0.35, rng)
        try_update(h_work, c5_work)

    # Phase 2: Try different grid sizes
    n_values = [46, 50, 54, 60, 80, 100, 120, 160, 200, 250, 300]
    phase2_time = budget_s * 0.15

    for n_try in n_values:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.82:
            break

        remaining = min(phase2_time, budget_s * 0.82 - elapsed)
        if remaining < 1.0:
            continue

        # Binary initialization
        h_bin = np.zeros(n_try)
        half = n_try // 2
        if n_try % 2 == 0:
            idx = rng.choice(n_try, half, replace=False)
            h_bin[idx] = 1.0
        else:
            idx = rng.choice(n_try, half, replace=False)
            h_bin[idx] = 1.0
            idx2 = rng.choice([i for i in range(n_try) if i not in set(idx.tolist())], 1, replace=False)
            h_bin[idx2] = 0.5

        # Interpolated initialization from best
        h_inter = h_bin.copy()
        if best_h is not None and len(best_h) != n_try:
            x_src = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
            x_dst = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_inter = np.interp(x_dst, x_src, best_h)
            h_inter = fix_sum(h_inter, n_try)

        # Run SA on both
        h_res1, c5_res1 = sa_optimize(h_bin, t_start + remaining * 0.5, rng)
        try_update(h_res1, c5_res1)

        if time.time() < t_start + budget_s * 0.82:
            h_res2, c5_res2 = sa_optimize(h_inter, t_start + min(remaining, (t_start + budget_s * 0.82) - time.time()), rng)
            try_update(h_res2, c5_res2)

    # Phase 3: Multi-restart on best solution's grid
    if best_h is not None:
        n_best = len(best_h)
        restarts = 5
        restart_time = (budget_s * 0.95 - (time.time() - t_start)) / restarts
        restart_time = max(restart_time, 2.0)

        for r in range(restarts):
            if time.time() - t_start > budget_s * 0.95:
                break
            # Perturb best solution
            h_pert = best_h.copy()
            n_pert = len(h_pert)
            for _ in range(n_pert // 3):
                i, j = rng.integers(0, n_pert, 2)
                if i != j:
                    h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
            h_pert = fix_sum(h_pert, n_pert)

            # Also try with random noise
            if rng.random() < 0.5:
                h_pert = best_h + rng.normal(0, 0.1, n_pert)
                h_pert = fix_sum(h_pert, n_pert)

            h_res, c5_res = sa_optimize(h_pert, t_start + budget_s * 0.95 - (time.time() - t_start - restart_time), rng)
            try_update(h_res, c5_res)

    # Phase 4: Final polish
    if best_h is not None and time.time() < t_start + budget_s * 0.97:
        h_res, c5_res = sa_optimize(best_h, t_start + budget_s * 0.99, rng)
        try_update(h_res, c5_res)

    # Final validation and fix
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, n)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
