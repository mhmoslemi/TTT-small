    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_polytope(v, target_sum):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target_sum}."""
    # Binary search for lambda
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

def logsumexp_smooth_max(cvals, temp):
    """Return (smooth_max_value, weights) where smooth_max approximates max(cvals)."""
    m = np.max(cvals)
    if temp < 1e-12:

        w = np.zeros_like(cvals)
        w[idx] = 1.0
        return m, w

    # In h[n+m-offset] = h[i]: n = i - m + offset, valid if 0 <= i-m+offset < N
    #   d[m] += (1-h[n]) = (1-h[i-m+offset])
    # In (1-h[n]) = (1-h[i]): n = i, h[n+m-offset] = h[i+m-offset]
    #   d[m] -= h[i+m-offset]

        if 0 <= n1 < N:

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_polytope(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_d_for_pair(h, i, j, N):
    offset = N - 1
    m = np.arange(2 * N - 1)
    d = np.zeros(2 * N - 1)
    
    n1 = i - m + offset
    v1 = (n1 >= 0) & (n1 < N)
    d[v1] += (1.0 - h[n1[v1]])
    
    n2 = i + m - offset
    v2 = (n2 >= 0) & (n2 < N)
    d[v2] -= h[n2[v2]]
    
    n1j = j - m + offset
    v1j = (n1j >= 0) & (n1j < N)
    d[v1j] -= (1.0 - h[n1j[v1j]])
    
    n2j = j + m - offset
    v2j = (n2j >= 0) & (n2j < N)
    d[v2j] += h[n2j[v2j]]
    
    return d

def minimize_1d_convex(cvals, d, t_lo, t_hi, n_eval=20):
    """Minimize max(cvals + t*d) for t in [t_lo, t_hi]."""

    # Also include 0 if in range
    if t_lo < 0 < t_hi:
        t_evals = np.append(t_evals, 0.0)
    
    best_t = 0.0
    best_val = np.max(cvals)  # Current value at t=0
    
    for t in t_evals:
        val = np.max(cvals + t * d)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_polytope(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_d_for_pair(h, i, j, N):
    offset = N - 1
    m = np.arange(2 * N - 1)
    d = np.zeros(2 * N - 1)
    n1 = i - m + offset
    v1 = (n1 >= 0) & (n1 < N)
    d[v1] += (1.0 - h[n1[v1]])
    n2 = i + m - offset
    v2 = (n2 >= 0) & (n2 < N)
    d[v2] -= h[n2[v2]]
    n1j = j - m + offset
    v1j = (n1j >= 0) & (n1j < N)
    d[v1j] -= (1.0 - h[n1j[v1j]])
    n2j = j + m - offset
    v2j = (n2j >= 0) & (n2j < N)
    d[v2j] += h[n2j[v2j]]
    return d
