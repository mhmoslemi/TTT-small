```python
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n_points):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    s = np.sum(h)
    if s > 0 and abs(s - target) > 1e-8:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-8:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def gen_random_h(n_points, rng):
    a = rng.choice(np.array([-1.0, 1.0]), size=n_points)
    s = int(np.sum(a))
    if s > 0:
        pos = np.where(a == 1.0)[0]
        a[rng.choice(pos, size=s//2, replace=False)] = -1.0
    elif s < 0:
        neg = np.where(a == -1.0)[0]
        a[rng.choice(neg, size=(-s)//2, replace=False)] = 1.0
    return (a + 1.0) / 2.0

def gen_sine_h(n_points, freq, amp, phase):
    x = np.arange(n_points) * (2.0 / n_points)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return repair_h(h, n_points)
```

Actually, I realize that for the projected gradient method, I need to:
1. Compute the correlation vector (O(n²) = O(86²) ≈ 7400 ops)
2. Find the argmax (O(n) = O(171))
3. Compute the gradient of the max (O(n) = O(86))
4. Take a step: h_new = h - lr * grad
5. Project onto the constraint set {h: 0 ≤ h_i ≤ 1, sum h_i = n/2}

The projection onto {0 ≤ h_i ≤ 1, sum h_i = n/2} is the "Euclidean projection onto the transportation polytope" (or the permutahedron). This can be done efficiently using a water-filling algorithm.

The water-filling algorithm:
- We want to find h_proj that minimizes ||h_proj - h||² subject to 0 ≤ h_proj_i ≤ 1 and sum h_proj_i = n/2.
- By KKT conditions: h_proj_i = clip(h_i - λ, 0, 1) for some λ such that sum(h_proj) = n/2.
- We can find λ by bisection.

This is O(n log(n/ε)) which is fast.

Let me implement this:

```python
def project_onto_constraint(h, n_points):
    """Project h onto {x: 0 <= x_i <= 1, sum x_i = n/2}."""
    target = n_points / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)
```

```python
def projected_gradient(h, n_points, dx, lr, n_iter):

        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, n_points)
        new_val = compute_c5(h_new, dx)
        if new_val < current_val:
            h = h_new
            current_val = new_val
        else:
            lr *= 0.9  # Backtrack
    return h, current_val
```
