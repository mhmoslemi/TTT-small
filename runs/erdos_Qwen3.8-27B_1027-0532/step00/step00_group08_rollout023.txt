def compute_c5_smooth(h, eps=0.1):
    """Smoothed version using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp for numerical stability
    m = np.max(corr)
    val = m + eps * np.log(np.sum(np.exp((corr - m) / eps)))
    return val * dx

def compute_grad_vectorized(h, eps=0.05):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Soft max weights
    c_max = np.max(corr)
    weights = np.exp((corr - c_max) / eps)
    weights /= np.sum(weights)
    
    # grad[j] = 1 - S1[j] - S2[j]
    # S1[j] = sum_k weights[k] * h[j - (k - n + 1)] (valid)
    # S2[j] = sum_k weights[k] * h[j + (k - n + 1)] (valid)
    
    grad = np.ones(n)
    
    for k in range(len(weights)):
        if weights[k] < 1e-12:
            continue
        m = k - (n - 1)
        wk = weights[k]
        # S1 contribution: wk * h[j-m] for valid j-m
        j_start = max(0, m)
        j_end = min(n, n + m)
        if j_start < j_end:
            grad[j_start:j_end] -= wk * h[j_start - m : j_end - m]
        # S2 contribution: wk * h[j+m] for valid j+m
        j_start = max(0, -m)
        j_end = min(n, n - m)
        if j_start < j_end:
            grad[j_start:j_end] -= wk * h[j_start + m : j_end + m]
    
    return grad

def repair(h, target):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h_proj = np.clip(h - mu, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 0.999999
        else:
            mask = h_proj > 1e-6
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_soft_max(h, eps):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    weights = np.exp((corr - c_max) / eps)
    weights /= np.sum(weights)
    
    grad = np.ones(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        # Subtract wk * h[j-m] for valid j
        if m >= 0:
            grad[m:] -= wk * h[:n-m]
        else:
            grad[:n+m] -= wk * h[-m:]
        # Subtract wk * h[j+m] for valid j
        if m >= 0:
            grad[:n-m] -= wk * h[m:]
        else:
            grad[-m:] -= wk * h[:n+m]

            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

import numpy as np
import time

def repair(h, target):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h_proj = np.clip(h - mu, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = h_proj < 0.999999 if diff > 0 else h_proj > 1e-6
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_soft_max(h, eps):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    weights = np.exp((corr - c_max) / eps)
    weights /= np.sum(weights)
    grad = np.ones(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
        else:
            grad[:n + m] += wk * (1.0 - h[-m:])
            grad[-m:] -= wk * h[:n + m]
    return grad

def subgrad_descent(h0, target, n_iter, lr, eps, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        grad = grad_soft_max(h, eps)
        h_new = h - lr * grad
        h_new = repair(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 100 == 0:
            lr *= 0.95
    return best_h, best_c5

def lp_solve(h0, target):
    import cvxpy as cp
    n = len(h0)
    corr0 = compute_corr(h0)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    for k in range(2 * n - 1):
        m = k - (n - 1)
        grad = np.zeros(n)
        if m >= 0:
            grad[m:] += (1.0 - h0[:n - m])
            grad[:n - m] += (-h0[m:])
        else:
            grad[:n + m] += (1.0 - h0[-m:])
            grad[-m:] += (-h0[:n + m])
        constraints.append(corr0[k] + grad @ (h_var - h0) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    if hasattr(cp, 'SOLVER_MAP') and hasattr(cp.SOLVER_MAP, 'installed_solvers'):
        installed = cp.SOLVER_MAP.installed_solvers()
        solvers = [s for s in solvers if s in installed] + installed
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                return np.asarray(h_var.value).ravel()
        except Exception:
            continue
    return None

def lp_descent(h0, target, n_iters, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        h_new_raw = lp_solve(h, target)
        if h_new_raw is None:
            break
        h_new = repair(h_new_raw, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if abs(c5_new - best_c5) < 1e-7 and it > 5:
            break
        h = h_new
    return best_h, best_c5

    start_time = time.time()
    np.random.seed(seed)

        if c51 < best_c5:

    if time_left() > 120:
        h3, c53 = lp_descent(best_h, target, n_iters=50, time_limit=min(time_left() - 20, 120))
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()

        h4, c54 = subgrad_descent(best_h, target, n_iter=15000, lr=0.0015, eps=0.008, time_limit=min(time_left() - 5, 50))
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4.copy()

    x = np.linspace(0.0, 2.0, n)
    alts = []
    g1 = np.exp(-((x - 1.0) ** 2) / 0.15)

    tri = np.maximum(0.0, 1.0 - np.abs(x - 1.0) * 2.0)
    alts.append(repair(tri, target))

    for ha in alts:

        if c57 < best_c5:
            best_c5 = c57
            best_h = h7.copy()

    robust = repair(best_h, target)
    final_c5 = compute_c5(robust)
    return robust, final_c5, n
