        s = shifts[k]
        js_f = np.arange(n) + s
        mask_f = (js_f >= 0) & (js_f < n)
        idx_f = np.where(mask_f)[0]
        j_fwd.append(idx_f)
        js_fwd.append(js_f[mask_f])
        
        js_b = np.arange(n) - s
        mask_b = (js_b >= 0) & (js_b < n)
        idx_b = np.where(mask_b)[0]
        j_bwd.append(idx_b)
        js_bwd.append(js_b[mask_b])

        return np.correlate(h, 1.0 - h, mode="full")

        return np.max(compute_corr(h)) * dx

        corr = compute_corr(h)
        max_c = np.max(corr)
        # Softmax weights
        scaled = beta * (corr - max_c)
        e = np.exp(scaled - np.max(scaled))
        w = e / np.sum(e)

            idx_f = j_fwd[k]

            # Backward term
            idx_b = j_bwd[k]
            if len(idx_b) > 0:
                grad[idx_b] -= wk * h[js_bwd[k]]

        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)

    j_fwd = []
    js_fwd = []
    j_bwd = []
    js_bwd = []
    for k in range(n_corr):
        s = shifts[k]
        js_f = np.arange(n) + s
        mask_f = (js_f >= 0) & (js_f < n)
        idx_f = np.where(mask_f)[0]
        j_fwd.append(idx_f)
        js_fwd.append(js_f[mask_f])
        js_b = np.arange(n) - s
        mask_b = (js_b >= 0) & (js_b < n)
        idx_b = np.where(mask_b)[0]
        j_bwd.append(idx_b)
        js_bwd.append(js_b[mask_b])

    def compute_corr(h):
        return np.correlate(h, 1.0 - h, mode="full")

    def compute_c5(h):
        return np.max(compute_corr(h)) * dx

    def subgradient(h, beta):
        corr = compute_corr(h)
        max_c = np.max(corr)
        scaled = beta * (corr - max_c)
        max_scaled = np.max(scaled)
        e = np.exp(scaled - max_scaled)
        w = e / np.sum(e)
        grad = np.zeros(n)
        for k in range(n_corr):
            wk = w[k]
            if wk < 1e-8:
                continue
            idx_f = j_fwd[k]
            if len(idx_f) > 0:
                grad[idx_f] += wk * (1.0 - h[js_fwd[k]])
            idx_b = j_bwd[k]
            if len(idx_b) > 0:
                grad[idx_b] -= wk * h[js_bwd[k]]
        return grad * dx

    def project(v):
        vmin = v.min()
        vmax = v.max()
        lo = vmin - 1.0
        hi = vmax + 1.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            h = np.clip(v - mid, 0.0, 1.0)
            s = np.sum(h)
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    h_cur = project(h_init.copy())
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur

    def local_opt(h0, tlim, lr0, beta0):
        h = h0.copy()
        c5 = compute_c5(h)
        bh = h.copy()
        bc5 = c5
        h_avg = h.copy()
        lr = lr0
        beta = beta0
        stuck = 0
        avg_decay = 0.99
        tl = time.time()
        while time.time() - tl < tlim:
            g = subgradient(h, beta)
            h_new = project(h - lr * g)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                lr = min(lr * 1.02, 0.05)
                stuck = 0
            else:
                lr *= 0.7
                stuck += 1
                if stuck > 30:
                    beta = min(beta * 2.0, 2000.0)
                    lr = max(lr * 3.0, lr0 * 0.1)
                    stuck = 0
            avg_decay = min(avg_decay * 1.001, 0.999)
            h_avg = avg_decay * h_avg + (1.0 - avg_decay) * h
            c5_avg = compute_c5(h_avg)
            if c5_avg < bc5:
                bc5 = c5_avg
                bh = h_avg.copy()
            if c5 < bc5:
                bc5 = c5
                bh = h.copy()
        return bh, bc5

    # Phase 0: Try random binary starts
    for trial in range(8):
        h_rand = np.zeros(n)
        ones_idx = rng.choice(n, n // 2, replace=False)
        h_rand[ones_idx] = 1.0
        h_rp = project(h_rand.copy())
        h_t, c5_t = local_opt(h_rp, 20.0, 0.01, 50.0)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 1: Local opt from initial
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_cur, c5_cur = local_opt(project(h_init.copy()), min(150.0, remaining * 0.4), 0.01, 30.0)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 2: Basin hopping
    temp = 0.04
    min_temp = 0.002
    while time.time() - t0 < budget_s - 60:
        remaining = budget_s - 60 - (time.time() - t0)
        if remaining < 15:
            break
        perturb_scale = temp
        if rng.random() < 0.05:
            perturb_scale = temp * 3.0
        h_pert = project(best_h + perturb_scale * rng.randn(n))
        opt_time = max(5.0, min(remaining / 5.0, 40.0))
        h_c, c5_c = local_opt(h_pert, opt_time, 0.005, 50.0)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
            temp = max(min_temp, temp * 0.97)
        else:
            d = c5_c - best_c5
            if d < temp * 20:
                ratio = np.exp(-d / max(temp, 1e-10))
                if rng.random() < ratio:
                    best_h = h_c.copy()
                    best_c5 = c5_c
            temp = max(min_temp, temp * 0.997)

    # Phase 3: Final refinement
    remaining = budget_s - 5 - (time.time() - t0)
    if remaining > 10:
        hf, cf = local_opt(best_h, remaining, 0.001, 300.0)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    return best_h, best_c5, n
