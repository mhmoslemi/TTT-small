def evaluate_c5(h, dx):

    return float(np.max(corr * dx))

def make_valid(h, n):
    """Ensure h is valid: in [0,1] and sums to n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    current = np.sum(h)
    if current > 0:
        # Redistribute
        deficit = target - current
        if deficit > 0:
            # Need to add; add to positions where h < 1
            mask = h < 1.0
            available = np.sum(1.0 - h[mask])
            if available > 0:
                h[mask] += deficit * (1.0 - h[mask]) / available
            else:
                h += deficit / n
        else:
            # Need to subtract
            mask = h > 0.0
            available = np.sum(h[mask])
            if available > 0:
                h[mask] += deficit * h[mask] / available
            else:
                h += deficit / n
    h = np.clip(h, 0, 1)
    # Final adjustment
    remaining = target - np.sum(h)
    if abs(remaining) > 1e-10:
        # Add to first available position
        for i in range(n):
            if remaining > 0 and h[i] < 1.0:
                add = min(remaining, 1.0 - h[i])
                h[i] += add
                remaining -= add
            elif remaining < 0 and h[i] > 0.0:
                sub = min(-remaining, h[i])
                h[i] -= sub
                remaining += sub
            if abs(remaining) < 1e-10:
                break

        j = np.random.randint(n)

        if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:

        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-12:
                h = h + diff * room / total
            else:
                h = h + diff / n

            if total > 1e-12:

import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-14:
                h = h + diff * room / total
            else:
                h = h + diff / n
        else:
            room = h.copy()
            total = room.sum()
            if total > 1e-14:
                h = h + diff * room / total
            else:
                h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        residual = target - h.sum()
        if abs(residual) > 1e-10:
            order = np.arange(n)
            for idx in order:
                if residual > 0 and h[idx] < 1.0:
                    add = min(residual, 1.0 - h[idx])
                    h[idx] += add
                    residual -= add
                elif residual < 0 and h[idx] > 0.0:
                    sub = min(-residual, h[idx])
                    h[idx] -= sub
                    residual += sub
                if abs(residual) < 1e-12:
                    break
    return h

def ils_search(h_init, dx, max_time, rng, step_lo=0.005, step_hi=0.12, perturb_rate=0.05):
    n = len(h_init)
    h = h_init.copy()
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    t0 = time.time()
    it = 0
    stagnation = 0

    while time.time() - t0 < max_time:
        it += 1
        progress = min(1.0, it / 8000.0)
        cur_hi = step_hi * (1.0 - 0.6 * progress)
        cur_lo = max(0.001, step_lo * (1.0 - 0.5 * progress))

        batch_size = 20
        improved_in_batch = False
        for _ in range(batch_size):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            step = rng.uniform(cur_lo, cur_hi)
            d = rng.choice([1, -1])
            delta = d * step
            nh_i = h[i] + delta
            nh_j = h[j] - delta
            if 0.0 <= nh_i <= 1.0 and 0.0 <= nh_j <= 1.0:
                h[i] = nh_i
                h[j] = nh_j
                c5 = evaluate_c5(h, dx)
                if c5 < current_c5:
                    current_c5 = c5
                    improved_in_batch = True
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        stagnation = 0
                else:
                    h[i] -= delta
                    h[j] += delta

        if not improved_in_batch:
            stagnation += 1
        else:
            stagnation = 0

        if stagnation > 5 or (perturb_rate > 0 and rng.random() < perturb_rate):
            n_pert = rng.randint(1, 4)
            for _ in range(n_pert):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                pstep = rng.uniform(0.02, step_hi)
                d = rng.choice([1, -1])
                delta = d * pstep
                nh_i = h[i] + delta
                nh_j = h[j] - delta
                if 0.0 <= nh_i <= 1.0 and 0.0 <= nh_j <= 1.0:
                    h[i] = nh_i
                    h[j] = nh_j
                    current_c5 = evaluate_c5(h, dx)
            stagnation = 0

    return best_h, best_c5

def gen_cos_h(n, rng, freq=1.0, amp=0.4):
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return make_feasible(h, n)

def gen_random_h(n, rng):
    h = np.full(n, 0.5)
    for _ in range(5 * n):

        j = rng.randint(n)

            delta = rng.uniform(-0.35, 0.35)
            if 0.0 <= h[i] + delta <= 1.0 and 0.0 <= h[j] - delta <= 1.0:
                h[i] += delta
                h[j] -= delta
    return make_feasible(h, n)
