import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_max_val(x, nu):
    m = np.max(x)
    shifted = x - m
    return float(m + np.log(np.sum(np.exp(nu * shifted))) / nu)

def project_to_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def interpolate_to_n(parent, n_parent, n_new):
    x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    f = interp1d(x_parent, parent, kind='linear', bounds_error=False, fill_value='extrapolate')
    new_h = f(x_new)
    return project_to_feasible(new_h, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = len(parent)
        best_h = project_to_feasible(parent, n_parent)
        best_c5 = compute_c5(best_h, n_parent)
    else:
        n_parent = 70
        parent = np.ones(n_parent) * 0.5
        best_h = parent.copy()
        best_c5 = compute_c5(best_h, n_parent)
    
    def elapsed():
        return time.time() - start_time
    
    def has_budget(remaining=30):
        return elapsed() < budget_s - remaining
    
    grid_sizes = [n_parent, 98, 140, 196]
    
    for n in grid_sizes:
        if not has_budget(120):
            break
        
        n_dx = 2.0 / n
        h0 = interpolate_to_n(parent, n_parent, n)
        
        constraints = [{'type': 'eq', 'fun': lambda h, n=n: np.sum(h) - n / 2.0}]
        bounds = [(0.0, 1.0)] * n
        
        for nu in [50, 100, 200]:
            if not has_budget(60):
                break
            
            def obj_func(h, n=n, n_dx=n_dx, nu=nu):
                corr = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max_val(corr * n_dx, nu)

                if not has_budget(30):
                    break

                    c5_val = compute_c5(h_opt, n)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_opt.copy()
                        h0 = h_opt.copy()
                except Exception:
                    pass
    
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_max_val(x, nu):
    m = np.max(x)
    shifted = x - m
    return float(m + np.log(np.sum(np.exp(nu * shifted))) / nu)

def project_to_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def interpolate_to_n(parent, n_parent, n_new):
    x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    f = interp1d(x_parent, parent, kind='linear', bounds_error=False, fill_value='extrapolate')
    new_h = f(x_new)
    return project_to_feasible(new_h, n_new)

    return np.sum(h) - n / 2.0
