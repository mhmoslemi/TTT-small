    return np.max(corr) * dx

    # Convert to 0-1 maintaining sum = n/2
    h = np.round(h)
    h = np.clip(h, 0, 1)
    # Adjust to get exactly n/2 ones
    current_sum = h.sum()
    target = n / 2
    if current_sum < target:
        # Need to flip some 0s to 1s
        need = int(target - current_sum)
        zero_indices = np.where(h == 0)[0]
        np.random.shuffle(zero_indices)
        h[zero_indices[:need]] = 1
    elif current_sum > target:
        need = int(current_sum - target)
        one_indices = np.where(h == 1)[0]
        np.random.shuffle(one_indices)
        h[one_indices[:need]] = 0
    return h

def simulated_annealing(h, n, dx, start_temp, end_temp, max_iters, rng):

    current_h = h.copy()

            current_c5 = new_c5

                best_h = current_h.copy()

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n, rng):
    """Convert h to a 0-1 array with exactly n//2 ones."""
    n_ones = n // 2
    h = np.array(h, dtype=float)
    # Sort by value descending, take top n_ones
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:n_ones]] = 1.0
    return result

def sa_step(h, a, b, n):
    """Apply swap of positions a and b (h[a]=1 -> 0, h[b]=0 -> 1)."""
    h[a] = 0.0
    h[b] = 1.0

def simulated_annealing(h, n, dx, temp_start, temp_end, n_iters, rng):
    """Simulated annealing with swap moves."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    n_ones = n // 2
    
    one_mask = cur_h == 1.0
    zero_mask = cur_h == 0.0
    
    for i in range(n_iters):
        # Pick random positions
        ones_idx = np.where(one_mask)[0]
        zeros_idx = np.where(zero_mask)[0]
        a = ones_idx[rng.integers(len(ones_idx))]
        b = zeros_idx[rng.integers(len(zeros_idx))]
        
        # Perform swap
        cur_h[a] = 0.0
        cur_h[b] = 1.0
        one_mask[a] = False
        one_mask[b] = True
        zero_mask[a] = True
        zero_mask[b] = False
        
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        
        t = temp_start * (temp_end / temp_start) ** (i / max(n_iters - 1, 1))
        
        if delta < 0 or rng.random() < math.exp(-delta / max(t, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # Revert
            cur_h[a] = 1.0
            cur_h[b] = 0.0
            one_mask[a] = True
            one_mask[b] = False
            zero_mask[a] = False
            zero_mask[b] = True
    
    return best_h, best_c5

                cur_h[a] = 0.0
                cur_h[b] = 1.0
            cur_c5 = compute_c5(cur_h)
        else:
            cur_h[best_a] = 0.0
            cur_h[best_b] = 1.0
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    
    return best_h, best_c5

def _scipy_eq(x, n):
    return np.sum(x[:n]) - n / 2.0

def _scipy_obj_and_grad(x, n):
    h = x[:n]
    C = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = C + 50.0 * max(0.0, float(np.max(corr * dx) - C))
    return val

def scipy_polish(h, n, max_iter=200):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    x0 = np.concatenate([h, [compute_c5(h) + 0.02]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def objective(x):
        hv = x[:n]
        C = x[n]
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        viol = max(0.0, float(np.max(corr * dx)) - C)
        return C + 100.0 * viol * viol + viol
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n / 2.0}]
    
    def make_ineq(k):
        def f(x):
            hv = x[:n]
            C = x[n]
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            return C - corr[k] * dx
        return f
    
    for k in range(n_corr):
        constraints.append({'type': 'ineq', 'fun': make_ineq(k)})
    
    try:
        result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-12})
        if result.fun < compute_c5(h) + 0.001:
            h_opt = np.clip(result.x[:n], 0, 1)
            h_bin = make_binary(h_opt, n)
            c5 = compute_c5(h_bin)
            if c5 < compute_c5(h):
                return h_bin, c5
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    def elapsed():
        return time.time() - t0
    
    def time_left():
        return budget_s - elapsed()
    
    # Phase 1: Initial construction
    if 'initial_h_values' in globals():
        try:
            h_init = np.array(initial_h_values, dtype=float)
            n_init = len(h_init)
            h_bin = make_binary(h_init, n_init)
            c5 = compute_c5(h_bin)
            best_c5 = c5
            best_h = h_bin.copy()
            best_n = n_init
        except Exception:
            pass
    
    # Phase 2: Multi-start SA
    n_list = []
    for n in [70, 84, 98, 112, 126, 140, 168, 196, 210, 245, 280, 315, 350, 420]:
        if n % 2 == 0:
            n_list.append(n)
    
    starts_per_n = 3
    
    for idx_n, n in enumerate(n_list):
        if time_left() < 250:
            break
        
        remaining_for_this_n = min(
            (time_left() - 250) / max(1, len(n_list) - idx_n),
            budget_s * 0.5
        )
        
        for s in range(starts_per_n):
            if time_left() < 250:
                break
            
            h_start = gen_pattern(n, (idx_n + s) % 5, rng)
            
            per_iter_time = 0.00005 * (1.0 + n / 100.0)
            n_iters = int(min(500000, max(10000, remaining_for_this_n / (starts_per_n - s) / per_iter_time * 0.7)))
            n_iters = min(n_iters, int(3000000 / max(1, n)))
            
            t_start = temp_start = 0.04
            temp_end = 0.0003
            h_sa, c5_sa = sa_optimize(h_start, n, temp_start, temp_end, n_iters, rng, two_swap_prob=0.12)
            
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
    
    # Phase 3: Local search on best
    if best_h is not None and time_left() > 60:
        n = best_n
        if n <= 120:
            rounds = 30
        elif n <= 180:
            rounds = 15
        else:
            rounds = 5
        h_ls, c5_ls = local_search(best_h, n, rounds, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
    
    # Phase 4: SA polish on improved
    if best_h is not None and time_left() > 30:
        remaining = time_left() - 10
        n_iters = int(remaining * 20000.0)
        n_iters = min(n_iters, 5000000)
        h_sa, c5_sa = sa_optimize(best_h, best_n, 0.008, 0.00005, n_iters, rng, two_swap_prob=0.08)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    
    # Phase 5: Local search again
    if best_h is not None and time_left() > 20:
        n = best_n
        rounds = 10 if n <= 140 else 3
        h_ls, c5_ls = local_search(best_h, n, rounds, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
    
    # Phase 6: Final SA
    if best_h is not None and time_left() > 5:
        n_iters = int((time_left() - 3) * 30000.0)
        n_iters = max(n_iters, 5000)
        h_sa, c5_sa = sa_optimize(best_h, best_n, 0.003, 0.00001, n_iters, rng, two_swap_prob=0.05)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    
    return best_h, best_c5, best_n
