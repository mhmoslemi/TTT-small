def _project(h, n, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)


def _score(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))
