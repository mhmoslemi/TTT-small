def grad_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    i = np.arange(n)
    valid1 = (i + lag >= 0) & (i + lag < n)
    grad[valid1] += (1.0 - h[i[valid1] + lag])
    valid2 = (i - lag >= 0) & (i - lag < n)
    grad[valid2] -= h[i[valid2] - lag]
    return grad

    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)

        h_trial = np.clip(h - step * g, 0.0, 1.0)
        h_trial = repair_sum(h_trial, target)
        c5_trial = np.max(corr_values(h_trial, dx))

        if c5_trial < c5_cur:
            h = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            step = max(step * 0.97, 0.002)

        if no_imp > patience:
            d = rng.normal(0, 0.05, n)
            d -= d.mean()
            h = repair_sum(h + d, target)
            no_imp = 0
            step = 0.025

    T = 0.003
    T_min = 3e-5
    sa_alpha = 0.9995
    noise = 0.018

    while time.time() < phase2_end:
        h_trial = h + rng.normal(0, noise, n)
        h_trial = repair_sum(h_trial, target)
        c5_t = np.max(corr_values(h_trial, dx))
        c5_c = np.max(corr_values(h, dx))

        if c5_t < c5_c or (T > T_min and rng.random() < np.exp(-(c5_t - c5_c) / T)):
            h = h_trial
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
        T = max(T * sa_alpha, T_min)
        noise = max(0.004, noise * 0.9998)

    noise2 = 0.006
    T2 = 1e-5
    while time.time() < deadline:
        for _ in range(3):
            idx = rng.integers(0, n)
            delta = rng.choice([-0.01, 0.01, -0.02, 0.02])
            partner = rng.integers(0, n)
            while partner == idx:
                partner = rng.integers(0, n)
            h_trial = h.copy()
            h_trial[idx] += delta
            h_trial[partner] -= delta
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = repair_sum(h_trial, target)
            c5_t = np.max(corr_values(h_trial, dx))
            c5_c = np.max(corr_values(h, dx))
            if c5_t < c5_c:
                h = h_trial
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()

    return best_h, best_c5, n
