import numpy as np
from scipy.optimize import minimize
import time

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return corr

def smooth_max(corr, alpha=100):
    # corr is array of overlap values for each lag
    # Return smooth max
    max_val = np.max(corr)
    shifted = alpha * (corr - max_val)
    exp_sum = np.sum(np.exp(shifted))
    return max_val + np.log(exp_sum) / alpha

def obj_and_grad(h, alpha=100, lam=1.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    
    # Smooth max
    max_val = np.max(corr)
    shifted = alpha * (corr - max_val)
    exp_shifted = np.exp(shifted)
    f = max_val + np.log(np.sum(exp_shifted)) / alpha
    
    # Gradient of smooth max w.r.t. each lag
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of each lag w.r.t. h
    # grad[k] = d(corr[k])/dh[j] for each j
    # We need the full gradient: sum_k weights[k] * d(corr[k])/dh
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # pairs (i, i+lag) for i = 0..n-1-lag
            # d/dh[i] = 1 - h[i+lag]
            # d/dh[i+lag] = -h[i]
            for i in range(n - lag):
                grad[i] += weights[k] * dx * (1 - h[i + lag])
                grad[i + lag] += weights[k] * dx * (-h[i])
        else:
            # lag < 0, let l = -lag
            l = -lag
            # pairs (i, i-lag) = (i, i+l) for i = l..n-1
            # which is (i, i-l) for i = l..n-1, i.e., (i-l, i) for i=l..n-1
            # d/dh[i] = 1 - h[i-l]
            # d/dh[i-l] = -h[i]
            for i in range(l, n):
                grad[i] += weights[k] * dx * (1 - h[i - l])
                grad[i - l] += weights[k] * dx * (-h[i])
    
    # Add penalty gradient for sum constraint
    sum_h = np.sum(h)
    target = n / 2.0
    grad += 2 * lam * (sum_h - target)
    f += lam * (sum_h - target) ** 2
    
    return f, grad

    x_new = np.linspace(0, 2.0, n_new, endpoint=False)

    """Adjust h to have the target sum while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    current = np.sum(h)
    diff = target_sum - current
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        mask = h < 1.0
        n_inc = np.sum(mask)
        if n_inc == 0:
            return h
        per = diff / n_inc
        h[mask] = np.minimum(h[mask] + per, 1.0)
    else:
        # Need to decrease sum: subtract from elements that are > 0
        mask = h > 0.0
        n_dec = np.sum(mask)
        if n_dec == 0:
            return h
        per = diff / n_dec
        h[mask] = np.maximum(h[mask] + per, 0.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    
    # Smooth max
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    f = cmax + np.log(exp_sum) / alpha
    
    # Penalty for sum constraint
    sum_h = np.sum(h)
    f += penalty * (sum_h - target_sum) ** 2

    grad = np.zeros(n)

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def upscale(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    current = np.sum(h)
    diff = target_sum - current
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        n_inc = np.sum(mask)
        if n_inc == 0:
            return h
        h[mask] += diff / n_inc
        h = np.minimum(h, 1.0)
    else:
        mask = h > 1e-12
        n_dec = np.sum(mask)
        if n_dec == 0:
            return h
        h[mask] += diff / n_dec
        h = np.maximum(h, 0.0)
    # Minor correction for numerical issues
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += residual / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

        if k >= 0:
            g[k:] += wi * (1.0 - h[:n - k]) * dx
            g[:n - k] += wi * (-h[k:]) * dx
        else:
            l = -k
            g[:n - l] += wi * (1.0 - h[l:]) * dx
            g[l:] += wi * (-h[:n - l]) * dx
    g += 2.0 * penalty * (sh - target_sum)
    return f, g


def local_search(h, target_sum, n_iter=200, step=0.05):
    n = len(h)
    best = h.copy()
    best_c5 = evaluate_c5(best)
    rng = np.random.default_rng(12345)
    for _ in range(n_iter):
        idx = rng.integers(0, n)
        direction = rng.choice([-1, 1])
        delta = direction * step * rng.uniform(0.3, 1.0)
        if best[idx] + delta < 0 or best[idx] + delta > 1:
            continue
        trial = best.copy()
        trial[idx] += delta
        # Compensate sum on a random other index
        others = [j for j in range(n) if j != idx]
        if not others:
            continue
        pad = others[rng.integers(0, len(others))]
        if padding_ok(trial, pad, -delta):
            trial[pad] -= delta
            trial = np.clip(trial, 0.0, 1.0)
            c5 = evaluate_c5(trial)
            if c5 < best_c5:
                best_c5 = c5
                best = trial
    return best, best_c5


def padding_ok(trial, idx, delta):
    v = trial[idx] + delta
    return 0.0 <= v <= 1.0


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    best_h = fix_sum(h_init.copy(), n0 / 2.0)
    best_c5 = evaluate_c5(best_h)

    resolutions = [n0]
    r = n0 * 2
    while r <= 360:
        resolutions.append(r)
        r *= 2

    for n_res in resolutions:
        if time.time() - t0 > budget_s - 60:
            break

        target_sum = n_res / 2.0
        h_cur = fix_sum(upscale(best_h, n_res), target_sum)
        cur_best = evaluate_c5(h_cur)

        stages = [
            (20, 50.0),
            (50, 200.0),
            (100, 500.0),
            (200, 1000.0),
            (400, 2000.0),
        ]

        time_left = budget_s - (time.time() - t0) - 40
        t_per = max(4.0, time_left / len(stages))

        for alpha, penalty in stages:
            if time.time() - t0 > budget_s - 15:
                break
            tl = min(t_per, budget_s - (time.time() - t0) - 10)
            if tl < 3:
                break
            try:
                res = minimize(
                    lambda x, a=alpha, p=penalty: obj_grad(x, target_sum, a, p),
                    h_cur,
                    jac=True,
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n_res,
                    options={'maxiter': 5000, 'maxfun': 10000, 'max_time': tl, 'gtol': 1e-9}
                )
                hc = fix_sum(np.clip(res.x, 0.0, 1.0), target_sum)
                c5 = evaluate_c5(hc)
                if c5 < cur_best:
                    h_cur = hc
                    cur_best = c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

        # Local search refinement
        if time.time() - t0 < budget_s - 20:
            n_ls = max(50, int(min(300, (budget_s - (time.time() - t0) - 10) * 20)))
            h_ls, c5_ls = local_search(h_cur, target_sum, n_iter=n_ls)
            if c5_ls < cur_best:
                h_cur = h_ls
                cur_best = c5_ls
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
