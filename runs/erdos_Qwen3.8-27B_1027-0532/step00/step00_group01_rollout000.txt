import numpy as np
import time

def compute_overlaps(h):
    d = 1.0 - h
    return np.correlate(h, d, mode='full')

def compute_c5(h, dx):
    corr = compute_overlaps(h)
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    lo, hi = -4.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Final correction for numerical precision
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 0) & (result < 1)
        n_mask = np.count_nonzero(mask)
        if n_mask > 0:
            result[mask] += diff / n_mask
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_gradient(h, T):
    n = len(h)
    corr = compute_overlaps(h)
    cmax = np.max(corr)
    weights = np.exp((corr - cmax) / T)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            nk = -k
            grad[:n - nk] += w * (1.0 - h[nk:])
            grad[nk:] -= w * h[:nk]
    return grad

def generate_alternative_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    h = project_box_sum(h + rng.randn(n) * 0.1, n / 2.0)
    return h

def optimize_from(h, dx, target_sum, budget_s, start_time, seed_offset):
    rng = np.random.RandomState(seed_offset)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    T = 0.03
    lr = 0.008
    momentum = np.zeros(n)
    beta = 0.9
    no_improve = 0
    max_no_improve = 500

    while time.time() - start_time < budget_s:
        grad = smooth_gradient(best_h, T)
        momentum = beta * momentum + (1 - beta) * grad
        step = momentum
        h_new = best_h - lr * step
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            no_improve = 0
            T = max(0.001, T * 0.9995)
        else:
            no_improve += 1
            lr = max(0.0001, lr * 0.9)
            T = max(0.001, T * 0.98)

            if no_improve > max_no_improve:
                # Large perturbation to escape
                scale = 0.05 + 0.15 * rng.random()
                h_pert = best_h + rng.randn(n) * scale
                h_pert = project_box_sum(h_pert, target_sum)
                c5_pert = compute_c5(h_pert, dx)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert
                    T = 0.02
                    lr = 0.005
                else:

                    h_pert2 = project_box_sum(h_pert2, target_sum)

                    if c5_pert2 < best_c5:
                        best_c5 = c5_pert2
                        best_h = h_pert2
                no_improve = 0
                T = 0.015
                lr = 0.004

    return best_h, best_c5

def local_search(h, dx, target_sum, start_time, budget_s, seed_offset):
