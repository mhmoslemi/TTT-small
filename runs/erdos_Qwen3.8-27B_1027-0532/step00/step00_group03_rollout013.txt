def evaluate_c5(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (corr * dx).max()

def minimize_1d_upper_envelope(C, D, t_lo, t_hi, max_iter=80):
    if t_lo >= t_hi - 1e-14:
        return t_lo
    for _ in range(max_iter):
        if t_hi - t_lo < 1e-12:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        vals = C + t_mid * D
        v_max = vals.max()
        # Find all active (within tolerance)
        active_mask = (v_max - vals) < 1e-9
        if active_mask.any():
            g_lo = D[active_mask].min()
            g_hi = D[active_mask].max()
            if g_hi < -1e-12:
                t_lo = t_mid
            elif g_lo > 1e-12:
                t_hi = t_mid
            else:
                break
        else:
            k_star = np.argmax(vals)
            if D[k_star] < 0:
                t_lo = t_mid
            else:
                t_hi = t_mid
    return 0.5 * (t_lo + t_hi)

def coordinate_descent(h, n, dx, max_sweeps=200, time_limit=None, start_time=None, rng=None, shake_interval=50, shake_magnitude=0.05):
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n, dx)
    
    m = 2 * n - 1  # number of correlation values
    lag = np.arange(m) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time_limit is not None:
            elapsed = time.time() - start_time
            if elapsed > time_limit:
                break
        
        # Optional shaking
        if rng is not None and sweep > 0 and sweep % shake_interval == 0:
            pert = rng.uniform(-shake_magnitude, shake_magnitude, n)
            h_pert = np.clip(h + pert, 0, 1)
            # Fix sum
            target = n / 2.0
            diff = target - h_pert.sum()
            # Distribute uniformly among coordinates that can absorb it
            if diff > 0:
                room = 1.0 - h_pert
                total_room = room.sum()
                if total_room > 0:
                    h_pert += diff * room / total_room
            elif diff < 0:
                excess = h_pert
                total_excess = excess.sum()
                if total_excess > 0:
                    h_pert += diff * excess / total_excess
            h_pert = np.clip(h_pert, 0, 1)
            c5_pert = evaluate_c5(h_pert, n, dx)
            if c5_pert < best_c5 * 1.05:  # accept if not too bad
                h = h_pert
        
        # Coordinate sweep
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            old_val = h[i]
            if old_val < 1e-12 or old_val > 1.0 - 1e-12:
                # Try moving away from boundary
                target_val = 0.5 if old_val < 0.5 else 0.5
                # Just compute the optimal
                pass
            t_lo = -old_val
            t_hi = 1.0 - old_val
            if t_hi - t_lo < 1e-14:
                continue
            
            C = np.correlate(h, 1.0 - h, mode="full")
            
            # Compute D[i, :]
            i_plus = i + lag
            i_minus = i - lag
            D = np.zeros(m)
            valid_p = (i_plus >= 0) & (i_plus < n)
            valid_m = (i_minus >= 0) & (i_minus < n)
            D[valid_p] = 1.0 - h[i_plus[valid_p]]
            D[valid_m] -= h[i_minus[valid_m]]

            h[i] = old_val + t_opt
        
        c5 = evaluate_c5(h, n, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def upscale(h, factor):
    """Upscale h by repeating each value 'factor' times."""
    n_old = len(h)
    n_new = n_old * factor
    h_new = np.repeat(h, factor)
    return h_new

        best_overall_h = h
        best_overall_c5 = evaluate_c5(h, n, dx)

def coordinate_descent(h, n, dx, max_sweeps=1000, time_limit=60.0, start_time=0.0, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    m = 2 * n - 1
    lag_arr = np.arange(m, dtype=np.float64) - (n - 1)
    C = np.correlate(h, 1.0 - h, mode="full")

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        if sweep > 0 and sweep % 60 == 0:
            mag = 0.015 * np.exp(-sweep / 300.0)
            if mag > 0.0005:
                pert = rng.uniform(-mag, mag, n)
                h_pert = np.clip(h + pert, 0.0, 1.0)
                h_pert = fix_sum(h_pert, n / 2.0)
                c5_pert = evaluate_c5(h_pert, dx)
                if c5_pert < best_c5 * 1.005:
                    h = h_pert
                    C = np.correlate(h, 1.0 - h, mode="full")

        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            t_lo = -old_val
            t_hi = 1.0 - old_val
            if t_hi - t_lo < 1e-14:
                continue

            i_plus = i + lag_arr
            i_minus = i - lag_arr
            D = np.zeros(m)
            vp = (i_plus >= 0) & (i_plus < n)
            vm = (i_minus >= 0) & (i_minus < n)
            D[vp] = 1.0 - h[i_plus[vp]]
            D[vm] -= h[i_minus[vm]]

            t_opt = minimize_1d(C, D, t_lo, t_hi)
            new_val = old_val + t_opt
            delta = new_val - old_val
            if abs(delta) > 1e-14:
                h[i] = new_val
                C += delta * D

        if sweep % 8 == 0:
            C = np.correlate(h, 1.0 - h, mode="full")

        if sweep % 4 == 0 or sweep == max_sweeps - 1:
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    c5_final = evaluate_c5(best_h, dx)
    return best_h, c5_final


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()

    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Stage 1: n=45
    if h0 is not None and len(h0) == 45:
        h = h0.copy()
    else:
        h = np.full(45, 0.5)
    n = 45
    dx = 2.0 / n
    h = fix_sum(h, n / 2.0)

    alloc = min(300, max(60, int((deadline - time.time()) * 0.25)))
    h, c5 = coordinate_descent(h, n, dx, max_sweeps=5000, time_limit=alloc, start_time=start_time, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

    # Restarts at n=45
    rng_r = np.random.RandomState(seed + 1000)
    for rs in range(4):
        if time.time() > deadline - 120:
            break
        h_r = best_h + rng_r.normal(0, 0.12, 45)
        h_r = np.clip(h_r, 0.0, 1.0)
        h_r = fix_sum(h_r, 22.5)
        tl = min(80, max(15, (deadline - time.time()) * 0.1))
        h_r, c5_r = coordinate_descent(h_r, 45, dx, max_sweeps=2000, time_limit=tl, start_time=start_time, seed=seed + 2000 + rs)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = 45

    # Try a structured "golden ratio" initialization
    n45 = 45
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h_golden = np.zeros(45)
    positions = []
    for j in range(45):
        frac = (j * (1.0 / phi)) % 1.0
        h_golden[j] = 1.0 if frac < 0.5 else 0.0
    h_golden = fix_sum(h_golden, 22.5)
    if time.time() < deadline - 60:
        h_g, c5_g = coordinate_descent(h_golden, 45, dx, max_sweeps=2000, time_limit=60, start_time=start_time, seed=seed + 3000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = 45

    # Stage 2: n=90
    if time.time() < deadline - 60:
        h90 = np.repeat(best_h, 2)
        dx90 = 2.0 / 90
        h90 = fix_sum(h90, 45.0)
        tl = min(200, max(30, (deadline - time.time()) * 0.25))
        h90, c5_90 = coordinate_descent(h90, 90, dx90, max_sweeps=3000, time_limit=tl, start_time=start_time, seed=seed + 4000)
        if c5_90 < best_c5:
            best_c5 = c5_90
            best_h = h90.copy()
            best_n = 90

    # Stage 3: n=180
    if time.time() < deadline - 45:
        factor = 180 // best_n
        if factor >= 1:
            h180 = np.repeat(best_h, factor)
            dx180 = 2.0 / 180
            h180 = fix_sum(h180, 90.0)
            tl = min(200, max(20, (deadline - time.time()) * 0.25))
            h180, c5_180 = coordinate_descent(h180, 180, dx180, max_sweeps=1000, time_limit=tl, start_time=start_time, seed=seed + 5000)
            if c5_180 < best_c5:
                best_c5 = c5_180
                best_h = h180.copy()
                best_n = 180

    # Stage 4: n=360
    if time.time() < deadline - 30:
        n_target = 360
        factor = n_target // best_n
        if factor >= 1:
            h360 = np.repeat(best_h, factor)
            dx360 = 2.0 / 360
            h360 = fix_sum(h360, 180.0)
            tl = min(150, max(15, (deadline - time.time()) * 0.3))
            h360, c5_360 = coordinate_descent(h360, 360, dx360, max_sweeps=500, time_limit=tl, start_time=start_time, seed=seed + 6000)
            if c5_360 < best_c5:
                best_c5 = c5_360
                best_h = h360.copy()
                best_n = n_target

    # Final polish: one more round at best_n if time remains
    if time.time() < deadline - 10:
        tl = max(5, (deadline - time.time()) * 0.6)
        best_h, c5_final = coordinate_descent(best_h, best_n, 2.0 / best_n, max_sweeps=5000, time_limit=tl, start_time=start_time, seed=seed + 7000)
        if c5_final < best_c5:
            best_c5 = c5_final

    if best_h is None:
        best_h = h
        best_c5 = evaluate_c5(best_h, 2.0 / len(best_h))
        best_n = len(best_h)

    return best_h, best_c5, best_n
