    corr = np.correlate(h, 1 - h, mode="full") * dx

def grad_correlation(h, l, dx):
    """Gradient of corr at lag l (where l = k - (n-1))"""
    n = len(h)
    grad = np.zeros(n)
    if l >= 0:
        grad[l:] += (1.0 - h[:n-l])
        grad[:n-l] -= h[l:]
    else:
        p = -l
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] -= h[:n-p]
    return grad * dx

    lo = h.min() - target_sum
    hi = h.max()

        proj = np.clip(h - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid
