def project_to_simplex(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

    n_parent = len(parent)
    x_parent = np.linspace(0.0, 2.0, n_parent)
    x_new = np.linspace(0.0, 2.0, n)
