Let me also add a "gradient step" approach as a fallback:
- Compute the subgradient of max(c_k) (just the gradient of the argmax component)
- Take a step in the negative direction
- Project onto {sum(h) = n/2, 0 <= h <= 1}

The projection onto this set can be done as follows:
Given a vector v, find h = argmin ||h - v||^2 subject to sum(h) = n/2, 0 <= h <= 1.
This is a convex QP with a box constraint and one equality constraint.
By KKT conditions: h_i = clip(v_i - λ, 0, 1) for some λ.
We find λ by bisection such that sum(clip(v_i - λ, 0, 1)) = n/2.

This is efficient (O(n log n) per projection).

```python
def project_onto_feasible(v, target):
    # Minimize ||h - v||^2 s.t. sum(h) = target, 0 <= h <= 1
    # KKT: h_i = clip(v_i - lam, 0, 1), find lam such that sum = target
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    return h

def subgradient_step(h, step_size):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    k_max = np.argmax(corr)
    s = k_max - (n-1)  # the shift
    # Gradient of c_s at h
    g = np.zeros(n)
    if s >= 0:
        g[0:n-s] += (1.0 - h[s:n])
        g[s:n] += (-h[0:n-s])
    else:
        g[-s:n] += (1.0 - h[0:n+s])
        g[0:n-s] += (-h[-s:n])
    # Step
    v = h - step_size * g
    # Project
    h_new = project_onto_feasible(v, n/2.0)
    return h_new, k_max
```

```python
            if time.time() - t0 > budget_s * 0.88:
                break
            hp = h_cur.copy()
            n_pert = max(3, tn // 12)
            np_idx = rng.randint(0, tn, size=n_pert)
            hp[np_idx] += rng.uniform(-0.12, 0.12, size=n_pert)
            hp = renorm(hp, tn)
            _g_beta = 150.0
            try:
                res = minimize(smooth_max_obj, hp, method='SLSQP',
                               jac=smooth_max_grad, bounds=bnds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-15})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = renorm(res.x, tn)
                    c5_new = compute_c5(h_new)
                    if c5_new < cur_c5:
                        h_cur = h_new
                        cur_c5 = c5_new
            except Exception:
                pass

        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5

    return best_h, compute_c5(best_h), len(best_h)
```
