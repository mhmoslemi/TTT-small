from scipy.optimize import minimize
import time

def repair_point(x, n, target_sum):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if abs(s - target_sum) > 1e-8 and s > 0:
        scale = target_sum / s
        x = x * scale
        x = np.clip(x, 0.0, 1.0)
        s2 = np.sum(x)
        diff = target_sum - s2
        if abs(diff) > 1e-8:
            if diff > 0:
                mask = x < 1.0 - 1e-12
            else:
                mask = x > 1e-12
            nf = np.sum(mask)
            if nf > 0:
                x[mask] += diff / nf
                x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj_and_grad(h, beta, dx, n):
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = beta * (vals - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))

        s = p - (n - 1)

import numpy as np
from scipy.optimize import minimize

def repair_point(x, n, target_sum):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if abs(s - target_sum) > 1e-8 and s > 0:
        x = x * (target_sum / s)
        x = np.clip(x, 0.0, 1.0)
        s2 = np.sum(x)
        diff = target_sum - s2
        if abs(diff) > 1e-8:
            if diff > 0:
                mask = x < 1.0 - 1e-12
            else:
                mask = x > 1e-12
            nf = int(np.sum(mask))
            if nf > 0:
                x[mask] += diff / nf
                x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, beta, dx, n):
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    L = m + np.log(sum_exp) / beta
    w = exp_shifted / sum_exp
    grad = np.zeros(n)
    for p in range(len(vals)):
        wp = w[p]
        if wp < 1e-10:
            continue
        s = p - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, s + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wp * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] += wp * (-h[lo2 + s:hi2 + s])
    grad *= dx
