def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_max_and_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    e = np.exp(a - a_max)
    e_sum = e.sum()
    w = e / e_sum
    obj = (a_max + np.log(e_sum)) / alpha * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-10:
            continue
        # Term 1: (1-h[j+s]) for j in [max(0,-s), min(n, n-s))
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 + s: hi1 + s])
        # Term 2: -h[j-s] for j in [max(0,s), min(n, n+s))
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 - s: hi2 - s]
    
    return obj, grad * dx

def projected_gradient_descent(h0, target_sum, alphas, steps_per_alpha, lr_schedule, time_limit, start_time):
    n = len(h0)
    best_h = np.copy(h0)
    best_c5 = compute_c5(h0)
    
    for alpha in alphas:
        h = np.copy(h0)
        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            obj, grad = smooth_max_and_grad(h, alpha)
            # Adaptive learning rate
            lr = lr_schedule(step, alpha)
            h_new = h - lr * grad
            h_new = project_to_constraint(h_new, target_sum)
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = np.copy(h_new)
            h = h_new
        h0 = np.copy(h)  # Use current as starting point for next alpha
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 1.0
_g_start = 0.0
_g_budget = 0.0
_g_n = 80

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    e = np.exp(a - a_max)
    e_sum = e.sum()
    w = e / e_sum
    obj = (a_max + np.log(e_sum)) / alpha * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 - s:hi2 - s]
    return obj, grad * dx

def _fun(x):
    if time.time() - _g_start > _g_budget - 15:
        return 1e10
    o, _ = smooth_max_and_grad(x, _g_alpha)
    return o

def _jac(x):
    if time.time() - _g_start > _g_budget - 15:
        return np.zeros(_g_n)
    _, g = smooth_max_and_grad(x, _g_alpha)
    return g

def _con_fun(x):
    return x.sum() - _g_n / 2.0

def _con_jac(x):
    return np.ones(_g_n)

def slsqp_solve(h0, alpha, maxiter):
    global _g_alpha, _g_n
    _g_alpha = alpha
    _g_n = len(h0)
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bounds = [(0.0, 1.0)] * len(h0)
    result = minimize(_fun, h0, jac=_jac, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    return result.x

def projected_grad_loop(h, alpha, n_steps, target_sum, start, budget):
    n = len(h)
    best_local = np.copy(h)
    best_c5_local = compute_c5(h)
    lr = 0.05
    for step in range(n_steps):
        if time.time() - start > budget - 10:
            break
        if step > 0 and step % 2000 == 0:
            lr = max(0.002, lr * 0.5)
        obj, grad = smooth_max_and_grad(h, alpha)
        h = h - lr * grad
        h = project_to_constraint(h, target_sum)
        if step % 1000 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5_local:
                best_c5_local = c5
                best_local = np.copy(h)
    return best_local, best_c5_local

def random_perturb(h, scale, target_sum, n):
    noise = np.random.randn(n) * scale
    h_new = h + noise
    return project_to_constraint(h_new, target_sum)

def run(seed=42, budget_s=1300, **kwargs):

    np.random.seed(seed)
    start_time = time.time()

    alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0, 600.0, 1200.0, 2500.0]

    for i in range(len(alphas)):

            best_h = np.copy(h_pg)

        h_pert = random_perturb(best_h, scale, target_sum, n)
        x_opt = slsqp_solve(h_pert, 300.0, 300)
        x_opt = project_to_constraint(x_opt, target_sum)
        c5 = compute_c5(x_opt)

        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = np.copy(h_fin)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
