def project_to_polytope(v, target_sum):
    """Project v onto {h : 0 <= h <= 1, sum(h) = target_sum}"""
    n = len(v)
    # Water filling
    v_sorted = np.sort(v)[::-1]
    # Find the threshold
    csum = np.cumsum(v_sorted)
    rho = np.arange(n, 0, -1)
    # Find where (csum - target) / rho changes sign
    # We want the largest rho such that (csum[rho-1] - target) / rho >= v_sorted[rho-1]
    # Actually, the projection is: h_i = clip(v_i - lambda, 0, 1) for some lambda
    # such that sum(h) = target
    
    # Binary search for lambda
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    return h

def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()
    
    def get_initial():
        try:
            return np.array(initial_h_values, dtype=float)
        except:
            return None

        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)

        n = len(h)
        center = n - 1
        total_lags = 2 * n - 1
        grad = np.zeros((n, total_lags))

            # Lag 0 (d = center): ∂/∂h[i] of sum h[k](1-h[k]) = 1-2h[i]
            grad[i, center] = 1.0 - 2.0 * h[i]

                # Positive lag (d = center + lag):
                # corr = sum_{k=0}^{n-1-lag} h[k+lag] * (1-h[k])
                # ∂/∂h[i]:
                #   h[i] appears in h[k+lag] when k = i-lag, valid if 0 <= i-lag <= n-1-lag, i.e., lag <= i
                #   h[i] appears in (1-h[k]) when k = i, valid if 0 <= i <= n-1-lag

`np.correlate(a, v, mode='full')` 
The result `c_k = sum_n a_{n+k} \cdot conj(v_n)`

Where k goes from `-(len(v)-1)` to `len(a)-1`.

The output has length `len(a) + len(v) - 1 = 2N - 1`.
Index 0 corresponds to k = -(N-1), index N-1 corresponds to k = 0, index 2N-2 corresponds to k = N-1.

So:
corr_full[N-1 + k] = sum_n h[n+k] * (1-h[n]) for k = -(N-1), ..., N-1

Where the sum is over valid n (0 ≤ n < N and 0 ≤ n+k < N).

So for k > 0 (positive lag): n ranges from 0 to N-1-k
  corr_full[N-1+k] = sum_{n=0}^{N-1-k} h[n+k] * (1-h[n])

For k < 0 (negative lag, let d = -k > 0): n ranges from d to N-1
  corr_full[N-1-d] = sum_{n=d}^{N-1} h[n-d] * (1-h[n])
  Let m = n-d: = sum_{m=0}^{N-1-d} h[m] * (1-h[m+d])

For k = 0:
  corr_full[N-1] = sum_{n=0}^{N-1} h[n] * (1-h[n])

Now the gradient:

For corr_full[N-1] (lag 0): 
  ∂/∂h[i] = 1 - 2h[i]

For corr_full[N-1+k] with k > 0:
