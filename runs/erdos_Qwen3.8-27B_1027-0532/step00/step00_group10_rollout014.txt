        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

At each iteration with current point h_0:
1. Compute c_k(h_0) for all k
2. Compute ∇c_k(h_0) for all k

5. Repeat

This is the standard "linearization + solve" approach for min-max problems. It should converge linearly (or at least much faster than subgradient).

The LP has:
- n+1 continuous variables (h[0],...,h[n-1], t)
- 2n-1 inequality constraints (one per lag)
- n+1 box constraints (0 ≤ h[i] ≤ 1 for each i)
- 1 equality constraint (sum h = target_sum)

import cvxpy as cp

def solve_lp_subproblem(h0, c0, grads, target_sum):
    n = len(h0)
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    for k in range(len(c0)):
        # c_k(h) ≈ c0[k] + grads[k] @ (h - h0)
        # = (c0[k] - grads[k] @ h0) + grads[k] @ h
        linear_part = c0[k] - np.dot(grads[k], h0)
        constraints.append(np.dot(grads[k], h) + linear_part <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)
    
    if h.value is not None:
        return np.clip(h.value, 0, 1), t.value
    return h0.copy(), c0.max()
