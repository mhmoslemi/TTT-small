def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def max_overlap(h, dx):
    return np.max(compute_overlaps(h, dx))

def project_box_sum(h, target_sum):
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    lo, hi = -1.0, 1.0
    try:
        lam = brentq(f, lo, hi, xtol=1e-15, rtol=1e-15)
    except:
        return np.clip(h, 0.0, 1.0)
    return np.clip(h - lam, 0.0, 1.0)

def overlap_grad(h, dx, weights):
    """Gradient of sum_k weights[k] * O_k w.r.t. h."""
    N = len(h)
    grad = np.zeros(N)

        if w < 1e-14:

        j_lo = max(0, t)
        j_hi = min(N, N + t)  # exclusive
        if j_lo < j_hi:
