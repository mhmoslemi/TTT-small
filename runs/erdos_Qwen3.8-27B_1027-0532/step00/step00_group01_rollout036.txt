        if s > target_sum:

    if abs(diff) > 1e-14:

    h1b, c5_1b = systematic_local_search(best_h.copy(), t0, tl, seed + 1)
    if c5_1b < best_c5:
        best_h, best_c5 = h1b, c5_1b

    # Phase 1c: Random local search
    tl = time.time() + remaining() * 0.10
    h1c, c5_1c = paired_local_search(best_h.copy(), t0, tl, seed + 2)
    if c5_1c < best_c5:
        best_h, best_c5 = h1c, c5_1c

    # Phase 2: Try higher resolution
    for n_new in [127, 255]:
        if remaining() < 60:
            break
        x_old = np.linspace(0, 2, len(best_h) + 1)[:-1]
        x_new = np.linspace(0, 2, n_new + 1)[:-1]
        h_new = np.interp(x_new, x_old, best_h)
        t_new = n_new / 2.0
        h_new = project_simplex(h_new, t_new)

        # Try quantized versions
        best_candidate = h_new
        best_candidate_c5 = c5_of(h_new)
        for thresh in [0.35, 0.45, 0.5, 0.55, 0.65]:
            hq = h_new.copy()
            hq[hq > thresh] = 1.0
            hq[hq < 1.0 - thresh] = 0.0
            hq = project_simplex(hq, t_new)
            cq = c5_of(hq)
            if cq < best_candidate_c5:
                best_candidate = hq
                best_candidate_c5 = cq
        h_new = best_candidate

        tl = time.time() + min(remaining() * 0.12, 150)
        h2, c5_2 = pgd_optimize(h_new, t_new, tl, seed + n_new)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

        tl = time.time() + min(remaining() * 0.08, 100)
        h2b, c5_2b = paired_local_search(best_h.copy(), t_new, tl, seed + 1000 + n_new)
        if c5_2b < best_c5:
            best_h, best_c5 = h2b, c5_2b

    # Phase 3: Random restarts from perturbed best
    if remaining() > 50:
        rng_r = np.random.RandomState(seed + 7777)
        n_b = len(best_h)
        t_b = n_b / 2.0
        for restart in range(5):
            if remaining() < 30:
                break
            h_r = best_h.copy()
            for _ in range(8):
                a, b = rng_r.randint(n_b), rng_r.randint(n_b)
                if a != b:
                    d = rng_r.uniform(-0.3, 0.3)
                    h_r[a] += d
                    h_r[b] -= d
            h_r = project_simplex(h_r, t_b)
            tl = time.time() + min(remaining() * 0.08, 80)
            h_r, c5_r = pgd_optimize(h_r, t_b, tl, seed + 2000 + restart)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 4: Final polish
    if remaining() > 20:
        n_f = len(best_h)
        t_f = n_f / 2.0
        tl = time.time() + remaining() * 0.8
        h_f, c5_f = paired_local_search(best_h.copy(), t_f, tl, seed + 9999)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    final_c5 = c5_of(best_h)
    return best_h, final_c5, len(best_h)
