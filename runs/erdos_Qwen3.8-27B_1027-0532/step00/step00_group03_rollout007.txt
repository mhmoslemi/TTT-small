    return np.correlate(h, 1 - h, mode='full') * dx

    """Adjust h to have the correct sum while staying in [0,1]."""
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    return np.clip(h, 0, 1)

    n = len(h)

    h_work = h.copy()

    h_bin = np.where(h_work >= 0.5, 1.0, 0.0)

    n_ones = int(np.sum(h_bin))
    target_ones = int(target_sum)  # For even n, this is exact

        excess = n_ones - int(target_sum)
        ones_idx = np.where(h_bin == 1)[0]
        to_remove = rng.choice(ones_idx, min(excess, len(ones_idx)), replace=False)
        h_bin[to_remove] = 0
    elif n_ones < int(target_sum):

        zeros_idx = np.where(h_bin == 0)[0]

        h_bin[to_add] = 1

Let me implement the fast swap evaluation:

When we swap h[i]=1 → 0 and h[j]=0 → 1, how does each overlap change?

For overlap at shift s (where s = m - (n-1) in the correlation index m):
overlap_s = sum_{valid i} h[i] * (1-h[i+s])

After the swap:
- If i is in the valid range for the "h" part and i+s is in the valid range:
  - The term h[i]*(1-h[i+s]) changes from 1*(1-h[i+s]) = (1-h[i+s]) to 0*(1-h[i+s]) = 0, change = -(1-h[i+s]) = -1 + h[i+s]
  - If h[i+s] = 0, change = -1; if h[i+s] = 1, change = 0

- If j is in the valid range for the "h" part and j+s is in the valid range:
  - The term h[j]*(1-h[j+s]) changes from 0*(1-h[j+s]) = 0 to 1*(1-h[j+s]) = (1-h[j+s]), change = 1-h[j+s]
  - If h[j+s] = 0, change = +1; if h[j+s] = 1, change = 0

- If i is in the valid range for the "(1-h)" part, i.e., i = i'+s for some valid i':
  - The term h[i']*(1-h[i]) changes from h[i']*1 = h[i'] to h[i']*0 = 0, change = -h[i']
  - h[i'] = h[i'-s], so change = -h[i-s]

- Similarly for j in the "(1-h)" part.

This is getting complex but doable. For a binary sequence, the changes are simple:
- Swapping h[i] from 1 to 0: 
  - For each shift s where i is in the "h" position: overlap decreases by (1-h[i+s]) = 1 if h[i+s]=0, 0 if h[i+s]=1
  - For each shift s where i is in the "(1-h)" position (i.e., i-s is in the h position): overlap decreases by h[i-s] = 1 if h[i-s]=1, 0 if h[i-s]=0

- Swapping h[j] from 0 to 1:
  - For each shift s where j is in the "h" position: overlap increases by (1-h[j+s])
  - For each shift s where j is in the "(1-h)" position: overlap increases by h[j-s]

So the net change in overlap at shift s is:
delta_s = -(1-h[i+s]) + (1-h[j+s]) + (-h[i-s]) + h[j-s]
(where terms are only included if the indices are valid)

For binary h: -(1-h[i+s]) = h[i+s]-1, so:
delta_s = (h[i+s]-1) + (1-h[j+s]) - h[i-s] + h[j-s]
= h[i+s] - h[j+s] - h[i-s] + h[j-s]

(All terms valid only when indices are in range)

This means we can compute the change in all overlaps in O(n) time (iterating over all shifts s), and check if the max decreases in O(n) time. Total: O(n) per swap evaluation, which for n=45 is very fast.

Actually, we can do even better. We only need to check if the NEW maximum is less than the CURRENT maximum. If we know the current maximum overlap value M and the shift s* that achieves it, we only need to check:
1. Does the overlap at s* decrease? (If yes, we need to find the new max)
2. Does any other overlap increase above M?

For (2), we need to check all shifts where the overlap increases. But with the formula above, we can compute all deltas in O(n) and check if any exceeds M - delta_{s*}.

This is still O(n) per evaluation, which is fine for n=45 (just 45 operations per swap).

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def make_binary(h, n, rng):
    """Convert h to a binary (or nearly binary) vector with sum = n/2."""
    target = n / 2.0
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    
    n_ones = int(np.sum(h_bin))
    if target == int(target):
        # Even n: need exactly n/2 ones
        while n_ones > int(target):
            idx = rng.choice(np.where(h_bin == 1)[0])
            h_bin[idx] = 0
            n_ones -= 1
        while n_ones < int(target):
            idx = rng.choice(np.where(h_bin == 0)[0])
            h_bin[idx] = 1
            n_ones += 1
    else:
        # Odd n: need n/2 - 0.5 ones and one 0.5
        # We'll handle this by having (n-1)/2 ones, one 0.5, (n-1)/2 zeros
        target_ones = int(target)  # floor(n/2)
        # First make it binary with target_ones ones
        while n_ones > target_ones:
            idx = rng.choice(np.where(h_bin == 1)[0])
            h_bin[idx] = 0
            n_ones -= 1
        while n_ones < target_ones:
            idx = rng.choice(np.where(h_bin == 0)[0])
            h_bin[idx] = 1
            n_ones += 1
        # Set one zero to 0.5
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) > 0:
            idx = rng.choice(zeros)
            h_bin[idx] = 0.5
    
    return h_bin

def projected_gradient(h, dx, time_budget, rng):
    """Projected gradient descent on the continuous problem."""
    n = len(h)
    target_sum = n / 2.0
    h_work = h.copy().clip(0, 1)
    h_work = fix_sum(h_work, target_sum)
    
    best_c5 = compute_c5(h_work, dx)
    best_h = h_work.copy()
    
    t_start = time.time()
    base_lr = 0.01
    momentum = 0.95
    velocity = np.zeros(n)
    it = 0
    
    while time.time() - t_start < time_budget:
        it += 1
        lr = base_lr / (1.0 + 0.005 * it)
        
        # Compute overlaps
        overlaps = np.correlate(h_work, 1.0 - h_work, mode='full') * dx
        
        # Use log-sum-exp for a smooth gradient
        alpha = 50.0
        c_max = np.max(overlaps)
        exp_vals = np.exp(alpha * (overlaps - c_max))
        weights = exp_vals / np.sum(exp_vals)
        
        # Compute gradient as weighted sum of gradients of each overlap
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            w = weights[m]
            if w < 1e-10:
                continue
            # corr[m] = sum_i h[i] * (1-h[i+n-1-m]) for valid i
            i_lo = max(0, m - n + 1)
            i_hi = min(n - 1, m)
            if i_lo > i_hi:
                continue
            for i in range(i_lo, i_hi + 1):
                j = i + n - 1 - m
                grad[i] += w * (1.0 - h_work[j])
                grad[j] += w * (-h_work[i])
        
        # Gradient step
        velocity = momentum * velocity - lr * grad
        h_work = h_work + velocity
        h_work = np.clip(h_work, 0.0, 1.0)
        h_work = fix_sum(h_work, target_sum)
        
        c5_new = compute_c5(h_work, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_work.copy()
    
    return best_h, best_c5
