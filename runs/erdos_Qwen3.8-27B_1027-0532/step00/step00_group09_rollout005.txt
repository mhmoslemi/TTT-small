```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _obj_fn(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _obj_grad_fn(h):
    n = len(h)
    dx = 2.0 / n
    idx = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr) * dx)
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        valid_f = (idx - k) >= 0
        valid_b = (idx + k) < n
        grad[valid_f] += (1.0 - h[idx[valid_f] - k])
        grad[valid_b] -= h[idx[valid_b] + k]
        grad *= dx
    return (val, grad)

def _cons_fun(h):
    return np.sum(h) - len(h) / 2.0

def _cons_jac(h):
    return np.ones(len(h))

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    h = h + (n / 2.0 - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    return h
