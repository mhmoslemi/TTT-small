import numpy as np
import math
import time


def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_simplex_box(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        alpha = 0.5 * (lo + hi)
        projected = np.clip(h - alpha, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = alpha
        else:
            hi = alpha
    alpha = 0.5 * (lo + hi)
    return np.clip(h - alpha, 0.0, 1.0)


def compute_grad_smooth(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones_like(corr) / len(corr)
    else:
        weights /= w_sum
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            i_start = 0
            i_end = n - d
        else:
            i_start = -d
            i_end = n
        if i_start >= i_end:
            continue
        i_idx = np.arange(i_start, i_end)
        j_idx = i_idx + d
        grad[j_idx] += w * one_minus_h[i_idx]
        grad[i_idx] -= w * h[j_idx]
    return grad * dx


def pgd_optimize(h0, n_iters, tau_init, tau_final, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_simplex_box(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    current_val = best_val
    start_time = time.time()
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max(1, n_iters - 1)
        tau = tau_init * (tau_final / tau_init) ** frac
        lr = (3.0 / n) * (1.0 - 0.7 * frac)
        grad = compute_grad_smooth(h, tau)
        h_new = project_simplex_box(h - lr * grad, target_sum)
        val = evaluate_c5(h_new)
        if val <= current_val + 1e-15:
            h = h_new
            current_val = val
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            improved = False
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_simplex_box(h - lr * shrink * grad, target_sum)
                val_try = evaluate_c5(h_try)
                if val_try <= current_val + 1e-15:
                    h = h_try
                    current_val = val_try
                    if val_try < best_val:
                        best_val = val_try
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                lr *= 0.5
    return best_h, best_val


def multi_start_pgd(h0, n_restarts, n_iters, tau_init, tau_final, time_limit, rng):
    n = len(h0)
    target_sum = n / 2.0
    best_h = project_simplex_box(h0.astype(float).copy(), target_sum)
    best_val = evaluate_c5(best_h)
    start_time = time.time()
    for r in range(n_restarts):
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            break
        if r == 0:
            h_start = best_h.copy()
        else:
            noise_scale = 0.05 * (1.0 + 0.5 * r)

            h_start = project_simplex_box(h_noisy, target_sum)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
    return best_h, best_val
