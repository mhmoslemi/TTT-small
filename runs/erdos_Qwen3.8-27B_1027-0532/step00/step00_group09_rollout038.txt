import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, t):
    max_c = np.max(corr)
    shifted = corr - max_c
    log_sum = np.log(np.sum(np.exp(np.clip(shifted, -500, 500))))
    return max_c + (log_sum - np.log(len(corr))) / t

def smooth_max_grad_full(h, t):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / (sum_exp * t)
    
    lags = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    
    for j in range(n):
        valid1 = (j - lags >= 0) & (j - lags < n)
        valid2 = (j + lags >= 0) & (j + lags < n)
        grad[j] = np.dot(weights[valid1], 1 - h[j - lags[valid1]]) - np.dot(weights[valid2], h[j + lags[valid2]])
    
    return grad

import time

    return np.sum(h) - _sum_target

def _constraint_jac(h):
    return np.ones(len(h))

def _smooth_obj(h, t_val):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    shifted = np.clip(shifted, -500.0, 500.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return max_c + (log_sum - np.log(len(corr))) / t_val

def _smooth_grad(h, t_val):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / (sum_exp * t_val)
    
    lags = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    
    for j in range(n):

        valid1 = (idx1 >= 0) & (idx1 < n)

        valid2 = (idx2 >= 0) & (idx2 < n)
        term1 = np.dot(weights[valid1], 1.0 - h[idx1[valid1]])
        term2 = np.dot(weights[valid2], h[idx2[valid2]])
        grad[j] = term1 - term2
