        if 0 <= j + k < n:
            val -= h[j + k]

                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                delta = (rng.rand() - 0.5) * 0.3
                h_t = best_local.copy()
                h_t[i] += delta
                h_t[j] -= delta
                h_t = np.clip(h_t, 0.0, 1.0)
                diff = target_sum - np.sum(h_t)
                if abs(diff) > 1e-10:
                    mask = (h_t > 0.01) & (h_t < 0.99)
                    nm = np.sum(mask)
                    if nm == 0:
                        continue
                    h_t[mask] += diff / nm
                    h_t = np.clip(h_t, 0.0, 1.0)
                c5 = compute_c5(h_t)
                if c5 < best_c5 - 1e-12:
                    best_local = h_t.copy()
                    best_h = h_t.copy()
                    best_c5 = c5
                    improved = True
            if not improved and it > 50:
                break

    # Final polish with SLSQP
    if time_left() > 10:
        try:
            res = try_optimize(best_h, compute_c5, c5_grad, maxiter=300)
            if res.x is not None:
                h_c = make_valid(res.x, target_sum)
                c5 = compute_c5(h_c)
                if c5 < best_c5:
                    best_h = h_c.copy()
                    best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5, n
