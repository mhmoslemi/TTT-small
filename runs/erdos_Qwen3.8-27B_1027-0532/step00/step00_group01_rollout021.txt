    corr = np.correlate(h, 1.0 - h, mode="full")

    k = max_idx - (n - 1)  # The lag
    
    grad = np.zeros(n)

                grad[i] += (1.0 - h[i + m])

                grad[i] += -h[i - m]

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n):
            if i >= k and (i - k) < (n - k):
                grad[i] += (1.0 - h[i - k])
            if i < (n - k) and (i + k) < n:
                grad[i] += -h[i + k]
    else:
        m = -k
        for i in range(n):
            if i < (n - m):
                grad[i] += (1.0 - h[i + m])
            if i >= m and (i - m) < (n - m):
                grad[i] += -h[i - m]
    return grad * dx

def penalized_objective(h, n, W):
    c5 = compute_c5(h)
    penalty = W * (np.sum(h) - n / 2.0) ** 2
    return c5 + penalty

def penalized_gradient(h, n, W):
    c5_grad = c5_gradient(h)
    pen_grad = 2.0 * W * (np.sum(h) - n / 2.0)
    return c5_grad + pen_grad
