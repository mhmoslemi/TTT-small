```python
import numpy as np
import time
from scipy.optimize import minimize_scalar

def project_onto_constraint(x, target_sum):
    """Project x onto {sum(x)=target_sum, 0<=x[i]<=1}"""
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target_sum
    def f(lam):
        return np.sum(np.clip(x - lam, 0, 1)) - target_sum
    # f is monotonically decreasing in lam
    # f(-inf) = sum(x) - target_sum (could be positive or negative)
    # f(+inf) = -target_sum (negative)
    # We need f(lam) = 0
    lo, hi = -2, 2  # wide range
    # Find bracket
    while f(lo) < 0:
        lo -= 1
    while f(hi) > 0:
        hi += 1
    res = minimize_scalar(f, bounds=(lo, hi), method='brent', options={'xatol': 1e-12})
    return np.clip(x - res.x, 0, 1)
```

```python
    c = corr * tau
    c_max = np.max(c)
```

```python
import numpy as np
import time
from scipy.optimize import minimize_scalar, minimize

def compute_c5(x):
    """Compute C5 value for a step function x"""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    return np.max(corr) * dx

def projection(x, target_sum):
    """Project x onto {sum=x = target_sum, 0<=x[i]<=1}"""
    def f(lam):
        return np.sum(np.clip(x - lam, 0, 1)) - target_sum
    # Find lambda
    lo, hi = -1.0, 1.0
    f_lo, f_hi = f(lo), f(hi)
    while f_lo < 0:
        lo -= 1.0
        f_lo = f(lo)
    while f_hi > 0:
        hi += 1.0
        f_hi = f(hi)
    res = minimize_scalar(f, bounds=(lo, hi), method='brent', options={'xatol': 1e-14})
    return np.clip(x - res.x, 0, 1)
```

```python
import numpy as np
import time
from scipy.optimize import minimize_scalar, minimize

def compute_c5(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def projection(x, target_sum):
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def corr_weighted_gradient(x, weights):
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-13:
            continue
        l = (n - 1) - k
        if l >= 0:
            grad[:n - l] += w * (1.0 - x[l:])
            grad[l:] -= w * x[:n - l]
        else:
            ln = -l
            grad[ln:] += w * (1.0 - x[:n - ln])
            grad[:n - ln] -= w * x[ln:]
    return grad

def grad_smoothed_max(x, tau):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = tau * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    probs = exp_s / np.sum(exp_s)
    return corr_weighted_gradient(x, probs)

def val_smoothed_max(x, tau):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = tau * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / tau

def projected_gradient(x0, target_sum, tau, lr, n_iter, time_limit, momentum=0.9):
    x = x0.copy()
    v = np.zeros_like(x)
    best = x.copy()
    best_c5 = compute_c5(x)
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        grad = grad_smoothed_max(x, tau)
        v = momentum * v - lr * grad
        x_new = projection(x + v, target_sum)
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best = x_new.copy()
            best_c5 = c5
        x = x_new
        lr *= 0.9998
    return best, best_c5

def slsqp_polish(x0, target_sum, n, time_limit):
    tau = 30.0
    t0 = time.time()
    def obj(z):
        return val_smoothed_max(z, tau)
    def obj_grad(z):
        return grad_smoothed_max(z, tau)
    def con_fun(z):
        return np.sum(z) - target_sum
    def con_jac(z):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    best = x0.copy()
    best_c5 = compute_c5(x0)
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-14})
        if time.time() - t0 > time_limit:
            return best, best_c5
        c5 = compute_c5(res.x)
        if c5 < best_c5:
            best = res.x.copy()
            best_c5 = c5
    except Exception:
        pass
    try:
        tau2 = 60.0
        def obj2(z):
            return val_smoothed_max(z, tau2)
        def obj2_grad(z):
            return grad_smoothed_max(z, tau2)
        res2 = minimize(obj2, best, jac=obj2_grad, method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 200, 'ftol': 1e-14})
        c5 = compute_c5(res2.x)
        if c5 < best_c5:
            best = res2.x.copy()
            best_c5 = c5
    except Exception:
        pass
    return best, best_c5

def upsample(x, factor):
    n = len(x)
    new_n = n * factor
    idx = np.arange(new_n) * n // new_n
    return x[idx]

def downsample(x, factor):
    n = len(x)
    new_n = n // factor
    total = np.zeros(new_n)
    for i in range(new_n):
        total[i] = np.mean(x[i * factor:(i + 1) * factor])
    return total

    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        if c5 < best_c5:

        best_x, best_c5 = slsqp_polish(x0, target_sum, n, min(120.0, remaining() * 0.3))

    # Phase 3: More gradient sweeps with smaller lr for fine-tuning
    for attempt in range(6):
        if remaining() < 30:

        tau = 50.0 + 30.0 * attempt
        lr = 0.3 / tau
        x0 = projection(best_x, target_sum)
        tlim = min(20.0, remaining() * 0.05)
        best_x, c5 = projected_gradient(x0, target_sum, tau, lr, 8000, tlim, momentum=0.8)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 4: Random restarts
    n_restarts = 40
    for r in range(n_restarts):
        if remaining() < 15:
            break
        if r % 4 == 0:
            noise = np.random.randn(n) * 0.03
        elif r % 4 == 1:
            mask = np.random.random(n) < 0.15
            noise = np.zeros(n)
            noise[mask] = np.random.choice([-0.2, 0.2], size=int(np.sum(mask)))
        elif r % 4 == 2:
            smooth_noise = np.random.randn(n)
            kernel = np.array([0.25, 0.25, 0.25, 0.25])
            smooth_noise = np.convolve(smooth_noise, kernel, mode='same')
            noise = smooth_noise * 0.08

            best_c5 = c5_down
```
