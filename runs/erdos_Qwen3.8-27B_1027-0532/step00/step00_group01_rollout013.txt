import numpy as np
import time

def compute_c5(h, dx):
    """Compute C5 value for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_to_sum(h, target_sum):
    """Adjust h to have the correct sum while keeping values in [0,1]."""
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    n = len(h)
    h_adjusted = h + diff / n
    # Clip and iterate
    for _ in range(100):
        h_adjusted = np.clip(h_adjusted, 0, 1)
        new_sum = np.sum(h_adjusted)
        diff = target_sum - new_sum
        if abs(diff) < 1e-10:
            break
        # Distribute only among non-saturated values
        mask = (h_adjusted > 0) & (h_adjusted < 1)
        if not np.any(mask):
            break
        h_adjusted[mask] += diff / np.sum(mask)
    return h_adjusted

def run(seed=42, budget_s=1300, **kwargs):

c[k] = sum_i h[i+k] * (1-h[i])

The output has length 2n-1, with k ranging from -(n-1) to n-1.
- k = -(n-1) is index 0
- k = 0 is index n-1
- k = n-1 is index 2n-2

So if max_idx = argmax(corr), then k = max_idx - (n-1).

For k > 0: c[k] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
For k = 0: c[0] = sum_{i=0}^{n-1} h[i] * (1-h[i])
For k < 0: let k' = -k. c[-k'] = sum_{i=k'}^{n-1} h[i] * (1-h[i-k'])

OK so for k > 0, the contributing terms are where h[i+k]=1 and h[i]=0, for i in [0, n-1-k].
For k < 0 (k' = -k), the contributing terms are where h[i]=1 and h[i-k']=0, for i in [k', n-1].

Let me rewrite:

For k > 0:
- Pairs (i, i+k) where h[i+k]=1, h[i]=0, i in [0, n-1-k]
- To fix: set h[i]=1, h[i+k]=0 (swap)
- This removes contribution 1 at lag k

For k < 0 (k' = -k):
- Pairs (i-k', i) where h[i]=1, h[i-k']=0, i in [k', n-1]
- Equivalent: pairs (j, j+k') where h[j+k']=1, h[j]=0, j in [0, n-1-k']
- To fix: set h[j]=1, h[j+k']=0 (swap)
- This removes contribution 1 at lag k

So in both cases, the structure is the same: find pairs (j, j+|k|) where h[j+|k|]=1, h[j]=0, and swap them.

    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def thue_morse_binary(n):
    idx = np.arange(n, dtype=np.int64)
    t = np.zeros(n, dtype=np.int64)
    for b in range(n.bit_length()):
        t = t ^ ((idx >> b) & 1)
    return (t & 1).astype(np.float64)

def binary_local_search(h, n, rng, max_time):
    start = time.time()
    dx = 2.0 / n
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    stall = 0
    
    while time.time() - start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        abs_k = abs(k)
        
        if abs_k == 0 or abs_k >= n:
            break
        
        mask = (h[abs_k:].astype(np.float64) == 1.0) & (h[:n - abs_k].astype(np.float64) == 0.0)
        violations = np.where(mask)[0]
        
        improved = False
        if len(violations) > 0:
            best_delta = current_c5
            best_j = -1
            
            for idx in violations:
                j = int(idx)
                h[j] = 1.0
                h[j + abs_k] = 0.0
                new_c5 = compute_c5(h, dx)
                h[j] = 0.0
                h[j + abs_k] = 1.0
                if new_c5 < best_delta:
                    best_delta = new_c5
                    best_j = j
            
            if best_j >= 0 and best_delta < current_c5 - 1e-12:
                h[best_j] = 1.0
                h[best_j + abs_k] = 0.0
                current_c5 = best_delta
                improved = True
                if best_delta < best_c5:
                    best_c5 = best_delta
                    best_h = h.copy()
        
        if improved:
            stall = 0
        else:
            stall += 1
            n_flips = min(1 + stall // 10, 4)
            for _ in range(n_flips):
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(rng.integers(len(ones)))
                j = int(rng.integers(len(zeros)))
                h[ones[i]] = 0.0
                h[zeros[j]] = 1.0
            current_c5 = compute_c5(h, dx)
            if stall % 200 == 0:
                h = best_h.copy()
                current_c5 = best_c5
                stall = 0
    
    return best_h, best_c5
