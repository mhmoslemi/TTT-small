    grad = np.zeros(n)
    A = 1 - h
    for s in range(-(n-1), n):
        w = weights[s + (n-1)]  # weight for shift s
        if w < 1e-15:
            continue
        # part1: w * A[i+s] for valid i+s
        # i+s in [0, n-1] means i in [-s, n-1-s], intersected with [0, n-1]
        i_start = max(0, -s)
        i_end = min(n-1, n-1-s)
        if i_start <= i_end:
            grad[i_start:i_end+1] += w * A[i_start+s:i_end+s+1]
        # part2: -w * h[i-s] for valid i-s
        # i-s in [0, n-1] means i in [s, s+n-1], intersected with [0, n-1]
        i_start2 = max(0, s)
        i_end2 = min(n-1, s+n-1)
        if i_start2 <= i_end2:
            grad[i_start2:i_end2+1] -= w * h[i_start2-s:i_end2-s+1]

    """Compute gradient of sum_k weights[k] * c[k] where c[k] = correlation at shift k-(n-1)"""
    grad = np.zeros(n)
    A = 1.0 - h
    for s in range(-(n-1), n):
        w = weights[s + (n-1)]
        if w < 1e-12:
            continue
        # part1: w * A[i+s] for i such that 0 <= i+s < n
        # i in [max(0,-s), min(n-1, n-1-s)]
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        if i_lo <= i_hi:

        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= w * h[i_lo2-s:i_hi2-s+1]
    return grad

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def project_to_feasible(y, target_sum, n):
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_gradient(h, weights, n):
    grad = np.zeros(n)
    A = 1.0 - h
    for s in range(-(n-1), n):

        if w < 1e-12:

def compute_c5(h, dx):

    return np.correlate(h, 1.0 - h, mode='full')

def project_to_feasible(y, target_sum, n):
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, weights, n):
    grad = np.zeros(n)
    A = 1.0 - h
    for s in range(-(n-1), n):
        w = weights[s + n - 1]
        if w < 1e-14:
            continue
        i_lo = max(0, s)
        i_hi = min(n-1, s + n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += w * A[i_lo - s:i_hi - s + 1]
        i_lo2 = max(0, -s)
        i_hi2 = min(n-1, n - 1 - s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= w * h[i_lo2 + s:i_hi2 + s + 1]
    return grad

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_c_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_feasible(y, target_sum, n):
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, weights, n):
    grad = np.zeros(n)
    A = 1.0 - h
    for s in range(-(n - 1), n):
        w = weights[s + n - 1]
        if w < 1e-14:
            continue
        i_lo = max(0, s)
        i_hi = min(n - 1, s + n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += w * A[i_lo - s:i_hi - s + 1]
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= w * h[i_lo2 + s:i_hi2 + s + 1]
    return grad
