```python
    # p[k] * dx * [(1-h[j-(k-N+1)]) * 1_{0≤j-(k-N+1)≤N-1} - h[j+(k-N+1)] * 1_{0≤j+(k-N+1)≤N-1}]
    # where k-N+1 is the actual shift
    
    # Let's use a different formulation. For shift s = k-(N-1), s ranges from -(N-1) to N-1.
    # ∂c_s/∂h[j] = (1-h[j-s]) * 1_{0≤j-s≤N-1} - h[j+s] * 1_{0≤j+s≤N-1}
    
    for k in range(2*N-1):
        s = k - (N-1)
        w = p[k] * dx
        if w < 1e-15:
            continue
        # (1-h[j-s]) for j-s in [0,N-1], i.e., j in [s, N-1+s]
        j_start = max(0, s)
        j_end = min(N-1, N-1+s)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end+1)
            grad[idx] += w * (1.0 - h[idx - s])
        # -h[j+s] for j+s in [0,N-1], i.e., j in [-s, N-1-s]
        j_start2 = max(0, -s)
        j_end2 = min(N-1, N-1-s)
        if j_start2 <= j_end2:
            idx2 = np.arange(j_start2, j_end2+1)
            grad[idx2] -= w * h[idx2 + s]
    
    return f, grad
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def compute_c5(h, dx):
    corr = correlate.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
```


```python
def smooth_c5_value(h, dx, temp):
    corr = correlate.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-12:
        return m
    shifted = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)))


def smooth_c5_gradient(h, dx, temp):
    N = len(h)
    corr = correlate.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    grad = np.zeros(N)
    if temp < 1e-12:
        max_k = int(np.argmax(corr))
        s_k = max_k - (N - 1)
        lo1 = max(0, s_k)
        hi1 = min(N - 1, N - 1 + s_k)
        if lo1 <= hi1:
            j = np.arange(lo1, hi1 + 1)
            grad[j] += dx * (1.0 - h[j - s_k])
        lo2 = max(0, -s_k)
        hi2 = min(N - 1, N - 1 - s_k)
        if lo2 <= hi2:
            j = np.arange(lo2, hi2 + 1)
            grad[j] -= dx * h[j + s_k]
        return grad
    shifted = (corr - m) / temp
    exp_s = np.exp(shifted)
    s_sum = np.sum(exp_s)
    p = exp_s / s_sum
    for k in range(2 * N - 1):
        s_k = k - (N - 1)
        w = p[k] * dx
        if w < 1e-18:
            continue
        lo1 = max(0, s_k)
        hi1 = min(N - 1, N - 1 + s_k)
        if lo1 <= hi1:
            j = np.arange(lo1, hi1 + 1)
            grad[j] += w * (1.0 - h[j - s_k])
        lo2 = max(0, -s_k)
        hi2 = min(N - 1, N - 1 - s_k)
        if lo2 <= hi2:
            j = np.arange(lo2, hi2 + 1)
            grad[j] -= w * h[j + s_k]
    return grad
```

```python
    def __init__(self, dx, temp):
        self.dx = dx
        self.temp = temp
    def __call__(self, h):
        return smooth_c5_value(h, self.dx, self.temp)
    def grad(self, h):
        return smooth_c5_gradient(h, self.dx, self.temp)


class ConstraintFunc:
    def __init__(self, target_sum):
        self.target_sum = target_sum
    def __call__(self, h):
        return np.sum(h) - self.target_sum


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return result


def optimize_single(state, h_start, n, temps, maxiter, t0, deadline, budget_s):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_start.copy()

            if res.success or res.fun < obj(h):

                if c < state['c5']:
```
