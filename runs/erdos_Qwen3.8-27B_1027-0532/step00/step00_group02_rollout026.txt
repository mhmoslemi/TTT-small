        x0 = _enforce_constraints(x0, target_sum)
        attempts.append(x0)

    # Also try structured starts
    for i in range(4):
        if not elapsed_ok(60):
            break
        freq = 1.0 + i * 0.5
        phase = rng.uniform(0, 2 * np.pi)
        t = np.linspace(0, 2, n)
        x0 = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t / 2.0 + phase)
        x0 = _enforce_constraints(x0, target_sum)
        attempts.append(x0)

    for idx, x0 in enumerate(attempts):
        if not elapsed_ok(60):
            break
        maxiter = 500 if idx == 0 else 300
        try:
            res = minimize(
                _compute_c5, x0, jac=_compute_grad,
                method='SLSQP', bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            h_sol = _enforce_constraints(res.x, target_sum)
            c5_val = _compute_c5(h_sol)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h_sol.copy()
        except Exception:
            pass

    # Polish with tighter SLSQP
    if elapsed_ok(120):
        try:
            res = minimize(
                _compute_c5, best_h, jac=_compute_grad,
                method='SLSQP', bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-15}
            )
            h_sol = _enforce_constraints(res.x, target_sum)
            c5_val = _compute_c5(h_sol)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h_sol.copy()
        except Exception:
            pass

    # Second round of perturbation from improved solution
    for i in range(10):
        if not elapsed_ok(120):
            break
        scale = 0.05 * (1.0 - i / 15.0)
        x0 = best_h + rng.uniform(-scale, scale, size=n)
        x0 = _enforce_constraints(x0, target_sum)
        try:
            res = minimize(
                _compute_c5, x0, jac=_compute_grad,
                method='SLSQP', bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-13}
            )
            h_sol = _enforce_constraints(res.x, target_sum)
            c5_val = _compute_c5(h_sol)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h_sol.copy()
        except Exception:
            pass

    # Coordinate descent polish
    if elapsed_ok(60):
        best_h, best_c5 = _coordinate_descent(best_h, target_sum, max_rounds=100)

    # Final SLSQP polish
    if elapsed_ok(30):
        try:
            res = minimize(
                _compute_c5, best_h, jac=_compute_grad,
                method='SLSQP', bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-15}
            )
            h_sol = _enforce_constraints(res.x, target_sum)
            c5_val = _compute_c5(h_sol)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h_sol.copy()
        except Exception:
            pass

    best_h = _enforce_constraints(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
