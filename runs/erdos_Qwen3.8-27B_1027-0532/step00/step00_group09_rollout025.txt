            if new_c5 < c5_curr:
                h_curr = new_h
                c5_curr = new_c5
        except Exception:
            pass
    return h_curr, c5_curr


def _coord_descent(h_best, c5_best, deadline, max_rounds=500):
    n = len(h_best)
    target = n / 2.0
    h = h_best.copy()
    c5 = c5_best
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for step in steps:
        for _ in range(max_rounds):
            if time.time() > deadline - 10:
                return h, c5
            improved = False
            for i in range(n):
                if time.time() > deadline - 10:
                    break
                for sign in [1.0, -1.0]:
                    new_val = h[i] + sign * step
                    if new_val < 0.0 or new_val > 1.0:
                        continue
                    comp = -sign * step
                    tried = 0
                    for j in range(n):
                        if j == i:
                            continue
                        new_j = h[j] + comp
                        if new_j < 0.0 or new_j > 1.0:
                            continue
                        h[i] = new_val
                        h[j] = new_j
                        new_c5 = _obj_raw(h)
                        if new_c5 < c5 - 1e-13:
                            c5 = new_c5
                            improved = True
                        else:
                            h[i] = h[i] - sign * step
                            h[j] = h[j] - comp
                        tried += 1
                        if tried >= 3:
                            break
    return h, c5


def _interpolate(h_src, n_new):
    n_src = len(h_src)
    dx_old = 2.0 / n_src
    dx_new = 2.0 / n_new
    x_old = np.arange(n_src) * dx_old + dx_old / 2.0
    x_new = np.arange(n_new) * dx_new + dx_new / 2.0
    h_new = np.interp(x_new, x_old, h_src, period=2.0)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0

    best_h = _project(h0, target)
    best_c5 = _obj_raw(best_h)

    # Phase 1: SLSQP from initial point with increasing sharpness
    best_h, best_c5 = _slsqp_optimize(best_h, n, [5, 10, 20, 50, 100, 200, 500, 1000], 3000, deadline)

    # Phase 2: Try higher resolutions via interpolation
    for n_try in [100, 150, 200, 300]:
        if time.time() > deadline - 120:
            break
        h_interp = _interpolate(best_h, n_try)
        h_interp = _project(h_interp, n_try / 2.0)
        h_new, c5_new = _slsqp_optimize(h_interp, n_try, [10, 25, 50, 100, 200], 1500, deadline)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Random perturbation restarts
    n_curr = len(best_h)
    for _ in range(30):
        if time.time() > deadline - 80:
            break
        h_pert = best_h + rng.normal(0, 0.03, n_curr)
        h_pert = _project(h_pert, n_curr / 2.0)
        h_new, c5_new = _slsqp_optimize(h_pert, n_curr, [50, 100, 200, 500], 800, deadline)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Coordinate descent refinement
    if time.time() < deadline - 20:
        best_h, best_c5 = _coord_descent(best_h, best_c5, deadline, max_rounds=300)

    # Final: one more SLSQP pass with high t for polish
    if time.time() < deadline - 15:
        n_f = len(best_h)
        best_h, best_c5 = _slsqp_optimize(best_h, n_f, [500, 1000, 2000], 2000, deadline)

    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _obj_raw(best_h)

    return (best_h, final_c5, n_final)
