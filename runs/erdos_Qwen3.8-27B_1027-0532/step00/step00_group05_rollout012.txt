            best_h = h_opt
        n = len(best_h)
        target = n / 2.0

    if time.time() - t0 < budget_s * 0.18:
        for c5_b, h_b in try_binary(best_h, target, 4):
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
                n = len(best_h)
                target = n / 2.0

    if time.time() - t0 < budget_s * 0.55:
        for restart in range(5000):
            if time.time() - t0 > budget_s * 0.55:
                break
            strategy = restart % 6
            if strategy == 0:
                h_try = best_h + rng.randn(n) * 0.03
            elif strategy == 1:
                h_try = best_h + rng.randn(n) * 0.1
            elif strategy == 2:
                h_try = best_h.copy()
                bs = rng.randint(2, max(3, n // 3))
                st = rng.randint(0, n - bs + 1)
                h_try[st:st + bs] += rng.randn(bs) * 0.25
            elif strategy == 3:
                h_try = generate_initial(n, 1, rng)
            elif strategy == 4:
                h_try = generate_initial(n, 3, rng)
            else:
                h_try = generate_initial(n, 4, rng)
            h_try = project_sum(h_try, target)
            max_a = 4 if rng.random() < 0.5 else 6
            h_opt = local_optimize(h_try, target, max_alphas=max_a)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    if time.time() - t0 < budget_s * 0.92:
        n_list = [28, 36, 56, 70, 84, 112, 140, 168, 196, 252, 336, 420]
        for n_try in n_list:
            if time.time() - t0 > budget_s * 0.92:
                break
            target_try = n_try / 2.0
            for rs in range(8):
                if time.time() - t0 > budget_s * 0.92:
                    break
                h_try = generate_initial(n_try, rs % 5, rng)
                h_try = project_sum(h_try, target_try)
                h_opt = local_optimize(h_try, target_try, max_alphas=5)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                    n = n_try
                    target = target_try

    if time.time() - t0 < budget_s * 0.98:
        best_h = coord_descent(best_h, target, n_sweeps=max(5, n // 5))
        c5_cd = compute_c5(best_h)
        if c5_cd < best_c5:
            best_c5 = c5_cd
        h_opt = local_optimize(best_h, target, max_alphas=5)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    for c5_b, h_b in try_binary(best_h, target, 3):
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    return best_h, best_c5, len(best_h)
