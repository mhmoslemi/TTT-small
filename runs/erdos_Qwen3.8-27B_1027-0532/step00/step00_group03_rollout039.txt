import numpy as np

def project_onto_feasible(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}."""
    n = len(h)
    # Bisection on threshold t
    lo, hi = -max(h) - 1, min(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h_clipped = np.clip(h - mid, 0, 1)
        s = np.sum(h_clipped)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune to get exact sum
    # Due to floating point, we might be slightly off
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-bound variables
        free = (h_proj > 0) & (h_proj < 1)
        if np.any(free):
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def compute_gradient(h):
    """Compute subgradient of max_k c[k]*dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    c_scale = c * dx
    max_val = np.max(c_scale)
    
    # Use softmax weighting for smoothness
    epsilon = 0.001
    weights = np.exp((c_scale - max_val) / epsilon)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        m = k - (n - 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_val = np.max(c)
    eps = max(1e-4, max_val * 0.1)
    w = np.exp((c - max_val) / eps)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += w[k] * (1.0 - h[:n-m])
            grad[:n-m] -= w[k] * h[m:]
        else:
            mp = -m
            grad[:n-mp] += w[k] * (1.0 - h[mp:])
            grad[mp:] -= w[k] * h[:n-mp]
    return grad * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hc = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hc)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h_proj = np.clip(h - t, 0.0, 1.0)
    return h_proj

def binary_local_search(h_init, n_evals_max, rng):
    """Local search on binary sequences."""
    n = len(h_init)
    target = n // 2
    # Ensure binary
    h = (h_init > 0.5).astype(float)
    # Fix to have exactly target ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        # Remove excess ones
        excess = len(ones) - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1
    
    best_c5 = compute_c5(h)
    
    for _ in range(n_evals_max):
        # Pick a random 1 and a random 0, swap them
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            h[i] = 1
            h[j] = 0
    
    return h, best_c5

def gradient_descent(h_init, n_iters, lr, time_limit=None, start_time=None):
    """Projected gradient descent."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    if start_time is None:
        start_time = time.time()
    
    for it in range(n_iters):
        if time_limit and (time.time() - start_time) > time_limit:
            break
        
        grad = compute_c5_grad(h)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_new = h - lr * 0.5 * grad
            h_new = project_feasible(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h = h_new
            # If still worse, reduce learning rate
            if c5 >= best_c5:
                lr *= 0.5
    
    return best_h, best_c5

def upsample(h, factor=2):
    """Upsample by repeating each element 'factor' times."""
    n = len(h)
    h_up = np.repeat(h, factor)
    return h_up

def interpolate_upsample(h, factor=2):
    """Upsample by linear interpolation."""
    n = len(h)
    n_new = n * factor
    x_old = np.arange(n)
    x_new = np.arange(n_new) / factor
    h_up = np.interp(x_new, x_old, h)
    return h_up

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(c)
    eps = max(1e-4, max_val * 0.08)
    w = np.exp((c - max_val) / eps)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        m = k - (n - 1)
        if m >= 0:
            if m < n:
                grad[m:] += wk * (1.0 - h[:n - m])
                grad[:n - m] -= wk * h[m:]
        else:
            mp = -m
            if mp < n:
                grad[:n - mp] += wk * (1.0 - h[mp:])
                grad[mp:] -= wk * h[:n - mp]
    return grad * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hc = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hc)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def make_binary(h, n_ones, rng):
    n = len(h)
    h = h.copy()
    sorted_idx = np.argsort(h)[::-1]
    h[:] = 0.0
    h[sorted_idx[:n_ones]] = 1.0
    return h

def binary_local_search(h_init, max_time, rng, start_time=None):
    n = len(h_init)
    target_ones = n // 2
    h = make_binary(h_init, target_ones, rng)
    best_c5 = compute_c5(h)
    if start_time is None:
        start_time = time.time()
    no_improve = 0
    while (time.time() - start_time) < max_time:
        batch_size = 500
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(batch_size):
            if (time.time() - start_time) > max_time:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                no_improve = 0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        if no_improve > 10000:
            break
    return h.copy(), best_c5

def gradient_descent(h_init, max_time, lr_init, start_time=None):
    n = len(h_init)
    target_sum = float(n) // 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    if start_time is None:
        start_time = time.time()
    lr = lr_init
    for _ in range(500000):
        if (time.time() - start_time) > max_time:
            break
        grad = compute_c5_grad(h)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            h_try = h - lr * 0.5 * grad
            h_try = project_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break
    return best_h, best_c5

def interpolate_upsample(h, factor=2):
    n = len(h)
    n_new = n * factor
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, n_new)
    h_up = np.interp(x_new, x_old, h)
    return h_up
