    """Compute gradient of c[k] with respect to h."""
    # c[k] = sum_i h[i+k] * (1-h[i]) for valid i
    # ∂c[k]/∂h[j] = (1-h[j-k]) - h[j+k] (when valid)

    # h[i+k] contributes to c[k] via (1-h[i])
    # h[i] contributes to c[k] via -h[i+k]

import numpy as np

def repair_vector(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = 0.5 * n
    diff = target - v.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = mask.sum()
        if cnt > 0:
            v[mask] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
        diff2 = target - v.sum()
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                m2 = v < 1.0
            else:
                m2 = v > 0.0
            c2 = m2.sum()
            if c2 > 0:
                v[m2] += diff2 / c2
                v = np.clip(v, 0.0, 1.0)
    return v

def eval_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx

def coord_descent_pass(h, n, dx, rng):
    c5_cur = eval_c5(h, dx)
    improved = False
    perm = rng.permutation(n)
    for j in perm:
        val = h[j]
        best_cv = val
        best_cc5 = c5_cur
        cands = [0.0, 1.0]
        if val > 0.01:
            cands.append(val - 0.05)
            cands.append(val - 0.02)
            cands.append(val - 0.1)
        if val < 0.99:
            cands.append(val + 0.05)
            cands.append(val + 0.02)
            cands.append(val + 0.1)
        for cv in cands:
            cv = max(0.0, min(1.0, cv))

                continue

            ht[j] = cv
            ht = repair_vector(ht, n)
            cc5 = eval_c5(ht, dx)
            if cc5 < best_cc5:
                best_cc5 = cc5
                best_cv = cv
        if best_cv != val:
            h[j] = best_cv
            h = repair_vector(h, n)
            c5_cur = eval_c5(h, dx)
            improved = True
    return h, c5_cur, improved
