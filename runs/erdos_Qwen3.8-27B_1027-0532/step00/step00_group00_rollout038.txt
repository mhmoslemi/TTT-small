```python
import numpy as np
import time
import random

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - 10 - (time.time() - start_time)
    
    def eval_c5(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def make_binary_balanced(n, rng):
        h = np.zeros(n)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
        return h
    
    def make_thue_morse(n):
        h = np.zeros(n)
        for i in range(n):
            # Count bits in i
            c = bin(i).count('1')
            h[i] = 1.0 if c % 2 == 1 else 0.0
        # Adjust to have exactly n/2 ones
        n_ones = int(np.sum(h))
        if n_ones > n//2:
            # Remove excess ones
            excess = n_ones - n//2
            one_positions = np.where(h == 1)[0]
            remove = np.random.choice(one_positions, size=excess, replace=False)
            h[remove] = 0
        elif n_ones < n//2:
            deficit = n//2 - n_ones
            zero_positions = np.where(h == 0)[0]
            add = np.random.choice(zero_positions, size=deficit, replace=False)
            h[add] = 1
        return h
    
    def local_search_binary(h, max_iters, rng, time_limit_fn):
        """Local search on binary sequence. Move: swap a 1 and a 0."""
        n = len(h)
        current = h.copy()
        current_c5 = eval_c5(current)
        
        for it in range(max_iters):
            if time_limit_fn() <= 0:
                break
            
            # Pick a random 1 position and a random 0 position
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            a = rng.choice(ones)
            b = rng.choice(zeros)
            
            # Try the swap
            current[a] = 0
            current[b] = 1
            new_c5 = eval_c5(current)
            
            if new_c5 < current_c5:
                current_c5 = new_c5
            else:
                # Revert
                current[a] = 1
                current[b] = 0
        
        return current, current_c5
    
    # Phase 1: Evaluate initial
    h0 = np.array(initial_h_values, dtype=float)
    best_h = h0.copy()
    best_c5 = eval_c5(best_h)
    
    # Phase 2: Try various n values with different initializations
    n_values = [44, 46, 48, 50, 52, 56, 60, 64, 72, 80, 96, 100, 128]
    
    for n in n_values:
        if time_left() <= 0:
            break
        
        # Try Thue-Morse
        h_tm = make_thue_morse(n)
        h_rs, c5_rs = local_search_binary(h_tm, 50000, np.random, time_left)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs
        
        # Try random
        for _ in range(3):
            h_rand = make_binary_balanced(n, np.random)
            h_rs, c5_rs = local_search_binary(h_rand, 50000, np.random, time_left)
            if c5_rs < best_c5:
```

```python
def _sa_continuous(h, max_iters):
    """SA for continuous h values."""
    n = len(h)
    cur = np.array(h, dtype=float)
    cur_val = _eval_c5(cur)
    best = cur.copy()
    best_val = cur_val
    
    T = 0.005
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        if _time_left() <= 0:
            break
        
        a = np.random.randint(n)
        b = np.random.randint(n)
        if a == b:
            continue
        delta = np.random.uniform(-0.03, 0.03)
        
        na = cur[a] + delta
        nb = cur[b] - delta
        if na < 0 or na > 1 or nb < 0 or nb > 1:
            continue
        
        cur[a] = na
        cur[b] = nb
        new_val = _eval_c5(cur)
        diff = new_val - cur_val
        
        if diff < 0 or np.random.random() < np.exp(-diff / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[a] -= delta
            cur[b] += delta
        
        T = max(T_min, T * cool)
    
    return best, best_val
```

This looks reasonable. But I'm worried about a few things:
1. The hill climbing with random swaps might be too slow to find good solutions
2. The SA parameters might not be well-tuned
3. I'm not doing "best-improvement" passes which are more effective

Let me add a "best-improvement" function that tries all swaps:

```python
def _best_improvement_pass(h):
    """Try all possible swaps, return best one."""
    n = len(h)
    cur = np.array(h, dtype=float)
    cur_max = _max_corr(cur)
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    best_delta = 0.0
    best_a, best_b = -1, -1
    
    for a in ones:
        for b in zeros:
            cur[a] = 0.0
            cur[b] = 1.0
            new_max = _max_corr(cur)
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_a, best_b = a, b
            cur[a] = 1.0
            cur[b] = 0.0
    
    if best_a >= 0:
        cur[best_a] = 0.0
        cur[best_b] = 1.0
        cur_max += best_delta
    
    return cur, cur_max, best_a >= 0
```
