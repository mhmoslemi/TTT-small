import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                room = 1.0 - h[mask]
                tr = np.sum(room)
                if tr > 0:
                    h[mask] += (diff / tr) * room
        else:
            mask = h > 0.0
            if np.any(mask):
                ta = np.sum(h[mask])
                if ta > 0:
                    h[mask] += (diff / ta) * h[mask]
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def obj_penalty(x, n, penalty):
    h = np.clip(x, 0.0, 1.0)
    c5 = compute_c5(h)
    violation = abs(np.sum(h) - n * 0.5)
    return c5 + penalty * violation

def soft_max_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mval = np.max(corr)
    exp_args = beta * (corr - mval)
    exp_args = np.clip(exp_args, -500, 0)
    log_sum = np.log(np.sum(np.exp(exp_args)))
    violation = abs(np.sum(h) - n * 0.5)
    return mval + (1.0 / beta) * log_sum + 10.0 * violation

def local_search(h, n, rng, num_steps):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    scale = 0.05
    for step in range(num_steps):
        h_try = best_h + scale * rng.randn(n)
        h_try = repair_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            scale = min(scale * 1.1, 0.15)
        else:
            scale = max(scale * 0.9, 0.01)
    return best_h, best_c5

def neld_mead_step(h, n, rng, maxiter):
    penalty = 50.0
    res = minimize(
        obj_penalty, h, args=(n, penalty),
        method='Nelder-Mead',
        options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True}
    )
    h_opt = repair_h(res.x, n)
    return h_opt
