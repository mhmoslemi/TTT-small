import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / alpha
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return smooth_val, grad

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _slsqp_obj(val):
    return val[-1]

def _slsqp_ineq(val, n, m_idx):
    h = val[:n]
    t = val[n]
    m = m_idx - (n - 1)
    dx = 2.0 / n
    c_val = 0.0
    for j in range(n):
        if 0 <= j + m < n:
            c_val += h[j + m] * (1.0 - h[j])
    return t - c_val * dx

def _slsqp_eq(val, n, _):
    return np.sum(val[:n]) - n / 2.0

def _run_slsqp(h_init, n, maxiter=300):
    target = n / 2.0
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = []
    for k in range(2 * n - 1):
        cons.append({'type': 'ineq', 'fun': _slsqp_ineq, 'args': (n, k)})
    cons.append({'type': 'eq', 'fun': _slsqp_eq, 'args': (n, 0)})
    res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_feasible(h_opt, target)
    return h_opt

def _gradient_descent(h, n, alpha_start, max_iter, deadline):
    target = n / 2.0
    alpha = alpha_start
    lr = 0.05
    best_local = compute_c5(h)
    best_local_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c5_val = compute_c5(h)
        if c5_val < best_local:
            best_local = c5_val
            best_local_h = h.copy()
        _, grad = compute_smooth_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_new = h - lr * grad / max(gnorm, 1e-10) * 0.5
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_val + 1e-12:
            h = h_new
            if it % 200 == 0 and it > 0:
                alpha = min(alpha * 1.3, 5000.0)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_local_h, best_local

def _make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 8)
    for b in range(n_blocks):
        start = int(n * b / n_blocks)
        length = max(1, n // (2 * n_blocks) + rng.randint(-1, 2))
        end = min(n, start + length)
        h[start:end] = 1.0

def _make_smooth_init(n, rng):
    t_arr = np.linspace(0, 2, n)
    n_freqs = rng.randint(2, 5)
    h = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.uniform(1, 6)
        p = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.sin(2 * np.pi * f * t_arr / 2.0 + p)
    h = 0.5 + h / (np.max(h.abs() if hasattr(h, 'abs') else np.abs(h)) + 1e-10)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)

def _make_binary_init(n, rng):

    except Exception:
        pass

    n_values = [42, 44, 40, 46, 48, 38, 50, 52, 56, 60, 64]
    if h0 is not None and len(h0) not in n_values:
        n_values.insert(0, len(h0))

    deadline = start_time + budget_s

    for n in n_values:
        if _time_left() < 50:
            break

        inits = []
        if h0 is not None and n == 42:
            inits.append(project_feasible(h0.copy(), n / 2.0))
            inits.append(project_feasible(h0[::-1].copy(), n / 2.0))
        for _ in range(2):
            inits.append(_make_block_init(n, rng))
        inits.append(_make_smooth_init(n, rng))
        inits.append(_make_binary_init(n, rng))

        for h_init in inits:
            if _time_left() < 15:
                break

            cur_h = h_init
            cur_c5 = compute_c5(cur_h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

            # SLSQP attempt
            if _time_left() > 30:
                try:
                    h_slsqp = _run_slsqp(cur_h, n, maxiter=200)
                    c5_s = compute_c5(h_slsqp)
                    if c5_s < best_c5:
                        best_c5 = c5_s
                        best_h = h_slsqp.copy()
                    if c5_s < cur_c5:
                        cur_h = h_slsqp
                        cur_c5 = c5_s
                except Exception:
                    pass

            # Gradient descent refinement
            if _time_left() > 10:
                h_gd, c5_gd = _gradient_descent(cur_h, n, 20.0, 3000, deadline)
                if c5_gd < best_c5:
                    best_c5 = c5_gd
                    best_h = h_gd.copy()

                # Second pass with higher alpha
                if _time_left() > 10:
                    h_gd2, c5_gd2 = _gradient_descent(h_gd, n, 100.0, 2000, deadline)
                    if c5_gd2 < best_c5:
                        best_c5 = c5_gd2
                        best_h = h_gd2.copy()

            # Perturbation escape
            if _time_left() > 20:
                h_pert = cur_h + rng.normal(0, 0.05, n)
                h_pert = project_feasible(h_pert, n / 2.0)
                h_gd3, c5_gd3 = _gradient_descent(h_pert, n, 20.0, 1500, deadline)
                if c5_gd3 < best_c5:
                    best_c5 = c5_gd3
                    best_h = h_gd3.copy()

    if best_h is None:
        n = 42
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
