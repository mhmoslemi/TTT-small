import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    s = tau * (scaled - m)
    es = np.exp(s)
    se = np.sum(es)
    obj = m + np.log(se) / tau
    w = es / se
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        mk = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        s1 = max(0, -mk)
        e1 = min(n, n - mk)
        if s1 < e1:
            grad[s1:e1] += wk * omh[s1 + mk:e1 + mk]
        s2 = max(0, mk)
        e2 = min(n, n + mk)
        if s2 < e2:
            grad[s2:e2] -= wk * h[s2 - mk:e2 - mk]
    return obj, grad

    lr = lr_init
    best_obj = _smooth_obj_grad(h, tau)[0]
    best_h = h.copy()
    prev_obj = best_obj
    
    for it in range(max_iter):
        obj, grad = _smooth_obj_grad(h, tau)
        h_new = h - lr * grad
        h_new = _project(h_new, n * 0.5)
        new_obj, _ = _smooth_obj_grad(h_new, tau)
        
        if new_obj < best_obj:

            best_h = h_new.copy()

        elif it % 200 == 0 and lr < lr_init:
            lr *= 1.05
        
        h = h_new
        prev_obj = new_obj
        if it > 100 and abs(prev_obj - best_obj) < 1e-10:
            break

    if old_n == new_n:
        return h.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    return np.interp(x_new, x_old, h)

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    s = tau * (scaled - m)
    es = np.exp(s)
    se = np.sum(es)
    obj = m + np.log(se) / tau
    w = es / se
    threshold = 1e-8
    significant = np.where(w > threshold)[0]
    grad = np.zeros(n)
    for k in significant:
        mk = k - (n - 1)
        wk = w[k]
        s1 = max(0, -mk)
        e1 = min(n, n - mk)
        if s1 < e1:
            grad[s1:e1] += wk * omh[s1 + mk:e1 + mk]
        s2 = max(0, mk)
        e2 = min(n, n + mk)
        if s2 < e2:
            grad[s2:e2] -= wk * h[s2 - mk:e2 - mk]
    return obj, grad

def _pgd(h, n, tau, max_iter, lr_init, momentum=0.9):
    h = _project(h.copy(), n * 0.5)
    lr = lr_init
    v = np.zeros(n)
    best_obj = _smooth_obj_grad(h, tau)[0]
    best_h = h.copy()
    
    for it in range(max_iter):
        obj, grad = _smooth_obj_grad(h, tau)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = _project(h_new, n * 0.5)
        new_obj, _ = _smooth_obj_grad(h_new, tau)
        
        if new_obj < best_obj:
            best_obj = new_obj
            best_h = h_new.copy()
        
        if new_obj > obj + 1e-12:
            lr *= 0.5
            v = v * 0.5
        elif it % 500 == 0 and lr < lr_init * 0.5:
            lr *= 1.2
        
        h = h_new
        if it > 200 and abs(new_obj - best_obj) < 1e-12:
            break
    
    return best_h, best_obj

def _upsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    return np.interp(x_new, x_old, h)

def _round_binary(h, n):
    threshold = np.sort(h)[n // 2 - 1] if n // 2 > 0 else 0.5
    h_bin = np.where(h > threshold, 1.0, 0.0)
    ones = np.sum(h_bin)
    if ones > n * 0.5:
        idx = np.where(h_bin == 1.0)[0]
        h_vals = h[idx]
        order = np.argsort(h_vals)[:ones - n * 0.5]
        h_bin[idx[order]] = 0.0
    elif ones < n * 0.5:
        idx = np.where(h_bin == 0.0)[0]
        h_vals = h[idx]
        order = np.argsort(-h_vals)[:n * 0.5 - ones]
        h_bin[idx[order]] = 1.0
    return h_bin

def _scipy_smooth_opt(h, n, tau, max_iter):
    def obj_func(x):
        o, _ = _smooth_obj_grad(x, tau)
        return o
    def grad_func(x):
        _, g = _smooth_obj_grad(x, tau)
        return g
    pen = 50.0
    def obj_pen(x):
        o, _ = _smooth_obj_grad(x, tau)
        s = np.sum(x) - n * 0.5
        return o + pen * s * s

        s = np.sum(x) - n * 0.5

                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        return _project(res.x, n * 0.5)
    except Exception:
        return h
