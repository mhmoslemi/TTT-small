    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k = active_lag  # in range -(n-1) to n-1
    for j in range(n):
        # Term 1: (1-h[j-k]) if 0 <= j-k < n
        if 0 <= j - k and j - k < n:
            grad[j] += (1 - h[j - k])
        # Term 2: -h[j+k] if 0 <= j+k < n
        if 0 <= j + k and j + k < n:
            grad[j] -= h[j + k]

def smooth_c5_and_grad(h, rho):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    # Shift for numerical stability
    corr_shifted = corr - np.max(corr)
    weights = np.exp(rho * corr_shifted)
    weights /= np.sum(weights)
    
    # Log-sum-exp value
    c5_smooth = np.max(corr) + np.log(np.sum(weights)) / rho
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - (n-1)
        # ∂overlap_k/∂h[j] for each j
        for j in range(n):
            if 0 <= j - k < n:
                grad[j] += weights[k_idx] * (1 - h[j-k]) * dx
            if 0 <= j + k < n:
                grad[j] += weights[k_idx] * (-h[j+k]) * dx
    
    return c5_smooth, grad

    for k_idx in np.where(significant)[0]:
        k = k_idx - (n - 1)
        grad += weights[k_idx] * gradient_for_lag(h, k)
    
    return c5_smooth, grad

def project_onto_constraints(h, target_sum):

    lo = np.min(h) - 2.0
    hi = np.max(h) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

import numpy as np
import time

def project_onto_constraints(h, target_sum):
    n = len(h)
    lo = np.min(h) - 2.0
    hi = np.max(h) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_for_lag(h, k, dx):
    n = len(h)
    j = np.arange(n)
    idx1 = j - k
    idx2 = j + k
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if np.any(mask1):
        term1[mask1] = 1.0 - h[idx1[mask1]]
    if np.any(mask2):
        term2[mask2] = h[idx2[mask2]]
    return dx * (term1 - term2)

def smooth_max_c5_and_grad(h, rho):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(corr)
    exp_shifted = np.exp(rho * (corr - max_val))
    weights = exp_shifted / np.sum(exp_shifted)
    c5_smooth = max_val + np.log(np.sum(exp_shifted)) / rho
    grad = np.zeros(n)
    threshold = np.max(weights) * 1e-8
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < threshold:
            continue
        k = k_idx - (n - 1)
        grad += w * gradient_for_lag(h, k, dx)
    return c5_smooth, grad

        v = momentum * v - lr * grad
        h = h + v
        h = project_onto_constraints(h, target_sum)
    return h
