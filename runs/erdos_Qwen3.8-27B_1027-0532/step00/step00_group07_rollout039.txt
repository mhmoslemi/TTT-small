def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # Log-sum-exp for numerical stability
    max_c = np.max(c)
    log_sum = max_c + np.log(np.sum(np.exp(beta * (c - max_c))))
    return (dx / beta) * log_sum

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # Softmax weights
    max_c = np.max(c)
    exp_vals = np.exp(beta * (c - max_c))
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient: for each j, sum over k
    grad = np.zeros(n)
    for j in range(n):
        # For each k (0 to 2n-2), s = k - (n-1)

        s_vals = np.arange(2*n-1) - (n-1)  # s = -(n-1), ..., n-1

import numpy as np
from scipy.optimize import minimize
import time

g_n = 0
g_target = 0.0
g_dx = 0.0
g_beta = 1.0
g_pen = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_grad(h, beta):
    n = g_n
    dx = g_dx
    c = np.correlate(h, 1 - h, mode='full')
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    obj = dx * (max_c + np.log(sum_exp) / beta)
    grad = np.zeros(n)
    s_vals = np.arange(2 * n - 1) - (n - 1)
    for j in range(n):
        js = j - s_vals
        jp = j + s_vals
        mask1 = (js >= 0) & (js < n)
        mask2 = (jp >= 0) & (jp < n)
        if np.any(mask1):
            grad[j] += np.dot(w[mask1], 1.0 - h[js[mask1]])
        if np.any(mask2):
            grad[j] -= np.dot(w[mask2], h[jp[mask2]])
    grad *= dx
    return obj, grad

def obj_func(x):
    x = np.clip(x, 0, 1)
    o, _ = smooth_obj_grad(x, g_beta)
    pen = g_pen * (np.sum(x) - g_target) ** 2
    return o + pen

def grad_func(x):
    x = np.clip(x, 0, 1)
    _, g = smooth_obj_grad(x, g_beta)
    g = g + 2.0 * g_pen * (np.sum(x) - g_target) / g_n
    return g

def project_feasible(h, target, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            h = np.clip(h + diff / n, 0, 1)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def lbgfsb_run(h0, start_time, budget_s):
    n = g_n
    target = g_target
    h_curr = project_feasible(h0.astype(float).copy(), target, n)
    betas = [2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 250.0, 500.0, 1000.0, 2000.0]
    for beta in betas:
        if budget_s - (time.time() - start_time) < 8.0:
            break
        g_beta = beta
        g_pen = 500.0 * beta
        bounds = [(0.0, 1.0)] * n
        maxiter = max(50, int(2000.0 / beta))
        try:
            res = minimize(obj_func, h_curr, method='L-BFGS-B', jac=grad_func,
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0})
            h_curr = res.x
        except Exception:
            break
        h_curr = project_feasible(h_curr, target, n)
    return h_curr

def slsqp_run(h0, start_time, budget_s):
    if budget_s - (time.time() - start_time) < 5.0:
        return h0
    g_beta = 500.0
    n = g_n
    target = g_target
    dx = g_dx

    def _slsqp_obj(x):
        x = np.clip(x, 0, 1)
        o, _ = smooth_obj_grad(x, g_beta)
        return o

    def _slsqp_grad(x):
        x = np.clip(x, 0, 1)
        _, g = smooth_obj_grad(x, g_beta)
        return g

    def _con_fun(x):
        return np.sum(x) - target

    def _con_jac(x):
        return np.ones(n) / n

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_curr = project_feasible(h0.astype(float).copy(), target, n)
    try:
        res = minimize(_slsqp_obj, h_curr, method='SLSQP', jac=_slsqp_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        h_curr = project_feasible(res.x, target, n)
    except Exception:
        pass
    return h_curr

    global g_n, g_target, g_dx, g_beta, g_pen

    g_n = len(initial_h_values)

    best_c5 = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    def time_ok(min_remaining):
        return (time.time() - start_time) < (budget_s - min_remaining)

    # Phase 1: Refine initial
    if time_ok(30):

        if c5 < best_c5:

            best_c5 = c5

    # Phase 3: Random restarts from best
    n_restarts = 80
    for i in range(n_restarts):

        frac = float(i) / float(n_restarts)
        scale = 0.25 * (1.0 - frac * 0.7) + 0.03
        perturb = rng.normal(0, scale, n)
        h_pert = best_h + perturb
        h_pert = project_feasible(h_pert, target, n)

        c5 = compute_c5(h_new)

            best_h = h_new
            best_c5 = c5

    # Phase 4: Try different initial structures
    if time_ok(30):

        h_uni = np.full(n, 0.5)
        h_uni += rng.normal(0, 0.1, n)
        h_uni = project_feasible(h_uni, target, n)

        c5 = compute_c5(h_new)

            best_c5 = c5

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
