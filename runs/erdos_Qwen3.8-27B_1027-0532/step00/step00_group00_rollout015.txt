def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_lags(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_to_sum(h, N):
    """Project h to satisfy sum(h) = N/2 while maintaining 0 <= h <= 1."""
    target = N / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum: add to elements that are < 1
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to decrease sum: subtract from elements that are > 0
            mask = h > 1e-10
            if not np.any(mask):
                break
            h[mask] -= (-diff) / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def simulated_annealing(h, N, rng, start_time, budget_s, time_frac=0.9):

    best_c5 = current_c5
    best_h = h.copy()
    
    T_start = 0.05
    T_end = T_start * 0.01
    total_iters = 500000

    best_c5 = current_c5

    # Phase 1: Greedy improvement
    n_greedy = 200000
    step_size = 0.1
    
    for it in range(n_greedy):
        if time.time() - start_time > budget_s * time_frac:
            break
        
        i = rng.integers(0, N)
        j = rng.integers(0, N)

            continue

        else:
