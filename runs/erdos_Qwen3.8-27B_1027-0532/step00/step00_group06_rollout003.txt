def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr


def project_onto_constraint(y, target_sum):
    N = len(y)
    
    def sum_for_lambda(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    
    lo = np.min(y) - 2.0
    hi = np.max(y) + 2.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_for_lambda(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    
    # Final micro-correction
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    
    return h


def optimize_h(h, target_sum, elapsed_so_far, time_budget, seed):
    """Optimize h using projected subgradient + smooth max + random restarts"""
    N = len(h)
    dx = 2.0 / N
    rng = np.random.RandomState(seed)
    
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    deadline = time.time() + time_budget
    
    # Phase 1: Smooth max gradient descent
    tau = 50.0  # Temperature for smooth max (higher = smoother)
    lr = 0.005
    patience = 5000
    no_improve = 0
    
    for it in range(patience):
        if time.time() > deadline * 0.6:
            break
        
        c5_val, corr = compute_c5(h)
        
        # Smooth max gradient: grad of (1/tau) log(sum exp(tau * corr[s]))
        # = exp(tau * corr) / sum(exp(tau * corr)) weighted average of individual gradients
        exp_tau = np.exp(tau * (corr - np.max(corr)))  # Subtract max for numerical stability
        weights = exp_tau / np.sum(exp_tau)
        
        # Compute gradient of each corr[s] w.r.t. h
        # For shift s with delta = s - (N-1):
        # corr[s] = sum_{i=lo}^{hi-1} h[i]*(1-h[i-delta])
        # d(corr[s])/d(h[j]) = (1-h[j-delta])*1_{j in [lo,hi)} - h[j+delta]*1_{j+delta in [lo,hi)}
        
        grad = np.zeros(N)
        for s in range(len(corr)):
            if weights[s] < 1e-10:
                continue
            delta = s - (N - 1)
            lo = max(0, delta)
            hi = min(N, N + delta)
            
            if hi > lo:
                idx1 = np.arange(lo, hi)
                grad[idx1] += weights[s] * (1.0 - h[idx1 - delta])
            
            j_lo = max(0, lo - delta)
            j_hi = min(N, hi - delta)
            if j_hi > j_lo:
                idx2 = np.arange(j_lo, j_hi)
                grad[idx2] += weights[s] * (-h[idx2 + delta])
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project
        h = project_onto_constraint(h_new, target_sum)
        
        # Check improvement
        c5_new, _ = compute_c5(h)
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt learning rate
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
        
        # Occasionally add noise to escape local optima
        if it % 500 == 100 and time.time() < deadline * 0.5:
            noise = rng.normal(0, 0.01, N)
            h = project_onto_constraint(h + noise, target_sum)
    
    # Phase 2: Coordinate descent / local polish
    if time.time() < deadline * 0.8:
        h = best_h.copy()
        h, c5_cd = coordinate_descent(h, target_sum, deadline)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()
    
    return best_h, best_c5


def coordinate_descent(h, target_sum, deadline):
    """Coordinate descent: optimize one coordinate at a time while maintaining sum constraint"""
    N = len(h)
    dx = 2.0 / N
    
    for sweep in range(200):
        if time.time() > deadline:
            break
        
        improved = False
        for j in range(N):
            if time.time() > deadline:
                break
            
            # Try changing h[j] and compensating by changing another coordinate
            # To maintain the sum constraint, if we increase h[j] by dv, we decrease another by dv
            # Or we can just change h[j] and re-project (less efficient but simpler)
            
            # Simple approach: try a few values of h[j], re-project, check improvement
            current_c5, _ = compute_c5(h)
            
            for dv in [0.01, -0.01, 0.05, -0.05, 0.02, -0.02]:
                h_trial = h.copy()
                h_trial[j] += dv
                h_trial = project_onto_constraint(h_trial, target_sum)
                c5_trial, _ = compute_c5(h_trial)
                if c5_trial < current_c5 - 1e-10:
                    h = h_trial
                    current_c5 = c5_trial
                    improved = True
                    break
        
        if not improved:
            break
    
    c5_final, _ = compute_c5(h)
    return h, c5_final

def project_onto_constraint(y, target_sum):

    for _ in range(80):
        mid = (lo + hi) / 2.0

        if cnt > 0:

def compute_smooth_max_grad(h, corr, tau):
    N = len(h)
    shifted = corr - np.max(corr)
    exp_vals = np.exp(tau * shifted)
    weights = exp_vals / np.sum(exp_vals)

        w = weights[s]

        delta = s - (N - 1)

        if j_hi > j_lo:

def pair_swap_optimize(h, target_sum, deadline, rng):

    attempts = 0

def upsample(h, N_new):
    N_old = len(h)
    if N_new == N_old:
        return h.copy()
    if N_new % N_old == 0:
        return np.repeat(h, N_new // N_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False) + 1.0 / N_old
    x_new = np.linspace(0, 2, N_new, endpoint=False) + 1.0 / N_new
    return np.interp(x_new, x_old, h)
