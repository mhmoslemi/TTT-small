import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(f) * dx)

def compute_f_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_value(f, tau):
    m = np.max(f)
    return m + np.log(np.sum(np.exp(tau * (f - m)))) / tau

def smooth_max_grad_f(f, tau):
    """Returns weights w_k such that sum w_k * f_k = smooth_max and grad = sum w_k * grad(f_k)"""
    m = np.max(f)
    w = np.exp(tau * (f - m))
    w /= np.sum(w)
    return w

def project_onto_constraint(h, c):
    """Project h onto {0 <= x <= 1, sum(x) = c}"""
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def gradient_objective(h, n, tau):
    """Compute gradient of smooth_max(f) * dx where f = correlate(h, 1-h, 'full')"""
    dx = 2.0 / n
    f = compute_f_values(h, n)
    w = smooth_max_grad_f(f, tau)
    
    grad = np.zeros(n)
    # For each k (shift), compute contribution to gradient
    # f_k = sum_i h[i+k]*(1-h[i]) for k>=0 (i from 0 to n-1-k)
    # f_k = sum_i h[i]*(1-h[i+k']) for k=-k'<0 (i from 0 to n-1-k')
    
    for idx in range(len(w)):
        k = idx - (n - 1)  # k ranges from -(n-1) to (n-1)
        wk = w[idx]
        if wk < 1e-15:
            continue
        if k >= 0:
            # f_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j] = (1-h[j-k]) for j in [k, n-1] - h[j+k] for j in [0, n-1-k]
            valid_j_pos = np.arange(k, n)
            grad[valid_j_pos] += wk * (1.0 - h[valid_j_pos - k])
            valid_j_neg = np.arange(0, n - k)
            grad[valid_j_neg] -= wk * h[valid_j_neg + k]

            valid_j_pos = np.arange(0, n - k_abs)

            valid_j_neg = np.arange(k_abs, n)
            grad[valid_j_neg] -= wk * h[valid_j_neg - k_abs]

def projected_gradient_descent(h0, n, max_iter=10000, tau_start=50, tau_end=500, 
                                lr_start=0.01, lr_end=0.0001, time_limit=100):
    h = h0.copy()
    c_target = n / 2.0
    start_time = time.time()
    
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Adaptive parameters
        progress = it / max_iter
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute gradient
        grad = gradient_objective(h, n, tau)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project onto constraint set
        h_new = project_onto_constraint(h_new, c_target)
        
        # Check improvement
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(f) * dx)

def smooth_max_value(f, tau):
    m = np.max(f)
    shifted = tau * (f - m)
    # Avoid overflow
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / tau

def gradient_c5_smooth(h, n, tau):
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(f)
    shifted = np.clip(tau * (f - m), -500, 500)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        k = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-14:
            continue
        if k >= 0:
            if k < n:
                gp = np.arange(k, n)
                grad[gp] += wk * (1.0 - h[gp - k])
                gn = np.arange(0, n - k)
                grad[gn] -= wk * h[gn + k]
        else:
            ka = -k
            if ka < n:
                gp = np.arange(0, n - ka)
                grad[gp] += wk * (1.0 - h[gp + ka])
                gn = np.arange(ka, n)
                grad[gn] -= wk * h[gn - ka]
    
    return grad * dx

def project_onto_constraint(h, c):
    """Project onto {0 <= x <= 1, sum(x) = c}"""
    # Bisection on the threshold
    lo, hi = -1.0, 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        h = h_new

            patience = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(f) * dx)

def smooth_max_value(f, tau):
    m = np.max(f)
    shifted = np.clip(tau * (f - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / tau

def gradient_c5_smooth(h, n, tau):
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(f)
    shifted = np.clip(tau * (f - m), -500.0, 500.0)
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(len(w)):
        k = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-14:
            continue
        if k >= 0:
            if k < n:
                gp = np.arange(k, n)
                grad[gp] += wk * (1.0 - h[gp - k])
                gn = np.arange(0, n - k)
                grad[gn] -= wk * h[gn + k]
        else:
            ka = -k
            if ka < n:
                gp = np.arange(0, n - ka)
                grad[gp] += wk * (1.0 - h[gp + ka])
                gn = np.arange(ka, n)
                grad[gn] -= wk * h[gn - ka]
    return grad * dx

def project_onto_constraint(h, c):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def projected_gradient_optimize(h0, n, time_limit=60.0, tau_init=20.0):
    h = h0.copy()
    c_target = n / 2.0
    start_time = time.time()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    tau = tau_init
    lr = 0.03
    patience = 0
    max_patience = 80
    vel = np.zeros(n)
    momentum = 0.9
    while time.time() - start_time < time_limit:
        grad = gradient_c5_smooth(h, n, tau)
        vel = momentum * vel + (1.0 - momentum) * grad
        for _ in range(40):
            h_new = h - lr * vel
            h_new = project_onto_constraint(h_new, c_target)
            c5_new = compute_c5(h_new, n)
            if c5_new <= best_c5 + 1e-12:
                break
            lr *= 0.7
        h = h_new
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0
            tau = min(tau * 1.03, 800.0)
            if lr < 0.05:
                lr = min(lr * 1.1, 0.05)
        else:
            patience += 1
            if patience > max_patience:
                tau = max(5.0, tau * 0.4)
                lr = 0.04
                vel = np.zeros(n)
                patience = 0
    return best_h, best_c5

def lbfgs_optimize(h0, n, time_limit=120.0):
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n)
    for tau in [30.0, 80.0, 200.0]:
        if time.time() - start_time > time_limit:
            break
        mu = 50.0
        def obj(x):
            f = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max_value(f, tau) * (2.0 / n) + mu * (np.sum(x) - n / 2.0) ** 2
        def obj_grad(x):
            g = gradient_c5_smooth(x, n, tau)
            g += 2.0 * mu * (np.sum(x) - n / 2.0) * np.ones(n)
            return g
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-8})
        if res.x is not None:
            h_opt = project_onto_constraint(res.x, n / 2.0)
            c5_opt = compute_c5(h_opt, n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return np.clip(h_new, 0.0, 1.0)

def local_binary_search(h, n, time_limit=10.0):
    start_time = time.time()
    c_target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    h_bin = project_onto_constraint(h_bin, c_target)
    c5_cur = compute_c5(h_bin, n)
    c5_best = c5_cur
    best_h = h_bin.copy()
    while time.time() - start_time < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            old_val = h_bin[i]
            new_val = 1.0 - old_val
            h_bin[i] = new_val
            h_bin = project_onto_constraint(h_bin, c_target)
            c5_try = compute_c5(h_bin, n)
            if c5_try < c5_cur - 1e-12:
                c5_cur = c5_try
                improved = True
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    best_h = h_bin.copy()
            else:
                h_bin[i] = old_val
        if not improved:
            break
    return best_h, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:

    else:

    best_h = h.copy()

    print(f"Initial: n={best_n}, C5={best_c5:.6f}")

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.35:

        h, c5 = projected_gradient_optimize(h, n, time_limit=t_alloc, tau_init=15.0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"PGD pass1: C5={best_c5:.6f}")

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.5:
        t_alloc = min(budget_s * 0.2, 120.0)
        h2, c5_2 = lbfgs_optimize(best_h, n, time_limit=t_alloc)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            print(f"L-BFGS: C5={best_c5:.6f}")

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.6:
        t_alloc = min(budget_s * 0.1, 80.0)
        h3, c5_3 = projected_gradient_optimize(best_h, n, time_limit=t_alloc, tau_init=100.0)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            print(f"PGD fine: C5={best_c5:.6f}")

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.75:
        t_alloc = min(budget_s * 0.1, 60.0)

            h_up = project_onto_constraint(h_up, n_new / 2.0)

            h_new, c5_new = projected_gradient_optimize(h_up, n_new, time_limit=t_alloc, tau_init=30.0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_new
