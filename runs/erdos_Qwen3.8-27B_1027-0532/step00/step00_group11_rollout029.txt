    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:

    # Distribute adjustment
    n = len(h)
    if diff > 0:
        # Need to add: distribute to positions with most room
        order = np.argsort(1.0 - h)[::-1]
    else:
        # Need to remove: distribute from positions with most value
        order = np.argsort(h)[::-1]
    for idx in order:
        if diff > 0:
            room = 1.0 - h[idx]
            add = min(room, diff)
            h[idx] += add
            diff -= add
        else:
            room = h[idx]
            rem = min(room, -diff)
            h[idx] -= rem
            diff += rem
        if abs(diff) < 1e-12:
            break
    h = np.clip(h, 0.0, 1.0)
    return h
