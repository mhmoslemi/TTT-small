def compute_gradient(h, n, dx, tau):
    """Compute gradient of smooth_max of correlation values"""
    # Compute correlation
    c = correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    m = np.max(c)
    exp_vals = np.exp(tau * (c - m))
    w = exp_vals / np.sum(exp_vals)
    
    # w has length 2n-1, corresponding to lags -(n-1), ..., 0, ..., (n-1)
    # w[k] corresponds to lag k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)  # actual lag
        if w[k] == 0:
            continue
        # Contribution to grad from this lag
        # ∂c[lag]/∂h[j] = (1-h[j-lag]) * 1_{valid} - h[j+lag] * 1_{valid}
        for j in range(n):
            term1 = 0.0
            term2 = 0.0
            if 0 <= j - lag <= n - 1:
                term1 = 1.0 - h[j - lag]
            if 0 <= j + lag <= n - 1:
                term2 = h[j + lag]
            grad[j] += w[k] * (term1 - term2)
    
    return grad
