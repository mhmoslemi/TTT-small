    """Repair a point to be feasible: 0<=x[i]<=1, sum(x)=n/2."""
    x = np.clip(x, 0, 1)

        return x

    # Better: water-filling
    diff = target - s

    x = x + per_elem
    x = np.clip(x, 0, 1)

    for _ in range(10):

        diff = target - s

        if np.any(free):
            x[free] += diff / np.sum(free)
            x = np.clip(x, 0, 1)
        else:
            break

    """Compute smooth max using log-sum-exp."""
    max_val = np.max(values)
    shifted = alpha * (values - max_val)
    log_sum = np.log(np.sum(np.exp(shifted)))
