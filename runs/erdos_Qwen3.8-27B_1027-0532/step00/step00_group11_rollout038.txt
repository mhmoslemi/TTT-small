import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def grad_at_shift(h, m):
    """
    Compute gradient of output[m] of np.correlate(h, 1-h, 'full') w.r.t. h.
    output[m] = sum_n h[n+p] * (1-h[n]) where p = m - (n-1)

    n = len(h)
    p = m - (n - 1)
    grad = np.zeros(n)
    
    # h[j] appears as h[n+p] when n = j-p, valid if 0 <= j-p < n and 0 <= n < n
    # i.e., 0 <= j-p < n => p <= j < p+n
    # Also need 0 <= n = j-p < n (always if 0 <= j-p < n)
    j_start = max(0, p)
    j_end = min(n - 1, n - 1 + p)
    for j in range(j_start, j_end + 1):
        grad[j] += (1.0 - h[j - p])
    
    # h[j] appears in (1-h[n]) when n = j, valid if 0 <= j < n and 0 <= j+p < n
    j_start2 = max(0, -p)
    j_end2 = min(n - 1, n - 1 - p)
    for j in range(j_start2, j_end2 + 1):
        grad[j] -= h[j + p]
    
    return grad

Wait, C₅ = max_m output[m] * dx where output[m] includes m=n-1 (the zero-lag) which is 0 for binary h.

For the other shifts:
For shift p (where p = m-(n-1), ranging from -(n-1) to n-1):
output[m] = #{i: 0≤i<n, 0≤i+p<n, h[i]=0, h[i+p]=1}

The maximum of this over all p is what we want to minimize.

This is exactly the "maximum cross-correlation" problem in coding theory. For a binary sequence of length n with weight k=n/2, the minimum possible maximum cross-correlation is related to the "cyclic autocorrelation" or "aperiodic autocorrelation" problem.

The aperiodic autocorrelation at lag p is:
R_p = Σ_{i=0}^{n-1-|p|} (2h[i]-1)(2h[i+p]-1) (for the "de Bruijn" or " Barker" type sequences)

But our quantity is different. Let me define x[i] = 2h[i]-1 ∈ {-1, +1}. Then:
h[i](1-h[i+p]) = (x[i]+1)/2 * (1-(x[i+p]+1)/2) = (x[i]+1)(1-x[i+p])/4

So output[m] = Σ_i (x[i]+1)(1-x[i+p])/4 = (1/4) Σ_i (x[i] - x[i]x[i+p] + 1 - x[i+p])
= (1/4) [Σ_i x[i] - Σ_i x[i]x[i+p] + (n-|p|) - Σ_i x[i+p]]

For a "balanced" sequence (equal number of +1 and -1), Σ_i x[i] = 0 (approximately, for the full range).

import numpy as np
from scipy.optimize import minimize
import time


def _project(v, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _grad_m(h, m):
    n = len(h)
    p = m - (n - 1)
    grad = np.zeros(n)
    js = max(0, p)
    je = min(n - 1, p + n - 1)
    if js <= je:
        idx = np.arange(js, je + 1)
        grad[idx] += 1.0 - h[idx - p]
    js2 = max(0, -p)
    je2 = min(n - 1, n - 1 - p)
    if js2 <= je2:
        idx2 = np.arange(js2, je2 + 1)
        grad[idx2] -= h[idx2 + p]
    return grad


def _psd(h0, n, max_iter, deadline, step_init=0.02, decay=0.95, decay_every=2000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    dx = 2.0 / n
    step = step_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = int(np.argmax(corr))
        c5 = float(corr[m] * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        grad = _grad_m(h, m)
        h = _project(h - step * grad * dx, n)
        if (it + 1) % decay_every == 0:
            step *= decay
    
    return best_h, best_c5


def _coby_five_obj(h):
    return _c5(h)


def _coby_five_eq(h):
    return 43.0 - h.sum()


def _coby_opt(h0, n, maxiter, rhobeg=0.05):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _coby_five_eq}]
    res = minimize(_coby_five_obj, h0, method='COBYLA', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'rhobeg': rhobeg})
    if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
        h_res = np.clip(res.x, 0.0, 1.0)
        h_res = _project(h_res, n)
        return h_res, _c5(h_res)
    return h0, _c5(h0)


def _local_search(h, n, iterations, deadline):
    best_h = h.copy()
    best_c5 = _c5(h)
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        # Pick a random coordinate to modify
        i = it % n
        # Try several values
        candidates = [0.0, 1.0]
        if best_h[i] > 0.05:
            candidates.append(max(0.0, best_h[i] - 0.1))
        if best_h[i] < 0.95:
            candidates.append(min(1.0, best_h[i] + 0.1))
        if 0 < best_h[i] < 1:
            candidates.append(best_h[i] - 0.02)
            candidates.append(best_h[i] + 0.02)
        
        for val in candidates:
            h_try = best_h.copy()
            h_try[i] = val
            h_try = _project(h_try, n)
            c5_try = _c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    return best_h, best_c5


def _perturbed_start(h, n, rng, scale=0.1):
    h_p = h + rng.normal(0, scale, n)
    return _project(h_p, n)
