            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pgd_optimize(h0, n, target_sum, deadline, max_iter=3000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = 15.0
    lr = 0.03
    stall = 0
    stall_thresh = 40
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, g = soft_max_obj_and_grad(h, tau)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > stall_thresh:
                tau *= 1.8
                lr = max(lr * 0.6, 0.002)
                stall = 0
                if lr < 0.002:
                    break
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, deadline, tau=80.0):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        if res.success or hasattr(res, 'x'):
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def worst_case_pgd(h0, n, target_sum, deadline, max_iter=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.95
            if lr < 0.001:
                lr = 0.02
                h = best_h + 0.005 * np.random.randn(n)
                h = project_onto_constraint(h, target_sum)
    return best_h, best_c5

def random_binary_init(n, target_sum):
    h = np.zeros(n)
    ones = np.random.choice(n, size=int(target_sum), replace=False)
    h[ones] = 1.0
    return h

def sinusoidal_init(n, target_sum, freq=2.0, phase=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_onto_constraint(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 15

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0

    # Phase 1: SLSQP from initial point
    if h_init is not None and time.time() < deadline - 100:
        n0 = len(h_init)
        h_opt, c5_opt = slsqp_optimize(h_init, n0, n0 / 2.0, deadline, tau=50.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: PGD with annealing from best
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        h_opt, c5_opt = pgd_optimize(best_h, n, n / 2.0, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Try other n values with SLSQP
    n_values_to_try = [42, 50, 60, 72, 80, 96, 100, 120]
    for n in n_values_to_try:
        if time.time() > deadline - 60:
            break
        target = n / 2.0
        if n == (len(best_h) if best_h is not None else -1):
            continue
        # Interpolate best solution to new size
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, best_h)
        else:
            h0 = np.full(n, 0.5)
        h0 = project_onto_constraint(h0, target)
        h_opt, c5_opt = slsqp_optimize(h0, n, target, deadline, tau=60.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: PGD from best with higher tau
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        h_opt, c5_opt = pgd_optimize(best_h, n, n / 2.0, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Random restarts with SLSQP
    if time.time() < deadline - 30:
        for n in [42, 60, 80, 100]:
            if time.time() > deadline - 15:
                break
            target = n / 2.0
            for attempt in range(4):
                if time.time() > deadline - 15:
                    break
                if attempt == 0:
                    h0 = random_binary_init(n, target)
                elif attempt == 1:
                    h0 = sinusoidal_init(n, target, freq=np.random.uniform(1, 5),
                                         phase=np.random.uniform(0, 2 * np.pi))
                else:
                    h0 = np.random.uniform(0.2, 0.8, n)
                    h0 = project_onto_constraint(h0, target)
                h_opt, c5_opt = slsqp_optimize(h0, n, target, deadline, tau=100.0)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 6: Final PGD polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_opt, c5_opt = pgd_optimize(best_h, n, n / 2.0, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final worst-case PGD
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        h_opt, c5_opt = worst_case_pgd(best_h, n, n / 2.0, deadline, max_iter=2000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5, len(best_h)
