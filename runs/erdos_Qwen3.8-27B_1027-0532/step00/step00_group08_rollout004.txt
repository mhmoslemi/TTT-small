import numpy as np
import time

def project_h(v, target_sum, n):
    """Project v onto {h: sum(h) = target_sum, 0 <= h <= 1}"""
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_smooth_gradient(h, dx, beta):
    """Compute gradient of soft-max of correlations"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros(n)
    n_corr = len(corr)
    for m in range(n_corr):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)  # lag
        # For this lag, grad[j] += w * [(1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)]
        for j in range(n):
            val = 0.0
            a = j - k
            if 0 <= a < n:
                val += (1.0 - h[a])
            b = j + k
            if 0 <= b < n:
                val -= h[b]
            grad[j] += w * val
    return grad

import time

    for _ in range(80):
