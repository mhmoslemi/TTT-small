    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-10:
        return np.full(n, 0.5)
    h = h * (target / s)

        else:

        if mask.any():

            h = np.clip(h, 0, 1)

    corr = compute_corr(h, n)

def generate_inits(n, rng):
    inits = []
    x = np.linspace(0, 2, n)
    inits.append(np.full(n, 0.5))
    for sigma in [0.15, 0.3, 0.5, 0.7, 1.0]:
        h_g = np.exp(-((x - 1.0) ** 2) / (2 * sigma ** 2))
        s = h_g.sum()
        if s > 0:
            h_g = h_g / s * (n / 2.0)
            inits.append(h_g)
    for _ in range(5):
        inits.append(rng.rand(n) * 0.9 + 0.05)
    h_alt = np.where(np.arange(n) % 2 == 0, 0.8, 0.2)
    inits.append(h_alt)
    h_alt3 = np.where(np.arange(n) % 3 == 0, 0.9, 0.35)
    inits.append(h_alt3)
    for c1 in [0.3, 0.5]:
        for c2 in [1.4, 1.7]:
            h_2b = np.exp(-((x - c1) ** 2) / 0.08) + np.exp(-((x - c2) ** 2) / 0.08)
            s = h_2b.sum()
            if s > 0:
                h_2b = h_2b / s * (n / 2.0)
                inits.append(h_2b)
    for freq in [1, 2, 3, 4, 6, 8]:
        h_sin = 0.5 + 0.25 * np.sin(2 * np.pi * freq * x)
        inits.append(h_sin)
    h_tri = 1.0 - np.abs(x - 1.0)
    s = h_tri.sum()
    if s > 0:
        h_tri = h_tri / s * (n / 2.0)
        inits.append(h_tri)
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    def elapsed():
        return time.time() - start_time
    
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        pass
    
    if h_init is not None:
        n = len(h_init)
        h = repair_h(h_init, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        if budget_s - elapsed() > 20:
            h_cd, c5_cd = coordinate_descent(h.copy(), n, max_sweeps=300, time_limit=min(40, budget_s * 0.06))
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n
    
    n_values = [63, 81, 99, 125, 155, 199, 255, 300]
    if best_n not in n_values:
        n_values.insert(0, best_n)
    
    for n_try in n_values:
        if budget_s - elapsed() < 15:
            break
        
        inits = generate_inits(n_try, rng)
        if best_h is not None and best_n != n_try:
            h_interp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, best_n), best_h)
            inits.append(h_interp)
            h_interp2 = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, len(best_h)), best_h)
            inits.append(h_interp2)
        
        per_init_time = min(5.0, (budget_s * 0.45) / (len(n_values) * len(inits)))
        
        for h_init_try in inits:
            if budget_s - elapsed() < 10:
                break
            h_try = repair_h(h_init_try.copy(), n_try)
            c5_try = compute_c5(h_try, n_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            
            time_left = budget_s - elapsed()
            tl = min(per_init_time, time_left * 0.3)
            if tl > 1.0:
                h_cd, c5_cd = coordinate_descent(h_try.copy(), n_try, max_sweeps=50, time_limit=tl)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n_try
    
    if best_h is not None and budget_s - elapsed() > 20:
        n_best = len(best_h)
        h_ref = best_h.copy()
        tl = min(60, (budget_s - elapsed()) * 0.3)
        h_ref, c5_ref = coordinate_descent(h_ref, n_best, max_sweeps=500, time_limit=tl)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_best
    
    if best_h is not None and budget_s - elapsed() > 15:
        n_best = len(best_h)
        for alpha in [20.0, 50.0, 100.0, 200.0]:
            if budget_s - elapsed() < 8:
                break
            h_sm = best_h.copy()
            try:
                result = minimize(sm_objective, h_sm, args=(n_best, alpha),
                                  method='L-BFGS-B',
                                  options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
                h_opt = repair_h(result.x, n_best)
                c5_opt = compute_c5(h_opt, n_best)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_best
            except Exception:
                pass
    
    if best_h is not None and budget_s - elapsed() > 10:
        n_best = len(best_h)
        for noise_scale in [0.05, 0.1, 0.15, 0.2]:
            if budget_s - elapsed() < 5:
                break
            h_pert = best_h + noise_scale * rng.randn(n_best)
            h_pert = repair_h(h_pert, n_best)
            tl = min(15, (budget_s - elapsed()) * 0.3)
            h_cd, c5_cd = coordinate_descent(h_pert, n_best, max_sweeps=100, time_limit=tl)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_best
    
    if best_h is not None:
        best_h = repair_h(best_h, len(best_h))
        best_c5 = compute_c5(best_h, len(best_h))
    
    if best_h is None:
        n = 63
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, n)
        best_n = n
    
    return best_h, best_c5, len(best_h)
