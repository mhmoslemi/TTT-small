import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def coord_step(h, j, n_candidates):
    n = len(h)
    old = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    base_max = float(np.max(corr))
    
    D = np.zeros(2*n-1)
    for ell in range(1, n):
        cp = 0.0
        if j + ell < n:
            cp += 1.0 - h[j+ell]
        if j - ell >= 0:
            cp -= h[j-ell]
        D[n-1+ell] = cp
        cn = 0.0
        if j - ell >= 0:
            cn += 1.0 - h[j-ell]
        if j + ell < n:
            cn -= h[j+ell]
        D[n-1-ell] = cn
    
    candidates = np.linspace(0.0, 1.0, n_candidates)
    best_t = old
    best_max = base_max
    
    for t in candidates:
        if abs(t - old) < 1e-14:
            continue
        dt = t - old
        new_vals = corr + dt * D
        new_vals[n-1] = corr[n-1] + (t - t*t) - (old - old*old)
        nm = float(np.max(new_vals))
        if nm < best_max:
            best_max = nm
            best_t = t
    return best_t

def coord_descent(h, max_sweeps, n_candidates, t_start, deadline):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            nv = coord_step(h, j, n_candidates)
            if abs(nv - h[j]) > 1e-13:
                h[j] = nv
                improved = True
        if not improved:
            break
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

    s = np.sum(h)

        else:
            ts = np.sum(h)
            if ts > 1e-12:
                h = h * (1.0 + diff / ts)
        h = np.clip(h, 0.0, 1.0)
    return h
