    """Compute all 2n-1 correlation values times dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

    """Compute smooth objective value and gradient"""

import numpy as np
from scipy.optimize import minimize
import time

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_grad(h, eps, target_sum):
    n = len(h)
    dx = 2.0 / n
    lags = compute_lags(h)
    
    m = np.max(lags)
    exp_shifted = np.exp((lags - m) / eps)
    sum_exp = np.sum(exp_shifted)
    obj_val = m + eps * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        wi = p[idx] * dx
        if wi < 1e-15:
            continue
        L = idx - (n - 1)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:n] += wi * (1.0 - h[:n-L])
            grad[:n-L] -= wi * h[L:]
        else:
            M = -L
            grad[:n-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:n-M]
    
    sum_dev = np.sum(h) - target_sum
    obj_val += 1000.0 * sum_dev * sum_dev
    grad += 2000.0 * sum_dev * np.ones(n)
    
    return obj_val, grad

def enforce(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            t = np.sum(room)
            if t > 1e-15:
                h += room * min(diff, t) / t
        else:
            room = h
            t = np.sum(room)
            if t > 1e-15:
                h -= room * min(-diff, t) / t
    return np.clip(h, 0.0, 1.0)

    for eps in epsilons:
        if time.time() - t0 > budget_s - 100:
            break
        
        for r in range(3):
            if time.time() - t0 > budget_s - 100:
                break
            
            if r == 0:
                h0 = best_h.copy()
            else:
                h0 = best_h + 0.05 * np.random.randn(n)
                h0 = enforce(h0, ts)
            
            def f(x):
                v, _ = smooth_obj_grad(x, eps, ts)
                return v
            def g(x):
                _, gr = smooth_obj_grad(x, eps, ts)
                return gr
            
            try:
                res = minimize(f, h0, jac=g, method='L-BFGS-B',
                            bounds=[(0,1)]*n,
                            options={'maxiter':500, 'ftol':1e-15, 'gtol':1e-12})
                h_new = enforce(res.x, ts)
                c5 = float(np.max(compute_lags(h_new)))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new
            except:
                pass
    
    # Coordinate descent refinement
    if time.time() - t0 < budget_s - 50:
        h_c = best_h.copy()
        c5_c = best_c5
        eps_cd = 0.005
        for _ in range(50000):
            if time.time() - t0 > budget_s - 50:
                break
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            for delta in [eps_cd, -eps_cd]:
