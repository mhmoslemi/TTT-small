import time

    """Project x onto {y: 0 <= y_i <= 1, sum(y) = s} using bisection."""
    lo = x.min() - 2.0
    hi = x.max() + 2.0

        proj = np.clip(x - mid, 0.0, 1.0)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    # Phase 1: Cyclic pair optimization with multiple gaps
    gap_cycle = [1, 2, 3, 5, 8, 13]
    stuck = 0
    gi = 0
    while not done(0.50):
        gap = gap_cycle[gi % len(gap_cycle)]
        for j in range(n):
            i = (j + gap) % n
            pair_optimize(h, j, i)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_h = h.copy()
            best_c5 = c5
            stuck = 0
        else:
            stuck += 1
        if stuck >= 60:
            gi += 1
            stuck = 0
        if stuck >= 200:
            h = best_h.copy()
            n_pert = max(2, n // 8)
            idx = np.random.choice(n, n_pert, replace=False)
            h[idx] += np.random.uniform(-0.2, 0.2, n_pert)
            h = project_onto_box_hyperplane(h, s)
            stuck = 0
            gi = 0

    # Phase 2: Random pair optimization
    h = best_h.copy()
    patience = 0
    count = 0
    while not done(0.78):
        j = np.random.randint(0, n)
        i = np.random.randint(0, n)
        if i == j:
            i = (i + 1) % n
        pair_optimize(h, j, i)
        count += 1
        if count % 25 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-13:
                best_h = h.copy()
                best_c5 = c5
                patience = 0
            elif c5 > best_c5 + 0.003:
                h = best_h.copy()
                patience = 0
            else:
                patience += 1
        if patience > 80:
            h = best_h.copy()
            n_pert = max(2, n // 10)
            idx = np.random.choice(n, n_pert, replace=False)
            h[idx] += np.random.uniform(-0.12, 0.12, n_pert)
            h = project_onto_box_hyperplane(h, s)
            patience = 0

    # Phase 3: Targeted optimization focused on worst lags
    h = best_h.copy()
    while not done(0.92):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max_arr = np.argsort(corr)[-3:]
        for km in k_max_arr:
            for j in range(0, n, max(1, n // 15)):
                i = (j + int(km) - (n - 1)) % n
                if 0 <= i < n and i != j:
                    pair_optimize(h, j, i)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_h = h.copy()
            best_c5 = c5
        elif c5 > best_c5 + 0.005:
            h = best_h.copy()

    # Phase 4: Fine-tuning with very small random steps + pair optimize
    h = best_h.copy()
    patience = 0
    while not done(0.96):
        n_pert = np.random.randint(1, 4)
        idx = np.random.choice(n, n_pert, replace=False)
        h[idx] += np.random.uniform(-0.03, 0.03, n_pert)
        h = project_onto_box_hyperplane(h, s)
        for _ in range(3):
            j = np.random.randint(0, n)
            i = np.random.randint(0, n)
            if i == j:
                i = (i + 1) % n
            pair_optimize(h, j, i)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_h = h.copy()
            best_c5 = c5
            patience = 0
        else:
            patience += 1
            if patience > 60:
                h = best_h.copy()
                patience = 0

    # Phase 5: Try upscaling
    n_curr = len(best_h)
    for factor in [2, 3, 4]:
        if done(0.99):
            break
        n_new = n_curr * factor
        if n_new > 400:
            continue
        x_old = (np.arange(n_curr) + 0.5) * (2.0 / n_curr)
        x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_onto_box_hyperplane(h_new, n_new / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            for _ in range(400):
                if done(0.99):
                    break
                j = np.random.randint(0, n_new)
                i = np.random.randint(0, n_new)
                if i == j:
                    i = (i + 1) % n_new
                pair_optimize(h_new, j, i)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
