                    h_trial = h_cur + alpha * (h_lp - h_cur)
                    h_trial = repair(h_trial, n_pts)
                    c = compute_c5(h_trial, 2.0 / n_pts)
                    if c <= compute_c5(h_cur, 2.0 / n_pts) + 1e-15:
                        best_step = h_trial
                        break
                if best_step is not None:
                    h_cur = best_step
            h_cur = try_improve(h_cur, n_pts)

        # SLSQP polish at this resolution
        if elapsed() < budget_s * 0.55:
            ho = slsqp_step(h_cur, n_pts, max_iter=5000)
            if ho is not None:
                try_improve(ho, n_pts)

    # Phase 5: Random restarts at best resolution
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    max_restarts = 5000
    for i in range(max_restarts):
        if elapsed() > budget_s * 0.88:
            break
        progress = i / max_restarts
        scale = 0.3 * (1.0 - progress) ** 1.5 + 0.01
        h_pert = best_h + rng.normal(0, scale, n_best)
        h_pert = repair(h_pert, n_best)
        ho = slsqp_step(h_pert, n_best, max_iter=300)
        if ho is not None:
            try_improve(ho, n_best)

    # Phase 6: Binary-like restarts
    if elapsed() < budget_s * 0.88:
        for _ in range(200):
            if elapsed() > budget_s * 0.92:
                break
            h_bin = np.zeros(n_best)
            n_ones = n_best // 2
            pos = rng.choice(n_best, n_ones, replace=False)
            h_bin[pos] = 1.0
            ho = slsqp_step(h_bin, n_best, max_iter=200)
            if ho is not None:
                try_improve(ho, n_best)

    # Phase 7: Final intensive polish
    n_final = len(best_h)
    for _ in range(20):
        if elapsed() > budget_s * 0.97:
            break
        ho = slsqp_step(best_h, n_final, max_iter=10000, ftol=1e-16)
        if ho is not None:
            c_new = compute_c5(repair(ho, n_final), 2.0 / n_final)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = repair(ho, n_final)
        # Also try LP
        h_lp = lp_direction(best_h, n_final)
        if h_lp is not None:
            for alpha in [1.0, 0.5, 0.1, 0.01]:
                h_trial = best_h + alpha * (h_lp - best_h)
                h_trial = repair(h_trial, n_final)
                c = compute_c5(h_trial, 2.0 / n_final)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_trial
                    break

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
