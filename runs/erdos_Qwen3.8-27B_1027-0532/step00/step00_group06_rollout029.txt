import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / T

def interpolate_to_size(parent, n_new):
    n_old = len(parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, parent, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    # Project back to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Adjust to satisfy sum constraint
    target = n_new / 2.0
    current = np.sum(h_new)
    if current > 0:
        h_new = h_new - (current - target) / n_new
        h_new = np.clip(h_new, 0, 1)
        # If clipping changed the sum, do a projection
        # Simple iteration
        for _ in range(100):
            diff = target - np.sum(h_new)
            if abs(diff) < 1e-10:
                break
            # Add/subtract uniformly from elements that can absorb it
            if diff > 0:
                mask = h_new < 1.0
            else:
                mask = h_new > 0.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h_new[mask] += step
            h_new = np.clip(h_new, 0, 1)
    return h_new

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    for _ in range(1000):
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

def optimize_grid(h_init, n, time_limit, T=50.0):
    """Run SLSQP optimization on a grid of size n"""
    bounds = [(0, 1)] * n
    
    def obj(x):
        return smooth_objective(x, T)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    best_h = h_init.copy()
    best_obj = compute_c5(best_h)
    
    start = time.time()
    while time.time() - start < time_limit:
        result = minimize(
            obj, best_h, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
