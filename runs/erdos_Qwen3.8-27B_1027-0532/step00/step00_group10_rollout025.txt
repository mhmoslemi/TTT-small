    overlaps = np.correlate(h, 1-h, mode='full') * dx

                    best = v
                    improved = True
                    break
        if not improved:
            break
    return h, best

def _refine(h, target, max_rounds=8):
    n = len(h)
    best = _max_ov(h)
    deltas = [0.02, 0.01, 0.005, -0.02, -0.01, -0.005]
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                h2 = h.copy()
                h2[i] = np.clip(h2[i] + d, 0.0, 1.0)
                h2 = _project(h2, target)
                v = _max_ov(h2)
                if v < best - 1e-10:
                    h = h2
                    best = v
                    improved = True
                    break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    n = 97
    target = n / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)[:n].copy()
    else:
        h = np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)
    h = _project(h, target)

    best_h = h.copy()
    best_val = _max_ov(h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient descent with momentum and restarts
    alpha = 25.0
    step = 0.06
    mom = np.zeros(n)
    beta = 0.9
    patience = 0
    phase1_limit = 0.55 * budget_s

    while (time.time() - t0) < phase1_limit:
        g, ov, cmx = _subgrad(h, alpha)
        g -= g.mean()
        mom = beta * mom + g
        d = -mom
        dn = np.linalg.norm(d)
        if dn < 1e-16:
            break
        d /= dn

        tried = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            h_new = _project(h + step * scale * d, target)
            v = _max_ov(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                patience = 0
                tried = True
                break
            elif v < cmx:
                h = h_new
                tried = True
                break
        if not tried:
            h = _project(h + step * 0.1 * d, target)
            patience += 1

        if patience > 60:
            scale = 0.02 + 0.06 * min(patience / 300.0, 1.0)
            h = _project(best_h + np.random.randn(n) * scale, target)
            mom = np.zeros(n)
            step = max(0.008, step * 0.88)
            alpha = min(alpha * 1.15, 250.0)
            patience = 0

    # Phase 2: Finer subgradient
    alpha = 100.0
    step = 0.02
    mom = np.zeros(n)
    patience = 0
    h = best_h.copy()

    while tl() > 60.0:
        g, ov, cmx = _subgrad(h, alpha)
        g -= g.mean()
        mom = beta * mom + g
        d = -mom
        dn = np.linalg.norm(d)
        if dn < 1e-16:
            break
        d /= dn

        improved = False
        for scale in [1.0, 0.5, 0.25]:
            h_new = _project(h + step * scale * d, target)
            v = _max_ov(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                improved = True
                patience = 0
                break
        if not improved:
            h = _project(h + step * 0.5 * d, target)
            if _max_ov(h) < best_val:
                best_val = _max_ov(h)
                best_h = h.copy()
                patience = 0
            else:
                patience += 1

        if patience > 100:
            h = _project(best_h + np.random.randn(n) * 0.015, target)
            mom = np.zeros(n)
            step = max(0.003, step * 0.85)
            patience = 0

    # Phase 3: Coordinate descent polish
    if tl() > 15.0:
        best_h, best_val = _coord_descent(best_h, target, max_rounds=12)

    # Phase 4: Fine refinement
    if tl() > 5.0:
        best_h, best_val = _refine(best_h, target, max_rounds=10)

    # Final random perturbations + polish
    while tl() > 3.0:
        h = _project(best_h + np.random.randn(n) * 0.01, target)
        v = _max_ov(h)
        if v < best_val:
            best_val = v
            best_h = h.copy()
            best_h, best_val = _refine(best_h, target, max_rounds=3)

    return best_h, best_val, n
