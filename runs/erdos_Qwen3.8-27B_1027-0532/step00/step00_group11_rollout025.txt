                h_new = fix_feasibility(res.x, _TARGET)
                c5_new = compute_c5(h_new, _DX)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  si={}, NM: {:.6f}".format(si, best_c5))
            except Exception:
                pass

    # Final polishing with very fine COBYLA
    if remaining() > 40:
        try:
            res = minimize(obj_raw, best_h, method='COBYLA',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 8000, 'rhobeg': 0.001})
            h_new = fix_feasibility(res.x, _TARGET)
            c5_new = compute_c5(h_new, _DX)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polish: {:.6f}".format(best_c5))
        except Exception:
            pass

    # One more round: SLSQP with very low temperature on the best
    if remaining() > 20:
        try:
            def final_obj(h):
                vals = compute_corr_vals(h, _DX)
                m = float(np.max(vals))
                t_val = 200.0
                shifted = t_val * (vals - m)
                shifted = np.clip(shifted, -500.0, 500.0)
                return m + (1.0 / t_val) * float(np.log(np.sum(np.exp(shifted))))
            res = minimize(final_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = fix_feasibility(res.x, _TARGET)
            c5_new = compute_c5(h_new, _DX)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Final polish: {:.6f}".format(best_c5))
        except Exception:
            pass

    best_h = fix_feasibility(best_h, _TARGET)
    final_c5 = compute_c5(best_h, _DX)
    print("Final: C5 = {:.6f}, sum = {:.6f}".format(final_c5, np.sum(best_h)))
    return (best_h, final_c5, _N)
