    j_pk = j + k  # j + k for each lag

    h_jk = np.where(valid_jk, h[np.clip(j_km, 0, N-1)], 0.0)

Step 1: Find the minimum of g(t) = max_{m≠N-1}(A_m + B_m * t) on [0,1].
- This is the minimum of the upper envelope of M-1 lines on [0,1].
- Algorithm: 
  a. At t=0, find the line with max A_m. Call it line i.
  b. At t=0, g'(0+) = B_i (the slope of the max line). If B_i >= 0, minimum is at t=0.
  c. If B_i < 0, find the next line j that intersects line i at the smallest t > 0. At that intersection, line j becomes the max (if it's above all others there).
  d. Continue until the slope becomes positive or we reach t=1.
  e. The minimum is at the last intersection before the slope becomes positive, or at t=1.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(v, total_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > total_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(v - t, 0.0, 1.0)

def optimize_one_coordinate(h, j, dx):
    N = len(h)
    t_old = h[j]
    corr = np.correlate(h, 1.0 - h, mode="full")
    N_lags = 2 * N - 1
    lags = np.arange(N_lags) - (N - 1)
    j_mk = j - lags
    j_pk = j + lags
    vm = (j_mk >= 0) & (j_mk < N)
    vp = (j_pk >= 0) & (j_pk < N)
    j_mk_c = np.clip(j_mk, 0, N - 1)
    j_pk_c = np.clip(j_pk, 0, N - 1)
    h_m = np.where(vm, h[j_mk_c], 0.0)
    h_p = np.where(vp, h[j_pk_c], 0.0)
    B = (1.0 - h_m) * vm.astype(np.float64) - h_p * vp.astype(np.float64)
    A = corr - t_old * B
    A[N - 1] = corr[N - 1] - t_old + t_old * t_old
    mask = np.ones(N_lags, dtype=bool)
    mask[N - 1] = False
    A_lin = A[mask]
    B_lin = B[mask]
    t_grid = np.linspace(0.0, 1.0, 1001)
    lin_max = np.max(A_lin[:, None] + B_lin[:, None] * t_grid[None, :], axis=0)
    quad = A[N - 1] + t_grid - t_grid * t_grid
    f_vals = dx * np.maximum(lin_max, quad)
    idx = np.argmin(f_vals)
    best_t = t_grid[idx]
    if 0.0 < best_t < 1.0:
        hw = (t_grid[1] - t_grid[0]) * 0.5
        t_fine = np.linspace(max(0.0, best_t - hw), min(1.0, best_t + hw), 301)
        lin_max_f = np.max(A_lin[:, None] + B_lin[:, None] * t_fine[None, :], axis=0)
        quad_f = A[N - 1] + t_fine - t_fine * t_fine
        f_vals_f = dx * np.maximum(lin_max_f, quad_f)
        best_t = t_fine[np.argmin(f_vals_f)]
    h[j] = best_t

def coordinate_descent(h, dx, n_passes=100):
    N = len(h)
    total_sum = N / 2.0
    h = h.astype(np.float64).copy()
    prev_c5 = compute_c5(h, dx)
    stagnation = 0
    for p in range(n_passes):
        for j in range(N):
            optimize_one_coordinate(h, j, dx)
        h = project_feasible(h, total_sum)
        c5 = compute_c5(h, dx)
        if c5 >= prev_c5 - 1e-12:
            stagnation += 1
        else:
            stagnation = 0
        if stagnation >= 5:
            break
        prev_c5 = c5
    return h, compute_c5(h, dx)

def interpolate_to_resolution(h_old, N_new):
    N_old = len(h_old)
    dx_old = 2.0 / N_old
    dx_new = 2.0 / N_new
    x_old = (np.arange(N_old) + 0.5) * dx_old
    x_new = (np.arange(N_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, N_new / 2.0)
    return h_new

def make_periodic_pattern(N, period, frac):
    h = np.zeros(N)
    for i in range(N):
        if (i % period) < frac:
            h[i] = 1.0
    return h

def make_random_binary(N, rng):
    n_ones = N // 2
    idx = rng.choice(N, size=n_ones, replace=False)
    h = np.zeros(N)
    h[idx] = 1.0
    return h
