def project_to_feasible(p, n):
    target = n / 2.0
    lo = -np.max(p) - 2.0
    hi = 1.0 - np.min(p) + 2.0
    for _ in range(80):

            hi = mid
        else:
            lo = mid
    tau = 0.5 * (lo + hi)
    return np.clip(p + tau, 0.0, 1.0)

def compute_subgrad_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        t = -s
        grad[:n - t] += (1.0 - h[t:])
        grad[t:] += (-h[:n - t])
