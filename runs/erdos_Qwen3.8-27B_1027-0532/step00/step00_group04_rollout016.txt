```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = 1e-10
    active = corr >= max_val - tol
    
    sg = np.zeros(n)

            continue

            kk = -k
            sg_k[:n-kk] += (1 - h[kk:])
            sg_k[kk:] -= h[:n-kk]
        sg += sg_k
```

```python
def project_feasible(y, n):
    # Project y onto {h : 0 ≤ h ≤ 1, sum(h) = n/2}
    # Binary search on lambda
    target = n / 2.0
    lo, hi = -1.0, 2.0  # lambda range (y[i] - lambda clipped to [0,1])
    # sum(clip(y - lambda, 0, 1)) is decreasing in lambda
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
```

Let me implement pair swaps properly:

For a pair (i, j) and shift δ (h[i] -= δ, h[j] += δ):
- For each lag m, c[m] changes by:
  Δc[m] = δ * (∂c[m]/∂h[j] - ∂c[m]/∂h[i])
- New c[m] = c[m] + δ * d_m where d_m = ∂c[m]/∂h[j] - ∂c[m]/∂h[i]
- New objective = max_m (c[m] + δ * d_m)
- This is a convex piecewise-linear function of δ
- Optimal δ: find where two "active" constraints cross, or at a boundary

The valid range of δ:
- h[i] - δ ≥ 0 → δ ≤ h[i]
- h[j] + δ ≤ 1 → δ ≤ 1 - h[j]
- h[i] - δ ≤ 1 → δ ≥ h[i] - 1 (always true since h[i] ≤ 1)
- h[j] + δ ≥ 0 → δ ≥ -h[j]

So δ ∈ [-h[j], min(h[i], 1-h[j])]

To find the optimal δ, I need to find the minimum of max_m (c[m] + δ * d_m) over δ ∈ [δ_min, δ_max].

- At δ_min or δ_max

```python
def pair_swap_minimize(corr, h, i, j, n):
    # corr is the current correlation array (length 2n-1)
    # Try swap: h[i] -= δ, h[j] += δ
    # Compute d_m = ∂c[m]/∂h[j] - ∂c[m]/∂h[i] for each m
    # Find δ that minimizes max_m (corr[m] + δ * d_m)
    
    # Compute d_m for all m
    d = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            # ∂c[m]/∂h[i] = (1-h[i-k])*[i≥k] - h[i+k]*[i<n-k]
            # ∂c[m]/∂h[j] = (1-h[j-k])*[j≥k] - h[j+k]*[j<n-k]
            gi = 0
            if i >= k:
                gi += (1 - h[i-k])
            if i+k < n:
                gi -= h[i+k]
            gj = 0
            if j >= k:
                gj += (1 - h[j-k])
            if j+k < n:
                gj -= h[j+k]
        else:
            kk = -k
            gi = 0
            if i+kk < n:
                gi += (1 - h[i+kk])
            if i-kk >= 0:
                gi -= h[i-kk]
            gj = 0
            if j+kk < n:
                gj += (1 - h[j+kk])
            if j-kk >= 0:
                gj -= h[j-kk]
        d[m] = gj - gi
    
    # Valid range for δ
    delta_min = -h[j]
    delta_max = min(h[i], 1 - h[j])
    
    if delta_min >= delta_max:
        return 0
    
    # Find δ that minimizes max_m (corr[m] + δ * d_m)
    # Use ternary search (convex function)
    a, b = delta_min, delta_max
    for _ in range(50):  # ~50 iterations of ternary search gives good precision
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        f1 = np.max(corr + m1 * d)
        f2 = np.max(corr + m2 * d)
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    delta_opt = (a + b) / 2
    return delta_opt
```

Let me also consider: what if I use the reformulation with auxiliary variable t?

minimize t
subject to:
  ∑_i h[i]*(1-h[i+k]) * dx ≤ t  for all k (these are the correlation constraints)
  ∑ h[i] = n/2
  0 ≤ h[i] ≤ 1

Each constraint ∑_i h[i]*(1-h[i+k]) * dx ≤ t is a quadratic inequality in h. This is a QCQP (quadratically constrained quadratic program). It's non-convex in general.

But wait... each individual constraint ∑_i h[i]*(1-h[i+k]) is it convex or concave in h?
∑_i h[i] - ∑_i h[i]*h[i+k]

The term -∑_i h[i]*h[i+k] is a quadratic form. The Hessian of this term with respect to h is:
- It's a banded matrix with -1 on the k-th sub/super diagonal (for the relevant positions)
- This matrix is indefinite (has both positive and negative eigenvalues)

So the constraints are non-convex. Standard convex solvers won't work directly.

However, for a FIXED k, if I fix all coordinates except h[i], the constraint becomes linear in h[i] (as I showed earlier). This is the key insight that makes coordinate descent work.

Let me go with the pair swap approach. It's simple, effective, and I can make it fast enough.

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def grad_col(h, n, i):
    """Compute ∂c[m]/∂h[i] for m = 0, ..., 2n-2."""
    col = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)

            g = 0.0
            if i >= k:
                g += (1.0 - h[i - k])
            if i + k < n:
                g -= h[i + k]
        else:
            kk = -k
            g = 0.0
            if i + kk < n:
                g += (1.0 - h[i + kk])
            if i - kk >= 0:
                g -= h[i - kk]
        col[m] = g
    return col

def find_optimal_delta(curr, d, dmin, dmax):
    """Find delta in [dmin, dmax] minimizing max(curr + delta * d)."""
    if dmin >= dmax:
        return 0.0
    a, b = dmin, dmax
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(curr + m1 * d) < np.max(curr + m2 * d):
            b = m2
        else:
            a = m1

                if abs(delta) < 1e-14:
                    continue

                h[i] = old_i - delta
                h[j] = old_j + delta

                curr = np.correlate(h, 1.0 - h, mode="full")
                new_val = np.max(curr) * dx
```
