import numpy as np
import time

def compute_all_overlaps(h, dx):
    """Compute all overlap values f(k) for k = -(n-1) to n-1."""
    n = len(h)
    # np.correlate(h, 1-h, mode='full') gives the values
    # The lag index: result[k] corresponds to lag = k - (n-1)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx  # shape (2n-1,)

def get_gradient_for_lag(h, lag, n):
    """Compute gradient of f(lag) with respect to h.
    lag is in range [-(n-1), n-1]
    f(lag) = sum_{i valid} h[i] * (1 - h[i+lag])
    where valid means 0 <= i < n and 0 <= i+lag < n
    """
    grad = np.zeros(n)
    if lag >= 0:
        # i ranges from 0 to n-1-lag
        # h[i] * (1 - h[i+lag]) for i = 0, ..., n-1-lag
        i_range = np.arange(0, n - lag)
        grad[i_range] += (1.0 - h[i_range + lag])
        # The -h[i] part: h[i+lag] appears, so for j = i+lag, i = j-lag
        # j ranges from lag to n-1
        j_range = np.arange(lag, n)
        grad[j_range] += -h[j_range - lag]
    else:
        # lag < 0, let l = -lag > 0
        # f(lag) = sum_{i valid} h[i] * (1 - h[i+lag])
        # i+lag >= 0 means i >= -lag = l
        # i+lag < n means i < n - lag = n + l
        # so i ranges from l to n-1
        l = -lag
        i_range = np.arange(l, n)
        grad[i_range] += (1.0 - h[i_range + lag])  # h[i-l]
        # -h[i+lag] = -h[i-l], for j = i-l, i = j+l
        # j ranges from 0 to n-1-l
        j_range = np.arange(0, n - l)
        grad[j_range] += -h[j_range + l]
    return grad

def project_to_sum(h, target_sum, n):
    """Project h onto {x: 0 <= x_i <= 1, sum x = target_sum}.
    Uses bisection on the shift parameter.
    """
    # We want to find t such that sum(clamp(h[i] + t, 0, 1)) = target_sum
    lo, hi = -max(h) - 1.0, 1.0 - min(h) + 1.0
    
    def sum_clamped(t):
        return np.sum(np.clip(h + t, 0.0, 1.0))
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_clamped(mid)
        if s > target_sum:
            lo = mid  # need to decrease values, so decrease t
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h + t, 0.0, 1.0)

    best_c5 = current_c5

    temperature = 0.05  # For softmax weighting of active constraints
    momentum = 0.3
    v = np.zeros(n)  # Momentum variable

        if elapsed > budget_s - 5:

            patience = 0
        else:

            h = project_to_sum(h, target_sum, n)
            patience = 0
            learning_rate = max(0.001, learning_rate * 0.5)
            temperature = max(0.01, temperature * 0.5)

        # Weight the gradients by softmax of overlaps
        # Use only the top few to be efficient
        weights = np.exp((overlaps - c5_val) / temperature)
        weights /= np.sum(weights)

        top_k = np.argsort(overlaps)[-10:][::-1]  # Top 10 lags
        top_w = weights[top_k]

            else:

        # Adaptive learning rate

    # Final evaluation
    final_overlaps = compute_all_overlaps(best_h, dx)
    final_c5 = np.max(final_overlaps)
    
    return best_h, final_c5, n

            grad[i] += (1.0 - h[i + lag])
            j = np.arange(ni)
            grad[j] -= h[j + l]
    return grad

def project_feasible(h, target, n):
    lo = -h.max() - 0.5
    hi = 1.0 - h.min() + 0.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h + t, 0.0, 1.0)

def subgradient_step(h, target, n, lr, momentum, v, temperature, top_k=15):
    y = h + momentum * v
    y = project_feasible(y, target, n)
    overlaps = compute_overlaps(y)
    c5 = np.max(overlaps)
    shifted = (overlaps - c5) / temperature
    max_sh = np.max(shifted)
    weights = np.exp(shifted - max_sh)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(len(overlaps))
        weights[np.argmax(overlaps)] = 1.0
    else:
        weights /= wsum
    idx = np.argsort(overlaps)[-top_k:][::-1]
    tw = weights[idx]
    tsum = tw.sum()
    if tsum < 1e-30:
        tw = np.zeros(len(idx))
        tw[0] = 1.0
    else:
        tw /= tsum
    grad = np.zeros(n)
    for ii in range(len(idx)):
        lag = int(idx[ii]) - (n - 1)
        grad += tw[ii] * gradient_at_lag(y, lag, n)
    v_new = momentum * v - lr * grad
    h_new = project_feasible(y + v_new, target, n)
    return h_new, v_new, c5, y

def optimize_from(h_init, target, n, budget_remaining, seed, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = np.max(compute_overlaps(h))
    v = np.zeros(n)
    start = time.time()
    phase = 0
    max_phases = 30
    while time.time() - start < budget_remaining - 5 and phase < max_phases:
        temperature = max(0.008, 0.06 * (0.88 ** phase))
        lr = 0.008 + 0.004 * (1.0 - phase / max_phases)
        mom = 0.3 + 0.25 * (1.0 - phase / max_phases)
        top_k = min(15 + 5 * (1.0 - phase / max_phases), 30)
        stuck = 0
        phase_start = time.time()
        phase_budget = min(80.0, budget_remaining - (time.time() - start) - 5)
        while time.time() - phase_start < phase_budget:
            h, v, c5, y = subgradient_step(h, target, n, lr, mom, v, temperature, int(top_k))
            h_c5 = np.max(compute_overlaps(h))
            if h_c5 < best_c5 - 1e-10:
                best_c5 = h_c5
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 400:
                h = best_h + 0.004 * rng.randn(n)
                h = project_feasible(h, target, n)
                v = np.zeros(n)
                stuck = 0
                lr = max(0.002, lr * 0.7)
                temperature = max(0.005, temperature * 0.6)
        h = best_h + 0.003 * rng.randn(n)
        h = project_feasible(h, target, n)
        v = np.zeros(n)
        phase += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global_start = time.time()
    rng = np.random.RandomState(seed)
    n = 97
    dx = 2.0 / n
    target = n / 2.0

    candidates = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64).flatten()
        if len(h0) == n:
            candidates.append(h0.copy())
        elif len(h0) > n:
            candidates.append(h0[:n].copy())
        else:
            tmp = np.zeros(n)
            tmp[:len(h0)] = h0
            candidates.append(tmp)

    for block in [3, 4, 5, 6, 7, 8, 9, 11, 13]:
        hb = np.zeros(n)
        pos = 0
        on = True
        while pos < n:
            b = min(block, n - pos)
            if on:
                hb[pos:pos+b] = 1.0
            pos += b
            on = not on
        candidates.append(hb.copy())

    for _ in range(4):
        t = np.linspace(0, 6.0 * np.pi, n)
        a1 = rng.uniform(0.2, 0.5)
        a2 = rng.uniform(0.1, 0.4)
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        f1 = rng.uniform(1, 3)
        f2 = rng.uniform(3, 7)
        hs = 0.5 + a1 * np.sin(f1 * t + ph1) + a2 * np.sin(f2 * t + ph2)
        candidates.append(np.clip(hs, 0, 1))

    for _ in range(3):
        candidates.append(rng.rand(n))

    best_h = None
    best_c5 = float('inf')
    for cand in candidates:
        hc = project_feasible(cand.copy(), target, n)
        c5v = np.max(compute_overlaps(hc))
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()

    elapsed = time.time() - global_start
    remaining = budget_s - elapsed - 10
    if remaining > 2:
        best_h, best_c5 = optimize_from(best_h, target, n, remaining, seed, rng)

    elapsed = time.time() - global_start
    remaining = budget_s - elapsed - 5
    if remaining > 10:
        for _ in range(min(8, int(remaining / 15))):
            h_pert = best_h + 0.002 * rng.randn(n)
            h_pert = project_feasible(h_pert, target, n)
            c5p = np.max(compute_overlaps(h_pert))
            if c5p < best_c5:
                h2, c5_2 = optimize_from(h_pert, target, n, min(60, remaining / 3), seed + 1, rng)
                if c5_2 < best_c5:
                    best_h = h2
                    best_c5 = c5_2
            remaining = budget_s - (time.time() - global_start) - 5
            if remaining < 5:
                break

    final_c5 = np.max(compute_overlaps(best_h))
    if not np.isfinite(final_c5) or final_c5 > 1.0:
        best_h = np.full(n, 0.5)
        final_c5 = np.max(compute_overlaps(best_h))

    return best_h, final_c5, n
