import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

def ensure_sum(h, n):
    """Ensure sum(h) == n/2 exactly, with h in [0,1]"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        total_cap = (1.0 - h[mask]).sum()
        if total_cap > 0:
            h[mask] += (1.0 - h[mask]) * (diff / total_cap)
    else:
        mask = h > 0.0
        total_cap = h[mask].sum()
        if total_cap > 0:
            h[mask] += h[mask] * (diff / total_cap)
    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff)
