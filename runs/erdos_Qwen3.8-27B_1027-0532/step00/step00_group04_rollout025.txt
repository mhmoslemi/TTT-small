When h[i] changes from old to new, how does corr change?
corr[k] = sum_j h[j] * (1 - h[j+d]) where d = k - (n-1)

The change in corr[k] is:

where B[k] is the same as before! So I can update corr incrementally:
corr_new = corr_old + (new - old) * B

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    current_sum = float(np.sum(h))
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-12:
        return h
    
    if diff > 0:
        room = 1.0 - h
        total_room = float(np.sum(room))
        if total_room < 1e-12:
            return h
        if diff >= total_room:
            h = np.clip(h + diff / n, 0, 1)
        else:
            h = h + room * (diff / total_room)
    else:
        total_room = float(np.sum(h))
        if total_room < 1e-12:
            return h
        if -diff >= total_room:
            h = np.clip(h + diff / n, 0, 1)
        else:
            h = h - h * (-diff / total_room)
    
    return np.clip(h, 0.0, 1.0)

def find_min_convex_max(A, B, lo=0.0, hi=1.0):
    a, b = lo, hi
    for _ in range(50):
        if b - a < 1e-15:
            break
        m = (a + b) / 2.0
        idx = np.argmax(A + B * m)
        if B[idx] < -1e-14:
            a = m
        elif B[idx] > 1e-14:
            b = m
        else:
            a = b = m
            break
    t = (a + b) / 2.0
    g_lo = float(np.max(A + B * lo))
    g_hi = float(np.max(A + B * hi))
    g_t = float(np.max(A + B * t))
    if g_lo <= g_hi and g_lo <= g_t:
        return lo
    elif g_hi <= g_lo and g_hi <= g_t:
        return hi
    else:

    h_new = np.interp(x_new, x_old, h)
    return h_new

import numpy as np
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    n = len(h)
    current_sum = float(np.sum(h))
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = float(np.sum(room))
        if total_room < 1e-12:
            return h
        if diff >= total_room:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h = h + room * (diff / total_room)
    else:
        total_room = float(np.sum(h))
        if total_room < 1e-12:
            return h
        if -diff >= total_room:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h = h - h * (-diff / total_room)
    return np.clip(h, 0.0, 1.0)


def find_min_convex_max(A, B, lo=0.0, hi=1.0):
    a, b = lo, hi
    for _ in range(50):
        if b - a < 1e-15:
            break
        m = (a + b) / 2.0
        idx = np.argmax(A + B * m)
        if B[idx] < -1e-14:
            a = m
        elif B[idx] > 1e-14:
            b = m
        else:
            a = b = m
            break
    t = (a + b) / 2.0
    g_lo = float(np.max(A + B * lo))
    g_hi = float(np.max(A + B * hi))
    g_t = float(np.max(A + B * t))
    if g_lo <= g_hi and g_lo <= g_t:
        return lo
    elif g_hi <= g_lo and g_hi <= g_t:
        return hi
    else:
        return t


def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def coordinate_descent_fast(h, max_passes=1000, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy().astype(float)
    target_sum = n / 2.0
    h = repair_sum(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_obj = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    d_arr = np.arange(-(n - 1), n)
    stall = 0
    for p in range(max_passes):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        for i in range(n):
            old_val = h[i]
            I1 = (i + d_arr >= 0) & (i + d_arr < n)
            I2 = (i - d_arr >= 0) & (i - d_arr < n)
            B = np.zeros(2 * n - 1)
            if np.any(I1):
                B[I1] += (1.0 - h[i + d_arr[I1]])
            if np.any(I2):
                B[I2] -= h[i - d_arr[I2]]
            A = corr - B * old_val
            t_opt = find_min_convex_max(A, B, 0.0, 1.0)
            delta = t_opt - old_val
            if abs(delta) > 1e-15:
                corr = corr + delta * B
                h[i] = t_opt
        old_sum = float(np.sum(h))
        if abs(old_sum - target_sum) > 1e-10:
            h = repair_sum(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
        obj = float(np.max(corr)) * (2.0 / n)
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
    return best_h, best_obj


def make_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h


def make_random(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return repair_sum(h, n / 2.0)


def make_greedy(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for _ in range(n):
        idx = rng.randint(n)
        h[idx] = min(1.0, h[idx] + 0.5)
    return repair_sum(h, n / 2.0)
