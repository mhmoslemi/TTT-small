import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def project_bounded_simplex(h, S):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    result = np.clip(h - tau, 0.0, 1.0)
    # Correct for any small error
    err = S - result.sum()
    if abs(err) > 1e-12:
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        if mask.any():
            result[mask] += err / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_gradient(h, beta):
    """Compute softmax-smoothed gradient of max overlap"""
    n = len(h)
    dx = 2.0 / n
    n_full = 2 * n - 1
    
    r = np.correlate(h, 1 - h, mode='full') * dx
    r_shifted = r - r.max()
    p = np.exp(beta * r_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    
    grad = np.zeros(n)
    for k_idx in range(n_full):
        lag = k_idx - (n - 1)
        pk = p[k_idx]
        if pk < 1e-15:
            continue
        # Term 1
        j1_lo = max(0, lag)
        j1_hi = min(n, n + lag)
        if j1_hi > j1_lo:
            js = np.arange(j1_lo, j1_hi)
            grad[js] += pk * dx * (1.0 - h[js - lag])
        # Term 2
        j2_lo = max(0, -lag)
        j2_hi = min(n, n - lag)
        if j2_hi > j2_lo:
            js = np.arange(j2_lo, j2_hi)
            grad[js] += pk * dx * (-h[js + lag])
    
    return grad

def optimize(n, h_init, time_limit, start_time):
    """Optimize h of length n starting from h_init"""
    dx = 2.0 / n
    S = n / 2.0
    
    h = h_init.copy().astype(float)
    h = project_bounded_simplex(h, S)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    elapsed = time.time() - start_time
    no_improve = 0
    
    while elapsed < time_limit:
        # Gradient descent with adaptive parameters
        alpha = np.random.uniform(0.01, 0.1)
        beta = np.random.choice([3, 5, 10, 20, 50])
        
        for _ in range(500):
            if elapsed >= time_limit:
                break
            grad = compute_gradient(h, beta)
            h_new = h - alpha * grad
            h_new = project_bounded_simplex(h_new, S)
            
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            h = h_new
            alpha *= 0.999
            if alpha < 0.001:
                break
        
        # Occasional restart with noise
        if no_improve > 500 and elapsed < time_limit * 0.9:
            noise_scale = np.random.uniform(0.01, 0.1)
            h = project_bounded_simplex(best_h + noise_scale * np.random.randn(n), S)
            no_improve = 0
        
        elapsed = time.time() - start_time
    
    return best_h, best_c5

Coordinate descent: for each coordinate j, find the optimal value v that minimizes max_k r[k], keeping all other coordinates fixed.

As I analyzed earlier, for fixed other coordinates, r[k] is linear in h[j]:
r[k] = a[k] + v * b[k]

where:
a[k] = r[k] - v_current * b[k] (the part not involving h[j])
b[k] = coefficient of h[j] in r[k]

We want to minimize max_k (a[k] + v * b[k]) over v ∈ [0, 1] (and also maintaining sum = S, but let's ignore that for now and project after).

The minimum of max of linear functions can be found by:
- The optimal v is where the "active" constraints (those achieving the max) intersect
- We can find it by: for each pair (i, j) of constraints, compute the intersection v* = (a[j] - a[i]) / (b[i] - b[j]) if b[i] ≠ b[j], check if v* is feasible and optimal

But with O(n) constraints, checking all pairs is O(n²). For n=45, that's 2025 per coordinate, times 45 coordinates = 91K per full pass. That's fast.

Actually, there's a simpler approach: the optimal v minimizes the maximum of O(n) linear functions. This is a 1D linear program, solvable in O(n log n) by finding the upper envelope. But for n=45, even O(n²) is fine.

    r = np.correlate(h, 1 - h, mode='full') * dx
    
    # Compute coefficient b[k] of h[j] in r[k]
    # b[k] = dx * (1-h[j-lag]) * [valid] - dx * h[j+lag] * [valid]
    b = np.zeros(n_full)
    for k_idx in range(n_full):
        lag = k_idx - (n - 1)
        coeff = 0.0
        if 0 <= j - lag < n:
            coeff += dx * (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            coeff -= dx * h[j + lag]
        b[k_idx] = coeff
    
    # a[k] = r[k] - v * b[k]
    a = r - v * b
    
    # Minimize max_k (a[k] + v' * b[k]) over v' in [0, 1]
    # This is a 1D LP: minimize t subject to a[k] + v'*b[k] <= t for all k, 0<=v'<=1
    # The optimal t is the minimum over v' of max_k (a[k] + v'*b[k])
    
    # Find the minimum of the upper envelope of linear functions
    # Method: check all intersection points and endpoints
    
    candidates = [0.0, 1.0]
    for i in range(n_full):
        for j2 in range(i+1, n_full):
            if abs(b[i] - b[j2]) < 1e-15:
                continue
            v_star = (a[j2] - a[i]) / (b[i] - b[j2])
            if -1e-10 <= v_star <= 1 + 1e-10:
                candidates.append(v_star)
    
    best_t = float('inf')
    best_v = 0.0
    for vc in candidates:
        vc = np.clip(vc, 0.0, 1.0)
        t = np.max(a + vc * b)
        if t < best_t:
            best_t = t
            best_v = vc
    
    # Update h[j]
    h_new = h.copy()
    h_new[j] = best_v
    # Project to maintain sum constraint
    h_new = project_bounded_simplex(h_new, S)
    
    return h_new, best_t

    n = len(h)

def project_bounded_simplex(h, S):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
