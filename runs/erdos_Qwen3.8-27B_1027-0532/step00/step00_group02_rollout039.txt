    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')  # before scaling by dx
    c = corr * dx
    # Softmax weights

    p = np.exp(c_shifted / t)
    p /= np.sum(p)
    
    grad = np.zeros(n)

        if p[k] < 1e-15:

        # First term: (1-h[i-offset]) for valid i-offset
        # i ranges over [0, n-1], i-offset must be in [0, n-1]
        # So i in [max(0, offset), min(n-1, n-1+offset)]

        i_hi = min(n-1, n-1+offset)
        if i_lo <= i_hi:

        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] -= wk * h[j_lo+offset:j_hi+1+offset]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def smooth_max_val(corr, t):
    m = np.max(corr)
    if t < 1e-12:
        return m
    return m + t * np.log(np.sum(np.exp((corr - m) / t)))

def gradient_smooth_max(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    p = np.exp((c - c_max) / t)
    p_sum = np.sum(p)
    if p_sum < 1e-30:
        return np.zeros(n)
    p /= p_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        offset = k - n + 1
        wk = p[k] * dx
        i_lo = max(0, offset)
        i_hi = min(n - 1, n - 1 + offset)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo - offset:i_hi + 1 - offset])
        j_lo = max(0, -offset)
        j_hi = min(n - 1, n - 1 - offset)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] -= wk * h[j_lo + offset:j_hi + 1 + offset]
    return grad
