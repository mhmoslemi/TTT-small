For the second case, let me substitute i = j+k:
When j=-k, i=0; when j=n-1, i=n-1+k
f_k = sum_{i=0}^{n-1+k} h[i] * (1-h[i-k]) * dx

So in both cases: f_k = sum_{i: 0<=i<=n-1, 0<=i-k<=n-1} h[i] * (1-h[i-k]) * dx

The gradient with respect to h[m]:
df_k/dh[m] = dx * [(1-h[m-k]) * I(0<=m-k<=n-1) - h[m+k] * I(0<=m+k<=n-1)]

Wait, that's not right either. Let me be more careful.

f_k = sum_i h[i] * (1-h[i-k]) * dx (over valid i)

So h[m] appears in two ways:
1. As h[i] when i = m: contributes (1-h[m-k]) * dx (if m-k is valid)
2. As h[i-k] when i-k = m, i.e., i = m+k: contributes -h[m+k] * dx (if m+k is valid)

So: df_k/dh[m] = dx * [(1-h[m-k]) * I(0<=m-k<=n-1) - h[m+k] * I(0<=m+k<=n-1)]

This is correct!

Now, for the subgradient of C5 = max_k f_k:
- Find the set of active indices: A = {k : f_k = C5}
- The subgradient is any convex combination of {grad(f_k) : k in A}
- A simple choice: use the gradient of the single k* that achieves the max (with ties broken arbitrarily)

For the local search, I can:
1. Compute C5 and find k*
2. Compute grad(f_{k*})
3. Take a small step in the -gradient direction
4. Project to maintain 0 <= h <= 1 and sum(h) = n/2
5. Repeat

The projection to sum(h) = n/2 after a step that changes the sum is a bit tricky. One approach:
- After taking the gradient step, adjust all elements uniformly to restore the sum
- Then clip to [0,1] and adjust again

Or better: decompose the gradient into a component parallel to the constraint (sum direction) and a component orthogonal to it, and only use the orthogonal component.

The "sum direction" is the vector (1,1,...,1). The projection of gradient g onto the orthogonal complement of (1,...,1) is:
g_proj = g - (sum(g)/n) * (1,1,...,1)

Then take a step in the -g_proj direction, which automatically preserves the sum.

Let me also handle the box constraint (0 <= h <= 1) by clipping after each step.

This gives a subgradient method that should converge (slowly, but it works).

- Switch to a "smoothing" approach when close to the optimum

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # array of length 2n-1

def subgradient_of_overlaps(h, k_idx):
    """Compute gradient of the k-th overlap with respect to h.
    k_idx is the index in the correlate output (0 to 2n-2).
    The actual shift is k = k_idx - (n-1).
    """
    n = len(h)
    dx = 2.0 / n
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        # Term 1: (1-h[m-k]) if 0 <= m-k <= n-1
        mk = m - k
        if 0 <= mk < n:
            g += (1.0 - h[mk])
        # Term 2: -h[m+k] if 0 <= m+k < n
        mk = m + k
        if 0 <= mk < n:
            g -= h[mk]
        grad[m] = g * dx
    return grad

def project_to_constraint(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute uniformly among elements with room
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 0:
                # Only adjust elements that have room
                mask = room > 0
                h[mask] += diff * room[mask] / total_room
            else:
                # All at 1, can't satisfy constraint (shouldn't happen)
                pass
        else:
            room = h.copy()
            total_room = room.sum()
            if total_room > 0:
                mask = room > 0
                h[mask] += diff * room[mask] / total_room
        h = np.clip(h, 0.0, 1.0)
        # Final correction
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            if diff > 0:
                idx = np.argmax(1.0 - h)
            else:
                idx = np.argmin(h)
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, n, max_iter=5000, step_init=0.01):
    """Run subgradient descent on C5."""
    h = h.copy().astype(float)
    current_c5 = evaluate_c5(h)
    step = step_init
    
    for it in range(max_iter):
        overlaps = compute_all_overlaps(h)
        k_idx = int(np.argmax(overlaps))
        current_c5 = overlaps[k_idx]
        
        grad = subgradient_of_overlaps(h, k_idx)
        
        # Remove mean to preserve sum
        grad -= np.mean(grad)
        
        # Normalized step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Take step
        h_new = h - step * grad / grad_norm
        
        # Project
        h_new = project_to_constraint(h_new, n)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            # Slowly decrease step
            step *= 0.999
        else:
            step *= 0.95
    
    return h, current_c5

def _eval_c5(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(corr) * dx)

def _eval_repaired(x, n):
    h = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-13:
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
            else:
                idx = int(np.argmin(h))
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
    return _eval_c5(h)

def _con_eq(x):
    return float(np.sum(x) - _g_target[0])

def _con_eq_jac(x):
    return np.ones(len(x))

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    return h

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def smooth_grad(h, n, alpha):
    corr = compute_overlaps(h)
    max_c = np.max(corr)
    w = np.exp(alpha * (corr - max_c))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(corr)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = i - (n - 1)
        for m in range(n):
            g = 0.0
            mk = m - k
            if 0 <= mk < n:
                g += (1.0 - h[mk])
            mk2 = m + k
            if 0 <= mk2 < n:
                g -= h[mk2]
            grad[m] += wi * g * dx
    grad -= np.mean(grad)
    return grad, float(max_c + np.log(w_sum) / alpha)

def sg_descent(h, n, max_iter, step_init, alpha):
    h = h.astype(float).copy()
    best_local = _eval_c5(h)
    step = step_init

        gnorm = np.linalg.norm(grad)

def try_nm(h, n, options):
