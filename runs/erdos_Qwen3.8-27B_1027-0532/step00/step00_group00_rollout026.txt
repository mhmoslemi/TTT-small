```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-10
        if room[mask].sum() >= diff:
            h[mask] += room[mask] * (diff / room[mask].sum())
    else:

        mask = room > 1e-10
        if room[mask].sum() >= -diff:
            h[mask] -= room[mask] * (-diff / room[mask].sum())

    if abs(diff) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
```

So a good move is: find i such that h[i] is large and h[i+s] is small, then increase h[i+s] and decrease some other h[j].

But we need to be careful not to increase other overlaps.

Let me add this as a "directed" phase:

```python
def directed_move(h, s, rng, n):
    """Try to reduce the overlap at shift s"""
    # Find positions where h[i] is large and h[i+s] is small
    # These contribute most to O_s
    contributions = np.zeros(n)
    if s > 0:
        for i in range(n - s):
            contributions[i] = h[i] * (1 - h[i+s])
        # Best target: increase h[i+s] for the i with largest contribution
        idx = np.argmax(contributions)
        target_idx = idx + s
    elif s < 0:
        s_pos = -s
        for i in range(s_pos, n):
            contributions[i] = h[i-s_pos] * (1 - h[i])
        # Best target: increase h[i] for the i with largest contribution
        # or equivalently, increase h[i] at the position with large h[i-s]
        idx = np.argmax(contributions)
        target_idx = idx  # we want to increase h[idx]
    else:
        return None  # shift 0 is always 0 for {0,1} valued h
    
    # Find a donor: decrease h[j] for some j
    # Best donor: j where h[j] is large but doesn't contribute much to overlaps
    # Simplified: just pick a random j with h[j] > 0.5
    donors = np.where(h > 0.3)[0]
    donors = donors[donors != target_idx]
    if len(donors) == 0:
        return None
    donor_idx = donors[rng.integers(0, len(donors))]
    
    delta = min(h[donor_idx] * 0.3, (1.0 - h[target_idx]) * 0.5)
    if delta < 0.01:
        return None
    
    new_h = h.copy()
    new_h[target_idx] += delta
    new_h[donor_idx] -= delta
    
    if 0 <= new_h[target_idx] <= 1 and 0 <= new_h[donor_idx] <= 1:
        return new_h
    
    return None
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-10
        total = room[mask].sum()
        if total > 1e-10:
            h[mask] += room[mask] * (diff / total)
    else:
        room = h.copy()
        mask = room > 1e-10
        total = room[mask].sum()
        if total > 1e-10:
            h[mask] -= room[mask] * (-diff / total)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def local_search(h_init, n, time_limit, target_sum, rng):
    h = h_init.copy()
    current_c5 = compute_c5(h)
    local_best_h = h.copy()
    local_best_c5 = current_c5
    step_size = 0.08
    no_improve = 0
    directed_interval = 500

    while time.time() < time_limit:
        do_directed = (no_improve > 100 and no_improve % directed_interval < 10)

        if do_directed:
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_k = int(np.argmax(corr))
            s = max_k - (n - 1)
            moved = False
            for attempt in range(5):
                if s > 0:
                    contrib = np.array([h[i] * (1.0 - h[i + s]) for i in range(n - s)])
                    idx = int(np.argmax(contrib))
                    target_idx = idx + s
                elif s < 0:
                    sp = -s
                    contrib = np.array([h[i - sp] * (1.0 - h[i]) for i in range(sp, n)])
                    idx = int(np.argmax(contrib)) + sp
                    target_idx = idx
                else:
                    break
                donors = np.where(h > 0.3)[0]
                donors = donors[donors != target_idx]
                if len(donors) == 0:
                    break
                donor_idx = int(donors[rng.integers(0, len(donors))])
                delta = min(h[donor_idx] * 0.4, (1.0 - h[target_idx]) * 0.6)
                delta = max(delta, 0.02)
                if h[target_idx] + delta > 1.0 or h[donor_idx] - delta < 0.0:
                    delta = min(1.0 - h[target_idx], h[donor_idx])
                    if delta < 0.01:
                        continue
                orig_t = h[target_idx]
                orig_d = h[donor_idx]
                h[target_idx] += delta
                h[donor_idx] -= delta
                new_c5 = compute_c5(h)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    no_improve = 0
                    moved = True
                    if new_c5 < local_best_c5:
                        local_best_c5 = new_c5
                        local_best_h = h.copy()
                else:
                    h[target_idx] = orig_t
                    h[donor_idx] = orig_d
            if moved:
                continue

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        delta = rng.uniform(-step_size, step_size)
        new_hi = h[i] + delta
        new_hj = h[j] - delta

        if 0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0:
            orig_i = h[i]
            orig_j = h[j]
            h[i] = new_hi
            h[j] = new_hj
            new_c5 = compute_c5(h)

            if new_c5 < current_c5:
                current_c5 = new_c5
                no_improve = 0
                if new_c5 < local_best_c5:
                    local_best_c5 = new_c5
                    local_best_h = h.copy()
            else:
                no_improve += 1
                accepted = False
                if no_improve > 500:
                    if rng.random() < 0.15:
                        accepted = True
                        no_improve = 0
                        step_size = max(step_size * 1.2, 0.15)
                elif no_improve > 100:
                    if rng.random() < 0.03:
                        accepted = True
                if not accepted:
                    h[i] = orig_i
                    h[j] = orig_j
                    if no_improve % 500 == 0:
                        step_size *= 0.95
                        if step_size < 0.01:
                            step_size = 0.08
        else:
            continue

    return local_best_h, local_best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)

    def update_best(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if h is not None and c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    if h_init is not None:
        n0 = len(h_init)
        target0 = n0 / 2.0
        h_start = make_feasible(h_init, target0)
        c5_start = compute_c5(h_start)
        update_best(h_start, c5_start, n0)

        phase1_end = start_time + budget_s * 0.45
        h_opt, c5_opt = local_search(h_start, n0, phase1_end, target0, rng)
        update_best(h_opt, c5_opt, n0)

    if time.time() < deadline - 30:
        n_candidates = [36, 40, 44, 48, 52, 56, 60, 64, 72, 80, 90, 100, 120, 144, 160]
        if h_init is not None:
            n_candidates = [x for x in n_candidates if x != len(h_init)]

        remaining = deadline - time.time() - 15
        time_per_n = remaining / max(1, len(n_candidates))

        for n in n_candidates:
            if time.time() > deadline - 10:
                break
            target = n / 2.0
            t_limit = time.time() + time_per_n
            h_rand = rng.uniform(0, 1, n)
            h_rand = make_feasible(h_rand, target)
            c5_rand = compute_c5(h_rand)
            h_opt, c5_opt = local_search(h_rand, n, t_limit, target, rng)
            update_best(h_opt, c5_opt, n)

    if best_h is not None and time.time() < deadline - 5:
        n_best = len(best_h)
        target_best = n_best / 2.0
        h_opt, c5_opt = local_search(best_h, n_best, deadline - 2, target_best, rng)
        update_best(h_opt, c5_opt, n_best)

    if best_h is None:
        n_fb = 45
        best_h = np.full(n_fb, 0.5)
        best_n = n_fb
        best_c5 = compute_c5(best_h)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
```
