    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    overlaps = corr * dx

    shifted = alpha * (overlaps - m)

    sum_exps = np.sum(exps)

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def overlap_gradient(h, k_index):
    N = len(h)
    dx = 2.0 / N
    lag = k_index - (N - 1)
    grad = np.zeros(N)
    if lag == 0:
        grad[:] = (1 - 2 * h)
    elif lag > 0:
        for m in range(N):
            if m >= lag:
                grad[m] += (1 - h[m - lag])
            if m < N - lag:
                grad[m] += (-h[m + lag])
    else:
        l = -lag
        for m in range(N):
            if m < N - l:
                grad[m] += (1 - h[m + l])
            if m >= l:
                grad[m] += (-h[m - l])
    return grad * dx

def smooth_max_and_grad(h, alpha):
    N = len(h)
    overlaps = compute_overlaps(h)
    n_terms = len(overlaps)
    
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    
    obj = m + (max_s + np.log(sum_exps)) / alpha
    
    grad = np.zeros(N)
    for k in range(n_terms):
        wk = w[k]
        if wk < 1e-12:
            continue
        grad += wk * overlap_gradient(h, k)
    
    return obj, grad

def project_feasible(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

def optimize_resolution(h_init, target_sum, time_start, time_budget, n_alphas=8):
    N = len(h_init)
    best_h = h_init.copy()
    best_c5 = float(np.max(compute_overlaps(best_h)))
    
    alphas = [5, 10, 20, 50, 100, 200, 500, 1000][:n_alphas]
    h_current = h_init.copy()
    
    for alpha in alphas:
        if time.time() - time_start > time_budget:
            break
        
        def obj(h):
            val, _ = smooth_max_and_grad(h, alpha)
            return val
        
        def obj_grad(h):
            _, g = smooth_max_and_grad(h, alpha)
            return g
        
        def eq_fun(h):
            return h.sum() - target_sum
        
        def eq_jac(h):
            return np.ones(N)
        
        constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        bounds = [(0.0, 1.0)] * N
        
        try:
            result = minimize(
                obj, h_current, jac=obj_grad,
                method='SLSQP', bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-12}
            )
            h_candidate = np.clip(result.x, 0, 1)
            h_candidate = project_feasible(h_candidate, target_sum)
            c5 = float(np.max(compute_overlaps(h_candidate)))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
            h_current = h_candidate
        except Exception:
            pass
    
    return best_h, best_c5

def projected_subgradient(h_init, target_sum, n_steps=5000, lr_init=0.05):
    N = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = float(np.max(compute_overlaps(h)))
    
    for t in range(n_steps):
        overlaps = compute_overlaps(h)
        c5 = float(np.max(overlaps))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Use gradient of active term (subgradient of max)
        max_k = int(np.argmax(overlaps))
        grad = overlap_gradient(h, max_k)
        
        # Diminishing step size
        lr = lr_init / (1 + t * 0.001)
        h = project_feasible(h - lr * grad, target_sum)
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)

    best_c5 = float(np.max(compute_overlaps(best_h)))
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Optimize at initial resolution using SLSQP
    if time_left() > 120:
        t_budget = min(time_left() - 60, 300)
        h_opt, c5_opt = optimize_resolution(h0, target_sum, time.time(), t_budget)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Try projected subgradient for additional improvement
    if time_left() > 60:
        t_budget = min(time_left() - 30, 60)
        h_sg, c5_sg = projected_subgradient(best_h, target_sum, n_steps=20000, lr_init=0.02)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Phase 3: Try higher resolutions
    for target_n in [90, 180, 360]:
        if time_left() < 150:
            break
        
        # Upsample using linear interpolation
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, target_n)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0, 1)
        target_sum_new = target_n / 2.0
        h_up = project_feasible(h_up, target_sum_new)
        
        # Optimize at new resolution
        t_budget = min(time_left() - 60, 300)
        h_opt, c5_opt = optimize_resolution(h_up, target_sum_new, time.time(), t_budget)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 4: Final projected subgradient polish at best resolution
    if time_left() > 30:
        N_final = len(best_h)
        target_sum_final = N_final / 2.0
        h_sg, c5_sg = projected_subgradient(best_h, target_sum_final, n_steps=50000, lr_init=0.01)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    N_final = len(best_h)
