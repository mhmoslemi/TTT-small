import numpy as np
import time

def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}."""
    h = np.clip(h, 0, 1)
    # Use bisection to find the right threshold
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            # Need to increase sum: add to entries not at 1
            mask = h < 1.0
            if np.sum(mask) == 0:
                break
            add = diff / np.sum(mask)
            add = min(add, 1.0 - h[mask].min())
            h[mask] += add
        else:
            # Need to decrease sum: subtract from entries not at 0
            mask = h > 0.0
            if np.sum(mask) == 0:
                break
            sub = -diff / np.sum(mask)
            sub = min(sub, h[mask].max())
            h[mask] -= sub
    # Final adjustment
    h = np.clip(h, 0, 1)
    # Fix any remaining sum discrepancy
    s = np.sum(h)
    if abs(s - target_sum) > 1e-10:
        # Distribute the difference
        diff = target_sum - s
        mask = (h > 0) & (h < 1)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def subgradient_step(h, target_sum, step_size):
    """Compute subgradient and take a step."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Find active (near-max) components
    threshold = max_val - 0.1
    active = np.where(corr >= threshold)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for m in active:
        lag = m - n + 1
        for j in range(n):
            # df/dh[j] = dx * [(1 - h[j-lag]) - h[j+lag]]
            term = 0.0
            jl = j - lag

            if 0 <= jl < n:
                term += (1.0 - h[jl])
            else:
                term += 1.0  # h[jl] = 0 outside
            if 0 <= jr < n:
                term += (-h[jr])
            else:
                term += 0.0
            grad[j] += term
    grad *= dx
    grad /= max(1, len(active))
    
    # Take step
    h_new = h - step_size * grad
    h_new = project_to_constraint(h_new, target_sum)

def coordinate_descent_step(h, j, target_sum):
    """Optimize coordinate j of h, maintaining the sum constraint approximately."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute all c_k values
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # For each lag k, compute A_k (coefficient of h[j] in c_k)
    num_lags = 2 * n - 1
    A = np.zeros(num_lags)
    B = np.zeros(num_lags)  # constant part (c_k - A_k * h[j])
    
    for m in range(num_lags):
        k = m - (n - 1)
        # c_k = sum_{n: valid} h[n+k] * (1-h[n])
        # Coefficient of h[j]: 
        #   (1-h[j-k]) if 0<=j-k<n (h[j] appears as h[n+k] with n=j-k)
        #   -h[j+k] if 0<=j+k<n (h[j] appears as h[n] with n=j, derivative of (1-h[n]) is -1)
        a_k = 0.0
        if 0 <= j - k < n:
            a_k += (1.0 - h[j - k])
        if 0 <= j + k < n:
            a_k -= h[j + k]
        A[m] = a_k * dx
        B[m] = corr[m] - A[m] * h[j]
    
    # Now minimize max_m (B[m] + A[m] * x) over x in [0, 1]
    # (approximately, ignoring the sum constraint for now)
    
    # Candidate x values: 0, 1, and breakpoints
    candidates = [0.0, 1.0]
    for i in range(num_lags):
        for j2 in range(i+1, num_lags):
            if abs(A[i] - A[j2]) > 1e-12:
                x = (B[j2] - B[i]) / (A[i] - A[j2])
                if 0 <= x <= 1:
                    candidates.append(x)
    
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        val = np.max(B + A * x)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x, best_val

    # Minimize max(a + b*x) over x in [0,1]
    
    # Evaluate at x=0
    vals_0 = a
    mx_0 = np.max(vals_0)
    act_0 = np.where(vals_0 >= mx_0 - 1e-14)[0]
    if np.min(b[act_0]) >= -1e-14:
        return 0.0
    
    # Evaluate at x=1
    vals_1 = a + b
    mx_1 = np.max(vals_1)
    act_1 = np.where(vals_1 >= mx_1 - 1e-14)[0]
    if np.max(b[act_1]) <= 1e-14:
        return 1.0
    
    # Binary search for the minimum
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        mx = np.max(vals)
        act = np.where(vals >= mx - 1e-14)[0]
        b_act = b[act]
        if b_act.min() >= -1e-14 and b_act.max() <= 1e-14:
            break
        elif b_act.min() > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def coordinate_sweep(h, target_sum, order):
    for j in order:
        new_val = optimize_single_coord(h, j)
        h[j] = new_val
    h = project_constraint(h, target_sum)
    return h
