        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        n_trials = min(500, len(ones_idx) * len(zeros_idx))
        for _ in range(n_trials):
            if time.time() - start > max_time:
                break
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stall = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            stall += 1
            if stall > 50:
                break
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
    return best_h, best_c5

def interpolated_h(coarse_h, fine_n):
    n_coarse = len(coarse_h)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False) + 1.0 / n_coarse
    x_fine = np.linspace(0, 2, fine_n, endpoint=False) + 1.0 / fine_n
    return np.interp(x_fine, x_coarse, coarse_h, period=2.0)

def fine_tune(h, target_sum, max_time, lr=0.005):
    n = len(h)
    h = project_h(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < max_time:
        grad = subgradient_max_corr(h)
        h_new = h - lr * grad
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.995
            if lr < 1e-8:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def remaining():
        return max(0, budget_s - (time.time() - start_time))

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 80
        h0 = np.full(n, 0.5)

    target_sum = n / 2.0
    h0 = project_h(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Adam on original grid
    t1 = min(300, budget_s * 0.25)
    h1, c5_1 = adam_optimize(h0, target_sum, t1, lr=0.05, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Binary local search
    t2 = min(300, budget_s * 0.25)
    h2, c5_2 = binary_local_search(best_h, target_sum, t2, seed=seed + 1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: Fine gradient tune
    t3 = min(200, budget_s * 0.15)
    h3, c5_3 = fine_tune(best_h, target_sum, t3, lr=0.005)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3

    # Phase 4: Try finer grid (n*2)
    if remaining() > 120:
        n_fine = n * 2
        target_fine = n_fine / 2.0
        h_fine = interpolated_h(best_h, n_fine)
        h_fine = project_h(h_fine, target_fine)

        t4 = min(150, remaining() * 0.4)
        hf1, cf1 = adam_optimize(h_fine, target_fine, t4, lr=0.03, seed=seed + 10)
        if cf1 < best_c5:
            best_c5 = cf1
            best_h = hf1
            n = n_fine
            target_sum = target_fine

        if remaining() > 60:
            t5 = min(100, remaining() * 0.4)
            hf2, cf2 = binary_local_search(best_h, target_sum, t5, seed=seed + 20)
            if cf2 < best_c5:
                best_c5 = cf2
                best_h = hf2

        if remaining() > 30:
            t6 = min(50, remaining() * 0.5)
            hf3, cf3 = fine_tune(best_h, target_sum, t6, lr=0.003)
            if cf3 < best_c5:
                best_c5 = cf3
                best_h = hf3

    # Phase 5: More Adam restarts on best grid
    if remaining() > 60:
        for restart in range(5):
            if remaining() < 30:
                break
            h_restart = best_h + np.random.randn(len(best_h)) * 0.05
            h_restart = project_h(h_restart, target_sum)
            t7 = min(40, remaining() * 0.4)
            hr, cr = adam_optimize(h_restart, target_sum, t7, lr=0.03, seed=seed + 100 + restart)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr

    # Final: ensure validity
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
