        # term1: 1-h[p-m] valid when 0 <= p-m < n → m in [p-(n-1), p] intersect [-(n-1), n-1]
        # term2: h[p+m] valid when 0 <= p+m < n → m in [-p, n-1-p] intersect [-(n-1), n-1]

            dlo = max(-hp, hq - 1.0)
            dhi = min(1.0 - hp, hq)
            if dhi - dlo < 1e-10:
                continue
            
            npts = 8
            deltas = np.linspace(dlo, dhi, npts)
            new_maxes = (C + D[None, :] * deltas[:, None]).max(axis=1)
            idx = new_maxes.argmin()
            imp = Cmax - new_maxes[idx]
            if imp > best_imp:
                best_imp = imp
                best_p = p
                best_q = q
                best_d = deltas[idx]
    
    return best_p, best_q, best_d, best_imp, Cmax
