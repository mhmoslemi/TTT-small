def project_onto_feasible(x, target_sum):
    """Project x onto {y: sum(y)=target_sum, 0<=y_i<=1} via bisection on lambda."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    """Compute C5 = max(corr) * dx"""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    j_vals = i_vals + d
    grad[i_vals] += (1.0 - h[j_vals]) * dx
    grad[j_vals] -= h[i_vals] * dx
    return grad

def project_onto_feasible(x, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(60):

        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
