def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return dx * np.max(corr)

    """Create random binary vector with exactly n//2 ones"""
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, ones, replace=False)

    """Upscale a binary parent to n_target using nearest neighbor"""
    n = len(parent)
    # Map each target index to source index
    ratios = np.arange(n_target) * n / n_target
    source_indices = np.clip(ratios.astype(int), 0, n - 1)

        current[j] = 1

                current[j] = 0

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))


def make_random_binary(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        indices = rng.choice(n, n // 2, replace=False)
        h[indices] = 1.0
    else:
        indices = rng.choice(n, (n - 1) // 2, replace=False)
        h[indices] = 1.0
        remaining = np.where(h == 0)[0]
        h[remaining[rng.randint(len(remaining))]] = 0.5
    return h


def local_search(h_init, budget_s, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    start = time.time()
    current = h_init.astype(float).copy()
    current_val = compute_c5(current)
    best = current.copy()
    best_val = current_val

    temp = 0.0015
    cooling = 0.99995
    min_temp = 1e-9
    stagnation_limit = 3000
    stagnation_count = 0

    while time.time() - start < budget_s:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or current[i] == current[j]:
            continue

        current[i], current[j] = current[j], current[i]
        new_val = compute_c5(current)

        if new_val <= current_val:
            current_val = new_val
            stagnation_count = 0
            if new_val < best_val:
                best = current.copy()
                best_val = new_val
        else:
            delta = new_val - current_val
            if temp > min_temp and rng.random() < np.exp(-delta / temp):
                current_val = new_val
            else:
                current[i], current[j] = current[j], current[i]

        stagnation_count += 1
        temp = max(temp * cooling, min_temp)

        if stagnation_count > stagnation_limit:
            temp = 0.0015
            stagnation_count = 0

    return best, best_val


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = make_random_binary(45, rng)
    n_parent = len(parent)

    elapsed = time.time() - start_time
    if budget_s - elapsed < 50:
        return parent, compute_c5(parent), n_parent

    # Phase 1: Local search at parent resolution
    phase1_budget = min(450, int((budget_s - elapsed) * 0.35))
    if phase1_budget > 30:
        h1, c5_1 = local_search(parent, phase1_budget, seed=seed)
        if c5_1 < best_c5:
            best_h, best_c5, best_n = h1, c5_1, n_parent
    else:
        best_h, best_c5, best_n = parent, compute_c5(parent), n_parent

    elapsed = time.time() - start_time
    if budget_s - elapsed < 100:
        return best_h, best_c5, best_n

    # Phase 2: Upscale to 2x and search
    h2_init = np.repeat(best_h, 2)
    n2 = len(h2_init)
    phase2_budget = min(350, int((budget_s - elapsed) * 0.3))
    if phase2_budget > 50:
        h2, c5_2 = local_search(h2_init, phase2_budget, seed=seed + 1001)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    elapsed = time.time() - start_time
    if budget_s - elapsed < 80:
        return best_h, best_c5, best_n

    # Phase 3: Upscale parent to 3x and search
    h3_init = np.repeat(parent, 3)
    n3 = len(h3_init)
    phase3_budget = min(250, int((budget_s - elapsed) * 0.25))
    if phase3_budget > 40:
        h3, c5_3 = local_search(h3_init, phase3_budget, seed=seed + 2001)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n3

    elapsed = time.time() - start_time
    if budget_s - elapsed < 60:
        return best_h, best_c5, best_n

    # Phase 4: Random restarts at parent resolution
    max_restarts = 15
    for r in range(max_restarts):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 30:
            break
        h_rand = make_random_binary(n_parent, rng)
        r_budget = min(80, int(budget_s - elapsed - 10))
        if r_budget < 20:

        h_r, c5_r = local_search(h_rand, r_budget, seed=seed + 5000 + r * 13)

        if c5_polish < best_c5:
            best_h, best_c5 = h_polish, c5_polish

    return best_h, best_c5, best_n
