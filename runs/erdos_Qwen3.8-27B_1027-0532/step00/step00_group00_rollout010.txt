```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n, rng):
    """Convert h to a binary sequence of length n with exactly n/2 ones."""
    # Threshold
    h_bin = (h > 0.5).astype(float)
    num_ones = int(np.sum(h_bin))
    target = n // 2
    
    if num_ones > target:
        excess = num_ones - target
        positions = np.where(h_bin == 1)[0]
        rng.shuffle(positions)
        for k in range(excess):
            h_bin[positions[k]] = 0
    elif num_ones < target:
        deficit = target - num_ones
        positions = np.where(h_bin == 0)[0]
        rng.shuffle(positions)
        for k in range(deficit):
            h_bin[positions[k]] = 1
    
    return h_bin

def local_search_binary(h, rng, time_limit, start_time):
    """Simulated annealing with swap moves."""
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Track 1-positions and 0-positions
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    T = 0.05
    T_min = 1e-7
    cooling_rate = 0.9995
    
    current_c5 = best_c5
    h_curr = h.copy()
    
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        # Pick random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        
        # Swap
        h_curr[i], h_curr[j] = h_curr[j], h_curr[i]
        new_c5 = compute_c5(h_curr)
        
        # Accept/reject
        delta = new_c5 - current_c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # Accept
            current_c5 = new_c5
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_curr.copy()
        else:
            # Revert
            h_curr[i], h_curr[j] = h_curr[j], h_curr[i]
        
        T = max(T_min, T * cooling_rate)
        iteration += 1
        
        # Restart if stuck
        if iteration % 50000 == 0 and T <= T_min * 10:
            T = 0.05
    
    return best_h, best_c5
```

Let me also consider: what's the theoretical lower bound on C5?

For a binary sequence of length n with n/2 ones:
- Sum over all shifts k (from -(n-1) to n-1) of corr[k] = sum_k sum_i h[i]*(1-h[i+k])
  = sum_i h[i] * sum_k (1-h[i+k])
  = sum_i h[i] * (2*sum_k 1 - sum_k h[i+k])
  
  Hmm, this is getting complicated. Let me think differently.

  sum_k corr[k] = sum_k sum_i h[i+k]*(1-h[i])
  = sum_i (1-h[i]) * sum_k h[i+k]
  = sum_i (1-h[i]) * sum_j h[j]  (since as k ranges over all valid values, i+k ranges over all j)
  = (sum_j h[j]) * (sum_i (1-h[i]))
  = (n/2) * (n/2)
  = n^2/4

So the average of corr[k] over all 2n-1 shifts is n^2/4 / (2n-1) ≈ n/8.

But we want to minimize the MAX, and the max must be at least the average. So C5 ≥ (2/n) * (n/8) = 1/4.

Wait, that's the average over ALL shifts including k=0. At k=0, corr[0] = sum h[i]*(1-h[i]) = 0 for binary. So the average over the remaining 2n-2 shifts is n^2/4 / (2n-2) ≈ n/8.

So C5 ≥ (2/n) * (n/8) = 1/4 = 0.25.

But this is a very loose bound. The actual minimum is probably higher.

For the best known result C5 ≤ 0.38085, we're at about 0.38. The theoretical lower bound of 0.25 suggests there's room for improvement.

Actually, I realize the bound might not be 0.25 because the correlations are not independent and the boundary effects matter. But it gives us a sense that 0.38 is not optimal.

Let me also think about the structure. For an alternating sequence 1,0,1,0,...:
- At shift 1: corr = sum h[i+1]*(1-h[i]) = n/2 (every other pair is (0,1) or (1,0))
  Wait: h = 1,0,1,0,... 
  h[i+1]*(1-h[i]): when h[i]=1, h[i+1]=0, so product = 0*(1-1) = 0. When h[i]=0, h[i+1]=1, product = 1*1 = 1.
  So corr[1] = n/2 (approximately).
  C5 contribution: (2/n) * (n/2) = 1. That's terrible!

So alternating is bad. We need a more "random-looking" sequence.

What about a sequence with long runs of 1s and 0s? Like 1111000011110000...
- At shift 1: within runs, h[i+1]*(1-h[i]) = 0 (since both are same). At boundaries: one transition 1→0 gives 0*1=0, one transition 0→1 gives 1*0=0. So corr[1] = 0!
- At shift 4: h[i+4]*(1-h[i]). In the pattern 11110000, shift 4 gives 00001111. So h[i+4]*(1-h[i]) = 1 when h[i]=0 and h[i+4]=1, which happens at the 0→1 boundary. There are 2 such positions per period of 8. So corr[4] ≈ n/4.
  C5 contribution: (2/n) * (n/4) = 0.5. Still bad.

So structured sequences with periodicity are bad because they create high correlations at the period.

The best sequences are probably "quasi-random" - they look random locally but have controlled global correlations. This is exactly what good pseudorandom sequences (like m-sequences or Barker sequences) aim for.

For our purposes, simulated annealing from a random start should be able to find good sequences. The key is having enough time to explore the space.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n, rng):
    h = np.array(h, dtype=float).reshape(-1)
    if len(h) != n:
        x_old = np.linspace(0, 2, len(h), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, h)
    h = np.clip(h, 0, 1)
    h_bin = (h > 0.5).astype(np.float64)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        positions = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(positions))
        for k in range(num_ones - target):
            h_bin[positions[perm[k]]] = 0.0
    elif num_ones < target:
        positions = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(positions))
        for k in range(target - num_ones):
            h_bin[positions[perm[k]]] = 1.0
    return h_bin

def local_search_binary(h, rng, duration, T_init=0.04, alpha=0.08):

    best_h = h.copy()
    h_curr = h.copy()

    zeros_arr = np.where(h == 0)[0].copy()

    if n_ones == 0 or n_zeros == 0:
        return best_h, best_c5

        if delta <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            ones_arr[io] = j

_arr[iz] = i
            since_improve += 1
            if new_c5 < best_c5:
                best_c5 = new_c5

                plateau += 1
        else:

def targeted_local_search(h, rng, duration):
    end_time = time.time() + duration
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h_curr = h.copy()

    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    current_c5 = best_c5

    while time.time() < end_time:
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        worst_idx = np.argmax(corr)
        shift = worst_idx - (n - 1)

        if shift == 0:
            io = rng.randint(n_ones)
            iz = rng.randint(n_zeros)
            i = ones_arr[io]
            j = zeros_arr[iz]
        else:
            candidates = []
            lo = max(0, -shift)
            hi = min(n, n - shift)
            for i in range(lo, hi):
                ip = i + shift
                if 0 <= ip < n:
                    if h_curr[i] == 0.0 and h_curr[ip] == 1.0:
                        candidates.append(i)
                    elif h_curr[i] == 1.0 and h_curr[ip] == 0.0:
                        candidates.append(ip)
            if len(candidates) == 0:
                io = rng.randint(n_ones)
                iz = rng.randint(n_zeros)
                i = ones_arr[io]
                j = zeros_arr[iz]
            else:
                i = candidates[rng.randint(len(candidates))]
                if h_curr[i] == 1.0:
                    j_idx = np.where(zeros_arr == i)[0]
                    if len(j_idx) == 0:
                        iz = rng.randint(n_zeros)
                        j = zeros_arr[iz]
                    else:
                        j = i
                        iz = j_idx[0]
                    io = np.where(ones_arr == i)[0][0]
                else:
                    io = np.where(ones_arr == i)[0][0] if len(np.where(ones_arr == i)[0]) > 0 else rng.randint(n_ones)
                    iz = np.where(zeros_arr == i)[0][0] if len(np.where(zeros_arr == i)[0]) > 0 else rng.randint(n_zeros)
                    j = i

        h_curr[i] = 1.0 - h_curr[i]
        h_curr[j] = 1.0 - h_curr[j]

        new_c5 = compute_c5(h_curr)
        if new_c5 <= current_c5:
            current_c5 = new_c5
            ones_arr = np.where(h_curr == 1)[0].copy()
            zeros_arr = np.where(h_curr == 0)[0].copy()
            n_ones = len(ones_arr)
            n_zeros = len(zeros_arr)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_curr.copy()
        else:
            h_curr[i] = 1.0 - h_curr[i]
            h_curr[j] = 1.0 - h_curr[j]

    return best_h, best_c5
```
