import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def compute_gradient(h, k_star, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    # Term 1
    m_min1 = max(0, n - 1 - k_star)
    m_max1 = min(n - 1, 2 * n - 2 - k_star)
    if m_min1 <= m_max1:
        j_start = m_min1 + k_star - (n - 1)
        j_end = m_max1 + k_star - (n - 1)
        grad[m_min1:m_max1 + 1] += (1 - h[j_start:j_end + 1])
    # Term 2
    m_min2 = max(0, k_star - (n - 1))
    m_max2 = min(n - 1, k_star)
    if m_min2 <= m_max2:
        j_start = m_min2 - k_star + (n - 1)
        j_end = m_max2 - k_star + (n - 1)
        grad[m_min2:m_max2 + 1] -= h[j_start:j_end + 1]
    return dx * grad

def project_feasible(v, n):
    target = n / 2.0
    # Find lambda such that sum(clamp(v - lambda, 0, 1)) = target
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(v - lam, 0, 1)
    # Fine-tune to get exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Adjust elements that are strictly in (0,1)
        free_mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(free_mask):
            n_free = np.sum(free_mask)
            h[free_mask] += diff / n_free
            h = np.clip(h, 0, 1)
    return h

def interpolate_to_grid(parent, n_new):
    n_old = len(parent)
    x_old = np.linspace(0.5 / n_old, 2 - 0.5 / n_old, n_old)
    x_new = np.linspace(0.5 / n_new, 2 - 0.5 / n_new, n_new)
    f = interp1d(x_old, parent, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    return np.clip(h_new, 0, 1)

def subgradient_optimize(h_init, n, time_limit):
    h = h_init.copy()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    start_time = time.time()
    T = 0
    h_sum = h.copy()
    
    while time.time() - start_time < time_limit:
        T += 1
        c = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(c)
        grad = compute_gradient(h, k_star, n)
        
        # Subgradient step
        step = 1.0 / np.sqrt(T)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        h_sum += h_new
        h = h_new
        
        c5_val = np.max(c) * (2.0 / n)
        avg_h = h_sum / T
        avg_c5 = compute_c5(avg_h, n)
        if avg_c5 < best_c5:
            best_c5 = avg_c5
            best_h = avg_h.copy()
    
    return best_h, best_c5

def slsqp_refine(h_init, n, time_limit):
    start_time = time.time()
    
    def objective(h):
        if time.time() - start_time > time_limit:
            return 1e10
        c = np.correlate(h, 1 - h, mode='full')
        # Use log-sum-exp for smoothness
        tau = 0.05

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_gradient_at_k(h, k, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    # Term 1: h[m] appears as first factor
    m_min1 = max(0, n - 1 - k)
    m_max1 = min(n - 1, 2 * n - 2 - k)
    if m_min1 <= m_max1:
        j_start = m_min1 + k - (n - 1)
        j_end = m_max1 + k - (n - 1)
        grad[m_min1:m_max1 + 1] += (1.0 - h[j_start:j_end + 1])
    # Term 2: h[m] appears as second factor (inside 1-h)
    m_min2 = max(0, k - (n - 1))
    m_max2 = min(n - 1, k)
    if m_min2 <= m_max2:
        j_start = m_min2 - k + (n - 1)
        j_end = m_max2 - k + (n - 1)
        grad[m_min2:m_max2 + 1] -= h[j_start:j_end + 1]
    return dx * grad

def project_feasible(v, n):
    target = n / 2.0
    # Binary search for lambda
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # Fix rounding to get exact sum
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-12:
        # Add/subtract to elements with room
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_mask = int(np.sum(mask))
        if n_mask > 0:
            h[mask] += diff / n_mask
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_grid(parent, n_new):
    n_old = len(parent)
    if n_old == n_new:
        return parent.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, parent)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

    best_c5 = compute_c5(h, n)

    T = 0
    h_avg = h.copy()
    
    while time.time() - start_time < time_limit:
        T += 1
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        f_val = float(c[k_star]) * dx
        grad = compute_gradient_at_k(h, k_star, n)

        if g_norm_sq < 1e-20:
            break

        h_new = h - step * grad

        # Running average

        h = h_new

        if c5_cur < best_c5:

    if n_old == n_new:
        h = parent.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
        x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
        h = np.interp(x_new, x_old, parent)
        h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n_new)

def subgrad_solve(h0, n, t_limit, rng):
    h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    h_sum = np.zeros(n)
    t0 = time.time()
    T = 0
    while time.time() - t0 < t_limit:
        T += 1
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        f_h = float(c[k_star]) * (2.0 / n)
        g = compute_gradient_at_k(h, k_star, n)
        g2 = float(np.dot(g, g))
        if g2 < 1e-20:
            break
        step = f_h / g2
        step = min(step, 0.5)
        if T % 5000 == 0:
            perturb = 0.01 * (1.0 - min(T / 50000.0, 1.0))
            noise = rng.standard_normal(n) * perturb
            h = project_feasible(h + noise, n)
        h = project_feasible(h - step * g, n)
        h_sum += h
        if T % 200 == 0 or T < 20:
            h_avg = h_sum / T
            c5a = compute_c5(h_avg, n)
            if c5a < best_c5:
                best_c5 = c5a
                best_h = h_avg.copy()
            c5c = compute_c5(h, n)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h.copy()
    return best_h, best_c5

def slsqp_solve(h0, n, t_limit):
    t0 = time.time()
    dx = 2.0 / n
    tau = 0.02

    def obj(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        cs = c / tau
        cm = np.max(cs)
        return dx * tau * (cm + np.log(np.sum(np.exp(cs - cm))))

    def obj_grad(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        cs = c / tau
        cm = np.max(cs)
        w = np.exp(cs - cm)
        ws = np.sum(w)
        w = w / ws
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-5:
                g += w[k] * compute_gradient_at_k(h, k, n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        if time.time() - t0 > t_limit:
            return h0, compute_c5(h0, n)
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': 300, 'ftol': 1e-10})
        if res.x is not None:
            res.x = project_feasible(res.x, n)
            c5r = compute_c5(res.x, n)
            c5_init = compute_c5(h0, n)
            if c5r < c5_init:
                return res.x, c5r
    except Exception:
        pass
    return h0, compute_c5(h0, n)

def local_search(h0, n, t_limit, rng):
    h = h0.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    t0 = time.time()
    scale = 0.1
    while time.time() - t0 < t_limit:
        scale = max(0.001, scale * 0.999)
        i = rng.randint(n)
        delta = rng.uniform(-scale, scale)
        h_tmp = h.copy()
        h_tmp[i] = np.clip(h_tmp[i] + delta, 0.0, 1.0)
        h_tmp = project_feasible(h_tmp, n)
        c5_tmp = compute_c5(h_tmp, n)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()
        h = h_tmp
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(70, 0.5)
    n_parent = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent, n_parent)
    best_n = n_parent

    candidate_grids = []
    for g in [70, 84, 98, 112, 126, 140, 168, 196, 224]:
        if g >= n_parent and g % 2 == 0 and g <= 250:
            candidate_grids.append(g)
    if not candidate_grids:
        candidate_grids = [max(n_parent, 70)]
    candidate_grids = sorted(set(candidate_grids))

    reserve = 60
    elapsed = time.time() - t_start
    avail = max(10, budget_s - elapsed - reserve)

    wts = np.array([float(g) ** 1.3 for g in candidate_grids])
    wts /= wts.sum()

    cur_parent = parent.copy()
    for idx, n in enumerate(candidate_grids):
        elapsed = time.time() - t_start
        if elapsed > budget_s - reserve:
            break
        t_alloc = avail * wts[idx]
        t_alloc = max(5.0, min(t_alloc, budget_s - elapsed - 10))
        h_init = interp_grid(cur_parent, n)
        h_opt, c5_opt = subgrad_solve(h_init, n, t_alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        cur_parent = h_opt.copy()

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 15:
        t_slsqp = min(remaining * 0.5, 100.0)
        h_fin, c5_fin = slsqp_solve(best_h, best_n, t_slsqp)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 10:
        t_ls = min(remaining - 3.0, 60.0)
        h_ls, c5_ls = local_search(best_h, best_n, t_ls, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 8:
        h_sg2, c5_sg2 = subgrad_solve(best_h, best_n, remaining - 2.0, rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
