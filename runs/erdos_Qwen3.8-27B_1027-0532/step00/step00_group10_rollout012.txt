def compute_swap_effects(h, p, q):
    n = len(h)
    d = np.arange(-(n-1), n)  # lags
    # F(p, d) for all d
    pd = p + d  # positions p+d
    qd = q + d  # positions q+d
    pm_d = p - d  # positions p-d
    qm_d = q - d  # positions q-d
    
    Fp = np.zeros(2*n-1)
    Fq = np.zeros(2*n-1)
    
    mask_pd = (pd >= 0) & (pd < n)
    mask_qd = (qd >= 0) & (qd < n)
    mask_pm = (pm_d >= 0) & (pm_d < n)
    mask_qm = (qm_d >= 0) & (qm_d < n)
    
    Fp[mask_pd] += (1 - h[pd[mask_pd]])
    Fp[mask_pm] += h[pm_d[mask_pm]]
    
    Fq[mask_qd] += (1 - h[qd[mask_qd]])
    Fq[mask_qm] += h[qm_d[mask_qm]]
    
    delta = (h[q] - h[p]) * (Fp - Fq)
    return delta

def compute_F_matrix(h):
    """Compute F[p, k] where k = d + (n-1), d is the lag.
    F[p, k] = (1-h[p+d])*[valid] + h[p-d]*[valid]"""
    n = len(h)
    F = np.zeros((n, 2*n-1))
    for k in range(2*n-1):

        for p in range(n):
            if 0 <= p+d < n:
                F[p, k] += (1 - h[p+d])
            if 0 <= p-d < n:
                F[p, k] += h[p-d]
    return F

```python
def compute_F_matrix(h):
    n = len(h)
    d = np.arange(-(n-1), n)  # shape (2n-1,)
    # p ranges over [0, n-1], shape (n,)
    p = np.arange(n)  # shape (n,)
    # p[..., None] + d[None, :] gives (n, 2n-1) array of p+d
    pd = p[:, None] + d[None, :]  # (n, 2n-1)
    pm = p[:, None] - d[None, :]  # (n, 2n-1)
    
    mask_pd = (pd >= 0) & (pd < n)  # (n, 2n-1)
    mask_pm = (pm >= 0) & (pm < n)  # (n, 2n-1)
    
    # (1-h[p+d]) where valid
    # We need to index h with pd, but only where mask_pd is True
    # Use np.take with clip
    pd_clipped = np.clip(pd, 0, n-1)
    pm_clipped = np.clip(pm, 0, n-1)
    
    term1 = (1 - h[pd_clipped]) * mask_pd  # (n, 2n-1)
    term2 = h[pm_clipped] * mask_pm  # (n, 2n-1)
    
    F = term1 + term2  # (n, 2n-1)
    return F
```

- Binary variables h[i] (for even n)
- Minimize t
- Subject to: for each lag d, sum_i h[i]*(1-h[i+d]) ≤ t/dx

The constraint is non-linear (quadratic). To linearize:
Let w[i,d] = h[i]*h[i+d] (for valid i,d)
Then h[i]*(1-h[i+d]) = h[i] - w[i,d]

Constraints:
- w[i,d] ≤ h[i]
- w[i,d] ≤ h[i+d]
- w[i,d] ≥ h[i] + h[i+d] - 1
- w[i,d] ≥ 0

And the objective constraint:
sum_i (h[i] - w[i,d]) ≤ t/dx for each d

def compute_F(h):
    n = len(h)
    d = np.arange(-(n-1), n)
    p = np.arange(n)
    pd = p[:, None] + d[None, :]
    pm = p[:, None] - d[None, :]
    mask_pd = (pd >= 0) & (pd < n)
    mask_pm = (pm >= 0) & (pm < n)
    pd_clipped = np.clip(pd, 0, n-1)
    pm_clipped = np.clip(pm, 0, n-1)
    term1 = (1 - h[pd_clipped]) * mask_pd
    term2 = h[pm_clipped] * mask_pm
    return term1 + term2

def greedy_step(h, corr, F, dx):
