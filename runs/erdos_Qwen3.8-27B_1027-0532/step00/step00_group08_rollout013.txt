    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        it += 1
        if new_c5 < best:
            best_h = h_try
            best = new_c5
        else:
            delta_c = new_c5 - best
            if delta_c < T:
                if np.random.random() < np.exp(-delta_c / max(T, 1e-12)):
                    best_h = h_try
        T = max(1e-8, T * 0.99995)
    return best_h, best

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    try:
        n = len(initial_h_values)
        best_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        n = 80
        best_h = np.zeros(n)
        best_h[::2] = 1.0

    target_sum = n / 2.0
    best_h = project_constraint(best_h, target_sum)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Softmax gradient descent
    time_left = budget_s - (time.time() - start_time)
    phase1_budget = time_left * 0.5
    best_h, best_c5 = local_search(best_h, n, target_sum, phase1_budget, start_time, best_c5)

    # Phase 2: Fine-grained transfer search
    time_left = budget_s - (time.time() - start_time)
    phase2_budget = time_left * 0.3
    n_iters = int(phase2_budget * 30000)
    best_h, best_c5 = transfer_search(best_h, n, target_sum, phase2_budget, start_time, best_c5, n_iters)

    # Phase 3: Try a few random restarts
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30:
        n_restarts = min(int(time_left / 30), 5)
        for _ in range(n_restarts):
            if time.time() - start_time > budget_s - 15:
                break
            h_rand = np.zeros(n)
            h_rand[np.random.choice(n, n // 2, replace=False)] = 1.0
            h_rand = project_constraint(h_rand, target_sum)
            h_rand_c5 = compute_c5(h_rand, n)
            if h_rand_c5 < best_c5:
                best_h = h_rand
                best_c5 = h_rand_c5
            t_alloc = min(20, budget_s - (time.time() - start_time) - 10)
            if t_alloc > 5:
                h_new, c5_new = local_search(h_rand, n, target_sum, t_alloc, time.time(), h_rand_c5)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Phase 4: Final polish with high beta
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10:
        dx = 2.0 / n
        step_size = 0.1
        for _ in range(int(time_left * 5000)):
            if time.time() - start_time > budget_s - 5:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            max_idx = np.argmax(corr)
            s = max_idx - (n - 1)
            g = dx * grad_shift(best_h, s, n)
            g -= np.mean(g)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break
            h_try = best_h - step_size * g / g_norm
            h_try = project_constraint(h_try, target_sum)
            new_c5 = compute_c5(h_try, n)
            if new_c5 < best_c5:
                best_h = h_try
                best_c5 = new_c5
            else:
                step_size *= 0.8
                if step_size < 0.001:
                    break

    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
