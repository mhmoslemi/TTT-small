    return float(m + np.log(np.sum(np.exp(scaled - m))) / beta)

def sum_eq(h, target):
    return float(np.sum(h) - target)

def project_feasible(h, target, n):
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

    target = n / 2.0
    h0 = np.array(initial_h_values, dtype=float).copy()

    for beta in [50, 100, 200]:

            cand = project_feasible(res.x, target, n)
            s = compute_score(cand)
            if s < best_score:

                best_score = s

    res = minimize(compute_score, best_h, method='SLSQP',

    if res.x is not None and np.isfinite(res.fun):

        s = compute_score(cand)
        if s < best_score:
            best_h = cand.copy()
            best_score = s
    
    # Phase 3: Random restarts
    rng = np.random.RandomState(seed)
    
    while time.time() - start_time < budget_s - 30:

        if r < 0.4:
            h_start = best_h + rng.normal(0, 0.05, n)
        elif r < 0.6:

                best_score = s
