    exp_c = np.exp(beta * (c - max_c))
    log_sum_exp = max_c + np.log(np.sum(exp_c)) / beta
    # Softmax weights
    p = exp_c / np.sum(exp_c)
    # Gradient of c_k with respect to h

        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

            h_pgd, c5_pgd = pgd_optimize(best_h, best_n, 200.0, 30000, 0.001, best_n / 2.0)
            if c5_pgd < best_c5:
                best_h, best_c5 = h_pgd, c5_pgd
                h_cand, c5_cand = slsqp_optimize(best_h, best_n, 10000, 1e-14)
                if c5_cand < best_c5:
                    best_h, best_c5 = h_cand, c5_cand

    # Final validation
    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
