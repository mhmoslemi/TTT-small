    """Project v onto {h: 0<=h[i]<=1, sum(h)=target}"""
    n = len(v)
    # Binary search for mu
    lo, hi = -1.0, 2.0
    
    def f(mu):
        return np.sum(np.clip(v - mu, 0, 1)) - target
    
    # If f(lo) < 0, we need to increase all, but they're capped at 1
    # If f(hi) > 0, we need to decrease all, but they're floored at 0

        else:
            hi = mid

    dx = 2.0 / n

    corr_scaled = corr * dx

    exp_vals = np.exp(beta * (corr_scaled - m))
    log_sum = np.log(np.sum(exp_vals)) / beta + m
    return log_sum

def soft_max_c5_gradient(h, n_points, beta=50):

        if weights[j] > 1e-10:
            s = j - (n - 1)
            grad += weights[j] * gradient_for_shift(h, s)

    return grad

def gradient_for_shift(h, s):
    n = len(h)
    grad = np.ones(n)
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if hi1 > lo1:
        grad[lo1:hi1] -= h[lo1+s:hi1+s]
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2-s:hi2-s]
    return grad

def project_onto_feasible(v, target):

        return np.sum(np.clip(v - mu, 0, 1)) - target

        return np.ones(n)
    if f(hi) > 0:  # even with mu very positive, sum > target (impossible if target >= 0)

        else:

import numpy as np
import time

def gradient_for_shift(h, s):
    n = len(h)
    grad = np.ones(n)
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if hi1 > lo1:
        grad[lo1:hi1] -= h[lo1 + s:hi1 + s]
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - s:hi2 - s]
    return grad

def project_onto_feasible(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_val = np.sum(np.clip(v - mid, 0, 1))
        if s_val > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2
    return np.clip(v - mu, 0, 1)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def c5_and_softgrad(h, n_points, beta):
    n = n_points
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c5 = np.max(corr)
    m = c5
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(len(corr)):
        w = weights[j]
        if w > 1e-14:
            s = j - (n - 1)
            grad += w * gradient_for_shift(h, s)
    grad *= dx
    return c5, grad

    def softgrad_optimize(h_start, beta, lr, max_iter, mom_gamma=0.0):
