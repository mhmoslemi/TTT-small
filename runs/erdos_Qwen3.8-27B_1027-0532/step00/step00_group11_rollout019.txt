```python
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum=43.0):

    # Use bisection to find the right threshold
    # h_proj[i] = clip(h[i] - tau, 0, 1) and sum = target_sum

        s = np.sum(h_proj)
        if s > target_sum:

        else:
            hi = mid
```

```python
    """Smooth approximation of C5 using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
```

Wait, let me verify:

- Third term: for j <= n-1-k, subtract h[j+k]. So grad[0] -= h[k], grad[1] -= h[k+1], ..., grad[n-1-k] -= h[n-1].
  In numpy: grad[:n-k] -= h[k:] ✓

Let j = n-m: c_{-m} = sum_{j=0}^{n-1-m} h[j] * (1 - h[j+m])
= sum_{j=0}^{n-1-m} h[j] - sum_{j=0}^{n-1-m} h[j] * h[j+m]

Now, for the subgradient method: we want to MINIMIZE c5 = max_k c_k * dx.

The subgradient of max_k c_k at a point where the max is uniquely achieved at k* is just the gradient of c_{k*}.

So the subgradient method for minimizing c5 is:
1. Find k* = argmax(corr)
2. Compute grad = grad_c_lag(h, k*)
3. Step: h = h - alpha * grad
4. Project: h = project_feasible(h)

This should work! Let me implement the full algorithm.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_c_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        k = lag
        if k < n:
            grad[k:] += 1.0
            if k > 0:
                grad[k:] -= h[:n-k]
            grad[:n-k] -= h[k:]
    else:
        m = -lag
        if m < n:
            grad[:n-m] += 1.0
            grad[:n-m] -= h[m:]
            if m > 0:
                grad[m:] -= h[:n-m]
    return grad

def project_feasible(h, target_sum=43.0):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def subgradient_descent(h_init, time_limit=30.0):
    n = len(h_init)
    h = project_feasible(h_init.astype(float))
    best_h = h.copy()
    best_val = compute_c5(h)
    
    start = time.time()
    alpha = 0.5
    stall = 0
    
    while time.time() - start < time_limit:
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        
        # Compute subgradient
        grad = grad_c_lag(h, lag)
        
        # Try several step sizes (line search)
        best_alpha = 0.0
        best_local_val = compute_c5(h)
        
        for mult in [0.01, 0.03, 0.1, 0.3, 0.5, 1.0, 1.5, 2.0]:
            a = alpha * mult
            h_try = project_feasible(h - a * grad)
            val = compute_c5(h_try)
            if val < best_local_val:
                best_local_val = val
                best_alpha = a
        
        if best_alpha > 0:
            h = project_feasible(h - best_alpha * grad)
            stall = 0
        else:
            alpha *= 0.7
            stall += 1
            if stall > 200:
                break
        
        if compute_c5(h) < best_val:
            best_val = compute_c5(h)
            best_h = h.copy()
    
    return best_h, best_val
```

```python
def coordinate_descent_polish(h, time_limit=10.0, step=0.01):
    n = len(h)
    h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            # Try increasing h[i] and decreasing h[i+1] (to maintain sum)
            # Or try changing h[i] and compensating in a random other coordinate
            for delta in [step, -step]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                # Need to compensate: change another coordinate
                # For simplicity, change the last coordinate
                j = (i + 1) % n
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                h[i] += delta
                h[j] -= delta
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved:
            step *= 0.8
            if step < 1e-4:
                break
    
    return h, best_val
```

```python
def smooth_c5_obj(h, temperature=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    T = 1.0 / temperature
    max_c = np.max(corr)
    # log-sum-exp for numerical stability
    log_sum = max_c + np.log(np.sum(np.exp(T * (corr - max_c)))) / T
    return log_sum * dx
```

```python
def subgradient_topk(h_init, time_limit=30.0, k=3):
    """Subgradient method using the top-k worst lags"""
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.astype(float))
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    
    start = time.time()
    alpha = 0.2
    stall = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        # Find top-k worst lags
        sorted_idx = np.argsort(corr)[::-1][:k]
        
        # Weighted gradient (weights proportional to how close each is to the max)
        grads = np.zeros(n)
        total_weight = 0.0
        for idx in sorted_idx:
            lag = int(idx - (n - 1))
            # Weight: how "close" this lag is to the worst
            weight = corr[idx]  # or use softmax weights
            g = grad_c_lag(h, lag)
            grads += weight * g
            total_weight += weight
        
        if total_weight > 0:
            grads /= total_weight
        
        # Line search
        best_a = 0.0
        best_mv = max_val
        for mult in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 1.5]:
            a = alpha * mult
            h_try = project_feasible(h - a * grads)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            mv = np.max(corr_try)
            if mv < best_mv:
                best_mv = mv
                best_a = a
        
        if best_a > 0:
            h = project_feasible(h - best_a * grads)
            stall = 0
        else:
            alpha *= 0.8
            stall += 1
            if stall > 200:
                break
        
        if best_mv * dx < best_val:
            best_val = best_mv * dx
            best_h = h.copy()
    
    return best_h, best_val
```

```python
    dx = 2.0 / n
    h = project_feasible(h_init.astype(float))
    best_h = h.copy()

    step_size = 0.05
    patience = 0
    
    while time.time() - start < time_limit:
        # Random perturbation
        n_perturb = np.random.randint(1, 5)
        indices = np.random.choice(n, n_perturb, replace=False)
        h_try = h.copy()
        for idx in indices:
            h_try[idx] += step_size * np.random.uniform(-1, 1)
        h_try = project_feasible(h_try)
        
        val = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx)
        
        if val < best_val:
            best_val = val
            best_h = h_try.copy()
            h = h_try.copy()
            patience = 0
        else:
            h = h_try.copy()  # Accept anyway (simulated annealing style)
            patience += 1
        
        if patience > 100:
            step_size *= 0.9
            patience = 0
    
    return best_h, best_val
```
