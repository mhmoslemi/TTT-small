        h_start = h_best + rng.normal(0, 0.12, n)
        h_start = repair(h_start, n)
        t_sub_end = min(t_phase2_end, time.time() + t_per_start)
        h_s, c5_s = sa_optimize(h_start, n, rng, t_sub_end, T_init=0.02)
        if c5_s < c5_best:
            h_best, c5_best = h_s, c5_s
        
        h_t2, c5_t2 = try_thresholds(h_best, n)
        if c5_t2 < c5_best:
            h_best, c5_best = h_t2, c5_t2
    
    t_final_end = t0 + budget_s - 5.0
    h_final, c5_final = sa_optimize(h_best, n, rng, t_final_end, T_init=0.005)
    if c5_final < c5_best:
        h_best, c5_best = h_final, c5_final
    
    h_thresh2, c5_thresh2 = try_thresholds(h_best, n)
    if c5_thresh2 < c5_best:
        h_best, c5_best = h_thresh2, c5_thresh2
    
    h_best = repair(h_best, n)
    c5_best = evaluate(h_best)
    return (h_best, c5_best, n)
