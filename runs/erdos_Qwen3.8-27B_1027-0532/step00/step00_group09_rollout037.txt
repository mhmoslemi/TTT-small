def obj_smooth(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    return m + float(np.log(np.sum(np.exp(shifted))) / beta)

def obj_smooth_grad(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        # dc[k]/dh[t] = (1-h[t-k+n-1]) - h[t+k-n+1] (with bounds)
        # Vectorized:
        # First term: for each t, 1-h[t-k+n-1] if 0<=t-k+n-1<n
        s = k - n + 1  # shift
        # t such that 0 <= t-s < n: max(0,s) <= t <= min(n-1, n-1+s)
        t_start = max(0, s)
        t_end = min(n, n + s)
        if t_start < t_end:
            grad[t_start:t_end] += w * (1.0 - h[t_start - s:t_end - s])
        # Second term: for each t, -h[t+s] if 0<=t+s<n
        t_start2 = max(0, -s)
        t_end2 = min(n, n - s)
        if t_start2 < t_end2:
            grad[t_start2:t_end2] += w * (-h[t_start2 + s:t_end2 + s])
    
    return grad

def constraint_fun(x, target):
    return np.sum(x) - target

def constraint_jac(x, target):
    return np.ones_like(x)

def normalize_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def obj_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))


def obj_smooth(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    shifted = beta * (corr - m)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return m + log_sum / beta


def obj_smooth_grad(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        s = k - n + 1
        # First term: (1 - h[t - s]) for valid t
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - s:hi1 - s])
        # Second term: -h[t + s] for valid t
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] += wk * (-h[lo2 + s:hi2 + s])
    return grad

    return float(np.sum(x) - target)

    return h0
