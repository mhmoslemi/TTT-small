def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(conv)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add diff total
        # Add to smallest values first (they have most room)
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            room = 1.0 - h[idx]
            add = min(room, remaining)
            h[idx] += add
            remaining -= add

The objective max_k g_k(h) where g_k(h) = sum_i h[i](1-h[i+k]) is a max of quadratic functions. Each g_k is:

def project_feasible(h, target_sum):
