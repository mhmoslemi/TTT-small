Actually, note that corr(h, 1-h) at lag k = corr(h, 1) at lag k - corr(h, h) at lag k = (sum of h over the overlapping region) - autocorr(h, h) at lag k.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_c5(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = alpha * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / alpha * dx)


def fix_sum(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        count = int(np.sum(mask))
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h


def sum_constraint(h, target_sum):
    return np.sum(h) - target_sum


def sum_constraint_jac(h, target_sum):
    return np.ones(len(h))


def try_slsqp(h_start, dx, target_sum, bounds, cons, alpha, maxiter):
    res = minimize(smooth_max_c5, h_start, args=(dx, alpha),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and res.x.shape == h_start.shape:
        h_out = fix_sum(res.x, target_sum)
        return h_out
    return None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    n_points = 86
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(n_points) * 0.5

    h0 = fix_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
             'args': (target_sum,)}]

    alphas_seq = [8, 15, 25, 40, 60, 90, 130, 180]
    for alpha in alphas_seq:
        if time.time() - t0 > budget_s - 30:
            break
        h_new = try_slsqp(best_h, dx, target_sum, bounds, cons, alpha, 250)
        if h_new is not None:
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    n_restarts = 80
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 15:
            break
        frac = i / max(1, n_restarts)
        scale = 0.02 + 0.15 * (1.0 - frac)
        h_trial = best_h + scale * np.random.randn(n_points)
        h_trial = fix_sum(h_trial, target_sum)
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

        alpha = 20 + int(60 * ((i * 7) % 5) / 4.0)
        h_new = try_slsqp(h_trial, dx, target_sum, bounds, cons, alpha, 100)
        if h_new is not None:
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                if time.time() - t0 < budget_s - 15:
                    h_new2 = try_slsqp(best_h, dx, target_sum, bounds, cons, 80, 120)
                    if h_new2 is not None:
                        c5_new2 = compute_c5(h_new2, dx)
                        if c5_new2 < best_c5:
                            best_h = h_new2
                            best_c5 = c5_new2

    if time.time() - t0 < budget_s - 120:
        n_sub_pts = n_points
        h_sub = best_h.copy()
        lr = 0.004
        best_lr = lr
        patience = 0

        for step in range(3000):
            if time.time() - t0 > budget_s - 20:
                break

            corr = np.correlate(h_sub, 1.0 - h_sub, mode="full")
            alpha_sg = 60
            scaled = alpha_sg * corr
            m = np.max(scaled)
            weights = np.exp(scaled - m)
            wsum = np.sum(weights)
            if wsum > 0:
                weights /= wsum

            grad = np.zeros(n_sub_pts)
            for k in range(len(corr)):
                if weights[k] < 1e-12:
                    continue
                w = weights[k]
                n_lo = max(0, -k)
                n_hi = min(n_sub_pts - 1, n_sub_pts - 1 - k)
                for n in range(n_lo, n_hi + 1):
                    j = n + k
                    if 0 <= j < n_sub_pts:
                        grad[j] += w * (1.0 - h_sub[n])
                        grad[n] -= w * h_sub[j]

            h_sub = h_sub - lr * grad
            h_sub = fix_sum(h_sub, target_sum)
            c5_sub = compute_c5(h_sub, dx)
            if c5_sub < best_c5:
                best_h = h_sub.copy()
                best_c5 = c5_sub
                patience = 0
                best_lr = lr
            else:
                patience += 1
                if patience > 200:
                    lr = best_lr * 0.7
                    h_sub = best_h + 0.01 * np.random.randn(n_sub_pts)
                    h_sub = fix_sum(h_sub, target_sum)
                    patience = 0

            lr = max(0.00005, lr * 0.9995)

    if time.time() - t0 < budget_s - 100:
        for n_hi in [172, 344]:
            if time.time() - t0 > budget_s - 50:
                break
            dx_hi = 2.0 / n_hi
            target_hi = n_hi / 2.0
            x_old = np.linspace(0.0, 2.0, n_points, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_hi, endpoint=False)
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = fix_sum(h_hi, target_hi)
            c5_hi = compute_c5(h_hi, dx_hi)

            bounds_hi = [(0.0, 1.0)] * n_hi
            cons_hi = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
                        'args': (target_hi,)}]

            if c5_hi < best_c5:
                best_h = h_hi.copy()
                best_c5 = c5_hi
                n_points = n_hi
                dx = dx_hi
                target_sum = target_hi

            if time.time() - t0 > budget_s - 40:
                break
            for alpha_hi in [30, 50, 80]:
                if time.time() - t0 > budget_s - 20:
                    break
                h_new = try_slsqp(h_hi, dx_hi, target_hi, bounds_hi, cons_hi, alpha_hi, 80)
                if h_new is not None:
                    c5_new = compute_c5(h_new, dx_hi)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        n_points = n_hi
                        dx = dx_hi
                        target_sum = target_hi

    best_h = fix_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n_points
