    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h, n):
    """Compute subgradient of max_k correlate(h, 1-h, 'full')[k]"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = corr.argmax()
    lag = idx - (n - 1)
    
    g = np.zeros(n)
    # For each j: g[j] = (1-h[j-lag])*[0<=j-lag<n] - h[j+lag]*[0<=j+lag<n]
    if lag >= 0:
        # j-lag >= 0 means j >= lag
        # j+lag < n means j < n-lag
        valid1 = np.arange(lag, n)  # j where j-lag is valid
        valid2 = np.arange(0, n - lag)  # j where j+lag is valid
        for j in valid1:
            g[j] += (1.0 - h[j - lag])
        for j in valid2:
            g[j] -= h[j + lag]
    else:
        # lag < 0, let lag_neg = -lag > 0
        lag_neg = -lag
        # j-lag = j+lag_neg >= 0 always (since j>=0, lag_neg>0)
        # j-lag < n means j < n - lag_neg
        # j+lag = j-lag_neg >= 0 means j >= lag_neg
        # j+lag < n always (since j<n, lag_neg>0)
        valid1 = np.arange(0, n - lag_neg)  # j where j-lag is valid
        valid2 = np.arange(lag_neg, n)  # j where j+lag is valid
        for j in valid1:
            g[j] += (1.0 - h[j + lag_neg])
        for j in valid2:
            g[j] -= h[j - lag_neg]

1. Find the worst lag k*
2. Try to reduce f_{k*}(h) by moving mass

To reduce f_{k*}(h) = sum_i h[i]*(1-h[i+k*]):
- Decrease h[i] where h[i] is high and h[i+k*] is low (contributes a lot to the sum)
- Increase h[i] where h[i] is low and h[i+k*] is high (reduces the contribution)

import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-10:
        mask = (projected > 1e-8) & (projected < 1.0 - 1e-8)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def compute_grad_at_lag(h, lag, n):
    """Compute gradient of f_lag(h) = sum_k h[k]*(1-h[k+lag]) w.r.t. h"""
    g = np.zeros(n)
    if lag >= 0:
        for i in range(n):
            val = 0.0
            if i - lag >= 0:
                val += (1.0 - h[i - lag])
            if i + lag < n:
                val -= h[i + lag]
            g[i] = val
    else:
        l = -lag
        for i in range(n):
            val = 0.0
            if i + l < n:
                val += (1.0 - h[i + l])
            if i - l >= 0:
                val -= h[i - l]
            g[i] = val
    return g
