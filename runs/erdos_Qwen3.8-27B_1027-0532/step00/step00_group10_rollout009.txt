```python
def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def subgradient_c5(h, n_points):
    """Returns (c5_value, subgradient) of the C5 objective."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode="full")
    k_star = np.argmax(corr)
    val = corr[k_star] * dx
    
    # Compute subgradient of corr[k_star] w.r.t. h
    n = n_points
    d = k_star - (n - 1)
    grad = np.zeros(n)
    
    if d >= 0:
        # corr = sum_{q=0}^{n-1-d} h[q+d]*(1-h[q])
        for q in range(n - d):
            grad[q+d] += (1 - h[q])
            grad[q] -= h[q+d]
    else:
        ad = -d
        # corr = sum_{q=0}^{n-1-ad} h[q]*(1-h[q+ad])
        for q in range(n - ad):
            grad[q] += (1 - h[q+ad])
            grad[q+ad] -= h[q]
    
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    
    if diff > 0:
        # Need to increase sum
        room = 1.0 - h
        if room.sum() < diff:
            return h  # infeasible
        # Add proportionally to room
        h += room * (diff / room.sum()) if room.sum() > 0 else 0
    else:
        # Need to decrease sum
        if h.sum() < -diff:
            return h  # infeasible
        h += h * (diff / h.sum()) if h.sum() > 0 else 0
    
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute to elements with room
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-10)[0]
            if len(idx) > 0:
                add = diff / len(idx)
                h[idx] += add
                h = np.clip(h, 0, 1)

                h[idx] -= sub
```

```python
def subgradient_topk(h, k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top-k indices
    top_idx = np.argsort(-corr)[:k]
    weights = corr[top_idx]
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)
        g = np.zeros(n)
        if d >= 0:
            for i in range(n - d):
                g[i + d] += (1.0 - h[i])
                g[i] -= h[i + d]
        else:
            ad = -d
            for i in range(n - ad):
                g[i] += (1.0 - h[i + ad])
                g[i + ad] -= h[i]
        grad += weights[top_idx.tolist().index(idx)] * g
    
    return grad * dx, float(np.max(corr) * dx)
```
