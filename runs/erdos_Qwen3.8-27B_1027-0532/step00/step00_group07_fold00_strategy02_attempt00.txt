def calculate_erdos_objective(h, n_points):
    """Compute the Erdős minimum overlap integral C_5 for a given h_vector.
    Returns: C_5 value (scalar)"""
    # Normalize vector and compute correlations for all shifts
    dx = 2.0 / n_points
    # Form the vector 1-h
    h_vector = np.array(h, dtype=float)
    g = 1 - h_vector
    
    # Use direct correlation calculation
    corr_val = np.correlate(h_vector, g, mode="full")
    max_corr_val = np.max(corr_val)
    c5 = max_corr_val * dx
    return c5

        # Merge new candidates
        current_candidates.extend(new_candidates[:n_points])
        elapsed = time.time() - mode_start_time
        
    total_time = time.time() - start_time
    
    # Check elapsed time
    if total_time > budget_s:
        # Return best found so far
        best_fitness = min([calculate_erdos_objective(c, n_points) for c in current_candidates]) if current_candidates else 1e8
        best_solution = [c for c in current_candidates if calculate_erdos_objective(c, n_points) == best_fitness][0]
    else:
        # Within time budget, prepare the fallback parent solution
        best_fitness = calculate_erdos_objective(initial_candidate, n_points)
        best_solution = initial_candidate
    
    # Ensure results meet constraints
    if not is_feasible(best_solution):
        adjusted, _ = repair_solution(best_solution)
        best_solution = adjusted
        best_fitness = calculate_erdos_objective(adjusted, n_points)
    
    return best_solution[:n_points], best_fitness, n_points
