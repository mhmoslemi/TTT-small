                        best_c5 = c5_test
                        improved = True
                        break
            if not improved:
                break

    # Final polish with SLSQP
    if remaining() > 15:
        h_cand = try_slsqp(best_h, target, n, 5000, maxiter=5000, ftol=1e-16)
        if h_cand is not None:
            c5 = compute_c5(h_cand, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()

    # Final check
    best_c5 = compute_c5(best_h, dx)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
