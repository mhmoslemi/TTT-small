def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_thue_morse_binary(N, rng):
    """Create a binary vector from Thue-Morse sequence, with exactly N/2 ones."""
    # Generate Thue-Morse
    tm = np.zeros(N, dtype=np.uint8)
    for i in range(N):
        # Count bits
        x = i
        count = 0
        while x:
            count += x & 1
            x >>= 1
        tm[i] = count & 1
    
    # We need N/2 ones. Thue-Morse has roughly N/2 ones.
    n_ones = int(np.sum(tm))
    target = N // 2
    if n_ones > target:
        ones = np.where(tm == 1)[0]
        to_remove = rng.choice(ones, n_ones - target, replace=False)
        tm[to_remove] = 0
    elif n_ones < target:
        zeros = np.where(tm == 0)[0]
        to_add = rng.choice(zeros, target - n_ones, replace=False)
        tm[to_add] = 1
    return tm.astype(float)
