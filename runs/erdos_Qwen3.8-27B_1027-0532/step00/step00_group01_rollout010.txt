_T = 0.0

def _cf(h):
    return np.sum(h) - _T

def _cj(h):
    return np.ones(len(h))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def _sog(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    sc = v / eps
    mx = np.max(sc)
    lg = sc - mx
    el = np.exp(lg)
    se = np.sum(el)
    obj = eps * (mx + np.log(se))
    w = el / se
    g = np.zeros(n)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                g[sh:] += wk * (1.0 - h[:n-sh])
                g[:n-sh] += wk * (-h[sh:])
        else:
            s = -sh
            if s < n:
                g[:n-s] += wk * (1.0 - h[s:])
                g[s:] += wk * (-h[:n-s])
    return obj, g

    return _sog(h, eps)[0]

            s = r.sum()
            if s < 1e-15:
