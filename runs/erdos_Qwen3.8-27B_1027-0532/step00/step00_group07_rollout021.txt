def lag_gradient(h, lag):

    n = len(h)
    grad = np.zeros(n)

    n_start = max(0, -lag)
    n_end = min(n, n - lag)  # exclusive
    
    for n_i in range(n_start, n_end):
        j1 = n_i + lag  # index in h from first factor
        j2 = n_i        # index in h from second factor
        grad[j1] += (1.0 - h[j2])
        grad[j2] += -h[j1]
    
    return grad

    best_h = h0.copy()

        if elapsed > time_budget:
            break
        
        # Find the lag that achieves the max
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)

        # Compute gradient
        grad = lag_gradient(h, lag)

        h_new = h + v
        h_new = project_feasible(h_new, target_sum)

            lr *= 0.5
        else:
            h = h_new
            lr = min(lr * 1.01, 0.2)
    
    return best_h, best_c5

    # Sort indices by value
    idx = np.argsort(h)
    h_new = np.zeros(n)

The gradient of f_β is:
∂f_β/∂h[i] = (1/β) * sum_k exp(β * corr[k] * dx) * β * dx * ∂corr[k]/∂h[i] / sum_k exp(β * corr[k] * dx)

where softmax_k = exp(β * corr[k] * dx) / sum_j exp(β * corr[j] * dx)

This gives a weighted average of the gradients at all lags, weighted by how close each lag is to the maximum. This is much smoother than the subgradient of the max.

The downside is computing all 2n-1 lags and their gradients, which is O(n^2) per iteration. But n=95, so 2n-1 = 189 lags, and each gradient is O(n), so total is O(n^2) ≈ 9000 per iteration. Same as computing the full correlation. So it's feasible.

def softmax_weighted_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Softmax weights
    shifted = beta * corr * dx
    shifted -= np.max(shifted)  # for numerical stability
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for k_out in range(2 * n - 1):
        w = weights[k_out]
        if w < 1e-10:
            continue
        lag = k_out - (n - 1)
        # Add gradient at this lag, weighted by w
        g = lag_gradient(h, lag)
        grad += w * g
    
    return grad, np.max(corr) * dx

        if l < n:
            j2 = np.arange(l, n)
            j1 = j2 - l
            grad[j1] += (1.0 - h[j2])
            grad[j2] += (-h[j1])
    return grad


def compute_soft_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * corr
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w > 1e-8:
            lag = k - (n - 1)
            grad += w * lag_gradient(h, lag)
    return grad, corr


def opt_pgd(h0, time_budget, beta_init=20.0, beta_max=500.0, lr_init=0.015):
    t_start = time.time()
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    step_count = 0
    stall_count = 0
    while time.time() - t_start < time_budget:
        progress = min(1.0, step_count / 2000.0)
        beta = beta_init + (beta_max - beta_init) * progress
        mom = momentum * (1.0 - 0.3 * progress)
        grad, corr = compute_soft_gradient(h, beta)
        c5_curr = float(np.max(corr)) * (2.0 / n)
        v = mom * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        c5_new = evaluate_c5(h_new)
        if np.isnan(c5_new) or np.isinf(c5_new):
            h_new = h.copy()
            c5_new = c5_curr
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        if c5_new > c5_curr + 1e-12:
            lr *= 0.5
            v *= 0.5
            if stall_count > 300:
                h = best_h.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5
                stall_count = 0
        else:
            h = h_new
            lr = min(lr * 1.01, 0.08)
        step_count += 1
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_init = len(h_init)
    target_sum = n_init / 2.0
    h_init = project_feasible(h_init, target_sum)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    best_n = n_init

    # Phase 1: PGD on initial solution
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    alloc1 = min(remaining * 0.40, 500.0)
    if alloc1 > 10:
        h_opt, c5_opt = opt_pgd(h_init, alloc1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_init

    # Phase 2: Basin hopping with perturbation
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    alloc2 = min(remaining * 0.25, 300.0)
    if alloc2 > 30:
        n = len(best_h)
        t_phase2 = time.time()
        n_restarts = 8
        for r in range(n_restarts):
            if time.time() - t_start > budget_s - 30:
                break
            elapsed_p2 = time.time() - t_phase2
            t_budget_local = min(35.0, alloc2 - elapsed_p2 - 5.0)
            if t_budget_local < 5:
                break
            scale = 0.04 + 0.08 * r / max(1, n_restarts - 1)
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = project_feasible(h_pert, n / 2.0)
            h_opt, c5_opt = opt_pgd(h_pert, t_budget_local, beta_init=50.0, beta_max=300.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

    # Phase 3: Try different n values
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    alloc3 = min(remaining * 0.20, 250.0)
    if alloc3 > 30:
        n_try_list = [n_init + 2, n_init + 4, n_init + 6, n_init + 10, n_init - 2, n_init + 14]
        for n_try in n_try_list:
            elapsed = time.time() - t_start
            remaining = budget_s - elapsed
            if remaining < 20 or n_try < 20:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_feasible(h_try, n_try / 2.0)
            t_local = min(remaining * 0.15, 60.0)
            if t_local < 5:
                break
            h_opt, c5_opt = opt_pgd(h_try, t_local)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 4: Final polish
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 10:
        h_opt, c5_opt = opt_pgd(best_h, remaining, beta_init=100.0, beta_max=500.0, lr_init=0.008)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
