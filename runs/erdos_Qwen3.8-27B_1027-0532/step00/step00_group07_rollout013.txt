    # Phase 3: Fine local search
    if tl() > 0:
        h = best.copy()
        for it in range(1000000):
            if tl() <= 0:
                break
            n_p = 1 if rng.random() < 0.7 else 2
            h_new = h.copy()
            for _ in range(n_p):
                idx = int(rng.integers(n))
                h_new[idx] += rng.standard_normal() * 0.008
            h_new = project_box_sum(h_new, ts)
            v = c5_of(h_new)
            if v < best_val:
                best = h_new.copy()
                best_val = v
                h = h_new
            elif it % 5000 == 0:
                h = best.copy()
                h += rng.standard_normal(n) * 0.005
                h = project_box_sum(h, ts)

    best = project_box_sum(best, ts)
    best_val = c5_of(best)
    return best, best_val, n
