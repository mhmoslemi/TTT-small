Actually, let me think about this more carefully. The correlation is:
corr[n-1+k] = sum_{i: 0≤i<n, 0≤i+k<n} h[i+k] * (1-h[i])

Let me define the valid range for i given k:
- i from max(0, -k) to min(n-1, n-1-k)

For k ≥ 0: i from 0 to n-1-k, number of terms = n-k
For k < 0: i from -k to n-1, number of terms = n+k = n-|k|

The gradient ∂corr[n-1+k]/∂h[j]:
- h[j] appears in h[i+k] when i = j-k (requires 0 ≤ j-k < n and 0 ≤ j-k+k < n, i.e., 0 ≤ j < n)
  → contributes (1-h[j-k]) if 0 ≤ j-k < n
- h[j] appears in h[i] when i = j (requires 0 ≤ j < n and 0 ≤ j+k < n)
  → contributes -h[j+k] if 0 ≤ j+k < n

So:
∂corr[n-1+k]/∂h[j] = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)

For the full gradient matrix, I can compute this row by row:

Row j: for k = -(n-1) to n-1:
- First term: (1-h[j-k]) if 0 ≤ j-k < n, i.e., if max(0, j-n+1) ≤ k ≤ min(n-1, j)

  Actually: 0 ≤ j-k < n → j-n < k ≤ j → k ∈ {j-n+1, ..., j} intersected with {-(n-1), ..., n-1}
  
- Second term: -h[j+k] if 0 ≤ j+k < n, i.e., -j ≤ k ≤ n-1-j
  k ∈ {-j, ..., n-1-j} intersected with {-(n-1), ..., n-1}

        #   new_corr = corr + delta

For a fixed i, the gradient contribution to all k is:
g_i[k] = (1-h[i-k]) * I(0≤i-k<n) - h[i+k] * I(0≤i+k<n)

I can precompute this for all i as a matrix G of shape (n, 2n-1).

Then for a swap (i,j): delta = G[i] - G[j], new_corr = corr + G[i] - G[j].

The cost of computing G is O(n^2). Then for each of the O(n^2) pairs, computing new_max is O(n). Total per iteration: O(n^2) + O(n^3) = O(n^3).

            h[i] += add
            diff -= add
            if diff <= 1e-10:
                break
    else:
        room = h
        order = np.argsort(room)

            sub = min(-diff, room[i])
            h[i] -= sub
            diff += sub

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        jm = j - k
        if 0 <= jm < n:
            val += 1.0 - h[jm]
        jp = j + k
        if 0 <= jp < n:
            val -= h[jp]
        g[j] = val

def binary_local_search(h_init, n, time_budget=60.0):

    h = h_init.copy()

    # Fix sum to target

        idx = np.random.choice(ones)
        h[idx] = 0
    while h.sum() < target:
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros)
        h[idx] = 1
    
    start_time = time.time()
    patience = 0
    best_score = compute_c5_score(h)
    best_h = h.copy()
    
    while time.time() - start_time < time_budget:
        corr = compute_corr(h)
        current_max = corr.max()
        if current_max < best_score - 1e-10:
            best_score = current_max
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 200:
            break

        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break

        G = np.zeros((n, 2*n-1))
        for j in range(n):
            for k in range(-(n-1), n):

                val = 0.0
                jm = j - k

                    val += 1.0 - h[jm]
                jp = j + k

                    val -= h[jp]

def continuous_refine(h_init, n, time_budget=60.0):
    target = n / 2.0
    h = h_init.copy()
    h = repair(h, target)
    
    dx = 2.0 / n
    best_score = compute_c5_score(h)
    best_h = h.copy()

    lr = 0.01
    top_k = 3
    
    while time.time() - start_time < time_budget:

        # Use top-k average for smoother objective
        sorted_corr = np.sort(corr)[::-1]

        active_k = [idx - (n-1) for idx in np.argsort(corr)[::-1][:top_k]]

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    if s > 0:
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 1e-10:

            sub = min(-diff, room[i])

            diff += sub

import numpy as np
import math
import time


def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    if s > 0:
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 1e-10:
                break
    else:
        room = h
        order = np.argsort(room)
        for i in order:
            sub = min(-diff, room[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-10:
                break
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def gradient_for_k(h, k):
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        jm = j - k
        if 0 <= jm < n:
            val += 1.0 - h[jm]
        jp = j + k
        if 0 <= jp < n:
            val -= h[jp]
        g[j] = val
    return g


def build_gradient_matrix(h):
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    for j in range(n):
        for k in range(-(n - 1), n):
            idx = n - 1 + k
            val = 0.0
            jm = j - k
            if 0 <= jm < n:
                val += 1.0 - h[jm]
            jp = j + k
            if 0 <= jp < n:
                val -= h[jp]
            G[j, idx] = val
    return G


def make_binary(h, n, rng):
    target = n // 2
    h = (h > 0.5).astype(float)
    while h.sum() > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones)] = 0.0
    while h.sum() < target:
        zeros = np.where(h == 0)[0]        h[rng.choice(zeros)] = 1.0
    return h


def binary_local_search(h_init, n, time_budget, seed):
    rng = np.random.RandomState(seed)
    h = make_binary(h_init, n, rng)
    start_time = time.time()
    best_h = h.copy()
    best_score = compute_c5_score(h)
    no_improve = 0
    temp = 0.4

    while time.time() - start_time < time_budget:
        corr = compute_corr(h)
        current_max = float(corr.max())
        if current_max < best_score - 1e-10:
            best_score = current_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        G = build_gradient_matrix(h)
        G_z = G[zeros]
        G_o = G[ones]
        new_max_arr = np.max(corr[None, None, :] + G_z[:, None, :] - G_o[None, :, :], axis=2)
        min_val = float(new_max_arr.min())
        if min_val < current_max - 1e-12:
            mi = np.unravel_index(np.argmin(new_max_arr), new_max_arr.shape)
            h[zeros[mi[0]]] = 1.0
            h[ones[mi[1]]] = 0.0
        else:
            ri = rng.randint(len(zeros))
            rj = rng.randint(len(ones))
            rand_max = float((corr + G_z[ri] - G_o[rj]).max())
            delta = rand_max - current_max
            if delta <= 0 or (temp > 0.005 and rng.random() < math.exp(-delta / temp)):
                h[zeros[ri]] = 1.0
                h[ones[rj]] = 0.0
            else:
                temp *= 0.95
                if temp < 0.005:
                    break
            no_improve += 1
        temp = max(temp * 0.999, 0.005)
    return best_h, best_score


def continuous_refine(h_init, n, time_budget, seed):
    rng = np.random.RandomState(seed)
    target = float(n) / 2.0
    h = repair(h_init.copy(), target)
    best_score = compute_c5_score(h)
    best_h = h.copy()
    start_time = time.time()
    lr = 0.03
    top_k = 3
    stuck_count = 0

    while time.time() - start_time < time_budget:
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:top_k]
        active_ks = [int(idx) - (n - 1) for idx in top_idx]
        g = np.zeros(n)
        for k in active_ks:
            g += gradient_for_k(h, k)
        g /= float(top_k)
        h_new = repair(h - lr * g, target)
        new_score = compute_c5_score(h_new)

            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.2)
            stuck_count = 0

            stuck_count += 1
            if stuck_count > 200:
                top_k = rng.randint(2, 7)
                stuck_count = 0
            if lr < 1e-7:
                break
    return best_h, best_score
