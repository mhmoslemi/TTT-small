def compute_gradient(h, weights, dx):
    n = len(h)
    # weights is array of length 2n-1, indexed by c = -(n-1), ..., n-1
    c = np.arange(-(n-1), n)  # shift values
    # For each j, compute:
    # grad[j] = sum_c weights_c * (1-h[j+c]) * 1_{0<=j+c<n} - sum_c weights_c * h[j-c] * 1_{0<=j-c<n}
    
    # Create index matrix: M[j, k] = c[k] + j for first term, c[k] - j for second
    j = np.arange(n)
    # First term: valid when 0 <= j + c < n
    jc = j[:, None] + c[None, :]  # shape (n, 2n-1)
    valid1 = (jc >= 0) & (jc < n)
    # (1-h[j+c]) where valid
    term1 = np.where(valid1, (1 - h[np.clip(jc, 0, n-1)]), 0) * weights[None, :]
    
    # Second term: valid when 0 <= j - c < n
    jm = j[:, None] - c[None, :]
    valid2 = (jm >= 0) & (jm < n)
    term2 = np.where(valid2, h[np.clip(jm, 0, n-1)], 0) * weights[None, :]
    
    grad = (term1.sum(axis=1) - term2.sum(axis=1)) * dx
    return grad

import numpy as np
import time

def project_to_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = project_to_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    n_pts = n
    c_vals = np.arange(-(n_pts - 1), n_pts)
    j_idx = np.arange(n_pts)
    jc = j_idx[:, None] + c_vals[None, :]
    jm = j_idx[:, None] - c_vals[None, :]
    valid1 = (jc >= 0) & (jc < n_pts)
    valid2 = (jm >= 0) & (jm < n_pts)
    clipped_jc = np.clip(jc, 0, n_pts - 1)
    clipped_jm = np.clip(jm, 0, n_pts - 1)

    phase1_end = start_time + budget_s * 0.35
    beta = 5.0
    lr = 0.04
    v = np.zeros(n_pts)
    momentum = 0.85

    while time.time() < phase1_end:
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        max_ov = np.max(overlaps)
        exp_vals = np.exp(beta * (overlaps - max_ov))
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp

        t1 = np.where(valid2, 1.0 - h[clipped_jm], 0.0)
        t2 = np.where(valid1, h[clipped_jc], 0.0)
        grad = ((t1 - t2) @ w) * dx

        h_new = h - lr * v

        c5 = compute_c5(h, dx)
        if c5 < best_c5:

            best_h = h.copy()

        beta = min(beta * 1.004, 80.0)

            tmp = h_try[i]
