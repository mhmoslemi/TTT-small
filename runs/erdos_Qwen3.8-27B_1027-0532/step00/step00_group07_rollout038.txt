import numpy as np

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _c5_of(h, n):
    return float(np.max(_corr_of(h)) * (2.0 / n))


def _to_binary(h, n, rng):
    h_b = np.where(h >= 0.5, 1.0, 0.0)
    cnt = int(np.sum(h_b))
    target = n // 2
    if cnt > target:
        idx = np.where(h_b == 1)[0]
        rng.shuffle(idx)
        h_b[idx[:cnt - target]] = 0.0
    elif cnt < target:
        idx = np.where(h_b == 0)[0]
        rng.shuffle(idx)
        h_b[idx[:target - cnt]] = 1.0
    return h_b


def _local_search_binary(h, n, rng, max_sweeps, time_limit_fn):
    current = h.astype(float).copy()
    current_corr = _corr_of(current)
    current_max = float(np.max(current_corr))
    dx = 2.0 / n

            swapped = False
