Feasibility check for a given V: does there exist t ∈ [0,1] such that a_d + b_d * t ≤ V for all d?

- And t ≥ (V - a_d) / b_d for all d with b_d < 0

def min_max_linear(a, b, mu=0.0, t_sum_offset=0.0):
    """
    Minimize max_d (a[d] + b[d]*t) + mu*(t + t_sum_offset)^2 over t in [0,1]
    If mu=0, it's just the min-max linear case.
    """
    if mu == 0:
        # Binary search on V
        # f(t) = max_d (a[d] + b[d]*t)
        # f(0) = max(a)
        # f(1) = max(a+b)
        lo = min(np.max(a), np.max(a + b))  # lower bound on min value
        hi = max(np.max(a), np.max(a + b))  # upper bound
        
        # Actually, let me just binary search
        # Check feasibility: exists t in [0,1] with a[d] + b[d]*t <= V for all d
        def feasible(V):
            # For b[d] > 0: t <= (V - a[d]) / b[d]
            # For b[d] < 0: t >= (V - a[d]) / b[d]  (note: dividing by negative flips inequality)
            # For b[d] == 0: need a[d] <= V
            pos = b > 1e-12
            neg = b < -1e-12
            zero = ~(pos | neg)
            
            if np.any(zero) and np.any(a[zero] > V + 1e-12):
                return False
            
            t_lo = 0.0
            t_hi = 1.0
            
            if np.any(pos):
                t_hi = min(t_hi, np.min((V - a[pos]) / b[pos]))
            if np.any(neg):
                t_lo = max(t_lo, np.max((V - a[neg]) / b[neg]))
            
            return t_lo <= t_hi + 1e-12
        
        # Binary search
        for _ in range(60):
            mid = (lo + hi) / 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid
        
        # Now find t at V = hi
        V = hi
        pos = b > 1e-12
        neg = b < -1e-12
        t_lo = max(0.0, np.max((V - a[neg]) / b[neg]) if np.any(neg) else 0.0)
        t_hi = min(1.0, np.min((V - a[pos]) / b[pos]) if np.any(pos) else 1.0)
        t = (t_lo + t_hi) / 2
        t = np.clip(t, 0, 1)
        return t, V * (2.0 / len(a))  # multiply by dx... wait, I'll handle dx outside
    
    else:
        # With penalty: minimize max_d(a[d] + b[d]*t) + mu*(t + t_sum_offset)^2
        # This is convex, use scipy or grid
        from scipy.optimize import minimize_scalar
        def f(t):
            return np.max(a + b * t) + mu * (t + t_sum_offset)**2
        res = minimize_scalar(f, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
        return res.x, res.fun

def repair_sum(h, target_sum):
    """Adjust h to have the target sum while keeping values in [0,1]."""
    h = h.copy()
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to add: distribute to coordinates with room

        total_val = np.sum(h)
        if total_val < 1e-10:
            return h
        h += h * (diff / total_val)
