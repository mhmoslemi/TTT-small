from scipy.optimize import minimize

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2
        p = np.clip(h - mid, 0, 1)
        s = np.sum(p)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target_sum - np.sum(p)
    if abs(diff) > 1e-12:
        mask = (p > 1e-10) & (p < 1 - 1e-10)
        if np.sum(mask) > 0:
            p[mask] += diff / np.sum(mask)
            p = np.clip(p, 0, 1)
    return p

def compute_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def compute_grad(h, k_active):
    n = len(h)
    dx = 2.0 / n
    alpha = n - 1 - k_active
    beta = k_active - n + 1
    grad = np.zeros(n)
    # First term: (1 - h[m + alpha]) for valid m

    grad *= dx
    return grad

    h_new = project_feasible(h_new, target_sum)
    return h_new

            lo = mid
        else:
            hi = mid

                break
            val_s, grad_s = smooth_obj_grad(h_r, tau_r)
            val_hard, _, _ = obj_and_grad(h_r)
            if val_hard < best_val - 1e-10:
                best_val = val_hard
                best_h = h_r.copy()
                stall_r = 0
            else:
                stall_r += 1
            if stall_r > 400:
                tau_r = min(tau_r * 1.4, 150.0)
                if tau_r >= 150.0:
                    lr_r *= 0.5
                stall_r = 0
                if lr_r < 1e-6:
                    break
            h_new_r = project_feasible(h_r - lr_r * grad_s, target_sum)
            new_val_s, _ = smooth_obj_grad(h_new_r, tau_r)
            if new_val_s <= val_s + 1e-8:
                h_r = h_new_r
            else:
                lr_r = max(lr_r * 0.8, 1e-7)
        n_restarts += 1

    # Phase 5: Final SLSQP from best
    if _remaining() > 15:
        try:
            res = minimize(lambda x: obj_and_grad(x)[0], best_h, method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15})
            if res.fun < best_val:
                best_val = res.fun
                best_h = project_feasible(res.x, target_sum)
        except Exception:
            pass

    best_h = project_feasible(best_h, target_sum)
    final_val, _, _ = obj_and_grad(best_h)
    return (best_h, final_val, n)
