            room = 1.0 - h

def get_worst_pair(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(corr))
    offset = s_star - n + 1
    i_start = max(0, -offset)
    i_end = min(n, n - offset) if offset >= 0 else n
    if i_start >= i_end:
        return None, None, 0.0
    i_idx = np.arange(i_start, i_end)
    j_idx = i_idx + offset
    valid = (j_idx >= 0) & (j_idx < n)
    i_idx = i_idx[valid]
    j_idx = j_idx[valid]
    if len(i_idx) == 0:
        return None, None, 0.0
    contribs = h[i_idx] * (1.0 - h[j_idx])
    best_local = int(np.argmax(contribs))
    return int(i_idx[best_local]), int(j_idx[best_local]), float(contribs[best_local])

def sa_optimize(h, rng, max_time=50.0):
    n = len(h)
    h = make_feasible(h.astype(float).copy(), n)
    current_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    T = 0.2
    T_min = 0.003
    start = time.time()
    no_improve = 0
    
    while time.time() - start < max_time:
        r = rng.random()
        
        if r < 0.4:
            i_b, j_b, contrib = get_worst_pair(h, n)
            if i_b is not None and contrib > 0.005:
                delta = min(h[i_b], 1.0 - h[j_b]) * rng.uniform(0.3, 1.0)
                if delta > 0.003:
                    h_new = h.copy()
                    h_new[i_b] -= delta
                    h_new[j_b] += delta
                    h_new = make_feasible(h_new, n)
                    new_c5 = eval_c5(h_new)
                    if new_c5 < current_c5 or (new_c5 - current_c5) < T * abs(rng.standard_normal()):
                        h = h_new
                        current_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                            no_improve = 0
                        continue
            no_improve += 1
            
        elif r < 0.6:
            i, j = rng.choice(n, size=2, replace=False)
            if h[i] <= h[j]:
                i, j = j, i
            gap = h[i] - h[j]
            if gap < 0.01:
                no_improve += 1
                continue
            delta = gap * rng.uniform(0.1, 0.8)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_c5 = eval_c5(h_new)
            if new_c5 < current_c5 or (new_c5 - current_c5) < T * abs(rng.standard_normal()):
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                
        elif r < 0.75:
            blen = rng.randint(2, max(3, n // 4))
            bstart = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[bstart:bstart+blen] = 1.0 - h_new[bstart:bstart+blen]
            h_new = make_feasible(h_new, n)
            new_c5 = eval_c5(h_new)
            if new_c5 < current_c5 or (new_c5 - current_c5) < T * abs(rng.standard_normal()):
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                
        elif r < 0.85:
            # Binary snap
            thresh = 0.5 + rng.normal(0, 0.15)
            h_new = (h > thresh).astype(float)
            h_new = make_feasible(h_new, n)
            new_c5 = eval_c5(h_new)
            if new_c5 < current_c5 or (new_c5 - current_c5) < T * abs(rng.standard_normal()):
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                
        else:
            # Large perturbation
            n_pert = max(3, n // 8)
            idx = rng.choice(n, size=n_pert, replace=False)
            h_new = h.copy()
            h_new[idx] = rng.random(n_pert)
            h_new = make_feasible(h_new, n)
            new_c5 = eval_c5(h_new)
            if new_c5 < current_c5 or (new_c5 - current_c5) < T * abs(rng.standard_normal()):
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Cooling
        if no_improve > 500:
            T = max(T_min, T * 0.8)
            if no_improve > 2000:
                T = min(0.2, T * 5.0)
                no_improve = 0
        T = max(T_min, T * 0.9995)
    
    return best_h, best_c5

def hill_climb(h, rng, max_time=20.0):
    n = len(h)
    h = make_feasible(h.astype(float).copy(), n)
    current_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    
    while time.time() - start < max_time:
        improved = False
        
        # Directed: move from worst contributor
        i_b, j_b, contrib = get_worst_pair(h, n)
        if i_b is not None and contrib > 0.001:
            delta = min(h[i_b], 1.0 - h[j_b])
            if delta > 0.001:
                h_new = h.copy()
                h_new[i_b] -= delta
                h_new[j_b] += delta
                h_new = make_feasible(h_new, n)
                new_c5 = eval_c5(h_new)
                if new_c5 < current_c5 - 1e-14:
                    h = h_new
                    current_c5 = new_c5
                    best_h = h.copy()
                    best_c5 = current_c5
                    improved = True
        
        # Try several random swaps
        if not improved:
            for _ in range(20):
                i, j = rng.choice(n, size=2, replace=False)
                if h[i] <= h[j]:
                    i, j = j, i
                gap = h[i] - h[j]
                if gap < 0.01:
                    continue
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    delta = gap * frac
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    new_c5 = eval_c5(h_new)
                    if new_c5 < current_c5 - 1e-14:
                        h = h_new
                        current_c5 = new_c5
                        best_h = h.copy()
                        best_c5 = current_c5
                        improved = True
                        break
                if improved:
                    break
        
        # Try block flips
        if not improved:
            for _ in range(10):
                blen = rng.randint(2, max(3, n // 4))
                bstart = rng.randint(0, n - blen + 1)
                h_new = h.copy()
                h_new[bstart:bstart+blen] = 1.0 - h_new[bstart:bstart+blen]
                h_new = make_feasible(h_new, n)
                new_c5 = eval_c5(h_new)
                if new_c5 < current_c5 - 1e-14:
                    h = h_new
                    current_c5 = new_c5
                    best_h = h.copy()
                    best_c5 = current_c5
                    improved = True
                    break

                h = h_new
                current_c5 = new_c5
