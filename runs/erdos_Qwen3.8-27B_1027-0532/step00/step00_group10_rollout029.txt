def project_feasible(x, target_sum):
    """Project x onto {h : 0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(x)
    # Find lambda such that sum(clip(x_i - lambda, 0, 1)) = target_sum
    lo, hi = x.min() - 1, x.max() + 1
    
    def sum_clipped(lam):
        return np.clip(x - lam, 0, 1).sum()
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_clipped(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    h = np.clip(x - lam, 0, 1)
    
    # Fix any rounding to ensure exact sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among entries not at bounds
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_gradient_at_lag(lag, h, n):
    """Compute gradient of correlate(h, 1-h, 'full')[n-1+lag] with respect to h"""
    grad = np.zeros(n)
    if lag >= 0:

        m = np.arange(lag, n)
        grad[m] += (1 - h[m - lag])
        # Second term: -h[m+lag] for 0 <= m <= n-1-lag
        m = np.arange(0, n - lag)
        grad[m] -= h[m + lag]

        # Second term: -h[m-lag_abs] for lag_abs <= m <= n-1
        m = np.arange(lag_abs, n)
        grad[m] -= h[m - lag_abs]
    return grad
