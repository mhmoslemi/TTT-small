import numpy as np
from scipy.optimize import minimize as scipy_minimize
import time

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}"""
    # Use bisection on Lagrange multiplier
    lo = h.min() - target_sum / len(h) - 1
    hi = h.max() - target_sum / len(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

        m_end2 = min(N, N + s)  # exclusive
        if m_start2 < m_end2:
            grad[m_start2:m_end2] -= w * dx * h[m_start2 - s : m_end2 - s]

    f = c * dx
    f_max = f.max()
    exp_arg = (f - f_max) / eps
    max_arg = exp_arg.max()
    exp_f = np.exp(exp_arg - max_arg)
    sum_exp = exp_f.sum()
    obj = f_max + eps * (max_arg + np.log(sum_exp))
    weights = exp_f / sum_exp

    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (N - 1)
        m_start = max(0, -s)
        m_end = min(N, N - s)
        if m_start < m_end:
            grad[m_start:m_end] += w * dx * (1.0 - h[m_start + s:m_end + s])
        m_start2 = max(0, s)
        m_end2 = min(N, N + s)
        if m_start2 < m_end2:
            grad[m_start2:m_end2] -= w * dx * h[m_start2 - s:m_end2 - s]

    return obj, grad

def optimize_gradient(h_init, dx, target, deadline, max_iter=10000):
    N = len(h_init)
    h = project_onto_constraint(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    momentum = np.zeros(N)
    beta = 0.9

    eps_schedule = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    iters_per_phase = max_iter // len(eps_schedule)
    
    for phase, eps in enumerate(eps_schedule):
        for it in range(iters_per_phase):
            if time.time() > deadline:
                return best_h, best_c5
            progress = (phase * iters_per_phase + it) / max_iter
            obj, grad = smooth_obj_grad(h, dx, eps)
            momentum = beta * momentum + (1 - beta) * grad
            effective_grad = momentum / (1 - beta ** (it + 1))

            lr = 0.02 * (1.0 - 0.5 * progress)
            improved = False
            for _ in range(12):
                h_new = project_onto_constraint(h - lr * effective_grad, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    break
                lr *= 0.5

            if not improved:
                lr2 = 0.01
                for _ in range(5):
                    h_new = project_onto_constraint(h - lr2 * grad, target)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5 - 1e-14:
                        h = h_new
                        best_c5 = c5_new
                        best_h = h.copy()
                        break
                    lr2 *= 0.3

    return best_h, best_c5

def coordinate_refine(h, dx, target, deadline, max_sweeps=200):
    N = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved_any = False
        for _ in range(N):
            m = _ % N
            current_val = h[m]
            f_sum_without = target - current_val
            lo = max(0.0, f_sum_without - (N / 2.0 - 1.0))
            hi = min(1.0, f_sum_without)
            if lo >= hi:
                continue
            best_x = current_val
            best_f = best_c5
            n_trials = 16
            for trial in range(n_trials):
                x = lo + (hi - lo) * trial / (n_trials - 1) if n_trials > 1 else (lo + hi) / 2
                h[m] = x
                c5_val = compute_c5(h, dx)
                if c5_val < best_f:
                    best_f = c5_val
                    best_x = x
            h[m] = best_x
            if best_f < best_c5:
                best_c5 = best_f
                improved_any = True
        if not improved_any:
            break
    
    return h, best_c5

def interpolate_h(h, n_new):
    N = len(h)
    if N == n_new:
        return h.copy()
    x_old = (np.arange(N) + 0.5) * (2.0 / N)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def binary_snap(h, target):
    N = len(h)
    h_snap = h.copy()
    h_snap[h_snap < 0.3] = 0.0
    h_snap[h_snap > 0.7] = 1.0
    h_snap = project_onto_constraint(h_snap, target)
    return h_snap

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20
    rng = np.random.RandomState(seed)

    h0 = initial_h_values.copy()
    N0 = len(h0)
    dx0 = 2.0 / N0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx0)
    best_N = N0

    # Phase 1: Optimize at original resolution
    if time.time() < deadline:
        h_opt, c5_opt = optimize_gradient(h0, dx0, N0 / 2.0, deadline, max_iter=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_N = N0

    # Phase 2: Coordinate refinement at original resolution
    if time.time() < deadline:
        h_ref, c5_ref = coordinate_refine(best_h, dx0, N0 / 2.0, deadline, max_sweeps=100)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_N = N0

    # Phase 3: Higher resolutions
    resolutions = []
    for factor in [2, 4, 8]:
        N = N0 * factor
        if N <= 720:
            resolutions.append(N)

    for N in resolutions:
        if time.time() > deadline - 30:
            break
        dx = 2.0 / N
        target = N / 2.0
        h_start = project_onto_constraint(interpolate_h(best_h, N), target)

        # Gradient optimization
        if time.time() < deadline - 10:
            h_opt, c5_opt = optimize_gradient(h_start, dx, target, deadline - 10, max_iter=8000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_N = N

        # Coordinate refinement
        if time.time() < deadline - 10:
            h_ref, c5_ref = coordinate_refine(best_h, dx, target, deadline - 10, max_sweeps=50)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
                best_N = N

        # Binary snap refinement
        if time.time() < deadline - 10:
            h_bin = binary_snap(best_h, target)
            h_bin_opt, c5_bin = optimize_gradient(h_bin, dx, target, deadline - 10, max_iter=3000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin_opt
                best_N = N

        # Random restart at this resolution
        if time.time() < deadline - 20:
            h_pert = project_onto_constraint(best_h + rng.uniform(-0.05, 0.05, N), target)
            h_pert_opt, c5_pert = optimize_gradient(h_pert, dx, target, deadline - 10, max_iter=5000)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert_opt
                best_N = N

    return best_h, best_c5, best_N
