def project_onto_box_sum(h, target_sum):
    """Project h onto {x : 0 ≤ x_i ≤ 1, Σ x_i = target_sum}"""
    # Bisection on lambda
    # We want: Σ clip(h[i] - lambda, 0, 1) = target_sum
    lo = h.min() - 1 - target_sum
    hi = h.max() + target_sum

    best_c5 = compute_c5(h1, dx1)

    deadline_a = t0 + budget_s * 0.25
    ha, ca = gradient_descent(h1, dx1, ts1, deadline_a, alpha=15.0, lr=0.003, mu=0.9)
    if ca < best_c5:
        best_c5 = ca
        best_h = ha

    deadline_b = t0 + budget_s * 0.45
    hb, cb = gradient_descent(ha, dx1, ts1, deadline_b, alpha=80.0, lr=0.001, mu=0.85)
    if cb < best_c5:
        best_c5 = cb
        best_h = hb

    deadline_c = t0 + budget_s * 0.55
    hc, cc = gradient_descent(hb, dx1, ts1, deadline_c, alpha=300.0, lr=0.0005, mu=0.8)
    if cc < best_c5:
        best_c5 = cc
        best_h = hc

    if time.time() - t0 < budget_s * 0.88:
        n2 = 193
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h2 = upsample(best_h, 2)
        h2 = project_onto_box_sum(h2, ts2)
        deadline_d = t0 + budget_s * 0.88
        hd, cd = gradient_descent(h2, dx2, ts2, deadline_d, alpha=15.0, lr=0.002, mu=0.9)
        if cd < best_c5:
            best_c5 = cd
            best_h = hd

    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        nf = len(best_h)
        dxf = 2.0 / nf
        tsf = nf / 2.0
        n_restarts = min(int(remaining / 12), 15)
        for i in range(n_restarts):
            if time.time() - t0 > budget_s - 8:
                break
            scale = 0.03 + 0.04 * np.random.random()
            hr_init = best_h + scale * np.random.randn(nf)
            hr_init = project_onto_box_sum(hr_init, tsf)
            alpha_r = 20.0 + 60.0 * np.random.random()
            lr_r = 0.002 + 0.003 * np.random.random()
            hr, cr = gradient_descent(hr_init, dxf, tsf, t0 + budget_s - 8,
                                       alpha=alpha_r, lr=lr_r, mu=0.9, stall_limit=200)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr

    best_h = project_onto_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
