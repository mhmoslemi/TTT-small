                continue
            h_trial[j] = np.clip(h_trial[j], 0.0, 1.0)
        
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            h = h_trial
            best_c5 = c5_trial
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    parent = np.asarray(initial_h_values, dtype=float)
    n_parent = len(parent)
    
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = n_parent
    
    def elapsed():
        return time.time() - start_time
    
    # Phase 1: SLSQP on parent grid
    if budget_s - elapsed() > 150:
        n = n_parent
        alpha_slsqp = 80.0
        
        def slsqp_fun(hv):
            return compute_smoothed_grad(hv, alpha_slsqp)[0]
        
        def slsqp_jac(hv):
            return compute_smoothed_grad(hv, alpha_slsqp)[1]
        
        cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - n / 2.0,
                 'jac': lambda hv: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(slsqp_fun, parent.copy(), slsqp_jac, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            h_test = project_box_sum(res.x, n / 2.0)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
                best_n = n
        except Exception:
            pass
    
    # Phase 2: Projected subgradient on multiple grids
    grids = [n_parent, 140, 200, 280]
    for n_try in grids:
        if budget_s - elapsed() < 100:
            break
        
        if n_try == best_n:
            h0 = best_h.copy()
        else:
            x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
            x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            h0 = np.interp(x_new, x_old, best_h)
            h0 = project_box_sum(h0, n_try / 2.0)
        
        remaining = budget_s - elapsed() - 60
        alloc = min(remaining, 250)
        if alloc < 30:
            break
        
        h_res, c5_res = projected_subgradient(h0, n_try, n_try / 2.0, alloc, rng)
        c5_final = compute_c5(h_res)
        if c5_final < best_c5:
            best_h = h_res.copy()
            best_c5 = c5_final
            best_n = n_try
    
    # Phase 3: Additional SLSQP from best found
    if budget_s - elapsed() > 80:
        n = best_n
        alpha2 = 100.0
        
        def slsqp_fun2(hv):
            return compute_smoothed_grad(hv, alpha2)[0]
        
        def slsqp_jac2(hv):
            return compute_smoothed_grad(hv, alpha2)[1]
        
        cons2 = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - n / 2.0,
                  'jac': lambda hv: np.ones(n)}]
        bnds2 = [(0.0, 1.0)] * n
        try:
            res2 = minimize(slsqp_fun2, best_h.copy(), slsqp_jac2, method='SLSQP',
                           bounds=bnds2, constraints=cons2,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_test = project_box_sum(res2.x, n / 2.0)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
                best_n = n
        except Exception:
            pass
    
    # Phase 4: Coordinate refinement
    if budget_s - elapsed() > 30:
        remaining = budget_s - elapsed() - 10
        h_res, c5_res = coordinate_refine(best_h, best_n, remaining, rng)
        c5_final = compute_c5(h_res)
        if c5_final < best_c5:
            best_h = h_res.copy()
            best_c5 = c5_final
            best_n = best_n
    
    # Final cleanup
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
