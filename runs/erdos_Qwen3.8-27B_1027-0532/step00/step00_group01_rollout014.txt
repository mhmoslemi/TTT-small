def try_swap_fast(h, i, j, R, n_evals=10):
    """Fast swap evaluation using precomputed correlation R.
    R is the current correlation array (2n-1 values, unscaled).
    Returns (dir, t, new_c5_unscaled, A, B) where A,B are the coefficient arrays."""
    n = len(h)
    dx = 2.0 / n
    current_max = np.max(R)
    
    # Compute A[k] for k = -(n-1) to n-1 (indexed 0 to 2n-2)
    # A[m] where m = n-1+k, k = m-(n-1)
    A = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        ipk = i + k
        jpk = j + k
        imk = i - k
        jmk = j - k
        a = 0.0
        if 0 <= jpk < n: a += h[jpk]
        if 0 <= ipk < n: a -= h[ipk]
        if 0 <= imk < n: a -= h[imk]
        if 0 <= jmk < n: a += h[jmk]
        A[m] = a
    
    # B: quadratic coefficient
    B = np.zeros(2*n-1)
    B[n-1] = 2.0  # k=0, index n-1
    if 0 <= j-i <= n-1:  # k = j-i, index n-1+(j-i) = n-1+j-i
        B[n-1+j-i] = -1.0
        # But if j-i == 0, we already set B[n-1] = 2, so it becomes 2-1=1
        # Actually if j==i we wouldn't be here, so j-i != 0
    if 0 <= i-j <= n-1:  # k = i-j, index n-1+(i-j)
        B[n-1+i-j] = -1.0
    
    # Now find optimal t
    best_dir = 0
    best_t = 0.0
    best_R_max = current_max
    new_R = R.copy()
    
    # Direction 1: h[i] += t, h[j] -= t
    t_max = min(h[j], 1.0 - h[i])
    if t_max > 1e-12:

                best_t = t

    # This is equivalent to direction 1 with i and j swapped, and t -> -t
    # Or: A' = -A (flipping the sign), B' = B (same quadratic terms)
    # Actually, direction 2 is: h[i] - t, h[j] + t

        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:
            break
        step = diff / nf
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    best_h = None
    best_c5 = 1.0
    
    # Get initial
    h0 = None
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h0 = np.array(globals()['initial_h_values'], dtype=float)

        best_c5 = c5
        best_h = h0.copy()
    
    # Phase 1: Guided search from initial (35% budget)
    if h0 is not None:
        tl = t0 + budget_s * 0.35
        h1, c5_1 = guided_search(h0.copy(), rng, tl, t0, n_evals=8)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
    
    # Phase 2: Multiple restarts (50% budget)
    tl2 = t0 + budget_s * 0.85
    n_attempts = 0
    while time.time() < tl2:
        n_attempts += 1
        # Choose n
        n_opts = [50, 63, 64, 72, 80, 100, 128, 160, 200]
        n_try = n_opts[rng.randint(len(n_opts))]
        
        # Random init
        h_try = rng.uniform(0.2, 0.8, size=n_try)
        h_try = project_h(h_try, n_try / 2.0, rng)
        
        c5_try = compute_c5(h_try)
        
        # Limited local search
        remaining = tl2 - time.time()
        alloc = max(3.0, min(15.0, remaining / max(1, 20 - n_attempts)))
        tl_try = t0 + (time.time() - t0) + alloc
        
        h_try, c5_try = guided_search(h_try, rng, tl_try, t0, n_evals=5)
        
        if c5_try < best_c5:
            best_c5 = c5_try

        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    
    return best_h, best_c5, len(best_h)
