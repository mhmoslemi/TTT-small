```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_details(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    return np.max(corr) * dx, max_idx

def project_onto_feasible(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return projected

def subgradient_descent(h, dx, n, target_sum, time_limit, start_time, lr_init=0.1):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    t = 0
    while True:
        if time.time() - start_time > time_limit - 5:
            break
        t += 1
        
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        c5 = corr[max_idx] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Subgradient
        s = max_idx - (n - 1)  # the shift
        subgrad = np.zeros(n)
        for i in range(n):
            if 0 <= i + s < n:
                subgrad[i] += (1 - h[i + s])
            if 0 <= i - s < n:
                subgrad[i] -= h[i - s]
        
        # Normalize and take step
        norm = np.linalg.norm(subgrad)
        if norm < 1e-10:
            break
        lr = lr_init / (1 + 0.01 * t)
        h = project_onto_feasible(h - lr * subgrad / norm, target_sum)
    
    return best_h, best_c5
```

```python
import time
import sys

def compute_overlaps(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_overlaps(h)) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_update(h, s, n):
    """Compute subgradient of O_s w.r.t. h"""
    sg = np.zeros(n)
    # For O_s = sum_i h[i]*(1-h[i+s]):
    # dO_s/dh[i] = (1-h[i+s])*[i+s<n] - h[i-s]*[i-s>=0]
    if s >= 0:
        for i in range(n - s):
            sg[i] += (1 - h[i + s])
        for i in range(s, n):
            sg[i] -= h[i - s]
    else:
        s_abs = -s
        for i in range(s_abs, n):
            sg[i] += (1 - h[i - s_abs])
        for i in range(n - s_abs):
            sg[i] -= h[i + s_abs]
    return sg

    h = h_init.copy().astype(float)
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start_time = time.time()
    
    t = 0
    lr = 0.5
    
    while time.time() - start_time < time_budget:
        t += 1
        overlaps = compute_overlaps(h)
        max_idx = np.argmax(overlaps)
        c5 = overlaps[max_idx] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        s = max_idx - (n - 1)
        sg = subgradient_update(h, s, n)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        
        # Step size decay
        lr = 0.5 / (1 + 0.05 * t)
        h = project_feasible(h - lr * sg / sg_norm, target_sum)
    
    return best_h, best_c5
```

```python
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
```

```python
import time

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, s, n):
    sg = np.zeros(n)
    if s >= 0:
        sg[:n-s] += (1.0 - h[s:])
        sg[s:] -= h[:n-s]
    else:
        sa = -s
        sg[sa:] += (1.0 - h[:n-sa])
        sg[:n-sa] -= h[sa:]
    return sg

def softmax_gradient(h, n, beta):
    overlaps = np.correlate(h, 1-h, mode="full")
    shifted = overlaps - np.max(overlaps)
    weights = np.exp(beta * shifted)
    wsum = weights.sum()
    weights /= wsum
    
    sg = np.zeros(n)
    for idx in range(len(overlaps)):
        if weights[idx] > 1e-5:
            s = idx - (n - 1)
            sg += weights[idx] * compute_subgrad(h, s, n)
    return sg
```

```python
def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, s, n):
    sg = np.zeros(n)
    if s >= 0:
        sg[:n-s] += (1.0 - h[s:])
        sg[s:] -= h[:n-s]
    else:
        sa = -s
        sg[sa:] += (1.0 - h[:n-sa])
        sg[:n-sa] -= h[sa:]
    return sg

def softmax_gradient(h, n, beta):
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    shifted = overlaps - np.max(overlaps)
    weights = np.exp(beta * shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    sg = np.zeros(n)
    for idx in range(len(overlaps)):
        if weights[idx] > 1e-6:
            s = idx - (n - 1)
            sg += weights[idx] * compute_subgrad(h, s, n)
    return sg
```
