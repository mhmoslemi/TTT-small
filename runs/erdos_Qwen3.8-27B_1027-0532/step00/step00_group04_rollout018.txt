    # The minimum is where d_k switches from negative to positive as we increase x

    lo, hi = -0.5, 0.5  # delta should be small

        max_idx = np.argmax(vals)
        # If multiple achieve max, return range of slopes
        active = np.where(vals == vals[max_idx])[0]
        return d_vals[active].min(), d_vals[active].max()

        if lo_slope > 0:
            hi = mid
        elif hi_slope < 0:
            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_c5(h, n, alpha=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # Log-sum-exp for stability
    shifted = (corr - max_corr) / alpha
    soft_max = max_corr + alpha * np.log(np.sum(np.exp(shifted)))
    return float(soft_max * dx)

def find_optimal_delta(f_vals, d_vals, delta_min, delta_max):
    if delta_min >= delta_max:
        return 0.0
    def slope_at(delta):
        vals = f_vals + delta * d_vals
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-12)[0]
        return d_vals[active].min(), d_vals[active].max()
    
    lo, hi = delta_min, delta_max
    for _ in range(50):
        mid = (lo + hi) / 2
        lo_s, hi_s = slope_at(mid)
        if lo_s >= 0:
            hi = mid
        elif hi_s <= 0:
            lo = mid
        else:
            return mid
        if hi - lo < 1e-12:
            break
    return (lo + hi) / 2

        valid2 = (idx2 >= 0) & (idx2 < n)
        idx1c = np.clip(idx1, 0, n-1)
        idx2c = np.clip(idx2, 0, n-1)
        part1 = np.where(valid1, 1 - h[idx1c], 0)
        part2 = np.where(valid2, h[idx2c], 0)
        return part1 - part2
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for pass_num in range(max_passes):
        f_vals = np.correlate(h, 1 - h, mode="full")
        current_max = float(np.max(f_vals))
        
        if current_max * dx <= best_c5 - 1e-12:
            best_c5 = current_max * dx
            best_h = h.copy()
        
        grads = compute_all_grads(h)
        
        improved = False
        for i in range(n):
            best_delta = 0.0
            best_j = -1
            best_new_max = current_max
            
            for j in range(n):
                if j == i:
                    continue
                d_min = max(-h[i], h[j] - 1)
                d_max = min(1 - h[i], h[j])
                if d_min >= d_max - 1e-15:
                    continue
                
                d_k = grads[i] - grads[j]
                opt_delta = find_optimal_delta(f_vals, d_k, d_min, d_max)
                
                if abs(opt_delta) < 1e-15:
                    continue
                    
                new_max = float(np.max(f_vals + opt_delta * d_k))
                if new_max < best_new_max - tol:
                    best_new_max = new_max
                    best_delta = opt_delta
                    best_j = j
            
            if best_j >= 0:
                h[i] = float(np.clip(h[i] + best_delta, 0, 1))
                h[best_j] = float(np.clip(h[best_j] - best_delta, 0, 1))
                f_vals = np.correlate(h, 1 - h, mode="full")
                current_max = float(np.max(f_vals))
                improved = True
        
        if not improved:
            break

Wait, the correction terms: when m=i and m+lag=j, the term h[i]*(1-h[j]) becomes (h[i]+delta)*(1-(h[j]-delta)) = h[i]*(1-h[j]) + delta*(1-h[j]) + h[i]*delta - delta^2

So there's a -delta^2 term when i+lag=j (i.e., when lag = j-i).

So there's a +delta^2 term when j+lag=i (i.e., when lag = i-j).

Δresult[k] = delta*(grad_i[k] - grad_j[k]) - delta^2*[lag = j-i] + delta^2*[lag = i-j]

The delta^2 terms only affect at most 2 lags (lag = j-i and lag = i-j). So the linear approximation is exact except for at most 2 lags. This means my approach is very accurate for small delta, and the error is bounded by delta^2.

                    continue
                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dlo >= dhi - 1e-14:
                    continue
                dk = G[i] - G[j]
                od = find_opt_delta(fv, dk, dlo, dhi)
                if abs(od) < 1e-14:
                    continue
                nm = float(np.max(fv + od * dk))
                if nm < bn - tol:
                    bn = nm
                    bd = od
                    bi = j
            if bi >= 0:
                h[i] = min(1.0, max(0.0, h[i] + bd))
                h[bi] = min(1.0, max(0.0, h[bi] - bd))
                fv = np.correlate(h, 1.0 - h, mode="full")
                cm = float(np.max(fv))
                imp = True
        if not imp:
            break
    
    return best_h, best_c5

def fix_constraint(h, n):
    ts = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = ts - h.sum()
        h += diff2 / n
    return h

def interp_h(h_old, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return fix_constraint(h_new, n_new)
