    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    active = np.argmax(corr)
    s = active - n + 1
    grad = np.ones(n)  # d/dh[j] of sum h[j] = 1

        else:
            hi = mid
    proj = np.clip(h - mid, 0.0, 1.0)
    return _repair(proj, target)

def _projected_subgrad(h, target, n_iters=2000):
    n = len(h)
    dx = 2.0 / n
    h_curr = _repair(h.copy(), target)
    best_c5 = _c5_obj(h_curr)
    step = 0.5
    for it in range(n_iters):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        max_val = np.max(corr)
        active_mask = corr >= max_val - 1e-5
        grad = np.zeros(n)
        count = 0
        for k in range(len(corr)):
            if active_mask[k]:
                s = k - n + 1
                for j in range(n):
                    g = 1.0
                    if 0 <= j + s < n:
                        g -= h_curr[j + s]
                    if 0 <= j - s < n:
                        g -= h_curr[j - s]
                    grad[j] += g
                count += 1
        if count > 0:
            grad /= count
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_new = _project_feasible(h_curr - step * grad, target)
        c5_new = _c5_obj(h_new)
        if c5_new < best_c5 - 1e-14:
            h_curr = h_new
            best_c5 = c5_new
            step = min(step * 1.1, 5.0)
        else:
            step *= 0.7
            if step < 1e-10:
                break
    return h_curr, _c5_obj(h_curr)

def _random_binary(h_base, rng, target):
    n = len(h_base)
    h = np.zeros(n)
    n_ones = int(target)
    frac = target - n_ones
    idx = rng.choice(n, size=n, replace=False)
    h[idx[:n_ones]] = 1.0
    if frac > 0:
        h[idx[n_ones]] = frac
    blend = 0.5
    h = blend * h + (1.0 - blend) * h_base
    return _repair(h, target)

def _random_smooth(rng, target, n):
    h = rng.uniform(0.0, 1.0, n)
    return _repair(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    global _target_sum
    rng = np.random.RandomState(seed)
    start_time = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(47, 0.5)

    n = len(h0)
    target = n / 2.0
    _target_sum = target
    h0 = _repair(h0, target)

    best_h = h0.copy()
    best_c5 = _c5_obj(best_h)

    def _remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: SLSQP from initial point
    if _remaining() > 30:
        h_opt = _slsqp_optimize(h0, n, 300)
        c5_val = _c5_obj(h_opt)
        if c5_val < best_c5:
            best_h = h_opt
            best_c5 = c5_val

    # Phase 2: Multiple SLSQP restarts with perturbations
    n_restarts = 30
    for i in range(n_restarts):
        if _remaining() < 5:
            break
        if i % 5 == 0:
            h_start = _random_smooth(rng, target, n)
        elif i % 5 == 1:
            h_start = best_h + rng.normal(0, 0.15, n)
            h_start = _repair(h_start, target)
        elif i % 5 == 2:
            h_start = _random_binary(best_h, rng, target)
        elif i % 5 == 3:
            h_start = best_h + rng.normal(0, 0.3, n)
            h_start = _repair(h_start, target)
        else:
            h_start = best_h.copy()
            swap_idx = rng.choice(n, size=4, replace=False)
            for j in range(0, len(swap_idx) - 1, 2):
                a, b = swap_idx[j], swap_idx[j + 1]
                h_start[a], h_start[b] = h_start[b], h_start[a]
                h_start = _repair(h_start, target)

        max_iter = 500 if i < 10 else 300
        h_opt = _slsqp_optimize(h_start, n, max_iter)
        c5_val = _c5_obj(h_opt)
        if c5_val < best_c5:
            best_h = h_opt
            best_c5 = c5_val
            if _remaining() < 10:
                break

    # Phase 3: Projected subgradient refinement on best
    if _remaining() > 20:
        h_ref, c5_ref = _projected_subgrad(best_h, target, 3000)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref

    # Phase 4: Try higher resolutions
    for n_try in [53, 61, 81, 101, 121, 151, 201]:
        if _remaining() < 30:
            break
        n_cur = len(best_h)
        target_try = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, n_cur)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = _repair(h_interp, target_try)

        for restart in range(3):
            if _remaining() < 15:
                break
            if restart == 0:
                h_start = h_interp
            else:
                h_start = h_interp + rng.normal(0, 0.1, n_try)
                h_start = _repair(h_start, target_try)

            max_iter = 200 if n_try <= 100 else 100
            h_opt = _slsqp_optimize(h_start, n_try, max_iter)
            c5_val = _c5_obj(h_opt)
            if c5_val < best_c5:
                best_h = h_opt
                best_c5 = c5_val

    # Phase 5: Final polishing with projected subgradient
    if _remaining() > 10:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_ref, c5_ref = _projected_subgrad(best_h, target_final, 5000)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref

    # Final repair
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = _repair(best_h, target_final)
    final_c5 = _c5_obj(best_h)

    return best_h, final_c5, n_final
