import time

def evaluate_erdos(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    dx = 2.0 / len(h)
    cor = np.correlate(h, 1.0 - h, mode="full")
    return np.max(cor) * dx

def project_feasible(h_raw):
    """Project h_raw to {0≤h≤1, sum(h)=n/2}"""
    n = len(h_raw)
    target = n / 2.0
    # Bisection on tau
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h_raw + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h_raw + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-9:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.count_nonzero(mask) > 0:
            h_proj[mask] += diff / np.count_nonzero(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_smooth_gradient(h, beta=50.0):
    """Compute gradient of smooth approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    cor = np.correlate(h, one_minus_h, mode="full")
    cor_scaled = cor * dx
    
    # Softmax weights
    max_c = np.max(cor_scaled)
    exps = np.exp(beta * (cor_scaled - max_c))
    weights = exps / np.sum(exps)
    
    # Compute gradient of each cor[p] w.r.t. h
    # For output index p, lag = p - (n-1)
    # cor[p] = sum over valid n of h[n+lag] * (1-h[n])
    # grad[h[i]] for a specific p:
    #   = sum over valid n where n+lag = i of (1-h[n])  [i.e., n = i-lag]
    #   - sum over valid n where n = i of h[n+lag]  [i.e., the term where i is the n index]
    
    # This is expensive to compute for all p simultaneously.
    # Instead, for each p, the gradient is:
    # grad_p[i] = (1-h[i-lag_p]) * indicator(valid) - h[i+lag_p] * indicator(valid)
    
    # Let me compute the weighted sum directly.
    grad = np.zeros(n)
    
    for p in range(len(cor)):
        if weights[p] < 1e-10:
            continue
        lag = p - (n - 1)
        # Valid range for n (the index into 1-h)
        if lag >= 0:
            n_lo, n_hi = 0, n - 1 - lag
        else:
            n_lo, n_hi = -lag, n - 1
        
        for n_idx in range(n_lo, n_hi + 1):
            i1 = n_idx + lag  # index in h (the "shifted" one)
            i2 = n_idx        # index in 1-h (the "original" one)
            grad[i1] += weights[p] * (1.0 - h[i2])
            grad[i2] += weights[p] * (-h[i1])
    
    return grad * dx

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:

    best_h = h.copy()
    best_c5 = evaluate_erdos(best_h)
    
    beta = 30.0
    alpha = 0.005
    no_improve = 0

        grad = compute_smooth_gradient(best_h, beta)
        
        # Take a step
        h_new = best_h - alpha * grad
        h_new = project_feasible(h_new)

            best_c5 = c5_new
            no_improve = 0
            alpha = min(alpha * 1.02, 0.02)
        else:

            alpha *= 0.9
            if no_improve > 500:

                c5_pert = evaluate_erdos(h_pert)

                    best_c5 = c5_pert
                alpha = 0.005
                no_improve = 0
                beta = min(beta * 1.5, 200.0)
                if no_improve > 1000:
                    break
    
    return best_h, best_c5, n

The gradient is:
∇F(h) = Σ_p softmax_p * dx * ∇cor_p(h)

where softmax_p = exp(β * dx * cor_p) / Σ_q exp(β * dx * cor_q)

And ∇cor_p(h) is the gradient of the p-th correlation element.

For a specific p with lag k = p - (n-1):
cor_p = Σ_{n=n_lo}^{n_hi} h[n+k] * (1-h[n])

∂cor_p/∂h[i] = (1-h[i-k]) * [i-k ∈ [n_lo, n_hi]] - h[i+k] * [i ∈ [n_lo, n_hi]]

(Using the fact that h[i] appears as h[n+k] when n = i-k, and as h[n] when n = i.)

Let me verify: if i = n+k for some n in [n_lo, n_hi], then ∂/∂h[i] of h[n+k]*(1-h[n]) = (1-h[n]) = (1-h[i-k]). ✓
If i = n for some n in [n_lo, n_hi], then ∂/∂h[i] of h[n+k]*(1-h[n]) = -h[n+k] = -h[i+k]. ✓

So:
∂cor_p/∂h[i] = (1-h[i-k]) * I(i-k ∈ [n_lo, n_hi]) - h[i+k] * I(i ∈ [n_lo, n_hi])

where n_lo and n_hi depend on k:
- If k ≥ 0: n_lo = 0, n_hi = n-1-k
- If k < 0: n_lo = -k, n_hi = n-1

And the conditions:
- i-k ∈ [n_lo, n_hi] means n_lo ≤ i-k ≤ n_hi, i.e., n_lo+k ≤ i ≤ n_hi+k
- i ∈ [n_lo, n_hi] means n_lo ≤ i ≤ n_hi

Let me define:
- For the first term: i ∈ [n_lo+k, n_hi+k]
- For the second term: i ∈ [n_lo, n_hi]

So:
∂cor_p/∂h[i] = (1-h[i-k]) * I(n_lo+k ≤ i ≤ n_hi+k) - h[i+k] * I(n_lo ≤ i ≤ n_hi)

def compute_gradient(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    cor = np.correlate(h, one_minus_h, mode="full")
    
    # Softmax weights
    cor_scaled = cor * dx
    max_c = np.max(cor_scaled)
    exps = np.exp(beta * (cor_scaled - max_c))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    
    for p in range(len(cor)):
        if weights[p] < 1e-12:
            continue
        k = p - (n - 1)
        if k >= 0:
            n_lo, n_hi = 0, n - 1 - k
        else:
            n_lo, n_hi = -k, n - 1
        
        # First term: (1-h[i-k]) for i in [n_lo+k, n_hi+k]
        i1_lo = max(n_lo + k, 0)
        i1_hi = min(n_hi + k, n - 1)
        if i1_lo <= i1_hi:
            # h[i-k] for i in [i1_lo, i1_hi], i.e., h[i1_lo-k : i1_hi-k+1]
            idx = np.arange(i1_lo, i1_hi + 1) - k
            grad[i1_lo:i1_hi+1] += weights[p] * (1.0 - h[idx])
        
        # Second term: -h[i+k] for i in [n_lo, n_hi]
        i2_lo = max(n_lo, 0)
        i2_hi = min(n_hi, n - 1)
        if i2_lo <= i2_hi:
            idx = np.arange(i2_lo, i2_hi + 1) + k
            grad[i2_lo:i2_hi+1] += weights[p] * (-h[idx])
    
    return grad * dx

def project_feasible(h_raw):
