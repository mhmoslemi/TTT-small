```python
        else:

    diff = target_sum - x.sum()

        if free.sum() > 0:

            x = np.clip(x, 0.0, 1.0)

    m = corr.max()
```

So: c_k = Σ_{i=lo}^{hi} h[i] * (1 - h[i+k-n+1])
where lo = max(0, n-1-k), hi = min(n-1, 2n-2-k)

Now, ∂c_k/∂h[j]:

Case 1: j is in [lo, hi] (i.e., j is one of the i values in the sum)
Then the term h[j]*(1-h[j+k-n+1]) contributes: (1 - h[j+k-n+1])

Case 2: j = i+k-n+1 for some i in [lo, hi]
Then i = j-k+n+1, and we need i ∈ [lo, hi], i.e., lo ≤ j-k+n+1 ≤ hi
The term h[i]*(1-h[j]) = h[j-k+n+1]*(1-h[j]) contributes: -h[j-k+n+1]

So: ∂c_k/∂h[j] = (1 - h[j+k-n+1]) * [j ∈ [lo, hi]] - h[j-k+n+1] * [j-k+n+1 ∈ [lo, hi]]

where the bracketed terms are indicators.

Now, j ∈ [lo, hi] means:
max(0, n-1-k) ≤ j ≤ min(n-1, 2n-2-k)

And j-k+n+1 ∈ [lo, hi] means:
max(0, n-1-k) ≤ j-k+n+1 ≤ min(n-1, 2n-2-k)
=> max(0, n-1-k) + k - n - 1 ≤ j ≤ min(n-1, 2n-2-k) + k - n - 1
=> max(k-n-1, -1) ≤ j ≤ min(k-2, 2k-n-3) ...

```python
def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(corr.max())
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n - 1, 2 * n - 2 - k)
        for i in range(lo, hi + 1):
            j2 = i + k - n + 1
            if 0 <= j2 < n:
                grad[i] += wk * (1.0 - h[j2])
                grad[j2] -= wk * h[i]
    
    grad *= dx
    return grad
```

```python
def project_onto_constraint(y, target_sum):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_soft_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(corr.max())
    weights = np.exp(beta * (corr - m))
    Z = float(weights.sum())

        wk = weights[k]
        if wk < 1e-14:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n - 1, 2 * n - 2 - k)

        j2 = idx + k - n + 1
        grad[idx] += wk * (1.0 - h[j2])
        grad[j2] -= wk * h[idx]
```
