        vals = corr * dx

import time
from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    """Project h onto {g: 0 <= g <= 1, sum(g) = target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradient(h, dx, alpha):
    """Compute gradient of smooth max of corr*dx."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    exp_vals = np.exp(alpha * (vals - m))
    weights = exp_vals / np.sum(exp_vals)
    
    n_corr = len(corr)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k in range(n_corr):
        lag = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-12:
            continue
        if lag == 0:
            grad += w * (one_minus_h - h)
        elif lag > 0:
            grad[lag:] += w * one_minus_h[:n - lag]
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * one_minus_h[l:]
            grad[l:] -= w * h[:n - l]
    
    return grad

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start = time.time()
    alpha = 50.0
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.9

        momentum = beta * momentum + (1 - beta) * grad
        h_new = project_onto_constraint(h - lr * momentum, target_sum)

        h = h_new

    return best_h, best_c5

def coordinate_descent_step(h, i, j):
    """Try to improve by adjusting h[i] and h[j] while keeping their sum constant."""
    n = len(h)
    a = h[i]
    b = h[j]
    s = a + b  # sum to preserve
    
    # t = new_a - a, so new_a = a + t, new_b = b - t
    # Constraints: 0 <= a+t <= 1, 0 <= b-t <= 1
    # t_min = max(-a, -(1-b)), t_max = min(1-a, b)
    t_min = max(-a, b - 1)
    t_max = min(1 - a, b)
    
    if t_min >= t_max:
        return h, 0.0
    
    def f(t):
        h_new = h.copy()
        h_new[i] = a + t
        h_new[j] = b - t
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        return np.max(corr)
    
    # Minimize f over [t_min, t_max]
    from scipy.optimize import minimize_scalar
    result = minimize_scalar(f, bounds=(t_min, t_max), method='bounded', options={'maxiter': 100})
    
    if result.fun < f(0):
        h[i] = a + result.x
        h[j] = b - result.x
        return h, f(0) - result.fun
    return h, 0.0

    t2 = min(budget_s * 0.15, remaining * 0.6, 200.0)
    if t2 > 10:
        h_try, c5_raw_try = gradient_refine(best_h, target_sum, max_time=t2, seed=seed + 1)
        c5_try = c5_raw_try * dx
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
        remaining = budget_s - elapsed() - 20

    # Phase 3: Random restarts with coord descent + gradient - 25% budget
    t3_total = min(budget_s * 0.25, remaining, 300.0)
    if t3_total > 30:
        n_restarts = max(3, int(t3_total / 40))
        t_per = t3_total / n_restarts
        for r in range(n_restarts):
            if elapsed() > budget_s - 20:
                break
            sigma = 0.04 + 0.03 * r
            h_pert = best_h + sigma * rng.standard_normal(n)
            h_pert = project_onto_constraint(h_pert, target_sum)
            h_try, c5_raw_try = coord_descent(h_pert, target_sum, max_time=t_per * 0.7, seed=seed + 100 + r)
            c5_try = c5_raw_try * dx
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            if t_per * 0.2 > 5:
                h_try2, c5_raw_try2 = gradient_refine(h_try, target_sum, max_time=t_per * 0.2, seed=seed + 200 + r)
                c5_try2 = c5_raw_try2 * dx
                if c5_try2 < best_c5:
                    best_h = h_try2
                    best_c5 = c5_try2

    remaining = budget_s - elapsed() - 20

    # Phase 4: Try different n values - 15% budget
    t4_total = min(budget_s * 0.15, remaining, 200.0)
    if t4_total > 40:
        n_candidates = [n + 2, n + 4, n - 2, n + 6, n + 8]
        n_candidates = [x for x in n_candidates if 50 <= x <= 120 and x != n]
        t_per_n = t4_total / max(len(n_candidates), 1)
        for n_try in n_candidates:
            if elapsed() > budget_s - 20:
                break
            if t_per_n < 10:
                break
            dx_try = 2.0 / n_try
            target_sum_try = n_try / 2.0

            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_onto_constraint(h_interp, target_sum_try)

            h_try, c5_raw_try = coord_descent(h_interp, target_sum_try, max_time=t_per_n * 0.7, seed=seed + 300 + n_try)
            c5_try = c5_raw_try * dx_try
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                n = n_try
                dx = dx_try

    # Final: one more gradient polish on the best
    remaining = budget_s - elapsed() - 5
    if remaining > 10:
        h_try, c5_raw_try = gradient_refine(best_h, target_sum, max_time=min(remaining, 60.0), seed=seed + 999)
        c5_try = c5_raw_try * (2.0 / len(best_h))
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    # Ensure feasibility
    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = np.max(compute_c(best_h)) * (2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
