import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, S):
    """Project h onto {x: sum(x)=S, 0<=x<=1} via bisection on shift"""
    lo = float(h.min()) - 1.0
    hi = float(h.max())
    for _ in range(100):
        mid = (lo + hi) * 0.5
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if val > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def solve_1d_cp(a_k, b_k):
    """
    Minimize max_k (a_k + b_k * x) over x in [0,1].
    Returns (x_opt, min_val).
    """
    n_k = len(a_k)
    
    # Binary search on t
    # Lower bound: min_k a_k (at x=0, max is at least min_k a_k... no)
    # Actually at x=0: max_k a_k
    # At x=1: max_k (a_k + b_k)
    # The minimum is at most min(max_k a_k, max_k(a_k+b_k))
    
    t_lo = float(np.min(a_k))  # absolute lower bound (may not be achievable)
    t_hi = float(np.min(np.max(a_k), np.max(a_k + b_k)))  # achievable upper bound
    
    # Binary search
    for _ in range(80):
        t_mid = (t_lo + t_hi) * 0.5
        # Check feasibility: exists x in [0,1] with a_k + b_k*x <= t_mid for all k
        lo_x = 0.0
        hi_x = 1.0
        # For b_k > 0: x <= (t_mid - a_k) / b_k
        pos = b_k > 1e-12
        if np.any(pos):
            hi_x = min(hi_x, float(np.min((t_mid - a_k[pos]) / b_k[pos])))
        # For b_k < 0: x >= (t_mid - a_k) / b_k
        neg = b_k < -1e-12
        if np.any(neg):
            lo_x = max(lo_x, float(np.max((t_mid - a_k[neg]) / b_k[neg])))
        
        if lo_x <= hi_x:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    t_opt = t_hi
    
    # Find optimal x
    lo_x = 0.0
    hi_x = 1.0
    pos = b_k > 1e-12
    if np.any(pos):
        hi_x = min(hi_x, float(np.min((t_opt - a_k[pos]) / b_k[pos])))
    neg = b_k < -1e-12
    if np.any(neg):
        lo_x = max(lo_x, float(np.max((t_opt - a_k[neg]) / b_k[neg])))
    lo_x = max(0.0, lo_x)
    hi_x = min(1.0, hi_x)
    x_opt = (lo_x + hi_x) * 0.5
    x_opt = max(0.0, min(1.0, x_opt))
    
    return x_opt, t_opt

def coordinate_descent(h, S, n_sweeps=200, tol=1e-10):
    """Exact coordinate descent for the minimax problem"""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    s = np.arange(-(n-1), n)  # shifts, shape (2n-1,)
    
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old_hi = h[i]
            g = 1.0 - h
            corr = np.correlate(h, g, mode="full") * dx
            
            # Compute slopes for this coordinate
            valid_upper = (i + s >= 0) & (i + s < n)
            upper_vals = np.zeros(2*n - 1)
            upper_vals[valid_upper] = 1.0 - h[i + s[valid_upper]]
            
            valid_lower = (i - s >= 0) & (i - s < n)
            lower_vals = np.zeros(2*n - 1)
            lower_vals[valid_lower] = h[i - s[valid_lower]]
            
            slopes = dx * (upper_vals - lower_vals)
            a_k = corr - slopes * old_hi
            b_k = slopes
            
            new_hi, _ = solve_1d_cp(a_k, b_k)
            h[i] = new_hi
            max_change = max(max_change, abs(new_hi - old_hi))
        
        if max_change < tol:
            break
    
    h = project_to_constraint(h, S)
    return h

def subgradient(h):
    """Compute subgradient of C5 at h (using average of active constraints)"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    max_val = np.max(corr)
    
    # Use all constraints within a small tolerance of the max
    tol = 0.01 * max_val + 1e-6
    active = corr >= (max_val - tol)
    n_active = np.sum(active)
    if n_active == 0:
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
        n_active = 1
    
    s = np.arange(-(n-1), n)
    grad = np.zeros(2*n - 1)  # temp
    grad_final = np.zeros(n)
    
    for i in range(n):
        valid_upper = (i + s >= 0) & (i + s < n)
        upper_vals = np.zeros(2*n - 1)
        upper_vals[valid_upper] = 1.0 - h[i + s[valid_upper]]
        
        valid_lower = (i - s >= 0) & (i - s < n)
        lower_vals = np.zeros(2*n - 1)
        lower_vals[valid_lower] = h[i - s[valid_lower]]
        
        slopes = dx * (upper_vals - lower_vals)
        grad_final[i] = float(np.sum(slopes[active])) / n_active
    
    return grad_final

        c5 = compute_c5(h_cd)
        if c5 < best_c5 - 1e-10:

            best_h = h_cd.copy()
        else:

    time_limit_2 = start_time + budget_s * 0.8
    attempt = 0

            grad = subgradient(h)
            h_new = h - alpha * grad
            h_new = project_to_constraint(h_new, S)
            c5_new = compute_c5(h_new)

                h_best_in_run = h_new.copy()

            # Polish with coordinate descent

        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

            hi_x = min(hi_x, float(np.min(upper_bounds)))
        
        neg_mask = b_k < -1e-12
        if np.any(neg_mask):
            lower_bounds = (t_mid - a_k[neg_mask]) / b_k[neg_mask]
            lo_x = max(lo_x, float(np.max(lower_bounds)))
        
        zero_mask = (b_k >= -1e-12) & (b_k <= 1e-12)
        if np.any(zero_mask):
            if float(np.max(a_k[zero_mask])) > t_mid:
                lo_x = 1.0
                hi_x = 0.0  # infeasible
        
        if lo_x <= hi_x:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    t_opt = t_hi
    
    # Find x_opt
    lo_x = 0.0
    hi_x = 1.0
    pos_mask = b_k > 1e-12
    if np.any(pos_mask):
        hi_x = min(hi_x, float(np.min((t_opt - a_k[pos_mask]) / b_k[pos_mask])))
    neg_mask = b_k < -1e-12
    if np.any(neg_mask):
        lo_x = max(lo_x, float(np.max((t_opt - a_k[neg_mask]) / b_k[neg_mask])))
    lo_x = max(0.0, lo_x)
    hi_x = min(1.0, hi_x)
    
    if lo_x > hi_x:
        x_opt = max(0.0, min(1.0, (lo_x + hi_x) * 0.5))
    else:
        x_opt = (lo_x + hi_x) * 0.5
    x_opt = max(0.0, min(1.0, x_opt))
    
    return x_opt, t_opt

def coordinate_descent(h, S, n_sweeps=200, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    s_shifts = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old_hi = h[i]
            g = 1.0 - h
            corr = np.correlate(h, g, mode="full") * dx
            
            # Slopes for coordinate i
            valid_upper = (i + s_shifts >= 0) & (i + s_shifts < n)
            upper_vals = np.zeros(2 * n - 1)
            upper_vals[valid_upper] = 1.0 - h[i + s_shifts[valid_upper]]
            
            valid_lower = (i - s_shifts >= 0) & (i - s_shifts < n)
            lower_vals = np.zeros(2 * n - 1)
            lower_vals[valid_lower] = h[i - s_shifts[valid_lower]]
            
            slopes = dx * (upper_vals - lower_vals)
            a_k = corr - slopes * old_hi
            b_k = slopes
            
            new_hi, _ = solve_1d_minimax(a_k, b_k)
            h[i] = new_hi
            change = abs(new_hi - old_hi)
            if change > max_change:
                max_change = change
        
        if max_change < tol:
            break
    
    h = project_to_constraint(h, S)
    return h

This means I need a different approach. Options:
1. **Pair coordinate descent**: Update pairs of coordinates (i, j) simultaneously, with the constraint h[i] + h[j] = constant.
2. **Penalty method**: Add a penalty term for violating the sum constraint.
3. **Lagrangian method**: Use a Lagrange multiplier for the sum constraint.

Let me use approach 3 (Lagrangian):

For a fixed λ, this is still a minimax problem with an added linear term. In coordinate descent, the 1D subproblem becomes:
minimize max_k (a_k + b_k * x) + λ * (x - c) for x in [0,1]
= maximize-then-minimize... 

Actually, max_k (a_k + (b_k + λ) * x - λ*c) = max_k (a_k - λ*c + (b_k + λ) * x)

So it's the same 1D minimax problem with modified slopes! I just add λ to all slopes.

Then I can use the same `solve_1d_minimax` function with b_k' = b_k + λ.

And I update λ using a subgradient method: if sum(h) > S, increase λ; if sum(h) < S, decrease λ.

def coordinate_descent_lagrangian(h0, S, n_sweeps=100, n_dual_iter=50, tol=1e-8):

    dx = 2.0 / n
    s_shifts = np.arange(-(n-1), n)
    h = h0.astype(float).copy()

    for dual_iter in range(n_dual_iter):
        for sweep in range(n_sweeps):

            for i in range(n):
                old_hi = h[i]
                g = 1.0 - h
                corr = np.correlate(h, g, mode="full") * dx

                upper_vals = np.zeros(2 * n - 1)

                lower_vals = np.zeros(2 * n - 1)
                lower_vals[valid_lower] = h[i - s_shifts[valid_lower]]

        sum_h = float(np.sum(h))
        lam *= 1.1  # increase step size
        if sum_h > S:
            lam += 0.1
        elif sum_h < S:
            lam -= 0.1
        lam = max(-1.0, min(1.0, lam))
    
    # Final projection
    h = project_to_constraint(h, S)
    return h
