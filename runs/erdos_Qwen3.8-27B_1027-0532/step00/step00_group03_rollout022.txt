import time

def compute_c5(h, dx):
    """Compute C5 = max of correlate(h, 1-h, 'full') * dx"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def local_search_binary(h, dx, time_limit=60, verbose=False):
    """Local search: try all swaps, accept best improving move."""
    start = time.time()
    n = len(h)
    best_h = h.astype(float).copy()
    best_val = compute_c5(best_h, dx)
    
    while time.time() - start < time_limit:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_gain = 1e-12
        best_swap = None
        
        for i in ones:
            for j in zeros:
                best_h[i], best_h[j] = 0.0, 1.0
                val = compute_c5(best_h, dx)
                best_h[i], best_h[j] = 1.0, 0.0
                
                gain = best_val - val
                if gain > best_gain:
                    best_gain = gain
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        best_h[i], best_h[j] = 0.0, 1.0
        best_val -= best_gain
        if verbose:
            print(f"  Swapped ({i},{j}), C5 = {best_val:.6f}")
    
    return best_h, best_val

import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def local_search(h_init, dx, time_limit=30):
    n = len(h_init)
    h = np.array(h_init, dtype=np.float64).copy()
    best_val = compute_c5(h, dx)
    start = time.time()

    while time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_gain = 1e-15
        best_i = -1
        best_j = -1

        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                val = compute_c5(h, dx)
                h[i], h[j] = 1.0, 0.0
                gain = best_val - val
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j

        if best_i < 0:
            break

        h[best_i], h[best_j] = 0.0, 1.0
        best_val -= best_gain

    return h, best_val


def wiggle_search(h_init, dx, time_limit=20, eps=0.1):
    n = len(h_init)
    h = np.array(h_init, dtype=np.float64).copy()
    best_val = compute_c5(h, dx)

            break

        j = rng.integers(0, n)
        if i == j:
            continue
        for sign in [1.0, -1.0]:
            step = sign * eps
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + step, 0.0, 1.0)
            actual_step = h_new[i] - h[i]
            h_new[j] = np.clip(h_new[j] - actual_step, 0.0, 1.0)
            actual_step2 = h_new[j] - h[j]
            h_new[i] = h[i] + actual_step2
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:

            val = compute_c5(h_new, dx)
            if val < best_val - 1e-12:
                best_val = val
                h = h_new.copy()

    return h, best_val

    h = np.zeros(n)
    positions = np.floor(np.arange(n_ones) * n / n_ones).astype(int)
    h[positions] = 1.0
    return h


def make_alternating(n, n_ones):
    h = np.zeros(n)
    h[::2] = 1.0
    actual = int(np.sum(h > 0.5))
    if actual != n_ones:

    return h


def make_shifted(n, n_ones, shift):

    h[shift::2] = 1.0
    actual = int(np.sum(h > 0.5))
    if actual != n_ones:

    return h
