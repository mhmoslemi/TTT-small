import time
import math

def project_to_feasible(a, n):
    """Project a onto {h : 0 <= h[i] <= 1, sum(h) = n/2}."""
    target = n / 2.0
    lo = a.min() - 2
    hi = a.max() + 2

        s = np.sum(np.clip(a - mid, 0, 1))
        if s > target:

        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(a - lam, 0, 1)

def compute_c5(h):
    """Compute C5 for given h array."""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def get_active_lags(h):
    """Return indices of all active (maximizing) lags."""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(overlaps)
    # Return all lags that are within tolerance of max
    active = np.where(overlaps >= max_val - 1e-10)[0]
    return active, overlaps

def subgradient_mean(h, active_lags):
    """Compute mean subgradient over active lags."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    count = len(active_lags)
    if count == 0:
        return grad
    for k in active_lags:
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1 - h[i + lag])
                grad[i + lag] += (-h[i])
        else:
            m = -lag
            for i in range(m, n):
                grad[i] += (1 - h[i - m])
                grad[i - m] += (-h[i])
    return grad * dx / count

def subgradient_descent(h, n, max_iter=50000, tol=1e-10):
    """Projected subgradient descent."""
    dx = 2.0 / n
    h_cur = h.copy()
    best = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    target_sum = n / 2.0
    
    # Step size schedule
    alpha0 = 0.1
    for t in range(max_iter):
        active, overlaps = get_active_lags(h_cur)
        c5_cur = np.max(overlaps)
        
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best = h_cur.copy()
        
        # Get subgradient

        g_norm = np.linalg.norm(g)
        if g_norm < tol:

        h_new = project_to_feasible(h_new, n)
        
        # Check improvement

    return best, best_c5

```python
def soft_max_objective(h, n, T=100):
    """Compute soft-max approximation of max overlap."""
    dx = 2.0 / n
    overlaps = np.correlate(h, 1-h, mode="full") * dx
    # Use log-sum-exp for numerical stability
    max_ov = np.max(overlaps)
    if T * max_ov > 500:
        # Use shifted version
        log_sum = max_ov + np.log(np.sum(np.exp(T * (overlaps - max_ov)))) / T
    else:
        log_sum = np.log(np.sum(np.exp(T * overlaps))) / T
    return log_sum

def soft_max_gradient(h, n, T=100):
    """Gradient of soft-max objective."""
    dx = 2.0 / n
    overlaps = np.correlate(h, 1-h, mode="full") * dx
    max_ov = np.max(overlaps)
    exp_vals = np.exp(T * (overlaps - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(overlaps)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += weights[k] * (1 - h[i + lag])
                grad[i + lag] += weights[k] * (-h[i])
        else:
            m = -lag
            for i in range(m, n):
                grad[i] += weights[k] * (1 - h[i - m])
                grad[i - m] += weights[k] * (-h[i])
    return grad * dx
```

import numpy as np
import time
import math

def project_to_feasible(a, n):
    target = n / 2.0
    lo = a.min() - 2.0
    hi = a.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(a - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(a - lam, 0.0, 1.0)
    # Fix any numerical drift
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Add/subtract from the entry closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(overlaps))

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def overlap_gradient_single_lag(h, n, lag):
    """Gradient of overlap at a single lag."""
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        m = -lag
        for i in range(m, n):
            grad[i] += (1.0 - h[i - m])
            grad[i - m] += (-h[i])
    return grad

def soft_max_grad(h, n, T):
    """Gradient of soft-max smoothed objective."""
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    max_ov = np.max(overlaps)
    exp_shifted = np.exp(T * (overlaps - max_ov))
    Z = np.sum(exp_shifted)
    weights = exp_shifted / Z
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        g = overlap_gradient_single_lag(h, n, lag)
        grad += w * g
    return grad * dx

    """Soft-max smoothed projected gradient descent."""
    start = time.time()
    h_cur = h.copy()
    target_sum = n / 2.0
    best_local = h_cur.copy()
    best_local_c5 = compute_c5(h_cur)

    n_iter = 2000
    
    for it in range(n_iter):
        if time.time() - start > max_time:

        # Anneal T

        h_cur = h_new

            h_cur = project_to_feasible(h_cur + noise, n)
    
    return best_local, best_local_c5

    """Interpolate from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old + 1)[:-1]  # centers of intervals? or just linspace

    h_new = np.interp(x_new, x_old, h_old)

def adam_projected(h, n, rng, max_time=120, T_schedule=None):
    """Adam optimizer with projection for the soft-max smoothed problem."""
    start = time.time()
    h_cur = h.copy()
    best = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    lr = 0.05
    
    T = 10.0
    it = 0
    
    while time.time() - start < max_time:
        it += 1
        # Update T periodically
        if it % 200 == 0:
            T *= 1.5
            if T > 1000:
                T = 1000
            lr = max(0.001, lr * 0.7)
        
        g = soft_max_grad(h_cur, n, T)
        
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g**2

        v_hat = v_v / (1 - beta2**it)

        h_new = project_to_feasible(h_new, n)

        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        
        h_cur = h_new
    
    return best, best_c5

import numpy as np
import time
import math

def project_to_feasible(a, n):
    target = n / 2.0
    lo = a.min() - 2.0
    hi = a.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(a - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(a - lam, 0.0, 1.0)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        m = -lag
        for i in range(m, n):
            grad[i] += (1.0 - h[i - m])
            grad[i - m] += (-h[i])
    return grad

def softmax_gradient(h, n, T):
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    max_ov = np.max(overlaps)
    exp_shifted = np.exp(T * (overlaps - max_ov))
    Z = np.sum(exp_shifted)
    if Z < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        grad += w * lag_gradient(h, n, lag)
    return grad * dx

def l2penalty_gradient(h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad += lag_gradient(h, n, lag)
    return grad * dx

def adam_optimize(h, n, rng, max_time, T_start=5.0, T_end=800.0, lr_init=0.08, n_restarts=6):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    restart_duration = max_time / n_restarts
    for restart in range(n_restarts):
        if time.time() - start > max_time * 0.95:
            break
        if restart > 0:
            noise_scale = 0.02 * (restart + 1)
            h_cur = project_to_feasible(best_h + rng.randn(n) * noise_scale, n)
        else:
            h_cur = h.copy()
        
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        lr = lr_init
        T = T_start
        it = 0
        plateau = 0
        plateau_best = compute_c5(h_cur)
        
        while time.time() - start < max_time * 0.95:
            it += 1
            if it % 150 == 0:

                lr = max(0.002, lr * 0.75)

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break

            v_v = beta2 * v_v + (1.0 - beta2) * (g * g)
            m_hat = m_v / (1.0 - beta1 ** it)
            v_hat = v_v / (1.0 - beta2 ** it)
            
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_to_feasible(h_cur - step, n)

                best_h = h_new.copy()
                plateau = 0

                plateau_best = c5_new

            if plateau > 3000:

    return best_h, best_c5

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def local_polish(h, n, rng, max_time=60.0):
    start = time.time()
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    
    while time.time() - start < max_time:

                continue
            s = h_cur[i] + h_cur[j]

                    h_cur = h_try
                    c5_cur = c5_try
                    improved = True
                    break
        if not improved:
            break
    
    return h_cur, c5_cur
