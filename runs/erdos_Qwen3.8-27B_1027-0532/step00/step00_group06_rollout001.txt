import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_arr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def swap_pair(h, i, j, amount):
    if amount > 0:
        amt = min(amount, h[i], 1.0 - h[j])
        if amt > 1e-15:
            h[i] -= amt
            h[j] += amt
    else:
        amt = min(-amount, h[j], 1.0 - h[i])
        if amt > 1e-15:
            h[j] -= amt
            h[i] += amt

def local_refine(h, max_time=5.0):
    n = len(h)
    start = time.time()
    curr_c5 = compute_c5(h)
    
    for iteration in range(200):
        if time.time() - start > max_time:
            break

        improved = False

            idxs_i = list(range(-max_shift, n))

        for i in idxs_i:
            if time.time() - start > max_time:
                break
            j = i + max_shift
            if j < 0 or j >= n:
                continue

                saved_j = h[j]
                h[i] -= amt

def sa_phase(h, h_best, c5_best, n_iter, T_start, T_min, swap_mag, max_time):
    n = len(h)
    start = time.time()
    curr_c5 = compute_c5(h)
    
    for it in range(n_iter):
        if time.time() - start > max_time:
            break
        frac = it / max(1, n_iter)
        T = T_min + (T_start - T_min) * (1.0 - frac) ** 2
        
        candidate = h.copy()
        for _ in range(2):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                amount = np.random.uniform(-swap_mag, swap_mag)
                swap_pair(candidate, i, j, amount)
        
        cand_c5 = compute_c5(candidate)
        delta = cand_c5 - curr_c5
        if delta < -1e-12 or (T > 1e-12 and np.random.random() < np.exp(-delta / T)):
            h = candidate
            curr_c5 = cand_c5
            if curr_c5 < c5_best:
                h_best = h.copy()
                c5_best = curr_c5
    return h, curr_c5, h_best, c5_best
