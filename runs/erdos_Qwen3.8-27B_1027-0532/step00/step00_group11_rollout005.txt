    corr = np.correlate(h, 1 - h, mode="full") * dx
    c_max = np.max(corr)

    exp_shifted = np.exp(shifted - np.max(shifted))  # extra stability

            if abs(residual) > 1e-12 and len(cands) > 1:

def projected_gradient(h0, n, dx, target_sum, alpha=100.0, max_iter=5000, lr=0.01):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    step = lr
    momentum_v = np.zeros(n)
    momentum = 0.9
    
    for it in range(max_iter):
        obj, grad = smooth_obj_and_grad(h, dx, alpha)
        momentum_v = momentum * momentum_v - step * grad
        h_new = project_feasible(h + momentum_v, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adaptive step
        if c5_new <= best_c5 + 1e-6:
            step = min(0.1, step * 1.02)
        else:
            step = max(1e-6, step * 0.8)
            h = best_h  # revert
            momentum_v = np.zeros(n)
        
        h = h_new if c5_new < best_c5 else h
    
    return best_h, best_c5
