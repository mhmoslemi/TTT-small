    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):

        p = np.clip(x - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def softmax_gradient(h, tau):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = dx * corr
    
    max_g = g.max()
    shifted = np.clip(tau * (g - max_g), -700.0, 0.0)
    exp_g = np.exp(shifted)
    total = exp_g.sum()
    
    if total < 1e-300:
        w = np.zeros(len(g))
        w[np.argmax(g)] = 1.0
    else:
        w = exp_g / total
    
    grad = np.zeros(n)
    j = np.arange(n)
    
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue

        vd = (j - lag >= 0) & (j - lag < n)

            grad[vd] += wk * (1.0 - h[j[vd] - lag])
        if np.any(vu):

        if c5_6 < best_c5:
            best_h, best_c5, best_n = h6, c5_6, len(h6)

    n = len(best_h)
    best_h = project_onto_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
