from scipy.optimize import minimize
from scipy.special import logsumexp

_STATE = {}

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def prepare_h(h, n_points, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    if len(h) != n_points:
        x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, h)
        h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
            else:
                break
        else:
            mask = h > 1e-10
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
            else:
                break
    return h

def obj_smoothed(x):

    vals = corr * dx
    return float((1.0 / beta) * logsumexp(beta * vals))

    return h
