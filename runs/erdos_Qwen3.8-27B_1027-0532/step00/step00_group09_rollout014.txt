- If j appears as the "first" factor: i = j, need 0≤j+s<n. Contribution: (1-h[j+s])
- If j appears as the "second" factor: i+s = j, i = j-s, need 0≤j-s<n. Contribution: -h[j-s]
- Note: if j appears in both roles (i.e., j and j+s are both in [0,n) AND j-s and j are both in [0,n)), both contributions apply.

Wait, can j appear in both? j as first factor: i=j, so the term is h[j]*(1-h[j+s]). j as second factor: i+s=j, i=j-s, so the term is h[j-s]*(1-h[j]). 

These are different terms (different i values) unless j = j-s, i.e., s=0. For s=0: the term is h[j]*(1-h[j]), and both "first" and "second" give the same contribution. Let's check: first gives (1-h[j]), second gives -h[j]. Total: 1-2h[j]. And d/dh[j] of h[j](1-h[j]) = 1-2h[j]. ✓

For s≠0, they're different terms, so both contributions apply independently.

My formula:
For s ≥ 0:
- Term 1 (j as first factor, need j+s < n): j in [0, n-1-s]. grad[j] += 1-h[j+s]. This is: g[0:n-s] += 1-h[s:n]
- Term 2 (j as second factor, need j-s ≥ 0): j in [s, n-1]. grad[j] -= h[j-s]. This is: g[s:n] -= h[0:n-s]

For s < 0, let s' = -s > 0:
- Term 1 (j as first factor, need j+s < n, i.e., j-s' < n, always true for j<n; AND j+s ≥ 0, i.e., j ≥ s'): j in [s', n-1]. grad[j] += 1-h[j-s']. This is: g[s':n] += 1-h[0:n-s']
- Term 2 (j as second factor, need j-s ≥ 0, i.e., j+s' ≥ 0, always true; AND j-s < n, i.e., j+s' < n, j < n-s'): j in [0, n-1-s']. grad[j] -= h[j+s']. This is: g[0:n-s'] -= h[s':n]

Let me verify with n=3, s=-1 (s'=1):
Term 1: g[1:3] += 1-h[0:2] = [1-a, 1-b]
Term 2: g[0:2] -= h[1:3] = [b, c]
Result: g = [-b, (1-a)-c, 1-b]

Checking: f_{-1} = h[i]*(1-h[i-1]) for i=1,2 = b(1-a) + c(1-b)
∂/∂a = -b ✓
∂/∂b = (1-a) - c ✓
∂/∂c = 1-b ✓

import numpy as np
import time


def project_with_sum(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - mid, 0.0, 1.0)


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float((c * dx).max())


def weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for s in range(n):
        wm = w[n - 1 + s]
        if wm < 1e-12:
            continue
        g[:n - s] += wm * (1.0 - h[s:])
        g[s:] -= wm * h[:n - s]
    for s in range(1, n):
        wm = w[n - 1 - s]
        if wm < 1e-12:
            continue
        g[s:] += wm * (1.0 - h[:n - s])
        g[:n - s] -= wm * h[s:]
    return g


def optimize_subgrad(h0, deadline, rng):
    n = len(h0)
    target = n * 0.5
    h = project_with_sum(h0, target)
    best_h = h.copy()
    best_c5 = c5_value(h)
    v = np.zeros(n)
    mu = 0.9
    alpha = 0.02
    temp = 0.04
    patience = 0
    max_patience = 600

    while time.time() < deadline:
        ov = np.correlate(h, 1.0 - h, 'full')
        cur_c5 = float((ov * (2.0 / n)).max())

        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = best_h + 0.003 * rng.randn(n)
                h = project_with_sum(h, target)
                v *= 0.1
                alpha = 0.02
                temp = 0.04
                patience = 0
                continue

        shifted = ov - ov.max()
        w = np.exp(shifted / temp)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(w))
            w[int(np.argmax(ov))] = 1.0
        else:
            w /= ws

        g = weighted_grad(h, w)
        v = mu * v - alpha * g
        h_new = project_with_sum(h + v, target)

        new_c5 = c5_value(h_new)
        if new_c5 <= cur_c5 + 1e-10:
            h = h_new
            alpha *= 0.999
            temp = max(temp * 0.997, 0.002)
        else:
            alpha *= 0.7
            v *= 0.3
            if alpha < 1e-9:
                break

    return best_h, best_c5


def parabolic_start(n):
    x = np.arange(n, dtype=float) / max(n - 1, 1)
    h = x * (1.0 - x)
    h = h / h.sum() * (n / 2.0)
    return np.clip(h, 0.0, 1.0)


def cos_start(n, f1, f2, a1, a2):
    x = (np.arange(n, dtype=float) + 0.5) * 2.0 / n
    h = 0.5 + a1 * np.cos(np.pi * f1 * x) + a2 * np.cos(np.pi * f2 * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_with_sum(h, n * 0.5)
    return h


def smooth_start(n):
    x = (np.arange(n, dtype=float) + 0.5) * 2.0 / n
    h = 0.5 + 0.3 * np.sin(np.pi * x) ** 2
    h = np.clip(h, 0.0, 1.0)
    h = project_with_sum(h, n * 0.5)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def update_best(h_opt, c5_opt):
        nonlocal best_h, best_c5
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Try initial_h_values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_with_sum(h_init, n_init * 0.5)
        elapsed = time.time() - t0
        alloc = min(max((budget_s - elapsed) * 0.3, 30), 400)
        deadline = time.time() + alloc
        ho, c5o = optimize_subgrad(h_init, deadline, rng)
        update_best(ho, c5o)
    except Exception:
        pass

    # Multiple n values and starts
    n_list = [64, 85, 100, 128, 160, 200]
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10
    if remaining < 20:
        n_list = n_list[:2]
    elif remaining < 100:
        n_list = n_list[:4]

    total_starts = len(n_list) * 5
    time_per = max(remaining / total_starts, 3.0)

    for n in n_list:
        if time.time() - t0 > budget_s - 15:
            break
        starts = [
            parabolic_start(n),
            cos_start(n, 1, 2, 0.25, 0.12),
            cos_start(n, 1, 3, 0.2, 0.1),
            smooth_start(n),
        ]
        hr = rng.uniform(0.0, 1.0, n)
        hr = project_with_sum(hr, n * 0.5)
        starts.append(hr)

        for hs in starts:
            if time.time() - t0 > budget_s - 15:
                break
            rem = budget_s - (time.time() - t0) - 10
            alloc = min(time_per, rem / 2.0)
            if alloc < 3.0:
                break
            deadline = time.time() + alloc
            ho, c5o = optimize_subgrad(hs, deadline, rng)
            update_best(ho, c5o)

    # Final refinement on best
    if best_h is not None and time.time() - t0 < budget_s - 5:
        rem = min(budget_s - (time.time() - t0) - 2, 200)
        deadline = time.time() + rem
        ho, c5o = optimize_subgrad(best_h, deadline, rng)
        update_best(ho, c5o)

    # Final polish with very small steps
    if best_h is not None and time.time() - t0 < budget_s - 3:
        rem = min(budget_s - (time.time() - t0) - 1, 100)
        deadline = time.time() + rem
        h_polish = best_h.copy()
        n = len(h_polish)
        target = n * 0.5
        alpha = 0.005
        v = np.zeros(n)
        for _ in range(2000):
            if time.time() > deadline:
                break
            ov = np.correlate(h_polish, 1.0 - h_polish, 'full')
            cur = float((ov * (2.0 / n)).max())
            if cur < best_c5 - 1e-9:
                best_c5 = cur
                best_h = h_polish.copy()
            shifted = ov - ov.max()
            w = np.exp(shifted / 0.01)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros(len(w))
                w[int(np.argmax(ov))] = 1.0
            else:
                w /= ws
            g = weighted_grad(h_polish, w)
            v = 0.95 * v - alpha * g
            h_polish = project_with_sum(h_polish + v, target)
            new_c = c5_value(h_polish)
            if new_c > cur + 1e-8:
                alpha *= 0.9
                v *= 0.5
                if alpha < 1e-10:
                    break

    if best_h is None:
        n = 85
        best_h = project_with_sum(parabolic_start(n), n * 0.5)
        best_c5 = c5_value(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_with_sum(best_h, len(best_h) * 0.5)
    best_c5 = c5_value(best_h)

    return best_h, best_c5, len(best_h)
