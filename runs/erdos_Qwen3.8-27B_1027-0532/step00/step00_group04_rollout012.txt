        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())


def sa_phase(h, n, target_sum, t_end, T_init, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    T_floor = 0.0001
    cooling = 0.9998
    no_improve = 0

    while time.time() < t_end:
        h_trial = h.copy()
        r = rng.random()
        if r < 0.03:
            k = rng.randint(10, 30)
            idx = rng.choice(n, size=k, replace=False)
            h_trial[idx] += rng.uniform(-0.8, 0.8, size=k)
        elif r < 0.25:
            k = rng.randint(2, 8)
            idx = rng.choice(n, size=k, replace=False)
            h_trial[idx] += rng.normal(0, 0.2, size=k)
        else:
            h_trial += rng.normal(0, 0.04, size=n)

        h_trial = project_box_hyperplane(h_trial, target_sum)
        c5_trial = compute_c5(h_trial)
        c5_cur = compute_c5(h)

        delta = c5_trial - c5_cur
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            h = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 5000:
            T = max(T, 0.015)
            no_improve = 0
        T = max(T_floor, T * cooling)

    return best_h, best_c5


def cd_phase(h, n, target_sum, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    while time.time() < t_end:
        improved = False
        c5_cur = compute_c5(h)
        for i in range(n):
            best_delta = 0.0
            best_c5_i = c5_cur
            for scale in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
                for s in [-1, 1]:
                    h_new = h.copy()
                    h_new[i] += s * scale
                    h_new = project_box_hyperplane(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5_i - 1e-12:
                        best_c5_i = c5_new
                        best_delta = s * scale
            if best_delta != 0.0:
                h[i] += best_delta
                h = project_box_hyperplane(h, target_sum)
                c5_cur = best_c5_i
                improved = True

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        if not improved:
            break

    return best_h, best_c5


def fine_cd_phase(h, n, target_sum, t_end):
    h = h.copy()

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    n = len(initial_h_values)
    target_sum = n / 2.0

    h = initial_h_values.astype(float).copy()
    h = project_box_hyperplane(h, target_sum)

    best_h, best_c5 = sa_phase(h, n, target_sum, t0 + budget_s * 0.5, 0.02, rng)
    best_h, best_c5 = cd_phase(best_h, n, target_sum, t0 + budget_s * 0.85)

    if time.time() < t0 + budget_s * 0.93:
        for _ in range(3):
            if time.time() > t0 + budget_s * 0.93:
                break
            h_r = best_h.copy()
            h_r += rng.normal(0, 0.15, size=n)
            h_r = project_box_hyperplane(h_r, target_sum)
            h_r, c5_r = sa_phase(h_r, n, target_sum, t0 + budget_s * 0.95, 0.03, rng)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    best_h, best_c5 = cd_phase(best_h, n, target_sum, t0 + budget_s * 0.97)
    best_h, best_c5 = fine_cd_phase(best_h, n, target_sum, t0 + budget_s * 0.995)

    best_h = project_box_hyperplane(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
