        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_penalty(h, n, tau, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    sm = smooth_max(corr, tau)
    cv = np.sum(h) - n / 2.0
    return sm + lam * cv * cv

def obj_slsqp(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, 8.0)

def c_eq(h):
    return np.sum(h) - len(h) / 2.0

def c_eq_jac(h):
    return np.ones(len(h))
