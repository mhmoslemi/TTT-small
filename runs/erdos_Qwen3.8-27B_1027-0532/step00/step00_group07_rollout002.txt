from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    log_sum = max_corr + np.log(np.sum(np.exp(shifted)))
    return float(log_sum / beta * dx)

def smooth_c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        # Term 1: (1-h[j-k]) for valid j (j-k in [0, n-1])
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += w * (1.0 - h[np.arange(js, je) - k])
        # Term 2: -h[j+k] for valid j (j+k in [0, n-1])  
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] += w * (-h[np.arange(js2, je2) + k])
    
    grad *= dx
    return grad

    return np.sum(h) - len(h) / 2.0

def _sum_jac(h):
    return np.ones(len(h))

        return np.clip(h, 0, 1)

        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)

                    if diff > 0:
                        other_mask = np.ones(n, dtype=bool)
                        other_mask[j] = False
                        others = np.where(other_mask)[0]
                        rooms = 1.0 - h[others]
                        if np.sum(rooms) < diff:
                            continue
                        # Distribute to the one with most room
                        best_other = others[np.argmax(rooms)]
                        h[best_other] += diff
                    else:
                        other_mask = np.ones(n, dtype=bool)
                        other_mask[j] = False
                        others = np.where(other_mask)[0]
                        if np.sum(h[others]) < -diff:
                            continue
                        best_other = others[np.argmax(h[others])]
                        h[best_other] += diff
                    if np.any(h < -1e-10) or np.any(h > 1 + 1e-10):
                        h[j] = best_val
                        np.sum(h)  # revert
                        continue
                    h = np.clip(h, 0, 1)
                
                c5_candidate = compute_c5(h)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_val = candidate
                    # Keep the modification
                else:
                    h[j] = best_val
                    # Need to revert the other element change too
                    # This is getting complicated...
            h[j] = best_val
            # Re-fix sum
            h = repair_sum(h, target)
            if compute_c5(h) < best_c5:
                improved = True
            best_c5 = compute_c5(h)
        if not improved:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    log_sum = max_corr + np.log(np.sum(np.exp(shifted)))
    return float(log_sum / beta * dx)

def smooth_c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += w * (1.0 - h[np.arange(js, je) - k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] += w * (-h[np.arange(js2, je2) + k])
    grad *= dx
    return grad

def _sum_constraint(h):
    return np.sum(h) - len(h) * 0.5

def _sum_jac(h):
    return np.ones(len(h))

def repair_sum(h, target):
    h = np.array(h, dtype=float)
    n = len(h)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def try_optimize(h0, beta, maxiter=500):
    n = len(h0)
    target = n * 0.5
    h0 = repair_sum(np.array(h0, dtype=float), target)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac}]
    result = minimize(
        smooth_c5, h0, jac=smooth_c5_grad, args=(beta,),
        method='SLSQP', bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    if result.x is not None:
        h_opt = repair_sum(result.x, target)
        return h_opt, compute_c5(h_opt)
    return h0, compute_c5(h0)

def try_rounding(h, target):
    n = len(h)
    candidates = []
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    if abs(s - target) < 1e-10:
        candidates.append(h_bin.copy())
    else:
        if s < target:
            needed = target - s
            zeros = np.where(h_bin == 0)[0]
            if len(zeros) > 0:
                order = zeros[np.argsort(-h[zeros])]
                for idx in order[:3]:
                    hc = h_bin.copy()
                    hc[idx] = needed
                    if 0 <= hc[idx] <= 1:
                        candidates.append(hc)
                candidates.append(h_bin.copy())
                h_bin2 = h_bin.copy()
                h_bin2[order[0]] = 0.5
                candidates.append(repair_sum(h_bin2, target))
        else:
            needed = s - target
            ones = np.where(h_bin == 1)[0]
            if len(ones) > 0:
                order = ones[np.argsort(h[ones])]
                for idx in order[:3]:
                    hc = h_bin.copy()
                    hc[idx] = 1.0 - needed
                    if 0 <= hc[idx] <= 1:
                        candidates.append(hc)
                h_bin2 = h_bin.copy()
                h_bin2[order[0]] = 0.5
                candidates.append(repair_sum(h_bin2, target))
    best_h = None
    best_c5 = float('inf')
    for hc in candidates:
        hc = repair_sum(hc, target)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    return best_h, best_c5

def local_refine(h, target, n_iter=200):
    n = len(h)
    h = np.array(h, dtype=float)
    step = 0.02
    for it in range(n_iter):
        improved = False
        j = it % n
        best_val = h[j]
        best_c5 = compute_c5(h)
        for delta in [0.01, 0.005, 0.001, -0.001, -0.005, -0.01, 0.02, 0.05, -0.02, -0.05]:
            new_val = h[j] + delta
            if new_val < 0 or new_val > 1:
                continue
            h[j] = new_val
            h = repair_sum(h, target)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_val = new_val
                improved = True
        h[j] = best_val
        h = repair_sum(h, target)
        if not improved and it > 50:
            if it > 100:
                break
    return h, compute_c5(h)
