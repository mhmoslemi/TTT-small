import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def sa_optimize(h0, rng, t_limit, max_stall=30000):
    n = len(h0)
    h = h0.astype(np.float64).copy()
    cur = compute_c5(h)
    best = h.copy()
    best_val = cur
    T = max(0.004, cur * 0.02)
    T_floor = 1e-10
    cool_rate = 0.99995
    t0 = time.time()
    stall = 0
    while time.time() - t0 < t_limit:
        if stall > max_stall:
            break
        oi = np.where(h > 0.5)[0]
        zi = np.where(h < 0.5)[0]
        a = oi[rng.integers(len(oi))]
        b = zi[rng.integers(len(zi))]
        if a == b:
            continue
        h[a] = 0.0; h[b] = 1.0
        nv = compute_c5(h)
        d = nv - cur
        if d < -1e-15:
            acc = True
        elif T > T_floor and d > 0:
            r = -d / T
            acc = r < 500 and rng.random() < math.exp(-r)
        else:
            acc = False
        if acc:
            cur = nv; stall = 0
            if nv < best_val:
                best_val = nv; best = h.copy()
        else:
            h[a] = 1.0; h[b] = 0.0; stall += 1
        T = max(T * cool_rate, T_floor)
    return best, best_val

def init_alternating(n):
    h = np.zeros(n)
    h[0::2] = 1.0
    tgt = n // 2
    cnt = int(h.sum())
    if cnt > tgt:
        op = np.where(h > 0.5)[0]
        h[op[tgt:]] = 0.0
    elif cnt < tgt:
        zp = np.where(h < 0.5)[0]
        h[zp[:tgt-cnt]] = 1.0
    return h

def init_sturmian(n, alpha):
    h = np.zeros(n)

        op = np.where(h > 0.5)[0]
        idx = np.argsort(np.abs(of - 0.5))[:cnt-tgt]
        h[op[idx]] = 0.0

        zp = np.where(h < 0.5)[0]
        idx = np.argsort(np.abs(zf - 0.5))[:tgt-cnt]
        h[zp[idx]] = 1.0
    return h
