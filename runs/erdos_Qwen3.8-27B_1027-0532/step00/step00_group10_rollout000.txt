    best_max = current_max

        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        temp = start_temp * (end_temp / start_temp) ** (it / n_iter)
        
        # Pick random pair
        i, j = np.random.permutation(n)[:2]
        
        delta_min = max(-h[i], h[j] - 1)
        delta_max = min(1 - h[i], h[j])
        if delta_min >= delta_max:
            continue
        
        c_i = compute_coeff(h, i, n)
        c_j = compute_coeff(h, j, n)
        c = c_i - c_j
        
        delta, new_max = find_best_delta(O, c, delta_min, delta_max)
        
        # Accept with Metropolis criterion
        if new_max < current_max:
            accept = True
        else:
            diff = new_max - current_max
            accept = np.random.random() < np.exp(-diff / (temp + 1e-10))
        
        if accept:
            h[i] += delta
            h[j] -= delta
            O = O + c * delta
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    
    return best_h, best_max

        return h
    # Uniform adjustment
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # If still not exact, do greedy fix
    diff = target_sum - np.sum(h)
    if abs(diff) >= 1e-12:
        if diff > 0:
            order = np.argsort(-h)  # smallest first (most room to increase)
        else:

                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def greedy_sweep(h, n, kp, O, time_limit):
    """One greedy sweep: try all pairs, take improving moves."""
    cur_max = np.max(O)
    
    for i in range(n - 1):
        if time.time() > time_limit:
            break
        for j in range(i + 1, n):
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-14:
                continue
            
            ci = compute_coeff(h, i, n, kp)
            cj = compute_coeff(h, j, n, kp)
            c = ci - cj
            
            delta, new_max = find_best_delta(O, c, d_lo, d_hi)
            
            if new_max < cur_max - 1e-12:
                h[i] += delta
                h[j] -= delta
                O += c * delta
                cur_max = new_max
    
    return h, O, cur_max

        if time.time() > time_limit:
            break
        
        temp = t_start * (t_end / t_start) ** (it / max(n_iter, 1))

        cj = compute_coeff(h, j, n, kp)
        c = ci - cj
        
        delta, new_max = find_best_delta(O, c, d_lo, d_hi)
        
        if new_max < cur_max:

        else:

            cur_max = new_max
            if cur_max < best_max:

                best_h = h.copy()
