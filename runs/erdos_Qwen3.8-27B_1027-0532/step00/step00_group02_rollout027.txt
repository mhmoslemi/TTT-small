    best_h = project_feasible(h0, target_sum)
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    sum_con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones(n)}

    def elapsed():
        return time.time() - start_time

    # Phase 1: SLSQP with decreasing log-sum-exp scale
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for scale in scales:
        if elapsed() > budget_s - 90:
            break

        def obj1(x):
            c = compute_c_array(x)
            return smooth_max(c, scale)

        try:
            res = minimize(obj1, best_h, method='SLSQP', bounds=bounds,
                           constraints=[sum_con], options={'maxiter': 500, 'ftol': 1e-15})
            hc = project_feasible(res.x, target_sum)
            c5 = compute_c5(hc)
            if c5 < best_c5 - 1e-14:
                best_h = hc
                best_c5 = c5
        except Exception:
            pass

    # Phase 2: Subgradient method with analytic gradients
    if elapsed() < budget_s - 60:
        alpha = 0.01
        for it in range(200000):
            if elapsed() > budget_s - 30:
                break
            if it % 2000 == 0:
                alpha = max(0.0005, 0.01 * 0.995 ** (it // 2000))

            c = compute_c_array(best_h)
            c_max = np.max(c)
            spread = c_max - np.min(c) + 1e-12
            thresh = c_max - 0.02 * spread
            active = np.where(c >= thresh)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])

            grad = np.zeros(n)
            for k in active:
                grad += grad_c_k(best_h, k, dx)
            grad /= len(active)

            # Backtracking line search
            step = alpha
            improved = False
            for _ in range(40):
                h_trial = project_feasible(best_h - step * grad, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c_max - 1e-14:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
                step *= 0.5

            if not improved and it % 5000 == 0:
                alpha = max(0.0001, alpha * 0.8)

    # Phase 3: SLSQP polish with very small scale
    if elapsed() < budget_s - 20:
        for scale in [0.0001, 0.00005, 0.00002]:
            if elapsed() > budget_s - 10:
                break
            def obj2(x):
                c = compute_c_array(x)
                return smooth_max(c, scale)
            try:
                res = minimize(obj2, best_h, method='SLSQP', bounds=bounds,
                               constraints=[sum_con], options={'maxiter': 1000, 'ftol': 1e-16})
                hc = project_feasible(res.x, target_sum)
                c5 = compute_c5(hc)
                if c5 < best_c5 - 1e-14:
                    best_h = hc
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Random restarts
    remaining = budget_s - elapsed()
    n_restarts = max(1, int(remaining / 80))
    for i in range(n_restarts):
        if elapsed() > budget_s - 15:
            break
        noise = 0.02 * (1.0 + np.random.rand())
        h_start = best_h + np.random.randn(n) * noise
        h_start = project_feasible(h_start, target_sum)

        def obj3(x):
            c = compute_c_array(x)
            return smooth_max(c, 0.005)
        try:
            res = minimize(obj3, h_start, method='SLSQP', bounds=bounds,
                           constraints=[sum_con], options={'maxiter': 300, 'ftol': 1e-14})
            hc = project_feasible(res.x, target_sum)
            c5 = compute_c5(hc)
            if c5 < best_c5 - 1e-14:
                best_h = hc
                best_c5 = c5
        except Exception:
            pass

    # Phase 5: Final subgradient pass
    if elapsed() < budget_s - 10:
        alpha = 0.005
        for it in range(100000):
            if elapsed() > budget_s - 2:
                break
            if it % 1000 == 0:
                alpha = max(0.0002, 0.005 * 0.99 ** (it // 1000))
            c = compute_c_array(best_h)
            c_max = np.max(c)
            spread = c_max - np.min(c) + 1e-12
            thresh = c_max - 0.01 * spread
            active = np.where(c >= thresh)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            grad = np.zeros(n)
            for k in active:
                grad += grad_c_k(best_h, k, dx)
            grad /= len(active)
            step = alpha
            for _ in range(30):
                h_trial = project_feasible(best_h - step * grad, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_h = h_trial
                    best_c5 = c5_trial
                    break
                step *= 0.5

    return best_h, best_c5, n
