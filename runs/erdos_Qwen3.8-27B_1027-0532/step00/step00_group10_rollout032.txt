import time

def project_to_box_sum(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(conv))

def compute_gradient_smooth(h, dx, n, beta):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(conv)
    exp_vals = np.exp(beta * (conv - max_val))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += w * dx * (1.0 - h[:n - s])
                grad[:n - s] += w * dx * (-h[s:])
        else:
            t = -s
            if t < n:
                grad[:n - t] += w * dx * (1.0 - h[t:])
                grad[t:] += w * dx * (-h[:n - t])
    return grad

def optimize_from(h0, dx, target_sum, n, beta_start, beta_end, budget_s, start_time):
    h = h0.copy()
    h = project_to_box_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    beta = beta_start
    alpha = 1.0 / (n * dx)  # initial step size
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 2.0:
            break
        
        # Compute gradient
        grad = compute_gradient_smooth(h, dx, n, beta)
        
        # Project gradient onto feasible directions (orthogonal to simplex constraint)
        # The tangent space of {sum = target, 0<=h<=1} at h is:
        # {d : sum(d) = 0, d_i = 0 where h_i is at bound}
        # Simple approach: subtract mean of grad for free variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            grad_proj = grad.copy()
            grad_proj[free] -= np.mean(grad[free])
            # Zero out gradient at bounds
            grad_proj[~free] = 0.0
        else:
            grad_proj = grad - np.mean(grad)
        
        # Line search: try decreasing step sizes
        success = False
        for trial in range(20):
            step = alpha / (trial + 1)
            h_trial = h - step * grad_proj
            h_trial = project_to_box_sum(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5 - 1e-10:
                h = h_trial
                best_h = h.copy()
                best_c5 = c5_trial
                alpha = min(alpha * 1.5, 1.0 / (n * dx * 0.1))
                success = True
                break
            elif c5_trial < compute_c5(h, dx):
                h = h_trial
                alpha = min(alpha * 1.2, 1.0 / (n * dx * 0.1))
                success = True
                break
        if not success:
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        # Increase beta periodically
        beta = min(beta * 1.05, beta_end)
    
    return best_h, best_c5

import numpy as np
import time


def project_to_box_sum(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h, dx):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(conv))


def compute_gradient_smooth(h, dx, n, beta):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(conv)
    exp_vals = np.exp(beta * (conv - max_val))
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_vals / total

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += w * dx * (1.0 - h[:n - s])
                grad[:n - s] += w * dx * (-h[s:])
        else:
            t = -s
            if t < n:
                grad[:n - t] += w * dx * (1.0 - h[t:])
                grad[t:] += w * dx * (-h[:n - t])
    return grad


def optimize_local(h, dx, target_sum, n, beta, deadline):
    h = project_to_box_sum(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    alpha = 0.3
    no_improve = 0

    while time.time() < deadline:
        grad = compute_gradient_smooth(h, dx, n, beta)

        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        grad_proj = grad.copy()
        if np.sum(free) > 1:
            grad_proj[free] -= np.mean(grad[free])
        grad_proj[~free] = 0.0

        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-14:
            break

        for trial in range(12):

            best_h = h_opt.copy()
            best_c5 = c5_opt

    best_h = project_to_box_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)

    return best_h, final_c5, n
