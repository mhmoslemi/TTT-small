    n = len(h)

    # log-sum-exp for numerical stability
    max_ov = np.max(overlap_raw)

    # For each k (lag), the contribution to grad[j] from weight[k] is:
    # weight[k] * (1-h[j-k]) * I[j-k in [0,n) and j in valid range for k]
    # - weight[k] * h[j+k] * I[j+k in [0,n) and j in valid range for k]

    # For term 1: sum_k weights[k] * (1-h[j-k]) * I[valid]
    #   = sum over valid (j,k) pairs of weights[k] * (1-h[j-k])
    #   For fixed j: sum over k where j-k is in [0,n-1] and (j,k) is a valid pair
    #   Valid pair means: max(0,-k) <= j-k < min(n, n-k) i.e. 0 <= j-k < n (since j is always valid)
    #   Wait, I need to re-derive. The valid range for n (= j-k) in the sum for lag k is:
    #   max(0,-k) <= n < min(n, n-k)
    #   Since n = j-k: max(0,-k) <= j-k < min(n, n-k)
    #   => j - max(0,-k) - k... this is getting complicated.

            # Term 2: j in [kk, n), contributes -h[j-kk]
            if kk < n:

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < best:
            best = c
            best_h = h.copy()
        elif temp > 1e-10:
            if np.random.random() < np.exp(-(c - best) / temp):
                pass
            else:
                h[i], h[j] = h[j], h[i]
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best


def gen_thue_morse(n):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    n1 = int(tm.sum())
    need = n // 2
    if n1 > need:
        oi = np.where(tm == 1.0)[0]
        tm[np.random.choice(oi, n1 - need, replace=False)] = 0.0
    elif n1 < need:
        zi = np.where(tm == 0.0)[0]
        tm[np.random.choice(zi, need - n1, replace=False)] = 1.0
    return tm


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    best_c5 = 1e10
    best_h = None
    best_n = 0

    h_init = None
    n_init = 0
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
    except NameError:
        pass

    if h_init is not None:
        c = compute_c5(h_init)
        best_c5 = c
        best_h = h_init.copy()
        best_n = n_init

        deadline1 = t0 + budget_s * 0.2
        for tau in [5.0, 10.0, 20.0, 50.0, 100.0]:
            if time.time() > deadline1:
                break
            ho, c = optimize_cont(h_init, n_init, tau, 0.08, 2000, 0.003)
            if c < best_c5:
                best_c5 = c
                best_h = ho.copy()
                best_n = n_init

    n_list = [20, 24, 28, 32, 36, 40, 42, 44, 48, 52, 56, 60, 64, 72, 80, 96]
    deadline_main = t0 + budget_s * 0.85

    for n in n_list:
        if time.time() > deadline_main:
            break
        if n % 2 != 0:
            continue

        tm = gen_thue_morse(n)
        ho, c = optimize_cont(tm, n, 10.0, 0.05, 1500, 0.005)
        if c < best_c5:
            best_c5 = c
            best_h = ho.copy()
            best_n = n

        hb = to_binary(ho, n)
        remaining = deadline_main - time.time()
        n_sa = min(10000, max(2000, int(remaining * 8)))
        hbo, c = binary_sa(hb, n, n_sa, 0.05, 0.001)
        if c < best_c5:
            best_c5 = c
            best_h = hbo.copy()
            best_n = n

        if time.time() < deadline_main:
            ho2, c2 = optimize_cont(hbo, n, 100.0, 0.01, 1000, 0.005)
            if c2 < best_c5:
                best_c5 = c2
                best_h = ho2.copy()
                best_n = n

        if time.time() < deadline_main:
            hr = np.zeros(n)
            hr[np.random.choice(n, n // 2, replace=False)] = 1.0
            hro, c = binary_sa(hr, n, 4000, 0.03, 0.001)
            if c < best_c5:
                best_c5 = c
                best_h = hro.copy()
                best_n = n

    if best_h is not None and time.time() < t0 + budget_s * 0.97:
        n = best_n
        for tau in [50.0, 100.0, 200.0]:
            if time.time() > t0 + budget_s * 0.97:
                break
            ho, c = optimize_cont(best_h, n, tau, 0.02, 2000, 0.005)
            if c < best_c5:
                best_c5 = c
                best_h = ho.copy()

        if np.max(best_h) > 0.99:
            hbo, c = binary_sa(best_h, n, 15000, 0.01, 0.0)
        else:
            hb = to_binary(best_h, n)
            hbo, c = binary_sa(hb, n, 15000, 0.02, 0.0)
        if c < best_c5:
            best_c5 = c
            best_h = hbo.copy()

        if time.time() < t0 + budget_s * 0.99:
            ho, c = optimize_cont(best_h, n, 300.0, 0.01, 1500, 0.01)
            if c < best_c5:
                best_c5 = c
                best_h = ho.copy()

    if best_h is None:
        n = 42
        best_h = np.full(n, 0.5)
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
