            h_start = 0.5 + 0.35 * np.random.randn(n)
            h_start = np.convolve(h_start, np.ones(3) / 3.0, mode='same')
        elif r % 4 == 2:
            half = n // 2
            h_start = np.concatenate([np.ones(half), np.zeros(half)])
            h_start = np.roll(h_start, np.random.randint(n))
        else:
            h_start = np.random.uniform(0.0, 1.0, size=n)

        h_start = project_feasible(h_start, n, target_sum)
        c5_start = compute_c5(h_start, dx)
        if c5_start < best_c5:
            best_h, best_c5 = h_start.copy(), c5_start

        _obj_temp = 0.0
        try:
            res = minimize(obj_func, h_start, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-12})
            hc = project_feasible(res.x, n, target_sum)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                best_h, best_c5 = hc.copy(), c5c
        except Exception:
            pass

    # Phase 3: Higher resolution attempts
    for n_try in [160, 240, 320, 400]:
        if time_left() < 120:
            break
        if n_try % 2 != 0:
            continue

        _obj_n = n_try
        _obj_target_sum = n_try / 2.0
        _obj_dx = 2.0 / n_try
        _obj_temp = 0.0

        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        h_interp = np.interp(np.linspace(0, len(best_h) - 1, n_try),
                             np.arange(len(best_h)), best_h)
        h_interp = project_feasible(h_interp, n_try, ts_try)

        constraints_try = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
        bounds_try = [(0.0, 1.0)] * n_try

        for temp in [0.5, 0.1, 0.0]:
            if time_left() < 60:
                break
            _obj_temp = temp
            try:
                res = minimize(obj_func, h_interp, method='SLSQP', bounds=bounds_try,
                              constraints=constraints_try,
                              options={'maxiter': 400, 'ftol': 1e-12})
                hc = project_feasible(res.x, n_try, ts_try)
                c5c = compute_c5(hc, dx_try)
                if c5c < best_c5:
                    best_h, best_c5, best_n = hc.copy(), c5c, n_try
                    h_interp = hc
            except Exception:
                pass

    # Phase 4: Final polish
    if time_left() > 30:
        best_n = len(best_h)
        _obj_n = best_n
        _obj_target_sum = best_n / 2.0
        _obj_dx = 2.0 / best_n
        _obj_temp = 0.0

        constraints_final = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
        bounds_final = [(0.0, 1.0)] * best_n

        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds_final,
                          constraints=constraints_final,
                          options={'maxiter': 1000, 'ftol': 1e-15})
            hc = project_feasible(res.x, best_n, best_n / 2.0)
            c5c = compute_c5(hc, 2.0 / best_n)
            if c5c < best_c5:
                best_h, best_c5 = hc.copy(), c5c
        except Exception:
            pass

    # Ensure exact feasibility
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
