def project_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_m = np.sum(mask)
        if n_m == 0:
            break
        step = diff / n_m
        if diff > 0:
            max_step = 1.0 - np.max(h[mask]) + 1e-12
            step = min(step, max_step)
        else:
            min_step = -(np.min(h[mask]) + 1e-12)
            step = max(step, min_step)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

    return np.max(c_vals)

def compute_all_c5(h, dx):

    m = np.max(c_vals)
    shifted = c_vals - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

    sm = smooth_max_fn(c_vals, beta)

    return sm + penalty

def constraint_sum(x, target_sum):
    return np.sum(x) - target_sum

def binarize(h, target_sum):
    """Convert to binary: pick the top target_sum values to be 1"""
    threshold = np.sort(h)[-int(target_sum)]
    h_bin = (h >= threshold).astype(float)
    # Handle ties
    n_ones = int(np.sum(h_bin))
    if n_ones > target_sum:
        # Need to reduce
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        # Remove the ones with smallest h values
        sorted_by_h = ones_idx[np.argsort(h[ones_idx])]
        for idx in sorted_by_h[:excess]:
            h_bin[idx] = 0.0
    elif n_ones < target_sum:

        zeros_idx = np.where(h_bin == 0.0)[0]
        sorted_by_h = zeros_idx[np.argsort(-h[zeros_idx])]
        for idx in sorted_by_h[:deficit]:
            h_bin[idx] = 1.0

            hc = project_sum(res.x, _target_sum)
            c5 = compute_c5(hc, _dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
            if remaining() > 12:
                hb2 = binarize(hc, _target_sum)
                deadline_ls = t0 + remaining() - 8
                hb2, c5b = local_search_binary(hb2, n, _dx, max_rounds=20, deadline=deadline_ls)
                if c5b < best_c5:
                    best_c5 = c5b
                    best_h = hb2

    # Phase 4: Double swap search from best binary
    if remaining() > 20:
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            bs = best_h.copy()
        else:
            bs = binarize(best_h, _target_sum)
        deadline_ds = t0 + remaining() - 5
        bs, c5_ds = double_swap_search(bs, n, _dx, max_iters=5000, deadline=deadline_ds)
        if c5_ds < best_c5:
            best_c5 = c5_ds
            best_h = bs

    # Phase 5: Final single-swap polish
    if remaining() > 5:
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            fs = best_h.copy()
        else:
            fs = binarize(best_h, _target_sum)
        deadline_fs = t0 + remaining() - 1
        fs, c5_fs = local_search_binary(fs, n, _dx, max_rounds=30, deadline=deadline_fs)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = fs

    # Phase 6: Try higher resolution if time allows
    if remaining() > 120:
        n2 = 172
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h_up = np.interp(np.linspace(0, 2, n2, endpoint=False),
                         np.linspace(0, 2, n, endpoint=False), best_h)
        h_up = project_sum(h_up, ts2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            _n_pts = n2
            _dx = dx2
            _target_sum = ts2
            _beta = 25.0
            bounds2 = [(0.0, 1.0)] * n2
            _con_fn_old = _con_fun
            res2 = minimize(_obj_fn, h_up, method='SLSQP', bounds=bounds2,
                           constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res2.x is not None:
                hc2 = project_sum(res2.x, ts2)
                c5_2 = compute_c5(hc2, dx2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = hc2
                    n = n2
        _n_pts = n
        _dx = 2.0 / n
        _target_sum = n / 2.0

    return best_h, best_c5, n
