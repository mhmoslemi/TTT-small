import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_constraint(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            remaining = diff
            for idx in order:
                room = 1.0 - h[idx]
                if room <= 1e-12:
                    break
                add = min(remaining, room)
                h[idx] += add
                remaining -= add
                if remaining < 1e-10:
                    break
        else:
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                avail = h[idx]
                if avail <= 1e-12:
                    break
                rem = min(remaining, avail)
                h[idx] -= rem
                remaining -= rem
                if remaining < 1e-10:
                    break
    return h


def local_search(h, n, rng, max_iter, start_temp, end_temp, deadline):

    stall_count = 0

            break

def interpolate_h(h_from, n_from, n_to, rng):
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_from, period=2.0)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    h_interp = fix_constraint(h_interp, n_to)
    return h_interp


def make_sine_init(n, freq, amplitude, phase, rng):
    x = np.arange(n, dtype=float)
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * freq * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    h = fix_constraint(h, n)
    return h


def make_random_init(n, rng):
    h = rng.uniform(0.1, 0.9, size=n)
    h = fix_constraint(h, n)
    return h
