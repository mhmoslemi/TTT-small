    h_dst = np.clip(h_dst, 0.0, 1.0)
    return project_feasible(h_dst, n_dst)


def smooth_objective(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    dx = 2.0 / n
    return dx * (cmax + np.log(np.sum(np.exp(T * (corr - cmax)))) / T)


def smooth_gradient(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    exp_vals = np.exp(T * (corr - cmax))
    sw = np.sum(exp_vals)
    weights = exp_vals / sw
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        w = weights[d + n - 1]
        if w < 1e-10:
            continue
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += w * (1.0 - h[idx + d])
        grad[idx + d] -= w * h[idx]
    return dx * grad


def projected_gradient(h, n, T, alpha, max_iter, t_start, time_limit):
    h = h.copy()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        grad = smooth_gradient(h, n, T)
        h_new = project_feasible(h - alpha * grad, n)
        delta = np.max(np.abs(h_new - h))
        h = h_new
        if delta < 1e-14:
            break
    return h


def try_slsqp(h0, n, T_use, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    constraint = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                   'jac': lambda h: np.ones(n)}]
    res = minimize(
        lambda h: smooth_objective(h, n, T_use),
        h0,
        jac=lambda h: smooth_gradient(h, n, T_use),
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = initial_h_values.astype(float).copy()
    best_n = len(best_h)
    best_c5 = evaluate_c5(best_h, best_n)

    def time_left():
        return max(0.0, budget_s - (time.time() - t0))

    T = 60.0

    # Phase 1: Optimize on original grid with SLSQP
    h_opt = try_slsqp(best_h, best_n, T, maxiter=5000)
    c5_opt = evaluate_c5(h_opt, best_n)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Projected gradient polish on original grid
    if time_left() > 50:
        h_pg = projected_gradient(best_h, best_n, T, alpha=0.002,
                                  max_iter=20000, t_start=time.time(),
                                  time_limit=min(50, time_left() * 0.3))
        c5_pg = evaluate_c5(h_pg, best_n)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 2: Try finer grids
    for n_try in [100, 140, 180, 240, 320]:
        if time_left() < 70:
            break

        h0 = interp_to_grid(best_h, best_n, n_try)

        if time_left() > 50:
            h_opt = try_slsqp(h0, n_try, T, maxiter=4000)
            c5_opt = evaluate_c5(h_opt, n_try)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

        if time_left() > 40:
            h_pg = projected_gradient(best_h, best_n, T, alpha=0.001,
                                      max_iter=30000, t_start=time.time(),
                                      time_limit=min(40, time_left() * 0.4))
            c5_pg = evaluate_c5(h_pg, best_n)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg

    # Phase 3: Random restarts on best grid
    for attempt in range(5):
        if time_left() < 50:
            break
        h_rand = np.clip(0.5 + 0.3 * rng.randn(best_n), 0.0, 1.0)
        h_rand = project_feasible(h_rand, best_n)
        # Blend with current best
        h_start = 0.7 * best_h + 0.3 * h_rand
        h_start = project_feasible(h_start, best_n)
        h_opt = try_slsqp(h_start, best_n, T, maxiter=2000)
        c5_opt = evaluate_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Final polish with higher T
    if time_left() > 30:
        h_opt = try_slsqp(best_h, best_n, T * 2.0, maxiter=2000)
        c5_opt = evaluate_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    return best_h, best_c5, best_n
