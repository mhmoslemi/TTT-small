    phase2_end = start_time + budget_s * 0.80
    best_h, best_c5 = optimize_from(best_h, n, rng, phase1_end, phase2_end, initial_step=0.015)

    phase3_end = start_time + budget_s * 0.98
    n_candidates = [n + 1, n + 2, n + 3, 100, 126, 128]
    n_candidates = [x for x in n_candidates if x != n and x > 10]
    n_candidates = list(dict.fromkeys(n_candidates))

    remaining = max(phase3_end - time.time(), 0)
    time_per = remaining / max(1, len(n_candidates))

    for n_try in n_candidates:
        if time.time() >= phase3_end or time_per < 5.0:
            break
        n_t = n_try
        ts = n_t / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_t)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_to_feasible(h_try, ts)
        t_end = min(time.time() + time_per, phase3_end)
        local_h, local_c5 = optimize_from(h_try, n_t, rng, time.time(), t_end, initial_step=0.02)
        if local_c5 < best_c5:
            best_h = local_h.copy()
            best_c5 = local_c5

    if time.time() < start_time + budget_s * 0.99:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_feasible(best_h, ts_final)
        best_c5 = evaluate_c5(best_h, n_final)

    best_c5 = evaluate_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
