import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, dx, mu):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    s = (corr - mc) / mu
    s = np.clip(s, -200, 200)
    return mc + mu * np.log(np.sum(np.exp(s)))

def lse_grad(h, dx, mu):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    s = (corr - mc) / mu
    s = np.clip(s, -200, 200)
    es = np.exp(s)
    total = np.sum(es)
    w = es / total
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            la = -lag
            grad[la:] += wk * (1.0 - h[:n - la])
            grad[:n - la] -= wk * h[la:]
    return grad

def lse_obj_grad(h, dx, mu):
    return lse_obj(h, dx, mu), lse_grad(h, dx, mu)

def project_sum(h, target_sum):

    best_c5 = compute_c5(h, dx)

            break
        for _ in range(max_iter_per_mu):

                break
            def fun(x):
                return lse_obj(x, dx, mu)
            def jac(x):
                return lse_grad(x, dx, mu)
            
            res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
            
            if res.fun < 1e6:
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = project_sum(h_new, target_sum)

                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
    
    return best_h, best_c5
