Let me also think about the pair ordering. Instead of trying all pairs randomly, I can be smarter:
- Identify the "worst" lag (where correlation is maximum)
- Try to reduce correlation at that lag
- This means: for the worst lag s, try to make h[n] and h[n+s] more similar

Actually, let me think about this more carefully. The correlation at lag s is:
c_s = sum_n h[n] * (1-h[n+s])

To decrease c_s, I want to either:
- Decrease h[n] where h[n+s] = 0 (reduces the product)
- Increase h[n+s] where h[n] = 1 (reduces the product)

So for the worst lag s, I should try to:
- Move mass from positions n where h[n]=1 and h[n+s]=0 to positions n+s

This is a targeted move! Let me implement this.

import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-15:
            h += (diff / total_room) * room
    else:
        total_val = np.sum(h)
        if total_val > 1e-15:
            h += (diff / total_val) * h
    return np.clip(h, 0.0, 1.0)


def targeted_local_search(h, n, time_limit, step=0.05):
    start = time.time()
    h = h.astype(float).copy()
    best_c = compute_c5(h, n)
    target = n / 2.0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_idx = np.argmax(corr)
        worst_lag = worst_idx - (n - 1)
        improved = False
        best_move_c = best_c

        if worst_lag >= 0:
            for m in range(n - worst_lag):
                if h[m] > 0.01 and h[m + worst_lag] < 0.99:
                    d = min(step, h[m], 1.0 - h[m + worst_lag])
                    if d < 1e-8:
                        continue
                    trial = h.copy()
                    trial[m] -= d
                    trial[m + worst_lag] += d
                    c = compute_c5(trial, n)
                    if c < best_move_c:
                        best_move_c = c
                        best_move = (m, m + worst_lag, d)
        else:
            s = -worst_lag
            for m in range(s, n):
                if h[m] > 0.01 and h[m - s] < 0.99:
                    d = min(step, h[m], 1.0 - h[m - s])
                    if d < 1e-8:
                        continue
                    trial = h.copy()
                    trial[m] -= d
                    trial[m - s] += d
                    c = compute_c5(trial, n)
                    if c < best_move_c:
                        best_move_c = c
                        best_move = (m, m - s, d)

        found = False
        if 'best_move' in dir() and best_move is not None:

            h[m1] -= d
            h[m2] += d
            best_c = best_move_c
            improved = True
            found = True

            for _ in range(200):
                i = np.random.randint(n)

                    continue

def simulated_annealing(h, n, time_limit, T_start=0.05, T_end=0.001):
    start = time.time()
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    target = n / 2.0

    t0 = time.time()
    for _ in range(200):
        _ = compute_c5(h, n)
    t1 = time.time()
    ips = 200.0 / max(t1 - t0, 1e-10)
    available = time_limit - (time.time() - start)
    n_iter = int(ips * available * 0.85)
    n_iter = max(n_iter, 1000)

    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        T = T_start * (T_end / T_start) ** frac

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = np.random.uniform(0.02, 0.15)
        trial = h.copy()
        trial[i] -= delta
        trial[j] += delta
        trial = np.clip(trial, 0.0, 1.0)
        trial = repair_sum(trial, target)
        c = compute_c5(trial, n)

        if c < current_c or np.random.random() < np.exp(-(c - current_c) / max(T, 1e-12)):
            h = trial
            current_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
