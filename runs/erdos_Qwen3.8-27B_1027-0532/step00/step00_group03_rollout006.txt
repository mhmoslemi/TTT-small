import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, temperature):
    v = corr / temperature
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    s = np.sum(exp_v)
    return temperature * (v_max + np.log(s)), exp_v / s

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        if lag >= 0:
            count = n - lag
            if count > 0:
                grad[:count] += w * (1.0 - h[lag:lag + count])
                grad[lag:] += w * (-h[:count])
        else:
            L = -lag
            count = n - L
            if count > 0:
                grad[L:] += w * (1.0 - h[:count])
                grad[:count] += w * (-h[L:L + count])
    grad *= dx
    return grad

    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    sm, _ = smooth_max(corr, temperature)
    target = n / 2.0
    sum_err = np.sum(h) - target
    return sm * dx + lambda_sum * sum_err ** 2

def grad_fn(h, temperature, lambda_sum):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    sm, weights = smooth_max(corr, temperature)
    grad = compute_gradient(h, weights, dx)
    target = n / 2.0
    sum_err = np.sum(h) - target
    grad += 2.0 * lambda_sum * sum_err
    return grad

    return np.repeat(h, factor)
