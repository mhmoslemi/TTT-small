import time

def compute_corr(h, n_points):
    """Compute all correlation values (unscaled by dx)."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return corr

def get_c5(corr, n_points):
    """Get C5 value from correlation array."""
    dx = 2.0 / n_points
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set: 0<=h[i]<=1, sum(h)=n/2."""
    target = n_points / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n_points, active_lags):
    """Compute gradient of f(m) averaged over active_lags."""
    n = n_points
    grad = np.zeros(n)
    for m in active_lags:
        # m is the actual lag (can be negative)
        for j in range(n):
            # Contribution from (1-h[j-m]) if j-m is valid
            jm = j - m
            if 0 <= jm < n:
                grad[j] += (1.0 - h[jm])
            # Contribution from -h[j+m] if j+m is valid
            jm = j + m
            if 0 <= jm < n:
                grad[j] -= h[jm]
    grad /= len(active_lags)
    return grad

def logsumexp_gradient(h, n_points, beta):
    """Compute gradient of smooth max approximation using log-sum-exp."""
    n = n_points
    corr = compute_corr(h, n)
    # Scale for numerical stability
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    # Gradient is weighted average of individual gradients
    # f(m) gradient for lag m at position j:
    # (1-h[j-m])*1[valid] - h[j+m]*1[valid]
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for idx in range(n_lags):
        m = idx - (n - 1)  # actual lag
        if weights[idx] < 1e-15:
            continue
        for j in range(n):
            jm = j - m
            if 0 <= jm < n:
                grad[j] += weights[idx] * (1.0 - h[jm])
            jm = j + m
            if 0 <= jm < n:
                grad[j] -= weights[idx] * h[jm]
    return grad
