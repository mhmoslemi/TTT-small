    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h, n):
    """Repair h to satisfy constraints: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0, 1)

    target = n / 2.0
    if s > 0:

        h = h * (target / s)
        h = np.clip(h, 0, 1)
        # If still not exact, adjust
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            # Distribute the difference
            if diff > 0:

                mask = h < 1.0
                if np.any(mask):

                    h[mask] = np.clip(h[mask] - sub, 0, 1)

    new_scores = scores.copy()

            continue
        # Select three distinct individuals

def project_feasible(h, n):

        else:
