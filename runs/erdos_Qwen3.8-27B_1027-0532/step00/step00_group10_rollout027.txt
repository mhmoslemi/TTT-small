    lo = y.min() - 1.0
    hi = y.max() + 1.0

        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _compute_overlaps(h):
    """Compute all overlap values (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _subgradient(h, shift):
    """Compute subgradient of overlap at given shift w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n - shift]
    else:

        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]

            result[idx[i]] = 0.5

        result[mid_idx] = np.clip(result[mid_idx] + diff, 0.0, 1.0)

        h_new[j] -= delta

Then the gradient is sum_k w[k] * subgradient(h, s_k).

import numpy as np
import time

def _project_feasible(y, target_sum):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _subgrad(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        g[:n-shift] += 1.0 - h[shift:]
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    return g

def _c5(h, dx):
    return np.max(_overlaps(h)) * dx

def _snap(h, target_sum):
    n = len(h)
    idx = np.argsort(h)
    r = np.zeros(n)
    half = n // 2  # 48
    # bottom half → 0, top half → 1, middle → 0.5
    r[idx[:half]] = 0.0
    r[idx[half:n-half]] = 0.5
    r[idx[n-half:]] = 1.0
    # Fix sum
    cur = np.sum(r)
    diff = target_sum - cur
    if abs(diff) > 1e-10:
        # Adjust middle values
        mid_idx = idx[half:n-half]
        for i in mid_idx:
            adj = min(abs(diff), abs(1.0 - r[i]))
            if diff > 0:
                r[i] += adj
                diff -= adj
            else:
                r[i] -= adj
                diff += adj
            if abs(diff) < 1e-10:
                break
    return r

def _random_feasible(rng, n, target_sum):
    h = rng.rand(n)
    return _project_feasible(h, target_sum)
