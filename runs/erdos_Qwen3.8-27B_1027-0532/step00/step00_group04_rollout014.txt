            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
    return best_h, best_c5


def upscale(h, n_from, n_to):
    x_from = np.linspace(0.0, 2.0, n_from)
    x_to = np.linspace(0.0, 2.0, n_to)
    h_up = np.interp(x_to, x_from, h)
    return h_up


def random_feasible(n, rng):
    h = rng.random(n)
    h = repair_vector(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h0 = repair_vector(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: COBYLA from initial
    if budget_s > 120:
        coby_budget = min(60.0, (budget_s - 120.0) * 0.1)
        try:
            res = minimize(coby_obj, h0, args=(n,), method='COBYLA',
                           options={'maxiter': int(coby_budget * 40), 'rhobeg': 0.003,
                                    'catol': 1e-10})
            if res.fun < best_c5:
                candidate = repair_vector(res.x, n)
                cand_c5 = compute_c5(candidate, n)
                if cand_c5 < best_c5:
                    best_h = candidate
                    best_c5 = cand_c5
        except Exception:
            pass

    # Phase 2: Intensive greedy pair search
    remaining = deadline - time.time()
    if remaining > 10:
        n_trials = int(remaining * 20000)
        best_h, best_c5 = greedy_pairs(best_h, n, n_trials, rng, 10, deadline)

    # Phase 3: Simulated annealing
    remaining = deadline - time.time()
    if remaining > 60:
        sa_time = remaining * 0.5
        sa_deadline = time.time() + sa_time
        n_iter = int(sa_time * 100000)
        best_h, best_c5 = sa_optimize(best_h, n, rng, 0.005, 0.00001,
                                      n_iter, sa_deadline)

    # Phase 4: Final greedy polish
    remaining = deadline - time.time()
    if remaining > 5:
        n_trials = int(remaining * 20000)
        best_h, best_c5 = greedy_pairs(best_h, n, n_trials, rng, 15, deadline)

    # Phase 5: Random restarts
    remaining = deadline - time.time()
    n_restarts = 3
    if remaining > 90:
        per_restart = remaining / (n_restarts * 2.0)
        for r in range(n_restarts):
            if time.time() > deadline:
                break
            h_r = random_feasible(n, rng)
            c5_r = compute_c5(h_r, n)
            if c5_r < best_c5 + 0.05:
                t_end = min(time.time() + per_restart * 0.5, deadline - 10)
                h_r, c5_r = greedy_pairs(h_r, n, int(per_restart * 0.3 * 10000),
                                         rng, 8, t_end)
                if c5_r < best_c5:
                    best_h = h_r
                    best_c5 = c5_r

    # Phase 6: Try doubled grid
    remaining = deadline - time.time()
    if remaining > 120:
        n2 = n * 2
        h2 = upscale(best_h, n, n2)
        h2 = repair_vector(h2, n2)
        c5_2 = compute_c5(h2, n2)
        sub_deadline = deadline - 10
        try:
            res = minimize(coby_obj, h2, args=(n2,), method='COBYLA',
                           options={'maxiter': 3000, 'rhobeg': 0.005})
            if res.fun < c5_2:
                h2 = repair_vector(res.x, n2)
                c5_2 = compute_c5(h2, n2)
        except Exception:
            pass
        if time.time() < sub_deadline:
            n_trials = int((sub_deadline - time.time()) * 8000)
            h2, c5_2 = greedy_pairs(h2, n2, n_trials, rng, 8, sub_deadline)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            n = n2

    final_h = repair_vector(best_h, len(best_h))
    final_c5 = compute_c5(final_h, len(best_h))
    return final_h, final_c5, len(best_h)
