import numpy as np
import time

def compute_c5_and_subgrad(h):
    """Compute C5 and the subgradient (gradient of the active correlation)"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_idx = np.argmax(corr)
    c5 = corr[max_idx] * dx
    
    # Compute subgradient: gradient of corr[max_idx] w.r.t. h
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    i_start = max(0, -lag)
    i_end = min(n, n - lag)
    for i in range(i_start, i_end):
        j = i
        k = i + lag
        if 0 <= k < n:
            # d/dh[j] of h[j]*(1-h[k])
            if j == k:
                grad[j] += (1 - 2*h[j])
            else:
                grad[j] += (1 - h[k])
                grad[k] += (-h[j])
    
    return c5, grad * dx

def project_to_feasible(h, n):
    """Project h onto {h : 0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        # Project onto hyperplane
        mean_shift = (np.sum(h) - target) / n
        h = h - mean_shift
        # Project onto box
        h = np.clip(h, 0.0, 1.0)
        # Check feasibility
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
    # Final adjustment
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase some values
        mask = h < 1.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    elif diff < 0:
        mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    best_c5 = compute_c5_and_subgrad(best_h)[0]

import numpy as np
from scipy.optimize import minimize
import time

def smooth_max(values, beta):
    """Smooth approximation of max using log-sum-exp"""
    m = np.max(values)
    if m == -np.inf:
        return 0.0
    shifted = values - m
    return m + np.log(np.sum(np.exp(shifted))) / beta

def objective_and_grad(h, n, beta=50.0, penalty=100.0):
    """Objective: smooth max of correlations + penalty for sum constraint"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    
    # Smooth max
    obj = smooth_max(corr, beta)
    
    # Penalty for sum constraint
    target = n / 2.0
    sum_penalty = penalty * (np.sum(h) - target) ** 2
    
    total = obj + sum_penalty
    
    # Gradient
    # Gradient of smooth max: softmax-like weights
    m = np.max(corr)
    weights = np.exp((corr - m) * beta) / np.sum(np.exp((corr - m) * beta))
    
    # Gradient of corr[k] w.r.t. h
    grad_obj = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue

        for i in range(i_start, i_end):
            j = i
            kp = i + lag

            else:
                grad_obj[j] += weights[k] * dx * (1 - h[kp])
                grad_obj[kp] += weights[k] * dx * (-h[j])
    
    # Gradient of penalty
    grad_penalty = 2 * penalty * (np.sum(h) - target) * np.ones(n)
    
    return total, grad_obj + grad_penalty

        return project_feasible(h, n)
    elif pattern_idx == 3:
        t = np.linspace(0, 1, n)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * t)
        return project_feasible(h, n)
    elif pattern_idx == 4:
        h = np.zeros(n)
        block = n // 5
        for i in range(0, n, block * 2):
            h[i:i + block] = 1.0
        return project_feasible(h, n)
    elif pattern_idx == 5:
        h = np.random.default_rng(rng.integers(0, 10000)).random(n)
        return project_feasible(h, n)
    else:
        return np.ones(n) * 0.5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    timeout = budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        n = 95
        h0 = np.ones(n) * 0.5

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    rng = np.random.default_rng(seed)

    if time.time() - t0 < timeout - 50:
        best_h, best_c5 = subgradient_optimize(best_h, n, max_iters=80000, time_limit=40)

    for beta in [10, 30, 100, 300, 1000, 3000]:
        if time.time() - t0 > timeout - 40:
            break
        h_opt, c5_opt = lbgfs_smooth(best_h, n, beta, time_limit=20)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.time() - t0 < timeout - 30:
            best_h, best_c5 = subgradient_optimize(best_h, n, max_iters=30000, time_limit=10)

    for pi in range(6):
        if time.time() - t0 > timeout - 60:
            break
        h_struct = make_structured_h(n, pi, rng)
        h_opt, c5_opt = subgradient_optimize(h_struct, n, max_iters=30000, time_limit=10)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.time() - t0 < timeout - 40:
            h_opt2, c5_opt2 = lbgfs_smooth(best_h, n, 100, time_limit=15)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    attempts = 0
    while time.time() - t0 < timeout - 50:
        attempts += 1
        base = best_h
        scale = 0.03 + 0.12 * rng.random()
        h_pert = base + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)

        h_opt, c5_opt = subgradient_optimize(h_pert, n, max_iters=30000, time_limit=12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if time.time() - t0 < timeout - 35:
            h_opt2, c5_opt2 = lbgfs_smooth(best_h, n, 200, time_limit=12)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    if time.time() - t0 < timeout - 15:
        best_h, best_c5 = subgradient_optimize(best_h, n, max_iters=100000, time_limit=15)

    if time.time() - t0 < timeout - 5:
        h_opt, c5_opt = lbgfs_smooth(best_h, n, 500, time_limit=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
