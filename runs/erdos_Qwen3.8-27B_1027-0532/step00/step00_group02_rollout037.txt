import numpy as np
import time

def project_onto_polytope(h, S):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, dx, alpha):
    c = compute_corr(h, dx)
    mc = c.max()
    return mc + np.log(np.exp(alpha * (c - mc)).sum()) / alpha

def compute_smoothed_gradient(h, c, dx, alpha):
    N = len(h)
    mc = c.max()
    e = np.exp(alpha * (c - mc))
    w = e / e.sum()
    omh = 1.0 - h
    g = np.zeros(N)
    for j in range(2 * N - 1):
        if w[j] < 1e-13:
            continue
        k = j - (N - 1)
        if k >= 0:
            g[0:N-k] += w[j] * omh[k:N]
            g[k:N] -= w[j] * h[0:N-k]
        else:
            g[-k:N] += w[j] * omh[0:N+k]
            g[0:N+k] -= w[j] * h[-k:N]
    return g * dx

def projected_gradient_optimize(h, dx, S, alpha, max_iter, t0, t_limit):
    h = project_onto_polytope(h, S)
    f = smooth_objective(h, dx, alpha)
    step = 0.05
    stall = 0
    best_f = f
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        c = compute_corr(h, dx)
        g = compute_smoothed_gradient(h, c, dx, alpha)
        h_new = project_onto_polytope(h - step * g, S)
        f_new = smooth_objective(h_new, dx, alpha)
        if f_new < f:
            f = f_new
            h = h_new
            stall = 0
            step = min(step * 1.04, 0.3)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            stall += 1
            step *= 0.7
            if step < 1e-10 or stall > 400:
                break
    return best_h, best_f

def coordinate_descent_refine(h, dx, S, t0, t_limit):
    N = len(h)
    h = project_onto_polytope(h, S)
    c_best = compute_corr(h, dx).max()
    for outer in range(500):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(N):
            if time.time() > t_limit:
                break
            h_orig = h[i]
            candidates = [0.0, 1.0, h_orig - 0.5, h_orig - 0.1, h_orig + 0.1, h_orig + 0.5]
            candidates = sorted(set(np.clip(candidates, 0.0, 1.0).tolist()))
            if h_orig not in candidates:
                candidates.append(h_orig)
            best_val = c_best
            best_candidate = h_orig
            for cv in candidates:
                if cv == h_orig:
                    continue
                h_trial = h.copy()
                h_trial[i] = cv
                h_trial = project_onto_polytope(h_trial, S)
                c_val = compute_corr(h_trial, dx).max()
                if c_val < best_val:
                    best_val = c_val
                    best_candidate = cv
            if best_candidate != h_orig:
                h[i] = best_candidate
                h = project_onto_polytope(h, S)
                c_best = best_val
                improved = True
        if not improved:
            break
    return h, c_best

        starts.append(project_onto_polytope(h, S))

        c_r = compute_corr(h_r, dx).max()
        if c_r < c_best:

            h_best = h_r.copy()

            h_best = h_cd.copy()

    h_best = project_onto_polytope(h_best, S)
    c_best = compute_corr(h_best, dx).max()
    return h_best, c_best, N
