def project(h, n):

    target = n / 2.0
    # Bisection on lambda
    h_clamped = np.clip(h, 0, 1)

    # As lambda -> -inf: sum -> n (all 1s)
    # As lambda -> +inf: sum -> 0

        g = 1.0 - h

        grad = np.zeros(n)

            for j in range(n):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)
