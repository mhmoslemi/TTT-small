    best_c5 = compute_c5(hb)
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    np.random.shuffle(ones)
    np.random.shuffle(zeros)
    for _ in range(50):
        if time.time() > deadline:
            break
        improved = False
        for i in ones:
            if time.time() > deadline:
                return hb, best_c5
            for j in zeros:
                hb[i] = 0.0
                hb[j] = 1.0
                obj = compute_c5(hb)
                if obj < best_c5 - 1e-12:
                    best_c5 = obj
                    improved = True
                    ones = np.where(hb == 1.0)[0]
                    zeros = np.where(hb == 0.0)[0]
                    break
                else:
                    hb[i] = 1.0
                    hb[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 42
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    dl1 = t0 + budget_s * 0.3
    h1, c5_1 = optimize_lse(h_init, n, dl1, max_iter=150)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    dl2 = t0 + budget_s * 0.5
    h2, c5_2 = optimize_coords(best_h, n, dl2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    hb = binary_round(best_h, n)
    hb, c5_b = binary_local_search(hb, n, t0 + budget_s * 0.55)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = hb

    dl3 = t0 + budget_s * 0.95
    while time.time() < dl3:
        h_p = best_h.copy()
        n_pert = np.random.randint(2, 6)
        for _ in range(n_pert):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                delta = np.random.uniform(-0.4, 0.4)
                h_p[i] = np.clip(h_p[i] + delta, 0.0, 1.0)
                h_p[j] = np.clip(h_p[j] - delta, 0.0, 1.0)
        h_p = project_feasible(h_p, n)

        if time.time() < dl3:
            h_p, c5_p = optimize_lse(h_p, n, min(time.time() + 20, dl3), max_iter=60)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

        h_p = best_h.copy()
        if time.time() < dl3:
            h_p, c5_p = optimize_coords(h_p, n, min(time.time() + 15, dl3))
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

        hb = binary_round(best_h, n)
        if time.time() < dl3:
            hb, c5_b = binary_local_search(hb, n, min(time.time() + 10, dl3))
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = hb

    return (best_h, best_c5, n)
