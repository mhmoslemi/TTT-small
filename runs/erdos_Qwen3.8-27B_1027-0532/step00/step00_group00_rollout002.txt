            i = rng.integers(0, n)
            j = rng.integers(0, n)

                j = rng.integers(0, n)

            candidate[j] += delta

        if stagnant % 5000 == 0 and stagnant > 0:
            ls_h, ls_val = local_search(current, max_rounds=50)
            if ls_val < best_val:
                best = ls_h.copy()
                best_val = ls_val
            current = ls_h.copy()
            current_val = ls_val
            stagnant = 0
    
    return best, best_val

def upsample(h, factor):
    n = len(h)
    new_n = n * factor
    old_idx = np.arange(n, dtype=float)
    new_idx = np.linspace(0.0, float(n - 1), int(new_n))
    h_up = np.interp(new_idx, old_idx, h)
    return project_feasible(h_up, int(new_n) / 2.0)
