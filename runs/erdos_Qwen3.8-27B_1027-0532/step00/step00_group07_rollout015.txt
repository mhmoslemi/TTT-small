    n = len(h)

def smooth_max_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)

    half = n - 1
    grad = np.zeros(n)

        k = idx - half

        else:
