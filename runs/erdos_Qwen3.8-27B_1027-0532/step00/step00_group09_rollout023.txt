def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def project_onto_feasible(v, n):
    target = n / 2.0
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Add small uniform correction
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    best_h = h.copy()
    best_c5 = compute_c5(h, n)[0]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_onto_feasible(v, n):
    target = n / 2.0
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune sum
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # Adjust elements that are not at bounds
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, n, temperature):
    """Compute smooth max of correlation and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    max_c = float(np.max(corr))
    exp_vals = np.exp(np.clip((corr - max_c) / temperature, -100, 100))
    weights = exp_vals / exp_vals.sum()
    # Smooth max value
    smooth_val = (max_c + temperature * np.log(exp_vals.sum())) * dx
    # Gradient
    grad = np.full(n, dx)  # from the "1" in 1 - h[j+d] - h[j-d]
    d_vals = np.arange(-(n-1), n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        d = d_vals[k]
        # h[j+d]: j ranges from max(0,-d) to min(n, n-d)-1
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo < hi:
            grad[lo:hi] -= w * h[lo+d:hi+d]
        # h[j-d]: j ranges from max(0,d) to min(n, n+d)-1
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-d:hi2-d]
    return smooth_val, grad, corr * dx

def proximal_gradient_optimize(h, n, time_limit, rng, verbosity=0):
    """Run proximal gradient with smooth max."""
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    temperature = max(compute_c5(h, n) * 0.5, 0.05)
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.time() < time_limit:
        t += 1
        val, grad, corr_vals = smooth_max_grad(h, n, temperature)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_onto_feasible(h_new, n)
        
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Anneal temperature
        if t % 200 == 0:
            temperature = max(temperature * 0.9, 0.01)
        
        # Reduce lr
        if t % 500 == 0:
            lr *= 0.95
    
    return best_h, best_c5

def to_binary(h, n):
    """Convert to binary maintaining sum constraint."""
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    # If n is odd, target = n//2 (floor)
    diff = target - h_bin.sum()
    if diff > 0:
        # Need to flip diff zeros to ones
        zeros = np.where(h_bin == 0)[0]
        # Sort by original h value (descending) - those closest to 0.5 from below
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        h_bin[zeros_sorted[:int(diff)]] = 1.0
    elif diff < 0:
        # Need to flip |diff| ones to zeros
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        h_bin[ones_sorted[:int(-diff)]] = 0.0
    return h_bin

def binary_local_search(h, n, time_limit, rng):
    """Local search on binary vectors via swaps."""
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    current_max = float(np.max(corr))
    
    d_vals = np.arange(-(n-1), n)
    
    while time.time() < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        n_trials = 500
        
        for _ in range(n_trials):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            
            # Compute delta
            idx_ip = i + d_vals
            idx_im = i - d_vals
            idx_jp = j + d_vals
            idx_jm = j - d_vals
            
            delta = np.zeros(2*n-1)
            m = (idx_ip >= 0) & (idx_ip < n)
            delta[m] += h[idx_ip[m]]
            m = (idx_im >= 0) & (idx_im < n)
            delta[m] += h[idx_im[m]]
            m = (idx_jp >= 0) & (idx_jp < n)
            delta[m] -= h[idx_jp[m]]
            m = (idx_jm >= 0) & (idx_jm < n)
            delta[m] -= h[idx_jm[m]]
            
            new_corr = corr + delta * dx
            new_max = float(np.max(new_corr))
            
            if new_max < current_max - 1e-12:
                # Accept
                h[i], h[j] = h[j], h[i]
                corr = new_corr
                current_max = new_max
                if current_max < best_c5:
                    best_c5 = current_max
                    best_h = h.copy()
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
        
        if not improved:
            # Perturbation: flip 2-4 random pairs
            n_flips = rng.randint(2, 5)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]

                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i != j:
                    h[i], h[j] = h[j], h[i]
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            current_max = float(np.max(corr))
    
    return best_h, best_c5

def project_onto_feasible(v, n):
    target = n / 2.0
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
