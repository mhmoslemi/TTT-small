from scipy.optimize import minimize

def project_to_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_correlations(h):
    """Compute all correlation values f(k) = dx * correlate(h, 1-h, full)[k]"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def compute_c5(h):
    """Compute C5 value"""
    c = compute_correlations(h)
    return float(np.max(c))

def compute_gradient(h, k, n):
    """Compute gradient of f(k) w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        # h[i] appears as first factor when i+k is valid
        if 0 <= i + k < n:
            grad[i] += dx * (1.0 - h[i + k])
        # h[i] appears as second factor when i-k is valid (i.e., h[i-k]*(1-h[i]))
        if 0 <= i - k < n:
            grad[i] -= dx * h[i - k]
    return grad

            best_h = h.copy()

        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def soft_max_obj(x, mu):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cmax = np.max(c)
    return mu * np.log(np.sum(np.exp((c - cmax) / mu))) + cmax

def soft_max_grad(x, mu):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / mu)
    total = exp_c.sum()
    weights = exp_c / total
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if w < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - x[k:])
            grad[k:] -= w * x[:n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - x[:n - kk])
            grad[:n - kk] -= w * x[kk:]
    return grad

def optimize_from_start(h0, n, mu_list, max_iter=200):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    for mu in mu_list:
        try:
            res = minimize(
                soft_max_obj, h, args=(mu,),
                jac=soft_max_grad,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter, 'iprint': -1}
            )
            h = res.x
        except Exception:
            pass
        h = project_to_feasible(h, n)
    return h

def bit_flip_search(h, n, max_rounds=30, time_limit=30.0, start_time=0.0):
    h = (h > 0.5).astype(float)
    diff = int(h.sum()) - n // 2
    if diff > 0:
        ones = list(np.where(h == 1)[0])
        for _ in range(diff):
            best_i = ones[0]
            best_val = float('inf')
            for i in ones:
                h[i] = 0
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_i = i
                h[i] = 1
            h[best_i] = 0
            ones.remove(best_i)
    elif diff < 0:
        zeros = list(np.where(h == 0)[0])
        for _ in range(-diff):
            best_j = zeros[0]
            best_val = float('inf')
            for j in zeros:
                h[j] = 1
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_j = j
                h[j] = 0
            h[best_j] = 1
            zeros.remove(best_j)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for i in ones:
            if time.time() - start_time > time_limit:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    return best_h, best_c5

def greedy_binary_construct(n, rng):
    h = np.zeros(n)
    target = n // 2
    c5 = compute_c5(h)
    for _ in range(target):
        best_pos = -1
        best_c5 = float('inf')
        candidates = np.where(h == 0)[0]
        if len(candidates) == 0:
            break
        for pos in candidates:
            h[pos] = 1
            val = compute_c5(h)
            if val < best_c5:
                best_c5 = val
                best_pos = pos
            h[pos] = 0
        if best_pos >= 0:
            h[best_pos] = 1
    return h

def generate_starts(n, rng, h_init, n_init):
    starts = []
    for _ in range(6):
        hr = rng.rand(n)
        starts.append(project_to_feasible(hr, n))
    if h_init is not None and n != n_init:
        x_old = np.linspace(0, 2, n_init + 1)[:-1] + 1.0 / n_init
        x_new = np.linspace(0, 2, n + 1)[:-1] + 1.0 / n
        hi = np.interp(x_new, x_old, h_init)
        starts.append(project_to_feasible(hi, n))
    ha = np.zeros(n)
    ha[::2] = 1.0
    starts.append(project_to_feasible(ha, n))
    for block in [3, 4, 5]:
        hb = np.zeros(n)
        for i in range(n):
            if (i % (2 * block)) < block:
                hb[i] = 1.0
        starts.append(project_to_feasible(hb, n))
    hg = greedy_binary_construct(n, rng)
    starts.append(hg)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except NameError:
        h_init = None
        n_init = 0

    best_h = None
    best_c5 = float('inf')

    if h_init is not None:
        best_c5 = compute_c5(h_init)
        best_h = h_init.copy()

    mu_coarse = [0.5, 0.2, 0.1, 0.05, 0.02]
    mu_fine = [0.01, 0.005, 0.002, 0.001, 0.0005]

    n_list = []
    if n_init > 0:
        n_list.insert(0, n_init)
    for n in [24, 30, 36, 42, 48, 54, 60, 66, 72, 80, 90, 100, 120, 140]:
        if n % 2 == 0 and n not in n_list:
            n_list.append(n)

    for n in n_list:
        if time.time() - start_time > budget_s * 0.65:
            break

        starts = generate_starts(n, rng, h_init, n_init)

        for hs in starts:
            if time.time() - start_time > budget_s * 0.65:
                break
            h_opt = optimize_from_start(hs, n, mu_coarse, max_iter=150)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        if best_h is not None and len(best_h) == n:
            if time.time() - start_time < budget_s * 0.8:
                h_fine = optimize_from_start(best_h, n, mu_fine, max_iter=200)
                c5 = compute_c5(h_fine)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_fine.copy()

            if n <= 72 and time.time() - start_time < budget_s * 0.92:
                t_limit = min(60.0, budget_s * 0.05)
                h_bin, c5_bin = bit_flip_search(
                    best_h, n, max_rounds=40,
                    time_limit=t_limit, start_time=start_time
                )
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()

    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        n_best = len(best_h)
        h_final = optimize_from_start(best_h, n_best, mu_fine, max_iter=300)
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()

        if n_best <= 72 and time.time() - start_time < budget_s * 0.98:
            t_limit = min(60.0, (budget_s - (time.time() - start_time)) * 0.5)
            h_bin, c5_bin = bit_flip_search(
                best_h, n_best, max_rounds=60,
                time_limit=t_limit, start_time=start_time
            )
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

    if best_h is None:
        n = 42
        best_h = np.ones(n) * 0.5
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
