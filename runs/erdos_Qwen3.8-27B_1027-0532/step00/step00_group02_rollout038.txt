But let me also think about a smarter approach: coordinate descent.

For coordinate descent, we fix all h[i] except h[j], and minimize the objective with respect to h[j].

The objective is C₅ = max_k c_k where c_k = sum_i h[i] * (1-h[i+k-(n-1)]) * dx.

For a fixed j, let's figure out which c_k depend on h[j]:
- c_k depends on h[j] if there exists i such that i = j or i + k - (n-1) = j
- Case 1: i = j, so c_k includes term h[j] * (1-h[j+k-(n-1)]) (if j+k-(n-1) is valid)
- Case 2: i + k - (n-1) = j, so i = j + n - 1 - k, and c_k includes term h[j+n-1-k] * (1-h[j]) (if j+n-1-k is valid)

So for each k, h[j] appears in at most 2 terms of c_k.

Let me denote:
- a_k = coefficient of h[j] in c_k (from case 1)
- b_k = coefficient of (1-h[j]) in c_k (from case 2)
- d_k = constant part of c_k (not involving h[j])

Then c_k = d_k + a_k * h[j] + b_k * (1-h[j]) = d_k + b_k + (a_k - b_k) * h[j]

So c_k is linear in h[j]! The objective is max_k c_k, which is the max of linear functions, i.e., a convex piecewise-linear function.

The minimum of this convex piecewise-linear function on [0,1] occurs at:
- A breakpoint (intersection of two linear pieces)
- Or at a boundary (0 or 1)

We can find it by:
1. Computing the value and slope of each c_k at h[j] = 0 and h[j] = 1
2. The active constraint at the minimum is where the max is minimized

This is a 1D problem that can be solved in O(n^2) or even O(n log n) time.

The smooth max objective is:

where c_k(h) = sum_i h[i] * (1-h[i+k-(n-1)]) (with valid index constraints)

The gradient of c_k with respect to h[j]:
∂c_k/∂h[j] = (1-h[j+m]) if j+m is valid (j is first index) + (-h[j-m]) if j-m is valid (j is second index)

where m = k - (n-1).

Actually, for each k (lag m), h[j] appears in c_k if:
- j + m is valid (i.e., 0 ≤ j+m < n): contributes (1-h[j+m])
- j - m is valid (i.e., 0 ≤ j-m < n): contributes (-h[j-m])

So the gradient of c_k with respect to h is a sparse vector with at most 2n non-zero entries.

For the smooth max, the gradient is a weighted sum of the gradients of the individual c_k, where the weights are softmax weights.

Let me implement this:

```python
def _smooth_max_grad(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = alpha * (vals - m)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of smooth max
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(len(corr)):
        if weights[k] < 1e-15:
            continue
        lag = k - (n - 1)
        for j in range(n):
            if 0 <= j + lag < n:
                grad[j] += weights[k] * dx * (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                grad[j] += weights[k] * dx * (-h[j - lag])
    
    return grad
```

```python
from scipy.optimize import minimize
import time

def _eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _slsqp(h_init, dx, n, target_sum, alpha=60, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        vals = corr * dx
        m = np.max(vals)
        shifted = alpha * (vals - m)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        obj_val = m + (1.0 / alpha) * np.log(np.sum(exp_vals))
        
        # Gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-15:
                continue
            lag = k - (n - 1)
            wk_dx = w * dx
            # h[j] appears in c_k with coefficient:
            # (1 - h[j+lag]) if 0 <= j+lag < n
            # (-h[j-lag]) if 0 <= j-lag < n
            j_first = np.arange(n)
            # First index: j + lag valid
            valid_first = (j_first + lag >= 0) & (j_first + lag < n)
            if np.any(valid_first):
                grad[j_first[valid_first]] += wk_dx * (1.0 - x[j_first[valid_first] + lag])
            # Second index: j - lag valid
            valid_second = (j_first - lag >= 0) & (j_first - lag < n)
            if np.any(valid_second):
                grad[j_first[valid_second]] += wk_dx * (-x[j_first[valid_second] - lag])
        
        return obj_val, grad
    
    result = minimize(obj_and_grad, h_init, method='SLSQP', bounds=bounds,
                     constraints=cons, jac=True,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
    return result.x, _eval_c5(result.x, dx)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def _obj_grad(x, dx, n, alpha):
    one_minus = 1.0 - x
    corr = np.correlate(x, one_minus, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = alpha * (vals - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj_val = m + (1.0 / alpha) * np.log(sum_exp)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        wk = w * dx
        # First index contribution: h[j] * (1-h[j+lag])
        # ∂/∂h[j] = (1-h[j+lag]) for valid j+lag
        j1 = np.arange(n)
        v1 = j1 + lag
        mask1 = (v1 >= 0) & (v1 < n)
        if np.any(mask1):
            grad[j1[mask1]] += wk * one_minus[v1[mask1]]
        # Second index contribution: h[j-lag] * (1-h[j])
        # ∂/∂h[j] = -h[j-lag] for valid j-lag
        v2 = j1 - lag
        mask2 = (v2 >= 0) & (v2 < n)
        if np.any(mask2):
            grad[j1[mask2]] += wk * (-x[v2[mask2]])
    
    return obj_val, grad

def _slsqp(h_init, dx, n, target_sum, alpha=60, maxiter=300, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    h_clip = np.clip(h_init, 0.0, 1.0)
    
    result = minimize(lambda x: _obj_grad(x, dx, n, alpha), h_clip,
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
    
    x = np.clip(result.x, 0.0, 1.0)
    # Repair sum
    diff = target_sum - np.sum(x)
    if abs(diff) < 0.5:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        # Second repair
        diff = target_sum - np.sum(x)
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    
    return x, _eval_c5(x, dx)

def _repair_sum(h, target_sum):

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:

        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        # Don't overshoot

            max_step = np.min(1.0 - h[mask])
        else:
            max_step = -np.min(h[mask])
        step = np.sign(step) * min(abs(step), abs(max_step) * 0.9)
        h[mask] += step
    return h
```

```python
import time

def _smooth_obj(x, dx, n, alpha):

    corr = np.correlate(x, one_minus, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = alpha * (vals - m)
    exp_s = np.exp(shifted)
    return m + (1.0 / alpha) * np.log(np.sum(exp_s))

def _smooth_grad(x, dx, n, alpha):
    one_minus = 1.0 - x
    corr = np.correlate(x, one_minus, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = alpha * (vals - m)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    j_idx = np.arange(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        wk = w * dx
        # First index: j + lag
        v1 = j_idx + lag
        m1 = (v1 >= 0) & (v1 < n)
        if np.any(m1):
            grad[j_idx[m1]] += wk * one_minus[v1[m1]]
        # Second index: j - lag

        m2 = (v2 >= 0) & (v2 < n)
        if np.any(m2):
            grad[j_idx[m2]] += wk * (-x[v2[m2]])
    return grad

def _slsqp(h_init, dx, n, target_sum, alpha, maxiter):

    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]
```
