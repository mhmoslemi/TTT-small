def compute_c5(h):
    """Compute C5 = max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_feasible(v, n):
    """Make v feasible: clip to [0,1] and adjust sum to n/2"""
    v = np.array(v, dtype=float).copy()
    v = np.clip(v, 0, 1)
    target = n / 2.0
    s = np.sum(v)
    if abs(s - target) < 1e-10:
        return v
    if s > target:
        # Need to reduce
        d = s - target
        # Proportional reduction
        mask = v > 0
        if np.sum(v[mask]) > 0:
            v[mask] *= (1.0 - d / np.sum(v[mask]))
        # Clip again
        v = np.clip(v, 0, 1)
    else:
        # Need to increase
        d = target - s
        mask = v < 1
        if np.sum(1.0 - v[mask]) > 0:
            v[mask] += d * (1.0 - v[mask]) / np.sum(1.0 - v[mask])
        v = np.clip(v, 0, 1)
    # Final adjustment for floating point
    s = np.sum(v)
    if s != target:
        diff = target - s
        # Distribute among entries that can absorb
        if diff > 0:
            mask = v < 1
        else:
            mask = v > 0
        if np.any(mask):
            v[mask] += diff / np.sum(mask)
            v = np.clip(v, 0, 1)
    return v

    v = rng.random(n)
    return make_feasible(v, n)

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(v, n):
    v = np.array(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    if s > target:
        d = s - target
        mask = v > 0
        total = np.sum(v[mask])
        if total > 0:
            v[mask] -= d * v[mask] / total
    else:
        d = target - s
        mask = v < 1
        total = np.sum(1.0 - v[mask])
        if total > 0:
            v[mask] += d * (1.0 - v[mask]) / total
    v = np.clip(v, 0.0, 1.0)
    # Fix any remaining error
    s = np.sum(v)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = v < 1
        else:
            mask = v > 0
        if np.any(mask):
            v[mask] += diff / np.sum(mask)
            v = np.clip(v, 0.0, 1.0)
    return v

def random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def random_feasible(n, rng):
    v = rng.random(n)
    return make_feasible(v, n)

def thue_morse(n, rng):
    v = np.zeros(n)
    for i in range(n):
        v[i] = bin(i).count('1') % 2
    # Adjust to have exactly n/2 ones
    num_ones = int(np.sum(v))
    target = n // 2
    if num_ones < target:
        zeros = np.where(v == 0)[0]
        flip = rng.choice(zeros, size=target - num_ones, replace=False)
        v[flip] = 1
    elif num_ones > target:
        ones = np.where(v == 1)[0]
        flip = rng.choice(ones, size=num_ones - target, replace=False)
        v[flip] = 0
    return v

def binary_local_search(h, n, max_steps, rng, cool=None):
    """Hill climbing / simulated annealing on binary sequences"""
    best = h.copy()
    best_val = compute_c5(h)
    current = h.copy()
    current_val = best_val
    
    if cool is None:
        cool = 0.05
    
    T = cool
    T_min = 0.001
    cooling = (T_min / T) ** (1.0 / max_steps)
    
    for step in range(max_steps):
        # Pick a random swap
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        current[i] = 0
        current[j] = 1
        new_val = compute_c5(current)
        
        if new_val < current_val or rng.random() < np.exp(-(new_val - current_val) / T):
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = current.copy()
        else:
            # Revert
            current[i] = 1
            current[j] = 0
        
        T *= cooling
    
    return best, best_val

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def slsqp_optimize(h0, n, maxiter=500):
    """SLSQP optimization"""
    def obj(x):
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    
    h_opt = np.clip(res.x, 0, 1)
    h_opt = make_feasible(h_opt, n)
    return h_opt, compute_c5(h_opt)

def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def make_feasible(v, n):
    v = np.array(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(v))
    if abs(s - target) < 1e-12:
        return v
    if s > target:
        d = s - target
        mask = v > 0
        total = float(np.sum(v[mask]))
        if total > 1e-12:
            v[mask] -= d * v[mask] / total
    else:
        d = target - s
        mask = v < 1
        total = float(np.sum(1.0 - v[mask]))
        if total > 1e-12:
            v[mask] += d * (1.0 - v[mask]) / total
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = v < 1
        else:
            mask = v > 0
        if np.any(mask):
            v[mask] += diff / float(np.sum(mask))
            v = np.clip(v, 0.0, 1.0)
    return v


def random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    v = np.zeros(n)
    v[idx] = 1.0
    return v


def thue_morse_seq(n, rng):
    v = np.zeros(n)
    for i in range(n):
        v[i] = bin(i).count('1') % 2
    num_ones = int(np.sum(v))
    target = n // 2
    if num_ones < target:
        zeros = np.where(v == 0)[0]
        flip = rng.choice(zeros, size=target - num_ones, replace=False)
        v[flip] = 1.0
    elif num_ones > target:
        ones = np.where(v == 1)[0]
        flip = rng.choice(ones, size=num_ones - target, replace=False)
        v[flip] = 0.0
    return v


def sawtooth_seq(n, rng):
    period = n // 2
    v = np.zeros(n)
    for i in range(n):
        phase = (i * period // n) % (2 * period)
        if phase < period:
            v[i] = 1.0
    num_ones = int(np.sum(v))
    target = n // 2
    if num_ones != target:
        if num_ones < target:
            zeros = np.where(v == 0)[0]
            flip = rng.choice(zeros, size=target - num_ones, replace=False)
            v[flip] = 1.0
        else:
            ones = np.where(v == 1)[0]
            flip = rng.choice(ones, size=num_ones - target, replace=False)
            v[flip] = 0.0
    return v


def binary_sa(h, n, time_limit, rng, cool=0.03):
    t_start = time.time()
    best = h.copy()
    best_val = compute_c5(h)
    current = h.copy()
    current_val = best_val
    T = cool
    T_min = 0.0003
    expected_steps = max(1000, int(time_limit * 8000))
    cooling = (T_min / T) ** (1.0 / expected_steps)
    no_improve = 0
    patience = 5000

    while time.time() - t_start < time_limit:
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])

        current[i] = 0.0
        current[j] = 1.0
        new_val = compute_c5(current)
        delta = new_val - current_val

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            current_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best = current.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > patience and T > T_min:
                T = max(T * 1.5, cool * 0.5)
                no_improve = 0
        else:
            current[i] = 1.0
            current[j] = 0.0

        T *= cooling

    return best, best_val


def binary_polish(h, n, max_passes=50):
    h = h.copy()
    val = compute_c5(h)
    for _ in range(max_passes):
        improved = False
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        for i in ones:
            if h[i] != 1:
                continue
            for j in zeros:
                if h[j] != 0:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_val = compute_c5(h)
                if new_val < val - 1e-12:
                    val = new_val
                    improved = True
                    zeros.remove(j)
                    zeros.append(i)
                    ones.remove(i)
                    ones.append(j)
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, val


def sum_constraint(x, n):
    return float(np.sum(x)) - n / 2.0


def slsqp_optimize(h0, n, maxiter=500):
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = make_feasible(h0, n)

    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        return compute_c5(x)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]

    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})

    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = make_feasible(h_opt, n)
    return h_opt, compute_c5(h_opt)


def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = random_binary(70, rng)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_val = compute_c5(h_init)

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 50
    if remaining > 5:
        t_alloc = remaining * 0.35
        h1, val1 = binary_sa(h_init, n_init, t_alloc, rng, cool=0.04)
        if val1 < best_val:
            best_val = val1
            best_h = h1

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 40
    if remaining > 5:
        h1, val1 = binary_polish(best_h, n_init, max_passes=100)
        if val1 < best_val:
            best_val = val1
            best_h = h1

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 40
    if remaining > 15:
        h2, val2 = slsqp_optimize(best_h, n_init, maxiter=600)
        if val2 < best_val:
            best_val = val2
            best_h = h2

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 40
    if remaining > 10:
        h_tm = thue_morse_seq(n_init, rng)
        t_alloc = min(remaining * 0.2, 60.0)
        h3, val3 = binary_sa(h_tm, n_init, t_alloc, rng, cool=0.03)
        if val3 < best_val:
            best_val = val3
            best_h = h3
        h3, val3 = binary_polish(h3, n_init, max_passes=100)
        if val3 < best_val:
            best_val = val3
            best_h = h3

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 30
    if remaining > 10:
        h_saw = sawtooth_seq(n_init, rng)
        t_alloc = min(remaining * 0.15, 40.0)
        h4, val4 = binary_sa(h_saw, n_init, t_alloc, rng, cool=0.02)
        if val4 < best_val:
            best_val = val4
            best_h = h4
        h4, val4 = binary_polish(h4, n_init, max_passes=100)
        if val4 < best_val:
            best_val = val4
            best_h = h4

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 30
    n_randoms = 3
    for r in range(n_randoms):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 20:
            break
        h_rand = random_binary(n_init, rng)
        t_alloc = (budget_s - elapsed - 20) / (n_randoms - r) * 0.3
        t_alloc = max(3.0, min(t_alloc, 60.0))
        h5, val5 = binary_sa(h_rand, n_init, t_alloc, rng, cool=0.03)
        if val5 < best_val:
            best_val = val5
            best_h = h5
        h5, val5 = binary_polish(h5, n_init, max_passes=80)
        if val5 < best_val:
            best_val = val5
            best_h = h5

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 20
    if remaining > 20:
        n_high = n_init * 2
        if n_high <= 280:
            h_high = upscale(best_h, n_high)
            h_high_bin = (h_high > 0.5).astype(float)
            h_high_bin = make_feasible(h_high_bin, n_high)
            t_alloc = min(remaining * 0.4, 120.0)
            h6, val6 = binary_sa(h_high_bin, n_high, t_alloc, rng, cool=0.02)
            if val6 < best_val:
                best_val = val6
                best_h = h6
            h6, val6 = binary_polish(h6, n_high, max_passes=50)
            if val6 < best_val:
                best_val = val6
                best_h = h6

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 10
    if remaining > 15:
        h_final = best_h
        if len(h_final) != n_init:
            h_final = upscale(h_final, n_init)
            h_final = (h_final > 0.5).astype(float)
            h_final = make_feasible(h_final, n_init)
        h7, val7 = slsqp_optimize(h_final, n_init, maxiter=400)
        if val7 < best_val:
            best_val = val7
            best_h = h7

    if len(best_h) != n_init:
        best_h = upscale(best_h, n_init)
        best_val = compute_c5(best_h)

    best_h = make_feasible(best_h, n_init)
    assert 0.0 <= float(np.min(best_h)) and float(np.max(best_h)) <= 1.0
    assert abs(float(np.sum(best_h)) - n_init / 2.0) < 1e-8
