import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary_random(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def make_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return h

def make_m_sequence(k, n):
    """Try to create an m-sequence-like initial condition."""
    # This is a simplified version - just use a LFSR
    # For now, just return random
    h = np.zeros(n)
    h[n//3:(2*n)//3] = 1.0  # block in the middle
    return h

def sa_optimize(h, n, rng, max_time=30.0, start_time=None):
    """Simulated annealing optimization."""
    if start_time is None:
        start_time = time.time()
    
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    temp = 0.025
    min_temp = 0.001
    coolspeed = 0.9998
    
    while time.time() - start_time < max_time:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - c5
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        temp = max(temp * coolspeed, min_temp)
    
    return best_h, best_c5

def local_search(h, n, rng, max_time=10.0, start_time=None):
    """Greedy local search with double swaps."""
    if start_time is None:
        start_time = time.time()
    
    c5 = compute_c5(h)
    
    stuck = 0
    while time.time() - start_time < max_time and stuck < 200:

        zeros = np.where(h == 0.0)[0]

            if new_c5 < c5:

                improved = True

            else:
                h[i] = 1.0
                h[j] = 0.0

                i1, i2 = ones[idx1[0]], ones[idx1[1]]
                j1, j2 = zeros[idx2[0]], zeros[idx2[1]]
