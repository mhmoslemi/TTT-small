import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = correlate(h, g, mode='full') * dx
    return float(np.max(c))

def c5_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return correlate(h, g, mode='full') * dx

def smooth_max_val(h, tau):
    c = c5_all_lags(h)
    c_max = np.max(c)
    if tau < 500:
        shifted = c - c_max
        return c_max + np.log(np.sum(np.exp(tau * shifted))) / tau
    return c_max

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = c5_all_lags(h)
    c_max = np.max(c)
    if tau < 500:
        shifted = c - c_max
        weights = np.exp(tau * shifted)
        sw = np.sum(weights)
        if sw < 1e-300:
            weights = np.zeros(len(c))
            weights[np.argmax(c)] = 1.0
        else:
            weights /= sw
    else:
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        idx1 = j - lag
        v1 = (idx1 >= 0) & (idx1 < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - h[idx1[v1]])
        idx2 = j + lag
        v2 = (idx2 >= 0) & (idx2 < n)
        if np.any(v2):
            grad[v2] -= w * h[idx2[v2]]
    grad *= dx
    return grad

def project_sum_box(h, target_sum):
    if h.sum() == target_sum and (h >= -1e-15).all() and (h <= 1 + 1e-15).all():
        return np.clip(h, 0, 1)
    theta_lo = -5.0
    theta_hi = 5.0
    for _ in range(100):
        theta = 0.5 * (theta_lo + theta_hi)
        projected = np.clip(h - theta, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            theta_lo = theta
        else:
            theta_hi = theta
    theta = 0.5 * (theta_lo + theta_hi)
    result = np.clip(h - theta, 0.0, 1.0)
    return result

def projected_grad_descent(h, target_sum, tau, max_iter=300, init_step=0.1):
    n = len(h)
    h_curr = h.copy()
    f_curr = smooth_max_val(h_curr, tau)
    step = init_step
    for it in range(max_iter):
        g = smooth_max_grad(h_curr, tau)
        alpha = step
        improved = False
        for ls in range(30):
            h_try = h_curr - alpha * g
            h_try = project_sum_box(h_try, target_sum)
            f_try = smooth_max_val(h_try, tau)
            if f_try <= f_curr - 1e-5 * alpha * np.dot(g, h_try - h_curr) + 1e-12:
                h_curr = h_try
                f_curr = f_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-10:
                break
        else:
            step = min(step * 1.1, 0.5)
    return h_curr

def slsqp_polish(h, target_sum, tau, maxiter=300):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smooth_max_val(x, tau),
                       h.copy(),
                       jac=lambda x: smooth_max_grad(x, tau),
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_sum_box(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        pass
    return h.copy()
