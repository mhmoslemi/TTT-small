def evaluate_c5(h):
    """Compute C5 for a given h."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    # Use bisection on Lagrange multiplier
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h + mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    projected = np.clip(h + (lo+hi)/2, 0, 1)
    # Fine adjustment
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        # Distribute the difference
        room_up = 1.0 - projected
        room_down = projected
        if diff > 0:
            total_room = np.sum(room_up)
            if total_room > 0:
                projected += diff / total_room * room_up
        else:
            total_room = np.sum(room_down)
            if total_room > 0:
                projected += diff / total_room * room_down
    return np.clip(projected, 0, 1)

def subgradient_step(h, step_size, active_indices=None):
    """One subgradient descent step."""
    N = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    
    # Find active shifts
    if active_indices is None:
        active_indices = np.where(corr > max_val - 1e-8)[0]
    
    # Compute subgradient (average gradient of active terms)
    grad = np.zeros(N)
    for k in active_indices:
        m = k - (N-1)  # shift amount
        # dc_k/dh[j]:
        # j is first index if 0<=j<N and 0<=j+m<N
        # j is second index if 0<=j<N and 0<=j-m<N
        first = (np.arange(N) >= 0) & (np.arange(N) + m < N) & (np.arange(N) + m >= 0)
        second = (np.arange(N) >= 0) & (np.arange(N) - m < N) & (np.arange(N) - m >= 0)
        # dc/dh[j] = (1-h[j+m])*first[j] - h[j-m]*second[j]
        g = np.zeros(N)
        j_first = np.where(first)[0]
        j_second = np.where(second)[0]
        for j in j_first:
            g[j] += (1 - h[j+m])
        for j in j_second:
            g[j] -= h[j-m]
        grad += g
    grad /= len(active_indices)
    
    # Step

    return h_new

            h_new[i] += delta

            if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:

                    h = h_new
                    best_c = c
                    improved = True

def project_feasible(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum} using bisection."""
    # Bisection on offset
    lo, hi = -max(h) - 1, max(1 - h) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(h + (lo+hi)/2, 0, 1)
    # Fine-tune to hit exact sum
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        # Adjust the least "committed" values
        room = np.where(proj < 1, 1 - proj, 0) + np.where(proj > 0, proj, 0)
        # Simple: distribute proportionally to available room
        if diff > 0:
            room_up = 1 - proj
            total = room_up.sum()
            if total > 1e-15:
                proj += diff * room_up / total
        else:
            room_dn = proj.copy()
            total = room_dn.sum()
            if total > 1e-15:
                proj += diff * room_dn / total
    return np.clip(proj, 0, 1)

    return np.correlate(h, 1-h, mode='full')

def compute_subgradient(h, top_k=5):

    N = len(h)
    corr = corr_values(h)
    max_val = np.max(corr)
    # Find correlations near the max
    threshold = max_val - 1e-6
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.where(corr == max_val)[0]
    
    grad = np.zeros(N)
    for k in active:
        m = k - (N - 1)  # shift in grid units
        # For correlation at shift m:
        # c = sum over valid i of h[i] * (1-h[i+m])
        # dc/dh[j] = (1-h[j+m]) for j where j,j+m valid as first index
        #           - h[j-m] for j where j-m, j valid as second index
        g = np.zeros(N)

        if m >= 0:

            for j in range(m, N):
                g[j] -= h[j - m]
        else:

            for j in range(m_abs, N):
                g[j] += (1 - h[j - m_abs])
            for j in range(N - m_abs):
                g[j] -= h[j + m_abs]
        grad += g

def soft_max_c5(h, tau=1.0):
    """Smooth approximation of C5 using log-sum-exp."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full')
    # Soft max
    max_corr = np.max(corr)
    log_sum = max_corr + tau * np.log(np.sum(np.exp((corr - max_corr) / tau)))
    return log_sum * dx

def soft_max_gradient(h, tau=1.0):
    """Gradient of soft max C5."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / tau)
    weights /= weights.sum()
    
    # Weighted sum of individual gradients
    grad = np.zeros(N)
    for k in range(len(corr)):
        s = k - (N - 1)
        if abs(weights[k]) < 1e-15:
            continue
        if s >= 0:
            for i in range(N - s):
                grad[i] += weights[k] * (1 - h[i + s])
            for i in range(s, N):
                grad[i] -= weights[k] * h[i - s]
        else:
            s_abs = -s
            for i in range(s_abs, N):
                grad[i] += weights[k] * (1 - h[i - s_abs])
            for i in range(N - s_abs):
                grad[i] -= weights[k] * h[i + s_abs]
    return grad * dx

def project_feasible(h, target_sum):

    for _ in range(80):

        if np.sum(proj) > target_sum:

def binary_overlaps(ones, N):
    """Compute all overlaps for binary h defined by ones (boolean array)."""
    # overlaps[s] for s = 0, ..., N-1 (we only need positive shifts by symmetry)
    # Actually, let's compute for all s from -(N-1) to N-1
    ones_idx = np.where(ones)[0]

            count = len(valid) - np.sum(ones[valid + s])
        else:
            valid = ones_idx[ones_idx + s >= 0]
            count = len(valid) - np.sum(ones[valid + s])
        overlaps[mid + s] = count
    return overlaps
