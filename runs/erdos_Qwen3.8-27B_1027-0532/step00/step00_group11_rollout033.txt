    h_bin = (h >= 0.5).astype(float)
    s = int(h_bin.sum())
    if s > target:
        idx = np.where(h_bin == 1)[0]
        n_remove = s - target
        order = np.argsort(h[idx])[:n_remove]
        h_bin[idx[order]] = 0

        n_add = target - s
        order = np.argsort(-h[idx])[:n_add]
