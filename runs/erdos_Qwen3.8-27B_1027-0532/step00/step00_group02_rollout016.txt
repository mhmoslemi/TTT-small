    m = np.max(scaled)
    return t * (m + np.log(np.sum(np.exp(scaled - m))))

def smooth_max_grad(h, dx, t):
    """Returns (value, gradient) of smooth_max(correlate(h,1-h), t) * dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr / t
    mx = np.max(scaled)
    weights = np.exp(scaled - mx)
    weights /= np.sum(weights)
    val = t * (mx + np.log(np.sum(np.exp(scaled - mx)))) * dx
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - n + 1
        wk = weights[k]
        if wk < 1e-10:
            continue
        # dC(m)/dh[j] = (1-h[j+m])*I(0<=j+m<n) - h[j-m]*I(0<=j-m<n)
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            # (1-h[j+m]) for j in [0, n-m-1]
            grad[:n-m] += wk * (1.0 - h[m:])
            # -h[j-m] for j in [m, n-1]

    grad *= dx
