def compute_R(h):
    """Compute autocorrelation R(m) for all m."""
    n = len(h)
    # R(m) = sum_j h[j] * h[j+m] for valid j
    # This is the autocorrelation
    R = np.correlate(h, h, mode='full')
    # R is of length 2n-1
    # R[0] corresponds to m = -(n-1), R[n-1] to m=0, R[2n-2] to m=n-1
    return R

def compute_c5_val(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def project_onto_simplex(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}."""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0, 1)
        s = np.sum(h_trial)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    return h

def gradient_descent(h, n_iters=10000, lr_init=0.1):
    """Perform gradient descent to minimize C5."""
    n = len(h)
    h = h.copy()
    target_sum = n / 2.0
    best_c5 = compute_c5_val(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        # Compute R(m) for all m
        R = np.correlate(h, h, mode='full')
        
        # Find m* = argmin R
        m_star_idx = np.argmin(R)
        m_star = m_star_idx - (n - 1)
        
        # Gradient direction: shift(h, m_star) + shift(h, -m_star)
        # This is the gradient of R(m_star) w.r.t. h
        # We want to INCREASE R(m_star), so we go in the +gradient direction
        direction = np.zeros(n)
        for j in range(n):
            jm = j + m_star
            if 0 <= jm < n:
                direction[j] += h[jm]
            jM = j - m_star
            if 0 <= jM < n:
                direction[j] += h[jM]
        
        # Adaptive learning rate
        lr = lr_init / (1 + it * 0.001)
        
        # Update
        h_new = h + lr * direction
        
        # Project
        h_new = project_onto_simplex(h_new, target_sum)
        
        # Check improvement
        c5 = compute_c5_val(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            # Reduce learning rate or revert
            h = h + lr * 0.5 * direction
            h = project_onto_simplex(h, target_sum)
    
    return best_h, best_c5

    return float(np.max(corr) * dx)

def coord_descent(h, budget_s=30):
    n = len(h)
    target_sum = n / 2.0
    nk = 2 * n - 1
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    
    start = time.time()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    no_improve = 0
    sweep = 0
    while time.time() - start < budget_s:
        sweep += 1
        improved = False
        for j in range(n):
            # Compute slopes
            slopes = np.zeros(nk)
            for k in range(nk):
                m = k - (n - 1)
                s = 0.0
                jm = j + m
                if 0 <= jm < n:
                    s += 1.0 - h[jm]
                jM = j - m
                if 0 <= jM < n:
                    s -= h[jM]
                slopes[k] = s
            
            # Check if coordinate is at boundary
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-14:
                continue
            
            # Golden section search
            a = lo
            b = hi
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = np.max(corr + slopes * c_pt)
            fd = np.max(corr + slopes * d_pt)
            
            for _ in range(30):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = np.max(corr + slopes * c_pt)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = np.max(corr + slopes * d_pt)
            
            delta_opt = (a + b) / 2.0
            new_val = h[j] + delta_opt
            
            if abs(new_val - h[j]) > 1e-10:
                delta = new_val - h[j]
                h[j] = new_val
                corr += slopes * delta
                improved = True
