```python
def smooth_gradient(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode="full") * dx
    
    # Softmax weights
    corr_centered = corr - corr.max()
    weights = np.exp(corr_centered / epsilon)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if weights[m] < 1e-10:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            k = lag
            for j in range(k, n):
                term = (1 - h[j-k])
                if j+k < n:
                    term -= h[j+k]
                grad[j] += weights[m] * dx * term
        else:
            l = -lag
            for j in range(n):
                term = 0
                if j+l < n:
                    term += (1 - h[j+l])
                if j >= l:
                    term -= h[j-l]
                grad[j] += weights[m] * dx * term
    
    return grad, corr.max()

def smooth_gradient(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode="full") * dx
    
    corr_centered = corr - corr.max()
    weights = np.exp(corr_centered / epsilon)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    
    for k in range(n):
        # Positive lag k
        w_pos = weights[n-1+k]
        if w_pos > 1e-12:
            # j in [k, n-1]: grad[j] += w_pos * dx * (1-h[j-k])
            grad[k:] += w_pos * dx * (1 - h[:n-k])
            # j in [0, n-1-k]: grad[j] -= w_pos * dx * h[j+k]
            grad[:n-k] -= w_pos * dx * h[k:]
        
        # Negative lag -k (skip k=0 to avoid double counting)
        if k > 0:
            w_neg = weights[n-1-k]
            if w_neg > 1e-12:
                # j in [0, n-1-k]: grad[j] += w_neg * dx * (1-h[j+k])
                grad[:n-k] += w_neg * dx * (1 - h[k:])
                # j in [k, n-1]: grad[j] -= w_neg * dx * h[j-k]
                grad[k:] -= w_neg * dx * h[:n-k]
    
    return grad, corr.max()
```

But computing A for all k at each coordinate is O(n^2) per coordinate, making the total O(n^3) per pass. For n=720, that's 720^3 ≈ 370M per pass. Too slow.

Better approach: compute corr once per pass, and for each j, compute A more efficiently.

Actually, A_k for coordinate j is just the gradient component I computed before. Let me precompute it.

For a given j, the gradient of g_k with respect to h[j] is:
For lag k >= 0:
  A_k = dx * [(1-h[j-k]) if j>=k else 0] - dx * h[j+k] if j+k<n else 0]
For lag -l (l>0):
  A_{-l} = dx * [(1-h[j+l]) if j+l<n else 0] - dx * h[j-l] if j>=l else 0]

I can compute all A_k for a fixed j in O(n) time using vectorized operations:

```python
def compute_A_for_j(h, j):
    n = len(h)
    dx = 2.0 / n
    A = np.zeros(2*n - 1)
    
    for k in range(n):  # positive lag k and negative lag -k
        m_pos = n - 1 + k
        A[m_pos] = 0
        if j >= k:
            A[m_pos] += (1 - h[j-k])
        if j + k < n:
            A[m_pos] -= h[j+k]
        A[m_pos] *= dx
        
        if k > 0:
            m_neg = n - 1 - k
            A[m_neg] = 0
            if j + k < n:
                A[m_neg] += (1 - h[j+k])
            if j >= k:
                A[m_neg] -= h[j-k]
            A[m_neg] *= dx
    
    return A
```

This is O(n) per coordinate j (with small constant), so O(n^2) per pass. For n=720, that's 518K per pass. Very fast.

Let me think about bounds:
- f(0) = max_k B_k
- f(1) = max_k (A_k + B_k)
- The minimum is <= min(f(0), f(1))

For binary search on t:
- Check: is there x in [0,1] such that A_k*x + B_k <= t for all k?
  For A_k > 0: x <= (t - B_k) / A_k
  For A_k < 0: x >= (t - B_k) / A_k
  For A_k = 0: need B_k <= t
  
  So: L = max(0, max_{A_k<0} (t-B_k)/A_k) and R = min(1, min_{A_k>0} (t-B_k)/A_k)
  Also need B_k <= t for all A_k = 0.
  Feasible iff L <= R and all A_k=0 have B_k <= t.

Binary search: ~50 iterations for double precision, each O(n). Total O(n) per coordinate.

So the full coordinate descent pass is O(n^2), and for n=720, each pass takes ~720 * 720 * 50 = 26M operations. In numpy, this should take well under a second per pass.

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # All saturated, distribute among all
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_nn(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    # Nearest neighbor: each old point maps to new_n/n new points
    indices = np.floor(np.arange(new_n) * n / new_n).astype(int)
    indices = np.clip(indices, 0, n-1)
    new_h = h[indices]
    return new_h

def make_default_h(n):
    h = np.zeros(n)
    # Alternating pattern
    h[::2] = 1.0
    if n % 2 == 1:
        # Need sum = n/2, currently sum = (n+1)/2
        # Need to reduce by 0.5
        h[n//2] = 0.0  # this makes sum = (n-1)/2, too low
        # Better: set one 1 to 0.5
        # Current sum = (n+1)/2, need n/2, difference = 0.5
        # Set h[0] = 0.5
        h[0] = 0.5
    # Actually for odd n, sum of alternating 1,0,1,0,...,1 is (n+1)/2
    # We need n/2, so reduce by 0.5: set one 1 to 0.5
    # For even n, sum = n/2 ✓
    return project_feasible(h, n)

def smooth_gradient(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Softmax weights (numerically stable)
    corr_shifted = corr - corr.max()
    weights = np.exp(corr_shifted / epsilon)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    
    for k in range(n):
        w_pos = weights[n - 1 + k]
        if w_pos > 1e-12:
            # Positive lag k
            if k == 0:
                grad += w_pos * dx * (1.0 - 2.0 * h)
            else:
                grad[k:] += w_pos * dx * (1.0 - h[:n-k])
                grad[:n-k] -= w_pos * dx * h[k:]
        
        if k > 0:
            w_neg = weights[n - 1 - k]
            if w_neg > 1e-12:
                # Negative lag -k
                grad[:n-k] += w_neg * dx * (1.0 - h[k:])
                grad[k:] -= w_neg * dx * h[:n-k]
    
    return grad, corr.max()

def optimize_smoothed(h, time_limit, seed):
    n = len(h)
    epsilon = 0.02
    lr = 0.05
    momentum = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    v = np.zeros(n)  # second moment
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() < time_limit:
        grad, _ = smooth_gradient(h, epsilon)
        t += 1
        
        # Adam update
        momentum = beta1 * momentum + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = momentum / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project
        h = project_feasible(h, n)
        
        # Track best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Schedule
        if t % 200 == 0:
            epsilon = max(epsilon * 0.9, 1e-3)
            lr = max(lr * 0.95, 0.005)
    
    return best_h

def solve_1d_max(A, B):
    """Find x in [0,1] minimizing max_k (A[k]*x + B[k])."""
    # Binary search on t
    lo = np.min(np.minimum(B, A + B))  # lower bound on achievable max
    hi = np.max(np.maximum(B, A + B))  # upper bound
    
    # Actually, better bounds:
    # At x=0: max is max(B)
    # At x=1: max is max(A+B)
    # The minimum of max_k(A_k*x + B_k) is between min(max(B), max(A+B)) and max(max(B), max(A+B))
    
    lo = min(np.max(B), np.max(A + B))
    hi = max(np.max(B), np.max(A + B))
    
    # But we need a tighter lower bound
    # The minimum is at least max_k B_k if all A_k >= 0
    # More generally, use the fact that at the optimum, at least two constraints are active
    
    # Binary search
    for _ in range(60):
        mid = (lo + hi) / 2.0
        # Check feasibility: exists x in [0,1] with A_k*x + B_k <= mid for all k
        # For A_k > 0: x <= (mid - B_k) / A_k
        # For A_k < 0: x >= (mid - B_k) / A_k
        # For A_k == 0: need B_k <= mid
        
        pos_mask = A > 1e-15
        neg_mask = A < -1e-15
        zero_mask = ~(pos_mask | neg_mask)
        
        if zero_mask.any() and np.max(B[zero_mask]) > mid:
            lo = mid
            continue
        
        L = 0.0
        R = 1.0
        
        if pos_mask.any():
            R = min(R, np.min((mid - B[pos_mask]) / A[pos_mask]))
        if neg_mask.any():
            L = max(L, np.max((mid - B[neg_mask]) / A[neg_mask]))
        
        if L <= R:
            hi = mid
        else:
            lo = mid
    
    # Now hi is approximately the optimal value, find x
    t = hi
    L = 0.0
    R = 1.0
    pos_mask = A > 1e-15
    neg_mask = A < -1e-15
    if pos_mask.any():
        R = min(R, np.min((t - B[pos_mask]) / A[pos_mask]))
    if neg_mask.any():
        L = max(L, np.max((t - B[neg_mask]) / A[neg_mask]))
    
    if L > R:
        # Infeasible at this t (shouldn't happen if convergence is good)
        return 0.5
    return (L + R) / 2.0
