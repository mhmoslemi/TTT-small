    h = np.clip(h, 0, 1)

    lo, hi = -2, 2

    bcorr = beta * corr * dx
    bcorr_shifted = bcorr - np.max(bcorr)

    grad = np.zeros(n)

            # Add (1-h[0:n-k]) to grad[k:n]
            grad[k:] += w * (1 - h[:n-k])

            grad[:n-kp] += w * (1 - h[kp:])
            # Subtract h[0:n-kp] from grad[kp:n]
            grad[kp:] -= w * h[:n-kp]

            momentum = mom_rate * momentum + (1.0 - mom_rate) * grad
            h_new = h - lr * momentum
            h_new = project_onto_box_sum(h_new, target_sum)
            c5_new = evaluate_h(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                stuck_count = 0
            else:
                h = h_new
                stuck_count += 1
                lr *= 0.93
                if stuck_count > 400:
                    break
    return best_h, best_c5


def thue_morse_h(n, seed=0):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] + (i & 1)
    tm = np.mod(tm, 2).astype(np.int32)
    target = n // 2
    ones = int(np.sum(tm))
    rng = np.random.RandomState(seed)
    if ones > target:
        idx = np.where(tm == 1)[0]
        rng.shuffle(idx)
        for j in range(ones - target):
            tm[idx[j]] = 0
    elif ones < target:
        idx = np.where(tm == 0)[0]
        rng.shuffle(idx)
        for j in range(target - ones):
            tm[idx[j]] = 1
    return tm.astype(float)


def upsample_h(h, new_n):
    n = len(h)
    old_x = np.arange(n, dtype=float)
    new_x = np.linspace(0, n - 1, new_n)
    return np.interp(new_x, old_x, h)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    rng = np.random.RandomState(seed)

    candidates = []
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) > 0:
            candidates.append(h0.copy())
    except Exception:
        pass

    candidates.append(thue_morse_h(42, seed=101))
    candidates.append(thue_morse_h(60, seed=202))
    candidates.append(thue_morse_h(84, seed=303))
    candidates.append(rng.rand(42))
    candidates.append(rng.rand(60))
    candidates.append(rng.rand(100))

    phase1_budget = min(300.0, budget_s * 0.35)
    per_candidate = phase1_budget / max(1, len(candidates))

    for hc in candidates:
        if time.time() - t_start > budget_s - 200:
            break
        n_h = len(hc)
        alloc = min(per_candidate, 100.0)
        if time.time() - t_start + alloc > budget_s - 200:
            alloc = max(5.0, budget_s - 200 - (time.time() - t_start))
        h_opt, c5_val = optimize_single(hc, alloc)
        if c5_val < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_val
            best_n = n_h

    if best_h is not None:
        current_best = best_h
        for factor in [2, 3, 4, 6]:
            if time.time() - t_start > budget_s - 100:
                break
            new_n = best_n * factor
            if new_n > 600:
                break
            h_up = upsample_h(current_best, new_n)
            h_up = project_onto_box_sum(h_up, new_n / 2.0)
            remaining_alloc = budget_s - 100 - (time.time() - t_start)
            alloc = min(180.0, remaining_alloc * 0.45)
            if alloc < 15:
                break
            h_opt, c5_val = optimize_single(h_up, alloc)
            if c5_val < best_c5 - 1e-8:
                best_h = h_opt.copy()
                best_c5 = c5_val
                best_n = new_n
                current_best = h_opt
            else:
                break

    res_list = [42, 60, 84, 120, 168, 200]
    for r in range(60):
        if time.time() - t_start > budget_s - 40:
            break
        n_try = res_list[r % len(res_list)]
        h_rand = rng.rand(n_try)
        remaining_alloc = budget_s - 40 - (time.time() - t_start)
        alloc = min(35.0, remaining_alloc * 0.15)
        if alloc < 8:
            break
        h_opt, c5_val = optimize_single(h_rand, alloc)
        if c5_val < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_val
            best_n = n_try

    remaining_time = budget_s - (time.time() - t_start)
    if remaining_time > 5 and best_h is not None:
        h_ls = best_h.copy()
        n_ls = best_n
        target_sum = n_ls / 2.0
        step = 0.004
        for _ in range(500000):
            if time.time() - t_start > budget_s - 1:
                break
            n_pert = max(1, int(0.25 * n_ls))
            idx = rng.randint(0, n_ls, size=n_pert)
            pert = np.zeros(n_ls)
            pert[idx] = rng.randn(n_pert) * step
            h_try = h_ls + pert
            h_try = project_onto_box_sum(h_try, target_sum)
            c5_try = evaluate_h(h_try)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
                h_ls = h_try
                step = min(step * 1.003, 0.04)
            else:
                step *= 0.996
                if step < 1e-7:
                    break

    if best_h is None:
        best_h = np.full(42, 0.5)
        best_n = 42
        best_c5 = evaluate_h(best_h)

    return best_h, best_c5, best_n
