    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:

        room = 1.0 - h
        order = np.argsort(-room)  # largest room first

    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
