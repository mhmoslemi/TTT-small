        try_solution(h2, n1)

    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        h3, c5_3 = coordinate_descent(h, n1, seed + 200, remaining * 0.2)
        try_solution(h3, n1)

    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        h_leg = generate_legendre(n1, seed)
        h_leg, c5_leg = local_search_binary(h_leg, n1, seed + 300, remaining * 0.15)
        try_solution(h_leg, n1)

    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        h4, c5_4 = coordinate_descent(best_h, best_n, seed + 400, remaining * 0.2)
        try_solution(h4, best_n)

    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        h5, c5_5 = simulated_annealing(best_h, best_n, seed + 500, remaining * 0.25)
        try_solution(h5, best_n)

    for n_try in [98, 194, 196]:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < 30:
            break
        h_try = generate_initial(n_try, seed + n_try)
        h_try = project_feasible(h_try, n_try)
        h_try, c5_try = local_search_binary(h_try, n_try, seed + n_try, remaining * 0.2)
        try_solution(h_try, n_try)

    if best_h is None:
        best_n = n1
        best_h = project_feasible(generate_initial(best_n, seed), best_n)
        best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
