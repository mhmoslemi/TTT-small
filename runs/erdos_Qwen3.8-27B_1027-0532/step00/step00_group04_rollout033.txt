    corr = np.correlate(h, 1.0 - h, mode='full')

    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return (2.0 / n) * float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def make_binary(h, n, seed=0):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        perm = rng.permutation(len(ones))
        h[ones[perm[:n_ones - target]]] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        perm = rng.permutation(len(zeros))
        h[zeros[perm[:target - n_ones]]] = 1.0
    return h

def compute_G(h):
    n = len(h)
    padded = np.pad(h, n - 1, 'constant')
    G = np.zeros((n, 2 * n - 1))
    for i in range(n):
        w = padded[i:i + 2*n - 1]
        G[i] = w + w[::-1]
    return G

def binary_local_search(h_init, max_sweeps=3000, time_limit=None, t_start=None, seed=0, use_sa=False):
    n = len(h_init)
    h = make_binary(h_init, n, seed)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    best = h.copy()
    best_max = max_corr
    rng = np.random.RandomState(seed + 12345)
    
    temp = 3.0 if use_sa else 0.0
    for sweep in range(max_sweeps):
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        G = compute_G(h)
        improved = False
        
        if use_sa:
            # Try random moves
            n_moves = min(50, len(ones_idx) * len(zeros_idx))
            for _ in range(n_moves):
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                if h[i] != 1 or h[j] != 0:
                    continue
                delta = G[i] - G[j]
                new_max = float(np.max(corr + delta))
                diff = new_max - max_corr
                if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-8)):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr + delta
                    max_corr = new_max
                    if max_corr < best_max:
                        best_max = max_corr
                        best = h.copy()
                    improved = True
                    ones_idx = np.where(h == 1)[0]
                    zeros_idx = np.where(h == 0)[0]
                    G = compute_G(h)
                    temp *= 0.995
        else:
            # Best improvement
            for i in ones_idx:
                if h[i] != 1:
                    continue
                deltas = G[i][None, :] - G[zeros_idx]
                new_maxes = np.max(corr[None, :] + deltas, axis=1)
                best_j = int(np.argmin(new_maxes))
                if new_maxes[best_j] < max_corr - 1e-12:
                    j = zeros_idx[best_j]
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr + G[i] - G[j]
                    max_corr = float(new_maxes[best_j])
                    if max_corr < best_max:
                        best_max = max_corr
                        best = h.copy()
                    improved = True
                    G = compute_G(h)
        
        if not improved:
            break
    
    return best, (2.0 / n) * best_max

def subgradient_optimize(h_init, max_steps=20000, time_limit=None, t_start=None):
    n = len(h_init)
    h = project_feasible(h_init.copy(), n)
    step = 10.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.3
    
    for it in range(max_steps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(np.max(corr))
        c5_val = (2.0 / n) * max_val
        if c5_val < best_val:
            best_val = c5_val
            best_h = h.copy()
        
        active = np.where(corr >= max_val - 1e-12)[0]
        grad = np.zeros(n)
        omh = 1.0 - h
        for k_idx in active:
            k = int(k_idx) - (n - 1)
            if k >= 0:
                grad[:n-k] += omh[k:]
                grad[k:] += -h[:n-k]
            else:
                l = -k
                grad[l:] += omh[:n-l]
                grad[:n-l] += -h[l:]
        grad *= (2.0 / n)
        
        momentum = beta * momentum - step * grad
        h = h + momentum
        h = project_feasible(h, n)
        
        if it % 500 == 0:
            step *= 0.85
    
    return best_h, best_val
