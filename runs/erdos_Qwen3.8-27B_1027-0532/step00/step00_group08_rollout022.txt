                t0 = time.time()
                h_tr, c5_tr = try_resolution(best_h, best_c5, n_target, t0, min(remaining * 0.3, 100), rng, None)
                if c5_tr < best_c5:
                    best_c5 = c5_tr
                    best_h = h_tr

    if best_h is not None:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining > 10:
            n = len(best_h)
            t0 = time.time()
            h_final, c5_final = optimize_phase(best_h, n / 2.0, 200.0, 3000.0, t0, remaining - 3, rng)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    if best_h is not None:
        try:
            fc = evaluate_erdos_solution(best_h)
            if fc is not None and fc < best_c5:
                best_c5 = fc
        except Exception:
            pass

    return (best_h, best_c5, len(best_h))
