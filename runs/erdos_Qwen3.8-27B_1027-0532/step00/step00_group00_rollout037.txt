import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def simulated_annealing_binary(n, initial=None, n_iter=1000000, T_start=1.0, T_end=0.001):
    if initial is not None:
        h = initial.copy()
    else:
        h = np.zeros(n)
        h[:n//2] = 1.0
        np.random.shuffle(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    for i in range(n_iter):
        T = T_start * (T_end / T_start) ** (i / n_iter)
        
        # Pick random swap
        i_idx = np.random.randint(n)
        j_idx = np.random.randint(n)
        if i_idx == j_idx or h[i_idx] == h[j_idx]:
            continue
        
        # Swap
        h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
        new_c5 = compute_c5(h)
        
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Undo
            h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def sa_optimize(h, n_iter, T_start, T_end, time_limit=None, start_time=None):
    n = len(h)
    h = h.astype(float).copy()
    current_c5, _ = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    no_improve = 0
    patience = 50000
    
    for i in range(n_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        T = T_start * (T_end / T_start) ** (i / n_iter)
        
        # Pick random positions
        while True:
            i_idx = np.random.randint(n)
            j_idx = np.random.randint(n)
            if i_idx != j_idx and h[i_idx] != h[j_idx]:
                break
        
        # Swap
        h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
        new_c5, _ = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            no_improve += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
        else:
            h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
        
        # Restart if stuck
        if no_improve > patience:
            h = best_h.copy()

            n_iter = max(10000, int(80000 * time_per_n / (15.0 * n_attempts)))
            T_s = 0.08 if n < 100 else 0.05
            T_e = 0.003

            h_sa, c5_sa = sa_optimize(h_init, n_iter, T_s, T_e, n_deadline)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
                best_n = n

            if time.time() < hard_deadline - 15:
                ls_dl = min(time.time() + 5.0, n_deadline)
                h_ls, c5_ls = local_search(h_sa, 30000, ls_dl)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls
                    best_n = n

    remaining = hard_deadline - time.time()
    if remaining > 10 and best_h is not None:
        n_iter = int(200000 * remaining / 120)
        h_final, c5_final = targeted_local_search(best_h, n_iter, hard_deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = len(h_final)

    remaining = hard_deadline - time.time()
    if remaining > 5 and best_h is not None:
        n_iter = int(100000 * remaining / 60)
        h_final, c5_final = local_search(best_h, n_iter, hard_deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = len(h_final)

    return best_h, best_c5, best_n
