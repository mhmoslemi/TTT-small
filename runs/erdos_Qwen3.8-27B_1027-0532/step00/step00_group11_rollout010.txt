- Update p based on which constraints are violated

def optimize_multiplicative_weights(h0, n, target_sum, n_iter=1000, rng=None):
    h = h0.copy()
    # c_k for k = 0, ..., 2n-2 (171 lags for n=86)
    n_lags = 2 * n - 1
    
    # p is a distribution over lags
    p = np.ones(n_lags) / n_lags
    
    for it in range(n_iter):
        # Compute current c_k values
        corr = np.correlate(h, 1-h, mode='full')
        # Find the max lag
        k_max = np.argmax(corr)
        
        # Update p (exponentiated gradient)
        p = p * np.exp(0.1 * corr / max(np.max(corr), 1e-10))
