```python
import numpy as np
import time
from scipy import optimize

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_all(h):
    """Compute all correlation values times dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def gradient_active_lag(h, shift):
    """
    Compute gradient of the correlation at a specific shift.
    shift = lag_index - (n-1), where lag_index is the index in the 'full' correlation output.
    If shift >= 0: c = sum_{i=0}^{n-1-shift} h[i] * (1 - h[i+shift])
    If shift < 0: c = sum_{i=-shift}^{n-1} h[i] * (1 - h[i+shift])
    """
    n = len(h)
    grad = np.zeros(n)
    if shift == 0:
        # c = sum_i h[i] * (1 - h[i]) = sum(h) - sum(h^2)
        # d/dh[j] = 1 - 2*h[j]
        grad = 1.0 - 2.0 * h
    elif shift > 0:
        # c = sum_{i=0}^{n-1-shift} h[i] * (1 - h[i+shift])
        # d/dh[j] = (1 - h[j+shift]) for j in [0, n-1-shift]
        #           - h[j-shift] for j in [shift, n-1]
        # Combined: for j in [0, n-1-shift]: grad[j] += (1 - h[j+shift])
        #           for j in [shift, n-1]: grad[j] -= h[j-shift]
        m = n - shift
        grad[:m] += (1.0 - h[shift:])
        grad[shift:] -= h[:m]
    else:
        s = -shift
        # c = sum_{i=s}^{n-1} h[i] * (1 - h[i-s])
        # d/dh[j] = (1 - h[j-s]) for j in [s, n-1]
        #           - h[j+s] for j in [0, n-1-s]
        m = n - s
        grad[s:] += (1.0 - h[:m])
        grad[:m] -= h[s:]
    return grad

def project_to_constraint(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 <= x_i <= 1}"""
    # Binary search for lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def interpolate_to_res(h, new_n):
    """Interpolate h from its current resolution to new_n points."""
    old_n = len(h)
    # Use linear interpolation
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    return h_interp

def gradient_optimize(h, budget_s, seed=42, initial_lr=0.05, min_lr=0.001):
    """Optimize h using subgradient method on active lags."""
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    h = project_to_constraint(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = initial_lr
    start_time = time.time()
    stall_count = 0
    patience = 500
    
    while time.time() - start_time < budget_s:
        # Compute all correlations
        c5_all = compute_c5_all(h)
        current_max = np.max(c5_all)
        
        # Find top-k active lags
        k_active = min(5, len(c5_all))
        top_indices = np.argsort(c5_all)[-k_active:]

        grad = np.zeros(n)

            shift = idx - (n - 1)
            grad += weight * gradient_active_lag(h, shift)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm
        
        # Take step

        h_new = project_to_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new)

            best_h = h_new.copy()
            h = h_new

                if c5_pert < best_c5 * 1.1:  # Accept if not too bad
                    h = h_perturbed

def pair_swap_optimize(h, budget_s, seed=42):
    """Local optimization using pair swaps."""
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    n = len(h)
    target_sum = n / 2.0
    
    h = project_to_constraint(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Try transferring from i to j or j to i
        for direction in [1, -1]:
            # Try different epsilon values
            for eps in [0.01, 0.02, 0.05, 0.1]:
                h_test = h.copy()
                h_test[i] += direction * eps
                h_test[j] -= direction * eps
                
                if h_test[i] < -1e-9 or h_test[i] > 1 + 1e-9:
                    continue
                if h_test[j] < -1e-9 or h_test[j] > 1 + 1e-9:
                    continue
                
                h_test = np.clip(h_test, 0, 1)
                # Re-project to maintain sum

                c5_test = compute_c5(h_test)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
                    h = h_test
                    break
        else:

        break
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 <= x_i <= 1}"""
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_max_corr(h, top_k=5):
    """
    Compute a subgradient of the max correlation.
    Returns gradient vector.
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Get top-k lags
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    
    # Compute weighted gradient
    grad = np.zeros(n)
    weights = corr[sorted_idx]
    weights = weights / (weights.sum() + 1e-15)  # normalize

        shift = idx - (n - 1)

            grad += w * (1.0 - 2.0 * h)
        elif shift > 0:

            grad[:m] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:m]
        else:

            m = n - s
            grad[s:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[s:]
```

```python
    weights = np.maximum(corr[top_idx], 0.0)

        weights = np.ones(top_k) / top_k

        weights /= ws
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        shift = int(idx) - (n - 1)
        grad += w * grad_shift(h, shift)
    return grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, shift):
    n = len(h)
    if shift == 0:
        return 1.0 - 2.0 * h
    elif shift > 0:
        m = n - shift
        g = np.zeros(n)
        g[:m] -= h[shift:]
        g[shift:] += 1.0 - h[:m]
        return g
    else:
        s = -shift
        m = n - s
        g = np.zeros(n)
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
        return g

def weighted_gradient(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    w = np.maximum(corr[idx], 0.0)
    ws = w.sum()
    if ws < 1e-14:
        w = np.ones(top_k) / top_k
    else:
        w /= ws
    g = np.zeros(n)
    for i, wi in zip(idx, w):
        g += wi * grad_shift(h, int(i) - (n - 1))
    return g

def optimize(h_init, time_budget, seed=42, top_k=5):
```
