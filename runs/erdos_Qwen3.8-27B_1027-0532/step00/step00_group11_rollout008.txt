            else:

            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h


def _obj_smooth(h):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * _dx_val
    shifted = _beta_val * corr
    m = np.max(shifted)
    if m > 500:
        return m / _beta_val
    return (m + np.log(np.sum(np.exp(shifted - m)))) / _beta_val


def _obj_direct(h):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * _dx_val


def _check_feasible(h, n, ts, tol=1e-6):
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - ts) > tol:
        return False
    return True


def _local_search(h, n, ts, dx, max_iters=2000, step_base=0.02):
    best_c5 = _compute_c5(h, dx)
    cur = h.copy()
    n_step = step_base
    no_improve = 0
    for it in range(max_iters):
        idx = np.random.randint(n)
        delta = np.random.choice([-1, 1]) * np.random.uniform(0.3, 1.0) * n_step
        h_try = cur.copy()
        h_try[idx] += delta
        h_try = _repair(h_try, n, ts)
        c5 = _compute_c5(h_try, dx)
        if c5 < best_c5:
            best_c5 = c5
            cur = h_try.copy()
            no_improve = 0
            n_step = min(0.05, n_step * 1.05)
        else:
            no_improve += 1
            if no_improve > 50:
                n_step *= 0.9
                no_improve = 0
                if n_step < 0.001:
                    break
    return cur, best_c5


def _nelder_mead(h, n, ts, dx, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint_sum}]
    try:
        res = minimize(_obj_direct, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        hc = _repair(res.x, n, ts)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)


def run(seed=42, budget_s=1300, **kwargs):
    global _target_sum, _dx_val, _beta_val, _n_pts
    np.random.seed(seed)
    t0 = time.time()

    n = 86
    dx = 2.0 / n
    ts = n / 2.0

    _target_sum = ts
    _dx_val = dx
    _n_pts = n

    h0 = np.array(initial_h_values, dtype=float)
    h0 = _repair(h0, n, ts)

    best_h = h0.copy()
    best_c5 = _compute_c5(h0, dx)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint_sum}]

    def elapsed():
        return time.time() - t0

    def time_ok(remaining=15):
        return elapsed() < (budget_s - remaining)

    # Phase 1: SLSQP from initial with increasing betas
    for beta in [40, 60, 100, 150, 250]:
        if not time_ok(60):
            break
        _beta_val = beta
        try:
            res = minimize(_obj_smooth, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-11})
            hc = _repair(res.x, n, ts)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 2: COBYLA from best
    if time_ok(80):
        try:
            res = minimize(_obj_direct, best_h, method='COBYLA',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'rhobeg': 0.03})
            hc = _repair(res.x, n, ts)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 3: SLSQP from perturbations of best
    for i in range(12):
        if not time_ok(60):
            break
        _beta_val = 80.0 + 20.0 * (i % 4)
        scale = 0.02 + 0.03 * (i % 5)
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = _repair(h_pert, n, ts)
        try:
            res = minimize(_obj_smooth, h_pert, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-10})
            hc = _repair(res.x, n, ts)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 4: Diverse starting points
    diverse_starts = []

    # Random smooth
    for i in range(3):
        t = np.linspace(0, 2, n)
        freq = np.random.uniform(1, 5, 3)
        amp = np.random.uniform(0.2, 0.5, 3)
        phase = np.random.uniform(0, 2 * np.pi, 3)
        h_smooth = np.zeros(n)
        for j in range(3):
            h_smooth += amp[j] * np.sin(freq[j] * t + phase[j])
        h_smooth = 0.5 + 0.3 * h_smooth
        h_smooth = _repair(h_smooth, n, ts)
        diverse_starts.append(h_smooth)

    # Anti-correlated with initial
    for i in range(3):
        h_anti = 0.5 + 0.5 * (1.0 - h0) + np.random.randn(n) * 0.05
        h_anti = _repair(h_anti, n, ts)
        diverse_starts.append(h_anti)

    # Step functions with structure
    for i in range(3):
        h_step = np.zeros(n)
        block_size = np.random.randint(3, 8)
        for start in range(0, n, block_size * 2):
            end = min(start + block_size, n)
            h_step[start:end] = 1.0
        h_step = _repair(h_step, n, ts)
        diverse_starts.append(h_step)

    # Periodic
    for i in range(2):
        period = np.random.randint(5, 12)
        duty = np.random.uniform(0.4, 0.6)
        h_periodic = np.zeros(n)
        for j in range(n):
            if (j % period) < int(period * duty):
                h_periodic[j] = 1.0
        h_periodic = _repair(h_periodic, n, ts)
        diverse_starts.append(h_periodic)

    for h_start in diverse_starts:
        if not time_ok(50):
            break
        _beta_val = 100.0
        try:
            res = minimize(_obj_smooth, h_start, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-10})
            hc = _repair(res.x, n, ts)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 5: Nelder-Mead on best
    if time_ok(60):
        hc, c5 = _nelder_mead(best_h, n, ts, dx, maxiter=3000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 6: Local search / simulated annealing
    if time_ok(30):
        best_h, best_c5 = _local_search(best_h, n, ts, dx, max_iters=5000, step_base=0.015)

    # Phase 7: SLSQP refine after local search
    if time_ok(60):
        for beta in [200, 400]:
            if not time_ok(40):
                break
            _beta_val = beta
            try:
                res = minimize(_obj_smooth, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-12})
                hc = _repair(res.x, n, ts)
                c5 = _compute_c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 8: More local search
    if time_ok(20):
        best_h, best_c5 = _local_search(best_h, n, ts, dx, max_iters=3000, step_base=0.008)

    # Phase 9: Final SLSQP
    if time_ok(30):
        _beta_val = 500.0
        try:
            res = minimize(_obj_smooth, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-13})
            hc = _repair(res.x, n, ts)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Final feasibility check and repair
    best_h = _repair(best_h, n, ts)
    best_c5 = _compute_c5(best_h, dx)

    # Verify
    if not _check_feasible(best_h, n, ts, tol=1e-4):
        best_h = _repair(best_h, n, ts)
        best_c5 = _compute_c5(best_h, dx)

    return (best_h, best_c5, n)
