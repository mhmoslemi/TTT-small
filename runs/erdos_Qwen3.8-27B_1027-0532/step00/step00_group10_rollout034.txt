```python
        noise_scale = 0.004 * max(0.0, 1.0 - 1.5 * progress)

        h_new = h - step * grad
        if noise_scale > 0:
            h_new = h_new + rng.randn(n) * noise_scale
        h_new = project_feasible(h_new, target)

        new_c5 = c5_of(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h = h_new

        if no_improve > 600:
            restart_count += 1
            if restart_count > 20:
                h = best_h + rng.randn(n) * 0.01
            else:
                scale = min(0.08, 0.02 + 0.005 * restart_count)
                h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            no_improve = 0

    if time.time() < deadline - 60:
        best_h, best_c5 = local_refine(best_h, dx, target, rng, max_iters=300)

    final_c5 = c5_of(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
```
