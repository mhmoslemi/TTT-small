def _repair(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference to elements that can accommodate it
    if diff > 0:
        # Need to increase sum
        mask = h < 1.0
        distributable = mask.astype(float)
        total_dist = distributable.sum()
        if total_dist > 0:
            h[mask] += diff * distributable[mask] / total_dist
    else:
        # Need to decrease sum
        mask = h > 0.0
        distributable = (h - 0).astype(float)  # h itself
        total_dist = distributable.sum()
        if total_dist > 0:
            h[mask] += diff * distributable[mask] / total_dist
    h = np.clip(h, 0, 1)
    # Final adjustment
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find one element to adjust
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0, 1)
    return h

def _repair(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        room[room < 1e-12] = 0.0
        total = room.sum()
        if total > 0:
            h += diff * room / total
    else:
        room = h.copy()
        room[room < 1e-12] = 0.0
        total = room.sum()
        if total > 0:
            h += diff * room / total
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
