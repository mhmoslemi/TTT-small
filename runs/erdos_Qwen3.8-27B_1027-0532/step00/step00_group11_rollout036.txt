def project_feasible(y, target_sum):

    n = len(y)
    lo = y.min() - 1.0

    diff = target_sum - h.sum()

def compute_c5(h, dx):

def compute_smooth_obj(h, alpha, dx):
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    return m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha

def compute_smooth_grad(h, alpha, dx, n):

    m = np.max(overlaps)
    weights = np.exp(alpha * (overlaps - m))

    weights /= sw
    
    # V is the weights array (length 2n-1)
    V = weights
    one_minus_h = 1.0 - h
    
    # Term 1: (V * (1-h))[i+n-1] for i=0..n-1
    conv1 = np.convolve(V, one_minus_h, mode='full')

    # Let me just use explicit indexing
    indices = np.arange(n)
    term2 = corr2[2*n-2 - indices]

For the first term: |∇²c_k| ≤ 2dx (roughly), so the first term is ≤ 2dx/α.
For the second term: |∇c_k|_2² ≤ n * (2dx)² = 4n dx², so the second term is ≤ 4n dx² / α².

Total L ≈ 2dx/α + 4n dx²/α² = 2dx/α + 4*(2/n)*n*(4/n²)/α² = 2dx/α + 16/(n*α²)

For n=86, dx=2/86≈0.023, α=100:
L ≈ 2*0.023/100 + 16/(86*10000) ≈ 0.00046 + 0.0000186 ≈ 0.00048

So step ≈ 1/L ≈ 2000. That's very large! This means the smooth objective is very flat for large α, and we can take large steps.

But wait, this doesn't account for the non-convexity. The Hessian can be negative in some directions (since c_k is indefinite). So the function is not convex, and large steps might jump over the minimum.

Let me use a more conservative step size: step = n / (4 * alpha * dx) = 86 / (4 * 100 * 0.023) ≈ 86/9.2 ≈ 9.3. Still large.

I think the practical approach is to use backtracking with an initial step of maybe 5-10 and let it adapt.

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, target_sum):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(300):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def smooth_obj_grad(h, alpha, dx, n):
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    e = np.exp(alpha * (overlaps - m))
    p = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    conv1 = np.convolve(p, 1.0 - h, mode='full')
    t1 = conv1[n - 1:2 * n - 1]
    corr2 = np.correlate(p, h, mode='full')
    t2 = corr2[2 * n - 2 - np.arange(n)]
    grad = (t1 - t2) * dx
    return obj, grad

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tl():
        return budget_s - (time.time() - start_time)

    # Phase 1: Multi-scale projected gradient with momentum
    alphas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for alpha in alphas:
        if tl() < 120:
            break

        h = best_h.copy()
        obj, grad = smooth_obj_grad(h, alpha, dx, n)
        step = max(0.1, 5.0 / (alpha * dx + 0.1))
        prev_grad = np.zeros(n)
        mu = 0.85

        for it in range(5000):
            if tl() < 40:
                break

            v = grad + mu * prev_grad
            prev_grad = grad.copy()

            s = step
            improved = False
            for _ in range(50):
                h_trial = project_feasible(h - s * v, target_sum)
                obj_trial, _ = smooth_obj_grad(h_trial, alpha, dx, n)
                if obj_trial < obj - 1e-15:
                    h = h_trial
                    obj = obj_trial
                    improved = True
                    break
                s *= 0.5

            if not improved:
                step *= 0.5
                if step < 1e-12:
                    break
            else:
                step = min(s * 1.5, 50.0 / (alpha * dx + 0.1))

            if (it + 1) % 200 == 0:
                c5_val = compute_c5(h, dx)
                if c5_val < best_c5:

        if c5_val < best_c5:

            best_c5 = c5_val

    # Phase 2: SLSQP refinement with smooth objective
    if tl() > 60:
        for alpha in [200.0, 500.0, 1000.0]:
            if tl() < 30:
                break

            def obj_func(x):
                o, _ = smooth_obj_grad(x, alpha, dx, n)
                return o

            def grad_func(x):
                _, g = smooth_obj_grad(x, alpha, dx, n)
                return g

            constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                            'jac': lambda x: np.ones(n)}]
            bounds = [(0.0, 1.0)] * n

            try:

                               jac=grad_func, bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 200, 'ftol': 1e-12})
                if res.success and res.x is not None:

            except Exception:

    if tl() > 30:

            j = rng.randint(n)

            if h[i] + delta < 0 or h[j] - delta > 1:

            h[i] += delta
            h[j] -= delta
            c5_t = compute_c5(h, dx)

            if c5_t < base_c5:
                base_c5 = c5_t
                if c5_t < best_c5:
                    best_h = h.copy()

                h[j] += delta

        if c5_final < best_c5:

            best_c5 = c5_final

    # Phase 4: Fine-grained coordinate optimization on worst lags
    if tl() > 20:
        h = best_h.copy()
        for it in range(200000):
            if tl() < 5:
                break

            overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
            k_worst = int(np.argmax(overlaps))
            k_val = k_worst - (n - 1)

            idx = np.arange(n)
            grad_k = np.zeros(n)
            v1 = (idx - k_val >= 0) & (idx - k_val < n)
            grad_k[v1] = (1.0 - h[idx[v1] - k_val])
            v2 = (idx + k_val >= 0) & (idx + k_val < n)
            grad_k[v2] -= h[idx[v2] + k_val]
            grad_k *= dx

            for s in [0.01, 0.005, 0.002, 0.001]:
                h_trial = project_feasible(h - s * grad_k, target_sum)
                c5_t = compute_c5(h_trial, dx)
                if c5_t < compute_c5(h, dx) - 1e-12:
                    h = h_trial
                    if c5_t < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_t
                    break

    return (best_h, best_c5, n)
