def project_onto_polytope(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = h.min() - target_sum/n, h.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune: distribute rounding error
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        # Add/subtract to elements not at boundary
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj

    return np.max(corr) * dx

    corr = np.correlate(h, 1 - h, mode="full")

    grad = np.zeros(N)

    vals = vals - np.max(vals)
    return float(np.log(np.sum(np.exp(vals))) / alpha)

def soft_max_gradient(h, dx, alpha):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = alpha * corr * dx
    vals = vals - np.max(vals)
    w = np.exp(vals)
    s = w.sum()
    if s < 1e-300:
        return np.zeros(N)
    w = w / s
    grad = np.zeros(N)
    one_h = 1.0 - h
    for k in range(2 * N - 1):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        shift = k - (N - 1)
        if shift >= 0:
            sv = shift
            if sv < N:
                grad[:N - sv] += wk * one_h[sv:N]
                grad[sv:] += -wk * h[:N - sv]
        else:
            sv = -shift
            if sv < N:
                grad[sv:] += wk * one_h[:N - sv]
                grad[:N - sv] += -wk * h[sv:]
    return grad

def subgradient_at_max(h, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_k = np.argmax(corr)
    shift = max_k - (N - 1)
    grad = np.zeros(N)
    one_h = 1.0 - h
    if shift >= 0:
        sv = shift
        if sv < N:
            grad[:N - sv] += one_h[sv:N]
            grad[sv:] += -h[:N - sv]
    else:
        sv = -shift
        if sv < N:
            grad[sv:] += one_h[:N - sv]
            grad[:N - sv] += -h[sv:]
    return dx * grad

def pair_swap_search(h, dx, target_sum, max_iters, time_fn):
    N = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    deltas = [0.02, 0.05, 0.1, 0.2, 0.5]
    for it in range(max_iters):
        if time_fn() < 15:
            break
        j, k = np.random.randint(0, N, 2)
        if j == k:
            continue
        save_j, save_k = best_h[j], best_h[k]
        best_delta = 0.0
        for d in deltas:
            for sign in [1, -1]:
                nj = save_j + sign * d
                nk = save_k - sign * d
                if 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
                    best_h[j] = nj
                    best_h[k] = nk
                    c5_test = compute_c5(best_h, dx)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_delta = sign * d
        best_h[j] = save_j + best_delta
        best_h[k] = save_k - best_delta
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    N = 95
    dx = 2.0 / N
    target_sum = N / 2.0
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != N:
            h0 = np.interp(np.linspace(0, len(h0) - 1, N), np.arange(len(h0)), h0)
        h0 = project_onto_polytope(h0, target_sum)
    except (NameError, AttributeError, TypeError):
        h0 = np.zeros(N)
        h0[:47] = 1.0
        h0[47] = 0.5
        h0[48:] = 0.0

    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    # Phase 1: Projected gradient with soft-max, increasing alpha
    alpha_schedule = [5, 10, 20, 50, 100, 200, 400, 800]
    for alpha in alpha_schedule:
        if time_left() < 60:
            break
        h_curr = best_h.copy()
        lr = 0.5 / alpha
        prev_obj = soft_max_value(h_curr, dx, alpha)
        max_stall = 200
        stall_count = 0
        for step in range(3000):
            if time_left() < 60:
                break
            grad = soft_max_gradient(h_curr, dx, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            h_new = h_curr - lr * grad
            h_new = project_onto_polytope(h_new, target_sum)
            obj_new = soft_max_value(h_new, dx, alpha)
            if obj_new < prev_obj:
                h_curr = h_new
                prev_obj = obj_new
                stall_count = 0
                lr = min(lr * 1.05, 2.0 / alpha)
            else:
                lr *= 0.5
                stall_count += 1
                if lr < 1e-8:
                    break
                h_new = h_curr - lr * grad
                h_new = project_onto_polytope(h_new, target_sum)
                obj_new = soft_max_value(h_new, dx, alpha)
                if obj_new < prev_obj:
                    h_curr = h_new
                    prev_obj = obj_new
                    stall_count = 0
            c5_val = compute_c5(h_curr, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_curr.copy()
            if stall_count > max_stall:
                break

    # Phase 2: Subgradient + projected gradient with exact max
    if time_left() > 90:
        h_curr = best_h.copy()
        lr = 0.01
        for step in range(5000):
            if time_left() < 30:
                break
            grad = subgradient_at_max(h_curr, dx)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            h_new = h_curr - lr * grad
            h_new = project_onto_polytope(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new <= compute_c5(h_curr, dx):
                h_curr = h_new
                lr = min(lr * 1.02, 0.1)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_curr.copy()
            else:
                lr *= 0.7
                if lr < 1e-10:
                    break

    # Phase 3: scipy SLSQP
    if time_left() > 120:
        def obj_fn(x):
            return compute_c5(x, dx)

        def obj_grad(x):
            return subgradient_at_max(x, dx)

        bounds = Bounds(0.0, 1.0)
        cons = LinearConstraint(np.ones((1, N)), target_sum, target_sum)
        try:
            res = minimize(obj_fn, best_h, method='SLSQP', jac=obj_grad,
                          bounds=bounds, constraints=[cons],
                          options={'maxiter': 500, 'ftol': 1e-12})
            if res.x is not None and np.all(res.x >= -0.1) and np.all(res.x <= 1.1):
                res.x = project_onto_polytope(res.x, target_sum)
                c5_val = compute_c5(res.x, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = res.x.copy()
        except Exception:
            pass

    # Phase 4: Pair-swap local search
    if time_left() > 30:
        best_h, best_c5 = pair_swap_search(best_h, dx, target_sum, 50000, time_left)

    # Phase 5: More projected gradient from best found
    if time_left() > 30:
        h_curr = best_h.copy()
        for alpha in [200, 500, 1000]:
            if time_left() < 20:
                break
            lr = 0.3 / alpha
            for step in range(2000):
                if time_left() < 20:
                    break
                grad = soft_max_gradient(h_curr, dx, alpha)
                h_new = h_curr - lr * grad
                h_new = project_onto_polytope(h_new, target_sum)
                obj_new = soft_max_value(h_new, dx, alpha)
                obj_old = soft_max_value(h_curr, dx, alpha)
                if obj_new < obj_old:
                    h_curr = h_new
                    lr = min(lr * 1.03, 1.0 / alpha)
                else:
                    lr *= 0.5
                    if lr < 1e-9:
                        break
                c5_val = compute_c5(h_curr, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_curr.copy()

    # Phase 6: Final pair-swap polish
    if time_left() > 15:
        best_h, best_c5 = pair_swap_search(best_h, dx, target_sum, 30000, time_left)

    # Final cleanup
    best_h = project_onto_polytope(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, N)
