    h_opt = project_feasible(res.x, target_sum)
    return h_opt, c5_value(h_opt)


def binary_local_search(h_bin, n_iter, rng):
    n = len(h_bin)
    h = h_bin.copy()
    best_val = c5_value(h)
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        val = c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_val


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError, TypeError):
        h_init = None

    if h_init is None:
        h_init = np.full(45, 0.5)

    n0 = len(h_init)
    ts0 = n0 / 2.0
    h_init = project_feasible(h_init, ts0)

    best_h = h_init.copy()
    best_c5 = c5_value(best_h)

    for alpha in [10, 50, 100, 300, 1000]:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 60:
            break
        h_opt, c5_opt = optimize_one(best_h, ts0, alpha, 300, 1000.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    n_restarts = 30
    for r in range(n_restarts):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 45:
            break
        scale = 0.1 * max(1.0, 1.0 - r / n_restarts)
        h_pert = best_h + rng.randn(n0) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_feasible(h_pert, ts0)
        h_opt, c5_opt = optimize_one(h_pert, ts0, 500, 200, 1000.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    resolutions = []
    for mult in [2, 4, 8, 16]:
        nt = n0 * mult
        if nt > 2000:
            break
        resolutions.append(nt)

    for n_target in resolutions:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 60:
            break
        n_old = len(best_h)
        x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        ts_new = n_target / 2.0
        h_interp = project_feasible(h_interp, ts_new)

        for alpha in [100, 500, 1000]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 30:
                break
            h_opt, c5_opt = optimize_one(h_interp, ts_new, alpha, 200, 1000.0)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            h_interp = best_h

        elapsed = time.time() - t_start
        if budget_s - elapsed < 30:
            continue
        if n_target % 2 == 0:
            target_ones = n_target // 2
            h_bin = np.zeros(n_target)
            sorted_idx = np.argsort(best_h)[::-1]
            h_bin[sorted_idx[:target_ones]] = 1.0
            n_ls = min(5000, max(500, int((budget_s - (time.time() - t_start)) / 10)))
            h_bin_result, c5_bin = binary_local_search(h_bin, n_ls, rng)
            if c5_bin < best_c5:
                best_h = h_bin_result
                best_c5 = c5_bin

    elapsed = time.time() - t_start
    if budget_s - elapsed > 20:
        n = len(best_h)
        ts = n / 2.0
        for alpha in [2000, 5000]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 15:
                break
            h_opt, c5_opt = optimize_one(best_h, ts, alpha, 500, 5000.0)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    return best_h, best_c5, len(best_h)
