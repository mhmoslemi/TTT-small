    return m

def repair(h, target):

    if s > target:
        # Scale down
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        # Fix residual
        s2 = np.sum(h)
        if s2 > target:
            # Reduce from elements that are not at 0
            diff = s2 - target
            mask = h > 1e-10
            if np.sum(mask) > 0:
                h[mask] -= diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)

        # Scale up
        diff = target - s
        room = 1.0 - h
        r = np.sum(room)
        if r > 1e-10:

            h = np.clip(h, 0.0, 1.0)
        # Fix residual
        s2 = np.sum(h)
        if s2 < target:
            diff2 = target - s2
            mask = h < 1.0 - 1e-10
            if np.sum(mask) > 0:
                h[mask] += diff2 / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h
