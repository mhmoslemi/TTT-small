import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return np.max(c)

def compute_subgradient(h, k):
    """Compute subgradient of c[k] with respect to h."""
    n = len(h)
    d = k - (n - 1)
    g = np.zeros(n)
    for m in range(n):
        # First term: I(m ∈ S_k) * (1 - h[m-d])
        if 0 <= m - d <= n - 1:
            g[m] += (1 - h[m - d])
        # Second term: -I(m+d ∈ [0,n-1]) * h[m+d]
        if 0 <= m + d <= n - 1:
            g[m] -= h[m + d]
    return g

def project_feasible(p, target_sum):
    """Project p onto {h : 0 ≤ h ≤ 1, sum(h) = target_sum}."""
    # Find λ such that sum(clip(p - λ, 0, 1)) = target_sum
    lo, hi = p.min() - target_sum, p.max() + target_sum

        s = np.sum(np.clip(p - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2

import numpy as np
import time

def _sg(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += 1.0 - h[:n - d]
            g[:n - d] -= h[d:]
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += 1.0 - h[dd:]
            g[dd:] -= h[:n - dd]
    return g

def _proj(p, S):
    lo = p.min() - 2.0
    hi = p.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(p - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) * 0.5, 0.0, 1.0)

def _c5val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n))

def _grad_and_val(h, n, beta):
    corr = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
    cmax = corr.max()
    w = np.exp(beta * (corr - cmax))
    wsum = w.sum()
    if wsum > 1e-300:
        w /= wsum
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            d = k - (n - 1)
            g += w[k] * _sg(h, d, n)
    return g, float(cmax)

def _optimize_start(h0, n, S, rng, t_end):
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5val(h)
    alpha = 0.08
    mom = 0.9
    v = np.zeros(n)
    beta = 5.0

        v = mom * v + g
        h_new = _proj(h - alpha * v, S)
        new_val = _c5val(h_new)
        if new_val < cur_val - 1e-13:
            h = h_new

                best_val = new_val

                v = np.zeros(n)
                alpha = 0.08
                beta = 5.0
                no_improve = 0
    return best_h, best_val
