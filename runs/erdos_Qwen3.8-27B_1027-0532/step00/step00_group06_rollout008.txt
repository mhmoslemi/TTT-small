    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_corr_full(h)) * dx)

def project_onto_box_sum(h, target_sum):
    n = len(h)
    target_sum = float(np.clip(target_sum, 0.0, float(n)))
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - h_proj.sum()
    if abs(residual) > 1e-14:
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        nf = int(free.sum())
        if nf > 0:
            h_proj[free] += residual / nf
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smoothed_subgradient(h, beta):
    n = len(h)
    corr = compute_corr_full(h)
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    weights = np.exp(shifted)
    wsum = weights.sum()
    weights /= wsum
    
    grad = np.zeros(n)
    j = np.arange(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-9:
            continue
        s = m - (n - 1)
        v1 = (j + s >= 0) & (j + s < n)
        if v1.any():
            grad[v1] += w * (1.0 - h[j[v1] + s])
        v2 = (j - s >= 0) & (j - s < n)
        if v2.any():
            grad[v2] -= w * h[j[v2] - s]
    return grad, float(max_c)

def opt_loop(h, dx, target_sum, time_fn, seed):
    n = len(h)
    np.random.seed(seed)
    beta = 30.0
    lr = 0.02
    momentum = 0.5
    v = np.zeros(n)
    c5 = compute_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    stall = 0
    
    while time_fn() > 10:
        grad, _ = smoothed_subgradient(h, beta)
        
        v = momentum * v + (1.0 - momentum) * grad
        gnorm = np.linalg.norm(v)
        if gnorm < 1e-14:
            break
        
        # Line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            h_trial = h - lr * factor * v
            h_trial = project_onto_box_sum(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < c5 - 1e-12:
                h = h_trial
                c5 = c5_trial
                improved = True
                break
        
        if improved:
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr = min(lr * 1.01, 0.2)
            if best_c5 < compute_c5(project_onto_box_sum(best_h + np.random.randn(n)*0.01, target_sum), dx):
                pass
        else:
            stall += 1
            lr = max(lr * 0.7, 0.0001)
            if stall > 300:
                # Restart with perturbation
                h = best_h + np.random.randn(n) * 0.05
                h = project_onto_box_sum(h, target_sum)
                c5 = compute_c5(h, dx)
                v = np.zeros(n)
                lr = 0.02
                stall = 0
                beta = min(beta * 1.3, 100.0)
    
    return best_h, best_c5

            h = np.clip(f(x_new), 0.0, 1.0)
