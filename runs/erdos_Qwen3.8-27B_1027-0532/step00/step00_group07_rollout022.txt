def alternating_step(h, n, target, max_iters=50):
    """Alternate between optimizing h given g=1-h"""
    h = h.copy()
    for it in range(max_iters):
        g = 1.0 - h
        # Solve LP: min t s.t. corr(h, g, k) <= t for all k, sum(h)=target, 0<=h<=1
        # Variables: h[0],...,h[n-1], t (index n)
        # Objective: [0,...,0, 1] (minimize t)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A @ x <= b
        # For each lag k (from -(n-1) to (n-1)):
        #   sum_i h[i+k]*g[i] - t <= 0
        #   => sum_i g[i]*[i+k valid] * h[i+k] - t <= 0
        #   In terms of variables h[0..n-1], t:
        #   sum_j (sum_{i: i+k=j} g[i]) * h[j] - t <= 0
        #   sum_j g[j-k] * h[j] - t <= 0  (where g[j-k] is 0 if j-k not in [0,n))
        #   => row: [g[j-k] for j=0..n-1, -1] dot [h, t] <= 0
        
        A_rows = []
        b_vals = []
        for k in range(-(n-1), n):
            row = np.zeros(n + 1)
            for j in range(n):
                i = j - k  # we need g[i] where i = j-k, valid if 0 <= i < n
                # Actually: corr at lag k = sum_i h[i+k]*g[i]
                # = sum_j h[j]*g[j-k] where j = i+k, so i = j-k
                # g[j-k] is valid when 0 <= j-k < n
                if 0 <= j - k < n:
                    row[j] = g[j - k]
            row[-1] = -1.0
            A_rows.append(row)
            b_vals.append(0.0)
        
        A = np.array(A_rows)
        b = np.array(b_vals)
        
        # Equality constraint: sum(h) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds: 0 <= h[i] <= 1, t is free (but will be >= 0 automatically)
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(0, None)],

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        for j in range(n):
            g = 0.0
            a = j - k
            b = j + k
            if 0 <= a < n:
                g += 1.0 - h[a]
            if 0 <= b < n:
                g -= h[b]
            grad[j] += g
    return grad / len(active) * dx

def sum_eq(x, target):
    return x.sum() - target

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        t[i] = float(bin(i).count('1') % 2)
    return t

def lse_obj(x, beta, n):
    c = np.correlate(x, 1 - x, 'full') * (2.0 / n)
    m = c.max()
    return m + float(np.log(np.sum(np.exp(beta * c - beta * m))) / beta)

def lse_grad(x, beta, n):
    c = np.correlate(x, 1 - x, 'full') * (2.0 / n)
    bc = beta * c
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()
    grad = np.zeros(n)
    for li in range(len(c)):
        if w[li] < 1e-12:
            continue
        k = li - (n - 1)
        for j in range(n):
            v = 0.0
            a = j - k
            b = j + k
            if 0 <= a < n:
                v += 1.0 - x[a]
            if 0 <= b < n:
                v -= x[b]
            grad[j] += w[li] * v
    return grad * (2.0 / n)

def alt_lp_refine(h, n, target, max_iters=80):
    h = h.copy()
    current_c5 = c5_value(h)
    for it in range(max_iters):
        g = 1.0 - h
        nv = n + 1
        c_obj = np.zeros(nv)
        c_obj[-1] = 1.0
        A_rows = []
        b_vals = []
        for k in range(-(n - 1), n):
            row = np.zeros(nv)
            for j in range(n):
                i = j - k
                if 0 <= i < n:
                    row[j] = g[i]
            row[-1] = -1.0
            A_rows.append(row)
            b_vals.append(0.0)
        A = np.array(A_rows)
        b = np.array(b_vals)
        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0, 1)] * n + [(0, None)], method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        c5_new = c5_value(h_new)
        if c5_new < current_c5 - 1e-12:
            h = h_new
            current_c5 = c5_new
        else:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    n = 95
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (target,)}]

    candidates = []
    try:
        candidates.append(np.array(initial_h_values, dtype=float))
    except NameError:
        pass

    tm = thue_morse(n)
    candidates.append(fix_sum(tm.copy(), target))
    candidates.append(fix_sum(1.0 - tm, target))
    tm_rev = tm[::-1].copy()
    candidates.append(fix_sum(tm_rev, target))
    candidates.append(fix_sum(1.0 - tm_rev, target))

    for _ in range(8):
        candidates.append(fix_sum(rng.random(n), target))

    for bs in [2, 3, 4, 5, 7, 11, 13]:
        pat = np.array([1.0 if (i // bs) % 2 == 0 else 0.0 for i in range(n)])
        candidates.append(fix_sum(pat, target))

    best_h = None
    best_c5 = 1.0
    for cand in candidates:
        cv = c5_value(cand)
        if cv < best_c5:
            best_c5 = cv
            best_h = cand.copy()

    def time_ok(min_remaining):
        return budget_s - (time.time() - t0) > min_remaining

    # SLSQP from best candidate
    if time_ok(30):
        res = minimize(c5_value, best_h, jac=c5_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-13})
        h_try = fix_sum(res.x, target)
        cv = c5_value(h_try)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_try.copy()

    # SLSQP from other promising candidates
    for cand in candidates:
        if not time_ok(20):
            break
        cv_cand = c5_value(cand)
        if cv_cand < best_c5 + 0.06 and cv_cand < best_c5 + 1e-9:
            continue
        if cv_cand < best_c5 + 0.06:
            res = minimize(c5_value, cand, jac=c5_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-13})
            h_try = fix_sum(res.x, target)
            cv = c5_value(h_try)
            if cv < best_c5:
                best_c5 = cv
                best_h = h_try.copy()

    # Alternating LP refinement
    if time_ok(20):
        h_alt = alt_lp_refine(best_h, n, target, max_iters=100)
        cv = c5_value(h_alt)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_alt.copy()

    # LSE annealing
    for beta in [10, 50, 200, 1000]:
        if not time_ok(20):
            break
        res = minimize(lse_obj, best_h, args=(beta, n), jac=lse_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-13})
        h_try = fix_sum(res.x, target)
        cv = c5_value(h_try)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_try.copy()
        # Follow up with direct max
        if time_ok(15):
            res2 = minimize(c5_value, best_h, jac=c5_grad, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 200, 'ftol': 1e-14})
            h_try2 = fix_sum(res2.x, target)
            cv2 = c5_value(h_try2)
            if cv2 < best_c5:
                best_c5 = cv2
                best_h = h_try2.copy()

    # More alternating LP rounds
    for _ in range(5):
        if not time_ok(15):
            break
        h_alt = alt_lp_refine(best_h, n, target, max_iters=50)
        cv = c5_value(h_alt)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_alt.copy()
        else:
            break

    # Multi-start SLSQP
    n_starts = 0
    while time_ok(15):
        scale = max(0.02, 0.12 * np.exp(-0.04 * n_starts))
        h_start = best_h + rng.normal(0, scale, n)
        h_start = fix_sum(h_start, target)
        res = minimize(c5_value, h_start, jac=c5_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-13})
        h_try = fix_sum(res.x, target)
        cv = c5_value(h_try)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_try.copy()
            n_starts = 0
        else:
            n_starts += 1
        if n_starts > 200:
            break

    # Final polish
    if time_ok(10):
        res = minimize(c5_value, best_h, jac=c5_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-15})
        h_try = fix_sum(res.x, target)
        cv = c5_value(h_try)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_try.copy()

    # Final LP polish
    if time_ok(5):
        h_alt = alt_lp_refine(best_h, n, target, max_iters=30)
        cv = c5_value(h_alt)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_alt.copy()

    best_h = fix_sum(best_h, target)
    final_c5 = c5_value(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
