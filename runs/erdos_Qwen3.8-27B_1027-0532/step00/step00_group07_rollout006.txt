```python
def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        h0 = np.full(95, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def project_onto_feasible(h):
        """Project h onto {sum(h) = target_sum, 0 <= h <= 1}"""
        h = h.copy()
        h = np.clip(h, 0.0, 1.0)
        for _ in range(1000):
            s = np.sum(h)
            diff = target_sum - s
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        # Final adjustment
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-14:
            h[int(np.argmax(h if diff < 0 else np.argmin(h)))] += diff
            h = np.clip(h, 0.0, 1.0)
        return h
    
    def compute_subgradient(h):
        """Compute subgradient of max_p output[p] at h"""
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # Find all active (maximal) lags
        active = np.where(np.abs(corr - max_val) < 1e-10)[0]
        grad = np.zeros(n)
        for p in active:
            d = p - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j - d < n:
                    g += 1.0
                if 0 <= j + d < n:
                    g -= h[j + d]
                if 0 <= j - d < n:
                    g -= h[j - d]
                grad[j] += g / len(active)
        return grad, max_val
    
    # Initialize
    h = project_onto_feasible(h0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Subgradient optimization
    step_size = 0.1
    for iteration in range(50000):
        if time.time() - start_time > budget_s - 30:
            break
        
        grad, max_val = compute_subgradient(h)
        
        # Project gradient to be orthogonal to constraint
        grad = grad - np.mean(grad)
        
        h_new = h - step_size * grad
        h_new = project_onto_feasible(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Reduce step size slightly when improving
            step_size *= 0.999

            h_test = project_onto_feasible(h + perturbation)
            if compute_c5(h_test) < best_c5:
                h = h_test
                best_h = h.copy()
                best_c5 = compute_c5(h)
    
    return (best_h, best_c5, n)
```

```python
def gradient_for_lag(h, d, n):
    grad = np.zeros(n)
    # Term 1: [0 <= j-d < n] for j in 0..n-1
    # j-d >= 0 → j >= d (if d > 0), or always true (if d <= 0)
    # j-d < n → j < n+d (if d > 0, always true since j < n; if d < 0, j < n+d)
    if d >= 0:
        mask1 = (np.arange(n) >= d)  # j >= d
        # j + d < n → j < n - d
        mask2 = (np.arange(n) < n - d)  # for h[j+d]
    else:
        mask1 = (np.arange(n) < n + d)  # j < n+d = n-|d|
        mask2 = (np.arange(n) + d >= 0)  # j >= -d = |d|
    
    grad[mask1] += 1.0
    if d >= 0:
        grad[mask2] -= h[d:]  # h[j+d] for j < n-d
    else:
        grad[mask2] -= h[:n+d]  # h[j+d] for valid j
    
    # Term 3: -h[j-d] for valid j
    if d >= 0:
        # j-d >= 0 → j >= d
        mask3 = (np.arange(n) >= d)
        grad[mask3] -= h[:n-d]  # h[j-d] for j >= d
    else:
        # j-d < n → j < n+d
        mask3 = (np.arange(n) < n + d)
        grad[mask3] -= h[-d:]  # h[j-d] for valid j
    
    return grad
```

```python
def gradient_for_lag(h, d, n):
    grad = np.zeros(n)
    j = np.arange(n)
    
    # [0 <= j-d < n] → [max(0,d) <= j < min(n, n+d)]
    lo1 = max(0, d)
    hi1 = min(n, n + d)
    grad[lo1:hi1] += 1.0
    
    # h[j+d] for [0 <= j+d < n] → [max(0,-d) <= j < min(n, n-d)]
    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    grad[lo2:hi2] -= h[lo2+d:hi2+d]
    
    # h[j-d] for [0 <= j-d < n] → [max(0,d) <= j < min(n, n+d)]
    lo3 = max(0, d)
    hi3 = min(n, n + d)
    grad[lo3:hi3] -= h[lo3-d:hi3-d]
    
    return grad
```
