def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

def round_to_binary(h, n):

    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

    x_old = np.arange(n_old) * (2.0 / n_old)
