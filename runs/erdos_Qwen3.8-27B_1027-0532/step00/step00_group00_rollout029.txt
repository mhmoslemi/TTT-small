import numpy as np
import time

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _eval_move(h, corr, n, i, j, d):
    """Evaluate max correlation after moving d from h[j] to h[i]."""
    i_lo = max(0, (n-1) - i)
    i_hi = min(2*n-2, (n-1) + i)
    j_lo = max(0, (n-1) - j)
    j_hi = min(2*n-2, (n-1) + j)
    
    lo = min(i_lo, j_lo)
    hi = max(i_hi, j_hi)
    
    # Get affected slice
    affected = corr[lo:hi+1].copy()
    
    ks_i = np.arange(i_lo, i_hi + 1)
    affected[ks_i - lo] += d * (1.0 - h[i + ks_i - (n-1)])
    
    ks_j = np.arange(j_lo, j_hi + 1)
    affected[ks_j - lo] -= d * (1.0 - h[j + ks_j - (n-1)])
    
    affected_max = np.max(affected)
    
    # Unaffected
    unaff_max = 0.0
    if lo > 0:
        unaff_max = max(unaff_max, np.max(corr[:lo]))
    if hi < 2*n-2:
        unaff_max = max(unaff_max, np.max(corr[hi+1:]))
    
    return max(affected_max, unaff_max)

def _greedy_search(h, corr, n, max_rounds=100000):
