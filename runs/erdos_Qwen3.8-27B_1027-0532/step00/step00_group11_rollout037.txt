def coordinate_minimize(h, j, raw_corrs):
    """Find optimal value of h[j] given all other coordinates fixed."""
    N = len(h)
    t_current = h[j]
    
    # Compute a_d and b_d for all lags
    # Lag d corresponds to raw_corrs index (d + N - 1)
    a = np.zeros(2*N - 1)
    b = np.zeros(2*N - 1)
    
    for d in range(-(N-1), N):
        m = d + N - 1  # index in raw_corrs
        # Current contribution of h[j] to this lag
        contrib = 0.0
        if 0 <= j - d < N:
            contrib += t_current * (1.0 - h[j - d])
        if 0 <= j + d < N:
            contrib += h[j + d] * (1.0 - t_current)
        
        a[m] = (1.0 - h[j-d]) if 0 <= j-d < N else 0.0
        if 0 <= j + d < N:
            a[m] -= h[j + d]
        
        b[m] = raw_corrs[m] - contrib
    
    # Now minimize max_m (a[m]*t + b[m]) subject to 0 <= t <= 1
    # Binary search on V
    
    def feasible(V):
        L = 0.0
        U = 1.0
        for m in range(2*N - 1):
            if a[m] > 1e-12:
                # a[m]*t + b[m] <= V → t <= (V-b[m])/a[m]
                U = min(U, (V - b[m]) / a[m])
            elif a[m] < -1e-12:
                # a[m]*t + b[m] <= V → t >= (V-b[m])/a[m]
                L = max(L, (V - b[m]) / a[m])
            else:
                # a[m] ≈ 0, need b[m] <= V
                if b[m] > V:
                    return False
        return L <= U
    
    # Binary search
    lo = raw_corrs.min()  # lower bound on objective
    hi = raw_corrs.max()  # upper bound (current value)
    
    # Actually, a better lower bound: min over t of max_m (a[m]*t+b[m])
    # Let's just use lo = min_m b[m] (since at some t, the max is at least this)
    # And hi = max_m (a[m]*0+b[m]) or max_m (a[m]*1+b[m]) (current bounds)
    lo = min(b.min(), max(a[m]*0+b[m] for m in range(2*N-1)), max(a[m]*1+b[m] for m in range(2*N-1)))
    hi = max(a*0+b).max()  # value at t=0 (upper bound on min)
    hi = min(hi, (a*1+b).max())  # also value at t=1
    
    for _ in range(80):
        mid = (lo + hi) / 2
        if feasible(mid):
            hi = mid
        else:
            lo = mid
    
    V_opt = hi
    # Now find the optimal t
    L = 0.0
    U = 1.0
    for m in range(2*N - 1):
        if a[m] > 1e-12:
            U = min(U, (V_opt - b[m]) / a[m])
        elif a[m] < -1e-12:
            L = max(L, (V_opt - b[m]) / a[m])
    
    if L > U:
        # Numerical issue, use midpoint
        t_opt = (L + U) / 2
    else:
        t_opt = (L + U) / 2
    
    t_opt = np.clip(t_opt, 0.0, 1.0)
    return t_opt

    def project(h):

        np.clip(h, 0.0, 1.0, out=h)
        for _ in range(200):
            diff = target_sum - np.sum(h)
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                mask = h < 1.0 - 1e-14
            else:
                mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
        return h

    def c5_val(h):
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    def update_coord(h, j):
        h = h.copy()
        raw = np.correlate(h, 1.0 - h, mode="full")
        t = h[j]

        jd = j - d_vals
        jpd = j + d_vals
        vm = (jd >= 0) & (jd < N)
        vp = (jpd >= 0) & (jpd < N)

        hp = np.where(vp, h[np.clip(jpd, 0, N-1)], 0.0)

        vmf = vm.astype(float)
        vpf = vp.astype(float)
        a = (1.0 - hm) * vmf - hp * vpf
        contrib = t * (1.0 - hm) * vmf + hp * (1.0 - t) * vpf
        b = raw - contrib

        lo = np.max(-np.abs(a) + b)
        hi = min(b.max(), (a + b).max())

        if lo >= hi:
            t_new = t
        else:
            for _ in range(50):
                mid = (lo + hi) * 0.5
                pos = a > 1e-10
                neg = a < -1e-10
                zero = ~(pos | neg)

                    ok = False
                if ok:
                    U = 1.0
                    L = 0.0

    best_c5 = c5_val(best_h)
