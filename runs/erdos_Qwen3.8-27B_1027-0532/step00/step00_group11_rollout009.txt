import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    n_points = 86
    dx = 2.0 / n_points
    target_sum = n_points / 2  # = 43
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        # Create a random binary-ish starting point
        h0 = np.zeros(n_points)
        idx = rng.permutation(n_points)
        h0[idx[:int(target_sum)]] = 1.0
    
    # Ensure constraints
    h0 = np.clip(h0, 0, 1)
    if abs(np.sum(h0) - target_sum) > 1e-10:
        # Adjust to fix sum
        diff = target_sum - np.sum(h0)
        # Distribute the difference
        if diff > 0:
            mask = h0 < 1
        else:
            mask = h0 > 0
        n_adj = np.sum(mask)
        h0[mask] += diff / n_adj
        h0 = np.clip(h0, 0, 1)
    
    best_h = h0.copy()
    best_val = c5_eval(h0)
    
    def c5_eval(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Optimization loop
    max_total_time = budget_s - 10  # leave buffer
    
    n_attempts = 0
    while time.time() - start_time < max_total_time:
        n_attempts += 1
        # Create perturbation
        if n_attempts == 1:
            h_start = best_h.copy()
        else:
            # Mix of perturbations
            pert_type = rng.integers(0, 4)
            if pert_type == 0:
                # Small Gaussian noise
                h_start = best_h + rng.normal(0, 0.05, n_points)
            elif pert_type == 1:
                # Flip some values
                h_start = best_h.copy()
                n_flips = rng.integers(2, 8)
                flip_idx = rng.choice(n_points, n_flips, replace=False)
                for i in flip_idx:
                    h_start[i] = 1 - h_start[i]
            elif pert_type == 2:
                # Swap values
                h_start = best_h.copy()
                i, j = rng.choice(n_points, 2, replace=False)
                h_start[i], h_start[j] = h_start[j], h_start[i]
            else:
                # Random restart (less aggressive)
                h_start = best_h + rng.normal(0, 0.15, n_points)
        
        h_start = np.clip(h_start, 0, 1)

        diff = target_sum - np.sum(h_start)

            else:

                h_start[mask] += diff / n_adj
                h_start = np.clip(h_start, 0, 1)

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                       'jac': lambda x: np.ones(n_points)}]

    return grad * dx

def project_feasible(h, target_sum):
    """Project h onto {x : 0≤x≤1, sum(x)=target_sum}"""
    # Binary search for the shift
    def sum_clipped(shift):
        return np.sum(np.clip(h - shift, 0, 1))
    
    # If sum is already correct
    s = np.sum(np.clip(h, 0, 1))
    if abs(s - target_sum) < 1e-10:
        return np.clip(h, 0, 1)
    
    # We need to find shift such that sum_clipped(shift) = target_sum
    # sum_clipped is decreasing in shift
    lo, hi = -2.0, 2.0
    # Make sure we bracket
    while sum_clipped(lo) < target_sum:
        lo -= 1.0
    while sum_clipped(hi) > target_sum:
        hi += 1.0
    
    shift = brentq(lambda s: sum_clipped(s) - target_sum, lo, hi)
    return np.clip(h - shift, 0, 1)

    dx = 2.0 / n

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-8)[0]
    grad = np.zeros(n)
