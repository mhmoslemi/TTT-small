import numpy as np
import time


def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def continuous_search(h, n, n_iters, step, rng):
    h_c = h.copy()
    c5_c = evaluate_c5(h_c, n)
    best_local_h = h_c.copy()
    best_local_c5 = c5_c
    for _ in range(n_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = step * (1.0 if rng.random() < 0.5 else -1.0)
        hi = h_c[i] + d
        hj = h_c[j] - d
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h_c[i] = hi
        h_c[j] = hj
        c5_new = evaluate_c5(h_c, n)
        if c5_new < c5_c:
            c5_c = c5_new
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local_h = h_c.copy()
        else:
            h_c[i] -= d
            h_c[j] += d
    return best_local_h, best_local_c5


def binary_search(h, n, n_iters, rng):

    c5_c = evaluate_c5(h_c, n)
    best_local_h = h_c.copy()

                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_c[i] = 0.0
            h_c[j] = 1.0
            c5_new = evaluate_c5(h_c, n)
            if c5_new < c5_c:
                c5_c = c5_new
                if c5_new < best_local_c5:
                    best_local_c5 = c5_new
                    best_local_h = h_c.copy()
            else:
                h_c[j] = 0.0

            zeros = np.where(h_c == 0)[0]

            h_c[i1] = 0.0
