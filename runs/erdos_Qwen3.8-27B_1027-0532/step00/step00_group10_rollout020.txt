import numpy as np
import time

def compute_c5(h, n):
    """Compute C5 value."""
    dx = 2.0 / n
    comp = np.correlate(h, 1 - h, mode='full')
    return np.max(comp) * dx

def compute_gradient_soft(h, n, temperature=50.0):
    """Compute gradient of soft-max of overlaps."""
    dx = 2.0 / n
    comp = np.correlate(h, 1 - h, mode='full') * dx
    # Soft-max weights
    shifted = comp - np.max(comp)
    weights = np.exp(temperature * shifted)
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for k_idx in range(len(comp)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        # ∂f_k/∂h[j] = dx * [(1-h[j-k]) if 0≤j-k<n else 0 - h[j+k] if 0≤j+k<n else 0]
        # j-k in [0,n): j in [k, k+n) intersect [0, n)
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if hi1 > lo1:
            js = np.arange(lo1, hi1)
            grad[js] += w * dx * (1 - h[js - k])
        # j+k in [0,n): j in [-k, n-k) intersect [0, n)
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            js = np.arange(lo2, hi2)
            grad[js] -= w * dx * h[js + k]
    
    return grad

def project_feasible(h, n, target_sum):
    """Project onto {h: 0≤h≤1, sum=h_target}."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:

    lo, hi = -1.5, 1.5

        s_mid = np.sum(np.clip(h - mid, 0, 1))
        if s_mid > target_sum:
            lo = mid
        else:

def compute_c5(h, n):
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(comp)) * dx

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h, n, overlaps, temp=50.0):
    dx = 2.0 / n
    max_ov = np.max(overlaps)
    exp_vals = np.exp(temp * (overlaps - max_ov))
    total = np.sum(exp_vals)
    if total == 0:
        return np.zeros(n)
    weights = exp_vals / total
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            grad[idx] += w * dx * (1.0 - h[idx - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= w * dx * h[idx + k]
    return grad
