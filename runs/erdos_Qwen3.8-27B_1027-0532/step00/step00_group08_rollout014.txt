    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def compute_soft_gradient(h, n, sharpness=20.0):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # Soft weighting
    temps = (corr - max_val) * sharpness
    weights = np.exp(temps)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    
    g = np.zeros(n)

        else:

            g[kk:] -= w * h[:n-kk]

    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def compute_soft_gradient(h, n, sharpness=30.0):
    """Compute soft gradient of max c_k"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    temps = (corr - max_val) * sharpness
    # Clip to avoid underflow
    temps = np.maximum(temps, -50)
    weights = np.exp(temps)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    
    g = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)
        w = weights[m]
        if k >= 0:
            g[k:] += w * (1.0 - h[:n-k])
            g[:n-k] -= w * h[k:]
        else:
            kk = -k
            g[:n-kk] += w * (1.0 - h[kk:])
            g[kk:] -= w * h[:n-kk]
    
    return g
