```python
def optimize_local(h, max_iters=10000, step_size=0.05):
    """Local optimization using projected gradient / coordinate descent."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        improved = False
        for i in range(n):

            for delta in [step_size, -step_size, 2*step_size, -2*step_size]:
                h_new = h.copy()
                h_new[i] = np.clip(h[i] + delta, 0, 1)
                # Adjust sum constraint: redistribute difference
                diff = h[i] - h_new[i]  # amount to redistribute
                if abs(diff) > 1e-10:

                    others = np.arange(n)
                    others = others[others != i]
                    # Add diff/n to others, but clip
                    h_new_others = np.clip(h_new[others] + diff/len(others), 0, 1)

                    # Re-adjust if needed (iteratively)
                    for _ in range(10):

                        target_sum = n / 2.0
                        error = target_sum - current_sum
                        if abs(error) < 1e-10:
                            break
                        # Distribute error
                        can_add = (h_new < 1)
                        can_sub = (h_new > 0)
                        if error > 0:
                            mask = can_add
                        else:
                            mask = can_sub
                        if mask.sum() == 0:

                        h_new[mask] += error / mask.sum()
                        h_new = np.clip(h_new, 0, 1)
                
                c5_new = compute_c5(h_new)

                    improved = True
                    break

    return h, best_c5
```

```python
import numpy as np
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return corr * dx

def compute_c5(h):
    return np.max(compute_overlaps(h))

def project_onto_constraint(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def gradient_overlap(h, m):
    """Compute gradient of overlap_m with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # ∂overlap_m/∂h[t] = (1-h[t-m])·[0 ≤ t-m < n] - h[t+m]·[0 ≤ t+m < n]
    # For t from 0 to n-1:
    #   First term: valid when 0 ≤ t-m < n, i.e., m ≤ t < m+n (and t < n), so max(0,m) ≤ t < n
    #   Second term: valid when 0 ≤ t+m < n, i.e., -m ≤ t < n-m (and t ≥ 0), so 0 ≤ t < min(n, n-m)
    
    if m >= 0:
        # First term: t from m to n-1
        if m < n:
            grad[m:] += (1 - h[:n-m])
        # Second term: t from 0 to n-1-m
        if m < n:
            grad[:n-m] -= h[m:]
    else:
        m_abs = -m
        # First term: t from 0 to n-1-m_abs
        if m_abs < n:
            grad[:n-m_abs] += (1 - h[m_abs:])
        # Second term: t from m_abs to n-1
        if m_abs < n:
            grad[m_abs:] -= h[:n-m_abs]
    
    return grad

def optimize_gradient(h, max_iters=5000, lr_schedule='decay'):
    """Optimize using projected gradient descent."""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy().astype(float)
    h = project_onto_constraint(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        overlaps = compute_overlaps(h)
        c5 = np.max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find which lags achieve (or are close to) the maximum
        max_val = np.max(overlaps)
        near_max = np.abs(overlaps - max_val) < 1e-8
        n_max = np.sum(near_max)
        
        # Compute gradient (average over near-max lags)
        grad = np.zeros(n)
        center = n - 1  # index of zero lag in the full correlation
        for idx in np.where(near_max)[0]:
            m = idx - center  # the actual lag
            grad += gradient_overlap(h, m)
        grad /= n_max
        
        # Learning rate
        lr = 0.1 / (1 + it * 0.01)
```

```python
import numpy as np
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return np.max(compute_overlaps(h))

def project_onto_constraint(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on lambda."""
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_gradient(h, temperature=0.005):
    """Compute weighted gradient using softmax over overlaps."""
    n = len(h)
    overlaps = compute_overlaps(h)
    max_val = np.max(overlaps)
    
    # Soft weights
    exp_vals = np.exp((overlaps - max_val) / temperature)
    weights = exp_vals / exp_vals.sum()
    
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-10:
            continue
        m = idx - center
        if m >= 0:
            if m < n:
                grad[m:] += w * (1.0 - h[:n-m])
                grad[:n-m] -= w * h[m:]
        else:
            ma = -m
            if ma < n:
                grad[:n-ma] += w * (1.0 - h[ma:])
                grad[ma:] -= w * h[:n-ma]
    
    return grad, max_val

def optimize_gradient(h0, budget_s, initial_lr=0.1):
    """Projected gradient descent with adaptive learning rate."""
    n = len(h0)
    target_sum = n / 2.0
    h = project_onto_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = initial_lr
    start = time.time()
    
    while time.time() - start < budget_s:
        grad, c5 = soft_gradient(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try the step
        h_new = project_onto_constraint(h - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            lr = min(lr * 1.05, initial_lr)
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
    
    return best_h, best_c5

def anneal(h0, budget_s, T_start=0.01, T_end=0.0001):
    """Simulated annealing with swap moves."""
    n = len(h0)
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    elapsed = 0.0
    
    # Precompute for speed
    while elapsed < budget_s:
        # Determine temperature
        progress = elapsed / budget_s
        T = T_start * (T_end / T_start) ** progress
        
        # Pick a move: swap two random positions
        i, j = np.random.choice(n, 2, replace=False)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        if abs(hi - hj) < 1e-10:
            continue
        
        # Swap
        h[i], h[j] = hj, hi
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            best_c5 = max(best_c5, 0)  # keep best
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj  # revert
        
        elapsed = time.time() - start
    
    return best_h, best_c5

def interpolate_up(h, new_n):
    """Interpolate h from its current resolution to new_n points."""
    n = len(h)
    old_x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # centers
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(new_x, old_x, h)
    # Ensure sum constraint
    h_new = project_onto_constraint(h_new, new_n / 2.0)
    return h_new
```

```python
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def get_gradient(h, temp=0.01):
    n = len(h)
    ov = compute_overlaps(h)
    mx = ov.max()
    w = np.exp((ov - mx) / temp)
    w /= w.sum()
    ctr = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        m = k - ctr
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            a = -m
            g[:n - a] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:n - a]
    return g, mx

def grad_optimize(h0, time_limit, lr0=0.05, mom=0.9, temp0=0.02):
    n = len(h0)
    tgt = n / 2.0
    h = project(h0.astype(float), tgt)
    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    temp = temp0
    t0 = time.time()
    no_improve = 0
    
    while time.time() - t0 < time_limit:
        g, c5 = get_gradient(h, temp)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        v = mom * v - lr * g
        h_new = project(h + v, tgt)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            lr = min(lr * 1.01, lr0 * 3.0)
        else:
            v *= 0.5
            lr *= 0.5
            if lr < 1e-15:
                lr = lr0 * 0.1
                v = np.zeros(n)
        
        temp = max(temp * 0.9995, 0.001)
    
    return best, best_c5

def anneal(h0, time_limit, T0=0.005, Tend=0.00005):
    n = len(h0)
    h = h0.astype(float).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    while time.time() - t0 < time_limit:
        frac = min(1.0, (time.time() - t0) / time_limit)
        T = max(T0 * (Tend / T0) ** frac, 1e-10)
        
        i, j = tuple(np.random.choice(n, 2, replace=False))
        if abs(h[i] - h[j]) < 1e-12:
            continue
        
        hi_s, hj_s = h[i], h[j]
        h[i], h[j] = hj_s, hi_s
        c5_new = compute_c5(h)
        
        if c5_new <= best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        else:
            h[i], h[j] = hi_s, hj_s
    
    return best, best_c5

def upsample(h, new_n):
    n = len(h)
    ox = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    nx = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(nx, ox, h)
    return project(h_new, new_n / 2.0)

def make_random_start(n, base=None):
    if base is not None and len(base) == n:
        h = base + 0.2 * np.random.randn(n)
    else:
        h = np.random.rand(n)
    return project(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(45, 0.5)

    n = len(h)
    h = project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def rem():
        return budget_s - (time.time() - t_start)

    # Phase 1: Optimize at native resolution
    if rem() > 200:
        h1, c1 = grad_optimize(best_h, min(250, rem() * 0.2))
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    if rem() > 100:
        h2, c2 = anneal(best_h, min(120, rem() * 0.15))
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 2: Multi-resolution
    for nn in [90, 180, 360]:
        if rem() < 150:
            break

        h_up = upsample(best_h, nn)
        c5_up = compute_c5(h_up)

        if rem() > 200:
            h3, c3 = grad_optimize(h_up, min(200, rem() * 0.25))
            if c3 < best_c5:
                best_h, best_c5 = h3, c3

        if rem() > 150:
            h_p = make_random_start(nn, h_up)
            h4, c4 = grad_optimize(h_p, min(120, rem() * 0.15))
            if c4 < best_c5:
                best_h, best_c5 = h4, c4

        if rem() > 100:
            h_p2 = make_random_start(nn, None)
            h5, c5v = grad_optimize(h_p2, min(100, rem() * 0.12))
            if c5v < best_c5:
                best_h, best_c5 = h5, c5v

        if rem() > 80:
            h_curr = best_h if len(best_h) == nn else h_up
            h6, c6 = anneal(h_curr, min(80, rem() * 0.12))
            if c6 < best_c5:
                best_h, best_c5 = h6, c4 if False else c6

        if rem() > 100:
            best_at_nn = best_h if len(best_h) == nn else h_up
            h7, c7 = grad_optimize(best_at_nn, min(100, rem() * 0.15))
            if c7 < best_c5:
                best_h, best_c5 = h7, c7

    # Phase 3: Final polish
    if rem() > 60:
        h8, c8 = grad_optimize(best_h, min(150, rem() * 0.6))
        if c8 < best_c5:
            best_h, best_c5 = h8, c8

    if rem() > 40:
        h9, c9 = anneal(best_h, min(50, rem() * 0.7), T0=0.002, Tend=0.0001)
        if c9 < best_c5:
            best_h, best_c5 = h9, c9

    return best_h, best_c5, len(best_h)
```
