from scipy.optimize import minimize
import time

def project_onto_constraint(h, S):
    """Project h onto {x: 0<=x_i<=1, sum(x)=S} using bisection."""
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smooth_obj(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    cmax = np.max(c)
    bc = beta * (c - cmax)
    m = np.max(bc)
    return (cmax + m / beta + np.log(np.sum(np.exp(bc - m))) / beta) * dx

def smooth_grad(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    cmax = np.max(c)
    bc = beta * (c - cmax)
    m = np.max(bc)
    e = np.exp(bc - m)
    p = e / np.sum(e)
    
    g = np.zeros(n)

        if p[j] < 1e-15:
            continue
        k = j - (n - 1)

        lo1 = max(0, k)
        hi1 = min(n - 1, n + k - 1)
        if lo1 <= hi1:

        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:

    return g * dx

Wait, I just realized there might be an issue with my gradient computation. Let me double-check it with a finite difference.

For h of length n, c = np.correlate(h, 1-h, mode="full").

The definition of np.correlate(a, v) in Python:
c_k = sum_n a_{n+k} * v_n

where the sum is over all n such that both indices are valid.

So c_k = sum_n h_{n+k} * (1-h)_n = sum_n h_{n+k} * (1-h_n)

For the full mode, k ranges from -(len(v)-1) to (len(a)-1), i.e., from -(n-1) to (n-1).
The output array has index j = k + (n-1), so j ranges from 0 to 2n-2.

Now, ∂c_j / ∂h_m where j corresponds to k = j - (n-1):

c_j = sum_n h_{n+k} * (1-h_n) for n where 0 ≤ n < n and 0 ≤ n+k < n

= sum_{n=max(0,-k)}^{min(n-1, n-1-k)} h_{n+k} * (1-h_n)

∂c_j / ∂h_m:
For each term h_{n+k} * (1-h_n):
- ∂/∂h_m of h_{n+k} is δ_{m, n+k}, giving (1-h_n) * δ_{m,n+k} = (1-h_{m-k}) * [m-k is valid]
- ∂/∂h_m of (1-h_n) is -δ_{m,n}, giving -h_{n+k} * δ_{m,n} = -h_{m+k} * [m+k is valid]

So: ∂c_j / ∂h_m = (1-h_{m-k}) * [0 ≤ m-k < n] - h_{m+k} * [0 ≤ m+k < n]

This is exactly what I have in my code. Good.

Now, the smooth max:
f(h) = (1/β) * log(∑_j exp(β * c_j))

∂f/∂h_m = (1/β) * (1/Z) * ∑_j exp(β*c_j) * β * ∂c_j/∂h_m
= ∑_j p_j * ∂c_j/∂h_m

where p_j = exp(β*c_j) / Z.

With the numerical stabilization (subtracting c_max):
p_j = exp(β*(c_j - c_max) - m) / ∑_l exp(β*(c_l - c_max) - m)

where m = max_j(β*(c_j - c_max)).

This is what I have in my code. Good.

                break
            h_try, c5_try = do_pg(h_try, n, beta, max_iter=400)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 4: Coordinate descent
    if time_ok(0.98):
        h_try, c5_try = coordinate_descent(best_h, n, n_rounds=2)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, target)
    final_c5 = c5_value(best_h)
    return (best_h, final_c5, n)
