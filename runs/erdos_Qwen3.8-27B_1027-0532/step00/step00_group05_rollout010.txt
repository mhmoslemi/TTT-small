import numpy as np
import time

def project_to_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def get_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_grad(h, s, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    sh = c - m
    w = np.exp(s * sh)
    sw = w.sum()
    w /= sw
    val = m + np.log(sw) / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        kv = k - n + 1
        la, ha = max(0, kv), min(n, n + kv)
        if ha > la:
            ix = np.arange(la, ha)
            grad[ix] += wk * dx * (1.0 - h[ix - kv])
        lb, hb = max(0, -kv), min(n, n - kv)
        if hb > lb:
            ix = np.arange(lb, hb)
            grad[ix] += wk * dx * (-h[ix + kv])
    return val, grad

def gradient_optimize(h, target, s_values, max_iters, dx, time_fn):

def binary_hill_climb(h_bin, target, max_rounds, time_fn):

def fix_sum_binary(h, target):
