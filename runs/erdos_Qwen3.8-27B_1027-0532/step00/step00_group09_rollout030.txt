    return np.correlate(h, 1.0 - h, mode='full')

def soft_max(cors, eps=0.01):
    shifted = cors / eps
    shifted = np.clip(shifted, -100, 100)  # avoid overflow
    return eps * (np.log(np.sum(np.exp(shifted))) - np.log(len(cors)))

def soft_max_grad(cors, eps=0.01):
    shifted = cors / eps
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    return weights / eps

def corr_grad(h, n_cors, shift):
    """Gradient of correlation at given shift"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i - shift < n:
            g += (1.0 - h[i - shift])
        if 0 <= i + shift < n:
            g -= h[i + shift]
        grad[i] = g
    return grad

    val = soft_max(cors, eps)
    
    n = len(h)
    weights = soft_max_grad(cors, eps)

import numpy as np
from scipy.optimize import minimize
import time

def compute_cors(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_at_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    i_start = max(0, shift)
    i_end = min(n, shift + n)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        grad[idx] += (1.0 - h[idx - shift])
    i_start2 = max(0, -shift)
    i_end2 = min(n, n - shift)
    if i_start2 < i_end2:
        idx2 = np.arange(i_start2, i_end2)
        grad[idx2] -= h[idx2 + shift]
    return grad

def obj_max_cors(h):
    return np.max(compute_cors(h))

def grad_max_cors(h):
    cors = compute_cors(h)
    n = len(h)
    max_val = np.max(cors)
    tol = 1e-6
    active = np.where(cors >= max_val - tol)[0]
    grad = np.zeros(n)
    for j in active:
        shift = j - (n - 1)
        grad += corr_grad_at_shift(h, shift)
    grad /= len(active)
    return grad

def soft_max_cors(h, eps):
    cors = compute_cors(h)
    scaled = cors / eps
    m = np.max(scaled)
    vals = np.exp(scaled - m)
    s = np.sum(vals)
    return eps * (m + np.log(s) - np.log(len(cors)))

def grad_soft_max_cors(h, eps):
    cors = compute_cors(h)
    n = len(h)
    total = 2 * n - 1
    scaled = cors / eps
    m = np.max(scaled)
    vals = np.exp(scaled - m)
    weights = vals / np.sum(vals)
    grad = np.zeros(n)
    for j in range(total):
        shift = j - (n - 1)
        grad += weights[j] * corr_grad_at_shift(h, shift)
    return grad

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(h):
    return np.ones(len(h))

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            avail = np.sum(1.0 - h[mask])
            if avail < 1e-15:
                break
            frac = min(1.0, diff / avail)
            h[mask] += frac * (1.0 - h[mask])
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            avail = np.sum(h[mask])
            if avail < 1e-15:
                break
            frac = min(1.0, (-diff) / avail)
            h[mask] -= frac * h[mask]
    return h

def evaluate_c5(h, n):
    dx = 2.0 / n
    return np.max(compute_cors(h)) * dx

def run_slsqp_from(h_start, bounds, constraints, maxiter, ftol):
    result = minimize(
        obj_max_cors, grad_max_cors, h_start, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0}
    )
    return result.x

def run_slsqp_soft(h_start, bounds, constraints, eps, maxiter):
    def obj_fn(h):
        return soft_max_cors(h, eps)
    def grad_fn(h):
        return grad_soft_max_cors(h, eps)

        bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
    )
    return result.x
