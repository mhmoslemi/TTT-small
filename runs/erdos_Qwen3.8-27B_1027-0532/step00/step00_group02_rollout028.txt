    def obj(x):
        return smooth_max_obj(x, dx, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        return project_sum(res.x, target_sum)
    return h

def local_search_pair(h, dx, target_sum, budget):
    start = time.time()
    n = len(h)
    best_c5 = compute_c5(h, dx)
    big_deltas = [0.5, 0.3, 0.2, 0.1, 0.05]
    small_deltas = [0.02, 0.01, 0.005]
    
    while time.time() - start < budget:
        improved = False
        for i in range(n):
            if time.time() - start > budget:
                break
            for j in range(i + 1, n):
                if time.time() - start > budget:
                    break
                for delta in big_deltas:
                    if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                        h[i] += delta
                        h[j] -= delta
                        c5 = compute_c5(h, dx)
                        if c5 < best_c5 - 1e-15:
                            best_c5 = c5
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c5 = compute_c5(h, dx)
                        if c5 < best_c5 - 1e-15:
                            best_c5 = c5
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    if improved:
                        break
                if not improved:
                    for delta in small_deltas:
                        if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                            h[i] += delta
                            h[j] -= delta
                            c5 = compute_c5(h, dx)
                            if c5 < best_c5 - 1e-15:
                                best_c5 = c5
                                improved = True
                            else:
                                h[i] -= delta
                                h[j] += delta
                        if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                            h[i] -= delta
                            h[j] += delta
                            c5 = compute_c5(h, dx)
                            if c5 < best_c5 - 1e-15:
                                best_c5 = c5
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                        if improved:
                            break
                if improved:
                    break
        if not improved:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    n = 47
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = project_sum(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    alphas = [10, 30, 80, 200, 500, 1000, 2000, 4000]
    for alpha in alphas:
        if time_left() < 150:
            break
        h_new = do_slsqp(h, n, dx, target_sum, alpha, maxiter=500)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = best_h.copy()
    
    if time_left() > 80:
        h_ls = best_h.copy()
        _, c5_ls = local_search_pair(h_ls, dx, target_sum, time_left() - 80)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    if time_left() > 60:
        h = best_h.copy()
        for alpha in [1000, 3000, 6000]:
            if time_left() < 30:
                break
            h_new = do_slsqp(h, n, dx, target_sum, alpha, maxiter=300)
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h = h_new.copy()
    
    if time_left() > 100:
        for restart in range(5):
            if time_left() < 60:
                break
            h_r = best_h.copy()
            noise = rng.uniform(-0.1, 0.1, n)
            h_r += noise
            h_r = project_sum(h_r, target_sum)
            h_r = do_slsqp(h_r, n, dx, target_sum, 200, maxiter=300)
            c5 = compute_c5(h_r, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
    
    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    
    return best_h, best_c5, n
