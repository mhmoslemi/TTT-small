    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:

    lam = (lo + hi) / 2

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def smooth_grad(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = T * c
    logsum = logsumexp(scaled)
    w = np.exp(scaled - logsum)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n - k])
            grad[:n - k] -= wm * h[k:]
        else:
            grad[:n + k] += wm * (1.0 - h[-k:])
            grad[-k:] -= wm * h[:n + k]
    return grad
