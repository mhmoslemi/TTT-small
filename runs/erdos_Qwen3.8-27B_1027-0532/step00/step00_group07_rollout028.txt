import numpy as np
from scipy.optimize import minimize

def compute_r(h):
    """Compute autocorrelation r(j) = sum_i h[i]*h[i+j] for j=0,...,n-1"""
    n = len(h)
    r = np.zeros(n)
    for j in range(n):
        r[j] = np.dot(h[:n-j], h[j:])
    return r

def smooth_min_r(h, beta):
    """Smooth approximation to -min_j r(j) using log-sum-exp"""
    r = compute_r(h)
    # -(1/beta) * log(sum exp(-beta * r))
    # Use log-sum-exp trick for numerical stability
    min_r = np.min(r)
    log_sum = min_r + np.log(np.sum(np.exp(-beta * (r - min_r))))

def project_feasible(h, target_sum):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Now adjust to meet the sum constraint
    # Find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:

    lo, hi = -1.0, 2.0
    for _ in range(100):

        s = np.sum(adjusted)
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2

    h = project_feasible(h, target_sum)

    best_c5 = c5_from_r(r, n)
    best_h = h.copy()

    def optimize(h_local, beta, lr, max_iters, momentum=0.85):
        nonlocal best_c5, best_h
        v = np.zeros(n)
        for it in range(max_iters):
            if time.time() - start_time > budget_s - 15:
                break
            r_loc = compute_r(h_local)
            min_r_val = float(np.min(r_loc))
            shifted = r_loc - min_r_val
            exp_vals = np.exp(-beta * shifted)
            w = exp_vals / float(np.sum(exp_vals))

            grad = compute_gradient(h_local, w)
            v = momentum * v + lr * grad
            h_trial = h_local - v
            h_trial = project_feasible(h_trial, target_sum)

            r_trial = compute_r(h_trial)
            c5_trial = c5_from_r(r_trial, n)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            h_local = h_trial
        return h_local

    phases = [
        (5.0, 0.2, 2000, 0.8),
        (10.0, 0.1, 3000, 0.85),
        (20.0, 0.05, 4000, 0.85),
        (50.0, 0.02, 5000, 0.9),
        (100.0, 0.01, 5000, 0.9),
        (300.0, 0.005, 5000, 0.9),
        (1000.0, 0.002, 5000, 0.9),
    ]

    for beta, lr, max_iters, mom in phases:
        if time.time() - start_time > budget_s - 30:
            break
        h = optimize(h, beta, lr, max_iters, mom)

    # Phase: fine-tune from best
    if time.time() - start_time < budget_s - 20:
        h = best_h.copy()
        h = optimize(h, 200.0, 0.003, 5000, 0.9)

    # Random restarts
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_restarts = min(int(remaining / 25), 15)
        scales = [0.03, 0.05, 0.08, 0.1, 0.15, 0.05, 0.08, 0.12, 0.1, 0.06, 0.2, 0.04, 0.15, 0.2, 0.1]
        for ri in range(n_restarts):
            if time.time() - start_time > budget_s - 10:
                break
            scale = scales[ri % len(scales)]
            h = best_h + scale * np.random.randn(n)
            h = project_feasible(h, target_sum)
            max_it = min(3000, int(max(100, (budget_s - (time.time() - start_time) - 5) * 500)))
            h = optimize(h, 100.0, 0.008, max_it, 0.88)

    # Final verification
    final_c5 = float(evaluate_erdos_solution(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Ensure feasibility one more time
    best_h = project_feasible(best_h, target_sum)
    final_c5 = float(evaluate_erdos_solution(best_h))

    return (best_h, final_c5, n)
