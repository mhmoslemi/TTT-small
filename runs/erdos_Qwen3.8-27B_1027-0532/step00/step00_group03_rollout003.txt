from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full") * dx
    return float(np.max(corr))

def project_simplex(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}."""
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = initial_h_values.copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx0)
    best_n = n0
    
    def elapsed():
        return time.time() - start_time
    
    def remaining():
        return budget_s - elapsed()
    
    # Optimization function for a given resolution
    def opt_at_res(h_init, n):
        dx = 2.0 / n
        target_sum = n / 2.0
        best_local = h_init.copy()
        best_local_c5 = compute_c5(h_init, dx)
        
        # Phase A: Coarse optimization with L-BFGS-B + penalty
        for beta in [10, 50, 200]:
            for mu in [100, 1000, 10000]:
                if remaining() < 30:
                    return best_local, best_local_c5
                
                def obj(x):
                    corr = np.correlate(x, 1-x, mode="full") * dx
                    m = corr.max()
                    smooth = np.log(np.sum(np.exp(corr - m))) / beta
                    return m + smooth + mu * (x.sum() - target_sum)**2
                
                res = minimize(obj, best_local, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)]*n,
                              options={'maxiter': 300, 'ftol': 1e-14})
                
                h_new = np.clip(res.x, 0, 1)
                h_new = project_simplex(h_new, target_sum)
                c5 = compute_c5(h_new, dx)
                if c5 < best_local_c5:
                    best_local_c5 = c5
                    best_local = h_new.copy()
        
        # Phase B: Fine optimization with SLSQP (handles constraint exactly)
        if remaining() > 60:
            for beta in [100, 500, 2000]:
                if remaining() < 30:
                    break

                    corr = np.correlate(x, 1-x, mode="full") * dx
                    m = corr.max()
                    return m + np.log(np.sum(np.exp(corr - m))) / beta
                
                def con(x):
                    return x.sum() - target_sum
                
                res = minimize(obj2, best_local, method='SLSQP',
                              bounds=[(0.0, 1.0)]*n,
                              constraints=[{'type': 'eq', 'fun': con}],
                              options={'maxiter': 500, 'ftol': 1e-14})
                
                if res.fun < 10:  # sanity check

                    h_new = project_simplex(h_new, target_sum)
                    c5 = compute_c5(h_new, dx)
                    if c5 < best_local_c5:

                        best_local = h_new.copy()

    best_n = n0

            best_h = h_opt

        current_n = n_new

    best_c5 = compute_c5(best_h, 2.0 / best_n)
    
    return best_h, best_c5, best_n

def gradient_of_smoothed_max(h, dx, beta):
    """Compute gradient of the smoothed max C5 objective."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = corr.max()
    e = np.exp(corr - m)
    e_sum = e.sum()
    w = e / e_sum
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wk = w[lag + n - 1] / beta
        if wk < 1e-14:
            continue
        if lag == 0:
            grad += wk * dx * (1 - 2*h)
        elif lag > 0:
            # (1-h[j-lag]) for j in [lag, n-1]
            grad[lag:] += wk * dx * (1 - h[:n-lag])
            # -h[j+lag] for j in [0, n-1-lag]
            grad[:n-lag] -= wk * dx * h[lag:]
        else:
            L = -lag
            # (1-h[j+L]) for j in [0, n-1-L]
            grad[:n-L] += wk * dx * (1 - h[L:])
            # -h[j-L] for j in [L, n-1]
            grad[L:] -= wk * dx * h[:n-L]
    
    return grad
