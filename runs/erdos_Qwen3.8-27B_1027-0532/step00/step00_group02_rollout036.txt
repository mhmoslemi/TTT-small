from scipy.optimize import minimize
from scipy.signal import correlate
import time

_target_sum = 23.5
_n_pts = 47

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1 - h, mode="full") * dx
    return np.max(c)

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-12:
                h = h + diff * room / total
        else:
            room = h.copy()
            total = room.sum()
            if total > 1e-12:
                h = h + diff * room / total
    return np.clip(h, 0.0, 1.0)

def _cons_fun(h):
    return np.sum(h) - _target_sum

def _cons_jac(h):
    return np.ones(len(h))
