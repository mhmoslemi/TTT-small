def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room < 1e-14:
                break
            h = np.minimum(1.0, h + diff * room / total_room)
        else:
            total_excess = np.sum(h)
            if total_excess < 1e-14:

            h = np.maximum(0.0, h + diff * h / total_excess)
