import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    full = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(full)

def project_onto_constraint(h, target_sum):
    """Project h onto {x in [0,1]^n, sum(x) = target_sum}"""
    n = len(h)
    # Bisection for t
    lo = h.min() - target_sum / n - 1
    hi = h.max() + target_sum / n + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0, 1)

def gradient_at_lag(h, k_star, n):
    """Compute gradient of g_k at h for the given lag k_star."""
    m = k_star - (n - 1)
    grad = np.zeros(n)
    idx1 = np.arange(n) + m
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += 1.0 - h[idx1[valid1]]
    idx2 = np.arange(n) - m
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return grad

def projected_gradient_optimize(h0, n, lr=0.01, n_iter=5000, target_sum=None):
    if target_sum is None:
        target_sum = n / 2.0
    h = h0.copy()

            best_h = h.copy()
        
        grad = gradient_at_lag(h, k_star, n)

- Variables: h[0], ..., h[n-1]
- Objective: smoothed_max = (1/T) * log(sum_k exp(T * g_k * dx)) where g_k = correlate(h, 1-h, "full")[k]
- Constraints: h[i] ∈ [0,1] (box), sum(h) = n/2 (equality)

The gradient of the smoothed max:
∂/∂h[j] of smoothed_max = sum_k w_k * ∂g_k/∂h[j] * dx
where w_k = exp(T * g_k * dx) / sum_l exp(T * g_l * dx)

This is a weighted average of the individual gradients.

    lo = h.min() - target_sum

def smooth_max_obj(h, n, T=10.0):
    """Smoothed max of correlations, scaled by dx."""
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    # Numerically stable log-sum-exp
    m = np.max(full)
    if T * m > 500:
        return m  # Just return the max if T is too large
    return m + np.log(np.sum(np.exp(T * (full - m)))) / T

def smooth_max_grad(h, n, T=10.0):
    """Gradient of smoothed max."""
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(full)
    scaled = T * (full - m)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(full)):
        if weights[k] < 1e-10:
            continue
        # m = k - (n-1)
        mm = k - (n - 1)
        idx1 = np.arange(n) + mm
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += weights[k] * (1.0 - h[idx1[valid1]])
        idx2 = np.arange(n) - mm
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] -= weights[k] * h[idx2[valid2]]
    grad *= dx
    return grad

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h, n):
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(full)

def project_onto_constraint(h, target_sum):
    n = len(h)
    lo = h.min() - target_sum
    hi = h.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def gradient_at_lag(h, k_star, n):
    m = k_star - (n - 1)
    grad = np.zeros(n)
    idx1 = np.arange(n) + m
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += 1.0 - h[idx1[valid1]]
    idx2 = np.arange(n) - m
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return grad

def thue_morse(n):
    """Generate first n values of Thue-Morse sequence (0s and 1s)."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def make_binary_from_pattern(pattern, n, target_sum):
    """Create binary-like h from a 0-1 pattern, adjusting to hit target_sum."""
    h = pattern.astype(float).copy()
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if diff > 0:
        # Need to change some 0s to 1s
        n_change = int(np.ceil(diff))
        zeros = np.where(h == 0)[0]
        for i in range(n_change):
            h[zeros[i]] = 1.0
        # Adjust fractional
        frac = target_sum - np.sum(h)
        if frac > 0:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[-1]] = 1.0 + frac  # This might exceed 1
                # Better: set a 0 to frac
                h[ones[-1]] = 1.0
                # Find a 0 and set to frac
                zeros2 = np.where(h == 0)[0]
                if len(zeros2) > 0:
                    h[zeros2[0]] = frac
    elif diff < 0:
        n_change = int(np.ceil(-diff))
        ones = np.where(h == 1)[0]
        for i in range(n_change):
            h[ones[i]] = 0.0
        frac = target_sum - np.sum(h)
        if frac < 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[-1]] = -frac  # This might exceed 1 if diff > 1
    # Final projection to ensure constraint is met
    return project_onto_constraint(h, target_sum)

def slqp_optimize(h0, n, time_limit=60):
    """SLSQP with smoothed max objective."""
    start = time.time()
    target_sum = n / 2.0
    
    def obj_and_grad(h, T):
        dx = 2.0 / n
        full = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(full)
        scaled = T * (full - m)
        scaled = np.clip(scaled, -400, 400)
        exp_scaled = np.exp(scaled)
        sum_exp = np.sum(exp_scaled)
        obj = m + np.log(sum_exp) / T
        
        weights = exp_scaled / sum_exp
        grad = np.zeros(n)
        for k in range(len(full)):
            w = weights[k]
            if w < 1e-12:
                continue
            mm = k - (n - 1)
            idx1 = np.arange(n) + mm
            valid1 = (idx1 >= 0) & (idx1 < n)
            grad[valid1] += w * (1.0 - h[idx1[valid1]])
            idx2 = np.arange(n) - mm
            valid2 = (idx2 >= 0) & (idx2 < n)
            grad[valid2] -= w * h[idx2[valid2]]
        grad *= dx
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum,
        'jac': lambda h: np.ones(n)
    }]
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, n)
    
    for T in [30.0, 15.0, 10.0, 7.5, 5.0, 3.0]:
        if time.time() - start > time_limit:
            break
        
        def fun(h):
            return obj_and_grad(h, T)[0]
        
        def jac(h):
            return obj_and_grad(h, T)[1]
        
        try:
            result = minimize(fun, best_h, jac=jac, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_onto_constraint(result.x, target_sum)
            c5_new = evaluate_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
