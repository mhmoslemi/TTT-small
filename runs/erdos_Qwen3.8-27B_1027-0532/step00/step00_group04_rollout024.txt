- Increase h[i] where h[i+k] is large (increase the "1-h[i]" factor... wait, we want to DECREASE the product, so we want to INCREASE h[i] to make (1-h[i]) smaller)

So the strategy is:

- But we need to preserve the sum constraint, so if we increase h[i], we should decrease some other h[j]

This suggests a "targeted pair swap": find (i, j) such that swapping mass from j to i (i.e., h[i] += d, h[j] -= d) reduces the max correlation.

The most promising candidates are:
- i is a position where h[i] is small but h[i+k] is large (so increasing h[i] reduces the product)
- j is a position where h[j] is large but h[j+k] is small (so decreasing h[j] doesn't hurt)

def targeted_move(h, n, dx, target_sum, step_size):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    
    # Compute the "pressure" at each position
    # pressure[i] = how much reducing corr[k_max] would benefit from changing h[i]
    pressure = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            # Term: h[i+lag] * (1-h[i])
            # Increasing h[i] reduces this term by h[i+lag]
            # Decreasing h[i+lag] reduces this term by (1-h[i])
            pressure[i] += h[i + lag]  # benefit of increasing h[i]
            pressure[i + lag] -= (1 - h[i])  # benefit of decreasing h[i+lag]
    # Note: negative pressure means we should decrease h at that position
    # Positive pressure means we should increase h at that position
    
    # Best move: increase h where pressure is most positive, decrease where most negative
    i = int(np.argmax(pressure))
    j = int(np.argmin(pressure))
    
    if i == j:
        return None
    if pressure[i] <= 0 or -pressure[j] <= 0:
        return None  # No beneficial direction
    
    d = min(step_size, (1 - h[i]) if pressure[i] > 0 else h[i], 
            h[j] if pressure[j] < 0 else (1 - h[j]))
    if d <= 0:
        return None
    
    h_trial = h.copy()
    h_trial[i] += d
    h_trial[j] -= d
    return repair(h_trial, target_sum)

import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) >= 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        new_val = h[idx] + diff
        if 0.0 <= new_val <= 1.0:
            h[idx] = new_val
    return h

def obj_smooth(h, args):
    beta, dx = args
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(corr))
    scaled = beta * (corr - cmax)
    scaled = np.clip(scaled, -500.0, 500.0)
    return dx * (cmax + float(np.log(np.sum(np.exp(scaled)))) / beta)
