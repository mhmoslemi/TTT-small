    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target_sum, n):

    h_c = np.clip(x, 1e-8, 1.0 - 1e-8)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target_sum, n):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

def _smooth_obj(x, dx, beta):
    h_c = np.clip(x, 1e-10, 1.0 - 1e-10)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    ms = np.max(shifted)
    e = np.exp(shifted - ms)
    se = np.sum(e)
    return dx * (m + ms / beta + np.log(se) / beta)

def _smooth_grad(x, dx, beta):
    h_c = np.clip(x, 1e-10, 1.0 - 1e-10)

    ms = np.max(shifted)
    e = np.exp(shifted - ms)
    se = np.sum(e)
    W = e / se
    g = 1.0 - 2.0 * h_c
