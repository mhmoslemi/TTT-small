def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5

        diff = target - np.sum(h)

            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corr(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def constraint_sum(h, n):
    return float(np.sum(h) - n * 0.5)

def objective_smooth(h, dx, beta):
    return smooth_max_corr(h, dx, beta)

def objective_penalty(h, dx, n, penalty):
    base = c5_value(h, dx)
    sum_viol = abs(np.sum(h) - n * 0.5)
    bound_viol = np.sum(np.maximum(0.0, h - 1.0)) + np.sum(np.maximum(0.0, -h))
    return base + penalty * (sum_viol + bound_viol)

def run_slsqp(h0, n, dx, beta, maxiter):

    return np.interp(x_dst, x_src, h_src)
