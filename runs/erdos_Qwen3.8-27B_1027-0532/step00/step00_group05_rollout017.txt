```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full") * dx

def local_search(h, max_iter=1000):
    n = len(h)
    dx = 2.0 / n
    current = h.copy().astype(float)
    current_c5 = evaluate_c5(current)
    
    ones = np.where(current > 0.5)[0]
    zeros = np.where(current <= 0.5)[0]
    
    for iteration in range(max_iter):
        best_c5 = current_c5
        best_swap = None
        
        for i in ones:
            for j in zeros:
                # Try swapping i and j
                old_i, old_j = current[i], current[j]
                current[i] = old_j
                current[j] = old_i
                new_c5 = evaluate_c5(current)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_swap = (i, j)
                current[i] = old_i
                current[j] = old_j
        
        if best_swap is None:
            break
        i, j = best_swap
        current[i], current[j] = current[j], current[i]
        current_c5 = best_c5
    
    return current, current_c5
```

```python
# h in [0,1]
constraints += [h >= 0, h <= 1]
# sum h = n/2
constraints += [cp.sum(h) == n/2]
# y constraints (McCormick)

    for k in range(n):
        if i != k:  # skip diagonal (h[i]*h[i] = h[i] for binary)
            constraints += [y[i,k] <= h[i], y[i,k] <= h[k], y[i,k] >= h[i] + h[k] - 1, y[i,k] >= 0]

    S = cp.sum(h[lag:])
    # A_lag = sum_{i=0}^{n-1-lag} h[i+lag]*h[i] = sum_{i=0}^{n-1-lag} y[i+lag, i]
    A = sum(y[lag+i, i] for i in range(n-lag))

    A2 = sum(y[j, j-lag] for j in range(lag, n))
    constraints += [S2 - A2 <= C/dx]
```

```python
    """Convert to binary with exactly n/2 ones."""
    b = (h > 0.5).astype(float)
```

Actually, for the best-improvement local search, we don't need to try ALL pairs. We can be smarter:
- The objective changes only for lags that involve the swapped positions
- A swap at positions i and j affects c[k] for k such that the correlation sum includes position i or j

For position i, it's included in c[k] when i is in the range of the sum. For the full correlation of h and 1-h:
c[k] = sum_{valid i'} h[i' + k - (n-1)] * (1 - h[i'])

Position i appears in h[i' + k - (n-1)] when i' + k - (n-1) = i, i.e., i' = i - k + (n-1). This is valid when 0 ≤ i - k + (n-1) < n, i.e., k - (n-1) ≤ i < k - (n-1) + n, i.e., k ≤ i + (n-1) and k > i + (n-1) - n = i - 1. So k = i, i+1, ..., i+(n-1).

Position i appears in (1-h[i']) when i' = i, which requires 0 ≤ i < n (always true) and 0 ≤ i + k - (n-1) < n, i.e., (n-1) - i ≤ k ≤ (n-1) - i + (n-1) = 2(n-1) - i. So k = (n-1)-i, (n-1)-i+1, ..., 2(n-1)-i.

So a swap at positions i and j affects c[k] for k in the union of:
- {i, i+1, ..., i+n-1} (from h part)
- {(n-1)-i, (n-1)-i+1, ..., 2(n-1)-i} (from 1-h part)
- {j, j+1, ..., j+n-1}
- {(n-1)-j, (n-1)-j+1, ..., 2(n-1)-j}

This is at most 4n values of k out of 2n-1 total. So roughly half the output changes.

This means we can't easily avoid recomputing the full correlation. Let's just use the direct approach.

Actually, I just realized there might be a much faster way. We can precompute the correlation and then update it incrementally when we make a swap.

If we change h[i] from 1 to 0 (and h[j] from 0 to 1), the change in c[k] is:
Δc[k] = (change in h[i] * (1-h[partner])) + (change in h[j] * (1-h[partner']))

where partner and partner' depend on k.

Specifically, for the full correlation:
c[k] = sum_i h[i+k-(n-1)] * (1-h[i])

If h[i] changes by δ (from 1 to 0, so δ = -1):
The term involving h[i] in the "first factor" (h[i+k-(n-1)]) changes when i+k-(n-1) = i_swap, i.e., k = i_swap + (n-1) - i. For this k, the change is δ * (1-h[i]).
The term involving h[i] in the "second factor" (1-h[i]) changes when i = i_swap, i.e., for all k where i_swap is valid as the second index. The change is -δ * h[i_swap + k - (n-1)] = δ * h[...].

```python
import numpy as np
import math
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full") * dx
    return float(np.max(c))

def thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def make_binary(values, n):
    """Create binary array of length n with exactly n//2 ones, close to values."""
    b = (values > 0.5).astype(float)
    n_ones = int(b.sum())
    target = n // 2
    if n_ones == target:
        return b
    # Sort by distance from 0.5
    diff = np.abs(values - 0.5)
    order = np.argsort(diff)
    if n_ones > target:
        to_flip = n_ones - target
        for idx in order:
            if int(b[idx]) == 1 and to_flip > 0:
                b[idx] = 0
                to_flip -= 1
    else:
        to_flip = target - n_ones
        for idx in order:
            if int(b[idx]) == 0 and to_flip > 0:
                b[idx] = 1
                to_flip -= 1
    return b

def local_search(b, max_iter=200):
    """Best-improvement local search."""
    n = len(b)
    b = b.copy().astype(float)
    cur = evaluate_c5(b)
    
    for _ in range(max_iter):
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        best = cur
        best_pair = None
        
        for i in ones:
            for j in zeros:
                b[i], b[j] = 0.0, 1.0
                val = evaluate_c5(b)
                b[i], b[j] = 1.0, 0.0
                if val < best - 1e-15:
                    best = val
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        i, j = best_pair
        b[i], b[j] = 0.0, 1.0
        cur = best

        t = t0 * (t1/t0) ** (it / n_iter)
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        i = rng.integers(len(ones))
```

```python
def project(h, target_sum):
    """Project h to {x : 0≤x≤1, sum(x)=target_sum}."""
    h = np.clip(h, 0, 1)
    for _ in range(1000):
        deficit = target_sum - h.sum()
        if abs(deficit) < 1e-10:
            break
        if deficit > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] = np.clip(h[mask] + deficit / mask.sum(), 0, 1)
    return h
```
