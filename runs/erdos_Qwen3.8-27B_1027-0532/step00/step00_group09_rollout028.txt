from scipy.optimize import minimize
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        n_free = np.sum(free)
        if n_free == 0:
            break
        h[free] = h[free] + diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h, *args):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def obj_lp_func(h, *args):
    p = args[0] if len(args) > 0 else 20
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.sum(np.power(corr, p)) ** (1.0 / p)

def eq_constraint(h, *args):
    target_sum = args[0]
    return np.sum(h) - target_sum

def do_slsqp(h_start, n, target_sum, max_iter, ftol, use_lp=False, lp_power=20):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_constraint, 'args': (target_sum,)}]
    if use_lp:
        result = minimize(obj_lp_func, h_start, method='SLSQP', bounds=bounds,
                          constraints=constraints, args=(lp_power,),
                          options={'maxiter': max_iter, 'ftol': ftol})
    else:
        result = minimize(obj_func, h_start, method='SLSQP', bounds=bounds,
                          constraints=constraints,

    return result.x
