    # Change in h[i] (decreased by delta): -delta * (1 - h[i + offset]) for valid i+offset

    idx_j = j + offset
    valid_j = (idx_j >= 0) & (idx_j < n)
    d[valid_j] += (1.0 - h[idx_j[valid_j]])

    # When k + offset = i (i.e., k = i - offset = i - (m-n+1) = i + n - 1 - m):
    #   The term (1-h[i]) appears, and changing h[i] by -delta changes this to +delta
    #   So contribution: +delta * h[k] where k = i - offset, valid if 0 <= k < n
    # When k + offset = j:
    #   Changing h[j] by +delta changes (1-h[j]) to (1-h[j]-delta), contribution: -delta * h[k]
    
    idx2_i = i - offset  # = i - (m - n + 1) = i + n - 1 - m
    valid2_i = (idx2_i >= 0) & (idx2_i < n)
    d[valid2_i] += h[idx2_i[valid2_i]]
    
    idx2_j = j - offset
    valid2_j = (idx2_j >= 0) & (idx2_j < n)
    d[valid2_j] += -h[idx2_j[valid2_j]]
    
    return d

def pair_step(h, i, j):

    n = len(h)
    dx = 2.0 / n
    n_c = 2 * n - 1

    d = compute_d(h, i, j)
    
    delta_lo = 0.0
    delta_hi = min(h[i], 1.0 - h[j])
    
    if delta_hi <= delta_lo + 1e-15:
        return h, float(np.max(c) * dx)
    
    # Find candidate breakpoints
    # c[m] + delta * d[m], m = 0, ..., n_c-1
    # Breakpoint between m and l: (c[m] - c[l]) / (d[l] - d[m]) when d[l] != d[m]
    
    # Vectorized computation
    c_vals = c  # shape (n_c,)
    d_vals = d  # shape (n_c,)
    
    # Compute all pairwise breakpoints
    # delta_ml = (c[l] - c[m]) / (d[m] - d[l])
    c_diff = c_vals[1:, np.newaxis] - c_vals[:n_c-1, np.newaxis].T  # c[l] - c[m] for m < l
    d_diff = d_vals[:n_c-1, np.newaxis].T - d_vals[1:, np.newaxis]  # d[m] - d[l] for m < l
    
    mask = np.abs(d_diff) > 1e-12
    bp = np.where(mask, c_diff / np.where(mask, d_diff, 1.0), np.inf)
    
    # Filter to valid range
    bp[(bp < delta_lo) | (bp > delta_hi)] = np.inf
    bp = bp.ravel()
    bp = bp[np.isfinite(bp)]
    
    candidates = np.concatenate(([delta_lo, delta_hi], bp))
    
    # Evaluate at all candidates
    scores = np.max(c_vals[None, :] + candidates[:, None] * d_vals[None, :], axis=1) * dx
    
    best_idx = np.argmin(scores)
    best_delta = candidates[best_idx]
    best_score = scores[best_idx]
    
    if best_delta > 1e-15:
        h = h.copy()
        h[i] -= best_delta
        h[j] += best_delta
    
    return h, float(best_score)

    n_c = 2 * n - 1
    d = np.zeros(n_c)

    offsets_cur = make_offsets(len(best_h))
    n_cur = len(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        no_improve = 0
        while time.time() - start_time < budget_s - 20 and no_improve < 6:
            h_pert = best_h + rng.randn(n_cur) * 0.03
            h_pert = repair_h(h_pert, n_cur)
            h_opt = coord_descent(h_pert, offsets_cur, max_sweeps=8, time_limit=budget_s, start_time=start_time)
            s_opt = evaluate_c5(h_opt)
            if s_opt < best_score - 1e-12:
                best_score = s_opt
                best_h = h_opt.copy()
                no_improve = 0
            else:
                no_improve += 1

    # Final refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_final = len(best_h)
        offsets_final = make_offsets(n_final)
        h_final = coord_descent(best_h, offsets_final, max_sweeps=5, time_limit=budget_s, start_time=start_time)
        s_final = evaluate_c5(h_final)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final.copy()

    best_h = repair_h(best_h, len(best_h))
    best_score = evaluate_c5(best_h)

    return best_h, best_score, len(best_h)
