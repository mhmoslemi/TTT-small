import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def interpolate_to_size(parent, n):
    n_parent = len(parent)
    if n == n_parent:
        return parent.copy().astype(float)
    x_parent = np.linspace(0, 2, n_parent)
    x_new = np.linspace(0, 2, n)
    return np.interp(x_new, x_parent, parent)

def subgradient_update(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    j_star = int(np.argmax(c))
    shift = j_star - (n - 1)
    i_arr = np.arange(n)
    grad = np.zeros(n)
    idx1 = i_arr - shift
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = i_arr + shift
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] -= h[idx2[mask2]]
    c5 = float(np.max(c) * dx)
    h_new = project_onto_feasible(h - alpha * grad, n)
    return h_new, c5

def smooth_obj(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c = np.maximum(c, 0.0)
    return float(np.sum(c ** p) ** (1.0 / p))

def smooth_grad(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_clipped = np.maximum(c, 1e-30)
    fp = np.sum(c_clipped ** p) ** (1.0 / p)
    coeff = (1.0 / p) * (np.sum(c_clipped ** p)) ** (1.0 / p - 1.0) * p * dx
    grad = np.zeros(n)
    i_arr = np.arange(n)
    for j in range(len(c)):
        w = c_clipped[j] ** (p - 1)
        if w < 1e-20:
            continue
        shift = j - (n - 1)
        idx1 = i_arr - shift
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += w * (1.0 - h[idx1[mask1]])
        idx2 = i_arr + shift
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] -= w * h[idx2[mask2]]
    grad *= coeff
    return grad

def run_slqp_polish(h0, n, time_limit, p=50):
    elapsed_start = time.time()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                   'jac': lambda x: np.ones(n)}]

    def obj_func(x):
        return smooth_obj(x, n, p)

    def grad_func(x):
        return smooth_grad(x, n, p)

    try:
        res = minimize(obj_func, h0, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.x is not None:
            h_opt = project_onto_feasible(res.x, n)
            return h_opt, compute_c5(h_opt, n)
    except Exception:
        pass
    return h0, compute_c5(h0, n)

def run_subgradient(h0, n, max_time):
    elapsed_start = time.time()
    h = h0.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    n_total = len(np.correlate(h, 1 - h, mode='full'))
    alpha0 = 0.02
    for it in range(100000):
        if time.time() - elapsed_start > max_time:
            break
        alpha = alpha0 / (1.0 + 0.01 * it) ** 0.7
        h, c5 = subgradient_update(h, n, alpha)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
