def compute_subgradient(h, n):

    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 1e-8)[0]
    
    sg = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                sg[:] += (1.0 - h)
                sg[:] -= h
            else:
                sg[lag:] += (1.0 - h[:n - lag])
                sg[:n - lag] -= h[lag:]
        else:
            l = -lag
            if l == 0:
                sg[:] += (1.0 - h)
                sg[:] -= h
            else:
                sg[:n - l] += (1.0 - h[l:])
                sg[l:] -= h[:n - l]
    
    dx = 2.0 / n
    sg *= dx
    return sg, max_corr * dx

        else:

            alpha *= 0.7
