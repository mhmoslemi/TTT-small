import numpy as np
import time

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def c5_of(h_arr, dx_val):
        corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
        return np.max(corr) * dx_val
    
    def coordinate_descent(h, dx_val, n, max_passes=200, time_limit=None):
        target_sum = n / 2.0
        best = h.copy()
        best_c5 = c5_of(best, dx_val)
        for pass_num in range(max_passes):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    s = best[i] + best[j]
                    lo = max(0.0, s - 1.0)
                    hi = min(1.0, s)
                    if hi - lo < 1e-10:
                        continue
                    orig_i, orig_j = best[i], best[j]
                    found = False
                    for frac in (0.0, 0.25, 0.5, 0.75, 1.0):
                        a = lo + frac * (hi - lo)
                        b = s - a
                        if abs(a - orig_i) < 1e-10:
                            continue
                        best[i], best[j] = a, b
                        c5 = c5_of(best, dx_val)
                        if c5 < best_c5 - 1e-12:
                            best_c5 = c5
                            improved = True
                            found = True
                            break
                    if not found:
                        best[i], best[j] = orig_i, orig_j
            if not improved:
                break
        return best, best_c5
    
    def simulated_annealing(h, dx_val, n, n_iter=100000, T_init=0.005, time_limit=None):
        target_sum = n / 2.0
        best = h.copy()
        best_c5 = c5_of(best, dx_val)
        curr = h.copy()
        T = T_init
        cooling = (1e-4 / T_init) ** (1.0 / max(n_iter, 1))
        for it in range(n_iter):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            old_i, old_j = curr[i], curr[j]
            delta = rng.uniform(-0.5, 0.5)
            new_i = np.clip(old_i + delta, 0.0, 1.0)
            new_j = old_i + old_j - new_i
            new_j = np.clip(new_j, 0.0, 1.0)
            curr[i], curr[j] = new_i, new_j
            c5_new = c5_of(curr, dx_val)
            if c5_new < best_c5:
                best = curr.copy()
                best_c5 = c5_new
            elif rng.uniform() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
                pass
            else:
                curr[i], curr[j] = old_i, old_j
            T *= cooling
        return best, best_c5
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(45)
        h0[1::2] = 1.0
        h0[0] = 0.5
    
    n = len(h0)

    # Phase 1: Coordinate descent on initial
    h_cd, c5_cd = coordinate_descent(h0, dx, n, max_passes=300, time_limit=budget_s - 300)
    
    # Phase 2: Simulated annealing
    h_sa, c5_sa = simulated_annealing(h_cd, dx, n, n_iter=200000, T_init=0.004, time_limit=budget_s - 150)
    
    best_h = h_sa if c5_sa < c5_cd else h_cd
    best_c5 = min(c5_sa, c5_cd)

    n_restarts = 10
