            g[idx] += 1.0 - 2.0 * h[h_idx]

import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * corr.max()


def _project(x, S):
    n = len(x)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def _grad_c5(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nk = min(top_k, len(corr))
    active = np.argsort(corr)[-nk:]
    g = np.zeros(n)
    for c in active:
        i_lo = max(0, c - n + 1)
        i_hi = min(c, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            g[idx] += 1.0 - 2.0 * h[c - idx]
    return (dx / nk) * g


def _lbfgs_obj(x, S, lam):
    h = np.clip(x, 0.0, 1.0)
    return _compute_c5(h) + lam * (h.sum() - S) ** 2


def _lbfgs_grad(x, S, lam, top_k):
    h = np.clip(x, 0.0, 1.0)
    g = _grad_c5(h, top_k=top_k)
    g += 2.0 * lam * (h.sum() - S)
    mask = (x > 0.0) & (x < 1.0)
    return g * mask


def _run_lbfgs(h0, S, maxiter=1000, lam=100.0, top_k=1):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_lbfgs_obj, h0, jac=_lbfgs_grad,
                   args=(S, lam, top_k), method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-7, 'maxls': 20})
    return _project(np.clip(res.x, 0.0, 1.0), S)


def _pgd(h0, S, nsteps=20000, top_k=3, alpha=0.03, mu=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_v = _compute_c5(h)
    best_h = h.copy()
    for _ in range(nsteps):
        g = _grad_c5(h, top_k=top_k)
        v = mu * v + (1.0 - mu) * g
        h_new = _project(h - alpha * v, S)
        val = _compute_c5(h_new)
        if val < best_v:
            best_v = val
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.002, 0.15)
        else:
            alpha *= 0.97
    return best_h, best_v
