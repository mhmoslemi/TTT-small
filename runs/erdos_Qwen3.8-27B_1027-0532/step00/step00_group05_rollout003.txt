    overlap = np.correlate(h, 1.0 - h, mode="full")

def project_to_feasible(h, n_points):
    """Project h to [0,1]^n with sum = n/2 using water-filling"""
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = (-diff) / cnt
            h[mask] = np.maximum(0.0, h[mask] - step)

                    s = float(np.sum(new_h))
                    diff = target_sum - s
                    if abs(diff) > 1e-12:
                        if diff > 0:
                            mask = new_h < 1.0
                        else:
                            mask = new_h > 0.0
                        cnt = int(np.sum(mask))
                        if cnt > 0:
                            new_h = new_h.copy()
                            new_h[mask] += diff / cnt
                            new_h = np.clip(new_h, 0.0, 1.0)
                    else:
                        new_h = new_h.copy()
                    new_val = compute_c5(new_h, n_points)
                    if new_val < current_val:
                        h = new_h
                        current_val = new_val
                        improved = True
                        if current_val < best_val:
                            best_val = current_val
                            best_h = h.copy()
                        break

    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except (NameError, AttributeError, TypeError):
        h_init = np.full(42, 0.5)
        n_init = 42

    best_h = h_init.copy()
    best_val = compute_c5(best_h, n_init)

    deadline_phase1 = start_time + budget_s * 0.35
    best_h, best_val = optimize_sa(best_h, n_init, deadline_phase1, rng, best_h, best_val)

    polish_deadline = start_time + budget_s * 0.45
    best_h, best_val = coordinate_polish(best_h, n_init, polish_deadline, best_h, best_val)

    higher_ns = []
    if n_init % 2 == 0:
        for f in [2, 3, 4, 6]:
            n = n_init * f
            if 2 <= n <= 300 and n % 2 == 0:
                higher_ns.append(n)

    if higher_ns:
        remaining = budget_s - (time.time() - start_time) - 10
        if remaining > 30:
            time_per = remaining / len(higher_ns)
            for n_target in higher_ns:
                if time.time() - start_time > budget_s - 15:
                    break
                factor = n_target // n_init
                h_up = np.repeat(best_h, factor)
                h_up = project_to_feasible(h_up, n_target)
                dead = min(time.time() + time_per * 0.8, start_time + budget_s - 20)
                best_h, best_val = optimize_sa(h_up, n_target, dead, rng, best_h, best_val)
                dead2 = min(time.time() + time_per * 0.2, start_time + budget_s - 10)
                best_h, best_val = coordinate_polish(best_h, n_target, dead2, best_h, best_val)

    n_final = len(best_h)
    best_h = project_to_feasible(best_h, n_final)
    final_val = compute_c5(best_h, n_final)

    if final_val > best_val:
        final_val = best_val

    return best_h, final_val, n_final
