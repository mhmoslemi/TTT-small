        init_h = np.array(initial_h_values, dtype=float)
    n_candidates = [64, 80, 96, 112, 128, 160, 192, 256]
    if init_h is not None and len(init_h) % 2 != 0:
        n_candidates.insert(0, len(init_h))
    total_time = budget_s
    time_per_n = total_time / len(n_candidates)
    for n in n_candidates:
        if time.time() - t_start > total_time - 20:
            break
        remaining = total_time - (time.time() - t_start)
        if remaining < 10:
            break
        m = n // 2
        if n % 2 == 0:
            n_starts = 2
            for s in range(n_starts):
                if time.time() - t_start > total_time - 20:
                    break
                if s == 0 and init_h is not None and len(init_h) == n:
                    h0 = (init_h > 0.5).astype(float)
                    if np.sum(h0) != m:
                        idx = np.argsort(h0)[:m]
                        h0 = np.zeros(n)
                        h0[idx] = 1.0
                    t_alloc = min(time_per_n * 0.5, remaining * 0.4)
                    if t_alloc < 2:
                        continue
                    h_best, c5_best = targeted_search(h0, t_alloc, seed + s * 1000)
                    if c5_best < best_c5:
                        best_c5 = c5_best
                        best_h = h_best.copy()
                else:
                    pos = rng.choice(n, m, replace=False)
                    h0 = np.zeros(n)
                    h0[pos] = 1.0
                    t_alloc = min(time_per_n * 0.4, remaining * 0.3)
                    if t_alloc < 2:
                        continue
                    h_best, c5_best = binary_local_search(h0, t_alloc, seed + s * 7 + n)
                    if c5_best < best_c5:
                        best_c5 = c5_best
                        best_h = h_best.copy()
            if time.time() - t_start < total_time - 20:
                if best_h is not None and len(best_h) == n:
                    t_alloc = min(time_per_n * 0.3, remaining * 0.3)
                    if t_alloc > 2:
                        h_best, c5_best = targeted_search(best_h.copy(), t_alloc, seed + n * 31)
                        if c5_best < best_c5:
                            best_c5 = c5_best
                            best_h = h_best.copy()
        else:
            if init_h is not None and len(init_h) == n:
                h0 = init_h.copy()
                t_alloc = min(time_per_n, remaining * 0.5)
                if t_alloc > 2:
                    h_best, c5_best = near_binary_search(h0, t_alloc, seed + n)
                    if c5_best < best_c5:
                        best_c5 = c5_best
                        best_h = h_best.copy()
    if best_h is None:
        n = 64
        h = np.zeros(n)
        h[:32] = 1.0
        best_h = h
        best_c5 = compute_c5(h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
