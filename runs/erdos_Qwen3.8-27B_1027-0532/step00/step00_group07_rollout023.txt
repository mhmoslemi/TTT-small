import numpy as np
import time
from scipy.optimize import minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def smoothed_obj(h, tau, dx):
    corr = compute_corr(h) * dx
    # Log-sum-exp for numerical stability
    m = np.max(corr)
    return tau * (m + np.log(np.sum(np.exp((corr - m) / tau))))

def smoothed_grad(h, tau, dx):
    corr = compute_corr(h) * dx
    m = np.max(corr)
    weights = np.exp((corr - m) / tau)
    weights /= np.sum(weights)
    # Gradient of log-sum-exp is weighted sum of gradients of each c(k)
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(2*n - 1):
        if weights[k] < 1e-15:
            continue
        lag = k - (n-1)
        js = max(0, lag)
        je = min(n, n+lag)
        if js < je:
            grad[js:je] += weights[k] * (1 - h[js-lag:je-lag])
        js2 = max(0, -lag)
        je2 = min(n, n-lag)
        if js2 < je2:
            grad[js2:je2] -= weights[k] * h[js2+lag:je2+lag]
    return grad

def project_constraint(h, target_sum):
    lo = h.min() - target_sum - 1
    hi = h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        if projected.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def repair(h, target_sum):

    s = h.sum()
    if abs(s - target_sum) > 1e-6:

    n = len(h)
    corr = compute_corr(h) * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / tau)
    sum_exp = np.sum(exp_vals)
    obj_val = tau * (m + np.log(sum_exp))
    weights = exp_vals / sum_exp

        w = weights[k]

            continue
