def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def thue_morse(n, target_ones):
    """Generate Thue-Morse based binary sequence of length n with exactly target_ones ones"""
    # Generate first n terms of Thue-Morse
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Count ones
    n_ones = int(np.sum(tm))
    # Adjust to get exactly target_ones
    if n_ones > target_ones:
        # Remove some ones (set to 0)
        ones_idx = np.where(tm == 1)[0]
        np.random.shuffle(ones_idx)
        tm[ones_idx[:n_ones - target_ones]] = 0

        zeros_idx = np.where(tm == 0)[0]
        np.random.shuffle(zeros_idx)
        tm[zeros_idx[:target_ones - n_ones]] = 1

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, s):
    """Project h onto {x: 0<=x_i<=1, sum(x)=s} using bisection"""
    n = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        total = projected.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_refine(h, n_steps=5000, base_lr=0.05, time_limit=60.0):
    """Subgradient method to minimize max correlation"""
    n = len(h)
    d = n - 1
    current = np.array(h, dtype=float)
    current_val = compute_c5(current)
    best = current.copy()
    best_val = current_val
    start = time.time()
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(current, 1.0 - current, mode="full")
        k_star = int(np.argmax(corr))
        shift = k_star - d
        
        # Compute gradient of corr at k_star
        grad = np.zeros(n)
        # First part: (1-h[j+shift]) for valid j+shift
        js = current[shift:] if shift >= 0 else current[:shift]  # h[j+shift] for valid j
        # Actually let me be more careful
        # For j from 0 to n-1, j+shift must be in [0, n-1]
        # If shift >= 0: j from 0 to n-1-shift
        # If shift < 0: j from -shift to n-1
        
        if shift >= 0:
            valid_j = np.arange(0, n - shift)
            grad[valid_j] += (1.0 - current[valid_j + shift])
        else:
            valid_j = np.arange(-shift, n)
            grad[valid_j] += (1.0 - current[valid_j + shift])
        
        # Second part: -h[j-shift] for valid j-shift
        if shift >= 0:
            valid_j = np.arange(shift, n)
            grad[valid_j] += (-current[valid_j - shift])
        else:
            valid_j = np.arange(0, n + shift)  # n + shift = n - |shift|
            grad[valid_j] += (-current[valid_j - shift])
        
        # Step size
        lr = base_lr / (1.0 + 0.01 * step)
        
        current = current - lr * grad
        current = project_polytope(current, n / 2.0)
        
        new_val = compute_c5(current)
        if new_val < best_val:
            best_val = new_val
            best = current.copy()
    
    return best, best_val

    elif n_ones < target:

    current_val = compute_c5(current)

    best_val = current_val
    deadline = time.time() + time_limit
    
    while time.time() < deadline:

        zeros = np.where(current == 0)[0]
        
        best_swap = None
        best_new_val = current_val

                current[i] = 0
                current[j] = 1

                current[i] = 1
                current[j] = 0

        i, j = best_swap
        current[i] = 0
        current[j] = 1
        current_val = best_new_val
        best = current.copy()
        best_val = current_val

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, s):
    n = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        total = projected.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_refine(h, n_steps=20000, base_lr=0.03, time_limit=60.0):
    n = len(h)
    d = n - 1
    current = np.array(h, dtype=float)
    current_val = compute_c5(current)
    best = current.copy()
    best_val = current_val
    start = time.time()
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(current, 1.0 - current, mode="full")
        k_star = int(np.argmax(corr))
        shift = k_star - d
        grad = np.zeros(n)
        if shift >= 0:
            vj = np.arange(0, n - shift)
            grad[vj] += (1.0 - current[vj + shift])
            vj2 = np.arange(shift, n)
            grad[vj2] += (-current[vj2 - shift])
        else:
            vj = np.arange(-shift, n)
            grad[vj] += (1.0 - current[vj + shift])
            vj2 = np.arange(0, n + shift)
            grad[vj2] += (-current[vj2 - shift])
        lr = base_lr / (1.0 + 0.02 * step)
        current = current - lr * grad
        current = project_polytope(current, n / 2.0)
        new_val = compute_c5(current)
        if new_val < best_val:

def steepest_ascent_binary(h, time_limit=30.0):
    n = len(h)
    current = make_binary(h, n)
    current_val = compute_c5(current)
    best = current.copy()
    best_val = current_val
    deadline = time.time() + time_limit

        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
