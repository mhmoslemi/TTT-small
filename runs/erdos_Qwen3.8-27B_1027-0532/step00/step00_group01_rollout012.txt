        if (time.time() - start) > time_limit * 0.5:
            temp *= 0.95
    return best_h, best_c5

def make_structured_init(n, seed, freq=None):
    rng = np.random.default_rng(seed)
    if freq is None:
        freq = rng.integers(2, 6)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(np.pi * freq * x)
    h += 0.1 * rng.normal(0, 1, n)
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    return project_to_feasible(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        pass
    best_h_overall = None
    best_c5_overall = float('inf')
    if h_init is not None:
        n_init = len(h_init)
        alloc1 = budget_s * 0.45
        h_r, c5_r = gradient_descent_phase(h_init, n_init, alloc1, seed)
        if c5_r < best_c5_overall:
            best_c5_overall = c5_r
            best_h_overall = h_r
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            alloc2 = min(remaining * 0.5, 200)
            h_r2, c5_r2 = local_search_phase(h_r, n_init, alloc2, seed + 1)
            if c5_r2 < best_c5_overall:
                best_c5_overall = c5_r2
                best_h_overall = h_r2
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        n_try = 64
        alloc3 = min(remaining * 0.4, 300)
        h_start = make_structured_init(n_try, seed + 100, freq=3)
        h_r3, c5_r3 = gradient_descent_phase(h_start, n_try, alloc3, seed + 100)
        if c5_r3 < best_c5_overall:
            best_c5_overall = c5_r3
            best_h_overall = h_r3
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_try2 = 128
        alloc4 = min(remaining * 0.4, 200)
        h_start2 = make_structured_init(n_try2, seed + 200, freq=5)
        h_r4, c5_r4 = gradient_descent_phase(h_start2, n_try2, alloc4, seed + 200)
        if c5_r4 < best_c5_overall:
            best_c5_overall = c5_r4
            best_h_overall = h_r4
    if best_h_overall is not None:
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            n_best = len(best_h_overall)
            h_fin, c5_fin = local_search_phase(best_h_overall, n_best, remaining, seed + 500)
            if c5_fin < best_c5_overall:
                best_c5_overall = c5_fin
                best_h_overall = h_fin
    if best_h_overall is None:
        h_fallback = np.full(64, 0.5)
        c5_fallback = compute_c5(h_fallback, 2.0 / 64)[0]
        return h_fallback, c5_fallback, 64
    return best_h_overall, float(best_c5_overall), len(best_h_overall)
