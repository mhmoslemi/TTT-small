        else:

        T *= cooling
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    init_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
        init_n = len(init_h)
        t_left = remaining()
        if t_left > 15:
            t_alloc = min(t_left * 0.04, 20.0)
            h_sol, c5_sol = local_search_best(init_h, init_n, time_limit=t_alloc)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                best_n = init_n
            h_sol2, c5_sol2 = local_search_first(init_h, init_n, time_limit=t_alloc, rng=rng)
            if c5_sol2 < best_c5:
                best_c5 = c5_sol2
                best_h = h_sol2.copy()
                best_n = init_n
            hout, cout = simulated_annealing(init_h, init_n, time_limit=t_alloc, rng=rng)
            if cout < best_c5:
                best_c5 = cout
                best_h = hout.copy()
                best_n = init_n
            if best_c5 < 0.3808:
                t_left2 = remaining()
                if t_left2 > 10:
                    h_sol3, c5_sol3 = local_search_best(best_h, best_n, time_limit=min(t_left2 * 0.1, 30.0))
                    if c5_sol3 < best_c5:
                        best_c5 = c5_sol3
                        best_h = h_sol3.copy()

    n_list = [42, 44, 46, 48, 50, 52, 54, 56, 58, 60, 64, 70, 80, 90, 100, 120, 140, 160, 200, 240]

    for n in n_list:
        if n == init_n:
            continue
        t_left = remaining()
        if t_left < 5:
            break
        n_remaining = len(n_list) - n_list.index(n)
        base_time = t_left / max(n_remaining, 1)
        t_alloc = min(base_time, 40.0)
        if n <= 100:
            n_restarts = max(3, int(t_alloc / 8.0))
        elif n <= 160:
            n_restarts = max(2, int(t_alloc / 12.0))
        else:
            n_restarts = max(2, int(t_alloc / 15.0))
        t_per = t_alloc / n_restarts
        strategy = "best" if n <= 120 else "first"

        for r in range(n_restarts):
            if remaining() < 3:
                break
            h = np.zeros(n)
            positions = rng.choice(n, n // 2, replace=False)
            h[positions] = 1.0
            if strategy == "best":
                h_sol, c5_sol = local_search_best(h, n, time_limit=t_per)
            else:
                h_sol, c5_sol = local_search_first(h, n, time_limit=t_per, rng=rng)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                best_n = n
            if c5_sol < 0.38085:
                h2, c5_2 = local_search_first(h_sol, n, time_limit=min(t_per * 0.5, 10.0), rng=rng)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = n

    t_left = remaining()
    if t_left > 10 and best_h is not None:
        n_best = best_n
        extra_restarts = int(min(t_left / 10.0, 20))
        for r in range(extra_restarts):
            if remaining() < 3:
                break
            h = np.zeros(n_best)
            positions = rng.choice(n_best, n_best // 2, replace=False)
            h[positions] = 1.0
            if n_best <= 120:
                h_sol, c5_sol = local_search_best(h, n_best, time_limit=8.0)
            else:
                h_sol, c5_sol = local_search_first(h, n_best, time_limit=8.0, rng=rng)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
            if c5_sol < best_c5 + 0.001:
                if remaining() > 5:
                    h2, c5_2 = local_search_best(h_sol, n_best, time_limit=min(remaining() * 0.3, 15.0))
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()

    if best_h is None:
        n = 42
        h = np.zeros(n)
        positions = rng.choice(n, n // 2, replace=False)
        h[positions] = 1.0
        best_h = h
        best_c5 = compute_c5(h, n)
        best_n = n

    return best_h, best_c5, best_n
