def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    # Bisection: find shift s such that sum(clip(h+s, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0)
        if np.sum(s) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

def get_subgradient(h, n, dx, k):
    """Compute subgradient of C[k]*dx w.r.t. h"""
    g = np.zeros(n)
    # C[k] = sum_{n: valid} h[n+k]*(1-h[n])
    # dC[k]/dh[j] = (1-h[j-k])*1[0<=j-k<n] - h[j+k]*1[0<=j+k<n]
    # First term: j-k in [0,n-1] => j in [k, k+n-1] intersect [0,n-1]
    if k >= 0:

        g[k:] += (1.0 - h[:n-k])
        # Second term: j+k in [0,n-1] => j in [0, n-1-k]
        lo2, hi2 = 0, n - k
        g[lo2:hi2] -= h[lo2+k:hi2+k]
    else:

        kp = -k

        g[lo1:hi1] += (1.0 - h[lo1+kp:hi1+kp])
        # j+k = j-kp in [0,n-1] => j in [kp, n-1]
        g[kp:] -= h[:n-kp]
    g *= dx
    return g
