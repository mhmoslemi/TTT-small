from scipy.optimize import minimize, line_search

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr

def compute_c5_only(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def gradient_of_corr_lag(h, m):
    """Compute gradient of c_m = sum_i h[i+m]*(1-h[i]) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j - m < n:
            g += (1 - h[j - m])
        if 0 <= j + m < n:
            g -= h[j + m]
        grad[j] = g
    return grad

def gradient_max_corr(h, n_points, top_k=5):
    """Subgradient of max_k c_k using top_k active lags."""
    dx = 2.0 / n_points
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Get top_k indices
    top_idx = np.argsort(corr)[-top_k:]
    # Weight by softmax for smoothness
    temps = 0.01
    weights = np.exp(corr[top_idx] / temps - np.max(corr[top_idx]) / temps)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        m = k - (n - 1)
        grad += w * gradient_of_corr_lag(h, m)
    return dx * grad

        return np.clip(h - theta, 0, 1).sum() - target

    return best_h, best_c5

    """Create a random perturbation of h."""
    h_new = h.copy()
    idx = np.random.choice(len(h), size=min(n_perturb, len(h)), replace=False)
    for i in idx:

    return np.correlate(h, 1 - h, mode="full")

def grad_corr_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    # h[i+m] term: j = i+m, so i = j-m, need 0<=j-m<n and 0<=j<n

    return grad

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    return float(np.max(compute_corr(h)) * dx)

def grad_corr_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    a = max(0, m)
    b = min(n, n + m)
    if a < b:
        grad[a:b] += (1.0 - h[a - m:b - m])
    a2 = max(0, -m)
    b2 = min(n, n - m)
    if a2 < b2:
        grad[a2:b2] -= h[a2 + m:b2 + m]
    return grad

def subgrad_max_corr(h, n_points, temperature=0.02):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    thresh = 1e-8
    for k in range(len(corr)):
        if w[k] > thresh:
            m = k - (n - 1)
            grad += w[k] * grad_corr_lag(h, m)
    return (2.0 / n_points) * grad

def project_feasible(h, n_points):

def perturb_h(h, n_points, seed, magnitude=0.05):
    rng = np.random.RandomState(seed)
