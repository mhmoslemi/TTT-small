def compute_c5(h, n):

def solve_1d(C0, Cs, as_, lam, n):
    """Solve the 1D subproblem for one coordinate.
    C0: zero-lag constant
    Cs, as_: arrays of length 2n-2 with Ck and ak for non-zero lags
    lam: Lagrange multiplier
    Returns optimal x in [0,1].

    # f(x) = max(C0 + (1+lam)*x - x^2, max_k(Ck + (ak+lam)*x))
    
    # Candidate points
    candidates = [0.0, 1.0]
    
    # Vertex of quadratic
    xv = (1.0 + lam) / 2.0
    if 0.0 <= xv <= 1.0:
        candidates.append(xv)
    
    # Quadratic-line intersections
    for Ck, ak in zip(Cs, as_):
        # C0 + (1+lam)*x - x^2 = Ck + (ak+lam)*x
        # -x^2 + (1+lam-ak-lam)*x + (C0-Ck) = 0
        # -x^2 + (1-ak)*x + (C0-Ck) = 0
        # x^2 - (1-ak)*x - (C0-Ck) = 0
        b = -(1.0 - ak)
        c = C0 - Ck
        disc = b*b - 4*c  # = (1-ak)^2 + 4*(C0-Ck)
        disc = (1.0 - ak)**2 + 4.0*(C0 - Ck)
        if disc >= 0:
            sq = np.sqrt(disc)
            x1 = ((1.0 - ak) + sq) / 2.0
            x2 = ((1.0 - ak) - sq) / 2.0
            if 0.0 <= x1 <= 1.0:
                candidates.append(x1)
            if 0.0 <= x2 <= 1.0:
                candidates.append(x2)
    
    # Evaluate at all candidates
    def f(x):
        q = C0 + (1.0 + lam) * x - x * x
        lines = Cs + (as_ + lam) * x
        return max(q, np.max(lines))
    
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        x = min(max(x, 0.0), 1.0)
        val = f(x)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x

def lagrangian_cd(h, n, lam, max_sweeps=5):

        for j in range(n):

            x_old = h[j]
            
            # Zero lag
            C0 = np.sum(h * (1.0 - h)) - x_old * (1.0 - x_old)
            
            # Non-zero lags

            as_ = np.zeros(2 * n - 2)
            idx = 0
            for lag in range(1, n):
                # Positive lag
                a = 0.0
                C = 0.0
                for i in range(n - lag):
                    if i + lag == j:
                        a += (1.0 - h[i])
                    elif i == j:
                        a -= h[i + lag]
                    else:
                        C += h[i + lag] * (1.0 - h[i])
                Cs[idx] = C
                as_[idx] = a
                idx += 1
                
                # Negative lag (-lag)
                a = 0.0
                C = 0.0
                for i in range(lag, n):
                    if i - lag == j:
                        a += (1.0 - h[i])
                    elif i == j:
                        a -= h[i - lag]
                    else:
                        C += h[i - lag] * (1.0 - h[i])
                Cs[idx] = C
                as_[idx] = a
                idx += 1
            
            x_new = solve_1d(C0, Cs, as_, lam, n)
            h[j] = x_new
    
    return h

import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, n):

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid

            hi = mid

def solve_1d_c0(C0, Cs, as_, lam):
    """Solve min_{x in [0,1]} max(C0 + (1+lam)*x - x^2, max_k(Ck+(ak+lam)*x))"""
    candidates = [0.0, 1.0]
    xv = (1.0 + lam) / 2.0
    if 0.0 <= xv <= 1.0:
        candidates.append(xv)
    
    # Quadratic-line intersections: C0+(1+lam)*x-x^2 = Ck+(ak+lam)*x
    # x^2 - (1-ak)*x - (C0-Ck) = 0

        b = 1.0 - ak
        c = C0 - Ck

        if disc >= 0:

                if 0.0 <= x <= 1.0:
                    candidates.append(float(x))

        for j in range(n):
