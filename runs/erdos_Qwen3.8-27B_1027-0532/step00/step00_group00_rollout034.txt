import time
import random

def compute_c5(h):
    """Compute C5 value for a given array h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binarize(h):
    """Convert h to binary maintaining sum."""
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def local_search_binary(h, seed=42, time_limit=100.0):
    """Local search on binary sequences."""
    rng = random.Random(seed)
    n = len(h)
    h = binarize(h).copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    
    start = time.time()
    no_improve = 0
    max_no_improve = 5000
    
    while time.time() - start < time_limit:
        ones_mask = h == 1.0
        zeros_mask = h == 0.0
        n_ones = int(np.sum(ones_mask))
        n_zeros = int(np.sum(zeros_mask))
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Pick random positions
        i = rng.choice(np.where(ones_mask)[0])
        j = rng.choice(np.where(zeros_mask)[0])
        
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Random restart perturbation
            for _ in range(10):
                i = rng.choice(np.where(h == 1)[0])
                j = rng.choice(np.where(h == 0)[0])
                h[i] = 0.0
                h[j] = 1.0
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            no_improve = 0
    
    return best_h, best_c

def continuous_optimize(h0, seed=42, time_limit=100.0):
    """Continuous optimization using subgradient method."""
    rng = random.Random(seed)
    n = len(h0)
    h = h0.astype(float).copy()
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h += (n / 2.0 - h_sum) / n
    h = np.clip(h, 0, 1)
    # Fix sum exactly
    diff = n / 2.0 - np.sum(h)
    # Adjust smallest entries
    if abs(diff) > 1e-10:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in idx:
            if diff > 0:
                adj = min(h_max - h[i], diff) if (h_max := 1.0) > h[i] else 0
            else:
                adj = max(-h[i], diff)
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-10:
                break
    
    best_c = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    lr = 0.01
    alpha = 50.0  # softmax temperature
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        # Softmax weights for the max
        corr_scaled = corr * alpha
        corr_max = np.max(corr_scaled)
        weights = np.exp(corr_scaled - corr_max)
        weights /= np.sum(weights)
        
        # Gradient
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-8:
                continue
            k = k_idx - (n - 1)
            for i in range(n):
                g = 0.0
                if 0 <= i - k <= n - 1:
                    g += (1.0 - h[i - k])
                if 0 <= i + k <= n - 1:
                    g += -h[i + k]
                grad[i] += weights[k_idx] * g

        h_new = h - lr * grad

        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target = n / 2.0
        if h_sum != target:

            h_new += (target - h_sum) / n
            h_new = np.clip(h_new, 0, 1)

            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr = max(lr * 0.95, 0.001)
        
        h = h_new
        alpha *= 1.01
    
    return best_h, best_c

def continuous_refine(h, seed=42, time_limit=20.0):
    """Refine a binary-ish solution using continuous optimization."""
    rng = random.Random(seed)
    n = len(h)
    h = h.astype(float).copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    step_size = 0.05
    alpha = 20.0  # temperature for softmax
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Compute softmax weights
        scaled = corr * alpha
        scaled -= np.max(scaled)
        weights = np.exp(scaled)
        weights /= np.sum(weights)
        
        # Compute gradient (weighted sum of per-shift gradients)
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            w = weights[k_idx]
            if w < 1e-6:
                continue
            k = k_idx - (n - 1)
            for i in range(n):
                g = 0.0
                if 0 <= i - k < n:
                    g += (1.0 - h[i - k])
                if 0 <= i + k < n:
                    g += -h[i + k]
                grad[i] += w * g
        
        # Gradient step with projection
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        target_sum = n / 2.0
        diff = target_sum - np.sum(h_new)
        if abs(diff) > 1e-10:
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step_size = min(step_size * 1.01, 0.2)
        else:
            step_size = max(step_size * 0.95, 0.005)
        
        alpha *= 1.005
    
    return best_h, best_c
