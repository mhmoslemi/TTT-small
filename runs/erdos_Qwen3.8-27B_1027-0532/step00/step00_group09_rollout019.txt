import numpy as np
import time

def _project(h, n, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _overlaps(h, n):
    o = np.empty(2 * n - 1)
    imh = 1.0 - h
    for s in range(-(n - 1), n):
        if s >= 0:
            o[s + n - 1] = np.dot(h[s:], imh[:n - s])
        else:
            o[s + n - 1] = np.dot(h[:n + s], imh[-s:])
    return o

def _c5_from_overlaps(ov, dx):
    return np.max(ov) * dx

def _c5(h, dx):
    return _c5_from_overlaps(_overlaps(h, len(h)), dx)

def _grad(h, imh, w, n):
    g = np.zeros(n)
    for si in range(2 * n - 1):
        wi = w[si]
        if wi < 1e-12:
            continue
        s = si - (n - 1)
        if s >= 0:
            m = n - s
            g[s:] += wi * imh[:m]
            g[:m] -= wi * h[s:]
        else:
            sp = -s
            m = n - sp
            g[:m] += wi * imh[sp:]
            g[sp:] -= wi * h[:m]
    return g

def _optimize(h0, n, dx, target, dur, rng, beta0=30.0, lr0=0.05):

    h = h0.copy()
    best_h = h.copy()

    best_c = _c5_from_overlaps(ov, dx)
    beta = beta0
    lr = lr0
    pat = 0

    while time.time() - t0 < dur:
        imh = 1.0 - h
        ov = _overlaps(h, n)
        sc = beta * ov
        sc -= np.max(sc)
        ex = np.exp(sc)
        es = ex.sum()
        if es < 1e-300:
            break
        ws = ex / es
        g = _grad(h, imh, ws, n)

        ov_new = _overlaps(h_new, n)
        c = _c5_from_overlaps(ov_new, dx)

            pat = 0
