def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

def compute_all_overlaps(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return corr * dx

def project_onto_constraints(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-10:
        return h
    # Need to adjust sum by -diff
    # Use binary search on threshold for water-filling
    if diff > 0:
        # Need to decrease sum: reduce elements that are above some threshold
        # h_new = min(h, threshold) won't work directly
        # Simple approach: reduce all elements proportionally where possible
        # Better: use the fact that we need to remove `diff` total
        # Reduce from elements closest to 1 first
        sorted_idx = np.argsort(h)[::-1]  # largest first
        remaining = diff
        h = h.copy()
        for idx in sorted_idx:
            if remaining <= 1e-12:
                break
            can_remove = h[idx]  # can remove up to h[idx] (down to 0)
            take = min(can_remove, remaining)
            h[idx] -= take
            remaining -= take
        if remaining > 1e-8:
            # Fallback
            h = np.clip(h - diff/n, 0, 1)
    else:
        # Need to increase sum: add elements that are below some threshold
        sorted_idx = np.argsort(h)  # smallest first
        remaining = -diff
        h = h.copy()
        for idx in sorted_idx:
            if remaining <= 1e-12:
                break
            can_add = 1.0 - h[idx]
            take = min(can_add, remaining)
            h[idx] += take
            remaining -= take
        if remaining > 1e-8:
            h = np.clip(h - diff/n, 0, 1)
    return h

            if remaining < 1e-12:

def full_gradient(h, dx, weights):
    """Compute weighted sum of gradients over all k."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)
        j_lo = max(0, m)
        j_hi = min(n, m + n)
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi)
            grad[idx] += w * (1.0 - h[idx - m])
        j_lo2 = max(0, -m)
        j_hi2 = min(n, n - m)
        if j_lo2 < j_hi2:
            idx = np.arange(j_lo2, j_hi2)
            grad[idx] += w * (-h[idx + m])
    return grad * dx

import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_constraints(h, target_sum):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        order = np.argsort(h)[::-1]
        remaining = diff
        for idx in order:
            if remaining < 1e-12:

            h[idx] -= take
            remaining -= take

        order = np.argsort(h)
        remaining = -diff

            if remaining < 1e-12:
                break
            take = min(1.0 - h[idx], remaining)
            h[idx] += take
            remaining -= take

    n_corr = 2 * n - 1
