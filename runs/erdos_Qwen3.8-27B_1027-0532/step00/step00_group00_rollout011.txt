def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation"""
    n_old = len(h)
    old_x = np.linspace(0, 2, n_old, endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(new_x, old_x, h)

    h_interp = np.clip(h_interp, 0, 1)

    target_sum = n_new / 2.0

    if abs(diff) > 1e-10:

        if diff > 0:

                h_interp[i] += add

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample_binary(h_old, n_new):
    """Upsample a binary (or step) function to higher resolution"""
    n_old = len(h_old)
    ratio = n_new / n_old
    # For each new position, find the corresponding old position
    indices = np.minimum((np.arange(n_new) * ratio).astype(int), n_old - 1)
    h_new = h_old[indices].copy()
    # Ensure sum is n_new/2
    target_sum = n_new / 2.0
    # If h_old is binary, h_new is binary, sum = n_new/2 if ratio is integer
    # For non-integer ratio, we might need to adjust
    if abs(h_new.sum() - target_sum) > 1e-10:
        # Adjust: flip some bits
        diff = target_sum - h_new.sum()
        if diff > 0:
            zeros = np.where(h_new == 0)[0]
            for i in zeros[:int(diff)]:
                h_new[i] = 1
        else:
            ones = np.where(h_new == 1)[0]
            for i in ones[:int(-diff)]:
                h_new[i] = 0
    return h_new

    n = len(h)
    h = h.copy()

        sorted_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sorted_idx[:n//2]] = 1.0

    best_c5 = current_c5
    best_h = h.copy()

    max_no_improve = 3000

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue

        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)))
    elif len(h) > n:
        h = h[:n]
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n, dtype=float)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def upsample_step(h_old, n_new):

    indices = np.minimum((np.arange(n_new) * ratio).astype(int), n_old - 1)
    h_new = h_old[indices].astype(float).copy()
