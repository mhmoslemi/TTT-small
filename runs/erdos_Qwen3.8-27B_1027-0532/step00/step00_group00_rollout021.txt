    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    if diff > 0:
        # Add mass to positions with most room
        order = np.argsort(1.0 - h)
        per = diff / n

            add = min(per, 1.0 - h[idx])
            h[idx] += add
            diff -= add

def repair_boundary(h):
    """Clip to [0,1] and fix sum."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Fix sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 0:
                h += diff * room / total_room
            h = np.clip(h, 0.0, 1.0)
        else:
            avail = h
            total_avail = avail.sum()
            if total_avail > 0:
                h += diff * avail / total_avail
            h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_step(h, dx, alpha, n):
    """One step of projected subgradient descent."""
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)  # the shift
    
    # Compute gradient of g_s at h
    grad = np.zeros(n)
    for i in range(n):
        # First term: (1-h[i-s]) if i-s is valid
        if 0 <= i - s <= n - 1:
            grad[i] += (1 - h[i - s])
        # Second term: -h[i+s] if i+s is valid
        if 0 <= i + s <= n - 1:
            grad[i] -= h[i + s]
    grad *= dx
    
    # Gradient descent step
    h_new = h - alpha * grad
    
    # Project onto [0,1]^n intersect {sum = n/2}
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    target = n / 2.0
    diff = target - np.sum(h_new)
    if diff > 0:
        # Add to positions with room
        room = 1.0 - h_new
        total_room = room.sum()
        if total_room > 1e-10:

    else:
        avail = h_new
        total_avail = avail.sum()

def targeted_greedy(h, dx, n, max_passes=1000, time_limit=120, t_start=None):
    """Targeted greedy: focus on reducing the max shift."""
    if t_start is None:
        t_start = time.time()
    h = h.copy()
    c5 = evaluate_c5(h, dx)
    
    for _ in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        
        # Find the active shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = np.argmax(corr)
        s = max_k - (n - 1)
        
        # For this shift, the gradient tells us which direction to move
        # We want to decrease g_s = sum_i h[i]*(1-h[i-s])
        # ∂g_s/∂h[i] = (1-h[i-s]) - h[i+s] (with boundary checks)
        # So we should increase h[i] where ∂g_s/∂h[i] < 0 and decrease where > 0
        
        # But we need to maintain the sum constraint
        # So: increase h[i] and decrease h[j] where ∂g_s/∂h[i] < 0 and ∂g_s/∂h[j] > 0
        
        # Find the best pair (i, j)
        best_c5 = c5
        best_move = None
        
        # Compute the "desirability" of increasing each h[i]
        # desirability[i] = -∂g_s/∂h[i] (we want to increase where this is positive)
        desirability = np.zeros(n)
        for i in range(n):
            g = 0.0
            if 0 <= i - s <= n - 1:
                g += (1.0 - h[i - s])
            if 0 <= i + s <= n - 1:
                g -= h[i + s]
            desirability[i] = -g  # positive means we should increase h[i]
        
        # Sort by desirability
        increase_order = np.argsort(-desirability)  # most desirable to increase first
        decrease_order = np.argsort(desirability)   # most desirable to decrease first
        
        # Try top pairs
        found = False
        for ii in range(min(10, n)):
            for jj in range(min(10, n)):
                i = increase_order[ii]
                j = decrease_order[jj]
                if i == j:

                for d in [0.02, 0.05, 0.1]:

                        h[i] += d

    h = np.clip(h, 0.0, 1.0)

def targeted_greedy(h, dx, n, max_passes=2000, time_limit=120, t_start=None):
    if t_start is None:
        t_start = time.time()
    h = h.copy()
    c5 = evaluate_c5(h, dx)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.2]
    
    for _ in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = int(np.argmax(corr))
        s = max_k - (n - 1)
        desirability = np.zeros(n)
        for i in range(n):
            g = 0.0
            if 0 <= i - s <= n - 1:
                g += (1.0 - h[i - s])
            if 0 <= i + s <= n - 1:
                g -= h[i + s]
            desirability[i] = -g
        inc_order = np.argsort(-desirability)
        dec_order = np.argsort(desirability)
        best_c5 = c5
        best_move = None
        for ii in range(min(15, n)):
            for jj in range(min(15, n)):
                i = int(inc_order[ii])
                j = int(dec_order[jj])
                if i == j:
                    continue
                for d in deltas:
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        break
                    h[i] += d
                    h[j] -= d
                    c5_new = evaluate_c5(h, dx)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_move = (i, j, d)
                    else:
                        h[i] -= d
                        h[j] += d
                        break
        if best_move is None:
            break
        i, j, d = best_move
        h[i] += d
        h[j] -= d
        c5 = best_c5
    return h, c5

def sa_optimize(h, dx, n, rng, time_limit, t_start):
    h = h.copy()
    c5 = evaluate_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    T = 0.005
    T_min = 1e-8
    cool = 0.99997
    stuck_count = 0
    
    while True:
        if time.time() - t_start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            T = max(T * cool, T_min)
            continue
        delta = rng.uniform(0.01, 0.4)
        if rng.random() < 0.5:
            new_i = h[i] + delta
            new_j = h[j] - delta
        else:
            new_i = h[i] - delta
            new_j = h[j] + delta
        if new_i < 0.0 or new_i > 1.0 or new_j < 0.0 or new_j > 1.0:
            T = max(T * cool, T_min)
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = new_i, new_j
        c5_new = evaluate_c5(h, dx)
        if c5_new <= c5 or (T > T_min * 100 and rng.random() < np.exp(-(c5_new - c5) / T)):
            c5 = c5_new
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                stuck_count = 0
            else:
                stuck_count += 1
            if stuck_count > 5000 and T > T_min * 10:
                T = max(T * 0.5, T_min)
                stuck_count = 0
        else:
            h[i], h[j] = old_i, old_j
        T = max(T * cool, T_min)
    return best_h, best_c5

def subgrad_descent(h, dx, n, max_iter=10000):
    h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    alpha = 0.05
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[::-1][:3]
        grad = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            for i in range(n):
                if 0 <= i - s <= n - 1:
                    grad[i] += (1.0 - h[i - s])
                if 0 <= i + s <= n - 1:
                    grad[i] -= h[i + s]
        grad *= dx / 3.0
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            alpha *= 0.9
        alpha = max(alpha * 0.9999, 0.001)
    return best_h, best_c5

def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def sine_init(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2.0 * np.pi)
    amp = rng.uniform(0.1, 0.4)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 45
    dx = 2.0 / n
    
    best_h = None

    try:

        if ihv is not None:

    elapsed = time.time() - t0

        if c5_2 < best_c5:

            best_h = h2
    
    elapsed = time.time() - t0
    if budget_s - elapsed > 100:
        tl = min(budget_s - elapsed - 200, 300)
        h3, c5_3 = sa_optimize(best_h, dx, n, rng, tl, time.time())
        if c5_3 < best_c5:

            best_h = h3
    
    elapsed = time.time() - t0
    if budget_s - elapsed > 30:

        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    
    for attempt in range(8):
        elapsed = time.time() - t0
        if budget_s - elapsed < 40:
            break
        if attempt % 2 == 0:
            h_rand = sine_init(n, rng)
        else:

        tl = min(budget_s - elapsed - 30, 80)
        h5, c5_5 = sa_optimize(h_rand, dx, n, rng, tl, time.time())
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            elapsed = time.time() - t0
            if budget_s - elapsed > 20:
                tl2 = min(budget_s - elapsed - 10, 60)
                h5b, c5_5b = targeted_greedy(h5, dx, n, time_limit=tl2, t_start=time.time())
                if c5_5b < best_c5:
                    best_c5 = c5_5b
                    best_h = h5b
    
    elapsed = time.time() - t0
    if budget_s - elapsed > 20:
        tl = min(budget_s - elapsed - 10, 120)
        h6, c5_6 = targeted_greedy(best_h, dx, n, time_limit=tl, t_start=time.time())
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6
    
    elapsed = time.time() - t0
    if budget_s - elapsed > 15:
        tl = min(budget_s - elapsed - 5, 60)
        h7, c5_7 = sa_optimize(best_h, dx, n, rng, tl, time.time())
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7
    
    elapsed = time.time() - t0

            best_h = h8

    best_c5 = evaluate_c5(best_h, dx)
    
    return best_h, best_c5, n
