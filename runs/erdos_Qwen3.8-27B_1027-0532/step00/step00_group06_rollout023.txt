import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    """Project h onto {sum(h)=n/2, 0<=h<=1}"""
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        # Distribute diff among non-saturated coordinates
        if diff > 0:
            mask = h < 1
        else:
            mask = h > 0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def subgradient(h, n, dx):
    """Compute subgradient of C5 at h"""
    corr = np.correlate(h, 1-h, mode='full')
    k_star = np.argmax(corr)
    grad = np.zeros(n)
    # ∂corr[k_star]/∂h[j]
    k = k_star
    for j in range(n):
        # Condition A: max(0, n-1-k) <= j <= min(n-1, 2n-2-k)

            grad[j] -= h[m]
    return grad / dx  # subgradient of C5 = max corr * dx

        if pk < 1e-14:
            continue
        j_min = max(0, n - 1 - k)
        j_max = min(n - 1, 2 * n - 2 - k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            ms = k + js - n + 1
            grad[js] += pk * (1.0 - h[ms])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            ms = js - k + n + 1 - 1
            grad[js] -= pk * h[ms]
    grad *= dx
    return grad

def optimize_smooth(h0, n, dx, time_limit, seed):
    rng = np.random.default_rng(seed)
    h = h0.astype(np.float64).copy()
    h = project_to_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    tau = 5.0
    alpha = 0.05
    no_improve = 0
    max_no_improve = 300
    
    t0 = time.time()
    while time.time() - t0 < time_limit:
        grad = compute_smooth_grad(h, n, dx, tau)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            no_improve = 0
            tau = min(tau * 1.02, 100.0)
        else:
            h = h_new
            no_improve += 1
            if no_improve == 100:
                alpha = max(alpha * 0.8, 0.001)
            elif no_improve >= max_no_improve:
                # Diversify
                n_pert = rng.integers(2, max(3, n // 12))
                idx = rng.choice(n, size=n_pert, replace=False)
                h[idx] += rng.uniform(-0.3, 0.3, size=n_pert)
                h = project_to_feasible(h, n)
                no_improve = 0
                tau = max(tau * 0.3, 3.0)
                alpha = 0.05
    
    return best_h, best_c5

def local_refine(h, n, dx, time_limit, seed):
    """Try coordinate-wise improvements"""
    rng = np.random.default_rng(seed)
    h = h.astype(np.float64).copy()
    h = project_to_feasible(h, n)
    best_c5 = compute_c5(h, dx)
    
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            if time.time() - t0 > time_limit:
                break
            old_val = h[j]
            h[j] = 0.0
            h = project_to_feasible(h, n)
            c5_0 = compute_c5(h, dx)
            
            h[j] = 1.0
            h = project_to_feasible(h, n)
            c5_1 = compute_c5(h, dx)
            
            h[j] = old_val
            h = project_to_feasible(h, n)
            
            if c5_0 < best_c5:
                h[j] = 0.0
                h = project_to_feasible(h, n)
                best_c5 = c5_0
                improved = True
            elif c5_1 < best_c5:
                h[j] = 1.0
                h = project_to_feasible(h, n)
                best_c5 = c5_1
                improved = True
            # else keep old_val (h is already restored)
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    def elapsed():

    def remaining():

                break

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, best_n
    
    # Stage 4: Try higher resolution
    for n_new in [best_n + 20, best_n + 40, best_n + 70]:
        if remaining() < 30:
            break
        if n_new > 300:
            continue
        dx_new = 2.0 / n_new
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False) + 1.0 / best_n
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_feasible(h_interp, n_new)
        
        t_stage = min(remaining() * 0.4, 120)
        if t_stage < 15:

        h_opt, c5_opt = optimize_smooth(h_interp, n_new, dx_new, t_stage, seed + n_new)
        if c5_opt < best_c5:

    # Final refinement
    if remaining() > 10:
        h_ref, c5_ref = local_refine(best_h, best_n, 2.0/best_n, min(remaining(), 60), seed+500)
        if c5_ref < best_c5:
            best_h, best_c5, best_n = h_ref, c5_ref, best_n
    
    return best_h, best_c5, best_n
