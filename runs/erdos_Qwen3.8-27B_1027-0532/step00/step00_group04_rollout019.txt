def project_onto_constraint(h, target_sum):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    exp_vals = np.exp(T * (corr - max_c))
    p = exp_vals / np.sum(exp_vals)
    
    # p[k] for k=0..2n-2, shift d_k = k-(n-1)
    # For each j: grad[j] = dx * (sum_k p[k]*(1-h[j+d_k])*[valid] - sum_k p[k]*h[j-d_k]*[valid])
    
    # Vectorize: create matrix M where M[j,k] = 1 if 0<=j+d_k<n
    # This is a band matrix
    
    j = np.arange(n)
    d = np.arange(2*n-1) - (n-1)  # shifts
    jd = j[:, None] + d[None, :]  # shape (n, 2n-1)
    jdm = j[:, None] - d[None, :]
    
    valid_jd = (jd >= 0) & (jd < n)
    valid_jdm = (jdm >= 0) & (jdm < n)
    
    # term1: (1-h[jd]) * valid_jd
    # We need h[jd] where valid
    # Create a padded h
    h_pad = np.zeros(3*n)  # pad to handle all shifts
    h_pad[n:2*n] = h
    
    # For jd: the actual index into h_pad is jd + n (shifted by n)
    # For jdm: the actual index into h_pad is jdm + n
    
    # But we need to be careful: jd can range from -(n-1) to 2n-2
    # h_pad[n:2*n] = h, so h_pad[n + idx] = h[idx] for idx in [0, n-1]
    # For jd in [0, n-1]: h[jd] = h_pad[n+jd]
    # For jd outside [0, n-1]: contribution is 0 (handled by valid mask)
    
    # Actually, let me just use fancy indexing with clipping
    jd_clipped = np.clip(jd, 0, n-1)
    jdm_clipped = np.clip(jdm, 0, n-1)
    
    term1 = np.where(valid_jd, 1.0 - h[jd_clipped], 0.0)
    term2 = np.where(valid_jdm, h[jdm_clipped], 0.0)
    
    grad = (term1 - term2) @ p * dx
    
    return grad
