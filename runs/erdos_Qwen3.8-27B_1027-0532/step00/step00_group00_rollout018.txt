        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta <= 0 or np.random.random() < np.exp(-delta / T):
            c5 = new_c5
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
            stall += 1

        T = max(T * 0.99999, 0.0001)
        iter_count += 1

        if stall > 80000:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    init_h = np.array(initial_h_values, dtype=float)
    best_h = init_h.copy()
    best_c5 = compute_c5(init_h)
    best_n = len(init_h)

    # Phase 1: Optimize on initial grid (n=45, odd)
    deadline1 = t0 + budget_s * 0.15
    n = len(init_h)
    target_sum = n / 2.0
    h = init_h.copy()
    c5 = best_c5
    T = 0.006
    stall = 0

    while time.time() < deadline1:
        move = np.random.random()
        if move < 0.55:
            i, j = np.random.choice(n, 2, replace=False)
            tmp = h[i]
            h[i] = h[j]
            h[j] = tmp
            new_c5 = compute_c5(h)
            if new_c5 <= c5 or np.random.random() < np.exp(-((new_c5 - c5) / T)):
                c5 = new_c5
                stall = 0
            else:
                h[j] = h[i]
                h[i] = tmp
                stall += 1
        elif move < 0.85:
            i = np.random.randint(n)
            old = h[i]
            h[i] = np.clip(old + np.random.randn() * 0.2, 0, 1)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-10:
                if diff > 0:
                    mask = h < 1
                else:
                    mask = h > 0
                if mask.sum() > 0:
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0, 1)
            new_c5 = compute_c5(h)
            if new_c5 <= c5 or np.random.random() < np.exp(-((new_c5 - c5) / T)):
                c5 = new_c5
                stall = 0
            else:
                h[i] = old
                diff = target_sum - h.sum()
                if abs(diff) > 1e-10:
                    if diff > 0:
                        mask = h < 1
                    else:
                        mask = h > 0
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                        h = np.clip(h, 0, 1)
                stall += 1
        else:
            i = int(np.argmin(h))
            j = int(np.argmax(h))
            if i == j:
                continue
            tmp = h[i]
            h[i] = h[j]
            h[j] = tmp
            new_c5 = compute_c5(h)
            if new_c5 <= c5 or np.random.random() < np.exp(-((new_c5 - c5) / T)):
                c5 = new_c5
                stall = 0
            else:
                h[j] = h[i]
                h[i] = tmp
                stall += 1

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        T = max(T * 0.9999, 0.001)
        if stall > 40000:
            stall = 0
            T = 0.02
            for _ in range(2):
                ii, jj = np.random.choice(n, 2, replace=False)
                h[ii], h[jj] = h[jj], h[ii]

    # Phase 2: Binary optimization on n=90
    deadline2 = t0 + budget_s * 0.72
    n_fine = 90
    count = n_fine // 2

    starts = []

    h_up = np.interp(np.linspace(0, 2, n_fine, endpoint=False),
                     np.linspace(0, 2, best_n, endpoint=False), best_h)
    starts.append(make_binary_topk(h_up, count))

    tm = thue_morse(n_fine).astype(np.float64)
    if tm.sum() != count:
        tm = fix_binary_count(tm, count)
    starts.append(tm.copy())

    hr = np.zeros(n_fine)
    hr[np.random.choice(n_fine, count, replace=False)] = 1
    starts.append(hr)

    tm2 = 1.0 - thue_morse(n_fine).astype(np.float64)
    if tm2.sum() != count:
        tm2 = fix_binary_count(tm2, count)
    starts.append(tm2.copy())

    h_cos = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_fine) / n_fine * 7)
    starts.append(make_binary_topk(h_cos, count))

    h_sin = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_fine) / n_fine * 5 + 1.3)
    starts.append(make_binary_topk(h_sin, count))

    time_per_start = max((deadline2 - time.time()) / len(starts), 10)

    for h_start in starts:
        if time.time() >= deadline2:
            break
        start_t = time.time()
        end_t = min(deadline2, start_t + time_per_start)

        c5_s = compute_c5(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()
            best_n = n_fine

        h_opt, c5_opt = anneal_binary(h_start, end_t)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_fine

    # Phase 3: Try n=180 if time remains
    deadline3 = t0 + budget_s * 0.95
    if time.time() < deadline3:
        n_fine2 = 180
        count2 = n_fine2 // 2

        h_up2 = np.interp(np.linspace(0, 2, n_fine2, endpoint=False),
                          np.linspace(0, 2, best_n, endpoint=False), best_h)
        h_start2 = make_binary_topk(h_up2, count2)
        c5_s2 = compute_c5(h_start2)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_start2.copy()
            best_n = n_fine2

        h_opt2, c5_opt2 = anneal_binary(h_start2, deadline3)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            best_n = n_fine2

    return best_h, best_c5, best_n
