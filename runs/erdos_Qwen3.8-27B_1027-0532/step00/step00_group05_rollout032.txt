    """Compute all correlation values (before multiplying by dx)."""
    return np.correlate(h, 1 - h, mode="full")

def compute_c5_value(h, n):
    """Compute C5 value."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}."""
    # Bisection on lambda
    lo = v.min() - 1
    hi = v.max() + 1

For the gradient of corr[k] with d = k - (n-1):

This is vectorizable! Let me rewrite:

```python
def gradient_smooth_max_vec(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[0:n-d] += w * (1.0 - h[d:n])
            grad[d:n] -= w * h[0:n-d]
        else:
            m = -d
            grad[m:n] += w * (1.0 - h[0:n-m])
            grad[0:n-m] -= w * h[m:n]
    
    grad *= dx
    return grad
```
