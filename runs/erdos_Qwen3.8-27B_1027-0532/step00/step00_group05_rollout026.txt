import numpy as np
import time


def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    lo = -h.max() - 0.5
    hi = -h.min() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def compute_grad_smooth_max(h, tau):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    scaled = tau * corr
    scaled -= scaled.max()
    weights = np.exp(scaled)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-12:
            continue
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        ta = max(0, n_lo + k)
        tb = min(n, n_hi + k + 1)
        if ta < tb:
            grad[ta:tb] += weights[m] * g[ta - k:tb - k]
        ta2 = max(0, n_lo)
        tb2 = min(n, n_hi + 1)
        if ta2 < tb2:
            grad[ta2:tb2] -= weights[m] * h[ta2 + k:tb2 + k]
    return grad * dx


def smooth_gradient_descent(h, n, target_sum, time_limit, tau_start=20.0, lr_start=0.05):
    tau = tau_start
    lr = lr_start
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    stall_count = 0
    while True:
        if time.time() > time_limit:
            break
        grad = compute_grad_smooth_max(h, tau)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                lr = max(0.0005, lr * 0.8)
                tau = min(500.0, tau * 1.05)
                if lr < 0.001:
                    break
                stall_count = 0
        h = h_new
        tau = min(500.0, tau * 1.003)
        if stall_count % 50 == 0:
            lr = max(0.0005, lr * 0.995)
    return best_h, best_c5


def binary_local_search(h, target_sum, time_limit):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = project_feasible(h, target_sum)
    c5 = evaluate_c5(h)
    improved = True
    while improved and time.time() < time_limit:
        improved = False
        ones = [i for i in range(n) if h[i] > 0.5]
        zeros = [i for i in range(n) if h[i] < 0.5]
        for i in ones:
            if time.time() > time_limit:
                return h, c5
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5_trial = evaluate_c5(h)
                if c5_trial < c5 - 1e-10:
                    c5 = c5_trial
                    improved = True
                    ones.remove(i)
                    zeros.remove(j)
                    ones = [x for x in range(n) if h[x] > 0.5]
                    zeros = [x for x in range(n) if h[x] < 0.5]
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return h, c5


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new


def continuous_local_search(h, n, target_sum, time_limit):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step_sizes = [0.1, 0.05, 0.02, 0.01]
    for ssize in step_sizes:
        while time.time() < time_limit:
            improved = False
            idx = np.random.randint(0, n)
            perturbation = (np.random.rand(n) - 0.5) * ssize
            perturbation[idx] = np.random.choice([-ssize, ssize])
            h_trial = project_feasible(h + perturbation, target_sum)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5 - 1e-10:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial
                improved = True
        h = best_h
    return best_h, best_c5

    start_time = time.time()

        return budget_s - (time.time() - start_time)

        h0 = None

    best_c5 = np.inf

        best_h = h.copy()

        best_n = n

        time_limit = start_time + remaining() * 0.3
        h_opt, c5_opt = smooth_gradient_descent(h, n, target_sum, time_limit)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

    n_values = [42, 56, 63, 70, 84, 98, 112, 126, 140, 168]
    if best_h is not None:
        n_values = [x for x in n_values if x >= best_n] or n_values

    for n in n_values:
        if remaining() < 60:
            break
        target_sum = n / 2.0
        h_init = None
        if best_h is not None:
            h_interp = interpolate_h(best_h, n)
            h_init = project_feasible(h_interp, target_sum)
        else:
            h_init = np.full(n, 0.5)
            h_init[:n // 2] = 1.0
            h_init[n // 2:] = 0.0
            h_init = project_feasible(h_init, target_sum)

        n_restarts = 3
        per_restart_time = min(40.0, remaining() * 0.6 / max(1, n_restarts))
        for r in range(n_restarts):
            if remaining() < 30:
                break
            if r == 0:
                h_start = h_init.copy()
            elif r == 1:
                h_rand = np.zeros(n)
                idx = np.random.permutation(n)[:n // 2]
                h_rand[idx] = 1.0
                h_start = project_feasible(h_rand, target_sum)
            else:
                h_start = h_init + np.random.randn(n) * 0.1
                h_start = project_feasible(h_start, target_sum)

            tl = start_time + min(per_restart_time, remaining() * 0.4)
            h_opt, c5_opt = smooth_gradient_descent(h_start, n, target_sum, tl)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

    if remaining() > 20 and best_h is not None:

        target_sum = n / 2.0
        h_bin, c5_bin = binary_local_search(best_h, target_sum, start_time + remaining() * 0.4)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

            best_c5 = c5_cont
            best_h = h_cont.copy()
            best_n = n
