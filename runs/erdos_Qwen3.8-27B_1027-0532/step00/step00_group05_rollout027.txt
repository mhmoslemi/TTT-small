import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_c5_value(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(t * (c - c_max)))) / t

def gradient_smooth_c5(h, t):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    c = np.correlate(h, b, mode="full") * dx
    c_max = np.max(c)
    exps = np.exp(t * (c - c_max))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        wr = w[r] * dx
        if wr < 1e-18:
            continue
        if s >= 0:
            j1s, j1e = s, n
            j2s, j2e = 0, n - s
            grad[j1s:j1e] += wr * (1.0 - h[j1s - s:j1e - s])
            grad[j2s:j2e] -= wr * h[j2s + s:j2e + s]
        else:
            sp = -s
            j1s, j1e = 0, n - sp
            j2s, j2e = sp, n
            grad[j1s:j1e] += wr * (1.0 - h[j1s + sp:j1e + sp])
            grad[j2s:j2e] -= wr * h[j2s - sp:j2e - sp]
    return grad

def projected_gradient(h0, n_max_iter, deadline, t_start=10.0):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    t = t_start
    step = 0.1
    best_local = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        c5_curr = compute_c5(h)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_local = h.copy()
        
        grad = gradient_smooth_c5(h, t)
        h_new = project_feasible(h - step * grad, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_curr:
            h = h_new
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.5
            if step < 1e-8:
                break
        if it > 50:
            t = max(t * 0.95, 50.0)
    
    return best_local, best_c5

def upsample_func(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, deadline):
    n = len(h0)
    target_sum = n / 2.0
    
    def objective(h):
        return compute_c5(h)
    
    def objective_grad(h):
        # Use smoothed gradient as proxy
        return gradient_smooth_c5(h, 50.0)
    
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 'jac': lambda h: np.ones(n)},
    ]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h0, method='SLSQP', jac=objective_grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or res.fun < objective(h0):
            h_res = project_feasible(res.x, target_sum)
            return h_res
    except Exception:
        pass
    return h0
