    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    s = k_star - (N - 1)  # the shift

    g = np.zeros(N)
    for i in range(N):

        if 0 <= i - s < N:

        # Contribution from (1-h[n]) term: when n = i
        if 0 <= i + s < N:
            g[i] -= h[i + s]
    
    # Projected gradient step

    else:
        if N + s > 0:
            g[:N+s] += (1.0 - h[-s:])
        if N + s > 0:
            g[-s:] -= h[:N+s]
    return g

def get_top_shifts(h, k=3):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[::-1][:k]
    shifts = [int(idx) - (N - 1) for idx in top_k]
    return corr, shifts

def optimize_subgradient(h0, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    N = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    c5_current = best_c5

    t0 = time.time()
    lr = 0.04
    noise = 0.04
    n_top = 3
    it = 0

    while time.time() - t0 < time_limit:
        it += 1
        corr, shifts = get_top_shifts(h, n_top)
        g = np.zeros(N)
        for s in shifts:
            g += make_subgradient(h, s)
        g /= n_top

        h_new = h - lr * g + noise * rng.randn(N)
        h_new = project_feasible(h_new)
        c5_new = evaluate_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        if c5_new <= c5_current:
            h = h_new
            c5_current = c5_new
        else:
            T = max(lr * 0.3, 0.0005)
            if rng.random() < np.exp(-(c5_new - c5_current) / T):
                h = h_new
                c5_current = c5_new

        lr = max(lr * 0.9997, 0.003)
        if it % 3000 == 0:
            noise = max(noise * 0.92, 0.005)
        if it % 20000 == 0:
            h = best_h.copy() + noise * rng.randn(N)
            h = project_feasible(h)
            c5_current = evaluate_c5(h)

    return best_h, best_c5

def pair_swap_search(h0, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    N = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    t0 = time.time()
    delta = 0.2
    no_improve = 0
    max_no_improve = 500

    while time.time() - t0 < time_limit:
        i, j = rng.randint(0, N, 2)
        if i == j:
            continue

        dirs = [1, -1]
        for d in dirs:
            di = d * delta
            dj = -d * delta
            if h[i] + di < -1e-10 or h[i] + di > 1.0 + 1e-10:
                continue
            if h[j] + dj < -1e-10 or h[j] + dj > 1.0 + 1e-10:
                continue

            h[i] += di
            h[j] += dj
            c5_new = evaluate_c5(h)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
                delta = min(delta * 1.2, 0.5)
            else:
                no_improve += 1
                delta = max(delta * 0.9, 0.05)

            h[i] -= di
            h[j] -= dj

        if no_improve > max_no_improve:
            h = best_h.copy()
            pert = rng.choice(4)
            idx1 = rng.randint(0, N)
            idx2 = rng.randint(0, N)
            if idx1 == idx2:
                idx2 = (idx2 + 1) % N
            amt = rng.choice([-0.4, -0.2, 0.2, 0.4])
            h[idx1] += amt
            h[idx2] -= amt
            h = project_feasible(h)
            no_improve = 0
            delta = 0.2

    return best_h, best_c5

def binary_round(h):
    N = len(h)
    h_bin = np.round(h)
    h_bin = np.clip(h_bin, 0.0, 1.0)
    n_ones = int(np.sum(h_bin))
    target = N // 2
    if n_ones > target:
        excess = n_ones - target
        ones_idx = np.where(h_bin == 1.0)[0]
        ones_idx = ones_idx[np.argsort(h[ones_idx])[::-1][:excess]]
        h_bin[ones_idx] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        zeros_idx = zeros_idx[np.argsort(h[zeros_idx])[::-1][:deficit]]
        h_bin[zeros_idx] = 1.0
    return h_bin

def interp_to_N(h_old, N_new):
    N_old = len(h_old)
    if N_old == N_new:
        return h_old.copy()
    x_old = np.arange(N_old, dtype=float) / N_old
    x_new = np.arange(N_new, dtype=float) / N_new
    h_new = np.zeros(N_new)
    for i in range(N_new):
        pos = x_new[i] * N_old
        i0 = int(pos) % N_old
        i1 = (i0 + 1) % N_old
        frac = pos - int(pos)
        h_new[i] = (1.0 - frac) * h_old[i0] + frac * h_old[i1]
    return project_feasible(h_new)

def random_binary(N, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(N)
    idx = rng.choice(N, size=N // 2, replace=False)
    h[idx] = 1.0
    return h

def structured_init(N, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(N)
    gap = N // (N // 2)
    count = 0
    start = rng.randint(0, max(gap, 1))
    for i in range(start, N, gap):
        if count < N // 2:
            h[i] = 1.0
            count += 1
    if count < N // 2:
        remaining = N // 2 - count
        available = np.where(h == 0.0)[0]
        chosen = rng.choice(available, size=min(remaining, len(available)), replace=False)
        h[chosen] = 1.0
    h += 0.01 * rng.randn(N)
    return project_feasible(h)

def run(seed=42, budget_s=1300, **kwargs):
    global_start = time.time()

    def elapsed():
        return time.time() - global_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = np.inf

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
        h_init = project_feasible(h_init)
    except (NameError, UnboundLocalError, TypeError):
        h_init = None

    if h_init is not None:
        c5_init = evaluate_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
        N_init = len(h_init)
    else:
        N_init = 45
        best_h = random_binary(N_init, seed=seed)
        best_c5 = evaluate_c5(best_h)

    # Phase 1: Subgradient optimization on initial
    t1 = remaining() * 0.30
    if t1 > 20:
        h_opt, c5_opt = optimize_subgradient(best_h, t1, seed=seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Pair swap refinement
    t2 = remaining() * 0.20
    if t2 > 15:
        h_opt, c5_opt = pair_swap_search(best_h, t2, seed=seed + 1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Try binary rounding and re-optimize
    h_bin = binary_round(best_h)
    c5_bin = evaluate_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    t3 = remaining() * 0.15
    if t3 > 15:
        h_opt, c5_opt = optimize_subgradient(best_h, t3, seed=seed + 2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        t3b = min(remaining() * 0.10, 60)
        if t3b > 10:
            h_opt, c5_opt = pair_swap_search(best_h, t3b, seed=seed + 3)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Try different N values
    N_candidates = [35, 40, 45, 50, 55, 60, 70]
    N_candidates = [N for N in N_candidates if N != len(best_h)]

    for N in N_candidates:
        if remaining() < 40:
            break
        t_alloc = min(remaining() * 0.25, 100)
        if t_alloc < 25:
            break

        # Interpolated start
        h_start = interp_to_N(best_h, N)
        t_sub = t_alloc * 0.5
        h_opt, c5_opt = optimize_subgradient(h_start, t_sub, seed=seed + N)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        t_swap = min(remaining() * 0.10, 50)
        if t_swap > 15:
            h_opt, c5_opt = pair_swap_search(best_h, t_swap, seed=seed + N + 500)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Random restarts at different N
    for N in [45, 50, 60]:
        if remaining() < 30:
            break
        for trial in range(2):
            if remaining() < 20:
                break
            h_rand = random_binary(N, seed=seed + 1000 + N * 7 + trial)
            h_struct = structured_init(N, seed=seed + 2000 + N * 7 + trial)
            # Pick the better of the two
            c5_r = evaluate_c5(h_rand)
            c5_s = evaluate_c5(h_struct)
            h_start = h_rand if c5_r < c5_s else h_struct
            t_trial = min(remaining() * 0.20, 80)
            if t_trial < 15:
                break
            h_opt, c5_opt = optimize_subgradient(h_start, t_trial * 0.6, seed=seed + 3000 + N + trial)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            t_swap = min(remaining() * 0.10, 40)
            if t_swap > 10:
                h_opt, c5_opt = pair_swap_search(best_h, t_swap, seed=seed + 4000 + N + trial)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 6: Final refinement
    if remaining() > 15:
        t_final = remaining() - 5
        h_opt, c5_opt = optimize_subgradient(best_h, t_final * 0.6, seed=seed + 7777)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if remaining() > 10:
            h_opt, c5_opt = pair_swap_search(best_h, remaining() - 5, seed=seed + 8888)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
