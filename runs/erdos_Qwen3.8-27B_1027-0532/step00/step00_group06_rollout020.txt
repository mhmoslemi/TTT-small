```python
    n = n_points
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    s = c * dx / tau
    M = np.max(s)
    p = np.exp(s - M)
    p /= np.sum(p)
    
    # Compute grad[j] = dx^2 * [correlation of p and (1-h) at lag j - correlation of p and h at lag j]
    # where the correlation is defined appropriately
    
    # Method: use FFT
    # We want for each j in [0, n-1]:
    # sum_{k} p[k] * (1-h[j-k]) * [0<=j-k<n] - p[k] * h[j+k] * [0<=j+k<n]
    
    # First term: sum_k p[k] * (1-h[j-k]) where 0<=j-k<n, i.e., j-n+1 <= k <= j
    # This is the cross-correlation of p (length 2n-1) and (1-h) (length n) evaluated at (n-1-j)
    # Hmm, this is getting complicated with the different lengths.
    
    # Let me just use the direct O(n^2) method since n is small (up to ~360)
    grad = np.zeros(n)

        if k_start <= k_end:
            m_indices = j - np.arange(k_start, k_end+1)  # j-k values
            grad[j] += np.dot(p[k_start:k_end+1], one_minus_h[m_indices])

        k_start2 = 0
        k_end2 = min(2*n-2, n-1-j)
        if k_start2 <= k_end2:
            m_indices2 = j + np.arange(k_start2, k_end2+1)  # j+k values
            grad[j] -= np.dot(p[k_start2:k_end2+1], h[m_indices2])
```

```python
def compute_gradient_fast(h, tau, n_points):
    n = n_points
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    s = c * dx / tau
    M = np.max(s)
    p = np.exp(s - M)
    p /= np.sum(p)
    
    # T1[j] = convolve(1-h, p)[j] for j=0..n-1
    # T2[j] = convolve(h, p_flipped)[2n-2+j] for j=0..n-1
    # grad[j] = dx^2 * (T1[j] - T2[j])
    
    N = 1
    while N < 3*n - 3:
        N *= 2
    
    H = np.fft.fft(1-h, N)
    Hh = np.fft.fft(h, N)
    P = np.fft.fft(p, N)
    P_flipped = np.fft.fft(p[::-1], N)
    
    T1_full = np.fft.ifft(H * P).real
    T2_full = np.fft.ifft(Hh * P_flipped).real
    
    T1 = T1_full[:n]
    T2 = T2_full[2*n-2 : 2*n-2+n]
    
    return dx * dx * (T1 - T2)
```
