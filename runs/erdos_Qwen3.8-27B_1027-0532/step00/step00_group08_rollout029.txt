```python
import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def softmax_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    logits = beta * (corr - c_max)
    exp_l = np.exp(logits)
    Z = np.sum(exp_l)
    p = exp_l / Z
    val = c_max + np.log(Z) / beta
    
    # Gradient
    grad = np.zeros(n)
    shifts = np.arange(-n+1, n)  # shift values from -(n-1) to n-1
    # p[k] corresponds to shift = k-(n-1) = shifts[k]
    for idx, s in enumerate(shifts):
        w = p[idx] * dx
        if w < 1e-15:
            continue
        # Contribution: w * [I[0≤j+s<n]*(1-h[j+s]) - I[0≤j-s<n]*h[j-s]]
        # First term: for j in [max(0,-s), min(n, n-s))
        j_start1 = max(0, -s)
        j_end1 = min(n, n - s)
        if j_end1 > j_start1:
            grad[j_start1:j_end1] += w * (1.0 - h[j_start1+s:j_end1+s])
        # Second term: for j in [max(0, s), min(n, n))  where j-s in [0,n)
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)  # j-s < n means j < n+s, and j>=0
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w * h[j_start2-s:j_end2-s]
    
    return val, grad

    beta = 50.0

            return compute_c5(np.clip(x, 0, 1))

        bnds = [(0.0, 1.0)] * n

            cand = project_feasible(res.x, target_sum)
            c = compute_c5(cand)
            if c < best_c5:

        except Exception:
            pass

                    h = h_new

                h = h_new
                lr = min(lr * 1.01, 0.01)
            else:
                lr *= 0.95
    
    return (best_h, best_c5, n)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def corr_gradient(h, k, n):
    shift = k - (n - 1)
    grad = np.zeros(n)
    j = np.arange(n)
    js = j + shift
    mask1 = (js >= 0) & (js < n)
    grad[mask1] = (1.0 - h[js[mask1]])
    js2 = j - shift
    mask2 = (js2 >= 0) & (js2 < n)
    grad[mask2] -= h[js2[mask2]]
    return grad

def softmax_grad(h, p, dx, n):
    grad = np.zeros(n)
    j = np.arange(n)
    n_corr = 2 * n - 1
    for idx in range(n_corr):
        w = p[idx]
        if w < 1e-12:
            continue
        shift = idx - (n - 1)
        ww = w * dx
        js = j + shift
        mask1 = (js >= 0) & (js < n)
        grad[mask1] += ww * (1.0 - h[js[mask1]])
        js2 = j - shift
        mask2 = (js2 >= 0) & (js2 < n)
        grad[mask2] -= ww * h[js2[mask2]]
    return grad

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    n_corr = 2 * n - 1

    best_h = project_feasible(h0, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP
    if remaining() > 200:
        def obj(x):
            return compute_c5(x)
        def con_eq(x):
            return np.sum(x) - target_sum
        def con_eq_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        bnds = [(0.0, 1.0)] * n

        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
            cand = project_feasible(res.x, target_sum)
            c = compute_c5(cand)
            if c < best_c5:
                best_h, best_c5 = cand, c
        except Exception:
            pass

    # Phase 2: Softmax-smoothed gradient descent with increasing beta
    if remaining() > 120:
        h = best_h.copy()
        for beta in [30.0, 80.0, 200.0, 500.0]:
            if remaining() < 30:
                break
            lr = 0.05
            stall = 0
            for it in range(30000):
                if remaining() < 10:
                    break
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                c_max = np.max(corr)
                logits = beta * (corr - c_max)
                logits = np.clip(logits, -500, 500)
                exp_l = np.exp(logits)
                Z = np.sum(exp_l)
                p = exp_l / Z
                grad = softmax_grad(h, p, dx, n)
                h_new = project_feasible(h - lr * grad, target_sum)
                new_c5 = compute_c5(h_new)
                if new_c5 < best_c5 - 1e-13:
                    best_h, best_c5 = h_new.copy(), new_c5
                    h = h_new
                    stall = 0
                else:
                    stall += 1
                    if stall % 30 == 0:
                        lr *= 0.5
                        if lr < 1e-8:
                            break

    # Phase 3: Exact subgradient method
    if remaining() > 40:
        h = best_h.copy()
        lr = 0.003
        for it in range(80000):
            if remaining() < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            max_val = np.max(corr)
            k_max = np.where(corr >= max_val - 1e-12)[0]
            grad = np.zeros(n)
            for k in k_max:
                grad += corr_gradient(h, k, n)
            grad /= len(k_max)
            grad *= dx
            h_new = project_feasible(h - lr * grad, target_sum)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5 - 1e-13:
                best_h, best_c5 = h_new.copy(), new_c5
                h = h_new
            else:
                if it % 200 == 0:
                    pert_scale = max(0.005, 0.02 * (1 - it / 80000))
                    pert = rng.randn(n) * pert_scale
                    h_pert = project_feasible(h + pert, target_sum)
                    c_pert = compute_c5(h_pert)
                    if c_pert < best_c5:
                        best_h, best_c5 = h_pert, c_pert
                        h = h_pert
                if it % 500 == 0:
                    lr *= 0.97

    # Phase 4: Local coordinate descent
    if remaining() > 20:
        h = best_h.copy()
        improved = True
        while improved and remaining() > 20:
            improved = False
            for i in range(n):
                if remaining() < 20:
                    break
                old = h[i]
                for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                    h[i] = np.clip(old + delta, 0.0, 1.0)
                    h_proj = project_feasible(h, target_sum)
                    c = compute_c5(h_proj)
                    if c < best_c5 - 1e-13:
                        best_h, best_c5 = h_proj, c
                        h = h_proj
                        improved = True
                        break
                h[i] = h[i] if improved else old
                if not improved:
                    h = best_h.copy()
                    break

    # Final validation
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
```
