def project_onto_simplex(h, target_sum, lo=0, hi=1):
    """Project h onto {x : lo <= x_i <= hi, sum(x) = target_sum}"""
    # Use bisection on the Lagrange multiplier
    def clip_sum(theta):
        x = np.clip(h - theta, lo, hi)
        return np.sum(x)
    
    lo_theta = h.min() - hi
    hi_theta = h.max() - lo
    
    for _ in range(100):
        mid = (lo_theta + hi_theta) / 2
        s = clip_sum(mid)
        if s > target_sum:
            lo_theta = mid
        else:
            hi_theta = mid
    
    theta = (lo_theta + hi_theta) / 2
    return np.clip(h - theta, lo, hi)

    d = 2.0 / n

    if k_star is None:
        k_star = int(np.argmax(c))

    m = k_star - (n - 1)

    idx1 = j + m
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1 - h[idx1[mask1]])
    
    idx2 = j - m
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] -= h[idx2[mask2]]
