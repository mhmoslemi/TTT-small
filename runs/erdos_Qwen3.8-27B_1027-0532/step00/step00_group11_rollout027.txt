import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        step = diff / float(mask.sum())
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_max(x, temp):
    x_max = x.max()
    return temp * (x_max + np.log(np.sum(np.exp((x - x_max) / temp))))


def penalty_obj(h, dx, target, penalty):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx + penalty * (h.sum() - target) ** 2


def smooth_obj(h, dx, target, temp, penalty):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return soft_max(corr, temp) * dx + penalty * (h.sum() - target) ** 2


def binary_project(h, target):
    n = len(h)
    k = int(round(target))
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:k]] = 1.0
    return h_bin


def coordinate_descent(h, dx, target, max_iter=500):
    h = fix_sum(h, target)
    n = len(h)
    best_val = compute_c5(h, dx)
    for it in range(max_iter):
        improved = False
        for i in range(n):
            for val in [0.0, 1.0, 0.5]:
                if abs(h[i] - val) < 1e-10:
                    continue
                h_new = h.copy()
                h_new[i] = val
                h_new = fix_sum(h_new, target)
                v = compute_c5(h_new, dx)
                if v < best_val - 1e-12:
                    h = h_new
                    best_val = v
                    improved = True
                    break
        if not improved:
            break
    return h, best_val
