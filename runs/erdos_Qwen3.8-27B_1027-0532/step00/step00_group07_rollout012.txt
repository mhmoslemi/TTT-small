import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(x - t, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_single_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smoothed_obj(h, dx, n, temperature, penalty_weight):
    h_eff = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_eff, 1.0 - h_eff, mode="full")
    scaled = temperature * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    obj = (m + np.log(np.sum(exp_scaled))) / temperature
    target_sum = n / 2.0
    obj += penalty_weight * (np.sum(h_eff) - target_sum) ** 2
    return obj

def smoothed_grad(h, dx, n, temperature, penalty_weight):
    h_eff = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_eff, 1.0 - h_eff, mode="full")
    scaled = temperature * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    wsum = np.sum(exp_scaled)
    weights = exp_scaled / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += weights[k] * grad_single_lag(h_eff, n, lag) * dx
    target_sum = n / 2.0
    grad += 2.0 * penalty_weight * (np.sum(h_eff) - target_sum)
    mask = (h >= 0.0) & (h <= 1.0)
    result = grad.copy()
    result[~mask] = 0.0
    return result

def local_optimize(h_start, dx, n, target_sum, bounds, maxiter, ftol, gtol, temperature, penalty_weight):
    obj_fn = partial(smoothed_obj, dx=dx, n=n, temperature=temperature, penalty_weight=penalty_weight)
    grad_fn = partial(smoothed_grad, dx=dx, n=n, temperature=temperature, penalty_weight=penalty_weight)
    try:
        res = minimize(obj_fn, h_start, method='L-BFGS-B', jac=grad_fn,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h_start, compute_c5(h_start, dx)

def random_restart_init(n, target_sum, rng, base_h=None):
    if base_h is not None and rng.random() < 0.7:
        scale = rng.uniform(0.05, 0.5)
        h_pert = base_h + rng.randn(n) * scale
    else:
        h_pert = rng.rand(n)
    return project_feasible(h_pert, target_sum)

def simulated_annealing_step(h, dx, n, target_sum, temperature_sa, rng):
    n_accept = 0
    for _ in range(50):
        idx = rng.randint(n)
        delta = rng.uniform(-0.1, 0.1)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c5_old = compute_c5(h, dx)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_old or rng.random() < np.exp(-(c5_new - c5_old) / max(temperature_sa, 1e-8)):
            h = h_new
            n_accept += 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h0 = np.full(95, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.25:
        for temp in [200.0, 100.0, 50.0, 25.0, 10.0, 5.0]:
            if time.time() - start_time > budget_s * 0.25:
                break
            h_new, c5_new = local_optimize(best_h, dx, n, target_sum, bounds,
                                           maxiter=300, ftol=1e-14, gtol=1e-10,
                                           temperature=temp, penalty_weight=200.0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    elapsed = time.time() - start_time
    while time.time() - start_time < budget_s * 0.75:
        h_pert = random_restart_init(n, target_sum, rng, base_h=best_h)
        for temp in [50.0, 10.0, 5.0]:
            if time.time() - start_time > budget_s * 0.75:
                break
            h_new, c5_new = local_optimize(h_pert, dx, n, target_sum, bounds,
                                           maxiter=200, ftol=1e-13, gtol=1e-9,
                                           temperature=temp, penalty_weight=300.0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_pert = h_new.copy()

    elapsed = time.temperature = time.time() - start_time
    while time.time() - start_time < budget_s * 0.90:
        h_pert = best_h + rng.randn(n) * 0.03
        h_pert = project_feasible(h_pert, target_sum)
        h_new, c5_new = local_optimize(h_pert, dx, n, target_sum, bounds,
                                       maxiter=100, ftol=1e-14, gtol=1e-10,
                                       temperature=5.0, penalty_weight=500.0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        best_h = project_feasible(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
