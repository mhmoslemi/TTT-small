Let me use a hybrid approach:

- Keep only the top M weights (e.g., M = 50 or 100)

```python
def compute_obj_grad(h, alpha, top_m=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    
    max_val = np.max(vals)
    shifted = alpha * (vals - max_val)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    total_exp = np.sum(exp_vals)
    weights = exp_vals / total_exp
    
    obj = max_val + np.log(total_exp) / alpha
    
    # Get top M indices by weight
    if top_m < len(weights):
        top_idx = np.argsort(weights)[::-1][:top_m]
        top_weights = weights[top_idx]
        # Renormalize
        top_weights = top_weights / np.sum(top_weights)
    else:
        top_idx = np.arange(len(weights))
        top_weights = weights
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, wk in zip(top_idx, top_weights):
        s = idx - (n - 1)
        if s >= 0:
            # First term: (1 - h[j+s]) for j in [0, n-s-1]
            grad[:n-s] += wk * (1.0 - h[s:])
            # Second term: -h[j-s] for j in [s, n-1]
            if s > 0:
                grad[s:] += -wk * h[:-s]
            else:
                grad[:] += -wk * h[:]
        else:
            sn = -s
            # First term: (1 - h[j+s]) for j in [sn, n-1], j+s in [0, n-sn-1]
            grad[sn:] += wk * (1.0 - h[:n-sn])
            # Second term: -h[j-s] = -h[j+sn] for j in [0, n-sn-1]
            grad[:n-sn] += -wk * h[sn:]
    
    return obj, grad
```

def project_box_sum(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mu = (lo + hi) / 2.0
        x = np.clip(h - mu, 0.0, 1.0)
        if x.sum() > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_obj_grad(h, alpha, top_m=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    max_val = np.max(vals)
    shifted = alpha * (vals - max_val)
    shifted = np.clip(shifted, -400, 0)
    exp_vals = np.exp(shifted)
    total_exp = np.sum(exp_vals)
    obj = max_val + np.log(total_exp) / alpha
    weights = exp_vals / total_exp
    M = min(top_m, 2 * n - 1)
    top_idx = np.argsort(-weights)[:M]
    top_w = weights[top_idx].copy()
    sw = top_w.sum()
    if sw < 1e-15:
        top_w = weights.copy()
        top_idx = np.arange(2 * n - 1)
        M = 2 * n - 1
    else:
        top_w /= sw
    grad = np.zeros(n, dtype=np.float64)
    for i in range(M):
        wk = top_w[i]
        if wk < 1e-12:
            continue
        s = top_idx[i] - (n - 1)
