    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    exp_C = np.exp(T * (C_raw - C_max))
    p = exp_C / exp_C.sum()
    return np.correlate(p, 1.0 - 2.0 * h, mode="valid")

    C_raw = np.correlate(h, 1.0 - h, mode="full")

    exp_C = np.exp(T * (C_raw - C_max))
    p = exp_C / exp_C.sum()
    return np.correlate(p, 1.0 - 2.0 * h, mode="valid")

def subgradient_step(h, n, alpha, target_sum):
    C_raw = np.correlate(h, 1.0 - h, mode="full")
    i_star = int(np.argmax(C_raw))
    grad = np.zeros(n)
    lo = max(0, i_star - n + 1)
    hi = min(n - 1, i_star)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 1.0 - 2.0 * h[i_star - idx]
    return project_feasible(h - alpha * grad, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    start_time = time.time()
    n = 86
    dx = 2.0 / n
    target_sum = n / 2.0
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - start_time

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(n, 0.5)

    h_start = project_feasible(h_start, target_sum)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start, dx)

    T_schedule = [3, 5, 10, 20, 50, 100, 200, 500]
    for T in T_schedule:
        if elapsed() > budget_s * 0.40:
            break
        h = best_h.copy()
        current_c5 = compute_c5(h, dx)
        alpha = 4.0 / (T * n)
        stall = 0
        for it in range(8000):
            if elapsed() > budget_s * 0.40:
                break
            grad = smooth_gradient(h, T, n)
            h_new = project_feasible(h - alpha * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                stall = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                stall += 1
                alpha *= 0.8
                if stall > 200:
                    break
            alpha = max(1e-6, min(alpha, 4.0 / (T * n)))

    if elapsed() < budget_s * 0.60:
        def nm_obj(x):
            return compute_c5(project_feasible(x, target_sum), dx)
        res = minimize(nm_obj, best_h, method='Nelder-Mead',
                      options={'maxiter': 30000, 'xatol': 1e-11, 'fatol': 1e-13})
        h_nm = project_feasible(res.x, target_sum)
        c5_nm = compute_c5(h_nm, dx)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()

    for T in [200, 500, 1000, 2000]:
        if elapsed() > budget_s * 0.75:
            break
        h = best_h.copy()
        current_c5 = compute_c5(h, dx)
        alpha = 4.0 / (T * n)
        stall = 0
        for it in range(5000):
            if elapsed() > budget_s * 0.75:
                break
            grad = smooth_gradient(h, T, n)
            h_new = project_feasible(h - alpha * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                stall = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                stall += 1
                alpha *= 0.9
                if stall > 300:
                    break
            alpha = max(1e-7, min(alpha, 4.0 / (T * n)))

    if elapsed() < budget_s * 0.90:
        h = best_h.copy()
        alpha = 0.03
        stall = 0
        for it in range(20000):
            if elapsed() > budget_s * 0.90:
                break
            h = subgradient_step(h, n, alpha, target_sum)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 500:
                alpha *= 0.7
                stall = 0
            alpha = max(1e-5, min(alpha, 0.03))

    if elapsed() < budget_s * 0.98:
        for restart in range(3):
            if elapsed() > budget_s * 0.98:
                break
            h_pert = best_h + rng.normal(0, 0.05, n)
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)
            h_cur = h_pert.copy() if c5_pert < best_c5 else best_h.copy()
            cur_c5 = compute_c5(h_cur, dx)
            T = 30
            alpha = 4.0 / (T * n)
            for it in range(3000):
                if elapsed() > budget_s * 0.98:
                    break
                grad = smooth_gradient(h_cur, T, n)
                h_new = project_feasible(h_cur - alpha * grad, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < cur_c5:
                    h_cur = h_new
                    cur_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_cur.copy()
                else:
                    alpha *= 0.9
                alpha = max(1e-6, min(alpha, 4.0 / (T * n)))

    final_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    else:
        best_h = final_h

    return best_h, best_c5, n
