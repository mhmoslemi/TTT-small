    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_overlap(h, k_star):
    n = len(h)
    s = k_star - (n - 1)
    grad = np.zeros(n)
    # h[j] appears as h[i] when i=j, contributing (1-h[j+s]) if j+s valid
    # h[j] appears as h[i+s] when i=j-s, contributing -h[j-s] if j-s valid
    for j in range(n):
        if 0 <= j + s < n:
            grad[j] += (1.0 - h[j + s])
        if 0 <= j - s < n:
            grad[j] -= h[j - s]
    return grad

def repair(h, target_sum):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s != target_sum:
        # Need to adjust
        # Simple approach: add (target_sum - s) / n to all, then re-clip
        # Better: iteratively adjust
        diff = target_sum - s
        if diff > 0:
            # Need to increase some values
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
        else:
            # Need to decrease some values
            mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
    # Fine adjustment
    diff = target_sum - np.sum(h)

        else:
            mask = h > 0.0
        if np.sum(mask) > 0:

            h = np.clip(h, 0, 1)
    return h

def gradient_component(h, m):
    """Gradient of C[m] = np.correlate(h, 1-h, 'full')[m] with respect to h."""
    n = len(h)
    s = m - (n - 1)
    grad = np.zeros(n)
    
    # First term: (1-h[j+s]) for j in [max(0,-s), min(n-1, n-1-s)]
    j_min1 = max(0, -s)
    j_max1 = min(n-1, n-1-s)
    if j_min1 <= j_max1:
        if s >= 0:
            grad[j_min1:j_max1+1] += 1.0 - h[j_min1+s:j_max1+s+1]
        else:
            grad[j_min1:j_max1+1] += 1.0 - h[j_min1+s:j_max1+s+1]
    
    # Second term: -h[j-s] for j in [max(0,s), min(n-1, n-1+s)]
    j_min2 = max(0, s)
    j_max2 = min(n-1, n-1+s)
    if j_min2 <= j_max2:
        grad[j_min2:j_max2+1] -= h[j_min2-s:j_max2-s+1]
    
    return grad
