import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_gradient(h, dx, tau=50.0):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Soft-max approximation
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(tau * shifted)
    weights /= np.sum(weights)
    
    # Smooth objective
    smooth_obj = (max_corr + np.log(np.sum(weights)) / tau) * dx
    
    # Gradient of smooth objective
    # ∂c[k]/∂h[m] = (1 - h[m - d]) - h[m + d] where d = k - (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        # Compute (1 - h[m - d]) - h[m + d] for all m
        # h[m-d] is h shifted right by d (or 0 if out of bounds)
        # h[m+d] is h shifted left by d (or 0 if out of bounds)
        
        one_minus_hshifted = np.ones(n)
        hshifted = np.zeros(n)
        
        if d >= 0:
            # h[m-d]: valid for m >= d, so hshifted[m] = h[m-d] for m >= d
            hshifted[d:] = h[:n-d]
            # h[m+d]: valid for m+d < n, so h[m+d] for m < n-d
            # This is h[d:][m] = h[m+d] for m < n-d
            hshifted_neg = np.zeros(n)
            hshifted_neg[:n-d] = h[d:]
        else:
            # d < 0, let d' = -d > 0
            dp = -d
            # h[m-d] = h[m+dp]: valid for m+dp < n, i.e., m < n-dp
            hshifted[:n-dp] = h[dp:]
            # h[m+d] = h[m-dp]: valid for m >= dp
            hshifted_neg = np.zeros(n)
            hshifted_neg[dp:] = h[:n-dp]
        
        grad += weights[k] * ((1 - hshifted) - hshifted_neg)
    
    grad *= dx
    return smooth_obj, grad

def project_onto_constraint(h, target_sum):
    # Project h onto {x : sum(x) = target_sum, 0 <= x <= 1}
    # Using bisection on lambda
    lo = -10.0
    hi = 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 80
    dx = 2.0 / n
    target_sum = n / 2.0  # = 40
    
    # Get initial solution
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
        dx = 2.0 / n
        target_sum = n / 2.0
    else:
        h = np.random.uniform(0, 1, n)
        h = project_onto_constraint(h, target_sum)
    
    # Evaluate initial
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with soft-max and momentum
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.5
    tau = 50.0
    
    no_improve_count = 0
    plateau_threshold = 200
    
    while time.time() - start_time < budget_s - 5:
        # Compute gradient
        obj, grad = compute_c5_and_gradient(h, dx, tau)
        
        # Update momentum
        momentum = beta * momentum + (1 - beta) * grad
        direction = momentum
        
        # Line search
        alpha = lr
        for _ in range(20):
            h_trial = h - alpha * direction
            h_trial = project_onto_constraint(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                h = h_trial
                best_h = h.copy()
                best_c5 = c5_trial
                no_improve_count = 0
                break
            alpha *= 0.5
        
        else:
            no_improve_count += 1
        
        # If stuck, perturb
        if no_improve_count > plateau_threshold:
            # Random perturbation
            perturbation = np.random.normal(0, 0.1, n)
            h = project_onto_constraint(h + perturbation, target_sum)
            c5_trial = compute_c5(h, dx)
            if c5_trial < best_c5:
                best_h = h.copy()
                best_c5 = c5_trial
            no_improve_count = 0
        
        # Adapt learning rate
        if no_improve_count > 50:
            lr *= 0.95
        elif no_improve_count == 0:
            lr = min(lr * 1.05, 2.0)
    
    return best_h, best_c5, n
