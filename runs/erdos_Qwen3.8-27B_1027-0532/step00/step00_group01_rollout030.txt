def make_feasible(h, n):
    """Project h to satisfy sum(h) = n/2, 0 <= h <= 1"""
    # First clip
    h = np.clip(h, 0, 1)
    # Then adjust sum
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to add: add to entries that are < 1
        addable = 1.0 - h
        total_addable = np.sum(addable)
        if total_addable > 0:
            h += diff / total_addable * addable

        removable = h
        total_removable = np.sum(removable)
        if total_removable > 0:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)

    grad = np.zeros(n)
    for p in range(n):
        # p as shifted index (m+lag = p, m = p-lag)
        if 0 <= p - lag < n:
            grad[p] += (1.0 - h[p - lag])
        # p as base index (m = p, m+lag = p+lag)
        if 0 <= p + lag < n:
            grad[p] -= h[p + lag]

    d = -grad
    
    # Project onto sum=0 hyperplane
    d = d - np.mean(d)
    
    # Also consider other active lags to avoid increasing them
    # (optional: weighted average of gradients for top few lags)
    
    # Normalize direction
    norm = np.linalg.norm(d)
    if norm < 1e-10:
        return h, max_val * dx
    
    d = d / norm

    new_h = h + alpha * d
