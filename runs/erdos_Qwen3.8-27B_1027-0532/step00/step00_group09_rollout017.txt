import time

    m = np.max(cor)

    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def smooth_gradient(h, beta, dx):
    n = len(h)
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cor)
    shifted = cor - m

        b_lo = max(i_lo, -k)

Wait, there's an issue with the flip deltas. When I flip h[i] from 1 to 0 AND h[j] from 0 to 1, the deltas interact! The delta D[i] assumes h[j] is still 0 (or 1 for D[j]). But after flipping both, the intermediate state is different.

Let me reconsider. The delta D[i, r] is computed assuming only h[i] changes. If I flip both i and j simultaneously, the total change is:
Δc_r = D[i, r] + D[j, r] (where D[j, r] is the delta for flipping j from 0 to 1, which is -D[j, r] as computed for 1→0)

Wait, I defined D[i, r] as the change when h[i] flips from 1 to 0. For h[j] flipping from 0 to 1, the change is the negative: -D[j, r] (if D was computed for the 1→0 direction at position j).

However, if i and j are "far apart" (|i-j| > some threshold), then the deltas don't interact and the total change is D[i] - D[j] (where D[j] is the 1→0 delta at j, and we negate for 0→1).

If i and j are close, there could be interaction. Let me check:

D[i, r] involves h[i+k] and h[i-k]. D[j, r] involves h[j+k] and h[j-k].

The interaction would occur if i+k = j or j+k = i or i-k = j or j-k = i, etc. This means |i-j| = |k| for some k in the relevant range.

For the total delta when flipping both i (1→0) and j (0→1):
The actual new c_r = sum_{valid} h_new[a+k]*(1-h_new[a])

The change is:
- For terms involving h[i]: was h[i]*(1-h[a]) or h[a+k]*h[i] type terms
- For terms involving h[j]: similar

If i and j don't interact (no index in the correlation sum involves both), then the deltas add.

If they do interact, we have a second-order term.

def binary_local_search_simple(h_bin, n, max_rounds=200, start_time=None, timeout=None):
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    cor = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(cor)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for round_idx in range(max_rounds):
        if start_time is not None and timeout is not None:
            if time.time() - start_time > timeout:
                break
        best_improvement = 0.0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Try the flip
                h[i] = 0.0
                h[j] = 1.0
                new_cor = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(new_cor)
                improvement = current_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pair = (i, j, new_cor.copy())
                # Revert
                h[i] = 1.0
                h[j] = 0.0
        
        if best_pair is not None:
            i, j, new_cor = best_pair
            h[i] = 0.0
            h[j] = 1.0
            cor = new_cor
            current_max = np.max(cor)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            break
    
    return h, current_max * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cor) * dx)

def smooth_objective(h, beta, dx):
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cor)
    shifted = cor - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def smooth_gradient(h, beta, dx):
    n = len(h)
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cor)
    shifted = cor - m
    ew = np.exp(beta * shifted)
    sw = ew.sum()
    w = ew / sw
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        wr = w[r] * dx
        if wr < 1e-14:
            continue
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo > i_hi:
            continue
        a_lo = max(0, i_lo + k)
        a_hi = min(n - 1, i_hi + k)
        if a_lo <= a_hi:
            grad[a_lo:a_hi + 1] += wr * (1.0 - h[a_lo - k:a_hi - k + 1])
        b_lo = max(i_lo, -k)
        b_hi = min(i_hi, n - 1 - k)
        if b_lo <= b_hi:
            grad[b_lo:b_hi + 1] -= wr * h[b_lo + k:b_hi + k + 1]
    return grad

    hi = float(np.max(h)) + 1.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def optimize_continuous(h0, n, max_time):
    """Optimize using L-BFGS-B with smooth max and penalty for sum constraint."""
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n)
    
    penalty = 100.0
    
    betas = [5, 10, 20, 40, 80, 150]
    max_iter_per_beta = 300
    
    for beta in betas:
        if time.time() - max_time[0] > max_time[1]:
            break
        
        def obj(h):
            return smooth_objective(h, beta, dx) + penalty * (np.sum(h) - target_sum) ** 2
        
        def grad(h):
            return smooth_gradient(h, beta, dx) + 2.0 * penalty * (np.sum(h) - target_sum) * np.ones(n)
        
        result = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'gtol': 1e-8})
        
        h_new = result.x
        # Project to satisfy sum constraint
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new, n)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5
