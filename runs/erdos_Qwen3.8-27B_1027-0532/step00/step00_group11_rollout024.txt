import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def smooth_max_val(corr, t):
    shifted = t * corr
    m = np.max(shifted)
    return (m + np.log(np.sum(np.exp(shifted - m)))) / t

def obj_grad(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    
    # Smooth max value
    shifted = t * corr
    m = np.max(shifted)
    sm_val = (m + np.log(np.sum(np.exp(shifted - m)))) / t
    
    # Softmax weights
    weights = np.exp(shifted - m)
    weights /= np.sum(weights)
    
    # Gradient: for each m in 0..n-1
    # d/dh[m] of result[j] = (1-h[m-lag_j])*I(0<=m-lag_j<n) - h[m+lag_j]*I(0<=m+lag_j<n)
    # where lag_j = j - (n-1)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        lag = j - (n - 1)
        w = weights[j] * dx
        if w < 1e-15:
            continue
        for i in range(n):
            # First term: (1-h[i-lag]) if 0<=i-lag<n
            if 0 <= i - lag < n:
                grad[i] += w * (1 - x[i - lag])
            # Second term: -h[i+lag] if 0<=i+lag<n
            if 0 <= i + lag < n:
                grad[i] -= w * x[i + lag]
    
    return sm_val, grad

    target_sum = n / 2.0
    corr = np.correlate(x, 1 - x, mode='full') * dx
    shifted = t * corr
    m = np.max(shifted)
    sm_val = (m + np.log(np.sum(np.exp(shifted - m)))) / t
    s = x.sum()
    sm_val += penalty * (s - target_sum) ** 2
    weights = np.exp(shifted - m)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j] * dx
        if w < 1e-15:
            continue
        lag = j - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - x[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * x[lo2 + lag:hi2 + lag]
    grad += 2.0 * penalty * (s - target_sum)
    return sm_val, grad


def repair_point(x, n, target_sum):
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if s > 1e-10:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        n_adjust = int(abs(diff)) + 1
        if diff > 0:
            idx = np.argsort(-x)[:n_adjust]
            for i in idx:
                x[i] = min(1.0, x[i] + diff / n_adjust)
        else:
            idx = np.argsort(x)[:n_adjust]
            for i in idx:
                x[i] = max(0.0, x[i] + diff / n_adjust)
    return x


def binary_local_search(h, n, rng, max_swaps=20000):
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        one_pos = np.where(h_bin == 1)[0]
        rng.shuffle(one_pos)
        for i in range(n_ones - target):
            h_bin[one_pos[i]] = 0
    elif n_ones < target:
        zero_pos = np.where(h_bin == 0)[0]
        rng.shuffle(zero_pos)
        for i in range(target - n_ones):
            h_bin[zero_pos[i]] = 1
    best_c5 = compute_c5(h_bin)
    for _ in range(max_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5 = compute_c5(h_bin)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, best_c5


def make_alternating(n, rng):
    h = np.zeros(n)
    pattern = np.ones(n // 2)
    pattern_alt = np.zeros(n // 2)
    h[0:n:2] = pattern
    h[1:n:2] = pattern_alt
    if n % 2 == 0:
        pass
    else:
        h[-1] = 0.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    n_points = 86
    target_sum = n_points / 2.0

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_init = np.full(n_points, 0.5)

    if h_init.shape[0] != n_points:
        h_init = np.full(n_points, 0.5)

    h_init = repair_point(h_init, n_points, target_sum)

    start_time = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n_points
    penalty = 5000.0

    t_values = [30, 60, 100, 200]

    for t in t_values:
        if time.time() - start_time > budget_s * 0.45:
            break
        for restart in range(5):
            if time.time() - start_time > budget_s * 0.55:
                break
            if restart == 0:
                x0 = best_h.copy()
            elif restart == 1:
                x0 = best_h + 0.08 * rng.randn(n_points)
                x0 = repair_point(x0, n_points, target_sum)
            elif restart == 2:
                mask = rng.rand(n_points) < 0.4
                x0 = best_h.copy()
                x0[mask] = rng.rand(int(mask.sum()))
                x0 = repair_point(x0, n_points, target_sum)
            elif restart == 3:
                x0 = (best_h > 0.5).astype(float)
                perm = rng.permutation(n_points)
                x0[perm[:10]] = 0.5
                x0 = repair_point(x0, n_points, target_sum)
            else:
                x0 = make_alternating(n_points, rng)
                x0 = repair_point(x0, n_points, target_sum)

            try:
                res = minimize(
                    obj_with_grad, x0, args=(t, penalty),
                    method='L-BFGS-B', jac=True,
                    bounds=bounds,
                    options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-9, 'iprint': -1}
                )
                if res.x is not None and not np.any(np.isnan(res.x)):
                    x_new = repair_point(res.x, n_points, target_sum)
                    c5_new = compute_c5(x_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = x_new.copy()
            except Exception:
                pass

    # Phase 2: Binary local search
    if time.time() - start_time < budget_s * 0.75:
        for it in range(5):
            if time.time() - start_time > budget_s * 0.80:
                break
            h_bin, c5_bin = binary_local_search(best_h, n_points, rng, max_swaps=30000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                try:
                    res = minimize(
                        obj_with_grad, best_h, args=(150, penalty),
                        method='L-BFGS-B', jac=True,
                        bounds=bounds,
                        options={'maxiter': 1000, 'ftol': 1e-14, 'gtol': 1e-9, 'iprint': -1}
                    )
                    if res.x is not None and not np.any(np.isnan(res.x)):
                        x_new = repair_point(res.x, n_points, target_sum)
                        c5_new = compute_c5(x_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = x_new.copy()
                except Exception:
                    pass
            else:
                h_bin2, c5_bin2 = binary_local_search(h_init, n_points, rng, max_swaps=20000)
                if c5_bin2 < best_c5:
                    best_c5 = c5_bin2
                    best_h = h_bin2.copy()

    # Phase 3: High-t refinement
    if time.time() - start_time < budget_s * 0.90:
        for t_refine in [300, 500, 800]:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                res = minimize(
                    obj_with_grad, best_h, args=(t_refine, penalty),
                    method='L-BFGS-B', jac=True,
                    bounds=bounds,
                    options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1}
                )
                if res.x is not None and not np.any(np.isnan(res.x)):
                    x_new = repair_point(res.x, n_points, target_sum)
                    c5_new = compute_c5(x_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = x_new.copy()
            except Exception:
                pass

    # Phase 4: Try different n_points if time remains
    if time.time() - start_time < budget_s * 0.70:
        for n_try in [100, 120, 172]:
            if time.time() - start_time > budget_s * 0.85:
                break
            target_try = n_try / 2.0
            dx_try = 2.0 / n_try
            # Interpolate from best_h to new length
            old_idx = np.linspace(0, 2.0, n_points, endpoint=False) + dx_try / 2
            new_idx = np.linspace(0, 2.0, n_try, endpoint=False) + 2.0 / n_try / 2
            h_interp = np.interp(new_idx, old_idx, best_h)
            h_interp = repair_point(h_interp, n_try, target_try)
            bounds_try = [(0.0, 1.0)] * n_try
            for t_try in [50, 100]:
                if time.time() - start_time > budget_s * 0.93:
                    break
                try:
                    res = minimize(
                        obj_with_grad, h_interp, args=(t_try, penalty),
                        method='L-BFGS-B', jac=True,
                        bounds=bounds_try,
                        options={'maxiter': 1000, 'ftol': 1e-13, 'gtol': 1e-8, 'iprint': -1}
                    )
                    if res.x is not None and not np.any(np.isnan(res.x)):
                        x_new = repair_point(res.x, n_try, target_try)
                        c5_new = compute_c5(x_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = x_new
                            n_points = n_try
                except Exception:
                    pass

    best_h = repair_point(best_h, len(best_h), len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
