import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, beta, penalty_w):
    h_c = np.clip(h, 0.0, 1.0)
    n = len(h_c)
    target_sum = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h_c, 1.0 - h_c, mode='full') * dx
    m = corr.max()
    smooth_max = m + np.log(np.sum(np.exp((corr - m) * beta))) / beta
    sum_pen = penalty_w * (h_c.sum() - target_sum) ** 2
    return smooth_max + sum_pen

def solve_lp_fix_g(g, n, target_sum):
    t = cp.Variable()
    h_var = cp.Variable(n)
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for lag in range(-(n - 1), n):
        j_min = max(0, -lag)
        j_max = n - 1 - max(0, lag)
        if j_min > j_max:
            continue
        length = j_max - j_min + 1
        start = j_min + lag
        coeffs = g[start:start + length]
        row = np.zeros(n)
        row[j_min:j_max + 1] = coeffs
        cons.append(row @ h_var <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['CLARABEL', 'ECOS', 'SOPHIA', 'GLPK']:
        try:
            prob.solve(solver=solver, verbose=False)
            if h_var.value is not None:
                return h_var.value
        except Exception:
            continue
    return None

def solve_lp_fix_h(h, n, target_sum):
    t = cp.Variable()
    g_var = cp.Variable(n)
    cons = [g_var >= 0, g_var <= 1, cp.sum(g_var) == target_sum]
    for lag in range(-(n - 1), n):
        i_min = max(0, lag)
        i_max = n - 1 - max(0, -lag)
        if i_min > i_max:
            continue
        length = i_max - i_min + 1
        start = i_min - lag
        coeffs = h[start:start + length]
        row = np.zeros(n)
        row[i_min:i_max + 1] = coeffs
        cons.append(row @ g_var <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['CLARABEL', 'ECOS', 'SOPHIA', 'GLPK']:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None:
                return g_var.value
        except Exception:
            continue
    return None

def alternating_lp(h_init, n_iter=15):
    n = len(h_init)
    target_sum = n / 2.0
    h = np.array(h_init, dtype=float).copy()
    for _ in range(n_iter):
        g = 1.0 - h
        h_new = solve_lp_fix_g(g, n, target_sum)
        if h_new is not None:
            h = project_feasible(h_new, target_sum)
        g_new = solve_lp_fix_h(h, n, target_sum)
        if g_new is not None:
            h = project_feasible(1.0 - g_new, target_sum)
    return h

def lbbgbs_refine(h_init, betas, target_sum, n, max_iter=150):
    best_h = np.array(h_init, dtype=float).copy()
    best_c5 = eval_c5(best_h)
    for beta in betas:
        for attempt in range(3):
            if attempt == 0:
                x0 = best_h.copy()
            else:
                x0 = best_h + np.random.randn(n) * 0.03
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(smooth_objective, x0, args=(beta, 80.0),
                               method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
                h_new = project_feasible(res.x, target_sum)
                c5_new = eval_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                continue
    return best_h, best_c5

def coordinate_descent(h_init, target_sum, n, n_sweeps=50):
    h = np.array(h_init, dtype=float).copy()
    dx = 2.0 / n
    best_c5 = eval_c5(h)
    for sweep in range(n_sweps):
        improved = False
        for i in range(n):
            old_val = h[i]
            for trial in [0.0, 1.0, old_val + 0.05, old_val - 0.05, old_val + 0.01, old_val - 0.01]:
                if trial < -1e-9 or trial > 1 + 1e-9:
                    continue
                trial = np.clip(trial, 0.0, 1.0)
                if abs(trial - old_val) < 1e-12:
                    continue
                h[i] = trial
                diff = target_sum - h.sum()
                if abs(diff) < 1e-10:
                    c5_trial = eval_c5(h)
                else:
                    h[i] = old_val
                    continue
                if c5_trial < best_c5 - 1e-10:
                    best_c5 = c5_trial
                    improved = True
                else:
                    h[i] = old_val
                    break
            else:
                h[i] = old_val
        if not improved:
            break
    return h, best_c5

def random_feasible(n, target_sum, rng):
    h = rng.random(n)
    h = project_feasible(h, target_sum)
    return h
