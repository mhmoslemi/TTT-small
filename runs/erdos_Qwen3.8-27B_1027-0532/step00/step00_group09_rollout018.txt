from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_objective(h, alpha=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = corr * dx
    max_val = np.max(scaled)
    if max_val == 0:
        return 0.0
    return (max_val + np.log(np.sum(np.exp(scaled - max_val))) / alpha)

def make_binary(h, n):
    """Convert to binary solution with exactly n/2 ones"""
    # Sort by value, keep top n/2 as 1
    threshold_idx = n // 2
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[sorted_idx[:threshold_idx]] = 1.0
    return binary

        return budget_s - (time.time() - start_time)

        h0 = np.array(initial_h_values, dtype=float)

            return smooth_objective(x, alpha=alpha)
        
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})
            score = exact_objective(res.x)
