                h = best_h + rng.randn(n) * 0.005
                h = project_onto_polytope(h, target)
                step = step0
                beta = beta0
                stuck = 0
    c5_final = compute_c5(best_h, dx)
    return best_h, min(best_c5, c5_final)


def interp_h(h_old, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(xn, xo, h_old)


def cos_construction(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    return project_onto_polytope(np.clip(h, 0.0, 1.0), n * 0.5)


def block_construction(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = np.zeros(n)
    h[(x < 0.5) | (x >= 1.5)] = 1.0
    return project_onto_polytope(h, n * 0.5)


def three_block_construction(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = np.zeros(n)
    h[(x < 1.0 / 3.0) | ((x >= 0.5) & (x < 5.0 / 6.0)) | (x >= 11.0 / 12.0)] = 1.0
    return project_onto_polytope(h, n * 0.5)


def rand_construction(n, seed=0):
    rng = np.random.RandomState(seed)
    return project_onto_polytope(rng.rand(n), n * 0.5)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = 1e9
    best_n = 0

    def time_left():
        return budget_s - (time.time() - t_start) - 10.0

    n0 = len(initial_h_values)
    h0 = initial_h_values.astype(np.float64).copy()
    h0 = project_onto_polytope(h0, n0 * 0.5)

    # Phase 1: Optimize at initial resolution
    tl = min(250.0, time_left() * 0.3)
    if tl > 30:
        h, c5 = optimize_pgd(h0, n0, tl, seed=seed)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n0

    # Phase 2: Scale up resolution
    for mult in [2, 4, 8]:
        nl = mult * n0 + (mult - 1)
        if nl > 1023 or time_left() < 120:
            break
        tl = min(250.0, time_left() * 0.35)
        if tl < 60:
            break
        hs = interp_h(best_h, best_n, nl)
        h, c5 = optimize_pgd(hs, nl, tl, seed=seed + mult * 10)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, nl

    # Phase 3: Alternative starting points at best_n
    nb = best_n
    if time_left() > 100:
        starts = [
            cos_construction(nb),
            block_construction(nb),
            three_block_construction(nb),
            rand_construction(nb, seed + 1),
            rand_construction(nb, seed + 2),
        ]
        t_each = time_left() * 0.55 / len(starts)
        if t_each > 25:
            for i, hs in enumerate(starts):
                if time_left() - t_each < 40:
                    break
                h, c5 = optimize_pgd(hs, nb, t_each, seed=seed + 300 + i)
                if c5 < best_c5:
                    best_h, best_c5, best_n = h, c5, nb

    # Phase 4: Final polish with higher beta
    if time_left() > 60:
        tl = max(time_left() - 15, 30)
        h, c5 = optimize_pgd(best_h, best_n, tl, seed=seed + 999, beta0=300.0, step0=0.01)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, best_n

    dx = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, best_n
