def c5_value(h):
    n = len(h)

def make_idx(n):
    a = np.arange(n)
    return a[None, :] - a[:, None] + (n - 1)

def smooth_grad(h, n, temp, idx):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    z = temp * c
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    M = w[idx]
