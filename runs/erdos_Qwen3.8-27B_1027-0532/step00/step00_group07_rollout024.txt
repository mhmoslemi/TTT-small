    """Project x onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(x + mid, 0, 1)
        s = np.sum(projected)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2, 0, 1)

def project_onto_feasible(x, n):

def compute_c5_and_gradient(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    m = np.argmax(corr)
    # m corresponds to shift m - (n-1)
    shift = m - (n - 1)
    # Gradient of c[shift] with respect to h
    # c[shift] = sum_k h[k+shift] * (1-h[k]) for valid k

            # -h[k] appears as -h[j] where j = k
            # Contribution to grad[j] from this term: -h[k+shift]
            grad[k] -= h[k + shift]
    return corr[m] * dx, grad * dx

    # Current correlations
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Compute delta for each shift

    # Phase 2: Simulated annealing
    if elapsed() < total_budget * 0.45:
        h2 = best_h.copy()
        A2 = build_A(h2, n)
        corr2 = np.correlate(h2, 1.0 - h2, mode="full")
        cm2 = float(np.max(corr2))
        t_limit2 = start_time + total_budget * 0.55
        best_h2, best_max2, corr2, A2 = simulated_annealing(
            h2, n, A2, corr2, cm2, 5000000, t_limit2, seed_offset=seed + 1
        )
        if best_max2 * (2.0 / n) < best_c5:
            best_c5 = best_max2 * (2.0 / n)
            best_h = h2

    # Phase 3: Local search again after SA
    if elapsed() < total_budget * 0.7:
        h3 = best_h.copy()
        n3 = len(h3)
        n_shifts3 = 2 * n3 - 1
        A3 = build_A(h3, n3)
        corr3 = np.correlate(h3, 1.0 - h3, mode="full")
        cm3 = float(np.max(corr3))
        t_limit3 = start_time + total_budget * 0.8
        best_h3, best_max3, _, _, _ = local_search_pair_swaps(
            h3, n3, A3, corr3, cm3, 20000000, t_limit3, seed_offset=seed + 2
        )
        if best_max3 * (2.0 / n3) < best_c5:
            best_c5 = best_max3 * (2.0 / n3)
            best_h = h3

    # Phase 4: Try different resolutions
    if elapsed() < total_budget * 0.9:
        current_n = len(best_h)
        for n_new in (int(current_n * 1.5), int(current_n * 2)):
            if n_new <= 10 or n_new > 511:
                continue
            if elapsed() > total_budget * 0.88:
                break
            old_x = np.linspace(0, 2, current_n, endpoint=False) + 1.0 / current_n
            new_x = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
            h_new = np.interp(new_x, old_x, best_h)
            h_new = project_onto_feasible(h_new, n_new)
            n_sh_new = 2 * n_new - 1
            A_new = build_A(h_new, n_new)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            cm_new = float(np.max(corr_new))
            t_lim = start_time + total_budget * 0.92
            bh, bm, _, _, _ = local_search_pair_swaps(
                h_new, n_new, A_new, corr_new, cm_new, 10000000, t_lim, seed_offset=seed + 10
            )
            c5_new = bm * (2.0 / n_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = bh

    # Final: ensure feasibility and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
