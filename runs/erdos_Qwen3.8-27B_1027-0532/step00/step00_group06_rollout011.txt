def apply_swap(h, i, j, corrs, n):
    """Apply swap of positions i (1->0) and j (0->1), update corrs in place."""
    # h[i] was 1, h[j] was 0
    # After swap: h[i] = 0, h[j] = 1
    for m in range(1, n):
        delta = 0.0
        # i as first factor: i <= n-1-m means m <= n-1-i
        if m <= n - 1 - i:
            delta -= (1 - h[i + m])  # h[i] goes 1->0, so term goes from (1-h[i+m]) to 0
        # i as second factor: i >= m means m <= i, k = i-m
        if m <= i:

        if m <= j:
            delta -= h[j - m]  # (1-h[j]) goes 1->0, term goes from h[j-m] to 0

    h[i] = 0
    h[j] = 1
