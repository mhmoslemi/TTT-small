        alt_starts.append(np.clip(0.5 + 0.4 * np.sin(np.pi * x), 0, 1))
        alt_starts.append(np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x), 0, 1))
        step_fn = np.zeros(n)
        step_fn[:n // 2] = 1.0
        step_fn[n // 2:] = 0.0
        alt_starts.append(step_fn.copy())
        for a_h in alt_starts:
            if time_left() < 60:
                break
            a_h = project_onto_feasible(a_h.copy(), target_sum)
            c5_a = compute_c5(a_h)
            if c5_a < best_c5:
                a_h, c5_a = optimize_h(a_h, target_sum, T, 60)
                if c5_a < best_c5:
                    best_h = a_h
                    best_c5 = c5_a

    if time_left() > 100:
        for n_try in [n + 13, n + 26, n + 39, n + 52]:
            if time_left() < 80:
                break
            if n_try > 250:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            ts_new = n_try / 2.0
            h_interp = project_onto_feasible(h_interp, ts_new)
            c5_interp = compute_c5(h_interp)
            opts_time = max(30, time_left() * 0.3)
            h_opt, c5_opt = optimize_h(h_interp, ts_new, 0.003, opts_time)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                n = len(best_h)
                target_sum = n / 2.0

    if time_left() > 30:
        best_h, best_c5 = optimize_h(best_h.copy(), target_sum, 0.002, time_left() - 15)

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_onto_feasible(best_h, ts_final)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
