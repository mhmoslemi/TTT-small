        # Need to flip some 1s to 0s
        ones_idx = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[ones_idx])

        n_tries = min(5000, len(zeros) * len(ones))

        best_swap = None

            j = ones[rng.randint(len(ones))]

            best_h[best_swap[1]] = 0

def project_feasible(h, target_sum):

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with moderate alpha to get a good continuous solution
    if tleft() > 300:
        h_opt = slsqp_optimize(best_h, n, target,
                               [100, 500, 2000],
                               maxiter=5000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=250)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 2: Projected gradient with Nesterov acceleration
    if tleft() > 300:
        h_pg, sc_pg = projected_gradient(best_h, n, target,
                                         time_limit=budget_s, start_time=start_time,
                                         deadline_offset=250)
        if sc_pg < best_score:
            best_score = sc_pg
            best_h = h_pg

    # Phase 3: High-alpha SLSQP (approximately solves exact minimax)
    if tleft() > 200:
        h_opt = slsqp_optimize(best_h, n, target,
                               [5000, 20000, 50000, 100000],
                               maxiter=5000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=150)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 4: Binary local search
    if tleft() > 150:
        h_bin, sc_bin = fast_binary_swap_search(best_h, n, target,
                                                max_rounds=40,
                                                time_limit=budget_s,
                                                start_time=start_time,
                                                deadline_offset=100,
                                                seed=seed + 1)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 5: More SLSQP polish after binary search
    if tleft() > 150:
        h_opt = slsqp_optimize(best_h, n, target,
                               [10000, 50000, 100000],
                               maxiter=5000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=100)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 6: Diversified restarts
    restart_configs = [(0.1, 1000), (0.2, 5000), (0.3, 10000), (0.5, 50000), (0.8, 100000)]
    for noise_level, alpha_main in restart_configs:
        if tleft() < 120:
            break
        h_pert = best_h + rng.randn(n) * noise_level
        h_pert = project_feasible(h_pert, target)
        alphas = [max(100, alpha_main // 10), alpha_main, alpha_main * 5]
        h_opt = slsqp_optimize(h_pert, n, target,
                               alphas,
                               maxiter=3000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=80)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

        if tleft() > 80:
            h_bin, sc_bin = fast_binary_swap_search(h_opt, n, target,
                                                    max_rounds=20,
                                                    time_limit=budget_s,
                                                    start_time=start_time,
                                                    deadline_offset=50,
                                                    seed=seed + 100 + int(noise_level * 1000))
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin

    # Phase 7: Try different resolutions
    for n_try in [500, 600, 300]:
        if tleft() < 200:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        h_opt = slsqp_optimize(h_interp, n_try, target_try,
                               [500, 5000, 20000, 50000],
                               maxiter=4000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=100)
        sc = compute_c5(h_opt)

        if tleft() > 100:
            h_bin, sc_bin = fast_binary_swap_search(h_opt, n_try, target_try,
                                                    max_rounds=25,
                                                    time_limit=budget_s,
                                                    start_time=start_time,
                                                    deadline_offset=60,
                                                    seed=seed + 500 + n_try)
            if sc_bin < sc:
                sc = sc_bin
                h_opt = h_bin

        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 8: Final polish at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_opt = slsqp_optimize(best_h, n_use, target_use,
                               [50000, 100000, 200000, 500000],
                               maxiter=5000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=20)
        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Final feasibility
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-10:
        h_final = project_feasible(h_final, n_use / 2.0)
        final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
