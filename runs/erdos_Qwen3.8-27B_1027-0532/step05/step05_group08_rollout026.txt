        h_new = h_cur.copy()
        h_new[idx] += delta
        h_new = project_feasible(h_new, target)

            if sc_new < best_score:

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_lag(h, k):
    """Gradient of correlation at lag k with respect to h"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    g *= dx
    return g

def logse_value_grad(h, alpha, active_idx=None):
    """Compute LogSE value and gradient"""
    n = len(h)
    corr = compute_corr(h)
    if active_idx is None:
        c_arr = corr
    else:
        c_arr = corr[active_idx]
    c_max = float(np.max(c_arr))
    shifted = np.clip(alpha * (c_arr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    val = c_max + float(np.log(exp_vals.sum())) / alpha
    
    w = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    
    if active_idx is None:
        idx_list = np.arange(len(corr))
    else:
        idx_list = active_idx
    
    for wk, m in zip(w, idx_list):
        if wk < 1e-15:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return val, g

def adam_optimize_logse(h, n, target, alpha, top_k, max_steps, lr, deadline):
    """Adam optimizer using LogSE objective with active set"""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr_cur = lr
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        # Update active set every 10 steps
        if step % 10 == 0 or step == 0:
            corr = compute_corr(h_cur)
            active = np.argsort(corr)[::-1][:top_k]
        
        _, g = logse_value_grad(h_cur, alpha, active)
        
        t += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * g * g
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        
        h_new = h_cur - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h_cur = h_new
        
        sc = compute_c5(h_cur)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_cur.copy()
            no_improve = 0
            lr_cur = min(lr_cur * 1.005, 0.08)
        else:
            no_improve += 1
            if no_improve > 100:
                lr_cur *= 0.85
                no_improve = 0
                if lr_cur < 1e-7:
                    break
    
    return best_h, best_score

def rmsprop_optimize(h, n, target, alpha, top_k, max_steps, lr, deadline):
    """RMSProp optimizer with LogSE"""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    v_buf = np.zeros(n)
    decay = 0.9
    lr_cur = lr
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() > deadline:
            break

            active = np.argsort(corr)[::-1][:top_k]

        h_cur = h_new

                    break

def slsqp_logse(h, n, target, alpha, maxiter, ftol):
    """SLSQP with LogSE"""
    def obj(x):
        v, _ = logse_value_grad(x, alpha)
        return v
    def grad(x):
        _, g = logse_value_grad(x, alpha)
        return g
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def lbfgsb_logse(h, n, target, alpha, mu, maxiter):
    """L-BFGS-B with LogSE + penalty"""
    def obj(x):
        v, _ = logse_value_grad(x, alpha)
        return v + mu * (np.sum(x) - target) ** 2
    def grad(x):
        _, g = logse_value_grad(x, alpha)
        return g + 2.0 * mu * (np.sum(x) - target)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def logse_value_grad(h, alpha, active_idx):
    n = len(h)
    corr = compute_corr(h)
    c_active = corr[active_idx]
    c_max = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    val = c_max + float(np.log(exp_vals.sum())) / alpha
    w = exp_vals
    w_sum = w.sum()
    if w_sum < 1e-300:
        return val, np.zeros(n)
    w = w / w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, active_idx):
        if wk < 1e-15:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return val, g


def adam_logse(h, n, target, alpha, top_k, max_steps, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    beta1 = 0.85
    beta2 = 0.999
    eps_adam = 1e-8
    lr_cur = lr_init
    no_improve = 0
    corr = compute_corr(h_cur)
    active = np.argsort(corr)[::-1][:top_k]
    for step in range(max_steps):
        if time.time() > deadline:
            break
        if step % 10 == 0:
            corr = compute_corr(h_cur)
            active = np.argsort(corr)[::-1][:top_k]
        _, g = logse_value_grad(h_cur, alpha, active)
        t += 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * g
        v_buf = beta2 * v_buf + (1.0 - beta2) * g * g
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)
        h_new = h_cur - lr_cur * m_hat / (np.sqrt(v_hat) + eps_adam)
        h_new = project_feasible(h_new, target)
        h_cur = h_new
        sc = compute_c5(h_cur)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_cur.copy()
            no_improve = 0
            lr_cur = min(lr_cur * 1.003, 0.06)
        else:
            no_improve += 1
            if no_improve > 150:
                lr_cur *= 0.8
                no_improve = 0
                if lr_cur < 1e-7:
                    break
    return best_h, best_score


def rmsprop_logse(h, n, target, alpha, top_k, max_steps, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    v_buf = np.zeros(n)
    decay = 0.9
    lr_cur = lr_init
    no_improve = 0
    corr = compute_corr(h_cur)
    active = np.argsort(corr)[::-1][:top_k]
    for step in range(max_steps):
        if time.time() > deadline:
            break
        if step % 15 == 0:
            corr = compute_corr(h_cur)
            active = np.argsort(corr)[::-1][:top_k]
        _, g = logse_value_grad(h_cur, alpha, active)
        v_buf = decay * v_buf + (1.0 - decay) * g * g
        h_new = h_cur - lr_cur * g / (np.sqrt(v_buf) + 1e-8)
        h_new = project_feasible(h_new, target)
        h_cur = h_new
        sc = compute_c5(h_cur)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr_cur *= 0.85
                no_improve = 0
                if lr_cur < 1e-8:
                    break
    return best_h, best_score


def slsqp_logse(h, n, target, alpha, maxiter, ftol):
    def obj_fun(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def obj_grad(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            return np.zeros(len(x))
        w = w / w_sum
        dx = 2.0 / len(x)
        g = np.zeros(len(x))
        for i in range(len(corr)):
            if w[i] < 1e-15:
                continue
            k = i - (n - 1)
            if k >= 0:
                g[:n - k] += w[i] * (1.0 - x[k:])
                g[k:] -= w[i] * x[:n - k]
            else:
                ell = -k
                g[ell:] += w[i] * (1.0 - x[:n - ell])
                g[:n - ell] -= w[i] * x[ell:]
        g *= dx
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fun, h.copy(), method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgsb_logse(h, n, target, alpha, mu, maxiter):
    def obj_fun(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha + mu * (np.sum(x) - target) ** 2

    def obj_grad(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            base_g = np.zeros(len(x))
        else:
            w = w / w_sum
            dx = 2.0 / len(x)
            base_g = np.zeros(len(x))
            for i in range(len(corr)):
                if w[i] < 1e-15:
                    continue
                k = i - (n - 1)
                if k >= 0:
                    base_g[:n - k] += w[i] * (1.0 - x[k:])
                    base_g[k:] -= w[i] * x[:n - k]
                else:
                    ell = -k
                    base_g[ell:] += w[i] * (1.0 - x[:n - ell])
                    base_g[:n - ell] -= w[i] * x[ell:]
            base_g *= dx
        return base_g + 2.0 * mu * (np.sum(x) - target) * np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h.copy(), method='L-BFGS-B', jac=obj_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e6})
        return project_feasible(res.x, target)
    except Exception:
        return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_adam(h, n, target, alpha, top_k, n_starts, steps_each, lr_init, deadline):
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(42)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        scale = 0.04 + 0.03 * s
        h_pert = h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_new, sc_new = adam_logse(h_pert, n, target, alpha, top_k, steps_each, lr_init, deadline)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    return best_h, best_score


def polish_active_set(h, n, target, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    c_max = float(np.max(corr))
    threshold = c_max * 0.999
    active_candidates = np.where(corr >= threshold)[0]
    top_k = min(len(active_candidates), 12)
    if top_k < 3:

    h_new, sc_new = adam_logse(h_cur, n, target, 10000, top_k, 300, 0.008, deadline)
