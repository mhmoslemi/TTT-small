```python
def linearized_active_set_step(h, active_lags, step_limit=0.05):
    """Take one step using the linearized active set method."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    # Compute gradients for active lags
    grads = []
    for m in active_lags:
        k = m - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            ell = -k
            g[ell:] += (1.0 - h[:n-ell])
            g[:n-ell] -= h[ell:]
        g *= dx
        grads.append(g)
    
    # The LP: min t s.t. grads[k]·d ≤ t - corr[active_lags[k]] for all k
    #          sum(d) = 0
    #          |d[i]| ≤ step_limit
    #          0 ≤ h[i] + d[i] ≤ 1
    #
    # This has n+1 variables (d[0],...,d[n-1], t)
    # Constraints:
    #  For each active lag k: sum_i grads[k][i]*d[i] - t ≤ -corr[active_lags[k]]
    #  sum_i d[i] = 0
    #  -step_limit ≤ d[i] ≤ step_limit
    #  -h[i] ≤ d[i] ≤ 1-h[i]
    #
    # Actually, let's simplify: just minimize the maximum of the linearized active lags
    # min max_k (corr[k] + grads[k]·d)
    # = min t s.t. corr[k] + grads[k]·d ≤ t
    #
    # Or equivalently: min t s.t. grads[k]·d - t ≤ -corr[k]
    
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag: grads[k]·d - t ≤ -corr[active_lags[k]]
    A_ub = []
    b_ub = []
    for k_idx, m in enumerate(active_lags):
        row = np.zeros(n + 1)
        row[:n] = grads[k_idx]
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-corr[m])
    
    # Box constraints on d: max(0, h[i]-step_limit) ≤ h[i]+d[i] ≤ min(1, h[i]+step_limit)
    # i.e., -min(h[i], step_limit) ≤ d[i] ≤ min(1-h[i], step_limit)
    lb_d = -np.min(h, step_limit)
    ub_d = np.min(1.0 - h, step_limit)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = list(zip(lb_d, ub_d)) + [(None, None)]  # t is unbounded
    
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        new_h = h + d
        return project_feasible(new_h, n / 2.0)
    return h
```

```python
def solve_active_set_lp(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    # Compute gradients for active lags
    grads = []
    for m in active_indices:
        k = int(m) - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            ell = -k
            g[ell:] += (1.0 - h[:n-ell])
            g[:n-ell] -= h[ell:]
        g *= dx
        grads.append(g)
    
    # LP: min t s.t. grads[k]·d ≤ t - corr[m_k] for k in active
    #     sum(d) = 0
    #     box on d
    n_active = len(active_indices)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for i, (g, m) in enumerate(zip(grads, active_indices)):
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[m]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    step_limit = 0.1
    lb = -np.minimum(h, step_limit)
    ub = np.minimum(1.0 - h, step_limit)
    bounds = list(zip(lb, ub)) + [(None, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad_fft(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    dx = 2.0 / n
    size = 4 * n
    W_fft = np.fft.rfft(w, size)
    A = 1.0 - h
    A_fft = np.fft.rfft(A, size)
    H_fft = np.fft.rfft(h, size)
    conv = np.fft.irfft(W_fft * A_fft, size)
    cross = np.fft.irfft(W_fft * np.conj(H_fft), size)
    T1 = conv[n - 1:2 * n - 1]
    T2 = cross[0:n][::-1]
    return dx * (T1 - T2)


def lse_grad_direct(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lag_gradient(h, m_idx):
    n = len(h)
    k = int(m_idx) - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g * (2.0 / n)


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return lse_grad_fft(x, alpha) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def obj(x):
        return lse_value(x, alpha)

    def jacob(x):
        return lse_grad_fft(x, alpha)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jacob,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def active_set_lp_step(h, active_indices, step_limit=0.05):
    n = len(h)
    corr = compute_corr(h)
    grads = []
    for m in active_indices:
        grads.append(lag_gradient(h, m))
    n_act = len(active_indices)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((n_act, n + 1))

    for i, (g, m) in enumerate(zip(grads, active_indices)):
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[int(m)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lb = -np.minimum(h, step_limit)
    ub = np.minimum(1.0 - h, step_limit)

    try:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    if n_steps < 100:
        n_steps = 100
    temps = np.geomspace(t_start, t_end, n_steps)

    for step, T in enumerate(temps):
        if time.time() > deadline:
            break
        scale = 0.015 + 0.02 * (step / max(n_steps - 1, 1))
        h_pert = curr_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        g = lse_grad_fft(h_pert, 800)
        h_pert = project_feasible(h_pert - 0.005 * g, target)
        h_pert = project_feasible(h_pert - 0.003 * lse_grad_fft(h_pert, 2000), target)
        new_score = compute_c5(h_pert)
        delta = new_score - curr_score
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            curr_h = h_pert
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_pert.copy()
    return best_h, best_score


def structured_perturb_reopt(h, target, deadline, rng, n_attempts=6):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    scales = [0.05, 0.1, 0.03, 0.15, 0.08, 0.12]

    for attempt in range(n_attempts):
        if time.time() > deadline:
            break
        if attempt % 3 == 0:
            shift = rng.randint(1, max(2, n // 20))
            h_pert = np.roll(best_h, shift)
        elif attempt % 3 == 1:
            h_pert = best_h + rng.randn(n) * scales[attempt % len(scales)]
        else:
            center = rng.randint(n // 4, 3 * n // 4)
            width = max(3, n // 15)
            bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            amp = rng.choice([-0.2, 0.2])
            h_pert = best_h + amp * bump
        h_pert = project_feasible(h_pert, target)
        h_ref = lbfgsb_round(h_pert, n, target, 500, 100.0, 150)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            h_ref2 = slsqp_round(best_h, n, target, 3000, 800, 1e-15)
            sc2 = compute_c5(h_ref2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    phase1_deadline = deadline - 500.0
    alphas_p1 = [20, 50, 100, 300, 1000, 3000, 8000]
    for alpha in alphas_p1:
        if time.time() > phase1_deadline:
            break
        for mu in [5.0, 50.0, 300.0]:
            if time.time() > phase1_deadline:
                break
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 250)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if time.time() < deadline - 350.0:
        h_as, sc_as = active_set_optimize(best_h, min(deadline - 200.0, time.time() + 120.0))
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    if time.time() < deadline - 250.0:
        h_wo, sc_wo = weighted_active_optimize(best_h, min(deadline - 150.0, time.time() + 100.0))
        if sc_wo < best_score:
            best_score = sc_wo
            best_h = h_wo

    if time.time() < deadline - 200.0:
        for alpha in [2000, 5000, 12000]:
            if time.time() > deadline - 200.0:
                break
            h_new = slsqp_round(best_h, n, target, alpha, 1500, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if time.time() < deadline - 180.0:
        h_sg, sc_sg = subgradient_polish(best_h, target, 800, 0.003, 0.004, 0.997, deadline - 180.0)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if time.time() < deadline - 150.0:
            h_sg2, sc_sg2 = subgradient_polish(best_h, target, 500, 0.001, 0.002, 0.998, deadline - 150.0)
            if sc_sg2 < best_score:
                best_score = sc_sg2
                best_h = h_sg2

    n_high = 600
    target_high = n_high / 2.0
    if time.time() < deadline - 280.0:
        h_high = interpolate_to(best_h, n_high, target_high)
        best_high = h_high.copy()
        best_score_high = compute_c5(h_high)
        for alpha in [30, 100, 500, 2000, 8000]:
            if time.time() > deadline - 150.0:
                break
            for mu in [10.0, 100.0, 500.0]:
                if time.time() > deadline - 150.0:
                    break
                h_high_new = lbfgsb_round(best_high, n_high, target_high, alpha, mu, 200)
                sc_high = compute_c5(h_high_new)
                if sc_high < best_score_high:
                    best_score_high = sc_high
                    best_high = h_high_new
        if time.time() < deadline - 150.0:
            h_high_as, sc_high_as = active_set_optimize(best_high, min(deadline - 100.0, time.time() + 80.0))
            if sc_high_as < best_score_high:
                best_score_high = sc_high_as
                best_high = h_high_as
        if time.time() < deadline - 120.0:
            for alpha in [3000, 8000]:
                if time.time() > deadline - 120.0:
                    break
                h_high_new = slsqp_round(best_high, n_high, target_high, alpha, 1000, 1e-15)
                sc_high = compute_c5(h_high_new)
                if sc_high < best_score_high:
                    best_score_high = sc_high
                    best_high = h_high_new
        if time.time() < deadline - 100.0:
            h_high_sg, sc_high_sg = subgradient_polish(best_high, target_high, 600, 0.002, 0.003, 0.997, deadline - 100.0)
            if sc_high_sg < best_score_high:
                best_score_high = sc_high_sg
                best_high = h_high_sg
        if time.time() < deadline - 80.0:
            h_high_surg, sc_high_surg = worst_lag_surgery(best_high, target_high, 200, 0.003, deadline - 80.0)
            if sc_high_surg < best_score_high:
                best_score_high = sc_high_surg
                best_high = h_high_surg
        if best_score_high < best_score:
            best_score = best_score_high
            best_h = best_high
            n = n_high
            target = target_high

    if time.time() < deadline - 80.0:
        h_sa, sc_sa = simulated_annealing(best_h, target, min(deadline - 40.0, time.time() + 60.0), rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    if time.time() < deadline - 50.0:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_sp, sc_sp = structured_perturb_reopt(best_h, tgt_cur, min(deadline - 30.0, time.time() + 40.0), rng, 5)
        if sc_sp < best_score:
            best_score = sc_sp
            best_h = h_sp

    if time.time() < deadline - 20.0:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_polish(best_h, tgt_cur, 500, 0.001, 0.002, 0.998, deadline - 15.0)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if time.time() < deadline - 10.0:
            h_surg, sc_surg = worst_lag_surgery(best_h, tgt_cur, 150, 0.002, deadline - 8.0)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    if time.time() < deadline - 5.0:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_new = slsqp_round(best_h, n_cur, tgt_cur, 10000, 800, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
