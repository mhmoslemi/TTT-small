            # Term 2: i in [k_abs, n-1], -h[i-k_abs]
            grad[k_abs:n] -= wm * h[0:n-k_abs]
    return lse_val, grad

def lbfgs_with_penalty(h0, target, alphas, mus, maxiter_per_call, deadline):
    """Run L-BFGS-B with log-sum-exp objective and penalty for sum constraint."""
    n = len(h0)
    h_cur = h0.copy()

- W_neg[k] = w[(n-1) - k] for k = 1, ..., n-1 (weights for negative shifts)

def projected_gradient(h, target, alpha, n_steps, lr, deadline):
    """Projected gradient descent with LSE smooth approximation."""
    n = len(h)
    h_cur = h.copy()
    best = h_cur.copy()
    best_score = compute_c5(best)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        val, grad = lse_value_and_grad(h_cur, alpha)
        # Penetration gradient: add penalty for constraint violation
        pen_grad = (np.sum(h_cur) - target) * np.ones(n)
        grad_total = grad + pen_grad * 100.0
        
        h_new = h_cur - lr * grad_total
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
    return best, best_score

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    """SLSQP with LSE smooth approximation and exact constraint."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for a in alphas:
        if time.time() > deadline:
            break
        
        def _obj(x, a=a):
            val, _ = lse_value_and_grad(x, a)
            return val
        
        def _grad(x, a=a):
            _, g = lse_value_and_grad(x, a)
            return g
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(_obj, h_cur, jac=_grad, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score
