            trial[j_rand] = v[j_rand]
            trial = _proj(trial, tgt)
            sc = _c5(trial)
            if sc <= scores[i]:
                pop[i] = trial
                scores[i] = sc
                if sc < best_sc:
                    best_sc = sc
                    best_h = trial.copy()
    return best_h, best_sc

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    deadline = start + budget_s - 20.0

    n_final = len(initial_h_values)
    tgt_final = n_final / 2.0
    h_best = _proj(np.array(initial_h_values, dtype=float), tgt_final)
    sc_best = _c5(h_best)

    dx_final = 2.0 / n_final

    # Phase 1: Quick subgradient at final resolution
    t1 = min(start + 60, deadline)
    h_sg, sc_sg = _subgrad_opt(h_best, n_final, tgt_final, 2000, dx_final, t1)
    if sc_sg < sc_best:
        sc_best = sc_sg
        h_best = h_sg

    # Phase 2: SLSQP with increasing alpha
    t2 = min(start + 160, deadline)
    h_sl = _slsqp(h_best, n_final, tgt_final, [100, 500, 2000, 8000], 2000, 1e-15, t2)
    sc_sl = _c5(h_sl)
    if sc_sl < sc_best:
        sc_best = sc_sl
        h_best = h_sl

    # Phase 3: Top-K SLSQP (focus on worst few shifts)
    t3 = min(start + 280, deadline)
    for K in [3, 5, 10]:
        if time.time() > t3:
            break
        h_tk = _topk_slsqp(h_best, n_final, tgt_final, K, 5000, 2000, t3)
        sc_tk = _c5(h_tk)
        if sc_tk < sc_best:
            sc_best = sc_tk
            h_best = h_tk

    # Phase 4: Worst-lag sequential surgery
    t4 = min(start + 400, deadline)
    h_ws, sc_ws = _worst_lag_seq(h_best, n_final, tgt_final, dx_final, 500, t4)
    if sc_ws < sc_best:
        sc_best = sc_ws
        h_best = h_ws

    # Phase 5: More SLSQP rounds
    t5 = min(start + 550, deadline)
    h_sl2 = _slsqp(h_best, n_final, tgt_final, [5000, 20000, 50000], 3000, 1e-16, t5)
    sc_sl2 = _c5(h_sl2)
    if sc_sl2 < sc_best:
        sc_best = sc_sl2
        h_best = h_sl2

    # Phase 6: Subgradient with more steps and adaptive K
    t6 = min(start + 700, deadline)
    h_sg2, sc_sg2 = _subgrad_opt(h_best, n_final, tgt_final, 5000, dx_final, t6)
    if sc_sg2 < sc_best:
        sc_best = sc_sg2
        h_best = h_sg2

    # Phase 7: DE at final resolution (short)
    t7 = min(start + 900, deadline)
    h_de, sc_de = _de_run(h_best, n_final, tgt_final, 12, 80, 0.5, 0.4, t7, rng)
    if sc_de < sc_best:
        sc_best = sc_de
        h_best = h_de
        h_sl3 = _slsqp(h_best, n_final, tgt_final, [1000, 5000, 20000], 2000, 1e-16, min(start + 1000, deadline))
        sc_sl3 = _c5(h_sl3)
        if sc_sl3 < sc_best:
            sc_best = sc_sl3
            h_best = h_sl3

    # Phase 8: Random restarts with structured perturbations
    t8 = min(start + 1100, deadline)
    for attempt in range(10):
        if time.time() > t8:
            break
        if attempt < 3:
            pert = h_best + rng.randn(n_final) * 0.02
        elif attempt < 6:
            freq = rng.randint(1, 3)
            ph = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.04)
            x = np.linspace(0, 2, n_final)
            pert = h_best + amp * np.sin(freq * np.pi * x + ph)
        elif attempt < 8:
            # Local perturbation
            center = rng.randint(0, n_final)
            width = rng.randint(20, 60)
            pert = h_best.copy()
            lo = max(0, center - width)
            hi = min(n_final, center + width)
            pert[lo:hi] += rng.randn(hi - lo) * 0.03
        else:
            pert = h_best + rng.randn(n_final) * 0.05
        h_pert = _proj(pert, tgt_final)
        h_ref = _slsqp(h_pert, n_final, tgt_final, [500, 2000, 10000], 1500, 1e-16, t8)
        sc_ref = _c5(h_ref)
        if sc_ref < sc_best:
            sc_best = sc_ref
            h_best = h_ref

    # Phase 9: Final polish
    if time.time() < deadline - 30:
        h_sg3, sc_sg3 = _subgrad_opt(h_best, n_final, tgt_final, 3000, dx_final, deadline - 30)
        if sc_sg3 < sc_best:
            sc_best = sc_sg3
            h_best = h_sg3

    if time.time() < deadline - 15:
        h_ws2, sc_ws2 = _worst_lag_seq(h_best, n_final, tgt_final, dx_final, 300, deadline - 15)
        if sc_ws2 < sc_best:
            sc_best = sc_ws2
            h_best = h_ws2

    if time.time() < deadline - 5:
        h_final_sl = _slsqp(h_best, n_final, tgt_final, [10000, 30000], 2000, 1e-17, deadline - 5)
        sc_final = _c5(h_final_sl)
        if sc_final < sc_best:
            sc_best = sc_final
            h_best = h_final_sl

    h_out = _proj(h_best, tgt_final)
    sc_out = _c5(h_out)
    return h_out, sc_out, n_final
