import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among non-bound values
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_fk(h, k, dx):
    """Gradient of f_k(h) = sum_{i=0}^{n-1-k} h[i](1-h[i+k]) * dx"""
    n = len(h)
    if k == 0:
        # f_0 = sum h[i](1-h[i]) * dx = dx * (sum h - sum h^2)
        # df_0/dh[i] = dx * (1 - 2*h[i])
        return dx * (1.0 - 2.0 * h)
    else:
        grad = np.zeros(n)
        upper = n - 1 - k
        # Term 1: sum_{i=0}^{upper} h[i] * (1-h[i+k])
        # d/dh[i] for i in [0, upper]: (1-h[i+k])
        # d/dh[i+k] for i in [0, upper], i.e., j in [k, n-1]: -h[j-k]
        idx1 = np.arange(0, upper + 1)
        grad[idx1] += dx * (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= dx * h[idx2 - k]
        return grad

    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        h_local = project_feasible(x, target)
        corr = compute_corr(h_local)
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def block_optimize(h, target, block_size=50, n_iter=100):

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n - 1:n - 1 + n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / int(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_fk(h, k, dx):
    n = len(h)
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    grad = np.zeros(n)
    upper = n - 1 - k
    idx1 = np.arange(0, upper + 1)
    grad[idx1] += dx * (1.0 - h[idx1 + k])
    idx2 = np.arange(k, n)
    grad[idx2] -= dx * h[idx2 - k]
    return grad

        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
