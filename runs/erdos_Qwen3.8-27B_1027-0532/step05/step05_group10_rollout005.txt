    obj = c_max + np.log(Z) / alpha
    
    # Gradient: (1/Z) * sum_k exp(alpha*corr[k]) * grad(corr[k]) * ... 
    # Actually, d/dh [c_max + log(Z)/alpha] where Z = sum exp(alpha*(corr-k - c_max))
    # = (1/Z) * sum_k exp(alpha*(corr[k]-c_max)) * alpha * grad(corr[k]) / alpha
    # = (1/Z) * sum_k exp(alpha*(corr[k]-c_max)) * grad(corr[k])
    # (the c_max derivative cancels)
    
    # corr[k] = dx * np.correlate(h, 1-h)[k]
    # grad(corr[k]) = dx * grad_raw[k]
    
    weights = exp_shifted / Z  # sum to 1
    grad = np.zeros(N)
    half = N - 1
    for idx in range(2*N-1):
        k = idx - half
        w = weights[idx]
        if w > 1e-10:
            grad += w * dx * grad_corr_at_lag(h, k, N)
    
    return obj, grad

# Wait, this gradient computation is O(N²) because we loop over all 2N-1 lags.
# For N=400, that's 800 calls to grad_corr_at_lag, each O(N), so O(N²) = 160,000.
# That's fine for SLSQP.

def slsqp_optimize(h_start, N, target, alpha, maxiter, ftol):
    """SLSQP optimization with smooth max objective."""
    dx = 2.0 / N
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        Z = np.sum(exp_shifted)
        weights = exp_shifted / Z
        grad = np.zeros(N)
        half = N - 1
        for idx in range(2*N-1):
            k = idx - half
            w = weights[idx]
            if w > 1e-10:
                grad += w * dx * grad_corr_at_lag(h, k, N)
        return grad
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(N)
    
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, target)

def lp_step(h, N, target, active_lags, solver='SCS', eps=1e-10):
    """One LP step: minimize t subject to c_k(h_new) <= t for active k, with h linearized."""
    dx = 2.0 / N
    g = 1.0 - h  # fixed complement
    
    h_var = cp.Variable(N)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    half = N - 1
    for k in active_lags:
        idx = k + half  # position in full corr
        # c_k = dx * sum_n h[n+k] * g[n] (for valid n)
        # Linearized: c_k ≈ dx * sum_n h_new[n+k] * g[n]
        # = dx * (linear function of h_new)
        lo = max(0, -k) if k < 0 else 0
        hi = min(N, N-k) if k >= 0 else N
        # When k >= 0: n from 0 to N-1-k, h indices from k to N-1
        # When k < 0: n from -k to N-1, h indices from 0 to N-1+k
        if k >= 0:
            # c_k = dx * sum_{n=0}^{N-1-k} h[n+k] * g[n]
            # = dx * sum_{m=k}^{N-1} h[m] * g[m-k]
            coeff = np.zeros(N)
            coeff[k:] = dx * g[:N-k]
        else:
            q = -k
            # c_k = dx * sum_{n=q}^{N-1} h[n-q] * g[n]
            # = dx * sum_{m=0}^{N-1-q} h[m] * g[m+q]
            coeff = np.zeros(N)
            coeff[:N-q] = dx * g[q:]
        
        constraints.append(coeff @ h_var <= t_var)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, eps=eps, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return h, None
    
    if h_var.value is None:
        return h, None
    
    h_new = np.clip(h_var.value, 0.0, 1.0)
    t_val = t_var.value if t_var.value is not None else compute_c5(h_new)
    return h_new, t_val

def alternating_lp(h_init, N, target, time_limit, n_active_start=50, n_active_max=200,
                   damping_start=0.7, reg_start=0.0):
    """Alternating LP with growing active set."""
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_start
    reg = reg_start
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_active = min(n_active_max, n_active_start + int(50 * (time.time() - start) / max(1, time_limit)))
        active_k = set(np.argsort(corr)[-n_active:].tolist())
        
        h_new, t_val = lp_step(h, N, target, active_k)
        if h_new is None:
            break
        
        # Damped update
        h_damped = (1 - damping) * h + damping * h_new
        h_damped = project_feasible(h_damped, target)
        
        c5_val = compute_c5(h_damped)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_damped.copy()
        
        h = h_damped
        
        # Adjust damping
        if c5_val > best_c5 + 1e-6:
            damping = max(0.3, damping * 0.8)
        else:
            damping = min(0.95, damping + 0.05)
    
    return best_h, best_c5
