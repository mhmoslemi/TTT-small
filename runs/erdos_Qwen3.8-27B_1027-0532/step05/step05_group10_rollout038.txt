import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using water-filling."""
    h = np.asarray(h, dtype=float).copy()
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_val(h, alpha):
    """Value of smooth max approximation."""
    dx = 2.0 / len(h)
    corr = compute_corr_array(h) * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_obj_grad(h, alpha):
    """Gradient of smooth max approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_array(h) * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    W = exp_shifted / np.sum(exp_shifted)
    
    # grad[m] = dx * sum_j W[j] * [(1-h[m+k_j]) - h[m-k_j]] (where valid)
    # k_j = j - (n-1)
    grad = np.zeros(n)
    nh = n - 1
    for j in range(2 * n - 1):
        if W[j] < 1e-12:
            continue
        k = j - nh
        # First term: (1-h[m+k])*dx for valid m
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += W[j] * (1.0 - h[lo1+k:hi1+k]) * dx
        # Second term: -h[m-k]*dx for valid m
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= W[j] * h[lo2-k:hi2-k] * dx
    return grad

def projected_gradient_descent(h_init, n, target, alphas, time_limit, start_time=None):
    """Optimize using projected gradient with analytical gradient."""
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for alpha in alphas:
        # Inner loop for this alpha
        max_inner = 2000
        for it in range(max_inner):
            if time.time() - start_time > time_limit:
                return best_h, best_val
            obj, grad = smooth_obj_val(h, alpha), smooth_obj_grad(h, alpha)
            
            # Backtracking line search
            lr = 1.0 / (alpha * max(np.max(np.abs(grad)), 1e-10) * 2.0 / n)
            # Actually, let me use a simpler step size
            # The gradient magnitude is at most sum_k |W_k| * dx * 2 = 2*dx = 4/n
            # So the Lipschitz constant is about alpha * 4/n
            # Step size: 1 / (alpha * 4/n) = n / (4*alpha)
            lr = n / (4.0 * alpha * 2.0)  # conservative
            
            for ls in range(30):
                h_new = h - lr * grad
                h_new = project_feasible(h_new, target)
                new_obj = smooth_obj_val(h_new, alpha)
                c5_new = compute_c5(h_new)
                if new_obj < obj - 1e-15 * abs(obj) or c5_new < best_val:
                    break
                lr *= 0.5
            
            h = project_feasible(h - lr * grad, target)
            
            c5_cur = compute_c5(h)
            if c5_cur < best_val:
                best_val = c5_cur
                best_h = h.copy()
    
    return best_h, best_val

def lp_constraint_generation(h_init, n, target, time_limit, n_active=20, reg=1e-4):
    """Alternating LP with constraint generation."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    active = set(np.argsort(corr)[-n_active:].tolist())

def slsqp_optimize(h_init, n, target, alpha, maxiter, ftol=1e-14):
    """SLSQP with analytical gradient."""
    def fun(h):
        return smooth_obj_val(h, alpha)
    
    def grad(h):
        return smooth_obj_grad(h, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    
    try:
        res = minimize(fun, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except:
        return h_init.copy()

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_obj_val(h, alpha):
    dx = 2.0 / len(h)
    corr = compute_corr_array(h) * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_array(h) * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    W = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    nh = n - 1

def lp_constraint_gen(h_init, n, target, time_limit, n_active=25, reg=1e-4, damping=0.75):

    nh = n - 1

    while time.time() - start < time_limit:

        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]

        lp_time = min(150, tleft() * 0.25)
        h_lp, c5_lp = lp_constraint_gen(best_h, n, target,
                                        time_limit=lp_time, n_active=25, reg=1e-4, damping=0.75)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 2: Projected gradient with analytical gradient
    if tleft() > 180:
        pg_time = min(150, tleft() * 0.3)
        h_pg, c5_pg = projected_gradient_opt(best_h, n, target,
                                             alphas=[200, 1000, 5000, 20000],
                                             time_limit=pg_time, t_start=start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 3: LP again
    if tleft() > 150:
        lp_time = min(120, tleft() * 0.28)
        h_lp, c5_lp = lp_constraint_gen(best_h, n, target,
                                        time_limit=lp_time, n_active=35, reg=1e-5, damping=0.85)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP with analytical gradient
    if tleft() > 100:
        for alpha in [5000, 20000, 100000]:
            if tleft() < 25:
                break
            h_s = slsqp_with_grad(best_h, n, target, alpha, maxiter=800)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase 5: Binary push + local search
    if tleft() > 80:
        h_bin, c5_bin = binary_push(best_h, n, target)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        # Binary local search
        if tleft() > 50:
            h_bs, c5_bs = binary_local_search(best_h, n, target,
                                              max_rounds=100,
                                              time_limit=min(35, tleft() * 0.5))
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs
        # Refine after binary
        if tleft() > 40:
            h_s = slsqp_with_grad(best_h, n, target, 10000, maxiter=500)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase 6: Multi-resolution exploration
    if tleft() > 250:
        n_low = 100
        target_low = n_low / 2.0
        x_low = np.linspace(0.0, 1.0, n_low)
        x_high = np.linspace(0.0, 1.0, n)
        h_low = np.interp(x_low, x_high, best_h)
        h_low = project_feasible(h_low, target_low)
        h_lp_low, c5_lp_low = lp_constraint_gen(h_low, n_low, target_low,
                                                time_limit=min(60, tleft() * 0.15),
                                                n_active=15, reg=1e-3, damping=0.7)
        h_up = interpolate_to(h_lp_low, n, target)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
        # Refine the upsolved
        if tleft() > 100:
            h_lp2, c5_lp2 = lp_constraint_gen(best_h, n, target,
                                              time_limit=min(80, tleft() * 0.25),
                                              n_active=25, reg=1e-4, damping=0.8)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2
        if tleft() > 50:
            h_s = slsqp_with_grad(best_h, n, target, 20000, maxiter=500)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
        # Try binary at this resolution too
        if tleft() > 50:
            h_bin2, c5_bin2 = binary_push(best_h, n, target)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2
            if tleft() > 30:
                h_bs2, c5_bs2 = binary_local_search(best_h, n, target,
                                                    max_rounds=80,
                                                    time_limit=min(20, tleft() * 0.5))
                if c5_bs2 < best_c5:
                    best_c5 = c5_bs2
                    best_h = h_bs2
                if tleft() > 20:
                    h_s2 = slsqp_with_grad(best_h, n, target, 10000, maxiter=300)
                    c5_s2 = compute_c5(h_s2)
                    if c5_s2 < best_c5:
                        best_c5 = c5_s2
                        best_h = h_s2

    # Phase 7: Perturbed restarts
    for noise in [0.05, 0.1, 0.2, 0.3]:
        if tleft() < 50:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_s = slsqp_with_grad(h_pert, n, target, 2000, maxiter=500)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
            if tleft() > 30:
                h_bin3, c5_bin3 = binary_push(best_h, n, target)
                if c5_bin3 < best_c5:
                    best_c5 = c5_bin3
                    best_h = h_bin3

    # Phase 8: Final LP polish
    if tleft() > 45:
        lp_time = min(35, tleft() * 0.6)
        h_lp, c5_lp = lp_constraint_gen(best_h, n, target,
                                        time_limit=lp_time, n_active=40, reg=1e-6, damping=0.9)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 9: Final SLSQP with very high alpha
    if tleft() > 25:
        h_s = slsqp_with_grad(best_h, n, target, 100000, maxiter=400)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 10: Final binary push and local search
    if tleft() > 20:
        h_bin4, c5_bin4 = binary_push(best_h, n, target)
        if c5_bin4 < best_c5:
            best_c5 = c5_bin4
            best_h = h_bin4
        if tleft() > 15:
            h_bs4, c5_bs4 = binary_local_search(best_h, n, target,
                                                max_rounds=50,
                                                time_limit=min(10, tleft() * 0.7))
            if c5_bs4 < best_c5:
                best_c5 = c5_bs4
                best_h = h_bs4
            if tleft() > 10:
                h_s4 = slsqp_with_grad(best_h, n, target, 50000, maxiter=200)
                c5_s4 = compute_c5(h_s4)
                if c5_s4 < best_c5:
                    best_c5 = c5_s4
                    best_h = h_s4

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_c5 = compute_c5(h_final)
    return h_final, final_c5, n
