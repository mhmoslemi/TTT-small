import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad


def _con_fun(h):
    return np.sum(h) - _global_target


def _con_jac(h):
    return np.ones(_global_n)


def slsqp_analytic(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), jac=_smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def projected_momentum(h_start, n, target, time_limit, max_steps, alpha, mu, base_lr):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = _smooth_grad(h, alpha)
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.01 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14 or c5_try < compute_c5(h) - 1e-14:
                h = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
    return best_h, best_c5


def improved_pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand, top_k):

    best_h = h.copy()

    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 300:
            break

        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_alpha = 50.0
        w_top = np.exp(soft_alpha * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi

                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)

                    best_j = j
                    best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Projected momentum with smooth objective
    if tleft() > 150:
        for alpha_m in [100, 500]:
            if tleft() < 50:
                break
            h_pm, c5_pm = projected_momentum(best_h, n, target,
                                             time_limit=min(60, tleft() * 0.3),
                                             max_steps=50000, alpha=alpha_m,
                                             mu=0.85, base_lr=0.05)
            if c5_pm < best_score:
                best_score = c5_pm
                best_h = h_pm

    # Phase 3: Improved pair CD with top-K soft gradient
    if tleft() > 150:
        cd_time = min(300, tleft() * 0.35)
        h_cd, c5_cd = improved_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=10000, n_cand=55, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP with high alphas
    if tleft() > 120:
        h_pol = slsqp_analytic(best_h, n, target,
                               [2000, 10000, 50000],
                               1500, 1e-16, min(100, tleft()))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Another round of pair CD
    if tleft() > 100:
        cd_time = min(200, tleft() * 0.35)
        h_cd, c5_cd = improved_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000, n_cand=70, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: SLSQP polish
    if tleft() > 80:
        h_pol = slsqp_analytic(best_h, n, target,
                               [5000, 20000],
                               1000, 1e-16, min(60, tleft()))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Multi-restart with structured perturbations
    rng = np.random.RandomState(seed)
    if tleft() > 150:
        for noise_scale in [0.03, 0.07, 0.12]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref = slsqp_analytic(h_pert, n, target,
                                   [100, 1000, 10000],
                                   1000, 1e-15, min(40, tleft()))
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: Local perturbation + pair CD
    if tleft() > 100:
        h_pert, c5_pert = local_perturbation_search(best_h, n, target,
                                                    max_tries=3000,
                                                    time_limit=min(30, tleft() * 0.2))
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert
        if tleft() > 50:
            cd_time = min(40, tleft() * 0.4)
            h_cd, c5_cd = improved_pair_cd(best_h, n, target,
                                           time_limit=cd_time,
                                           max_sweeps=5000, n_cand=50, top_k=3)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 9: Try higher resolution
    if tleft() > 150:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_analytic(h_interp, n_try, target_try,
                                     [50, 500, 5000],
                                     1000, 1e-15, min(60, tleft()))
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 10: Final pair CD at best resolution
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = improved_pair_cd(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=5000, n_cand=60, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_analytic(best_h, n_use, target_use,
                                 [10000, 50000],
                                 800, 1e-16, min(20, tleft()))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
