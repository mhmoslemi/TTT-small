from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def exact_coord_descent(h_init, time_limit=600, max_sweeps=200, n_cand_limit=None):
    """Exact coordinate descent using binary search on t."""
    start = time.time()
    N = len(h_init)
    h = h_init.copy()
    n_corr = 2 * N - 1
    shifts = np.arange(-(N-1), N)
    dx = 2.0 / N
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    
    no_improve_sweeps = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        sweep_changed = False
        for j in range(N):
            # Compute B for this coordinate
            js_minus = j - shifts
            js_plus = j + shifts
            
            valid_minus = (js_minus >= 0) & (js_minus < N)
            valid_plus = (js_plus >= 0) & (js_plus < N)
            
            B = np.zeros(n_corr)
            B[valid_minus] = 1.0 - h[js_minus[valid_minus]]
            B[valid_plus] -= h[js_plus[valid_plus]]
            
            A = corr - B * h[j]

            t_lo = 0.0

            cmax_val = np.max(A + B * h[j])

                if np.min(active_B) > 1e-8:
                    # All active have positive slope → should decrease x
                    # Try x = 0
                    pass  # Let binary search handle it
                elif np.max(active_B) < -1e-8:
                    # All active have negative slope → should increase x
                    pass
                else:

                    continue
            
            # Also check boundary: if h[j] == 0 and all active B >= 0, stay
            if h[j] <= 1e-12:

                    continue

                    continue
            
            # Binary search
            for _ in range(50):

                pos_mask = B > 1e-12

                    x_lower = -1e10

                hi_eff = min(1.0, x_upper)

                else:
                    t_lo = t_mid

            pos_mask = B > 1e-12
            neg_mask = B < -1e-12
            
            if pos_mask.any():

            else:
                x_upper = 1.0
            
            if neg_mask.any():
                x_lower = float(np.max((t_opt - A[neg_mask]) / B[neg_mask]))
            else:
                x_lower = 0.0
            
            x_lo = max(0.0, x_lower)

            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

            break
    
    return best_h, best_c5

_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def soft_subgradient(h, n, target, time_limit, max_steps=20000, n_top=5, base_lr=0.03):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()

        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 80.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k_star_safe(h, idxs, k)])
            grad[v2] -= wi * h[idxs[v2] - k_star_safe(h, idxs, -k)]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved and step > 500:
            break
    return best_h, best_c5

def momentum_subgradient(h, n, target, time_limit, max_steps=20000, mu=0.85, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, n_dst / 2.0)


def pnorm_optimize(h_init, n, target, p, time_limit, max_iter=5000, lr=0.01):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        weights = corr ** (p - 1)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        grad = np.zeros(n)
        for s_idx in range(len(corr)):
            w = weights[s_idx]
            if w < 1e-10:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        for lr_mult in [1.0, 0.5, 0.25]:
            lr_try = lr * lr_mult
            h_try = h - lr_try * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                break

    return best_h, best_c5

            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc_final = slsqp_round(best_h, n_use, target_use,
                                        [10000, 50000, 200000],
                                        2000, 1e-16)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
