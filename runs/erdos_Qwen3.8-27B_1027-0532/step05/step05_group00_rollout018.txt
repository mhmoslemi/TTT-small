    # delta[k] for shift k (index k in corr array is k-(n-1))
    # For shift k:
    #   terms involving i: -(1-h[i+k]) if valid, +h[i-k] if valid
    #   terms involving j: +(1-h[j+k]) if valid, -h[j-k] if valid
    
    delta = np.zeros(2*n - 1)

    lo = max(-i, -(n-1))
    hi = min(n-1-i, n-1)
    for k in range(lo, hi+1):
        delta[k + (n-1)] -= (1 - h[i+k])
    
    # Shift k where i-k is in [0, n): k in [i-(n-1), i]
    # delta[k] += h[i-k]

def compute_delta_vec(h, i, j, n):
    delta = np.zeros(2*n - 1)
    offset = n - 1
    
    # For position i (1 -> 0):
    # - (1 - h[i+k]) for k where 0 <= i+k < n, i.e., k in [-i, n-1-i]
    lo = max(-i, -offset)
    hi = min(n-1-i, offset)
    if lo <= hi:
        idx = np.arange(lo + offset, hi + offset + 1)
        h_idx = np.arange(i + lo, i + hi + 1)
        delta[idx] -= (1 - h[h_idx])
    
    # + h[i-k] for k where 0 <= i-k < n, i.e., k in [i-(n-1), i]
    lo = max(i - offset, -offset)
    hi = min(i, offset)
    if lo <= hi:
        idx = np.arange(lo + offset, hi + offset + 1)
        h_idx = np.arange(i - hi, i - lo + 1)  # i-k for k in [lo, hi]
        delta[idx] += h[h_idx]
    
    # For position j (0 -> 1):
    # + (1 - h[j+k]) for k where 0 <= j+k < n
    lo = max(-j, -offset)
    hi = min(n-1-j, offset)
    if lo <= hi:
        idx = np.arange(lo + offset, hi + offset + 1)

    hi = min(j, offset)
    if lo <= hi:
        idx = np.arange(lo + offset, hi + offset + 1)
        h_idx = np.arange(j - hi, j - lo + 1)
