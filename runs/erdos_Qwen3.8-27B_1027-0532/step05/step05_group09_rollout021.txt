from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    n = len(h)
    h = h.astype(float).copy()
    # Project onto {x: 0 <= x[i] <= 1, sum(x) = target}
    # Using bisection on the Lagrange multiplier
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def compute_grad_direct(h, n, k):
    """Compute gradient of f_k with respect to h"""
    dx = 2.0 / n

        return dx * (1.0 - 2.0 * h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    n = len(h)
    h = h.astype(float).copy()

    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]

    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def grad_fk(h, n, k):
    dx = 2.0 / n
    if k == 0:
        return dx * (1.0 - 2.0 * h)

    np.add(grad[:n-k], dx * (1.0 - h[k:]), out=grad[:n-k])
    if n - k <= 0:
        return grad
    np.subtract(grad[k:], dx * h[:n-k], out=grad[k:])
    return grad
