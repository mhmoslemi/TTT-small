    #              = (let j = i+lag) j from 0 to n-1+lag, sum h[j]*(1-h[j-lag])
    
    # Gradient with respect to h[j]:
    # For lag >= 0:
    #   h[j] appears as h[i+lag] when i = j-lag, valid if 0 <= j-lag <= n-1-lag, i.e., lag <= j <= n-1
    #     contribution: dx * (1-h[j-lag])
    #   h[j] appears as h[i] when i = j, valid if 0 <= j <= n-1-lag
    #     contribution: dx * h[j+lag] * (-1) = -dx * h[j+lag]
    #
    # For lag < 0 (let p = -lag > 0):
    #   i from p to n-1, sum h[i-p]*(1-h[i])
    #   h[j] appears as h[i-p] when i = j+p, valid if p <= j+p <= n-1, i.e., 0 <= j <= n-1-p
    #     contribution: dx * (1-h[j+p])
    #   h[j] appears as h[i] when i = j, valid if p <= j <= n-1
    #     contribution: dx * h[j-p] * (-1) = -dx * h[j-p]
    
    # So in both cases, for |lag| = p:
    #   h[j] appears in "h[advanced]" position: valid if 0 <= j <= n-1-p, contribution dx*(1-h[j±p])
    #   h[j] appears in "h[base]" position: valid if p <= j <= n-1, contribution -dx*h[j∓p]
    
    # Let me just compute it numerically for safety and speed:
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        # d = dx * sum h[i](1-h[i]), grad[i] = dx*(1-2*h[i])
        grad[:] = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        # d = dx * sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        upper = n - lag
        # h[j] in h[i+lag] role: j = i+lag, i from 0 to n-1-lag, so j from lag to n-1
        #   contribution: dx*(1-h[j-lag])
        grad[lag:] += dx * (1.0 - h[:-lag])
        # h[j] in h[i] role: j = i, i from 0 to n-1-lag, so j from 0 to n-lag-1
        #   contribution: -dx*h[j+lag]
        grad[:upper] -= dx * h[lag:upper]
    else:
        p = -lag
        # d = dx * sum_{i=p}^{n-1} h[i-p]*(1-h[i])
        # h[j] in h[i-p] role: j = i-p, i from p to n-1, so j from 0 to n-1-p
        #   contribution: dx*(1-h[j+p])
        upper = n - p
        grad[:upper] += dx * (1.0 - h[p:])
        # h[j] in h[i] role: j = i, i from p to n-1, so j from p to n-1
        #   contribution: -dx*h[j-p]
        grad[p:] -= dx * h[:upper]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr_full(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def cons_eq(x):
    return float(np.sum(x)) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_step(h, n_active, temp, lr):
    """One subgradient step using top active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    c_curr = float(np.max(corr)) * dx
    top_idx = np.argsort(corr)[::-1][:n_active]
    diffs = corr[top_idx] - corr[top_idx[0]]
    weights = np.exp(-diffs / temp)  # negative so top gets highest weight
    weights /= weights.sum()
    grad = np.zeros(n)
    for wk, k in zip(weights, top_idx):
        k = int(k)
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            upper = n - lag
            grad[lag:] += wk * dx * (1.0 - h[:-lag])
            grad[:upper] -= wk * dx * h[lag:]
        else:
            p = -lag
            upper = n - p
            grad[:upper] += wk * dx * (1.0 - h[p:])
            grad[p:] -= wk * dx * h[:upper]
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n / 2.0)
    return h_new, c_curr

def subgradient_loop(h, n_steps, n_active, lr_init, lr_decay, momentum, temp, time_limit, start_time):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr_full(h_cur)
        c_curr = float(np.max(corr))
        dx = 2.0 / n
        c_curr_dx = c_curr * dx
        if c_curr_dx < best_score:
            best_score = c_curr_dx
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(-diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            lag = k - (n - 1)
            if lag == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            elif lag > 0:
                upper = n - lag
                grad[lag:] += wk * dx * (1.0 - h_cur[:-lag])
                grad[:upper] -= wk * dx * h_cur[lag:]
            else:
                p = -lag
                upper = n - p
                grad[:upper] += wk * dx * (1.0 - h_cur[p:])
                grad[p:] -= wk * dx * h_cur[:upper]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, n / 2.0)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr_dx:
            h_cur = h_new
            lr = min(lr * 1.01, 0.005)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def binary_local_search(h, n_iterations, rng, time_limit, start_time):
    """Local search on binary h via random swaps."""
    n = len(h)
    target = n / 2.0
    # Start from binary version
    h_bin = (h > 0.5).astype(float)
    # Fix to have exactly n/2 ones
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    if len(ones) > n/2:
        excess = len(ones) - n//2
        to_flip = rng.choice(ones, excess, replace=False)
        h_bin[to_flip] = 0.0
    elif len(ones) < n/2:
        deficit = n//2 - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h_bin[to_flip] = 1.0
    
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    
    for it in range(n_iterations):
        if time.time() - start_time > time_limit:
            break
        # Try a random swap
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        trial = best_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        sc = compute_c5(trial)
        if sc < best_score - 1e-15:
            best_h = trial
            best_score = sc
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)

```python
def swap_delta(h, i, j, n):
    """Compute the change in full correlation when h[i]: 1->0 and h[j]: 0->1."""
    # h[i] was 1, becomes 0
    # h[j] was 0, becomes 1
    # delta_C[m] = contribution_changes
    # For h[i]: 1->0
    #   As h[a] with a=i: m = i - b + (n-1), b ranges so that a=i is valid
    #     Actually: C[m] includes term h[a]*(1-h[b]) where a-b = m-(n-1)
    #     So a = i means m-(n-1) = i-b, m = i-b+(n-1)
    #     Valid b: 0<=b<=n-1, and m in [0, 2n-2]
    #     m = i-b+(n-1), so as b goes from 0 to n-1, m goes from i+(n-1) to i
    #     Valid m: max(0, i) <= m <= min(2n-2, i+n-1)
    #     For each such m, b = i-m+(n-1), delta from h[i] 1->0: was 1*(1-h[b]), now 0, so delta = -(1-h[b])
    #   As h[b] with b=i: m-(n-1) = a-i, m = a-i+(n-1)
    #     Valid a: 0<=a<=n-1, m in [0, 2n-2]
    #     m = a-i+(n-1), as a goes from 0 to n-1, m goes from -i+(n-1) to n-1-i+(n-1) = n-1-i+n... wait
    #     m = a - i + (n-1), a from 0 to n-1, so m from (n-1-i) to (n-1-i+n-1) = 2n-2-i
    #     Valid m: max(0, n-1-i) <= m <= min(2n-2, 2n-2-i)
    #     For each such m, a = m-i+(n-1), delta from h[i] 1->0: was h[a]*(1-1)=0, now h[a]*(1-0)=h[a], so delta = h[a]
    
    # Similarly for h[j]: 0->1
    #   As h[a] with a=j: delta = +(1-h[b]) for each valid b (since h'[j]=1, was 0)
    #   As h[b] with b=j: delta = -h[a] for each valid a (since 1-h'[j]=0, was 1)
    
    delta = np.zeros(2*n-1)
    
    # h[i]: 1 -> 0
    # As h[a=i]: m = i - b + (n-1) for b in [0, n-1]
    #   delta[m] -= (1 - h[b]) where b = i - m + (n-1)
    for b in range(n):
        m = i - b + (n-1)
        if 0 <= m < 2*n-1:
            delta[m] -= (1.0 - h[b])
    # As h[b=i]: m = a - i + (n-1) for a in [0, n-1]
    #   delta[m] += h[a] where a = m - i + (n-1)
    for a in range(n):
        m = a - i + (n-1)
        if 0 <= m < 2*n-1:
            delta[m] += h[a]
    
    # h[j]: 0 -> 1
    # As h[a=j]: m = j - b + (n-1) for b in [0, n-1]
    #   delta[m] += (1 - h[b]) (since h'[j]=1 contributes (1-h[b]), was 0)
    for b in range(n):
        m = j - b + (n-1)
        if 0 <= m < 2*n-1:
            delta[m] += (1.0 - h[b])
    # As h[b=j]: m = a - j + (n-1) for a in [0, n-1]
    #   delta[m] -= h[a] (since 1-h'[j]=0, was 1, so h[a]*1 becomes h[a]*0)
    for a in range(n):
        m = a - j + (n-1)
        if 0 <= m < 2*n-1:
            delta[m] -= h[a]
    
    return delta
```

```python
def swap_delta_vec(h, i, j, n):
    """Vectorized computation of correlation change for swap i:1->0, j:0->1."""
    delta = np.zeros(2*n-1)
    
    # h[i]: 1 -> 0, as h[a=i]: delta[m] -= (1-h[b]) where m = i-b+(n-1)
    # b ranges 0..n-1, m = i-b+(n-1) = (i+n-1) - b
    # So m ranges from (i+n-1)-(n-1)=i to i+n-1
    # m_low = i, m_high = i+n-1, clipped to [0, 2n-2]
    m_lo = max(0, i)
    m_hi = min(2*n-2, i+n-1)
    if m_lo <= m_hi:
        b_vals = (i + n - 1) - np.arange(m_lo, m_hi+1)
        delta[m_lo:m_hi+1] -= (1.0 - h[b_vals])
    
    # h[i]: 1 -> 0, as h[b=i]: delta[m] += h[a] where m = a-i+(n-1)
    # a ranges 0..n-1, m = a-i+(n-1) = (n-1-i) + a
    # m ranges from (n-1-i) to (n-1-i)+(n-1) = 2n-2-i
    m_lo = max(0, n-1-i)
    m_hi = min(2*n-2, 2*n-2-i)
    if m_lo <= m_hi:
        a_vals = np.arange(m_lo, m_hi+1) - (n-1-i)
        delta[m_lo:m_hi+1] += h[a_vals]
    
    # h[j]: 0 -> 1, as h[a=j]: delta[m] += (1-h[b]) where m = j-b+(n-1)
    m_lo = max(0, j)
    m_hi = min(2*n-2, j+n-1)
    if m_lo <= m_hi:
        b_vals = (j + n - 1) - np.arange(m_lo, m_hi+1)
        delta[m_lo:m_hi+1] += (1.0 - h[b_vals])
    
    # h[j]: 0 -> 1, as h[b=j]: delta[m] -= h[a] where m = a-j+(n-1)
    m_lo = max(0, n-1-j)
    m_hi = min(2*n-2, 2*n-2-j)
    if m_lo <= m_hi:
        a_vals = np.arange(m_lo, m_hi+1) - (n-1-j)
        delta[m_lo:m_hi+1] -= h[a_vals]
    
    return delta
```
