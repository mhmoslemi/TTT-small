import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_init, n, target, time_limit, rng, max_rounds=1000000):
    """Simulated annealing on binary solutions."""
    start = time.time()
    # Round to binary
    h = (h_init >= 0.5).astype(float)
    # Fix sum
    ones = np.sum(h)
    if ones > target:
        # Remove excess ones
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:int(ones-target)]] = 0
    elif ones < target:
        # Add needed ones
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:int(target-ones)]] = 1
    h = project_feasible(h, target)
    
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    T = 0.01
    T_min = 1e-8
    cooling = 0.999
    
    for step in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if T < T_min:
            break
            
        # Pick random swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        
        h[i] = 0
        h[j] = 1
        
        new_score = compute_c5(h)
        delta = new_score - current_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        T *= cooling
    
    return best_h, best_score

def cutting_plane_lp(h_init, n, target, time_limit, eps=1e-10):
    """Solve the LP relaxation using cutting-plane method."""
    start = time.time()
    h = h_init.copy()
    active_lags = []
    
    while time.time() - start < time_limit:
        # Find current worst lags
        overlaps = compute_overlaps(h)
        c5_val = np.max(overlaps)
        
        # Find all lags within eps of the max
        threshold = c5_val - eps
        active_mask = overlaps >= threshold
        active_lags = np.where(active_mask)[0] - (n - 1)  # Convert to actual lag values
        
        if len(active_lags) == 0:
            break

        # min_t t

        #      0 <= h <= 1
        #      sum(h) = target

        # Fix g = 1-h, then L_k = sum_i h[i]*g[i+k]*dx (linear in h)

        return h

    else:
        idx = np.where(h > 1e-10)[0]

        return h

    h = np.clip(h, 0.0, 1.0)

    h_cur = h_start.copy()
    for alpha in alphas:
        def obj(x, a=alpha):
            val, _ = smooth_max_obj(x, a)
            return val
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def lp_refined(h_init, n, target, max_iter, time_limit, n_top=150,
               damping=0.9, reg=1e-5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        for k in lags:
            k = int(k)
            i_start = max(0, -k)
            i_end = min(n, n - k)
            if i_end > i_start:
                coeff = np.zeros(n)
                coeff[i_start:i_end] = g[i_start + k:i_end + k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0, budget_s - 30.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Generate diverse binary candidates
    if tleft() > 50:
        candidates = []
        # From initial
        c1 = to_binary(best_h, n, target, rng)
        candidates.append(c1)
        # Sturmian with golden ratio
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
        c2 = sturmian(n, alpha)
        c2 = to_binary(c2, n, target, rng)
        candidates.append(c2)
        # Sturmian with other alpha
        c3 = sturmian(n, 0.382)
        c3 = to_binary(c3, n, target, rng)
        candidates.append(c3)
        # Random
        c4 = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        c4[idx] = 1.0
        candidates.append(c4)
        # Quadratic residue style
        c5 = quadratic_residue_pattern(n, 17)
        c5 = to_binary(c5, n, target, rng)
        candidates.append(c5)
        # Coarse optimization
        if tleft() > 200:
            n_c = 80
            target_c = n_c / 2.0
            h_c = interpolate_to(best_h, n_c, target_c)
            h_c_bin = to_binary(h_c, n_c, target_c, rng)
            h_c_bin, _ = hill_climb_binary(h_c_bin, n_c, target_c, min(20, tleft() * 0.1), rng, mode='first', max_rounds=200)
            c6 = to_binary(interpolate_to(h_c_bin, n, target), n, target, rng)
            candidates.append(c6)

    # Phase 2: Hill climbing on each candidate
    for ci, cand in enumerate(candidates):
        if tleft() < 30:
            break
        t_alloc = min(30, tleft() * 0.15)
        hc, sc = hill_climb_binary(cand, n, target, t_alloc, rng, mode='best', max_rounds=30)
        if sc < best_score:
            best_score = sc
            best_h = hc

    # Phase 3: SA on best binary and a few others
    binary_starts = [best_h]
    for cand in candidates[:3]:
        binary_starts.append(to_binary(cand, n, target, rng))

    for si, bs in enumerate(binary_starts):
        if tleft() < 60:
            break
        t_alloc = min(120, tleft() * 0.3)
        # First quick hill climb
        bs_hc, bs_sc = hill_climb_binary(bs, n, target, min(15, t_alloc * 0.2), rng, mode='first', max_rounds=200)
        if bs_sc < best_score:
            best_score = bs_sc
            best_h = bs_hc
        # Then SA
        sa_t = min(t_alloc, tleft() * 0.4)
        hs, ss = sa_binary(bs_hc, n, target, sa_t, rng, t_init=0.003 + 0.001 * si)
        if ss < best_score:
            best_score = ss
            best_h = hs

    # Phase 4: Additional SA rounds on best with different temps
    for round_idx in range(3):
        if tleft() < 60:
            break
        t_alloc = min(80, tleft() * 0.25)
        t_init_val = 0.002 * (0.5 ** round_idx)
        hs, ss = sa_binary(best_h, n, target, t_alloc, rng, t_init=t_init_val)
        if ss < best_score:
            best_score = ss
            best_h = hs

    # Phase 5: Continuous refinement via SLSQP
    if tleft() > 80:
        h_cont = best_h.copy()
        h_cont = project_feasible(h_cont, target)
        t_alloc = min(150, tleft() * 0.3)
        # Time-based: run SLSQP with increasing alpha
        alphas = [50, 200, 1000, 5000, 20000]
        per_alpha = t_alloc / len(alphas)
        for alpha in alphas:
            if tleft() < 20:
                break
            t_step = min(per_alpha, tleft() * 0.3)
            max_it = max(500, int(t_step * 500))
            h_new = slsqp_run(h_cont, n, target, [alpha], maxiter=max_it, ftol=1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cont = h_new

    # Phase 6: LP refinement
    if tleft() > 60:
        t_alloc = min(80, tleft() * 0.3)
        h_lp, c5_lp = lp_refined(best_h, n, target, max_iter=2000,
                                  time_limit=t_alloc, n_top=120,
                                  damping=0.9, reg=1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Round to binary, SA again, then refine
    if tleft() > 60:
        h_bin = to_binary(best_h, n, target, rng)
        h_bin, sc_bin = hill_climb_binary(h_bin, n, target, min(20, tleft() * 0.2), rng, mode='first', max_rounds=100)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
        if tleft() > 40:
            hs, ss = sa_binary(h_bin, n, target, min(40, tleft() * 0.3), rng, t_init=0.002)
            if ss < best_score:
                best_score = ss
                best_h = hs

    # Phase 8: Final SLSQP polish
    if tleft() > 40:
        h_final = best_h.copy()
        h_final = project_feasible(h_final, target)
        alphas_final = [1000, 10000, 50000]
        for alpha in alphas_final:
            if tleft() < 15:
                break
            h_new = slsqp_run(h_final, n, target, [alpha], maxiter=1500, ftol=1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_final = h_new

    # Phase 9: Final LP polish
    if tleft() > 30:
        t_alloc = min(25, tleft() * 0.6)
        h_lp2, c5_lp2 = lp_refined(best_h, n, target, max_iter=1000,
                                    time_limit=t_alloc, n_top=100,
                                    damping=0.95, reg=1e-6)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 10: Try higher resolution if time
    if tleft() > 150:
        for n_h in [500, 600]:
            if tleft() < 100:
                break
            target_h = n_h / 2.0
            h_hi = interpolate_to(best_h, n_h, target_h)
            h_hi, c5_hi = lp_refined(h_hi, n_h, target_h, max_iter=1500,
                                      time_limit=min(60, tleft() * 0.3), n_top=150,
                                      damping=0.9, reg=1e-4)
            h_hi = slsqp_run(h_hi, n_h, target_h, [500, 5000], maxiter=1000)
            sc = compute_c5(h_hi)
            if sc < best_score:
                best_score = sc
                best_h = h_hi
                break

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
