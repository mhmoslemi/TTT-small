def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_f_and_g(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    fval = cmax + np.log(np.sum(np.exp(shifted))) / alpha
    s = float(np.sum(x))
    fval += mu * (s - target) ** 2
    w = np.exp(shifted)
    ws = w.sum()
    if ws > 1e-300:
        w /= ws
    else:
        w = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for ki in active:
        wi = w[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += wi * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= wi * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += wi * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= wi * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return fval, grad

    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: lse_f_and_g(x, alpha, target, mu)[0],
                       h.copy(),
                       jac=lambda x: lse_f_and_g(x, alpha, target, mu)[1],
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 30})
        return project_feasible(res.x, target)
    except Exception:
        return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)

def fast_coord_paste(h, target, n_inner=30):
    """One pass of fast coordinate descent maintaining sum constraint via projection at end."""
    n = len(h)
    h_cur = h.copy()
    corr = get_corr(h_cur)
    
    for j in range(n):
        old_val = h_cur[j]
        m_range = np.arange(-(n-1), n)
        coeff = np.zeros(2 * n - 1)
        vp = (j + m_range >= 0) & (j + m_range < n)
        coeff[vp] += (1.0 - h_cur[j + m_range[vp]])
        vn = (j - m_range >= 0) & (j - m_range < n)
        coeff[vn] -= h_cur[j - m_range[vn]]
        
        # Binary search for optimal t in [-old_val, 1-old_val]
        t_lo = -old_val
        t_hi = 1.0 - old_val
        for _ in range(n_inner):
            t1 = t_lo + (t_hi - t_lo) * 0.3333
            t2 = t_hi - (t_hi - t_lo) * 0.3333
            f1 = np.max(corr + t1 * coeff)
            f2 = np.max(corr + t2 * coeff)
            if f1 < f2:

            else:

            h_cur[j] = v_opt
            corr = corr + (v_opt - old_val) * coeff

    return h_cur

def topk_subgrad_step(h, target, n_top, step):
    n = len(h)
    corr = get_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_top]
    top_vals = corr[top_idx]
    spread = top_vals[0] - top_vals[-1]
    if spread < 1e-12:
        weights = np.ones(n_top) / n_top
    else:
        weights = (top_vals - top_vals[-1] + 1e-10) / (spread + 1e-10)
        weights /= weights.sum()
    grad = np.zeros(n)
    for i, ki in enumerate(top_idx):
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += weights[i] * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= weights[i] * h[j2 + lag]
        else:

            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= weights[i] * h[j2 - m]

        sc = compute_c5(h_new)
        if sc < cur:

            cur = sc
    return h_cur, cur

        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_f_and_g(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    fval = cmax + np.log(np.sum(np.exp(shifted))) / alpha
    s = float(np.sum(x))
    fval += mu * (s - target) ** 2
    w = np.exp(shifted)
    ws = w.sum()
    if ws > 1e-300:
        w = w / ws
    else:
        w = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for ki in active:
        wi = w[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += wi * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= wi * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += wi * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= wi * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return fval, grad

def lbfgs_opt(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: lse_f_and_g(x, alpha, target, mu)[0],
                       h.copy(),
                       jac=lambda x: lse_f_and_g(x, alpha, target, mu)[1],
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 30})
        return project_feasible(res.x, target)
    except Exception:
        return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)

def fast_coord_descent(h, target, n_inner=35):
    n = len(h)
    h_cur = h.copy()
    corr = get_corr(h_cur)
    m_range = np.arange(-(n - 1), n)
    for j in range(n):
        old_val = h_cur[j]
        coeff = np.zeros(2 * n - 1)
        vp = (j + m_range >= 0) & (j + m_range < n)
        coeff[vp] += (1.0 - h_cur[j + m_range[vp]])
        vn = (j - m_range >= 0) & (j - m_range < n)
        coeff[vn] -= h_cur[j - m_range[vn]]
        t_lo = -old_val
        t_hi = 1.0 - old_val
        for _ in range(n_inner):
            t1 = t_lo + (t_hi - t_lo) / 3.0
            t2 = t_hi - (t_hi - t_lo) / 3.0
            f1 = np.max(corr + t1 * coeff)
            f2 = np.max(corr + t2 * coeff)
            if f1 < f2:
                t_hi = t2
            else:
                t_lo = t1
        t_opt = (t_lo + t_hi) / 2.0
        v_opt = np.clip(t_opt + old_val, 0.0, 1.0)
        if abs(v_opt - old_val) > 1e-12:
            h_cur[j] = v_opt
            corr = corr + (v_opt - old_val) * coeff
    h_cur = project_feasible(h_cur, target)
    return h_cur

def topk_subgrad(h, target, n_top, step):
    n = len(h)
    corr = get_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_top]
    top_vals = corr[top_idx]
    spread = top_vals[0] - top_vals[-1]
    if spread < 1e-12:
        weights = np.ones(n_top) / n_top
    else:
        weights = (top_vals - top_vals[-1] + 1e-10) / (spread + 1e-10)
        weights /= weights.sum()
    grad = np.zeros(n)
    for i, ki in enumerate(top_idx):
        wi = weights[i]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += wi * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= wi * h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += wi * (1.0 - h[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= wi * h[j2 - m]
    h_new = h - step * grad
    return project_feasible(h_new, target)

def annealed_topk(h, target, n_top, step0, step1, n_iter, deadline):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        frac = i / max(1, n_iter - 1)
        step = step0 * (step1 / step0) ** frac
        h_new = topk_subgrad(h_cur, target, n_top, step)
        sc = compute_c5(h_new)
        if sc < cur:
            h_cur = h_new
            cur = sc
    return h_cur, cur

def binary_snap(h, target):
    n = len(h)
    best = h.copy()
    best_sc = compute_c5(best)
    for thresh in [0.05, 0.1, 0.15, 0.2, 0.3]:
        h_snap = h.copy()
        h_snap[h_snap < thresh] = 0.0
        h_snap[h_snap > 1.0 - thresh] = 1.0
        h_snap = project_feasible(h_snap, target)
        sc = compute_c5(h_snap)
        if sc < best_sc:
            best = h_snap
            best_sc = sc
    return best, best_sc
