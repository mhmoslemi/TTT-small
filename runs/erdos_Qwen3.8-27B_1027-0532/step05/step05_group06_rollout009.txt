If the maximum is achieved at lag k, we can try to simultaneously reduce corr_k and corr_{-k} (since they're related by the symmetry of correlation). This gives a 2-constraint problem that might be easier to solve.

**Yet another idea: "Binary relaxation"**

The optimal solution might be close to binary. Let me:
1. Round h to the nearest binary (with the constraint ∑h = n/2)
2. Optimize the binary solution using local search (swap pairs of 0s and 1s)
3. Use the binary solution as a starting point for continuous optimization

The binary local search is:
- For each pair (i, j) where h[i]=1 and h[j]=0, try swapping them
- Accept if it improves the max correlation
- This is O(n²) per sweep but very effective for binary problems

```python
def binary_local_search(h, target, n_sweeps=10, deadline=None):
    """Local search on the binary relaxation."""
    n = len(h)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix the sum constraint
    ones = int(h_bin.sum())
    target_ones = int(target)  # n/2
    if ones > target_ones:
        # Need to flip some 1s to 0s
        set_to_one = np.where(h_bin == 1)[0]
        # Flip the ones where h was smallest
        order = np.argsort(h[set_to_one])
        flip_count = ones - target_ones
        for i in range(flip_count):
            h_bin[set_to_one[order[i]]] = 0
    elif ones < target_ones:
        set_to_zero = np.where(h_bin == 0)[0]
        order = np.argsort(-h[set_to_zero])
        flip_count = target_ones - ones
        for i in range(flip_count):
            h_bin[set_to_zero[order[i]]] = 1
    
    best_score = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones_indices = np.where(h_bin == 1)[0]
        zeros_indices = np.where(h_bin == 0)[0]
        
        for i_idx in range(len(ones_indices)):
            for j_idx in range(len(zeros_indices)):
                i = ones_indices[i_idx]
                j = zeros_indices[j_idx]
                h_test = h_bin.copy()
                h_test[i] = 0
                h_test[j] = 1
                sc = compute_c5(h_test)
                if sc < best_score - 1e-14:
                    h_bin = h_test
                    best_score = sc
                    best_h = h_bin.copy()
                    improved = True
                    # Update indices
                    ones_indices = np.where(h_bin == 1)[0]
                    zeros_indices = np.where(h_bin == 0)[0]
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-16, 'maxls': 80})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def adaptive_subgradient(h, target, n_iters, lr_base, momentum, n_active, deadline):
    h_cur = h.copy()
    n = len(h)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    vel = np.zeros(n)
    lr = lr_base
    patience = 0
    max_patience = 200

    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_active]
        top_vals = corr[top_idx]
        shifted_tv = top_vals - top_vals.max()
        weights = np.exp(shifted_tv * 5.0)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, ki in zip(weights, top_idx):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]

        s = np.sum(h_cur)
        grad += 2.0 * 200.0 * (s - target)

        vel = momentum * vel - lr * grad
        h_new = h_cur + vel
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_new.copy()
            patience = 0
            lr = min(lr * 1.01, lr_base * 3.0)
        else:
            patience += 1
            lr *= 0.97
            if patience > max_patience:
                break
        h_cur = h_new

    return best_h, best_score


def pnorm_subgradient(h, target, p_vals, n_iters_per_p, lr, deadline):
    h_cur = h.copy()
    n = len(h)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    for p in p_vals:
        if time.time() > deadline:
            break
        for it in range(n_iters_per_p):
            if time.time() > deadline:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            corr_pos = np.maximum(corr, 1e-30)
            weights = corr_pos ** p
            wsum = weights.sum()
            if wsum < 1e-300:
                weights = np.ones(len(corr)) / len(corr)
            else:
                weights = weights / wsum

            grad = np.zeros(n)
            active = np.where(weights > 1e-8)[0]
            for ki in active:
                w = weights[ki]
                lag = int(ki) - (n - 1)
                if lag >= 0:
                    j1 = np.arange(lag, n)
                    grad[j1] += w * (1.0 - h_cur[j1 - lag])
                    if lag > 0:
                        j2 = np.arange(0, n - lag)
                        grad[j2] -= w * h_cur[j2 + lag]
                else:
                    m = -lag
                    j1 = np.arange(0, n - m)
                    grad[j1] += w * (1.0 - h_cur[j1 + m])
                    if m > 0:
                        j2 = np.arange(m, n)
                        grad[j2] -= w * h_cur[j2 - m]

            s = np.sum(h_cur)
            grad += 2.0 * 200.0 * (s - target)

            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new

    return best_h, best_score


def coordinate_descent(h, target, n_sweeps=5, max_inner=30):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(max_inner):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def binary_round_search(h, target, n_trials=200, deadline=None):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    target_ones = int(round(target))

    for trial in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        h_bin = (h_cur > 0.5).astype(float)
        ones = int(h_bin.sum())
        if ones > target_ones:
            set_one = np.where(h_bin == 1)[0]
            order = np.argsort(h_cur[set_one])
            for i in range(ones - target_ones):
                h_bin[set_one[order[i]]] = 0.0
        elif ones < target_ones:
            set_zero = np.where(h_bin == 0)[0]
            order = np.argsort(-h_cur[set_zero])
            for i in range(target_ones - ones):
                h_bin[set_zero[order[i]]] = 1.0

        improved = True
        while improved:
            if deadline is not None and time.time() > deadline - 5:
                break
            improved = False
            ones_idx = np.where(h_bin == 1)[0]
            zeros_idx = np.where(h_bin == 0)[0]
            for i in ones_idx:
                for j in zeros_idx:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    sc = compute_c5(h_bin)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_h = h_bin.copy()
                        improved = True
                        break
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
                if improved:
                    break
        h_cur = project_feasible(0.5 * (best_h + h_cur), target)

    return best_h, best_score
```
