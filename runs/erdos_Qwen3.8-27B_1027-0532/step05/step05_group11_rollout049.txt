        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        n_half = n - 1
        
        # Find active lags (top K)
        K = 20
        top_idx = np.argsort(overlaps)[::-1][:K]
        max_val = overlaps[top_idx[0]]
        
        # Compute subgradient (average of gradients of active lags)
        g = np.zeros(n)

            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:

                # grad[j] = dx * (1 - 2*h[j])

                # h[j] in (1-h[i+k]) part: j-k valid if for k>0: j >= k, for k<0: j <= n-1-m
                
                if k > 0:
                    # j in h[i]: j from 0 to n-1-k, contributes (1-h[j+k])
                    # j in (1-h[i+k]): i=j-k, j from k to n-1, contributes -h[j-k]

                    for j in range(k, n):
                        g[j] += dx * (-h[j - k])
                else:
                    # k < 0, m = -k
                    # j in h[i]: j from m to n-1, contributes (1-h[j-m])
                    # j in (1-h[i-m]): i=j+m, j from 0 to n-1-m, contributes -h[j+m]
                    for j in range(m, n):
                        g[j] += dx * (1.0 - h[j - m])
                    for j in range(n - m):

    else:

        return h

    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

    """Build LP with linearized constraints at fixed point h_fixed"""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0

    for lag in lags:
        k = lag
        abs_k = abs(k)
        if abs_k == 0:
            coeff = (1.0 - h_fixed) * dx
            constraints.append(coeff @ h_var <= t_var)
        elif k > 0:

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_subgradient(h, active_indices, n):
    dx = 2.0 / n
    n_half = n - 1
    g = np.zeros(n)
    for idx in active_indices:
        k = idx - n_half
        abs_k = abs(k)
        if abs_k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n - k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n - k]) * dx
        else:
            m = -k
            g[m:] += (1.0 - h[:n - m]) * dx
            g[:n - m] += (-h[m:]) * dx
    return g / len(active_indices)


def frank_wolfe(h_init, n, target, time_limit, max_iter=400000, K=40):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = compute_subgradient(h, top_idx, n)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 20000 and t - best_t > 150000:
            break
    return best_h, best_c5


def frank_wolfe_adaptive(h_init, n, target, time_limit, max_iter=400000, K=40):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        current_max = float(np.max(overlaps))
        if current_max <= best_c5 + 1e-14:
            best_c5 = current_max
            best_h = h.copy()
            best_t = t
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = compute_subgradient(h, top_idx, n)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        h = h_new
        if t > 30000 and t - best_t > 200000:
            break
    return best_h, best_c5


def build_lp_linear(h_fixed, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        abs_k = abs(k)
        if abs_k == 0:
            coeff = (1.0 - h_fixed) * dx
            constraints.append(coeff @ h_var <= t_var)
        elif k > 0:
            coeff = np.zeros(n)
            coeff[:n - k] = (1.0 - h_fixed[k:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        else:
            m = -k
            coeff = np.zeros(n)
            coeff[m:] = (1.0 - h_fixed[:n - m]) * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints


def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=250,
                   damping_start=0.9, damping_end=0.8,
                   reg_start=1e-3, reg_end=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 100 and stuck > 200:
            break
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        h_var, t_var, constraints = build_lp_linear(g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _g_target


def cons_jac(h):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter=2000, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def random_binary_init(n, target, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:int(target)]] = 1.0
    return h


def structured_init(n, target, rng, shift):
    idx = np.arange(n)
    h = np.zeros(n)
    block = int(target // 4)
    offsets = [0, block, 2 * block, 3 * block]
    for off in offsets:
        start = (off + shift) % n
        for j in range(block):
            h[(start + j) % n] = 1.0
    return project_feasible(h, target)


def perturb_block(h, n, target, rng, block_len, n_perturb):
    h_new = h.copy()
    for _ in range(n_perturb):
        start = rng.randint(0, n - block_len)
        h_new[start:start + block_len] = 1.0 - h_new[start:start + block_len]
    return project_feasible(h_new, target)


def time_remaining(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)
