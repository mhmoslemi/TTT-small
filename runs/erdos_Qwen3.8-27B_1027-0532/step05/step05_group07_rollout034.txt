def nesterov_smooth(h, n, target, deadline, alpha, learning_rate, momentum):
    dx = 2.0 / n
    h_t = h.copy()
    h_prev = h.copy()
    mu = momentum  # momentum parameter (0.9 typically)
    
    for t in range(max_iters):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead
        y = h_t + mu * (h_t - h_prev)
        y = np.clip(y, 0, 1)
        
        # Compute gradient of smooth approximation
        corr = compute_corr(y) * dx
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700, 0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Gradient of smooth approximation = weighted sum of correlation gradients
        grad = np.zeros(n)
        for k, w in enumerate(weights):
            if w < 1e-10:
                continue
            grad += w * corr_gradient(y, k, n)
        
        # Gradient step
        h_t_new = y - learning_rate * grad
        h_t_new = project_feasible(h_t_new, target)
        
        # Update momentum
        h_prev = h_t
        h_t = h_t_new
    
    return h_t
```

4. Then project: h_proj = clip(h - λ, 0, 1)

def toeplitz_matvec(t_vec, h, n):
    """
    Compute A @ h where A is Toeplitz with t[d] as the (d+1)-th diagonal.
    t_vec has length 2n-1, t_vec[k] = t[k-(n-1)] for k=0,...,2n-2.
    """
    from scipy.fft import fft, ifft
    
    # First row: t[0], t[1], ..., t[n-1] = t_vec[n-1], t_vec[n], ..., t_vec[2n-2]
    first_row = t_vec[n-1:2*n-1]
    # First column: t[0], t[-1], ..., t[-(n-1)] = t_vec[n-1], t_vec[n-2], ..., t_vec[0]
    first_col = t_vec[0:n][::-1]
    
    # FFT-based Toeplitz mat-vec
    N = 2 * n - 1
    c_ext = np.zeros(N)
    c_ext[0:n] = first_col
    c_ext[n:] = first_row[1:][::-1]  # r[1], r[2], ..., r[n-1] reversed
    
    h_ext = np.zeros(N)
    h_ext[0:n] = h
    
    result = ifft(fft(c_ext) * fft(h_ext))[n-1:2*n-1]
    return result.real
```
