    """Compute F_alpha(h) = (1/alpha) * log(sum(exp(alpha * O_k)))"""
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/len(h))
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

    n = len(h)

    # Actually corr has length 2n-1, index 0 corresponds to lag -(n-1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):

    corr = corr * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)

        w[max_idx] = 1.0
    else:
        w /= sw

        hi2 = min(n, n - k)

            grad[lo2:hi2] -= wk * h[np.arange(lo2, hi2) + k]
    return grad

    h = project_feasible(h, target)

    best_h = h.copy()
    best_val = compute_c5(h)
    v = np.zeros(n)

        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (t-1)**2)) / 2.0

            if compute_c5(h_new) < compute_c5(h):

        except Exception:
            pass

    # = sum_i h[i+k]*w_signal[i]

    # We want A[m] = sum_k w_at_lag_k * h[m-k] = sum_k w_signal[n-1+k] * h[m-k]
    # Let j = m-k: = sum_j w_signal[n-1+m-j] * h[j]
    # From np.correlate: sum_m h[m] * w_signal[m-k] 
    # Setting m -> j: sum_j h[j] * w_signal[j-k]
    # We want: sum_j h[j] * w_signal[n-1+m-j]
    # So we need j-k = n-1+m-j, i.e., k = 2j-n-1+m. Not a constant. 
    
    # OK I give up on the FFT approach for now. The direct method is fine for n=400.
    
    # Just do the direct computation with a precomputed structure:
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    
    for j in range(len(lags)):
        if w[j] < 1e-14:
            continue
        k = lags[j]
        wk = w[j] * dx
        # (1-h[m-k]) term: valid when 0<=m-k<n, i.e., k<=m<k+n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            grad[lo:hi] += wk * (1.0 - h[np.arange(lo, hi) - k])
        # -h[m+k] term: valid when 0<=m+k<n, i.e., -k<=m<n-k
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[np.arange(lo2, hi2) + k]
    
    return grad

def step_function_overlaps(steps, values, n_eval=2000):
    """Compute the maximum overlap for a step function.
    steps: array of breakpoints [x_0, x_1, ..., x_M] where x_0=0, x_M=2
    values: array of function values [v_0, v_1, ..., v_{M-1}]
    """
    n = n_eval
    x = np.linspace(0, 2, n)
    h = np.searchsorted(steps[1:], x, side='right')  # index into values
    h = values[np.clip(h, 0, len(values)-1)]

        mask = (x >= lo) & (x < hi)
        h[mask] = values[i]

This can be computed exactly for each pair (i,j), giving O(M²) work per lag, and O(M² * num_lags) total.
