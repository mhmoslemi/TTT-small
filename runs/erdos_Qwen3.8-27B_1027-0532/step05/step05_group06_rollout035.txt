def structured_perturb(h, rng, target, mode):
    n = len(h)
    h_p = h.copy()
    if mode == 'gauss':
        h_p += rng.randn(n) * 0.1
    elif mode == 'flip':
        i1 = rng.randint(0, n // 2)
        i2 = rng.randint(i1 + n // 10, n)
        h_p[i1:i2] = 1.0 - h_p[i1:i2]
    elif mode == 'mirror':
        i1 = rng.randint(0, n // 3)
        i2 = rng.randint(i1 + n // 10, n)
        h_p[i1:i2] = h_p[i1:i2][::-1]
    elif mode == 'shift':
        sh = rng.randint(1, n // 8 + 1)
        h_p = np.roll(h_p, sh)
    elif mode == 'smooth':
        ksize = 5
        kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2)
        kernel /= kernel.sum()
        h_p = np.convolve(h_p, kernel, mode='same')
    elif mode == 'binary':
        h_p = (h_p > 0.5).astype(float)
    elif mode == 'sparse':
        thresh = np.sort(h_p)[int(n * 0.35)]
        h_p[h_p < thresh] = 0.0
    elif mode == 'gauss2':
        h_p += rng.randn(n) * 0.25
    return project_feasible(np.clip(h_p, 0.0, 1.0), target)

    alphas_1 = [3, 10, 30, 100, 400, 1500, 6000, 25000, 100000]
    for a in alphas_1:
        if time.time() > deadline - 100:
            break
        h_new = lbfgs_lse(best_h, target, a, mu, 1000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Top-k optimization (new approach)
    if time.time() < deadline - 80:
        h_tk, sc_tk = topk_optimize(best_h, target, mu, k=10,
                                    n_rounds=8, maxiter_each=400,
                                    deadline=deadline)
        if sc_tk < best_score:
            best_score = sc_tk
            best_h = h_tk
        h_tk, sc_tk = topk_optimize(best_h, target, mu, k=5,
                                    n_rounds=6, maxiter_each=400,
                                    deadline=deadline)
        if sc_tk < best_score:
            best_score = sc_tk
            best_h = h_tk
        h_tk, sc_tk = topk_optimize(best_h, target, mu, k=3,
                                    n_rounds=5, maxiter_each=300,
                                    deadline=deadline)
        if sc_tk < best_score:
            best_score = sc_tk
            best_h = h_tk

    # Phase 3: Subgradient phase
    if time.time() < deadline - 80:
        h_sg, sc_sg = subgradient_phase(best_h, target, n_top=5,
                                        n_steps=300, deadline=deadline)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 4: L-BFGS polish at high alpha
    if time.time() < deadline - 60:
        h_new = lbfgs_lse(best_h, target, 50000, mu, 1500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Diverse restarts
    modes_list = ['gauss', 'flip', 'mirror', 'smooth', 'binary',
                  'sparse', 'gauss2', 'shift', 'flip', 'mirror', 'gauss', 'sparse']
    for mode in modes_list:
        if time.time() > deadline - 200:
            break
        h_pert = structured_perturb(best_h, rng, target, mode)
        h_new = lbfgs_lse(h_pert, target, 100, mu, 300)
        h_new = lbfgs_lse(h_new, target, 5000, mu, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if time.time() > deadline - 200:
            break

    # Phase 6: Top-k after restarts
    if time.time() < deadline - 60:
        h_tk, sc_tk = topk_optimize(best_h, target, mu, k=5,
                                    n_rounds=5, maxiter_each=300,
                                    deadline=deadline)
        if sc_tk < best_score:
            best_score = sc_tk
            best_h = h_tk

    # Phase 7: Upsample to 2x
    n2 = 2 * n
    target2 = n2 / 2.0
    if time.time() < deadline - 350:
        h_fine = upsample(best_h, n2)
        alphas_fine = [10, 50, 200, 1000, 5000, 20000]
        for a in alphas_fine:
            if time.time() > deadline - 100:
                break
            h_new = lbfgs_lse(h_fine, target2, a, mu, 500)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        if time.time() < deadline - 60:
            h_tk, sc_tk = topk_optimize(best_h, target2, mu, k=8,
                                        n_rounds=5, maxiter_each=300,
                                        deadline=deadline)
            if sc_tk < best_score:
                best_score = sc_tk
                best_h = h_tk

    # Phase 8: Try 4x resolution
    n4 = 4 * n
    target4 = n4 / 2.0
    if time.time() < deadline - 400:
        h_quad = upsample(best_h, n4)
        alphas_quad = [20, 100, 500, 3000, 10000]
        for a in alphas_quad:
            if time.time() > deadline - 100:
                break
            h_new = lbfgs_lse(h_quad, target4, a, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        if time.time() < deadline - 60:
            h_tk, sc_tk = topk_optimize(best_h, target4, mu, k=6,
                                        n_rounds=3, maxiter_each=200,
                                        deadline=deadline)
            if sc_tk < best_score:
                best_score = sc_tk
                best_h = h_tk

    # Phase 9: Final coordinate descent
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if time.time() < deadline - 80:
        h_cd, sc_cd = coordinate_descent(best_h, target_cur, n_sweeps=2,
                                         deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 10: SLSQP polish
    if time.time() < deadline - 40:
        h_sl = slsqp_polish(best_h, target_cur, 5000, 2000)
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    # Phase 11: Final top-k + L-BFGS
    if time.time() < deadline - 30:
        h_tk, sc_tk = topk_optimize(best_h, target_cur, mu, k=3,
                                    n_rounds=3, maxiter_each=200,
                                    deadline=deadline)
        if sc_tk < best_score:
            best_score = sc_tk
            best_h = h_tk
        h_new = lbfgs_lse(best_h, target_cur, 100000, mu, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
