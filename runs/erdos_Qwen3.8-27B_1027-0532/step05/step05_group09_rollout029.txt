from scipy.optimize import minimize, golden
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    # Projection onto {h : 0 <= h <= 1, sum(h) = target}
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k < n:
            lower = max(h_sorted[k-1] - 1.0, h_sorted[k])
        else:
            lower = h_sorted[k-1] - 1.0
        upper = h_sorted[k-1]
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            return np.clip(h - tau, 0.0, 1.0)
    # Bisection fallback
    lo, hi = h.min() - 1.0, h.max() + 0.5
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def golden_section_1d(f, a, b, tol=1e-10, max_iter=80):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ≈ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

def lse_value(overlaps, alpha):
    """Compute log-sum-exp of overlaps scaled by alpha."""
    m = np.max(overlaps)
    return m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha

def full_1d_cs(h, target, alpha, mu, max_sweeps=50, deadline=None):
    """Full 1D coordinate descent with golden section search."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    overlaps_cur = compute_overlaps(h_cur)
    sum_cur = np.sum(h_cur)
    obj_cur = lse_value(overlaps_cur, alpha) + mu * (sum_cur - target)**2
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            # Compute coefficients for how each overlap changes with h[i]
            old_hi = h_cur[i]
            if old_hi < 1e-12 or old_hi > 1.0 - 1e-12:
                continue  # at boundary, skip for now
            
            # coeff[k] = dx * [(1-h[i+k]) * 1(k<=n-1-i) - h[i-k] * 1(k<=i)]
            # For k=0: dx * (1-2*old_hi)
            # For k>0: 
            #   if k <= n-1-i: += dx*(1-h[i+k])
            #   if k <= i: += -dx*h[i-k]
            
            # To save computation, let's precompute the effect
            # new_overlaps[k] = overlaps_cur[k] + coeff[k] * delta
            # where delta = new_hi - old_hi
            
            # Objective as function of delta:
            #   g(delta) = lse_value(overlaps_cur + coeff*delta, alpha) + mu*(sum_cur + delta - target)^2
            #   domain: delta in [-old_hi, 1-old_hi]
            
            # Build coeff array
            coeff = np.zeros(n)
            idx = np.arange(n)
            mask1 = idx <= (n - 1 - i)  # k <= n-1-i
            mask2 = idx <= i  # k <= i
            # For mask1: coeff[k] += dx * (1 - h[i+k])
            # For mask2: coeff[k] += -dx * h[i-k]
            # Need to be careful: h[i+k] valid when i+k < n, i.e., k < n-i, i.e., k <= n-1-i ✓
            # h[i-k] valid when i-k >= 0, i.e., k <= i ✓
            
            k_valid1 = np.where(mask1)[0]
            k_valid2 = np.where(mask2)[0]
            
            # coeff for k in valid1: dx * (1 - h[i+k])
            coeff[k_valid1] += dx * (1.0 - h_cur[i + k_valid1])
            # coeff for k in valid2: -dx * h[i-k]
            coeff[k_valid2] += -dx * h_cur[i - k_valid2]
            
            # For k=0, both masks include it, so coeff[0] = dx*(1-h[i]) - dx*h[i] = dx*(1-2h[i]) ✓
            
            a_bound = -old_hi
            b_bound = 1.0 - old_hi
            
            def g(delta):
                new_overlaps = overlaps_cur + coeff * delta
                new_sum = sum_cur + delta
                return lse_value(new_overlaps, alpha) + mu * (new_sum - target)**2
            
            # Find minimum
            best_delta = golden_section_1d(g, a_bound, b_bound, tol=1e-10, max_iter=60)
            
            if abs(best_delta) > 1e-12:
                h_cur[i] = old_hi + best_delta
                # Update overlaps
                overlaps_cur = overlaps_cur + coeff * best_delta
                sum_cur += best_delta
                obj_cur = lse_value(overlaps_cur, alpha) + mu * (sum_cur - target)**2
                improved = True
        
        if not improved:
            break
    
    # Final projection
    h_cur = project_feasible(h_cur, target)
    final_score = compute_c5(h_cur)
    return h_cur, final_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k < n:
            lower = max(h_sorted[k-1] - 1.0, h_sorted[k])
        else:
            lower = h_sorted[k-1] - 1.0
        upper = h_sorted[k-1]
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            return np.clip(h - tau, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def lse_smooth(overlaps, alpha):
    m = np.max(overlaps)
    if alpha * (m - np.min(overlaps)) > 500:
        # Too large, just use max
        return m
    shifted = alpha * (overlaps - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def compute_grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full')[n-1:] * dx
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)

    # d(overlap_k)/d(h[i]) = dx * [(1-h[i+k]) if i+k < n AND i+k valid] - dx * [h[i-k] if i-k >= 0]

    #   If i in [0, n-1-k]: dx*(1-h[i+k])
    #   If i-k in [0, n-1-k] (i.e., k <= i): -dx*h[i-k]

def lbfgs_phase(h, target, alpha, mu, time_limit):

            d = a_lo + gr * (a_hi - a_lo)

                    a_lo = c

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k < n:
            lower = max(h_sorted[k-1] - 1.0, h_sorted[k])
        else:
            lower = h_sorted[k-1] - 1.0
        upper = h_sorted[k-1]
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            return np.clip(h - tau, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lse_smooth(overlaps, alpha):

    spread = alpha * (m - np.min(overlaps))
    if spread > 500.0:
        return m
    shifted = np.clip(alpha * (overlaps - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def fista_short(h0, target, alpha, max_iters, deadline):

    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    lr = 1.0 / (alpha * 0.05) if alpha > 1 else 1.0
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_c = np.clip(y, 0.0, 1.0)
        overlaps = np.correlate(y_c, 1.0 - y_c, mode='full')[n-1:] * (2.0/n)
        m = np.max(overlaps)

        se = np.exp(e)

        omh = 1.0 - y_c

        h_new, sc = coord_descent_golden(best_h, target, 50000, deadline - 5, mu=1.0)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
