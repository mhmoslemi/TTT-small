        val = c_max + float(np.log(wsum)) / alpha
        grad = np.zeros(n)
        for ii, ki in enumerate(idx):
            w = weights[ii]
            if w < 1e-12:
                continue
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * x[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - x[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * x[j2 - m]
    else:
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum > 1e-300:
            weights = weights / wsum
        else:
            weights = np.ones(len(corr)) / len(corr)
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for ki in active:
            w = weights[ki]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * x[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - x[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, top_k=None):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu, top_k)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu, top_k)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-12, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def active_set_descent(h, target, n_iters=300, tol_frac=1e-3):
    n = len(h)
    h_cur = h.copy()
    best_score, _ = compute_c5(h_cur)
    lr = 0.005
    momentum = 0.3
    vel = np.zeros(n)
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        threshold = c_max - tol_frac * c_max
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:5]
        elif len(active) > 30:
            active = np.argsort(corr)[::-1][:30]
        grad = np.zeros(n)
        for ki in active:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= len(active)
        vel = momentum * vel - lr * grad
        h_try = project_feasible(h_cur + vel, target)
        sc_try, _ = compute_c5(h_try)
        if sc_try < best_score:
            h_cur = h_try
            best_score = sc_try
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.85
            vel *= 0.5
            if lr < 1e-10:
                break
    return h_cur, best_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def coordinate_descent(h, target, n_sweeps=5):
    n = len(h)
    h_cur = h.copy()
    cur_score, _ = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_tmp = h_cur.copy()
                h_tmp[j] = v
                h_tmp = project_feasible(h_tmp, target)
                corr_tmp = np.correlate(h_tmp, 1.0 - h_tmp, mode='full')
                return float(np.max(corr_tmp))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)[0]
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def paired_lag_optimize(h, target, max_iter=100):
    n = len(h)
    h_cur = h.copy()
    best_score, _ = compute_c5(h_cur)
    lr = 0.003
    for it in range(max_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        sorted_idx = np.argsort(corr)[::-1]
        top2 = sorted_idx[:2]
        c1, c2 = corr[top2[0]], corr[top2[1]]
        if c2 > c1 * 0.999:
            w1, w2 = 0.5, 0.5
        else:
            w1, w2 = 0.7, 0.3
        grad = np.zeros(n)
        for ki, w in [(top2[0], w1), (top2[1], w2)]:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_try = project_feasible(h_cur - lr * grad, target)
        sc_try, _ = compute_c5(h_try)
        if sc_try < best_score:
            h_cur = h_try
            best_score = sc_try
        else:
            lr *= 0.8
            if lr < 1e-12:
                break
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score, _ = compute_c5(best_h)
    mu = 200.0

    # Phase 1: Progressive LSE optimization
    alphas = [5, 20, 100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if tleft() < 30:
            break
        maxiter = 1200 if a < 1000 else 800
        tk = None
        if a > 5000:
            tk = 50
        h_new = lbfgs_optimize(best_h, target, a, mu, maxiter, top_k=tk)
        sc, _ = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Active-set descent with momentum
    if tleft() > 150:
        h_as, sc_as = active_set_descent(best_h, target, n_iters=400, tol_frac=2e-3)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as
        h_new = lbfgs_optimize(best_h, target, 50000, mu, 600, top_k=30)
        sc, _ = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Paired lag optimization
    if tleft() > 80:
        h_pl, sc_pl = paired_lag_optimize(best_h, target, max_iter=200)
        if sc_pl < best_score:
            best_score = sc_pl
            best_h = h_pl

    # Phase 4: More active-set with tighter tolerance
    if tleft() > 100:
        h_as2, sc_as2 = active_set_descent(best_h, target, n_iters=300, tol_frac=5e-4)
        if sc_as2 < best_score:
            best_score = sc_as2
            best_h = h_as2

    # Phase 5: LSE at very high alpha with top-K
    if tleft() > 60:
        h_new = lbfgs_optimize(best_h, target, 500000, mu, 500, top_k=20)
        sc, _ = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Coordinate descent
    if tleft() > 200:
        h_cd, sc_cd = coordinate_descent(best_h, target, n_sweeps=6)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 7: SLSQP polish
    if tleft() > 50:
        h_slq = slsqp_polish(best_h, target, 10000, 8000)
        sc, _ = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 8: Final active-set
    if tleft() > 60:
        h_as3, sc_as3 = active_set_descent(best_h, target, n_iters=200, tol_frac=1e-4)
        if sc_as3 < best_score:
            best_score = sc_as3
            best_h = h_as3

    # Phase 9: Coordinate descent final
    if tleft() > 120:
        h_cd2, sc_cd2 = coordinate_descent(best_h, target, n_sweeps=4)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 10: SLSQP final
    if tleft() > 40:
        h_final_slq = slsqp_polish(best_h, target, 50000, 10000)
        sc, _ = compute_c5(h_final_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_final_slq

    # Phase 11: One more coordinate descent
    if tleft() > 80:
        h_cd3, sc_cd3 = coordinate_descent(best_h, target, n_sweeps=3)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    # Phase 12: Final SLSQP
    if tleft() > 30:
        h_final = slsqp_polish(best_h, target, 200000, 15000)
        sc, _ = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    h_final = project_feasible(best_h, target)
    final_score, _ = compute_c5(h_final)
    return h_final, final_score, n
