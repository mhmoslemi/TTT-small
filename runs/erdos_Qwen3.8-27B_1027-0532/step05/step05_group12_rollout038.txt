    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)

    # If j - k is valid (0 ≤ j-k ≤ n-1): contributes -h[j-k]

    # This is a sliding window sum of w * (1-h)

    # This is a sliding window sum of w * h

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_smooth_grad(h, alpha):
    """Compute gradient of smooth max using FFT-based operations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    omh = 1.0 - h
    # A_j = corr_full[2n-2-j] where corr_full = correlate(weights, omh, 'full')
    corr_full = np.correlate(weights, omh, mode='full')
    # B_j = conv_full[n-1+j] where conv_full = convolve(h, weights, 'full')
    conv_full = np.convolve(h, weights, mode='full')
    
    A = corr_full[2*n-2 : n-2 : -1]  # length n
    B = conv_full[n-1 : 2*n-1]  # length n
    
    return (A - B) * dx

def fista_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30):
    """FISTA (accelerated projected gradient) with smooth max objective."""
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    z = h.copy()  # FISTA momentum variable
    t_k = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    
    alpha_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    max_iter_per_alpha = 2000
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        z = best_h.copy()
        h = z.copy()
        t_k = 1.0
        
        for it in range(max_iter_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            
            # Compute gradient at z
            grad = fast_smooth_grad(z, alpha)
            
            # Lipschitz constant estimate (upper bound on alpha for the smooth part)
            # The smooth max has Lipschitz constant ~ alpha * dx * n (rough bound)
            # Use a safe step size
            L = alpha * dx * n  # conservative Lipschitz bound
            step = 1.0 / L
            
            # Gradient step
            h_new = z - step * grad
            h_new = project_feasible(h_new, target)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            
            z_new = h_new + (t_k - 1.0) / t_new * (h_new - h)
            z_new = project_feasible(z_new, target)
            
            h = h_new
            z = z_new
            t_k = t_new
            
            # Check improvement
            score = compute_c5(h)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h.copy()
        
        # Reset t_k for next alpha
        h = best_h.copy()

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    omh = 1.0 - h
    corr_full = np.correlate(weights, omh, mode='full')
    conv_full = np.convolve(h, weights, mode='full')
    A = corr_full[2*n-2 : n-2 : -1]
    B = conv_full[n-1 : 2*n-1]
    return (A - B) * dx

def fista_smooth(h_init, n, target, alphas, max_iters, start_time, time_limit, deadline_offset=30):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        z = best_h.copy()
        h = z.copy()
        t_k = 1.0
        stalled = 0
        
        for it in range(max_iters):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            grad = fast_smooth_grad(z, alpha)
            # Adaptive step size
            # Lipschitz constant of smooth max gradient is ~ alpha * max_k ||grad c_k||
            # ||grad c_k||_inf <= 2*dx (each component is at most dx*(1+1))
            # So L_inf <= alpha * 2 * dx * sqrt(n) (converting to L2)
            # But for projected gradient, we need step < 2/L where L is the L2 Lipschitz constant
            # Let's use a conservative step
            L = alpha * 2.0 * (2.0/n) * np.sqrt(n)  # = alpha * 4 / sqrt(n)
            step = 2.0 / (L + 1e-10)
            
            h_new = z - step * grad
            h_new = project_feasible(h_new, target)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
            z_new = h_new + (t_k - 1.0) / t_new * (h_new - h)
            z_new = project_feasible(z_new, target)
            
            h = h_new
            z = z_new
            t_k = t_new
            
            score = compute_c5(h)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h.copy()
                stalled = 0
            else:
                stalled += 1
                if stalled > 100:
                    break
    
    return best_h, best_score

def alternating_qp(h_init, n, target, max_iter, start_time, time_limit, deadline_offset=30,
                   damping=0.7, reg=1e-3):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in range(-(n-1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas, maxiter, start_time, time_limit, deadline_offset=30):
    # Use smooth objective with SLSQP
    _n = n
    _target = target
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        alpha = alphas[0]  # Use fixed alpha
        shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    def obj_grad(h):
        return fast_smooth_grad(h, alphas[0])
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    
    h_cur = h_start.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        # Redefine with current alpha
        def obj_a(h, a=alpha):
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -600.0, 0.0)
            return c_max + np.log(np.sum(np.exp(shifted))) / a
        
        def obj_grad_a(h, a=alpha):
            return fast_smooth_grad(h, a)
        
        try:
            res = minimize(obj_a, h_cur, jac=obj_grad_a,

            if res.x is not None:

        except Exception:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_local_search(h_init, n, target, n_iterations=50000, start_time=None, time_limit=100, deadline_offset=30):
    """Local search on binary solutions using random pair swaps."""
    # Round to binary
    h = np.where(h_init > 0.5, 1.0, 0.0)
    # Fix to have exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > target:
        # Remove excess ones (keep the ones at positions where h_init is largest)
        ones_idx = np.where(h == 1.0)[0]
        excess = n_ones - target
        scores = h_init[ones_idx]
        remove_idx = ones_idx[np.argsort(scores)[:excess]]
        h[remove_idx] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0.0)[0]
        deficit = target - n_ones
        scores = h_init[zeros_idx]
        add_idx = zeros_idx[np.argsort(scores)[-deficit:]]
        h[add_idx] = 1.0
    
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    zero_idx = np.where(h == 0.0)[0]
    one_idx = np.where(h == 1.0)[0]
    
    for it in range(n_iterations):
        if start_time is not None and time.time() - start_time > time_limit - deadline_offset:
            break
        
        if len(zero_idx) == 0 or len(one_idx) == 0:
            break
        
        i = zero_idx[np.random.randint(len(zero_idx))]
        j = one_idx[np.random.randint(len(one_idx))]
        
        # Try the swap
        h[i] = 1.0
        h[j] = 0.0
        score = compute_c5(h)
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            # Update index sets
            zero_idx = zero_idx[zero_idx != i]
            zero_idx = np.append(zero_idx, j)
            one_idx = one_idx[one_idx != j]
            one_idx = np.append(one_idx, i)
        else:
            # Revert
            h[i] = 0.0
            h[j] = 1.0
    
    return best_h, best_score

def project_feasible(h, target_sum):

def fast_smooth_grad(h, alpha):

                            deadline_offset=40)
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    # Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        h_fin = slsqp_polish(best_h, n_use, n_use / 2.0,
                             alphas=[5000, 20000],
                             maxiter=300,
                             start_time=start_time, time_limit=budget_s,
                             deadline_offset=10)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
