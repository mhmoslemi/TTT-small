    """corr is the full correlation array (length 2n-1).
    h is the current h array (BEFORE the update to position j).
    delta is the change to h[j]."""

        return

    p_start = -j
    p_end = n - 1 - j

    m_end = n - 1 + p_end

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_max(corr):
    return float(np.max(corr))


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        if time.time() > deadline:
            return h_cur
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def fast_coord_descent(h, target, n_sweeps=10, n_points=None):
    n = len(h)
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full').astype(float)
    cur_score = float(np.max(corr))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    h_rev = h_cur[::-1].copy()
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old = h_cur[j]
            L = n - 1
            a_slice = slice(L - j, L - j + n)
            b_slice = slice(j, j + n)
            dc_unit = np.zeros(2 * n - 1)
            dc_unit[a_slice] += (1.0 - h_cur)
            dc_unit[b_slice] -= h_cur[::-1]
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = float(np.max(corr + (c - old) * dc_unit))
            fd = float(np.max(corr + (d - old) * dc_unit))
            for _ in range(28):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = float(np.max(corr + (c - old) * dc_unit))
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = float(np.max(corr + (d - old) * dc_unit))
            best_v = (a + b) / 2.0

                if new_score < cur_score - 1e-13:
                    cur_score = new_score
                    improved = True
        if not improved:
            break
    return h_cur, cur_score * (2.0 / n)


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        if time.time() > deadline:
            return h_cur
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def subgradient_top_lags(h, target, n_steps=500, n_top=10, lr_init=0.02, momentum=0.85):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    v = np.zeros(n)
    lr = lr_init
    for step in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        grads = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            w = max(w, 0.0)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grads[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grads[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grads[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grads[j2] -= w * h_cur[j2 - m]
        v = momentum * v - lr * grads
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            cur_score = sc_new
            h_cur = h_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.9
            v *= 0.5
            if lr < 1e-10:
                break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def structured_perturbation(h, rng, mode, intensity):
    n = len(h)
    h_pert = h.copy()
    if mode == 'gauss':
        h_pert += rng.randn(n) * intensity
    elif mode == 'smooth':
        freq = np.fft.rfftfreq(n)
        cutoff = max(1, int(n * intensity))
        spec = np.fft.rfft(h_pert)
        noise = np.zeros(len(spec))
        for i in range(1, min(cutoff, len(spec))):
            noise[i] = rng.randn() + 1j * rng.randn()
        perturb = np.fft.irfft(spec + noise * 0.1, n)
        h_pert = perturb
    elif mode == 'bump':
        center = rng.randint(0, n)
        width = max(1, int(n * intensity * 0.1))
        for i in range(n):
            dist = min(abs(i - center), n - abs(i - center))
            h_pert[i] -= 0.1 * intensity * np.exp(-0.5 * (dist / width) ** 2)
    elif mode == 'swap':
        n_swap = max(1, int(n * intensity * 0.05))
        for _ in range(n_swap):
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i != j:
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
    return h_pert

            break
        h_pert = structured_perturbation(best_h, rng, strat, param)
        h_pert = project_feasible(h_pert, target)
        h_cd, sc_cd = fast_coord_descent(h_pert, target, n_sweeps=8)
        h_new = lbfgs_optimize(h_cd, target, 500, mu, 300, start_time + budget_s)
        h_new = lbfgs_optimize(h_new, target, 10000, mu, 300, start_time + budget_s)
        h_new = fast_coord_descent(h_new, target, n_sweeps=5)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Try doubling resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 300:
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n2)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, target2)
        alphas_fine = [50, 500, 5000, 50000]
        for a in alphas_fine:
            if tleft() < 60:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, mu, 300, start_time + budget_s)
            h_fine, _ = fast_coord_descent(h_fine, target2, n_sweeps=5)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 7: Final subgradient + coordinate descent
    if tleft() > 80:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_top_lags(best_h, tgt_cur, n_steps=300, n_top=5, lr_init=0.02)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_cd, sc_cd = fast_coord_descent(best_h, tgt_cur, n_sweeps=20)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 8: Final L-BFGS with very high alpha
    if tleft() > 40:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_new = lbfgs_optimize(best_h, tgt_cur, 500000, mu, 500, start_time + budget_s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_cd, sc_cd = fast_coord_descent(best_h, tgt_cur, n_sweeps=10)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: SLSQP polish
    if tleft() > 30:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_slq = slsqp_polish(best_h, tgt_cur, 10000, 200)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq
        h_cd, sc_cd = fast_coord_descent(best_h, tgt_cur, n_sweeps=5)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Final projection and verification
    n_final = len(best_h)
    tgt_final = n_final / 2.0
    h_final = project_feasible(best_h, tgt_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
