                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n / 2.0)
        return h_new
    except Exception:
        return h


def subgradient_polish(h, n_steps, n_active, lr_init, lr_decay, lr_max, temperature, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    stagnant = 0

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_curr = c_max * dx

        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 80:
            lr *= 0.85
            stagnant = 0
            if lr < 1e-12:
                break

        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temperature)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            npos = n - k
            grad[:npos] += wk * (1.0 - h_cur[k:])
            if k > 0:
                grad[k:] -= wk * h_cur[:npos]
        grad *= dx

        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.015, lr_max)
        else:
            lr *= lr_decay

    return best_h, best_score


def worst_lag_surgery(h, max_steps, eps_init, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = eps_init

    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        k_star = int(np.argmax(corr))
        npos = n - k_star
        grad = np.zeros(n)
        grad[:npos] += (1.0 - h_cur[k_star:])
        if k_star > 0:
            grad[k_star:] -= h_cur[:npos]
        grad *= dx

        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.04, 0.02)
        else:
            eps *= 0.5
            if eps < 1e-10:
                break

    return h_cur, cur_score


def top_k_surgery(h, max_steps, n_top, eps_init, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = eps_init

    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_top]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            npos = n - k
            grad[:npos] += (1.0 - h_cur[k:])
            if k > 0:
                grad[k:] -= h_cur[:npos]
        grad *= dx / n_top

        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.015)
        else:
            eps *= 0.5
            if eps < 1e-10:
                break

    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_mu
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 20.0

    def tleft():
        return hard_deadline - time.time()

    n_parent = len(initial_h_values)
    target_parent = n_parent / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_parent)
    best_h_parent = h_parent.copy()
    best_score_parent = compute_c5(best_h_parent)

    n = 500
    target = n / 2.0
    h = interpolate_to(h_parent, n, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    # Phase 1: Quick SLSQP
    if tleft() > 200:
        for a in [200.0, 500.0]:
            if tleft() < 30:
                break
            h = slsqp_optimize(h, a, 300, start_time + 50)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    # Phase 2: L-BFGS-B with increasing alpha
    if tleft() > 300:
        for a in [100.0, 300.0, 1000.0, 3000.0, 10000.0]:
            if tleft() < 30:
                break
            h = lbfgsb_lse(h, a, 300, start_time + 50, mu=1.0)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
            h = lbfgsb_lse(h, a, 200, start_time + 50, mu=10.0)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    # Phase 3: Subgradient
    if tleft() > 200:
        h, sc = subgradient_polish(best_h, 600, 5, 0.003, 0.97, 0.015, 0.005, start_time + 120)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 4: SLSQP polish
    if tleft() > 150:
        for a in [2000.0, 5000.0]:
            if tleft() < 40:
                break
            h = slsqp_optimize(best_h, a, 600, start_time + 60)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    # Phase 5: L-BFGS-B high alpha
    if tleft() > 150:
        for a in [5000.0, 15000.0, 50000.0]:
            if tleft() < 30:
                break
            h = lbfgsb_lse(best_h, a, 250, start_time + 40, mu=5.0)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
            h = lbfgsb_lse(best_h, a, 150, start_time + 40, mu=50.0)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    # Phase 6: Subgradient + surgery
    if tleft() > 120:
        h, sc = subgradient_polish(best_h, 400, 3, 0.002, 0.98, 0.008, 0.003, start_time + 60)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        h, sc = worst_lag_surgery(best_h, 300, 0.002, start_time + 40)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 7: SLSQP
    if tleft() > 120:
        for a in [10000.0, 30000.0]:
            if tleft() < 40:
                break
            h = slsqp_optimize(best_h, a, 800, start_time + 60)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    # Phase 8: Top-k surgery + subgradient
    if tleft() > 100:
        h, sc = top_k_surgery(best_h, 400, 4, 0.001, start_time + 40)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        h, sc = subgradient_polish(best_h, 300, 5, 0.001, 0.98, 0.005, 0.004, start_time + 40)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 9: L-BFGS-B very high alpha
    if tleft() > 100:
        for a in [20000.0, 50000.0, 100000.0]:
            if tleft() < 30:
                break
            h = lbfgsb_lse(best_h, a, 200, start_time + 35, mu=100.0)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    # Phase 10: SLSQP final
    if tleft() > 60:
        h = slsqp_optimize(best_h, 50000.0, 500, hard_deadline - 20)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Phase 11: Final surgery
    if tleft() > 30:
        h, sc = worst_lag_surgery(best_h, 200, 0.0005, hard_deadline - 15)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Try random restarts if time allows
    if tleft() > 80:
        for attempt in range(3):
            if tleft() < 35:
                break
            scale = 0.03 + 0.02 * attempt
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, target)
            h_pert = lbfgsb_lse(h_pert, 5000.0, 200, start_time + 30, mu=10.0)
            h_pert = slsqp_optimize(h_pert, 10000.0, 400, start_time + 30)
            sc = compute_c5(h_pert)
            if sc < best_score:
                best_score = sc
                best_h = h_pert.copy()

    # Final subgradient
    if tleft() > 25:
        h, sc = subgradient_polish(best_h, 200, 3, 0.001, 0.99, 0.003, 0.002, hard_deadline - 10)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Compare with parent resolution
    h_parent_interp = interpolate_to(best_h, n_parent, target_parent)
    sc_parent = compute_c5(h_parent_interp)
    if sc_parent < best_score_parent:
        best_h_parent = h_parent_interp
        best_score_parent = sc_parent

    if best_score_parent < best_score:
        best_h = best_h_parent
        best_score = best_score_parent

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
