    # Phase 2: SLSQP with extensive alpha schedule
    if tleft() > 180:
        alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
        h_ref = slsqp_polish(best_h, n, target, alphas, 5000, 1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Subgradient on active set
    if tleft() > 180:
        deadline_sg = start_time + (budget_s - 15.0) * 0.42
        h_sg, sc_sg = speed_max(best_h, n, target, deadline_sg, rng, scale=0.005, eta0=0.3)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 4: More CD after subgradient
    if tleft() > 150:
        deadline_p4 = start_time + (budget_s - 15.0) * 0.52
        h_cd, sc_cd = paired_cd(best_h, n, target, deadline_p4, rng, recompute_interval=10)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: SLSQP again
    if tleft() > 120:
        alphas2 = [1000, 5000, 20000, 100000]
        h_ref = slsqp_polish(best_h, n, target, alphas2, 5000, 1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Coarse-to-fine
    if tleft() > 250:
        n_coarse = max(n // 4, 50)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        deadline_coarse = start_time + (budget_s - 15.0) * 0.65
        h_coarse, sc_coarse = paired_cd(h_coarse, n_coarse, target_coarse,
                                        deadline_coarse, rng, recompute_interval=10)
        h_up = interpolate_to(h_coarse, n, target)
        h_up = slsqp_polish(h_up, n, target, [500, 2000, 10000, 50000], 4000, 1e-16)
        deadline_up = start_time + (budget_s - 15.0) * 0.73
        h_up, sc_up = paired_cd(h_up, n, target, deadline_up, rng, recompute_interval=10)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 7: Structured perturbations + CD
    deadline_p7 = start_time + (budget_s - 15.0) * 0.88
    x_grid = np.linspace(0, 2 * np.pi, n, endpoint=False)
    for attempt in range(12):
        if tleft() < 60:
            break
        if attempt < 4:
            freq = 1 + attempt
            pert = 0.03 * np.sin(freq * x_grid + rng.uniform(0, 2 * np.pi))
        elif attempt < 8:
            pert = rng.randn(n) * 0.03
        else:
            block_size = rng.randint(20, 80)
            start_idx = rng.randint(0, n - block_size)
            pert = np.zeros(n)
            pert[start_idx:start_idx + block_size] = rng.uniform(-0.05, 0.05)
        h_pert = project_feasible(best_h + pert, target)
        h_res, sc_res = paired_cd(h_pert, n, target, deadline_p7, rng, recompute_interval=15)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    # Phase 8: Final SLSQP + subgradient + CD
    if tleft() > 80:
        alphas3 = [2000, 10000, 50000, 200000]
        h_final = slsqp_polish(best_h, n, target, alphas3, 5000, 1e-17)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 40:
        h_sg2, sc_sg2 = speed_max(best_h, n, target, start_time + budget_s - 30, rng,
                                  scale=0.003, eta0=0.2)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    if tleft() > 20:
        h_final, sc_final = paired_cd(best_h, n, target,
                                      start_time + budget_s - 10, rng, recompute_interval=8)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
