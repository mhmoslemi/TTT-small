    best_h = h.copy()

For the non-smooth max, we can use:

```python
def frank_wolfe_step(h, k_star, n, target):
    """Solve the linear min problem for the Frank-Wolfe method"""
    dx = 2.0 / n
    # Compute gradient of f_{k_star}
    grad = np.zeros(n)
    k = k_star - (n - 1)
    if k >= 0:
        # f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
        # df_k/dh[i] = (1-h[i+k])*dx for i < n-k
        # df_k/dh[i] = -h[i-k]*dx for i >= k
        grad[:n-k] += (1.0 - h[k:]) * dx
        grad[k:] -= h[:n-k] * dx
    else:
        m = -k
        # f = sum_{j=0}^{n-1-m} h[j+m]*(1-h[j]) * dx
        # df/dh[j] = -h[j+m]*dx for j < n-m
        # df/dh[j] = (1-h[j-m])*dx for j >= m
        grad[:n-m] -= h[m:] * dx
        grad[m:] += (1.0 - h[:n-m]) * dx
    
    # Linear minimization: min grad · h' s.t. 0≤h'≤1, sum(h')=target
    # Solution: sort by grad, set smallest to 1, largest to 0, one fractional
    order = np.argsort(grad)
    h_w = np.zeros(n)
    h_w[order[:int(target)]] = 1.0
    remainder = target - int(target)
    if remainder > 1e-12 and int(target) < n:
        h_w[order[int(target)]] = remainder
    return h_w
```

            h_var >= 0.0,
            h_var <= 1.0

            res = minimize(_slsqp_obj_grad, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_top_m
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Active-set LP to quickly improve
    if tleft() > 200:
        h_lp, c5_lp = active_set_lp(best_h, n, target, top_k=25, max_iter=300,
                                    time_limit=budget_s, start_time=start_time,
                                    deadline_s=budget_s - 200, damping=0.65, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Nesterov smooth gradient descent
    if tleft() > 250:
        h_ns, c5_ns = nesterov_smooth_descent(best_h, n, target,
                                              time_limit=budget_s, start_time=start_time,
                                              deadline_s=budget_s - 250,
                                              alphas=[20, 50, 100, 200, 500, 1000, 2000, 5000, 10000],
                                              steps_per_alpha=1500, top_m=60)
        if c5_ns < best_score:
            best_score = c5_ns
            best_h = h_ns

    # Phase 3: SLSQP with moderate alpha
    if tleft() > 150:
        h_s = slsqp_polish(best_h, n, target,
                           [100, 300, 1000, 3000, 10000],
                           maxiter=1500, time_limit=budget_s, start_time=start_time,
                           deadline_s=budget_s - 150, top_m=None)
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score = sc
            best_h = h_s

    # Phase 4: SLSQP with very high alpha
    if tleft() > 100:
        h_s = slsqp_polish(best_h, n, target,
                           [10000, 30000, 100000, 300000],
                           maxiter=2000, time_limit=budget_s, start_time=start_time,
                           deadline_s=budget_s - 100, top_m=None)
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score = sc
            best_h = h_s

    # Phase 5: Local perturbation restarts
    for noise in [0.005, 0.01, 0.02, 0.03]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_s = slsqp_polish(h_pert, n, target,
                           [500, 2000, 10000, 50000],
                           maxiter=1000, time_limit=budget_s, start_time=start_time,
                           deadline_s=budget_s - 80, top_m=None)
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score = sc
            best_h = h_s

    # Phase 6: Higher resolution attempts
    for n_try in [600, 800, 1000]:
        if tleft() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        # Quick Nesterov at higher res
        h_hr, c5_hr = nesterov_smooth_descent(h_interp, n_try, target_try,
                                              time_limit=budget_s, start_time=start_time,
                                              deadline_s=budget_s - 100,
                                              alphas=[50, 200, 1000, 5000, 20000, 50000],
                                              steps_per_alpha=800, top_m=80)
        if tleft() > 60:
            h_hr = slsqp_polish(h_hr, n_try, target_try,
                                [5000, 20000, 50000],
                                maxiter=1000, time_limit=budget_s, start_time=start_time,
                                deadline_s=budget_s - 50, top_m=None)
        sc = compute_c5(h_hr)
        if sc < best_score:
            best_score = sc
            best_h = h_hr

    # Phase 7: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        h_fin = slsqp_polish(best_h, n_use, n_use / 2.0,
                             [20000, 50000, 200000, 500000],
                             maxiter=1500, time_limit=budget_s, start_time=start_time,
                             deadline_s=15, top_m=None)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Ensure feasibility
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
