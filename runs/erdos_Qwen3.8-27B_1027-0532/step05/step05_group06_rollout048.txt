import numpy as np

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def obj_grad_lse(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            if lag > 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - x[j1 - lag])
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
            else:
                grad += w * (1.0 - x)
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f_wrapper(x):
        return obj_grad_lse(x, alpha, target, mu)[0]
    
    def g_wrapper(x):
        return obj_grad_lse(x, alpha, target, mu)[1]
    
    try:
        res = minimize(f_wrapper, h.copy(), jac=g_wrapper, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-18, 'maxls': 20})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()
