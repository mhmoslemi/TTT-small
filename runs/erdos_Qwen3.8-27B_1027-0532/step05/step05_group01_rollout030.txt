2. The function f(δ) = max_m (corr[m] + δ * g[m]) is piecewise linear

4. This is equivalent to: find δ such that the maximum of (corr + δ*g) is minimized

- Computing all values corr[m] + δ * g[m] for a few candidate δ values

def compute_g(h, n, i, j):
    """Compute the direction vector g for changing h[i]-=1, h[j]+=1."""
    n_corr = 2 * n - 1
    g = np.zeros(n_corr)
    # h[i] contribution: for m in [i, min(i+n-1, n_corr-1)]
    m_lo = i
    m_hi = min(i + n - 1, n_corr - 1)
    if m_lo <= m_hi:
        idx = np.arange(m_lo, m_hi + 1)
        h_idx = i - idx + n - 1  # index into (1-h)
        g[idx] -= (1.0 - h[h_idx])
    # h[j] contribution: for m in [j, min(j+n-1, n_corr-1)]
    m_lo = j
    m_hi = min(j + n - 1, n_corr - 1)
    if m_lo <= m_hi:
        idx = np.arange(m_lo, m_hi + 1)
        h_idx = j - idx + n - 1
        g[idx] += (1.0 - h[h_idx])
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    return np.sum(h) - _global_target

    return np.ones(_global_n)

    return float(c_max + np.sum(np.exp(shifted)) / alpha)

    _global_target = target
