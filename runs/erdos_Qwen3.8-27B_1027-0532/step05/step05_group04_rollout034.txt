```python
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def antisymetric_projection(h, n):
    # Project h onto the subspace where h[j + n//2] = 1 - h[j]
    m = n // 2
    h_sym = np.zeros(n)
    for j in range(m):
        h_sym[j] = (h[j] + (1 - h[j + m])) / 2
        h_sym[j + m] = 1 - h_sym[j]
    return project_feasible(h_sym, n / 2.0)

def lse_objective(h, n, target, alpha):
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-10:
            continue
        # d corr[k] / d h[j]
        for j in range(min(n - k, n)):
```

```python
def lag_cycling(h, n, target, n_rounds, lr_init, deadline):
    h_cur = h.copy()
    best = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        # Sort lags by correlation value (worst first)
        order = np.argsort(over)[::-1]
        for k in order:
            k = int(k)
            # Gradient descent step on this specific lag
            g = np.zeros(n)
            g[:n-k] += (1.0 - h_cur[k:])
            g[k:] -= h_cur[:n-k]
            h_new = h_cur - lr * g * dx
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < compute_c5(h_cur):
                h_cur = h_new
                if sc_new < best_score:
                    best_score = sc_new
                    best = h_cur.copy()
            else:
                lr = max(lr * 0.9, 1e-8)
        lr = min(lr * 1.05, lr_init)
    
    return best, best_score
```

```python
def compute_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_corr_grad(h, n, weights):
    """Compute weighted sum of gradients of corr[k] for all k."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    return grad * dx

def lse_value_and_weights(h, n, alpha):
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

    log_sum = float(np.log(np.sum(exp_vals)))
    val = c_max + log_sum / alpha
    weights = exp_vals / exp_vals.sum()
    return val, weights

def nesterov_lse(h, n, target, n_steps, alpha, lr_init, momentum_init, deadline):
    """Nesterov accelerated gradient on LSE objective."""
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    mu = momentum_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead

def slsqp_polish(h, n, target, alpha, maxiter, ftol, deadline):

    x0 = h.copy()
    
    def obj(x):
        h_t = np.clip(x, 0.0, 1.0)
        val, _ = lse_value_and_weights(h_t, n, alpha)
        return val
```

```python
import numpy as np

_g_target = 0.0
```

```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def weighted_corr_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        grad[:n - k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n - k]
    return grad * dx


def lse_value_and_weights(h, n, alpha):
    corr = compute_corr(h)
    over = corr[n - 1:n - 1 + n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    val = c_max + float(np.log(s)) / alpha
    weights = exp_vals / s
    return val, weights

    h_t = np.clip(x, 0.0, 1.0)
    val, _ = lse_value_and_weights(h_t, _g_n, _g_alpha)

    h_t = np.clip(x, 0.0, 1.0)
    _, weights = lse_value_and_weights(h_t, _g_n, _g_alpha)
    return weighted_corr_grad(h_t, _g_n, weights)
```
