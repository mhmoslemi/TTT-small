def weighted_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-14:
            continue
        nk = n - k
        grad[:nk] += wk * (1.0 - h[k:])
        if k > 0:
            grad[k:] -= wk * h[:nk]
    return grad


def mw_gd(h, n, target, n_outer, eta_init, eta_growth, lr_init, n_inner, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    lr = lr_init
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        eta = eta_init * (eta_growth ** outer)
        ov = compute_overlaps(h_cur)
        max_ov = float(np.max(ov))
        weights = np.exp(eta * (ov - max_ov))
        tw = float(np.sum(weights))
        if tw < 1e-300:
            weights = np.ones(n) / n
        else:
            weights = weights / tw
        for inner in range(n_inner):
            if time.time() > deadline:
                break
            grad = weighted_gradient(h_cur, weights)
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            if sc <= cur_score + 1e-15:
                h_cur = h_new
                cur_score = sc
                if inner % 30 == 29:
                    lr = min(lr * 1.005, 0.03)
            else:
                lr = max(lr * 0.88, 1e-9)
                if lr < 1e-11:
                    break
    return best_h, best_score


def _slsqp_full_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _slsqp_full_eq(x):
    return np.sum(x) - _g_target


def _slsqp_full_eq_jac(x):
    return np.ones(_g_n)


def slsqp_full(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_full_eq, 'jac': _slsqp_full_eq_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_slsqp_full_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def _active_obj(x):
    h_full = _g_h_base.copy()
    h_full[_g_free_idx] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _active_eq(x):
    return np.sum(x) - _g_free_target


def _active_eq_jac(x):
    return np.ones(len(x))


def slsqp_active(h, n, target, alphas, maxiter, ftol, deadline, tol=1e-3):
    global _g_h_base, _g_free_idx, _g_free_target, _g_alpha
    _g_h_base = h.copy()
    free_mask = (h > tol) & (h < 1.0 - tol)
    free_idx = np.where(free_mask)[0]
    n_free = len(free_idx)
    if n_free < 2:
        return h.copy(), compute_c5(h)
    _g_free_idx = free_idx
    fixed_mask = ~free_mask
    free_target = target - np.sum(h[fixed_mask])
    if free_target < 0 or free_target > n_free:
        return h.copy(), compute_c5(h)
    _g_free_target = free_target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    x0 = h[free_idx].copy()
    bounds = [(0.0, 1.0)] * n_free
    cons = [{'type': 'eq', 'fun': _active_eq, 'jac': _active_eq_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_active_obj, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = h_cur.copy()
            h_new[free_idx] = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                x0 = h_cur[free_idx].copy()
        except Exception:
            pass
    return h_cur, cur_score


def binary_local_search(h, n, target, max_rounds, deadline):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    cur_score = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_score = cur_score
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h_bin)
        worst_k = int(np.argmax(ov[:n]))
        improved = False
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        rng_local = np.random.RandomState(rnd + 1)
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            i = int(ones[rng_local.randint(len(ones))])
            j = int(zeros[rng_local.randint(len(zeros))])
            if i == j:
                continue
            h_test = h_bin.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-14:
                h_bin = h_test
                cur_score = sc
                ones = np.where(h_bin > 0.5)[0]
                zeros = np.where(h_bin < 0.5)[0]
                improved = True
                break
        if not improved:
            break
        if cur_score < best_score:
            best_score = cur_score
            best_h = h_bin.copy()
    return best_h, best_score


def subgradient_focus(h, n, target, n_steps, n_active, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h_cur)
        top_idx = np.argsort(ov)[::-1][:n_active]
        diffs = ov[top_idx] - ov[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights = weights / weights.sum()
        w_full = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            w_full[int(ki)] = wi
        grad = weighted_gradient(h_cur, w_full)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if sc <= cur_score + 1e-15:
            h_cur = h_new
            cur_score = sc
        else:
            lr = max(lr * 0.95, 1e-10)
            if lr < 1e-12:
                break
    return best_h, best_score


def periodic_optimize(p, n, target, deadline):
    if n % p != 0:
        return None
    reps = n // p
    ptarget = target / reps
    best_c5 = 0.5
    best_c = None
    c0 = np.full(p, 0.5)
    for trial in range(50):
        if time.time() > deadline:
            break
        if trial == 0:
            c = c0.copy()
        else:
            c = 0.5 + 0.3 * np.sin(2.0 * np.pi * np.arange(p) / p + trial * 0.7)
            c = np.clip(c, 0.0, 1.0)
            c = project_feasible(c, ptarget)
        bounds = [(0.0, 1.0)] * p
        for a in [100.0, 500.0]:
            if time.time() > deadline:
                break
            def _per_obj(x, c_pat=c.copy(), _p=p, _reps=reps, _n=n):
                h_full = np.tile(np.clip(x, 0.0, 1.0), _reps)[:_n]
                corr = np.correlate(h_full, 1.0 - h_full, mode='full')
                c_max = float(np.max(corr))
                shifted = a * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / a
            def _per_eq(x, _pt=ptarget):
                return np.sum(x) - _pt
            def _per_eq_jac(x, _p=p):
                return np.ones(_p)
            cons = [{'type': 'eq', 'fun': _per_eq, 'jac': _per_eq_jac}]
            try:
                res = minimize(_per_obj, c, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
                c_new = project_feasible(res.x, ptarget)
                h_full = np.tile(c_new, reps)[:n]
                h_full = project_feasible(h_full, target)
                sc = compute_c5(h_full)
                if sc < best_c5:
                    best_c5 = sc
                    best_c = c_new
                    c = c_new.copy()
            except Exception:
                pass
    if best_c is not None:
        h_full = np.tile(best_c, reps)[:n]
        return project_feasible(h_full, target)
    return None


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_weights
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 30.0 - elapsed()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Try periodic patterns for diverse starts
    if tleft() > 300:
        for p in [4, 5, 8, 10, 20, 40, 50, 80, 100, 200]:
            if tleft() < 250:
                break
            h_per = periodic_optimize(p, n, target, deadline=start_time + elapsed() + 15.0)
            if h_per is not None:
                sc = compute_c5(h_per)
                if sc < best_score:
                    best_score = sc
                    best_h = h_per.copy()

    # Phase 2: Main multiplicative weights gradient descent
    if tleft() > 200:
        deadline_p2 = start_time + min(300, budget_s - 300)
        h_mw, sc_mw = mw_gd(best_h, n, target, n_outer=80, eta_init=50.0,
                            eta_growth=1.15, lr_init=0.005, n_inner=50,
                            deadline=deadline_p2)
        if sc_mw < best_score:
            best_score = sc_mw
            best_h = h_mw.copy()

    # Phase 3: SLSQP full polish
    if tleft() > 150:
        deadline_p3 = start_time + min(450, budget_s - 250)
        h_sl, sc_sl = slsqp_full(best_h, n, target, [200.0, 800.0, 3000.0],
                                  3000, 1e-16, deadline_p3)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl.copy()

    # Phase 4: Active-set SLSQP (faster, focuses on free variables)
    if tleft() > 120:
        deadline_p4 = start_time + min(600, budget_s - 200)
        h_act, sc_act = slsqp_active(best_h, n, target,
                                      [500.0, 2000.0, 8000.0, 30000.0],
                                      3000, 1e-17, deadline_p4, tol=1e-4)
        if sc_act < best_score:
            best_score = sc_act
            best_h = h_act.copy()

    # Phase 5: Subgradient focus on worst lags
    if tleft() > 100:
        deadline_p5 = start_time + min(700, budget_s - 180)
        h_sg, sc_sg = subgradient_focus(best_h, n, target, n_steps=800,
                                         n_active=2, lr_init=0.003,
                                         deadline=deadline_p5)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg.copy()

    # Phase 6: Binary local search
    if tleft() > 100:
        deadline_p6 = start_time + min(800, budget_s - 150)
        h_bin, sc_bin = binary_local_search(best_h, n, target,
                                             max_rounds=50, deadline=deadline_p6)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
        # Unfreeze binary and re-optimize
        if tleft() > 80:
            h_unf = best_h + 0.05 * rng.randn(n)
            h_unf = project_feasible(h_unf, target)
            deadline_p6b = start_time + min(850, budget_s - 130)
            h_mw2, sc_mw2 = mw_gd(h_unf, n, target, n_outer=40, eta_init=200.0,
                                   eta_growth=1.2, lr_init=0.002, n_inner=30,
                                   deadline=deadline_p6b)
            if sc_mw2 < best_score:
                best_score = sc_mw2
                best_h = h_mw2.copy()

    # Phase 7: Second MW pass with different parameters
    if tleft() > 100:
        deadline_p7 = start_time + min(950, budget_s - 120)
        h_mw3, sc_mw3 = mw_gd(best_h, n, target, n_outer=60, eta_init=500.0,
                               eta_growth=1.25, lr_init=0.001, n_inner=40,
                               deadline=deadline_p7)
        if sc_mw3 < best_score:
            best_score = sc_mw3
            best_h = h_mw3.copy()

    # Phase 8: Active-set SLSQP final
    if tleft() > 80:
        deadline_p8 = start_time + min(1050, budget_s - 100)
        h_act2, sc_act2 = slsqp_active(best_h, n, target,
                                        [5000.0, 20000.0, 80000.0],
                                        5000, 1e-17, deadline_p8, tol=1e-5)
        if sc_act2 < best_score:
            best_score = sc_act2
            best_h = h_act2.copy()

    # Phase 9: Final subgradient micro-polish
    if tleft() > 40:
        deadline_p9 = start_time + budget_s - 30
        h_sg2, sc_sg2 = subgradient_focus(best_h, n, target, n_steps=500,
                                           n_active=1, lr_init=0.001,
                                           deadline=deadline_p9)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2.copy()

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
