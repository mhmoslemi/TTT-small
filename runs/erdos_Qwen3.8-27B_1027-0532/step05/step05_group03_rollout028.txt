    n = len(h)
    best_score = compute_c5(h)
    while time_remaining():
        corr = np.correlate(h, 1-h, mode='full')
        over = corr[n-1:]  # lags 0 to n-1
        k_star = argmax(over)
        current_max = over[k_star]
        
        best_swap = None
        best_delta = 0
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for a in zeros:  # a: 0 -> 1
            for b in ones:  # b: 1 -> 0
                # Compute delta for lag k_star
                delta = 0
                if a <= n-1-k_star: delta += 1 - h[a+k_star]
                if a >= k_star: delta -= h[a-k_star]
                if b <= n-1-k_star: delta += h[b+k_star] - 1
                if b >= k_star: delta += h[b-k_star]
                
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (a, b)
        
        if best_swap is None:
            break  # no improving swap
        
        # Apply the swap
        a, b = best_swap
        h[a] = 1
        h[b] = 0
        
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
        else:
            # Revert (the swap improved worst lag but worsened another)
            h[a] = 0
            h[b] = 1
            break
    return h, best_score

from scipy.optimize import minimize
import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_relaxation(n, max_time=60):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    # z variables for each (i, k) pair where i + k < n
    # z_{i,k} represents h_i * h_{i+k}
    z = {}
    constraints = [
        cp.sum(h) == n / 2.0,
        h >= 0,
        h <= 1
    ]
    
    for k in range(1, n):
        for i in range(n - k):
            z[(i, k)] = cp.Variable()
            constraints.append(z[(i, k)] <= h[i])
            constraints.append(z[(i, k)] <= h[i + k])
            constraints.append(z[(i, k)] >= h[i] + h[i + k] - 1)
            constraints.append(z[(i, k)] >= 0)
    
    # Objective constraints: for each k, sum_i h[i]*(1-h[i+k]) <= t/dx
    for k in range(1, n):
        # sum_{i=0}^{n-1-k} h[i] - z[i,k] <= t/dx
        left = cp.sum(h[:n-k]) - cp.sum([z[(i, k)] for i in range(n - k)])
        constraints.append(left <= t / dx)
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    best_val = None
    best_h = None
    for solver in ['GLOP', 'GLPK', 'CLARABEL', 'ECOS']:
        try:
            problem.solve(solver=solver, time_limit=max_time)
            if problem.status == 'optimal' or problem.status == 'optimal_inaccurate':
                val = t.value
                if best_val is None or val < best_val:
                    best_val = val
                    best_h = h.value.copy()
        except Exception:
            pass
    
    if best_h is None:
        return None, None
    return best_h, best_val * dx  # Convert to C5 scale

Let me think about option 2: incremental update.

When we swap h[a] (0→1) and h[b] (1→0), the change in c_k for each k is:
Δc_k = I(a ≤ n-1-k)*(1-h[a+k]) - I(a ≥ k)*h[a-k] + I(b ≤ n-1-k)*(h[b+k]-1) + I(b ≥ k)*h[b-k]

This is O(1) per k, so O(n) for all k.

So the algorithm becomes:
1. Maintain the full c vector
2. For each candidate swap, compute Δc (O(n))
3. Check if max(c + Δc) < max(c) (O(n))
4. If accepted, update c += Δc

Total per iteration: O(n) for delta + O(n) for max check = O(n)
For 50000 iterations with n=400: 20 million operations. Feasible!

Actually, let me reconsider. The delta for a swap (a,b) at all lags is:

f(a, k) = I(a+k < n) * (1-h[a+k]) - I(a ≥ k) * h[a-k]
g(b, k) = I(b+k < n) * (h[b+k]-1) + I(b ≥ k) * h[b-k]

I can precompute f(a, :) and g(b, :) for the chosen a and b in O(n) each. Then Δc = f + g, and I check if max(c + f + g) < c_max.
