    constraint_violation = np.sum(h) - target

            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k]) * dx

            grad[:n-m] += w * (1.0 - h[m:]) * dx
            grad[m:] -= w * h[:n-m] * dx
    
    # Add augmented Lagrangian gradient

    return grad

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha, dx):

    c_max = np.max(corr)

def smooth_max_grad(h, alpha, dx, top_m=None):

    c_max = np.max(corr)

    if wsum < 1e-300:

    weights = exp_shifted / wsum

def s_lep_step(h, n, target, dx, solver='SCS', max_iters=200000, eps=1e-10):
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        if solver == 'SCS':
            prob.solve(solver=cp.SCS, max_iters=max_iters, eps=eps, verbose=False)
        elif solver == 'ECOS':
            prob.solve(solver=cp.ECOS, max_iters=max_iters, verbose=False)
        elif solver == 'CLARABEL':
            prob.solve(solver=cp.CLARABEL, verbose=False)
        else:
            prob.solve()
    except Exception:
        # Try fallback
        try:
            prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
        except Exception:
            return h.copy()
    
    if h_var.value is None:
        return h.copy()
    
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def lbfgs_augmented_lagrangian(h_init, n, target, alpha, time_limit, start_time, 
                                max_outer=20, max_iter_inner=500, rho_init=1.0):

    best_h = h.copy()

            best_c5 = c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def smooth_max_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx

            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx

def s_lep_single_step(h, n, target, dx, max_iters=200000, eps=1e-10):
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=max_iters, eps=eps, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=max_iters, verbose=False)
        except Exception:
            return h.copy()
    if h_var.value is None:
        return h.copy()
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_opt, c5_opt = lbfgs_al_optimize(
            best_h, n_use, target_use, 50000,
            time_limit=min(15, tleft() * 0.5),
            start_time=start_time, max_outer=8, max_iter_inner=150)
        if c5_opt < best_score:
            best_score = c5_opt
            best_h = h_opt

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
