    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_objective(x, alpha, target, mu)
    
    def g(x):
        return lse_gradient(x, alpha, target, mu)
    
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': pgtol, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def optimize_topk(h, target, K, mu, maxiter, deadline, pgtol=1e-10):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return topk_obj_and_grad(x, K, target, mu)[0]
    
    def g(x):
        return topk_obj_and_grad(x, K, target, mu)[1]
    
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': pgtol, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def equalize_step(h, target, n_top, n_steps, lr_init, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = float(np.mean(top_vals))
        top_std = float(np.std(top_vals))
        if top_std < 1e-14:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            grad = grad / gnorm
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    return h_cur, cur_score

def make_block_start(n, target, rng, block_len):
    n_blocks = max(1, n // block_len)
    total_ones_needed = int(target)
    # Create alternating blocks
    h = np.zeros(n)
    # Greedily assign ones to blocks
    blocks = []
    pos = 0
    while pos < n:
        blen = min(block_len, n - pos)
        blocks.append((pos, pos + blen))
        pos += blen
    # Shuffle block order
    rng.shuffle(blocks)
    ones_assigned = 0
    for (s, e) in blocks:
        remaining = total_ones_needed - ones_assigned
        assign = min(remaining, e - s)
        if assign > 0:
            # Randomly place ones within the block
            idx = rng.choice(e - s, size=assign, replace=False)
            h[s + idx] = 1.0
            ones_assigned += assign
    return project_feasible(h, target)
