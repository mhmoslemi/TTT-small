import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_lse_objective(x, alpha, target, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val


def compute_lse_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad


def compute_topk_obj_grad(x, target, mu, k_top):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[::-1][:k_top]
    val = float(np.sum(corr[top_idx])) / k_top
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    grad = np.zeros(n)
    for ki in top_idx:
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += (1.0 - x[j1 - lag]) / k_top
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= x[j2 + lag] / k_top
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += (1.0 - x[j1 + m]) / k_top
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= x[j2 - m] / k_top
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_step(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return compute_lse_objective(x, alpha, target, mu)

    def g(x):
        return compute_lse_grad(x, alpha, target, mu)

    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 80,
                                'gtol_x': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h


def topk_lbfgs_step(h, target, mu, k_top, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return compute_topk_obj_grad(x, target, mu, k_top)[0]

    def g(x):
        return compute_topk_obj_grad(x, target, mu, k_top)[1]

    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 80})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_step(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return compute_lse_objective(x, alpha, target, 1e10)

    def g(x):
        return compute_lse_grad(x, alpha, target, 1e10)

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h


def equalize_active(h, target, n_top, n_steps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.003
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-10:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (spread + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_try = h_cur - lr * grad
        h_try = project_feasible(h_try, target)
        sc_try = compute_c5(h_try)
        if sc_try < cur_score:
            h_cur = h_try
            cur_score = sc_try
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h_cur, cur_score


def upsample(h_old, n_new):
    n_old = len(h_old)
    target_new = n_new / 2.0
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, target_new)
    return h_new


def coordinate_descent_fast(h, target, n_sweeps, deadline):

    h_cur = h.copy()

        improved = False
        for j in range(n):

            hd[j] = d
            hd = project_feasible(hd, target)
            corr_d = np.correlate(hd, 1.0 - hd, mode='full')
            fd = float(np.max(corr_d))
            for _ in range(20):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    hc2 = h_cur.copy()
                    hc2[j] = c
                    hc2 = project_feasible(hc2, target)
                    corr_c2 = np.correlate(hc2, 1.0 - hc2, mode='full')
                    fc = float(np.max(corr_c2))
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    hd2 = h_cur.copy()
                    hd2[j] = d
                    hd2 = project_feasible(hd2, target)
                    corr_d2 = np.correlate(hd2, 1.0 - hd2, mode='full')
                    fd = float(np.max(corr_d2))
            best_v = (a + b) / 2.0
            h_new = h_cur.copy()
            h_new[j] = best_v

def diverse_restart(base_h, n, target, rng, strategy, param):
    h_pert = base_h.copy()
    if strategy == 'gauss':
        h_pert += rng.randn(n) * param
    elif strategy == 'sparse_flip':
        n_flip = max(1, int(n * param))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_pert[idx] = 1.0 - h_pert[idx]
    elif strategy == 'smooth_pert':
        freq = np.fft.rfftfreq(n)
        n_freq = len(freq)
        mask = np.zeros(n_freq, dtype=bool)
        n_low = max(2, int(n_freq * param))
        mask[:n_low] = True
        pert_spec = np.where(mask, rng.randn(n_freq) + 1j * rng.randn(n_freq), 0)
        pert = np.fft.irfft(pert_spec, n)
        h_pert += pert * 0.1
    elif strategy == 'block_shift':
        block_size = max(4, n // 20)
        n_shifts = 3
        for _ in range(n_shifts):
            start = rng.randint(0, n - block_size)
            end = start + block_size
            shift = rng.randint(1, block_size // 2)
            h_pert[start:end] = np.roll(h_pert[start:end], shift)
    elif strategy == 'binarize_noisy':
        h_pert = (h_pert > 0.5).astype(float)
        h_pert += rng.randn(n) * 0.1 * param
    elif strategy == 'compress_expand':
        n_coarse = max(50, n // 4)
        x_coarse = np.linspace(0, 1, n_coarse)
        h_coarse = np.interp(x_coarse, np.linspace(0, 1, n), h_pert)
        x_fine = np.linspace(0, 1, n)
        h_pert = np.interp(x_fine, x_coarse, h_coarse)
        h_pert += rng.randn(n) * 0.02 * param
    h_pert = project_feasible(h_pert, target)
    return h_pert

    return h_final, final_score, len(h_final)
