def coord_descent_step(h, j, n, dx, overlaps):
    """Minimize the objective w.r.t. h[j] given all other h fixed."""
    # Compute A(k, j) for all lags k = -(n-1), ..., 0, ..., n-1
    # A(k,j) = (1-h[j+k]) if 0<=j+k<n else 0, minus h[j-k] if 0<=j-k<n else 0
    
    n_lags = 2 * n - 1
    N_half = n - 1
    
    # (1-h[j+k]) for valid k: j+k in [0, n-1] → k in [-j, n-1-j]
    A = np.zeros(n_lags)
    
    # Term 1: (1-h[j+k]) for k where 0 <= j+k < n
    lo1 = max(0, -j)  # k >= -j (but k >= -(n-1) always)
    hi1 = min(n-1, n-1-j)  # j+k <= n-1 → k <= n-1-j
    if hi1 >= lo1:
        A[lo1:hi1+1] += (1.0 - h[j + np.arange(lo1, hi1+1)])
    
    # Term 2: -h[j-k] for k where 0 <= j-k < n
    lo2 = max(0, j-(n-1))  # j-k >= 0 → k <= j; j-k <= n-1 → k >= j-(n-1)
    hi2 = min(n-1, j)
    if hi2 >= lo2:

    # Now f(z) = max_k (overlaps[k] + dx * (z - h[j]) * A[k])
    # = max_k (overlaps[k] - dx*h[j]*A[k] + dx*A[k]*z)
    # = max_k (e_k + f_k * z) where e_k = overlaps[k] - dx*h[j]*A[k], f_k = dx*A[k]

    e = overlaps - dx * h_j * A
    f = dx * A

    # Use grid + refinement
    z_grid = np.linspace(0, 1, 50)
    # Matrix: (n_lags, 50)
    M = e[:, None] + f[:, None] * z_grid[None, :]
    vals = np.max(M, axis=0)
    z_best = z_grid[np.argmin(vals)]
    
    # Refine with bisection (3 iterations)
    lo, hi = 0.0, 1.0

        v_mid = np.max(e + f * mid)

            lo = mid
        else:
            hi = mid
    
    z_opt = (lo + hi) / 2
    return z_opt

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

```python
def lbflobs_optimize(h_init, n, target, dx, alphas, maxiter, penalty=1e4):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for alpha in alphas:
        def obj(h):
            overlaps = compute_all_overlaps(h)
            max_val = np.max(overlaps)
            shifted = alpha * (overlaps - max_val)
            shifted = np.clip(shifted, -50, 0)
            smooth_max = max_val + np.log(np.sum(np.exp(shifted))) / alpha
            penalty_term = penalty * (np.sum(h) - target) ** 2
            return smooth_max + penalty_term
        
        def grad(h):
            overlaps = compute_all_overlaps(h)
            max_val = np.max(overlaps)
            shifted = alpha * (overlaps - max_val)
            shifted = np.clip(shifted, -50, 0)
            weights = np.exp(shifted)
            weights /= weights.sum()
            
            N_half = n - 1
            g = np.zeros(n)
            for m, w in enumerate(weights):
                if w < 1e-10:
                    continue
                k = m - N_half
                # A(k, j) for all j
                # Term 1: (1-h[j+k]) for valid j
                j_lo1 = max(0, -k)
                j_hi1 = min(n-1, n-1-k)
                if j_hi1 >= j_lo1:
                    jj = np.arange(j_lo1, j_hi1+1)
                    g[jj] += w * (1.0 - h[jj + k])
                # Term 2: -h[j-k] for valid j
                j_lo2 = max(0, k)
                j_hi2 = min(n-1, n-1+k)
                if j_hi2 >= j_lo2:
                    jj = np.arange(j_lo2, j_hi2+1)
                    g[jj] += w * (-h[jj - k])
            
            g *= dx
            g += 2 * penalty * (np.sum(h) - target) * (1.0 / n)
            
            return g
        
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=[(0,1)]*n,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
```

One more consideration: the `minimize_upper_envelope` function. Let me double-check the logic.

We want to minimize f(z) = max_k (e_k + f_k * z) for z ∈ [0, 1].

f(z) is convex (max of linear functions). It's piecewise linear with knots at the intersection points.

At z, the subdifferential is {f_k : k ∈ active(z)} where active(z) = {k : e_k + f_k * z = f(z)}.

f(z) is minimized at z* iff 0 ∈ ∂f(z*), i.e., min_{k∈active} f_k ≤ 0 ≤ max_{k∈active} f_k.

Bisection:
- If min_{k∈active(mid)} f_k > 0: all active slopes are positive, so f is increasing at mid, minimum is to the left → z_hi = mid
- If max_{k∈active(mid)} f_k < 0: all active slopes are negative, so f is decreasing at mid, minimum is to the right → z_lo = mid
- Otherwise: 0 is in the subdifferential, mid is optimal → stop

This is correct.

Edge case: if the function is flat (all f_k = 0), then any z is optimal. The bisection will stop immediately since min_slope = max_slope = 0.

Another edge case: if e + f*0 and e + f*1 are both equal (the function is constant), same thing.
