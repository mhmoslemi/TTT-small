def project_feasible(h, target):
    """Euclidean projection onto {0<=x<=1, sum(x)=target}"""
    h = np.array(h, dtype=float)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def project_binary(h, target):
    h_r = np.round(h).astype(int)
    # Adjust to get exactly target ones
    s = h_r.sum()
    if s > target:
        excess = s - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        h_r[ones[order[:excess]]] = 0
    elif s < target:
        deficit = target - s
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
