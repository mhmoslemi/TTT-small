OK let me just carefully derive compute_delta.

The full correlation c[m] for m = 0, ..., 2n-2 is defined as:
c[m] = sum over valid i of h[i] * (1-h)[i + m - (n-1)]

Let me define the shift: for output index m, we're computing sum_i h[i] * (1-h[i + m - n + 1])

Let s = m - (n-1), so s ranges from -(n-1) to n-1.
c[m] = sum_i h[i] * (1-h[i+s]) where the sum is over i such that both i and i+s are in [0, n-1].

Now, we swap: h[i] goes from 1 to 0, h[j] goes from 0 to 1.

The change in c[m] = change from h[i] terms + change from h[j] terms.

**Change from h[i]: 1 → 0**

Terms where h[i] appears as the first factor: h[i] * (1-h[i+s])
Before: 1 * (1-h[i+s]) = 1 - h[i+s]
After: 0 * (1-h[i+s]) = 0
Change: -(1 - h[i+s]) = h[i+s] - 1

This is valid when i+s is in [0, n-1], i.e., -i ≤ s ≤ n-1-i.

Terms where h[i] appears as the second factor (inside 1-h): h[i-s] * (1-h[i])
Before: h[i-s] * (1-1) = 0
After: h[i-s] * (1-0) = h[i-s]
Change: +h[i-s]

This is valid when i-s is in [0, n-1], i.e., i-n+1 ≤ s ≤ i.

**Change from h[j]: 0 → 1**

Terms where h[j] appears as the first factor: h[j] * (1-h[j+s])
Before: 0 * (1-h[j+s]) = 0
After: 1 * (1-h[j+s]) = 1 - h[j+s]
Change: +(1 - h[j+s]) = 1 - h[j+s]

Valid when j+s ∈ [0, n-1], i.e., -j ≤ s ≤ n-1-j.

Terms where h[j] appears as the second factor: h[j-s] * (1-h[j])
Before: h[j-s] * (1-0) = h[j-s]
After: h[j-s] * (1-1) = 0
Change: -h[j-s]

Valid when j-s ∈ [0, n-1], i.e., j-n+1 ≤ s ≤ j.

So the total change in c[m] (where s = m - (n-1)):
Δc[m] = [h[i+s] - 1 if 0 ≤ i+s < n else 0]
       + [h[i-s] if 0 ≤ i-s < n else 0]
       + [1 - h[j+s] if 0 ≤ j+s < n else 0]
       + [-h[j-s] if 0 ≤ j-s < n else 0]

def full_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Distribute the difference among the most "partial" elements
    if diff > 0:
        # Need to increase sum: prioritize elements closest to 1
        order = np.argsort(-h)
    else:
        # Need to decrease sum: prioritize elements closest to 0
        order = np.argsort(h)
    h = h.copy()
    for idx in order:
        if diff > 0:
            add = min(1.0 - h[idx], diff)
            h[idx] += add
            diff -= add
        else:
            sub = min(h[idx], -diff)
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-14:
            break
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def sturmian_init(n, alpha, rho=0.0):
    """Generate a Sturmian sequence."""
    i = np.arange(n)
    return np.floor((i+1)*alpha + rho) - np.floor(i*alpha + rho)

def make_initial(n, rng):
    """Create a diverse initial binary solution."""
    # Try several constructions and pick a reasonable one
    # 1. Alternating with small perturbation
    h1 = np.zeros(n)
    h1[::2] = 1.0  # 101010...
    # Perturb randomly
    for _ in range(n//10):
        i = rng.randint(n)
        j = rng.randint(n)

    h2 = project_binary(h2, n//2)
    
    # 3. Random

    # Return the one with best C5
    scores = [compute_c5(h) for h in [h1, h2, h3]]
    return [h1, h2, h3][np.argmin(scores)]

def greedy_optimize(h, max_iters, rng, time_limit, start_time):
    """Fast greedy: always make the swap that most reduces max corr."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fft(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        
        max_val = float(full.max())
        if max_val < 1e-10:
            break
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        
        if k_star == 0:
            stall += 1
            continue
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute A[i] = h[i+k] + h[i-k] (contribution to reducing corr at k_star)
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        
        # For moving 1->0 at position i: reduction in corr at k_star = A[i]
        # We want to maximize A[i] for i in ones
        # For moving 0->1 at position j: we don't want to increase corr at k_star
        # The increase = (1-h[j-k])_valid + (1-h[j+k])_valid... 
        # Actually let's just check the full delta
        
        # Top candidates: ones with highest A (most reduction), zeros with lowest "cost"
        top_i = ones[np.argsort(-A[ones])[:min(30, len(ones))]]
        
        # For zeros, compute the "cost" of turning on at j
        cost = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        cost[valid] += (1.0 - h[idx[valid]])
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        cost[valid] += (1.0 - h[idx[valid]])
        
        top_j = zeros[np.argsort(cost[zeros])[:min(30, len(zeros))]]
        
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        
        if not found:

            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

                new_max = float((full + delta).max())

                    h[j] = 1.0

                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    
    return best_h, best_c5

def sa_optimize(h, n_iters, t_init, t_final, rng, time_limit, start_time, block_size=1):
    """Simulated annealing with optional block moves."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fft(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 100000
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < block_size or len(zeros) < block_size:
            break
        
        if block_size == 1:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Block move: swap block_size pairs
            indices = rng.permutation(len(ones))[:block_size]
            jndices = rng.permutation(len(zeros))[:block_size]
            i_list = ones[indices]
            j_list = zeros[jndices]
            
            delta_total = np.zeros(2*n-1)
            for ii, jj in zip(i_list, j_list):
                delta_total += compute_delta(h, ii, jj)
            
            new_max = float((full + delta_total).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                for ii, jj in zip(i_list, j_list):
                    h[ii] = 0.0
                    h[jj] = 1.0
                full += delta_total
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def sa_focused(h, n_iters, t_init, t_final, rng, time_limit, start_time):

        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # 50% of the time, focus on the worst lag
        if it % 2 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                # Find ones whose removal most reduces corr at k_star

                # i should be a 1 with high A[i]

                B = np.zeros(n)
                idx = positions + k_star

                valid = (idx >= 0) & (idx < n)
                B[valid] += (1.0 - h[idx[valid]])

                if t > 0.01:

                    i_probs = np.exp(i_scores / t)
                    i_probs /= i_probs.sum()
                    j_probs = np.exp(-j_scores / t)
                    j_probs /= j_probs.sum()
                    i = ones[rng.choice(len(ones), p=i_probs)]
                    j = zeros[rng.choice(len(zeros), p=j_probs)]
                else:
                    i = ones[np.argmax(i_scores)]
                    j = zeros[np.argmin(j_scores)]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

    return best_h, best_c5

# For SLSQP
_cons_n = 0
_cons_target = 0.0

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, alphas, maxiter=3000, ftol=1e-16, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    n = len(h_start)
    target = n / 2.0
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    
    return h_cur, curr_score

def bfgs_refine(h_start, alphas, maxiter=5000, time_limit=None, start_time=None):
    """Try BFGS with penalty method for the constraint."""
    n = len(h_start)
    target = n / 2.0
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            # Use a large penalty for the constraint
            penalty = 10000.0
            def obj_with_pen(x):
                c = smooth_obj(x, alpha)
                c += penalty * (np.sum(x) - target) ** 2
                return c
            
            # Project to [0,1] bounds using a transformation
            # Use sigmoid: x = sigmoid(y), so y = log(x/(1-x))
            # Start from current h
            eps = 1e-6
            h_safe = np.clip(h_cur, eps, 1-eps)
            y0 = np.log(h_safe / (1 - h_safe))
            
            res = minimize(obj_with_pen, y0, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16})

            diff -= add
        else:

            diff += sub

def coord_descent_continuous(h, n_passes, time_limit, start_time):
    """Coordinate descent exploiting linearity of corr in each coordinate."""
    n = len(h)
    h = h.copy().astype(float)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    full = full_corr_fft(h)
    for p in range(n):
        if time.time() - start_time > time_limit:
            break
        k_offsets = np.arange(2*n-1) - (n-1)
        # Slope for each correlation bin: s[m] = 1 - h[p+k_m] - h[p-k_m]
        idx_plus = p + k_offsets
        idx_minus = p - k_offsets
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        h_plus = np.where(mask_plus, h[np.clip(idx_plus, 0, n-1)], 0.0)
        h_minus = np.where(mask_minus, h[np.clip(idx_minus, 0, n-1)], 0.0)
        slopes = (1.0 - h_plus - h_minus) * (mask_plus | mask_minus)
        # For bins where neither neighbor is valid, slope is 0 (no contribution)
        # Actually if mask is false, the term doesn't exist, so slope should be 0
        slopes = np.where(mask_plus | mask_minus, 1.0 - h_plus - h_minus, 0.0)
        # Intercept: c[m] - h[p] * slopes[m]
        intercepts = full - h[p] * slopes
        # Minimize max_m (slopes[m]*t + intercepts[m]) for t in [0,1]
        # Ternary search
        lo = 0.0
        hi = 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(slopes * m1 + intercepts))
            f2 = float(np.max(slopes * m2 + intercepts))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        new_val = (lo + hi) / 2.0
        if new_val != h[p]:
            delta = (new_val - h[p]) * slopes
            full += delta
            h[p] = new_val
    # Project to ensure sum constraint
    h = project_feasible(h, target)
    full = full_corr_fft(h)
    cur_c5 = float(full.max()) * dx
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    return best_h, best_c5

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    best_continuous_h = h_init.copy()
    best_continuous_score = compute_c5(h_init)
    if best_continuous_score < best_score:
        best_score = best_continuous_score
        best_h = h_init.copy()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    # Phase 1: Binary optimization from initial
    allowed = min(60, tleft() * 0.10)
    h_g, c5_g = greedy_optimize(best_h if (best_h == 1.0).any() else project_binary(h_init, target_init),
                                 max_iters=50000, rng=rng, time_limit=allowed, start_time=start_time)
    h_bin_best = h_g if c5_g < best_score else best_h
    bin_score = min(c5_g, compute_c5(h_bin_best))

    # Phase 2: SA from best binary
    allowed = min(120, tleft() * 0.18)
    n_sa = int(min(8000000, allowed * 60000))
    h_sa, c5_sa = sa_focused(h_bin_best, n_sa, 0.025, 0.00005, rng,
                             time_limit=allowed, start_time=start_time)
    if c5_sa < bin_score:
        bin_score = c5_sa
        h_bin_best = h_sa

    # Phase 3: Iterated greedy + SA with perturbation
    for it in range(8):
        if tleft() < 60:
            break
        # Perturb
        n_pert = rng.randint(3, 10)
        h_pert = h_bin_best.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
        # Greedy
        allowed_g = min(20, tleft() * 0.05)
        h_g2, c5_g2 = greedy_optimize(h_pert, max_iters=15000, rng=rng,
                                       time_limit=allowed_g, start_time=start_time)
        if c5_g2 < bin_score:
            bin_score = c5_g2
            h_bin_best = h_g2
        # SA
        allowed_s = min(50, tleft() * 0.08)
        n_sa2 = int(min(3000000, allowed_s * 50000))
        h_sa2, c5_sa2 = sa_focused(h_g2, n_sa2, 0.02, 0.0001, rng,
                                    time_limit=allowed_s, start_time=start_time)
        if c5_sa2 < bin_score:
            bin_score = c5_sa2
            h_bin_best = h_sa2

    # Phase 4: Coordinate descent on continuous relaxation
    if tleft() > 80:
        # Start from the binary best (as continuous)
        h_cd = h_bin_best.copy().astype(float)
        target_cd = n_init / 2.0
        allowed_cd = min(100, tleft() * 0.15)
        h_cd_out, cd_score = coord_descent_continuous(h_cd, n_passes=5,
                                                       time_limit=allowed_cd, start_time=start_time)
        if cd_score < best_score:
            best_score = cd_score
            best_h = h_cd_out.copy()
            best_continuous_h = h_cd_out.copy()
            best_continuous_score = cd_score

    # Phase 5: SLSQP refinement on continuous
    if tleft() > 60:
        alphas = [30, 80, 200, 500, 1000, 3000]
        allowed_slsqp = min(150, tleft() * 0.25)
        h_ref, ref_score = slsqp_refine(best_continuous_h, alphas, maxiter=5000,
                                         ftol=1e-16, time_limit=allowed_slsqp, start_time=start_time)
        if ref_score < best_score:
            best_score = ref_score
            best_h = h_ref
            best_continuous_h = h_ref
            best_continuous_score = ref_score

    # Phase 6: More coordinate descent passes from the SLSQP result
    if tleft() > 40:
        allowed_cd2 = min(60, tleft() * 0.15)
        h_cd2, cd2_score = coord_descent_continuous(best_continuous_h, n_passes=3,
                                                     time_limit=allowed_cd2, start_time=start_time)
        if cd2_score < best_score:
            best_score = cd2_score
            best_h = h_cd2.copy()
            best_continuous_h = h_cd2
            best_continuous_score = cd2_score

    # Phase 7: Second SLSQP pass with finer alphas
    if tleft() > 30:
        alphas2 = [5000, 10000, 20000]
        allowed2 = min(50, tleft() * 0.20)
        h_ref2, ref2_score = slsqp_refine(best_continuous_h, alphas2, maxiter=8000,
                                           ftol=1e-17, time_limit=allowed2, start_time=start_time)
        if ref2_score < best_score:
            best_score = ref2_score
            best_h = h_ref2
            best_continuous_h = h_ref2
            best_continuous_score = ref2_score

    # Phase 8: Try slightly different n values
    for n_try in [n_init + 4, n_init - 4, n_init + 8]:
        if tleft() < 40:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_continuous_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed_g = min(15, tleft() * 0.06)
        h_g3, c5_g3 = greedy_optimize(h_bin_try, max_iters=10000, rng=rng,
                                       time_limit=allowed_g, start_time=start_time)
        allowed_cd3 = min(20, tleft() * 0.06)
        h_cd3, c5_cd3 = coord_descent_continuous(h_g3, n_passes=3,
                                                  time_limit=allowed_cd3, start_time=start_time)
        sc3 = min(c5_g3, c5_cd3)
        h3 = h_g3 if c5_g3 < c5_cd3 else h_cd3
        if tleft() > 20:
            allowed_s3 = min(20, tleft() * 0.08)
            h_ref3, c5_r3 = slsqp_refine(h3, [500, 2000, 5000], maxiter=3000,
                                          ftol=1e-16, time_limit=allowed_s3, start_time=start_time)
            if c5_r3 < sc3:
                sc3 = c5_r3
                h3 = h_ref3
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 9: Final coordinate descent + SLSQP polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        best_h = project_feasible(best_h, target_use)
        allowed_final_cd = min(30, tleft() * 0.3)
        h_fcd, fcd_score = coord_descent_continuous(best_h, n_passes=3,
                                                     time_limit=allowed_final_cd, start_time=start_time)
        if fcd_score < best_score:
            best_score = fcd_score
            best_h = h_fcd
        if tleft() > 15:
            n_use2 = len(best_h)
            allowed_fin_sls = min(20, tleft() * 0.4)
            h_fs, fs_score = slsqp_refine(best_h, [5000, 15000], maxiter=5000,
                                           ftol=1e-17, time_limit=allowed_fin_sls, start_time=start_time)
            if fs_score < best_score:
                best_score = fs_score
                best_h = h_fs

    # Final validation
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
