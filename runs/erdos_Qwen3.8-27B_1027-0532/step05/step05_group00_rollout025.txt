    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(np.sum(h_r))
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Verify
    while int(np.sum(h_r)) != target_sum:
        if np.sum(h_r) > target_sum:
            h_r[np.where(h_r == 1)[0][0]] = 0
        else:
            h_r[np.where(h_r == 0)[0][0]] = 1
    return h_r

def swap_delta(h, i, j):
    """Compute delta in correlation when h[i]:1->0 and h[j]:0->1"""
    n = len(h)
    k = np.arange(-(n-1), n, dtype=np.int64)
    delta = np.zeros(2*n-1)
    
    # h[i] goes 1->0: contribution changes
    # At shift k, the term involving h[i] is:
    #   h[i] * (1-h[i+k]) if 0<=i+k<n  (i is the "1" position)
    #   (1-h[i-k]) * h[i] if 0<=i-k<n  (i is the "0" position in the pair)
    # Wait, let me be more careful.
    # corr[m] = sum over i of h[i] * (1-h[i + m - (n-1)])
    # Let shift = m - (n-1) = k
    # corr at index m = corr at shift k = sum_i h[i] * (1-h[i+k])
    
    # If h[i] changes from 1 to 0:
    #   Term where i is the first index: was 1*(1-h[i+k]), now 0. Delta = -(1-h[i+k])
    #   Term where i is the second index (i+k' = i, so k' = 0... no)
    #   Actually: corr[k] = sum_j h[j] * (1-h[j+k])
    #   If h[i] = 1 -> 0:
    #     - j = i: term was 1*(1-h[i+k]), now 0*(1-h[i+k]) = 0. Delta = -(1-h[i+k])
    #     - j+k = i, i.e., j = i-k: term was h[i-k]*(1-1) = 0, now h[i-k]*(1-0) = h[i-k]. Delta = +h[i-k]
    
    mask_pos = (i + k >= 0) & (i + k < n)
    delta[mask_pos] -= (1.0 - h[i + k[mask_pos]])
    mask_neg = (i - k >= 0) & (i - k < n)
    delta[mask_neg] += h[i - k[mask_neg]]
    
    # h[j] goes 0->1:
    #     - j = j: term was 0*(1-h[j+k]), now 1*(1-h[j+k]). Delta = +(1-h[j+k])
    #     - j+k = j... wait, j is the position that changes
    #     - j+k' = j means k'=0, that's the same position
    #     Actually: in corr[k] = sum_j h[j] * (1-h[j+k]):
    #     - When the "first" factor changes (h[j] 0->1): delta = (1-h[j+k])
    #     - When the "second" factor changes (h[j+k] 0->1, i.e., j+k = j... no)
    #     Let me redo: h[j] changes from 0 to 1.
    #     In corr[k] = sum_i h[i] * (1-h[i+k]):
    #     - If i = j: term changes from 0*(1-h[j+k]) to 1*(1-h[j+k]). Delta = (1-h[j+k])
    #     - If i+k = j, i.e., i = j-k: term changes from h[j-k]*(1-0) to h[j-k]*(1-1) = 0. Delta = -h[j-k]
    
    mask_pos2 = (j + k >= 0) & (j + k < n)
    delta[mask_pos2] += (1.0 - h[j + k[mask_pos2]])
    mask_neg2 = (j - k >= 0) & (j - k < n)
    delta[mask_neg2] -= h[j - k[mask_neg2]]
    
    return delta
