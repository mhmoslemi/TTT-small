            new_counts = counts.copy()
            for k in range(1, n - p):
                new_counts[n - 1 + k] += (1 - h[p + k])
            for k in range(1, p + 1):
                new_counts[n - 1 - k] += (1 - h[p - k])
            s = float(np.max(new_counts))
            if s < best_score:
                best_score = s
                best_p = p
        if best_p < 0:
            best_p = int(candidates[rng.randint(len(candidates))])
        h[best_p] = 1
        for k in range(1, n - best_p):
            counts[n - 1 + k] += (1 - h[best_p + k])
        for k in range(1, best_p + 1):
            counts[n - 1 - k] += (1 - h[best_p - k])
    return h.astype(float)

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def obj_soft_max_p(x, penalty):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha
    s = float(np.sum(h))
    val += penalty * (s - _g_target) ** 2
    return val

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score

def worst_lag_surgery(h, max_steps=300, initial_epsilon=0.0015):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        if k_star == 0:
            grad = dx * (1.0 - 2.0 * h_cur)
        else:
            grad = np.zeros(n)
            upper = n - 1 - k_star
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += dx * (1.0 - h_cur[idx1 + k_star])
            idx2 = np.arange(k_star, n)
            grad[idx2] -= dx * h_cur[idx2 - k_star]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.01)
        else:
            eps *= 0.85
            if eps < 1e-9:
                break
    return h_cur, cur_score

def top_lag_surgery(h, max_steps=200, initial_epsilon=0.001, n_lags=3):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        top_lags = np.argsort(over)[::-1][:n_lags]
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k == 0:
                grad += dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= dx * h_cur[idx2 - k]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.015, 0.008)
        else:
            eps *= 0.88
            if eps < 1e-9:
                break
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def try_rounding_improve(h, target):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(h)
    for thresh in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        h_r = binarize_h(h, target)
        h_r[h >= thresh] = 1.0
        h_r[h < thresh] = 0.0
        n_ones = int(h_r.sum())
        if n_ones != int(target):
            if n_ones > int(target):
                idx = np.where(h_r == 1.0)[0]
                order = np.argsort(h[idx])[:n_ones - int(target)]
                h_r[idx[order]] = 0.0
            else:
                idx = np.where(h_r == 0.0)[0]
                order = np.argsort(-h[idx])[:int(target) - n_ones]
                h_r[idx[order]] = 1.0
        h_r = project_feasible(h_r, target)
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best = h_r.copy()
    return best, best_score

def tleft(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Generate diverse binary initializations
    candidates = []
    h_bin1 = binarize_h(h_init, target)
    candidates.append(h_bin1)
    h_bin2 = greedy_binary(n, rng)
    candidates.append(h_bin2)
    h_bin3 = binarize_h(h_init + rng.randn(n) * 0.1, target)
    candidates.append(h_bin3)
    h_bin4 = binarize_h(h_init - rng.randn(n) * 0.08, target)
    candidates.append(h_bin4)

    for h_c in candidates:
        sc = compute_c5(h_c)
        if sc < best_score:
            best_score = sc
            best_h = h_c.copy()

    # Phase 2: Binary SA on best binary candidate
    if tleft(start_time, budget_s) > 400:
        h_bin_best = binarize_h(best_h, target)
        h_sa, sc_sa = binary_sa(h_bin_best, 30000, rng, temp_init=10.0, temp_floor=1.0)
        sc_sa_c5 = sc_sa * (2.0 / n)
        if sc_sa_c5 < best_score:
            best_score = sc_sa_c5
            best_h = h_sa

        h_tbs, sc_tbs = targeted_binary_search(best_h, 15000, rng)
        sc_tbs_c5 = sc_tbs * (2.0 / n)
        if sc_tbs_c5 < best_score:
            best_score = sc_tbs_c5
            best_h = h_tbs

    # Phase 3: SLSQP from binary starting point
    if tleft(start_time, budget_s) > 300:
        h_ref, sc = slsqp_optimize(best_h, [50, 200, 1000, 5000], 3000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Rounding + re-optimization
    if tleft(start_time, budget_s) > 250:
        h_round, sc_round = try_rounding_improve(best_h, target)
        if sc_round < best_score:
            best_score = sc_round
            best_h = h_round
        h_ref, sc = slsqp_optimize(best_h, [500, 2000, 10000], 2500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Subgradient with momentum (multiple settings)
    if tleft(start_time, budget_s) > 200:
        h_sg, sc_sg = subgradient_momentum(best_h, 800, 5, 0.004, 0.997, 0.3, 0.004)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sg2, sc_sg2 = subgradient_momentum(best_h, 600, 6, 0.003, 0.998, 0.4, 0.002)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: High-alpha SLSQP
    if tleft(start_time, budget_s) > 150:
        h_ref, sc = slsqp_optimize(best_h, [10000, 50000, 200000], 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Targeted surgery
    if tleft(start_time, budget_s) > 100:
        h_surg, sc_surg = top_lag_surgery(best_h, max_steps=300, initial_epsilon=0.001, n_lags=3)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, max_steps=300, initial_epsilon=0.002)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2
        h_ref, sc = slsqp_optimize(best_h, [50000, 200000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Multi-resolution
    if tleft(start_time, budget_s) > 300:
        for n_fine in [n + 100, n + 200, n + 400]:
            if tleft(start_time, budget_s) < 120:
                break
            target_fine = n_fine / 2.0
            if target_fine != int(target_fine):
                continue
            h_fine = interpolate_to(best_h, n_fine, int(target_fine))
            h_fine = binarize_h(h_fine, int(target_fine))
            h_fine, sc_fine = binary_sa(h_fine, 10000, rng, temp_init=6.0, temp_floor=0.5)
            sc_fine_c5 = sc_fine * (2.0 / n_fine)
            h_fine_fine = h_fine
            if sc_fine_c5 < best_score:
                best_score = sc_fine_c5
                best_h = h_fine_fine.copy()
                n = n_fine
                target = target_fine
            h_ref_fine, sc_rf = slsqp_optimize(h_fine_fine, [200, 1000, 5000], 1500, 1e-15)
            if sc_rf < best_score:
                best_score = sc_rf
                best_h = h_ref_fine
                n = n_fine
                target = target_fine
            h_sg_fine, sc_sgf = subgradient_momentum(best_h, 400, 5, 0.003, 0.997, 0.3, 0.003)
            if sc_sgf < best_score:
                best_score = sc_sgf
                best_h = h_sg_fine
            h_ref_fine2, sc_rf2 = slsqp_optimize(best_h, [10000, 50000], 1000, 1e-16)
            if sc_rf2 < best_score:
                best_score = sc_rf2
                best_h = h_ref_fine2

    # Phase 9: Random restarts
    deadline_p9 = start_time + (budget_s - 30.0) * 0.82
    for attempt in range(8):
        if time.time() > deadline_p9:
            break
        if tleft(start_time, budget_s) < 60:
            break
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pert = best_h.copy()
        if attempt % 4 == 0:
            h_pert += rng.randn(n_cur) * 0.05
        elif attempt % 4 == 1:
            block = max(10, n_cur // 25)
            for b in range(n_cur // block):
                if rng.random() < 0.3:
                    shift = rng.uniform(-0.1, 0.1)
                    h_pert[b*block:min((b+1)*block, n_cur)] += shift
        elif attempt % 4 == 2:
            h_pert = project_feasible(binarize_h(h_pert, target_cur) * 0.3 + best_h * 0.7, target_cur)
        else:
            n_flips = max(1, n_cur // 15)
            flip_idx = rng.choice(n_cur, n_flips, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_feasible(h_pert, target_cur)
        h_bin_pert = binarize_h(h_pert, target_cur)
        h_bin_pert, sc_bp = binary_sa(h_bin_pert, 5000, rng, temp_init=5.0, temp_floor=0.5)
        sc_bp_c5 = sc_bp * (2.0 / n_cur)
        h_start = h_bin_pert if sc_bp_c5 < compute_c5(h_pert) else h_pert
        h_ref, sc = slsqp_optimize(h_start, [500, 2000, 10000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final subgradient passes
    if tleft(start_time, budget_s) > 60:
        h_sg, sc_sg = subgradient_momentum(best_h, 600, 5, 0.002, 0.998, 0.3, 0.003)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sg2, sc_sg2 = subgradient_momentum(best_h, 400, 7, 0.001, 0.999, 0.5, 0.001)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 11: Final high-precision SLSQP
    if tleft(start_time, budget_s) > 40:
        h_ref, sc = slsqp_optimize(best_h, [50000, 200000, 500000], 1500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 12: Final surgery
    if tleft(start_time, budget_s) > 20:
        h_surg, sc_surg = worst_lag_surgery(best_h, max_steps=150, initial_epsilon=0.001)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = top_lag_surgery(best_h, max_steps=100, initial_epsilon=0.0008, n_lags=4)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    # Phase 13: One last SLSQP
    if tleft(start_time, budget_s) > 10:
        h_final, sc_final = slsqp_optimize(best_h, [100000, 500000], 500, 1e-17)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
