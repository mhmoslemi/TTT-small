```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.99999
        else:
            mask = h > 1e-09
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coord_descent(h, n, target, n_passes, deadline, reproject_every=5):
    h_cur = h.copy()
    c = compute_corr(h_cur)
    dx = 2.0 / n
    best_score = float(np.max(c)) * dx
    best_h = h_cur.copy()
    k_vals = np.arange(2 * n - 1)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            old_val = h_cur[j]
            idx1 = j + k_vals - (n - 1)
            idx2 = j + (n - 1) - k_vals
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            slopes = np.zeros(2 * n - 1)
            if np.any(valid1):
                slopes[valid1] += (1.0 - h_cur[idx1[valid1]])
            if np.any(valid2):
                slopes[valid2] -= h_cur[idx2[valid2]]
            b = c - slopes * old_val
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = float(np.max(b + slopes * m1))
                f2 = float(np.max(b + slopes * m2))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            f_opt = float(np.max(b + slopes * t_opt))
            f_0 = float(np.max(b))
            f_1 = float(np.max(b + slopes))
            if f_0 < f_opt:
                t_opt = 0.0
                f_opt = f_0
            if f_1 < f_opt:
                t_opt = 1.0
                f_opt = f_1
            if abs(t_opt - old_val) > 1e-13:
                delta = t_opt - old_val
                h_cur[j] = t_opt
                c += slopes * delta
        if (p + 1) % reproject_every == 0:
            h_cur = project_feasible(h_cur, target)
            c = compute_corr(h_cur)
        new_score = float(np.max(c)) * dx
        if new_score < best_score:
            best_score = new_score
            best_h = h_cur.copy()
    h_cur = project_feasible(h_cur, target)
    final_score = float(np.max(compute_corr(h_cur))) * dx
    if final_score < best_score:
        best_score = final_score
        best_h = h_cur.copy()
    return best_h, best_score


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def equalize_subgradient(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        top_vals = corr[top_idx]
        c_star = top_vals[0]
        weights = np.exp((top_vals - c_star) / 0.001)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            m = k - (n - 1)
            if m >= 0:
                j_pos = np.arange(0, n - m)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + m])
                j_neg = np.arange(m, n)
                grad[j_neg] -= wk * h_cur[j_neg - m]
            else:
                mm = -m
                j_pos = np.arange(0, n - mm)
                grad[j_pos + mm] += wk * (1.0 - h_cur[j_pos])
                j_neg = np.arange(0, n - mm)
                grad[j_neg] -= wk * h_cur[j_neg + mm]
                j_pos2 = np.arange(0, n - mm)
                grad2 = np.zeros(n)
                for i in range(n - mm):
                    grad2[i] += wk * (1.0 - h_cur[i + mm])
                    grad2[i + mm] -= wk * h_cur[i]
                grad = grad2
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()

            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 5
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 8)
                amp = rng.uniform(0.1, 0.5)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.3, 0, 1), target)
            else:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.5, 0.05, 0.95), target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score
```
