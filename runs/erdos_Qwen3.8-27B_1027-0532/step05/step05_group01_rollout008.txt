```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def slp_optimize(h_init, n, target, time_limit, M_start=5, M_end=30):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_iters = 0
    
    while time.time() - start < time_limit:
        # Adapt M
        frac = (time.time() - start) / time_limit
        M = int(M_start + (M_end - M_start) * frac)
        M = max(3, min(40, M))
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Get top M lags
        top_idx = np.argsort(corr)[-M:]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for li in top_idx:
            s = int(li) - (n - 1)
            i_lo = max(0, -s)
            i_hi = min(n - 1, n - 1 - s)
            if i_hi < i_lo:
                continue
            vi = np.arange(i_lo, i_hi + 1)
            
            # B value and gradient
            B_val = float(np.sum(h[vi] * h[vi + s]))
            gB = np.zeros(n)
            gB[vi] += h[vi + s]
            gB[vi + s] += h[vi]
            
            # A coefficients
            cA = np.zeros(n)
            cA[vi + s] = 1.0
            
            # Constraint: cA @ h_new - (B_val + gB @ (h_new - h)) <= t
            # cA @ h_new - gB @ h_new <= t + B_val - gB @ h
            lhs_c = cA - gB
            rhs_b = B_val - float(np.dot(gB, h))
            constraints.append(lhs_c @ h_var - t_var <= rhs_b)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value.flatten(), 0.0, 1.0)
        
        # Step size
        n_iters += 1
        lam = max(0.05, 1.0 * (0.98 ** n_iters))
        h_new = (1.0 - lam) * h + lam * h_lp
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_now - 1e-15:
            h = h_new
        elif n_iters > 100:
            # Try full step
            h_try = project_feasible(h_lp, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now:
                h = h_try
    
    return best_h, best_c5
```
