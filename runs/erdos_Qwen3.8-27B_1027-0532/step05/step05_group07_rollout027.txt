def smooth_obj(x, alpha, n, dx):

def slsqp_optimize(h, n, target, alphas, maxiter=5000, ftol=1e-18):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for a in alphas:
        def obj(x):
            return smooth_obj(x, a, n, dx)
        def con_fun(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

def active_set_refinement(h, n, target, max_rounds=50, tol=1e-12):

            b_val -= h[idx]
        idx = i - m + offset
        if 0 <= idx < n:

        B[m] = b_val

    nl1 = 2 * n - 1
    offset = n - 1
    B = np.zeros(nl1)
    m = np.arange(nl1)
    idx1 = j + m - offset
    idx2 = j - m + offset
    idx3 = i + m - offset
    idx4 = i - m + offset
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    mask3 = (idx3 >= 0) & (idx3 < n)
    mask4 = (idx4 >= 0) & (idx4 < n)
    if np.any(mask1):
        B[mask1] += h[idx1[mask1]]
    if np.any(mask2):
        B[mask2] += h[idx2[mask2]]
    if np.any(mask3):
        B[mask3] -= h[idx3[mask3]]
    if np.any(mask4):
        B[mask4] -= h[idx4[mask4]]
    return B

def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=40):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep = 0

    while sweep < (n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break
        sweep += 1

        if sweep % recompute_interval == 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            B = compute_B(h_cur, i, j, n)
            old_i = h_cur[i]
            u_min = t_min - old_i
            u_max = t_max - old_i

            pos = B > 1e-14
            neg = B < -1e-14
            zero = ~(pos | neg)

            hi = float(np.max(corr + u_max * B))
            lo = 0.0

            for _ in range(50):
                mid = (lo + hi) * 0.5
                if np.any(zero) and np.max(corr[zero]) > mid + 1e-14:
                    lo = mid
                    continue
                ub = u_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - corr[pos]) / B[pos])))
                lb = u_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - corr[neg]) / B[neg])))
                if lb <= ub + 1e-14:
                    hi = mid
                else:
                    lo = mid

            v = hi
            ub = u_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - corr[pos]) / B[pos])))
            lb = u_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - corr[neg]) / B[neg])))
            ub = min(ub, u_max)
            lb = max(lb, u_min)

            if lb > ub + 1e-14:
                u_star = (u_min + u_max) * 0.5
            else:
                u_star = (lb + ub) * 0.5

            if abs(u_star) > 1e-14:
                h_cur[i] = old_i + u_star
                h_cur[j] = c - h_cur[i]
                corr += u_star * B

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 200 and sweep > 100:
            pert = rng.randn(n) * 0.005
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def smooth_obj_lse(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx

def smooth_obj_pnorm(x, p, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = (np.sum(np.power(np.maximum(corr, 0.0), p))) ** (1.0 / p)
    return val * dx

def slsqp_full(h, n, target, alphas, maxiter=5000, ftol=1e-18):
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for a in alphas:
        def obj_lse(x):
            return smooth_obj_lse(x, a, dx)
        def con_eq(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_lse, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

def active_set_refinement(h, n, target, max_rounds=30, tol=1e-13):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)

    for r in range(max_rounds):
        if time.time() > active_set_refinement._deadline:
            break
        free_mask = (h_cur > 5e-7) & (h_cur < 1.0 - 5e-7)
        free_indices = np.where(free_mask)[0]
        n_free = len(free_indices)

        if n_free == 0:
            break
        if n_free > 380:
            return h_cur, best_score

        fixed_sum = np.sum(h_cur[~free_mask])
        free_target = target - fixed_sum
        h_free = h_cur[free_indices].copy()
        bounds_free = [(0.0, 1.0)] * n_free

        for alpha in [500, 2000, 8000, 30000]:
            if time.time() > active_set_refinement._deadline:
                break

            def obj_sub(x, _h_cur=h_cur, _free_indices=free_indices, _alpha=alpha, _dx=dx, _n=n):
                ht = _h_cur.copy()
                ht[_free_indices] = x
                corr = np.correlate(ht, 1.0 - ht, mode='full')
                c_max = np.max(corr)
                shifted = _alpha * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                val = c_max + np.log(np.sum(np.exp(shifted))) / _alpha
                return val * _dx

            def con_sub(x, _ft=free_target):
                return np.sum(x) - _ft

            def jac_sub(x, _nf=n_free):
                return np.ones(_nf)

            try:
                res = minimize(obj_sub, h_free, method='SLSQP', bounds=bounds_free,
                               constraints=[{'type': 'eq', 'fun': con_sub, 'jac': jac_sub}],
                               options={'maxiter': 5000, 'ftol': 1e-18})
                h_new_free = project_feasible_1d(res.x, free_target)
                h_test = h_cur.copy()
                h_test[free_indices] = h_new_free
                sc = compute_c5(h_test)
                if sc < best_score - 1e-16:
                    best_score = sc
                    h_cur = h_test
                    break
            except Exception:
                pass

    h_out = project_feasible(h_cur, target)
    return h_out, compute_c5(h_out)

def structured_perturbation(h, n, rng, mode='sine'):
    if mode == 'sine':
        freq = rng.randint(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        x = np.arange(n)
        pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 'multi_sine':
        n_modes = rng.randint(2, 5)
        pert = np.zeros(n)
        for _ in range(n_modes):
            freq = rng.choice(np.arange(1, n // 2 + 1))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.04)
            x = np.arange(n)
            pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 'block':
        width = rng.randint(n // 8, n // 3)
        start = rng.randint(0, n - width)
        pert = np.zeros(n)
        pert[start:start + width] = rng.choice([1, -1]) * rng.uniform(0.03, 0.08)
    elif mode == 'gaussian':
        pert = rng.randn(n) * rng.uniform(0.02, 0.06)
    else:
        pert = rng.randn(n) * 0.03
    return h + pert

def trust_constr_optimize(h, n, target, dx, maxiter=3000):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = 5000.0 * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + np.log(np.sum(np.exp(shifted))) / 5000.0
        return val * dx

    from scipy.optimize import LinearConstraint, Bounds
    A = np.ones((1, n))
    lin_con = LinearConstraint(A, target, target)
    bounds = Bounds(0.0, 1.0)
    try:
        res = minimize(obj, h, method='trust-constr', bounds=bounds,
                       constraints=[lin_con], options={'maxiter': maxiter, 'gtol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def block3_coord_desc(h, n, target, deadline, rng, n_sweeps=50, recompute_interval=20):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    stall_count = 0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        indices = np.arange(n)
        rng.shuffle(indices)

        for p in range(n // 3):
            i = int(indices[3 * p])
            j = int(indices[3 * p + 1])
            k = int(indices[3 * p + 2])
            s = h_cur[i] + h_cur[j] + h_cur[k]
            if s < 2e-14 or s > 3.0 - 2e-14:
                continue

            Bi = compute_B(h_cur, i, k, n)
            Bj = compute_B(h_cur, j, k, n)
            A_mat = Bi - Bj
            C_mat = Bj  # B_j - B_k already computed as B_j in the call above
            # Actually: B_i - B_k = Bi (computed as compute_B(h,i,k,n))
            # B_j - B_k = Bj (computed as compute_B(h,j,k,n))
            # Change in corr = a*(B_i - B_k) + b*(B_j - B_k) where u_i=a, u_j=b, u_k=-(a+b)

            # Feasible region for (a, b):
            # -h_i <= a <= 1 - h_i
            # -h_j <= b <= 1 - h_j
            # h_k - 1 <= a + b <= h_k
            a_lo = -h_cur[i]
            a_hi = 1.0 - h_cur[i]
            b_lo = -h_cur[j]
            b_hi = 1.0 - h_cur[j]
            ab_lo = h_cur[k] - 1.0
            ab_hi = h_cur[k]

            # Grid search in 2D: 10x10 grid
            a_grid = np.linspace(a_lo, a_hi, 10)
            b_grid = np.linspace(b_lo, b_hi, 10)
            best_val = np.inf
            best_a, best_b = (a_lo + a_hi) * 0.5, (b_lo + b_hi) * 0.5

            for a_v in a_grid:
                b_clamp_lo = max(b_lo, ab_lo - a_v)
                b_clamp_hi = min(b_hi, ab_hi - a_v)
                if b_clamp_lo > b_clamp_hi:
                    continue
                for b_v in np.linspace(b_clamp_lo, b_clamp_hi, 10):
                    val = np.max(corr + a_v * A_mat + b_v * C_mat)
                    if val < best_val:
                        best_val = val
                        best_a, best_b = a_v, b_v

            if best_val < float(np.max(corr)) - 1e-14:
                h_cur[i] = np.clip(h_cur[i] + best_a, 0.0, 1.0)
                h_cur[j] = np.clip(h_cur[j] + best_b, 0.0, 1.0)
                h_cur[k] = np.clip(s - h_cur[i] - h_cur[j], 0.0, 1.0)
                h_cur[i] = np.clip(h_cur[i], 0.0, 1.0)
                h_cur[j] = np.clip(h_cur[j], 0.0, 1.0)
                h_cur[k] = np.clip(s - h_cur[i] - h_cur[j], 0.0, 1.0)
                # Recompute corr change properly
                corr = compute_corr(h_cur)

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 50 and sweep > 20:
            pert = rng.randn(n) * 0.01
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Initial paired coordinate descent (25% time)
    deadline_p1 = start_time + (budget_s - 15.0) * 0.20
    h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p1, rng, recompute_interval=30)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: SLSQP polish (10% time)
    if tleft() > 200:
        h_sl, sc_sl = slsqp_full(best_h, n, target, [200, 1000, 5000], 3000, 1e-17)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 3: Active set refinement (15% time)
    if tleft() > 300:
        active_set_refinement._deadline = start_time + (budget_s - 15.0) * 0.45
        h_as, sc_as = active_set_refinement(best_h, n, target, max_rounds=25)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    # Phase 4: Block-3 coordinate descent (15% time)
    if tleft() > 300:
        deadline_p4 = start_time + (budget_s - 15.0) * 0.55
        h_b3, sc_b3 = block3_coord_desc(best_h, n, target, deadline_p4, rng, n_sweeps=40)
        if sc_b3 < best_score:
            best_score = sc_b3
            best_h = h_b3

    # Phase 5: Structured perturbation restarts (25% time)
    if tleft() > 300:
        deadline_p5 = start_time + (budget_s - 15.0) * 0.78
        modes = ['sine', 'multi_sine', 'block', 'gaussian', 'sine', 'multi_sine', 'block', 'gaussian']
        for attempt in range(8):
            if tleft() < 100:
                break
            mode = modes[attempt % len(modes)]
            h_pert = structured_perturbation(best_h, n, rng, mode=mode)
            h_pert = project_feasible(h_pert, target)
            # Quick CD from perturbed start
            dl = min(deadline_p5, start_time + (budget_s - 15.0) * (0.78 - 0.02 * attempt))
            h_res, sc_res = paired_coord_desc(h_pert, n, target, dl, rng, recompute_interval=25)
            if sc_res < best_score:
                best_score = sc_res
                best_h = h_res
            # Quick SLSQP on the best perturbation result
            if sc_res < best_score + 0.001 and tleft() > 100:
                h_sl, sc_sl = slsqp_full(h_res, n, target, [500, 2000, 8000], 2000, 1e-17)
                if sc_sl < best_score:
                    best_score = sc_sl
                    best_h = h_sl

    # Phase 6: Active set refinement again (10% time)
    if tleft() > 150:
        active_set_refinement._deadline = start_time + (budget_s - 15.0) * 0.88
        h_as, sc_as = active_set_refinement(best_h, n, target, max_rounds=30)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as

    # Phase 7: High-precision SLSQP final polish (5% time)
    if tleft() > 60:
        h_final, sc_final = slsqp_full(best_h, n, target, [2000, 8000, 30000, 100000], 5000, 1e-19)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 8: Final paired CD (5% time)
    if tleft() > 30:
        h_final, sc_final = paired_coord_desc(best_h, n, target,
                                              start_time + budget_s - 12.0, rng,
                                              recompute_interval=15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final active set
    if tleft() > 15:
        active_set_refinement._deadline = start_time + budget_s - 8.0
        h_final, sc_final = active_set_refinement(best_h, n, target, max_rounds=10)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
