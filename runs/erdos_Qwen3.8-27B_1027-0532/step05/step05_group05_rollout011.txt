new_corr[k] = corr[k] - dx * h_j * (1-h[j+k]) * I(j ≤ n-1-k) - dx * h[j-k] * (1-h_j) * I(j ≥ k) + dx * x * (1-h[j+k]) * I(j ≤ n-1-k) + dx * h[j-k] * (1-x) * I(j ≥ k)

= corr[k] + dx * (x - h_j) * [(1-h[j+k]) * I(j ≤ n-1-k) - h[j-k] * I(j ≥ k)]

So new_corr[k] = corr[k] + a_k * (x - h_j) where a_k = dx * [(1-h[j+k]) * I(j ≤ n-1-k) - h[j-k] * I(j ≥ k)]

And we want to minimize max_k new_corr[k] = max_k (corr[k] + a_k * (x - h_j)) subject to 0 ≤ x ≤ 1 and the sum constraint.

If we ignore the sum constraint (handle it separately), this is:
minimize max_k (a_k * x + (corr[k] - a_k * h_j)) subject to 0 ≤ x ≤ 1

This is a 1D convex piecewise linear minimization. The minimum is at:
- x = 0 if all a_k ≥ 0 (function is increasing)
- x = 1 if all a_k ≤ 0 (function is decreasing)
- Otherwise, at the intersection point of two lines

The intersection of lines a_i * x + b_i and a_j * x + b_j is at x = (b_j - b_i) / (a_i - a_j).

We can find the optimal x by:
1. Compute all lines: c_k(x) = a_k * x + d_k where d_k = corr[k] - a_k * h_j
2. Find x* = argmin_{x∈[0,1]} max_k c_k(x)

This can be done by:
- Evaluating at x=0 and x=1
- Finding all intersection points in [0,1] and evaluating there
- Taking the best

For n=400, there are ~399 lines, so O(n²) intersection points to check. That's 160,000 evaluations per coordinate, and 400 coordinates, so 64M per sweep. That might be a bit slow for multiple sweeps.

def optimize_coordinate(h, j, n, dx, corr):

    for k in range(1, n):
        val = 0.0
        if j <= n - 1 - k:
            val += (1.0 - h[j+k])
        if j >= k:
            val -= h[j-k]
        a[k] = dx * val

        m2 = hi - (hi - lo) / 3
        if g(m1) < g(m2):
            hi = m2
        else:
            lo = m1
    x_opt = (lo + hi) / 2

def compute_corr(h):

def project_feasible(h, target_sum):

        else:

        if c_max < best_score:

            best_h = h_cur.copy()

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_alpha = 0.0
_g_target = 0.0
_g_pen = 0.0


def compute_corr(h):
    n = len(h)
    corr = np.zeros(n)
    for k in range(1, n):
        corr[k] = np.dot(h[:n-k], 1.0 - h[k:])
    return corr


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr[1:])) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _lse_core(h, n, alpha, target, pen):
    dx = 2.0 / n
    corr = np.zeros(n)
    for k in range(1, n):
        corr[k] = np.dot(h[:n-k], 1.0 - h[k:]) * dx
    active = corr[1:]
    c_max = float(np.max(active))
    shifted = alpha * (active - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    Z = float(np.sum(np.exp(shifted)))
    w = np.exp(shifted) / Z
    obj = c_max + float(np.log(Z)) / alpha
    if pen > 0:
        obj += pen * (np.sum(h) - target) ** 2
    grad = np.zeros(n)
    for k in range(1, n):
        wk = w[k - 1]
        if wk < 1e-15:
            continue
        m = n - k
        grad[:m] += wk * dx * (1.0 - h[k:k + m])
        grad[k:] -= wk * dx * h[:m]
    if pen > 0:
        grad += 2.0 * pen * (np.sum(h) - target)
    return obj, grad


def _lse_obj_wrapper(x):
    obj, grad = _lse_core(x, _g_n, _g_alpha, _g_target, _g_pen)
    return obj


def _lse_grad_wrapper(x):
    obj, grad = _lse_core(x, _g_n, _g_alpha, _g_target, _g_pen)
    return grad


def lbfgsb_optimize(h, n, target, alpha, pen, maxiter, maxfun, deadline):
    global _g_n, _g_alpha, _g_target, _g_pen
    if time.time() > deadline:
        return h
    _g_n = n
    _g_alpha = float(alpha)
    _g_target = float(target)
    _g_pen = float(pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj_wrapper, h, method='L-BFGS-B',
                       jac=_lse_grad_wrapper, bounds=bounds,
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-18, 'gtol': 1e-10, 'iprint': -1,
                                'maxls': 50})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h


def coordinate_descent(h, n, target, n_sweeps, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur) * dx
        for j in range(n):
            if time.time() > deadline:
                break
            a_k = np.zeros(n)
            for k in range(1, n):
                val = 0.0
                if j <= n - 1 - k:
                    val += (1.0 - h_cur[j + k])
                if j >= k:
                    val -= h_cur[j - k]
                a_k[k] = val * dx
            d_k = corr[1:] - a_k[1:] * h_cur[j]

            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                g1 = float(np.max(a_k[1:] * m1 + d_k))
                g2 = float(np.max(a_k[1:] * m2 + d_k))
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo + hi) / 2.0
            h_cur[j] = x_opt
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur) * dx
        c_max = float(np.max(corr))
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.004)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            m = n - k
            grad[:m] += wk * dx * (1.0 - h_cur[k:k + m])
            grad[k:] -= wk * dx * h_cur[:m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_max:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 4.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur) * dx
        k_star = int(np.argmax(corr))
        if k_star == 0:

        m = n - k_star
        grad = np.zeros(n)
        grad[:m] += dx * (1.0 - h_cur[k_star:k_star + m])
        grad[k_star:] -= dx * h_cur[:m]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:

            cur_score = sc_new
            eps = min(eps * 1.05, 0.02)
