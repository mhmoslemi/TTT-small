    grad = np.zeros(n)
    for k in range(n):

            jn = np.arange(k, n)
            grad[jn] -= wk * h[jn - k]
    grad *= dx
    return grad


def adaptive_wgd(h, n, target, n_steps, beta_init, beta_max, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init

    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        grad = weighted_grad(h_cur, n, weights, dx)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
            h_cur = h_new
            beta = min(beta * 1.03, beta_max)
        else:
            lr *= 0.92
            if lr < 1e-9:
                lr = lr_init * 0.5
                beta = min(beta * 1.8, beta_max)
    return best_h, best_score


def slsqp_polish(h, n, target, alphas, maxiter_per_alpha, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]

    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break

        def lse_obj(x, alpha=alpha, n=n, dx=dx):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        def lse_jac(x, alpha=alpha, n=n, dx=dx):
            h = np.clip(x, 0.0, 1.0)
            corr = compute_corr(h)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            w = np.exp(shifted)
            w /= w.sum()
            g = np.zeros(n)
            for k in range(n):
                wk = w[k]
                if wk < 1e-12:
                    continue
                jp = np.arange(0, n - k)
                g[jp] += wk * (1.0 - h[jp + k])
                if k > 0:
                    jn = np.arange(k, n)
                    g[jn] -= wk * h[jn - k]
            return g * dx

        try:
            res = minimize(lse_obj, h_cur, jac=lse_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def binary_quantize(h, n, target):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:int(target)]
    h_bin[idx] = 1.0
    return h_bin


def binary_local_search(h_bin, n, target, n_trials=5000, deadline=None, rng_seed=42):
    rng = np.random.RandomState(rng_seed)
    h_cur = h_bin.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    corr_cur = compute_corr(h_cur)
    over_cur = corr_cur[n - 1:n - 1 + n]
    cur_max = float(np.max(over_cur))

    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]

    for trial in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = int(ones[i_idx])
        j = int(zeros[j_idx])

        h_test = h_cur.copy()
        h_test[i] = 0.0
        h_test[j] = 1.0
        corr_test = compute_corr(h_test)
        over_test = corr_test[n - 1:n - 1 + n]
        new_max = float(np.max(over_test))
        if new_max < cur_max - 1e-12:
            h_cur = h_test
            cur_max = new_max
            corr_cur = corr_test
            over_cur = over_test
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
    return h_cur, cur_max * dx


def bump_parametrization_optimize(n, target, n_bumps, rng, deadline, n_inner=500):
    dx = 2.0 / n
    x = np.linspace(0.0, 2.0, n)
    best_h = None
    best_score = float('inf')

    for trial in range(3):
        centers = np.sort(rng.uniform(0.2, 1.8, n_bumps))
        widths = rng.uniform(0.05, 0.4, n_bumps)
        heights = np.ones(n_bumps)

        params = np.concatenate([centers, widths, heights])
        bounds = []
        for _ in range(n_bumps):
            bounds.append((0.01, 1.99))
        for _ in range(n_bumps):
            bounds.append((0.02, 0.8))
        for _ in range(n_bumps):
            bounds.append((0.1, 2.0))

        def obj(params):
            c = params[:n_bumps]
            w = params[n_bumps:2 * n_bumps]
            a = params[2 * n_bumps:]
            h = np.zeros(n)
            for bi in range(n_bumps):
                h += a[bi] * np.exp(-((x - c[bi]) / w[bi]) ** 2)
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            return compute_c5(h)

        def h_from_params(params):
            c = params[:n_bumps]
            w = params[n_bumps:2 * n_bumps]
            a = params[2 * n_bumps:]
            h = np.zeros(n)
            for bi in range(n_bumps):
                h += a[bi] * np.exp(-((x - c[bi]) / w[bi]) ** 2)
            h = np.clip(h, 0.0, 1.0)
            return project_feasible(h, target)

        try:
            res = minimize(obj, params, method='Nelder-Mead',
                           bounds=bounds,
                           options={'maxiter': n_inner, 'xatol': 1e-8, 'fatol': 1e-10})
            h_trial = h_from_params(res.x)
            sc = compute_c5(h_trial)
            if sc < best_score:
                best_score = sc
                best_h = h_trial
        except Exception:
            pass

    if best_h is None:
        return None, float('inf')
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_perturb(h, n, target, n_peaks, amp, rng):
    x = np.linspace(0.0, 1.0, n)
    pert = np.zeros(n)
    for _ in range(n_peaks):
        c = rng.uniform(0, 1)
        w = rng.uniform(0.05, 0.3)
        s = rng.choice([-1, 1])
        pert += s * amp * np.exp(-((x - c) / w) ** 2)
    h_new = h + pert
    return project_feasible(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_dx
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    _g_n = n
    _g_target = target
    _g_dx = 2.0 / n

    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 30.0 - elapsed()

    # Phase 1: Coordinate descent for quick improvement (0-60s)
    deadline1 = start_time + 60
    h_cd1, sc_cd1 = coordinate_descent_fast(best_h, n, target, max_cycles=300, deadline=deadline1)
    if sc_cd1 < best_score:
        best_score = sc_cd1
        best_h = h_cd1

    # Phase 2: Weighted gradient descent (60-180s)
    if tleft() > 100:
        deadline2 = start_time + 180
        h_wgd, sc_wgd = adaptive_wgd(best_h, n, target, n_steps=2000,
                                     beta_init=50, beta_max=5000,
                                     lr_init=0.002, deadline=deadline2)
        if sc_wgd < best_score:
            best_score = sc_wgd
            best_h = h_wgd
        # Follow with coordinate descent
        deadline2b = start_time + 220
        h_cd, sc_cd = coordinate_descent_fast(best_h, n, target, max_cycles=200, deadline=deadline2b)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: SLSQP with moderate alpha (220-450s)
    if tleft() > 200:
        deadline3 = start_time + 450
        h_sls, sc_sls = slsqp_polish(best_h, n, target,
                                     alphas=[200, 1000, 5000],
                                     maxiter_per_alpha=1500,
                                     deadline=deadline3)
        sc_sls = compute_c5(h_sls)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls
        # Coordinate descent after SLSQP
        deadline3b = start_time + 500
        h_cd, sc_cd = coordinate_descent_fast(best_h, n, target, max_cycles=300, deadline=deadline3b)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Bump parametrization for diversification (500-600s)
    if tleft() > 150:
        deadline4 = start_time + 600
        h_bump, sc_bump = bump_parametrization_optimize(n, target, n_bumps=3, rng=rng,
                                                        deadline=deadline4, n_inner=300)
        if h_bump is not None and sc_bump < best_score:
            best_score = sc_bump
            best_h = h_bump
        # Also try 4 bumps
        if tleft() > 100:
            h_bump4, sc_bump4 = bump_parametrization_optimize(n, target, n_bumps=4, rng=rng,
                                                              deadline=start_time + 650, n_inner=300)
            if h_bump4 is not None and sc_bump4 < best_score:
                best_score = sc_bump4
                best_h = h_bump4

    # Phase 5: Smooth perturbations + coordinate descent (650-800s)
    if tleft() > 150:
        deadline5 = start_time + 800
        for pert_trial in range(5):
            if time.time() > deadline5:
                break
            h_pert = smooth_perturb(best_h, n, target, n_peaks=3, amp=0.15, rng=rng)
            h_cd, sc_cd = coordinate_descent_fast(h_pert, n, target, max_cycles=150,
                                                  deadline=deadline5)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

    # Phase 6: Binary quantization + local search (800-1000s)
    if tleft() > 150:
        deadline6 = start_time + 1000
        h_bin = binary_quantize(best_h, n, target)
        h_bin_ls, sc_bin = binary_local_search(h_bin, n, target, n_trials=8000,
                                               deadline=deadline6, rng_seed=seed + 1)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin_ls.copy()
            # Interpolate back to continuous and refine
            h_interp = h_bin_ls.copy()
            deadline6b = start_time + 1050
            h_cd, sc_cd = coordinate_descent_fast(h_interp, n, target, max_cycles=200,
                                                  deadline=deadline6b)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd
        # Also try binary from a perturbed start
        if tleft() > 80:
            h_pert2 = smooth_perturb(best_h, n, target, n_peaks=5, amp=0.3, rng=rng)
            h_bin2 = binary_quantize(h_pert2, n, target)
            h_bin2_ls, sc_bin2 = binary_local_search(h_bin2, n, target, n_trials=5000,
                                                     deadline=start_time + 1100, rng_seed=seed + 2)
            if sc_bin2 < best_score:
                best_score = sc_bin2
                best_h = h_bin2_ls.copy()
                h_cd, sc_cd = coordinate_descent_fast(best_h, n, target, max_cycles=150,
                                                      deadline=start_time + 1130)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_h = h_cd

    # Phase 7: SLSQP with high alpha (1100-1250s)
    if tleft() > 100:
        deadline7 = start_time + 1250
        h_sls2, sc_sls2 = slsqp_polish(best_h, n, target,
                                       alphas=[10000, 50000, 200000],
                                       maxiter_per_alpha=2000,
                                       deadline=deadline7)
        sc_sls2 = compute_c5(h_sls2)
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2

    # Phase 8: Final coordinate descent + SLSQP (1250-1300s)
    if tleft() > 30:
        deadline8 = start_time + 1280
        h_cd_final, sc_final = coordinate_descent_fast(best_h, n, target, max_cycles=200,
                                                       deadline=deadline8)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_cd_final
        if tleft() > 15:
            h_sls_final, sc_slsf = slsqp_polish(best_h, n, target, alphas=[100000],
                                                maxiter_per_alpha=1000,
                                                deadline=start_time + 1295)
            sc_slsf = compute_c5(h_sls_final)
            if sc_slsf < best_score:
                best_score = sc_slsf
                best_h = h_sls_final

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, len(h_final)
