def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_lse(x):

    shifted = _g_alpha * (corr - c_max)

    return np.ones(_g_n)

    _g_target = target

def smooth_perturb(h, n, target, sigma, rng):

    phase = rng.uniform(0, 2*np.pi, 5)

        pert += a * np.sin(2*np.pi*f*x + p)
    pert *= sigma
    h_new = h + pert
    return project_feasible(h_new, target)

def block_shift_perturb(h, n, target, rng):
    """Shift a random block of h values."""
    # Choose a block
    block_len = rng.randint(n//20, n//5)
    start = rng.randint(0, n - block_len)

        h_new[start+shift:start+block_len+shift] = h[start:start+block_len]

    elif shift > 0:

        h_new[start+shift:n] = h[start:n-shift]
        h_new[start:start+shift] = 0.0
    h_new = project_feasible(h_new, target)
    return h_new

def bump_perturb(h, n, target, rng):

    best_h = h_cur.copy()

    dx = 2.0 / n

    lr = lr_init
    stuck = 0

            break

        max_over = float(np.max(over))
        mean_over = float(np.mean(over))

        spread = max_over - mean_over
        beta_eff = beta * (1.0 + 10.0 * spread)
        
        weights = np.exp(beta_eff * (over - max_over))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(n):

                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc

            v *= 0.5
            stuck = 0

            lr = min(lr * 1.05, 0.005)
        
        h_cur = h_new
        # Gradually increase beta
        beta = min(beta * beta_growth, 50000)
    
    return best_h, best_score

        grad[j_neg] -= wk * h[j_neg - k]

def wgd_with_momentum(h, n, target, deadline, beta_init=100, beta_growth=1.15,
                      n_steps=5000, lr_init=0.001, momentum=0.9):
    """Weighted gradient descent with momentum and adaptive beta."""
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    v = np.zeros(n)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init
    dx_val = 2.0 / n
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        over = overlaps_of(h_cur)
        max_over = float(np.max(over))
        
        # Compute weights
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        
        # Compute gradient
        grad = gradient_of_weighted_overlaps(h_cur, weights, dx_val)
        
        # Momentum update
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        # Adapt learning rate
        if sc < compute_c5(h_cur):
            h_cur = h_new
            lr = min(lr * 1.01, 0.005)
        else:
            lr = max(lr * 0.9, 1e-6)
            v *= 0.5
        
        # Adapt beta
        beta = min(beta * beta_growth, 100000.0)
    
    return best_h, best_score


def block_slsqp_descent(h, n, target, block_size, n_rounds, alpha, deadline=None):
    """Optimize blocks of size `block_size` using SLSQP, maintaining feasibility."""
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = (n + block_size - 1) // block_size
    
    def _block_obj(x):
        h_full = h_cur.copy()
        h_full[:len(x)] = np.clip(x, 0.0, 1.0)
        # For blocks not at the start, we need to handle differently
        # Actually, let's just optimize the whole thing with SLSQP
        return _obj_lse(h_full)
    
    # Instead of complicated block handling, do cycling SLSQP on windows
    for rd in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        for bi in range(n_blocks):
            if deadline is not None and time.time() > deadline:
                break
            s = bi * block_size
            e = min(s + block_size, n)
            
            # Optimize the sub-block using SLSQP on the full vector
            # with only the sub-block as free variables
            # This is complex, so instead we do a Nelder-Mead on the sub-block
            x0 = h_cur[s:e].copy()
            
            def sub_obj(x_sub):
                h_full = h_cur.copy()
                h_full[s:e] = np.clip(x_sub, 0.0, 1.0)
                h_full = project_feasible(h_full, target)
                return _obj_lse(h_full)
            
            try:
                res = minimize(sub_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
                h_new = h_cur.copy()
                h_new[s:e] = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
            except Exception:
                pass
    return h_cur, cur_score


def sa_optimize(h, n, target, deadline, t0=0.01, t_min=0.0001, n_restarts=10, rng=np.random.RandomState(0)):
    """Simulated annealing with informed perturbations."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx_val = 2.0 / n
    
    for restart in range(n_restarts):

            break

        T = t0 * (0.8 ** restart)
        h_cur = best_h.copy() if restart > 0 else h.copy()
        cur_score = compute_c5(h_cur)
        
        max_iters = 2000
        for it in range(max_iters):

                break

            over = overlaps_of(h_cur)
            max_over = float(np.max(over))

            n_active = max(3, n // 10)
            top_idx = np.argsort(over)[::-1][:n_active * 2]
            active = rng.choice(top_idx, size=n_active, replace=False)
            
            # Compute gradient for these active shifts
            grad = np.zeros(n)
            for k in active:
                k = int(k)
                j_pos = np.arange(0, n - k)
                grad[j_pos] += (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= h_cur[j_neg - k]
            grad *= dx_val / n_active
            
            # Add noise
            noise = rng.randn(n) * T * 0.1
            step = -grad * T * 10 + noise
            
            h_new = project_feasible(h_cur + step, target)
            new_score = compute_c5(h_new)

            if delta < 0 or rng.rand() < np.exp(-delta / T):
                h_cur = h_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h_cur.copy()
            
            T = max(T * 0.9995, t_min)

            max_over = float(np.max(over))
            n_active = max(3, n // 15)
            top_idx = np.argsort(over)[::-1][:n_active * 3]
            active = rng.choice(top_idx, size=n_active, replace=False)

            grad = np.zeros(n)
            for k in active:
                k = int(k)
                j_pos = np.arange(0, n - k)
                grad[j_pos] += (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= h_cur[j_neg - k]
            grad *= dx / n_active

            r = rng.rand()
            if r < 0.7:
                noise = T * 0.5 * rng.randn(n)
                step = -grad * T * 15 + noise
            else:
                step = T * rng.randn(n)

            h_new = project_feasible(h_cur + step, target)
            new_score = compute_c5(h_new)
            delta = new_score - cur_score

            if delta < 0:
                h_cur = h_new
                cur_score = new_score
                stuck = 0
                if new_score < best_score:
                    best_score = new_score
                    best_h = h_cur.copy()
            elif rng.rand() < np.exp(-delta / max(T, 1e-15)):
                h_cur = h_new
                cur_score = new_score
                stuck += 1
            else:
                stuck += 1

            T = max(T * 0.9997, t_min)

            if stuck > stuck_limit:
                h_cur = best_h.copy()
                cur_score = best_score
                T = max(T * 2.0, t_min * 10)
                stuck = 0

    return best_h, best_score


def equalize_top(h, n, target, deadline, n_iters=300, top_m=5, lr_init=0.004):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init

    for it in range(n_iters):
        if time.time() > deadline:
            break
        over = overlaps_of(h_cur)
        top_idx = np.argsort(over)[::-1][:top_m]
        margins = over[top_idx] - over[top_idx[0]]
        weights = np.exp(margins / 0.003)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx

        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.02, 0.015)
        else:
            h_cur = h_new
            lr *= 0.92
            if lr < 1e-8:
                lr = lr_init * 0.5
                h_cur = best_h.copy()

    return best_h, best_score


def block_nm_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = (n + block_size - 1) // block_size

    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = min(s + block_size, n)

            def _subobj(x_sub, s=s, e=e):
                h_full = h_cur.copy()
                h_full[s:e] = np.clip(x_sub, 0.0, 1.0)
                h_full = project_feasible(h_full, target)
                corr = compute_corr(h_full)
                c_max = float(np.max(corr))
                shifted = _g_alpha * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

            x0 = h_cur[s:e].copy()
            try:
                res = minimize(_subobj, x0, method='Nelder-Mead',
                               options={'maxiter': 400, 'xatol': 1e-10, 'fatol': 1e-12})
                h_new = h_cur.copy()
                h_new[s:e] = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
            except Exception:
                pass
    return h_cur, cur_score


def multi_start_explore(h_init, n, target, rng, deadline, n_starts=4):
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    time_per = max(5.0, (deadline - time.time()) / max(1, n_starts))
    x = np.linspace(0, 1, n)

    for s in range(n_starts):
        if time.time() > deadline:
            break
        t_dl = time.time() + time_per

        if s == 0:
            h_start = h_init.copy()
        elif s == 1:
            h_start = project_feasible(h_init + 0.03 * rng.randn(n), target)
        elif s == 2:
            n_peaks = rng.randint(3, 6)
            h_alt = np.zeros(n)
            for _ in range(n_peaks):
                c = rng.uniform(0.05, 0.95)
                w = rng.uniform(0.02, 0.06)
                h_alt += np.exp(-50 * (x - c) ** 2)
            h_alt = project_feasible(h_alt, target)
            mix = 0.3
            h_start = project_feasible((1 - mix) * h_init + mix * h_alt, target)
        else:
            h_start = project_feasible(h_init * (0.8 + 0.4 * rng.rand(n)), target)

        h_res, sc = sa_informed(h_start, n, target, t_dl, rng,
                                t0=0.003, t_min=0.00001, n_restarts=3, stuck_limit=30)
        if sc < best_score:
            best_score = sc
            best_h = h_res
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    # Phase 1: Initial SLSQP baseline
    if tleft() > 60:
        h1, sc1 = slsqp_lse(best_h, n, target, [100, 500, 2000], 3000, 1e-15,
                            deadline=start_time + 60)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Top-penalty SLSQP
    if tleft() > 60:
        cur_c5 = compute_c5(best_h)
        over = overlaps_of(best_h)
        top_val = float(np.max(over))
        h2, sc2 = slsqp_top_penalty(best_h, n, target, top_m=4,
                                    target_val=top_val * 0.99,
                                    maxiter=5000, ftol=1e-16,
                                    deadline=start_time + 120)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Main WGD with momentum
    if tleft() > 250:
        h3, sc3 = wgd_momentum(best_h, n, target, start_time + 400,
                               beta_init=100, beta_growth=1.06,
                               n_steps=15000, lr_init=0.002, momentum=0.83)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Equalize top overlaps
    if tleft() > 100:
        h4, sc4 = equalize_top(best_h, n, target, start_time + 500,
                               n_iters=500, top_m=5, lr_init=0.004)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: SLSQP with escalating alpha
    if tleft() > 100:
        h5, sc5 = slsqp_lse(best_h, n, target, [3000, 10000, 40000], 3000, 1e-16,
                            deadline=start_time + 600)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Multi-start SA exploration
    if tleft() > 200:
        h6, sc6 = multi_start_explore(best_h, n, target, rng, start_time + 800, n_starts=4)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Second WGD round with higher beta
    if tleft() > 150:
        h7, sc7 = wgd_momentum(best_h, n, target, start_time + 950,
                               beta_init=1000, beta_growth=1.08,
                               n_steps=8000, lr_init=0.001, momentum=0.9)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Block descent
    if tleft() > 80:
        h8, sc8 = block_nm_descent(best_h, n, target, 4, 3, 5000.0, start_time + 1030)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final SLSQP with very high alpha
    if tleft() > 50:
        h9, sc9 = slsqp_lse(best_h, n, target, [50000, 200000], 3000, 1e-17,
                            deadline=start_time + budget_s - 70)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Final micro WGD
    if tleft() > 25:
        h10, sc10 = wgd_momentum(best_h, n, target, start_time + budget_s - 30,
                                 beta_init=10000, beta_growth=1.03,
                                 n_steps=3000, lr_init=0.0003, momentum=0.95)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
