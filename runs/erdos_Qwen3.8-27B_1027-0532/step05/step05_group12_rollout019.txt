def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=30):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    
    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0/n)
    
    def cons_fun(h):
        return np.sum(h) - _global_target
    
    def cons_jac(h):
        return np.ones(_global_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def spectral_init(n, target, n_modes=5):
    """Create a spectral initialization"""
    h = np.full(n, 0.5)
    x = np.arange(n)
    for m in range(1, n_modes + 1):
        a = 0.3 / m
        phi = np.pi * m / (2 * n_modes)
        h += a * np.sin(2 * np.pi * m * x / n + phi)
    h = project_feasible(h, target)
    return h

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_gradient_weighted(h, weights, top_indices, dx, n_half):
    n = len(h)
    grad = np.zeros(n)
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += w * dx * (1.0 - h[k:])
                grad[k:] += -w * dx * h[:nk]
        else:
            m = -k
            nm = n - m
            if nm > 0:
                grad[m:] += w * dx * (1.0 - h[:nm])
                grad[:nm] += -w * dx * h[m:]
    return grad


def active_set_lp(h_current, n, target, top_m, max_iters, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_current.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1

    for it in range(max_iters):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        all_c = compute_all_corr(h)
        top_indices = np.argsort(all_c)[-top_m:]

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for idx_k in top_indices:
            k = idx_k - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                if k >= 0:
                    coeff[k:hi] = g[:hi - k] * dx
                else:
                    coeff[:hi] = g[-k:] * dx
                constraints.append(coeff @ h_var <= t_var)

        reg = 0.01
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = project_feasible(h_new, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def subgradient_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    n_half = n - 1

    alpha_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    top_m_schedule = [5, 10, 15, 20, 30, 50, 80, 120, 200, 300, 500, 800]

    momentum_v = np.zeros(n)
    beta = 0.85

        if time.time() - start_time > time_limit - deadline_offset:

        top_m = min(top_m_schedule[ai], 2 * n - 1)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=30):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    dx = 2.0 / n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    def cons_fun(h):
        return np.sum(h) - _global_target

    def cons_jac(h):
        return np.ones(_global_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def spectral_perturb(h_base, n, rng, n_modes=4):
    h = h_base.copy()
    x = np.arange(n, dtype=float)
    for m in range(1, n_modes + 1):
        a = rng.uniform(0.05, 0.2) / m
        phi = rng.uniform(0, 2 * np.pi)
        h += a * np.sin(2 * np.pi * m * x / n + phi)
    return project_feasible(h, n / 2.0)
