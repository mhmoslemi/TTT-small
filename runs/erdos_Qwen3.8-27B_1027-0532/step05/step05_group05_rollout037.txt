d(over[k])/d(h[j]):
- h[j] appears as h[i+k] when i=j-k (valid if 0<=j-k<=n-1-k, i.e., k<=j<=n-1): contributes (1-h[j-k])
- h[j] appears as h[i] when i=j (valid if 0<=j<=n-1-k, i.e., j<=n-1-k): contributes -h[j+k]

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_alpha = 0.0
_g_target = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr[n-1:])) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj(x):
    n = _g_n
    corr = compute_corr(x)
    over = corr[n-1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha) * (2.0 / n)


def _grad(x):
    n = _g_n
    dx = 2.0 / n
    corr = compute_corr(x)
    over = corr[n-1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    sig = np.where(weights > 1e-8)[0]
    for k in sig:
        k = int(k)
        w = weights[k] * dx
        if k == 0:
            grad[:] += w * (1.0 - 2.0 * x[:])
        else:
            grad[k:] += w * (1.0 - x[:n-k])
            grad[:n-k] -= w * x[k:]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_alpha, _g_target
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_obj, h_cur, method='SLSQP', jac=_grad,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_adaptive(h, n, target, n_steps, deadline, lr_init=0.005, momentum=0.85):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        c_max = float(np.max(over))
        alpha = max(50.0, 200.0 * (step / n_steps))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()
        grad = np.zeros(n)
        sig = np.where(weights > 1e-6)[0]
        for k in sig:
            k = int(k)
            w = weights[k] * dx
            if k == 0:
                grad[:] += w * (1.0 - 2.0 * h_cur[:])
            else:
                grad[k:] += w * (1.0 - h_cur[:n-k])
                grad[:n-k] -= w * h_cur[k:]
        v = momentum * v - lr * grad
        h_new = project_feasible(h_cur + v, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= 0.9
            v *= 0.5
            if lr < 1e-12:
                lr = lr_init * 0.1
    return best_h, best_score


def coordinate_descent(h, n, target, n_iters, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:10]
        active_positions = set()
        for k in top_idx:
            k = int(k)
            active_positions.update(range(max(0, k-5), min(n, n-k+5)))
        active_list = sorted(active_positions)
        if len(active_list) == 0:
            active_list = list(range(n))
        rng_local = np.random.RandomState(it)
        rng_local.shuffle(active_list)
        improved = False
        for j in active_list:
            if time.time() > deadline:
                break
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_try = h_cur.copy()
                h_try[j] = h_cur[j] + delta
                if h_try[j] < 0 or h_try[j] > 1:
                    continue
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < c_curr - 1e-14:
                    h_cur = h_try
                    c_curr = sc
                    improved = True
                    break
    return best_h, best_score


def fourier_perturb(h, n, target, rng, n_terms=15, amplitude=0.02):
    x = np.linspace(0, 2, n)
    coeffs_a = np.zeros(n_terms)
    coeffs_b = np.zeros(n_terms)
    for k in range(1, n_terms + 1):
        coeffs_a[k-1] = np.mean(h * np.cos(k * np.pi * x))
        coeffs_b[k-1] = np.mean(h * np.sin(k * np.pi * x))
    phase_shifts = (rng.rand(n_terms) - 0.5) * amplitude
    h_new = np.full(n, np.mean(h))
    for k in range(1, n_terms + 1):
        a = coeffs_a[k-1]
        b = coeffs_b[k-1] + phase_shifts[k-1]
        h_new += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    return project_feasible(h_new, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
