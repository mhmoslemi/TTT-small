        idx2 = np.arange(k, n)
        grad[idx2] -= dx * h[idx2 - k]

            break
        h_new = project_feasible(h_cur - eps * grad, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.008)
        else:
            eps *= 0.75
            if eps < 1e-10:
                break
    return h_cur, cur_score

def top_lags_descent(h, max_steps, eps_init, n_lags):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = eps_init
    for _ in range(max_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:2*n-1]
        top = np.argsort(over)[::-1][:n_lags]
        max_val = float(over[top[0]])
        diffs = over[top] - max_val
        wts = np.exp(diffs * 50.0)
        wts /= wts.sum()
        grad = np.zeros(n)
        for wk, k in zip(wts, top):
            grad += wk * compute_grad_lag(h_cur, int(k), n, dx)
        grad = project_grad(grad, n)
        if np.linalg.norm(grad) < 1e-15:
            break
        h_new = project_feasible(h_cur - eps * grad, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.006)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, cur_score

def coordinate_round(h, target, thresh):
    h_r = h.copy()
    h_r[h_r < thresh] = 0.0
    h_r[h_r > 1.0 - thresh] = 1.0
    return project_feasible(h_r, target)

def try_rounding(h, target):
    best = h.copy()
    best_score = compute_c5(h)
    for thresh in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
        h_r = coordinate_round(h, target, thresh)
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best = h_r.copy()
    return best, best_score

def polish_directional(h, target, n_dirs, step_sizes, n_iter):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    rng = np.random.RandomState(12345)
    for _ in range(n_iter):
        d = rng.randn(n)
        d = project_grad(d, n)
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-10:
            continue
        d = d / dnorm
        for ss in step_sizes:
            for sign in [1.0, -1.0]:
                h_try = project_feasible(h_cur + sign * ss * d, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-15:
                    h_cur = h_try
                    best_score = sc
        for ss in step_sizes:
            h_try = project_feasible(h_cur + ss * d, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-15:
                h_cur = h_try
                best_score = sc
    return h_cur, best_score

def polish_coordinate(h, target, n_sweeps):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        for i in range(n):
            best_local = h_cur[i]
            best_local_sc = best_score
            for candidate in [0.0, 1.0, h_cur[i] + 0.05, h_cur[i] - 0.05,
                              h_cur[i] + 0.01, h_cur[i] - 0.01, h_cur[i] + 0.001, h_cur[i] - 0.001]:
                candidate = float(np.clip(candidate, 0.0, 1.0))
                h_try = h_cur.copy()
                h_try[i] = candidate
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_local_sc - 1e-15:
                    best_local_sc = sc
                    best_local = candidate
                    best_local_h = h_try.copy()
            if best_local_sc < best_score - 1e-15:
                h_cur = best_local_h
                best_score = best_local_sc
    return h_cur, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    if tleft() > 500:
        h_ref, sc = slsqp_run(best_h, [30, 100, 300, 1000, 5000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 400:
        h_rd, sc_rd = try_rounding(best_h, target)
        if sc_rd < best_score:
            best_score = sc_rd
            best_h = h_rd
        h_ref, sc = slsqp_run(best_h, [500, 2000, 10000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 300:
        h_sg, sc_sg = subgrad_optimize(best_h, 400, 4, 0.004, 0.3, 0.01, 0.999)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sg2, sc_sg2 = subgrad_optimize(best_h, 300, 6, 0.003, 0.4, 0.005, 0.999)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2
        h_ref, sc = slsqp_run(best_h, [2000, 10000, 50000], 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 200:
        h_sz, sc_sz = worst_lag_descent(best_h, 300, 0.002)
        if sc_sz < best_score:
            best_score = sc_sz
            best_h = h_sz
        h_sz2, sc_sz2 = top_lags_descent(best_h, 200, 0.0015, 3)
        if sc_sz2 < best_score:
            best_score = sc_sz2
            best_h = h_sz2
        h_ref, sc = slsqp_run(best_h, [10000, 50000, 200000], 1500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 150:
        h_pol, sc_pol = polish_directional(best_h, target, 80, [0.003, 0.001, 0.0003, 0.0001], 30)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
        h_ref, sc = slsqp_run(best_h, [50000, 200000], 1000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 120:
        for attempt in range(5):
            if tleft() < 60:
                break
            n_cur = len(best_h)
            target_cur = n_cur / 2.0
            h_pert = best_h.copy()
            if attempt % 3 == 0:
                h_pert += rng.randn(n_cur) * 0.03
            elif attempt % 3 == 1:
                block = max(5, n_cur // 40)
                for b in range(n_cur // block):
                    h_pert[b*block:min((b+1)*block, n_cur)] += rng.uniform(-0.08, 0.08)
            else:
                flip_n = n_cur // 20
                fi = rng.choice(n_cur, flip_n, replace=False)
                h_pert[fi] = 1.0 - h_pert[fi]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc = slsqp_run(h_pert, [500, 2000, 10000], 1200, 1e-16)
            h_sz, sc2 = worst_lag_descent(h_ref, 100, 0.001)
            if sc2 < sc:
                h_ref, sc = h_sz, sc2
            h_ref, sc = slsqp_run(h_ref, [50000], 800, 1e-17)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_sg, sc_sg = subgrad_optimize(best_h, 300, 5, 0.002, 0.35, 0.003, 0.999)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sz, sc_sz = worst_lag_descent(best_h, 200, 0.001)
        if sc_sz < best_score:
            best_score = sc_sz
            best_h = h_sz

    if tleft() > 30:
        h_ref, sc = slsqp_run(best_h, [100000, 500000], 1000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        h_sz, sc_sz = worst_lag_descent(best_h, 100, 0.0005)
        if sc_sz < best_score:
            best_score = sc_sz
            best_h = h_sz
        h_pol, sc_pol = polish_directional(best_h, target, 40, [0.001, 0.0003, 0.0001], 20)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 15:
        h_final, sc_final = slsqp_run(best_h, [200000, 1000000], 500, 1e-17)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
