```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def build_full_lp(h, dx, n, target, reg, h_center):
    """Build LP with ALL overlap constraints."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h
    n_half = n - 1
    for k in range(-n_half, n_half + 1):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var
    if reg > 0 and h_center is not None:
        objective = objective + reg * cp.sum_squares(h_var - h_center)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    return prob, h_var, t_var


def alternating_lp_all_constraints(h_init, n, target, max_iter, time_limit,
                                     damping=0.95, reg_start=1e-4, reg_end=1e-10):
    """Alternating LP using all constraints."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 200:
            break

        progress = min(1.0, it / max(1, max(10, max_iter * 0.3)))
        reg = reg_start * (reg_end / reg_start) ** progress
        damping_cur = damping

        prob, h_var, t_var = build_full_lp(h, dx, n, target, reg, h)

        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping_cur) * h + damping_cur * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_subproblem(h, dx, n, target):
    g = 1.0 - h
    n_half = n - 1
    n_cons = 2 * n_half + 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((n_cons, n + 1))
    b_ub = np.zeros(n_cons)
    for row in range(n_cons):
        k = row - n_half
        if k >= 0:
            hi = n
            A_ub[row, k:hi] = g[:hi - k] * dx
        else:
            hi = n + k
            A_ub[row, :hi] = g[-k:] * dx
        A_ub[row, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status != 0:
        return None
    return res.x[:n]


def alternating_lp_fast(h_init, n, target, max_iter, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 250:
            break
        if it < 50:
            damping = 0.85 + 0.003 * it
        elif it < 150:
            damping = 1.0
        else:
            damping = 1.0
        h_lp = solve_lp_subproblem(h, dx, n, target)
        if h_lp is None:
            break
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
            h = h_new
            stuck = 0
        else:
            h = h_new
            stuck += 1
    return best_h, best_c5


def subgradient_refine(h_init, n, target, max_steps, time_limit, n_top=30, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        w = 0.8 ** np.arange(n_top)
        w = w / w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            wk = w[i]
            if k >= 0:
                idx1 = idxs[idxs + k < n]
                grad[idx1] += wk * (1.0 - y[idx1 + k])
                idx2 = idxs[idxs >= k]
                grad[idx2] += wk * (-y[np.clip(idx2 - k, 0, n - 1)])
            else:
                mm = -k
                idx1 = idxs[idxs >= mm]
                grad[idx1] += wk * (1.0 - y[idx1 - mm])
                idx2 = idxs[idxs + mm < n]
                grad[idx2] += wk * (-y[np.clip(idx2 + mm, 0, n - 1)])
        grad *= dx
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-16:
            break
        c5_y = float(np.max(overlaps))
        gap = c5_y - best_c5
        if gap > 0:
            lr = min(gap / (norm_grad ** 2 + 1e-20), 0.3)
            lr = max(lr, 0.005)
        else:
            lr = 0.01
        v = momentum * v + (1.0 - momentum) * grad
        h_try = h - lr * v
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()
            v *= 0.5
    return best_h, best_c5


def smooth_objective(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)
    pen = 100.0 * (np.sum(h) - target) ** 2
    return val + pen


def l_bfgs_polish(h_start, n, target, alphas, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_objective, h_cur, args=(alpha, n, target),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': ftol,
                                    'gtol': 1e-14, 'maxfun': maxiter * 2})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def slsqp_polish(h_start, n, target, alphas, maxiter=1000):
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alphas[-1] * (corr - c_max), -500.0, 0.0)
        return (c_max + np.log(np.sum(np.exp(shifted))) / alphas[-1]) * (2.0 / n)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def cvxpy_lp_refine(h_init, n, target, max_iter, time_limit, n_top=200,
                    damping=0.9, reg_start=1e-4, reg_end=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 200:
            break
        g = 1.0 - h
        progress = min(1.0, it / max(1, max_iter * 0.4))
        reg = reg_start * (reg_end / reg_start) ** progress
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            if k >= 0:
                lo, hi = k, n
                coeff = np.zeros(n)
                coeff[lo:hi] = g[:hi - lo] * dx
            else:
                hi = n + k
                coeff = np.zeros(n)
                coeff[:hi] = g[-k:] * dx
            constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

    if tleft() > 120:
        h_cvx, c5_cvx = cvxpy_lp_refine(best_h, n, target,
                                         max_iter=500, time_limit=min(100, tleft() * 0.4),
                                         n_top=250, damping=0.85,
                                         reg_start=1e-3, reg_end=1e-8)
        if c5_cvx < best_score:
            best_score = c5_cvx
            best_h = h_cvx

    # Phase 6: L-BFGS-B polish again
    if tleft() > 150:
        h_lbfgs2 = l_bfgs_polish(best_h, n, target,
                                 [50, 200, 500, 2000, 10000, 50000],
                                 maxiter=2000)
        sc = compute_c5(h_lbfgs2)
        if sc < best_score:
            best_score = sc
            best_h = h_lbfgs2

    # Phase 7: Subgradient with different parameters
    if tleft() > 100:
        h_sg2, c5_sg2 = subgradient_refine(best_h, n, target,
                                           max_steps=50000, time_limit=min(80, tleft() * 0.3),
                                           n_top=50, momentum=0.85)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    # Phase 8: Final alternating LP
    if tleft() > 80:
        h_lp3, c5_lp3 = alternating_lp_fast(best_h, n, target,
                                            max_iter=50000, time_limit=min(60, tleft() * 0.5))
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 9: SLSQP final polish
    if tleft() > 80:
        h_slsqp = slsqp_polish(best_h, n, target,
                               [200, 1000, 5000, 20000], maxiter=800)
        sc = compute_c5(h_slsqp)
        if sc < best_score:
            best_score = sc
            best_h = h_slsqp

    # Phase 10: Try slightly different resolution
    if tleft() > 120:
        for n_trial in [440, 360]:
            if tleft() < 80:
                break
            target_t = n_trial / 2.0
            h_t, sc_t = try_different_resolution(best_h, n_trial, target_t,
                                                 time_limit=min(100, tleft() * 0.5))
            if sc_t < best_score:
                best_score = sc_t
                best_h = h_t
                n = n_trial
                target = target_t
                break

    # Final: one more LP pass if time allows
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, c5_final = alternating_lp_fast(best_h, n_use, target_use,
                                                max_iter=20000, time_limit=min(30, tleft() * 0.6))
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final
        else:
            best_h = project_feasible(best_h, target_use)
            best_score = compute_c5(best_h)

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, len(h_final)
```
