def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_value_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])

                grad[j2] -= w * x[j2 + lag]

import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def time_left(start_time, budget_s, reserve=40.0):
    return budget_s - reserve - (time.time() - start_time)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_value_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def _lse_obj(x):
    return lse_value_grad(x, _OPT['alpha'], _OPT['target'], _OPT['mu'])[0]

def _lse_grad(x):
    return lse_value_grad(x, _OPT['alpha'], _OPT['target'], _OPT['mu'])[1]

def _eq_fun(x):
    return np.sum(x) - _OPT['target']

def _eq_jac(x):
    return np.ones(len(x))

def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    _OPT['alpha'] = alpha
    _OPT['target'] = target
    _OPT['mu'] = mu
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    _OPT['alpha'] = alpha
    _OPT['target'] = target
    _OPT['mu'] = 1e8
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgradient_average(h, target, n_iters=600, top_k=10, eta0=0.008):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    h_sum = np.zeros(n)
    for t in range(1, n_iters + 1):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for ki in top_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= top_k
        eta = eta0 / np.sqrt(t)
        h_new = h_cur - eta * grad
        h_new = project_feasible(h_new, target)
        h_cur = h_new
        h_sum += h_cur
        if t % 40 == 0:
            h_avg = h_sum / t
            h_avg = project_feasible(h_avg, target)
            sc = compute_c5(h_avg)
            if sc < best_score:
                best_score = sc
                best_h = h_avg.copy()
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score

def active_set_equalize(h, target, n_top=8, n_iters=400):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    lr = 0.015
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-13:
            break
        grad_var = np.zeros(n)
        grad_max = np.zeros(n)
        for ki in top_idx:
            diff = (corr[ki] - top_mean) / (top_std + 1e-15)
            w = 1.0 / n_top
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                g_here = (1.0 - h_cur[j1 - lag])
                grad_var[j1] += 2.0 * diff * g_here
                grad_max[j1] += w * g_here
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    g_here2 = h_cur[j2 + lag]
                    grad_var[j2] -= 2.0 * diff * g_here2
                    grad_max[j2] -= w * g_here2
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                g_here = (1.0 - h_cur[j1 + m])
                grad_var[j1] += 2.0 * diff * g_here
                grad_max[j1] += w * g_here
                if m > 0:
                    j2 = np.arange(m, n)
                    g_here2 = h_cur[j2 - m]
                    grad_var[j2] -= 2.0 * diff * g_here2
                    grad_max[j2] -= w * g_here2
        total_grad = grad_var + 0.3 * grad_max
        h_new = h_cur - lr * total_grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
            lr = min(lr * 1.03, 0.06)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return h_cur, best_score

def coordinate_descent(h, target, n_sweeps=3):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            h_saved = h_cur.copy()
            h_cur[j] = c
            h_c = project_feasible(h_cur, target)
            fc = compute_c5(h_c)
            h_cur[j] = d
            h_d = project_feasible(h_cur, target)
            fd = compute_c5(h_d)
            for _ in range(35):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    h_cur = h_saved.copy()
                    h_cur[j] = c
                    h_c = project_feasible(h_cur, target)
                    fc = compute_c5(h_c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    h_cur = h_saved.copy()
                    h_cur[j] = d
                    h_d = project_feasible(h_cur, target)
                    fd = compute_c5(h_d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_saved[j]) > 1e-10:
                h_saved[j] = best_v
                h_new = project_feasible(h_saved, target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    h_cur = h_new
                    improved = True
                else:
                    h_cur = h_saved
            else:
                h_cur = h_saved
        if not improved:
            break
    return h_cur, cur_score

def p_norm_value_grad(x, p, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    corr_pos = np.maximum(corr, 0.0)
    if p > 50:
        c_max = float(np.max(corr_pos))
        shifted = p * (corr_pos - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / p
        weights = np.exp(shifted)
    else:
        val = float(np.sum(corr_pos ** p)) ** (1.0 / p)
        weights = corr_pos ** (p - 1)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def pnorm_optimize(h, target, p, mu, maxiter):
    n = len(h)
    _OPT['alpha'] = p
    _OPT['target'] = target
    _OPT['mu'] = mu

    def _pobj(x):
        return p_norm_value_grad(x, p, target, mu)[0]

    def _pgrad(x):
        return p_norm_value_grad(x, p, target, mu)[1]

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pobj, h, jac=_pgrad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h
