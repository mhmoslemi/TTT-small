**Let me also think about a different angle: the Lagrangian dual.**

The primal problem is:
minimize max_k c_k(h) subject to 0 ≤ h ≤ 1, sum(h) = n/2

The Lagrangian dual is:
maximize min_h [max_k c_k(h) + λ*(sum(h) - n/2)]
subject to ...

This isn't directly useful. But a related formulation: introduce t and write:
minimize t
subject to c_k(h) ≤ t for all k
         0 ≤ h ≤ 1
         sum(h) = n/2

The Lagrangian (for the inequality constraints c_k(h) ≤ t) is:
L(h, t, μ) = t + Σ_k μ_k * (c_k(h) - t) = (1-Σμ)t + Σμ_k c_k(h)
subject to 0 ≤ h ≤ 1, sum(h) = n/2, μ ≥ 0, Σμ = 1

The dual function is:
g(μ) = min_{h: 0≤h≤1, sum(h)=n/2} [Σμ_k c_k(h)]
      (since we need 1-Σμ = 0 for the dual to be bounded below, so Σμ = 1)

So the dual is:
maximize g(μ) = min_h Σμ_k c_k(h)
subject to μ ≥ 0, Σμ = 1

And g(μ) is the solution of a convex optimization problem (since c_k(h) are... wait, are they convex in h? No, they're quadratic!)

So the inner minimization is non-convex. This dual isn't directly useful.

However, if we use the linearization at h₀:
g(μ) ≈ min_h [corr(h₀) + Σμ_k G_k · (h-h₀)]
      = corr(h₀) - h₀·(Σμ_k G_k) + min_{h: 0≤h≤1, sum(h)=n/2} h·(Σμ_k G_k)

The inner problem is a linear program! And the dual becomes:
maximize corr(h₀) - h₀·(Σμ_k G_k) + min_{h: 0≤h≤1, sum(h)=n/2} h·(Σμ_k G_k)
subject to μ ≥ 0, Σμ = 1

This is a non-convex problem in μ (because of the min over h), but we can solve it by:
1. For a given μ, solve the inner LP to get h*(μ)
2. Update μ (e.g., using the subgradient of the outer problem)

This is essentially the same as the SCA approach but from a dual perspective. The SCA approach is simpler to implement, so let me stick with it.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, dx):
    """Compute the full gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            G[m, k:] += (1.0 - h[:n-k])
            G[m, :n-k] -= h[k:]
        else:
            ell = -k
            G[m, ell:] += (1.0 - h[:n-ell])
            G[m, :n-ell] -= h[ell:]
    G *= dx
    return G

def sca_step(h, target, dx, all_lags=True):
    """One SCA iteration: compute LP and return proposed step."""
    n = len(h)
    corr = compute_corr(h)
    G = compute_grad_matrix(h, dx)
    
    if not all_lags:
        top_k = 30
        top_idx = np.argsort(corr)[::-1][:top_k]
        G_used = G[top_idx]
        corr_used = corr[top_idx]
    else:
        G_used = G
        corr_used = corr
    
    n_lags = G_used.shape[0]
    
    # LP: minimize t
    # s.t. G_used @ h - t <= G_used @ h_old - corr_used  (for each lag)
    #      sum(h) = target
    #      0 <= h <= 1
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G_used
    A_ub[:, n] = -1.0
    b_ub = G_used @ h - corr_used
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n].copy(), float(res.x[n])
    return None, None

def sca_line_search(h, target, dx, h_proposed, max_tries=20):
    """Line search: find best step size from h toward h_proposed."""
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    for i in range(max_tries):
        lam = 2.0 ** (-i)
        h_trial = h + lam * (h_proposed - h)
        h_trial = project_feasible(h_trial, target)
        c_trial = compute_c5(h_trial)
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
        else:
            break
    return best_h, best_c

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.01)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_projection(h, target):
    """Project h to nearest binary vector with sum = target."""
    n = len(h)
    # Sort by value, set top target to 1, rest to 0
    threshold = np.sort(h)[::-1][int(target)]
    h_bin = (h >= threshold).astype(float)
    if h_bin.sum() > target:
        # Need to flip some 1s to 0s
        ones = np.where(h_bin == 1)[0]
        excess = int(h_bin.sum() - target)
        # Flip the ones with smallest h values
        flip_order = ones[np.argsort(h[ones])]
        h_bin[flip_order[:excess]] = 0
    elif h_bin.sum() < target:
        zeros = np.where(h_bin == 0)[0]
        deficit = int(target - h_bin.sum())
        flip_order = zeros[np.argsort(-h[zeros])]
        h_bin[flip_order[:deficit]] = 1
    return h_bin
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[m, k:] += (1.0 - h[:n - k])
            G[m, :n - k] -= h[k:]
        else:
            ell = -k
            G[m, ell:] += (1.0 - h[:n - ell])
            G[m, :n - ell] -= h[ell:]
    G *= dx
    return G

def sca_lp_solve(h, target, dx, top_k=None):
    n = len(h)
    corr = compute_corr(h)
    G = compute_grad_matrix(h, dx)
    if top_k is not None and top_k < 2 * n - 1:
        top_idx = np.argsort(corr)[::-1][:top_k]
        G_use = G[top_idx]
        corr_use = corr[top_idx]
    else:
        G_use = G
        corr_use = corr
    n_lags = G_use.shape[0]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G_use
    A_ub[:, n] = -1.0
    b_ub = G_use @ h - corr_use
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None

def sca_line_search(h, target, dx, h_prop, max_tries=20):
    best_h = h.copy()
    best_c = compute_c5(h)
    for i in range(max_tries):
        lam = 2.0 ** (-i)
        h_trial = h + lam * (h_prop - h)
        h_trial = project_feasible(h_trial, target)
        c_trial = compute_c5(h_trial)
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
        else:
            break
    return best_h, best_c

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgs_obj(x, alpha, mu, target):
    return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

def lbfgs_obj_grad(x, alpha, mu, target):
    return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, args=(alpha, mu, target), method='L-BFGS-B',
                       jac=lbfgs_obj_grad, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_obj(x, alpha):
    return lse_value(x, alpha)

def slsqp_obj_grad(x, alpha):
    return lse_grad(x, alpha)

_slsqp_target = 0.0

def slsqp_cons_fun(x):
    return np.sum(x) - _slsqp_target

def slsqp_cons_jac(x):
    return np.ones(len(x))

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    global _slsqp_target
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    try:
        res = minimize(slsqp_obj, h, args=(alpha,), method='SLSQP',
                       jac=slsqp_obj_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.012)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_proj(h, target):
    n = len(h)
    sorted_h = np.sort(h)[::-1]
    if int(target) >= n:
        return np.ones(n)
    threshold = sorted_h[int(target) - 1]
    h_bin = (h >= threshold).astype(float)
    s = h_bin.sum()
    if s > target:
        ones = np.where(h_bin == 1)[0]
        excess = int(s - target)
        flip = ones[np.argsort(h[ones])]
        h_bin[flip[:excess]] = 0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        deficit = int(target - s)
        flip = zeros[np.argsort(-h[zeros])]
        h_bin[flip[:deficit]] = 1
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SCA with full LP (all lags)
    for it in range(60):
        if time.time() > deadline - 450:
            break
        h_prop, t_prop = sca_lp_solve(h, target, dx, top_k=None)
        if h_prop is None:
            break
        h_new, sc_new = sca_line_search(h, target, dx, h_prop, max_tries=15)
        if sc_new < best_score - 1e-15:
            h = h_new
            best_score = sc_new
            best_h = h_new.copy()
        else:
            h_prop2, _ = sca_lp_solve(h, target, dx, top_k=25)
            if h_prop2 is not None:
                h_new2, sc_new2 = sca_line_search(h, target, dx, h_prop2, max_tries=12)
                if sc_new2 < best_score - 1e-15:
                    h = h_new2
                    best_score = sc_new2
                    best_h = h_new2.copy()
                    continue
            # Try even smaller top_k
            h_prop3, _ = sca_lp_solve(h, target, dx, top_k=8)
            if h_prop3 is not None:
                h_new3, sc_new3 = sca_line_search(h, target, dx, h_prop3, max_tries=12)
                if sc_new3 < best_score - 1e-15:
                    h = h_new3
                    best_score = sc_new3
                    best_h = h_new3.copy()
                    continue
            break

    # Phase 2: SCA with top-k lags (faster convergence on active set)
    for top_k in [15, 10, 5]:
        for it in range(40):
            if time.time() > deadline - 380:
                break
            h_prop, _ = sca_lp_solve(h, target, dx, top_k=top_k)
            if h_prop is None:
                break
            h_new, sc_new = sca_line_search(h, target, dx, h_prop, max_tries=12)
            if sc_new < best_score - 1e-15:
                h = h_new
                best_score = sc_new
                best_h = h_new.copy()
            else:
                break

    # Phase 3: Multi-start SCA (perturb and re-optimize)
    for attempt in range(3):
        if time.time() > deadline - 320:
            break
        scales = [0.05, 0.08, 0.03]
        h_pert = project_feasible(best_h + rng.randn(n) * scales[attempt], target)
        for it in range(20):
            if time.time() > deadline - 300:
                break
            h_prop, _ = sca_lp_solve(h_pert, target, dx, top_k=15)
            if h_prop is None:
                break
            h_new, sc_new = sca_line_search(h_pert, target, dx, h_prop, max_tries=10)
            if sc_new < compute_c5(h_pert):
                h_pert = h_new
            else:
                break
        sc_pert = compute_c5(h_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_h = h_pert.copy()
            h = h_pert

    # Phase 4: L-BFGS-B soft-max continuation
    alphas_p4 = [30, 100, 500, 2000, 8000, 30000]
    for alpha in alphas_p4:
        if time.time() > deadline - 250:
            break
        for mu in [5.0, 50.0, 200.0]:
            if time.time() > deadline - 240:
                break
            h_new = lbfgsb_round(h, n, target, alpha, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new

    # Phase 5: SLSQP refinement
    if time.time() < deadline - 150:
        for alpha in [1000, 5000, 20000, 80000]:
            if time.time() > deadline - 100:
                break
            h_new = slsqp_round(h, n, target, alpha, 3000, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new

    # Phase 6: Subgradient with decreasing temperature
    if time.time() < deadline - 100:
        for temp in [0.01, 0.004, 0.001, 0.0005]:
            if time.time() > deadline - 80:
                break
            h_sg, sc_sg = subgradient_polish(h, n, target, 2000, temp, 0.003, 0.997, deadline - 80)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg.copy()
                h = h_sg

    # Phase 7: Worst-lag surgery
    if time.time() < deadline - 60:
        h_surg, sc_surg = worst_lag_surgery(h, n, target, 500, 0.003, deadline - 60)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg.copy()
            h = h_surg

    # Phase 8: SCA again after all polishing
    if time.time() < deadline - 40:
        for it in range(30):
            if time.time() > deadline - 30:
                break
            h_prop, _ = sca_lp_solve(best_h, target, dx, top_k=12)
            if h_prop is None:
                break
            h_new, sc_new = sca_line_search(best_h, target, dx, h_prop, max_tries=10)
            if sc_new < best_score:
                best_h = h_new.copy()
                best_score = sc_new
            else:
                break

    # Phase 9: Binary projection check
    if time.time() < deadline - 30:
        h_bin = binary_proj(best_h, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
            h_new = lbfgsb_round(best_h, n, target, 2000, 100.0, 200)
            sc_pol = compute_c5(h_new)
            if sc_pol < best_score:
                best_score = sc_pol
                best_h = h_new.copy()

    # Phase 10: Try higher resolution
    if time.time() < deadline - 180:
        n2 = int(n * 1.5)
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        # SCA at higher res
        for it in range(30):
            if time.time() > deadline - 60:
                break
            h2_prop, _ = sca_lp_solve(h2, target2, dx2, top_k=20)
            if h2_prop is None:
                break
            h2_new, sc2_new = sca_line_search(h2, target2, dx2, h2_prop, max_tries=10)
            if sc2_new < best_score2 - 1e-15:
                h2 = h2_new
                best_score2 = sc2_new
                best_h2 = h2_new.copy()
            else:
                break
        # L-BFGS-B at higher res
        if time.time() < deadline - 50:
            for alpha in [50, 500, 3000]:
                if time.time() > deadline - 40:
                    break
                h2_new = lbfgsb_round(h2, n2, target2, alpha, 50.0, 200)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new.copy()
                    h2 = h2_new
        # Subgradient at higher res
        if time.time() < deadline - 30:
            h2_sg, sc2_sg = subgradient_polish(h2, n2, target2, 800, 0.003, 0.003, 0.997, deadline - 30)
            if sc2_sg < best_score2:
                best_score2 = sc2_sg
                best_h2 = h2_sg.copy()
        # Surgery at higher res
        if time.time() < deadline - 20:
            h2_surg, sc2_surg = worst_lag_surgery(h2, n2, target2, 200, 0.002, deadline - 20)
            if sc2_surg < best_score2:
                best_score2 = sc2_surg
                best_h2 = h2_surg.copy()
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 11: Final SCA + surgery on best solution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    dx_cur = 2.0 / n_cur
    if time.time() < deadline - 15:
        for it in range(20):
            if time.time() > deadline - 8:
                break
            h_prop, _ = sca_lp_solve(best_h, target_cur, dx_cur, top_k=10)
            if h_prop is None:
                break
            h_new, sc_new = sca_line_search(best_h, target_cur, dx_cur, h_prop, max_tries=8)
            if sc_new < best_score:
                best_h = h_new.copy()
                best_score = sc_new
            else:
                break
    if time.time() < deadline - 5:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 200, 0.001, deadline - 3)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg.copy()

    # Final L-BFGS-B polish
    if time.time() < deadline - 2:
        h_new = lbfgsb_round(best_h, n_cur, target_cur, 50000, 500.0, 100)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
