        if stall > 15000:
            break
        
        ratio = min(1.0, it / max(max_iters - 1, 1))
        t = t_init * (t_end / t_init) ** ratio

        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        r = rng.random()
        
        if r < 0.4:
            # Targeted swap: focus on worst shift
            worst_k = int(np.argmax(corr)) - (n - 1)
            ip = [i for i in ones if (0 <= i + worst_k < n) or (0 <= i - worst_k < n)]
            jp = [j for j in zeros if (0 <= j + worst_k < n) or (0 <= j - worst_k < n)]
            if ip and jp:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
        elif r < 0.7:

            h_new[ones[rng.randint(len(ones))]] = 0
            h_new[zeros[rng.randint(len(zeros))]] = 1
        else:

            idx1 = rng.choice(len(ones), 2, replace=False)
            idx2 = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[idx1[0]]] = 0
            h_new[ones[idx1[1]]] = 0
            h_new[zeros[idx2[0]]] = 1
            h_new[zeros[idx2[1]]] = 1
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)

            corr = corr_new

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr:
                curr = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr

def irrational_rotation_init(n, alpha, beta=0.0):
    """Generate h[i] = 1 if frac(i*alpha + beta) < 0.5"""
    i = np.arange(n)
    vals = np.mod(i * alpha + beta, 1.0)
    h = (vals < 0.5).astype(float)
    return h

def legendre_init(n, rng):
    # Find smallest odd prime >= n
    p = max(n, 3)
    while True:
        if p % 2 == 0:
            p += 1
        is_p = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_p = False
                break
        if is_p:
            break
        p += 2
    
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
    
    seq = (leg >= 0).astype(float)
    if p >= n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
    
    return project_binary(h, n // 2)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
