def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_vector(h):

    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_coefficients(h, k_idx, n):
    half_n = n - 1
    k = k_idx - half_n
    lo = max(0, -k)
    hi = min(n, n - k)
    if hi <= lo:
        return None
    dx = 2.0 / n
    coeff = np.zeros(n)
    coeff[lo:hi] = (1.0 - h[lo - min(0, k):hi - min(0, k)]) * dx
    return coeff

            h_var <= 1.0

        for k_idx in sorted(active_k):

            lo = max(0, -k)
            hi = min(n, n - k)

                coeff[lo:hi] = (1.0 - h[max(0, lo - (-k)):hi - (-k) if k < 0 else lo + k]) * dx
                if k >= 0:
                    coeff[lo:hi] = (1.0 - h[lo+k:hi+k]) * dx
                else:
                    idx_src = lo + k
                    coeff[lo:hi] = (1.0 - h[idx_src:idx_src + (hi - lo)]) * dx
                constraints.append(coeff @ h_var <= t_var)

def smooth_obj_lse(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    global _global_n, _global_target, _global_alpha
    start = time.time()
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj_lse, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})

            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_improve(h, n, target, max_steps, time_limit, top_m=5):

        grad = np.zeros(n)
        if np.any(idx_valid1):
