def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_lag_gradient(h, k, n):
    """Compute gradient of the k-th lag correlation with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        # corr at lag 0 = sum(h[i] * (1-h[i])) = sum(h[i]) - sum(h[i]^2)
        # d/dh[m] = 1 - 2*h[m]
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        # positive lag k: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # d/dh[m] = (1-h[m+k]) if m < n-k, minus h[m-k] if m >= k
        j_pos = np.arange(0, n - k)
        grad[j_pos] += (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= h[j_neg - k]
        grad *= dx
    return grad

    h_cur = h.copy()
    v = np.zeros(n)  # momentum buffer
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    momentum = momentum_init
    lr = lr_init

            break

        c_max = float(np.max(corr))
        c_curr = c_max * (2.0 / n)
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Focus on top-k lags
        top_idx = np.argsort(corr)[::-1][:top_k]
        # Use softmax weighting
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k) - (n - 1)  # Convert to actual lag (can be negative)
            if k == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur) * (2.0/n)
            elif k > 0:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k]) * (2.0/n)
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k] * (2.0/n)

                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k_abs]) * (2.0/n)
                j_neg = np.arange(k_abs, n)
                grad[j_neg] -= wk * h_cur[j_neg - k_abs] * (2.0/n)

        h_new = h_cur + v

            momentum *= 0.99
            v *= 0.5  # Damp momentum on failure

    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-15:

        else:

            break

        h = np.clip(h, 0.0, 1.0)

    """Gradient of correlation at lag k (positive lag) w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h)
    else:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    return grad * dx

    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    mom = mom_init
    lr = lr_init

        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_curr = c_max * (2.0 / n)

            best_score = c_curr
            best_h = h_cur.copy()

        raw = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(raw / 0.001)
        weights /= weights.sum()

            lag = int(idx) - (n - 1)
            k = abs(lag)
            g = get_lag_gradient(h_cur, k, n)
            grad += wk * g
        
        v = mom * v - lr * grad
        h_new = project_feasible(h_cur + v, target)
        sc = compute_c5(h_new)

            h_cur = h_new
