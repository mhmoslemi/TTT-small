def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:

    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        if need <= len(ones):

def compute_swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    k_lo = max(-i, -(n - 1))
    k_hi = min(n - 1 - i, n - 1)
    if k_lo <= k_hi:
        delta[k_lo + n - 1:k_hi + n] -= (1.0 - h[i + k_lo:i + k_hi + 1])
    k_lo = max(i - (n - 1), -(n - 1))
    k_hi = min(i, n - 1)
    if k_lo <= k_hi:
        vals = h[i - k_lo:i - k_hi + 1][::-1]
        delta[k_lo + n - 1:k_hi + n] += vals
    k_lo = max(-j, -(n - 1))
    k_hi = min(n - 1 - j, n - 1)
    if k_lo <= k_hi:
        delta[k_lo + n - 1:k_hi + n] += (1.0 - h[j + k_lo:j + k_hi + 1])

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (corr - float(corr.max()))
    exp_sum = float(np.sum(np.exp(np.clip(shifted, -500, 0))))
    return float(corr.max()) + float(np.log(exp_sum)) / alpha

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            g_time = min(30, tleft() * 0.04)
            h_pg, c5_pg = fast_greedy(h_pert, 15000, g_time, start_time, rng, target_init)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

            if tleft() < 40:
                break
            sa_time = min(40, tleft() * 0.04)
            max_it6 = int(min(80000, sa_time * 2500))
            h_ps, c5_ps = fast_sa(h_pg, target_init, max_it6, 0.006, 0.00002, rng,
                                  sa_time, start_time, next_shifts=5)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 600, 1500, 5000, 15000]
        slsqp_time = min(250, tleft() - 30)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 8000, 1e-15,
                                 slsqp_time, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 50:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        sa_time = min(100, tleft() - 20)
        max_it7 = int(min(200000, sa_time * 2500))
        h_sf, c5_sf = fast_sa(h_bin_f, target_use, max_it7, 0.005, 0.00001, rng,
                              sa_time, start_time, next_shifts=10)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [10000, 30000, 80000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 5000, 1e-16,
                                   max(5.0, tleft() - 5), start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = project_feasible(best_h, target_final)
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
