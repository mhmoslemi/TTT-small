    if abs(diff) < 1e-14:

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:

            h_r[idx[0]] = 0
            final_sum -= 1

            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1

def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]

            break

        if r < 0.6:

            worst_5 = np.argsort(corr)[-5:]
            k = worst_5[rng.randint(len(worst_5))] - (n - 1)
            p_pool = np.array([p for p in ones if 0 <= p + k < n and h[p+k] == 0])
            q_pool = np.array([q for q in zeros if 0 <= q + k < n and h[q+k] == 1])

                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
            else:
                p = p_pool[rng.randint(len(p_pool))]
                q = q_pool[rng.randint(len(q_pool))]
        elif r < 0.85:

            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
        else:

                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
            else:

                q1_idx, q2_idx = rng.choice(len(zeros), 2, replace=False)
                p1, p2 = ones[p1_idx], ones[p2_idx]
                q1, q2 = zeros[q1_idx], zeros[q2_idx]
                # Apply double swap (handle as two sequential single swaps)
                # For simplicity, just do a random single swap
                p = p1
                q = q1
                
        if p == q:
            continue
        
        # Evaluate
        d = delta_c5_single_swap(h, p, q, corr)
        new_corr = corr + d
        new_c5 = float(new_corr.max()) * dx
        delta_c5 = new_c5 - current_c5

                best_c5 = current_c5
                best_h = h.copy()
                best_corr = corr.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_corr, best_c5
