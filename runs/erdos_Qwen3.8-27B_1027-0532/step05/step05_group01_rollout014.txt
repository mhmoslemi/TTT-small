def compute_c5(h):
    n = len(h)

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

            stall = 0
        else:
            stall += 1
        if stall > 200:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def topk_subgradient(h, n, target, time_limit, n_top=8, alpha_w=30.0, base_lr=0.04):
    start_t = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    stall = 0

    for step in range(50000):
        if time.time() - start_t > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break

        top_k_idx = np.argsort(corr)[-n_top:]
        w = np.exp(alpha_w * (corr[top_k_idx] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k_star if False else idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.125]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved and step > 500:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish at initial resolution
    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [200, 1000, 5000, 20000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at initial resolution
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.35)
        h_cd, c5_cd = fast_cd(best_h, n, target, time_limit=cd_time,
                              n_cand_start=50, n_cand_min=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP after CD
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [2000, 10000, 50000], 1200, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Another CD round
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = fast_cd(best_h, n, target, time_limit=cd_time,
                              n_cand_start=45, n_cand_min=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Interpolate to higher resolution and optimize
    for n_hires in [600, 800]:
        if tleft() < 250:
            break
        target_hires = n_hires / 2.0
        h_interp = interpolate_to(best_h, n_hires, target_hires)
        sc_interp = compute_c5(h_interp)

        h_hires = h_interp
        best_hires_score = sc_interp

        if tleft() > 150:
            h_pol, sc = slsqp_optimize(h_hires, n_hires, target_hires,
                                       [200, 1000, 5000], 1000, 1e-15)
            if sc < best_hires_score:
                best_hires_score = sc
                h_hires = h_pol

        if tleft() > 100:
            cd_t = min(150, tleft() * 0.4)
            h_cd, c5_cd = fast_cd(h_hires, n_hires, target_hires,
                                  time_limit=cd_t, n_cand_start=40, n_cand_min=15)
            if c5_cd < best_hires_score:
                best_hires_score = c5_cd
                h_hires = h_cd

        if tleft() > 60:
            h_pol, sc = slsqp_optimize(h_hires, n_hires, target_hires,
                                       [5000, 20000, 80000], 800, 1e-15)
            if sc < best_hires_score:
                best_hires_score = sc
                h_hires = h_pol

        if tleft() > 40:
            cd_t = min(40, tleft() * 0.5)
            h_cd, c5_cd = fast_cd(h_hires, n_hires, target_hires,
                                  time_limit=cd_t, n_cand_start=35, n_cand_min=12)
            if c5_cd < best_hires_score:
                best_hires_score = c5_cd
                h_hires = h_cd

        if best_hires_score < best_score:
            best_score = best_hires_score
            best_h = h_hires

    # Phase 6: Top-k subgradient at best resolution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = topk_subgradient(best_h, n_use, target_use,
                                       time_limit=min(50, tleft() * 0.4),
                                       n_top=8, alpha_w=30.0, base_lr=0.04)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Perturbation restarts
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.08, 0.15]:
        if tleft() < 60:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = project_feasible(best_h + rng.randn(n_use) * noise_scale, target_use)
        h_ref, sc = slsqp_optimize(h_pert, n_use, target_use,
                                   [1000, 5000, 20000], 800, 1e-15)
        sc = compute_c5(h_ref)
        if tleft() > 30 and sc < best_score:
            cd_t = min(25, tleft() * 0.4)
            h_cd, c5_cd = fast_cd(h_ref, n_use, target_use,
                                  time_limit=cd_t, n_cand_start=30, n_cand_min=10)
            if c5_cd < sc:
                sc = c5_cd
                h_ref = h_cd
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [10000, 50000, 200000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)

    return best_h, final_score, len(best_h)
