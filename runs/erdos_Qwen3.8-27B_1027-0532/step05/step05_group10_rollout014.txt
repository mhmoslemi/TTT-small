When h[i] changes by Δ:

So the change in f_k = dx * A_k * Δ.

This means for a given coordinate i, f_k as a function of a = h[i] is:
f_k(a) = dx * (A_k * a + B_k)

And the max over k is a convex piecewise-linear function of a.

Now, to find the optimal a for coordinate i:
1. Compute A_k and B_k for all k (this is O(n) work)
2. Find min_{a∈[0,1]} max_k dx*(A_k*a + B_k)

The optimal a is at a breakpoint or at a boundary. The breakpoints are at a = (B_k2 - B_k1)/(A_k1 - A_k2) for pairs (k1, k2) where A_k1 ≠ A_k2.

But finding ALL breakpoints is O(n²) which is too much. Instead:
- Evaluate at a = 0: max_k dx*B_k
- Evaluate at a = 1: max_k dx*(A_k + B_k)
- Find the k* that maximizes at a=0 and at a=1
- The optimal a is between 0 and 1 where the "active" k changes

A practical heuristic:
- Find k* = argmax_k (A_k*0 + B_k) = argmax_k B_k [the worst shift at a=0]
- Find k** = argmax_k (A_k + B_k) [the worst shift at a=1]
- Try a = 0, a = 1, and a = clip(-B_{k*}/A_{k*}, 0, 1) [makes f_{k*} = 0]
- Also try a = clip(-B_{k**}/A_{k**}, 0, 1)
- Pick the best

This is O(n) per coordinate, so O(n²) per sweep. For n=400, that's 160,000 operations per sweep. Very fast!

But wait, I need to recompute A_k and B_k for each i. Computing A_k is O(1) per k (just look up h[i+k] and h[i-k]). Computing B_k is O(1) if I can extract it from the current correlation: B_k = (corr[n-1+k] - A_k * h[i]) / dx.

1. Get current correlation (or maintain it incrementally)
2. For each k: A_k = 1 - h[i+k] - h[i-k] (with boundary), B_k = (corr[n-1+k] - A_k*h[i]) / dx
3. Find candidate a values
4. Evaluate max_k dx*(A_k*a + B_k) for each candidate
5. Pick best a
6. Update correlation: corr[n-1+k] += A_k * (new_a - old_a) * ...

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {0≤h≤1, sum(h)=target_sum} using bisection on Lagrange multiplier."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final adjustment
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h_proj < 1.0)[0]
        else:
            idx = np.where(h_proj > 0.0)[0]
        if len(idx) > 0:
            h_proj[idx] += residual / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max_and_grad(h, alpha):
    """Compute smooth max of f_k and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    
    c_max = np.max(f)
    shifted = alpha * (f - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    weights = exp_shifted / Z
    val = c_max + np.log(Z) / alpha
    
    # Gradient: grad[i] = dx * sum_k weights[k] * (1 - h[i+k] - h[i-k])
    # Use only significant weights for efficiency
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    if len(active) == 0:
        return val, np.zeros(n)
    
    half_n = n - 1
    ks = active - half_n  # actual shift values
    w = weights[active]
    
    grad = np.zeros(n)
    # grad[i] = dx * sum_k w_k * (1 - h[i+k]) [if valid] - dx * sum_k w_k * h[i-k] [if valid]
    
    # Vectorized: for each k, add w_k to grad[i] for the right i
    # Method: for each (k, w_k), grad[i] += w_k * (1-h[i+k]) for valid i
    #          grad[i] += -w_k * h[i-k] for valid i
    
    # This is O(n * n_active) which for n=400, n_active~50 is 20,000. Fine.
    for idx in range(len(ks)):
        k = ks[idx]
        wk = w[idx]
        # Source: grad[i] += wk * (1 - h[i+k]) for 0≤i<n, 0≤i+k<n
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            grad[lo:hi] += wk * (1.0 - h[lo+k:hi+k])
        # Target: grad[i] += -wk * h[i-k] for 0≤i<n, 0≤i-k<n
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2-k:hi2-k]
    
    grad *= dx
    return val, grad

def nesterov_optimize(h_init, target, alpha_start, alpha_end, max_iters, time_limit,
                       momentum=0.9, backtrack_factor=0.5, max_backtracks=20):
    """Nesterov-accelerated gradient descent on smooth max."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    y = h.copy()  # Nesterov's y
    v = np.zeros(n)  # momentum
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    L = 1.0 / dx  # initial step size estimate (rough Lipschitz constant)
    # Actually, the Lipschitz constant of the gradient of the smooth max
    # is bounded by alpha * (max eigenvalue of the Hessian of f_k)
    # For our problem, the Hessian of f_k has entries involving products of h values
    # A rough bound: L ≈ alpha * 2 / dx (very loose)
    # Let's use backtracking to handle this.
    
    step = 1.0
    n_iter = 0
    last_alpha = alpha_start
    
    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        
        # Update alpha (exponential schedule)
        progress = min(1.0, it / max(1, max_iters * 0.7))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Nesterov extrapolation
        # Standard: y = x + momentum * (x - x_old)
        # Simplified: y = h + momentum * v (where v is the "velocity")
        # Actually, let's use the standard Nesterov scheme:
        # y = h + (t-1)/(t) * (h - h_old) where t = (1+sqrt(1+4t_old²))/2
        # But this is complex. Let's use a simpler momentum scheme:
        # v = momentum * v + grad
        # h_new = h - step * v
        
        val, grad = smooth_max_and_grad(y, alpha)
        
        # Backtracking line search
        step = 1.0
        for bt in range(max_backtracks):
            h_try = y - step * grad
            h_try = project_feasible(h_try, target)
            # Check sufficient decrease (Armijo condition)
            # f(h_try) ≤ f(y) + grad·(h_try - y) + (1/(2*step)) * ||h_try - y||²
            # Simplified: just check that f(h_try) < f(y)
            val_try = compute_c5(h_try)  # Use exact C5 for the check
            if val_try < val + 1e-10:
                break
            step *= backtrack_factor
        
        v = momentum * v + grad
        h = project_feasible(y - step * v, target)
        
        # Update Nesterov y
        y = project_feasible(h + momentum * (h - y), target)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Bisection on lambda for sum constraint
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-9:
        if residual > 0:
            idx = np.where(h_proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h_proj > 1e-12)[0]
        if len(idx) > 0:
            h_proj[idx] += residual / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

        any_improved = False
        for i in range(n):

            # Compute A vector: A[m] = coefficient of h[i] in corr[m]

def smooth_max_grad(h, alpha):
    n = len(h)

    Z = np.sum(exp_s)
    weights = exp_s / Z
    val = c_max + np.log(Z) / alpha
    
    half_n = n - 1
    active = np.where(weights > 1e-12)[0]

        return val, np.zeros(n)

        wk = w[idx]

    return val, grad
```

Actually, the numpy docs say:

where the sum is over all n such that both a_{n+k} and v_n are defined.

c_k = sum_n h_{n+k} * (1-h_n) [where n+k < len(a) and n < len(v)]

With mode='full', k ranges from -(len(v)-1) to len(a)-1.
Since len(a) = len(v) = n:
k ranges from -(n-1) to n-1.

The output array c has length 2n-1, with index 0 corresponding to k = -(n-1) and index 2n-2 corresponding to k = n-1.

So output[m] corresponds to k = m - (n-1).

f_k = c_{k} * dx = output[k + (n-1)] * dx

For k = 0: f_0 = output[n-1] * dx = sum_n h_n * (1-h_n) * dx = sum_n h_n - h_n² * dx

This is the "self-overlap" - how much h disagrees with itself.

For k = 1: f_1 = output[n] * dx = sum_n h_{n+1} * (1-h_n) * dx

This measures: where h[x] is high, how much is h[x+1] low? (shifted by 1)

Wait, I need to be more careful. c_k = sum_n a_{n+k} * v_n = sum_n h_{n+k} * (1-h_n)

This is the correlation at lag k: it measures the "cross-correlation" between h shifted by k and (1-h).

If h_{n+k} is high and h_n is low (so 1-h_n is high), this contributes positively.
If h_{n+k} is low and h_n is high (so 1-h_n is low), this contributes 0.
If h_{n+k} is high and h_n is high (so 1-h_n is low), this contributes 0.
If h_{n+k} is low and h_n is low, this contributes 0.

So f_k measures: "at lag k, how much does h(x+k) tend to be high where h(x) is low?"

We want this to be small for ALL k. That is, we want h(x) and h(x+k) to be simultaneously high or simultaneously low for all k.

This is exactly the condition for h to be "self-similar" at all scales. The only function that satisfies this perfectly is a constant (h = c for all x), but that gives f_k = c(1-c)*2 which is maximized at c=1/2, giving f_k = 1/2.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-9:
        if residual > 0:
            idx = np.where(h_proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h_proj > 1e-12)[0]
        if len(idx) > 0:
            h_proj[idx] += residual / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def coord_descent(h_start, target, max_sweeps=300, time_limit=120.0, project_every=5):
    n = len(h_start)
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_start.copy(), target)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    start = time.time()

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        any_improved = False
        for i in range(n):
            old_a = h[i]
            ks = np.arange(2 * n - 1) - half_n

            A = np.zeros(2 * n - 1)
            mask1 = (i + ks >= 0) & (i + ks < n)
            mask2 = (i - ks >= 0) & (i - ks < n)
            A[mask1] += 1.0 - h[i + ks[mask1]]
            A[mask2] -= h[i - ks[mask2]]

            if np.max(np.abs(A)) < 1e-14:
                continue

            lo_a = 0.0
            hi_a = 1.0
            for _ in range(30):
                mid_a = (lo_a + hi_a) / 2.0
                vals = corr + A * (mid_a - old_a)
                m_star = int(np.argmax(vals))
                slope = A[m_star]
                if slope > 1e-12:
                    hi_a = mid_a
                elif slope < -1e-12:
                    lo_a = mid_a
                else:
                    break

            new_a = (lo_a + hi_a) / 2.0
            new_a = float(np.clip(new_a, 0.0, 1.0))

            if abs(new_a - old_a) > 1e-11:
                delta = new_a - old_a
                corr += A * delta
                h[i] = new_a
                any_improved = True

        if not any_improved:
            break

        if (sweep + 1) % project_every == 0:
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)

        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()

    return best_h, best_c5

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    c_max = np.max(f)
    shifted = np.clip(alpha * (f - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    weights = exp_s / Z
    val = c_max + np.log(Z) / alpha

    half_n = n - 1
    active = np.where(weights > 1e-12)[0]
    if len(active) == 0:
        return val, np.zeros(n)

    ks = active - half_n
    w = weights[active]
    grad = np.zeros(n)

    for idx in range(len(ks)):
        k = ks[idx]
        wk = w[idx]
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 - k:hi2 - k]

    grad *= dx
    return val, grad

def gradient_refine(h_start, target, alphas, max_iters_per_alpha, time_limit, momentum=0.9):
    n = len(h_start)
    dx = 2.0 / n
    best_h = project_feasible(h_start.copy(), target)
    best_c5 = compute_c5(best_h)
    start = time.time()

    for alpha in alphas:
        h = best_h.copy()
        v = np.zeros(n)
        L = 4.0 * alpha / n
        step = 0.5 / L
        no_improve = 0

        for it in range(max_iters_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_c5

            val, grad = smooth_max_and_grad(h, alpha)
            v = momentum * v + (1.0 - momentum) * grad
            h_try = project_feasible(h - step * v, target)
            c5_try = compute_c5(h_try)

            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                step = min(step * 1.02, 2.0 / L)
                no_improve = 0
            else:
                h = h_try
                no_improve += 1
                step *= 0.85
                if no_improve > 80:
                    break

    return best_h, best_c5

def multi_restart(h_base, target, noise_levels, n_restarts_each, time_budget):
    n = len(h_base)
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    start = time.time()

    for noise in noise_levels:
        for r in range(n_restarts_each):
            if time.time() - start > time_budget:
                return best_h, best_c5
            rng = np.random.RandomState(12345 + int(noise * 1000) + r * 7)
            h_pert = h_base + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
            h_cd, c5_cd = coord_descent(h_pert, target, max_sweeps=40,
                                        time_limit=min(20, (time_budget - (time.time() - start)) * 0.3))
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    return best_h, best_c5

def binary_push(h_start, target):
    n = len(h_start)
    h_bin = (h_start > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    c5_bin = compute_c5(h_bin)

    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)

    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    for blend in [0.75, 0.6, 0.4, 0.25]:
        h_blend = blend * best_h + (1.0 - blend) * h_start
        h_blend = project_feasible(h_blend, target)
        sc = compute_c5(h_blend)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_blend

    return best_h, best_c5
```
