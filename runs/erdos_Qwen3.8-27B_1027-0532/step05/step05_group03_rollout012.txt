            if time.time() - start > time_limit:
                break
            if worst_k == 0:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
            else:
                if k_for_forward := (worst_k < n):
                    m = n - worst_k
                    if m > 0:
                        bad_fwd = (h_cur[:m] == 1) & (h_cur[worst_k:] == 0)
                        bad_idx = np.where(bad_fwd)[0]
                        if len(bad_idx) > 0:
                            i = int(bad_idx[rng.randint(len(bad_idx))])
                        else:
                            i = int(ones[rng.randint(len(ones))])
                    else:
                        i = int(ones[rng.randint(len(ones))])
                else:
                    i = int(ones[rng.randint(len(ones))])
                jc = zeros[zeros != i]
                if len(jc) == 0:
                    break
                j = int(jc[rng.randint(len(jc))])

            if palindromic:
                ji = n - 1 - i
                jj = n - 1 - j
                if h_cur[ji] != 0 or h_cur[jj] != 1:
                    continue

            if palindromic:
                h_cur[i] = 0
                h_cur[n - 1 - i] = 0
                h_cur[j] = 1
                h_cur[n - 1 - j] = 1
            else:
                h_cur[i] = 0
                h_cur[j] = 1

            new_overlaps = compute_overlaps(h_cur)
            new_score = float(np.max(new_overlaps))
            if new_score < cur_score - 1e-15:
                cur_overlaps = new_overlaps
                cur_score = new_score
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h_cur.copy()
                    no_improve = 0
                break
            else:
                if palindromic:
                    h_cur[i] = 1
                    h_cur[n - 1 - i] = 1
                    h_cur[j] = 0
                    h_cur[n - 1 - j] = 0
                else:
                    h_cur[i] = 1
                    h_cur[j] = 0

        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            if no_improve % 100 == 0:
                n_pairs = rng.randint(1, 4)
                fs = rng.choice(ones, n_pairs, replace=False)
                fz = rng.choice(zeros, n_pairs, replace=False)
                if palindromic:
                    valid = True
                    for a, b in zip(fs, fz):
                        if h_cur[n - 1 - a] != 1 or h_cur[n - 1 - b] != 0:
                            valid = False
                            break
                    if valid:
                        for a in fs:
                            h_cur[a] = 0
                            h_cur[n - 1 - a] = 0
                        for b in fz:
                            h_cur[b] = 1
                            h_cur[n - 1 - b] = 1
                    else:
                        continue
                else:
                    h_cur[fs] = 0
                    h_cur[fz] = 1
                cur_overlaps = compute_overlaps(h_cur)
                cur_score = float(np.max(cur_overlaps))

    return best_h, best_score

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_overlaps(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol, time_limit=300):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    t0 = time.time()
    for a in alphas:
        if time.time() - t0 > time_limit:
            break
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp, time_limit=60):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    t0 = time.time()
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr * dx
        c_curr = float(np.max(over))
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k < n:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
            if k >= n:
                k2 = k - (n - 1)
                idx1 = np.arange(k2, n)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 - k2])
                idx2 = np.arange(0, n - k2)
                grad[idx2] -= wk * dx * h_cur[idx2 + k2]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.006)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def worst_lag_surgery(h, max_steps=200, initial_epsilon=0.001, time_limit=30):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    t0 = time.time()
    for step in range(max_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr * dx
        k_star = int(np.argmax(over))
        if k_star < n:
            upper = n - 1 - k_star
            grad = np.zeros(n)
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += dx * (1.0 - h_cur[idx1 + k_star])
            idx2 = np.arange(k_star, n)
            grad[idx2] -= dx * h_cur[idx2 - k_star]
        else:
            k2 = k_star - (n - 1)
            grad = np.zeros(n)
            idx1 = np.arange(k2, n)
            grad[idx1] += dx * (1.0 - h_cur[idx1 - k2])
            idx2 = np.arange(0, n - k2)
            grad[idx2] -= dx * h_cur[idx2 + k2]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.006)
        else:
            eps *= 0.8
            if eps < 1e-8:
                break
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: SLSQP refinement of continuous solution
    h_ref, sc = slsqp_optimize(best_h, [50, 200, 1000], 2000, 1e-15, time_limit=200)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 2: Binary rounding and local search
    if tleft() > 600:
        h_bin, sc_bin = smart_binary_round(best_h, target)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()

        h_bs, sc_bs = binary_local_search(h_bin, time_limit=150, top_k=6, seed=seed, palindromic=False)
        if sc_bs < best_score:
            best_score = sc_bs
            best_h = h_bs.copy()

        # Try palindromic variant
        h_pal = make_palindromic(h_bin, target)
        h_pal_ls, sc_pal = binary_local_search(h_pal, time_limit=120, top_k=6, seed=seed+1, palindromic=True)
        if sc_pal < best_score:
            best_score = sc_pal
            best_h = h_pal_ls.copy()

    # Phase 3: SLSQP refinement after binary improvements
    if tleft() > 400:
        h_ref, sc = slsqp_optimize(best_h, [500, 2000, 10000], 2000, 1e-16, time_limit=180)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Subgradient with momentum
    if tleft() > 300:
        h_sg, sc_sg = subgradient_momentum(best_h, 400, 5, 0.003, 0.997, 0.3, 0.004, time_limit=60)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sg2, sc_sg2 = subgradient_momentum(best_h, 300, 8, 0.002, 0.998, 0.5, 0.002, time_limit=50)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 5: Second binary round + search
    if tleft() > 250:
        h_bin2, sc_bin2 = smart_binary_round(best_h, target)
        if sc_bin2 < best_score:
            best_score = sc_bin2
            best_h = h_bin2.copy()
        h_bs2, sc_bs2 = binary_local_search(h_bin2, time_limit=120, top_k=8, seed=seed+100, palindromic=False)
        if sc_bs2 < best_score:
            best_score = sc_bs2
            best_h = h_bs2.copy()
        h_pal2 = make_palindromic(h_bin2, target)
        h_pal2_ls, sc_pal2 = binary_local_search(h_pal2, time_limit=100, top_k=8, seed=seed+200, palindromic=True)
        if sc_pal2 < best_score:
            best_score = sc_pal2
            best_h = h_pal2_ls.copy()

    # Phase 6: High-alpha SLSQP
    if tleft() > 150:
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000], 1500, 1e-16, time_limit=120)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Surgery on worst lags
    if tleft() > 80:
        h_surg, sc_surg = worst_lag_surgery(best_h, max_steps=200, initial_epsilon=0.0008, time_limit=30)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 8: Third binary round
    if tleft() > 120:
        h_bin3, sc_bin3 = smart_binary_round(best_h, target)
        if sc_bin3 < best_score:
            best_score = sc_bin3
            best_h = h_bin3.copy()
        h_bs3, sc_bs3 = binary_local_search(h_bin3, time_limit=100, top_k=10, seed=seed+500, palindromic=False)
        if sc_bs3 < best_score:
            best_score = sc_bs3
            best_h = h_bs3.copy()

    # Phase 9: Random restarts with perturbations
    if tleft() > 100:
        for attempt in range(5):
            if tleft() < 50:
                break
            n_cur = len(best_h)
            target_cur = n_cur / 2.0
            if attempt % 2 == 0:
                h_pert = best_h + rng.randn(n_cur) * 0.04
            else:
                block = max(10, n_cur // 20)
                h_pert = best_h.copy()
                for b in range(n_cur // block):
                    if rng.random() < 0.5:
                        h_pert[b * block:min((b + 1) * block, n_cur)] += rng.uniform(-0.12, 0.12)
            h_pert = project_feasible(h_pert, target_cur)
            h_bin_r, sc_br = smart_binary_round(h_pert, target_cur)
            h_bs_r, sc_bsr = binary_local_search(h_bin_r, time_limit=40, top_k=6, seed=seed + 1000 + attempt, palindromic=False)
            if sc_bsr < best_score:
                best_score = sc_bsr
                best_h = h_bs_r.copy()

    # Phase 10: Final SLSQP
    if tleft() > 40:
        h_ref, sc = slsqp_optimize(best_h, [10000, 50000], 1000, 1e-17, time_limit=30)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final binary polish
    if tleft() > 20:
        h_bin_f, sc_bf = smart_binary_round(best_h, target)
        h_bs_f, sc_bsf = binary_local_search(h_bin_f, time_limit=10, top_k=6, seed=seed+9999, palindromic=False)
        if sc_bsf < best_score:
            best_score = sc_bsf
            best_h = h_bs_f.copy()

    # Phase 12: Final surgery
    if tleft() > 10:
        h_surg_f, sc_sf = worst_lag_surgery(best_h, max_steps=100, initial_epsilon=0.0005, time_limit=5)
        if sc_sf < best_score:
            best_score = sc_sf
            best_h = h_surg_f.copy()

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
