def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):

    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]

def subgradient_refine(h, target_sum, max_iters, time_limit=None, start_time=None, 
                       lr_init=0.02, top_k=5):

    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        # Find top K shifts
        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = top_indices - (n - 1)
        top_corr_vals = corr[top_indices]
        
        # Compute weighted subgradient
        weights = top_corr_vals / top_corr_vals.sum()
        subgrad = np.zeros(n)
        
        for wi, k in zip(weights, top_shifts):
            for i in range(n):
                if 0 <= i + k < n:
                    subgrad[i] += wi * (1.0 - h[i + k])
                if 0 <= i - k < n:
                    subgrad[i] += wi * (-h[i - k])

            best_c5 = c5_val
            best_h = h_new.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:

        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]

        if len(ones) == 0 or len(zeros) == 0:

        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]

def subgradient_refine(h, target_sum, max_iters, time_limit=None, start_time=None,
                       lr_init=0.015, top_k=8, restart_every=500):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        if it > 0 and it % restart_every == 0:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')

        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = (top_indices - (n - 1)).astype(int)
        top_corr_vals = corr[top_indices].astype(float)
        weights = top_corr_vals / max(top_corr_vals.sum(), 1e-10)

        subgrad = np.zeros(n)
        for wi, k in zip(weights, top_shifts):
            i_lo = max(0, -k)
            i_hi = min(n, n - k)
            for i in range(i_lo, i_hi):
                subgrad[i] += wi * (1.0 - h[i + k])
            i_lo2 = max(0, k)
            i_hi2 = min(n, n + k)
            for i in range(i_lo2, i_hi2):
                subgrad[i] += wi * (-h[i - k])

        lr = lr_init / (1.0 + 0.0005 * it) ** 0.7
        h_new = h - lr * subgrad
        h_new = project_feasible(h_new, target_sum)

        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_val = float(corr.max()) * dx

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()

        h = h_new

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _global_target

    return np.ones(_global_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:

        try:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def legendre_init(n, rng):
    p = n if n % 2 == 1 else n + 1
    while True:
        is_prime = True
        if p < 2:
            is_prime = False
        elif p == 2:
            is_prime = True
        else:
            if p % 2 == 0:
                is_prime = False
            else:
                for d in range(3, int(np.sqrt(p)) + 1, 2):
                    if p % d == 0:
                        is_prime = False
                        break
        if is_prime:
            break
        p += 2

    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0

    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        remaining = n - p
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)

        h_leg = legendre_init(n_best, rng)
        allowed = min(40, tleft() * 0.08)
        h_g3, c5_g3 = greedy_optimize(h_leg, max_iters=20000, time_limit=allowed,
                                      start_time=start_time, scan_width=20)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()
        if tleft() > 60:
            allowed2 = min(60, tleft() * 0.10)
            max_it3 = int(min(150000, allowed2 * 1200))
            h_sa5, c5_sa5 = sa_blocks(h_g3, n_best // 2, max_it3, 0.010, 0.00005, rng,
                                      time_limit=allowed2, start_time=start_time)
            if c5_sa5 < best_score:
                best_score = c5_sa5
                best_h = h_sa5.copy()

    # Phase 7: ILS - perturbation + local search
    if tleft() > 80:
        n_best = len(best_h)
        target_best = n_best // 2
        for pert_iter in range(4):
            if tleft() < 40:
                break
            n_pert = 2 + pert_iter * 2
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(25, tleft() * 0.05)
            h_pg, c5_pg = greedy_optimize(h_pert, max_iters=10000, time_limit=allowed,
                                          start_time=start_time, scan_width=20)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

            if tleft() < 35:
                break
            allowed2 = min(30, tleft() * 0.05)
            max_it4 = int(min(80000, allowed2 * 1500))
            h_ps, c5_ps = sa_blocks(h_pg, target_best, max_it4, 0.008, 0.00003, rng,
                                    time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 8: SLSQP continuous refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 500, 1500, 4000, 10000, 25000, 50000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 8000, 1e-15,
                                 time_limit=min(tleft() - 20, 250), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Binary SA on refined solution
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(80, tleft() - 15)
        max_it5 = int(min(250000, allowed * 1500))
        h_sf, c5_sf = sa_blocks(h_bin_f, target_use, max_it5, 0.006, 0.00002,
                                rng, time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 10: Final SLSQP polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [10000, 30000, 80000, 200000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 5000, 1e-16,
                                   time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = project_feasible(best_h, target_final)
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
