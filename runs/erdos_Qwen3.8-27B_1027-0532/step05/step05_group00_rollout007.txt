def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:

            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

def interpolate_to(h_src, n_dst, target_dst):

        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def legendre_init(n, rng):
    p = 1
    while p < n:
        p += 2
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime and p >= n:
            break
    if p < n:
        p = n + 1 if (n + 1) % 2 == 1 else n + 2
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2
    
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
    
    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        remaining = n - p
        h[p:] = 0.5
    
    target = n // 2
    h = project_binary(h, target)
    return h

                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def legendre_init(n, rng):
    p = 3
    while p < n:
        p += 2
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime and p >= n:
            break
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2

    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0

    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)
    return h


def random_structured_init(n, rng):
    h = np.zeros(n)
    period = rng.randint(max(4, n // 50), max(5, n // 20) + 1)
    duty = rng.uniform(0.3, 0.7)
    block = np.zeros(period)
    k_ones = max(1, int(period * duty))
    block[:k_ones] = 1.0
    h = np.tile(block, n // period + 1)[:n]
    for _ in range(n // 5):
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
    target = n // 2
    return project_binary(h, target)
