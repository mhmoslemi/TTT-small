def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def legendre_init(n, rng):
    p = n + 1
    if p % 2 == 0:
        p += 1
    while True:
        is_prime = True
        if p < 2:
            is_prime = False
        elif p < 3:
            is_prime = True
        else:
            for d in range(3, int(np.sqrt(p)) + 1, 2):
                if p % d == 0:
                    is_prime = False
                    break
        if is_prime:
            break
        p += 2

    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0

    seq = (leg >= 0).astype(float)
    if p >= n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    # Phase 1: Fast local search from initial
    allowed = min(80, tleft() * 0.12)
    h_g, c5_g = fast_local_search(best_h, max_iters=80000, time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA from local search result
    if tleft() > 100:
        allowed = min(250, tleft() * 0.22)
        max_it = max(10000, int(allowed * 1800))
        h_sa, c5_sa = sa_search(best_h, target_init, max_it, 0.01, 0.00002, rng,
                                time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: ILS
    if tleft() > 200:
        allowed = min(350, tleft() * 0.30)
        h_ils, c5_ils = ils_loop(best_h, best_score, target_init, rng,
                                  start_time + allowed, start_time, n_rounds=10)
        if c5_ils < best_score:
            best_score = c5_ils
            best_h = h_ils.copy()

    # Phase 4: Try different n values
    n_candidates = []
    for dn in [2, -2, 4, -4, 6, -6, 8]:
        nc = n_init + dn
        if nc >= 200 and nc % 2 == 0 and nc != n_init and nc not in n_candidates:
            n_candidates.append(nc)

    for n_try in n_candidates[:4]:
        if tleft() < 80:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(40, tleft() * 0.12)
        h_g2, c5_g2 = fast_local_search(h_bin, max_iters=30000, time_limit=allowed, start_time=start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 50:
            continue
        allowed2 = min(50, tleft() * 0.10)
        max_it2 = max(5000, int(allowed2 * 1500))
        h_sa3, c5_sa3 = sa_search(h_g2, target_try, max_it2, 0.008, 0.00005, rng,
                                   time_limit=allowed2, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 5: Legendre start + optimization
    if tleft() > 150:
        h_leg = legendre_init(n_init, rng)
        allowed = min(50, tleft() * 0.12)
        h_g4, c5_g4 = fast_local_search(h_leg, max_iters=40000, time_limit=allowed, start_time=start_time)
        if c5_g4 < best_score:
            best_score = c5_g4
            best_h = h_g4.copy()
        if tleft() > 80:
            allowed2 = min(80, tleft() * 0.15)
            max_it3 = max(10000, int(allowed2 * 1500))
            h_sa5, c5_sa5 = sa_search(h_g4, target_init, max_it3, 0.01, 0.00003, rng,
                                       time_limit=allowed2, start_time=start_time)
            if c5_sa5 < best_score:
                best_score = c5_sa5
                best_h = h_sa5.copy()

    # Phase 6: Second SA pass with different parameters on current best
    if tleft() > 100:
        allowed = min(150, tleft() * 0.20)
        max_it4 = max(20000, int(allowed * 2000))
        h_sa6, c5_sa6 = sa_search(best_h, int(len(best_h) / 2), max_it4, 0.005, 0.00001, rng,
                                   time_limit=allowed, start_time=start_time)
        if c5_sa6 < best_score:
            best_score = c5_sa6
            best_h = h_sa6.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 500, 1500, 5000, 12000, 30000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-15,
                                  time_limit=min(tleft() - 20, 250), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final local search on binary projection of refined
    if tleft() > 40:
        n_use = len(best_h)
        target_int = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_int)
        allowed = min(50, tleft() - 10)
        h_final_ls, sc_ls = fast_local_search(h_bin_f, max_iters=30000, time_limit=allowed,
                                              start_time=start_time)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_final_ls

    # Phase 9: Final SLSQP polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [8000, 20000, 50000, 100000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 2000, 1e-16,
                                    time_limit=max(5.0, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
