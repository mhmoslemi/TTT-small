            else:
                worst = np.argsort(corr)[-5:]
                k = int(worst[rng.randint(len(worst))]) - (n - 1)
                i_cand, j_cand = get_candidates(h, k, n)
                if len(i_cand) >= 2 and len(j_cand) >= 2:
                    i1, i2 = rng.choice(len(i_cand), 2, replace=False)
                    j1, j2 = rng.choice(len(j_cand), 2, replace=False)
                    h_new[i_cand[i1]] = 0
                    h_new[i_cand[i2]] = 0
                    h_new[j_cand[j1]] = 1
                    h_new[j_cand[j2]] = 1
                    moved = True

        if not moved:
            continue

        corr_new = get_corr(h_new)
        c5_new = float(corr_new.max())
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted)) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, float(n) / 2.0)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def legendre_init(n, rng):
    p = max(n, 7)
    while True:
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2

    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0

    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy
    allowed = min(30, max(5, tleft() * 0.05))
    h_g, c5_g = greedy_search(best_h, 100000, allowed, start_time, topM=5, scan_width=50)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g

    # Phase 2: SA round 1 (higher temperature)
    if tleft() > 100:
        allowed = min(200, tleft() * 0.20)
        max_it = int(min(500000, allowed * 3000))
        h_sa, c5_sa = sa_search(best_h, target_init, max_it, 0.015, 0.0001, rng,
                                allowed, start_time, use_2swap=True)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 3: Greedy after SA
    if tleft() > 80:
        allowed = min(40, tleft() * 0.08)
        h_g2, c5_g2 = greedy_search(best_h, 100000, allowed, start_time, topM=8, scan_width=60)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2

    # Phase 4: SA round 2 (lower temperature, more targeted)
    if tleft() > 150:
        allowed = min(250, tleft() * 0.22)
        max_it = int(min(500000, allowed * 3000))
        h_sa2, c5_sa2 = sa_search(best_h, target_init, max_it, 0.008, 0.00003, rng,
                                  allowed, start_time, use_2swap=True)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2

    # Phase 5: ILS restarts
    if tleft() > 150:
        for pert_iter in range(8):
            if tleft() < 60:
                break
            n_pert = 3 + pert_iter * 2
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(20, tleft() * 0.04)
            h_pg, c5_pg = greedy_search(h_pert, 50000, allowed, start_time, topM=5, scan_width=40)
            if tleft() > 40:
                allowed2 = min(30, tleft() * 0.06)
                max_it2 = int(min(100000, allowed2 * 3000))
                h_ps, c5_ps = sa_search(h_pg, target_init, max_it2, 0.008, 0.00005, rng,
                                        allowed2, start_time, use_2swap=True)
                if c5_ps < best_score:
                    best_score = c5_ps
                    best_h = h_ps
            elif c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg

    # Phase 6: Try different n values
    if tleft() > 120:
        n_options = [n_init + 4, n_init - 4, n_init + 8, n_init - 8]
        n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n_init]
        for n_try in n_options:
            if tleft() < 50:
                break
            target_try = int(n_try / 2)
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin = project_binary(h_interp, target_try)

            allowed = min(20, tleft() * 0.05)
            h_g3, c5_g3 = greedy_search(h_bin, 50000, allowed, start_time, topM=5, scan_width=40)
            if c5_g3 < best_score:
                best_score = c5_g3
                best_h = h_g3

            if tleft() > 30:
                allowed2 = min(30, tleft() * 0.06)
                max_it3 = int(min(80000, allowed2 * 3000))
                h_sa3, c5_sa3 = sa_search(h_g3, target_try, max_it3, 0.01, 0.0001, rng,
                                          allowed2, start_time, use_2swap=True)
                if c5_sa3 < best_score:
                    best_score = c5_sa3
                    best_h = h_sa3

    # Phase 7: Legendre initialization
    if tleft() > 100:
        h_leg = legendre_init(n_init, rng)
        allowed = min(30, tleft() * 0.06)
        h_g4, c5_g4 = greedy_search(h_leg, 50000, allowed, start_time, topM=5, scan_width=50)
        if tleft() > 60:
            allowed2 = min(60, tleft() * 0.12)
            max_it4 = int(min(200000, allowed2 * 3000))
            h_sa4, c5_sa4 = sa_search(h_g4, target_init, max_it4, 0.012, 0.00005, rng,
                                      allowed2, start_time, use_2swap=True)
            if c5_sa4 < best_score:
                best_score = c5_sa4
                best_h = h_sa4
        elif c5_g4 < best_score:
            best_score = c5_g4
            best_h = h_g4

    # Phase 8: SLSQP multi-restart
    if tleft() > 80:
        alphas = [30, 80, 200, 500, 1500, 5000, 15000, 40000, 80000]
        n_restarts = 3
        for rs in range(n_restarts):
            if tleft() < 30:
                break
            if rs == 0:
                h_start_slsqp = best_h.copy()
            else:
                h_start_slsqp = best_h.copy()
                ones = np.where(h_start_slsqp > 0.5)[0]
                zeros = np.where(h_start_slsqp <= 0.5)[0]
                n_swaps = rs
                for _ in range(n_swaps):
                    if len(ones) > 0 and len(zeros) > 0:
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        h_start_slsqp[i] = 0.0
                        h_start_slsqp[j] = 1.0
                        ones = np.where(h_start_slsqp > 0.5)[0]
                        zeros = np.where(h_start_slsqp <= 0.5)[0]

            allowed = min(150, tleft() * 0.15)
            h_ref, sc = slsqp_refine(h_start_slsqp, alphas, 8000, 1e-14,
                                     time_limit=allowed, start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 9: SA on binary projection of SLSQP result
    if tleft() > 60:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(80, tleft() * 0.15)
        max_it5 = int(min(300000, allowed * 3000))
        h_sf, c5_sf = sa_search(h_bin_f, target_use, max_it5, 0.006, 0.00003, rng,
                                allowed, start_time, use_2swap=True)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 10: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        alphas2 = [10000, 30000, 60000, 100000, 200000]
        h_ref2, sc2 = slsqp_refine(best_h, alphas2, 5000, 1e-15,
                                   time_limit=max(10, tleft() - 10), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
