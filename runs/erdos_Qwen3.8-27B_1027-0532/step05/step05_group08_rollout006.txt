        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        
        g = np.zeros(n)
        if k_star >= 0:
            length = n - k_star
            g[:length] -= h_cur[k_star:k_star+length]
            g[k_star:] += (1.0 - h_cur[:length])
        else:
            ell = -k_star
            length = n - ell
            g[:length] += (1.0 - h_cur[ell:ell+length])
            g[ell:] -= h_cur[:length]
        g *= dx
        
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_score - 1e-15:
            h_cur = h_new
            best_h = h_cur.copy()
            best_score = c5_new
            eps = min(eps * 1.015, 0.02)
        else:
            eps *= 0.92
            if eps < 1e-10:
                break
    return best_h, best_score

def multi_lag_steps(h, n, target, m_active=10, n_steps=1000, eps_init=0.003, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:m_active]
        
        g = np.zeros(n)
        for m in top_idx:
            k = int(m) - (n - 1)
            if k >= 0:
                length = n - k
                g[:length] -= h_cur[k:k+length]
                g[k:] += (1.0 - h_cur[:length])
            else:
                ell = -k
                length = n - ell
                g[:length] += (1.0 - h_cur[ell:ell+length])
                g[ell:] -= h_cur[:length]
        g *= dx / m_active
        
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_score - 1e-15:
            h_cur = h_new
            best_h = h_cur.copy()
            best_score = c5_new
            eps = min(eps * 1.01, 0.015)
        else:
            eps *= 0.93
            if eps < 1e-10:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
