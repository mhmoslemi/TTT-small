```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = beta * (corr - c_max)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # grad[j] = dx * sum_k weights[k] * [(1-h[j+shift_k])*[j+shift_k<n] - h[j-shift_k]*[j-shift_k>=0]]

    for k in range(2*n - 1):
        s = k - (n - 1)  # shift
        wk = weights[k]
        if wk < 1e-15:
            continue
        # (1-h[j+s]) for valid j: j in [max(0,-s), min(n, n-s))
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_end > j_start:

            grad[idx] += wk * (1.0 - h[idx + s])
        # -h[j-s] for valid j: j in [max(0,s), min(n, n+s))
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)
        if j_end2 > j_start2:
            idx2 = np.arange(j_start2, j_end2)
            grad[idx2] -= wk * h[idx2 - s]
    
    return dx * grad
```

```python
def smooth_grad_direct(h, beta, target, penalty):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    one_minus_h = 1.0 - h
    
    for j in range(N):

        idx1 = j - t + N - 1  # for (1-h[t]) term
        idx2 = t - j + N - 1  # for h[t] term
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    
    # Add penalty gradient
    sum_diff = np.sum(h) - target
    grad += 2.0 * penalty * sum_diff
    
    return dx * grad
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def _smooth_val(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

def _smooth_val_pen(h, beta, target, penalty):
    v = _smooth_val(h, beta)
    d = np.sum(h) - target
    return v + penalty * d * d

def _smooth_grad(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    one_minus_h = 1.0 - h
    t_arr = np.arange(N)
    
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (np.sum(h) - target)
    return grad

def lbfgs_continuous(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smooth_val_pen(x, beta, target, penalty)
    
    def grad(x):
        return _smooth_grad(x, beta, target, penalty)
    
    h_cur = h_start.copy()
    for _ in range(3):  # Try up to 3 times
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < compute_c5(h_cur):
                h_cur = h_candidate
        except Exception:
            pass
    return h_cur

def binary_local_search(h_start, target_sum, max_iters, time_limit, start_time, use_gradient=True):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 200
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates that reduce the worst correlation
        i_cand = []
        j_cand = []
        for i in ones:
            if 0 <= i + k_star < n and h[i + k_star] < 0.5:
                i_cand.append(i)
        for j in zeros:
            if 0 <= j + k_star < n and h[j + k_star] > 0.5:
                j_cand.append(j)
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            # Try random swaps
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
        else:
            # Score candidates by their effect
            # For the worst shift, the improvement is:
            # Removing h[i]=1 at position i: reduces corr[k_star] by (1-h[i+k_star]) = 1 (since h[i+k_star]=0)
            # But we need to check other shifts too.
            # Simple heuristic: pick the i with the smallest "impact" on other shifts
            # and the j with the largest impact
            i = i_cand[np.random.randint(len(i_cand))]
            j = j_cand[np.random.randint(len(j_cand))]
        
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(corr.max()) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)

    best_h = project_box_sum(h_init, target)
    best_c5 = compute_c5(best_h)

    c5_bin_init = compute_c5(h_bin_init)
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin_init.copy()
    
    # Phase 1: Continuous L-BFGS-B optimization with increasing beta
    betas = [3, 5, 10, 20, 50, 100, 200, 500]
    h_curr = best_h.copy()
    for beta in betas:
        if tleft() < 60:

        time_alloc = min(80, tleft() * 0.1)
        h_opt = lbfgs_continuous(h_curr, beta, target, penalty=500.0,
                                  maxiter=500, time_limit=time_alloc, start_time=start_time)
        h_proj = project_box_sum(h_opt, target)
        sc = compute_c5(h_proj)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_proj.copy()
```

```python
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)

    for j in range(N):

        idx2 = t_arr - j + N - 1

        c5_new = float(corr_new.max()) * dx

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin_init = project_binary(h_init, target_int)
    c5_bin_init = compute_c5(h_bin_init)
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin_init.copy()

    # Phase 1: Continuous L-BFGS-B with increasing beta
    h_cont = best_h.copy()
    beta_schedule = [3, 5, 10, 20, 50, 100, 200, 500, 1000]
    for beta in beta_schedule:
        if tleft() < 50:
            break
        time_alloc = min(60, tleft() * 0.12)
        h_opt = lbfgs_continuous(h_cont, beta, target, penalty=500.0,
                                  maxiter=300, time_limit=time_alloc, start_time=start_time)
        sc = compute_c5(h_opt)
        h_cont = h_opt
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()

    # Phase 2: Binary local search from best
    if tleft() > 50:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(100, tleft() * 0.2)
        h_ls, c5_ls = greedy_binary(h_bin, target_int, time_limit=time_alloc,
                                     start_time=start_time, scan_width=25)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 3: SA from best binary
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(150, tleft() * 0.25)
        max_it = int(min(200000, time_alloc * 2000))
        h_sa, c5_sa = sa_binary(h_bin, target_int, max_it, 0.008, 0.00002,
                                 rng, time_limit=time_alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Continuous refinement from best
    if tleft() > 60:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [50, 100, 300, 700, 1500]:
            if tleft() < 25:
                break
            time_alloc = min(40, tleft() * 0.15)
            h_opt = lbfgs_continuous(h_cont, beta, target, penalty=1000.0,
                                      maxiter=250, time_limit=time_alloc, start_time=start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 5: Multi-restart with different betas
    if tleft() > 80:
        for restart in range(3):
            if tleft() < 40:
                break
            # Perturb the continuous solution
            perturbation = rng.normal(0, 0.05, n)
            h_pert = project_box_sum(best_h + perturbation, target)
            beta_r = [100, 300, 800]
            h_r = h_pert
            for beta_r_val in beta_r:
                if tleft() < 20:
                    break
                time_alloc = min(30, tleft() * 0.2)
                h_opt = lbfgs_continuous(h_r, beta_r_val, target, penalty=800.0,
                                          maxiter=200, time_limit=time_alloc, start_time=start_time)
                h_r = h_opt
                sc = compute_c5(h_r)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_r.copy()

    # Phase 6: Try different grid sizes
    n_options = [n + 2, n + 4, n - 2, n + 6, n + 8]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n]

    for n_try in n_options:
        if tleft() < 50:
            break
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)

        time_alloc = min(30, tleft() * 0.15)
        h_opt = lbfgs_continuous(h_interp, 50, target_try, penalty=500.0,
                                  maxiter=200, time_limit=time_alloc, start_time=start_time)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()

        if tleft() < 30:
            continue
        h_bin_try = project_binary(h_opt, target_try_int)
        time_alloc2 = min(25, tleft() * 0.1)
        h_ls2, c5_ls2 = greedy_binary(h_bin_try, target_try_int,
                                       time_limit=time_alloc2, start_time=start_time, scan_width=15)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 7: Final SA polish on binary projection of best
    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        time_alloc = min(60, tleft() - 10)
        max_it = int(min(100000, time_alloc * 2500))
        h_sf, c5_sf = sa_binary(h_bin_final, target_use, max_it, 0.004, 0.00001,
                                 rng, time_limit=time_alloc, start_time=start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 8: Final continuous polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_final_cont = project_box_sum(best_h.copy(), target_use)
        for beta_f in [300, 800, 2000]:
            if tleft() < 10:
                break
            time_alloc = min(20, tleft() * 0.4)
            h_polished = lbfgs_continuous(h_final_cont, beta_f, target_use,
                                           penalty=2000.0, maxiter=150,
                                           time_limit=time_alloc, start_time=start_time)
            h_final_cont = h_polished
            sc = compute_c5(h_polished)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_polished.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
```
