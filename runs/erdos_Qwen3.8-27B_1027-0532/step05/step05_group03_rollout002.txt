            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x):
    h_proj = np.clip(x, 0.0, 1.0)
    corr = compute_corr(h_proj)
    over = corr[_g_n - 1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _eq_fun(x):
    return np.sum(x) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def _qp_obj(w):
    A = _qp_A
    return 0.5 * float(w @ A @ w)

def _qp_cons_fun(w):
    return np.sum(w) - 1.0

def slsqp_opt(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(_sm_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < best_score:
                h_cur = h_new
                best_score = sc
        except Exception:
            pass
    return h_cur, best_score

def compute_weighted_grad(h, over, n_lags, top_k):
    n = len(h)
    c_max = float(np.max(over))
    shifted = n_lags * (over - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(over)[::-1][:top_k]
    for k in top_idx:
        k = int(k)
        w = weights[k]
        if w < 1e-12:
            continue
        if k == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            idx1 = np.arange(0, n - k)
            grad[idx1] += w * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= w * h[idx2 - k]
    return grad, weights, top_idx

def equalize_dir(h, over, m):
    global _qp_A
    n = len(h)
    top_idx = np.argsort(over)[::-1][:m]
    grads = np.zeros((n, m))
    for j, k in enumerate(top_idx):
        k = int(k)
        if k == 0:
            grads[:, j] = 1.0 - 2.0 * h
        else:
            idx1 = np.arange(0, n - k)
            grads[idx1, j] += (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grads[idx2, j] -= h[idx2 - k]
    A = grads.T @ grads
    _qp_A = A
    w0 = np.ones(m) / m
    bounds_qp = [(0.0, 1.0)] * m
    cons_qp = [{'type': 'eq', 'fun': _qp_cons_fun, 'jac': lambda w: np.ones(m)}]
    try:
        res = minimize(_qp_obj, w0, method='SLSQP', bounds=bounds_qp,
                      constraints=cons_qp, options={'maxiter': 200})
        w_opt = np.clip(res.x, 0.0, 1.0)
        w_opt /= w_opt.sum()
    except Exception:
        w_opt = w0
    d = -grads @ w_opt
    d = d - np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return None
    return d / dn

def equalization_search(h, target, n_iter, m_lags, lr_init):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        cur_score = float(np.max(over)) * dx
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        d = equalize_dir(h_cur, over, m_lags)
        if d is None:
            break
        h_new = project_feasible(h_cur + lr * d, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.85
            if lr < 1e-9:
                break
    return best_h, best_score

def fine_local_search(h, target, n_iter, alpha, top_k, lr_init):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        cur_score = float(np.max(over)) * dx
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        grad, weights, top_idx = compute_weighted_grad(h_cur, over, alpha, top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        direction = -grad / gn
        h_new = project_feasible(h_cur + lr * direction, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, 0.008)
        else:
            lr *= 0.88
            if lr < 1e-10:
                break
    return best_h, best_score

def l_bfgs_opt(h, alpha, target, maxiter):
    n = len(h)
    global _g_n, _g_alpha, _g_target, _g_rho
    _g_n = n
    _g_alpha = alpha
    _g_target = target
    _g_rho = 100.0

    def obj_pen(x):
        h_c = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h_c)
        over = corr[n-1:]
        c_max = float(np.max(over))
        shifted = _g_alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        sm = c_max + float(np.log(np.sum(np.exp(shifted))) / _g_alpha)
        pen = _g_rho * (np.sum(h_c) - target) ** 2
        return sm + pen

    def obj_grad(x):
        val = obj_pen(x)
        h_c = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h_c)
        over = corr[n-1:]
        c_max = float(np.max(over))
        shifted = _g_alpha * (over - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        top_idx = np.argsort(over)[::-1][:80]
        for k in top_idx:
            k = int(k)
            w = weights[k]
            if w < 1e-10:
                continue
            if k == 0:
                grad[:] += w * (1.0 - 2.0 * h_c)
            else:
                idx1 = np.arange(0, n - k)
                grad[idx1] += w * (1.0 - h_c[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= w * h_c[idx2 - k]
        pen_grad = 2.0 * _g_rho * (np.sum(h_c) - target)
        grad += pen_grad
        return val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def perturb_and_opt(h, target, rng, strength, n_coords):
    n = len(h)
    h_pert = h.copy()
    idx = rng.choice(n, size=n_coords, replace=False)
    h_pert[idx] += rng.randn(n_coords) * strength
    h_pert = project_feasible(h_pert, target)
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_rho, _qp_A
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 500:
        h, sc = slsqp_opt(best_h, [50, 200, 1000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 400:
        h, sc = slsqp_opt(best_h, [5000, 20000, 100000], 3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 300:
        h, sc = equalization_search(best_h, target, 500, 4, 0.004)
        if sc < best_score:
            best_score = sc
            best_h = h
        h, sc = slsqp_opt(best_h, [10000, 50000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 250:
        h, sc = equalization_search(best_h, target, 400, 6, 0.003)
        if sc < best_score:
            best_score = sc
            best_h = h
        h, sc = fine_local_search(best_h, target, 1000, 2000, 20, 0.003)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 200:
        h, sc = l_bfgs_opt(best_h, 5000, target, 500)
        if sc < best_score:
            best_score = sc
            best_h = h
        h, sc = slsqp_opt(best_h, [50000, 200000], 1500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 150:
        h, sc = equalization_search(best_h, target, 300, 5, 0.002)
        if sc < best_score:
            best_score = sc
            best_h = h
        h, sc = fine_local_search(best_h, target, 500, 5000, 30, 0.002)
        if sc < best_score:
            best_score = sc
            best_h = h

    n_attempts = 6
    for att in range(n_attempts):
        if tleft() < 80:
            break
        if att % 3 == 0:
            h_pert = perturb_and_opt(best_h, target, rng, 0.04, n // 8)
        elif att % 3 == 1:
            h_pert = perturb_and_opt(best_h, target, rng, 0.02, n // 4)
        else:
            n_flips = n // 12
            flip_idx = rng.choice(n, n_flips, replace=False)
            h_pert = best_h.copy()
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
            h_pert = project_feasible(h_pert, target)
        h, sc = slsqp_opt(h_pert, [1000, 10000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h
        h, sc = fine_local_search(best_h, target, 300, 5000, 20, 0.002)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 120:
        n_fine = n + 100
        target_fine = n_fine / 2.0
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n_fine)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, target_fine)
        h_fine, sc_fine = slsqp_opt(h_fine, [200, 1000, 5000], 1500, 1e-14)
        h_fine, sc_fine = fine_local_search(h_fine, target_fine, 500, 2000, 20, 0.003)
        h_fine, sc_fine = equalization_search(h_fine, target_fine, 200, 5, 0.003)
        h_fine, sc_fine = slsqp_opt(h_fine, [20000, 100000], 1000, 1e-15)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n_fine
            target = target_fine

    if tleft() > 60:
        h, sc = equalization_search(best_h, target, 300, 8, 0.0015)
        if sc < best_score:
            best_score = sc
            best_h = h
        h, sc = fine_local_search(best_h, target, 500, 10000, 40, 0.001)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 30:
        h, sc = slsqp_opt(best_h, [100000, 500000], 1000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h

    if tleft() > 15:
        h, sc = fine_local_search(best_h, target, 300, 50000, 50, 0.0005)
        if sc < best_score:
            best_score = sc
            best_h = h

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
