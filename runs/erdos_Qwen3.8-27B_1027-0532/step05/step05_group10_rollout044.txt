    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_FULL(h_old, n, target, solver_preference):
    """Solve the full linearized LP. Returns new h or None."""
    dx = 2.0 / n
    h_old = np.asarray(h_old, dtype=float)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
        t_var >= 0.0
    ]
    
    max_lag = n - 1
    for k in range(-max_lag, max_lag + 1):
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi <= lo:
            continue
        coeff = (1.0 - h_old[lo + k: hi + k]) * dx
        if np.max(coeff) < 1e-14:
            continue
        constraints.append(coeff @ h_var[lo:hi] <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for solver in solver_preference:
        try:
            if solver == 'SCS':
                prob.solve(solver=cp.SCS, eps=1e-10, max_iters=300000, verbose=False)
            elif solver == 'CLARABEL':
                prob.solve(solver=cp.CLARABEL, verbose=False)
            elif solver == 'ECOS':
                prob.solve(solver=cp.ECOS, max_iter=300000)
            elif solver == 'GLPK':
                prob.solve(solver=cp.GLPK, verbose=False)
            else:
                prob.solve(solver=solver, verbose=False)
            if h_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.asarray(h_var.value).ravel()
        except Exception:
            continue
    return None

def alt_lp(h_init, n, target, max_iter, time_limit, damping, solver_pref):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        h_lp = solve_lp_FULL(h, n, target, solver_pref)
        if h_lp is None:
            break
        h_lp = np.clip(h_lp, 0.0, 1.0)
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -300.0, 0.0)
    dx = 2.0 / n
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def slsqp_polish(h_init, n, target, alphas, maxiter, time_limit):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    con = ({'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t,
            'jac': lambda x, t=target: np.ones(len(x))})
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=[con],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def subgrad(h_init, n, target, max_steps, time_limit):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    half_n = n - 1
    idxs = np.arange(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k = int(np.argmax(corr)) - half_n
        idx_v1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_v2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[idx_v1] += (1.0 - h_cur[idxs[idx_v1] - k])
        grad[idx_v2] -= h_cur[idxs[idx_v2] + k]
        grad *= dx
        
        improved = False
        for lr in [0.001, 0.003, 0.005, 0.01, 0.03, 0.05]:
            h_try = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 100:
            break
    return best_h, best_c5

        h2, s2 = alt_lp_full(best_h, n, target, 80, min(120, tleft() * 0.25), 0.5, solvers)
        if s2 < best_score:
            best_score, best_h = s2, h2

    # Phase 3: SLSQP medium alpha
    if tleft() > 80:
        h3, s3 = slsqp_polish(best_h, n, target, [100, 500, 2000], 1500, min(60, tleft() * 0.4))
        if s3 < best_score:
            best_score, best_h = s3, h3

    # Phase 4: Full LP, high damping (converge to fixed point)
    if tleft() > 150:
        h4, s4 = alt_lp_full(best_h, n, target, 80, min(120, tleft() * 0.25), 0.7, solvers)
        if s4 < best_score:
            best_score, best_h = s4, h4

    # Phase 5: SLSQP high alpha
    if tleft() > 80:
        h5, s5 = slsqp_polish(best_h, n, target, [5000, 20000, 100000], 2000, min(60, tleft() * 0.4))
        if s5 < best_score:
            best_score, best_h = s5, h5

    # Phase 6: Subgradient descent
    if tleft() > 60:
        h6, s6 = subgrad_descent(best_h, n, target, 5000, min(40, tleft() * 0.3))
        if s6 < best_score:
            best_score, best_h = s6, h6

    # Phase 7: Full LP, very high damping (polish)
    if tleft() > 100:
        h7, s7 = alt_lp_full(best_h, n, target, 60, min(80, tleft() * 0.3), 0.9, solvers)
        if s7 < best_score:
            best_score, best_h = s7, h7

    # Phase 8: SLSQP very high alpha
    if tleft() > 60:
        h8, s8 = slsqp_polish(best_h, n, target, [50000, 200000, 1000000], 3000, min(50, tleft() * 0.4))
        if s8 < best_score:
            best_score, best_h = s8, h8

    # Phase 9: Try higher resolution
    for n_try in [500, 502, 600]:
        if tleft() < 120:
            break
        if n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h9, s9 = alt_lp_full(h_interp, n_try, target_try, 50, min(60, tleft() * 0.3), 0.5, solvers)
        if s9 < best_score:
            best_score = s9
            best_h = h9
            n = n_try
            target = target_try
        if tleft() > 40:
            h10, s10 = slsqp_polish(best_h, n, target, [2000, 10000], 1000, min(30, tleft() * 0.4))
            if s10 < best_score:
                best_score, best_h = s10, h10

    # Phase 10: Perturbed restarts
    for noise in [0.01, 0.03, 0.08, 0.15]:
        if tleft() < 80:
            break
        h_pert = project_feasible(best_h + rng.randn(len(best_h)) * noise, target)
        n_use = len(best_h)
        h11, s11 = alt_lp_full(h_pert, n_use, target, 30, min(30, tleft() * 0.2), 0.5, solvers)
        if s11 < best_score:
            best_score, best_h = s11, h11
        if tleft() > 40:
            h12, s12 = slsqp_polish(best_h, n_use, target, [5000, 50000], 1000, min(25, tleft() * 0.4))
            if s12 < best_score:
                best_score, best_h = s12, h12

    # Phase 11: Final LP polish
    if tleft() > 60:
        n_use = len(best_h)
        h13, s13 = alt_lp_full(best_h, n_use, n_use / 2.0, 40, min(40, tleft() * 0.4), 0.95, solvers)
        if s13 < best_score:
            best_score, best_h = s13, h13

    # Phase 12: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        h14, s14 = slsqp_polish(best_h, n_use, n_use / 2.0, [100000, 500000], 2000, min(25, tleft() * 0.6))
        if s14 < best_score:
            best_score, best_h = s14, h14

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
