import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Handle residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.8, reg=1e-4, n_active=50, solver='GLOP'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    solvers_to_try = [solver, 'ECOS', 'CBC']
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        for s in solvers_to_try:
            if time.time() - start > time_limit:
                break
            try:
                if s == 'GLOP':
                    prob.solve(solver=cp.GLOP)
                elif s == 'ECOS':
                    prob.solve(solver=cp.ECOS, max_iter=10000)
                elif s == 'CBC':
                    prob.solve(solver=cp.CBC)
                else:
                    prob.solve(solver=s)
                solved = True
                break
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.92)
        else:
            damping = min(0.95, damping + 0.02)

        if it % 50 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.7)

    return best_h, best_c5


def bottleneck_slsqp(h_start, n, target, alphas, top_M, maxiter_per_alpha, ftol):
    """SLSQP that only smooths the top M lags for better focus."""
    dx = 2.0 / n
    half_n = n - 1
    x_all = np.arange(2 * n - 1)
    
    # Precompute lag information
    lags = x_all - half_n  # lags from -(n-1) to (n-1)
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        # Compute current corr and find top M
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_indices = np.argsort(corr)[-top_M:]
        
        def obj(h):
            corr_val = np.correlate(h, 1.0 - h, mode='full')
            top_vals = corr_val[top_indices] * dx
            c_max = np.max(top_vals)
            shifted = np.clip(alpha * (top_vals - c_max), -400.0, 0.0)
            smooth = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            # Penalty for non-top lags exceeding smooth value
            other_vals = np.delete(corr_val * dx, top_indices)
            penalty = np.max(np.maximum(other_vals - smooth, 0)) ** 2
            return smooth + 0.1 * penalty
        
        def obj_grad(h):
            corr_val = np.correlate(h, 1.0 - h, mode='full')
            vals = corr_val * dx
            top_vals = vals[top_indices]
            c_max = np.max(top_vals)
            shifted = np.clip(alpha * (top_vals - c_max), -400.0, 0.0)
            weights = np.exp(shifted) / np.sum(np.exp(shifted))
            
            grad = np.zeros(n)
            for j, idx in enumerate(top_indices):
                k = int(lags[idx])
                w = weights[j] * alpha / alpha  # = weights[j]
                # d(corr[idx])/d(h[i]) for valid i
                # corr[idx] = sum_i h[i] * (1-h[i+k]) * dx (approximately)
                # d/dh[i] = (1-h[i+k])*dx - h[i-k]*dx
                valid1 = (0 <= np.arange(n) - k) & ((np.arange(n) - k) < n)
                valid2 = (0 <= np.arange(n) + k) & ((np.arange(n) + k) < n)
                grad[valid1] += w * (1.0 - h[np.arange(n)[valid1] - k]) * dx
                grad[valid2] -= w * h[np.arange(n)[valid2] + k] * dx
            
            # Add gradient of penalty (approximate)
            return grad
        
        def cons_fun(h):
            return np.array([np.sum(h) - target])
        
        def cons_jac(h):
            return np.array([np.ones(n)])
        
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        try:
            res = minimize(obj, h_cur, jac=obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score
