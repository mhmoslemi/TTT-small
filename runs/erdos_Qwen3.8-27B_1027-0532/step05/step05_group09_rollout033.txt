def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    return obj, grad, corr

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad, corr

def fista_phase(h0, target, alpha, deadline, step_init=None):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    if step_init is None:
        _, g0, _ = lse_obj_grad_fft(x, alpha)
        gnorm = np.linalg.norm(g0)
        if gnorm < 1e-10:
            step_init = 1.0
        else:
            step_init = 0.9 * dx / gnorm
    step = min(step_init, 2.0 * dx)
    max_step = 5.0 * dx
    stall = 0
    stall_threshold = 300
    for it in range(500000):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_c = np.clip(y, 0.0, 1.0)
        obj_y, grad_y, _ = lse_obj_grad_fft(y_c, alpha)
        x_new = project_feasible(y_c - step * grad_y, target)
        obj_new, _, _ = lse_obj_grad_fft(x_new, alpha)
        linear_bound = obj_y + np.dot(grad_y, x_new - y_c) + 0.5 / step * np.dot(x_new - y_c, x_new - y_c)
        if obj_new > linear_bound + 1e-14:
            for _ in range(20):
                step *= 0.5
                x_new = project_feasible(y_c - step * grad_y, target)
                obj_new, _, _ = lse_obj_grad_fft(x_new, alpha)
                linear_bound = obj_y + np.dot(grad_y, x_new - y_c) + 0.5 / step * np.dot(x_new - y_c, x_new - y_c)
                if obj_new <= linear_bound + 1e-14:
                    break
        step = min(step * 1.05, max_step)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score - 1e-12:
            best_score = sc
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_threshold:
                t = 1.0
                x_prev = x.copy()
                stall = 0
    return best_x, best_score

def nesterov_subgrad(h0, target, deadline, step_init=0.02):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    step = step_init
    fail_count = 0
    for it in range(300000):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_c = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y_c, 1.0 - y_c, mode='full')[n-1:]
        max_val = float(np.max(corr))
        active = np.where(corr >= max_val - 0.8)[0]
        grad = np.zeros(n)
        for ki in active:
            ki = int(ki)
            if ki == 0:
                grad[:] += y_c * (1.0 - y_c)
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += (1.0 - y_c[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= y_c[jn - ki]
        grad /= len(active)
        grad *= dx
        x_new = project_feasible(y_c - step * grad, target)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score - 1e-13:
            best_score = sc
            best_x = x.copy()
            fail_count = 0
            step = min(step * 1.01, 0.15)
        else:
            fail_count += 1
            step *= 0.98
            if fail_count > 800:
                fail_count = 0
                step = max(step, 0.005)
                t = 1.0
                x_prev = x.copy()
            if step < 1e-7:
                break
    return best_x, best_score

def perturbation_restart(h0, target, seed_val, deadline, alpha=20000):
    n = len(h0)
    rng = np.random.RandomState(seed_val)
    scale = 0.03
    h_pert = h0 + rng.randn(n) * scale
    h_pert = project_feasible(h_pert, target)
    h_new, sc_new = fista_phase(h_pert, target, alpha, deadline, step_init=None)
    return h_new, sc_new
