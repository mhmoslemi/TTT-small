```python
def coord_optimal_x(h, i, active_lags, n, dx, current_c5_vals):
    """Find optimal h[i] given all other coords fixed.
    active_lags: list of lag values (not indices)
    current_c5_vals: the current overlap values for each active lag
    """
    x0 = h[i]
    # For each active lag k, compute A_k = coefficient of (x - x0) in overlap_k
    A = np.zeros(len(active_lags))
    for idx, k in enumerate(active_lags):
        a_val = 0.0
        if 0 <= i + k and i + k < n:
            a_val += (1.0 - h[i + k])
        if 0 <= i - k and i - k < n:
            a_val -= h[i - k]
        A[idx] = a_val * dx
    
    C = np.array(current_c5_vals)  # current overlap values
    
    # Minimize max_k (C[k] + y*A[k]) for y in [-x0, 1-x0]
    y_lo = -x0
    y_hi = 1.0 - x0
    
    # Evaluate at endpoints
    f_lo = np.max(C + y_lo * A)
    f_hi = np.max(C + y_hi * A)
    
    # Find breakpoints in [y_lo, y_hi]
    m = len(A)
    best_y = y_lo if f_lo < f_hi else y_hi
    best_val = min(f_lo, f_hi)
    
    for p in range(m):
        for q in range(p+1, m):
            if abs(A[p] - A[q]) < 1e-15:
                continue
            y_bp = (C[q] - C[p]) / (A[p] - A[q])
            if y_lo <= y_bp <= y_hi:
                v = np.max(C + y_bp * A)
                if v < best_val:
                    best_val = v
                    best_y = y_bp
    
    return x0 + best_y
```

```python
def coord_descent(h_init, n, target, time_limit, n_active=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    overlaps = np.correlate(h, 1.0-h, mode='full') * dx  # shape (2n-1,)
    
    for sweep in range(10000):
        if time.time() - start > time_limit:
            break
        
        # Select active lags
        active_idx = np.argsort(overlaps)[-n_active:]
        active_lags = active_idx - (n-1)  # convert to lag values
        active_vals = overlaps[active_idx].copy()
        
        # Cycle through coordinates
        for i in range(n):
            if time.time() - start > time_limit:
                break
            
            x0 = h[i]
            # Compute A_k for each active lag
            A = np.zeros(n_active)
            for idx, k in enumerate(active_lags):
                a_val = 0.0
                if 0 <= i + k and i + k < n:
                    a_val += (1.0 - h[i + k])
                if 0 <= i - k and i - k < n:
                    a_val -= h[i - k]
                A[idx] = a_val * dx
            
            # Minimize max_k (active_vals[k] + y*A[k]) for y in [-x0, 1-x0]
            y_lo = -x0
            y_hi = 1.0 - x0
            m = n_active
            
            # Quick check: if all A[k] == 0, no change needed
            if np.max(np.abs(A)) < 1e-15:
                continue
            
            f_lo = np.max(active_vals + y_lo * A)
            f_hi = np.max(active_vals + y_hi * A)
            
            best_y = y_lo if f_lo <= f_hi else y_hi
            best_val = min(f_lo, f_hi)
            
            # Check breakpoints
            for p in range(m):
                for q in range(p+1, m):
                    da = A[p] - A[q]
                    if abs(da) < 1e-15:
                        continue
                    y_bp = (active_vals[q] - active_vals[p]) / da
                    if y_lo < y_bp < y_hi:
                        v = np.max(active_vals + y_bp * A)
                        if v < best_val:
                            best_val = v
                            best_y = y_bp
            
            new_x = x0 + best_y
            if abs(new_x - x0) < 1e-10:
                continue
            new_x = max(0.0, min(1.0, new_x))
            
            # Update h[i] and adjust h[j] to maintain sum
            delta = new_x - x0
            # Find j to adjust
            # Try j = (i+1) % n first, then others
            adjusted = False
            for offset in range(1, n):
                j = (i + offset) % n
                if j == i:
                    continue
                new_j = h[j] - delta
                if 0.0 <= new_j <= 1.0:
                    h[j] = new_j
                    adjusted = True
                    break
            
            if not adjusted:

                continue
            
            h[i] = new_x

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def coord_optimize(h, i, active_lags, active_vals, n, dx):
    x0 = h[i]
    m = len(active_lags)
    
    # Compute A vector
    A = np.zeros(m)
    for idx, k in enumerate(active_lags):
        a_val = 0.0
        if 0 <= i + k and i + k < n:
            a_val += (1.0 - h[i + k])
        if 0 <= i - k and i - k < n:
            a_val -= h[i - k]
        A[idx] = a_val * dx
    
    if np.max(np.abs(A)) < 1e-14:
        return x0
    
    C = active_vals
    y_lo = -x0
    y_hi = 1.0 - x0
    
    # Evaluate at endpoints
    f_lo = float(np.max(C + y_lo * A))
    f_hi = float(np.max(C + y_hi * A))
    
    best_y = y_lo if f_lo <= f_hi else y_hi
    best_val = min(f_lo, f_hi)
    
    # Vectorized breakpoint computation
    pi, qi = np.triu_indices(m, 1)
    da = A[pi] - A[qi]
    valid = np.abs(da) > 1e-14
    if np.any(valid):
        y_bps = np.zeros(np.sum(valid))
        y_bps[:] = (C[qi[valid]] - C[pi[valid]]) / da[valid]
        in_range = (y_bps > y_lo) & (y_bps < y_hi)
        if np.any(in_range):
            y_valid = y_bps[in_range]
            # Compute max(C + y*A) for each y
            vals = np.max(C[None, :] + y_valid[:, None] * A[None, :], axis=1)
            idx_best = np.argmin(vals)
            if vals[idx_best] < best_val:
                best_val = vals[idx_best]
                best_y = y_valid[idx_best]
    
    return x0 + best_y
```

Let me also think about the SLSQP approach. The smooth max approximation is:
smooth_max(corr, alpha) = (1/alpha) * log(sum(exp(alpha * corr)))

But this can overflow for large alpha * corr. A numerically stable version:
smooth_max(corr, alpha) = c_max + (1/alpha) * log(sum(exp(alpha * (corr - c_max))))

where c_max = max(corr). This is what the previous code does.

For SLSQP, the gradient of the objective w.r.t. h[i] is:
d/dh[i] of smooth_max = sum_k (alpha * exp(alpha*(c_k - c_max)) / sum_j exp(alpha*(c_j - c_max))) * dc_k/dh[i]

where dc_k/dh[i] is the partial derivative of overlap_k w.r.t. h[i], which is exactly A_k as computed before!

So the gradient is:
grad[i] = sum_k w_k * A_{k,i}

where w_k = alpha * exp(alpha*(c_k - c_max)) / sum_j exp(alpha*(c_j - c_max))

This is computable in O(n * n_lags) per gradient evaluation.

Actually, SLSQP in scipy computes finite differences for the gradient unless we provide it. Providing the analytic gradient would be much faster and more accurate. Let me do that.

```python
def slsqp_smooth(h_init, n, target, time_limit, alphas):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0-x, mode='full') * dx
            c_max = np.max(corr)
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500, 0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
        
        def grad(x):
            corr = np.correlate(x, 1.0-x, mode='full') * dx
            c_max = np.max(corr)
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted) / np.sum(np.exp(shifted))
            w *= alpha
            # grad[i] = sum_k w_k * A_{k,i}
            g = np.zeros(n)
            for k_idx in range(len(corr)):
```

```python
import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    for i in idx:
        h[i] += per
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            per2 = residual / len(idx2)
            for i in idx2:
                h[i] += per2
            h = np.clip(h, 0.0, 1.0)
    return h


def build_overlap_coeff(h, k, n, dx):
    g = 1.0 - h
    coeff = np.zeros(n)
    if k >= 0:
        if k < n:
            coeff[:n - k] = g[k:] * dx
    else:
        kk = -k
        if kk < n:
            coeff[kk:] = g[:n - kk] * dx
    return coeff


def alternating_lp(h_init, n, target, time_limit, n_active=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(int(x) for x in np.argsort(corr)[-n_active:])

    for it in range(100000):
        if time.time() - start > time_limit:
            break

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]

        for k_idx in sorted(active_k):
            k = k_idx - half_n
            coeff = build_overlap_coeff(h, k, n, dx)
            constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, abstol=1e-10, reltol=1e-10)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(solver=cp.CLARABEL)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        damping = 0.75
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(int(x) for x in np.argsort(corr)[-n_active:])
        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(int(x) for x in np.argsort(corr)[-3 * n_active:])

    return best_h, best_c5


def coord_optimize_x(h, i, active_lags, active_vals, n, dx):
    x0 = h[i]
    m = len(active_lags)
    A = np.zeros(m)
    for idx in range(m):
        k = active_lags[idx]
        a_val = 0.0
        if 0 <= i + k and i + k < n:
            a_val += (1.0 - h[i + k])
        if 0 <= i - k and i - k < n:
            a_val -= h[i - k]
        A[idx] = a_val * dx

    if np.max(np.abs(A)) < 1e-14:
        return x0

    C = active_vals
    y_lo = -x0
    y_hi = 1.0 - x0

    f_lo = float(np.max(C + y_lo * A))
    f_hi = float(np.max(C + y_hi * A))

    best_y = y_lo if f_lo <= f_hi else y_hi
    best_val = min(f_lo, f_hi)

    pi, qi = np.triu_indices(m, 1)
    da = A[pi] - A[qi]
    valid = np.abs(da) > 1e-14
    if np.any(valid):
        y_bps = np.empty(np.sum(valid))
        y_bps[:] = (C[qi[valid]] - C[pi[valid]]) / da[valid]
        in_range = (y_bps > y_lo + 1e-12) & (y_bps < y_hi - 1e-12)
        if np.any(in_range):
            y_valid = y_bps[in_range]
            vals = np.max(C[None, :] + y_valid[:, None] * A[None, :], axis=1)
            idx_best = np.argmin(vals)
            if vals[idx_best] < best_val:
                best_val = vals[idx_best]
                best_y = y_valid[idx_best]

    new_x = x0 + best_y
    return max(0.0, min(1.0, new_x))


def coord_descent(h_init, n, target, time_limit, n_active=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1

    no_improve_sweeps = 0
    max_no_improve = 30

    for sweep in range(100000):
        if time.time() - start > time_limit:
            break
        if no_improve_sweeps > max_no_improve:
            break

        overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        active_idx = np.argsort(overlaps)[-n_active:]
        active_lags = (active_idx - half_n).astype(int)
        active_vals = overlaps[active_idx].copy()

        for i in range(n):
            if time.time() - start > time_limit:
                break

            new_x = coord_optimize_x(h, i, active_lags, active_vals, n, dx)
            x0 = h[i]
            delta = new_x - x0
            if abs(delta) < 1e-12:
                continue

            adjusted = False
            for offset in range(1, n):
                j = (i + offset) % n
                if j == i:
                    continue
                new_j = h[j] - delta
                if -1e-12 <= new_j <= 1.0 + 1e-12:
                    h[j] = max(0.0, min(1.0, new_j))
                    adjusted = True
                    break

            if not adjusted:
                continue
            h[i] = new_x
            active_vals += delta * (np.zeros(n_active))

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

    return best_h, best_c5


def smooth_max_and_grad(x, alpha, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    w_sum = float(np.sum(exp_shifted))
    obj = c_max + np.log(w_sum) / alpha

    if w_sum < 1e-300:
        return float(obj), np.zeros(n)

    w = exp_shifted / w_sum
    half_n = n - 1
    grad = np.zeros(n)

    for k_idx in range(len(corr)):
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        k = k_idx - half_n
        if k >= 0:
            if k < n:
                grad[:n - k] += wi * (1.0 - x[k:])
                grad[:n - k] -= wi * x[:n - k]
        else:
            kk = -k
            if kk < n:
                grad[kk:] += wi * (1.0 - x[:n - kk])
                grad[:n - kk] -= wi * x[kk:]

    grad *= dx
    return float(obj), grad


def slsqp_optimize(h_init, n, target, time_limit, alphas):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break

        def obj_and_grad(x):
            val, g = smooth_max_and_grad(x, alpha, n, dx)
            return val, g

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]

        try:
            res = minimize(obj_and_grad, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 3: LP again after coordinate descent
    if tleft() > 150:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     time_limit=min(150, tleft() * 0.35), n_active=60)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 4: Coordinate descent with more active lags
    if tleft() > 200:
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                    time_limit=min(250, tleft() * 0.4), n_active=70)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if tleft() > 100:
        h_sl, c5_sl = slsqp_optimize(best_h, n, target,
                                     time_limit=min(100, tleft() * 0.4),
                                     alphas=[50, 200, 500])
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    # Phase 6: Coordinate descent final
    if tleft() > 100:
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                    time_limit=min(150, tleft() * 0.5), n_active=80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 7: Try perturbing and re-optimizing
    for noise in [0.03, 0.08, 0.15, 0.25]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        if tleft() > 40:
            h_cd, c5_cd = coord_descent(h_pert, n, target,
                                        time_limit=min(50, tleft() * 0.5), n_active=50)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 8: Final LP
    if tleft() > 60:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     time_limit=min(60, tleft() * 0.7), n_active=60)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 9: Final coordinate descent
    if tleft() > 30:
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                    time_limit=max(15, tleft() * 0.8), n_active=90)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_c5:
        best_c5 = final_score

    return h_final, best_c5, len(h_final)
```
