def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, target, max_sweeps, time_limit, start_time):

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:

        h_cur = best_h.copy()
        any_improved = False
        for i in range(n):

                    best_h = h_cur.copy()
                    any_improved = True

            elif old_val > 1.0 - 1e-10:
                # Try setting to 0

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_gradient(h, alpha, n):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)

    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / sum_exp
    
    A = np.zeros(n)
    B = np.zeros(n)
    half_n = n - 1

            B[k:] += w * h[:n - k]
        else:
            ka = -k
            A[ka:] += w * h[:n - ka]
            B[:n - ka] += w * h[ka:]
    
    return dx * (1.0 - A - B)

        res = minimize(lse_objective, h_start, args=(alpha, n),
                       jac=lambda x: lse_gradient(x, alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start.copy()
