Actually, one more important consideration: the `find_t_star` function. Let me make sure the binary search is correct and efficient.

The key insight: we're minimizing f(t) = max_m (A[m] + C[m]*t) over t ∈ [t_min, t_max].

f is convex and piecewise linear. The minimum occurs where:
- The subgradient 0 ∈ ∂f(t*), i.e., there exist m₁, m₂ such that C[m₁] > 0, C[m₂] < 0, and A[m₁] + C[m₁]*t* = A[m₂] + C[m₂]*t* = f(t*), OR
- t* is at a boundary (t_min or t_max).

The binary search on v finds the minimum value v* = f(t*). Then t* is found from the feasible interval at v = v*.

The number of binary search iterations determines the precision of v*. With 30 iterations, we get precision of about (v_hi - v_lo) / 2³⁰. If v_hi - v_lo is about 1.0, that's about 1e-9 precision. That's plenty.

Actually, I realize I can do better: instead of binary search on v, I can use ternary search directly on t (since f is convex). Ternary search on a 1D convex function:
- Evaluate f at two interior points
- Discard the half with the larger value
- Repeat

Each evaluation is O(n) (computing max of A + C*t for a given t). With 50 iterations, precision is (t_max - t_min) / (3/2)⁵⁰ ≈ 1 / 10¹⁹. More than enough.

But binary search on v is also fine and might be more robust. Let me stick with it.

Let me also add a special case: if C is all zeros (or nearly all zeros), then f(t) = max(A) for all t, and any t in [t_min, t_max] is optimal. In this case, we should just keep h[i] and h[j] unchanged.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_B(h, n):
    B = np.zeros((n, 2 * n - 1))
    B[:, n - 1] = 1.0 - 2.0 * h
    for s in range(1, n):
        mp = n - 1 + s
        mn = n - 1 - s
        i1 = np.arange(n - s)
        i2 = np.arange(s, n)
        B[i1, mp] += 1.0 - h[i1 + s]
        B[i2, mp] -= h[i2 - s]
        B[i2, mn] += 1.0 - h[i2 - s]
        B[i1, mn] -= h[i1 + s]
    return B

def find_t_star(A, C, t_min, t_max, n_iter=30):
    pos = C > 1e-14
    neg = C < -1e-14
    has_pos = np.any(pos)
    has_neg = np.any(neg)
    
    if not has_pos and not has_neg:
        return (t_min + t_max) * 0.5
    
    # Binary search on v = min f(t)
    f0 = float(np.max(A + C * t_min))
    f1 = float(np.max(A + C * t_max))
    v_lo = min(f0, f1) - 0.05
    v_hi = max(f0, f1)
    
    for _ in range(n_iter):
        vm = (v_lo + v_hi) * 0.5
        lb = t_min
        if has_neg:
            lb = max(lb, float(np.max((vm - A[neg]) / C[neg])))
        ub = t_max
        if has_pos:
            ub = min(ub, float(np.min((vm - A[pos]) / C[pos])))
        if lb <= ub + 1e-12:
            v_hi = vm
        else:
            v_lo = vm
    
    v = v_hi
    lb = t_min
    if has_neg:
        lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
    ub = t_max
    if has_pos:
        ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
    if lb > ub:
        return (t_min + t_max) * 0.5
    return (lb + ub) * 0.5

def fast_coord_desc(h_start, n, target, deadline, rng, recompute_interval=10):
    h_cur = h_start.copy()
    corr = compute_corr(h_cur)
    B = compute_all_B(h_cur, n)
    best_score = float(np.max(corr)) * (2.0 / n)
    best_h = h_cur.copy()
    stall = 0
    sweep = 0
    dx = 2.0 / n
    
    while time.time() < deadline:
        sweep += 1
        if sweep % recompute_interval == 0:
            corr = compute_corr(h_cur)
            B = compute_all_B(h_cur, n)
        
        perm = rng.permutation(n)
        
        for p in range(n // 2):
            i = int(perm[2 * p])
            j = int(perm[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            
            C = B[i] - B[j]
            A = corr - h_cur[i] * C
            t_star = find_t_star(A, C, t_min, t_max, 25)
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        
        score = float(np.max(corr)) * dx
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 300:
            sc = 0.005 + 0.005 * min(stall / 200.0, 1.0)
            h_cur = project_feasible(h_cur + rng.randn(n) * sc, target)
            corr = compute_corr(h_cur)
            B = compute_all_B(h_cur, n)
            stall = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def slsqp_obj(x):
    corr = compute_corr(x)
    amax = float(np.max(corr))
    shifted = _g_alpha * (corr - amax)
    shifted = np.clip(shifted, -700.0, 0.0)
    return amax + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_grad(x):
    corr = compute_corr(x)
    amax = float(np.max(corr))
    shifted = _g_alpha * (corr - amax)
    shifted = np.clip(shifted, -700.0, 0.0)
    es = np.exp(shifted)
    p = es / float(np.sum(es))
    B = compute_all_B(x, len(x))
    return B.T @ p

def slsqp_cons(x):
    return np.sum(x) - _g_target

def slsqp_cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h, n, target, alphas, maxiter=2000):

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP',
                          jac=slsqp_grad, bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur
