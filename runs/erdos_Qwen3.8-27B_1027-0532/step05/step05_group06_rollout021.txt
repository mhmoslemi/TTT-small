```python
    """Compute the objective as a function of h[j], returning the slope and intercept for each active lag."""
    n = len(h)
    dx = 2.0 / n
    old_val = h[j]
    
    # Precompute the current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')

    # delta * (1 - h[j + (k - (n-1))]) - delta * h[j - (k - (n-1))] (if indices valid)

    # - h[j] appears as (1-h[i + k - (n-1)]) when i + k - (n-1) = j, i.e., i = j - k + (n-1):
    #   contribution -delta * h[j - k + (n-1)]

    # where _valid means the index is in range, else 0
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fast_coordinate_descent(h, target, n_sweeps=50, n_evals=7, tol=1e-10):
    """Fast coordinate descent exploiting piecewise-linear structure."""
    n = len(h)
    h_cur = h.copy()
    h_cur = project_feasible(h_cur, target)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_score = float(np.max(corr)) * (2.0 / n)
    
    # Precompute the lag indices
    lags = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        improved = 0.0
        for j in range(n):
            old = h_cur[j]
            # Compute coefficient vector for coordinate j
            # corr_new[k] = corr[k] + (v - old) * coeff[k]
            # where coeff[k] = (1-h[j+lag_k])_valid - h[j-lag_k]_valid
            j_plus = j + lags  # index j + lag
            j_minus = j - lags  # index j - lag
            
            coeff = np.zeros(2*n-1)
            valid_p = (j_plus >= 0) & (j_plus < n)
            coeff[valid_p] += (1.0 - h_cur[j_plus[valid_p]])
            valid_m = (j_minus >= 0) & (j_minus < n)
            coeff[valid_m] -= h_cur[j_minus[valid_m]]
            
            # f_j(v) = max_k (corr[k] + (v-old)*coeff[k]) 
            # = max_k (coeff[k]*v + (corr[k] - old*coeff[k]))
            intercept = corr - old * coeff
            # f_j(v) = max_k (coeff[k]*v + intercept[k])
            
            # Find minimizer over v ∈ [0,1]
            # Evaluate at several points

            vals = np.max(evals[:, None] * coeff[None, :] + intercept[None, :], axis=1)
            best_idx = np.argmin(vals)
            best_v = evals[best_idx]

            if 0 < best_idx < n_evals - 1:
                lo = evals[best_idx - 1]
                hi = evals[best_idx + 1]
                fine_evals = np.linspace(lo, hi, 7)
                fine_vals = np.max(fine_evals[:, None] * coeff[None, :] + intercept[None, :], axis=1)
                best_v = fine_evals[np.argmin(fine_vals)]
            elif best_idx == 0:

                best_v = fine_evals[np.argmin(fine_vals)]
            elif best_idx == n_evals - 1:

                delta = best_v - old
                corr += delta * coeff
                h_cur[j] = best_v

        h_cur = project_feasible(h_cur, target)
        # Recompute corr after projection
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        new_score = float(np.max(corr)) * (2.0 / n)
        
        if new_score < cur_score - 1e-14:
            cur_score = new_score
        if improved < tol:
            break
    
    return h_cur, cur_score
```

Actually, I can be even more efficient. Instead of evaluating at multiple points and then refining, I can use the fact that the minimum of a convex PL function can be found more directly.

For f(v) = max_k (a_k * v + b_k), the minimum over [0,1] is:
- If the function is decreasing on [0,1], the min is at v=1
- If it's increasing, the min is at v=0
- Otherwise, the min is at a breakpoint

The breakpoints are where two lines cross: a_i * v + b_i = a_j * v + b_j → v = (b_j - b_i) / (a_i - a_j)

But computing all pairwise crossings is O(n²) which is too expensive.

Alternative: use the fact that we can compute the subgradient (as a set) at any point in O(n), and use bisection on the subgradient.

At point v, the active set is A(v) = {k : a_k*v + b_k = max_j(a_j*v + b_j)}.
The subgradient is [min_{k∈A} a_k, max_{k∈A} a_k].
If 0 ∈ subgradient, v is optimal.
If min subgradient > 0, we need to go left.
If max subgradient < 0, we need to go right.

So the algorithm:
1. v_lo = 0, v_hi = 1
2. For a few iterations:
   a. v_mid = (v_lo + v_hi) / 2
   b. Compute active set at v_mid: A = argmax(a*v_mid + b)
   c. sg_lo = min(a[A]), sg_hi = max(a[A])
   d. If sg_lo <= 0 <= sg_hi: v_mid is optimal (or close enough)
   e. If sg_lo > 0: v_hi = v_mid (need to go left)
   f. If sg_hi < 0: v_lo = v_mid (need to go right)
3. Return v_mid

Each iteration is O(n) (compute a*v_mid + b, find argmax, get min/max of active a's). With 20 iterations, that's 20*O(n) per coordinate.

```python
def lse_objective(x, alpha, mu, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = c_max + log_sum / alpha
    val += mu * (np.sum(x) - target) ** 2
    return val

def lse_grad(x, alpha, mu, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = weights / wsum
    
    grad = np.zeros(n)
    for ki in range(len(corr)):
        if weights[ki] < 1e-10:
            continue
        w = weights[ki]
        lag = ki - (n - 1)
        if lag >= 0:
            # corr[ki] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # d/dh[j] = sum over terms where j is involved
            # h[j] as first factor: j in [0, n-1-lag], contributes (1-h[j+lag])
            # h[j] as second factor (in 1-h[j]): j+lag in [0, n-1], i.e., j in [lag, n-1], contributes -h[j-lag]
            # For coordinate j:
            #   if 0 <= j <= n-1-lag: grad[j] += w * (1-h[j+lag])
            #   if lag <= j <= n-1: grad[j] -= w * h[j-lag]
            j1 = np.arange(0, n - lag)
            grad[j1] += w * (1.0 - x[j1 + lag])
            if lag > 0:
                j2 = np.arange(lag, n)
                grad[j2] -= w * x[j2 - lag]
        else:
            m = -lag  # m > 0
            # corr[ki] = sum_{i=m}^{n-1} h[i] * (1-h[i-m])
            # For coordinate j:
            #   if m <= j <= n-1: grad[j] += w * (1-h[j-m])
            #   if 0 <= j-m <= n-1: grad[j] -= w * h[j+m]
            j1 = np.arange(m, n)
            grad[j1] += w * (1.0 - x[j1 - m])
            if m > 0:
                j2 = np.arange(0, n - m)
                grad[j2] -= w * x[j2 + m]
    
    grad += 2.0 * mu * (np.sum(x) - target)
    return grad
```

```python
def lse_func_and_grad(x, alpha, mu, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(corr.max())
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = weights / wsum
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-8:
            continue
        L = int(lags[ki])
        j_plus_arr = np.arange(n) + L  # j + L for j=0..n-1
        valid_p = (j_plus_arr >= 0) & (j_plus_arr < n)
        grad[valid_p] += w * (1.0 - x[j_plus_arr[valid_p]])
        j_minus_arr = np.arange(n) - L  # j - L for j=0..n-1
        valid_m = (j_minus_arr >= 0) & (j_minus_arr < n)
        grad[valid_m] -= w * x[j_minus_arr[valid_m]]
    
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_polish(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:

            h.copy(),

            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12, 'maxls': 40}
        )
        return project_feasible(res.x, target)
    except Exception:
        return h
```
