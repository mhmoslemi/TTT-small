```python
def compute_swap_delta(h, a, b, n):
    """Compute delta in over[] when h[a] goes 1->0 and h[b] goes 0->1."""
    # a: position changing from 1 to 0
    # b: position changing from 0 to 1
    delta = np.zeros(n)
    
    # Effect of a (1->0):
    # As i+k term: for k in [0, a], contribution is h[a-k]-1
    # As i term: for k in [0, n-1-a], contribution is h[a+k]
    k_range_a1 = min(a, n-1)  # max k where a-k >= 0
    if a >= 0:
        ks = np.arange(0, min(a, n-1) + 1)
        idx = a - ks  # indices a-k
        delta[ks] += h[idx] - 1.0
    
    k_max_a2 = n - 1 - a
    if k_max_a2 >= 0:
        ks = np.arange(0, k_max_a2 + 1)
        idx = a + ks  # indices a+k
        delta[ks] += h[idx]
    
    # Effect of b (0->1):
    # As i+k term: for k in [0, b], contribution is 1-h[b-k]
    # As i term: for k in [0, n-1-b], contribution is -h[b+k]
    if b >= 0:
        ks = np.arange(0, min(b, n-1) + 1)
        idx = b - ks
        delta[ks] += 1.0 - h[idx]
    
    k_max_b2 = n - 1 - b
    if k_max_b2 >= 0:
        ks = np.arange(0, k_max_b2 + 1)
        idx = b + ks
        delta[ks] -= h[idx]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_over(h):
    """Compute the overlap correlations (lags 0 to n-1)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:]  # lags 0 to n-1

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def swap_delta(h, a, b):
    """
    Compute delta in over[] (length n) when h[a] goes 1->0 and h[b] goes 0->1.
    h is assumed binary (0 or 1).
    """
    n = len(h)
    delta = np.zeros(n)
    
    # a goes 1 -> 0:
    # As (i+k) term: for k in [0, a], i = a-k, contribution was h[a]*(1-h[a-k]) = 1-h[a-k], now 0
    #   delta = -(1-h[a-k]) = h[a-k] - 1
    k_max1 = min(a, n-1)
    if k_max1 >= 0:
        ks = np.arange(0, k_max1 + 1)
        delta[ks] += h[a - ks] - 1.0
    
    # As i term: for k in [0, n-1-a], i = a, contribution was h[a+k]*(1-h[a]) = 0, now h[a+k]
    #   delta = h[a+k] - 0 = h[a+k]
    k_max2 = n - 1 - a
    if k_max2 >= 0:
        ks = np.arange(0, k_max2 + 1)
        delta[ks] += h[a + ks]
    
    # b goes 0 -> 1:
    # As (i+k) term: for k in [0, b], i = b-k, contribution was 0*(1-h[b-k]) = 0, now 1*(1-h[b-k])
    #   delta = 1-h[b-k]
    k_max3 = min(b, n-1)
    if k_max3 >= 0:
        ks = np.arange(0, k_max3 + 1)
        delta[ks] += 1.0 - h[b - ks]
    
    # As i term: for k in [0, n-1-b], i = b, contribution was h[b+k]*(1-0) = h[b+k], now 0
    #   delta = 0 - h[b+k] = -h[b+k]
    k_max4 = n - 1 - b
    if k_max4 >= 0:
        ks = np.arange(0, k_max4 + 1)
        delta[ks] -= h[b + ks]
    
    return delta
```
