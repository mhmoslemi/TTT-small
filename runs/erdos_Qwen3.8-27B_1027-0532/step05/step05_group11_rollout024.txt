def project_feasible_waters(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1} using water-filling."""
    h = np.clip(h, 0.0, 1.0)
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    
    # Bisection for lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

def soft_max_eval(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    sh = np.clip(alpha * (corr - mx), -500.0, 0.0)
    exps = np.exp(sh)
    se = np.sum(exps)
    return float(mx + np.log(se) / alpha)

def soft_max_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    sh = np.clip(alpha * (corr - mx), -500.0, 0.0)
    exps = np.exp(sh)
    se = np.sum(exps)
    weights = exps / se
    grad = np.zeros(n)
    n_half = n - 1
    thresh = 1.0 / (n * 100)
    sig = np.where(weights > thresh)[0]
    for ki in sig:
        m = ki - n_half
        w = weights[ki] * dx
        if m >= 0:
            hi = n - m
            grad[:hi] += w * (1.0 - h[m:m+hi])
            grad[m:] += w * (-h[:n-m])
        else:
            p = -m
            grad[p:] += w * (1.0 - h[:n-p])
