    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(300):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)

    if lr_init is None:
        lr_init = 1.0 / max(alpha * 0.03, 0.5)
    step = lr_init

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        accepted = False
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink) - 1e-15:
                accepted = True
                break
            step *= (1.0 - shrink)
            if step < 1e-14:
                break
            x_new = project_feasible(y_clipped - step * grad_y, target)
        if not accepted:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        if k % 50 == 0 or k < 10:
            sc = compute_c5(x)
            if sc < best_obj:
                best_obj = sc
                best_x = x.copy()
    sc = compute_c5(x)
    if sc < best_obj:
        best_obj = sc
        best_x = x.copy()
    return best_x, best_obj

def active_set_fista(h0, target, deadline, num_active=4, momentum=True):
    n = len(h0)
    dx = 2.0 / n
    h_cur = project_feasible(h0, target).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step = dx * n * 0.1
    t_mom = 1.0
    x_prev = h_cur.copy()
    y = h_cur.copy()

    for it in range(200000):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        if momentum:
            y = h_cur + ((t_mom - 1.0) / t_new) * (h_cur - x_prev)
        else:
            y = h_cur.copy()
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')[n-1:]
        c_max = np.max(corr)

        top_idx = np.argsort(corr)[-num_active:]
        min_top = corr[top_idx[-1]]
        diffs = corr[top_idx] - min_top
        scale = max(0.002, 0.01 * c_max)
        w = np.exp(diffs / scale)
        w /= w.sum()

        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            if ki == 0:
                grad += wi * (1.0 - 2.0 * y)
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += wi * (1.0 - y[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= wi * y[jn - ki]
        grad *= dx

        obj_y = compute_c5(y)
        for ls in range(25):
            h_new = project_feasible(y - step * grad, target)
            sc_new = compute_c5(h_new)
            if sc_new < obj_y - 1e-12:
                break
            step *= 0.5
        if sc_new >= obj_y - 1e-12:
            h_new = project_feasible(y - step * 0.5 * grad, target)
            sc_new = compute_c5(h_new)
        step = min(step * 1.15, dx * n * 0.3)
        x_prev = h_cur
        h_cur = h_new
        t_mom = t_new

        if it % 30 == 0:
            sc = compute_c5(h_cur)
            if sc < best_score:
                best_score = sc
                best_h = h_cur.copy()

    sc = compute_c5(h_cur)
    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()
    return best_h, best_score

def binary_local_search(h0, target, deadline, max_rounds=50):
    n = len(h0)
    h_cur = np.where(h0 > 0.5, 1.0, 0.0)
    h_cur = project_feasible(h_cur, target)
    h_cur = np.where(h_cur > 0.5, 1.0, 0.0)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    rng = np.random.RandomState(777)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old = h_cur[idx]
            new = 1.0 - old
            h_cur[idx] = new
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = h_cur.copy()
                improved = True
            else:
                h_cur[idx] = old
                h_cur = project_feasible(h_cur, target)
        if not improved:
            break
    h_relax = project_feasible(best_h + 0.1 * (np.random.RandomState(42).rand(n) - 0.5), target)
    return best_h, best_score

def sinusoidal_init(n, freq, phase, power):
    x = np.linspace(0, 2.0, n, endpoint=False)
    vals = np.abs(np.sin(np.pi * freq * x + phase)) ** power
    vals = vals / vals.sum() * n / 2.0
    return np.clip(vals, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline_final = start_time + budget_s - 10.0

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline_final - time.time()

    def try_update(h_candidate, score_candidate):
        nonlocal best_h, best_score
        if score_candidate < best_score:
            best_score = score_candidate
            best_h = h_candidate.copy()

    # Phase 1: FISTA continuation with multiple alpha levels
    alpha_schedule = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
    time_per_phase = [30, 30, 35, 40, 40, 45, 50, 60]
    for i, (a, td) in enumerate(zip(alpha_schedule, time_per_phase)):
        if tleft() < td + 5:
            break
        dl = start_time + (sum(time_per_phase[:i+1]))
        if dl > deadline_final - 200:
            dl = deadline_final - 200
        h_new, sc_new = fista_optimize(best_h, target, a, 500000, dl)
        try_update(h_new, sc_new)

    # Phase 2: Active-set FISTA polish
    if tleft() > 80:
        h_new, sc_new = active_set_fista(best_h, target, start_time + 80, num_active=3)
        try_update(h_new, sc_new)
        if tleft() > 60:
            h_new, sc_new = active_set_fista(best_h, target, start_time + 60, num_active=5)
            try_update(h_new, sc_new)

    # Phase 3: More FISTA at very high alpha
    if tleft() > 100:
        h_new, sc_new = fista_optimize(best_h, target, 50000, 500000, start_time + 60)
        try_update(h_new, sc_new)
        if tleft() > 40:
            h_new, sc_new = fista_optimize(best_h, target, 100000, 500000, start_time + 40)
            try_update(h_new, sc_new)

    # Phase 4: Active-set polish again
    if tleft() > 60:
        h_new, sc_new = active_set_fista(best_h, target, start_time + 60, num_active=3)
        try_update(h_new, sc_new)

    # Phase 5: Structured restarts
    rng = np.random.RandomState(seed)
    restart_configs = []
    for freq in range(2, 9):
        for phase in [0.0, 0.5, 1.0, 1.5]:
            for power in [1.5, 2.0, 3.0]:
                restart_configs.append((freq, phase, power))

    max_restarts = min(len(restart_configs), 12)
    selected_configs = rng.choice(len(restart_configs), size=max_restarts, replace=False)
    for idx in selected_configs:
        if tleft() < 35:
            break
        f, p, pw = restart_configs[int(idx)]
        h_struct = sinusoidal_init(n, f, p, pw)
        h_struct = project_feasible(h_struct, target)
        h_struct = fista_optimize(h_struct, target, 200, 100000, start_time + 15)[0]
        h_struct, sc_struct = active_set_fista(h_struct, target, start_time + 15, num_active=3)
        if sc_struct < best_score:
            best_score = sc_struct
            best_h = h_struct.copy()

    # Phase 6: Random perturbations from current best
    if tleft() > 100:
        for scale in [0.02, 0.05, 0.1, 0.15]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, target)
            h_new, sc_new = fista_optimize(h_pert, target, 3000, 500000, start_time + 12)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            if tleft() > 15:
                h_new, sc_new = active_set_fista(best_h, target, start_time + 15, num_active=3)
                try_update(h_new, sc_new)

    # Phase 7: Binary search + relax
    if tleft() > 60:
        h_bin, sc_bin = binary_local_search(best_h, target, start_time + 20)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
        h_relax = project_feasible(h_bin + 0.05 * (rng.rand(n) - 0.5), target)
        h_relax, sc_relax = fista_optimize(h_relax, target, 5000, 500000, start_time + 15)
        if sc_relax < best_score:
            best_score = sc_relax
            best_h = h_relax
        if tleft() > 15:
            h_new, sc_new = active_set_fista(best_h, target, start_time + 15, num_active=3)
            try_update(h_new, sc_new)

    # Phase 8: Final fine-tuning
    if tleft() > 40:
        h_new, sc_new = active_set_fista(best_h, target, start_time + 20, num_active=3)
        try_update(h_new, sc_new)
        if tleft() > 15:
            h_new, sc_new = fista_optimize(best_h, target, 80000, 500000, start_time + 15)
            try_update(h_new, sc_new)

    # Phase 9: Ultimate active-set polish
    if tleft() > 20:
        h_new, sc_new = active_set_fista(best_h, target, start_time + max(tleft() - 5, 10), num_active=3)
        try_update(h_new, sc_new)

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
