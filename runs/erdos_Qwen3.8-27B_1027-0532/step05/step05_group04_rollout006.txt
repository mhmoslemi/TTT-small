        res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:

    except Exception:

def reweight_descent(h, n, target, n_outer, beta_init, beta_growth, 
                     inner_steps, lr_init, top_k, deadline=None):
    """Iterative reweighted gradient descent focusing on top-k worst lags"""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init
    
    for outer in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        
        # Compute all overlaps to find worst k
        over_full = compute_corr_overlaps_full(h_cur)
        max_over = float(np.max(over_full))
        
        # Weight the overlaps
        weights = np.exp(beta * (over_full - max_over))
        weights /= weights.sum()
        
        # Gradient using weighted sum
        grad = np.zeros(n)
        for k in range(n):
            if weights[k] < 1e-10:
                continue
            grad += weights[k] * grad_for_k(h_cur, k)
        grad *= dx
        
        # Line search
        improved = False
        lr_cur = lr
        for attempt in range(20):
            h_new = h_cur - lr_cur * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc_new
                improved = True
                break
            lr_cur *= 0.7
        if not improved:
            lr *= 0.6
            if lr < 1e-8:
                break
        else:
            lr = min(lr * 1.02, 0.05)
        
        beta = min(beta * beta_growth, 100000.0)
    
    return h_cur, cur_score

def single_worst_descent(h, n, target, n_steps, lr_init, deadline=None):
    """Focus on the single worst k at each step"""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init

        worst_k = int(np.argmax(over))

        lr_cur = lr
        improved = False
        for attempt in range(15):

            if sc_new < cur_score - 1e-15:

                cur_score = sc_new
                improved = True
                break
            lr_cur *= 0.6
        if not improved:
            lr *= 0.5
            if lr < 1e-9:
                break
    
    return h_cur, cur_score

def momentum_reweight(h, n, target, n_steps, beta, lr, momentum, deadline=None):
    """Momentum-based reweighted gradient descent"""
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        over = compute_corr_overlaps_full(h_cur)
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(n):
            if weights[k] < 1e-10:
                continue
            grad += weights[k] * grad_for_k(h_cur, k)
        grad *= dx
        
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            v = momentum * v * 0.8
    
    return h_cur, cur_score

Let me think about a different approach entirely.

**Alternative: Use the structure of the problem**

The overlap for shift k is:
overlap_k = ∑_{i=0}^{n-1-k} h[i](1-h[i+k])

For k=0: overlap_0 = ∑h[i](1-h[i]) = n/2 - ∑h[i]² (since ∑h[i] = n/2)

So overlap_0 = n/2 - ∑h[i]². To minimize overlap_0, we want to maximize ∑h[i]², which means making h more "binary" (clustering values toward 0 and 1).

For k=1: overlap_1 = ∑h[i](1-h[i+1]) = ∑_{i=0}^{n-2} h[i] - ∑h[i]h[i+1]

The term ∑_{i=0}^{n-2} h[i] = n/2 - h[n-1] (approximately, since the last element might not be counted).

Actually: ∑_{i=0}^{n-2} h[i] = (n/2) - h[n-1]

So overlap_1 = (n/2 - h[n-1]) - ∑_{i=0}^{n-2} h[i]h[i+1]

To minimize overlap_1, we want to maximize ∑h[i]h[i+1] (adjacent products) while keeping h[n-1] small.

This suggests that the optimal h should be "blocky" - long runs of 1s followed by long runs of 0s - to maximize adjacent products. But we also need to do well for all k simultaneously.

This is related to the concept of "low autocorrelation" sequences. For binary sequences, the autocorrelation at shift k is related to the number of matching pairs at that distance.

**Insight**: The optimal h might have a specific structure. Let me think about what structure minimizes max_k overlap_k.

For a binary h with n/2 = 200 ones:
- overlap_k = (number of ones in positions where h[i+k]=0, for i=0..n-1-k)
- = (number of 1-0 pairs at distance k)

We want to minimize the maximum number of 1-0 pairs at any distance k.

For a random binary sequence with 200 ones out of 400, the expected number of 1-0 pairs at distance k is approximately:
- (n-k) * P(h[i]=1) * P(h[i+k]=0) ≈ (n-k) * (1/2) * (1/2) = (n-k)/4

For k=0: n/4 = 100 (but we computed overlap_0 = n/2 - ∑h[i]² = 200 - 200 = 0 for binary!)

Wait, for binary h, overlap_0 = ∑h[i](1-h[i]) = 0 since each h[i] is either 0 or 1. So k=0 is automatically optimal for binary!

For k=1: overlap_1 = number of adjacent 1-0 or 0-1 transitions... no wait.
overlap_1 = ∑_{i=0}^{n-2} h[i](1-h[i+1])

For binary h, this counts the number of positions where h[i]=1 and h[i+1]=0. This is exactly half the number of transitions (each 1→0 transition is counted, and each 0→1 transition is counted in overlap_1 for the reverse direction... no, overlap_1 only counts 1→0).

Actually, the total number of transitions in a binary sequence of length n with 200 ones and 200 zeros is at most 400 (alternating), and overlap_1 counts the 1→0 transitions specifically. If the sequence alternates perfectly, overlap_1 = 200.

To minimize overlap_1, we want to minimize the number of 1→0 transitions. This means having long blocks of 1s and long blocks of 0s.

But we need to do well for ALL k simultaneously. The trade-off is:
- Long blocks minimize small-k overlaps (few transitions)
- But long blocks create large overlaps at k = block_length (the overlap between a block of 1s and the adjacent block of 0s)

This is the fundamental tension in the problem!

**Optimal structure hypothesis**: The optimal h might consist of blocks of specific lengths that balance the trade-off.

For example, if h consists of:
- Block of a ones, block of b zeros, block of a ones, block of b zeros, ...
- Total: 200 ones, 200 zeros

Then for k < a: overlap_k counts the transitions at block boundaries
For k ≈ a: overlap_k is large (overlap of ones with zeros at the next block)
For k ≈ a+b: overlap_k is large again (overlap of ones with zeros at the block after next)

The minimum of max_k overlap_k would be achieved when the "peaks" at k=a, k=a+b, k=2a+b, etc. are all equal and as low as possible.

This is getting complex. Let me just focus on practical numerical optimization.

**My final approach**:

1. Start from initial_h_values
2. Use a combination of:
   - SLSQP with log-sum-exp (to handle the max smoothly)
   - Subgradient methods focusing on the single worst k
   - Momentum-based reweighted descent
   - Periodic "restart" with perturbation to escape local optima

3. Key improvements over previous:
   - More aggressive adaptive learning rate (backtracking line search)
   - Better active set management (recompute periodically)
   - Add a "perturbation + re-optimize" step to escape local optima
   - Use multiple alpha values in SLSQP (annealing)

Let me also add a **binary search on threshold** approach:
- For a given threshold τ (in overlap units, not C₅ units), define:
  f_τ(h) = ∑_k max(overlap_k - τ, 0)²
- Minimizing f_τ pushes all overlaps below τ
- If we can find h with f_τ(h) = 0, then C₅ ≤ τ * dx
- Binary search on τ to find the minimum achievable value

This is a smooth objective (C²) that's easier to optimize than the max!

Let me implement this:

```python
def threshold_obj(h, n, tau):
    """Sum of squared excesses above threshold"""
    over = compute_corr_overlaps_full(h)
    excess = np.maximum(over - tau, 0.0)
    return np.sum(excess**2)

def threshold_grad(h, n, tau):
    """Gradient of threshold objective"""
    over = compute_corr_overlaps_full(h)
    excess = np.maximum(over - tau, 0.0)
    active = excess > 0
    grad = np.zeros(n)
    for k in range(n):
        if active[k]:
            grad += 2.0 * excess[k] * grad_for_k(h, k)
    return grad
```

This is great because:
1. The objective is smooth (C²)
2. The gradient is well-defined
3. We can use L-BFGS or other quasi-Newton methods
4. Binary search on τ gives a certificate

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr_overlaps_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    return grad

def threshold_solver(h, n, target, tau, maxiter, deadline=None):
    """Solve min sum(max(overlap_k - tau, 0)^2) using L-BFGS"""
    dx = 2.0 / n
    
    def obj(x):
        over = compute_corr_overlaps_full(x)
        excess = np.maximum(over - tau, 0.0)
        return float(np.sum(excess**2))
    
    def grad(x):
        over = compute_corr_overlaps_full(x)
        excess = np.maximum(over - tau, 0.0)
        g = np.zeros(n)
        for k in range(n):
            if excess[k] > 0:
                g += 2.0 * excess[k] * grad_for_k(x, k)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h.copy()

def binary_search_tau(h, n, target, tau_lo, tau_hi, n_iters, deadline=None):
    """Binary search on threshold to find minimum achievable max overlap"""
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        tau = (tau_lo + tau_hi) / 2.0
        h_new = threshold_solver(h, n, target, tau, maxiter=500, deadline=deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        over = compute_corr_overlaps_full(h_new)
        if np.max(over) <= tau:
            tau_hi = tau
        else:
            tau_lo = tau
    
    return best_h, best_score

def slsqp_lse(h, n, target, alpha, maxiter, ftol, deadline=None):
    """SLSQP with log-sum-exp objective"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    try:
        res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new
    except Exception:
        pass
    return h_cur

def subgrad_single(h, n, target, n_steps, lr_init, deadline=None):
    """Subgradient descent on single worst k"""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    lr = lr_init
    
    for step in range(n_steps):

        lr_cur = lr
        improved = False

                cur_score = sc_new
                improved = True
                break
            lr_cur *= 0.6
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return h_cur, cur_score

def momentum_rw(h, n, target, n_steps, beta, lr, momentum, deadline=None):
    """Momentum reweighted gradient descent"""
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        over = compute_corr_overlaps_full(h_cur)
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(n):
            if weights[k] < 1e-8:
                continue
            grad += weights[k] * grad_for_k(h_cur, k)
        grad *= dx
        
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            v *= 0.85
    
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr_overlaps_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    return grad


def slsqp_lse(h, n, target, alpha, maxiter, ftol, deadline=None):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        grad = np.zeros(n)
        for k in range(n):
            if weights[k] > 1e-12:
                grad += weights[k] * grad_for_k(x, k)
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    try:
        res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < cur_score:
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                return h_new
    except Exception:
        pass
    return h_cur


def threshold_proj_grad(h, n, target, tau, n_steps, lr_init, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        over = compute_corr_overlaps_full(h_cur)
        excess = np.maximum(over - tau, 0.0)
        n_active = int(np.sum(excess > 0))
        if n_active == 0:
            break
        grad = np.zeros(n)
        for k in range(n):
            if excess[k] > 0:
                grad += 2.0 * excess[k] * grad_for_k(h_cur, k)
        grad *= dx
        lr_cur = lr
        improved = False
        for attempt in range(15):
            h_new = h_cur - lr_cur * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc_new
                improved = True
                break
            lr_cur *= 0.6
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h_cur, cur_score


def momentum_rw(h, n, target, n_steps, beta, lr, momentum, deadline=None):
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        over = compute_corr_overlaps_full(h_cur)
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum
        grad = np.zeros(n)
        for k in range(n):
            if weights[k] > 1e-10:
                grad += weights[k] * grad_for_k(h_cur, k)
        grad *= dx
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
        else:
            v *= 0.85
    return h_cur, cur_score


def subgrad_topk(h, n, target, n_steps, top_k, lr_init, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        over = compute_corr_overlaps_full(h_cur)
        top_idx = np.argsort(over)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            grad += grad_for_k(h_cur, k)
        grad *= dx / top_k
        lr_cur = lr
        improved = False
        for attempt in range(15):
            h_new = h_cur - lr_cur * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc_new
                improved = True
                break
            lr_cur *= 0.6
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
        else:
            lr = min(lr * 1.03, 0.05)
    return h_cur, cur_score


def binary_search_tau(h, n, target, tau_lo, tau_hi, n_iters, deadline=None):
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(n_iters):

        over = compute_corr_overlaps_full(h_new)
        if float(np.max(over)) <= tau:
            tau_hi = tau
        else:
            tau_lo = tau
    return best_h, best_score
