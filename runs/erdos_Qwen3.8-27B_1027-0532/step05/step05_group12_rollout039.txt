import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_and_grad(h, alpha, top_m=None):
    n = len(h)
    dx = 2.0 / n
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    total_lags = 2 * n - 1
    if top_m is not None and top_m < total_lags:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(total_lags)
        sub_corr = corr
    c_max = np.max(sub_corr)
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        val = c_max * dx
        weights = np.zeros(len(sub_corr))
    else:
        val = (c_max + np.log(wsum) / alpha) * dx
        weights = exp_shifted / wsum
    grad = np.zeros(n)
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return val, grad

def lbfgs_optimize(h_start, n, target, alpha, top_m, penalty, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        val, grad_c5 = smooth_c5_and_grad(x, alpha, top_m)
        diff = np.sum(x) - target
        val_p = val + 0.5 * penalty * diff * diff
        grad_p = grad_c5 + penalty * diff
        return val_p, grad_p
    try:
        result = minimize(obj, h_start, method='L-BFGS-B', jac=True, bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
        h_new = project_feasible(np.asarray(result.x), target)
        return h_new
    except Exception:
        return h_start

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def greedy_flip_improve(h, n, target, max_flips=500, time_limit_s=30, start_time=None):
    if start_time is None:

        if time.time() - start_time > time_limit_s:
            break
        improved = False
        best_flip_idx = -1

            h_try[i] = 1.0 - h_try[i]

            h_cur = project_feasible(h_cur, target)
            best_score = best_flip_val
            improved = True
            flips_done += 1
    return h_cur, best_score
