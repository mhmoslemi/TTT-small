    n = len(h)

    dc = np.zeros(2*n - 1)
    nm1 = n - 1
    
    # Flip i: 0 -> 1 (delta = +1)

    hi1 = 2*nm1 - i

    # For k in [lo1, hi1]: h[k - lo1] ... wait

    best_h = h.copy()
    best_max = cur_max
