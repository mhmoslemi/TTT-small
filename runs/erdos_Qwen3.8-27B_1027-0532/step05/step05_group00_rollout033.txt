def compute_c5(h):
    n = len(h)

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, alpha, target_sum, penalty):
    """Log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    log_sum = float(np.log(sum_exp))
    
    # Objective = dx * (c_max + log_sum / alpha) + penalty * (sum - target)^2
    sum_diff = float(np.sum(h)) - target_sum
    obj = dx * (c_max + log_sum / alpha) + penalty * sum_diff * sum_diff
    
    # Gradient

from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))

    log_sum = float(np.log(sum_exp))
    sum_diff = float(np.sum(h)) - target
    obj = dx * (c_max + log_sum / alpha) + penalty * sum_diff * sum_diff
    p = exp_vals / sum_exp

        if k >= 0:

            grad[k:] -= w * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w * (1.0 - h[:nv])
            grad[:nv] -= w * h[kk:n]
    grad += 2.0 * penalty * sum_diff
    return obj, grad
