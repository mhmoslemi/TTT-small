def smooth_max_val_grad(h, alpha):

def bb_gradient_descent(h_init, n, target, time_limit, alpha_start=50.0, alpha_end=5000.0, n_alphas=20):
    start_t = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Schedule alphas
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        # Do several BB steps at this alpha
        h_prev = None
        g_prev = None
        s_prev = None
        y_prev = None
        step_size = 0.01
        
        for step in range(200):
            if time.time() - start_t > time_limit:
                break
            
            val, g = smooth_max_val_grad(h, alpha)
            # Project gradient to feasible direction
            g = g - np.mean(g)
            
            # BB step size
            if h_prev is not None and g_prev is not None:
                s = h - h_prev
                y = g - g_prev
                sy = np.dot(s, y)
                ss = np.dot(s, s)
                yy = np.dot(y, y)
                if sy > 1e-12:
                    bb_plus = ss / sy
                    bb_minus = sy / yy if yy > 1e-12 else bb_plus
                    step_size = 0.5 * (bb_plus + bb_minus)
                    step_size = np.clip(step_size, 1e-6, 0.1)
            
            # Try the step
            h_new = project_feasible(h - step_size * g, target)
            c5_new = compute_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h_prev = h
            g_prev = g
            h = h_new
            s_prev = h - h_prev
            y_prev = g  # will update next iteration
    
    return best_h, best_c5

def multiplicative_weights_opt(h_init, n, target, time_limit, eta_init=2.0):
