import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_lp(h):
    """Build the LP to find the steepest descent direction for min max corr."""
    n = len(h)
    corr = compute_corr(h)
    
    # Installation: for each j in 0..2n-2, compute grad_j
    # and form the constraint: grad_j · δ - t <= -corr[j]
    
    n_vars = n + 1  # δ[0..n-1], t
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n_vars))
    b = np.zeros(n_constraints)
    
    for j in range(2 * n - 1):
        k = j - (n - 1)
        if k >= 0:
            # corr[n-1+k]: grad[i] = (1-h[i-k]) for i >= k, minus h[i+k] for i < n-k
            if k == 0:
                A[j, :n] = 1.0 - h  # (1-h[i]) for all i
                A[j, :n] -= h  # minus h[i+k] = h[i] for all i
                # So A[j, :n] = 1 - 2*h[i]
            else:
                # (1-h[i-k]) for i in [k, n-1]
                A[j, k:n] += (1.0 - h[:n-k])
                # -h[i+k] for i in [0, n-1-k]
                A[j, 0:n-k] -= h[k:]
        else:
            m = -k
            # corr[n-1-m]: grad[i] = (1-h[i+m]) for i < n-m, minus h[i-m] for i >= m
            A[j, 0:n-m] += (1.0 - h[m:])
            A[j, m:n] -= h[:n-m]
        
        A[j, n] = -1.0  # coefficient of t
        b[j] = -corr[j]
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds for δ[i]: -h[i] ≤ δ[i] ≤ 1-h[i]
    # Bound for t: unbounded below (or we can set a lower bound of the current max)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((None, None))
    
    return c_obj, A, b, A_eq, b_eq, bounds, corr

def lp_descent_step(h, max_backtrack=20, backtrack_factor=0.5):
    """One step of LP-based descent."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    current_c5 = compute_c5(h)
    
    c_obj, A, b, A_eq, b_eq, bounds, corr = build_lp(h)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return h, current_c5
    
    delta = res.x[:n]
    
    # Check if delta is meaningful
    if np.max(np.abs(delta)) < 1e-15:
        return h, current_c5
    
    # Backtracking line search
    for bt in range(max_backtrack):
        alpha = backtrack_factor ** bt
        h_new = h + alpha * delta
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < current_c5 - 1e-14:
            return h_new, c5_new
        if alpha < 0.01:
            break
    
    # If no improvement found, try a smaller fixed step
    alpha = 0.01
    h_new = project_feasible(h + alpha * delta, target)
    c5_new = compute_c5(h_new)
    if c5_new < current_c5:
        return h_new, c5_new
    
    return h, current_c5

def subgradient_step(h, n_active=5, temp=0.01):
    """One subgradient step on the max of corr."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = compute_corr(h)
    
    # Find top n_active lags
    sorted_idx = np.argsort(corr)[::-1][:n_active]
    max_val = corr[sorted_idx[0]]
    diffs = corr[sorted_idx] - max_val
    weights = np.exp(diffs / temp)
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for w, j in zip(weights, sorted_idx):
        k = int(j) - (n - 1)
        if k >= 0:
            if k == 0:
                grad[:] += w * dx * (1.0 - 2.0 * h)
            else:
                grad[k:] += w * dx * (1.0 - h[:n-k])
                grad[:n-k] -= w * dx * h[k:]
        else:
            m = -k
            grad[:n-m] += w * dx * (1.0 - h[m:])
            grad[m:] -= w * dx * h[:n-m]
    
    return grad

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_and_solve_lp(h):
    """Solve the LP for steepest descent direction. Returns (delta, t_opt) or (None, None)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_entries = 2 * n - 1
    
    # Build constraint matrix
    # Variables: delta[0..n-1], t
    # Minimize t
    # Subject to: for each j: grad_j · delta - t <= -corr[j]
    #             sum(delta) = 0
    #             -h[i] <= delta[i] <= 1-h[i]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Build A_ub (n_entries x (n+1)) and b_ub (n_entries)
    # For efficiency, build row by row using numpy operations
    A_ub = np.zeros((n_entries, n + 1))
    b_ub = -corr
    
    for j in range(n_entries):
        k = j - (n - 1)
        if k >= 0:
            # grad[i] = (1-h[i-k]) for i >= k, minus h[i+k] for i < n-k
            if k == 0:
                A_ub[j, :n] = 1.0 - 2.0 * h
            else:
                # (1-h[i-k]) for i in [k, n-1]
                A_ub[j, k:] += (1.0 - h[:n-k])
                # -h[i+k] for i in [0, n-1-k]
                A_ub[j, :n-k] -= h[k:]
        else:
            m = -k
            # grad[i] = (1-h[i+m]) for i < n-m, minus h[i-m] for i >= m
            A_ub[j, :n-m] += (1.0 - h[m:])
            A_ub[j, m:] -= h[:n-m]
        A_ub[j, n] = -1.0
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lower = -h.copy()
    upper = 1.0 - h.copy()
    bounds = list(zip(lower, upper))
    bounds.append((None, None))  # t is unbounded
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n], corr
    else:
        return None, None, corr
