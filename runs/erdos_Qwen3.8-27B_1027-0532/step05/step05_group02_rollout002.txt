    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]

    if abs(diff) < 1e-14:

    else:
        idx = np.where(h > 0.0)[0]

        return h_r.astype(float)

    zeros = np.where(h_r == 0)[0]

        j_end2 = min(n, n + k)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w * h[j_start2-k:j_end2-k]
    return grad

def proj_grad_descent(h_start, alpha_init, alpha_final, max_iters, lr_init, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stall = 0
    lr_reduce_streak = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 80:
            break
            
        ratio = it / max(max_iters - 1, 1)
        alpha = alpha_init * (alpha_final / alpha_init) ** ratio
        
        grad = continuous_gradient(h, alpha, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
            
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            stall = 0
            lr_reduce_streak = 0
        else:
            stall += 1
            lr_reduce_streak += 1
            if lr_reduce_streak > 8:
                lr *= 0.7
                lr_reduce_streak = 0
        
        h = h_new
        
        if it % 20 == 19:
            h = project_feasible(h, target)
    
    return best_h, best_score

def binary_greedy(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        
        if not found:
            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    
    return best_h, best_c5

def binary_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 50000:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if it % 7 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def block_swap_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 30000:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        if it % 3 == 0:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            block_size = rng.randint(1, min(4, n // 50 + 1))
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < block_size or len(zeros) < block_size:
                stall += 1
                continue
            i_start = ones[rng.randint(len(ones) - block_size + 1)]
            i_block = np.where(np.arange(n) >= i_start)[0][:block_size]
            valid_one_block = i_block[i_block < n]
            if len(valid_one_block) < block_size:
                stall += 1
                continue
            if not np.all(h[valid_one_block[:block_size]] == 1.0):
                stall += 1
                continue
            j_start = zeros[rng.randint(len(zeros) - block_size + 1)]
            j_block = np.arange(j_start, min(j_start + block_size, n))
            if len(j_block) < block_size or not np.all(h[j_block] == 0.0):
                stall += 1
                continue
            
            delta = np.zeros(2*n-1)
            for idx_i in valid_one_block[:block_size]:
                for idx_j in j_block[:block_size]:
                    delta += compute_delta(h, idx_i, idx_j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / (t * block_size))):
                for idx_i in valid_one_block[:block_size]:
                    h[idx_i] = 0.0
                for idx_j in j_block[:block_size]:
                    h[idx_j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def multi_start_binary(n, seed, rng, time_limit, start_time):
    target = n // 2
    best_h = None
    best_score = 1.0
    
    strategies = []
    
    h_rand = np.zeros(n)
    perm = rng.permutation(n)
    h_rand[perm[:target]] = 1.0
    strategies.append(('random', h_rand))
    
    h_block = np.zeros(n)
    block_len = max(1, n // 20)
    for i in range(0, n, 2 * block_len):
        end = min(i + block_len, n)
        h_block[i:end] = 1.0
    h_block = project_binary(h_block, target)
    strategies.append(('block', h_block))
    
    h_periodic = np.zeros(n)
    period = max(2, n // 50)
    for i in range(n):
        if (i // period) % 2 == 0:
            h_periodic[i] = 1.0
    h_periodic = project_binary(h_periodic, target)
    strategies.append(('periodic', h_periodic))
    
    for name, h_start in strategies:
        if time.time() - start_time > time_limit:
            break
        h_g, c5_g = binary_greedy(h_start, 5000, rng, time_limit=30, start_time=start_time)
        h_s, c5_s = binary_sa(h_g, 100000, 0.02, 0.001, rng, time_limit=30, start_time=start_time)
        if c5_s < best_score:
            best_score = c5_s
            best_h = h_s.copy()
    
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    
    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick binary greedy to polish parent
    allowed = min(40, tleft() * 0.10)
    h_g, c5_g = binary_greedy(best_h, 50000, rng, time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()
    
    # Phase 2: Continuous gradient descent (main new component)
    if tleft() > 60:
        allowed = min(120, tleft() * 0.20)
        max_iters = int(min(2000, allowed * 50))
        h_cont, c5_cont = proj_grad_descent(best_h, 10.0, 5000.0, max_iters, 0.02,
                                            time_limit=allowed, start_time=start_time)
        if c5_cont < best_score:
            best_score = c5_cont
            best_h = h_cont.copy()
        
        # Project to binary and refine
        h_bin = project_binary(h_cont, target_init)
        h_gr, c5_gr = binary_greedy(h_bin, 30000, rng, time_limit=min(20, tleft()*0.08), start_time=start_time)
        if c5_gr < best_score:
            best_score = c5_gr
            best_h = h_gr.copy()
    
    # Phase 3: Focused SA
    if tleft() > 80:
        allowed = min(150, tleft() * 0.20)
        n_sa = int(min(3000000, allowed * 40000))
        h_sa, c5_sa = binary_sa(best_h, n_sa, 0.015, 0.00005, rng,
                                time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 4: Iterate gradient descent + binary refinement
    for round_i in range(5):
        if tleft() < 100:
            break
        
        # Continuous gradient descent
        allowed = min(80, tleft() * 0.12)
        max_iters = int(min(1500, allowed * 40))
        h_cont2, c5_cont2 = proj_grad_descent(best_h, 5.0, 2000.0, max_iters, 0.015,
                                              time_limit=allowed, start_time=start_time)
        
        # Binary projection and SA
        h_bin2 = project_binary(h_cont2, target_init)
        allowed2 = min(30, tleft() * 0.05)
        h_gr2, c5_gr2 = binary_greedy(h_bin2, 20000, rng, time_limit=allowed2, start_time=start_time)
        allowed3 = min(40, tleft() * 0.06)
        n_sa2 = int(min(500000, allowed3 * 30000))
        h_sa2, c5_sa2 = binary_sa(h_gr2, n_sa2, 0.01, 0.0001, rng,
                                  time_limit=allowed3, start_time=start_time)
        if c5_sa2 < c5_gr2:
            h_try = h_sa2
            c5_try = c5_sa2
        else:
            h_try = h_gr2
            c5_try = c5_gr2
        
        if c5_cont2 < best_score:
            best_score = c5_cont2
            best_h = h_cont2.copy()
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()
    
    # Phase 5: Block swap SA
    if tleft() > 80:
        allowed = min(80, tleft() * 0.10)
        n_bs = int(min(200000, allowed * 20000))
        h_bs, c5_bs = block_swap_sa(best_h, n_bs, 0.01, 0.001, rng,
                                    time_limit=allowed, start_time=start_time)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs.copy()
    
    # Phase 6: Multi-start at different resolutions
    for n_try in [n_init + 4, n_init + 8, n_init - 4]:
        if tleft() < 80:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, n_init)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = project_binary(h_interp, target_try)
        
        allowed = min(30, tleft() * 0.06)
        h_g3, c5_g3 = binary_greedy(h_bin_try, 10000, rng, time_limit=allowed * 0.5, start_time=start_time)
        allowed2 = min(25, tleft() * 0.04)
        n_sa3 = int(min(200000, allowed2 * 20000))
        h_s3, c5_s3 = binary_sa(h_g3, n_sa3, 0.01, 0.0001, rng,
                                time_limit=allowed2, start_time=start_time)
        
        if c5_s3 < c5_g3:
            h_try = h_s3
            c5_try = c5_s3
        else:
            h_try = h_g3
            c5_try = c5_g3
        
        # Also try gradient descent at this resolution
        if tleft() > 50:
            allowed_gd = min(30, tleft() * 0.04)
            h_gd, c5_gd = proj_grad_descent(h_try, 5.0, 1000.0, 500, 0.01,
                                            time_limit=allowed_gd, start_time=start_time)
            h_gd_bin = project_binary(h_gd, target_try)
            c5_gd_bin = compute_c5(h_gd_bin)
            if c5_gd_bin < c5_try:
                c5_try = c5_gd_bin
                h_try = h_gd_bin
        
        if c5_try < best_score:
            best_score = c5_try
            # Resample back to n_init
            x_new = np.linspace(0.0, 1.0, n_init)
            x_old = np.linspace(0.0, 1.0, n_try)
            h_resampled = np.interp(x_new, x_old, h_try)
            best_h = project_binary(h_resampled, target_init)
            
            # Quick greedy at native resolution
            if tleft() > 30:
                h_gf, c5_gf = binary_greedy(best_h, 15000, rng, time_limit=min(15, tleft()*0.03), start_time=start_time)
                if c5_gf < best_score:
                    best_score = c5_gf
                    best_h = h_gf.copy()
    
    # Phase 7: Final gradient descent burst
    if tleft() > 50:
        allowed = min(tleft() - 20, 80)
        h_cont_f, c5_cont_f = proj_grad_descent(best_h, 3.0, 3000.0, 1000, 0.01,
                                                time_limit=allowed, start_time=start_time)
        if c5_cont_f < best_score:
            best_score = c5_cont_f
            best_h = h_cont_f.copy()
    
    # Phase 8: Final binary SA polish
    if tleft() > 30:
        h_bin_final = project_binary(best_h, target_init)
        n_sa_f = int(min(500000, max(50000, (tleft() - 15) * 30000)))
        h_sa_f, c5_f = binary_sa(h_bin_final, n_sa_f, 0.005, 0.00001,
                                  rng, time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()
    
    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    # Also check binary projection
    h_bin_check = project_binary(best_h, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_check)
    if bin_score < final_score:
        h_final = h_bin_check
        final_score = bin_score
    
    return h_final, final_score, n_final
