def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def compute_delta(h, i, j):

    result = np.zeros(2 * n - 1)

    idx = i + k

    result[mask] += h[idx[mask]]
    idx = i - k

    curr_score = compute_c5(h_cur)
    alphas_full = [30, 80, 200, 500]
    for alpha in alphas_full:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    alphas_top = [200, 500, 1500, 5000]
    for alpha in alphas_top:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj_top, h_cur, args=(alpha, 8),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2.0)

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast greedy
    allowed1 = min(50, tleft() * 0.06)
    h_g, c5_g = greedy_topk(best_h, max_iters=30000, rng=rng,
                            time_limit=allowed1, start_time=start_time, top_k=5, cand_size=25)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Main adaptive SA with diversification
    if tleft() > 80:
        n_rounds = 14
        sa_time = tleft() * 0.50
        per_round = sa_time / n_rounds
        h_cur = best_h.copy()
        for r in range(n_rounds):
            if tleft() < 60:
                break
            t_init = 0.015 + 0.01 * rng.random()
            n_iters = int(min(800000, per_round * 25000))
            h_sa, c5_sa = adaptive_sa(h_cur, n_iters, t_init, rng,
                                      per_round * 0.85, time.time(),
                                      target_acc=0.45, focused_prob=0.35)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()
            h_cur = h_sa.copy()
            # Diversify
            n_pert = rng.randint(8, 20)
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                ones = np.where(h_cur == 1.0)[0]
                zeros = np.where(h_cur == 0.0)[0]
            # Quick greedy after perturbation
            if tleft() > 30:
                h_g2, c5_g2 = greedy_topk(h_cur, max_iters=5000, rng=rng,
                                          time_limit=per_round * 0.15, start_time=time.time(),
                                          top_k=3, cand_size=15)
                if c5_g2 < best_score:
                    best_score = c5_g2
                    best_h = h_g2.copy()
                h_cur = h_g2

    # Phase 3: Greedy polish
    if tleft() > 50:
        allowed3 = min(35, tleft() * 0.08)
        h_g3, c5_g3 = greedy_topk(best_h, max_iters=40000, rng=rng,
                                  time_limit=allowed3, start_time=start_time,
                                  top_k=6, cand_size=30)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

    # Phase 4: Continuous SLSQP
    if tleft() > 60:
        target_use = float(n) / 2.0
        allowed4 = min(tleft() - 30, 150)
        h_ref, sc = slsqp_refine(best_h, n, target_use, allowed4, time.time())
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Binary SA on best
    if tleft() > 30:
        h_bin = project_binary(best_h, target)
        if tleft() > 35:
            allowed5 = max(5, tleft() - 15)
            h_sa5, c5_5 = adaptive_sa(h_bin, 5000000, 0.008, rng,
                                      allowed5, time.time(),
                                      target_acc=0.50, focused_prob=0.4)
            if c5_5 < best_score:
                best_score = c5_5
                best_h = h_sa5

    # Phase 6: Final SLSQP squeeze
    if tleft() > 20:
        target_use = float(n) / 2.0
        allowed6 = max(5, tleft() - 10)
        h_ref6, sc6 = slsqp_refine(best_h, n, target_use, allowed6, time.time())
        if sc6 < best_score:
            best_score = sc6
            best_h = h_ref6

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
