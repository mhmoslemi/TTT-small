```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    """Project h onto {x : 0<=x<=1, sum(x)=target}"""
    h = h.astype(float).copy()
    n = len(h)
    # Bisection on the shift
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha, active_lags=None):
    """Compute LSE objective and gradient over active lags (or all lags)."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    corr = corr_raw * dx
    
    if active_lags is not None:
        corr_active = corr[active_lags]
        c_max = corr_active.max()
        shifted = alpha * (corr_active - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = (exp_shifted / sum_exp).astype(float)
        obj = (c_max + np.log(sum_exp) / alpha)
        
        # Gradient: for each active lag k with weight w_k
        # d(corr[k])/d(h[i]) = dx * (1-h[i+k]) if i < n-k (i is first index)
        #                    - dx * h[i-k] if i >= k (i is second index, i.e., i-k+k=i)
        grad = np.zeros(n)
        for wi, ki in zip(weights, active_lags):
            ki = int(ki)
            # Term from h[i] as first factor: (1-h[i+ki]) for i = 0..n-1-ki
            jp = np.arange(0, n - ki)
            grad[jp] += wi * dx * (1.0 - h[jp + ki])
            # Term from h[i] as second factor: -h[i-ki] for i = ki..n-1
            jn = np.arange(ki, n)
            grad[jn] -= wi * dx * h[jn - ki]
    else:
        c_max = float(corr.max())
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        obj = (c_max + np.log(sum_exp) / alpha)
        
        # FFT-based gradient for all lags
        w_flipped = weights[::-1].copy()
        fsize = 2 * n
        Wf = np.fft.rfft(w_flipped, fsize)
        V = np.fft.rfft(1.0 - h, fsize)
        W = np.fft.rfft(weights, fsize)
        H = np.fft.rfft(h, fsize)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad = dx * (corr_pos - corr_neg)
    
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, active_lags=None):
    n = len(h0)
    x = project_feasible(h0.copy(), target)
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    obj0, grad0 = lse_obj_and_grad(x, alpha, active_lags)
    L0 = 0.0
    for i in range(n):
        L0 = max(L0, abs(grad0[i]))
    L0 = max(L0, 1e-8)
    step = 1.0 / (alpha * 0.01 + L0)
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha, active_lags)
        
        # Backtracking line search
        step_trial = step
        x_new = project_feasible(y_clipped - step_trial * grad_y, target)
        for _ in range(50):
            obj_new, _ = lse_obj_and_grad(x_new, alpha, active_lags)
            linear_decrease = obj_y + np.dot(grad_y, x_new - y_clipped)
            if obj_new <= linear_decrease + 1e-4 * step_trial * np.dot(grad_y, grad_y):
                break
            step_trial *= 0.5
            x_new = project_feasible(y_clipped - step_trial * grad_y, target)
        else:
            step_trial = max(step_trial, 1e-15)
            x_new = project_feasible(y_clipped - step_trial * grad_y, target)
        
        step = min(step_trial * 1.5, 1.0 / max(L0, 1e-15))
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

    n = len(h)

    best_score = compute_c5(h_cur)
    
    alphas = [alpha_start]

            best_score = sc
            h_cur = h_new
    
    return h_cur, best_score

def coordinate_descent(h, target, deadline, max_sweeps=500):
    """Coordinate descent with multiple trial values."""
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(777)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            # Try several values
            trials = []
            if old_val < 0.05:
                trials = [0.0, 0.01, 0.02, 0.05]
            elif old_val > 0.95:
                trials = [1.0, 0.99, 0.98, 0.95]
            else:
                step_size = 0.1
                trials = [old_val + step_size, old_val - step_size,
                         old_val + 0.5*step_size, old_val - 0.5*step_size,
                         0.0, 1.0]
                trials = [max(0, min(1, t)) for t in trials]
            
            best_trial = None
            best_trial_val = old_val
            best_trial_score = cur_score
            for trial in trials:
                if abs(trial - old_val) < 1e-10:
                    continue
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = trial
                    best_trial_val = trial
            
            if best_trial is not None:
                h_cur[idx] = best_trial
                h_cur = project_feasible(h_cur, target)
                cur_score = best_trial_score
                improved = True
            else:
                h_cur[idx] = old_val
                h_cur = project_feasible(h_cur, target)
        
        if not improved:
            break
    
    return h_cur, cur_score
```

coeff_k = -h[i-k] for k in (n-1-i, i] (if i > n-1-i)

This gives us: new_corr[k] = old_corr[k] + d * coeff_k for all k.

We want to choose d (so that v' = v + d is in [0,1]) to minimize max_k new_corr[k].

This is a 1D optimization problem! We can solve it by:
1. Computing the function f(d) = max_k (old_corr[k] + d * coeff_k)
2. This is the maximum of linear functions of d, so it's convex in d
3. We can find the minimizer by ternary search or by finding where the two "active" lines cross

This is much more efficient than evaluating the full correlation for each trial value!

```python
def optimal_coord_change(corr, i, h):
    """Find optimal value for h[i] given current corr vector."""
    n = len(h)
    v = h[i]
    
    # Compute coeff_k for each k
    max_k_first = n - 1 - i  # k where first factor applies: 0 to max_k_first
    max_k_second = i  # k where second factor applies: 0 to max_k_second
    
    coeff = np.zeros(n)
    overlap_end = min(max_k_first, max_k_second)
    
    # Overlap region: both factors
    for k in range(overlap_end + 1):
        coeff[k] = (1.0 - h[i + k]) - h[i - k]
    
    # Only first factor (if any)
    for k in range(overlap_end + 1, max_k_first + 1):
        coeff[k] = (1.0 - h[i + k])
    
    # Only second factor (if any)
    for k in range(overlap_end + 1, max_k_second + 1):
        coeff[k] = -h[i - k]
    
    # Now minimize max_k (corr[k] + d * coeff[k]) over d in [-v, 1-v]
    # This is convex in d, so use ternary search
    lo_d = -v
    hi_d = 1.0 - v
    
    # Ternary search
    for _ in range(50):
        m1 = lo_d + (hi_d - lo_d) / 3
        m2 = hi_d - (hi_d - lo_d) / 3
        f1 = np.max(corr + m1 * coeff)
        f2 = np.max(corr + m2 * coeff)
        if f1 < f2:
            hi_d = m2
        else:
            lo_d = m1
    
    d_opt = (lo_d + hi_d) / 2
    v_opt = v + d_opt
    return v_opt
```

Actually, the ternary search can be replaced with finding the vertex more directly. Since f(d) = max of linear functions, it's piecewise linear and convex. The minimum is where the slope changes from negative to positive. We can find this by:
1. Computing all the lines: y_k(d) = corr[k] + d * coeff[k]
2. Finding the lower envelope... no, the upper envelope (since we want min of max)
3. The minimum of the upper envelope is at a vertex where two lines intersect

But with n=400 lines, finding the vertex of the upper envelope is O(n log n) using a convex hull trick, which is more complex. For n=400, ternary search is fine.

Actually, even simpler: since the function is convex and piecewise linear, we can binary search on the value. For a given target t, check if there exists d such that corr[k] + d*coeff[k] <= t for all k. This is a feasibility problem in 1D:
- For each k: if coeff[k] > 0, then d <= (t - corr[k]) / coeff[k]
- For each k: if coeff[k] < 0, then d >= (t - corr[k]) / coeff[k]
- For each k: if coeff[k] = 0, need corr[k] <= t

So the feasible range for d is [max_{k: coeff[k]<0} (t-corr[k])/coeff[k], min_{k: coeff[k]>0} (t-corr[k])/coeff[k]]

And we need this range to be non-empty and intersect with [-v, 1-v].

We can binary search on t to find the minimum feasible t. This is O(n) per binary search step, O(log(precision)) steps, so O(n * log(precision)) total.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_active(h, alpha, active):
    """LSE over active lags, returns (obj, grad)."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    corr = corr_raw * dx
    corr_a = corr[active]
    c_max = corr_a.max()
    shifted = alpha * (corr_a - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(exp_s.sum())
    w = exp_s / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for wi, ki in zip(w, active):
        ki = int(ki)
        jp = np.arange(0, n - ki)
        grad[jp] += wi * dx * (1.0 - h[jp + ki])
        jn = np.arange(ki, n)
        grad[jn] -= wi * dx * h[jn - ki]
    return obj, grad

def fista_active(h0, target, alpha, active, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0.copy(), target)
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    step = 0.01
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        
        obj_y, grad_y = lse_active(y, alpha, active)
        
        # Backtracking
        s = step
        x_new = project_feasible(y - s * grad_y, target)
        for _ in range(30):
            obj_new, _ = lse_active(x_new, alpha, active)
            if obj_new <= obj_y + np.dot(grad_y, x_new - y) + 1e-4 * s * np.dot(grad_y, grad_y):
                break
            s *= 0.5
            x_new = project_feasible(y - s * grad_y, target)
        
        step = min(s * 1.3, 0.1)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    
    return best_x, best_score

def get_active_set(corr, max_lags=10):
    sorted_idx = np.argsort(corr)[::-1]

    threshold = c_max - 0.005
    active = sorted_idx[corr[sorted_idx] >= threshold]
```

```python
def linprog_step(h, target, active, deadline):
    """Solve LP to find a descent direction."""
    n = len(h)
    from scipy.optimize import linprog
    
    corr = compute_corr(h)
    c_max = corr.max()
    
    # Linearize corr_k around h
    # corr_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
    # grad_corr_k[i] = dx * (1-h[i+k]) if i <= n-1-k (first factor)
    #                - dx * h[i-k] if i >= k (second factor)
    
    # Variables: delta[0], ..., delta[n-1], t
    # Minimize: t
    # Subject to:
    #   sum_i grad_corr_k[i] * delta[i] + corr[k] - t <= 0 for k in active
    #   -h[i] <= delta[i] <= 1-h[i] for i = 0,...,n-1
    #   sum(delta) = 0
    
    n_vars = n + 1  # delta[0..n-1] and t
    # c_i for objective: minimize t = c^T x
    c = np.zeros(n_vars)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k: sum_i grad_corr_k[i] * delta[i] - t <= -corr[k]
    A_rows = []
    b_vals = []
    for k in active:
        row = np.zeros(n_vars)
        ki = int(k)
        dx = 2.0 / n
        for i in range(n - ki):
            row[i] += dx * (1.0 - h[i + ki])
        for i in range(ki, n):
            row[i] -= dx * h[i - ki]
        row[n] = -1.0  # -t
        A_rows.append(row)
        b_vals.append(-corr[k] * 1.0)  # Wait, corr[k] already includes dx
    
    # Actually, corr[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
    # So the linearization is: corr[k] + sum_i grad_corr_k[i] * delta[i]
    # where grad_corr_k[i] = dx*(1-h[i+k]) - dx*h[i-k] (as appropriate)
    # And we want: corr[k] + sum_i grad_corr_k[i] * delta[i] <= t
    # i.e.: sum_i grad_corr_k[i] * delta[i] - t <= -corr[k]
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Equality constraint: sum(delta) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -h[i] <= delta[i] <= 1-h[i], t >= 0
    bounds = []
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    bounds.append((0.0, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[:n]
        # Line search along delta
        h_cur = h.copy()
        best_sc = compute_c5(h_cur)
        for alpha_ls in [0.1, 0.5, 1.0, 0.05, 0.2]:
            h_trial = h_cur + alpha_ls * delta
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < best_sc:
                best_sc = sc
                h_cur = h_trial
        return h_cur, best_sc
    return h, compute_c5(h)
```

```python
            break
        order = rng.permutation(n)
        sweep_improved = 0
        for idx in order:
            if time.time() > deadline:
                break
            v = h_cur[idx]
            max_kf = n - 1 - idx
            max_ks = idx
            overlap = min(max_kf, max_ks)

            coeff = np.zeros(n)
            if overlap >= 0:
                kr = np.arange(overlap + 1)
                coeff[kr] = (1.0 - h_cur[idx + kr]) - h_cur[idx - kr]
            if max_kf > overlap:
                kr2 = np.arange(overlap + 1, max_kf + 1)
                coeff[kr2] = 1.0 - h_cur[idx + kr2]
            if max_ks > overlap:
                kr3 = np.arange(overlap + 1, max_ks + 1)
                coeff[kr3] = -h_cur[idx - kr3]

            lo_d = -v
            hi_d = 1.0 - v
            if hi_d - lo_d < 1e-12:
                continue

            best_val = float(np.max(corr))
            for _ in range(35):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(corr + m1 * coeff))
                f2 = float(np.max(corr + m2 * coeff))
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1

            d_opt = (lo_d + hi_d) / 2.0
            f_opt = float(np.max(corr + d_opt * coeff))
            if f_opt >= best_val - 1e-10:
                continue

            h_save = h_cur.copy()
            corr_save = corr.copy()
            h_cur[idx] = v + d_opt
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            sc = float(np.max(corr))
            if sc < cur_score - 1e-12:
                cur_score = sc
                sweep_improved += 1
            else:
                h_cur = h_save
                corr = corr_save

        if sweep_improved == 0:
            break
    return h_cur, cur_score

def get_active_set(corr, max_lags=10, gap=0.005):
    sorted_idx = np.argsort(corr)[::-1]
    c_max = float(corr[sorted_idx[0]])
    threshold = c_max - gap
    active = sorted_idx[corr[sorted_idx] >= threshold]
    if len(active) > max_lags:
        active = sorted_idx[:max_lags]
    return active

def linprog_step(h, target, active):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    A_rows = []
    b_vals = []
    for ki in active:
        ki = int(ki)
        row = np.zeros(n_vars)
        for i in range(n - ki):
            row[i] += dx * (1.0 - h[i + ki])
        for i in range(ki, n):
            row[i] -= dx * h[i - ki]
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(-float(corr[ki]))
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
    except Exception:
        return h, compute_c5(h)
    if res.success:
        delta = res.x[:n]
        h_cur = h.copy()
        best_sc = compute_c5(h_cur)
        for a_ls in [0.3, 0.1, 0.5, 0.05, 0.15, 1.0]:
            h_trial = project_feasible(h_cur + a_ls * delta, target)
            sc = compute_c5(h_trial)
            if sc < best_sc:
                best_sc = sc
                h_cur = h_trial
        return h_cur, best_sc
    return h, compute_c5(h)

def pair_swap(h, target, deadline):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(12345)
    n_trials = 8000
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(h_cur[i] - h_cur[j]) < 1e-6:
            continue
        vi = h_cur[i]
        vj = h_cur[j]
        h_cur[i] = vj
        h_cur[j] = vi
        sc = compute_c5(h_cur)
        if sc < cur_score - 1e-10:
            cur_score = sc
        else:
            h_cur[i] = vi
            h_cur[j] = vj
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 12.0
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: FISTA continuation with growing active set
    alphas_schedule = [100, 300, 1000, 3000, 8000, 20000, 50000]
    durations = [50, 45, 45, 50, 50, 50, 50]
    for i, (a, dur) in enumerate(zip(alphas_schedule, durations)):
        if tleft() < dur + 5:
            break
        corr = compute_corr(best_h)
        active = get_active_set(corr, max_lags=min(10, 5 + i), gap=0.008 - i * 0.0005)
        h_new, sc_new = fista_active(best_h, target, a, active, 50000, start_time + dur)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 2: Fast coordinate descent
    if tleft() > 120:
        h_new, sc_new = fast_coord_descent(best_h, target, start_time + min(110, tleft() - 5), max_sweeps=500)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: LP-based steps
    if tleft() > 100:
        lp_deadline = start_time + min(80, tleft() - 5)
        for _ in range(200):
            if time.time() > lp_deadline:
                break
            corr = compute_corr(best_h)
            active = get_active_set(corr, max_lags=8, gap=0.003)
            h_new, sc_new = linprog_step(best_h, target, active)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 4: Coordinate descent after LP
    if tleft() > 80:
        h_new, sc_new = fast_coord_descent(best_h, target, start_time + min(70, tleft() - 5), max_sweeps=400)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 5: More FISTA with high alpha
    if tleft() > 80:
        corr = compute_corr(best_h)
        active = get_active_set(corr, max_lags=6, gap=0.002)
        h_new, sc_new = fista_active(best_h, target, 80000, active, 30000, start_time + min(60, tleft() - 5))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        if tleft() > 30:
            h_new, sc_new = fast_coord_descent(best_h, target, start_time + min(25, tleft() - 5), max_sweeps=300)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 6: Perturbation restarts
    rng = np.random.RandomState(seed)
    if tleft() > 100:
        scales = [0.005, 0.01, 0.003, 0.015, 0.02, 0.008, 0.025]
        for s in scales:
            if tleft() < 35:
                break
            h_pert = best_h + rng.randn(n) * s
            h_pert = project_feasible(h_pert, target)
            corr = compute_corr(h_pert)
            active = get_active_set(corr, max_lags=10, gap=0.005)
            h_new, sc_new = fista_active(h_pert, target, 3000, active, 10000, start_time + 12)
            h_new2, sc_new2 = fast_coord_descent(h_new, target, start_time + 8, max_sweeps=200)
            if sc_new2 < best_score:
                best_score = sc_new2
                best_h = h_new2
            elif sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 7: Pair swap
    if tleft() > 50:
        h_new, sc_new = pair_swap(best_h, target, start_time + min(40, tleft() - 5))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 8: Final coordinate descent
    if tleft() > 30:
        h_new, sc_new = fast_coord_descent(best_h, target, start_time + min(25, tleft() - 3), max_sweeps=500)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 9: Final pair swap + coord
    if tleft() > 20:
        h_new, sc_new = pair_swap(best_h, target, start_time + 8)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        if tleft() > 10:
            h_new, sc_new = fast_coord_descent(best_h, target, start_time + 5, max_sweeps=200)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
