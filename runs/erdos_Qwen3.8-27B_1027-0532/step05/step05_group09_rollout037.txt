import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr_init is None:
        lr_init = 1.0 / max(alpha * 0.05, 1.0)
    step = lr_init
    last_x = x.copy()
    last_grad = None
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        
        # BB step size update
        if last_grad is not None:
            s_vec = x - last_x
            y_vec = grad_y - last_grad
            s_norm = np.dot(s_vec, s_vec)
            y_norm = np.dot(s_vec, y_vec)
            if y_norm > 1e-10:
                bb_step = s_norm / y_norm
                step = np.clip(bb_step, 0.1 * lr_init, 10.0 * lr_init)
        last_x = x.copy()
        last_grad = grad_y.copy()
        
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Backtracking
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            if obj_new <= obj_y - 1e-4 * step * np.dot(grad_y, x_new - y_clipped):
                break
            step *= (1.0 - shrink)
            shrink = min(shrink + 0.1, 0.92)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step / (1.0 - shrink), 10.0 * lr_init)
        
        # Momentum restart
        sc = compute_c5(x_new)
        if sc > best_obj + 0.01:
            t = 1.0  # restart momentum
        
        x_prev = x
        x = x_new
        t = t_new
        
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def exact_subgrad_refine(h, target, deadline, max_iters=10000):
    """Subgradient method on the exact max with backtracking."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best = compute_c5(h_cur)
    best_h = h_cur.copy()
    step = 0.05
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_max = float(np.max(corr))
        sc = c_max * dx
        
        # Find active lags
        tol = max(0.5 / n, 1e-6)
        active_mask = corr >= c_max - tol
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        
        # Weighted subgradient (softmax with small temp)
        diffs = corr[active_idx] - c_max
        temp = 0.3 / n
        w = np.exp(diffs / temp)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, active_idx):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        # Backtracking
        accepted = False
        for bt in range(40):
            h_try = project_feasible(h_cur - step * grad, target)
            sc_try = compute_c5(h_try)
            if sc_try < best - 1e-15:
                h_cur = h_try
                best = sc_try
                best_h = h_cur.copy()
                accepted = True
                break
            step *= 0.7
        if not accepted:
            step = max(step, 1e-6)
            h_try = project_feasible(h_cur - step * grad, target)
            sc_try = compute_c5(h_try)
            if sc_try < best:
                h_cur = h_try
                best = sc_try
                best_h = h_cur.copy()
        
        if accepted:
            step = min(step * 1.05, 0.5)
    
    return best_h, best

def stochastic_transfers(h, target, deadline, n_trials=500000):
    """Stochastic local search using pair transfers."""
    n = len(h)
    h_cur = h.copy()
    best = compute_c5(h_cur)
    best_h = h_cur.copy()
    rng = np.random.RandomState(777)
    no_improve = 0
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Try transferring mass from i to j or vice versa
        for direction in [1, -1]:
            hi = h_cur[i]
            hj = h_cur[j]
            if direction == 1:
                # transfer from i to j: h[i]-=d, h[j]+=d
                delta = min(hi, 1.0 - hj)
            else:
                # transfer from j to i: h[j]-=d, h[i]+=d
                delta = min(hj, 1.0 - hi)
            if delta < 1e-12:
                continue
            
            # Try a few fraction of delta
            for frac in [1.0, 0.5, 0.25]:
                d = delta * frac
                if d < 1e-14:
                    continue
                h_try = h_cur.copy()
                if direction == 1:
                    h_try[i] -= d
                    h_try[j] += d
                else:
                    h_try[j] -= d
                    h_try[i] += d
                # Check feasibility (sum is preserved, need to check bounds)
                if h_try[i] < -1e-12 or h_try[j] < -1e-12 or h_try[i] > 1+1e-12 or h_try[j] > 1+1e-12:
                    continue
                h_try = np.clip(h_try, 0, 1)
                sc = compute_c5(h_try)
                if sc < best - 1e-15:
                    h_cur = h_try
                    best = sc
                    best_h = h_cur.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
                if no_improve > 5000:
                    # Perturb to escape
                    pert = rng.randn(n) * 0.01
                    h_cur = project_feasible(h_cur + pert, target)
                    no_improve = 0
    
    return best_h, best

def block_optimize(h, target, deadline, block_size=10, n_rounds=500):
    """Optimize blocks of consecutive coordinates."""
    n = len(h)
    h_cur = h.copy()
    best = compute_c5(h_cur)
    best_h = h_cur.copy()
    rng = np.random.RandomState(999)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        
        # Pick a random block
        start = rng.randint(0, n - block_size + 1)
        end = start + block_size
        block = h_cur[start:end].copy()
        
        # Try to optimize the block
        def block_obj(x):
            h_test = h_cur.copy()
            h_test[start:end] = np.clip(x, 0, 1)
            sc = compute_c5(h_test)
            pen = np.sum(np.clip(-x, 0, None)**2) + np.sum(np.clip(x-1, 0, None)**2)
            # Penalty for sum deviation
            sum_dev = np.sum(h_test) - target
            pen += 100.0 * sum_dev**2
            return sc + pen
        
        res = minimize(block_obj, block, method='Nelder-Mead',
                      options={'maxiter': 50, 'xatol': 1e-6, 'fatol': 1e-8})
        
        h_test = h_cur.copy()
        h_test[start:end] = np.clip(res.x, 0, 1)
        h_test = project_feasible(h_test, target)
        sc = compute_c5(h_test)
        if sc < best - 1e-15:
            h_cur = h_test
            best = sc
            best_h = h_cur.copy()
    
    return best_h, best

def make_structural_h(n, seed=0):
    """Create diverse structural initializations."""
    rng = np.random.RandomState(seed)
    variants = []
    
    # Variant 1: Banded (high in middle)

    variants.append(h)
    
    # Variant 2: Two peaks
    h = np.exp(-((x - 0.25)**2) / 0.03) + np.exp(-((x - 0.75)**2) / 0.03)
    variants.append(h)
    
    # Variant 3: Periodic
    h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x)
    variants.append(h)
    
    # Variant 4: Staircase
    h = np.zeros(n)
    quarter = n // 4
    for i in range(n):
        if (i % (2*quarter)) < quarter:
            h[i] = 1.0
    variants.append(h)
    
    # Variant 5: Random smooth
    t = np.linspace(0, 1, 50)
    vals = rng.randn(50).cumsum()
    vals = (vals - vals.min()) / (vals.max() - vals.min() + 1e-10)
    h = np.interp(x, t, vals)
    variants.append(h)
    
    # Project all to feasible
    target = n / 2.0
    projected = []
    for h in variants:
        h = project_feasible(h, target)
        projected.append(h)
    
    return projected

            for ki in active:
                ki = int(ki)

                jn = np.arange(ki, n)

            j = rng.randint(n)
        
        if i == j:
            continue
        
        # Transfer from i to j (reduce h[i], increase h[j])
        delta = min(h_cur[i], 1.0 - h_cur[j])
        if delta < 1e-12:

            delta = min(h_cur[j], 1.0 - h_cur[i])
            if delta < 1e-12:

            i, j = j, i

        h_cur[j] += d
        
        sc = compute_c5(h_cur)
        if sc < best - 1e-15:
            best = sc
            best_h = h_cur.copy()
        else:
            # Revert
            h_cur[i] += d
            h_cur[j] -= d
    
    return best_h, best

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:

        hi = h.max() + 0.5

            if s > target:

            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad
