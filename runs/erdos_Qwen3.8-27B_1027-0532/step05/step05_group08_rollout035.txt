What if I formulate the problem as a second-order cone program (SOCP) or semidefinite program (SDP)?

The constraint is: corr[k] ≤ T for all k (for a target T)

The term h[i]*h[i+k] is bilinear, making this non-convex. However, I can use the S-lemma or other relaxations.

Actually, for a fixed k, the constraint corr[k] ≤ T is:
sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k]) ≤ T

This is equivalent to:
sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i] * h[i+k] ≤ T

The term sum h[i] * h[i+k] is the "autocorrelation" at lag k. For the constraint to be satisfied, we need the autocorrelation to be large enough.

If I fix the sum (which is n/2), then I need:
sum h[i] * h[i+k] ≥ n/2 - T (approximately, adjusting for the range)

This is a quadratic constraint. The feasible set is non-convex in general (the set {h: h^T A h ≥ c} is non-convex for positive semidefinite A).

        h[idx] = 1.0
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def swap_anneal(h, n, target, n_iters, deadline, rng, init_temp, fin_temp, batch):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters, 1)
        temp = init_temp * (fin_temp / init_temp) ** frac
        h_new = h_cur.copy()
        for _ in range(batch):
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i != j:
                h_new[i], h_new[j] = h_new[j], h_new[i]
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
        else:
            diff = sc_new - compute_c5(h_cur)
            if rng.random() < np.exp(-diff / (temp + 1e-15)):
                h_cur = h_new
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def active_set_optimize(h, n, target, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = 0.02
    for step in range(5000):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr)) * (2.0 / n)
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
            lr = min(lr * 1.02, 0.06)
        top_idx = np.argsort(corr)[::-1][:4]
        g = np.zeros(n)
        for m in top_idx:
            lag = int(m) - (n - 1)
            g += grad_at_lag(h_cur, lag, n)
        g /= 4.0
        h_new = project_feasible(h_cur - lr * g, target)
        c_new = compute_c5(h_new)
        if c_new < c_max - 1e-15:
            h_cur = h_new
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _rng
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)
    _rng = rng

    n = len(initial_h_values)
    target = n / 2.0
    h_start = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_start.copy()
    best_score = compute_c5(best_h)

    k_sched = [12, 6, 3, 1]

    # Phase 1: Top-k momentum at original resolution
    if time.time() < deadline - 700:
        h_opt, score_opt = topk_momentum_optimize(best_h, n, target, k_sched, 8000, deadline - 700, 0.04, 0.84)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt

    # Phase 2: Active set refinement
    if time.time() < deadline - 550:
        h_act, score_act = active_set_optimize(best_h, n, target, deadline - 550)
        if score_act < best_score:
            best_score = score_act
            best_h = h_act

    # Phase 3: SLSQP with increasing alpha
    if time.time() < deadline - 400:
        h_new = lse_slsqp(best_h, n, target, 3000, 2000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if time.time() < deadline - 300:
        h_new = lse_slsqp(best_h, n, target, 10000, 1500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Swap annealing for escape
    if time.time() < deadline - 350:
        h_an, sc_an = swap_anneal(best_h, n, target, 3000, deadline - 350, rng, 0.008, 0.0005, 3)
        if sc_an < best_score:
            best_score = sc_an
            best_h = h_an
            h_ref = lse_slsqp(best_h, n, target, 5000, 1000)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 5: Binary init alternative
    if time.time() < deadline - 400:
        h_bin, score_bin = binary_random_init(n, target, 300, rng)
        if score_bin < best_score + 0.03:
            h_rel = h_bin + 0.12 * rng.randn(n)
            h_rel = project_feasible(h_rel, target)
            h_opt2, score_opt2 = topk_momentum_optimize(h_rel, n, target, k_sched, 3000, deadline - 300, 0.03, 0.85)
            if score_opt2 < best_score:
                best_score = score_opt2
                best_h = h_opt2
            h_new2 = lse_slsqp(best_h, n, target, 5000, 1000)
            sc2 = compute_c5(h_new2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_new2

    # Phase 6: Upscale to n=800 and optimize
    n2 = 800
    target2 = n2 / 2.0
    if time.time() < deadline - 250:
        h2 = interpolate_to(best_h, n2, target2)
        score2 = compute_c5(h2)
        if score2 < best_score + 0.01:
            h2_opt, score2_opt = topk_momentum_optimize(h2, n2, target2, [6, 4, 2, 1], 5000, deadline - 100, 0.03, 0.86)
            if score2_opt < score2:
                score2 = score2_opt
                h2 = h2_opt
            if time.time() < deadline - 80:
                h2_act, score2_act = active_set_optimize(h2, n2, target2, deadline - 60)
                if score2_act < score2:
                    score2 = score2_act
                    h2 = h2_act
            if time.time() < deadline - 40:
                h2_new = lse_slsqp(h2, n2, target2, 5000, 1500)
                sc2 = compute_c5(h2_new)
                if sc2 < score2:
                    score2 = sc2
                    h2 = h2_new
            if score2 < best_score:
                best_score = score2
                best_h = h2
            h2_an, sc2_an = swap_anneal(h2, n2, target2, 2000, deadline - 20, rng, 0.005, 0.0003, 3)
            if sc2_an < score2:
                score2 = sc2_an
                h2 = h2_an
            if score2 < best_score:
                best_score = score2
                best_h = h2

    # Phase 7: Final polish at whatever resolution
    n_final = len(best_h)
    target_final = n_final / 2.0
    if time.time() < deadline - 10:
        h_pol, sc_pol = active_set_optimize(best_h, n_final, target_final, deadline - 5)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
