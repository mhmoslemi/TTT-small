    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    
    consecutive_no_improve = 0

            break
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > 500:
                break
        
        # Compute gradient at the max shift
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        # This is the gradient of c[k_star] w.r.t. h
        # We want to DECREASE c[k_star], so we move h in -grad direction
        # Positions with high grad should decrease, low grad should increase
        
        # Precompute A matrix
        A = compute_A_matrix(h)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]  # these should increase
        high_set = sorted_idx[-n_cand:]  # these should decrease
        
        # Try all pairs
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                # v = A[j] - A[i] (effect on corr when h[i]-=δ, h[j]+=δ)
                # We want to minimize max(corr + δ * v)
                
                # Bounds on δ
                delta_lo = max(0.0, -h[i])  # h[i] - δ ≥ 0 → δ ≤ h[i]
                delta_hi = min(h[i], 1.0 - h[j])  # h[j] + δ ≤ 1 → δ ≤ 1-h[j]
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                v = A[j] - A[i]
                
                # Binary search for optimal δ
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * v
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(v[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * v) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

    return best_h, best_c5

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_A_matrix(h):
    n = len(h)
    n_corr = 2 * n - 1
    A = np.zeros((n, n_corr))
    for s in range(-(n-1), n):
        m = s + n - 1
        p_lo1 = max(0, -s)
        p_hi1 = min(n, n - s)
        if p_lo1 < p_hi1:
            A[p_lo1:p_hi1, m] += (1.0 - h[p_lo1 + s : p_hi1 + s])
        p_lo2 = max(0, s)
        p_hi2 = min(n, n + s)
        if p_lo2 < p_hi2:
            A[p_lo2:p_hi2, m] -= h[p_lo2 - s : p_hi2 - s]
    return A

            # (decrease h at the highest gradient position)

        corr = np.correlate(hv, 1.0 - hv, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target,
             'jac': lambda hv: np.ones(n)}]

    h_cur = h.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: obj_smooth(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Main vectorized pair CD at original resolution
    if tleft() > 200:
        t1 = min(300, tleft() * 0.4)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                         time_limit=t1, n_cand=40, n_shifts=3,
                                         max_sweeps=100000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Multi-gradient refinement
    if tleft() > 100:
        t2 = min(80, tleft() * 0.2)
        h_mg, c5_mg = multi_gradient_step(best_h, n, target,
                                          time_limit=t2, max_steps=10000,
                                          n_top=5, base_lr=0.02)
        if c5_mg < best_score:
            best_score = c5_mg
            best_h = h_mg

    # Phase 3: Another round of pair CD with different params
    if tleft() > 150:
        t3 = min(200, tleft() * 0.35)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                         time_limit=t3, n_cand=30, n_shifts=5,
                                         max_sweeps=100000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Binary simulated annealing for structure search
    if tleft() > 120:
        t4 = min(100, tleft() * 0.25)
        h_sa, c5_sa = binary_sa(best_h, n, target,
                                time_limit=t4, max_steps=30000,
                                T_init=0.03, T_final=0.0005)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 5: Pair CD after SA
    if tleft() > 100:
        t5 = min(100, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                         time_limit=t5, n_cand=40, n_shifts=3,
                                         max_sweeps=100000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Try higher resolution
    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            sc_interp = compute_c5(h_interp)

            t6 = min(100, tleft() * 0.3)
            h_cd, c5_cd = vectorized_pair_cd(h_interp, n_try, target_try,
                                             time_limit=t6, n_cand=35, n_shifts=3,
                                             max_sweeps=100000)
            sc_cd = c5_cd

            if tleft() > 50:
                t6b = min(50, tleft() * 0.2)
                h_cd2, c5_cd2 = vectorized_pair_cd(h_cd, n_try, target_try,
                                                   time_limit=t6b, n_cand=30, n_shifts=5,
                                                   max_sweeps=100000)
                if c5_cd2 < sc_cd:
                    sc_cd = c5_cd2
                    h_cd = h_cd2

            if min(sc_interp, sc_cd) < best_score:
                best_score = min(sc_interp, sc_cd)
                best_h = h_cd if sc_cd <= sc_interp else h_interp
                n = n_try
                target = target_try

    # Phase 7: Final pair CD at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t7 = min(80, tleft() * 0.5)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n_use, target_use,
                                         time_limit=t7, n_cand=40, n_shifts=3,
                                         max_sweeps=100000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Multi-gradient final polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t8 = min(30, tleft() * 0.4)
        h_mg, c5_mg = multi_gradient_step(best_h, n_use, target_use,
                                          time_limit=t8, max_steps=5000,
                                          n_top=5, base_lr=0.015)
        if c5_mg < best_score:
            best_score = c5_mg
            best_h = h_mg

    # Phase 9: Final tiny pair CD
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t9 = min(20, tleft() * 0.5)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n_use, target_use,
                                         time_limit=t9, n_cand=30, n_shifts=3,
                                         max_sweeps=50000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final projection and verification
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
