import numpy as np
from scipy.optimize import minimize

def compute_c5(h):

    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    total = 2 * n - 1
    if top_m is not None and top_m < total:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(total)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-15:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            N = n
            if N - k > 0:
                grad[:N - k] += w * (1.0 - h[k:]) * dx
                grad[:N - k] -= w * h[k:] * dx
                grad[k:] += w * (1.0 - h[:N - k]) * dx
                grad[k:] -= 0.0
        else:
            m = -k
            N = n
            if N - m > 0:
                grad[:N - m] += w * (1.0 - h[m:]) * dx
                grad[:N - m] -= w * h[m:] * dx
                grad[m:] += w * (1.0 - h[:N - m]) * dx
    return grad


def nesterov_smooth(h_start, n, target, time_limit, start_time, offset=30):
    dx = 2.0 / n
    h = h_start.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    top_m_schedule = [200, 150, 100, 80, 60, 40, 30, 20, 15, 10]
    iters_per_alpha = 15000

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - offset:
            break
        top_m = top_m_schedule[min(ai, len(top_m_schedule) - 1)]
        top_m = min(top_m, 2 * n - 1)

        L_est = 4.0 * alpha / n + 1.0
        step = 1.0 / max(L_est, 1e-6)
        step = min(step, 5.0)

        x = best_h.copy()
        y = x.copy()
        t = 1.0

            x_new = project_feasible(x_new, target)

def backtracking_nesterov(h_start, n, target, time_limit, start_time, offset=30):

        t = 1.0
        x = best_h.copy()
        y = x.copy()
        c_prev = compute_c5(x)

        for k_iter in range(iters_per_alpha):

            c_prev = c_trial

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, offset=30,
                   damping=0.7, reg=1e-4):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > time_limit - offset:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def smooth_obj_wrapper(h, alpha, dx, top_m):
    corr = compute_corr(h)
    c_max = np.max(corr)
    total = 2 * len(h) - 1
    if top_m is not None and top_m < total:
        top_idx = np.argsort(corr)[-top_m:]
        sub = corr[top_idx]
    else:
        sub = corr
    shifted = np.clip(alpha * (sub - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for ai, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit - offset:
            break
        top_m = max(10, (2 * n - 1) // (ai + 2))
        top_m = min(top_m, 2 * n - 1)
        try:
            def obj_func(x):
                return smooth_obj_wrapper(x, alpha, dx, top_m)

            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def binary_collapse(h, target, threshold_lo=0.03, threshold_hi=0.97):
    h_new = h.copy()
    h_new[h_new < threshold_lo] = 0.0
    h_new[h_new > threshold_hi] = 1.0
    h_new = project_feasible(h_new, target)
    return h_new


def structured_perturbation(h, n, target, rng, ptype):
    h_pert = h.copy()
    if ptype == 0:
        block = rng.randint(n // 10, n // 3)
        start = rng.randint(0, n - block)
        h_pert[start:start + block] = rng.uniform(0, 1, block)
    elif ptype == 1:
        for _ in range(rng.randint(3, 8)):
            idx = rng.randint(0, n)
            h_pert[idx] = rng.uniform(0, 1)
    elif ptype == 2:
        period = rng.randint(3, 10)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h_pert += amp * np.sin(2 * np.pi * x / period + phase)
    elif ptype == 3:
        mask = rng.random(n) > 0.7
        h_pert[mask] = rng.uniform(0, 1, np.sum(mask))
    else:
        h_pert += rng.randn(n) * rng.uniform(0.02, 0.08)
    return project_feasible(h_pert, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Short LP for warm start
    if tleft() > 200:
        lp_time = min(120, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                     max_iter=5000, time_limit=lp_time,
                                     start_time=start_time, offset=budget_s - lp_time + 60,
                                     damping=0.7, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Nesterov with backtracking (main workhorse)
    if tleft() > 300:
        pg_time = min(400, tleft() * 0.45)
        h_pg, c5_pg = backtracking_nesterov(best_h, n, target,
                                            time_limit=pg_time,
                                            start_time=start_time,
                                            offset=budget_s - pg_time + 60)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 3: Binary collapse + re-optimize
    if tleft() > 200:
        h_bin = binary_collapse(best_h, target, 0.05, 0.95)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
        # Re-optimize from binary point
        pg_time = min(150, tleft() * 0.3)
        h_pg, c5_pg = backtracking_nesterov(h_bin, n, target,
                                            time_limit=pg_time,
                                            start_time=start_time,
                                            offset=budget_s - pg_time + 60)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 4: Nesterov with higher alpha
    if tleft() > 200:
        pg_time = min(200, tleft() * 0.35)
        h_pg, c5_pg = nesterov_smooth(best_h, n, target,
                                      time_limit=pg_time,
                                      start_time=start_time,
                                      offset=budget_s - pg_time + 60)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 5: Multi-start with structured perturbations
    n_restarts = 8
    for ri in range(n_restarts):
        if tleft() < 80:
            break
        ptype = ri % 5
        h_pert = structured_perturbation(best_h, n, target, rng, ptype)
        # Short Nesterov from perturbation
        pg_time = min(60, tleft() * 0.15)
        h_pg, c5_pg = backtracking_nesterov(h_pert, n, target,
                                            time_limit=pg_time,
                                            start_time=start_time,
                                            offset=budget_s - pg_time + 40)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 6: SLSQP polish
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slq = slsqp_optimize(best_h, n_use, target_use,
                               [200, 1000, 5000, 20000, 100000],
                               maxiter=3000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               offset=50)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 7: Try higher resolution
    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            c5_interp = compute_c5(h_interp)

            pg_time = min(120, tleft() * 0.35)
            h_pg, c5_pg = backtracking_nesterov(h_interp, n_try, target_try,
                                                time_limit=pg_time,
                                                start_time=start_time,
                                                offset=budget_s - pg_time + 80)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg

            if tleft() > 100:
                h_slq = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                       [500, 2000, 10000, 50000],
                                       maxiter=2000, ftol=1e-18,
                                       time_limit=budget_s, start_time=start_time,
                                       offset=40)
                sc = compute_c5(h_slq)
                if sc < best_score:
                    best_score = sc
                    best_h = h_slq

    # Phase 8: Final Nesterov + SLSQP
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        pg_time = min(80, tleft() * 0.5)
        h_pg, c5_pg = nesterov_smooth(best_h, n_use, target_use,
                                      time_limit=pg_time,
                                      start_time=start_time,
                                      offset=budget_s - pg_time + 20)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slq = slsqp_optimize(best_h, n_use, target_use,
                               [5000, 20000, 100000],
                               maxiter=2000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               offset=10)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Final projection
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
