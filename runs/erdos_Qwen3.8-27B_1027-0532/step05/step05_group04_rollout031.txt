def _lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

def _fourier_obj(params):
    n = _g_fc_n
    target = _g_fc_target
    nf = _g_fc_nf
    x_f = _g_fc_x
    h = np.full(n, params[0])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_lse(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass

def active_lag_descent(h, n, target, n_steps, n_active, temp, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            k = int(ki)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wi * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wi * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h
_new)
        if sc_new <= c_curr:
            h_cur = h_new
        else:
            lr = max(lr * lr_decay, 1e-12)
            if lr < 1e-12:
                break
    return best_h, best_score

def weighted_lag_descent(h, n, target, n_outer, beta_init, beta_growth, lr_init, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init
    for it in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        total_w = weights.sum()
        if total_w < 1e-30:
            break
        weights /= total_w
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            h_new2 = h_cur - 0.5 * lr * grad
            h_new2 = project_feasible(h_new2, target)
            sc_new2 = compute_c5(h_new2)
            if sc_new2 < cur_score:
                h_cur = h_new2
                cur_score = sc_new2
            else:
                lr = max(lr * 0.7, 1e-10)
                if lr < 1e-10:
                    break
        beta = min(beta * beta_growth, 100000.0)
    return h_cur, cur_score

def de_search(n, target, pop_size, max_gen, rng, deadline=None, F=0.8, CR=0.9):

            break

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fourier_opt(n, target, nf, x0, deadline):

            h += res.x[2 * i] * np.sin(np.pi * i * x_f)

    except Exception:
        return None
