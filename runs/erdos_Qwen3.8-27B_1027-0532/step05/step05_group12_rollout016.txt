```python
def solve_1d_minimax(vals, coefs, t_min, t_max):
    # Find t in [t_min, t_max] minimizing max_k (vals[k] + coefs[k] * t)
    n_active = len(vals)
    if n_active == 0:
        return (t_min + t_max) / 2
    
    best_t = t_min
    best_val = np.max(vals + coefs * t_min)
    
    # Check boundaries
    val_at_max = np.max(vals + coefs * t_max)
    if val_at_max < best_val:
        best_val = val_at_max
        best_t = t_max
    
    # Check pairwise intersections
    for i in range(n_active):
        for j in range(i+1, n_active):
            if abs(coefs[i] - coefs[j]) < 1e-15:
                continue
            t_int = (vals[j] - vals[i]) / (coefs[i] - coefs[j])
            if t_min <= t_int <= t_max:
                val_int = np.max(vals + coefs * t_int)
                if val_int < best_val:
                    best_val = val_int
                    best_t = t_int
    return best_t
```

```python
def solve_1d_minimax(vals, coefs, lo, hi):
    """Find t in [lo, hi] minimizing max_k (vals[k] + coefs[k]*t)"""
    n = len(vals)
    if n == 0:
        return (lo + hi) / 2.0
    
    best_t = lo
    best_val = np.max(vals + coefs * lo)
    
    # Check hi
    val_hi = np.max(vals + coefs * hi)
    if val_hi < best_val:
        best_val = val_hi
        best_t = hi
    
    # Check pairwise intersections within [lo, hi]
    # Only consider pairs where the intersection is in range
    for i in range(n):
        ci = coefs[i]
        vi = vals[i]
        for j in range(i + 1, n):
            cj = coefs[j]
            if abs(ci - cj) < 1e-15:

                v = np.max(vals + coefs * t_int)
                if v < best_val - 1e-15:
                    best_val = v
                    best_t = t_int
    
    return best_t
```

```python
    vi = vals[:, None]  # (n, 1)
    vj = vals[None, :]  # (1, n)
    ci = coefs[:, None]
    cj = coefs[None, :]
    
    denom = ci - cj  # (n, n)
    numer = vj - vi  # (n, n)
    
    # Avoid division by zero
    mask = np.abs(denom) > 1e-15
    with np.errstate(divide='ignore', invalid='ignore'):
        t_mat = np.where(mask, numer / np.where(mask, denom, 1.0), np.nan)
    
    # Only consider upper triangle (i < j) to avoid duplicates
    tri_mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    t_mat = np.where(tri_mask, t_mat, np.nan)
    
    # Filter to [lo, hi]
    t_mat = np.where((t_mat >= lo) & (t_mat <= hi), t_mat, np.nan)
    
    # For each candidate t, compute max_k (vals[k] + coefs[k]*t)
    valid = ~np.isnan(t_mat)
    if not np.any(valid):
        # No valid intersection, check boundaries
        val_lo = np.max(vals)
        val_hi = np.max(vals + coefs * hi)
        return lo if val_lo <= val_hi else hi
    
    t_candidates = t_mat[valid]
    # For each candidate, compute the objective
    # obj = max_k (vals[k] + coefs[k] * t)
    obj_vals = np.max(vals[None, :] + coefs[None, :] * t_candidates[:, None], axis=1)
    
    # Also check boundaries
    val_lo = np.max(vals)
    val_hi = np.max(vals + coefs * hi)
    
    all_vals = np.concatenate([obj_vals, [val_lo, val_hi]])
    all_ts = np.concatenate([t_candidates, [lo, hi]])
    
    idx = np.argmin(all_vals)
    return all_ts[idx]
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
