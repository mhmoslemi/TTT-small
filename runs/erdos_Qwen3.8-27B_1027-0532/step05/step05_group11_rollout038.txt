import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_alternating_lp(h_init, n, target, max_iter, time_limit,
                        damping=0.99, reg=1e-6, solver='ECOS'):
    """Alternating LP using ALL lags."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 200:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # All lag constraints
        n_half = n - 1
        for lag in range(-n_half, n_half + 1):
            if lag >= 0:
                # sum_{i=0}^{n-1-lag} h[i] * g[i+lag] * dx <= t
                if n - 1 - lag >= 0:
                    coeff = g[lag:] * dx  # length n-lag
                    # This is a partial sum, need to be careful with cvxpy
                    constraints.append(cp.sum(h_var[:n-lag] * coeff) <= t_var)
            else:
                # lag < 0, let k = -lag
                k = -lag
                # sum_{i=k}^{n-1} h[i] * g[i-lag] * dx = sum_{i=k}^{n-1} h[i] * g[i+k] * dx
                if n - 1 - k >= 0:
                    coeff = g[:n-k] * dx  # length n-k
                    constraints.append(cp.sum(h_var[k:] * coeff) <= t_var)
        
        if reg > 0:
            objective = t_var + reg * cp.sum_squares(h_var - h)
        else:
            objective = t_var
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                try:
                    prob.solve(solver='SCS', verbose=False)
                except Exception:
                    break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

Actually, I just realized: the alternating LP might not converge in the traditional sense because it's like a "saddle point" problem. Let me think...

The problem is: min_h max_k overlap_k(h, 1-h)

The alternating method does:
1. Fix g = 1-h, minimize max_k overlap_k(h_new, g) → gives h_new
2. Set h = h_new, go to 1

This is like: min_h max_k f(h, 1-h) where f(h,g) = overlap_k(h,g)

The key issue: when we fix g and optimize h, we're not actually optimizing the original objective (which has 1-h in both places). We're optimizing a related but different objective.

However, at a fixed point of the alternating method, we'd have: h* is optimal for g* = 1-h*, AND g* is optimal for h*. This is a "saddle point" or "Nash equilibrium" of the game where player 1 chooses h and player 2 chooses g, and the payoff is max_k overlap_k(h,g).

If such a fixed point exists and the alternating method converges to it, then at the fixed point, max_k overlap_k(h*, g*) = max_k overlap_k(h*, 1-h*), which is exactly our objective! So the fixed point of the alternating method gives the correct answer.

The question is whether it converges. In practice, alternating minimization for such problems often converges, especially with damping. Let me implement it with a small damping factor and see.

import numpy as np
from scipy.optimize import linprog
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_overlap_matrix(g, n, dx):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for idx in range(n_lags):
        lag = idx - (n - 1)
        if lag >= 0:
            A[idx, :n - lag] = g[lag:] * dx
        else:
            k = -lag
            A[idx, k:] = g[:n - k] * dx
    return A

def solve_lp_for_h(g, n, target, dx):
    A = build_overlap_matrix(g, n, dx)
    n_lags = 2 * n - 1
    A_ineq = np.hstack([A, -np.ones((n_lags, 1))])
    b_ineq = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy()
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='simplex')
        if res.success:
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def solve_lp_for_g(h, n, target, dx):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for idx in range(n_lags):
        lag = idx - (n - 1)
        if lag >= 0:
            A[idx, :n - lag] = h[lag:] * dx
        else:
            k = -lag
            A[idx, k:] = h[:n - k] * dx
    A_ineq = np.hstack([A, -np.ones((n_lags, 1))])
    b_ineq = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def full_lp_alternating(h_init, n, target, time_limit, damping=1.0,
                        max_stuck=500, alternate_sides=True):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    side = 0

    while time.time() - start < time_limit:
        if alternate_sides:
            if side % 2 == 0:
                g = 1.0 - h
                new_val = solve_lp_for_h(g, n, target, dx)
            else:
                g = 1.0 - h
                new_g = solve_lp_for_g(h, n, target, dx)
                if new_g is None:
                    side += 1
                    continue
                new_val = 1.0 - new_g
        else:
            g = 1.0 - h
            new_val = solve_lp_for_h(g, n, target, dx)

        if new_val is None:
            break
        side += 1

        if damping < 1.0:
            new_val = (1.0 - damping) * h + damping * new_val
        new_val = project_feasible(new_val, target)

        c5_new = compute_c5(new_val)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = new_val.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > max_stuck:
                break
        h = new_val

    return best_h, best_c5

def binary_round(h, target):
    n = len(h)
    n_ones = int(target)
    idx = np.argpartition(h, n - n_ones)[n - n_ones:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def binary_local_search(h_bin, n, target, time_limit, rng, max_rounds=100000):
    start = time.time()
    dx = 2.0 / n
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0

    for step in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break

        max_tries = 20
        for _ in range(max_tries):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]

            h[i] = 0.0
            h[j] = 1.0
            c5_try = compute_c5(h)

            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h.copy()
                no_improve = 0
                improved = True
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
                break
            else:
                h[i] = 1.0
                h[j] = 0.0

            no_improve += 1

        if not improved and no_improve > 200:
            h = best_h.copy()
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            if rng.random() < 0.15:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0.0, 1.0
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
            no_improve = 0

    return best_h, best_c5

def targeted_binary_search(h_bin, n, target, time_limit, rng):
    start = time.time()
    dx = 2.0 / n
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    while time.time() - start < time_limit:
        if no_improve > 3000:
            break

        overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        n_lags = len(overlaps)
        n_half = n - 1
        top_k_idx = np.argsort(overlaps)[::-1][:5]
        top_lags = top_k_idx - n_half

        improved = False
        for lag in top_lags:
            if time.time() - start > time_limit:
                break
            if lag >= 0:
                pos_1 = [i for i in range(n - lag) if h[i] == 1.0 and h[i + lag] == 0.0]
                pos_0 = [i for i in range(lag, n) if h[i] == 0.0 and h[i - lag] == 1.0]
            else:
                k = -lag
                pos_1 = [i for i in range(k, n) if h[i] == 1.0 and h[i - k] == 0.0]
                pos_0 = [i for i in range(0, n - k) if h[i] == 0.0 and h[i + k] == 1.0]

            if not pos_1 or not pos_0:
                continue

            n_tries = min(10, len(pos_1) * len(pos_0))
            for _ in range(n_tries):
                i = pos_1[rng.randint(len(pos_1))]
                j = pos_0[rng.randint(len(pos_0))]
                if i == j:
                    continue
                h[i], h[j] = 0.0, 1.0
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    no_improve = 0
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
                    no_improve += 1
            if improved:
                break

        if not improved:
            no_improve += 1

    return best_h, best_c5

def smooth_obj_wrapper(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def slsqp_polish(h_start, n, target, alphas_list, maxiter=1000):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    from scipy.optimize import minimize

    def obj_func(x):
        return smooth_obj_wrapper(x, 100, n, target) * 1.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas_list[:6]:
        try:
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def make_structured_init(n, rng, target):
