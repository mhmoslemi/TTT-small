def alternating_lp_full(h_init, n, target, max_iter, time_limit,
                        damping_start=1.0, damping_end=0.3):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_history = []

    for it in range(max_iter):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        damping = max(damping_end, damping)

        h_new = solve_full_lp(h, n, target)
        if h_new is None or np.all(h_new == h):
            continue

        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        c5_history.append(c5_val)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        if len(c5_history) > 20:
            recent = c5_history[-20:]
            if max(recent) - min(recent) < 1e-10:
                break

    return best_h, best_c5


def _obj_slsqp(h):
    n = _slsqp_n
    alpha = _slsqp_alpha
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    logsum = float(np.log(np.sum(np.exp(shifted)))) / alpha
    return float(c_max + logsum)


def _grad_slsqp(h):
    n = _slsqp_n
    alpha = _slsqp_alpha
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    idx = np.arange(n)
    half_n = n - 1
    for m in range(len(corr)):
        if weights[m] < 1e-15:
            continue
        L = m - half_n
        j_min = max(0, L)
        j_max = min(n - 1, n - 1 + L)
        for j in range(j_min, j_max + 1):
            grad[j] += weights[m] * (1.0 - h[j - L]) * dx
            grad[j] -= weights[m] * h[j + L] * dx if 0 <= j + L < n else 0.0

    return grad

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1

        if time.time() - start > time_limit:

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idxs = np.arange(n)
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False

                best_h = h_try.copy()

                break

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def periodic_pattern_search(h_init, n, target, periods, time_limit):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for p in periods:
        if time.time() - start > time_limit:
            break
        if p < 2 or p > n:
            continue

        num_reps = n // p
        remaining = n - num_reps * p
        if num_reps < 2:
            continue

        pattern = np.zeros(p)
        for i in range(p):
            positions = list(range(i, n, p))
            pattern[i] = np.mean(h_init[positions])

        target_pattern = (target - remaining * 0.5) / num_reps
        target_pattern = max(1.0, min(p - 1.0, target_pattern))

        pattern = project_feasible(pattern, p / 2.0)

        h_var = cp.Variable(p)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == p / 2.0,
            h_var >= 0,
            h_var <= 1
        ]

        scale = 2.0 / p
        for k in range(p):
            coeff = np.zeros(p)
            for r in range(p):
                coeff[r] = (1.0 - pattern[(r + k) % p]) * scale
            constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                continue

        if h_var.value is not None:
            opt_pattern = np.clip(h_var.value, 0.0, 1.0)
            opt_pattern = project_feasible(opt_pattern, p / 2.0)

            h_tiled = np.zeros(n)
            for i in range(num_reps):
                h_tiled[i * p:(i + 1) * p] = opt_pattern
            if remaining > 0:
                h_tiled[num_reps * p:] = opt_pattern[:remaining]
            h_tiled = project_feasible(h_tiled, target)

            sc = compute_c5(h_tiled)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_tiled.copy()

    return best_h, best_c5

    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Full-constraint alternating LP (main workhorse)
    if tleft() > 300:
        lp_time = min(400, tleft() * 0.4)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                          max_iter=5000, time_limit=lp_time,
                                          damping_start=1.0, damping_end=0.3)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 2: SLSQP refinement
    if tleft() > 100:
        h_ref, sc = slsqp_optimize(best_h, n, target,
                                   [50, 200, 1000], 2000, 1e-14)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref

    # Phase 3: Second LP round with different damping
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.25)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                          max_iter=5000, time_limit=lp_time,
                                          damping_start=0.9, damping_end=0.5)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 4: Periodic pattern search for diversity
    if tleft() > 200:
        periods = [4, 6, 8, 10, 12, 14, 16, 20, 24, 25, 28, 30, 32, 40]
        h_per, c5_per = periodic_pattern_search(best_h, n, target,
                                                periods, time_limit=min(60, tleft() * 0.3))
        if c5_per < best_c5:
            best_c5 = c5_per
            best_h = h_per

        if c5_per < best_c5 + 0.01 and tleft() > 150:
            lp_time = min(100, tleft() * 0.3)
            h_lp, c5_lp = alternating_lp_full(h_per, n, target,
                                              max_iter=3000, time_limit=lp_time,
                                              damping_start=1.0, damping_end=0.4)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    # Phase 5: SLSQP with high alpha
    if tleft() > 100:
        h_ref, sc = slsqp_optimize(best_h, n, target,
                                   [1000, 5000], 1500, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref

    # Phase 6: Multi-start perturbations
    for noise in [0.02, 0.05, 0.1, 0.2]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        lp_time = min(60, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_full(h_pert, n, target,
                                          max_iter=2000, time_limit=lp_time,
                                          damping_start=1.0, damping_end=0.5)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        if tleft() > 50:
            h_ref, sc = slsqp_optimize(best_h, n, target,
                                       [500, 2000], 800, 1e-14)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_ref

    # Phase 7: Try higher resolution
    if tleft() > 250:
        n_high = n * 2
        target_high = n_high / 2.0
        h_interp = interpolate_to(best_h, n_high, target_high)
        lp_time = min(150, tleft() * 0.4)
        h_lp, c5_lp = alternating_lp_full(h_interp, n_high, target_high,
                                          max_iter=3000, time_limit=lp_time,
                                          damping_start=1.0, damping_end=0.4)
        if c5_lp < best_c5:
            best_h = h_lp
            best_c5 = c5_lp
            n = n_high
            target = target_high
            if tleft() > 60:
                h_ref, sc = slsqp_optimize(best_h, n, target,
                                           [500, 2000], 800, 1e-14)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_ref

    # Phase 8: Subgradient refinement
    if tleft() > 50:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                          max_steps=3000,
                                          time_limit=min(30, tleft() * 0.5))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 9: Final LP polish
    if tleft() > 80:
        lp_time = min(60, tleft() * 0.7)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                          max_iter=3000, time_limit=lp_time,
                                          damping_start=0.95, damping_end=0.7)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 10: Final SLSQP
    if tleft() > 40:
        h_ref, sc = slsqp_optimize(best_h, n, target,
                                   [2000, 10000], 1000, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref

    # Phase 11: Final subgradient
    if tleft() > 15:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                          max_steps=2000,
                                          time_limit=min(12, tleft() * 0.8))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Final feasibility check
    h_final = project_feasible(best_h, n / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
