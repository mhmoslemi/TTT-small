        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 2: High-alpha FISTA
    for a, td in [(3000, 45), (10000, 45), (30000, 40)]:
        if time.time() > deadline_final - 250:
            break
        h_new, sc_new = fista_run(best_h, target, a, time.time() + td)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = targeted_optimize(best_h, target, 3, time.time() + 20)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: Targeted subgradient focused
    if time.time() < deadline_final - 200:
        h_new, sc_new = targeted_optimize(best_h, target, 5, time.time() + 50, temp_base=0.02)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = targeted_optimize(best_h, target, 3, time.time() + 40, temp_base=0.005)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = targeted_optimize(best_h, target, 2, time.time() + 30, temp_base=0.002)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 4: Multi-scale
    if time.time() < deadline_final - 150:
        h_new, sc_new = multi_scale_optimize(best_h, n, target, time.time() + 80)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
            h_new, sc_new = fista_run(best_h, target, 10000, time.time() + 40)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 5: Structured restarts
    if time.time() < deadline_final - 100:
        h_new, sc_new = structured_restarts(best_h, n, target, time.time() + 70)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 6: Final high-alpha + targeted
    if time.time() < deadline_final - 60:
        h_new, sc_new = fista_run(best_h, target, 50000, time.time() + 35)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = targeted_optimize(best_h, target, 3, time.time() + 20, temp_base=0.003)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Coordinate descent final polish
    if time.time() < deadline_final - 10:
        h_new, sc_new = coord_polish(best_h, target, deadline_final - 5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
