        h_res, sc = slsqp_optimize(h_pert, n, target, alpha_schedule, maxiter_list, 60)
        if sc < best_score:
            best_score = sc
            best_h = h_res
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    t0 = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Linearized constraint generation LP at base resolution
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = linearized_cg_lp(best_h, n, target,
                                       max_iter=5000, time_limit=lp_time, damp=0.7)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: SLSQP continuation with moderate alpha
    if tleft() > 100:
        h_sl, sc_sl = slsqp_optimize(best_h, n, target,
                                     [30, 100, 300, 1000], [1500, 1500, 1500, 1500],
                                     min(80, tleft() * 0.5))
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 3: Another LP round
    if tleft() > 150:
        lp_time = min(150, tleft() * 0.3)
        h_lp, c5_lp = linearized_cg_lp(best_h, n, target,
                                       max_iter=5000, time_limit=lp_time, damp=0.85)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: High-alpha SLSQP
    if tleft() > 80:
        h_sl, sc_sl = slsqp_optimize(best_h, n, target,
                                     [500, 2000, 10000, 50000], [1000, 1000, 1000, 1000],
                                     min(70, tleft() * 0.5))
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 5: Subgradient
    if tleft() > 40:
        h_sg, c5_sg = subgradient_refine(best_h, n, target,
                                         max_steps=3000, time_limit=min(30, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Polish with SLSQP after subgradient
    if tleft() > 60:
        h_sl, sc_sl = slsqp_optimize(best_h, n, target,
                                     [2000, 10000, 50000, 200000], [800, 800, 800, 800],
                                     min(50, tleft() * 0.6))
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 7: Try higher resolution
    if tleft() > 250:
        n2 = 600
        if n2 % 2 == 0:
            target2 = n2 / 2.0
            h_interp = interpolate_to(best_h, n2, target2)
            sc_interp = compute_c5(h_interp)
            h_lp2, c5_lp2 = linearized_cg_lp(h_interp, n2, target2,
                                             max_iter=3000, time_limit=min(120, tleft() * 0.4), damp=0.7)
            if c5_lp2 < best_score:
                best_score = c5_lp2
                best_h = h_lp2
                n = n2
                target = target2
            if tleft() > 80:
                h_sl2, sc_sl2 = slsqp_optimize(best_h, n, target,
                                               [100, 500, 2000, 10000], [1000, 1000, 1000, 1000],
                                               min(70, tleft() * 0.5))
                if sc_sl2 < best_score:
                    best_score = sc_sl2
                    best_h = h_sl2

    # Phase 8: Another LP at current best resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(80, tleft() * 0.5)
        h_lp, c5_lp = linearized_cg_lp(best_h, n_use, target_use,
                                       max_iter=3000, time_limit=lp_time, damp=0.9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 9: Perturbation restarts
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        for s in [seed + 1, seed + 2, seed + 3]:
            if tleft() < 50:
                break
            h_p, sc_p = perturb_and_optimize(best_h, n_use, target_use, s,
                                             [100, 500, 2000], [800, 800, 800],
                                             None)
            if sc_p < best_score:
                best_score = sc_p
                best_h = h_p

    # Phase 10: Final high-alpha SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sl, sc_sl = slsqp_optimize(best_h, n_use, target_use,
                                     [5000, 20000, 100000, 500000], [600, 600, 600, 600],
                                     min(35, tleft() * 0.8))
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 11: Final subgradient
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_refine(best_h, n_use, target_use,
                                         max_steps=2000, time_limit=min(15, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 12: Final LP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp, c5_lp = linearized_cg_lp(best_h, n_use, target_use,
                                       max_iter=2000, time_limit=min(20, tleft() * 0.7), damp=0.95)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
