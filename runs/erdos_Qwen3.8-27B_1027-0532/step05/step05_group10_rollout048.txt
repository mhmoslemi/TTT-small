        h_s, sc_s = slsqp_analytic(best_h, n, target,
                                    [30000, 100000, 300000], 1500, 1e-16, min(60, _tl() * 0.8))
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 8: Final LP polish
    if _tl() > 25:
        h_lp, c5_lp = lp_alternate(best_h, n, target, time_limit=18, damping=0.95, n_active=60)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 9: Last SLSQP polish
    if _tl() > 20:
        h_s, sc_s = slsqp_analytic(best_h, n, target,
                                    [50000, 200000], 1000, 1e-16, min(18, _tl() * 0.8))
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    h_final = project_feasible(best_h, n / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
