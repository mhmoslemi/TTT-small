    """Active set method for minimax"""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for outer in range(50):
        if deadline is not None and time.time() > deadline:
            break
        
        over = corr_over_lags(h_cur)
        max_val = float(np.max(over))
        
        # Identify active set
        active_idx = np.where(over >= max_val - active_tol)[0]
        if len(active_idx) > max_active:
            active_idx = np.argsort(over)[::-1][:max_active]

        # min t s.t. t >= f_k(h) for k in active_idx, sum h = target, 0 <= h <= 1
        n_vars = n + 1  # h (n vars) + t (1 var)
        
        def obj(x):
            return x[n]  # minimize t
        
        def obj_grad(x):
            g = np.zeros(n_vars)
            g[n] = 1.0
            return g
        
        cons = []
        for k in active_idx:
            k = int(k)
            def con(x, k=k):

                h_part = x[:n]
                f_k_val = float(np.sum(h_part[:n-k] * (1 - h_part[k:]))) * (2.0/n)
                return -(x[n] - f_k_val)
            def con_jac(x, k=k):
                gj = np.zeros(n_vars)
                h_part = x[:n]
                j_pos = np.arange(0, n - k)
                gj[j_pos] -= (1.0 - h_part[j_pos + k]) * (2.0/n)
                j_neg = np.arange(k, n)
                gj[j_neg] -= (-h_part[j_neg - k]) * (2.0/n)
                gj[n] = -1.0
                return gj
            cons.append({'type': 'ineq', 'fun': con, 'jac': con_jac})
        
        # Equality constraint: sum h = target
        def eq_con(x):
            return np.sum(x[:n]) - target
        def eq_jac(x):
            gj = np.zeros(n_vars)
            gj[:n] = 1.0
            return gj
        cons.append({'type': 'eq', 'fun': eq_con, 'jac': eq_jac})
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        x0 = np.concatenate([h_cur, [cur_score / (2.0/n)]])

            h_new = project_feasible(res.x[:n], target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def exact_project(h, target):
    """Exact Euclidean projection onto {x : 0 <= x <= 1, sum x = target}"""
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, k, n, dx):
    """Gradient of the k-th lag correlation with respect to h"""
    grad = np.zeros(n)
    if k == 0:
        # corr[0] = sum_j h[j](1-h[j]), grad = (1-2h[j]) * dx
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        j_pos = np.arange(0, n - k)
        grad[j_pos] += (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= h[j_neg - k]
        grad *= dx
    return grad

```python
def min_norm_subgrad(over, h, n, dx, tol=1e-6):
    """Find the minimum-norm subgradient of the max correlation"""
    max_val = float(np.max(over))
    active = np.where(over >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(over))])
    
    if len(active) == 1:
        return grad_lag(h, int(active[0]), n, dx)
    
    # QP: min ||sum p_k g_k||^2 s.t. p >= 0, sum p = 1
    # = min p^T (G^T G) p s.t. p >= 0, p^T 1 = 1
    # where G has columns g_k
    
    grads = np.array([grad_lag(h, int(k), n, dx) for k in active])
    # G is (n x m) where m = len(active)
    GtG = grads.T @ grads  # (m x m)
    m = len(active)
    
    # Solve: min (1/2) p^T GtG p s.t. GtG_1 = GtG @ 1 (this is the "least squares" solution)
    # Using KKT: GtG p + lambda * 1 + mu = 0, p >= 0, mu >= 0, p_i * mu_i = 0
    # Simple approach: use scipy minimize
    
    from scipy.optimize import minimize as sp_min
    
    obj = lambda p: 0.5 * p @ GtG @ p
    obj_grad = lambda p: GtG @ p
    cons = [{'type': 'eq', 'fun': lambda p: np.sum(p) - 1.0}]
    bounds = [(0.0, 1.0)] * m
    p0 = np.ones(m) / m
    
    res = sp_min(obj, p0, jac=obj_grad, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
    
    p_opt = np.clip(res.x, 0.0, None)
    p_opt /= p_opt.sum()
    
    subgrad = grads.T @ p_opt
    return subgrad
```

This computes the minimum-norm subgradient, which gives the steepest descent direction for the max function. This is more efficient than just averaging the active gradients.
