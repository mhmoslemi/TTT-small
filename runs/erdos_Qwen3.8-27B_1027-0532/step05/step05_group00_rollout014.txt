    #   new_corr[m] -= h[j-k] if j-k in [0,n-1]

    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:

    #   if 0 <= i-k < n: delta[m] += h[i-k]

    ks2 = np.arange(k_min_i2, k_max_i2 + 1)
    m_idx2 = n - 1 + ks2
    delta[m_idx2] += h[i - ks2]

    k_min_j = -j

    m_idx4 = n - 1 + ks4
