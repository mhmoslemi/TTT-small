from scipy.optimize import minimize, linprog

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m]
        if wk < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[k:] += wk * (1.0 - h[:n - k])
            g[:n - k] -= wk * h[k:]
        else:
            ell = -k
            g[:n - ell] += wk * (1.0 - h[ell:])
            g[ell:] -= wk * h[:n - ell]
    return g * dx


def obj_lbfgs(x, args):
    alpha = args[0]
    return lse_value(x, alpha)


def grad_lbfgs(x, args):
    alpha = args[0]
    return lse_grad(x, alpha)


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def obj_slsqp(x, args):
    alpha = args[0]
    return lse_value(x, alpha)


def grad_slsqp(x, args):
    alpha = args[0]
    return lse_grad(x, args[0])


def cons_sum_fun(x, args):
    target = args[0]
    return np.sum(x) - target


def cons_sum_jac(x, args):
    return np.ones(len(x))


def slsqp_round(h, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_sum_fun, 'jac': cons_sum_jac, 'args': (target,)}]
    try:
        res = minimize(obj_slsqp, h, method='SLSQP', jac=grad_slsqp,
                       args=(alpha,), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def single_lag_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        ell = -k
        g[:n - ell] += (1.0 - h[ell:])
        g[ell:] -= h[:n - ell]
    return g * dx


def lp_trust_region(h, n, target, m_active, delta, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    dx = 2.0 / n
    for outer in range(60):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:m_active]
        A_ub_rows = []
        b_ub_vals = []
        for m in top_m:
            k = int(m) - (n - 1)
            g_k = single_lag_grad(h_cur, k)
            c_k = float(corr[int(m)])
            d_k = c_k * dx - float(np.dot(g_k, h_cur))
            A_ub_rows.append(np.concatenate([g_k, np.array([-1.0])]))
            b_ub_vals.append(-d_k)
        A_ub = np.array(A_ub_rows) if A_ub_rows else None
        b_ub = np.array(b_ub_vals) if b_ub_vals else None
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        lb = np.maximum(0.0, h_cur - delta)
        ub = np.minimum(1.0, h_cur + delta)
        bounds_lp = list(zip(lb, ub))
        bounds_lp.append((None, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if res.success:
                h_lp = res.x[:n]
                C_lp = res.x[n]
                for t in [1.0, 0.5, 0.25, 0.1]:
                    h_try = (1.0 - t) * h_cur + t * h_lp
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-15:
                        best_score = sc
                        best_h = h_try.copy()
                        h_cur = h_try.copy()
                        break
        except Exception:
            break
    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        ws = weights.sum()
        if ws < 1e-300:
            continue
        weights /= ws
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            g_k = single_lag_grad(h_cur, k)
            g += wk * g_k
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = single_lag_grad(h_cur, k_star)
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            best_h = h_cur.copy()
            eps = min(eps * 1.01, 0.02)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def active_set_slsqp(h, n, target, m_active, maxiter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for outer in range(20):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:m_active]
        c_max_curr = float(corr[top_m[0]])
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        active_cons = []
        for m in top_m:
            k = int(m) - (n - 1)
            def con_fun(x, args):
                kk = args[0]
                g_k = single_lag_grad(x, kk)
                return float(np.dot(g_k, x)) - (c_max_curr + 0.0)
            def con_jac(x, args):
                kk = args[0]
                return single_lag_grad(x, kk)
            active_cons.append({'type': 'ineq', 'fun': con_fun, 'jac': con_jac,
                                'args': (k, c_max_curr)})
        all_cons = [{'type': 'eq', 'fun': cons_sum_fun, 'jac': cons_sum_jac, 'args': (target,)}]
        all_cons.extend(active_cons)
        try:
            res = minimize(lse_value, h_cur, method='SLSQP',
                           jac=lambda x: lse_grad(x, 5000.0),
                           bounds=bounds, constraints=all_cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_candidate = project_feasible(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < best_score:
                best_score = sc
                best_h = h_candidate.copy()
                h_cur = h_candidate.copy()
        except Exception:
            break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    target0 = n0 / 2.0
    h0 = project_feasible(np.array(initial_h_values, dtype=float), target0)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Work at n=800
    n = 800
    target = n / 2.0
    h800 = interpolate_to(h0, n, target)
    best_h800 = h800.copy()
    best_score800 = compute_c5(h800)

    # Phase 2: L-BFGS-B continuation at n=800
    alphas_p1 = [20, 60, 200, 800, 3000, 10000, 30000]
    for alpha in alphas_p1:
        if time.time() > deadline - 400:
            break
        for mu in [20.0, 100.0, 500.0]:
            if time.time() > deadline - 400:
                break
            h_new = lbfgsb_round(best_h800, n, target, alpha, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score800:
                best_score800 = sc
                best_h800 = h_new

    # Phase 3: LP trust region at n=800
    if time.time() < deadline - 300:
        for m_act in [8, 5, 3]:
            if time.time() > deadline - 200:
                break
            h_lp, sc_lp = lp_trust_region(best_h800, n, target, m_act, 0.08, deadline - 200)
            if sc_lp < best_score800:
                best_score800 = sc_lp
                best_h800 = h_lp

    # Phase 4: More L-BFGS-B at higher alpha
    for alpha in [50000, 100000, 200000]:
        if time.time() > deadline - 200:
            break
        h_new = lbfgsb_round(best_h800, n, target, alpha, 1000.0, 200)
        sc = compute_c5(h_new)
        if sc < best_score800:
            best_score800 = sc
            best_h800 = h_new

    # Phase 5: Subgradient polish at n=800
    if time.time() < deadline - 150:
        for temp in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline - 120:
                break
            h_sg, sc_sg = subgradient_polish(best_h800, n, target, 2000, temp, 0.005, 0.996, deadline - 120)
            if sc_sg < best_score800:
                best_score800 = sc_sg
                best_h800 = h_sg

    # Phase 6: Worst lag surgery at n=800
    if time.time() < deadline - 80:
        h_surg, sc_surg = worst_lag_surgery(best_h800, n, target, 500, 0.005, deadline - 80)
        if sc_surg < best_score800:
            best_score800 = sc_surg
            best_h800 = h_surg

    # Phase 7: More LP trust region after surgery
    if time.time() < deadline - 60:
        h_lp2, sc_lp2 = lp_trust_region(best_h800, n, target, 5, 0.04, deadline - 60)
        if sc_lp2 < best_score800:
            best_score800 = sc_lp2
            best_h800 = h_lp2

    # Phase 8: Final subgradient at n=800
    if time.time() < deadline - 30:
        h_sg2, sc_sg2 = subgradient_polish(best_h800, n, target, 1500, 0.001, 0.002, 0.998, deadline - 30)
        if sc_sg2 < best_score800:
            best_score800 = sc_sg2
            best_h800 = h_sg2

    # Phase 9: Final surgery at n=800
    if time.time() < deadline - 15:
        h_surg2, sc_surg2 = worst_lag_surgery(best_h800, n, target, 300, 0.002, deadline - 15)
        if sc_surg2 < best_score800:
            best_score800 = sc_surg2
            best_h800 = h_surg2

    # Update global best
    if best_score800 < best_score:
        best_score = best_score800
        best_h = best_h800.copy()

    # Phase 10: Also try optimizing at original n=400 with LP trust region
    if time.time() < deadline - 10:
        h400 = h0.copy()
        sc400 = compute_c5(h400)
        best_h400 = h400.copy()
        best_sc400 = sc400
        alphas_400 = [50, 200, 1000, 5000, 20000]
        for alpha in alphas_400:
            if time.time() > deadline - 10:
                break
            h_new = lbfgsb_round(best_h400, n0, target0, alpha, 100.0, 300)
            sc = compute_c5(h_new)
            if sc < best_sc400:
                best_sc400 = sc
                best_h400 = h_new
        if time.time() < deadline - 10:
            h_lp400, sc_lp400 = lp_trust_region(best_h400, n0, target0, 5, 0.06, deadline - 10)
            if sc_lp400 < best_sc400:
                best_sc400 = sc_lp400
                best_h400 = h_lp400
        if time.time() < deadline - 10:
            h_sg400, sc_sg400 = subgradient_polish(best_h400, n0, target0, 1000, 0.002, 0.003, 0.997, deadline - 10)
            if sc_sg400 < best_sc400:
                best_sc400 = sc_sg400
                best_h400 = h_sg400
        if best_sc400 < best_score:
            best_score = best_sc400
            best_h = best_h400.copy()

    # Final projection and evaluation
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
