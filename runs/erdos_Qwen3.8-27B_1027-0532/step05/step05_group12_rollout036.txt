    # Round to binary
    h = (h_init >= 0.5).astype(float)
    # Adjust to have exactly n//2 ones
    num_ones = int(np.sum(h))
    if num_ones > n // 2:
        # Remove extra ones
        excess = num_ones - n // 2
        one_indices = np.where(h == 1)[0]
        h[one_indices[:excess]] = 0
    elif num_ones < n // 2:

        zero_indices = np.where(h == 0)[0]
        h[zero_indices[:need]] = 1

        zeros = np.where(h == 0)[0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, n):
    """Round h to binary with exactly n//2 ones."""
    h = h.copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:

    elif num_ones < target:
        need = target - num_ones
        zero_idx = np.where(h == 0)[0]

    return h

def smooth_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    
    grad = np.zeros(n)
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = int(idx_k) - half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0 and k < n:
            grad[k:] += w * (1.0 - h[:n-k]) * dx
            grad[:n-k] -= w * h[k:] * dx
        elif k < 0 and -k < n:
            m = -k
            grad[:n-m] += w * (1.0 - h[m:]) * dx
            grad[m:] -= w * h[:n-m] * dx
    return grad
