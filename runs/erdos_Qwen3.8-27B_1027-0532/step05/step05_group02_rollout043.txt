def compute_corr_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def project_binary(h, target_sum):
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    h_r = (h_sorted >= 0.5).astype(float)
    # Use a smarter rounding: take top target_sum values
    h_r = np.zeros(n)
    idx = np.argsort(h)[::-1][:target_sum]
    h_r[idx] = 1.0
    return h_r

def compute_swap_delta(corr, h, i, j, n):
    """Compute the delta in the full correlation when h[i]:1->0, h[j]:0->1"""
    delta = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]
    
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]
    
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    
    return delta

def compute_dual_swap_delta(h, i1, i2, j1, j2, n):
    """Compute delta for dual swap: h[i1]->0, h[i2]->0, h[j1]->1, h[j2]->1"""
    delta = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    
    for i in (i1, i2):
        idx = i + k
        mask = (idx >= 0) & (idx < n)
        delta[mask] += h[idx[mask]]
        idx = i - k
        mask = (idx >= 0) & (idx < n)
        delta[mask] += h[idx[mask]]
    
    for j in (j1, j2):
        idx = j - k
        mask = (idx >= 0) & (idx < n)
        delta[mask] -= h[idx[mask]]
        idx = j + k
        mask = (idx >= 0) & (idx < n)
        delta[mask] -= h[idx[mask]]
    
    return delta

    positions = np.arange(n)

        if time.time() - start_time > time_limit:
            break
        if stall > 200:

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))

        if k_star == 0:

            continue
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if k_star > 0:

            new_c5 = float(corr.max()) * dx

def _smooth_corr_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_con_fun(h, target):
    return np.sum(h) - target

def slsqp_refine(h_start, target, alphas, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = float(np.max(np.correlate(h_cur, 1.0 - h_cur, mode='full'))) * (2.0 / n)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(
                _smooth_corr_obj, h_cur, args=(alpha, n),
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': lambda x: np.ones(n), 'args': (target,)}],
                options={'maxiter': maxiter, 'ftol': 1e-14}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = h_new - (np.sum(h_new) - target) / n
            h_new = np.clip(h_new, 0.0, 1.0)
            sc = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))) * (2.0 / n)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        h_sorted = np.sort(h_src)[::-1]
        h_r = np.zeros(n_dst)
        idx = np.argsort(h_src)[::-1][:target_dst]
        h_r[idx] = 1.0
        return h_r
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    h_r = np.zeros(n_dst)
    idx = np.argsort(h_out)[::-1][:target_dst]
    h_r[idx] = 1.0
    return h_r

def tabu_search(h, n_iters, tabu_size, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    corr = compute_corr_full(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    tabu = {}
    tabu_counter = 0
    stall = 0
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        
        # Expire old tabu entries
        if tabu_counter % 100 == 0:
            expired = [k for k, v in tabu.items() if v < tabu_counter - tabu_size]
            for k in expired:
                del tabu[k]
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Generate candidates targeting worst lag
        candidates = []
        if k_star != 0:
            kk = abs(k_star)
            if k_star > 0:
                i_cands = ones[(ones + kk < n) & (h[ones + kk] == 0)]
            else:
                i_cands = ones[(ones - kk >= 0) & (h[ones - kk] == 0)]
        else:
            i_cands = ones
        
        n_try = min(15, len(i_cands))
        if n_try == 0:
            i_cands = ones
            n_try = min(15, len(ones))
        
        for _ in range(n_try):
            i = i_cands[rng.randint(len(i_cands))]
            j = zeros[rng.randint(len(zeros))]
            key = (min(i, j), max(i, j))
            if key not in tabu:
                candidates.append((i, j))
            else:
                # Aspiration: allow if it improves the best
                delta = compute_swap_delta(corr, h, i, j, n)
                new_max = float((corr + delta).max())
                if new_max < best_c5 / dx * 0.99:
                    candidates.append((i, j))
        
        if len(candidates) == 0:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            candidates = [(i, j)]
        
        # Pick best candidate
        best_new_max = max_val + 1
        best_pair = candidates[0]
        for (i, j) in candidates[:min(10, len(candidates))]:
            delta = compute_swap_delta(corr, h, i, j, n)
            new_max = float((corr + delta).max())
            if new_max < best_new_max:
                best_new_max = new_max
                best_pair = (i, j)
        
        i, j = best_pair
        delta = compute_swap_delta(corr, h, i, j, n)
        h[i] = 0.0
        h[j] = 1.0
        corr += delta
        tabu[(min(i, j), max(i, j))] = tabu_counter
        tabu_counter += 1
        
        new_c5 = float(corr.max()) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return max(0.0, budget_s - 10.0 - (time.time() - start_time))
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    
    # Initial binary projection
    h = project_binary(h_init, target)
    corr = compute_corr_full(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    # Phase 1: Aggressive greedy (20s)
    t1 = min(20.0, tleft() * 0.10)
    h1, c5_1 = greedy_phase(best_h, max_iters=100000, rng=rng,
                             time_limit=t1, start_time=start_time)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Focused SA with dual swaps (150s)
    t2 = min(150.0, tleft() * 0.25)
    n_sa = int(min(4000000, t2 * 30000))
    h2, c5_2 = sa_focused_dual(best_h, max(100000, n_sa), 0.015, 0.00005,
                                rng, time_limit=t2, start_time=time.time(), dual_prob=0.35)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Tabu search (100s)
    t3 = min(100.0, tleft() * 0.20)
    h3, c5_3 = tabu_search(best_h, max(50000, int(t3 * 5000)), tabu_size=200,
                            rng=rng, time_limit=t3, start_time=time.time())
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Iterated local search (remaining bulk)
    t4 = tleft() * 0.30
    h4, c5_4 = iterated_ls(best_h, rng, time_limit=max(30.0, t4),
                            start_time=time.time(), max_rounds=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: More SA from best with different params
    if tleft() > 60:
        t5 = min(80.0, tleft() * 0.30)
        rng2 = np.random.RandomState(seed + 1)
        h5, c5_5 = sa_with_dual(best_h, max(100000, int(t5 * 25000)),
                                 0.01, 0.0001, rng2,
                                 time_limit=t5, start_time=time.time(), dual_prob=0.5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Phase 6: Multi-resolution attempts
    for n_try in [n + 4, n + 8, n - 4, n + 12]:
        if tleft() < 40:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        allowed = min(30.0, tleft() * 0.15)
        if allowed < 10:
            break
        hg, cg = greedy_phase(h_interp, max_iters=50000, rng=rng,
                               time_limit=allowed * 0.4, start_time=time.time())
        allowed2 = max(5.0, allowed * 0.6)
        hs, cs = sa_focused_dual(hg, max(50000, int(allowed2 * 20000)),
                                  0.01, 0.0001, rng,
                                  time_limit=allowed2, start_time=time.time(), dual_prob=0.3)
        c5_try = min(cs, cg)
        h_try = h_interp if c5_try == cg else (hs if c5_try == cs else h_interp)
        # Convert back to original n for comparison
        if n_try != n:
            h_back = interpolate_to(h_try, n, target)
            c5_back = float(np.max(np.correlate(h_back, 1.0 - h_back, mode='full'))) * dx
            if c5_back < best_c5:
                best_c5 = c5_back
                best_h = h_back.copy()
        else:
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 7: SLSQP continuous refinement
    if tleft() > 40:
        target_float = float(n) / 2.0
        allowed = min(tleft() - 20, 100.0)
        h_ref, sc = slsqp_refine(best_h, target_float,
                                  [30, 100, 300, 1000, 3000],
                                  maxiter=1500, time_limit=allowed, start_time=time.time())
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref.copy()
    
    # Phase 8: Final focused SA burst on binary
    if tleft() > 20:
        h_bin = project_binary(best_h, target)
        n_sa_final = max(50000, int((tleft() - 15) * 30000))
        allowed = max(5.0, tleft() - 12)
        h_fin, c5_fin = sa_focused_dual(h_bin, n_sa_final, 0.005, 0.00002,
                                         rng, time_limit=allowed, start_time=time.time(),
                                         dual_prob=0.4)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Final: SLSQP if time remains
    if tleft() > 15:
        target_float = float(n) / 2.0
        h_ref2, sc2 = slsqp_refine(best_h, target_float,
                                    [500, 2000, 5000],
                                    maxiter=1000, time_limit=max(5.0, tleft() - 8),
                                    start_time=time.time())
        if sc2 < best_c5:
            best_c5 = sc2
            best_h = h_ref2.copy()
    
    # Ensure feasibility
    h_final = np.clip(best_h, 0.0, 1.0)
    # Fix sum
    diff = target - np.sum(h_final)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h_final < 1.0)[0]
            h_final[idx] += diff / len(idx)
        else:
            idx = np.where(h_final > 0.0)[0]
            h_final[idx] += diff / len(idx)
        h_final = np.clip(h_final, 0.0, 1.0)
    
    final_score = float(np.max(np.correlate(h_final, 1.0 - h_final, mode='full'))) * dx
    return h_final, final_score, n
