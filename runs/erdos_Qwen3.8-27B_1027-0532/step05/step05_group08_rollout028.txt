        else:

        cnt = int(mask.sum())
        if cnt == 0:

    return g

            # corr_k(t) = A_k + B_k*t + C_k*t^2
            # For k=0: C_j = -1 (from t*(1-t) term), B_0 = 1
            # For k!=0: C_k = 0

            best_val = None

                candidates.append(h_old - 0.05)
            if h_old < 0.95:
                candidates.append(h_old + 0.1)
                candidates.append(h_old + 0.05)
            candidates.append(0.0)
            candidates.append(1.0)
            candidates.append(0.5)
            
            # For each candidate, compute the resulting max corr
            # But we also need to maintain the sum constraint
            # So if we change h[j] from h_old to t, we need to adjust others
            # For simplicity, let's ignore the sum constraint in the 1D subproblem
            # and re-project after each update
            
            for t in candidates:

                h_cur[j] = t
                corr = compute_corr(h_cur)
                val = float(np.max(corr))
                if best_val is None or val < best_val:
                    best_val = val
                    best_t = t

            h_cur[j] = h_old

                h_cur[j] = t
                corr = compute_corr(h_cur)
                return float(np.max(corr))

                best_val = res.fun

            if abs(best_t - h_old) > 1e-10:
                improved = True

        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            h = h_cur.copy()
        else:

            # Just clip
            h_cur = np.clip(h_cur, 0.0, 1.0)
            # Re-adjust sum

            if abs(diff) > 1e-12:
                mask = (h_cur < 1.0) if diff > 0 else (h_cur > 0.0)
                if mask.sum() > 0:
                    h_cur[mask] += diff / mask.sum()
                    h_cur = np.clip(h_cur, 0.0, 1.0)
            sc = compute_c5(h_cur)
            if sc < best_score:
                best_score = sc
                h = h_cur.copy()
        
        if not improved:
            break
    
    return h, best_score

from scipy.optimize import minimize, minimize_scalar
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):

        if wk > 1e-14:

        h_new = slsqp_round(best_h, n_cur, target_cur, 30000, 2000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
