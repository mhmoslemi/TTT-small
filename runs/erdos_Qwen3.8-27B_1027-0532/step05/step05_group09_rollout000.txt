import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    lr_init = 1.0 / (max(alpha * 0.03, 1.0))
    step = lr_init
    stale = 0
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(25):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / max(1.0 - shrink, 0.01), lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj - 1e-12:
            best_obj = sc
            best_x = x.copy()
            stale = 0
        else:
            stale += 1
        if stale > 2000:
            break
    return best_x, best_obj

def subgrad_polish(h, target, deadline, num_active=5):
    """Direct subgradient polishing with momentum"""
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h, target).copy()
    best_sc = compute_c5(h_cur)
    best_h = h_cur.copy()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    
    for it in range(10000):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_max = np.max(corr)
        temp = 0.003
        w = np.exp((corr - c_max) / temp)
        w = w / w.sum()
        
        top_idx = np.argsort(corr)[::-1][:num_active]
        w_top = w[top_idx]
        w_top = w_top / w_top.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w_top, top_idx):
            ki = int(ki)
            if ki == 0:
                grad += wi * (1.0 - 2.0 * h_cur) * dx
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += wi * (1.0 - h_cur[jp + ki]) * dx
                jn = np.arange(ki, n)
                grad[jn] -= wi * h_cur[jn - ki] * dx
        
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h_cur - lr * momentum
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr = max(lr * 0.9, 0.001)
    
    return best_h, best_sc
