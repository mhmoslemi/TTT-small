    best_h = h0.copy()
    best_score = compute_c5(best_h)
    
    for alpha in alphas:
        # Use L-BFGS-B with penalty for sum constraint
        h_cur = best_h.copy()
        
        def obj(x):
            x_clip = np.clip(x, 0, 1)
            # Add penalty for sum constraint
            sum_pen = (np.sum(x_clip) - target) ** 2 * 1e4
            return smooth_obj(x_clip, alpha, n) + sum_pen

            x_clip = np.clip(x, 0, 1)
            g = smooth_grad(x_clip, alpha)
            g += 2 * 1e4 * (np.sum(x_clip) - target)
            return g
        
        res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': maxiter_each, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
