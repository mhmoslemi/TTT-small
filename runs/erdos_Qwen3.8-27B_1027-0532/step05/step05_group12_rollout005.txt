import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, target, max_passes, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    q_coarse = np.linspace(0.0, 1.0, 100)
    q_fine = np.linspace(0.0, 1.0, 200)
    
    for p in range(max_passes):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        improved = False
        for i in range(n):
            old_q = h[i]
            
            # Compute coeff
            coeff = np.zeros(2*n-1)
            coeff[n-1-i : 2*n-1-i] += (1.0 - h)
            coeff[i : n+i] -= h[::-1]
            
            # Grid search
            dq = q_fine - old_q
            f_vals = corr[np.newaxis, :] + coeff[np.newaxis, :] * dq[:, np.newaxis]
            f_0 = corr[n-1] + dq * (1.0 - q_fine - old_q)
            f_vals[:, n-1] = f_0
            new_maxes = np.max(f_vals, axis=1)
            best_idx = np.argmin(new_maxes)
            best_q = q_fine[best_idx]
            
            if abs(best_q - old_q) > 1e-10:
                h[i] = best_q
                corr += coeff * (best_q - old_q)
                corr[n-1] -= (best_q - old_q) ** 2
                improved = True
        
        # Check improvement
        current_c5 = float(np.max(corr)) * dx
        if current_c5 < best_c5 - 1e-12:
            best_c5 = current_c5
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def binary_push(h, n, target, time_limit, start_time, deadline_offset=30):
    """Push values toward 0 or 1, then re-optimize."""
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Try pushing each coordinate to 0 or 1 (whichever is closer)
    # But maintain the sum constraint

        changed = False
        for i in range(n):

                h = project_feasible(h, target)

                    best_h = h.copy()
                    changed = True
                else:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_from_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent(h_init, n, target, max_passes, time_limit, start_time, deadline_offset, rng):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    corr = corr_from_h(h)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    
    q_grid = np.linspace(0.0, 1.0, 300)
    last_improve_pass = 0
    
    for p in range(max_passes):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        order = rng.permutation(n)
        pass_improved = False
        
        for i in order:
            old_q = h[i]
            
            # Compute linear coefficients for each shift
            coeff = np.zeros(2 * n - 1)
            # First factor: positions [n-1-i, 2n-2-i], values (1-h)
            coeff[n-1-i : 2*n-1-i] += (1.0 - h)
            # Second factor: positions [i, n+i-2], values h[::-1]  (negated)
            coeff[i : n+i] -= h[::-1]
            
            # Grid search for optimal q
            dq = q_grid - old_q
            # f_k(q) for all k simultaneously: shape (G, 2n-1)
            f_vals = corr[np.newaxis, :] + coeff[np.newaxis, :] * dq[:, np.newaxis]
            # Fix k=0 (index n-1) with exact quadratic
            f0_vals = corr[n-1] + dq * (1.0 - q_grid - old_q)
            f_vals[:, n-1] = f0_vals
            new_maxes = np.max(f_vals, axis=1)
            best_idx = np.argmin(new_maxes)
            best_q = q_grid[best_idx]
            
            if abs(best_q - old_q) > 1e-10:
                h[i] = best_q
                corr += coeff * (best_q - old_q)
                corr[n-1] -= (best_q - old_q) ** 2
                pass_improved = True
        
        if pass_improved:
            last_improve_pass = p
            current_c5 = float(np.max(corr)) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # No improvement in this pass
            if p - last_improve_pass > 20:
                break
            # Try a perturbation to escape local optimum
            if p % 50 == 0 and p > 0:
                perturbation = rng.randn(n) * 0.01
                h = project_feasible(h + perturbation, target)
                corr = corr_from_h(h)
    
    return best_h, best_c5
