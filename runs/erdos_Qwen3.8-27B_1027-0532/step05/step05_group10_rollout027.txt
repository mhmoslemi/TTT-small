def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    """Project to binary with exact sum."""
    h = h.copy()
    n = len(h)
    # Start with threshold
    b = (h > 0.5).astype(float)
    # Adjust sum
    diff = target - np.sum(b)
    if diff > 0:
        # Need more 1s: flip some 0s to 1s, starting with highest h value
        zeros = np.where(b == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for i in range(int(diff)):
            if i < len(zeros):

        ones = ones[np.argsort(h[ones])]
        for i in range(int(-diff)):
            if i < len(ones):

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    n = len(h)
    b = (h > 0.5).astype(float)
    diff = int(round(target - np.sum(b)))
    if diff > 0:
        zeros = np.where(b == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for i in range(min(diff, len(zeros))):
            b[zeros[i]] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for i in range(min(-diff, len(ones))):
            b[ones[i]] = 0.0
    return b

def binary_local_search(b, target, time_limit, rng):
    start = time.time()
    n = len(b)
    b = b.copy()
    best_b = b.copy()
    best_c5 = compute_c5(b)
    patience = 0
    max_patience = 200

    while time.time() - start < time_limit:
        corr = np.correlate(b, 1.0 - b, mode='full')
        current_max = float(np.max(corr))
        best_new_max = current_max
        best_i = -1
        best_j = -1

        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        n_ones = len(ones)
        n_zeros = len(zeros)

        for ii in range(n_ones):
            i = ones[ii]
            for jj in range(n_zeros):
                j = zeros[jj]
                b[i] = 0
                b[j] = 1
                nm = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
                b[j] = 0
                b[i] = 1
                if nm < best_new_max - 1e-12:
                    best_new_max = nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            b[best_i] = 0
            b[best_j] = 1
            new_c5 = best_new_max * (2.0 / n)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_b = b.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

    return best_b, best_c5

def smooth_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj(h, alpha, n, target):
    return smooth_logsumexp(h, alpha)

def cons_eq(h):
    return np.array([np.sum(h) - h.shape[0] / 2.0])

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit):
    start = time.time()
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(
                lambda x: smooth_logsumexp(x, alpha),
                h_cur,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass

    return h_cur, compute_c5(h_cur)

def alternating_lp(h_init, n, target, max_iter, time_limit, n_active=35, damping=0.85, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    current_reg = reg

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=80000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=80000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

        if it % 100 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.85)

    return best_h, best_c5

def subgradient_step(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 100:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
