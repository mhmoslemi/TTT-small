import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_full(h, weights, n):
    """Compute gradient of sum_i weights[i]*C[i] where C=correlate(h,1-h,'full')."""
    grad = np.zeros(n)
    # Lag 0
    w0 = weights[n - 1]
    if abs(w0) > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    # Positive lags k=1..n-1 (indices n..2n-2)
    for k in range(1, n):
        w = weights[n - 1 + k]
        if abs(w) < 1e-14:
            continue
        grad[k:] += w
        grad[:n-k] -= w * h[k:]
        grad[k:] -= w * h[:n-k]
    # Negative lags p=1..n-1 (indices 0..n-2)
    for p in range(1, n):
        w = weights[n - 1 - p]
        if abs(w) < 1e-14:
            continue
        grad[:n-p] += w * (1.0 - h[p:])
        grad[p:] -= w * h[:n-p]
    return grad

def reweight(corr_full, n, beta):
    """Compute softmax weights over the full correlation vector."""
    c = corr_full.copy()
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def adam_wgd(h, n, target, n_steps, beta_init, beta_growth, lr, deadline=None):
    """Weighted gradient descent with Adam optimizer."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    beta = beta_init
    cur_score = best_score
    stalled = 0
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        t += 1
        corr = compute_corr(h_cur)
        cur_score = float(np.max(corr)) * (2.0 / n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h_cur.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 200:
                # Restart with perturbation
                noise = 0.01 * (np.random.rand(n) - 0.5)
                h_cur = project_feasible(h_cur + noise, target)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stalled = 0
                continue
        
        weights = reweight(corr, n, beta)
        g = grad_full(h_cur, weights, n) * (2.0 / n)
        
        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = h_cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h_cur = h_new
        else:
            # Try smaller step
            h_try = h_cur - 0.5 * lr * m_hat / (np.sqrt(v_hat) + eps)
            h_try = project_feasible(h_try, target)
            if compute_c5(h_try) < cur_score:
                h_cur = h_try
            else:
                lr = max(lr * 0.9, 1e-6)
        
        # Grow beta
        if step % 50 == 0:
            beta = min(beta * beta_growth, 100000.0)
    
    return best_h, best_score

def slsqp_ls(h, n, target, alpha, maxiter, deadline=None):
    """SLSQP with log-sum-exp approximation."""
    def obj(x):
        corr = compute_corr(x)
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700, 0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    def cons_fun(x):
        return np.sum(x) - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except:
        return h

def pair_transfer_search(h, n, target, max_passes=10, deadline=None):
    """Local search: try transferring mass from high to low positions."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Sort indices by h value
        order = np.argsort(h_cur)
        # Try transferring from top to bottom
        for a in range(n // 2, n, n // 20):  # top indices
            if deadline is not None and time.time() > deadline:
                break
            for b in range(0, n // 2, n // 20):  # bottom indices
                i = order[a]
                j = order[b]
                if h_cur[i] - h_cur[j] < 0.01:
                    continue
                # Try small transfer
                for delta in [0.02, 0.05, 0.1, 0.2]:
                    h_new = h_cur.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    if h_new[i] < 0 or h_new[j] > 1:
                        continue
                    h_new = project_feasible(h_new, target)
                    sc = compute_c5(h_new)
                    if sc < cur_score - 1e-10:
                        h_cur = h_new
                        cur_score = sc
                        improved = True
                        break
        if not improved:
            break
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    n = len(initial_h_values)

    # Phase 1: Adam weighted gradient descent (main workhorse)
    deadline1 = start_time + 300

        best_h = h1

                          deadline=start_time + budget_s - 60)
            s2 = compute_c5(h2)
            if s2 < best_score:

                best_h = h2

                          beta_growth=1.08, lr=0.001, deadline=deadline3)

            best_score = s3
            best_h = h3
    
    # Phase 4: Pair transfer local search
    if tleft() > 100:
        h4, s4 = pair_transfer_search(best_h, n, target, max_passes=20,
                                      deadline=start_time + budget_s - 100)

                          beta_growth=1.1, lr=0.0005, deadline=deadline5)
        if s5 < best_score:
            best_score = s5
            best_h = h5

        if s6 < best_score:
            best_score = s6

                                      deadline=start_time + budget_s - 20)
        if s7 < best_score:
            best_score = s7
            best_h = h7
    
    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
