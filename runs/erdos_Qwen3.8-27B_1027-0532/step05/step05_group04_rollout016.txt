def _weighted_grad(h, n, target, weights):
    """Compute weighted gradient of overlap sum."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-10:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    return grad

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

def irls_optimize(h_init, n, target, n_outer, beta_schedule, lr_init,
                  inner_iters, momentum=0.8, deadline=None):
    """
    Iteratively reweighted least squares with momentum.
    """
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    v = np.zeros(n)
    
    for it in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        
        beta = beta_schedule[it] if it < len(beta_schedule) else beta_schedule[-1]
        
        # Compute importance weights
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        max_o = float(np.max(over))
        weights = np.exp(beta * (over - max_o))
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum
        
        for _ in range(inner_iters):
            if deadline is not None and time.time() > deadline:
                break
            
            # Compute gradient
            grad = np.zeros(n)
            for k in range(n):
                wk = weights[k]
                if wk < 1e-10:
                    continue
                jp = np.arange(0, n - k)
                grad[jp] += wk * (1.0 - h_cur[jp + k])
                jn = np.arange(k, n)
                grad[jn] -= wk * h_cur[jn - k]
            grad *= dx
            
            # Momentum update
            v = momentum * v - lr * grad
            h_new = h_cur + v
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            
            if sc_new < cur_score:
                h_cur = h_new
                cur_score = sc_new
                lr = min(lr * 1.03, 0.01)
            else:
                v *= 0.8
                lr = max(lr * 0.92, 1e-7)
    
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    for _ in range(10):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return h

def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def sa_adaptive(h_init, n, target, n_steps, t_init, t_min, rng,

    score_cur = compute_c5(h_cur)
    score_best = score_cur
    t = t_init
    cool_rate = (t_min / t_init) ** (1.0 / max(n_steps, 1))
    accept_buf = []
    noise_mult = 1.0
    step = 0
    while step < n_steps:
        if deadline is not None and time.time() > deadline:
            break
        scale = (noise_floor + 0.04 * (t / t_init) ** 0.4) * noise_mult
        bl = int(rng.randint(1, max_block + 1))
        start = int(rng.randint(0, max(n - bl, 1)))

        h_new[start:start + bl] += rng.randn(bl) * scale

def sa_spectral(h_init, n, target, n_steps, t_init, t_min, rng,

    for step in range(n_steps):

def _weighted_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-10:
            continue
        jp = np.arange(0, n - k)
        grad[jp] += wk * (1.0 - h[jp + k])
        jn = np.arange(k, n)
        grad[jn] -= wk * h[jn - k]
    grad *= dx
    return grad

def irls_momentum(h_init, n, target, n_outer, beta_sched, lr_init,
                  inner_iters, momentum, deadline=None):
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    v = np.zeros(n)
    for it in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        beta = beta_sched[min(it, len(beta_sched) - 1)]
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        max_o = float(np.max(over))
        weights = np.exp(beta * (over - max_o))
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum
        for _ in range(inner_iters):
            if deadline is not None and time.time() > deadline:
                break
            grad = _weighted_grad(h_cur, n, weights)
            v = momentum * v - lr * grad
            h_new = h_cur + v
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score:
                h_cur = h_new
                cur_score = sc_new
                lr = min(lr * 1.02, 0.008)
            else:
                v *= 0.75
                lr = max(lr * 0.93, 1e-7)
    return h_cur, cur_score

    global_score = compute_c5(best_h)

        h_p = best_h + rng.randn(n) * scale
        h_p = project_feasible(h_p, target)
        starts.append(h_p)
    per_start_budget = None
    if deadline is not None:
        remaining = deadline - time.time()
        per_start_budget = remaining / len(starts)
    for idx in range(len(starts)):
        if deadline is not None and time.time() > deadline:
            break
        dl = None

            max_block=25, noise_floor=0.002)
        if s_res < global_score:
            global_score = s_res
            global_best = h_res
    return global_best, global_score
