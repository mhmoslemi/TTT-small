def compute_influence(h):
    """Compute influence matrix infl[i,k] = change in corr[k] if h[i] increases by 1"""
    n = len(h)
    infl = np.zeros((n, n))
    for i in range(n):
        for k in range(n):
            # h[i]*(1-h[i+k]) contributes if i+k < n
            if i + k < n:
                infl[i, k] += (1 - h[i+k])
            # h[i-k]*(1-h[i]) contributes if i >= k
            if i >= k:
                infl[i, k] -= h[i-k]
    return infl
```

def binary_local_search(h, n_steps, T_init, T_final):
    n = len(h)
    target = n // 2
    h = h.copy().astype(float)
    # Make binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_feasible(h, target)  # should already be feasible if binary
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]  # lags 0 to n-1
    current_max = float(np.max(corr)) * dx
    
    infl = compute_influence(h)
    
    # Get indices of 0s and 1s
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    best_h = h.copy()
    best_score = current_max
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / n_steps)
    
    for step in range(n_steps):
        T = max(T * cooling, T_final)
        
        # Try random pair
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute change vector
        delta = infl[i] - infl[j]
        new_corr = corr + delta
        new_max = float(np.max(new_corr)) * dx
        
        dE = (new_max - current_max) / dx  # in units of correlation sum
        
        if dE < 0 or np.random.random() < np.exp(-dE / T):
            # Accept
            h[i] = 1.0  # was 0
            h[j] = 0.0  # was 1
            corr = new_corr
            current_max = new_max
            
            # Update influence matrix
            # Position i: h[i] went from 0 to 1
            update_influence(infl, h, i)
            # Position j: h[j] went from 1 to 0
            update_influence(infl, h, j)
            
            # Update zero/one sets
            # (this is tricky with arrays; might need to use a different data structure)
            
            if current_max < best_score:
                best_score = current_max
                best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_score
```

        new_max = float(np.max(new_corr))
        
        dE = new_max - current_max
        if dE < 0 or (T > 1e-12 and rng.random() < np.exp(-dE / T)):
            corr = new_corr
            current_max = new_max
            for (pos, d) in flips:
                h[pos] += d
                h[pos] = np.clip(h[pos], 0.0, 1.0)
                update_influence_fast(infl, pos, d, n)
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    
    return best_h, best_max * dx

def slsqp_soft_max(h, alphas, max_iter, ftol, n_active_top=0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        h_tmp = project_feasible(x, target)
        corr = compute_full_corr(h_tmp)
        c_max = float(np.max(corr))
        a = alphas[-1]
        shifted = a * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / a
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h.copy()
    cur_score = evaluate_c5(h_cur)
    
    for a in alphas:
        alphas[-1] = a
        try:
            res = minimize(obj_func_wrapper, h_cur, args=(a, target, n, dx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = evaluate_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
        alphas[-1] = a
    
    return h_cur, cur_score

def obj_func_wrapper(x, alpha, target, n, dx):
    h_tmp = project_feasible(x, target)
    corr = compute_full_corr(h_tmp)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h, alphas, max_iter, ftol):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h.copy()
    cur_score = evaluate_c5(h_cur)
    
    for a in alphas:
        try:
            def obj(x):
                h_tmp = project_feasible(x, target)
                corr = compute_full_corr(h_tmp)
                c_max = float(np.max(corr))
                shifted = a * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted))) / a)
            
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = evaluate_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def round_to_binary(h, target):
    n = len(h)
    h_r = h.copy()
    h_r[h_r >= 0.5] = 1.0
    h_r[h_r < 0.5] = 0.0
    return project_feasible(h_r, target)

def try_rounding(h, target):
    n = len(h)
    best = h.copy()
    best_score = evaluate_c5(h)
    for thresh in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        h_r = h.copy()
        h_r[h_r >= thresh] = 1.0
        h_r[h_r < thresh] = 0.0
        h_r = project_feasible(h_r, target)
        sc = evaluate_c5(h_r)
        if sc < best_score:
            best_score = sc
            best = h_r.copy()
    return best, best_score

def subgradient_all_lags(h, n_steps, n_active, lr, momentum, temp):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    nc = 2 * n - 1
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = evaluate_c5(h_cur)
    velocity = np.zeros(n)
    
    for step in range(n_steps):
        corr = compute_full_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        infl_cached = build_influence_fast(h_cur)
        for wk, L in zip(weights, top_idx):
            L = int(L)
            d = L - (n - 1)
            for i in range(n):
                jd = i - d
                jd2 = i + d
                g = 0.0
                if 0 <= jd <= n - 1:
                    g += (1.0 - h_cur[jd])
                if 0 <= jd2 <= n - 1:
                    g -= h_cur[jd2]
                grad[i] += wk * dx * g
        
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = evaluate_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= 0.995
            if lr < 1e-12:
                break
    
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    n_init = len(initial_h_values)
    target_init = n_init / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target_init)
    
    best_h = h_init.copy()
    best_score = evaluate_c5(best_h)
    
    # Phase 1: SLSQP refinement (all lags)
    if tleft() > 800:
        h_ref, sc = slsqp_refine(best_h, [50, 200, 1000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 2: Round and binary anneal (all lags)
    if tleft() > 600:
        target_cur = len(best_h) / 2.0
        h_bin = round_to_binary(best_h, target_cur)
        sc_bin = evaluate_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
        
        h_ann, sc_ann = binary_anneal(h_bin, 80000, 0.5, 0.01, rng)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann.copy()
        
        # SLSQP on the binary solution
        if tleft() > 500:
            h_ref, sc = slsqp_refine(best_h, [200, 1000, 5000], 2000, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    
    # Phase 3: More binary annealing from refined solution
    if tleft() > 500:
        target_cur = len(best_h) / 2.0
        h_rounded, sc_rounded = try_rounding(best_h, target_cur)
        if sc_rounded < best_score:
            best_score = sc_rounded
            best_h = h_rounded
        
        h_bin = round_to_binary(best_h, target_cur)
        h_ann, sc_ann = binary_anneal(h_bin, 120000, 0.3, 0.005, rng)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann.copy()
    
    # Phase 4: Block moves
    if tleft() > 400:
        target_cur = len(best_h) / 2.0
        h_bin = round_to_binary(best_h, target_cur)
        h_block, sc_block = binary_anneal_block(h_bin, 50000, 0.4, 0.01, rng, block_size=4)
        if sc_block < best_score:
            best_score = sc_block
            best_h = h_block.copy()
    
    # Phase 5: SLSQP refinement
    if tleft() > 300:
        h_ref, sc = slsqp_refine(best_h, [500, 2000, 10000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 6: Another round of binary search + SLSQP
    if tleft() > 250:
        target_cur = len(best_h) / 2.0
        h_rounded, sc_rounded = try_rounding(best_h, target_cur)
        if sc_rounded < best_score:
            best_score = sc_rounded
            best_h = h_rounded
        
        h_bin = round_to_binary(best_h, target_cur)
        h_ann, sc_ann = binary_anneal(h_bin, 100000, 0.2, 0.003, rng)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann.copy()
        
        if tleft() > 150:
            h_ref, sc = slsqp_refine(best_h, [2000, 10000, 50000], 1500, 1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    
    # Phase 7: Try finer grid
    if tleft() > 300:
        n_fine = len(best_h) + 100
        h_fine = interpolate_to(best_h, n_fine)
        target_fine = n_fine / 2.0
        h_ref, sc_fine = slsqp_refine(h_fine, [100, 500, 2000], 1000, 1e-14)
        
        h_bin_fine = round_to_binary(h_ref, target_fine)
        h_ann_fine, sc_ann_fine = binary_anneal(h_bin_fine, 60000, 0.3, 0.01, rng)
        if sc_ann_fine < sc_fine:
            h_fine = h_ann_fine
            sc_fine = sc_ann_fine
        
        h_ref2, sc_fine2 = slsqp_refine(h_fine, [5000, 20000], 1000, 1e-15)
        if sc_fine2 < sc_fine:
            h_fine = h_ref2
            sc_fine = sc_fine2
        
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
    
    # Phase 8: Subgradient with all lags
    if tleft() > 150:
        h_sg, sc_sg = subgradient_all_lags(best_h, 300, 5, 0.003, 0.4, 0.005)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    
    # Phase 9: Final binary anneal
    if tleft() > 100:
        target_cur = len(best_h) / 2.0
        h_bin = round_to_binary(best_h, target_cur)
        h_ann, sc_ann = binary_anneal(h_bin, 100000, 0.1, 0.001, rng)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann.copy()
    
    # Phase 10: Final SLSQP polish
    if tleft() > 50:
        h_ref, sc = slsqp_refine(best_h, [5000, 20000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 11: One more binary pass
    if tleft() > 30:
        target_cur = len(best_h) / 2.0
        h_bin = round_to_binary(best_h, target_cur)
        h_ann, sc_ann = binary_anneal(h_bin, 50000, 0.05, 0.0005, rng)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann.copy()
    
    # Final: ensure feasibility
    final_target = len(best_h) / 2.0
    h_out = project_feasible(best_h, final_target)
    final_score = evaluate_c5(h_out)
    
    if final_score > best_score:
        h_out = project_feasible(best_h, final_target)
        final_score = evaluate_c5(h_out)
    
    return h_out, final_score, len(h_out)
