  = R(k) + dx * 1(j<n-k) * h[j] * (1-h[j+k]) + dx * 1(j>=k) * h[j-k] - dx * 1(j>=k) * h[j-k] * h[j]
  
  = [R(k) + dx * 1(j>=k) * h[j-k]] + h[j] * [dx * 1(j<n-k) * (1-h[j+k]) - dx * 1(j>=k) * h[j-k]]
  
  So O(k) = B(k) + A(k) * h[j]
  
  where:
  B(k) = R(k) + dx * 1(j>=k) * h[j-k]
  A(k) = dx * [1(j<n-k) * (1-h[j+k]) - 1(j>=k) * h[j-k]]
  
  Great! So for each lag k, O(k) is linear in h[j]: O(k) = B(k) + A(k) * h[j]
  
  And we want to minimize max_k O(k) over h[j] ∈ [0, 1].
  
  This is a 1D min-max of linear functions! The optimal h[j] is:
  - If all A(k) = 0: any h[j] works (doesn't affect the max)
  - Otherwise: find h[j] that minimizes max_k (B(k) + A(k) * h[j])
  
  The optimum is at the point where the two "highest" lines cross, or at a boundary (0 or 1).
  
  Specifically, define f_j(t) = max_k (B(k) + A(k) * t) for t ∈ [0,1].
  This is a convex piecewise-linear function of t. Its minimum is at:
  - t = 0 if f is decreasing on [0,1] (i.e., the max is always achieved by a line with negative slope)
  - t = 1 if f is increasing on [0,1]
  - Otherwise, at the intersection of the two lines that are highest at the optimum
  
  To find this efficiently:
  1. Compute f(0) = max_k B(k) and f(1) = max_k (B(k) + A(k))
  2. If f(0) <= f(1) and the argmax at 0 has A < 0, try interior
  3. More robustly: binary search or direct computation
  
  Actually, the simplest approach: since f is convex piecewise linear, its minimum is where the subgradient contains 0. The subgradient at t is {sum of A(k) for k in argmax}. So we want t where some A(k) in the argmax is ≤ 0 and some is ≥ 0 (or all are 0).
  
  A practical approach:
  1. Evaluate f at t = 0, 0.5, 1 to get a rough idea
  2. Use ternary search or golden section on [0,1] (since f is convex)
  3. Or: find the two "critical" lines and their intersection
  
  For n=400, evaluating f(t) = max_k (B(k) + A(k)*t) takes O(n) time. Ternary search with 50 iterations gives ~0.05% precision, which is more than enough. That's 50*400 = 20000 ops per coordinate. For 400 coordinates, that's 8M ops per full sweep. Very fast!

```python
def coordinate_step(h, j, target, n, dx):
    """Optimize coordinate j of h to minimize max_k O(k), keeping sum constraint approximately."""
    # Compute A(k) and B(k) for each k
    A = np.zeros(n)
    B = np.zeros(n)
    
    for k in range(n):
        # O(k) = dx * sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
        # Contribution of h[j]:
        # as first factor: if j < n-k, term is dx * h[j] * (1-h[j+k])
        # as second factor (i+k=j, i=j-k): if j>=k and j-k < n-k (always true since j<n),
        #   term is dx * h[j-k] * (1-h[j])
        
        # Full O(k):
        # R(k) = dx * sum_{i in {0,...,n-1-k} \ {j, j-k}} h[i] * (1-h[i+k])
        
        # Actually let me just compute A and B directly:
        # A(k) = dx * [1(j < n-k) * (1-h[j+k]) - 1(j>=k) * h[j-k]]
        # B(k) = O(k) - A(k) * h[j]  (current value minus h[j]'s contribution)
        
        a_k = 0.0
        if j < n - k:
            a_k += (1.0 - h[j+k])
        if j >= k:
            a_k -= h[j-k]
        A[k] = dx * a_k
        B[k] = get_overlaps(h)[k] - A[k] * h[j]
    
    # Now minimize max_k (B[k] + A[k] * t) for t in [0, 1]
    # Use ternary search (since the max of linear functions is convex)
    
    def f(t):
        return float(np.max(B + A * t))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(80):

            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    t_opt = (a + b) / 2.0
    h[j] = t_opt
    # Rebalance sum
    # (We'll handle the sum constraint separately)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_overlaps(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]


def smooth_max(overlaps, alpha):
    o_max = float(np.max(overlaps))
    shifted = alpha * (overlaps - o_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return o_max + log_sum / alpha


def smooth_max_grad(h, alpha, dx):
    n = len(h)
    overlaps = get_overlaps(h)
    o_max = float(np.max(overlaps))
    shifted = alpha * (overlaps - o_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    total = w.sum()
    if total < 1e-300:
        w = np.ones(n) / n
    else:
        w = w / total
    
    grad = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk < 1e-12:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    sm_val = o_max + float(np.log(total)) / alpha
    return sm_val, grad


def lbfgs_round(h, n, target, alpha, maxiter, penalty_mu, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj_grad(x):
        sm_val, sm_grad = smooth_max_grad(x, alpha, dx)
        sum_diff = np.sum(x) - target
        penalty_val = 0.5 * penalty_mu * sum_diff * sum_diff
        penalty_grad = penalty_mu * sum_diff
        val = sm_val + penalty_val
        grad = sm_grad + penalty_grad
        return val, grad
    
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxfun': maxiter * 3})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        return h_new, sc
    except Exception:
        return h.copy(), compute_c5(h)


def coord_descent_sweep(h, n, target, dx):
    """One full pass of coordinate descent."""
    h_cur = h.copy()
    # Precompute overlaps for efficiency
    for j in range(n):
        overlaps = get_overlaps(h_cur)
        # Compute A(k) and B(k) for coordinate j
        A = np.zeros(n)
        for k in range(n):
            a_k = 0.0
            if j < n - k:
                a_k += (1.0 - h_cur[j + k])
            if j >= k:
                a_k -= h_cur[j - k]
            A[k] = dx * a_k
        B = overlaps - A * h_cur[j]
        
        # Golden section search on [0, 1]
        def f(t):
            return float(np.max(B + A * t))
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, 1.0
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc = f(c)
        fd = f(d)
        for _ in range(50):
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                fc = f(c)
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = f(d)
        
        h_cur[j] = (a + b) / 2.0
    
    # Project to feasibility
    h_cur = project_feasible(h_cur, target)
    return h_cur


def golden_section_minimize(f, a, b, tol=1e-8):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    while abs(b - a) > tol:
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_overlaps(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n - 1:n - 1 + n]


def smooth_max_value(overlaps, alpha):
    o_max = float(np.max(overlaps))
    shifted = alpha * (overlaps - o_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return o_max + log_sum / alpha


def smooth_max_grad_vec(h, alpha, dx):
    n = len(h)
    overlaps = get_overlaps(h)
    o_max = float(np.max(overlaps))
    shifted = alpha * (overlaps - o_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    total = w.sum()
    if total < 1e-300:
        w = np.ones(n) / n
    else:
        w = w / total
    grad = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk < 1e-12:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    sm_val = o_max + float(np.log(total)) / alpha
    return sm_val, grad


def lbfgs_round(h, n, target, alpha, maxiter, penalty_mu, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj_grad(x):
        sm_val, sm_grad = smooth_max_grad_vec(x, alpha, dx)
        sum_diff = np.sum(x) - target
        penalty_val = 0.5 * penalty_mu * sum_diff * sum_diff
        val = sm_val + penalty_val
        grad = sm_grad + penalty_mu * sum_diff
        return val, grad

    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxfun': maxiter * 4})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        return h_new, sc
    except Exception:
        return h.copy(), compute_c5(h)


def coord_descent_sweep(h, n, target, dx, n_golden=35, random_order=False, rng=None):
    h_cur = h.copy()
    overlaps = get_overlaps(h_cur)
    if random_order and rng is not None:
        order = rng.permutation(n)
    else:
        order = range(n)
    for j in order:
        A = np.zeros(n)
        for k in range(n):
            a_k = 0.0
            if j < n - k:
                a_k += (1.0 - h_cur[j + k])
            if j >= k:
                a_k -= h_cur[j - k]
            A[k] = dx * a_k
        B = overlaps - A * h_cur[j]

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, 1.0
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc = float(np.max(B + A * c))
        fd = float(np.max(B + A * d))
        for _ in range(n_golden):
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                fc = float(np.max(B + A * c))
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = float(np.max(B + A * d))
        h_cur[j] = (a + b) / 2.0
    h_cur = project_feasible(h_cur, target)
    return h_cur

def de_diversify(n, target, pop_size, n_gen, rng, deadline, F=0.8, CR=0.85):
    pop = []
    for i in range(pop_size):
        hh = rng.rand(n)
        hh = project_feasible(hh, target)
        pop.append(hh)
    scores = [compute_c5(hh) for hh in pop]
    order = np.argsort(scores)
    pop = [pop[i] for i in order]
    scores = [scores[i] for i in order]
    best = pop[0].copy()
    best_sc = scores[0]
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = project_feasible(trial, target)
            ts = compute_c5(trial)
            if ts <= scores[i]:
                pop[i] = trial
                scores[i] = ts
                if ts < best_sc:
                    best_sc = ts
                    best = trial.copy()
    return best, best_sc


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
