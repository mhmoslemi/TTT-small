import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

# Global state for gradient computation
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_dx = 0.0

def _smooth_obj(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:_g_n - 1 + _g_n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _smooth_grad(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:_g_n - 1 + _g_n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(_g_n)
    weights = exp_s / s
    grad = np.zeros(_g_n)
    for k in range(_g_n):
        w = weights[k]
        if w < 1e-12:
            continue
        jp = np.arange(0, _g_n - k)
        grad[jp] += w * (1.0 - x[jp + k])
        jn = np.arange(k, _g_n)
        grad[jn] -= w * x[jn - k]
    grad *= _g_dx
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_with_grad(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha, _g_dx
    _g_n = n
    _g_target = target
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_dx = 0.0

def _smooth_obj(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:_g_n - 1 + _g_n]
    c_max = np.max(over)
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha

def _smooth_grad(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:_g_n - 1 + _g_n]
    c_max = np.max(over)
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(_g_n)
    weights = exp_s / s
    grad = np.zeros(_g_n)
    for k in range(_g_n):
        w = weights[k]
        if w < 1e-12:
            continue
        jp = np.arange(0, _g_n - k)
        grad[jp] += w * (1.0 - x[jp + k])
        jn = np.arange(k, _g_n)
        grad[jn] -= w * x[jn - k]
    grad *= _g_dx
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_grad(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha, _g_dx
    _g_n = n
    _g_target = target
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def active_set_descent(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):

            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.001)
        weights /= weights.sum()

            cur_score = sc_new

            break

            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)

                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score

def momentum_grad(h, n, target, n_steps, alpha, lr, moment, deadline=None):
    global _g_n, _g_alpha, _g_dx
    _g_n = n
    _g_dx = 2.0 / n

        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

            cur_score = sc_new
        else:
            v *= 0.5
    return h_cur, cur_score

    best_score = compute_c5(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_dx = 0.0

def _full_smooth_obj(x):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha

def _full_smooth_grad(x):
    n = _g_n
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_s / s
    grad = np.zeros(n)
    w0 = weights[n - 1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * x)
    for k in range(1, n):
        wk_pos = weights[n - 1 + k]
        wk_neg = weights[n - 1 - k]
        if wk_pos > 1e-12:
            jn = np.arange(k, n)
            grad[jn] += wk_pos * (1.0 - x[jn - k])
            jp = np.arange(0, n - k)
            grad[jp] -= wk_pos * x[jp + k]
        if wk_neg > 1e-12:
            jp = np.arange(0, n - k)
            grad[jp] += wk_neg * (1.0 - x[jp + k])
            jn = np.arange(k, n)
            grad[jn] -= wk_neg * x[jn - k]
    grad *= _g_dx
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_gradient(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha, _g_dx
    _g_n = n
    _g_target = target
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        for attempt in range(3):
            if deadline is not None and time.time() > deadline:
                break
            try:
                res = minimize(_full_smooth_obj, h_cur, jac=_full_smooth_grad,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
                    break
            except Exception:
                pass
    return h_cur

def grad_projected(h, n, target, n_steps, alpha, lr_init, lr_decay, deadline=None):
    global _g_n, _g_alpha, _g_dx
    _g_n = n
    _g_dx = 2.0 / n
    _g_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    stall = 0
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        grad = _full_smooth_grad(h_cur)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            stall = 0
        else:
            stall += 1
            lr *= lr_decay
            if stall > 50 or lr < 1e-14:
                break
    return h_cur, cur_score

def active_set_descent(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stall = 0
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, idx in zip(weights, top_idx):
            k = int(idx) - (n - 1)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            elif k > 0:
                jn = np.arange(k, n)
                grad[jn] += wk * (1.0 - h_cur[jn - k])
                jp = np.arange(0, n - k)
                grad[jp] -= wk * h_cur[jp + k]
            else:
                ka = -k
                jp = np.arange(0, n - ka)
                grad[jp] += wk * (1.0 - h_cur[jp + ka])
                jn = np.arange(ka, n)
                grad[jn] -= wk * h_cur[jn - ka]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            stall = 0
        else:
            stall += 1
            lr *= lr_decay
            if stall > 30 or lr < 1e-14:
                break
    return h_cur, cur_score

def momentum_grad(h, n, target, n_steps, alpha, lr, moment, deadline=None):
    global _g_n, _g_alpha, _g_dx
    _g_n = n
    _g_dx = 2.0 / n
    _g_alpha = alpha
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        grad = _full_smooth_grad(h_cur)
        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
        else:
            v *= 0.5
    return h_cur, cur_score

def de_search(n, target, pop_size, max_generations, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
