Wait, I realize there might be an issue with the LP formulation. Let me double-check.

We want: maximize the minimum overlap, i.e., max_h min_k c_k(h)

The LP relaxation at point h₀ is:
maximize s
subject to L_k(h; h₀) >= s for all k

where L_k(h; h₀) = c_k(h₀) + ∇c_k(h₀)^T (h - h₀)

Now, ∇c_k(h₀) is the gradient of c_k at h₀. Since c_k(h) = dx * sum_{valid x} h[x]*h[x+k], the gradient is:
∂c_k/∂h[j] = dx * (h₀[j+k] if valid + h₀[j-k] if valid)

So L_k(h; h₀) = c_k(h₀) + dx * sum_j (h₀[j+k]·1_{valid} + h₀[j-k]·1_{valid}) * (h[j] - h₀[j])
= c_k(h₀) + dx * sum_j (h₀[j+k]·1_{valid} + h₀[j-k]·1_{valid}) * h[j] - dx * sum_j (h₀[j+k]·1_{valid} + h₀[j-k]·1_{valid}) * h₀[j]

The last term is: dx * sum_j [h₀[j+k]*h₀[j]*1_{valid} + h₀[j-k]*h₀[j]*1_{valid}]
= dx * [sum_{valid j} h₀[j+k]*h₀[j] + sum_{valid j} h₀[j-k]*h₀[j]]
= c_k(h₀) + c_{-k}(h₀)

Wait, that's not right. Let me be more careful.

c_k(h) = dx * sum_{x: 0<=x<n, 0<=x+k<n} h[x] * h[x+k]

∂c_k/∂h[j] = dx * [h[j+k] (if 0<=j+k<n) + h[j-k] (if 0<=j-k<n)]

So ∇c_k(h₀) is a vector where component j is:
dx * [h₀[j+k] · 1_{0<=j+k<n} + h₀[j-k] · 1_{0<=j-k<n}]

Note that j always satisfies 0<=j<n (since j is an index into h).

So: ∇c_k(h₀)[j] = dx * [h₀[j+k] (if j+k < n) + h₀[j-k] (if j-k >= 0)]

Now, L_k(h; h₀) = c_k(h₀) + ∇c_k(h₀)^T (h - h₀)
= c_k(h₀) + sum_j ∇c_k(h₀)[j] * h[j] - sum_j ∇c_k(h₀)[j] * h₀[j]

The constant term:
sum_j ∇c_k(h₀)[j] * h₀[j] = dx * sum_j [h₀[j+k]·1_{j+k<n} + h₀[j-k]·1_{j-k>=0}] * h₀[j]
= dx * [sum_{j=0}^{n-1-k} h₀[j+k]*h₀[j] + sum_{j=k}^{n-1} h₀[j-k]*h₀[j]]
= dx * [c_k(h₀)/dx + c_k(h₀)/dx]  (since both sums equal the same thing by reindexing)

Wait: sum_{j=0}^{n-1-k} h₀[j+k]*h₀[j] = sum_{i=k}^{n-1} h₀[i]*h₀[i-k] = c_k(h₀)/dx

And sum_{j=k}^{n-1} h₀[j-k]*h₀[j] = same thing = c_k(h₀)/dx

So sum_j ∇c_k(h₀)[j] * h₀[j] = dx * 2 * c_k(h₀)/dx = 2 * c_k(h₀)

Therefore:
L_k(h; h₀) = c_k(h₀) + sum_j ∇c_k(h₀)[j] * h[j] - 2*c_k(h₀)
= sum_j ∇c_k(h₀)[j] * h[j] - c_k(h₀)

So the constraint is: sum_j ∇c_k(h₀)[j] * h[j] - c_k(h₀) >= s

Or: (∇c_k(h₀))^T h >= s + c_k(h₀)

This confirms my earlier formula. The coefficient vector for h in the k-th constraint is ∇c_k(h₀), and the RHS constant is c_k(h₀) (moved to the other side).

Great, so my LP formulation is correct.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_overlaps(h, dx):
    """Compute c_k(h) = dx * sum h[x]*h[x+k] for all k."""
    n = len(h)
    corr = np.correlate(h, h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def build_lp_constraint(h, n, k, dx):
    """Build coefficient vector for the k-th linearized constraint."""
    coeff = np.zeros(n)
    const = 0.0
    if k >= 0:
        nparts = n - k
        coeff[:nparts] += h[k:] * dx
        coeff[k:] += h[:nparts] * dx
        const = dx * np.dot(h[:nparts], h[k:])
    else:
        m = -k
        nparts = n - m
        coeff[m:] += h[:nparts] * dx
        coeff[:nparts] += h[m:] * dx
        const = dx * np.dot(h[m:], h[:nparts])
    return coeff, const


def ccp_lp_method(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30):
    """CCP with LP subproblems (Dinkelbach-style)."""
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        h_var = cp.Variable(n)
        s_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        
        for k in range(-(n-1), n):
            coeff, const = build_lp_constraint(h, n, k, dx)
            constraints.append(coeff @ h_var - const >= s_var)
        
        prob = cp.Problem(cp.Maximize(s_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(np.asarray(h_var.value).ravel(), 0, 1)
        h_lp = project_feasible(h_lp, target)
        
        c5_try = compute_c5(h_lp)
        improved = False
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_lp.copy()
            h = h_lp.copy()
            improved = True
        else:
            # Try damped steps
            for damp in [0.5, 0.25, 0.1, 0.05]:
                h_try = (1-damp)*h + damp*h_lp
                h_try = project_feasible(h_try, target)
                c5_d = compute_c5(h_try)
                if c5_d < best_c5 - 1e-14:
                    best_c5 = c5_d
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def proj_grad_method(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000]
    steps_per_alpha = 2000
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        momentum_v = np.zeros(n)
        beta = 0.9
        stalled = 0
        
        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            
            # Compute smooth gradient
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            top_m = min(50, len(corr))
            top_indices = np.argsort(corr)[-top_m:]
            sub_corr = corr[top_indices]
            shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
            exp_shifted = np.exp(shifted)
            wsum = np.sum(exp_shifted)
            if wsum < 1e-300:
                break
            weights = exp_shifted / wsum
            
            grad = np.zeros(n)
            n_half = n - 1
            for idx_k, w in zip(top_indices, weights):
                if w < 1e-14:
                    continue
                k = idx_k - n_half
                if k == 0:
                    grad += w * (1.0 - 2.0 * h) * dx
                elif k > 0:
                    grad[k:] += w * (1.0 - h[:n - k]) * dx
                    grad[:n - k] -= w * h[k:] * dx
                else:
                    m = -k
                    grad[:n - m] += w * (1.0 - h[m:]) * dx
                    grad[m:] -= w * h[:n - m] * dx
            
            momentum_v = beta * momentum_v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-12:
                break
            
            lr = min(1.0 / (alpha * gnorm + 1e-10), 1.0)
            
            improved = False
            for _ in range(30):
                h_try = project_feasible(h - lr * momentum_v, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
                lr *= 0.5
            
            if not improved:
                stalled += 1
                if stalled > 30:
                    break
    
    return best_h, best_score

def project_feasible(h, target_sum):

            return h
        h = h.copy()

            c_max = np.max(corr)
            top_m = min(30, len(corr))

            sub_corr = corr[top_indices]
            shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
            exp_shifted = np.exp(shifted)
            wsum = np.sum(exp_shifted)
            if wsum < 1e-300:

                    h = h_try
                    improved = True
                    break

        prob = cp.Problem(cp.Maximize(s_var), constraints)

            prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_lp = np.clip(np.asarray(h_var.value).ravel(), 0, 1)
        h_lp = project_feasible(h_lp, target)

        c5_try = compute_c5(h_lp)
        improved = False
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_lp.copy()
            h = h_lp.copy()
            improved = True
        else:
            for damp in [0.5, 0.25, 0.1]:
                h_try = (1.0 - damp) * h + damp * h_lp
                h_try = project_feasible(h_try, target)
                c5_d = compute_c5(h_try)
                if c5_d < best_c5 - 1e-14:
                    best_c5 = c5_d
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, deadline):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def proj_grad_method(h_init, n, target, deadline):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    steps_per_alpha = 3000

    for alpha in alpha_schedule:
        if time.time() > deadline:
            break

        momentum_v = np.zeros(n)
        beta = 0.9
        stalled = 0

        for step in range(steps_per_alpha):
            if time.time() > deadline:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            top_m = min(30, len(corr))
            top_indices = np.argsort(corr)[-top_m:]
            sub_corr = corr[top_indices]
            shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
            exp_shifted = np.exp(shifted)
            wsum = np.sum(exp_shifted)
            if wsum < 1e-300:
                break
            weights = exp_shifted / wsum

            grad = np.zeros(n)
            n_half = n - 1
            for idx_k, w in zip(top_indices, weights):
                if w < 1e-14:
                    continue
                k = idx_k - n_half
                if k == 0:
                    grad += w * (1.0 - 2.0 * h) * dx
                elif k > 0:
                    grad[k:] += w * (1.0 - h[:n - k]) * dx
                    grad[:n - k] -= w * h[k:] * dx
                else:
                    m = -k
                    grad[:n - m] += w * (1.0 - h[m:]) * dx
                    grad[m:] -= w * h[:n - m] * dx

            momentum_v = beta * momentum_v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-12:
                break

            lr = min(1.0 / (alpha * gnorm + 1e-10), 1.0)

            improved = False
            for _ in range(30):
                h_try = project_feasible(h - lr * momentum_v, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
                lr *= 0.5

            if not improved:
                stalled += 1
                if stalled > 40:
                    break

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    global_deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    phase_dur = budget_s * 0.22
    d1 = start_time + phase_dur
    h_ccp, c5_ccp = ccp_lp_method(h_init, n, target, max_iter=200, deadline=d1)
    if c5_ccp < best_score:
        best_score = c5_ccp
        best_h = h_ccp

    if time.time() < global_deadline - 180:
        d2 = min(start_time + budget_s * 0.35, global_deadline - 180)
        h_slq = slsqp_optimize(best_h, n, target,
                               [100, 500, 2000, 10000, 50000],
                               maxiter=2000, ftol=1e-16, deadline=d2)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    if time.time() < global_deadline - 150:
        d3 = min(start_time + budget_s * 0.45, global_deadline - 150)
        h_pg, c5_pg = proj_grad_method(best_h, n, target, deadline=d3)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    if time.time() < global_deadline - 150:
        d4 = min(start_time + budget_s * 0.52, global_deadline - 150)
        h_ccp, c5_ccp = ccp_lp_method(best_h, n, target, max_iter=100, deadline=d4)
        if c5_ccp < best_score:
            best_score = c5_ccp
            best_h = h_ccp

    if time.time() < global_deadline - 350:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        d5 = min(start_time + budget_s * 0.70, global_deadline - 350)
        h_ccp, c5_ccp = ccp_lp_method(h_interp, n_try, target_try,
                                       max_iter=100, deadline=d5)
        if time.time() < global_deadline - 120:
            d6 = min(start_time + budget_s * 0.78, global_deadline - 120)
            h_ccp = slsqp_optimize(h_ccp, n_try, target_try,
                                   [200, 1000, 5000],
                                   maxiter=1500, ftol=1e-16, deadline=d6)
            c5_ccp = compute_c5(h_ccp)
        if c5_ccp < best_score:
            best_score = c5_ccp
            best_h = h_ccp
            n = n_try
            target = target_try

    if time.time() < global_deadline - 300 and len(best_h) <= 600:
        n_try = 800
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        d7 = min(start_time + budget_s * 0.85, global_deadline - 300)
        h_ccp, c5_ccp = ccp_lp_method(h_interp, n_try, target_try,
                                       max_iter=60, deadline=d7)
        if time.time() < global_deadline - 80:
            d8 = min(start_time + budget_s * 0.91, global_deadline - 80)
            h_ccp = slsqp_optimize(h_ccp, n_try, target_try,
                                   [500, 2000],
                                   maxiter=1000, ftol=1e-16, deadline=d8)
            c5_ccp = compute_c5(h_ccp)
        if c5_ccp < best_score:
            best_score = c5_ccp
            best_h = h_ccp
            n = n_try
            target = target_try

    n_use = len(best_h)
    target_use = n_use / 2.0

    if time.time() < global_deadline - 40:
        h_fin = slsqp_optimize(best_h, n_use, target_use,
                               [5000, 20000, 100000],
                               maxiter=3000, ftol=1e-16, deadline=global_deadline - 20)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    if time.time() < global_deadline - 10:
        n_use = len(best_h)
        h_pg, c5_pg = proj_grad_method(best_h, n_use, n_use / 2.0, deadline=global_deadline)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
