    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]

        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h_fixed)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    return prob, h_var, t_var

    # Use random Fourier coefficients to create a smooth pattern
    freqs = rng.randint(1, n // 4, size=n_frequencies)
    phases = rng.uniform(0, 2 * np.pi, size=n_frequencies)
    amps = rng.uniform(0.3, 0.7, size=n_frequencies)
    
    x = np.arange(n) / n
    val = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        val += a * np.sin(2 * np.pi * f * x + p)
    
    # Threshold to get binary-like
    h = (val > 0).astype(float)
    
    # Fix sum
    target = n / 2.0
    while h.sum() > target:
        idx = rng.choice(np.where(h == 1)[0])
        h[idx] = 0
    while h.sum() < target:
        idx = rng.choice(np.where(h == 0)[0])
        h[idx] = 1
    
    return h


def block_init(n, rng, n_blocks=4):
    """Generate a block-structured initialization."""
    h = np.zeros(n)
    block_size = n // (2 * n_blocks)
    for i in range(n_blocks):
        start = i * 2 * block_size
        h[start:start + block_size] = 1.0
    # Shuffle blocks with rng

    h = np.zeros(n)
    n_ones = n // 2
    positions = rng.choice(n, size=n_ones, replace=False)
    h[positions] = 1.0
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
