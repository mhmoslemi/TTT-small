            if m > 0:
                grad[k:] += w * (1.0 - h[:m]) * dx
                grad[:m] -= w * h[k:] * dx
        else:
            m = -k
            if m > 0:
                grad[:m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:m] * dx
    return grad


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _cons_fun(h):
    return np.sum(h) - _state['target']


def _cons_jac(h):
    return np.ones(_state['n'])


def _slsqp_obj(h):
    n = len(h)
    alpha = _state['alpha']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def alternating_lp(h_init, n, target, top_k, max_iters, deadline):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = get_corr(h) * dx
        c_max = np.max(corr)
        idx = np.argsort(corr)[-top_k:]
        shifts = idx - (n - 1)
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for k in shifts:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var + 1e-4 * cp.sum_squares(h_var - h)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=8000, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.4 * h + 0.6 * h_new
        h = project_feasible(h, target)
        sc = compute_c5(h)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    return best_h, best_c5


def projected_grad(h_init, n, target, deadline, schedule):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    for k_top, beta, lr_scale, n_steps, gap_tol in schedule:
        if time.time() > deadline:
            break
        v = np.zeros(n)
        stall = 0
        for step in range(n_steps):
            if time.time() > deadline:
                break
            corr = get_corr(h) * dx
            c_max = np.max(corr)
            top_indices = np.argsort(corr)[-k_top:]
            top_shifts = top_indices - (n - 1)
            gaps = np.maximum(corr[top_indices] - (c_max - gap_tol), 0.0)
            if gaps.sum() < 1e-14:
                weights = np.ones(k_top) / k_top
            else:
                weights = gaps / gaps.sum()
            g = weighted_shift_grad(h, top_shifts, dx, weights)
            v = beta * v + g
            gnorm = np.max(np.abs(v))
            if gnorm < 1e-13:
                break
            lr = lr_scale / (gnorm + 1e-10)
            improved = False
            for _ in range(25):
                h_try = project_feasible(h - lr * v, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stall = 0
                    break
                lr *= 0.5
            if not improved:
                stall += 1
                if stall > 50:
                    break
    return best_h, best_score


def slsqp_polish(h_init, n, target, alphas, deadline, maxiter=300):
    _state['n'] = n
    _state['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h = h_init.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        _state['alpha'] = alpha
        try:
            res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h) - 1e-15:
                    h = h_new
        except Exception:
            pass
    return h


def steepest_subgrad_step(h, n, target, dx, k_top, gap_tol):
    corr = get_corr(h) * dx
    c_max = np.max(corr)
    top_indices = np.argsort(corr)[-k_top:]
    top_shifts = top_indices - (n - 1)
    gaps = np.maximum(corr[top_indices] - (c_max - gap_tol), 0.0)
    if gaps.sum() < 1e-14:
        weights = np.ones(k_top) / k_top
    else:
        weights = gaps / gaps.sum()
    g = weighted_shift_grad(h, top_shifts, dx, weights)
    return g, c_max


def projected_grad_v2(h_init, n, target, deadline):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    phases = [
        (80, 0.80, 1.0, 0.02, 400),
        (40, 0.85, 0.7, 0.01, 400),
        (25, 0.85, 0.5, 0.005, 400),
        (15, 0.90, 0.4, 0.003, 400),
        (10, 0.90, 0.3, 0.002, 400),
        (7, 0.90, 0.2, 0.001, 400),
        (5, 0.90, 0.15, 0.0005, 400),
        (3, 0.88, 0.1, 0.0002, 400),
        (2, 0.85, 0.06, 0.0001, 400),
        (1, 0.85, 0.03, 0.0, 400),
    ]

    for k_top, beta, lr_scale, gap_tol, n_steps in phases:
        if time.time() > deadline:
            break
        v = np.zeros(n)
        stall = 0
        for step in range(n_steps):
            if time.time() > deadline:
                break
            g, c_max = steepest_subgrad_step(h, n, target, dx, k_top, gap_tol)
            v = beta * v + g
            gnorm = np.max(np.abs(v))
            if gnorm < 1e-13:
                break
            lr = lr_scale / (gnorm + 1e-10)
            improved = False
            for _ in range(25):
                h_try = project_feasible(h - lr * v, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stall = 0
                    break
                lr *= 0.5
            if not improved:
                stall += 1
                if stall > 40:
                    break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Phase 1: LP with top constraints (focus on most active)
    if tleft() > 200:
        deadline = start_time + min(140, tleft() * 0.22)
        for tk in [20, 15]:
            if time.time() > deadline:
                break
            h_lp, c5_lp = alternating_lp(best_h, n, target, top_k=tk,
                                         max_iters=2000, deadline=deadline)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    # Phase 2: Main gradient optimization
    if tleft() > 120:
        deadline = start_time + min(380, tleft() * 0.42)
        h_pg, c5_pg = projected_grad_v2(best_h, n, target, deadline=deadline)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 3: SLSQP polish
    if tleft() > 80:
        deadline = start_time + min(120, tleft() * 0.25)
        h_slq = slsqp_polish(best_h, n, target,
                             [100, 500, 2000, 10000, 50000], deadline=deadline, maxiter=400)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 4: Second gradient round with tighter schedule
    if tleft() > 100:
        deadline = start_time + min(200, tleft() * 0.35)
        h_pg, c5_pg = projected_grad_v2(best_h, n, target, deadline=deadline)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 5: Small random restarts
    for noise in [0.003, 0.007, 0.012, 0.02, 0.035]:
        if tleft() < 35:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        deadline = start_time + min(50, tleft() * 0.4)
        h_pg, c5_pg = projected_grad_v2(h_pert, n, target, deadline=deadline)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg
        if tleft() > 35 and c5_pg < best_score + 0.002:
            deadline2 = start_time + min(25, tleft() * 0.3)
            h_slq = slsqp_polish(h_pg, n, target, [5000, 20000], deadline=deadline2, maxiter=200)
            sc = compute_c5(h_slq)
            if sc < best_score:
                best_score = sc
                best_h = h_slq

    # Phase 6: Higher resolution attempts
    if tleft() > 100:
        for n_try in [480, 560, 640]:
            if tleft() < 70:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            deadline = start_time + min(55, tleft() * 0.45)
            h_pg, c5_pg = projected_grad_v2(h_interp, n_try, target_try, deadline=deadline)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg

    # Phase 7: Final SLSQP at best resolution
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        deadline = start_time + min(25, tleft() * 0.6)
        h_fin = slsqp_polish(best_h, n_use, target_use, [10000, 50000],
                             deadline=deadline, maxiter=200)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility enforcement
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
