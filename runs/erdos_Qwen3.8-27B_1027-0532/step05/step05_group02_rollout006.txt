def gradient_smoothed_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = c_max + np.log(sum_exp) / alpha
    w = exp_shifted / sum_exp

    grad = np.zeros(n)
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)

    for li in range(n_lags):
        wl = w[li]
        if wl < 1e-18:
            continue
        lag = lags[li]
        if lag >= 0:
            grad[:n - lag] += wl * (1.0 - h[lag:])
            grad[lag:] -= wl * h[:n - lag]
        else:
            grad[-lag:] += wl * (1.0 - h[:n + lag])
            grad[:n + lag] -= wl * h[-lag:]

    return obj, grad

    return obj + penalty, grad

                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def multi_start_continuous(h_start, n_starts, alphas, maxiter_per, time_limit, start_time):
    n = len(h_start)
    best_h = h_start.copy()
    best_score = compute_c5(h_start)
    target = n / 2.0

    for s in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        if s == 0:
            h_init = h_start.copy()
        else:
            h_init = best_h.copy()
            n_pert = rng.randint(2, 6)
            for _ in range(n_pert):
                i = rng.randint(n)
                if h_init[i] > 0.5:
                    h_init[i] = max(0.0, h_init[i] - 0.1)
                else:
                    h_init[i] = min(1.0, h_init[i] + 0.1)
            h_init = project_feasible(h_init, target)

        h_res, score = slsqp_with_gradient(h_init, alphas, maxiter_per, time_limit, start_time)
        if score < best_score:
            best_score = score
            best_h = h_res.copy()

    return best_h, best_score

def correlated_sa_move(h, full, dx, rng, positions):
    n = len(h)
    top_k = 3
    top_indices = np.argsort(full)[-top_k:]
    k_star = top_indices[rng.randint(top_k)] - (n - 1)
    if k_star == 0:
        k_star = 1 + rng.randint(n - 1)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    A = np.zeros(n)
    idx = positions + k_star
    valid = (idx >= 0) & (idx < n)
    A[valid] += h[idx[valid]]
    idx = positions - k_star
    valid = (idx >= 0) & (idx < n)
    A[valid] += h[idx[valid]]
    B = np.zeros(n)
    idx = positions - k_star
    valid = (idx >= 0) & (idx < n)
    B[valid] += h[idx[valid]]
    idx = positions + k_star
    valid = (idx >= 0) & (idx < n)
    B[valid] += h[idx[valid]]
    i_cands = ones[A[ones] <= 1.0]
    j_cands = zeros[B[zeros] >= 1.0]
    if len(i_cands) > 0 and len(j_cands) > 0:
        i = i_cands[rng.randint(len(i_cands))]
        j = j_cands[rng.randint(len(j_cands))]
    else:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
    return i, j

def sa_correlated(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fft(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i, j = correlated_sa_move(h, full, dx, rng, positions)
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy to improve binary starting point
    allowed = min(60, tleft() * 0.08)
    h_g, c5_g = fast_greedy(best_h, max_iters=30000, rng=rng,
                            time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Correlated SA for binary improvement
    allowed = min(150, tleft() * 0.15)
    n_sa = int(min(3000000, allowed * 35000))
    h_sa, c5_sa = sa_correlated(best_h, n_sa, 0.015, 0.00005, rng,
                                time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Continuous optimization (main workhorse)
    # Multi-round SLSQP with increasing alpha
    n_use = len(best_h)
    target_use = float(n_use) / 2.0

    alphas_round1 = [10, 50, 200, 500]
    allowed = min(200, tleft() * 0.25)
    h_cont, sc_cont = slsqp_with_gradient(best_h, alphas_round1, 1500,
                                          time_limit=allowed, start_time=start_time)
    if sc_cont < best_score:
        best_score = sc_cont
        best_h = h_cont.copy()

    # Phase 4: More aggressive continuous optimization
    alphas_round2 = [1000, 3000, 10000]
    allowed = min(200, tleft() * 0.20)
    h_cont2, sc_cont2 = slsqp_with_gradient(best_h, alphas_round2, 2000,
                                            time_limit=allowed, start_time=start_time)
    if sc_cont2 < best_score:
        best_score = sc_cont2
        best_h = h_cont2.copy()

    # Phase 5: L-BFGS-B with penalty (different optimizer, may find different minima)
    allowed = min(100, tleft() * 0.12)
    h_lb, sc_lb = lbfgs_continuous(best_h, alpha=500, mu=100.0, maxiter=500,
                                   time_limit=allowed, start_time=start_time)
    if sc_lb < best_score:
        best_score = sc_lb
        best_h = h_lb.copy()

    # Refine L-BFGS-B result with SLSQP
    if sc_lb < best_score + 0.001:
        allowed = min(60, tleft() * 0.05)
        h_lb2, sc_lb2 = slsqp_with_gradient(h_lb, [2000, 5000], 1000,
                                            time_limit=allowed, start_time=start_time)
        if sc_lb2 < best_score:
            best_score = sc_lb2
            best_h = h_lb2.copy()

    # Phase 6: Binary SA from best continuous solution
    if tleft() > 60:
        h_bin = project_binary(best_h, target_init)
        allowed = min(120, tleft() * 0.15)
        n_sa2 = int(min(4000000, allowed * 30000))
        h_sa2, c5_sa2 = sa_correlated(h_bin, n_sa2, 0.012, 0.00003, rng,
                                      time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 7: Another round of continuous from the best binary
    if tleft() > 60:
        allowed = min(120, tleft() * 0.15)
        alphas_final = [500, 2000, 5000, 10000]
        h_final_cont, sc_final = slsqp_with_gradient(best_h, alphas_final, 2000,
                                                     time_limit=allowed, start_time=start_time)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_cont.copy()

    # Phase 8: Multi-start continuous (try different perturbations)
    if tleft() > 80:
        allowed = min(tleft() - 30, 100)
        n_starts = max(2, int(allowed / 30))
        h_ms, sc_ms = multi_start_continuous(best_h, n_starts,
                                             [500, 2000, 5000], 800,
                                             time_limit=allowed, start_time=start_time)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms.copy()

    # Phase 9: Final SLSQP polish with very high alpha
    if tleft() > 20:
        allowed = max(5.0, tleft() - 10)
        h_polish, sc_polish = slsqp_with_gradient(best_h, [5000, 10000, 20000], 1000,
                                                  time_limit=allowed, start_time=start_time)
        if sc_polish < best_score:
            best_score = sc_polish
            best_h = h_polish.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
