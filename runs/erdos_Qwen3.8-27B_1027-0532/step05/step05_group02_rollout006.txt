import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N, dtype=np.float64)
    a[:n] = h
    b = np.zeros(N, dtype=np.float64)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_gradient_corr(h, weights):
    """Compute gradient of weighted sum of correlations.
    
    grad[i] = 1 - S_plus[i] - S_minus[i]
    where S_plus[i] = sum_k weights[k] * h[i + lag_k]
    and S_minus[i] = sum_k weights[k] * h[i - lag_k]
    
    lag_k = k - (n-1) for k in 0..2n-2
    """

    # Let's define w_rev[n-1-l] = w[n-1+l] for l in -(n-1)..(n-1)
    # Then S_plus[i] = sum_l w_rev[n-1-l] * h[i+l]
    # = (w_rev * h)[2n-2-i] ... hmm this is getting confusing
    
    # Let me think differently.
    # S_plus[i] = sum_{l: i+l in [0,n-1]} w[n-1+l] * h[i+l]
    # Let j = i+l, so l = j-i
    # S_plus[i] = sum_{j: j in [0,n-1], j-i in [-(n-1), n-1]} w[n-1+j-i] * h[j]
    # = sum_j h[j] * w[n-1+(j-i)]

    # Then S_plus[i] = sum_j h[j] * w_shifted[j-i]
    # This is the cross-correlation (h ⊛ w_shifted)[i]

    N = 1

    # In FFT: ifft(fft(h) * conj(fft(w_shifted)))

    # w_shifted[0] = w[n-1] (lag 0)
    # w_shifted[1] = w[n] (lag +1)
    # ...

    # ...
    # w_shifted[N-1] = w[n-2] (lag -1)

    w_shifted[0:n] = w[n-1:]  # lags 0 to n-1

    return 1.0 - S_plus - S_minus

def smooth_objective_and_grad(h, alpha):
    """Compute smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fast(h)
    max_corr = float(np.max(corr))
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    obj = (max_corr + log_sum_exp / alpha) * dx
    
    # Softmax weights
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = compute_gradient_corr(h, weights) * dx
    
    return obj, grad

def project_simplex(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}."""
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-14:
        return h
    
    # Bisection on lambda
    # h_i(lambda) = clip(h_i - lambda, 0, 1)
    # We need sum(h_i(lambda)) = target_sum
    # This is monotonically decreasing in lambda
    
    lo = -max(h) - 1.0
    hi = 1.0 - min(h) + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(h - mid, 0.0, 1.0)
        s_mid = np.sum(h_mid)
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    return h_proj

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())

        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum

            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

        else:
            zeros2 = np.where(h_r == 0)[0]

def compute_gradient_direct(h, weights):
    n = len(h)
    # weights has length 2n-1, indexed 0 to 2n-2
    # lag l = k - (n-1) for k in 0..2n-2
    
    # S_plus[i] = sum_l weights[n-1+l] * h[i+l]
    # S_minus[i] = sum_l weights[n-1+l] * h[i-l]
    
    S_plus = np.zeros(n)
    S_minus = np.zeros(n)
    
    for l in range(-(n-1), n):  # l from -(n-1) to n-1
        w_val = weights[n-1+l]
        if w_val < 1e-15:
            continue
        # S_plus[i] += w_val * h[i+l] for i+l in [0, n-1]
        # i ranges from max(0, -l) to min(n-1, n-1-l)
        if l >= 0:
            i_start = 0
            i_end = n - l
        else:
            i_start = -l
            i_end = n
        S_plus[i_start:i_end] += w_val * h[i_start:i_end]
        
        # S_minus[i] += w_val * h[i-l] for i-l in [0, n-1]
        # i ranges from max(0, l) to min(n-1, n-1+l)
        if l >= 0:
            i_start = l
            i_end = n
        else:
            i_start = 0
            i_end = n + l
        S_minus[i_start:i_end] += w_val * h[i_start:i_end]

```python
def compute_grad_direct(h, weights):
    n = len(h)
    S_plus = np.zeros(n)
    S_minus = np.zeros(n)
    for l in range(-(n-1), n):
        w_val = weights[n-1+l]
        if abs(w_val) < 1e-18:
            continue
        if l >= 0:
            S_plus[:n-l] += w_val * h[l:]
            S_minus[l:] += w_val * h[:n-l]
        else:
            S_plus[-l:] += w_val * h[:n+l]
            S_minus[:n+l] += w_val * h[-l:]
    return 1.0 - S_plus - S_minus
```
