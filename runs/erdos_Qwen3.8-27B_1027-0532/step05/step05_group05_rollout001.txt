import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lse_objective(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def threshold_objective(h, t, lam):
    corr = compute_corr(h)
    excess = np.maximum(corr - t, 0.0)
    return t + lam * float(np.sum(excess ** 2))

def lbfgsb_lse(h, n, target, alpha, maxiter, deadline):
    def obj(x):
        h_x = project_feasible(x, target)
        return lse_objective(h_x, alpha)
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def slsqp_lse(h, n, target, alpha, maxiter, ftol, deadline):
    def obj(x):
        return lse_objective(x, alpha)
    def cons_eq(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def threshold_optimize(h, n, target, n_iter, deadline, rng):
    h_cur = h.copy()
    t_init = float(np.max(compute_corr(h_cur)))
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lam = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        t = float(np.percentile(corr, 95.0))
        for _ in range(3):
            def obj(x):
                h_x = np.clip(x, 0.0, 1.0)
                hx = project_feasible(h_x, target)
                corr_x = compute_corr(hx)
                excess = np.maximum(corr_x - t, 0.0)
                return t + lam * float(np.sum(excess ** 2))
            try:
                res = minimize(obj, h_cur, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 500, 'ftol': 1e-15})
                h_cur = project_feasible(res.x, target)
            except Exception:
                break
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        lam *= 1.3
    return best_h, best_score

def binary_local_search(h_bin, n, target, max_rounds, deadline, rng):
    h = h_bin.copy().astype(float)
    n_ones = int(round(target))
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_score = compute_c5(h)
    best_h = h.copy()
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        max_c = float(np.max(corr))
        improved = False
        n_ones_list = ones.tolist()
        n_zeros_list = zeros.tolist()
        rng.shuffle(n_ones_list)
        rng.shuffle(n_zeros_list)
        max_swaps = min(len(n_ones_list) * len(n_zeros_list), 2000)
        swapped = 0
        for a in n_ones_list:
            if swapped >= max_swaps:
                break
            for b in n_zeros_list:
                if swapped >= max_swaps or abs(a - b) < 1:
                    continue
                h[a] = 0.0
                h[b] = 1.0
                corr_new = compute_corr(h)
                sc_new = float(np.max(corr_new))
                h[a] = 1.0
                h[b] = 0.0
                swapped += 1
                if sc_new < max_c - 1e-10:
                    h[a] = 0.0
                    h[b] = 1.0
                    ones = ones[(ones != a)]
                    zeros = zeros[(zeros != b)]
                    ones = np.append(ones, b)
                    zeros = np.append(zeros, a)
                    n_ones_list = ones.tolist()
                    n_zeros_list = zeros.tolist()
                    max_c = sc_new
                    improved = True
                    break
        if not improved:
            break
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score

def multi_start_binary(h_init, n, target, n_starts, deadline, rng):
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    h_bin = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h_bin > 0.5))
    target_ones = int(round(target))
    if n_ones != target_ones:
        for _ in range(abs(n_ones - target_ones)):
            if n_ones > target_ones:
                idx = rng.choice(np.where(h_bin > 0.5)[0])
                h_bin[idx] = 0
                n_ones -= 1
            else:
                idx = rng.choice(np.where(h_bin < 0.5)[0])
                h_bin[idx] = 1
                n_ones += 1
    for s in range(n_starts):
        if time.time() > deadline:
            break
        if s == 0:
            h_s = h_bin.copy()
        else:
            h_s = h_bin.copy()
            n_flips = rng.randint(5, 20)
            ones = np.where(h_s > 0.5)[0]
            zeros = np.where(h_s < 0.5)[0]
            flip_ones = rng.choice(ones, size=min(n_flips, len(ones)), replace=False)
            flip_zeros = rng.choice(zeros, size=min(n_flips, len(zeros)), replace=False)
            h_s[flip_ones] = 0
            h_s[flip_zeros] = 1
        h_s, sc_s = binary_local_search(h_s, n, target, 50, min(time.time() + 30, deadline), rng)
        h_relax = project_feasible(h_s * 0.99 + 0.01 * 0.5, target)
        h_relax = slsqp_lse(h_relax, n, target, 200, 1000, 1e-14, min(time.time() + 20, deadline))
        sc_relax = compute_c5(h_relax)
        if sc_relax < best_score:
            best_score = sc_relax
            best_h = h_relax.copy()
    return best_h, best_score

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stale = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
            stale = 0
        else:
            stale += 1
            if stale > 200:
                break
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k_idx in zip(weights, top_idx):
            k_full = int(k_idx)
            if k_full < n - 1:
                k = k_full - (n - 1)
            else:
                k = k_full - (n - 1)
            if k >= 0:
                j_pos = np.arange(0, n - k)
                grad[j_pos + k] += wk * (1.0 - h_cur[j_pos])
                grad[j_pos] -= wk * h_cur[j_pos + k]
            else:
                k_abs = -k
                j_pos = np.arange(0, n - k_abs)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k_abs])
                grad[j_pos + k_abs] -= wk * h_cur[j_pos]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def adversarial_lp(h, n, target, deadline):
    from scipy.optimize import linprog
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    corr = compute_corr(h_cur)
    max_val = float(np.max(corr))
    t = max_val * 0.999
    for it in range(50):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        active = np.where(corr > t * 0.99)[0]
        if len(active) == 0:
            break
        def obj(x):
            hx = np.clip(x, 0.0, 1.0)
            hx = project_feasible(hx, target)
            corr_x = compute_corr(hx)
            excess = np.maximum(corr_x[active] - t, 0.0)
            return float(np.sum(excess ** 2))
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_cur = project_feasible(res.x, target)
        except Exception:
            break
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        t = float(np.max(compute_corr(h_cur)))
        if t < max_val * 0.98:
            break
    return best_h, best_score
