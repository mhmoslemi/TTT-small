def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, m_idx in zip(weights, top_idx):
            m_idx = int(m_idx)
            lag = m_idx - (n - 1)
            if lag >= 0:
                j_pos = np.arange(lag, n)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos - lag])
                j_neg = np.arange(0, n - lag)
                grad[j_neg] -= wk * h_cur[j_neg + lag]
            else:
                l = -lag
                j_pos = np.arange(0, n - l)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + l])
                j_neg = np.arange(l, n)
                grad[j_neg] -= wk * h_cur[j_neg - l]
        
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    bounds = [(0.0, 1.0)] * n
    
    def _cons_eq(x):
        return np.sum(x) - target
    
    def _cons_jac(x):
        return np.ones(n)
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if time.time() > deadline:
            break
        
        def _obj(x, alpha=a):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur

            grad[j_pos] += (1.0 - h_cur[j_pos + l])
            j_neg = np.arange(l, n)
            grad[j_neg] -= h_cur[j_neg - l]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.05)
        else:
            eps *= 0.5
            if eps < 1e-10:
                break
    return h_cur, cur_score


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break

        def _obj(x, alpha=a):
            corr = compute_corr(x)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        def _cons_eq(x):
            return np.sum(x) - target

        def _cons_jac(x):
            return np.ones(n)

        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def de_optimize(n, target, pop_size, f_factor, cr, deadline, rng, init_pop=None):
    x = np.linspace(0, 2, n)
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        for i in range(pop_size):
            r = rng.rand()
            if r < 0.3:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r < 0.5:
                freq = rng.randint(1, 5)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r < 0.7:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            else:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.25, target)
            pop[i] = project_feasible(pop[i], target)

    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    while time.time() < deadline:
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 10.0

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    x_orig = np.linspace(0.0, 1.0, n_final)
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return hard_deadline - time.time()

    # Phase 1: Adam with alpha annealing at full resolution
    alpha_schedule_1 = [100, 300, 800, 2000, 5000, 12000]
    lr_schedule_1 = [0.02, 0.015, 0.01, 0.008, 0.006, 0.004]
    for i, a in enumerate(alpha_schedule_1):
        if tleft() < 80:
            break
        dl = min(start_time + 60 * (i + 1), hard_deadline)
        h_adam, sc = adam_optimize(best_h, n_final, target_final, a, 5000, lr_schedule_1[i], dl)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    # Phase 2: Subgradient polish
    if tleft() > 100:
        dl = min(start_time + 400, hard_deadline)
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final, 2000, 5, 0.004, 0.998, dl)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 3: Worst lag surgery
    if tleft() > 60:
        dl = min(start_time + 470, hard_deadline)
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final, 400, 0.003, dl)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 4: Low-resolution DE search
    n1 = 40
    t1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, t1)
    if tleft() > 120:
        pop1 = [h1.copy()]
        for _ in range(19):
            pop1.append(project_feasible(rng.rand(n1) * 0.6 + 0.2, t1))
        dl1 = min(start_time + 580, hard_deadline)
        h1_de, sc1 = de_optimize(n1, t1, 20, 0.8, 0.7, dl1, rng, init_pop=pop1)
        h1_adam, sc1a = adam_optimize(h1_de, n1, t1, 500, 3000, 0.05, min(start_time + 620, hard_deadline))
        h1_up = interpolate_to(h1_adam, n_final, target_final)
        sc1_up = compute_c5(h1_up)
        if sc1_up < best_score:
            best_score = sc1_up
            best_h = h1_up

    # Phase 5: Mid-resolution DE
    n2 = 160
    t2 = n2 / 2.0
    x2 = np.linspace(0.0, 1.0, n2)
    h2_init = project_feasible(np.interp(x2, x_orig, best_h), t2)
    if tleft() > 150:
        pop2 = [h2_init.copy()]
        for _ in range(14):
            p = h2_init + rng.randn(n2) * rng.uniform(0.03, 0.15)
            pop2.append(project_feasible(p, t2))
        dl2 = min(start_time + 750, hard_deadline)
        h2_de, sc2 = de_optimize(n2, t2, 15, 0.65, 0.55, dl2, rng, init_pop=pop2)
        h2_adam, sc2a = adam_optimize(h2_de, n2, t2, 500, 3000, 0.03, min(start_time + 800, hard_deadline))
        h2_up = project_feasible(np.interp(x_orig, x2, h2_adam), target_final)
        sc2_up = compute_c5(h2_up)
        if sc2_up < best_score:
            best_score = sc2_up
            best_h = h2_up

    # Phase 6: SLSQP at full resolution
    if tleft() > 100:
        dl = min(start_time + 920, hard_deadline)
        h_ref = slsqp_optimize(best_h, n_final, target_final, [500, 2000, 8000], 2000, 1e-16, dl)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: High-alpha Adam
    if tleft() > 80:
        for a, lr_val in [(10000, 0.004), (20000, 0.003), (50000, 0.002)]:
            if tleft() < 30:
                break
            hl = tleft()
            dl = min(start_time + 920 + int(hl * 0.3), hard_deadline)
            h_adam, sc = adam_optimize(best_h, n_final, target_final, a, 3000, lr_val, dl)
            if sc < best_score:
                best_score = sc
                best_h = h_adam

    # Phase 8: Subgradient + surgery
    if tleft() > 50:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final, 1000, 3, 0.003, 0.999, min(start_time + 1050, hard_deadline))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final, 300, 0.002, min(start_time + 1100, hard_deadline))
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 9: Diverse restarts with Adam
    if tleft() > 80:
        for attempt in range(5):
            if tleft() < 40:
                break
            scales = [0.03, 0.06, 0.1, 0.05, 0.08]
            h_pert = best_h + rng.randn(n_final) * scales[attempt]
            h_pert = project_feasible(h_pert, target_final)
            h_adam, sc = adam_optimize(h_pert, n_final, target_final, 1000, 3000, 0.01, min(start_time + 1100 + 30 * (attempt + 1), hard_deadline))
            if sc < best_score:
                best_score = sc
                best_h = h_adam

    # Phase 10: Final polish
    if tleft() > 15:
        h_adam, sc = adam_optimize(best_h, n_final, target_final, 30000, 2000, 0.003, hard_deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
