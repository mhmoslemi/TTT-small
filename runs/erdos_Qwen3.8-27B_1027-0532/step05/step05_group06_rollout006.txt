def subgradient_max(h, dx, n_active=5):
    """Compute subgradient of max_k corr[k] * dx, averaging over top n_active."""
    n = len(h)
    corr = compute_corr(h)
    # Get top active indices
    top_idx = np.argsort(corr)[::-1][:n_active]

        if lag >= 0:

            grad[j1] += (1.0 - h[j1 - lag])
            if lag > 0:

                grad[j2] -= h[j2 + lag]

            j1 = np.arange(0, n - m)
            grad[j1] += (1.0 - h[j1 + m])

                j2 = np.arange(m, n)
                grad[j2] -= h[j2 - m]
    # Normalize: average over active set
    grad /= len(top_idx)
    return grad, float(np.max(corr)) * dx

c_full[n-1-k] = ∑_i a[i+k] * v[i] for k = 0, ..., n-1 (negative lags, v shifted left by k)

With a = h, v = 1-h:
c_full[n-1+k] = ∑_i h[i] * (1-h[i+k]) for i = 0 to n-1-k
c_full[n-1-k] = ∑_i h[i+k] * (1-h[i]) for i = 0 to n-1-k

Both of these are non-negative (since h ∈ [0,1]).

Note that c_full[n-1] = ∑ h[i]*(1-h[i]) = n/2 - ∑h[i]² (using ∑h = n/2).

To minimize this, we want to maximize ∑h[i]², i.e., make h as binary as possible.

For binary h: c_full[n-1] = 0.
For h = 0.5 everywhere: c_full[n-1] = n * 0.25 = n/4, so C₅ ≥ (n/4)*dx = 1/2.

So binary solutions have C₅ < 0.5 from lag 0 alone. The other lags determine the actual C₅.

    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val

def lse_gradient(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')

    n = len(h)

    cur_f = compute_c5(h_cur)
    lr = lr_init

            break

                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_f:
                    cur_f = sc
