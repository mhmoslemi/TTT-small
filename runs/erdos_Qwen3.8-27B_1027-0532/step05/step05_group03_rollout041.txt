  3. Try to reduce the lags in A simultaneously
  4. Use the gradient of Σ_{k∈A} f_k(h) as the descent direction

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_overall_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    return corr[n-1:n-1+n]

def smooth_objective(h, alpha):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    over = compute_overall_corr(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha, over

def smooth_gradient(h, alpha):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    over = compute_overall_corr(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w = w / w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk < 1e-15:
            continue
        if k == 0:
            g += wk * dx * (1.0 - 2.0 * h)
        else:
            g[k:] += wk * dx * (1.0 - h[:n-k])
            g[:n-k] -= wk * dx * h[k:]
    return g

def lbfgs_optimize(h_init, alpha, maxiter, ftol):
    n = len(h_init)
    target = n / 2.0

    def obj(x):
        h = project_feasible(x, target)
        over = compute_overall_corr(h)
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def jac(x):
        return smooth_gradient(x, alpha)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10,
                                'maxfun': maxiter * 4, 'disp': False})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        return h_new, sc
    except Exception:
        return h_init, compute_c5(h_init)

def active_lag_descent(h, n_active, n_steps, lr_init, lr_decay):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        over = compute_overall_corr(h_cur)
        top_idx = np.argsort(over)[::-1][:n_active]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            if k == 0:
                grad += dx * (1.0 - 2.0 * h_cur)
            else:
                grad[k:] += dx * (1.0 - h_cur[:n-k])
                grad[:n-k] -= dx * h_cur[k:]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        c_curr = float(np.max(over)) * dx
        if sc_new < c_curr:
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
            lr = min(lr * 1.005, 0.01)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def nesterov_subgrad(h, n_steps, n_active, lr_init, momentum, temp):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    vel = np.zeros(n)
    m = momentum
    for step in range(n_steps):
        over = compute_overall_corr(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        w = np.exp(diffs / temp)
        w = w / w.sum()
        grad = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                grad[k:] += wk * dx * (1.0 - h_cur[:n-k])
                grad[:n-k] -= wk * dx * h_cur[k:]
        vel = m * vel + (1.0 - m) * grad
        h_new = h_cur - lr * vel
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.008)
            m = min(m + 0.001, 0.85)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    return best_h, best_score

def lag_balanced_descent(h, n_steps, lr_init, n_lags_full):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        over = compute_overall_corr(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        c_max = float(np.max(over))
        threshold = c_max - 0.02 * (c_max - float(np.min(over)))
        active = np.where(over >= threshold)[0]
        if len(active) > n_lags_full:
            active = np.argsort(over)[::-1][:n_lags_full]
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            if k == 0:
                grad += dx * (1.0 - 2.0 * h_cur)
            else:
                grad[k:] += dx * (1.0 - h_cur[:n-k])
                grad[:n-k] -= dx * h_cur[k:]
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        h_new = h_cur - lr * grad / grad_norm
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.008, 0.012)
        else:
            lr *= 0.92
            if lr < 1e-11:
                break
    return best_h, best_score

def try_finer_grid(h_src, n_dst, budget_left):
    target_dst = n_dst / 2.0
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_fine = np.interp(x_new, x_old, h_src)
    h_fine = project_feasible(h_fine, target_dst)
    h_fine, sc = lbfgs_optimize(h_fine, 200, 800, 1e-12)
    h_fine, sc = lbfgs_optimize(h_fine, 1000, 600, 1e-13)
    h_fine, sc = active_lag_descent(h_fine, 5, 200, 0.003, 0.97)
    h_fine, sc = lbfgs_optimize(h_fine, 5000, 400, 1e-14)
    return h_fine, sc

def try_coarser_grid(h_src, n_dst, budget_left):

    h_coarse, sc = lbfgs_optimize(h_coarse, 10000, 500, 1e-14)
    return h_coarse, sc

def bang_bang_local_search(h, target, max_rounds=20):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for rnd in range(max_rounds):
        changed = False
        for j in range(n):
            for trial in [0.0, 1.0, 1.0 - h_cur[j]]:
                if trial == h_cur[j]:
                    continue
                h_trial = h_cur.copy()
                h_trial[j] = trial
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_score - 1e-12:
                    h_cur = h_trial
                    best_score = sc
                    best_h = h_cur.copy()
                    changed = True
                    break
        if not changed:
            break
    return best_h, best_score
