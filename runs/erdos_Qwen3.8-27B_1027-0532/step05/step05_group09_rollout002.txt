            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        g = lse_grad_fft(y, alpha)
        obj_y = lse_obj(y, alpha)
        x_new = project_feasible(y - step * g, target)
        
        for bt in range(25):
            obj_new = lse_obj(x_new, alpha)
            dec = obj_y - obj_new
            quad = 0.5 / step * np.dot(x_new - y, x_new - y)
            if dec >= quad * 0.8:
                break
            step *= 0.5
            x_new = project_feasible(y - step * g, target)
        
        step = min(step * 1.6, max_step)
        x_prev = x
        x = x_new
        t = t_new
        
        c5 = compute_c5(x)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count == stall_threshold:
                rng = np.random.RandomState(k)
                x = x + rng.randn(n) * 0.003
                x = project_feasible(x, target)
                x_prev = x.copy()
                y = x.copy()
                t = 1.0
                best_x = x.copy()
                best_c5 = compute_c5(x)
                stall_count = 0
                stall_threshold = int(stall_threshold * 1.5)
    
    return best_x, best_c5

def peak_shave(h, target, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    rng = np.random.RandomState(777)
    
    for it in range(10000):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        cmax = float(np.max(corr))
        active = np.where(corr > cmax - 0.005 / dx)[0]
        na = min(len(active), 6)
        active = active[np.argsort(corr[active])[::-1][:na]]
        
        grad = np.zeros(n)
        for ki in active:
            ki = int(ki)
            if ki == 0:
                grad += (1.0 - 2.0 * h_cur)
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += (1.0 - h_cur[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= h_cur[jn - ki]
        grad *= dx / len(active)
        
        improved = False
        for eps in [0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002]:
            h_new = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_new)
            if sc < cur - 1e-14:
                h_cur = h_new
                cur = sc
                improved = True
                break
        if not improved:
            h_pert = h_cur + rng.randn(n) * 0.004
            h_pert = project_feasible(h_pert, target)
            sc = compute_c5(h_pert)
            if sc < cur:
                h_cur = h_pert
                cur = sc
    
    return h_cur, cur

def periodic_pattern(n, period, duty=0.5, offset=0):
    x = np.arange(n)
    h = np.where(((x + offset) % period) < (period * duty), 1.0, 0.0)
    return h

def coord_polish(h, target, deadline):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur = compute_c5(h_cur)
    rng = np.random.RandomState(31415)
    stalled = 0
    
    for sweep in range(5000):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for idx in order:
            if time.time() > deadline:
                break
            v = h_cur[idx]
            if 0.01 < v < 0.99:
                trial = 1.0 if v > 0.5 else 0.0
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur - 1e-14:
                    cur = sc
                    improved = True
                else:
                    h_cur[idx] = v
                    h_cur = project_feasible(h_cur, target)
            elif v >= 0.99:
                h_cur[idx] = 0.0
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur - 1e-14:
                    cur = sc
                    improved = True
                else:
                    h_cur[idx] = 1.0
                    h_cur = project_feasible(h_cur, target)
            elif v <= 0.01:
                h_cur[idx] = 1.0
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur - 1e-14:
                    cur = sc
                    improved = True
                else:
                    h_cur[idx] = 0.0
                    h_cur = project_feasible(h_cur, target)
        if not improved:
            stalled += 1
            if stalled > 3:
                h_pert = h_cur + rng.randn(n) * 0.008
                h_pert = project_feasible(h_pert, target)
                sc = compute_c5(h_pert)
                if sc < cur:
                    h_cur = h_pert
                    cur = sc
                    stalled = 0
                else:
                    break
        else:
            stalled = 0
    
    return h_cur, cur

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    deadline = start + budget_s - 10.0
    
    n = len(initial_h_values)
    target = n / 2.0
    h0 = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    def tleft():
        return deadline - time.time()
    
    # Phase 1: FISTA continuation with careful schedule
    schedule = [(20, 45), (80, 45), (300, 55), (1500, 70), (8000, 80), (40000, 80)]
    for alpha, td in schedule:
        if tleft() < td + 10:
            break
        h_new, sc = fista_solve(best_h, target, alpha, start + td, lr_scale=1.0)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    
    # Phase 2: Peak shaving
    if tleft() > 50:
        h_new, sc = peak_shave(best_h, target, start + 50)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    
    # Phase 3: Very high alpha FISTA
    if tleft() > 80:
        h_new, sc = fista_solve(best_h, target, 60000, start + 60, lr_scale=2.0)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
        if tleft() > 35:
            h_new, sc = peak_shave(best_h, target, start + 35)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new
    
    # Phase 4: Structured multi-starts
    if tleft() > 150:
        periods_to_try = [7, 11, 13, 17, 19, 23, 31]
        for p in periods_to_try:
            if tleft() < 25:
                break
            h_p = periodic_pattern(n, p, 0.5, 0)
            h_p = project_feasible(h_p, target)
            h_new, sc = fista_solve(h_p, target, 300, start + 22, lr_scale=1.0)
            if sc > best_c5:
                continue
            if tleft() < 20:
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_new
                break
            h_new, sc2 = fista_solve(h_new, target, 5000, start + 18, lr_scale=1.5)
            if sc2 < best_c5:
                best_c5 = sc2
                best_h = h_new
        
        # Random binary patterns
        for trial in range(4):
            if tleft() < 22:
                break
            rng = np.random.RandomState(seed + 500 + trial * 100)
            h_r = (rng.rand(n) < 0.5).astype(float)
            h_r = project_feasible(h_r, target)
            h_new, sc = fista_solve(h_r, target, 500, start + 20, lr_scale=1.0)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new
    
    # Phase 5: Perturbation-based restarts from best
    if tleft() > 100:
        scales = [0.03, 0.05, 0.07, 0.04, 0.06, 0.09]
        for s in scales:
            if tleft() < 22:
                break
            rng = np.random.RandomState(seed + int(s * 10000))
            h_p = best_h + rng.randn(n) * s
            h_p = project_feasible(h_p, target)
            h_new, sc = fista_solve(h_p, target, 3000, start + 20, lr_scale=1.5)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new
    
    # Phase 6: Final high-alpha FISTA
    if tleft() > 60:
        h_new, sc = fista_solve(best_h, target, 100000, start + 45, lr_scale=3.0)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    
    # Phase 7: Peak shave final
    if tleft() > 25:
        h_new, sc = peak_shave(best_h, target, start + 25)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    
    # Phase 8: Coordinate descent polish
    if tleft() > 15:
        h_new, sc = coord_polish(best_h, target, start + max(tleft() - 3, 10))
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    
    h_final = project_feasible(best_h, target)
    final_c5 = compute_c5(h_final)
    return h_final, final_c5, n
