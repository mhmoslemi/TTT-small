import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, p):
    n = len(h)
    k = (n - 1) - p
    g = np.zeros(n)
    if k >= 0:
        i1 = np.arange(k, n)
        g[i1] += (1.0 - h[i1 - k])
        i2 = np.arange(0, n - k)
        g[i2] -= h[i2 + k]
    else:
        m = -k
        i1 = np.arange(0, n - m)
        g[i1] += (1.0 - h[i1 + m])
        i2 = np.arange(m, n)
        g[i2] -= h[i2 - m]
    return g

def lse_obj(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_val = np.exp(shifted)
    total = np.sum(exp_val)
    weights = exp_val / total
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-12:
            continue
        grad += w * corr_grad_single(x, p)
    return grad

def slsqp_with_grad(h, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lse_obj, h, args=(alpha,), jac=lambda x: lse_grad(x, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_no_grad(h, n, target, alpha, maxiter, ftol):
