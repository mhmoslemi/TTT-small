def slp_step(h, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2*n - 1
    mid = n - 1
    
    # Compute gradients for all shifts
    # grad[k, i] = d(corr[k])/d(h[i])
    idxs = np.arange(n)
    
    # For shift m = k - mid, the gradient is:
    # d(corr[k])/d(h[i]) = (1-h[i-m]) * 1[m<=i<=n-1] - h[i+m] * 1[0<=i<=n-1-m] for m>=0
    #                        (1-h[i+|m|]) * 1[0<=i<=n-1-|m|] - h[i-|m|] * 1[|m|<=i<=n-1] for m<0
    
    # This is a (n_corr x n) matrix. For n=400, that's 799 x 400 = 319600 elements.
    # We can compute it efficiently.
    
    grads = np.zeros((n_corr, n))
    for m in range(-(n-1), n):
        k = m + mid
        if m >= 0:
            # d(corr[k])/d(h[i]) = (1-h[i-m]) for i in [m, n-1], minus h[i+m] for i in [0, n-1-m]
            if m < n:
                v1 = np.arange(m, n)
                grads[k, v1] += (1.0 - h[v1 - m])
            if m > 0:
                v2 = np.arange(0, n - m)
                grads[k, v2] -= h[v2 + m]
            # If m == 0: v1 = [0, n-1], v2 = [0, n-1] (same range)
            # grads[k, v] = (1-h[v]) - h[v] = 1 - 2h[v]
            # Wait, that's not right for m=0.
            # For m=0: corr[k] = sum_{j=0}^{n-1} h[j](1-h[j])
            # d/dh[i] = (1-h[i]) - h[i] = 1 - 2h[i]
            # From formula: v1 = [0, n-1], grads += (1-h[0:n]) = 1-h
            #              v2 = [0, n-1], grads -= h[0:n] = h
            # Total: 1 - h - h = 1 - 2h. Correct!
        else:
            m_abs = -m
            if m_abs < n:
                v1 = np.arange(0, n - m_abs)
                grads[k, v1] += (1.0 - h[v1 + m_abs])
            if m_abs > 0:
                v2 = np.arange(m_abs, n)
                grads[k, v2] -= h[v2 - m_abs]
    
    # Now solve LP: minimize t
    # s.t. corr[k] + grads[k] @ delta <= t, for all k
    #      -h[i] <= delta[i] <= 1-h[i]
    #      sum(delta) = 0
    #      |delta[i]| <= step_size (optional)
    
    delta_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = []
    for k in range(n_corr):
        constraints.append(corr[k] + grads[k] @ delta_var <= t_var)
    constraints.append(delta_var >= -h + 1e-10)
    constraints.append(delta_var <= 1 - h - 1e-10)
    constraints.append(cp.sum(delta_var) == 0)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)  # or CPLEX, CBC, etc.
        if prob.status == 'optimal':
            delta_opt = delta_var.value
            h_new = h + delta_opt
            h_new = np.clip(h_new, 0, 1)
            # Project to maintain exact sum
            h_new = project_feasible(h_new, target)
            return h_new
    except:
        pass
    return h

For the block coordinate descent, the idea is:

    dx = 2.0 / n

    n_corr = 2*n - 1

    idxs = np.arange(n)

    grads = np.zeros((n_corr, n))
    for k in range(n_corr):
        m = k - mid
        if m >= 0:
            if m < n:
                v1 = np.arange(m, n)
                grads[k, v1] += (1.0 - h[v1 - m])
            if m > 0:
                v2 = np.arange(0, n - m)
                grads[k, v2] -= h[v2 + m]
        else:
            m_abs = -m
            if m_abs < n:
                v1 = np.arange(0, n - m_abs)
                grads[k, v1] += (1.0 - h[v1 + m_abs])
            if m_abs > 0:
                v2 = np.arange(m_abs, n)
                grads[k, v2] -= h[v2 - m_abs]

    delta_var = cp.Variable(n)
    t_var = cp.Variable()

        corr[i] + grads[i] @ delta_var <= t_var for i in range(n_corr)

    constraints.append(delta_var >= -h)
    constraints.append(delta_var <= 1.0 - h)
    # Trust region
    constraints.append(cp.norm(delta_var, np.inf) <= step_scale * dx)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)

                    return h_try

def compute_grad_matrix(h, n):
    """Compute the gradient matrix for all correlations.
    Returns grads[k, i] = d(corr[k])/d(h[i]) for k=0..2n-2, i=0..n-1.
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2*n - 1
    mid = n - 1
    grads = np.zeros((n_corr, n))
    
    # Shift m = 0 (k = mid)
    # corr[mid] = sum_j h[j](1-h[j])
    # d/dh[i] = 1 - 2h[i]
    grads[mid] = 1.0 - 2.0 * h
    
    # Positive shifts m = 1, ..., n-1 (k = mid + m)
    for m in range(1, n):
        k = mid + m
        # corr[k] = sum_{j=m}^{n-1} h[j](1-h[j-m])
        # d/dh[i] = (1-h[i-m]) for i in [m, n-1]
        #          - h[i+m] for i in [0, n-1-m]
        grads[k, m:] += (1.0 - h[:n-m])
        grads[k, :n-m] -= h[m:]
    
    # Negative shifts m = -1, ..., -(n-1) (k = mid - |m|)
    for m in range(1, n):
        k = mid - m
        # corr[k] = sum_{j=0}^{n-1-m} h[j](1-h[j+m])
        # d/dh[i] = (1-h[i+m]) for i in [0, n-1-m]
        #          - h[i-m] for i in [m, n-1]
        grads[k, :n-m] += (1.0 - h[m:])
        grads[k, m:] -= h[:n-m]
    
    return corr, grads

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    grads = np.zeros((n_corr, n))
    grads[mid] = 1.0 - 2.0 * h
    for m in range(1, n):
        k = mid + m
        grads[k, m:] += (1.0 - h[:n-m])
        grads[k, :n-m] -= h[m:]
        k2 = mid - m
        grads[k2, :n-m] += (1.0 - h[m:])
        grads[k2, m:] -= h[:n-m]
    return corr, grads

    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_budget * 0.9:
            break
        if time.time() - start > time_budget:
            break
            
        corr, grads = compute_grad_matrix(h, n)
        n_corr = 2 * n - 1

        delta_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = []
        for k in range(n_corr):
            constraints.append(corr[k] + grads[k] @ delta_var <= t_var)
        constraints.append(cp.sum(delta_var) == 0)
        constraints.append(delta_var >= -h + 1e-12)
        constraints.append(delta_var <= 1.0 - h - 1e-12)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)

            if prob.status in ('optimal', 'optimal_inaccurate') and delta_var.value is not None:
                delta = delta_var.value

                            best_h = h.copy()
                        break
        except Exception:
            pass

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    grads = np.zeros((n_corr, n))
    grads[mid] = 1.0 - 2.0 * h
    for m in range(1, n):
        k_pos = mid + m
        grads[k_pos, m:] += (1.0 - h[:n-m])
        grads[k_pos, :n-m] -= h[m:]
        k_neg = mid - m
        grads[k_neg, :n-m] += (1.0 - h[m:])
        grads[k_neg, m:] -= h[:n-m]
    return corr, grads

def slp_refine(h, n, target, time_budget, max_iters=100, solver=cp.CBC):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_budget * 0.95:
            break
        
        corr, grads = compute_grad_matrix(h, n)
        n_corr = 2 * n - 1
        
        delta_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = []
        # Use all correlation constraints
        for k in range(n_corr):
            constraints.append(corr[k] + grads[k, :] @ delta_var <= t_var)
        constraints.append(cp.sum(delta_var) == 0)
        constraints.append(delta_var >= -h + 1e-10)
        constraints.append(delta_var <= 1.0 - h - 1e-10)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and delta_var.value is not None:
                delta = np.asarray(delta_var.value).flatten()
                h_cur_c5 = compute_c5(h)
                improved = False
                for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                    h_try = np.clip(h + alpha * delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < h_cur_c5 - 1e-14:
                        h = h_try
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h.copy()
                        improved = True
                        break
                if not improved and it > 5:
                    break
        except Exception:
            break
    
    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _global_target

def _cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def momentum_subgradient(h, n, target, time_limit, max_steps=50000, mu=0.9, base_lr=0.01):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        
        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        
        if no_improve > 5000:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
