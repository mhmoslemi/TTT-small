        if abs(diff) < 1e-14:
            break

        else:

import numpy as np

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_objective(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val


def lse_gradient(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            idx1 = np.arange(lag, n)
            grad[idx1] += w * (1.0 - x[idx1 - lag])
            if lag > 0:
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= w * x[idx2 + lag]
        else:

            grad[idx1] += w * (1.0 - x[idx1 - m])
            if m > 0:
                idx2 = np.arange(0, n - m)
                grad[idx2] -= w * x[idx2 + m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad

    g_obj = partial(lse_gradient, alpha=alpha, target=target, mu=mu)
    try:

                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-15, 'maxls': 80, 'maxfun': maxiter * 4})
        return project_feasible(res.x, target)
    except Exception:
        return h


def _sum_eq(x, target):
    return np.sum(x) - target


def _sum_eq_jac(x):
    return np.ones(len(x))


def run_slsqp(h, target, alpha, maxiter):

    bounds = [(0.0, 1.0)] * n
    f_obj = partial(lse_objective, alpha=alpha, target=target, mu=1e8)
    g_obj = partial(lse_gradient, alpha=alpha, target=target, mu=1e8)

        h_new = run_lse(h_bin, target, 1000, mu, 400)
        h_new = run_lse(h_new, target, 50000, mu, 400)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_eq, sc_eq = active_set_equalize(best_h, target, n_top=7, n_steps=200)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    # Phase 5: Upsample to 2x resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    cur_n = len(best_h)
    cur_target = cur_n / 2.0
    if time.time() < deadline - 250:
        h_fine = upsample(best_h, cur_n, n2, target2)
        alphas_fine = [50, 500, 5000, 50000, 200000, 500000]
        for a in alphas_fine:
            if time.time() > deadline - 150:
                break
            h_fine = run_lse(h_fine, target2, a, mu, 600)
            sc = compute_c5(h_fine)
            if sc < best_score:
                best_score = sc
                best_h = h_fine
                cur_n = n2
                cur_target = target2

    # Phase 6: Equalization at current resolution
    if time.time() < deadline - 150:
        h_eq, sc_eq = active_set_equalize(best_h, cur_target, n_top=10, n_steps=300)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = run_lse(best_h, cur_target, 500000, mu, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Diverse restarts
    restart_list = [
        ('gauss', 0.03), ('gauss', 0.08), ('gauss', 0.15),
        ('swap', 15), ('swap', 40),
        ('block', 8), ('block', 25),
        ('binary', 0.0)
    ]
    for strat, param in restart_list:
        if time.time() > deadline - 100:
            break
        h_pert = best_h.copy()
        if strat == 'gauss':
            h_pert += rng.randn(cur_n) * param
        elif strat == 'swap':
            h_pert = perturb_swap(best_h, cur_target, rng, param)
        elif strat == 'block':
            h_pert = perturb_block(best_h, cur_target, rng, param)
        elif strat == 'binary':
            h_pert = make_binary(best_h, cur_target, rng)
        h_pert = project_feasible(h_pert, cur_target)
        h_new = run_lse(h_pert, cur_target, 500, mu, 300)
        h_new = run_lse(h_new, cur_target, 50000, mu, 300)
        h_new = run_lse(h_new, cur_target, 300000, mu, 200)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Coordinate descent
    if time.time() < deadline - 80:
        n_cd = min(cur_n, 800)
        if cur_n <= 800:
            h_cd, sc_cd = coord_descent(best_h, cur_target, n_sweeps=3)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd
        else:
            h_cd, sc_cd = coord_descent(best_h[:n_cd], cur_target, n_sweeps=2)
            if sc_cd < compute_c5(best_h[:n_cd]):
                best_h[:n_cd] = h_cd
                best_h = project_feasible(best_h, cur_target)
                best_score = compute_c5(best_h)

    # Phase 9: Final equalization + LSE
    if time.time() < deadline - 50:
        h_eq, sc_eq = active_set_equalize(best_h, cur_target, n_top=5, n_steps=200)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = run_lse(best_h, cur_target, 500000, mu, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_eq2, sc_eq2 = active_set_equalize(best_h, cur_target, n_top=3, n_steps=100)
        if sc_eq2 < best_score:
            best_score = sc_eq2
            best_h = h_eq2

    # Phase 10: SLSQP polish
    if time.time() < deadline - 30:
        h_slq = run_slsqp(best_h, cur_target, 100000, 400)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 11: Final tiny coordinate descent
    if time.time() < deadline - 15:
        h_cd, sc_cd = coord_descent(best_h, cur_target, n_sweeps=1)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    h_final = project_feasible(best_h, cur_target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
