def softmax_gradient_descent(h, n, target, time_limit, max_steps, T_schedule, lr_schedule):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Temperature schedule
        T = max(0.01, T_schedule * max(0, 1 - step / (max_steps * 0.8)))
        
        # Softmax weights
        shifted_corr = (corr - c_max) / T
        shifted_corr = np.clip(shifted_corr, -50, 0)
        w = np.exp(shifted_corr)
        w = w / w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        # For efficiency, only use top shifts
        top_m = min(20, n_corr)
        top_idx = np.argsort(corr)[-top_m:]
        w_top = w[top_idx]
        w_top = w_top / w_top.sum()
        
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)
        
        # Adaptive step size
        lr = lr_schedule * max(0.01, 1.0 - step / max_steps)
        
        # Backtracking line search
        for attempt in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                break
            lr *= 0.5
        
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

    best_lr = 0.05
    no_improve_count = 0

            break

                best_lr = lr * 1.1

                break
            lr *= 0.5

def enhanced_pair_cd(h, n, target, time_limit, max_sweeps=5000,

            n_cand = max(n_cand - 3, n_cand_start)

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)

def momentum_subgradient(h, n, target, time_limit, max_steps=20000,

        grad = np.zeros(n)

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h = h_gt

    # Phase 2: L-BFGS-B polish (faster than SLSQP)
    if tleft() > 150:
        for alpha in [100, 500, 2000, 10000]:
            if tleft() < 40:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 3: Enhanced pair CD (main refinement)
    if tleft() > 200:
        cd_time = min(400, tleft() * 0.35)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=10000,
                                       n_cand_start=25,
                                       n_cand_max=80,
                                       patience=200)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Game-theoretic again after CD
    if tleft() > 100:
        gt_time = min(80, tleft() * 0.15)
        h_gt, c5_gt = game_theoretic_descent(best_h, n, target,
                                             time_limit=gt_time,
                                             max_steps=8000,
                                             eta_base=2.0, m_top=12)
        if c5_gt < best_score:
            best_score = c5_gt
            best_h = h_gt

    # Phase 5: L-BFGS-B + SLSQP hybrid
    if tleft() > 120:
        for alpha in [5000, 20000, 100000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
        if tleft() > 60:
            h_pol = slsqp_round(best_h, n, target,
                                [5000, 20000, 100000],
                                1000, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 6: Second round of pair CD
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000,
                                       n_cand_start=40,
                                       n_cand_max=100,
                                       patience=150)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Momentum subgradient for escape
    if tleft() > 80:
        h_mom, c5_mom = momentum_subgradient(best_h, n, target,
                                             time_limit=min(50, tleft() * 0.2),
                                             max_steps=15000, mu=0.8,
                                             base_lr=0.04)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 8: Multi-restart with diverse perturbations
    rng = np.random.RandomState(seed)
    if tleft() > 100:
        for noise_scale in [0.03, 0.08, 0.12, 0.2]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfags_polish(h_pert, n, target, 5000, maxiter=800)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                               time_limit=min(40, tleft() * 0.2),
                                               max_sweeps=2000,
                                               n_cand_start=30,
                                               n_cand_max=60,
                                               patience=100)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Phase 9: Local perturbation search
    if tleft() > 60:
        h_pert, c5_pert = local_perturbation_search(best_h, n, target,
                                                    max_tries=3000,
                                                    time_limit=min(40, tleft() * 0.3),
                                                    seed=seed + 1)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 10: Try higher resolution
    if tleft() > 180:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfags_polish(h_interp, n_try, target_try, 2000, maxiter=1000)
            h_hires = enhanced_pair_cd(h_hires, n_try, target_try,
                                       time_limit=min(60, tleft() * 0.3),
                                       max_sweeps=2000,
                                       n_cand_start=30,
                                       n_cand_max=60,
                                       patience=100)[0]
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 11: Final polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000, 200000],
                              800, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
