import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(800):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[n - 1] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n - k]
        else:
            ell = -k
            g[ell:] += wm * (1.0 - h[:n - ell])
            g[:n - ell] -= wm * h[ell:]
    g *= dx
    return g


def topk_lse_value(h, alpha, top_k):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    c_active = corr[top_idx]
    am = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
    return am + float(np.log(np.sum(np.exp(shifted)))) / alpha


def topk_lse_grad(h, alpha, top_k):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    c_active = corr[top_idx]
    am = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(top_k)
        w[0] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    g = np.zeros(n)
    for wm, m in zip(w, top_idx):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n - k]
        else:
            ell = -k
            g[ell:] += wm * (1.0 - h[:n - ell])
            g[:n - ell] -= wm * h[ell:]
    g *= dx
    return g


def lbfgsb_lse(h, n, target, alpha, mu, maxiter):
    def obj_fn(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg_fn(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, method='L-BFGS-B', jac=objg_fn,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_lse(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    def obj_fn(x):
        return lse_value(x, alpha)
    def objg_fn(x):
        return lse_grad(x, alpha)
    try:
        res = minimize(obj_fn, h, method='SLSQP', jac=objg_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_topk(h, n, target, alpha, top_k, maxiter):
    def obj_fn(x):
        return topk_lse_value(x, alpha, top_k)
    def objg_fn(x):
        return topk_lse_grad(x, alpha, top_k)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', jac=objg_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h


def momentum_subgrad(h, n, target, n_steps, temp, lr_init, lr_decay, momentum, deadline, top_k=5):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        am = corr[top_idx[0]]
        diffs = corr[top_idx] - am
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        v = momentum * v + (1.0 - momentum) * g
        h_new = h_cur - lr * v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def block_perturbation_search(h, n, target, n_trials, block_sizes, deadline, rng):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n

        if time.time() > deadline:
            break
        B = block_sizes[trial % len(block_sizes)]
        i = rng.randint(0, max(1, n - B))
        h_trial = h_cur.copy()
        block = h_trial[i:i+B].copy()
        block_mass = block.sum()
        if block_mass < 1e-10 or block_mass > B - 1e-10:
            continue
        mode = rng.randint(3)
        if mode == 0:
            h_trial[i:i+B] = block_mass / B
        elif mode == 1:
            shift = rng.choice([-2, -1, 1, 2])
            h_trial[i:i+B] = np.roll(block, shift)
            h_trial[i:i+B] = np.clip(h_trial[i:i+B], 0.0, 1.0)
        else:
            noise = rng.randn(B) * 0.15
            h_trial[i:i+B] = block + noise
            h_trial[i:i+B] = np.clip(h_trial[i:i+B], 0.0, 1.0)
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < best_score:
            best_score = sc
            h_cur = h_trial
    return h_cur, best_score


def pair_swap_search(h, n, target, n_trials, deadline, rng):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def symmetry_averaging(h, n, target):
    h_sym = (h + h[::-1]) / 2.0
    h_sym = project_feasible(h_sym, target)
    return h_sym

            best_h = h_sym
            h_new = slsqp_lse(best_h, n0, target0, 20000, 500, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: Block perturbation + pair swap at n0
    if time.time() < deadline - 100:
        block_sizes = [3, 5, 8, 10, 15]
        h_bp, sc_bp = block_perturbation_search(best_h, n0, target0, 3000, block_sizes, deadline - 50, rng)
        if sc_bp < best_score:
            best_score = sc_bp
            best_h = h_bp
        h_ps, sc_ps = pair_swap_search(best_h, n0, target0, 5000, deadline - 30, rng)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps

    # Phase 7: Higher resolution n=800
    n1 = 800
    target1 = n1 / 2.0
    if time.time() < deadline - 200:
        h1 = interpolate_to(best_h, n1, target1)
        best_h1 = h1.copy()
        best_score1 = compute_c5(h1)
        # Optimize at higher resolution
        for alpha in [30, 100, 400, 1500, 5000]:
            if time.time() > deadline - 80:
                break
            for mu in [20.0, 200.0]:
                if time.time() > deadline - 60:
                    break
                h_new = lbfgsb_lse(best_h1, n1, target1, alpha, mu, 100)
                sc = compute_c5(h_new)
                if sc < best_score1:
                    best_score1 = sc
                    best_h1 = h_new
        if time.time() < deadline - 60:
            h_new = slsqp_lse(best_h1, n1, target1, 5000, 1000, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score1:
                best_score1 = sc
                best_h1 = h_new
        if time.time() < deadline - 40:
            h_sg1, sc_sg1 = momentum_subgrad(best_h1, n1, target1, 1000, 0.002, 0.004, 0.995, 0.85, deadline - 40, top_k=4)
            if sc_sg1 < best_score1:
                best_score1 = sc_sg1
                best_h1 = h_sg1
        if time.time() < deadline - 25:
            block_sizes1 = [3, 5, 8, 12]
            h_bp1, sc_bp1 = block_perturbation_search(best_h1, n1, target1, 2000, block_sizes1, deadline - 20, rng)
            if sc_bp1 < best_score1:
                best_score1 = sc_bp1
                best_h1 = h_bp1
        if best_score1 < best_score:
            best_score = best_score1
            best_h = best_h1

    # Phase 8: Even higher resolution n=1200 if time permits
    if time.time() < deadline - 120:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [50, 200, 800, 3000]:
            if time.time() > deadline - 50:
                break
            h_new = lbfgsb_lse(best_h2, n2, target2, alpha, 100.0, 80)
            sc = compute_c5(h_new)
            if sc < best_score2:
                best_score2 = sc
                best_h2 = h_new
        if time.time() < deadline - 30:
            h_sg2, sc_sg2 = momentum_subgrad(best_h2, n2, target2, 800, 0.002, 0.003, 0.995, 0.85, deadline - 25, top_k=3)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 9: Final aggressive polish at the best resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if time.time() < deadline - 30:
        h_new = slsqp_lse(best_h, n_cur, target_cur, 50000, 1000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if time.time() < deadline - 15:
        h_sg, sc_sg = momentum_subgrad(best_h, n_cur, target_cur, 1000, 0.001, 0.002, 0.997, 0.9, deadline - 10, top_k=3)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    if time.time() < deadline - 8:
        h_ps, sc_ps = pair_swap_search(best_h, n_cur, target_cur, 3000, deadline - 5, rng)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps

    # Phase 10: Final SLSQP
    if time.time() < deadline - 3:
        h_new = slsqp_lse(best_h, n_cur, target_cur, 100000, 500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
