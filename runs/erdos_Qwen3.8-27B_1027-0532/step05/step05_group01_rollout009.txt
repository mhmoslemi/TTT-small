    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_weighted(h, alpha, top_k):
    """Compute gradient of the log-sum-exp smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    idxs = np.arange(n)
    
    order = np.argsort(corr)
    active = order[-top_k:]
    c_max = corr[active[-1]]
    w = np.exp(alpha * (corr[active] - c_max))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, s in zip(w, active):
        k = int(s) - mid
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            v1 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] + k])
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v2] -= wi * h[idxs[v2] - k]
    
    grad *= dx
    return grad

def lbfgs_round(h_start, n, target, alpha, top_k, maxiter=3000):
    """Single L-BFGS-B optimization round."""
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mid = n - 1
        idxs = np.arange(n)
        
        order = np.argsort(corr)
        active = order[-top_k:]
        c_max = corr[active[-1]]
        exp_arg = np.clip(alpha * (corr[active] - c_max), -500.0, 0.0)
        log_sum = np.log(np.sum(np.exp(exp_arg)))
        f = c_max + log_sum / alpha
        
        w = np.exp(exp_arg)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, s in zip(w, active):
            k = int(s) - mid
            if k == 0:
                grad += wi * (1.0 - 2.0 * x)
            else:
                v1 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += wi * (1.0 - x[idxs[v1] + k])
                v2 = (idxs - k >= 0) & (idxs - k < n)
                grad[v2] -= wi * x[idxs[v2] - k]
        grad *= (2.0 / n)
        
        return f, grad
    
    res = minimize(obj_and_grad, h_start, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
    h_new = project_feasible(res.x, target)
    return h_new
