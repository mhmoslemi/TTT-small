def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def reweighting_optimize(h, target, alpha, n_outer, deadline):
    """Iterative reweighting method for min max problem."""
    n = len(h)
    h_cur = h.copy()
    c_max_uncorr = 2 * n - 1  # number of correlation lags
    weights = np.ones(c_max_uncorr) / c_max_uncorr
    
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        # Step: minimize weighted sum of correlations
        # S(h) = sum_k w_k * [sum_i h[i] - h[i]*h[(i+k)%n]] * dx
        # = constant - sum_k w_k * sum_i h[i]*h[(i+k)%n] * dx
        # So we want to maximize sum_k w_k * (h⊛h)_k
        # This is a quadratic form: h^T (sum_k w_k * A_k) h
        # where A_k is the shift matrix for lag k
        
        # Gradient of S w.r.t. h:
        # dS/dh[i] = -sum_k w_k * [sum of h[j] where j is related to i by lag k]
        
        # For simplicity, use L-BFGS-B on the smooth objective
        def smooth_obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return float(np.sum(weights * corr)) + 1e6 * (np.sum(x) - target)**2
        
        def smooth_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            grad = np.zeros(n)
            for ki in range(len(corr)):
                w = weights[ki]
                if w < 1e-10:
                    continue
                lag = ki - (n - 1)
                if lag >= 0:
                    j1 = np.arange(lag, n)
                    grad[j1] += w * (1.0 - x[j1 - lag])
                    if lag > 0:
                        j2 = np.arange(0, n - lag)
                        grad[j2] -= w * x[j2 + lag]
                else:
                    m = -lag
                    j1 = np.arange(0, n - m)
                    grad[j1] += w * (1.0 - x[j1 + m])
                    if m > 0:
                        j2 = np.arange(m, n)
                        grad[j2] -= w * x[j2 - m]
            grad += 2e6 * (np.sum(x) - target)
            return grad
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj, h_cur, jac=smooth_grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 200, 'gtol': 1e-12})
            h_cur = project_feasible(res.x, target)
        except Exception:
            pass
        
        # Update weights
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        weights = np.exp(alpha * (corr - corr.max()))
        weights = weights / weights.sum()
    
    return h_cur

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_step(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    x0 = h.copy()
    
    def f(x):
        return lse_obj_grad(x, alpha, target, mu)[0]
    
    def g(x):
        return lse_obj_grad(x, alpha, target, mu)[1]
    
    try:
        if time.time() > deadline:
            return x0
        res = minimize(f, x0, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 100})
        return project_feasible(res.x, target)
    except Exception:
        return x0

    cur_score = compute_c5(h_b)

    best_score = cur_score

def coordinate_descent_fast(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h_cur, cur_score
            # Golden section search on h[j]
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            
            h_cur[j] = c
            h_temp = project_feasible(h_cur, target)
            fc = compute_c5(h_temp)
            h_cur[j] = d
            h_temp = project_feasible(h_cur, target)
            fd = compute_c5(h_temp)
            
            for _ in range(20):
                if fc < fd:
                    b = d
                    d = c
                    c = b - gr * (b - a)
                    h_cur[j] = c
                    h_temp = project_feasible(h_cur, target)
                    fc = compute_c5(h_temp)
                else:
                    a = c
                    c = d
                    d = a + gr * (b - a)
                    h_cur[j] = d
                    h_temp = project_feasible(h_cur, target)
                    fd = compute_c5(h_temp)
            
            best_v = (a + b) / 2.0
            h_cur[j] = best_v
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < cur_score - 1e-14:
                cur_score = sc
                improved = True
        if not improved:
            break
    return h_cur, cur_score


def top_k_equalize(h, target, n_top, n_steps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.01
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        top_mean = top_vals.mean()
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return h_cur, cur_score


def upsample(h, n_new, target_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h, kind='cubic')
    h_new = project_feasible(h_new, target_new)
    return h_new

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def lse_obj_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]

    alphas_fine = [50, 500, 5000, 50000, 200000]
    for a in alphas_fine:
        if time.time() > phase4_deadline:
            break
        h_fine = lbfgs_step(h_fine, target2, a, 100.0, 400, phase4_deadline)
    sc_fine = compute_c5(h_fine)
    if sc_fine < best_score:
        best_score = sc_fine
        best_h = h_fine
        n = n2
        target = target2

    phase5_deadline = min(start_time + 1150.0, deadline - 80.0)
    h_bin3, sc_bin3 = binary_anneal(best_h, target, n_iter=30000, deadline=phase5_deadline, rng=rng)
    h_unf3 = unfreeze_optimize(h_bin3, target, rng, phase5_deadline)
    sc3 = compute_c5(h_unf3)
    if sc3 < best_score:
        best_score = sc3
        best_h = h_unf3

    h_bin4, sc_bin4 = binary_steepest(best_h, target, n_sweeps=5, deadline=phase5_deadline, rng=rng)
    if sc_bin4 < best_score:
        h_unf4 = unfreeze_optimize(h_bin4, target, rng, phase5_deadline)
        sc4 = compute_c5(h_unf4)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_unf4

    h_eq2, sc_eq2 = top_k_equalize(best_h, target, n_top=4, n_steps=150, deadline=deadline - 40.0)
    if sc_eq2 < best_score:
        best_score = sc_eq2
        best_h = h_eq2
    for a in [200000, 1000000]:
        if time.time() > deadline - 30.0:
            break
        h_new = lbfgs_step(best_h, target, a, 200.0, 500, deadline - 30.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if time.time() < deadline - 30.0:
        h_cd, sc_cd = coordinate_descent_fast(best_h, target, n_sweeps=2, deadline=deadline - 10.0)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if time.time() < deadline - 5.0:
        h_slq = slsqp_polish(best_h, target, 3000, deadline - 2.0)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
