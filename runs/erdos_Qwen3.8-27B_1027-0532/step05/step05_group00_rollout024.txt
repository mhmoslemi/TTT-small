def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[np.random.choice(len(idx))] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[np.random.choice(len(idx))] = 1
            final_sum += 1
    return h_r.astype(float)

    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    c_max = np.max(corr)

            else:

                grad[kk:] -= w * h[:n-kk]

                  time_limit=None, start_time=None, rng=None):

    m = np.zeros(n)
    v = np.zeros(n)

    best_score = compute_c5(h)
    
    binary_check_interval = max(1, n_iters // 100)
    best_binary_h = project_binary(h, int(target_sum))
    best_binary_score = compute_c5(best_binary_h)
    
    for t in range(1, n_iters + 1):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit + start_time:
                break
        
        # Get beta for this iteration

        # Interpolate through beta_schedule
        if len(beta_schedule) == 1:
            beta = beta_schedule[0]
        else:
            seg = progress * (len(beta_schedule) - 1)
            i = int(seg)
            frac = seg - i
            if i >= len(beta_schedule) - 1:
                beta = beta_schedule[-1]
            else:
                beta = beta_schedule[i] * (1 - frac) + beta_schedule[i+1] * frac
        
        obj, grad = smooth_max_and_grad(h, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
        
        # Periodic binary check
        if t % binary_check_interval == 0:
            h_bin = project_binary(h, int(target_sum))
            sc = compute_c5(h_bin)
            if sc < best_binary_score:
                best_binary_score = sc
                best_binary_h = h_bin.copy()
        
        if t % 500 == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    
    return best_h, best_score, best_binary_h, best_binary_score

        else:
            idx2 = np.where(h > 1e-10)[0]
        if len(idx2) > 0:
            h[idx2] = np.clip(h[idx2] + diff2 / len(idx2), 0.0, 1.0)
    return h

def project_binary(h, target_sum, rng=None):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if rng is None:
        rng = np.random.RandomState(0)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Exact sum check
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[rng.choice(len(idx))] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[rng.choice(len(idx))] = 1
            final_sum += 1
    return h_r.astype(float)

def smooth_max_and_grad(h, beta, top_k=50):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    c_max = float(np.max(corr))
    shifted = beta * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    obj = c_max + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    
    # Use only top-k weights
    if top_k < 2 * n - 1:
        top_idx = np.argsort(-weights)[:top_k]
        top_idx = top_idx[::-1]  # sort by weight ascending for early break
        top_w = weights[top_idx]
        top_idx = top_idx[::-1]
        top_w = top_w[::-1]
    else:
        top_idx = np.arange(2 * n - 1)
        top_w = weights
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, top_w):
        if w < 1e-12:
            break
        k = idx - (n - 1)
        if k >= 0:
            if k > 0:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
            else:
                grad[:] += w * (1.0 - h[:])
                grad[:] -= w * h[:]
        else:
            kk = -k
            grad[:n-kk] += w * (1.0 - h[kk:])
            grad[kk:] -= w * h[:n-kk]
    
    return obj, grad

def adam_continuous(h_init, target_sum, n_iters, beta_schedule, lr=0.05,
                    beta1=0.9, beta2=0.999, eps=1e-8,
                    time_limit=None, start_time=None, rng=None,
                    binary_check_every=2000):
    n = len(h_init)
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_cont_h = h.copy()
    best_cont_score = compute_c5(h)
    best_bin_h = project_binary(h, int(target_sum))
    best_bin_score = compute_c5(best_bin_h)
    
    for t in range(1, n_iters + 1):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        # Beta schedule
        progress = (t - 1) / max(n_iters - 1, 1)
        if len(beta_schedule) == 1:
            beta = beta_schedule[0]
        else:
            seg = progress * (len(beta_schedule) - 1)
            i = int(seg)
            frac = seg - i
            if i >= len(beta_schedule) - 1:
                beta = beta_schedule[-1]
            else:
                beta = beta_schedule[i] + (beta_schedule[i+1] - beta_schedule[i]) * frac
        
        obj, grad = smooth_max_and_grad(h, beta)
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
        
        # Check
        if t % binary_check_every == 0:
            sc = compute_c5(h)
            if sc < best_cont_score:
                best_cont_score = sc
                best_cont_h = h.copy()
            h_bin = project_binary(h, int(target_sum))
            sc_bin = compute_c5(h_bin)
            if sc_bin < best_bin_score:
                best_bin_score = sc_bin
                best_bin_h = h_bin.copy()
    
    # Final check
    sc = compute_c5(h)
    if sc < best_cont_score:
        best_cont_score = sc
        best_cont_h = h.copy()
    h_bin = project_binary(h, int(target_sum))
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_bin_score:
        best_bin_score = sc_bin
        best_bin_h = h_bin.copy()
    
    return best_cont_h, best_cont_score, best_bin_h, best_bin_score

def greedy_binary_swap(h, max_iters=10000, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        if stall > 200:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates that can reduce the max
        i_cand = [i for i in ones if 0 <= i + k_star < n]
        j_cand = [j for j in zeros if 0 <= j - k_star < n]
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 10
            continue
        
        # Limit candidates for speed
        max_cand = 30
        i_cand = i_cand[:max_cand]
        j_cand = j_cand[:max_cand]
        
        # Find the best swap
        best_improve = 0
        best_i = -1
        best_j = -1
        
        # Use top shifts for evaluation
        top_thresh = max_val * 0.9
        top_indices = np.where(corr >= top_thresh)[0]
        
        for i in i_cand:
            for j in j_cand:
                # Evaluate new max after swap
                new_max = -1
                for tidx in top_indices:
                    k = tidx - (n - 1)
                    delta = 0
                    if 0 <= i + k < n:
                        delta -= (1 - h[i + k])
                    if 0 <= i - k < n:
                        delta += h[i - k]
                    if 0 <= j + k < n:
                        delta += (1 - h[j + k])
                    if 0 <= j - k < n:
                        delta -= h[j - k]
                    v = corr[tidx] + delta
                    if v > new_max:
                        new_max = v
                if new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
