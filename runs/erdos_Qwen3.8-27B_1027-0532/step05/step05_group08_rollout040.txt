def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        if diff > 0:
            max_step = (1.0 - h[mask].min())
            step = min(step, max_step)
        else:
            max_step = (-h[mask].min())
            step = max(step, -max_step)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = min(12, 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.ones(top_k) / top_k
        else:
            weights /= wsum
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-14:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-16:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.015, 0.012)
        else:
            eps *= 0.88
            if eps < 1e-13:
                break
    return h_cur, best_score


def structured_annealing(h, n, target, t_start, t_end, n_iters, deadline, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    dx = 2.0 / n
    for i in range(n_iters):
        if time.time() > deadline:
            break
        frac = i / max(1, n_iters - 1)
        t = t_start * (t_end / t_start) ** frac
        move_type = rng.random()
        if move_type < 0.35:
            h_try = h_cur + rng.randn(n) * (0.02 * t + 0.002)
        elif move_type < 0.60:
            corr = compute_corr(h_cur)
            m_star = int(np.argmax(corr))
            k_star = m_star - (n - 1)
            h_try = h_cur.copy()
            if k_star >= 0:
                scale = 0.015 * t + 0.001
                idx1 = rng.randint(0, n - k_star)
                idx2 = k_star + rng.randint(0, n - k_star)
                h_try[idx1] -= scale
                h_try[idx2] += scale
            else:
                ell = -k_star
                scale = 0.015 * t + 0.001
                idx1 = ell + rng.randint(0, n - ell)
                idx2 = rng.randint(0, n - ell)
                h_try[idx1] -= scale
                h_try[idx2] += scale
        elif move_type < 0.80:
            h_try = h_cur.copy()
            i1 = rng.randint(0, n)
            i2 = rng.randint(0, n)
            if i1 != i2:
                tmp = h_try[i1]
                h_try[i1] = h_try[i2]
                h_try[i2] = tmp
        else:
            corr = compute_corr(h_cur)
            m_star = int(np.argmax(corr))
            k_star = m_star - (n - 1)
            g = np.zeros(n)
            if k_star >= 0:
                g[:n - k_star] += (1.0 - h_cur[k_star:])
                g[k_star:] -= h_cur[:n - k_star]
            else:
                ell = -k_star
                g[ell:] += (1.0 - h_cur[:n - ell])
                g[:n - ell] -= h_cur[ell:]
            g *= dx
            step_size = 0.01 * t + 0.0005
            h_try = h_cur - step_size * g + rng.randn(n) * 0.003 * t
        h_try = project_feasible(h_try, target)
        new_score = compute_c5(h_try)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            h_cur = h_try
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
    return best_h, best_score


def multi_lag_balance(h, n, target, m_active, alpha, maxiter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for outer in range(25):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:m_active]
        def obj_ml(x):
            c = compute_corr(x)
            c_active = c[top_m]
            am = float(np.max(c_active))
            shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
            return am + float(np.log(np.sum(np.exp(shifted)))) / alpha + 1e-8 * float(np.max(c))
        def objg_ml(x):
            c = compute_corr(x)
            c_active = c[top_m]
            am = float(np.max(c_active))
            shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-300:
                w = np.ones(m_active) / m_active
            else:
                w /= ws
            g = np.zeros(n)
            for wk, m in zip(w, top_m):
                m = int(m)
                k = m - (n - 1)
                if k >= 0:
                    g[:n - k] += wk * (1.0 - x[k:])
                    g[k:] -= wk * x[:n - k]
                else:
                    ell = -k
                    g[ell:] += wk * (1.0 - x[:n - ell])
                    g[:n - ell] -= wk * x[ell:]
            g *= dx
            return g
        bounds = [(0.0, 1.0)] * n
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_ml, h_cur, method='SLSQP', jac=objg_ml,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_candidate = project_feasible(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < best_score:
                best_score = sc
                h_cur = h_candidate
        except Exception:
            break
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuation at initial resolution
    alphas_p1 = [30, 100, 500, 2000, 8000, 30000]
    for alpha in alphas_p1:
        if time.time() > deadline - 500:
            break
        for mu in [50.0, 200.0]:
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 250)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP at initial resolution
    for alpha in [2000, 10000, 40000]:
        if time.time() > deadline - 400:
            break
        h_new = slsqp_round(best_h, n, target, alpha, 2000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Subgradient at initial resolution
    if time.time() < deadline - 350:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, 1500, 0.005, 0.004, 0.997, deadline - 300)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 4: Move to n=800
    n2 = 800
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)
    best_h2 = h2.copy()
    best_score2 = compute_c5(h2)

    if time.time() < deadline - 300:
        for alpha in [20, 60, 200, 800, 3000, 12000]:
            if time.time() > deadline - 100:
                break
            for mu in [30.0, 100.0, 300.0]:
                h2_new = lbfgsb_round(best_h2, n2, target2, alpha, mu, 150)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new

    if time.time() < deadline - 200:
        for alpha in [2000, 8000, 30000]:
            if time.time() > deadline - 80:
                break
            h2_new = slsqp_round(best_h2, n2, target2, alpha, 1500, 1e-15)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

    if best_score2 < best_score:
        best_score = best_score2
        best_h = best_h2.copy()
    n_cur = len(best_h)
    target_cur = n_cur / 2.0

    # Phase 5: Multi-lag balancing at working resolution
    if time.time() < deadline - 200:
        h_ml, sc_ml = multi_lag_balance(best_h, n_cur, target_cur, 10, 3000, 400, deadline - 120)
        if sc_ml < best_score:
            best_score = sc_ml
            best_h = h_ml

    # Phase 6: Structured simulated annealing
    if time.time() < deadline - 150:
        h_sa, sc_sa = structured_annealing(best_h, n_cur, target_cur, 0.4, 0.001, 4000, deadline - 80, rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 7: Subgradient polish
    if time.time() < deadline - 100:
        for temp in [0.004, 0.002, 0.001, 0.0005]:
            if time.time() > deadline - 50:
                break
            h_sg, sc_sg = subgradient_polish(best_h, n_cur, target_cur, 2000, temp, 0.003, 0.998, deadline - 50)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 8: Worst-lag surgery
    if time.time() < deadline - 60:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 600, 0.003, deadline - 30)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 9: Second annealing round (cooler)
    if time.time() < deadline - 50:
        h_sa2, sc_sa2 = structured_annealing(best_h, n_cur, target_cur, 0.1, 0.0003, 2000, deadline - 25, rng)
        if sc_sa2 < best_score:
            best_score = sc_sa2
            best_h = h_sa2

    # Phase 10: Final SLSQP polish
    if time.time() < deadline - 15:
        h_new = slsqp_round(best_h, n_cur, target_cur, 50000, 500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 11: Final micro subgradient
    if time.time() < deadline - 5:
        h_sg, sc_sg = subgradient_polish(best_h, n_cur, target_cur, 500, 0.0005, 0.0008, 0.999, deadline - 2)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 12: Try complement symmetry
    if time.time() < deadline - 3:
        h_comp = 1.0 - best_h
        h_comp = project_feasible(h_comp, target_cur)
        sc_comp = compute_c5(h_comp)
        if sc_comp < best_score:
            best_score = sc_comp
            best_h = h_comp

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
