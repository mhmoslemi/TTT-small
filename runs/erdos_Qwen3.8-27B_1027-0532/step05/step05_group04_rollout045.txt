        else:

- For h[i] ∈ (0,1), the gradient of the Lagrangian = 0

        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_worst = int(np.argmax(over))
        grad = np.zeros(n)
        grad[:n - k_worst] -= h_cur[k_worst:]
        grad[k_worst:] += (1.0 - h_cur[:n - k_worst])
        grad *= dx
        improved = False
        for trial in [eps, eps * 0.5, eps * 2.0, eps * 0.25, eps * 4.0]:
            h_new = h_cur - trial * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-13:
                h_cur = h_new
                cur_score = sc_new
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h_cur.copy()
                if trial > eps:
                    eps = min(trial * 1.3, 0.01)
                break
        if not improved:
            eps *= 0.55
            if eps < 1e-8:
                break
    return best_h, best_score

def multi_lag_target(h, n, target, n_rounds, top_k, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    dx = 2.0 / n
    eps = 0.001
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        top_idx = np.argsort(over)[::-1][:top_k]
        grad = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            grad[:n - ki] -= h_cur[ki:]
            grad[ki:] += (1.0 - h_cur[:n - ki])
        grad /= top_k
        grad *= dx
        improved = False
        for trial in [eps, eps * 0.5, eps * 2.0]:
            h_new = h_cur - trial * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-13:
                h_cur = h_new
                cur_score = sc_new
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h_cur.copy()
                if trial > eps:
                    eps = min(trial * 1.2, 0.008)
                break
        if not improved:
            eps *= 0.6
            if eps < 1e-8:
                break
    return best_h, best_score

def perturbation_refine(h, n, target, n_trials, sigma, deadline):
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(12345)
    for t in range(n_trials):
        if time.time() > deadline:
            break
        noise = rng.randn(n) * sigma
        h_pert = h + noise
        h_pert = project_feasible(h_pert, target)
        h_new = slsqp_lse(h_pert, n, target, alpha=2000, maxiter=2000,
                          ftol=1e-15, deadline=time.time() + 40)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm-up with increasing alpha
    ph1_dead = start_time + 80
    for alpha in [20, 80, 300, 800]:
        if time.time() > ph1_dead:
            break
        h_new = slsqp_lse(best_h, n, target, alpha, maxiter=4000,
                          ftol=1e-15, deadline=ph1_dead)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Weighted gradient descent (main workhorse)
    ph2_dead = start_time + 220
    h_new, sc_new = wgd_momentum(best_h, n, target, n_steps=2500,
                                  beta_init=20, beta_growth=1.2,
                                  lr_init=0.0015, momentum=0.85,
                                  deadline=ph2_dead)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    # SLSQP polish after WGD
    h_new = slsqp_lse(best_h, n, target, alpha=2000, maxiter=4000,
                      ftol=1e-16, deadline=start_time + 260)
    sc = compute_c5(h_new)
    if sc < best_score:
        best_score = sc
        best_h = h_new

    # Phase 3: Multi-lag targeting
    ph3_dead = start_time + 360
    h_new, sc_new = multi_lag_target(best_h, n, target, n_rounds=600,
                                      top_k=3, deadline=ph3_dead)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    # SLSQP
    h_new = slsqp_lse(best_h, n, target, alpha=5000, maxiter=4000,
                      ftol=1e-16, deadline=start_time + 400)
    sc = compute_c5(h_new)
    if sc < best_score:
        best_score = sc
        best_h = h_new

    # Phase 4: Second WGD with higher beta
    ph4_dead = start_time + 520
    h_new, sc_new = wgd_momentum(best_h, n, target, n_steps=2000,
                                  beta_init=200, beta_growth=1.35,
                                  lr_init=0.0008, momentum=0.8,
                                  deadline=ph4_dead)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    # Phase 5: High-alpha SLSQP sequence
    ph5_dead = start_time + 680
    for alpha in [10000, 25000, 50000]:
        if time.time() > ph5_dead:
            break
        h_new = slsqp_lse(best_h, n, target, alpha, maxiter=4000,
                          ftol=1e-16, deadline=ph5_dead)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Target worst lag (fine)
    ph6_dead = start_time + 760
    h_new, sc_new = target_worst_lag(best_h, n, target, n_rounds=400,
                                      deadline=ph6_dead)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    # Phase 7: Perturbation refinement
    ph7_dead = start_time + 880
    h_new, sc_new = perturbation_refine(best_h, n, target, n_trials=4,
                                         sigma=0.005, deadline=ph7_dead)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    # Phase 8: Final WGD with very high beta
    ph8_dead = start_time + 980
    h_new, sc_new = wgd_momentum(best_h, n, target, n_steps=1200,
                                  beta_init=2000, beta_growth=1.4,
                                  lr_init=0.0003, momentum=0.6,
                                  deadline=ph8_dead)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    # Phase 9: Final SLSQP with very high alpha
    ph9_dead = start_time + budget_s - 40
    for alpha in [80000, 150000, 300000]:
        if time.time() > ph9_dead:
            break
        h_new = slsqp_lse(best_h, n, target, alpha, maxiter=4000,
                          ftol=1e-17, deadline=ph9_dead)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 10: Final micro-surgery
    if time.time() < start_time + budget_s - 15:
        h_new, sc_new = target_worst_lag(best_h, n, target, n_rounds=200,
                                          deadline=start_time + budget_s - 5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
