        if abs(delta) > 1e-15:

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc_pass(h, n, dx, corr, rng, n_inner=22):
    order = rng.permutation(n)
    h = h.copy()
    k = np.arange(-(n - 1), n)
    for i in order:
        ip = i + k
        im = i - k
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        S = np.zeros(2 * n - 1)
        S[valid_p] += (1.0 - h[ip[valid_p]])
        S[valid_m] -= h[im[valid_m]]
        S *= dx
        old_val = h[i]
        C = corr - old_val * S
        val_0 = np.max(C)
        val_1 = np.max(C + S)
        if val_1 < val_0 - 1e-13:
            z_new = 0.0
        elif val_0 < val_1 - 1e-13:
            z_new = 1.0
        else:
            lo = min(val_0, val_1)
            hi = max(val_0, val_1)
            for _ in range(n_inner):
                mid = (lo + hi) * 0.5
                pos = S > 1e-14
                neg = S < -1e-14
                zero = ~(pos | neg)
                if np.any(zero) and np.max(C[zero]) > mid + 1e-14:
                    lo = mid
                    continue
                z_u = 1.0
                if np.any(pos):
                    z_u = min(z_u, np.float64(np.min((mid - C[pos]) / S[pos])))
                z_l = 0.0
                if np.any(neg):
                    z_l = max(z_l, np.float64(np.max((mid - C[neg]) / S[neg])))
                if z_l <= z_u and z_u >= -1e-12 and z_l <= 1.0 + 1e-12:
                    hi = mid
                else:
                    lo = mid
            T = (lo + hi) * 0.5
            pos = S > 1e-14
            neg = S < -1e-14
            z_u = 1.0
            if np.any(pos):
                z_u = min(z_u, np.float64(np.min((T - C[pos]) / S[pos])))
            z_l = 0.0
            if np.any(neg):
                z_l = max(z_l, np.float64(np.max((T - C[neg]) / S[neg])))
            z_new = np.clip((z_l + z_u) * 0.5, 0.0, 1.0)
        delta = z_new - old_val
        if abs(delta) > 1e-15:
            h[i] = z_new
            corr += delta * S
    return h, corr

def coord_desc_optimize(h_init, n, target, time_limit, rng, n_passes_max=20000, recompute_every=8):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    corr = compute_corr_vec(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_t = time.time()
    for p in range(n_passes_max):
        if time.time() - start_t > time_limit:
            break
        h, corr = coord_desc_pass(h, n, dx, corr, rng)
        h = project_feasible(h, target)
        if (p + 1) % recompute_every == 0:
            corr = compute_corr_vec(h)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_slq_target = 0.0

def _slq_cons(h):
    return np.sum(h) - _slq_target

def _slq_jac(h):
    return np.ones(_slq_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _slq_n, _slq_target
    _slq_n = n
    _slq_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_cons, 'jac': _slq_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternating_lp(h_init, n, target, time_limit, n_top=200,
                   damping_start=0.95, damping_end=0.85,
                   reg_start=1e-4, reg_end=1e-8, max_iter=5000):
    start_t = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        if stuck > 800:
            break
        g = 1.0 - h
        corr_raw = np.correlate(h, g, mode='full')
        n_half = n - 1
        top_idx = np.argsort(corr_raw)[::-1][:n_top]
        lags = top_idx - n_half
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        progress = min(1.0, it / max(1, max_iter * 0.3))
        reg = reg_start * (reg_end / reg_start) ** progress
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        damping = damping_start + (damping_end - damping_start) * progress
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

def subgradient_step(h, n, target, n_top=30, lr=None, rng=None):
    dx = 2.0 / n
    overlaps = compute_corr_vec(h)
    n_half = n - 1
    top_idx = np.argsort(overlaps)[::-1][:n_top]
    top_vals = overlaps[top_idx]
    w = 0.85 ** np.arange(n_top)
    w = w / w.sum()
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        k = m - n_half
        if k >= 0:
            idx1 = np.arange(0, n - k)
            grad[idx1] += w[i] * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] += w[i] * (-h[idx2 - k])
        else:
            kk = -k
            idx1 = np.arange(0, n - kk)
            grad[idx1] += w[i] * (1.0 - h[idx1 - kk])
            idx2 = np.arange(kk, n)
            grad[idx2] += w[i] * (-h[idx2 + kk])
    grad *= dx
    norm_g = np.linalg.norm(grad)
    if norm_g < 1e-16:
        return h, None
    if lr is None:
        c5 = float(np.max(overlaps))
        lr = min(0.5, max(0.05, (c5 * 2.0) / (norm_g ** 2 + 1e-20)))
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new, lr

            if tleft() > 40:
                cd_t2 = min(30, tleft() * 0.2)
                h_hi, c5_hi = coord_desc_optimize(h_hi, n_hi, target_hi, cd_t2, rng,
                                                  n_passes_max=10000, recompute_every=4)
            sc = compute_c5(h_hi)
            if sc < best_score - 1e-6:
                best_score = sc
                best_h = h_hi
                break

    # Phase 8: Shake and re-optimize (different starting points)
    for attempt in range(4):
        if tleft() < 80:
            break
        noise = 0.02 + attempt * 0.015
        h_shake = best_h.copy()
        for _ in range(5):
            mask = rng.random(len(h_shake)) < 0.25
            n_mask = mask.sum()
            if n_mask > 0:
                h_shake[mask] += rng.randn(n_mask) * noise
            h_shake = project_feasible(h_shake, len(h_shake) / 2.0)
        n_s = len(h_shake)
        t_s = n_s / 2.0
        cd_t = min(40, tleft() * 0.3)
        h_s, c5_s = coord_desc_optimize(h_shake, n_s, t_s, cd_t, rng,
                                        n_passes_max=15000, recompute_every=5)
        if tleft() > 50:
            h_s, c5_s = alternating_lp(h_s, n_s, t_s, min(30, tleft() * 0.3),
                                       n_top=150, damping_start=0.9,
                                       damping_end=0.8, reg_start=1e-3,
                                       reg_end=1e-6, max_iter=1500)
        if c5_s < best_score:
            best_score = c5_s
            best_h = h_s

    # Phase 9: Final coordinate descent polish
    if tleft() > 40:
        n_f = len(best_h)
        t_f = n_f / 2.0
        cd_t = min(35, tleft() * 0.7)
        h_f, c5_f = coord_desc_optimize(best_h, n_f, t_f, cd_t, rng,
                                        n_passes_max=20000, recompute_every=4)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_f = len(best_h)
        t_f = n_f / 2.0
        h_f = slsqp_optimize(best_h, n_f, t_f,
                              [200, 1000, 5000, 20000],
                              maxiter=1000, ftol=1e-15)
        sc = compute_c5(h_f)
        if sc < best_score:
            best_score = sc
            best_h = h_f
        # One more CD after SLSQP
        if tleft() > 15:
            h_f, c5_f = coord_desc_optimize(best_h, n_f, t_f, min(12, tleft() * 0.7),
                                            rng, n_passes_max=10000, recompute_every=3)
            if c5_f < best_score:
                best_score = c5_f
                best_h = h_f

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
