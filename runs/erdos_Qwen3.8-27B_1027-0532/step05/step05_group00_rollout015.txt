**Projected Gradient Method:**
1. Compute gradient of smooth objective (without constraint, just the log-sum-exp)
2. Take a step: h_new = h - lr * grad
3. Project onto feasible set: 0≤h≤1, sum(h)=n/2
4. Repeat

The projection onto {h: 0≤h≤1, sum(h)=target} can be done efficiently:
- First clip to [0,1]
- Then adjust to satisfy sum constraint (similar to what project_feasible does but more carefully)

For a more accurate projection:
- We want to find h' = argmin ||h' - h_new||² s.t. 0≤h'≤1, sum(h')=target
- This is a quadratic program with box constraints and one equality constraint
- Can be solved by bisection on the Lagrange multiplier for the equality constraint

```python
def project_box_sum(h_new, target_sum):
    """Project h_new onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(h_new)
    # Bisection on lambda: h*_i = clip(h_new_i - lambda, 0, 1)
    # We need sum(h*_i) = target_sum
    # sum(h*_i) is monotonically decreasing in lambda
    
    lo = min(h_new) - 1.0  # ensures sum > target
    hi = max(h_new) + 1.0  # ensures sum < target
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h_new - mid, 0.0, 1.0)
        s = float(np.sum(h_proj))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h_new - (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment for numerical precision
    diff = target_sum - float(np.sum(h_proj))
    if abs(diff) > 1e-12:
        # Distribute the difference among non-boundary elements
        interior = np.where((h_proj > 1e-10) & (h_proj < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            h_proj[interior] += diff / len(interior)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj
```

```python
def smooth_obj_penalty(h, alpha, mu, target, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr * dx - cmax * dx), -500.0, 0.0)

    pen = mu * (float(np.sum(h)) - target) ** 2

def smooth_grad_penalty(h, alpha, mu, target, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr * dx - cmax * dx), -500.0, 0.0)
    exp_s = np.exp(shifted)
    Z = float(np.sum(exp_s))
    p = exp_s / Z  # probability weights over correlation bins
    
    grad = np.zeros(n)
    # For each shift s (indexed by k = s + n - 1):
    for k in range(2*n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        s = k - (n - 1)
        # Term 1: (1-h[j+s]) for j where 0<=j+s<n
        # Term 2: -h[j-s] for j where 0<=j-s<n
        if s >= 0:
            # Term 1: j from 0 to n-1-s
            grad[:n-s] += dx * pk * (1.0 - h[s:])
            # Term 2: j from s to n-1
            grad[s:] += dx * pk * (-h[:n-s])
        else:
            ss = -s  # positive
            # Term 1: j from ss to n-1
            grad[ss:] += dx * pk * (1.0 - h[:-ss])
            # Term 2: j from 0 to n-1-ss
            grad[:n-ss] += dx * pk * (-h[ss:])
    
    # Penalty gradient
    grad += 2.0 * mu * (float(np.sum(h)) - target)
    
    return grad
```

```python
def sa_block(h, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    """SA with block moves."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0/n)
    best_h = h.copy()
    curr_c5 = best_c5
    stall = 0
    max_stall = 50000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        r = rng.random()
        h_new = h.copy()
        moved = False
        
        if r < 0.4:
            # Single swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
                moved = True
        elif r < 0.7:
            # Pair swap (2-for-2)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        else:
            # Block swap: swap a block of 1s with a block of 0s of the same length
            block_len = rng.randint(2, 5)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= block_len and len(zeros) >= block_len:
                # Find consecutive blocks
                i_start = ones[rng.randint(len(ones))]
                j_start = zeros[rng.randint(len(zeros))]
                # Check if block of consecutive ones starting at i_start exists
                block_i = [i_start + k for k in range(block_len) if 0 <= i_start+k < n and h[i_start+k] == 1]
                block_j = [j_start + k for k in range(block_len) if 0 <= j_start+k < n and h[j_start+k] == 0]
                if len(block_i) == block_len and len(block_j) == block_len:
                    for k in block_i:
                        h_new[k] = 0
                    for k in block_j:
                        h_new[k] = 1
                    moved = True
        
        if not moved:
            stall += 1
            continue
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0/n)
        delta = c5_new - curr_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

            grad[:n-ss] += dx * pk * (-h[ss:])
