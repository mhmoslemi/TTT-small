import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # For numerical stability, subtract max
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    obj = dx * (c_max + log_sum / alpha)
    
    # Gradient
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)  # softmax weights
    
    # ∂c_k/∂h[i] where k = m - (n-1), m is index into corr
    # corr[m] = sum over valid i of h[i] * (1-h[i+m-(n-1)])
    # Let k = m - (n-1)
    # c_k = sum_{i: 0<=i<n, 0<=i+k<n} h[i]*(1-h[i+k])
    # ∂c_k/∂h[i] = (1-h[i+k])*[0<=i+k<n] + (-h[i-k])*[0<=i-k<n]
    
    # Efficient computation:
    # grad[i] = dx * sum_k p_k * ((1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n])
    
    # For each k, the contribution to grad[i] is:
    #   (1-h[i+k]) for i in [0, n-1-k] (if k >= 0)
    #   -h[i-k] for i in [k, n-1] (if k >= 0)
    #   (1-h[i+k]) for i in [-k, n-1] i.e. i in [0, n-1] where i+k in [0,n-1] i.e. i < n-k... 
    #   Wait, let me be more careful.
    #
    # c_k = sum_{i=max(0,-k)}^{min(n-1, n-1-k)} h[i]*(1-h[i+k])
    #
    # ∂c_k/∂h[i] for a specific i:
    #   From h[i] term: (1-h[i+k]) if 0<=i+k<n
    #   From -h[i-k]*h[i] term (where the sum index is i-k): -h[i-k] if 0<=i-k<n
    #
    # So grad[i] = dx * sum_k p_k * [(1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]]
    
    # This is essentially a weighted sum of shifts.
    # For efficiency, we can compute it as a correlation-like operation.
    
    # Let A_k = p_k for each shift k
    # Then grad[i] = dx * sum_k A_k * (1-h[i+k]) * [valid] - dx * sum_k A_k * h[i-k] * [valid]
    #              = dx * (sum_k A_k * (1-h) shifted by -k) - dx * (sum_k A_k * h shifted by k)
    #              = dx * (correlation of A with (1-h)) - dx * (correlation of A with h)
    
    # Actually, let me think of it differently.
    # Let's define the shifts: k ranges from -(n-1) to n-1
    # For a given i, the term is: sum over k where the pair (i, i+k) is valid
    #   (1-h[i+k]) * p_k  - h[i-k] * p_k where (i-k, i) is valid
    
    # Let me just compute it directly using numpy operations
    # For efficiency with n=606, we can use the fact that only a few p_k are significant
    
    # Get top shifts
    n_top = min(50, len(p))
    top_idx = np.argsort(p)[-n_top:][::-1]
    top_p = p[top_idx]
    top_shifts = top_idx - (n - 1)  # actual k values
    
    grad = np.zeros(n)
    for idx in range(len(top_shifts)):
        k = top_shifts[idx]
        pk = top_p[idx]
        if pk < 1e-10:
            continue
        if k >= 0:
            # (1-h[i+k]) for i in [0, n-1-k]
            if n - 1 - k >= 0:
                grad[:n-k] += pk * (1.0 - h[k:n])
            # -h[i-k] for i in [k, n-1]
            if k > 0:
                grad[k:] -= pk * h[:n-k]
        else:
            k_abs = -k
            # (1-h[i+k]) = (1-h[i-k_abs]) for i in [k_abs, n-1]
            if n - 1 - k_abs >= 0:
                grad[k_abs:] += pk * (1.0 - h[:n-k_abs])
            # -h[i-k] = -h[i+k_abs] for i in [0, n-1-k_abs]
            if k_abs > 0:
                grad[:n-k_abs] -= pk * h[k_abs:]
    
    grad *= dx
    
    return obj, grad

def lbfgs_refine(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj_and_grad, h_cur, 
                         args=(alpha,),
                         method='L-BFGS-B',
                         jac=True,
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = np.clip(res.x, 0, 1)
            # Feasibility: adjust sum
            diff = target - np.sum(h_new)
            if abs(diff) > 1e-10:
                # Distribute adjustment
                if diff > 0:
                    idx = np.where(h_new < 1.0)[0]
                else:
                    idx = np.where(h_new > 0.0)[0]
                if len(idx) > 0:
                    h_new[idx] += diff / len(idx)
                    h_new = np.clip(h_new, 0, 1)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def sa_block_moves(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):

                h_new[zeros[j_idx]] = 1

            i = ones[rng.randint(len(ones))]

def greedy_balanced(h, max_iters, time_limit, start_time, scan_width=25):
    """Greedy that considers not just the max shift but balances all top shifts."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 2000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        
        # Get top shifts with weights
        top_count = min(20, len(corr))
        top_idx = np.argsort(corr)[-top_count:][::-1]
        top_shifts = top_idx - (n - 1)
        top_vals = corr[top_idx].astype(float)
        
        # Weighted objective: minimize sum of top vals (approximating max + variance)
        # Actually let's minimize the max while considering second-worst
        # Use a "penalized max": max + 0.1 * second_max
        sorted_top = np.sort(top_vals)[::-1]
        obj_val = sorted_top[0] + 0.15 * sorted_top[1] + 0.05 * sorted_top[2] if len(sorted_top) > 2 else sorted_top[0]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # For each candidate swap, compute new obj_val
        # Focus on candidates that affect the worst shifts
        k_star = top_shifts[0]
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        
        # Also add candidates from second worst
        if len(top_shifts) > 1:
            k_star2 = top_shifts[1]
            for i in ones:
                if 0 <= i + k_star2 < n and h[i + k_star2] == 0:
                    if i not in i_cand:
                        i_cand.append(i)
            for j in zeros:
                if 0 <= j + k_star2 < n and h[j + k_star2] == 1:
                    if j not in j_cand:
                        j_cand.append(j)
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        
        best_obj = obj_val
        best_i = -1
        best_j = -1
        
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                # Evaluate new top values
                new_vals = top_vals.copy()
                for t_idx in range(len(top_shifts)):
                    k = top_shifts[t_idx]
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    new_vals[t_idx] = top_vals[t_idx] + d
                
                sv = np.sort(new_vals)[::-1]
                new_obj = sv[0] + 0.15 * sv[1] + 0.05 * sv[2] if len(sv) > 2 else sv[0]
                
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multistart_sa(h_good, n, target, n_starts, time_per_start, rng, start_time, time_limit):
    best_h = h_good.copy()
    best_score = compute_c5(h_good)
    
    for s in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        
        # Create perturbed starting point
        h_pert = h_good.copy()
        n_flips = rng.randint(5, 15)
        for _ in range(n_flips):
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
        
        # Run SA on perturbed
        max_it = int(time_per_start * 1500)
        t_end = min(time_limit, time_per_start)
        h_new, sc = sa_block_moves(h_pert, target, max_it, 0.01, 0.00002, rng,
                                    t_end, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def _smooth_obj_constrained(h, alpha, target, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    total_exp = float(np.sum(exp_shifted))
    log_sum = float(np.log(total_exp))
    obj_dx = dx * (c_max + log_sum / alpha)
    sum_h = float(np.sum(h))
    penalty = lam * (sum_h - target) ** 2
    obj = obj_dx + penalty
    p = exp_shifted / total_exp
    top_count = min(60, len(p))
    top_idx = np.argsort(p)[-top_count:][::-1]
    top_p = p[top_idx]
    top_shifts = top_idx - (n - 1)
    grad = np.zeros(n)
    for idx in range(len(top_shifts)):
        k = top_shifts[idx]
        pk = top_p[idx]
        if pk < 1e-12:
            continue
        if k >= 0:
            if n - 1 - k >= 0:
                grad[:n - k] += pk * (1.0 - h[k:n])
            if k > 0:
                grad[k:] -= pk * h[:n - k]
        else:
            k_abs = -k
            if n - 1 - k_abs >= 0:
                grad[k_abs:] += pk * (1.0 - h[:n - k_abs])
            if k_abs > 0:
                grad[:n - k_abs] -= pk * h[k_abs:]
    grad *= dx
    grad += 2.0 * lam * (sum_h - target)
    return obj, grad


def lbfgs_refine(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        lam = max(1000.0, alpha * 10.0)
        try:
            res = minimize(_smooth_obj_constrained, h_cur,
                           args=(alpha, target, lam),
                           method='L-BFGS-B', jac=True,
                           bounds=bounds,
                           options={'maxiter': maxiter_per_alpha,
                                    'ftol': 1e-15, 'gtol': 1e-10,
                                    'maxfun': maxiter_per_alpha * 3})
            h_new = np.clip(res.x, 0.0, 1.0)
            diff = target - np.sum(h_new)
            if abs(diff) > 1e-8:
                if diff > 0:
                    idx = np.where(h_new < 1.0)[0]
                else:
                    idx = np.where(h_new > 0.0)[0]
                if len(idx) > 0:
                    h_new[idx] += diff / len(idx)
                    h_new = np.clip(h_new, 0.0, 1.0)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

        if not moved:
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def greedy_balanced(h, max_iters, time_limit, start_time, scan_width=25):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 3000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        top_count = min(25, len(corr))
        top_idx = np.argsort(corr)[-top_count:][::-1]
        top_shifts = top_idx - (n - 1)
        top_vals = corr[top_idx].astype(float)
        sorted_top = np.sort(top_vals)[::-1]
        w1, w2, w3 = 1.0, 0.12, 0.04
        obj_val = sorted_top[0] * w1 + sorted_top[1] * w2 + (sorted_top[2] * w3 if len(sorted_top) > 2 else 0.0)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = []
        j_cand = []
        for t_idx in range(min(3, len(top_shifts))):
            k = top_shifts[t_idx]
            for i in ones:
                if 0 <= i + k < n and h[i + k] == 0 and i not in i_cand:
                    i_cand.append(i)
            for j in zeros:
                if 0 <= j + k < n and h[j + k] == 1 and j not in j_cand:
                    j_cand.append(j)

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        best_obj = obj_val
        best_i = -1
        best_j = -1

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                new_vals = top_vals.copy()
                for t_idx in range(len(top_shifts)):
                    k = top_shifts[t_idx]
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n and k != 0:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n and k != 0:
                        d -= h[j - k]
                    new_vals[t_idx] = top_vals[t_idx] + d
                sv = np.sort(new_vals)[::-1]
                new_obj = sv[0] * w1 + sv[1] * w2 + (sv[2] * w3 if len(sv) > 2 else 0.0)
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def greedy_standard(h, max_iters, time_limit, start_time, scan_width=20):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 2000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue

        top_threshold = max_val * 0.85
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_shifts = top_indices - (n - 1)
        top_vals = corr.astype(float)

        best_improve = 0.0
        best_i = -1
        best_j = -1

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = top_vals[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multistart_sa(h_good, n, target, n_starts, rng, start_time, time_limit):
    best_h = h_good.copy()
    best_score = compute_c5(h_good)

    for s in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        time_per = (time_limit - (time.time() - start_time) - 10.0) / max(n_starts - s, 1)
        if time_per < 10:
            break

        h_pert = h_good.copy()
        n_flips = rng.randint(8, 20)
        for _ in range(n_flips):
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1

        max_it = int(time_per * 1200)
        t_end = min(allowed_global[0], time_per)
        h_new, sc = sa_block_moves(h_pert, target, max_it, 0.012, 0.00003, rng,
                                   t_end, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    return best_h, best_score


allowed_global = [1e9]


def run(seed=42, budget_s=1300, **kwargs):
    global allowed_global
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuous with escalating alpha
    allowed = min(180, tleft() * 0.18)
    allowed_global[0] = allowed
    alphas = [25, 60, 150, 400, 1000, 3000, 8000]
    h_lb, sc_lb = lbfgs_refine(best_h, n_init, target_init, alphas, 1500, allowed, start_time)
    if sc_lb < best_score:
        best_score = sc_lb
        best_h = h_lb.copy()

    # Binary projection and greedy
    best_h_bin = project_binary(best_h, target_init)
    sc_bin = compute_c5(best_h_bin)

    # Phase 2: Balanced greedy
    allowed = min(70, tleft() * 0.08)
    h_g, c5_g = greedy_balanced(best_h_bin, max_iters=25000, time_limit=allowed,
                                start_time=start_time, scan_width=20)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()
    else:
        if c5_g < sc_bin:
            best_h_bin = h_g.copy()

    # Phase 3: Standard greedy for additional refinement
    allowed = min(40, tleft() * 0.05)
    h_gs, c5_gs = greedy_standard(best_h_bin, max_iters=20000, time_limit=allowed,
                                  start_time=start_time, scan_width=18)
    if c5_gs < best_score:
        best_score = c5_gs
        best_h = h_gs.copy()

    # Phase 4: SA with block moves
    allowed = min(200, tleft() * 0.20)
    max_it = int(min(200000, allowed * 1200))
    h_sa, c5_sa = sa_block_moves(best_h, target_init, max_it, 0.012, 0.00003, rng,
                                 allowed, start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 5: Second L-BFGS-B round
    if tleft() > 80:
        allowed = min(100, tleft() * 0.10)
        alphas2 = [200, 800, 3000, 12000]
        h_lb2, sc_lb2 = lbfgs_refine(best_h, n_init, target_init, alphas2, 1200, allowed, start_time)
        if sc_lb2 < best_score:
            best_score = sc_lb2
            best_h = h_lb2.copy()
        h_bin2 = project_binary(best_h, target_init)
        sc_bin2 = compute_c5(h_bin2)
        if sc_bin2 < best_score:
            best_score = sc_bin2
            best_h = h_bin2.copy()
        # Quick greedy after
        if tleft() > 40:
            allowed_g = min(30, tleft() * 0.04)
            h_g2, c5_g2 = greedy_balanced(best_h, max_iters=15000, time_limit=allowed_g,
                                          start_time=start_time, scan_width=18)
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()

    # Phase 6: Multi-start SA
    if tleft() > 120:
        allowed = min(120, tleft() * 0.12)
        allowed_global[0] = allowed
        h_ms, sc_ms = multistart_sa(best_h, n_init, target_init, 4, rng,
                                    start_time, allowed)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms.copy()

    # Phase 7: Try different n values
    n_options = []
    for delta in [-4, -2, 2, 4, 6, 8, 10, 14, 18]:
        n_try = n_init + delta
        if n_try >= 300 and n_try % 2 == 0 and n_try != n_init:
            n_options.append(n_try)

    for n_try in n_options:
        if tleft() < 60:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        allowed_g = min(20, tleft() * 0.04)
        h_g3, c5_g3 = greedy_balanced(h_bin_try, max_iters=12000, time_limit=allowed_g,
                                      start_time=start_time, scan_width=15)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

        if tleft() < 40:
            continue
        allowed_sa = min(30, tleft() * 0.05)
        max_it2 = int(min(40000, allowed_sa * 1200))
        h_sa3, c5_sa3 = sa_block_moves(h_g3, target_try, max_it2, 0.008, 0.00005, rng,
                                       allowed_sa, start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 8: Final L-BFGS-B polish
    if tleft() > 35:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas3 = [3000, 10000, 30000]
        h_ref, sc_ref = lbfgs_refine(best_h, n_use, target_use, alphas3, 1500,
                                     min(tleft() - 10, 50), start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 9: Final SA polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        sc_bin_f = compute_c5(h_bin_f)
        allowed = min(25, tleft() - 8)
        max_it3 = int(min(80000, allowed * 1500))
        h_sf, c5_sf = sa_block_moves(h_bin_f, target_use, max_it3, 0.004, 0.00001, rng,
                                     allowed, start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
