def compute_c5_binary(h):
    """Compute c5 for binary h (fast)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    """Local search on binary h using swap moves."""
    n = len(h)

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_val = corr.max()
        
        # Find the lags that are at or near the max
        threshold = max_val - 0.5  # lags within 0.5 of max
        hot_lags = np.where(corr >= threshold)[0]

            a = ones[rng.randint(len(ones))]

            h_test[a] = 0.0

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)

    return float(np.max(corr)) * dx


def corr_grad_topk(h, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for ki in top_idx:
        m = int(ki) - (n - 1)
        if m > 0:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
        elif m < 0:
            p = -m
            grad[p:] += (1.0 - h[:n-p])
            grad[:n-p] -= h[p:]
        else:
            grad[:] += (1.0 - 2.0 * h)
    return corr, grad, top_idx
