        h_sl, sc_sl = slsqp_polish(best_h, best_n, best_n / 2.0, [8000, 25000], 5000, 1e-18)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    if tleft() > 30:
        h_cd, sc_cd = paired_coord_desc(best_h, best_n, best_n / 2.0, start_time + budget_s - safety, rng, n_bisect=25, pert_scale=0.01, stall_limit=50)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    h_out = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
