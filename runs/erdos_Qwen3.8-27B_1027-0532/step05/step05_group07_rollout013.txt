```python
def compute_grad(h, n):
    """Compute Jacobian of corr w.r.t. h. Shape: (2n-1, n)"""
    nl1 = 2 * n - 1
    grad = np.zeros((nl1, n))
    for k in range(nl1):
        s = k - (n - 1)
        # First term: (1-h[j+s]) for valid j where 0<=j<n and 0<=j+s<n
        # j+s >= 0 => j >= -s, and j+s < n => j < n-s
        # Also 0 <= j < n
        j_start = max(0, -s)
        j_end = min(n, n - s)  # exclusive
        if j_end > j_start:
            grad[k, j_start:j_end] = 1.0 - h[j_start+s:j_end+s]
        # Second term: -h[j-s] for valid j where 0<=j<n and 0<=j-s<n
        # j-s >= 0 => j >= s, and j-s < n => j < n+s
        # Also 0 <= j < n
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)  # exclusive
        if j_end2 > j_start2:
            grad[k, j_start2:j_end2] -= h[j_start2-s:j_end2-s]
    return grad
```

```python
def frank_wolfe(h_init, n, target, max_iter, alpha_schedule, deadline):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for t in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute correlation
        corr = compute_corr(h)
        
        # Choose alpha (smoothing parameter)
        alpha = alpha_schedule(t)
        
        # Compute weights
        max_corr = np.max(corr)
        weights = np.exp(alpha * (corr - max_corr))
        weights /= weights.sum()
        
        # Compute gradient
        grad = compute_grad(h, n)
        g = weights @ grad  # Shape: (n,)
        
        # Solve linear subproblem
        h_new = fw_linear_subproblem(g, n, target)
        
        # Step size
        gamma = 2.0 / (t + 2)
        
        # Update
        h = (1 - gamma) * h + gamma * h_new
        h = project_feasible(h, target)
        
        # Check improvement
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score
```

```python
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, n):
    """Jacobian of corr w.r.t. h. Shape: (2n-1, n)"""
    nl1 = 2 * n - 1
    grad = np.zeros((nl1, n))
    for s in range(-(n-1), n):
        k = s + n - 1
        js = max(0, -s)
        je = min(n, n - s)
        if je > js:
            grad[k, js:je] = 1.0 - h[js+s:je+s]
        js2 = max(0, s)
        je2 = min(n, n + s)
        if je2 > js2:
            grad[k, js2:je2] -= h[js2-s:je2-s]
    return grad

def fw_step(g, n, target):
    """Solve min <g, h> s.t. sum(h)=target, 0<=h<=1"""
    idx = np.argsort(g)
    h_new = np.zeros(n)
    h_new[idx[:int(target)]] = 1.0
    return h_new

def frank_wolfe_anneal(h_init, n, target, deadline, rng):

    best_h = h.copy()

        t += 1
        corr = compute_corr(h)

        alpha = min(alpha, 50000.0)

        shifted = alpha * (corr - max_c)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_score
```

```python
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad(h, n):
    nl1 = 2 * n - 1
    grad = np.zeros((nl1, n))
    for s in range(-(n - 1), n):
        k = s + n - 1
        js = max(0, -s)
        je = min(n, n - s)
        if je > js:
            grad[k, js:je] = 1.0 - h[js + s:je + s]
        js2 = max(0, s)
        je2 = min(n, n + s)
        if je2 > js2:
            grad[k, js2:je2] -= h[js2 - s:je2 - s]
    return grad


def fw_linear_subproblem(g, n, target):
    idx = np.argsort(g)
    h_new = np.zeros(n)
    h_new[idx[:int(target)]] = 1.0
    return h_new


def safe_frank_wolfe(h_init, n, target, deadline, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    stalled = 0
    step_sizes = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]

    while time.time() < deadline:
        t += 1
        corr = compute_corr(h)
        alpha = 20.0 * (1.08 ** (t // 20))
        alpha = min(alpha, 100000.0)
        max_c = np.max(corr)
        shifted = alpha * (corr - max_c)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= ws
        grad = compute_grad(h, n)
        g = weights @ grad
        h_v = fw_linear_subproblem(g, n, target)
        improved = False
        for gamma in step_sizes:
            h_try = (1.0 - gamma) * h + gamma * h_v
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                stalled = 0
                break
        if not improved:
            stalled += 1
            if stalled > 80:
                scale = 0.02 + 0.03 * (stalled // 200)
                scale = min(scale, 0.15)
                h = project_feasible(h + rng.randn(n) * scale, target)
                stalled = 0
    return best_h, best_score


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=40):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep_count = 0

    for sweep in range(n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break
        sweep_count += 1
        if sweep % recompute_interval == 0 and sweep > 5:
            corr = compute_corr(h_cur)
        rng.shuffle(pair_positions)
        max_delta = 0.0

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            lo = 0.0
            if np.any(C > 0):
                hi = float(np.max(A[C > 0] + C[C > 0] * t_max))
            else:
                hi = float(np.max(A + C * t_max))

            for _ in range(40):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid

            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
                max_delta = max(max_delta, abs(delta))

        new_score = float(np.max(corr)) * dx
        if new_score < best_score:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 150 and sweep > 50:
            pert_scale = 0.005 + 0.01 * min(stall_count // 100, 5)
            h_cur = project_feasible(h_cur + rng.randn(n) * pert_scale, target)
            corr = compute_corr(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=5000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def gradient_descent_smoothed(h_init, n, target, deadline, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    stalled = 0
    while time.time() < deadline:
        t += 1
        corr = compute_corr(h)
        alpha = 50.0 * (1.06 ** (t // 30))
        alpha = min(alpha, 200000.0)
        max_c = np.max(corr)
        shifted = alpha * (corr - max_c)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= ws
        grad = compute_grad(h, n)
        g = weights @ grad
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            stalled += 50
        else:
            g = g / g_norm
        lr = 0.02 / (1.0 + 0.01 * t)
        h_try = h - lr * g
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_try.copy()
            h = h_try
            stalled = 0
        else:
            lr = lr * 0.5
            h_try = h - lr * g
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_try.copy()
                h = h_try
                stalled = 0
            else:
                stalled += 1
        if stalled > 60:
            scale = 0.01 + 0.02 * min(stalled // 50, 5)
            h = project_feasible(h + rng.randn(n) * scale, target)
            stalled = 0
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Safe Frank-Wolfe (global exploration)
    p1_frac = 0.25
    deadline_p1 = start_time + (budget_s - 15.0) * p1_frac
    h_fw, sc_fw = safe_frank_wolfe(best_h, n, target, deadline_p1, rng)
    if sc_fw < best_score:
        best_score = sc_fw
        best_h = h_fw

    # Phase 2: Gradient descent with smoothing
    if tleft() > 200:
        deadline_p2 = start_time + (budget_s - 15.0) * 0.40
        h_gd, sc_gd = gradient_descent_smoothed(best_h, n, target, deadline_p2, rng)
        if sc_gd < best_score:
            best_score = sc_gd
            best_h = h_gd

    # Phase 3: Paired coordinate descent (local refinement)
    if tleft() > 250:
        deadline_p3 = start_time + (budget_s - 15.0) * 0.62
        h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p3, rng, recompute_interval=30)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: SLSQP with progressive alpha
    if tleft() > 180:
        h_ref = slsqp_polish(best_h, n, target, [100, 300, 800, 2000, 5000, 12000], 4000, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: More coordinate descent
    if tleft() > 120:
        deadline_p5 = start_time + (budget_s - 15.0) * 0.82
        h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p5, rng, recompute_interval=25)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 6: High-alpha SLSQP
    if tleft() > 90:
        h_ref = slsqp_polish(best_h, n, target, [10000, 30000, 80000, 200000], 5000, 1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final coordinate descent
    if tleft() > 40:
        h_final, sc_final = paired_coord_desc(best_h, n, target, start_time + budget_s - 12.0, rng, recompute_interval=20)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
