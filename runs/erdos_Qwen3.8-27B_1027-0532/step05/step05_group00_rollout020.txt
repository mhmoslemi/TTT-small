But here's a subtle point: the "delta" in SA is new_c5 - old_c5. If we're using the max, this delta can be very small (or zero most of the time, since most swaps don't change the max). This means the SA will rarely accept moves, making it very slow to explore.

A better approach: use the sum of the top K values (or the soft max) as the SA energy. This gives more granular differences between moves.

Let me use: energy = sum of top K correlation values (after sorting). This is a smooth-ish function that correlates with the max but gives more signal for acceptance/rejection.

Actually, even better: use the soft max with a fixed alpha. For alpha=100, the soft max is a good approximation to the max but is smooth and gives useful gradients.

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            # Remove the ones with smallest h value
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Fix any remaining sum mismatch
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[-1]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def swap_delta(corr, h, i, j):
    """Compute the delta to corr when swapping h[i]:1->0 and h[j]:0->1."""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # Effect of h[i] going 1->0 (dv = -1)
    # Direct: for m in [n-1-i, 2n-2-i], delta[m] += -1 * (1-h[i + (m-(n-1))])
    m_start1 = n - 1 - i
    m_end1 = 2 * n - 2 - i
    ks1 = np.arange(m_start1, m_end1 + 1) - (n - 1)
    # ks1 = k values where h[i+k] is valid
    # 1-h[i+k] for these k values
    idx_direct_i = i + ks1  # = i + (m - (n-1)) for m in range
    delta[m_start1:m_end1+1] += -(1.0 - h[idx_direct_i])
    
    # Shifted: for m in [i, i+n-1], delta[m] += +1 * h[i - (m-(n-1))]
    m_start2 = i
    m_end2 = i + n - 1
    ks2 = np.arange(m_start2, m_end2 + 1) - (n - 1)
    idx_shift_i = i - ks2  # = i - k for k in range
    delta[m_start2:m_end2+1] += h[idx_shift_i]
    
    # Effect of h[j] going 0->1 (dv = +1)
    m_start3 = n - 1 - j
    m_end3 = 2 * n - 2 - j
    ks3 = np.arange(m_start3, m_end3 + 1) - (n - 1)
    idx_direct_j = j + ks3
    delta[m_start3:m_end3+1] += (1.0 - h[idx_direct_j])
    
    m_start4 = j
    m_end4 = j + n - 1
    ks4 = np.arange(m_start4, m_end4 + 1) - (n - 1)
    idx_shift_j = j - ks4
    delta[m_start4:m_end4+1] += -h[idx_shift_j]
    
    return delta

def apply_swap(h, corr, i, j):
    """Apply swap: h[i] 1->0, h[j] 0->1. Update corr in place."""
    delta = swap_delta(corr, h, i, j)
    corr += delta
    h[i] = 0
